/usr/bin/bash -lc "python3 - <<'PY'
import gzip
from pathlib import Path

def norm(h):
    h=h.split()[0]
    if h.startswith('@'): h=h[1:]
    if h.endswith('/1') or h.endswith('/2'): h=h[:-2]
    return h
names=set(Path('work/minimap2_hg38_proper_pair.qnames').read_text().splitlines())
seen=ret=rem=0
with gzip.open('data/inputs/short_reads/pair/forward/forward.fastqsanger.gz','rt') as f:
    while True:
        rec=[f.readline() for _ in range(4)]
        if not rec[0]: break
        seen += 1
        if norm(rec[0].rstrip()) in names: rem += 1
        else: ret += 1
print(f'input_pairs={seen}')
print(f'remove_names={len(names)}')
print(f'removed_by_input_membership={rem}')
print(f'expected_retained={ret}')
PY"
