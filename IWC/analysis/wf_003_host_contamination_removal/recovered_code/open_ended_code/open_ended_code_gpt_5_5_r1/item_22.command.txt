/usr/bin/bash -lc "set -euo pipefail
mkdir -p /workspace/work/ref /workspace/work/filter /workspace/run_trace
{
  printf 'start_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  printf '"'aligner=BWA\\n'
  printf 'forward_input=%s\\n' '/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz'
  printf 'reverse_input=%s\\n' '/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz'
  /opt/conda/bin/bwa 2>&1 | head -3 || true
} > /workspace/run_trace/execution.log
cd /workspace/work/ref
if [ "'! -s hg38.fa.gz ]; then
  curl -L --fail --retry 5 --retry-delay 10 -o hg38.fa.gz https://hgdownload.soe.ucsc.edu/goldenPath/hg38/bigZips/hg38.fa.gz
fi
printf '"'%s  hg38.fa.gz\\n' '1c9dcaddfa41027f17cd8f7a82c7293b' > hg38.fa.gz.md5.expected
md5sum -c hg38.fa.gz.md5.expected | tee /workspace/run_trace/hg38_md5check.log
if [ "'! -s hg38.fa ]; then
  gzip -dc hg38.fa.gz > hg38.fa
fi
printf '"'reference_ready_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /workspace/run_trace/execution.log
ls -lh hg38.fa.gz hg38.fa >> /workspace/run_trace/execution.log'
