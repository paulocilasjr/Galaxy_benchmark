/usr/bin/bash -lc 'set -euo pipefail
FQ=/workspace/work/filter/validation/filtered_forward.qnames
RQ=/workspace/work/filter/validation/filtered_reverse.qnames
if diff -q "$FQ" "$RQ" >/dev/null; then
  printf '"'qnames_synchronized\\tOK\\n' >> /workspace/run_trace/validation_summary.txt
else
  printf 'qnames_synchronized\\tFAIL\\n' >> /workspace/run_trace/validation_summary.txt
  exit 1
fi
printf 'end_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /workspace/run_trace/execution.log
tail -20 /workspace/run_trace/validation_summary.txt'
