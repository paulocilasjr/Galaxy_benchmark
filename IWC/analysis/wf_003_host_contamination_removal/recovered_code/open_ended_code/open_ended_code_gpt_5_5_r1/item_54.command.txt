/usr/bin/bash -lc 'set -euo pipefail
export PATH=/opt/conda/bin:$PATH
R1=/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz
R2=/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz
REF=/workspace/work/ref/hg38.fa
cd /workspace/work/filter
printf '"'alignment_start_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /workspace/run_trace/execution.log
bwa mem -t 8 "$REF" "$R1" "$R2" 2> /workspace/run_trace/bwa_mem.stderr.log '"\\
  | perl -F'\\t' -lane 'next if /"'^@/; next if @F < 2; $flag = $F[1] + 0; print $F[0] if (($flag & 0x2) && !($flag & 0x100) && !($flag & 0x800));'"' \\
  | LC_ALL=C sort -u > remove_proper_pair_qnames.txt
printf 'alignment_done_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /workspace/run_trace/execution.log
wc -l remove_proper_pair_qnames.txt | tee /workspace/run_trace/removed_qnames.count'
