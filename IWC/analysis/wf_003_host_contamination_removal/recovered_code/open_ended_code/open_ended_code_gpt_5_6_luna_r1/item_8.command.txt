/usr/bin/bash -lc "set -o pipefail
printf '%s\\n' 'Tool availability:'
for t in bowtie2 bowtie2-build samtools bwa bwa-mem2 minimap2 pigz gzip md5sum awk; do
  if command -v \""'$t" >/dev/null 2>&1; then printf '"'%-14s %s\\n' \""'$t" "$(command -v "$t")"; else printf '"'%-14s missing\\n' \""'$t"; fi
done
printf '"'%s\\n' 'Disk:'
df -h /workspace /tmp | sed -n '1,4p'
printf '%s\\n' 'Likely reference locations:'
for d in /reference /references /ref /refs /data /opt /usr/local/share; do
  if [ -d \""'$d" ]; then find "$d" -maxdepth 3 -type f '"\\( -iname '*hg38*' -o -iname 'hg38.fa*' \\) -print 2>/dev/null | sed -n '1,40p'; fi
done"
