/usr/bin/bash -lc 'set -o pipefail
work=/workspace/.host_filter_work
fwd=/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz
rev=/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz
PATH=/workspace/.host_filter_env/bin:$PATH bowtie2 '"\\
  -x \""'$work/hg38" '"\\
  -1 \""'$fwd" '"\\
  -2 \""'$rev" '"\\
  --threads 8 \\
  --un-conc-gz \""'$work/retained.fastq.gz" '"\\
  -S /dev/null \\
  2> \""'$work/bowtie2.log"
printf '"'%s\\n' 'Bowtie2 summary:'
sed -n '1,120p' \""'$work/bowtie2.log"
printf '"'%s\\n' 'Retained outputs:'
ls -lh \""'$work"/retained.fastq.gz* 2>/dev/null || true'
