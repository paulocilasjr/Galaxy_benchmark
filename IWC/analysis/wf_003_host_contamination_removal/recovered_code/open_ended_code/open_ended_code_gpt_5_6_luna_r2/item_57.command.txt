/usr/bin/bash -lc "printf '%s\\n' '---BUILD PROCESSES---'
for proc in /proc/[0-9]*; do
  cmdline="'$(tr '"'\\0' ' ' < \""'$proc/cmdline" 2>/dev/null)
  case "$cmdline" in
    *bowtie2-build*|*bowtie2-build-s*)
      pid=${proc##*/}
      ppid=$(awk '"'{print "'$4}'"' \""'$proc/stat" 2>/dev/null)
      printf '"'pid=%s ppid=%s %s\\n' \""'$pid" "$ppid" "$cmdline"
      ;;
  esac
done
printf '"'%s\\n' '---INDEX LOG TIMES---'
stat -c '%y %s %n' /workspace/work_hg38/index.log /workspace/work_hg38/hg38*.bt2.tmp 2>/dev/null | sed -n '1,80p'
printf '%s\\n' '---LOG HEAD/TAIL---'
sed -n '1,12p' /workspace/work_hg38/index.log
tail -n 20 /workspace/work_hg38/index.log"
