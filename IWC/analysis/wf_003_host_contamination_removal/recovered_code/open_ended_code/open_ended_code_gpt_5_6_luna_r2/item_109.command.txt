/usr/bin/bash -lc "gzip -cd /workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/all_forward_qnames_v1.txt
gzip -cd /workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/all_reverse_qnames_v1.txt
sort -u /workspace/work_hg38/retained_qnames_v1.txt > /workspace/work_hg38/retained_qnames_sorted_v1.txt
sort -u /workspace/work_hg38/all_forward_qnames_v1.txt > /workspace/work_hg38/all_forward_qnames_sorted_v1.txt
sort -u /workspace/work_hg38/all_reverse_qnames_v1.txt > /workspace/work_hg38/all_reverse_qnames_sorted_v1.txt
cmp /workspace/work_hg38/all_forward_qnames_v1.txt /workspace/work_hg38/all_reverse_qnames_v1.txt
cmp /workspace/work_hg38/retained_qnames_v1.txt /workspace/work_hg38/retained_qnames_v1_r.txt
printf 'input_qnames_synchronized=yes\\nretained_bowtie2_qnames_synchronized=yes\\n'
printf 'input_unique_forward_qnames='; wc -l < /workspace/work_hg38/all_forward_qnames_sorted_v1.txt
printf 'input_unique_reverse_qnames='; wc -l < /workspace/work_hg38/all_reverse_qnames_sorted_v1.txt
printf 'retained_unique_qnames='; wc -l < /workspace/work_hg38/retained_qnames_sorted_v1.txt
printf 'retained_duplicate_qnames='; sort /workspace/work_hg38/retained_qnames_v1.txt | uniq -d | wc -l
cmp /workspace/work_hg38/retained_qnames_sorted_v1.txt <(comm -12 /workspace/work_hg38/retained_qnames_sorted_v1.txt /workspace/work_hg38/all_forward_qnames_sorted_v1.txt)
printf '%s\\n' 'retained_qnames_are_input_qnames=yes'
awk 'NR==FNR {keep["'$1]=1; next} FNR%4==1 {id=$1; sub(/''^@/,"",id); take=(id in keep)} take'"' \\
  /workspace/work_hg38/retained_qnames_v1.txt \\
  <(gzip -cd /workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz) \\
  > /workspace/work_hg38/expected_forward_original_v1.fastq
awk 'NR==FNR {keep["'$1]=1; next} FNR%4==1 {id=$1; sub(/''^@/,"",id); take=(id in keep)} take'"' \\
  /workspace/work_hg38/retained_qnames_v1.txt \\
  <(gzip -cd /workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz) \\
  > /workspace/work_hg38/expected_reverse_original_v1.fastq
gzip -cd /workspace/work_hg38/expected_forward_original_v1.fastq | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/expected_forward_qnames_v1.txt
gzip -cd /workspace/work_hg38/expected_reverse_original_v1.fastq | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/expected_reverse_qnames_v1.txt
cmp /workspace/work_hg38/retained_qnames_sorted_v1.txt <(sort -u /workspace/work_hg38/expected_forward_qnames_v1.txt)
cmp /workspace/work_hg38/expected_forward_qnames_v1.txt /workspace/work_hg38/expected_reverse_qnames_v1.txt
printf 'expected_forward_records='; awk 'END {print NR/4}' /workspace/work_hg38/expected_forward_original_v1.fastq
printf 'expected_reverse_records='; awk 'END {print NR/4}' /workspace/work_hg38/expected_reverse_original_v1.fastq
printf '%s\\n' 'original_record_selection_set_and_pair_order_verified=yes'"
