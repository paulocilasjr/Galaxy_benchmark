/usr/bin/bash -lc "set -o pipefail
gzip -t final_answer/filtered_forward.fastqsanger.gz
gzip -t final_answer/filtered_reverse.fastqsanger.gz
zcat final_answer/filtered_forward.fastqsanger.gz | awk 'NR%4==1 {n++; if(substr("'$0,1,1)!="@") bad_header++} NR%4==2 {seq_len=length($0)} NR%4==3 {if(substr($0,1,1)!="+") bad_plus++} NR%4==0 {if(length($0)!=seq_len) bad_length++} END {print "forward_records=" n, "lines=" NR, "bad_headers=" (bad_header+0), "bad_plus=" (bad_plus+0), "sequence_quality_length_mismatches=" (bad_length+0); if(NR%4 || bad_header || bad_plus || bad_length) exit 1}'"'
zcat final_answer/filtered_reverse.fastqsanger.gz | awk 'NR%4==1 {n++; if(substr("'$0,1,1)!="@") bad_header++} NR%4==2 {seq_len=length($0)} NR%4==3 {if(substr($0,1,1)!="+") bad_plus++} NR%4==0 {if(length($0)!=seq_len) bad_length++} END {print "reverse_records=" n, "lines=" NR, "bad_headers=" (bad_header+0), "bad_plus=" (bad_plus+0), "sequence_quality_length_mismatches=" (bad_length+0); if(NR%4 || bad_header || bad_plus || bad_length) exit 1}'"'
paste <(zcat final_answer/filtered_forward.fastqsanger.gz | awk 'NR%4==1{sub(/"'^@/,""); print $1}'"') <(zcat final_answer/filtered_reverse.fastqsanger.gz | awk 'NR%4==1{sub(/"'^@/,""); print $1}'"') | awk '"'$1!=$2{bad++} END{print "paired_rows=" NR, "mate_id_mismatches=" (bad+0); if(bad) exit 1}'"'
zcat final_answer/filtered_forward.fastqsanger.gz | awk 'NR%4==1{sub(/"'^@/,""); print $1}'"' | LC_ALL=C sort -u > run_work/alignment/retained_ids_check.txt
printf 'proper_retained_overlap='; comm -12 run_work/alignment/proper_pair_ids.txt run_work/alignment/retained_ids_check.txt | wc -l
printf 'classified_plus_retained_unique='; LC_ALL=C sort -u run_work/alignment/proper_pair_ids.txt run_work/alignment/retained_ids_check.txt | wc -l
perl -MJSON::PP -e '"'$x=decode_json(do { local $/; <> }); die unless ref($x) eq "HASH" && keys(%$x)==1 && $x->{aligner} eq "BWA-MEM"; print "method_json_valid=1'"\\n\"' final_answer/method.json
sha256sum final_answer/filtered_forward.fastqsanger.gz final_answer/filtered_reverse.fastqsanger.gz final_answer/method.json
find final_answer -maxdepth 1 -type f -printf '%f\\t%s bytes\\n' | sort"
