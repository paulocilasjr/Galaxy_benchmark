/usr/bin/bash -c "python3 - <<'PY'
import gzip
valid=set(); primary=set()
with open('/workspace/analysis/bowtie2.sam') as f:
    for line in f:
        if line.startswith('@'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<2: continue
        flag=int(p[1])
        if not (flag & 0x100) and not (flag & 0x800):
            primary.add(p[0])
            if flag & 0x2:
                valid.add(p[0])
ret=set()
with gzip.open('/workspace/analysis/unconc_1.gz','rt') as f:
    while True:
        h=f.readline(); s=f.readline(); plus=f.readline(); q=f.readline()
        if not h: break
        ret.add(h.rstrip('\\n').split()[0])
print('primary',len(primary),'valid',len(valid),'retained',len(ret),'complement_ok', valid.isdisjoint(ret) and (primary-ret)==valid and ret==primary-valid)
PY"
