/usr/bin/bash -c "python3 - <<'PY'
import gzip
valid=set(); primary=set()
with open('/workspace/analysis/bowtie2.sam') as f:
    for line in f:
        if line.startswith('@'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<2: continue
        flag=int(p[1])
        if not (flag & 0x100) and not (flag & 0x800):
            primary.add(p[0])
            if flag & 0x2: valid.add(p[0])
ret=set()
with gzip.open('/workspace/analysis/unconc_1.gz','rt') as f:
    while True:
        h=f.readline(); s=f.readline(); plus=f.readline(); q=f.readline()
        if not h: break
        ret.add(h.rstrip('\\n').split()[0])
print('primary',len(primary),'valid',len(valid),'retained',len(ret))
print('valid disjoint ret', len(valid & ret), valid.isdisjoint(ret))
print('primary-ret',len(primary-ret),'same as valid',(primary-ret)==valid)
print('ret-primary',len(ret-primary))
print('ret == primary-valid', ret==primary-valid)
print('extra retained not primary',list(ret-primary)[:5])
print('valid missing from primary-ret',list(valid-(primary-ret))[:5])
print('primary-ret not valid',list((primary-ret)-valid)[:5])
PY"
