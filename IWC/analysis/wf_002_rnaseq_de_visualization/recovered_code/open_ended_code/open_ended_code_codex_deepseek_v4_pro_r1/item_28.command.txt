/usr/bin/bash -lc "/usr/bin/micromamba run -p /tmp/de-env Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2))
ref_dir <- '/workspace/data/inputs/counts_from_reference_condition'
chg_dir <- '/workspace/data/inputs/counts_from_changed_condition'
files_ref <- list.files(ref_dir, pattern='[.]dat"'$'"', recursive=TRUE, full.names=TRUE)
files_chg <- list.files(chg_dir, pattern='[.]dat"'$'"', recursive=TRUE, full.names=TRUE)
files <- c(files_ref, files_chg)
# sort to deterministic ref1 ref2 chg1 chg2 by path; identifiers in filename
files <- files[order(files)]
read_one <- function(f) {
  x <- read.delim(f, check.names=FALSE, stringsAsFactors=FALSE)
  stopifnot(ncol(x)==2, identical(colnames(x)[1], 'Geneid'))
  # second col may be sample ID
  x
}
lst <- lapply(files, read_one)
print(sapply(lst, dim))
print(sapply(lst, function(x) colnames(x)[2]))
# check IDs identical
ids <- lapply(lst, function(x) x[[1]])
stopifnot(all(vapply(ids, identical, logical(1), ids[[1]])))
counts <- do.call(cbind, lapply(lst, function(x) as.integer(x[[2]])))
sample_names <- gsub('[.]dat"'$'"', '', basename(files))
colnames(counts) <- sample_names
rownames(counts) <- ids[[1]]
print(head(counts))
print(dim(counts))
keep <- rowSums(counts >= 10) >= 2
cat('retained', sum(keep), '\\n')
counts_f <- counts[keep, , drop=FALSE]
condition <- factor(rep(c('reference','changed'), each=2), levels=c('reference','changed'))
col_data <- data.frame(condition=condition, row.names=colnames(counts_f))
dds <- DESeqDataSetFromMatrix(countData=counts_f, colData=col_data, design=~condition)
dds <- DESeq(dds)
res <- results(dds, contrast=c('condition','changed','reference'), independentFiltering=FALSE, cooksCutoff=FALSE)
cat('res dim', dim(res), '\\n')
cat('NAs pvalue', sum(is.na(res"'$pvalue)), '"'NAs padj', sum(is.na(res"'$padj)), '"'\\n')
cat('nonfinite lfc', sum("'!is.finite(res$log2FoldChange)), '"'\\n')
print(table(is.finite(res"'$log2FoldChange), is.finite(res$pvalue), useNA='"'ifany'))
print(res[which("'!is.finite(res$log2FoldChange) | !is.finite(res$pvalue))[1:min(10, sum(!is.finite(res$log2FoldChange) | !is.finite(res$pvalue)))],])
cat('"'range pvalue excluding NA', range(res"'$pvalue, na.rm=TRUE), '"'\\n')
cat('max abs lfc', max(abs(res"'$log2FoldChange), na.rm=TRUE), '"'\\n')
cat('sample names:', sample_names, '\\n')
cat('summary padj', summary(res"'$padj), '"'\\n')
RS"
