/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv, json, math, os, glob
import pandas as pd
p='/workspace/final_answer/differential_expression.tsv'
df=pd.read_csv(p, sep='\\t', dtype={'gene_id':str})
print(df.shape)
print(list(df.columns))
print('duplicates', df['gene_id'].duplicated().sum())
print('nulls', df.isna().sum().to_dict())
print('finite lfc', df['log2_fold_change'].apply(lambda x: isinstance(x,float) and math.isfinite(x)).all())
print('p in [0,1]', (df['p_value']>=0).all() and (df['p_value']<=1).all())
print('fdr in [0,1]', (df['fdr']>=0).all() and (df['fdr']<=1).all())
# Compare IDs to input order
inputs=glob.glob('/workspace/data/inputs/*/*/*.dat')
base=pd.read_csv(sorted(inputs)[0], sep='\\t', usecols=[0], dtype=str)
print('id subset', df['gene_id'].isin(base.iloc[:,0]).all())
# filtering recompute using pd
frames=[]
for f in sorted(inputs):
    x=pd.read_csv(f, sep='\\t')
    x.columns=['gene_id','count']
    frames.append(x.set_index('gene_id')['count'])
mat=pd.concat(frames, axis=1)
keep=(mat>=10).sum(axis=1)>=2
print('expected retained', int(keep.sum()), 'actual', len(df))
print('gene sets identical', set(df['gene_id'])==set(mat.index[keep]))
print('method json', json.load(open('/workspace/final_answer/method.json')))
PY"
