/usr/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
from pathlib import Path
samples={'SRR5085167':'reference','SRR5085168':'reference','SRR5085169':'changed','SRR5085170':'changed'}
dfs=[]
for s, cond in samples.items():
    group='counts_from_reference_condition' if cond=='reference' else 'counts_from_changed_condition'
    p=next(Path('/workspace/data/inputs').joinpath(group).glob(f'{s.lower()}_counts_table/*.dat'))
    d=pd.read_csv(p,sep='\\t')
    d=d.rename(columns={d.columns[1]:s})
    dfs.append(d)
df=dfs[0]
for d in dfs[1:]:
    df=df.merge(d[['Geneid',d.columns[1]]],on='Geneid',how='outer')
df=df.set_index('Geneid')
print(df.shape)
print(df.head())
print('dups', df.index.duplicated().sum())
print('noninteger?', ((df.to_numpy(dtype=float)%1)"'!=0).sum())
print('"'NA', df.isna().sum().sum())
print(df.columns.tolist())
print('min per sample',df.min().to_dict())
keep=(df>=10).sum(axis=1)>=2
print('keep',keep.sum())
print('sum counts retained', df[keep].values.sum(), 'total', df.values.sum())
print('genes retained zero all', (df[keep].sum(axis=1)==0).sum())
print('condition groups changed sum', df[keep][['SRR5085169','SRR5085170']].sum().sum(), 'ref', df[keep][['SRR5085167','SRR5085168']].sum().sum())
PY"
