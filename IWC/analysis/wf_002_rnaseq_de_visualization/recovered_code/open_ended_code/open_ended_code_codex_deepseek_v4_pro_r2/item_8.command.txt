/usr/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
from pathlib import Path
ref=['SRR5085167','SRR5085168']
changed=['SRR5085169','SRR5085170']
files={}
for s in ref:
    p=next(Path('/workspace/data/inputs/counts_from_reference_condition').glob(f'{s.lower()}_counts_table/*.dat'))
    files[s]=pd.read_csv(p, sep='\\t')
for s in changed:
    p=next(Path('/workspace/data/inputs/counts_from_changed_condition').glob(f'{s.lower()}_counts_table/*.dat'))
    files[s]=pd.read_csv(p, sep='\\t')
df=files[ref[0]][['Geneid']].copy()
for s in ref+changed:
    d=files[s].rename(columns={s:'count'})
    df=df.merge(d[['Geneid','count']],on='Geneid',how='outer')
df=df.set_index('Geneid')
print(df.shape)
print(df.head())
print('dups', df.index.duplicated().sum())
print('noninteger?', ((df.to_numpy(dtype=float)%1)"'!=0).sum())
print('"'NA', df.isna().sum().sum())
print('gene names')
print(df.columns.tolist())
print('min per sample',df.min().to_dict())
# filtering
keep=(df>=10).sum(axis=1)>=2
print('keep',keep.sum())
print('sum counts retained', df[keep].values.sum(), 'total', df.values.sum())
print('genes with zero all', (df[keep].sum(axis=1)==0).sum())
PY"
