/usr/bin/bash -lc 'PATH=/tmp/codex_r_base_env/bin:$PATH R_LIBS_USER=/tmp/codex_rnaseq_Rlib /tmp/codex_r_base_env/bin/Rscript --vanilla - <<'"'RS'
files <- c(
  'data/inputs/counts_from_changed_condition/srr5085169_counts_table/srr5085169_counts_table.dat',
  'data/inputs/counts_from_changed_condition/srr5085170_counts_table/srr5085170_counts_table.dat',
  'data/inputs/counts_from_reference_condition/srr5085167_counts_table/srr5085167_counts_table.dat',
  'data/inputs/counts_from_reference_condition/srr5085168_counts_table/srr5085168_counts_table.dat'
)
tab <- lapply(files, read.delim, header=TRUE, sep='\\t', quote='', comment.char='', check.names=FALSE)
stopifnot(all(vapply(tab, function(x) ncol(x)==2L && names(x)[1]=='Geneid', logical(1))))
stopifnot(all(vapply(tab, function(x) identical(x"'$Geneid, tab[[1]]$Geneid), logical(1))))
counts <- do.call(cbind, lapply(tab, `[[`, 2L))
ids <- tab[[1]]$Geneid
keep <- rowSums(counts >= 10) >= 2
o <- read.delim('"'final_answer/differential_expression.tsv', header=TRUE, sep='\\t', quote='', check.names=FALSE)
stopifnot(identical(names(o), c('gene_id','log2_fold_change','p_value','fdr')))
stopifnot(nrow(o)==sum(keep), "'!anyDuplicated(o$gene_id), identical(o$gene_id, ids[keep]))
stopifnot(all(is.finite(o$log2_fold_change)), all(is.finite(o$p_value)), all(is.finite(o$fdr)))
stopifnot(all(o$p_value >= 0 & o$p_value <= 1), all(o$fdr >= 0 & o$fdr <= 1))
stopifnot(max(abs(o$fdr - p.adjust(o$p_value, method='"'BH'))) < 1e-12)
method_text <- paste(readLines('final_answer/method.json', warn=FALSE), collapse='')
stopifnot(identical(method_text, '{\"method\":\"DESeq2\"}'))
cat('VALIDATION_OK\\n')
cat('rows=', nrow(o), '\\n', sep='')
cat('columns=', paste(names(o), collapse=','), '\\n', sep='')
cat('lfc_finite=', all(is.finite(o"'$log2_fold_change)), '"'\\n', sep='')
cat('p_range=', format(min(o"'$p_value), digits=16), '"',', format(max(o"'$p_value), digits=16), '"'\\n', sep='')
cat('fdr_range=', format(min(o"'$fdr), digits=16), '"',', format(max(o"'$fdr), digits=16), '"'\\n', sep='')
RS
printf '%s\\n' '--- concise execution log ---'
sed -n '1,40p' run_trace/execution.log
printf '%s\\n' '--- final file set ---'
find final_answer -maxdepth 1 -type f -printf '%f\\n' | sort"
