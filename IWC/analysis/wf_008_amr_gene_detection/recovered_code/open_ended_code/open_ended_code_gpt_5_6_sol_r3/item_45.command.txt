/usr/bin/bash -lc "set -e
/opt/conda/bin/python3.11 - <<'PY'
import csv, json
from pathlib import Path
out = Path('final_answer/resistance_determinants.tsv')
with out.open(newline='') as fh:
    rows = list(csv.DictReader(fh, delimiter='\\t'))
assert rows and list(rows[0]) == ['determinant']
vals = [r['determinant'] for r in rows]
assert len(vals) == 13 and len(vals) == len(set(vals))
expected = [
    'aadA5', \"aac(6')-Ib-cr\", 'blaCTX-M-15', 'blaOXA-1', 'mph(A)',
    'catB3', 'sul1', 'tet(B)', 'dfrA17', 'gyrA (S83L)',
    'gyrA (D87N)', 'parC (S80I)', 'parE (S458T)'
]
assert vals == expected
D = json.load(open('analysis_work/results/results.json'))
acq = []
for x in D['seq_regions'].values():
    if 'ResFinder-2.4.0' in x.get('ref_database', []):
        assert x['identity'] >= 90 and x['coverage'] >= 60
        if x['name'] not in acq:
            acq.append(x['name'])
assert acq == expected[:9], (acq, expected[:9])
point = []
for x in D['seq_variations'].values():
    region = D['seq_regions'][x['seq_regions'][0]]
    assert region['coverage'] >= 95
    point.append(f\"{region['name']} ({x['seq_var'].removeprefix('p.')})\")
assert point == expected[9:], (point, expected[9:])
print(f'validated_rows\\t{len(rows)}')
print(f'acquired_unique\\t{len(acq)}')
print(f'point_mutations\\t{len(point)}')
PY
find final_answer -maxdepth 1 -type f -printf '%f\\n' | sort
sha256sum final_answer/resistance_determinants.tsv
sed -n '1,30l' final_answer/resistance_determinants.tsv
sed -n '1,30p' run_trace/execution.log"
