/usr/bin/bash -lc "work=/tmp/amr_analysis_20260828
printf 'Repository snapshots:\\n'
for d in \""'$work/resfinder_db" "$work/pointfinder_db" "$work/resfinder" "$work/pointfinder"; do
  echo "$d"
  git -C "$d" status --short --branch
  git -C "$d" log -1 --format='"'%H %s'
done
printf '\\nResFinder files:\\n'; find \""'$work/resfinder" -maxdepth 3 -type f -printf '"'%P\\n' | sort | sed -n '1,240p'
printf '\\nPointFinder files:\\n'; find \""'$work/pointfinder" -maxdepth 3 -type f -printf '"'%P\\n' | sort | sed -n '1,240p'
printf '\\nDatabase files:\\n'; find \""'$work/resfinder_db" "$work/pointfinder_db" -maxdepth 3 -type f -printf '"'%P\\n' | sort | sed -n '1,300p'"
