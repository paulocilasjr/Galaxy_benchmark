  https://bitbucket.org/genomicepidemiology/resfinder_db.git \\

  https://github.com/genomicepidemiology/ResFinder.git \\
  https://github.com/genomicepidemiology/PointFinder.git; do
  echo \"URL "'$url"
  git ls-remote "$url" 2>&1 | sed -n '"'1,8p'
done"
