/usr/bin/bash -lc "micromamba run -p /workspace/tools/amr_env python - <<'PY'
import csv
want={'gyrA','parC','parE'}
best={}
for r in csv.reader(open('work/amr/pointfinder_ecoli_vs_genome.tsv'), delimiter='\\t'):
    gene=r[0].rsplit('_',2)[0]
    if gene not in want: continue
    p=float(r[2]); qcov=100*(abs(int(r[7])-int(r[6]))+1)/int(r[10]); bits=float(r[13])
    if gene not in best or (qcov,p,bits)>best[gene][:3]:
        best[gene]=(qcov,p,bits,r[0],r[1])
for gene,val in sorted(best.items()):
    print(gene, f'coverage={val[0]:.2f}', f'identity={val[1]:.3f}', val[3], val[4])
PY"
