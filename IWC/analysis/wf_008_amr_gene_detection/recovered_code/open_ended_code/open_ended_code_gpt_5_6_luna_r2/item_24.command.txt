/usr/bin/bash -lc 'set -e
mkdir -p work/db
if [ ! -d work/db/resfinder_db/.git ]; then git clone --filter=blob:none --no-checkout https://bitbucket.org/genomicepidemiology/resfinder_db.git work/db/resfinder_db; fi
if [ ! -d work/db/pointfinder_db/.git ]; then git clone --filter=blob:none --no-checkout https://bitbucket.org/genomicepidemiology/pointfinder_db.git work/db/pointfinder_db; fi
for spec in '"'work/db/resfinder_db d1e607b' 'work/db/pointfinder_db 694919f'; do set -- "'$spec; echo "fetch $1 $2"; git -C "$1" fetch --no-tags origin "$2"; git -C "$1" rev-parse FETCH_HEAD; done'
