mkdir -p work/resfinder_run
PYTHONPATH="$PWD/work/resfinder/src:$PWD/work/pydeps" /opt/conda/bin/python -m resfinder '"\\
  -ifa \""'$PWD/work/genome.fna" '"\\
  -o \""'$PWD/work/resfinder_run" '"\\
  -s 'Escherichia coli' \\
  -db_res \""'$PWD/work/db/resfinder_db" '"\\
  -db_point \""'$PWD/work/db/pointfinder_db" '"\\
  -acquired -point \\
  -l 0.60 -t 0.90 -l_p 0.95 -t_p 0.90 \\
  -b /opt/conda/bin/blastn \\
  2>&1 | tee work/resfinder_run.log"
