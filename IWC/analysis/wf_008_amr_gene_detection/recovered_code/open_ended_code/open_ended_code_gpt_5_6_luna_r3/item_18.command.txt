/usr/bin/bash -lc 'workdir=$(cat /workspace/.amr_work_path)
gzip -cd /workspace/data/inputs/input_sequence_fasta/GCA_053668395.1_ASM5366839v1_genomic.fna.gz > "$workdir/input.fna"
mkdir -p "$workdir/run"
set -o pipefail
/opt/conda/bin/python3 -m resfinder '"\\
  -ifa \""'$workdir/input.fna" '"\\
  -o \""'$workdir/run" '"\\
  -s \"Escherichia coli\" \\
  -db_res \""'$workdir/resfinder_db" '"\\
  -db_point \""'$workdir/pointfinder_db" '"\\
  -b /opt/conda/bin/blastn \\
  --acquired \\
  --point \\
  -t 0.9 \\
  -l 0.6 \\
  -t_p 0.9 \\
  -l_p 0.95 \\
  > \""'$workdir/run/stdout.log" 2> "$workdir/run/stderr.log"
status=$?
printf '"'exit_status=%s\\n' \""'$status"
printf '"'%s\\n' '--- stderr ---'
sed -n '1,240p' \""'$workdir/run/stderr.log"
printf '"'%s\\n' '--- stdout ---'
sed -n '1,240p' \""'$workdir/run/stdout.log"
printf '"'%s\\n' '--- output inventory ---'
find \""'$workdir/run" -maxdepth 3 -type f -printf '"'%P\\t%s bytes\\n' | sort | sed -n '1,240p'
exit \""'$status"'
