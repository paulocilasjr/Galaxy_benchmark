/usr/bin/bash -lc 'workdir=$(mktemp -d /workspace/amr_work.XXXXXX)
printf '"'%s\\n' \""'$workdir" > /workspace/.amr_work_path
GIT_TERMINAL_PROMPT=0 git clone --quiet https://bitbucket.org/genomicepidemiology/resfinder_db.git "$workdir/resfinder_db"
GIT_TERMINAL_PROMPT=0 git clone --quiet https://bitbucket.org/genomicepidemiology/pointfinder_db.git "$workdir/pointfinder_db"
GIT_TERMINAL_PROMPT=0 git clone --quiet https://bitbucket.org/genomicepidemiology/resfinder.git "$workdir/resfinder"
printf '"'%s\\n' '--- revisions ---'
for repo in resfinder_db pointfinder_db resfinder; do
  printf '%s\\t' \""'$repo"
  git -C "$workdir/$repo" rev-parse HEAD
  git -C "$workdir/$repo" show-ref | sed -n '"'1,10p'
done
printf '%s\\n' '--- target commits ---'
for repo_commit in 'resfinder_db d1e607b' 'pointfinder_db 694919f'; do
  set -- "'$repo_commit
  printf '"'%s\\t' \""'$1"
  git -C "$workdir/$1" cat-file -t "$2" 2>&1 || true
done
printf '"'%s\\n' '--- top-level inventories ---'
find \""'$workdir/resfinder_db" -maxdepth 2 -type f | sort | sed -n '"'1,160p'
find \""'$workdir/pointfinder_db" -maxdepth 3 -type f | sort | sed -n '"'1,160p'
printf '%s\\n' '--- resfinder inventory ---'
find \""'$workdir/resfinder" -maxdepth 2 -type f | sort | sed -n '"'1,200p'"
