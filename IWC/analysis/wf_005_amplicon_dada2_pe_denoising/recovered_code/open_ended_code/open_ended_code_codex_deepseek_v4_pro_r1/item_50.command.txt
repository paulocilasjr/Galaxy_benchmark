/usr/bin/bash -c "python3 - <<'PY'
import csv, re, sys, os
p='final_answer/asv_abundance.tsv'
expected=['mFMT_cecal_1191_2','mFMT_cecal_1191_3','mFMT_cecal_1275_1','mFMT_cecal_1275_2','mFMT_cecal_1275_4','hFMT_cecal_1296_1','hFMT_cecal_1296_2','hFMT_cecal_1296_3','hFMT_cecal_1297_1','hFMT_cecal_1297_2','hFMT_cecal_1297_4','noFMT_cecal_1192_1','noFMT_cecal_1192_3','noFMT_cecal_1294_1','noFMT_cecal_1294_2','noFMT_cecal_1294_3','noFMT_cecal_1295_4']
with open(p, newline='') as f:
    rows=list(csv.reader(f, delimiter='\\t'))
hdr=rows[0]
assert hdr[0]=='sequence'
assert hdr[1:]==expected, (hdr[1:], expected)
seqs=[]
for i,row in enumerate(rows[1:],1):

    seqs.append(seq)
    assert re.fullmatch(r'[ACGT]+', seq), f'bad sequence row {i}'
    for j,v in enumerate(row[1:],1):
        assert re.fullmatch(r'\\d+', v), (i,j,v)
        assert int(v)>=0
    assert sum(map(int,row[1:]))>0
assert len(seqs)==len(set(seqs))
print('PASS')
print('rows', len(seqs))
print('columns', len(hdr)-1)
print('total_counts', sum(sum(map(int,r[1:])) for r in rows[1:]))
print('seq_lengths', sorted(set(map(len,seqs))))
print('samples', len(expected))
PY"
