from pathlib import Path
import gzip, statistics
samples=[]
for d in sorted(Path('data/inputs/paired_input_data').iterdir()):
    if d.is_dir(): samples.append(d)
for mate in ['forward','reverse']:
    lens=[]; qsums=[]; qcounts=[]; reads=0
    # gather all reads, data is small enough
    for d in samples:
        fp=d/mate/(f'{mate}.fastqsanger.gz')
        with gzip.open(fp,'rt') as fh:
            while True:
                h=fh.readline()
                if not h: break
                seq=fh.readline().strip(); fh.readline(); qual=fh.readline().strip()
                reads+=1; lens.append(len(seq))
                if len(qsums)<len(qual):
                    qsums.extend([0]*(len(qual)-len(qsums))); qcounts.extend([0]*(len(qual)-len(qcounts)))
                for i,ch in enumerate(qual):
                    qsums[i]+=ord(ch)-33; qcounts[i]+=1
    means=[qsums[i]/qcounts[i] for i in range(len(qsums))]
    print(mate, 'reads', reads, 'lengths', sorted(set(lens))[:10], 'n_lengths', len(set(lens)))
    for pos in [1,100,150,200,220,230,240,250]:
        if pos <= len(means): print(mate, pos, round(means[pos-1],2))
