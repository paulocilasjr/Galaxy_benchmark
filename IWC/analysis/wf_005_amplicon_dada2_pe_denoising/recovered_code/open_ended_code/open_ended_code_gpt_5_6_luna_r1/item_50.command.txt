/usr/bin/bash -lc "/workspace/.denoise/bin/Rscript - <<'RS'
suppressPackageStartupMessages(library(dada2))
f <- 'data/inputs/paired_input_data/hfmt_cecal_1296_1/forward/forward.fastqsanger.gz'; r <- 'data/inputs/paired_input_data/hfmt_cecal_1296_1/reverse/reverse.fastqsanger.gz'
for (m in c(TRUE,FALSE)) {
  ff <- paste0('work/one_F_',m,'.fastq.gz'); rr <- paste0('work/one_R_',m,'.fastq.gz')
  x <- tryCatch(filterAndTrim(f,ff,r,rr,truncLen=c(240,200),maxN=0,maxEE=c(2,2),truncQ=2,rm.phix=FALSE,compress=TRUE,multithread=TRUE,matchIDs=m,verbose=TRUE),error=function(e)e)
  print(m); print(x)
}
RS
ls -lh work/one_* 2>/dev/null || true"
