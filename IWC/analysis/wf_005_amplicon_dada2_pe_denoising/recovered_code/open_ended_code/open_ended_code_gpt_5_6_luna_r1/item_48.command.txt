/usr/bin/bash -lc "/workspace/.denoise/bin/Rscript - <<'RS'
suppressPackageStartupMessages(library(dada2))
f <- 'data/inputs/paired_input_data/hfmt_cecal_1296_1/forward/forward.fastqsanger.gz'
for (qt in c('Auto','Sanger','Illumina')) {
  out <- paste0('work/test_', qt, '.fastq.gz')
  x <- tryCatch(fastqFilter(f, out, truncLen=240, maxN=0, maxEE=Inf, truncQ=0, rm.phix=FALSE, compress=TRUE, verbose=TRUE, qualityType=qt), error=function(e)e)
  print(qt); print(x)
}
RS
ls -l work/test_* 2>/dev/null || true"
