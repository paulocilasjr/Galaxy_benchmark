/usr/bin/bash -lc "mkdir -p work/testfilter; /workspace/.mamba/envs/dada2/bin/Rscript - <<'RS'
library(dada2)
f <- 'data/inputs/paired_input_data/mfmt_cecal_1191_2/forward/forward.fastqsanger.gz'
r <- 'data/inputs/paired_input_data/mfmt_cecal_1191_2/reverse/reverse.fastqsanger.gz'
for (m in c(FALSE,TRUE)) {
  tag <- if (m) 'true' else 'false'
  o <- filterAndTrim(f,paste0('work/testfilter/f_',tag,'.fastq.gz'),r,paste0('work/testfilter/r_',tag,'.fastq.gz'),truncLen=c(0,0),truncQ=2,maxEE=c(2,2),maxN=0,minLen=20,rm.phix=FALSE,matchIDs=m,multithread=FALSE,verbose=TRUE)
  cat('matchIDs',m,'\\n'); print(o)
}
RS
ls -lh work/testfilter"
