/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
from pathlib import Path
import csv, collections
p=Path('work/dada2_run/asv_abundance.tsv')
with p.open() as h:
    rd=csv.reader(h,delimiter='\\t'); hdr=next(rd); rows=list(rd)
print('header',hdr)
print('rows',len(rows),'ncols',len(hdr))
lengths=collections.Counter(map(lambda x:len(x[0]), rows)); print('lengths',dict(sorted(lengths.items())))
print('alphabet',sorted(set(''.join(x[0] for x in rows))))
print('unique seq',len({x[0] for x in rows}), 'dups',len(rows)-len({x[0] for x in rows}))
print('totals',dict(zip(hdr[1:], [sum(int(r[i]) for r in rows) for i in range(1,len(hdr))])))
print('nonzero ASVs per sample',dict(zip(hdr[1:], [sum(int(r[i])>0 for r in rows) for i in range(1,len(hdr))])))
print('max count',max(int(x) for r in rows for x in r[1:]))
PY"
