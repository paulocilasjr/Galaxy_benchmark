/usr/bin/bash -c "python3 - <<'PY'
import gzip, json
s=json.load(open('qc_summary.json'))
comp=str.maketrans('ACGT','TGCA')
def rc(x): return x.translate(comp)[::-1]
for sample in ['mfmt_cecal_1191_2','hfmt_cecal_1296_1']:
    f=s[sample+':forward']['first_seqs'][0]
    r=s[sample+':reverse']['first_seqs'][0]
    rr=rc(r)
    print(sample, 'lens',len(f),len(rr))
    # find overlap exact between f suffix and rr prefix
    for k in range(min(len(f),len(rr)),0,-1):
        if f[-k:]==rr[:k]:
            print('overlap',k)
            break
    print('f:',f)
    print('rc(r):',rr)
PY"
