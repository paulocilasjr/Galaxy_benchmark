/usr/bin/bash -lc "/opt/conda/bin/python3.11 - <<'PY'
import h5py, numpy as np, pandas as pd
p='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
def dec(x):
    if isinstance(x, bytes): return x.decode()
    return x
with h5py.File(p,'r') as f:
    for k in ['obs/_index','var/gene_symbol','var/ensembl_gene_id']:
        a=f[k][()]
        print(k, a.shape, a.dtype, [dec(x) for x in a[:10]])
    for col in ['cell_type','disease','individual','sex']:
        cat=[dec(x) for x in f[f'obs/{col}/categories'][()]]
        codes=f[f'obs/{col}/codes'][()]
        vals=np.array(cat,dtype=object)[codes]
        print('\\n',col,'categories=',cat)
        print(pd.Series(vals).value_counts().to_string())
    data=f['layers/counts/data'][()]
    print('\\ncounts data:', data.dtype, data.shape, 'min', data.min(), 'max', data.max(), 'integer?', np.all(data==np.floor(data)), 'sum', data.sum(), 'nonzero', len(data))
    indptr=f['layers/counts/indptr'][()]
    totals=np.add.reduceat(data, indptr[:-1])
    totals[indptr[:-1]==indptr[1:]]=0
    print('per-cell totals min/max/median',totals.min(),totals.max(),np.median(totals),'zeros',np.sum(totals==0))
PY"
