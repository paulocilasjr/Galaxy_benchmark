/usr/bin/bash -lc "/opt/conda/bin/python3.11 -u - <<'PY'
from pathlib import Path
print('start',flush=True)
import h5py, numpy as np, pandas as pd, math, json
from scipy.sparse import csr_matrix
print('imports',flush=True)
def dec(a): return np.asarray([x.decode() if isinstance(x,bytes) else str(x) for x in a], dtype=object)
def clean(x): return x.replace(' ','_').replace('-','_')
src='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
with h5py.File(src,'r') as f:
 print('open',flush=True)
 ctcat=dec(f['obs/cell_type/categories'][()]); dcat=dec(f['obs/disease/categories'][()]); icat=dec(f['obs/individual/categories'][()]); print('cats',flush=True)
 ct=ctcat[f['obs/cell_type/codes'][()]]; disease=dcat[f['obs/disease/codes'][()]]; ind=icat[f['obs/individual/codes'][()]]; genes=dec(f['var/gene_symbol'][()]); print('meta',flush=True)
 raw=f['layers/counts/data'][()]; X=csr_matrix((raw.astype(np.int64),f['layers/counts/indices'][()],f['layers/counts/indptr'][()]),shape=(len(ind),len(genes))); print('X',X.nnz,flush=True)
meta=pd.DataFrame({'individual':ind,'cell_type':ct,'disease':disease}); combos=list(meta.groupby(['individual','cell_type'],sort=False,observed=True).size().loc[lambda s:s>=10].index); print('combos',len(combos),flush=True)
cols=[clean(i)+'_'+clean(c) for i,c in combos]; PB=np.vstack([np.asarray(X[(meta.individual==i)&(meta.cell_type==c)].sum(axis=0)).ravel() for i,c in combos]).astype(np.int64); print('PB',PB.shape,PB.sum(),flush=True)
tot=PB.sum(axis=1); M=float(np.median(tot)); S=math.ceil(10+0.7*(len(combos)-10)); threshold=10/M*1_000_000; mask=((PB/tot[:,None]*1_000_000)>=threshold).sum(axis=0)>=S; mask &= PB.sum(axis=0)>=1000; expected_genes=genes[mask]; expected=PB[:,mask].T; print('expected',len(expected_genes),flush=True)
pbt=pd.read_csv('final_answer/pseudobulk_counts.tsv',sep='\\t'); print('pbt',pbt.shape,flush=True)
print('cols equal',list(pbt.columns)==['gene']+cols, 'genes equal',pbt.gene.tolist()==expected_genes.tolist(), 'vals equal',np.array_equal(pbt.iloc[:,1:].to_numpy(),expected),flush=True)
det=pd.read_csv('final_answer/differential_expression.tsv',sep='\\t'); print('det',det.shape, flush=True)
p=det.p_value.to_numpy(float); order=np.argsort(p,kind='mergesort'); q=p[order]*len(p)/np.arange(1,len(p)+1); q=np.minimum.accumulate(q[::-1])[::-1]; q=np.clip(q,0,1); eq=np.empty_like(p); eq[order]=q; print('bh',np.allclose(det.fdr.to_numpy(float),eq,rtol=0,atol=1e-14),flush=True)
print('PASS',flush=True)
PY
