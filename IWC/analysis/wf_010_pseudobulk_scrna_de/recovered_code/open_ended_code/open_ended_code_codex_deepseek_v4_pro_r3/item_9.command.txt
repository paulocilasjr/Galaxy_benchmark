/usr/bin/bash -c "python - <<'PY'
import h5py, json
p='/workspace/data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'

    print('root', list(f.keys()))
    print('obs', list(f['obs'].keys()))
    print('var', list(f['var'].keys()))
    for k in f['obs'].keys():
        ds=f['obs'][k]
        print('obs',k, ds.shape, ds.dtype, ds[:10] if ds.dtype.kind in 'OUS' else ds[:10])
    for k in f['var'].keys():
        ds=f['var'][k]
        print('var',k, ds.shape, ds.dtype, ds[:10] if ds.dtype.kind in 'OUS' else ds[:10])
    print('X attrs', dict(f['X'].attrs))
    X=f['X']
    print('X type', type(X), X.shape, X.dtype)
    print('layers', list(f['layers'].keys()) if 'layers' in f else None)
    for k in (f['layers'].keys() if 'layers' in f else []):
        ds=f['layers'][k]
        print('layer',k, type(ds), ds.shape, ds.dtype, dict(ds.attrs) if hasattr(ds,'attrs') else None)
    print('obs _index?', f['obs'].keys())
PY"
