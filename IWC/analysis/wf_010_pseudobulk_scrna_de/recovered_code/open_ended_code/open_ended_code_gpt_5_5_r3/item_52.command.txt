import json
import numpy as np
import pandas as pd
pb = pd.read_csv('final_answer/pseudobulk_counts.tsv', sep='\\t')

print('counts integer', bool(np.equal(vals, np.rint(vals)).all()))
print('lfc finite', bool(np.isfinite(de['log2_fold_change']).all()))
print('p valid', bool(((de['p_value'] >= 0) & (de['p_value'] <= 1) & np.isfinite(de['p_value'])).all()))
print('fdr valid', bool(((de['fdr'] >= 0) & (de['fdr'] <= 1) & np.isfinite(de['fdr'])).all()))
print('method', json.load(open('final_answer/method.json')))

print('fdr min max', float(de['fdr'].min()), float(de['fdr'].max()))
