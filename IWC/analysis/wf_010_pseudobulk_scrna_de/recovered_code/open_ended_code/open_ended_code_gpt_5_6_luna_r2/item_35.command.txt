import h5py, numpy as np, pandas as pd
from scipy.sparse import csr_matrix
p='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
def dec(d):
 a=d[()]
 return [x.decode() if isinstance(x,(bytes,np.bytes_)) else str(x) for x in a]
def san(x): return x.replace(' ','_').replace('-','_')
with h5py.File(p,'r') as f:
 g=f['layers/counts']; shape=tuple(map(int,g.attrs['shape']))
 X=csr_matrix((g['data'][()].astype(np.int64),g['indices'][()],g['indptr'][()]),shape=shape)
 inds=dec(f['obs/individual/categories']); ic=f['obs/individual/codes'][()].astype(int)
 cts=dec(f['obs/cell_type/categories']); cc=f['obs/cell_type/codes'][()].astype(int)
