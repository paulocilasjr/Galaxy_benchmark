/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,statistics
final=set(r['peptide'] for r in csv.DictReader(open('final_answer/verified_peptides.tsv'),delimiter='\\t'))
peps={}
with open('data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular') as f:
 for r in csv.DictReader(f,delimiter='\\t'):
  seq=r['Sequence']
  if seq not in peps or float(r['PEP'])<peps[seq]: peps[seq]=float(r['PEP'])
vals=[peps[p] for p in final if p in peps]
print('final MQ PEP max',max(vals), 'median',statistics.median(vals), 'n',len(vals))
for p,v in sorted(peps.items(), key=lambda x:-x[1])[:20]:
 if p in final: print('final high',p,v)
PY"
