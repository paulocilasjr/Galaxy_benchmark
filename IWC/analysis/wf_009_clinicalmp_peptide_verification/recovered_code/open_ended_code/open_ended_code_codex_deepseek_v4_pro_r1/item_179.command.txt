/usr/bin/bash -lc "micromamba run -n pepquery python3 - <<'PY'
import json
p='run_trace/analysis_steps.jsonl'
objs=[]
with open(p) as f:
    for i,line in enumerate(f,1):
        line=line.strip()
        if not line: continue
        try: o=json.loads(line)
        except Exception as e: print('bad',i,e); raise
        req={'stage','status','tool_or_command','effective_parameters','inputs','outputs','validation'}
        assert req <= o.keys(), (i,req-o.keys())
        objs.append(o)
print('jsonl objects',len(objs))
success=[o['stage'] for o in objs if o.get('successful_chain')]
failed=[o['stage'] for o in objs if o['status']=='failed']
print('successful_chain',success)
print('failed',failed)
PY"
