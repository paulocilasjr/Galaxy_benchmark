from pathlib import Path
p=Path('final_answer/verified_peptides.tsv')
lines=p.read_text().splitlines()
print('lines',len(lines))
assert lines[0]=='peptide\\tprotein_id'
seen=set(); bad=[]
for i,line in enumerate(lines[1:],1):
    cols=line.split('\\t')
    if len(cols)"'!=2 or not all(cols): bad.append((i,line)); continue
    pep,pid=cols
    if pep!=pep.upper() or any(c in pep for c in '"'[]()_.:+-* '): bad.append((i,'badpep',line))
    if (pep,pid) in seen: bad.append((i,'dup',line))
    seen.add((pep,pid))
print('unique pairs',len(seen),'bad',bad[:10])
# verified peptide count from psm files
from pathlib import Path as P
peps=set()
for f in sorted(P('work').glob('pepquery_out/F*_nodb_5g/psm_rank.txt')):
    with open(f) as fh:
        h=fh.readline().rstrip('\\n').split('\\t'); ci=h.index('confident'); pi=h.index('peptide')
        for line in fh:
            c=line.rstrip('\\n').split('\\t')
            if c[ci]=='Yes': peps.add(c[pi])
print('verified count',len(peps),'rows peptides',len({x[0] for x in seen}))
assert len(peps)==len({x[0] for x in seen})
PY"
