def readfa(p):
    d={}; order=[]; h=None; s=[]
    with open(p) as f:
      for line in f:
        if line.startswith('>'):
          if h is not None:
            d[s]=h; order.append((h,s))
          h=line.rstrip('\\n'); s=''
        else: s+=line.strip()
      if h is not None: d[s]=h; order.append((h,s))
    return d, order
