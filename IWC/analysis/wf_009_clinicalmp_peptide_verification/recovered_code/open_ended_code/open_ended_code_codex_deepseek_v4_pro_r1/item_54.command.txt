/usr/bin/bash -lc "micromamba run -n pepquery python3 - <<'PY'
import zipfile,re
p='/home/benchmark/.local/share/mamba/envs/pepquery/share/pepquery-2.0.2-0/pepquery-2.0.2.jar'
with zipfile.ZipFile(p) as z:
  data=z.read('main/java/pg/PeptideSearchMT.class')
  s=re.findall(rb'[\\x20-\\x7e]{2,}', data)
  text=b' '.join(s).decode('latin1','ignore')
  # find all option-like strings in help
  for pat in ['do_modification_search_validation','do_modification','um ','-um','Unrestricted','restriction','addAAsubstitution']:
    starts=[m.start() for m in re.finditer(re.escape(pat), text, re.I)]
    print('\\nPAT',pat,len(starts))
    for st in starts[:10]:
      print(text[max(0,st-500):st+1000].replace('  ',' ')[:1500])
PY"
