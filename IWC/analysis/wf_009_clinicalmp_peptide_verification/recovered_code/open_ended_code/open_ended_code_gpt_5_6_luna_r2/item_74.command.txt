import csv,collections,re
p='/workspace/analysis_work/pq_all_n100_fast/psm_rank.txt'
rows=list(csv.DictReader(open(p),delimiter='\\t'))
ver=sorted({r['peptide'] for r in rows if r['confident']=='Yes'})
print('verified unique',len(ver))
print('\\n'.join(ver))
# summarize candidate output mapping
mq={}
for r in csv.DictReader(open('data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular'),delimiter='\\t'):
 if r['Sequence'] in ver: mq.setdefault(r['Sequence'],[]).append(r)
sg={}
for r in csv.DictReader(open('data/inputs/sgps_peptide_report/SGPS_peptide-report.tabular'),delimiter='\\t'):
 if r['Sequence'] in ver: sg.setdefault(r['Sequence'],[]).append(r)
print('--- mapping summary')
for x in ver:
 prots=[]
 for r in mq.get(x,[]): prots.extend(r['Proteins'].split(';') if r['Proteins'] else [])
 for r in sg.get(x,[]): prots.extend(r['Protein(s)'].split(','))
 prots=[p for p in prots if p]
 print(x,'MQrows',len(mq.get(x,[])),'SGrows',len(sg.get(x,[])),'proteins',sorted(set(prots)))
print('missing assignment', [x for x in ver if not any((r['Proteins'] for r in mq.get(x,[]))) and not sg.get(x)])
PY"
