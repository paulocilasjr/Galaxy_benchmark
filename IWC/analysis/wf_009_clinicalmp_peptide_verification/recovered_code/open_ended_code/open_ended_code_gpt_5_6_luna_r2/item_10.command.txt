mq='data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular'

print('SG',len(rows2),'matching',sum(r['Sequence'] in D for r in rows2),'unique',len(set(r['Sequence'] for r in rows2 if r['Sequence'] in D)))
for r in rows2:
 if r['Sequence'] in D:
  print(r['Sequence'], '\\t',r['Protein(s)'],'\\t',r['Protein Group(s)'],'\\t',r['Validation'],'\\t',r['#Validated PSMs'])
