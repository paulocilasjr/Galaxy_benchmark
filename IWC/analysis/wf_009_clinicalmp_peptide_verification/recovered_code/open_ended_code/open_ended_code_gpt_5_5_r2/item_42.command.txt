/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,re
path='final_answer/verified_peptides.tsv'
rows=list(csv.DictReader(open(path),delimiter='\\t'))
print('columns',rows and list(rows[0].keys()))
print('bad_peptides',[r['peptide'] for r in rows if not re.fullmatch(r'[A-Z]+',r['peptide'])][:10])
print('blank_fields',sum(1 for r in rows if not r['peptide'] or not r['protein_id']))
print('duplicate_rows',len(rows)-len({(r['peptide'],r['protein_id']) for r in rows}))
PY"
