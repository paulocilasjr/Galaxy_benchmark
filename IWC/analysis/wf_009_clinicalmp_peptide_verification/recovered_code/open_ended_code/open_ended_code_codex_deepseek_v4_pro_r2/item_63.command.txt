/usr/bin/bash -lc "micromamba run -n pepq python - <<'PY'
from pathlib import Path
import re
exp=Path('work/expected_report_pairs.tabular').read_text().splitlines()
pairs=[]
for line in exp[1:]:
    p,prot=line.split('\\t',1)
    # workflow normalization
    for item in prot.split(';'):
        item=re.sub(r'(tr|sp)[|]','',item)
        item=re.sub(r'[ ]+','',item)
        item=re.sub(r'[|].*"'$'"','',item)
        for sub in item.split(','):
            if sub:
                pairs.append((p,sub))
# distinct preserving order
seen=set(); uniq=[]
for p,pid in pairs:
    if (p,pid) not in seen:
        seen.add((p,pid)); uniq.append((p,pid))
print('normalized pairs',len(pairs),'unique',len(uniq),'peptides',len(set(p for p,_ in uniq)),'proteins',len(set(pid for _,pid in uniq)))
print('protein ids sorted')
ids=sorted(set(pid for _,pid in uniq))
print('\\n'.join(ids))
Path('work/expected_normalized_pairs.tsv').write_text(''.join(f'{p}\\t{pid}\\n' for p,pid in uniq))
PY
wc -l work/expected_normalized_pairs.tsv; head -20 work/expected_normalized_pairs.tsv; tail -20 work/expected_normalized_pairs.tsv"
