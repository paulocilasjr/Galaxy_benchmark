/usr/bin/bash -lc "micromamba run -n pepq python - <<'PY'
from pathlib import Path
import csv
s=Path('data/inputs/sgps_peptide_report/SGPS_peptide-report.tabular')
m=Path('data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular')
rows=[]
with s.open(newline='') as f:
    r=csv.reader(f,delimiter='\\t')
    next(r,None)
    for c in r:
        if len(c)>=6:
            rows.append((c[5],c[1]))
with m.open(newline='') as f:
    r=csv.reader(f,delimiter='\\t')
    next(r,None)
    for c in r:
        if len(c)>=35:
            rows.append((c[0],c[34]))
print('report rows',len(rows))
Path('work/peptide_protein_reports.tsv').write_text(''.join(f'{p}\\t{pr}\\n' for p,pr in rows))
# count unique peptide/prot
print('unique pairs',len(set(rows)),'unique pep',len(set(p for p,_ in rows)),'unique prot raw',len(set(pr for _,pr in rows)))
# compare to fetched expected concatenated test-data maybe hash?
PY
wc -l work/peptide_protein_reports.tsv; head -5 work/peptide_protein_reports.tsv; tail -5 work/peptide_protein_reports.tsv"
