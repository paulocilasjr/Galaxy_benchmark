/usr/bin/bash -lc "micromamba run -n pepq python - <<'PY'
from pathlib import Path
files=[Path('work/human_uniprot_isoforms.fasta'),Path('data/inputs/crap/Contaminants%28cRAP%29-Protein-Database.fasta')]
seen=set(); out=Path('work/human_crap_unique.fasta')
with out.open('w') as fo:
    n=0; skipped=0
    for f in files:
        header=None; seq=[]
        def flush():
            nonlocal header,seq
            global n,skipped
            if header is not None:
                s=''.join(seq).strip().upper()
                if s and s not in seen:
                    seen.add(s)
                    fo.write(header+'\\n')
                    for i in range(0,len(s),80): fo.write(s[i:i+80]+'\\n')
                    n+=1
                elif s:
                    skipped+=1
                header=None; seq=[]
        with f.open() as fh:
            for line in fh:
                if line.startswith('>'):
                    flush(); header=line.rstrip('\\n')
                else:
                    seq.append(line.strip())
            flush()
print('unique entries',n,'duplicates skipped',skipped)
PY
grep -c '"'^>'"' work/human_crap_unique.fasta; ls -lh work/human_crap_unique.fasta"
