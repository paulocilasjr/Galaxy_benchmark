from pathlib import Path
files=[Path('work/pepquery_f10_v2/psm_rank.txt'),Path('work/pepquery_f11/psm_rank.txt'),Path('work/pepquery_f13/psm_rank.txt'),Path('work/pepquery_f15/psm_rank.txt')]

    lines=f.read_text().splitlines()

print('confident rows',len(peps),'unique peptides',len(set(peps)))
from collections import Counter
print('peptides',sorted(set(peps)))
Path('work/verified_peptides_from_pepquery.txt').write_text('\\n'.join(sorted(set(peps)))+'\\n')
# counts by file
for f in files:
    lines=f.read_text().splitlines(); header=lines[0].split('\\t'); ci=header.index('confident'); pi=header.index('peptide')
    yes=[l.split('\\t')[pi] for l in lines[1:] if l.split('\\t')[ci]=='Yes']
    print(f.name, len(yes), 'unique',len(set(yes)), sorted(set(yes)))
