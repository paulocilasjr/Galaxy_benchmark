/usr/bin/bash -lc "# derive accession sets from each report for the chosen set and show per-peptide counts
awk -F '\\t' 'NR==FNR{v["'$1]=1;next} NR>1&&($1 in v){n=split($35,a,";");for(i=1;i<=n;i++)if(a[i]~ /''^tr'"\\|/){split(a[i],b,\"|\");print "'$1 "'"\\t\" b[2] \"\\tMQ\"}}' /tmp/verified_allz3.txt data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular > /tmp/assign_mq.txt
awk -F '\\t' 'NR==FNR{v["'$1]=1;next} NR>1&&($6 in v){n=split($2,a,",");for(i=1;i<=n;i++){gsub(/''^ +| +$/,"",a[i]);print $6 "'"\\t\" a[i] \"\\tSGPS\"}}' /tmp/verified_allz3.txt data/inputs/sgps_peptide_report/SGPS_peptide-report.tabular > /tmp/assign_sgps.txt
cat /tmp/assign_mq.txt /tmp/assign_sgps.txt | sort -u | awk -F '\\t' '{k="'$1 SUBSEP $2;src[k]=src[k] (src[k]?",":"") $3; n[$1]++; seen[$1]=1} END{for(k in src){split(k,a,SUBSEP);print a[1] "'"\\t\" a[2] \"\\t\" src[k]}}' | sort -k1,1 -k2,2 | awk -F '\\t' '"'$3 ~ /MQ/ && $3 ~ /SGPS/ {next} {print}'"' | head -60; echo 'MQ assignments='"'$(sort -u /tmp/assign_mq.txt | wc -l) '"'SGPS assignments='"'$(sort -u /tmp/assign_sgps.txt | wc -l) '"'union='"'$(cat /tmp/assign_mq.txt /tmp/assign_sgps.txt | sort -u | wc -l)'
