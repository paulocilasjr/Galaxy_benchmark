/usr/bin/bash -lc "set -euo pipefail
printf 'completion_summary\\n'
for f in /workspace/work/pepquery_logs/*.log; do
  stem="'$(basename "$f" .log)
  printf '"'%s\\t' \""'$stem"
  perl -pe '"'s/\\e\\[[0-9;]*m//g' \""'$f" | awk '"'/Matched spectra:/{matched="'$NF} /Identified peptides after PTM searching:/{verified=$NF} /End'"\\./{done=\"yes\"} END{print \"matched_spectra=\" matched \"\\tverified_peptides=\" verified \"\\tcompleted=\" done}'
done
printf '\\nresult_files\\n'; find /workspace/work/pepquery_by_mgf -mindepth 2 -maxdepth 2 -type f -printf '%h/%f\\t%s\\n' | sort
printf '\\npsm examples\\n'; for f in /workspace/work/pepquery_by_mgf/*/psm.txt; do printf '\\n%s\\n' \""'$f"; awk '"'NR<=4{print} END{print \"lines=\" NR}' \""'$f"; done
printf '"'\\nparameter_diff_check\\n'; sha256sum /workspace/work/pepquery_by_mgf/*/parameter.txt"
