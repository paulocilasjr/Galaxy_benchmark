[ "$(find final_answer -maxdepth 1 -type f | wc -l)" -eq 1 ]
[ -f final_answer/verified_peptides.tsv ]
[ "$(head -n 1 final_answer/verified_peptides.tsv)" = $'"'peptide\\tprotein_id' ]
