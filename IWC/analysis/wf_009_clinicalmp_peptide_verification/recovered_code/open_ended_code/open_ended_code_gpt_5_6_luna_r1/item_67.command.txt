/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,re,collections
cs=[x.strip() for x in open('data/inputs/distinct_peptides_for_pepquery/Distinct_Peptides_for_PepQuery.tabular') if x.strip()]
by=collections.defaultdict(set); sources=collections.defaultdict(set)
for r in csv.DictReader(open('data/inputs/sgps_peptide_report/SGPS_peptide-report.tabular'),delimiter='\\t'):
 if r['Sequence'] in cs:
  for x in r['Protein(s)'].split(','):
   if x.strip(): by[r['Sequence']].add(x.strip()); sources[r['Sequence']].add('sgps')
for r in csv.DictReader(open('data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular'),delimiter='\\t'):
 if r['Sequence'] in cs:
  for x in r['Proteins'].split(';'):
   x=x.strip()
   if x and not x.startswith('REV__'):
    m=re.match(r'(?:sp|tr)\\|(["'^|]+)'"\\|',x)
    by[r['Sequence']].add(m.group(1) if m else x); sources[r['Sequence']].add('maxquant')
print('candidate with mapped microbial ids',len(by),'pairs',sum(len(x) for x in by.values()))
for p in cs:
 print(p,'\\t',';'.join(sorted(by[p])),'\\t',','.join(sorted(sources[p])))
PY"
