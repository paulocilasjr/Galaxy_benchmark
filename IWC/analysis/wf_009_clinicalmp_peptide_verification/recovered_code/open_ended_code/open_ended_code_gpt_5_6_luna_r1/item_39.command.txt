/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,collections
cs=[x.strip() for x in open('data/inputs/distinct_peptides_for_pepquery/Distinct_Peptides_for_PepQuery.tabular') if x.strip()]
rows=list(csv.DictReader(open('data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular'),delimiter='\\t'))
for p in cs:
 rr=[r for r in rows if r['Sequence']==p]
 if rr:
  r=rr[0]
  print(p,'\\t',r['Reverse'],r['Potential contaminant'],r['Proteins'],r['Leading razor protein'],r['PEP'],r['Score'])
print('statuses')
c=collections.Counter()
for p in cs:
 rr=[r for r in rows if r['Sequence']==p]
 if not rr: c['none']+=1
 else:
  r=rr[0];c[('reverse' if r['Reverse'] else 'target', 'contam' if r['Potential contaminant'] else 'notcontam','prot' if r['Proteins'] else 'no-prot')]+=1
print(c)
PY"
