import sys,re,os,bisect,math
sys.path.insert(0,'.')
from work.fragment_score import theoretical_ions,score_variant,AA,H2O,PROTON,TMT,CAM,OX
path='data/inputs/tandem_mass_spectrometry_ms_ms_datasets/ptrc_skubitz_plex2_f15_9aug19_rage_rep_19_06_08_mgf/ptrc_skubitz_plex2_f15_9aug19_rage_rep_19_06_08_mgf.mgf'

with open(path) as f:
 title=None;peaks=[];mz=None;ch=None
 for line in f:
  if line.startswith('BEGIN IONS'):title=None;peaks=[];mz=ch=None
  elif line.startswith('TITLE='):title=line.rstrip()[6:]
  elif line.startswith('PEPMASS='):mz=float(line.split('=',1)[1].split()[0])
  elif line.startswith('CHARGE='):ch=int(re.search(r'\\d+',line).group())
  elif line and (line[0].isdigit() or line[0]=='.'):
   a=line.split();peaks.append((float(a[0]),float(a[1])))

   peaks=sorted(peaks);ms=[x[0] for x in peaks]
   print('mz',mz,'charge',ch,'peaks',len(peaks),'top',sorted(peaks,key=lambda x:x[1],reverse=True)[:10])
   seq='GIYPSSANFTTDLHSLGQYIQEGLR'

      # no K/C/M
      for z in range(1,fc+1):

       ii=bisect.bisect_left(ms,t)

       if cand and min(x[0] for x in cand)<=tol:
        er,j=min(cand);mt.append((er,l,t,peaks[j][0],peaks[j][1]))
      print(name,'fc',fc,'tol',tol,'count',len(mt),'errors',sorted(round(x[0],4) for x in mt)[:20])

   print('all', ' '.join(f'{x:.4f}' for x,_ in peaks))
