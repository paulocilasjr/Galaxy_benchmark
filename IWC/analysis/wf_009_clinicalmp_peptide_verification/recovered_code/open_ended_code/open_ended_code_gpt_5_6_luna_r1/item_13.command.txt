/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv, re, collections
base='data/inputs'
with open(base+'/distinct_peptides_for_pepquery/Distinct_Peptides_for_PepQuery.tabular') as f:
    cand=[x.strip().upper() for x in f if x.strip()]
print('candidate count',len(cand),'unique',len(set(cand)))
with open(base+'/sgps_peptide_report/SGPS_peptide-report.tabular', newline='') as f:
    rows=list(csv.DictReader(f,delimiter='\\t'))
print('sgps rows',len(rows), 'fields',list(rows[0]))
by=collections.defaultdict(list)
for r in rows: by[r['Sequence'].strip().upper()].append(r)
found=[p for p in cand if p in by]
print('candidate in SGPS',len(found),'not',len(cand)-len(found))
print('not found', [p for p in cand if p not in by])
print('\\nSGPS candidate assignments:')
for p in cand:
    rr=by.get(p,[])
    if rr:
        for r in rr:
            print(p, '\\t', r['Protein(s)'], '\\t', r['Validation'], '\\t', r['#PSMs'], '\\t', r['#Validated PSMs'], '\\t', r['Modified Sequence'])
with open(base+'/maxquant_peptide_report/MaxQuant-peptide-report.tabular', newline='') as f:
    mq=list(csv.DictReader(f,delimiter='\\t'))
print('\\nMQ rows',len(mq),'fields',len(mq[0]))
mb=collections.defaultdict(list)
for r in mq: mb[r['Sequence'].strip().upper()].append(r)
foundmq=[p for p in cand if p in mb]
print('candidate in MQ',len(foundmq),'not',len(cand)-len(foundmq))
print('MQ found assignments:')
for p in foundmq:
  for r in mb[p]: print(p, '\\t', r['Proteins'], '\\t', r['Potential contaminant'], '\\t', r['Reverse'], '\\t', r['Score'], '\\t', r['PEP'], '\\t', r['Charges'])
PY"
