/usr/bin/bash -lc 'export PATH=/opt/conda/bin:$PATH; python - <<'"'PY'
from pathlib import Path
for fn,out in [('assembly/mito_hifiasm_150.bp.r_utg.gfa','assembly/mito_r_utg.fa'),('assembly/mito_hifiasm_150.bp.p_ctg.gfa','assembly/mito_p_ctg.fa')]:
    seqs=[]
    for line in Path(fn).read_text().splitlines():
        if line.startswith('S'):
            p=line.split('\\t'); seqs.append((p[1],p[2]))
    with open(out,'w') as f:
        for name,seq in seqs:
            f.write(f'>{name}\\n{seq}\\n')
    print(out, [(n,len(s)) for n,s in seqs])
PY
