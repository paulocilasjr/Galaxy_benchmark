/usr/bin/bash -lc 'PATH=/workspace/tools/mitoenv/bin:$PATH /workspace/tools/mitoenv/bin/minimap2 -x map-hifi -t4 work/kmc/unitigs_ci300_cx600.fasta data/inputs/collection_of_pacbio_data/err13917223/err13917223.fastqsanger.gz > work/reads/raw_vs_unitigs_ci300_cx600.paf 2> work/logs/raw_vs_unitigs_ci300_cx600.log
/workspace/tools/mitoenv/bin/python - <<'"'PY'
from pathlib import Path
from collections import Counter,defaultdict
ids=set(); unit=Counter(); rows=0; kept=0
for line in Path('work/reads/raw_vs_unitigs_ci300_cx600.paf').open():
    f=line.split('\\t')
    if len(f)<12: continue
    rows+=1
    q,ql,qs,qe,strand,t,tl,ts,te,match,alen,mapq=f[:12]
    match=int(match); alen=int(alen); mapq=int(mapq)
    ident=match/alen if alen else 0
    if alen>=1000 and ident>=0.97 and mapq>=20:
        ids.add(q); unit[t]+=1; kept+=1
Path('work/reads/unitigs_ci300_cx600_read_ids.txt').write_text('\\n'.join(sorted(ids))+'\\n')
print('paf_rows',rows,'kept_alignments',kept,'selected_ids',len(ids),'unitigs_hit',len(unit))
for u,c in unit.most_common(30): print(u,c)
PY
/workspace/tools/mitoenv/bin/seqkit grep -f work/reads/unitigs_ci300_cx600_read_ids.txt data/inputs/collection_of_pacbio_data/err13917223/err13917223.fastqsanger.gz | /workspace/tools/mitoenv/bin/seqkit fq2fa > work/reads/unitigs_ci300_cx600_reads.fasta
/workspace/tools/mitoenv/bin/seqkit stats -a work/reads/unitigs_ci300_cx600_reads.fasta > work/reads/unitigs_ci300_cx600_reads.stats.tsv
cat work/reads/unitigs_ci300_cx600_reads.stats.tsv"
