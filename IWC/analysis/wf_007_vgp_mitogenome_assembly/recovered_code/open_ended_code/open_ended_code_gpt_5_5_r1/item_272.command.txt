from pathlib import Path
from collections import defaultdict,Counter,deque
# load unitig metadata from fasta headers
meta={}
for line in Path('work/kmc/unitigs_ci300_cx600.fasta').open():
    if line.startswith('>'):
        parts=line[1:].split()
        uid=parts[0]
        d={}
        for p in parts[1:]:
            if '=' in p:
                k,v=p.split('=',1); d[k]=v

byread=defaultdict(list)
for line in Path('work/reads/raw_vs_unitigs_ci300_cx600.paf').open():
    f=line.split('\\t')
    if len(f)<12: continue
    q,ql,qs,qe,strand,t,tl,ts,te,match,alen,mapq=f[:12]
    ql=int(ql); qs=int(qs); qe=int(qe); tl=int(tl); ts=int(ts); te=int(te); match=int(match); alen=int(alen); mapq=int(mapq)
    ident=match/alen if alen else 0
    if alen>=500 and ident>=0.97 and mapq>=20:
        byread[q].append((qs,qe,strand,t,tl,ts,te,alen,ident))
# cooccurrence for reads with >=2 unitigs
edges=Counter(); directed=Counter(); reads_multi=0
for r,als in byread.items():
    # keep best aln per target to avoid duplicates
    best={}

        if a[3] not in best or a[7]>best[a[3]][7]: best[a[3]]=a

for k,c in directed.most_common(80): print(k,c)
