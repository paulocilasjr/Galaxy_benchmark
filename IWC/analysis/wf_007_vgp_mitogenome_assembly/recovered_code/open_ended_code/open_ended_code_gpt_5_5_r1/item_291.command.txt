from pathlib import Path
from collections import defaultdict,deque,Counter
COMP=str.maketrans('ACGT','TGCA')
def rc(s): return s.translate(COMP)[::-1]
kmers=[]; counts={}
for line in Path('work/kmc/kmers_ci300_cx600.txt').open():
    k,c=line.split(); c=int(c); kmers.append(k); counts[k]=c
adj=defaultdict(set); edge_count=0
for k in kmers:
    for s in (k, rc(k)):
        a=s[:-1]; b=s[1:]
        adj[a].add(b); adj[b].add(a); edge_count+=1
seen=set(); comps=[]
for n in adj:
    if n in seen: continue
    q=deque([n]); seen.add(n); nodes=[]; edges=0
    while q:
        x=q.popleft(); nodes.append(x); edges += len(adj[x])
        for y in adj[x]:
            if y not in seen: seen.add(y); q.append(y)
    comps.append((len(nodes),edges//2,nodes[0]))
print('kmers',len(kmers),'nodes',len(adj),'components',len(comps))
for c in sorted(comps, reverse=True)[:30]: print(c)
