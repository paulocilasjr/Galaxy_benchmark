/usr/bin/bash -lc "/workspace/tools/mitoenv/bin/python - <<'PY'
from pathlib import Path
comp=set('unitig_5 unitig_7 unitig_8 unitig_11 unitig_13 unitig_15 unitig_19 unitig_26 unitig_27 unitig_38 unitig_40 unitig_56'.split())
# component fasta
out=[]; keep=False
for line in Path('work/kmc/unitigs_ci300_cx600.fasta').read_text().splitlines():
    if line.startswith('>'):
        keep=line[1:].split()[0] in comp
    if keep: out.append(line)
Path('work/kmc/mito_component_unitigs.fasta').write_text('\\n'.join(out)+'\\n')
ids=set(); unit_counts={u:0 for u in comp}
for line in Path('work/reads/raw_vs_unitigs_ci300_cx600.paf').open():
    f=line.split('\\t')
    if len(f)<12: continue
    q,ql,qs,qe,strand,t,tl,ts,te,match,alen,mapq=f[:12]
    match=int(match); alen=int(alen); mapq=int(mapq)
    ident=match/alen if alen else 0
    if t in comp and alen>=500 and ident>=0.97 and mapq>=20:
        ids.add(q); unit_counts[t]+=1
Path('work/reads/mito_component_read_ids.txt').write_text('\\n'.join(sorted(ids))+'\\n')
print('component_ids',len(ids))
print(unit_counts)
PY
/workspace/tools/mitoenv/bin/seqkit grep -f work/reads/mito_component_read_ids.txt data/inputs/collection_of_pacbio_data/err13917223/err13917223.fastqsanger.gz | /workspace/tools/mitoenv/bin/seqkit fq2fa > work/reads/mito_component_reads.fasta
/workspace/tools/mitoenv/bin/seqkit stats -a work/kmc/mito_component_unitigs.fasta work/reads/mito_component_reads.fasta > work/reads/mito_component.stats.tsv
cat work/reads/mito_component.stats.tsv"
