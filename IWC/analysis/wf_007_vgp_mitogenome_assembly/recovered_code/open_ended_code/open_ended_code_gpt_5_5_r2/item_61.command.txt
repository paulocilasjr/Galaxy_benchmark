/usr/bin/bash -lc "micromamba run -p /workspace/.micromamba_env python - <<'PY'
from pathlib import Path
hist=[]
for line in Path('work/kmc/agrius_k31.hist').read_text().splitlines():
    c,n=map(int,line.split())
    hist.append((c,n))
print('bins',len(hist),'min',hist[0],'max',hist[-1])
for lo,hi in [(3,10),(11,20),(21,40),(41,80),(81,160),(161,320),(321,640),(641,1280),(1281,2560),(2561,5120),(5121,10000)]:
    uniq=sum(n for c,n in hist if lo<=c<=hi)
    total=sum(c*n for c,n in hist if lo<=c<=hi)
    print(lo,hi,uniq,total)
# local maxima above 50
peaks=[]
for i in range(1,len(hist)-1):
    if hist[i][0] > 50 and hist[i][1] >= hist[i-1][1] and hist[i][1] >= hist[i+1][1]:
        peaks.append(hist[i])
print('top peaks >50 by num kmers')
for c,n in sorted(peaks,key=lambda x:x[1], reverse=True)[:20]:
    print(c,n)
print('tail nonzero bins')
for c,n in hist[-100:]:
    if n:
        print(c,n)
PY"
