/usr/bin/bash -lc 'set +e
for t in hifiasm flye canu miniasm minimap2 samtools seqkit blastn makeblastdb meryl jellyfish fastqc bcftools wtdbg2 raven racon; do if command -v "$t" >/dev/null 2>&1; then printf '"'%-12s ' \""'$t"; ("$t" --version 2>&1 || "$t" -V 2>&1 || true) | head -n 1; else printf '"'%-12s absent\\n' \""'$t"; fi; done
echo '"'--- python ---'
python3 --version
for t in conda mamba micromamba pip3 apt-get; do command -v \""'$t" || true; done
echo '"'--- resources ---'
nproc; free -h; df -h ."
