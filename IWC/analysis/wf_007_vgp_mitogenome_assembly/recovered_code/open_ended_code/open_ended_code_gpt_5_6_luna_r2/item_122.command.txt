/usr/bin/bash -lc "/workspace/mito_env/bin/seqkit subseq -r 1:15448 work/mt_unit_cons2_bayes.fasta | awk 'NR==1{print \">Agrius_convolvuli_mitochondrion\"; next}{print}' > work/final_unit_bayes.fasta; /workspace/mito_env/bin/seqkit seq -w 0 work/final_unit_bayes.fasta | awk 'NR==1{print;next}{print "'$0$0}'"' > work/final_unit_bayes2.fasta; PATH=/workspace/mito_env/bin:"'$PATH /workspace/mito_env/bin/minimap2 -x map-hifi --secondary=no -t8 -a work/final_unit_bayes2.fasta work/mt_reads.fastq | /workspace/mito_env/bin/samtools view -b -F4 - | /workspace/mito_env/bin/samtools sort -@4 -o work/final_unit_bayes2.bam -; echo MAP_EXIT=${PIPESTATUS[*]}; /workspace/mito_env/bin/samtools index -@4 work/final_unit_bayes2.bam; /workspace/mito_env/bin/samtools flagstat work/final_unit_bayes2.bam; /workspace/mito_env/bin/samtools coverage work/final_unit_bayes2.bam; /workspace/mito_env/bin/samtools consensus -f FASTA -a -m bayesian -p --min-MQ 20 --min-BQ 20 -T work/final_unit_bayes2.fasta -o work/final_recons_bayes2.fasta work/final_unit_bayes2.bam; echo RECONS_EXIT=$?; /workspace/mito_env/bin/samtools consensus -f FASTA -a -m simple -q -p --min-MQ 20 --min-BQ 20 -T work/final_unit_bayes2.fasta -o work/final_recons_simple2.fasta work/final_unit_bayes2.bam; echo SIMPLE_RECONS_EXIT=$?; for f in work/final_unit_bayes.fasta work/final_recons_bayes2.fasta work/final_recons_simple2.fasta; do echo "### $f"; /workspace/mito_env/bin/seqkit stats -T "$f"; awk '"'BEGIN{n=0;bad=0} /"'^>/{next}{for(i=1;i<=length;i++){c=toupper(substr($0,i,1));if(c=="N")n++;if(c!~/[ACGT]/)bad++}}END{print "N",n,"bad",bad}'"' \""'$f"; done'
