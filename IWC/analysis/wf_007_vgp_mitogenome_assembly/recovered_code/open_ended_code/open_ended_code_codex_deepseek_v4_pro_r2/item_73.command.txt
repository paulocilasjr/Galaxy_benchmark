/usr/bin/bash -lc "/workspace/mitoenv/bin/python - <<'PY'
from Bio import pairwise2, SeqIO
seed = str(next(SeqIO.parse('work/mito_seed.fa','fasta')).seq)
cons = str(next(SeqIO.parse('work/polish_mito37/mito_consensus.fa','fasta')).seq)
aln = pairwise2.align.globalms(seed, cons, 2, -1, -2, -1, one_alignment_only=True)[0]
print('score', aln.score, 'seed_len', len(seed), 'cons_len', len(cons), 'aln_len', len(aln[0]))
# Print first 300 columns and last 300 columns, with positions
sa, ca = aln[0], aln[1]
for region in ['first','last']:
    start = 0 if region=='first' else max(0,len(sa)-300)
    end = 300 if region=='first' else len(sa)
    print('\\n--',region,'--')
    print('seedpos', start)
    print('S', sa[start:end])
    print('C', ca[start:end])
PY"
