/usr/bin/bash -lc "awk -F'\\t' 'FNR==NR{split("'$1,a," ");g[a[1]]=$3;next}{gc=g[$2]+0; print $0"'"\\t\"gc}' /workspace/work_mito/sample16_gc.tsv /workspace/work_mito/sample16_orfs_all.tsv > /workspace/work_mito/sample16_orfs_gc.tsv
printf '%s\\n' '--- ORF diversity by GC bin ---'
awk -F'\\t' '{gc="'$10+0; bin=int(gc/5)*5; if($9>maxu[bin])maxu[bin]=$9; if($5>maxo[bin])maxo[bin]=$5; n[bin]++} END{for(i=0;i<=75;i+=5)if(n[i])printf "GC%d-%d n=%d max_unique=%d max_orfs=%d'"\\n\",i,i+4,n[i],maxu[i],maxo[i]}' /workspace/work_mito/sample16_orfs_gc.tsv
printf '%s\\n' '--- low-GC high-diversity ORF candidates ---'
awk -F'\\t' '"'$10<35 && $9>=8 {print}'"' /workspace/work_mito/sample16_orfs_gc.tsv | sort -t"'$'"'\\t' -k9,9nr -k4,4nr | sed -n '1,120p'
printf '%s\\n' '--- global high-diversity candidates ---'
awk -F'\\t' '"'$9>=12 {print}'"' /workspace/work_mito/sample16_orfs_gc.tsv | sort -t"'$'"'\\t' -k9,9nr -k4,4nr | sed -n '1,80p'"
