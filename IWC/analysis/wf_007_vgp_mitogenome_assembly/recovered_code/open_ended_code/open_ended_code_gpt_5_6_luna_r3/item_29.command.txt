/usr/bin/bash -lc "micromamba run -p /workspace/.mt_env python - <<'PY'
from pathlib import Path

def readfa(p):
    s=''.join(x.strip() for x in Path(p).read_text().splitlines() if not x.startswith('>'))
    return s.upper()
seq=readfa('/workspace/work_mito/seed.fa')
comp=str.maketrans('ACGT','TGCA')
# invertebrate mitochondrial translation
code={}
for codon,aa in {
'F':['TTT','TTC'],'L':['TTA','TTG','CTT','CTC','CTA','CTG'],'I':['ATT','ATC','ATA'],'M':['ATG'],'V':['GTT','GTC','GTA','GTG'],
'S':['TCT','TCC','TCA','TCG','AGT','AGC','AGA','AGG'],'P':['CCT','CCC','CCA','CCG'],'T':['ACT','ACC','ACA','ACG'],
'A':['GCT','GCC','GCA','GCG'],'Y':['TAT','TAC'],'*':['TAA','TAG'],'H':['CAT','CAC'],'Q':['CAA','CAG'],
'N':['AAT','AAC'],'K':['AAA','AAG'],'D':['GAT','GAC'],'E':['GAA','GAG'],'C':['TGT','TGC'],'W':['TGA','TGG'],
'R':['CGT','CGC','CGA','CGG'],'G':['GGT','GGC','GGA','GGG']}.items():
    for c in aa: code[codon]=codon
# reverse dict easier
am={}
for aa,cods in {
'F':['TTT','TTC'],'L':['TTA','TTG','CTT','CTC','CTA','CTG'],'I':['ATT','ATC','ATA'],'M':['ATG'],'V':['GTT','GTC','GTA','GTG'],
'S':['TCT','TCC','TCA','TCG','AGT','AGC','AGA','AGG'],'P':['CCT','CCC','CCA','CCG'],'T':['ACT','ACC','ACA','ACG'],
'A':['GCT','GCC','GCA','GCG'],'Y':['TAT','TAC'],'*':['TAA','TAG'],'H':['CAT','CAC'],'Q':['CAA','CAG'],
'N':['AAT','AAC'],'K':['AAA','AAG'],'D':['GAT','GAC'],'E':['GAA','GAG'],'C':['TGT','TGC'],'W':['TGA','TGG'],
'R':['CGT','CGC','CGA','CGG'],'G':['GGT','GGC','GGA','GGG']}.items():
    for c in cods: am[c]=aa
for strand,x in [('+',seq),('-',seq.translate(comp)[::-1])]:
    hits=[]
    for f in range(3):
      aa=''.join(am.get(x[i:i+3],'X') for i in range(f,len(x)-2,3))
      start=0
      for j,a in enumerate(aa+'*'):
        if a=='*':
          if j-start>=100: hits.append((j-start,f+1,start*3+f,j*3+f,aa[start:j][:20]))
          start=j+1
    print(strand, sorted(hits,reverse=True)[:30])
PY"
