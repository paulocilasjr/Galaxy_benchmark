/usr/bin/bash -lc 'for acc in SRP637459 SRP630949; do
  echo "=== $acc ==="
  curl --http1.1 -L --fail --silent --show-error --retry 5 --retry-delay 2 "https://www.ebi.ac.uk/ena/portal/api/filereport?accession=${acc}&result=read_run&fields=run_accession%2Cfastq_ftp%2Cfastq_bytes%2Cfastq_md5&format=tsv&download=false" -o "/tmp/${acc}.ena.tsv" || true
  echo bytes=$(wc -c < "/tmp/${acc}.ena.tsv" 2>/dev/null || echo 0) lines=$(wc -l < "/tmp/${acc}.ena.tsv" 2>/dev/null || echo 0)
  sed -n '"'1,4p' \"/tmp/"'${acc}.ena.tsv" 2>/dev/null || true
 done'
