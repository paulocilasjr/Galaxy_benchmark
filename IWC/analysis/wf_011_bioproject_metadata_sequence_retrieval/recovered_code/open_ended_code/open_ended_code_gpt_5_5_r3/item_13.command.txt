for acc in "'$(sed '"'/"'^$/d'"' data/inputs/bioproject_ids/bioproject_ids.txt); do
  curl -fsSL \"https://trace.ncbi.nlm.nih.gov/Traces/sra/sra.cgi?save=efetch&db=sra&rettype=runinfo&term="'${acc}" -o "work/${acc}.runinfo.csv"
  printf '"'%s\\t%s\\n' \""'$acc" "$(($(wc -l < work/${acc}.runinfo.csv)-1))"
done'
