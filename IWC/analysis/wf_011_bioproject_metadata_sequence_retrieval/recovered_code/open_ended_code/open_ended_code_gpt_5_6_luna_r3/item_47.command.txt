import csv,gzip
from pathlib import Path
root=Path('/workspace/final_answer')
for mp in sorted((root/'metadata').glob('*.tsv')):
    with mp.open(newline='') as f:
        rows=list(csv.DictReader(f,delimiter='\\t'))
    print(mp.name)
    for row in rows:
        files=[root/'reads'/f\"{row['run_accession']}_1.fastq.gz\"]
        if row['library_layout']=='PAIRED': files.append(root/'reads'/f\"{row['run_accession']}_2.fastq.gz\")
        counts=[]; bases=[]
        for p in files:
            n=b=0
            with gzip.open(p,'rt',encoding='ascii',newline='') as f:
                while True:
                    h=f.readline()
                    if not h: break
                    s=f.readline().rstrip('\\r\\n'); f.readline(); q=f.readline().rstrip('\\r\\n')
                    n+=1; b+=len(s)
            counts.append(n); bases.append(b)
        expected=int(row['run_total_bases'])
        assert sum(bases)==expected, (row['run_accession'],sum(bases),expected)
        print(row['run_accession'], row['library_layout'], 'records='+','.join(map(str,counts)), 'bases='+str(sum(bases)))
PY"
