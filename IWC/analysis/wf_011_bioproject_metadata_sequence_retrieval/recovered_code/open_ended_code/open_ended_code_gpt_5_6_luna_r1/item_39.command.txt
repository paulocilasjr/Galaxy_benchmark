/usr/bin/bash -lc "/opt/conda/bin/python3.11 - <<'PY'
import xml.etree.ElementTree as ET
for file in ['work/ncbi_sra_PRJNA1339024.xml','work/ncbi_sra_PRJNA1354216.xml']:
 print('\\n',file)
 root=ET.parse(file).getroot()
 for p in root.findall('EXPERIMENT_PACKAGE')[:2]:
  run=p.find('RUN_SET/RUN')
  print(run.get('accession'), 'run attrs', {k:run.get(k) for k in ('total_spots','total_bases','is_public')})
  print('stats', ET.tostring(run.find('Statistics'),encoding='unicode') if run.find('Statistics') is not None else None)
  print('layout', ET.tostring(p.find('EXPERIMENT/DESIGN/LIBRARY_DESCRIPTOR/LIBRARY_LAYOUT'),encoding='unicode'))
PY"
