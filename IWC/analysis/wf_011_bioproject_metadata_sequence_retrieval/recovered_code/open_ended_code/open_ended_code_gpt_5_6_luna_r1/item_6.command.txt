/usr/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET
from pathlib import Path
for path in sorted(Path('work').glob('*.sra.xml')):
    root=ET.parse(path).getroot()
    print('\\n',path)
    print('packages',len(root.findall('EXPERIMENT_PACKAGE')))
    for p in root.findall('EXPERIMENT_PACKAGE'):
        exp=p.find('EXPERIMENT'); run=p.find('RUN_SET/RUN'); study=p.find('STUDY'); sample=p.find('SAMPLE')
        lib=exp.find('DESIGN/LIBRARY_DESCRIPTOR') if exp is not None else None
        layout=lib.find('LIBRARY_LAYOUT') if lib is not None else None
        layout_name='PAIRED' if layout is not None and layout.find('PAIRED') is not None else 'SINGLE' if layout is not None and layout.find('SINGLE') is not None else '?'
        print('\\t'.join([run.get('accession','') if run is not None else '', exp.get('accession',''), study.get('accession',''), sample.get('accession',''), next((x.text for x in sample.findall('IDENTIFIERS/EXTERNAL_ID') if x.get('namespace')=='BioSample'),''), layout_name, run.get('total_spots',''), run.get('total_bases',''), exp.findtext('PLATFORM/*/INSTRUMENT_MODEL',''), lib.findtext('LIBRARY_STRATEGY',''), lib.findtext('LIBRARY_SOURCE',''), lib.findtext('LIBRARY_SELECTION','')]))
PY"
