/usr/bin/bash -c "python - <<'PY'
import json
path='/workspace/run_trace/analysis_steps.jsonl'
final_success_stages={'trim_nextera_adapters_and_quality_filter','align_filtered_reads','filter_alignments_and_remove_duplicates','call_atac_peaks'}
out=[]
with open(path) as f:
    for line in f:
        line=line.strip()
        if not line: continue
        rec=json.loads(line)
        if rec.get('status')=='success':
            rec['successful_chain'] = rec['stage'] in final_success_stages
        out.append(rec)
with open(path,'w') as f:
    for rec in out:
        f.write(json.dumps(rec, sort_keys=True)+'\\n')
print('updated', len(out))
PY"
