/usr/bin/bash -lc "set -euo pipefail
PATH=/workspace/tools/bin:/opt/conda/bin:/usr/bin:/bin
samtools view -h -f 2 -F 2304 /workspace/- \\
| awk 'BEGIN { OFS=\"\\t\"; pending=0 }
       /"'^@/ { print; next }
       {
         if (!pending) {
           first=$0; qname=$1; chr=$3; mapq=$5+0; pending=1
         } else {
           if ($1 != qname) {
             print "ERROR: non-adjacent mates after primary proper-pair filter: " qname " vs " $1 > "/dev/stderr"
             exit 2
           }
           if (mapq >= 30 && ($5+0) >= 30 && chr != "chrM" && $3 != "chrM") {
             print first
             print $0
           }
           pending=0
         }
       }
       END {
         if (pending) {
           print "ERROR: unpaired terminal SAM record: " qname > "/dev/stderr"
           exit 3
         }
       }'"' \\
| samtools sort -n -@ 2 -m 768M -T /workspace/work/tmp/namesort - \\
| samtools fixmate -m -u - - \\
| samtools sort -@ 2 -m 768M -T /workspace/work/tmp/coordsort - \\
| samtools markdup -@ 2 -r -s -f /workspace/work/qc/markdup_stats.txt - /workspace/work/filtered.bam"
