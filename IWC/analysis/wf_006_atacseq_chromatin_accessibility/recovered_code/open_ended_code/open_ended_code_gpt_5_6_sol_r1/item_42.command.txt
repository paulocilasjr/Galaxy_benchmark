/usr/bin/bash -lc "set -euo pipefail
/opt/conda/bin/python -m json.tool /workspace/final_answer/method.json >/dev/null
cmp /workspace/final_answer/peaks.bed <(cut -f1-3 /workspace/work/peaks/atac_peaks.narrowPeak)
awk -F '\\t' 'NF "'!= 3 {bad_nf++} $2 !~ /''^[0-9]+$/ || $3 !~ /''^[0-9]+$/ {bad_num++} ($2+0) < 0 || ($3+0) <= ($2+0) {bad_span++} $1 == "chrM" {mt++} END {printf "records=%d bad_nf=%d bad_num=%d bad_span=%d chrM=%d'"\\n\", NR, bad_nf+0, bad_num+0, bad_span+0, mt+0; if (bad_nf || bad_num || bad_span || mt) exit 1}' /workspace/final_answer/peaks.bed
awk -F '\\t' '{k="'$1 FS $2 FS $3; n[k]++} END {for(k in n){u++; if(n[k]>1){dup_intervals++; extra+=n[k]-1; if(n[k]>max)max=n[k]}} printf "unique_coordinates=%d duplicated_coordinate_sets=%d extra_records_at_shared_coordinates=%d max_records_per_coordinate=%d'"\\n\",u,dup_intervals,extra,max}' /workspace/final_answer/peaks.bed
awk -F '\\t' 'NR==1{min="'$3-$2;max=min;sum=0} {w=$3-$2;if(w<min)min=w;if(w>max)max=w;sum+=w} END{printf "peak_width_min=%d peak_width_mean=%.1f peak_width_max=%d'"\\n\",min,sum/NR,max}' /workspace/final_answer/peaks.bed
awk -F '\\t' 'BEGIN{min=1e9} {if("'$9<min)min=$9} END{printf "minimum_minus_log10_q=%.5f (q=%.7f)'"\\n\",min,10"'^(-min)}'"' /workspace/work/peaks/atac_peaks.narrowPeak
printf '%s\\n' '--- deliverables ---'
find /workspace/final_answer -maxdepth 1 -type f -printf '%f\\t%s bytes\\n' | sort
printf '%s\\n' '--- checksums ---'
sha256sum /workspace/final_answer/method.json /workspace/final_answer/peaks.bed
printf '%s\\n' '--- method ---'
sed -n '1,20p' /workspace/final_answer/method.json
printf '%s\\n' '--- BED edges ---'
sed -n '1,3p' /workspace/final_answer/peaks.bed
tail -3 /workspace/final_answer/peaks.bed"
