/usr/bin/bash -lc "WORK=/tmp/atac_pipeline_srr891269_v3
printf '%s\\n' '--- final files ---'
find \""'$WORK" -maxdepth 1 -type f -printf '"'%s %p\\n' | sort -n | tail -30
printf '%s\\n' '--- final BAM/index ---'
ls -lh \""'$WORK/final_deduplicated.bam" "$WORK/final_deduplicated.bam.bai" "$WORK/coordinate_sorted.bam" "$WORK/name_sorted.bam"
printf '"'%s\\n' '--- fastp summary ---'
rg -n 'total reads:|reads passed filter:|reads failed due to|reads with adapter trimmed:|bases trimmed due to adapters:|Insert size peak|fastp v' \""'$WORK/../atac_pipeline_srr891269_v1/fastp.stderr" | sed -n '"'1,100p'
printf '%s\\n' '--- pair filter ---'
cat \""'$WORK/pair_filter.stderr"
printf '"'%s\\n' '--- duplicate stats ---'
cat \""'$WORK/markdup.stats"
printf '"'%s\\n' '--- flagstat ---'
cat \""'$WORK/final.flagstat"
printf '"'%s\\n' '--- idxstats chrM/total ---'
rg -n '"'^chrM'"\\s|"'^'"\\*\\s' \""'$WORK/final.idxstats" || true
printf '"'%s\\n' '--- MACS2 stderr/stdout ---'
sed -n '1,220p' \""'$WORK/macs2.stderr"
sed -n '"'1,120p' \""'$WORK/macs2.stdout"
printf '"'%s\\n' '--- BED validation ---'
wc -l /workspace/final_answer/peaks.bed \""'$WORK/macs2/SRR891269_peaks.narrowPeak"
sed -n '"'1,5p' /workspace/final_answer/peaks.bed
tail -5 /workspace/final_answer/peaks.bed
awk -F '\\t' 'NF < 3 || "'$1 == "" || $2 !~ /''^[0-9]+$/ || $3 !~ /''^[0-9]+$/ {bad++} END {print "bad_records=" (bad+0)}'"' /workspace/final_answer/peaks.bed
cut -f1-3 \""'$WORK/macs2/SRR891269_peaks.narrowPeak" | cmp -s - /workspace/final_answer/peaks.bed; echo "bed_projection_match=$?"'
