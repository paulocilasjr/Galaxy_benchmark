/usr/bin/bash -lc "printf '%s\\n' '--- conda tools ---'
ls -1 /opt/conda/bin | sed -n '1,240p'
printf '%s\\n' '--- conda metadata relevant ---'
find /opt/conda/conda-meta -maxdepth 1 -type f -printf '%f\\n' | rg 'bowtie|bwa|samtools|macs|fastp|cutadapt|bedtools|htslib|biopython|numpy|scipy|pysam|perl|r-base|mamba|micromamba' | sort
printf '%s\\n' '--- resources ---'
df -h /workspace /tmp /opt/conda
grep -E 'MemTotal|MemAvailable|"'^CPU'"\\(' /proc/meminfo /proc/cpuinfo 2>/dev/null | sed -n '1,20p'"
