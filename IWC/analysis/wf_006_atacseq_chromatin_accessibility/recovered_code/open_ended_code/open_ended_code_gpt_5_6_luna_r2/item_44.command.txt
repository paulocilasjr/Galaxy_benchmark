/usr/bin/bash -lc 'export PATH="/opt/conda/bin:/workspace/atac_env/bin:$PATH"
set -o pipefail
bwa mem -t 8 -M /workspace/atac_work/ref/hg19.fa   /workspace/atac_work/test/trimmed_R1.fastq.gz   /workspace/atac_work/test/trimmed_R2.fastq.gz 2> /workspace/atac_work/test/bwa.stderr |
samtools view -u -F 2304 - 2> /workspace/atac_work/test/view.stderr |
samtools sort -n -@ 4 -m 512M -T /workspace/atac_work/test/name_tmp   -o /workspace/atac_work/test/name.bam - 2> /workspace/atac_work/test/name_sort.stderr
samtools fixmate -m /workspace/atac_work/test/name.bam - 2> /workspace/atac_work/test/fixmate.stderr |
/opt/conda/bin/python /workspace/atac_work/filter_valid_pairs.py - 2> /workspace/atac_work/test/filter.stderr |
samtools sort -@ 4 -m 512M -T /workspace/atac_work/test/coord_tmp   -o /workspace/atac_work/test/coord.bam - 2> /workspace/atac_work/test/coord_sort.stderr
samtools quickcheck -v /workspace/atac_work/test/name.bam /workspace/atac_work/test/coord.bam
printf '"'%s\\n' '--- counts ---'
samtools view -c /workspace/atac_work/test/name.bam
samtools view -c /workspace/atac_work/test/coord.bam
samtools view -c -f 2 -q 30 /workspace/atac_work/test/coord.bam
printf '%s\\n' '--- filter ---'
cat /workspace/atac_work/test/filter.stderr
printf '%s\\n' '--- bwa ---'
sed -n '1,80p' /workspace/atac_work/test/bwa.stderr"
