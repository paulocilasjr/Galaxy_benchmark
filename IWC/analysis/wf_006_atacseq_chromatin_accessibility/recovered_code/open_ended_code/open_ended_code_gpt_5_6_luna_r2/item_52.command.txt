/usr/bin/bash -lc 'mkdir -p /tmp/atac_fifo_test2
mkfifo /tmp/atac_fifo_test2/R1.fq /tmp/atac_fifo_test2/R2.fq
cat /tmp/atac_fifo_test2/R1.fq > /tmp/atac_fifo_test2/R1.capture &
p1=$!
cat /tmp/atac_fifo_test2/R2.fq > /tmp/atac_fifo_test2/R2.capture &
p2=$!
/workspace/atac_env/bin/fastp   --in1 /workspace/data/inputs/pe_fastq_input/srr891269/forward/forward.fastqsanger.gz   --in2 /workspace/data/inputs/pe_fastq_input/srr891269/reverse/reverse.fastqsanger.gz   --out1 /tmp/atac_fifo_test2/R1.fq   --out2 /tmp/atac_fifo_test2/R2.fq   --reads_to_process 100000   --detect_adapter_for_pe   --adapter_sequence CTGTCTCTTATACACATCT   --adapter_sequence_r2 CTGTCTCTTATACACATCT   --qualified_quality_phred 20   --unqualified_percent_limit 70   --length_required 15   --n_base_limit 50   --disable_trim_poly_g   --thread 4   --json /tmp/atac_fifo_test2/fastp.json   --html /tmp/atac_fifo_test2/fastp.html   2> /tmp/atac_fifo_test2/fastp.stderr
fp=$?
wait $p1
s1=$?
wait $p2
s2=$?
printf '"'statuses fastp=%s r1=%s r2=%s\\n' \""'$fp" "$s1" "$s2"
wc -l /tmp/atac_fifo_test2/R1.capture /tmp/atac_fifo_test2/R2.capture
sed -n '"'1,12p' /tmp/atac_fifo_test2/fastp.stderr"
