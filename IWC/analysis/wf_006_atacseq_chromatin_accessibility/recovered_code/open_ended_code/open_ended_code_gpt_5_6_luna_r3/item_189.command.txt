/usr/bin/bash -lc "set -o pipefail
/opt/conda/bin/python /workspace/work/scripts/filter_pairs.py \\
  /workspace/work/align/raw.name.bam \\
  /workspace/work/align/valid.name.bam \\
  /workspace/work/align/valid_pair_filter.json \\
  --min-mapq 30 \\
  --mitochondrial chrM
/opt/conda/bin/samtools fixmate -m -@ 8 \\
  /workspace/work/align/valid.name.bam \\
  /workspace/work/align/valid.fixmate.bam
/opt/conda/bin/samtools sort -@ 8 \\
  -o /workspace/work/align/valid.coord.bam \\
  /workspace/work/align/valid.fixmate.bam
/opt/conda/bin/samtools markdup -r -@ 8 -s -f /workspace/work/align/markdup.stats \\
  /workspace/work/align/valid.coord.bam \\
  /workspace/work/align/final.bam
/opt/conda/bin/samtools index -@ 8 /workspace/work/align/final.bam
/opt/conda/bin/samtools flagstat /workspace/work/align/final.bam > /workspace/work/align/final.flagstat.txt
/opt/conda/bin/samtools stats /workspace/work/align/final.bam > /workspace/work/align/final.stats.txt"
