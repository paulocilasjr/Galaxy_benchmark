/usr/bin/bash -lc 'export PATH=/workspace/env/bin:$PATH; '"\\
  /workspace/env/bin/samtools sort -n -@ 4 -m 1G -o work/filtered.namesorted.bam work/filtered.sorted.bam && \\
  /workspace/env/bin/samtools fixmate -m -u work/filtered.namesorted.bam - \\
  | /workspace/env/bin/samtools sort -@ 4 -m 1G -o work/fixmate.sorted.bam - && \\
  /workspace/env/bin/samtools markdup -r -s -@ 8 work/fixmate.sorted.bam work/dedup.bam \\
  > work/markdup.stats.txt 2> work/markdup.log
status="'$?
echo PIPELINE_EXIT:$status
cat work/markdup.log
cat work/markdup.stats.txt
ls -lh work/dedup.bam work/fixmate.sorted.bam work/filtered.namesorted.bam'
