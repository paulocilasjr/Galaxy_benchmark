/usr/bin/bash -lc 'export PATH=/workspace/env/bin:$PATH; /workspace/env/bin/fastp '"\\
  -i data/inputs/pe_fastq_input/srr891269/forward/forward.fastqsanger.gz \\
  -I data/inputs/pe_fastq_input/srr891269/reverse/reverse.fastqsanger.gz \\
  -o work/trimmed_R1.fastq.gz \\
  -O work/trimmed_R2.fastq.gz \\
  --detect_adapter_for_pe \\
  --adapter_sequence CTGTCTCTTATACACATCT \\
  --adapter_sequence_r2 CTGTCTCTTATACACATCT \\
  --qualified_quality_phred 20 \\
  --unqualified_percent_limit 70 \\
  --n_base_limit 50 \\
  --length_required 15 \\
  --thread 8 \\
  --json work/fastp.json \\
  --html work/fastp.html \\
  --report_title 'SRR891269 paired-end ATAC-seq fastp' \\
  > work/fastp.log 2>&1
status="'$?
echo FASTP_EXIT:$status
tail -100 work/fastp.log'
