/usr/bin/bash -lc "gzip -t final_answer/trimmed_forward.fastqsanger.gz && gzip -t final_answer/trimmed_reverse.fastqsanger.gz && perl -e '
use strict; use warnings;
open(my "'$f1, "-|", "gzip", "-cd", $ARGV[0]) or die $!;
open(my $f2, "-|", "gzip", "-cd", $ARGV[1]) or die $!;
my ($pairs,$min1,$min2,$max1,$max2,$trimmed1,$trimmed2)=(0,10**9,10**9,0,0,0,0);
while (1) {
  my $h1=<$f1>; my $h2=<$f2>;
  die "mate count mismatch'"\\n\" if defined("'$h1) != defined($h2);
  last unless defined $h1;
  my $s1=<$f1>; my $p1=<$f1>; my $q1=<$f1>;
  my $s2=<$f2>; my $p2=<$f2>; my $q2=<$f2>;
  die "truncated FASTQ'"\\n\" unless defined("'$q1) && defined($q2);
  s/'"\\r?\\n"'$// for ($h1,$s1,$p1,$q1,$h2,$s2,$p2,$q2);
  die "invalid FASTQ markers'"\\n\" unless "'$h1 =~ /''^'"\\@/ && "'$h2 =~ /''^'"\\@/ && "'$p1 =~ /''^'"\\+/ && "'$p2 =~ /''^'"\\+/;
  my ("'$id1)=split /'"\\s+/, substr("'$h1,1); my ($id2)=split /'"\\s+/, substr("'$h2,1);
  $id1 =~ s{/1$}{}; $id2 =~ s{/2$}{};
  die "ID mismatch: $id1 $id2'"\\n\" unless "'$id1 eq $id2;
  die "sequence/quality length mismatch'"\\n\" unless length("'$s1)==length($q1) && length($s2)==length($q2);
  my ($l1,$l2)=(length($s1),length($s2));
  my $n1=()=$s1=~/N/ig; my $n2=()=$s2=~/N/ig;
  my $low1=grep { ord($_)-33 < 20 } split //,$q1;
  my $low2=grep { ord($_)-33 < 20 } split //,$q2;
  die "filter violation at $id1'"\\n\" if "'$l1<15 || $l2<15 || $n1>5 || $n2>5 || 100*$low1>40*$l1 || 100*$low2>40*$l2;
  $min1=$l1 if $l1<$min1; $min2=$l2 if $l2<$min2; $max1=$l1 if $l1>$max1; $max2=$l2 if $l2>$max2;
  $trimmed1++ if $l1<101; $trimmed2++ if $l2<101; $pairs++;
}
close $f1; close $f2;
print "validated_pairs=$pairs min_lengths=$min1,$min2 max_lengths=$max1,$max2 reads_shorter_than_101=$trimmed1,$trimmed2'"\\n\";
' final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz && sha256sum final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz && stat -c '%n %s bytes' final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz"
