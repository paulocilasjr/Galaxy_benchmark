/usr/bin/bash -lc "perl -Mstrict -Mwarnings -e '
for my "'$f (@ARGV) {
  open my $fh, "-|", "gzip", "-cd", $f or die "$f: $!";
  my ($records,$min,$max,$bad_len,$bad_n,$bad_q,$bad_shape)=(0,undef,0,0,0,0,0);
  while (defined(my $h=<$fh>)) {
    my $s=<$fh>; my $p=<$fh>; my $q=<$fh>;
    if (!defined($s) || !defined($p) || !defined($q)) { $bad_shape++; last }
    chomp($h,$s,$p,$q);
    $records++;
    my $len=length($s);
    $min=$len if !defined($min) || $len<$min;
    $max=$len if $len>$max;
    $bad_shape++ if $h !~ /''^'"\\@/ || "'$p !~ /''^'"\\+/ || "'$len != length($q);
    $bad_len++ if $len < 15;
    my $n=()=$s=~/N/ig;
    $bad_n++ if $n > 5;
    my $low=grep { $_ < 53 } unpack("C*", $q);
    $bad_q++ if $low*100 > 40*$len;
  }
  close $fh or die "gzip failed for $f";
  print join("'"\\t\","'$f,"records=$records","min_len=$min","max_len=$max","bad_fastq=$bad_shape","bad_len=$bad_len","bad_N=$bad_n","bad_Q20_fraction=$bad_q"),"'"\\n\";
}' final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz"
