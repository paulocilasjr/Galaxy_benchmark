/usr/bin/bash -lc "gzip -t final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz && perl -MIO::Uncompress::Gunzip=:all -e '
my ("'$fp,$rp)=@ARGV;
my $f=IO::Uncompress::Gunzip->new($fp) or die $GunzipError;
my $r=IO::Uncompress::Gunzip->new($rp) or die $GunzipError;
my ($pairs,$minf,$minr,$maxnf,$maxnr,$maxlf,$maxlr)=(0,10**9,10**9,0,0,0,0);
while (1) {
  my @a=map { scalar <$f> } 1..4; my @b=map { scalar <$r> } 1..4;
  last if !defined($a[0]) && !defined($b[0]);
  die "mate count mismatch'"\\n\" if "'!defined($a[0]) || !defined($b[0]);
  die "truncated FASTQ'"\\n\" if grep { "'!defined } (@a,@b);
  chomp(@a,@b);
  die "invalid FASTQ framing'"\\n\" unless "'$a[0]=~/''^'"\\@/ && "'$a[2]=~/''^'"\\+/ && "'$b[0]=~/''^'"\\@/ && "'$b[2]=~/''^'"\\+/;
  my ("'$ia)=split /'"\\s+/,substr("'$a[0],1); my ($ib)=split /'"\\s+/,substr("'$b[0],1); $ia=~s{/[12]$}{}; $ib=~s{/[12]$}{};
  die "mate ID mismatch at pair ".($pairs+1)."'"\\n\" if "'$ia ne $ib;
  die "sequence/quality mismatch'"\\n\" if length("'$a[1])!=length($a[3]) || length($b[1])!=length($b[3]);
  my @x=([$a[1],$a[3]],[$b[1],$b[3]]);
  for my $m (0,1) {
    my ($s,$q)=@{$x[$m]}; my $len=length($s); die "retained read <15'"\\n\" if "'$len<15;
    my $n=()=$s=~/N/ig; die "retained read >5 Ns'"\\n\" if "'$n>5;
    my $low=0; $low++ for grep { ord($_)-33<20 } split //,$q;
    die "retained read >40% low-quality'"\\n\" if "'$low*100>$len*40;
    if ($m==0) {$minf=$len if $len<$minf; $maxnf=$n if $n>$maxnf; my $p=$low/$len; $maxlf=$p if $p>$maxlf}
    else {$minr=$len if $len<$minr; $maxnr=$n if $n>$maxnr; my $p=$low/$len; $maxlr=$p if $p>$maxlr}
  }
  $pairs++;
}
print "pairs=$pairs min_length_forward=$minf min_length_reverse=$minr max_N_forward=$maxnf max_N_reverse=$maxnr max_low_fraction_forward=",sprintf("%.6f",$maxlf)," max_low_fraction_reverse=",sprintf("%.6f",$maxlr),"'"\\n\";
' final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz
sha256sum final_answer/trimmed_forward.fastqsanger.gz final_answer/trimmed_reverse.fastqsanger.gz
find final_answer -maxdepth 1 -type f -printf '%f\\n' | sort"
