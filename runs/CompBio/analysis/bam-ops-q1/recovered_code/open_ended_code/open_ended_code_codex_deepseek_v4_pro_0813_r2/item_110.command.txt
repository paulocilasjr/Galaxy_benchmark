/bin/bash -c "python - <<'PY'
import pysam
bam='mt.sorted.bam'
cnt_all=0; cnt_min30=0; cnt_avg30=0; cnt_min20=0; cnt_q30flag2=0
for r in pysam.AlignmentFile(bam,'rb'):
    if r.is_proper_pair:
        if r.mapping_quality>=30:
            cnt_q30flag2+=1
        q=r.query_qualities
        if q is not None and len(q)>0:
            if min(q)>=30: cnt_min30+=1
            if sum(q)/len(q)>=30: cnt_avg30+=1
print('q30flag2 all records', cnt_q30flag2)
print('proper pair min base q>=30', cnt_min30)
print('proper pair avg base q>=30', cnt_avg30)
# primary only
cnt=0;cnt2=0;cnt3=0
for r in pysam.AlignmentFile(bam,'rb'):
    if r.is_proper_pair and not r.is_secondary and not r.is_supplementary:
        if r.mapping_quality>=30: cnt+=1
        q=r.query_qualities
        if q is not None and len(q)>0 and min(q)>=30: cnt2+=1
        if q is not None and len(q)>0 and sum(q)/len(q)>=30: cnt3+=1
print('primary mapq30 proper',cnt,'primary minbase30',cnt2,'primary avgbase30',cnt3)
PY"
