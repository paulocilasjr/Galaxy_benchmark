/bin/bash -lc "python3 - <<'PY'
import subprocess, collections, gzip, re
# get transcript -> gene from fasta headers? transcripts.fa header contains gene symbol etc
name2 = {}
for line in open('gencode.v47.transcripts.fa'):
    if line.startswith('>'):
        parts=line[1:].strip().split('|')
        # fields: ENST|ENSG|OTTHUMG|OTTHUMT|tag|gene|length|type
        name=parts[0]
        gene=parts[5] if len(parts)>5 else '?'
        typ=parts[7] if len(parts)>7 else '?'
        name2[name]=gene
# parse primary SAM (no -F? use sorted bam primary)
counts=collections.Counter()
typ=collections.defaultdict(set)
for line in subprocess.Popen(['samtools','view','-F','256','trans_aln.sorted.bam'],stdout=subprocess.PIPE,text=True).stdout:
    fields=line.split('\\t')
    if len(fields)<3: continue
    ref=fields[2]
    if ref=='*': continue
    gene=name2.get(ref,'?')
    counts[gene]+=1
for gene,c in counts.most_common(50):
    print(f'{gene}\\t{c}')
print('total genes',len(counts))
PY"
