/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
chrom='chr9'
for start,end in [(111664528,111664548),(111664580,111664600),(111666806,111666826)]:
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={start};end={end}'
    with urllib.request.urlopen(url) as r:d=json.load(r)
    print('range',start,end,d['dna'])
    for i,ch in enumerate(d['dna'],start+1): print(i,ch)
    print()
PY"
