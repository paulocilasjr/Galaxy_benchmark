/bin/bash -lc "python3 - <<'PY'
import subprocess,collections,re
name2={}
for line in open('gencode.v47.transcripts.fa'):
    if line.startswith('>'):
        h=line[1:].strip();p=h.split('|');name2[h]=p[5] if len(p)>5 else '?'
c=collections.Counter()
for line in subprocess.Popen(['samtools','view','-F','256','trans_aln.sorted.bam'],stdout=subprocess.PIPE,text=True).stdout:
    f=line.split('\\t')
    if len(f)<3 or f[2]=='*':continue
    c[name2.get(f[2],'?')]+=1
# get coding gene info
infos={}
for line in open('gencode.v47.annotation.gtf'):
    if line.startswith('#'):continue
    f=line.rstrip('\\n').split('\\t')
    if len(f)<9 or f[2]"'!='"'gene':continue
    attrs=dict(re.findall(r'(\\S+?) \"(["'^"]*)"'"',f[8])); infos[attrs.get('gene_name','')]=(attrs.get('gene_type',''),int(f[3]),int(f[4]))
coding=[(g,n) for g,n in c.items() if infos.get(g,(None,))[0]=='protein_coding']
coding.sort(key=lambda x:x[1],reverse=True)
for rank,(g,n) in enumerate(coding[:2000],1):
    if rank in [100,500,1000,1500,2000]: print(rank,g,n)
print('total coding detected',len(coding))
print('rank/expression of candidates')
for g in ['EEF1A1','RPS4X','HLA-C','CD74','PHB2','STAT1','GLS']:
    rank=next((i+1 for i,(gg,n) in enumerate(coding) if gg==g),None)
    n=next((n for gg,n in coding if gg==g),0)
    print(g,rank,n)
PY"
