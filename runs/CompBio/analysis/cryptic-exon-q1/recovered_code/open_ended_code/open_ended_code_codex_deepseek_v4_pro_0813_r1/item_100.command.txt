/bin/bash -lc "python3 - <<'PY'
import subprocess,re,collections,pickle
known=set(pickle.load(open('known_all.pkl','rb')))
hits=[]; novel_counts=collections.Counter()
for line in subprocess.Popen(['samtools','view','-F','4','unmapped_cand_k7.bam'],stdout=subprocess.PIPE,text=True).stdout:
    f=line.split('\\t')
    if len(f)<6:continue
    flag=int(f[1]);mapq=int(f[4]);cigar=f[5]
    if flag&0x900 or mapq<10:continue
    m=re.match(r'"'^([''^|]+)'"\\|(["'^:]+):('"\\d+)-(\\d+)"'$'"',f[2])
    if not m:continue
    label=m.group(1);chrom=m.group(2);ref_start=int(m.group(3));pos=int(f[3])
    ops=re.findall(r'(\\d+)([MIDNSHP=X])',cigar);left=0;ints=[]
    for n,op in ops:
        n=int(n)
        if op=='N':
            gs=ref_start+1+(pos-1)+left;ge=gs+n-1;nov=(chrom,gs,ge) not in known
            ints.append((gs,ge,n,nov))
            if nov:novel_counts[(label,chrom,gs,ge)]+=1
            left+=n
        elif op in 'MDN=X':left+=n
    for i in range(len(ints)-1):
        a=ints[i];b=ints[i+1]
        if a[3] and b[3]:
            exon_len=b[0]-a[1]-1
            hits.append((label,chrom,pos,mapq,cigar,ints,a[:3],b[:3],exon_len,f[0]))
print('adjacent novel-novel pairs',len(hits))
for h in hits:print(h)
print('novel introns')
for j,c in novel_counts.most_common(100):print(j,c)
PY"
