/bin/bash -lc "python3 - <<'PY'
import gzip
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s.translate(comp)[::-1]
kmers=['CAAGGAGTAAATAATGCTGA', 'ATGCTGAAAAATTTGGCTATGTC', 'CACTAATAAAGCAAGGAGTAAATAATGCTGAAAAATTTGGCTAT']
allk=set(kmers)|set(rc(k) for k in kmers)
counts={k:0 for k in allk}
with gzip.open('cryptic.exon.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h: break
        s=f.readline().strip(); f.readline(); f.readline()
        for k in allk:
            if k in s: counts[k]+=1
for k,c in counts.items(): print(k,c)
PY"
