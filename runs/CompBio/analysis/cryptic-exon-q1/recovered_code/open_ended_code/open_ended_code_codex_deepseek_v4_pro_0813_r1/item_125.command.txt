for line in open('gencode.v47.annotation.gtf'):

    if attrs.get('gene_name') in ['GNG10','DNAJC25-GNG10']:
        print(f[0],f[3],f[4],f[6],attrs.get('gene_name'),attrs.get('gene_type'))
        print(line)
