/bin/bash -lc "cat > fetch_candidates.py <<'PY'
import subprocess, collections, re, urllib.request, json, concurrent.futures, sys, time
# gather gene counts
name2={}
for line in open('gencode.v47.transcripts.fa'):
    if line.startswith('>'):
        h=line[1:].strip(); p=h.split('|')
        name2[h]=p[5] if len(p)>5 else '?'
counts=collections.Counter()
for line in subprocess.Popen(['samtools','view','-F','256','trans_aln.sorted.bam'],stdout=subprocess.PIPE,text=True).stdout:
    f=line.split('\\t')
    if len(f)<3 or f[2]=='*': continue
    counts[name2.get(f[2],'?')]+=1
geneinfo={}
for line in open('gencode.v47.annotation.gtf'):
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    if len(f)<9 or f[2]"'!='"'gene': continue
    attrs=dict(re.findall(r'(\\S+?) \"(["'^"]*)"'"', f[8]))
    geneinfo[attrs.get('gene_name','')]=(f[0], int(f[3]), int(f[4]), attrs.get('gene_type',''))
# collect top N unique coding genes
top=[]
seen=set()
for g,c in counts.most_common():
    info=geneinfo.get(g)
    if info and info[3]=='protein_coding' and g not in seen:
        seen.add(g); top.append((g,c,info))
    if len(top)>=int(sys.argv[1]): break
print('top',len(top))
# fetch region with flank
def fetch(item):
    g,c,info=item
    chrom,start,end=info[:3]
    st=max(1,start-200); en=end+200
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={st};end={en}'
    for attempt in range(3):
        try:
            with urllib.request.urlopen(url, timeout=120) as r:
                d=json.load(r)
            dna=d['dna'].upper()
            return (g,chrom,st,en,dna)
        except Exception as e:
            if attempt==2: print('fail',g,e,file=sys.stderr); return None
            time.sleep(1)
with open('candidate_genes.fa','w') as out:
    # sequential/thread
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as ex:
        for res in ex.map(fetch, top):
            if res is None: continue
            g,chrom,st,en,dna=res
            out.write(f'>{g}|{chrom}:{st}-{en}\\n')
            for i in range(0,len(dna),80): out.write(dna[i:i+80]+'\\n')
print('done')
PY
python3 fetch_candidates.py 300 && ls -lh candidate_genes.fa"
