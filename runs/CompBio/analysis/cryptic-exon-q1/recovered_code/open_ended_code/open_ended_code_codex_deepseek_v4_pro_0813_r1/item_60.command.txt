/bin/bash -lc "python3 - <<'PY'
# extract STAT1 fasta from candidate_genes.fa
out=open('stat1.fa','w'); keep=False
for line in open('candidate_genes.fa'):
    if line.startswith('>'):
        keep=line.startswith('>STAT1|')
    if keep: out.write(line)
out.close()
PY
ls -lh stat1.fa
minimap2 -t 4 -k 7 -w 3 -d stat1_k7.mmi stat1.fa 2>&1 | tail -5
minimap2 -t 4 -ax splice -k 7 -w 3 --secondary=no stat1_k7.mmi <(gunzip -c cryptic.exon.q1.fq.gz) | samtools view -bS - > stat1_aln_k7.bam
samtools view -F 4 stat1_aln_k7.bam | awk '"'$6 ~ /N/ {print}'"' | head -20"
