/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
chrom='chr9'
def get(start,end):
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={start-1};end={end}'
    with urllib.request.urlopen(url,timeout=20) as r:d=json.load(r)
    return d['dna']
for gs,ge in [(111661716,111664536),(111664590,111666814)]:
    left=get(gs-5,gs+4); right=get(ge-3,ge+6)
    print('intron',gs,ge)
    print('lower',[(gs-4+i,c) for i,c in enumerate(left)])
    print('upper',[(ge-2+i,c) for i,c in enumerate(right)])
PY"
