/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for start,end in [(190908990,190909010),(190974820,190974840)]:
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr2;start={start};end={end}'
    with urllib.request.urlopen(url) as r:d=json.load(r)
    print(start,end,d['dna'])
    for i,ch in enumerate(d['dna'],start+1): print(i,ch)
    print()
PY"
