/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# Known EEF1A1 introns: (73517935,73518029), (73518265,73518353), (73518611,73518697), etc
chrom='chr6'
def get(start,end):
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={start-1};end={end}'
    with urllib.request.urlopen(url) as r:d=json.load(r)
    return d['dna']
for gs,ge in [(73517935,73518029),(73518265,73518353),(73518611,73518697)]:
    left=get(gs-3,gs+2); right=get(ge-2,ge+3)
    print('intron',gs,ge,'lower',left,'upper',right)
    # 1-based positions for left: start-3..start+2
    print('lower_pos',[(gs-3+i,c) for i,c in enumerate(left)])
    print('upper_pos',[(ge-2+i,c) for i,c in enumerate(right)])
PY"
