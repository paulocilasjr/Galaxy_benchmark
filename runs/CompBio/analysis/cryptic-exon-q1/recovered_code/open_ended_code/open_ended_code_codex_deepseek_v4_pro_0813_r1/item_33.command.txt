/bin/bash -lc "python3 - <<'PY'
import subprocess, collections, re
name2={}
for line in open('gencode.v47.transcripts.fa'):
    if line.startswith('>'):
        h=line[1:].strip(); p=h.split('|')
        name2[h]=p[5] if len(p)>5 else '?'
counts=collections.Counter()
for line in subprocess.Popen(['samtools','view','-F','256','trans_aln.sorted.bam'],stdout=subprocess.PIPE,text=True).stdout:
    f=line.split('\\t')
    if len(f)<3 or f[2]=='*': continue
    counts[name2.get(f[2],'?')]+=1
# parse gtf gene entries to get gene_name, chr, start,end, type
geneinfo={}
for line in open('gencode.v47.annotation.gtf'):
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    if len(f)<9 or f[2]"'!='"'gene': continue
    attrs=dict(re.findall(r'(\\S+?) \"(["'^"]*)"'"', f[8]))
    gname=attrs.get('gene_name','')
    gtype=attrs.get('gene_type','')
    geneinfo[gname]=(f[0], int(f[3]), int(f[4]), gtype)
# print top 150 coding genes
for rank,(g,c) in enumerate(counts.most_common(250),1):
    info=geneinfo.get(g)
    if info and info[3]=='protein_coding':
        print(f'{rank}\\t{g}\\t{c}\\t{info[0]}\\t{info[1]}\\t{info[2]}')
PY"
