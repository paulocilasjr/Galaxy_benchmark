/bin/bash -lc "cat > scripts/prepare_genome.py <<'PY'
import gzip
keep=False
with gzip.open('reference/hg38.fa.gz','rt') as f, open('reference/hg38.primary.fa','w') as o:
 for l in f:
  if l.startswith('>'):keep=l[1:].strip() in {'chr'+str(i) for i in range(1,23)}|{'chrX','chrY','chrM'}
  if keep:o.write(l)
PY"
