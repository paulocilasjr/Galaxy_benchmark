/bin/bash -lc 'mv hg38.complete.fa.gz hg38.reference.fa.gz
mv gencode.v49.complete.annotation.gtf.gz gencode.v49.annotation.gtf.gz
rm hg38.fa.gz gencode.v49.annotation.gtf.gz.partial 2>/dev/null || true
rm hg38.part.* gtf.part.*
gzip -cd hg38.reference.fa.gz > hg38.reference.fa &
pid_fa=$!
python3 gtf_to_junction_bed.py gencode.v49.annotation.gtf.gz > gencode.v49.junctions.bed &
pid_bed=$!
wait "$pid_fa"
wait "$pid_bed"
ls -lh hg38.reference.fa hg38.reference.fa.gz gencode.v49.junctions.bed
wc -l gencode.v49.junctions.bed
df -h .'
