/bin/bash -lc 'genome_size=983659424
gtf_size=93374019
parts=8
for i in $(seq 0 7); do
  start=$((i * genome_size / parts))
  end=$((((i + 1) * genome_size / parts) - 1))
  if [ "$i" -eq 7 ]; then end=$((genome_size - 1)); fi
  curl -sS -L --fail --retry 3 -r "${start}-${end}" -o "hg38.part.${i}" https://hgdownload.soe.ucsc.edu/goldenPath/hg38/bigZips/hg38.fa.gz &
  start2=$((i * gtf_size / parts))
  end2=$((((i + 1) * gtf_size / parts) - 1))
  if [ "$i" -eq 7 ]; then end2=$((gtf_size - 1)); fi
  curl -sS -L --fail --retry 3 -r "${start2}-${end2}" -o "gtf.part.${i}" https://ftp.ebi.ac.uk/pub/databases/gencode/Gencode_human/release_49/gencode.v49.annotation.gtf.gz &
done
wait
wc -c hg38.part.* gtf.part.*'
