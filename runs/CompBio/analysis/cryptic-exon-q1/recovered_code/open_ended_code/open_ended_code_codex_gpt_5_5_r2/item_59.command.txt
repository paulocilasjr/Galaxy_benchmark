/bin/bash -lc "awk '"'!seen[$4]++ && $1!="chrM"{print}'"' results/top300_coding_genes.bed > work/top_coding_unique.bed
: > ref/top_coding_loci.fa
while read chrom start end gene count strand; do
  dna="'$(curl -s "https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=${chrom};start=${start};end=${end}" | jq -r '"'.dna // empty')
  if [ -n \""'$dna" ]; then
    printf '"'>%s|%s:%s-%s|%s|count=%s\\n' \""'$gene" "$chrom" "$start" "$end" "$strand" "$count" >> ref/top_coding_loci.fa
    printf '"'%s\\n' \""'$dna" | fold -w 80 >> ref/top_coding_loci.fa
  else
    printf '"'failed %s %s %s %s\\n' \""'$gene" "$chrom" "$start" "$end" >&2
  fi
done < work/top_coding_unique.bed
awk '"'/"'^>/{n++; print; if(n==5) exit}'"' ref/top_coding_loci.fa
wc -c ref/top_coding_loci.fa"
