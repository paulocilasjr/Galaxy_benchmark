/bin/bash -lc "python3 - <<'PY'
import pysam, collections, re
sam=pysam.AlignmentFile('aligned.bam','rb')
juncs=collections.Counter()
read_juncs=collections.defaultdict(list)
for r in sam.fetch('5',150400000,150407000):
    if r.is_unmapped: continue
    ref=r.reference_start
    q=0
    for op,l in r.cigartuples:
        if op==0:
            q+=l; ref+=l
        elif op==1:
            q+=l
        elif op==2:
            ref+=l
        elif op==3:
            # intron starts at ref (0-based) and ends at ref+l (0-based exclusive), next base ref+l
            # 1-based intron start = ref+1; end = ref+l
            start1=ref+1; end1=ref+l
            j=(r.reference_name,start1,end1, '-' if r.is_reverse else '+')
            juncs[j]+=1
            read_juncs[r.query_name].append(j)
            ref+=l
        elif op==4:
            pass
        elif op==5:
            pass
        else:
            pass
print('junction counts near CD74:')
for j,c in sorted(juncs.items(), key=lambda x:x[1], reverse=True):
    print(j,c)
print('\\nreads with >=2 junctions:')
multi=0
for name,js in read_juncs.items():
    if len(js)>=2:
        multi+=1
        print(name, js)
print('multi count',multi)
# reads with novel N=542 or N=2461?
for name,js in read_juncs.items():
    if any(j[1:3] in [(150402019,150402562),(150402624,150405084)] for j in js):
        print('novel-read',name,js)
PY"
