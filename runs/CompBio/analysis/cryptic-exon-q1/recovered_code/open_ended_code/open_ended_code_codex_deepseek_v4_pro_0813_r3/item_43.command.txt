/bin/bash -lc "python3 - <<'PY'
import csv, collections
cands=['UBC','HLA-A','HLA-C','AHNAK','ARPC5','CD74','DNAJC25-GNG10','GNG10','HLA-B','HMGB1','NUP210','PCNX1','STAMBPL1','STEAP1B']
counts={}
with open('gene_counts.txt') as f:
 for line in f:
  if line.startswith('#'): continue
  parts=line.rstrip().split('\\t')
  if parts[0]=='Geneid': continue
  counts[parts[0]]=int(parts[-1])
print('candidate counts:')
for g in sorted(cands, key=lambda x:-counts.get(x,0)):
 print(g, counts.get(g,0))
print('\\ntop 50 genes:')
for i,(g,c) in enumerate(sorted(counts.items(), key=lambda kv:-kv[1])[:50],1):
 print(i,g,c)
PY"
