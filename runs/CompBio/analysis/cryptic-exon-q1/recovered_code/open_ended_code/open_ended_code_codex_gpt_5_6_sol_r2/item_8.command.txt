/bin/bash -lc "gzip -cd Homo_sapiens.GRCh38.cdna.all.fa.gz | awk '/"'^>/{id=$1; sub(/''^>/,"",id); sym=""; bio=""; for(i=2;i<=NF;i++){if($i ~ /''^gene_symbol:/){sym=$i; sub(/''^gene_symbol:/,"",sym)} if($i ~ /''^gene_biotype:/){bio=$i; sub(/''^gene_biotype:/,"",bio)}} if(sym!="") print id"'"\\t\"sym\"\\t\"bio}' > transcript_gene.tsv
samtools view -F 260 cdna.alignments.bam | awk '{c["'$3]++} END{for(k in c) print k"'"\\t\"c[k]}' | sort -k1,1 > transcript_counts.tsv
sort -k1,1 transcript_gene.tsv | join -t "'$'"'\\t' - transcript_counts.tsv | awk -F '\\t' '{c["'$2]+=$4; bio[$2]=$3} END{for(g in c) print c[g],g,bio[g]}'"' | sort -nr | head -50"
