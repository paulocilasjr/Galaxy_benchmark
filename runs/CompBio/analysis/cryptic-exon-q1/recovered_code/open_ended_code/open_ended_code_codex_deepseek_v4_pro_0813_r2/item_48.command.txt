/bin/bash -lc "python3 - <<'PY'
import collections, gzip
# load novel pc grouped
g=collections.defaultdict(list)
for line in open('novel_junctions_pc.tsv'):
    if line.startswith('gene_id'): continue
    p=line.rstrip().split('\\t')
    if len(p)<7: continue
    gid,gname,chrom,strand,start,end,cnt=p
    g[(gid,gname)].append((chrom,strand,int(start),int(end),int(cnt)))
expr={}
for line in open('gene_expression.tsv'):
    if line.startswith('gene_id'): continue
    p=line.rstrip().split('\\t')
    if len(p)<7: continue
    expr[p[0]]=(p[1],int(p[6]))
# sort by expression
rows=[]
for key,lst in g.items():
    if len(lst)"'!=2: continue
    gid,name=key
    e=expr.get(gid,(name,0))[1]
    rows.append((e,gid,name,lst))
for e,gid,name,lst in sorted(rows, reverse=True):
    print(f'"'{name}\\t{gid}\\texpr={e}\\t{lst}')
PY"
