/bin/bash -lc "gzip -cd data/gencode.vM31.annotation.gtf.gz | awk -F '\\t' '"'$3 == "exon" && $9 ~ /gene_name "Lepr"(;| )/ {print $1, $4, $5, $7}'"' | sort -k1,1 -k2,2n -k3,3n > data/lepr_exons.tsv
printf 'exon records: '; wc -l < data/lepr_exons.tsv
printf 'transcripts: '; gzip -cd data/gencode.vM31.annotation.gtf.gz | awk -F '\\t' '"'$3 == "exon" && $9 ~ /gene_name "Lepr"(;| )/ {if (match($9,/transcript_id "[''^"]+"/)) {x=substr($9,RSTART+15,RLENGTH-16); print x}}'"' | sort -u | wc -l
head -5 data/lepr_exons.tsv
tail -5 data/lepr_exons.tsv"
