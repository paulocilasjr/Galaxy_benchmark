/bin/bash -c "python3 - <<'PY'
import gzip
all_int=[]; pc_int=[]
with gzip.open('gencode.vM31.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        if p[2]=='exon' and 'gene_name \"Lepr\"' in p[8]:
            d={}
            for item in p[8].split(';'):
                item=item.strip()
                if ' ' in item:
                    k,v=item.split(' ',1)
                    d[k]=v.strip().strip('\"')
            all_int.append((int(p[3]),int(p[4]),d.get('transcript_type'),d.get('transcript_id')))
            if d.get('transcript_type')=='protein_coding':
                pc_int.append((int(p[3]),int(p[4])))
def union_len(arr):
    if not arr:return 0
    s=sorted(arr)
    cur=None; total=0
    for a,b in s:
        if cur is None or a>cur[1]:
            if cur: total += cur[1]-cur[0]+1
            cur=[a,b]
        elif b>cur[1]: cur[1]=b
    total += cur[1]-cur[0]+1
    return total
print('all',len(all_int),union_len([(a,b) for a,b,_ in all_int]),'pc',len(pc_int),union_len(pc_int))
print('non-pc intervals:')
for x in all_int:
    if x[2]"'!='"'protein_coding': print(x)
PY"
