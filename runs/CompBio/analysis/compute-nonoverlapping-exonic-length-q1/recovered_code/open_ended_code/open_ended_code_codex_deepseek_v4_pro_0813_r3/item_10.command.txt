/bin/bash -c "python3 - <<'PY'
import gzip
intervals=[]
with gzip.open('gencode.vM31.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        if p[2]=='exon' and 'gene_name \"Lepr\"' in p[8]:
            intervals.append((int(p[3]), int(p[4])))
intervals=sorted(set(intervals))
# merge overlapping and adjacent intervals (adjacent contributes same total)
merged=[]
for a,b in intervals:
    if merged and a <= merged[-1][1]:
        merged[-1][1]=max(merged[-1][1],b)
    else:
        merged.append([a,b])
total=sum(b-a+1 for a,b in merged)
sum_raw=sum(b-a+1 for a,b in intervals)
print('unique_exon_records',len(intervals))
print('raw_sum',sum_raw)
print('merged_intervals',len(merged))
print('total_union',total)
print(merged)
PY"
