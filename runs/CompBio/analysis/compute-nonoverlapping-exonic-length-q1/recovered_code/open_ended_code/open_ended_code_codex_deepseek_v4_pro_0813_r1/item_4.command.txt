/bin/bash -lc "python3 - <<'PY'
import gzip,re
fn='gencode.vM31.annotation.gtf.gz'
count=0
rows=0
gene_ids=set()
with gzip.open(fn,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        attrs=p[8]
        if 'gene_name \"Lepr\"' in attrs:
            rows+=1
            if p[2]=='exon': count+=1
            m=re.search(r'gene_id \"(["'^"]+)"'"',attrs)
            if m: gene_ids.add(m.group(1))
print('all Lepr rows', rows, 'exons', count, 'gene_ids', gene_ids)
PY"
