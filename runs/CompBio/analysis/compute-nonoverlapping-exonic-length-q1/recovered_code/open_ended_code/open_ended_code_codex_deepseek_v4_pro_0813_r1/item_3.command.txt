/bin/bash -lc "python3 - <<'PY'
import gzip, re, collections
fn='gencode.vM31.annotation.gtf.gz'
exons=[]
trans=set()
gene=None
with gzip.open(fn,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<9: continue
        chrom,src,feat,start,end,score,strand,frame,attrs=parts
        if feat"'!='"'exon': continue
        # check gene_name Lepr; parse attrs robust
        if 'gene_name \"Lepr\"' in attrs:
            exons.append((int(start),int(end)))
            m=re.search(r'transcript_id \"(["'^"]+)"'"',attrs)
            if m: trans.add(m.group(1))
            gid=re.search(r'gene_id \"(["'^"]+)"'"',attrs)
            if gid: gene=gid.group(1)
# merge
exons.sort()
merged=[]
for s,e in exons:
    if not merged or s>merged[-1][1]:
        merged.append([s,e])
    else:
        if e>merged[-1][1]: merged[-1][1]=e
union=sum(e-s+1 for s,e in merged)
raw=sum(e-s+1 for s,e in exons)
print('gene_id', gene)
print('exon_lines', len(exons), 'transcripts', len(trans), 'raw_sum', raw, 'union', union, 'merged_count', len(merged))
print('merged', merged)
PY"
