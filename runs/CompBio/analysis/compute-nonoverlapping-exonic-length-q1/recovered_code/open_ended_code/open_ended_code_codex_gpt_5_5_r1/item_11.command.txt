/bin/bash -c "python3 - <<'PY'
import gzip,re
ids=set(); n=0
attr_re=re.compile(r'(\\S+) \"(["'^"]*)"'"')
with gzip.open('gencode.vM31.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)==9 and p[2]=='exon':
            d=dict(attr_re.findall(p[8]))
            if d.get('gene_name')=='Lepr':
                ids.add(d.get('gene_id'))
                n+=1
print(n)
print('\\n'.join(sorted(ids)))
PY"
