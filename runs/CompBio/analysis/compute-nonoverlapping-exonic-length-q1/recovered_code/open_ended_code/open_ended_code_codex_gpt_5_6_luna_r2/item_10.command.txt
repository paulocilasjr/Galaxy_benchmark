/bin/bash -lc "cut -f1,4,5 lepr_exons.gtf | sort -u > lepr_unique_intervals.tsv

awk 'BEGIN{OFS=\"\\t\"} NR==1 {chrom="'$1; start=$2; end=$3; next} {if ($1==chrom && $2<=end+1) {if ($3>end) end=$3} else {print chrom,start,end,end-start+1; chrom=$1; start=$2; end=$3}} END {if (NR) print chrom,start,end,end-start+1}'"' lepr_unique_sorted.tsv > lepr_merged_intervals.tsv
printf 'records=%s unique_intervals=%s merged_intervals=%s\\n' \""'$(wc -l < lepr_exons.gtf)" "$(wc -l < lepr_unique_intervals.tsv)" "$(wc -l < lepr_merged_intervals.tsv)"
cat lepr_merged_intervals.tsv
awk '"'{sum+="'$4} END{print "union_length=" sum}'"' lepr_merged_intervals.tsv
awk '{sum+="'$3-$2+1} END{print "sum_unique_exon_lengths=" sum}'"' lepr_unique_sorted.tsv"
