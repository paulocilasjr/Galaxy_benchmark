/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
p='models/model0_best.h5'
with h5py.File(p,'r') as f:
    total_all=0; total_train=0; total_nontrain=0
    def visit(name,obj):
        global total_all,total_train,total_nontrain
        if isinstance(obj,h5py.Dataset):
            n=int(np.prod(obj.shape))
            total_all += n
            bn=name.split('/')[-1].split(':')[0]
            if bn in ('moving_mean','moving_variance'):
                total_nontrain += n
            else:
                total_train += n
    f['model_weights'].visititems(visit)
    print('all datasets elements', total_all)
    print('exclude moving mean/var', total_train)
    print('moving stat elements', total_nontrain)
    print('number datasets', sum(1 for _ in f['model_weights'].values() if isinstance(_,h5py.Dataset)))
PY"
