/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
for p in ['models/model0_best.h5','models/model1_best.h5']:
    with h5py.File(p,'r') as f:
        total_all=0; total_train=0; moving=0; names=[]
        def visit(name,obj):
            global total_all,total_train,moving
            if isinstance(obj,h5py.Dataset):
                n=int(np.prod(obj.shape))
                total_all+=n
                bn=name.split('/')[-1].split(':')[0]
                if bn in ('moving_mean','moving_variance'):
                    moving+=n
                else:
                    total_train+=n
                names.append(name)
        f['model_weights'].visititems(visit)
        print('\\n',p)
        print('all',total_all,'trainable_excluding_moving',total_train,'moving',moving,'datasets',len(names))
        # head-related last datasets
        print('last 15 names:')
        for n in names[-15:]: print(' ', n)
PY"
