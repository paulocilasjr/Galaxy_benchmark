/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, json
for p in ['models/model0_best.h5','models/model1_best.h5']:
 with h5py.File(p,'r') as f:
  cfg=json.loads(f.attrs['model_config'])
  finals=[l for l in cfg['config']['layers'] if l['class_name']=='Dense']
  hname=finals[-1]['name']
  head=0; names=[]
  def visit(name,obj):
   if isinstance(obj,h5py.Dataset):
    names.append(name)
    # dataset names relative to model_weights: e.g. dense_20/dense_20/kernel:0
    parts=name.split('/')
    if parts and parts[0]==hname:
     head += int(np.prod(obj.shape))
  f['model_weights'].visititems(visit)
  full=0; moving=0
  for n in names:
   ds=f['model_weights'][n]; nn=int(np.prod(ds.shape)); full+=nn
   if n.split('/')[-1].split(':')[0] in ('moving_mean','moving_variance'): moving+=nn
  print('\\n',p,'hname',hname,'head',head,'full',full,'trainable excl moving',full-moving,'nonhead train',full-moving-head)
  print('head datasets', [n for n in names if n.split('/')[0]==hname])
PY"
