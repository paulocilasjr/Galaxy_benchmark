/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, json, collections
for p in ['models/model0_best.h5','models/model1_best.h5']:
 with h5py.File(p,'r') as f:
  print('\\n',p)
  counts=collections.OrderedDict()
  def visit(name,obj):
   if isinstance(obj,h5py.Dataset):
    top=name.split('/')[0]
    n=int(np.prod(obj.shape))
    if name.split('/')[-1].split(':')[0] in ('moving_mean','moving_variance'):
     key='NONTRAIN_'+top
    else:
     key='TRAIN_'+top
    counts[key]=counts.get(key,0)+n
  f['model_weights'].visititems(visit)
  for k,v in counts.items():
   print(f'{k:40s} {v:12d}')
  train=sum(v for k,v in counts.items() if k.startswith('TRAIN_'))
  print('TRAIN TOTAL', train)
PY"
