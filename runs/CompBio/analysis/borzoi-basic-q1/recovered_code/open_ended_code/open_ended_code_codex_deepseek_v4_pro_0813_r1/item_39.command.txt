/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, json
for p in ['models/model0_best.h5','models/model1_best.h5']:
 with h5py.File(p,'r') as f:
  cfg=json.loads(f.attrs['model_config'])
  finals=[l for l in cfg['config']['layers'] if l['class_name']=='Dense']
  hname=finals[-1]['name']
  state={'head':0,'full':0,'moving':0,'names':[]}
  def visit(name,obj):
   if isinstance(obj,h5py.Dataset):
    state['names'].append(name)
    nn=int(np.prod(obj.shape)); state['full']+=nn
    if name.split('/')[0]==hname: state['head']+=nn
    if name.split('/')[-1].split(':')[0] in ('moving_mean','moving_variance'): state['moving']+=nn
  f['model_weights'].visititems(visit)
  s=state
  print('\\n',p,'hname',hname)
  print('head',s['head'],'full',s['full'],'moving',s['moving'],'trainable excl moving',s['full']-s['moving'],'nonhead train',s['full']-s['moving']-s['head'])
  print('head datasets', [n for n in s['names'] if n.split('/')[0]==hname])
PY"
