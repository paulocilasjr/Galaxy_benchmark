/bin/bash -lc "python - <<'PY'
import h5py, json
for p in ['models/model0_best.h5','models/model1_best.h5']:
 with h5py.File(p,'r') as f:
  cfg=json.loads(f.attrs['model_config'])
  finals=[l for l in cfg['config']['layers'] if l['class_name']=='Dense']
  print('\\n',p, 'last dense:', finals[-1]['name'], finals[-1]['config']['units'])
  crop=[l for l in cfg['config']['layers'] if l['class_name']=='Cropping1D']
  print('crop', crop[0]['config']['cropping'] if crop else None)
  names=[]
  def visit(name,obj):
   if isinstance(obj,h5py.Dataset): names.append(name)
  f['model_weights'].visititems(visit)
  # identify head final layer prefix from last dense name
  headprefix='/'.join(['model_weights', finals[-1]['name'], finals[-1]['name']])
  nonhead=[n for n in names if not n.startswith(headprefix+'/')]
  print('nonhead datasets',len(nonhead), 'headprefix', headprefix)
PY"
