from collections import defaultdict

def datasets(fn):
    out=[]
    with h5py.File(fn,'r') as f:
        f.visititems(lambda n,o: out.append((n,o.shape,math.prod(o.shape))) if isinstance(o,h5py.Dataset) else None)
    return sorted(out)

def is_nontrainable(n):
    return n.endswith('moving_mean:0') or n.endswith('moving_variance:0')

for label,fn in [('human','data/borzoi_f0_human_model0_best.h5'),('mouse','data/borzoi_f0_mouse_model1_best.h5')]:

    train=sum(c for n,_,c in ds if not is_nontrainable(n))
    non=sum(c for n,_,c in ds if is_nontrainable(n))
    print(label, 'datasets',len(ds),'total',total,'nontrain_bn_moving',non,'trainable_by_exclusion',train)
    # layer totals top-level after model_weights/<layer>/
