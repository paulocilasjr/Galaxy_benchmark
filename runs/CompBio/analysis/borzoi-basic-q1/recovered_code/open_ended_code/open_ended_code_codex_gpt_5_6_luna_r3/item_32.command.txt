/bin/bash -lc "python - <<'PY'
import h5py
with h5py.File('borzoi-replicate0-human.h5','r') as f:
 g=f['model_weights']; total_all=total_train=0
 for k in sorted(g.keys()):
  ds=[]; g[k].visititems(lambda n,o: ds.append((n,tuple(o.shape))) if isinstance(o,h5py.Dataset) else None)
  if ds:
   print(k,ds)
   for n,sh in ds:
    c=1
    for x in sh:c*=x
    total_all+=c
    if not ('moving_mean' in n or 'moving_variance' in n): total_train+=c
 print('ALL',total_all,'TRAIN',total_train)
PY"
