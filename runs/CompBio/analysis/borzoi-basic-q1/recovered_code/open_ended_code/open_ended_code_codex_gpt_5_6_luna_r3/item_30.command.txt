/bin/bash -lc "python - <<'PY'
import h5py
p='borzoi-replicate0-human.h5'
with h5py.File(p,'r') as f:
 g=f['model_weights']
 for k in list(g.keys()):
  sub=g[k]
  if 'weight_names' in sub.attrs:
   names=[x.decode() if isinstance(x,bytes) else x for x in sub.attrs['weight_names']]
   print(k, names)
  else:
   # recursively data paths
   ds=[]
   sub.visititems(lambda n,o: ds.append((n,tuple(o.shape))) if isinstance(o,h5py.Dataset) else None)
   if ds: print(k,ds[:5], 'n',len(ds))
PY"
