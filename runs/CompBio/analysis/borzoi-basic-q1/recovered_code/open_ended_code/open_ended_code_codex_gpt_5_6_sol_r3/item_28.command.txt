import json
import numpy as np
from baskerville.seqnn import SeqNN
p=json.load(open('published_params.json'))['model']
p['verbose']=False
m=SeqNN(p)
for i,model in enumerate(m.models):
 print(i, model.count_params(), sum(np.prod(v.shape) for v in model.trainable_weights), sum(np.prod(v.shape) for v in model.non_trainable_weights))
seen={}
for model in m.models:
 for v in model.trainable_weights:
  seen[v.ref()]=int(np.prod(v.shape))
print('union_trainable',sum(seen.values()),'vars',len(seen))
seenall={}
for model in m.models:
 for v in model.weights:
  seenall[v.ref()]=int(np.prod(v.shape))
print('union_all',sum(seenall.values()),'vars',len(seenall))
PY"
