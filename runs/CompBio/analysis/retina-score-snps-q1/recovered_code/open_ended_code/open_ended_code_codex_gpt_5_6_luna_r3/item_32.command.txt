/bin/bash -lc "python - <<'PY'
import json,urllib.request,urllib.parse
rows=[('chr1',20014826,'T'),('chr1',20015797,'A'),('chr1',2448840,'A')]
for chrom,pos,ref in rows:
 print('\\n',chrom,pos,ref)
 for genome in ['hg19','hg38']:
  for coord in ['0based','1based']:
   st=pos if coord=='0based' else pos-1
   url='https://api.genome.ucsc.edu/getData/sequence?'+urllib.parse.urlencode({'genome':genome,'chrom':chrom,'start':st,'end':st+1})
   try:
    d=json.load(urllib.request.urlopen(url,timeout=20))
    print(genome,coord,d.get('dna'),url)
   except Exception as e: print(genome,coord,'ERR',e)
PY"
