from concurrent.futures import ThreadPoolExecutor, as_completed

    i, r=row

      for mode,st in [('0based',int(r.start)),('1based',int(r.start)-1)]:
        url='https://api.genome.ucsc.edu/getData/sequence?'+urllib.parse.urlencode({'genome':genome,'chrom':r.chr,'start':st,'end':st+1})
        try:
          out[genome+'_'+mode]=json.load(urllib.request.urlopen(url,timeout=30))['dna'].upper()
        except Exception as e: out[genome+'_'+mode]='ERR'
    return out
with ThreadPoolExecutor(max_workers=20) as ex:
