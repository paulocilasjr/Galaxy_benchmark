/bin/bash -lc "python -u - <<'PY'
import torch,time
from score_rod import load_model
for th in [1,2,4,8]:
 torch.set_num_threads(th)
 m=load_model('fold0_Rod.h5')
 for bs in [1,2,4,8]:
  x=torch.zeros((bs,4,2114),dtype=torch.float32)
  t=time.time()
  with torch.no_grad(): y=m(x)
  print('threads',th,'bs',bs,'secs',round(time.time()-t,3),flush=True)
  del m
  m=load_model('fold0_Rod.h5')
 del m
PY"
