/bin/bash -lc "python -u - <<'PY'
import pandas as pd, urllib.request, urllib.parse, json, time
from concurrent.futures import ThreadPoolExecutor
x=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t').head(20)
print('starting',len(x),flush=True)
def get(t):
 i,r=t
 st=int(r.start)-1100
 url='https://api.genome.ucsc.edu/getData/sequence?'+urllib.parse.urlencode({'genome':'hg19','chrom':r.chr,'start':st,'end':st+2200})
 with urllib.request.urlopen(url,timeout=30) as f: d=json.load(f)
 return i,d['dna']
with ThreadPoolExecutor(max_workers=4) as ex:
 ys=[]
 for j,z in enumerate(ex.map(get,list(x.iterrows()))):
  ys.append(z)
  print('done',j,flush=True)
print('finished',len(ys),flush=True)
print(ys[0],flush=True)
PY"
