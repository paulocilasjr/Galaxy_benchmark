/bin/bash -lc "python -u - <<'PY'
import pandas as pd
from pyliftover import LiftOver
import pyfaidx
x=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
lo=LiftOver('hg19ToHg38.over.chain.gz'); f=pyfaidx.Fasta('hg38.fa')
comp=str.maketrans('ACGT','TGCA')
out=[]; unm=[]; mism=[]
for i,r in x.iterrows():
 z=lo.convert_coordinate(r.chr,int(r.start))
 if not z: unm.append((i,r.chr,r.start)); continue
 c,p,strand,_=z[0]
 ref=str(r.ref).upper(); alt=str(r.alt).upper()
 if strand=='-': ref=ref.translate(comp)[::-1]; alt=alt.translate(comp)[::-1]
 b=str(f[c][p:p+1]).upper()
 if b"'!=ref: mism.append((i,r.chr,int(r.start),c,p,strand,r.ref,ref,b))
 out.append((i,c,p,ref,alt,int(r.label),strand))
print('"'mapped',len(out),'unmapped',unm,'ref_mismatches',len(mism))
print('mismatch examples',mism[:10])
print('minus',sum(a[-1]=='-' for a in out))
pd.DataFrame([(c,p,ref,alt,label,i) for i,c,p,ref,alt,label,s in out],columns=['chr','start','ref','alt','label','orig_index']).to_csv('retina_hg38_lifted.tsv',sep='\\t',index=False)
PY"
