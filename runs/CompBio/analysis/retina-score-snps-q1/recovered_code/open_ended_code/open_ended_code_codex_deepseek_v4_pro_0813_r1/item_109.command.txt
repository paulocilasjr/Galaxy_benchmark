/bin/bash -lc "python - <<'PY'
p='torch_score.py'
s=open(p).read()
# replace w/b definitions
s=s.replace(\"\"\"    def w(name):\\n        return torch.from_numpy(np.array(h[f'model_weights/{name}/{name}/kernel:0'],dtype=np.float32))\\n    def b(name):\\n        return torch.from_numpy(np.array(h[f'model_weights/{name}/{name}/bias:0'],dtype=np.float32))\\n\"\"\", \"\"\"    def w_np(name):\\n        return np.array(h[f'model_weights/{name}/{name}/kernel:0'],dtype=np.float32)\\n    def b_np(name):\\n        return np.array(h[f'model_weights/{name}/{name}/bias:0'],dtype=np.float32)\\n\"\"\")
# replace assignments
s=s.replace(\"\"\"    model.conv0.weight.data=w('conv1d').permute(2,1,0).contiguous()\\n\"\"\", \"\"\"    model.conv0.weight.data=torch.from_numpy(np.ascontiguousarray(np.transpose(w_np('conv1d'),(2,1,0))))\\n\"\"\")
s=s.replace(\"\"\"    model.conv0.bias.data=b('conv1d')\\n\"\"\", \"\"\"    model.conv0.bias.data=torch.from_numpy(b_np('conv1d'))\\n\"\"\")
s=s.replace(\"\"\"        model.blocks[i]['conv'].weight.data=w(nm).permute(2,1,0).contiguous()\\n\"\"\", \"\"\"        model.blocks[i]['conv'].weight.data=torch.from_numpy(np.ascontiguousarray(np.transpose(w_np(nm),(2,1,0))))\\n\"\"\")
s=s.replace(\"\"\"        model.blocks[i]['conv'].bias.data=b(nm)\\n\"\"\", \"\"\"        model.blocks[i]['conv'].bias.data=torch.from_numpy(b_np(nm))\\n\"\"\")
s=s.replace(\"\"\"    model.fc.weight.data=w('logcounts').squeeze(-1).unsqueeze(0).contiguous()\\n\"\"\", \"\"\"    model.fc.weight.data=torch.from_numpy(np.ascontiguousarray(w_np('logcounts').T))\\n\"\"\")
s=s.replace(\"\"\"    model.fc.bias.data=b('logcounts')\\n\"\"\", \"\"\"    model.fc.bias.data=torch.from_numpy(b_np('logcounts').reshape(-1))\\n\"\"\")
# input transpose
s=s.replace(\"\"\"        xs=torch.from_numpy(x).permute(0,2,1).contiguous().to(DEV)\\n\"\"\", \"\"\"        xs=torch.from_numpy(np.ascontiguousarray(np.transpose(x,(0,2,1)))).to(DEV)\\n\"\"\")
open(p,'w').write(s)
PY
sed -n '35,70p' torch_score.py"
