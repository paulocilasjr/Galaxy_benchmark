/bin/bash -lc "python - <<'PY'
import pandas as pd, requests
from pyliftover import LiftOver
s=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
lo=LiftOver('hg19ToHg38.over.chain.gz')
url='https://api.genome.ucsc.edu/getData/sequence'
def getbase(g,chrom,p): # p 1-based
    r=requests.get(url,params={'genome':g,'chrom':chrom,'start':p-1,'end':p},timeout=20)
    d=r.json()
    return d.get('dna','ERR').upper() if r.status_code==200 else f\"ERR{r.status_code}\"
for i,row in s.head(30).iterrows():
    res=lo.convert_coordinate(row['chr'],int(row['start'])-1)
    if not res:
        print(i,row['chr'],row['start'],row['ref'],'no lift'); continue
    chr2,p0,strand,score=res[0]
    p2=p0+1
    base=getbase('hg38',chr2,p2)
    print(i,row['chr'],row['start'],row['ref'],'->',chr2,p2,strand,'hg38',base,'match',base==row['ref'])
PY"
