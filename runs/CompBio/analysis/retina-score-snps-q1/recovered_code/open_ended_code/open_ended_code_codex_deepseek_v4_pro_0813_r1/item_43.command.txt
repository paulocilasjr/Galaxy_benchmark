/bin/bash -lc "python - <<'PY'
import pandas as pd, requests
s=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
url='https://api.genome.ucsc.edu/getData/sequence'
for i,row in s.head(15).iterrows():
    out=[]
    for mode in ['P-1','P','P+1']:
        p=int(row['start']); api_start={'P-1':p-1,'P':p,'P+1':p+1}[mode]; api_end=api_start+1
        r=requests.get(url,params={'genome':'hg19','chrom':row['chr'],'start':api_start,'end':api_end},timeout=20)
        dna=r.json().get('dna','ERR').upper() if r.status_code==200 else 'ERR'
        out.append(f'{mode}:{dna}')
    print(i,row['chr'],row['start'],row['ref'],' '.join(out))
PY"
