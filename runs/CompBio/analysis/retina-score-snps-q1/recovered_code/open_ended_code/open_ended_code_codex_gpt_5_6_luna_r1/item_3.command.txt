/bin/bash -lc "python - <<'PY'
import json, urllib.request
for rec in ['6330053','6796067']:
    url=f'https://zenodo.org/api/records/{rec}'
    with urllib.request.urlopen(url) as resp:
        data=json.load(resp)
    print('\\nRECORD', rec, data.get('metadata',{}).get('title'))
    print('files:')
    for f in data.get('files',[]):
        print(f['key'], f.get('size'), f.get('links',{}).get('self'), f.get('links',{}).get('content'))
PY"
