/bin/bash -lc "python - <<'PY'
import json
p=json.load(open('snp_sequences.json'))
rows=p['rows']; seqs=p['sequences']['hg19']; idx=2114//2-1
comp=str.maketrans('ACGT','TGCA')
from collections import Counter
for pos in [idx,idx+1,idx-1]:
 print('position',pos,'ref match',sum(s[pos]==r['ref'] for s,r in zip(seqs,rows)),'alt match',sum(s[pos]==r['alt'] for s,r in zip(seqs,rows)))
print('ref-or-complement',sum(s[idx]==r['ref'] or s[idx]==r['ref'].translate(comp)[::-1] for s,r in zip(seqs,rows)))
print('allele counter',Counter((s[idx],r['ref'],r['alt']) for s,r in zip(seqs,rows)).most_common(20))
for i,(s,r) in enumerate(zip(seqs,rows)):
 if s[idx]"'!=r['"'ref']:
  print(i,r['chr'],r['start'],r['ref'],r['alt'],'genome',s[idx], 'alt?',s[idx]==r['alt'], 'comp',r['ref'].translate(comp)[::-1]);
  if i>30: break
PY"
