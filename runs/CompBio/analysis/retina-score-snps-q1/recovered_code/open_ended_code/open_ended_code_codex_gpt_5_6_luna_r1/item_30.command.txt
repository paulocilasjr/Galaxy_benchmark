/bin/bash -lc "python - <<'PY'
import tensorflow as tf
print('tf',tf.__version__,'keras',tf.keras.__version__)
p='rod_models/fold0/Rod.h5'
for kwargs in [dict(compile=False),dict(compile=False,safe_mode=False)]:
    try:
        m=tf.keras.models.load_model(p,**kwargs)
        print('loaded',kwargs, m.input_shape, m.output_shape, len(m.weights))
        print(m.inputs, m.outputs)
        break
    except Exception as e:
        print('FAILED',kwargs,type(e).__name__,e)
PY"
