/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from sklearn.metrics import average_precision_score, auc, precision_recall_curve
x=pd.read_csv('rod_scores.tsv',sep='\\t')
y=x['label']; s=x['abs_Rod_fold_avg_lfc']
p,r,_=precision_recall_curve(y,s)
print('sklearn_average_precision',format(average_precision_score(y,s),'.15f'))
print('trapezoidal_pr_auc',format(auc(r,p),'.15f'))
print('ties',500-s.nunique())
print(x.sort_values('abs_Rod_fold_avg_lfc',ascending=False).head(10).to_string(index=False))
PY"
