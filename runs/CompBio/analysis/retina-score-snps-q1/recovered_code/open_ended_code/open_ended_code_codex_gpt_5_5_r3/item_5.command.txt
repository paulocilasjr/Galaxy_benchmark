/bin/bash -lc "mkdir -p zenodo && curl -L -s https://zenodo.org/api/records/6330053 -o zenodo/6330053.json && curl -L -s https://zenodo.org/api/records/6796067 -o zenodo/6796067.json && python - <<'PY'
import json
for rec in ['6330053','6796067']:
    data=json.load(open(f'zenodo/{rec}.json'))
    print('RECORD', rec, data.get('metadata',{}).get('title'))
    for f in data.get('files',[]):
        print(f.get('key'), f.get('size'), f.get('links',{}).get('self'))
PY"
