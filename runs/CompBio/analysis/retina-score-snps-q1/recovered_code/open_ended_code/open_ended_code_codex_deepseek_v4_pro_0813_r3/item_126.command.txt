/bin/bash -c "python - <<'PY'
import numpy as np,pandas as pd
from sklearn.metrics import average_precision_score
out=pd.read_pickle('work/snps_final_hg38_with_seq.pkl')
refs=[]; alts=[]
for f in range(5):
    refs.append(np.load(f'work/ref_pred_fold{f}.npy').astype(np.float64))
    alts.append(np.load(f'work/alt_pred_fold{f}.npy').astype(np.float64))
refs=np.array(refs); alts=np.array(alts)
lfc=(alts-refs).mean(axis=0)
ap=average_precision_score(out['label'], np.abs(lfc))
print(repr(ap), np.floor(ap*100)/100)
# compare float32
refs32=np.array([np.load(f'work/ref_pred_fold{f}.npy') for f in range(5)])
alts32=np.array([np.load(f'work/alt_pred_fold{f}.npy') for f in range(5)])
lfc32=(alts32-refs32).mean(axis=0)
ap32=average_precision_score(out['label'], np.abs(lfc32))
print(repr(ap32), np.floor(ap32*100)/100)
print('max diff',np.max(np.abs(lfc-lfc32)))
PY"
