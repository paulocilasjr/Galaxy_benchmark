/bin/bash -c "python - <<'PY'
import urllib.request,json
def get(build,chrom,start,end):
    return json.load(urllib.request.urlopen(f'https://api.genome.ucsc.edu/getData/sequence?genome={build};chrom={chrom};start={start};end={end}'))['dna']
def rc(s): return s.upper().translate(str.maketrans('ACGT','TGCA'))[::-1]
sites=[('chr10',51540905,46054916,'C','A'),('chr10',51505476,46090345,'C','A')]
for chrom,b,p0,ref,alt in sites:
    flank=20
    a=get('hg19',chrom,b-flank,b+flank+1)
    c=get('hg38',chrom,p0-flank,p0+flank+1)
    print('\\n',chrom,b,p0,'len',len(a),len(c))
    print('src',a.upper()); print('rc ',rc(a)); print('tgt',c.upper()); print('eq',rc(a)==c.upper())
    print('src center',a[flank], 'rc center?', rc(a)[flank], 'tgt center', c[flank])
PY"
