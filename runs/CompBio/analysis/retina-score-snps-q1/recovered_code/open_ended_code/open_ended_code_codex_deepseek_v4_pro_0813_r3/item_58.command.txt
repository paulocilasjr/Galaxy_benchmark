/bin/bash -c "python - <<'PY'
import gzip
path='ref/hg19ToHg38.over.chain.gz'
qpos=51540905; tpos=46054917
cands=[]
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        tName=p[2]; tSize=int(p[3]); tStrand=p[4]; tStart=int(p[5]); tEnd=int(p[6])
        qName=p[7]; qSize=int(p[8]); qStrand=p[9]; qStart=int(p[10]); qEnd=int(p[11])
        if qName=='chr10' and qStart<=qpos<=qEnd and tName=='chr10' and tStart<=tpos<=tEnd:
            cands.append(line.strip())
            print(line.strip())
print('num cands',len(cands))
PY"
