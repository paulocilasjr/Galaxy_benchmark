/bin/bash -c "python - <<'PY'
import urllib.request,json
for st,en in [(1,2),(1,3),(0,2)]:
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom=chr1;start={st};end={en}'
    try:
        d=json.load(urllib.request.urlopen(url)); print(st,en,len(d['dna']),d['dna'],d.get('start'),d.get('end'))
    except Exception as e: print(st,en,'ERR',e)
PY"
