/bin/bash -c "python - <<'PY'
import pandas as pd, urllib.request,json,concurrent.futures
df=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
def check(r):
    i,row=r
    chrom=row['chr']; p=row['start']
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={p};end={p+1}'
    try:b=json.load(urllib.request.urlopen(url))['dna'].upper()
    except Exception as e:b='ERR'
    ref=row['ref'].upper()
    return i,b,ref
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:
    res=list(ex.map(check,df.iterrows()))
mism=[]
for i,b,ref in res:
    if b=='ERR' or (ref"'!='"'N' and b"'!=ref): mism.append((i,b,ref))
print('"'mismatches',len(mism)); print(mism[:20])
PY"
