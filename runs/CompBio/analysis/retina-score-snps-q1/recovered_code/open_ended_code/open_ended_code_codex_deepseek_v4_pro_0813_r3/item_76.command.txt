/bin/bash -c "python - <<'PY'
import urllib.request,json
for build,pos in [('hg19',51540905),('hg38',46054917)]:
    chrom='chr10'
    for mode in ['a','b']:
        if mode=='a': start=pos-2; end=pos+1
        else: start=pos-1-1; end=pos+1
        url=f'https://api.genome.ucsc.edu/getData/sequence?genome={build};chrom={chrom};start={start};end={end}'
        d=json.load(urllib.request.urlopen(url))['dna']
        print(build,pos,mode,'start',start,'end',end,'len',len(d),d,'seq2',d[2] if len(d)>2 else None)
PY"
