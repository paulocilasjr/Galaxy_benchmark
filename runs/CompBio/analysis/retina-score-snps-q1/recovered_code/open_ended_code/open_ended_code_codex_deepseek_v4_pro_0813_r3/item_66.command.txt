/bin/bash -c "python - <<'PY'
import gzip
qtarget=46054916
path='ref/hg19ToHg38.over.chain.gz'
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        if p[2]=='chr10' and int(p[5])<=51540904<=int(p[6]):
            header=p; blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            break
t=int(header[5]); q=int(header[10])
found=False
for j,b in enumerate(blocks):
    parts=b.split()
    if len(parts)==3: size,dt,dq=map(int,parts)
    else: size=int(parts[0]); dt=dq=0
    if q <= qtarget < q+size:
        off=qtarget-q
        print('qtarget block',j,b,'t',t,'q',q,'size',size,'off',off,'t0',t+off,'t1',t+off+1)
        found=True; break
    t += size+dt; q += size+dq
print('found',found,'end t,q',t,q)
PY"
