/bin/bash -lc "tf_arm_env/bin/python - <<'PY'
import math
import numpy as np
import pandas as pd
from sklearn.metrics import average_precision_score
x=pd.read_csv('retina.score.snps.q1.rod_scored.tsv',sep='\\t')
y=x.label.to_numpy(); s=np.abs(x.Rod_fold_avg_lfc.to_numpy())
manual=0.0; tp=0; seen=0; positives=y.sum()
for threshold in np.unique(s)[::-1]:
 mask=s==threshold
 group_n=int(mask.sum()); group_pos=int(y[mask].sum())
 tp += group_pos; seen += group_n
 manual += (group_pos/positives)*(tp/seen)
lib=average_precision_score(y,s)
assert len(x)==500 and positives==250 and np.isfinite(s).all()
assert abs(manual-lib)<1e-15
assert math.floor(lib*100)/100==0.67
print(f'rows={len(x)} positives={positives} sklearn={lib:.15f} manual={manual:.15f} floor={math.floor(lib*100)/100:.2f}')
PY"
