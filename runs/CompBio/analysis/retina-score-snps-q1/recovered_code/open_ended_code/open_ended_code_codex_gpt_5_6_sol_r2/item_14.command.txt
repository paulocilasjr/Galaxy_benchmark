/bin/bash -lc "mkdir -p rod_models
PYTHONPATH=.deps python - <<'PY'
from remotezip import RemoteZip
from pathlib import Path
url='https://zenodo.org/api/records/6330053/files/retina_models.zip/content'
out=Path('rod_models')
with RemoteZip(url) as z:
 for fold in range(5):
  name=f'retina_models/fold{fold}/Rod.h5'
  dest=out/f'fold{fold}_Rod.h5'
  print('extracting',name,flush=True)
  with z.open(name) as src, dest.open('wb') as dst:
   while chunk:=src.read(8*1024*1024): dst.write(chunk)
  print(dest, dest.stat().st_size, flush=True)
PY
shasum rod_models/*.h5"
