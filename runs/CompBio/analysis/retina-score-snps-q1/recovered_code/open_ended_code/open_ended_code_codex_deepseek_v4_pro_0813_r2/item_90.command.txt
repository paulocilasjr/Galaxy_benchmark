/bin/bash -lc "python3 - <<'PY'
import pyfaidx, pandas as pd
from pyliftover import LiftOver
f19=pyfaidx.Fasta('genomes/hg19.fa'); f38=pyfaidx.Fasta('genomes/hg38.fa')
lo=LiftOver('genomes/hg19ToHg38.over.chain.gz')
df=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
for row in df.itertuples():
 if (row.chr,row.start) in [('chr10',46090343),('chr10',46054914)]:
  pass
# find rows whose liftover starts with these
for r in df.itertuples():
 res=lo.convert_coordinate(r.chr,r.start,strand='+')
 if res and res[0][1] in [46090344,46054915]:
  print('original',r.chr,r.start,r.ref,r.alt,'hg19 base0',f19[r.chr][r.start:r.start+1],'lift',res[0][1],'hg38 base0',f38[r.chr][res[0][1]:res[0][1]+1])
PY"
