b=open('zenodo/retina_models.tail','rb').read(); pos=b.find(b'PK\\x01\\x02',900000); end=b.rfind(b'PK\\x05\\x06')

while pos<end:
 if b[pos:pos+4]"'!=b'"'PK\\x01\\x02': print('bad',pos); break
 vals=struct.unpack_from('<4s6H3L5H2L',b,pos)
 # signature ver ver flags method time date crc csize usize nlen elen clen disk intattr extattr offset
 _,v,vn,fl,method,tm,dt,crc,cs,us,nl,el,cl,disk,ia,ea,off=vals
 name=b[pos+46:pos+46+nl].decode('utf8','replace')
 print(name, 'usize',us,'csize',cs,'method',method,'localoff',off)
 pos += 46+nl+el+cl; i+=1
print('count',i)
PY"
