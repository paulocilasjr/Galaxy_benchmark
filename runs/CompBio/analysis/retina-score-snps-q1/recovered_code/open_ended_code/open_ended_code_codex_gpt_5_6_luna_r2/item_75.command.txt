/bin/bash -lc "python - <<'PY'
import numpy as np
z=np.load('zenodo/rod_scores_center1057.npz'); y=z['labels'].astype(int); s=np.abs(z['lfcs'])
# Equivalent AP: precision at each positive's rank, with ties grouped in descending threshold order.
order=np.argsort(-s,kind='mergesort'); ss=s[order]; yy=y[order]
# sklearn groups ties; all values are effectively unique except exact ties
ap=0.; pos=0; prev=None
for j in range(len(y)):
 if prev is None or ss[j]"'!=prev:
  if prev is not None:
   pass
  prev=ss[j]
 # add at end of each tied group if any positive, recall jump * precision
 end=j==len(y)-1 or ss[j+1]!=ss[j]
 if end:
  group=yy[:j+1].sum()-pos; new=yy[:j+1].sum()
  if group: ap += (new/float(j+1))*((new-pos)/y.sum())
  pos=new
print('"'manual AP',repr(ap),'positives',y.sum())
print('floor',format(np.floor(ap*100)/100,'.2f'))
PY"
