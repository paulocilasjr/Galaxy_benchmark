/bin/bash -lc "native/python/bin/python3 - <<'PY'
import json,math,re
import numpy as np
import pandas as pd
from sklearn.metrics import average_precision_score
original=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
scored=pd.read_csv('retina.scored.tsv',sep='\\t')
assert original.equals(scored[original.columns])
preds=np.stack([np.load(f'fold{i}_predictions.npy') for i in range(5)])
assert preds.shape==(5,500,2) and np.isfinite(preds).all()
lfc=(preds[:,:,1]-preds[:,:,0]).mean(axis=0)
np.testing.assert_allclose(lfc,scored.Rod_fold_avg_lfc,rtol=1e-6,atol=1e-9)
y=original.label.to_numpy();score=np.abs(lfc)
order=np.argsort(-score,kind='stable')
ends=np.r_[np.flatnonzero(np.diff(score[order])),499]
tp=np.cumsum(y[order])[ends]
rank_ap=np.sum((tp/(ends+1))*np.diff(np.r_[0,tp/250]))
ap=average_precision_score(y,score)
assert abs(rank_ap-ap)<1e-12
answer=f'{math.floor(ap*100)/100:.2f}'
assert re.fullmatch(r'0\\.\\d{2}',answer)
assert answer==json.load(open('metrics.json'))['rounded_down']
print('Verified:',rank_ap,answer,'unique scores:',len(ends))
PY"
