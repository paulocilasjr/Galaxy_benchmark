/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
d1 <- GetAssayData(integrated, assay='RNA', layer='data.s1')
d2 <- GetAssayData(integrated, assay='RNA', layer='data.s2')
expr <- cbind(d1,d2)
genes <- rownames(expr)
pairs <- list(
 Horizontal=c('ONECUT1','LHX1'),
 Tcell=c('CD3D','CD3E'),
 Bcell=c('MS4A1','CD79A'),
 Microglia=c('C1QA','P2RY12'),
 Macrophage=c('CD163','MRC1'),
 RPE=c('RPE65','MLANA'),
 Endothelial=c('PECAM1','VWF'),
 Pericyte=c('PDGFRB','RGS5'),
 Fibroblast=c('DCN','COL1A1'),
 Schwann=c('MPZ','SOX10'),
 Astrocyte=c('GFAP','AQP4'),
 Muller=c('RLBP1','GLUL'),
 RGC=c('RBPMS','NEFL'),
 Cone=c('ARR3','GNAT2'),
 Bipolar=c('VSX2','GRM6'),
 Rod=c('RHO','NRL')
)
for(nm in names(pairs)){
 gs <- pairs[[nm]]
 if("'!all(gs %in% genes)) {cat(nm,'"'missing\\n'); next}
 for(thr in c(0.5,1)){
  mask <- expr[gs[1],] >= thr & expr[gs[2],] >= thr
  cat(sprintf('%s %s>=%g: ', nm, paste(gs,collapse='&'), thr))
  print(table(integrated"'$sample[mask]))
 }
}
EOF
Rscript [REDACTED LOCAL PATH]'
