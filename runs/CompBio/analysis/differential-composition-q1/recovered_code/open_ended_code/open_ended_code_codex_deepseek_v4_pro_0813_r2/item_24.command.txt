/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
d1 <- GetAssayData(integrated, assay='RNA', layer='data.s1')
d2 <- GetAssayData(integrated, assay='RNA', layer='data.s2')
expr <- cbind(d1,d2)
# ensure same cell order as meta? Create Seurat object after integration likely same colnames order maybe all s1 then s2.
stopifnot(all(colnames(expr) == colnames(integrated)))
genes <- rownames(expr)
count_expr <- function(genes_vec, cutoff=0.5) {
 g <- genes_vec[genes_vec %in% genes]
 if(length(g)==0) return(rep(0,ncol(expr)))
 x <- as.matrix(expr[g,,drop=FALSE])
 Matrix::colSums(x >= cutoff) > 0
}
marker_groups <- list(
 Tcell=c('CD3D','CD3E','CD3G','TRAC','LCK','IL7R'),
 Bcell=c('MS4A1','CD79A','CD19','BANK1'),
 Microglia=c('C1QA','C1QB','C1QC','CSF1R','P2RY12','TYROBP','CX3CR1','AIF1'),
 Macrophage=c('CD163','MRC1','LYZ','FCGR3A','S100A8','S100A9'),
 RPE=c('RPE65','BEST1','TTR','PMEL','MLANA','DCT','SERPINF1'),
 Endothelial=c('PECAM1','VWF','CLDN5','FLT1','KDR','EGFL7'),
 Pericyte=c('PDGFRB','RGS5','NOTCH3','ACTA2','CSPG4'),
 Fibroblast=c('DCN','COL1A1','COL1A2','LUM','COL3A1','COL6A1','COL6A2','PDGFRA'),
 Schwann=c('MPZ','PLP1','MBP','SOX10','GAP43'),
 Astrocyte=c('GFAP','GJA1','ALDH1L1','AQP4'),
 Muller=c('RLBP1','GLUL','AQP4','SOX9','SLC1A3','CLU','DKK3'),
 Horizontal=c('ONECUT1','ONECUT2','LHX1','CALB1'),
 Amacrine=c('TFAP2A','GAD1','GAD2','SLC6A9','SLC32A1','CALB2'),
 Bipolar=c('VSX2','GRM6','TRPM1','ISL1','GRIK1','VSX1','LHX4'),
 RGC=c('RBPMS','SLC17A6','POU4F1','POU4F2','THY1','NEFL','NEFM','SNCG'),
 Cone=c('ARR3','GNAT2','PDE6H','OPN1SW','OPN1MW','OPN1LW'),
 Rod=c('RHO','NRL','NR2E3','GNAT1','PDE6B','SAG')
)
for(nm in names(marker_groups)){
 pos <- count_expr(marker_groups[[nm]], cutoff=0.5)
 cat('\\n###', nm, 'positive cells by sample:', table(integrated"'$sample[pos]), '"'\\n')
 if(sum(pos)>0) print(table(integrated"'$sample[pos], integrated$seurat_clusters[pos]))
}
cat('"'\\nONECUT1 > 1 by sample and cluster\\n')
print(table(integrated"'$sample[expr['"'ONECUT1',] > 1], integrated"'$seurat_clusters[expr['"'ONECUT1',] > 1]))
cat('\\nLHX1 > 1 by sample and cluster\\n')
print(table(integrated"'$sample[expr['"'LHX1',] > 1], integrated"'$seurat_clusters[expr['"'LHX1',] > 1]))
cat('\\nONECUT1 >0.5 & LHX1 >0.5 by sample/cluster\\n')
mask <- expr['ONECUT1',] > 0.5 & expr['LHX1',] > 0.5
print(table(integrated"'$sample[mask], integrated$seurat_clusters[mask]))
cat('"'\\nT cell CD3D >0.5 by sample/cluster\\n')
mask <- expr['CD3D',] > 0.5
print(table(integrated"'$sample[mask], integrated$seurat_clusters[mask]))
EOF
Rscript [REDACTED LOCAL PATH]'
