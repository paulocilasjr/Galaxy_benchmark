/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
d1 <- GetAssayData(integrated, assay='RNA', layer='data.s1')
d2 <- GetAssayData(integrated, assay='RNA', layer='data.s2')
expr <- cbind(d1,d2)
genes <- rownames(expr)
# print per-sample number of cells with specific genes above thresholds and clusters distribution
spec <- c('CD3D','CD3E','CD3G','MS4A1','CD79A','CD19','C1QA','C1QB','C1QC','P2RY12','CSF1R','AIF1','CX3CR1','LYZ','CD163','MRC1','RPE65','BEST1','TTR','PMEL','MLANA','VWF','PECAM1','CLDN5','PDGFRB','RGS5','DCN','COL1A1','MPZ','SOX10','GFAP','AQP4','RLBP1','GLUL','ONECUT1','ONECUT2','LHX1','CALB1','TFAP2A','GAD1','SLC6A9','SLC32A1','RBPMS','SLC17A6','POU4F1','THY1','RHO','NRL','ARR3','GNAT2','PDE6H','VSX2','GRM6','TRPM1','ISL1','GRIK1')
for(g in spec){
 if("'!(g %in% genes)) {cat(g,'"'missing\\n'); next}
 vals <- expr[g,]
 for(thr in c(1,2)){
  mask <- vals >= thr
  if(sum(mask)>0){
   tb <- table(integrated"'$sample[mask])
   cat(sprintf('"'%s >= %g: ', g, thr)); print(tb)
   # top clusters
   tbc <- table(integrated"'$sample[mask], integrated$seurat_clusters[mask])
   if(sum(mask)<=30) { print(tbc) } else { print(rowSums(tbc)) }
  }
 }
}
EOF
Rscript [REDACTED LOCAL PATH]'
