/bin/bash -lc "Rscript - <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
d1c <- GetAssayData(integrated, assay='RNA', layer='counts.s1')
d2c <- GetAssayData(integrated, assay='RNA', layer='counts.s2')
for(g in c('ONECUT1','ONECUT2','LHX1','CALB1','TFAP2A','GAD1')){
 cat(g,'raw positives sample1',sum(d1c[g,]>0),'sample2',sum(d2c[g,]>0),'\\n')
 cat('sum counts s1',sum(d1c[g,]),'s2',sum(d2c[g,]),'\\n')
}
# cells with ONECUT1 raw >=1 and LHX1 raw >=1
m1 <- d1c['ONECUT1',]>=1 & d1c['LHX1',]>=1
m2 <- d2c['ONECUT1',]>=1 & d2c['LHX1',]>=1
cat('ONECUT1&LHX1 raw >=1: sample1',sum(m1),'sample2',sum(m2),'\\n')
# top ONECUT1 raw in sample2
ord <- order(d2c['ONECUT1',], decreasing=TRUE)
print(head(d2c['ONECUT1',ord],10))
print(head(d2c['LHX1',ord],10))
EOF"
