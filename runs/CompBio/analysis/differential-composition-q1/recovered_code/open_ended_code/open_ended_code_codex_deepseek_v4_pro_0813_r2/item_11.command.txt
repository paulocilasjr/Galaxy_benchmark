/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
suppressPackageStartupMessages(library(Matrix))
genes <- read.delim(gzfile('differential.composition.q1.genes.txt.gz'), stringsAsFactors=FALSE)
cat('genes dim', dim(genes), 'head', head(genes"'$gene_symbols), '"'\\n')
mat1 <- Matrix::readMM(gzfile('differential.composition.q1.1.mtx.gz'))
mat2 <- Matrix::readMM(gzfile('differential.composition.q1.2.mtx.gz'))
cat('mat dims', dim(mat1), dim(mat2), '\\n')
# check row sums and column sums
cat('row sums sample1', sum(Matrix::rowSums(mat1)>0), 'of', nrow(mat1), '\\n')
cat('col sums sample1 min max median', min(Matrix::colSums(mat1)), max(Matrix::colSums(mat1)), median(Matrix::colSums(mat1)), '\\n')
cat('col sums sample2 min max median', min(Matrix::colSums(mat2)), max(Matrix::colSums(mat2)), median(Matrix::colSums(mat2)), '\\n')
# check gene order names; no barcodes
colnames(mat1) <- paste0('s1_', seq_len(ncol(mat1)))
colnames(mat2) <- paste0('s2_', seq_len(ncol(mat2)))
rownames(mat1) <- genes"'$gene_symbols
rownames(mat2) <- genes$gene_symbols
# save as RDS perhaps
saveRDS(list(mat1=mat1, mat2=mat2, genes=genes), '"'mats.rds')
cat('saved\\n')
EOF
Rscript [REDACTED LOCAL PATH]"
