/bin/bash -lc "python - <<'PY'
import gzip
import numpy as np, pandas as pd
from scipy.io import mmread

genes=pd.read_csv('differential.composition.q1.genes.txt.gz',sep='\\t',compression='gzip')['gene_symbols'].str.upper().to_numpy()
g={x:i for i,x in enumerate(genes)}
sets={
'mac_no_apoe':['LYZ','CD68','CD163','MRC1','MSR1','MARCO'],
'mac_core':['LYZ','CD68','CD163','MRC1'],
'micro_core':['P2RY12','CX3CR1','TMEM119','AIF1','C1QA','C1QB','TYROBP'],
'apoe':['APOE'],
'cone':['ARR3','GNAT2','PDE6C','OPN1LW','OPN1MW','OPN1SW'],
'muller':['RLBP1','GLUL','SLC1A3','VIM','AQP4','CRYAB','CLU'],
}
idx={k:[g[x] for x in v if x in g] for k,v in sets.items()}
for s,path in [('s1','differential.composition.q1.1.mtx.gz'),('s2','differential.composition.q1.2.mtx.gz')]:
 with gzip.open(path,'rb') as f: X=mmread(f).tocsc().astype(np.float32)
 lib=np.asarray(X.sum(axis=0)).ravel(); scale=10000/np.maximum(lib,1)
 Xn=X.copy()
 for j in range(Xn.shape[1]): Xn.data[Xn.indptr[j]:Xn.indptr[j+1]]*=scale[j]
 Xn.data=np.log1p(Xn.data)
 print('\\n',s, X.shape, 'median lib', np.median(lib))
 for name,rows in idx.items():
  score=np.asarray(Xn[rows,:].mean(axis=0)).ravel() if rows else np.zeros(X.shape[1])
  det=np.asarray((X[rows,:]>0).sum(axis=0)).ravel() if rows else np.zeros(X.shape[1])
  print(name,'markers', [genes[i] for i in rows], 'mean_score',score.mean(), 'cells_score>0.25',int((score>0.25).sum()), 'cells_score>0.5',int((score>0.5).sum()), 'cells_2+det',int((det>=2).sum()), 'cells_3+det',int((det>=3).sum()))
 # top cells by mac_no_apoe
 rows=idx['mac_no_apoe']; score=np.asarray(Xn[rows,:].mean(axis=0)).ravel(); order=np.argsort(score)[-20:][::-1]
 print('top mac_no_apoe scores')
 for j in order[:10]:
  vals={genes[i]:float(Xn[i,j]) for i in rows if Xn[i,j]>0}
  print(j, score[j], vals)
PY"
