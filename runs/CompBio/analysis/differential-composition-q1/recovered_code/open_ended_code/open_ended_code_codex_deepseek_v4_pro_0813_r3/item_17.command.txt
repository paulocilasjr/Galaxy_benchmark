/bin/bash -lc "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('processed.h5ad')
for genes in [['ONECUT1','ONECUT2','LHX1','CALB1','C1QL1','PRDM13','TNR','SEMA5A'], ['ARR3','GNAT2','PDE6C','PDE6H','RXRG'], ['RHO','NRL','PDE6A','CNGB1','SAG','RCVRN'], ['RBPMS','SNCG','NEFL','POU4F1'], ['VSX2','OTX2','GRM6','TRPM1','CABP5']]:
 print('\\n', genes)
 sub=adata[:, [g for g in genes if g in adata.var_names]]
 X=sub.X.toarray() if hasattr(sub.X,'toarray') else sub.X
 df=pd.DataFrame(X,index=adata.obs_names,columns=sub.var_names)
 df['leiden']=adata.obs['leiden'].values
 print(df.groupby('leiden').mean().round(3).to_string())
 print('pct positive')
 print((df.drop(columns='leiden')>0).groupby(df['leiden']).mean().round(3).to_string())
PY"
