/bin/bash -lc "python3 - <<'PY'
import gzip, scipy.io, numpy as np, pandas as pd, scanpy as sc, os
from scipy.sparse import csr_matrix, hstack, vstack
import anndata as ad
f1='differential.composition.q1.1.mtx.gz'; f2='differential.composition.q1.2.mtx.gz'
m1=scipy.io.mmread(gzip.open(f1,'rt')).tocsr()
m2=scipy.io.mmread(gzip.open(f2,'rt')).tocsr()
print(m1.shape, m2.shape, m1.nnz, m2.nnz)
genes=[]
with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
 f.readline()
 for line in f:
  genes.append(line.rstrip('\\n').split('\\t')[1])
print(len(genes), genes[:2], genes[-2:])
adata1=ad.AnnData(m1.T, obs=pd.DataFrame(index=[f'individual1_{i}' for i in range(m1.shape[1])]), var=pd.DataFrame(index=genes))
adata1.obs['individual']='individual1'
adata2=ad.AnnData(m2.T, obs=pd.DataFrame(index=[f'individual2_{i}' for i in range(m2.shape[1])]), var=pd.DataFrame(index=genes))
adata=ad.concat([adata1, adata2], join='outer', merge='same')
adata.obs_names_make_unique()
print(adata)
# basic stats
sc.pp.calculate_qc_metrics(adata, inplace=True, percent_top=None)
print(adata.obs[['n_genes_by_counts','total_counts']].describe())
# save
adata.write_h5ad('combined_raw.h5ad')
PY"
