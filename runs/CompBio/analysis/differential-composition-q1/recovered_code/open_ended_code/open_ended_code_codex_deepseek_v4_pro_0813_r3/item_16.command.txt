/bin/bash -lc "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('processed.h5ad')
# define marker genes for each cell type in list
markers={
'astrocyte':['GFAP','AQP4','S100B','ALDH1L1'],
'microglial cell':['C1QA','C1QB','C1QC','P2RY12','CSF1R','AIF1'],
'RPE cell':['RPE65','BEST1','TYR','PMEL','MLANA','RGR','SERPINF1'],
'cone cell':['ARR3','GNAT2','PDE6C','PDE6H','OPN1SW','OPN1MW','RXRG'],
'rod cell':['RHO','NRL','PDE6A','PDE6B','GNAT1','CNGA1','CNGB1','SAG','RCVRN'],
'horizontal cell':['ONECUT1','ONECUT2','LHX1','CALB1','PRDM13','C1QL1'],
'bipolar cell':['VSX2','OTX2','GRM6','TRPM1','CABP5','GRIK1'],
'RGC':['RBPMS','SNCG','NEFL','NEFM','POU4F1','POU4F2','THY1','GAP43'],
'T cells':['CD3D','CD3E','CD3G','TRAC','LCK','IL7R'],
'macrophage':['CD68','CD163','LYZ','MRC1'],
'Schwann cell':['MPZ','PMP22','SOX10','PLP1'],
'Pericyte':['RGS5','PDGFRB','NOTCH3','CSPG4'],
'B cell':['CD79A','CD79B','MS4A1','CD19','PAX5'],
'fibroblast':['COL1A1','COL1A2','DCN','LUM'],
'endothelial cell':['PECAM1','VWF','CDH5','FLT1','KDR','CLDN5'],
'muller glia cell':['RLBP1','GLUL','SLC1A3','AQP4','VIM','CLU','DKK3','SOX9'],
}
# compute mean expression per cluster and pct
for ct, genes in markers.items():
    genes=[g for g in genes if g in adata.var_names]
    if not genes: continue
    X=adata[:,genes].X
    if hasattr(X,'toarray'): X=X.toarray()
    mean=pd.Series(X.mean(axis=1), index=adata.obs_names)
    grouped=mean.groupby(adata.obs['leiden']).mean()
    top=grouped.sort_values(ascending=False).head(5)
    print('\\n',ct, top.round(2).to_dict())
    # pct cells with any? maybe
    expr=pd.Series((X>0).sum(axis=1), index=adata.obs_names)
    pct=expr.groupby(adata.obs['leiden']).mean()/len(genes)*100
    print('pcttop', pct.sort_values(ascending=False).head(5).round(1).to_dict())
PY"
