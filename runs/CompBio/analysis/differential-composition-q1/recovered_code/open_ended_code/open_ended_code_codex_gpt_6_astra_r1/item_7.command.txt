/bin/bash -lc "cat > verify.py <<'PY'
import numpy as np
import pandas as pd
from scipy.sparse import load_npz
G=pd.read_csv('differential.composition.q1.genes.txt.gz',sep='\\t').gene_symbols.values
markers=['ONECUT1','ONECUT2','LHX1','PROX1','CALB1','GAD1','GAD2','TFAP2A','TFAP2B','LHX2']
for s in [1,2]:
 X=load_npz(f'counts{s}.npz')
 A=X[:,[np.where(G==g)[0][0] for g in markers]].toarray()
 positive=(A[:,:3]>0).sum(1)
 print('sample',s,'horizontal-specific markers detected:',np.bincount(positive,minlength=4))
 mask=positive>=2
 print('horizontal candidate count',sum(mask),'marker means',dict(zip(markers,np.round(A[mask].mean(0),2))))
 print('LHX1 positive counts',A[A[:,2]>0,:3].tolist()[:15])
 print('median UMIs',np.median(np.asarray(X.sum(1)).ravel()))
PY
python verify.py"
