/bin/bash -lc "python3 -u - <<'PY'
import gzip, scipy.io, numpy as np, pandas as pd
base='differential.composition.q1'
with gzip.open(base+'.genes.txt.gz','rt') as f:
 f.readline(); genes=[line.rstrip('\\n').split('\\t')[1] for line in f]
Ms=[]; libs=[]
for j in [1,2]:
 m=scipy.io.mmread(base+f'.{j}.mtx.gz').tocsr().astype(np.float32); lib=np.asarray(m.sum(axis=0)).ravel(); libs.append(lib)
 norm=m.multiply(1e4/lib).tocsr(); Ms.append(norm)
a=np.asarray(Ms[0].mean(axis=1)).ravel(); b=np.asarray(Ms[1].mean(axis=1)).ravel()
detect=[]
for m in [Ms[0],Ms[1]]: detect.append(np.asarray((m>0).mean(axis=1)).ravel())
# genes common detected / major expression
for title,vals in [('sample1 mean',a),('sample2 mean',b),('diff b-a',b-a),('log2ratio',np.log2((b+.01)/(a+.01)))]:
 print('\\nTOP',title)
 for i in np.argsort(-vals)[:50]: print(genes[i],round(float(vals[i]),3), 'det1',round(float(detect[0][i]),3),'det2',round(float(detect[1][i]),3),'means',round(float(a[i]),3),round(float(b[i]),3))
print('\\nTOP absolute standardized differences?')
# require gene mean > .1 and top relative diff
mask=(a>0.1)|(b>0.1)
for i in np.argsort(-(b-a))[mask[np.argsort(-(b-a))]][:100]: print('up2',genes[i],round(float(a[i]),3),round(float(b[i]),3),round(float(b[i]-a[i]),3),round(float(np.log2((b[i]+.01)/(a[i]+.01))),2))
print('--- up1')
for i in np.argsort(a-b)[mask[np.argsort(a-b)]][:100]: print('up1',genes[i],round(float(a[i]),3),round(float(b[i]),3),round(float(a[i]-b[i]),3),round(float(np.log2((a[i]+.01)/(b[i]+.01))),2))
PY"
