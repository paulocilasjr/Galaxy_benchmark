import gzip, scipy.io, numpy as np, time
from scipy.sparse import vstack
base='differential.composition.q1'
with gzip.open(base+'.genes.txt.gz','rt') as f: f.readline(); genes=[x.rstrip().split('\\t')[1] for x in f]

A=[]
for j in [1,2]:
 print('reading',j,flush=True); m=scipy.io.mmread(base+f'.{j}.mtx.gz').tocsr().astype(np.float32); print('read',m.nnz,flush=True); lib=np.asarray(m.sum(axis=0)).ravel(); x=m.multiply(1e4/(lib+1e-9)).tocsr(); x.data=np.log1p(x.data); A.append(x.T.tocsr())
X=vstack(A).tocsr(); print('X',X.shape,X.nnz,flush=True)

from sklearn.decomposition import TruncatedSVD
print('fit',flush=True); t=time.time(); svd=TruncatedSVD(20,random_state=1,n_iter=2).fit(X[:,hvg]); print('fitdone',time.time()-t,flush=True); print(svd.transform(X[:,hvg]).shape,flush=True)
