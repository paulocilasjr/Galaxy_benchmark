/bin/bash -lc "python3 -u - <<'PY'
import gzip, scipy.io, numpy as np
base='differential.composition.q1'
with gzip.open(base+'.genes.txt.gz','rt') as f:
 f.readline(); genes=[line.rstrip('\\n').split('\\t')[1] for line in f]
print('genes',len(genes),flush=True)
idx={g:i for i,g in enumerate(genes)}
for j in [1,2]:
 print('reading',j,flush=True)
 m=scipy.io.mmread(base+f'.{j}.mtx.gz').tocsr().astype(np.float32)
 print('matrix',m.shape,m.nnz,flush=True)
 lib=np.asarray(m.sum(axis=0)).ravel()
 print('lib',lib.mean(),flush=True)
 for g in ['GFAP','AQP4','RPE65','RHO','ARR3','CD3D','LYZ','RBPMS','RGS5','COL1A1','EMCN','RLBP1']:
  i=idx.get(g)
  print(g, i, float(m[i].sum()) if i is not None else None, int((m[i]>0).sum()) if i is not None else None,flush=True)
PY"
