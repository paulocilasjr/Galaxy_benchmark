/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
from scipy.io import mmread
from collections import defaultdict

with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
    f.readline(); genes=[line.rstrip('\\n').split('\\t')[1] for line in f]
sym=defaultdict(list)
for i,g in enumerate(genes): sym[g.upper()].append(i)
sets={
'astrocyte':['GFAP','AQP4','ALDH1L1','SLC1A3','SLC1A2','GJA1'],
'muller':['RLBP1','GLUL','VIM','SLC1A3','SLC1A2','APOE','CLU','AQP4','SOX9'],
'horizontal':['PROX1','ONECUT1','ONECUT2','LHX1','CALB1','CALB2','TFAP2A','TFAP2B'],
'horizontal2':['LHX1','ONECUT1','ONECUT2','PROX1','CALB1'],
'macrophage':['CD68','LYZ','CD14','FCGR3A','LST1','S100A8','S100A9','MSR1'],
'microglia':['P2RY12','CX3CR1','C1QA','C1QB','C1QC','AIF1','TMEM119','TYROBP'],
'T':['CD3D','CD3E','CD3G','TRAC','LCK','CD2'],
'B':['MS4A1','CD79A','CD79B','CD19','IGHM'],
'endothelial':['PECAM1','VWF','CLDN5','FLT1','ESAM','RAMP2'],
'pericyte':['RGS5','PDGFRB','CSPG4','MCAM','NOTCH3'],
'RPE':['RPE65','BEST1','PMEL','TYR','MITF','LRAT','RDH5'],
'RGC':['RBPMS','POU4F1','POU4F2','SNCG','THY1'],
'cone':['OPN1LW','OPN1MW','OPN1SW','GNAT2','ARR3','PDE6H','PDE6C'],
'bipolar':['VSX2','GRM6','TRPM1','PKD2L1','CABP5','PRKCA','PCP2'],
'rod':['RHO','NRL','GNAT1','PDE6A','PDE6B','SAG','GNGT1']
}
for lab,fn in [('1','differential.composition.q1.1.mtx.gz'),('2','differential.composition.q1.2.mtx.gz')]:
    X=mmread(fn).tocsc()
    print('\\nSample', lab, 'cells', X.shape[1])
    for name,ms in sets.items():
        idx=[i for m in ms for i in sym.get(m.upper(),[])]
        sub=X[idx,:].tocsc()
        # number marker genes detected per cell and total counts markers
        det=np.diff(sub.astype(bool).tocsc().indptr) # wait indptr over columns if csc rows selected ok
        total=np.asarray(sub.sum(axis=0)).ravel()
        counts=[]
        for n in [1,2,3,4]: counts.append((det>=n).sum())
        print(f'{name:12s} markers {len(idx):2d} det>=1/2/3/4 {counts} total>=3/5/10 {[(total>=t).sum() for t in [3,5,10]]}')
        # individual gene expr counts
        ge=[]
        for m in ms:
            ii=sym.get(m.upper(),[])
            if ii:
                c=(np.asarray(X[ii,:].sum(axis=0)).ravel()>0).sum()
                ge.append(f'{m}:{c}')
        print(' ', ' '.join(ge))
PY"
