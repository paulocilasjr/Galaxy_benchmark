/bin/bash -lc "python3 - <<'PY'
# quick exact same reproduce and print imbalanced top markers incl known genes for k=24, maybe previous label IDs shifted due unstable KMeans threading
import gzip, numpy as np
from scipy.io import mmread
from scipy import sparse
from sklearn.decomposition import TruncatedSVD
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from collections import defaultdict
with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f: f.readline(); genes=[l.rstrip('\\n').split('\\t')[1] for l in f]
sym=defaultdict(list)
for i,g in enumerate(genes): sym[g.upper()].append(i)
X1=mmread('differential.composition.q1.1.mtx.gz').tocsc().astype(np.float32); X2=mmread('differential.composition.q1.2.mtx.gz').tocsc().astype(np.float32)
X=sparse.hstack([X1,X2],format='csc').T.tocsr(); batch=np.array([0]*X1.shape[1]+[1]*X2.shape[1])
lib=np.asarray(X.sum(axis=1)).ravel(); Xn=X.copy().astype(np.float32); rows=np.repeat(np.arange(Xn.shape[0]), np.diff(Xn.indptr)); Xn.data=np.log1p(Xn.data*10000/lib[rows])
mean=np.asarray(Xn.mean(axis=0)).ravel(); mean2=np.asarray(Xn.power(2).mean(axis=0)).ravel(); var=mean2-mean**2; nnz=np.diff(Xn.tocsc().indptr); mask=(mean>0.005)&(mean<5)&(nnz>=20); score=np.zeros_like(mean); score[mask]=var[mask]/(mean[mask]+1e-3); hvg=np.argsort(score)[-3000:]
Z=StandardScaler().fit_transform(TruncatedSVD(n_components=50,random_state=1).fit_transform(Xn[:,hvg]))
lab=KMeans(n_clusters=24,n_init=20,random_state=42).fit_predict(Z)
markers={'H':['PROX1','ONECUT1','ONECUT2','LHX1','CALB1','CALB2','TFAP2A','TFAP2B'], 'Bip':['VSX2','GRM6','TRPM1','PKD2L1','CABP5','PRKCA','PCP2'], 'Muller':['RLBP1','GLUL','VIM','SLC1A3','SLC1A2','APOE','CLU','AQP4','SOX9'], 'Astro':['GFAP','AQP4','ALDH1L1','SLC1A3','SLC1A2','SOX9'], 'Rod':['RHO','SAG','GNAT1','NRL'], 'Cone':['ARR3','PDE6H','GNAT2','OPN1LW']}
mid={k:[i for m in ms for i in sym.get(m,[])] for k,ms in markers.items()}
rows=[]
for c in range(24):
 cells=lab==c; n1=((cells)&(batch==0)).sum(); n2=((cells)&(batch==1)).sum(); n=n1+n2
 p1=n1/X1.shape[1]; p2=n2/X2.shape[1]
 if n<20: continue
 ratios=(p1+1e-9)/(p2+1e-9)
 scores={k:float(np.asarray(Xn[cells][:,idx].mean(axis=1)).mean()) for k,idx in mid.items()}
 top=max(scores,key=scores.get)
 rows.append((max(ratios,1/ratios),c,n,n1,n2,ratios,top,scores))
for imb,c,n,n1,n2,ratio,top,scores in sorted(rows, reverse=True)[:15]:
 cells=lab==c; mi=np.asarray(Xn[cells].mean(axis=0)).ravel(); mo=np.asarray(Xn[~cells].mean(axis=0)).ravel(); order=np.argsort(mi-mo)[::-1]
 print('\\ncl',c,'n',n,'n1',n1,'n2',n2,'ratio',ratio,'top',top,scores)
 print('topgenes', ', '.join(genes[i] for i in order[:12] if mi[i]>.05))
PY"
