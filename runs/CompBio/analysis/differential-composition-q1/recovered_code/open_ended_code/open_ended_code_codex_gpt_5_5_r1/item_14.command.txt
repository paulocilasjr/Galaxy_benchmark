/bin/bash -lc "cat > marker_gene_stats.py <<'PY'
import gzip
import numpy as np
from scipy.io import mmread

with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
    next(f); genes=[line.rstrip('\\n').split('\\t')[1].upper() for line in f]
idx={g:i for i,g in enumerate(genes)}
with gzip.open('differential.composition.q1.1.mtx.gz','rb') as f: X1=mmread(f).tocsc().astype(np.float32)
with gzip.open('differential.composition.q1.2.mtx.gz','rb') as f: X2=mmread(f).tocsc().astype(np.float32)
libs=[np.asarray(X.sum(axis=0)).ravel() for X in [X1,X2]]
sets={
'RPE_specific':['RPE65','BEST1','PMEL','TYR','MITF','MERTK','RGR','LRAT','RDH5','RBP1'],
'rod_specific':['RHO','NRL','GNAT1','PDE6A','PDE6B','GNGT1','CNGA1','ROM1','SAG'],
'cone_specific':['OPN1SW','OPN1MW','OPN1LW','ARR3','GNAT2','PDE6C','GNGT2','CNGB3'],
'muller_specific':['GLUL','APOE','CLU','VIM','CRYAB','SLC1A3','SLC1A2','RLBP1'],
'horizontal_specific':['ONECUT1','ONECUT2','LHX1','PROX1','CALB1'],
'bipolar_specific':['VSX2','TRPM1','GRM6','PRKCA','CABP5','PCP2'],
'RGC_specific':['RBPMS','SNCG','NEFL','NEFM','POU4F1','POU4F2'],
}
print('gene\\ts1_sum\\ts2_sum\\ts1_cpm\\ts2_cpm\\tratio\\tpos1\\tpos2\\tq99_1\\tq99_2')
for name,gs in sets.items():
    print('##',name)
    for g in gs:
        if g not in idx: continue
        vals=[]
        for X,lib in zip([X1,X2],libs):
            row=np.asarray(X[idx[g],:].todense()).ravel()
            norm=np.log1p(row/np.maximum(lib,1)*1e4)
            vals.append((row.sum(), row.sum()/lib.sum()*1e6, (row>0).mean(), np.quantile(norm,.99), np.quantile(norm,.999), norm.max()))
        print(f\"{g}\\t{vals[0][0]:.0f}\\t{vals[1][0]:.0f}\\t{vals[0][1]:.2f}\\t{vals[1][1]:.2f}\\t{(vals[1][1]+1e-9)/(vals[0][1]+1e-9):.2f}\\t{vals[0][2]:.3f}\\t{vals[1][2]:.3f}\\t{vals[0][3]:.2f}\\t{vals[1][3]:.2f}\\tmax {vals[0][5]:.2f}/{vals[1][5]:.2f}\")
    inds=[idx[g] for g in gs if g in idx]
    for X,lib,lab in [(X1,libs[0],'s1'),(X2,libs[1],'s2')]:
        m=np.asarray(X[inds,:].sum(axis=0)).ravel()
        norm=np.log1p(m/np.maximum(lib,1)*1e4)
        print(name,lab,'set_pos', (m>0).mean(), 'set_q90/q95/q99', np.quantile(norm,[.9,.95,.99]), 'cells norm>1/2/3', [(norm>t).mean() for t in [1,2,3,4]])
PY
python3 marker_gene_stats.py"
