import gzip, numpy as np, math, time

def load(path):
    with gzip.open(path,'rt') as f:
        f.readline()
        for l in f:
            if not l.startswith('%'):
                nrow,ncol=map(int,l.split()); break
        data=np.fromiter((float(x) for x in f), dtype=np.float64, count=nrow*ncol)

def pca_scores(X, ncomp=50):
    # cells x genes input
    Xc=X-X.mean(axis=0)
    U,S,Vt=np.linalg.svd(Xc, full_matrices=False)
    return U[:,:ncomp]*S[:ncomp], S

def kmeans(X,k,ninit=20,max_iter=200,seed=1):
    rng=np.random.default_rng(seed)
    best=None
    n=X.shape[0]
    x2=(X*X).sum(1)

def silhouette_sample(X, labels, seed=2, maxn=1500):
    rng=np.random.default_rng(seed); n=X.shape[0]
