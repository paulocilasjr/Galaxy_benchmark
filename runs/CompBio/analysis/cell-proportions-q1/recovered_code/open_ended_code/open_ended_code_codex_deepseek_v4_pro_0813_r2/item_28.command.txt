/bin/bash -c "python3 - <<'PY'
import scanpy as sc, numpy as np, gzip
from scipy.io import mmread
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f: M=mmread(f)
X=np.asarray(M.T,dtype=float)
adata=sc.read('adata_unfiltered.h5ad'); cl=adata.obs['leiden_0.5'].astype(int).values
lib=X.sum(1); Xn=np.log1p(X/lib[:,None]*10000)
# mean normalized expression by 50-gene blocks across all cells and clusters
for start in range(0,1000,50):
 m=Xn[:,start:start+50].mean()
 print(f'{start:3d}-{start+49:3d} allmean {m:.3f}')
# For each cluster, mean block scores (top A,B,C and other)
blocks={}
for b in range(0,1000,100): blocks[f'{b}-{b+99}']=Xn[:,b:b+100].mean(1)
for c in sorted(np.unique(cl)):
 print('\\ncluster',c)
 vals=sorted(blocks.items(), key=lambda kv: -np.mean(kv[1][cl==c]))
 for name,v in vals[:5]: print(name, np.mean(v[cl==c]).round(3))
PY"
