import gzip, numpy as np
from scipy.io import mmread
from scipy import stats
path='cell.proportions.q1.mtx.gz'
with gzip.open(path,'rb') as f:
    M=mmread(f).astype(float)
# genes x cells -> cells x genes
X=M.T # 5350x1000
lib=np.sum(X,axis=1)
genes=np.sum(X>0,axis=1)
for arr,name in [(lib,'lib'),(genes,'genes')]:
 print(name,'min q01 q05 q10 median q90 q95 q99 max')
 qs=[0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]
 print([np.quantile(arr,q) for q in qs])
# cells with low/high lib sorted
idx=np.argsort(lib)
print('lowest lib')
for i in idx[:20]: print(i, lib[i], genes[i])
print('highest lib')
for i in idx[-20:]: print(i, lib[i], genes[i])
# number detected vs lib correlation
print('corr lib/genes', np.corrcoef(lib,genes)[0,1])
# Determine potential QC thresholds common median +/- 3 MAD etc
for arr,name in [(lib,'lib'),(genes,'genes')]:
 med=np.median(arr); mad=np.median(np.abs(arr-med))
 print(name,'median',med,'mad',mad,'low med-3mad',med-3*mad,'high med+3mad',med+3*mad,'low count',(arr<med-3*mad).sum(),'high count',(arr>med+3*mad).sum())
# top expressed genes mean, variance
gm=X.mean(axis=0)
gv=X.var(axis=0)
idx=np.argsort(gm)[::-1]
print('top genes by mean')
for i in idx[:30]: print(i, 'mean',gm[i], 'var',gv[i], 'detect',(X[:,i]>0).mean())
