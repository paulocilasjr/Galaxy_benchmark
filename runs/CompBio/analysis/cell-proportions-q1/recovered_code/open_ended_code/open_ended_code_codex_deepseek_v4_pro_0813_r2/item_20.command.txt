import scanpy as sc, numpy as np, gzip
from scipy.io import mmread
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f:
    M=mmread(f)
adata=sc.AnnData(np.asarray(M.T,dtype='float32'))
adata.obs_names=[f'cell{i}' for i in range(adata.n_obs)]
adata.var_names=[f'gene{i}' for i in range(adata.n_vars)]
adata.obs['nCount']=np.asarray(adata.X.sum(1)).ravel()
adata.obs['nFeature']=np.asarray((adata.X>0).sum(1)).ravel()

adata=adata[adata.obs.nFeature<=929].copy()
print('after',adata.n_obs)
sc.pp.normalize_total(adata,target_sum=1e4)
sc.pp.log1p(adata)
adata.raw=adata
sc.pp.scale(adata,max_value=10)
sc.tl.pca(adata,n_comps=50,svd_solver='arpack')
sc.pp.neighbors(adata,n_neighbors=15,n_pcs=20)
sc.tl.umap(adata)
for res in [0.1,0.2,0.3,0.5,0.8,1.0]:
 sc.tl.leiden(adata,resolution=res,key_added=f'leiden_{res}')
 print('res',res, np.bincount(adata.obs[f'leiden_{res}'].astype(int)))
# silhouette kmeans on PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

for k in range(2,8):
 km=KMeans(n_clusters=k,random_state=0,n_init=10).fit(Z)
 print('kmeans k',k,np.bincount(km.labels_),silhouette_score(Z,km.labels_,sample_size=2000,random_state=0))
