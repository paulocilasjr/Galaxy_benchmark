import gzip,numpy as np
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import confusion_matrix
with gzip.open('cell.proportions.q1.mtx.gz','rt') as f:
 for line in f:
  if not line.startswith('%'): break
 dims=tuple(map(int,line.split())); vals=np.fromiter((float(x) for x in f),dtype=np.float32)
X=vals.reshape(dims,order='F').T; lib=X.sum(1); det=(X>0).sum(1)
Y=np.log1p(X/lib[:,None]*1e4).astype(np.float32)
idx=np.argsort(Y.var(0))[-300:]
p=PCA(n_components=20,random_state=1,svd_solver='randomized').fit_transform(Y[:,idx])
for k in [3,4,5,6,7,8]:
 lab=KMeans(n_clusters=k,n_init=50,random_state=1).fit_predict(p)
 print('\\nK',k)
 for c in sorted(set(lab), key=lambda z:-sum(lab==z)):
  ii=np.where(lab==c)[0]; print(c,len(ii),'range',ii.min(),ii.max(),'mean idx',ii.mean(),'lib',round(lib[ii].mean(),1),'det',round(det[ii].mean(),1),'pcmean',np.round(p[ii].mean(0)[:4],2))
 # counts by contiguous blocks 250/500/1000 perhaps
 print('labels first 100',lab[:100].tolist())
PY"
