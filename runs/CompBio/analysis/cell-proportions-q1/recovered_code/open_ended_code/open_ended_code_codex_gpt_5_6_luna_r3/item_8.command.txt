/bin/bash -lc "python -u - <<'PY'
import gzip, numpy as np, time
p='cell.proportions.q1.mtx.gz'
t=time.time()
with gzip.open(p,'rt') as f:
    for line in f:
        if not line.startswith('%'): break
    dims=tuple(map(int,line.split()))
    data=np.fromiter((float(line) for line in f),dtype=np.float32)
print('dims',dims,'n',data.size,'time',time.time()-t,'minmax',data.min(),data.max(),'mean',data.mean(),'zeros',np.count_nonzero(data==0))
X=data.reshape(dims,order='F') # genes x cells
cs=X.sum(axis=0); nn=(X>0).sum(axis=0); rs=X.sum(axis=1); rn=(X>0).sum(axis=1)
for name,a in [('cell totals',cs),('cell detected',nn),('gene totals',rs),('gene detected',rn)]:
 q=np.percentile(a,[0,0.1,1,5,10,25,50,75,90,95,99,99.9,100])
 print(name, np.array2string(q,precision=3))
print('first cells totals',cs[:10]); print('first genes totals',rs[:10])
PY"
