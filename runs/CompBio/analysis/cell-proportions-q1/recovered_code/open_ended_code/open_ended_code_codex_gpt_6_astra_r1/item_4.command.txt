/bin/bash -lc "mkdir -p analysis && cat > analysis/explore.py <<'PY'
import os
os.environ['MPLCONFIGDIR']='analysis/mpl'
os.environ['OMP_NUM_THREADS']='4'
import numpy as np
import scipy.io as sio
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
X=np.asarray(sio.mmread('cell.proportions.q1.mtx.gz')).T
np.save('analysis/counts.npy',X)
n=X.sum(1); g=(X>0).sum(1)
print('shape',X.shape,'range',X.min(),X.max(),'noninteger',np.sum(X"'!=np.round(X)))
print('"'quantiles', np.array([np.quantile(a,[0,.001,.01,.025,.05,.1,.25,.5,.75,.9,.95,.975,.99,1]) for a in [n,g]]))
Y=np.log1p(X/np.maximum(n[:,None],1)*1e4)
pca=PCA(50,random_state=1); P=pca.fit_transform(Y)
np.save('analysis/pca.npy',P)
print('PC explained',pca.explained_variance_ratio_[:20])
fig,axs=plt.subplots(2,2,figsize=(12,10))
axs[0,0].scatter(n,g,s=3);axs[0,0].set_xscale('log');axs[0,0].set_xlabel('counts');axs[0,0].set_ylabel('genes')
axs[0,1].hist(np.log10(n+1),bins=100)
axs[1,0].scatter(P[:,0],P[:,1],s=3,c=np.log10(n+1));axs[1,1].scatter(P[:,1],P[:,2],s=3,c=g)
fig.savefig('analysis/initial.png',dpi=150)
for k in range(2,11):
 km=KMeans(k,n_init=10,random_state=1).fit(P[:,:20]); labels=km.labels_
 print('K',k,'sizes',np.bincount(labels),'sil',silhouette_score(P[:,:20],labels,sample_size=2500,random_state=1),flush=True)
 np.save('analysis/kmeans_'+str(k)+'.npy',labels)
PY
OPENBLAS_NUM_THREADS=4 python analysis/explore.py"
