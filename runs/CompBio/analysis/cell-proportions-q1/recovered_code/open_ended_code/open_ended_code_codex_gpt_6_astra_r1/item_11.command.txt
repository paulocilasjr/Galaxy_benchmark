/bin/bash -lc "cat > analysis/validate.py <<'PY'
import numpy as np
from scipy.optimize import nnls
X=np.load('analysis/counts.npy'); labels=np.load('analysis/kmeans_6.npy')
means=np.array([X[labels==k].mean(0) for k in range(6)])
profiles=means/means.sum(axis=1,keepdims=True)
for k in [3,4,5]:
 w,_=nnls(profiles[:3].T,profiles[k])
 pred=w@profiles[:3]
 r2=1-np.sum((profiles[k]-pred)**2)/np.sum((profiles[k]-profiles[k].mean())**2)
 print('Suspect cluster',k,'mixture weights',w,'mixture fit R2',r2)
keep=np.isin(labels,[0,1,2])
counts=np.sort(np.bincount(labels[keep]))[::-1]
proportions=counts/counts.sum()*100
answer=';'.join(str(int(5*np.round(p/5))) for p in proportions)
print('Retained counts:',counts,'Percentages:',proportions,'Answer:',answer)
with open('analysis/answer.txt','w') as f:f.write(answer+'\\n')
PY
OPENBLAS_NUM_THREADS=4 python analysis/validate.py"
