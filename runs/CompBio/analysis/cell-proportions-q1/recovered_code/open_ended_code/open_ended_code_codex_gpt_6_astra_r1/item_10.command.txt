/bin/bash -lc "cat > analysis/qc.py <<'PY'
import os
os.environ['MPLCONFIGDIR']='analysis/mpl'
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
X=np.load('analysis/counts.npy');P=np.load('analysis/pca.npy');c=np.load('analysis/kmeans_6.npy')
n=X.sum(1);g=(X>0).sum(1)
fig,axs=plt.subplots(1,3,figsize=(16,5))
for k in range(6):
 s=c==k
 print('cluster',k,'N',s.sum(),'counts',np.round(np.quantile(n[s],[0,.1,.5,.9,1]),2),'genes',np.quantile(g[s],[0,.1,.5,.9,1]),'PC means',P[s,:5].mean(0),'index range',np.where(s)[0][[0,-1]])
 axs[0].scatter(P[s,0],P[s,1],s=5,label=f'{k}: {s.sum()}')
 axs[1].scatter(P[s,1],P[s,2],s=5)
 axs[2].scatter(n[s],g[s],s=5)
axs[0].legend();fig.savefig('analysis/clusters.png',dpi=150)
means=np.array([X[c==k].mean(0) for k in range(6)])
print('mean correlations',np.corrcoef(means))
for k in range(6):
 print(k,'zero fraction',np.mean(X[c==k]==0),'mean gene cv',np.mean(X[c==k].std(0)/(means[k]+.01)))
PY
OPENBLAS_NUM_THREADS=4 python -u analysis/qc.py"
