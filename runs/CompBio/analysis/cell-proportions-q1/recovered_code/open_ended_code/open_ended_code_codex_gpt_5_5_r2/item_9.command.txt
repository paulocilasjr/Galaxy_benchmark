/bin/bash -lc "python - <<'PY'
import numpy as np
from scipy.io import mmread
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering, DBSCAN
from sklearn.metrics import silhouette_score, calinski_harabasz_score, davies_bouldin_score
from sklearn.mixture import GaussianMixture

X = np.asarray(mmread('cell.proportions.q1.mtx.gz'), dtype=np.float64).T # cells x genes
lib = X.sum(1)
# normalize to median lib, log1p
Y = np.log1p(X / lib[:,None] * np.median(lib))
# choose hvg by dispersion-ish variance on log norm
vars = Y.var(0)
idx = np.argsort(vars)[-500:]
Z = StandardScaler().fit_transform(Y[:,idx])
pca = PCA(n_components=50, random_state=1)
PC = pca.fit_transform(Z)
print('PCA var first 15', np.round(pca.explained_variance_ratio_[:15],4), 'cum10', pca.explained_variance_ratio_[:10].sum(), 'cum20', pca.explained_variance_ratio_[:20].sum())
for use_n in [10,20,30,50]:
    A=PC[:,:use_n]
    print('\\nPCs', use_n)
    for k in range(2,9):
        km=KMeans(n_clusters=k, n_init=50, random_state=42).fit(A)
        lab=km.labels_
        sizes=np.bincount(lab)
        sil=silhouette_score(A, lab)
        ch=calinski_harabasz_score(A, lab)
        db=davies_bouldin_score(A, lab)
        print('k',k,'sizes',sorted(sizes, reverse=True),'pct', [round(s/len(lab)*100,1) for s in sorted(sizes, reverse=True)], 'sil',round(sil,3),'ch',round(ch,1),'db',round(db,3))
PY"
