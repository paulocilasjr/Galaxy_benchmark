/bin/bash -lc "python - <<'PY'
import numpy as np
from scipy.io import mmread
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
X=np.asarray(mmread('cell.proportions.q1.mtx.gz')).T
genes=(X>0).sum(1); lib=X.sum(1)
mask=genes<930
sf=lib[mask]/np.median(lib[mask])
Y=np.log1p(X[mask]/sf[:,None])
idx=np.argsort(Y.var(0))[-500:]
pc=PCA(n_components=10, random_state=0).fit_transform(StandardScaler().fit_transform(Y[:,idx]))
lab=KMeans(3,n_init=100,random_state=0).fit_predict(pc)
counts=sorted(np.bincount(lab), reverse=True)
print(mask.sum(), counts, [round(c/mask.sum()*100) for c in counts], [round(round(c/mask.sum()*100/5)*5) for c in counts])
PY"
