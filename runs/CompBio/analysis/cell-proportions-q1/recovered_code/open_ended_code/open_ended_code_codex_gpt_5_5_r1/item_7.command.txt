/bin/bash -lc "python - <<'PY'
import numpy as np
from scipy.io import mmread
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
X=np.asarray(mmread('cell.proportions.q1.mtx.gz')).T
lib=X.sum(1); genes=(X>0).sum(1)
sf=lib/np.median(lib)
Y=np.log1p(X/sf[:,None])
idx=np.argsort(Y.var(0))[-500:]
pc=PCA(n_components=10, random_state=0).fit_transform(StandardScaler().fit_transform(Y[:,idx]))
lab=KMeans(6,n_init=100,random_state=0).fit_predict(pc)
for c in sorted(range(6), key=lambda c: -np.sum(lab==c)):
    g=genes[lab==c]
    print(c, 'n', len(g), 'pct', len(g)/len(genes)*100, 'genes range/quant', int(g.min()), int(np.quantile(g,.05)), int(np.median(g)), int(np.quantile(g,.95)), int(g.max()), 'lib median', np.median(lib[lab==c]))
for th in [910,915,920,925,930,935,940]:
    print('genes>=',th, int((genes>=th).sum()))
PY"
