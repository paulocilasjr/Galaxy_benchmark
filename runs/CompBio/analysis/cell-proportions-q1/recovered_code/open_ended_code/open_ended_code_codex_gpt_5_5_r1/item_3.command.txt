/bin/bash -lc "python - <<'PY'
import numpy as np
from scipy.io import mmread
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score, calinski_harabasz_score, davies_bouldin_score

Xg = mmread('cell.proportions.q1.mtx.gz')  # genes x cells ndarray?
X = np.asarray(Xg).T  # cells x genes
print('shape', X.shape, 'dtype', X.dtype, 'minmax', X.min(), X.max(), 'sum', X.sum())
lib = X.sum(axis=1)
genes = (X>0).sum(axis=1)
print('lib quantiles', np.quantile(lib, [0, .001,.005,.01,.02,.05,.1,.25,.5,.75,.9,.95,.98,.99,.995,.999,1]))
print('genes quantiles', np.quantile(genes, [0,.001,.005,.01,.02,.05,.1,.25,.5,.75,.9,.95,.98,.99,.995,.999,1]))
print('zero cells', np.sum(lib==0), 'low lib<500', np.sum(lib<500), '<1000', np.sum(lib<1000), 'low genes<200', np.sum(genes<200), '<500', np.sum(genes<500))
# normalize log1p CPM/size factor median
sf = lib / np.median(lib[lib>0])
Y = np.log1p(X / sf[:,None])
# select variable genes?
var = Y.var(axis=0)
idx = np.argsort(var)[-500:]
Ys = StandardScaler().fit_transform(Y[:, idx])
pc = PCA(n_components=50, random_state=1).fit_transform(Ys)
print('pca var first10', PCA(n_components=10, random_state=1).fit(Ys).explained_variance_ratio_)
for qcname, mask in [('all', np.ones(X.shape[0], dtype=bool)), ('lib_p1p99', (lib>=np.quantile(lib,.01)) & (lib<=np.quantile(lib,.99))), ('lib_p05p995', (lib>=np.quantile(lib,.005)) & (lib<=np.quantile(lib,.995))), ('genes_p1p99', (genes>=np.quantile(genes,.01)) & (genes<=np.quantile(genes,.99)) )]:
    print('\\nQC', qcname, 'n', mask.sum(), 'drop', (~mask).sum())
    Z = pc[mask, :20]
    for k in range(2,9):
        lab=KMeans(n_clusters=k, n_init=50, random_state=42).fit_predict(Z)
        sil=silhouette_score(Z, lab)
        ch=calinski_harabasz_score(Z, lab)
        db=davies_bouldin_score(Z, lab)
        counts=np.bincount(lab)
        print(k, 'sil', round(sil,3), 'ch', int(ch), 'db', round(db,3), 'sizes', sorted(counts, reverse=True), 'pct', [round(c/mask.sum()*100,1) for c in sorted(counts, reverse=True)])
PY"
