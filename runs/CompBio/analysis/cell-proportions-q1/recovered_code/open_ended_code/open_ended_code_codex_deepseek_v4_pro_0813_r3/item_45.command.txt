/bin/bash -lc "python - <<'PY'
import scipy.io,gzip,numpy as np
from sklearn.mixture import GaussianMixture
Z=np.load('pca_standardized.npy')
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f:A=scipy.io.mmread(f)
A=np.asarray(A).astype(float)
X=np.log1p(A/A.sum(0)*1e4)
g=GaussianMixture(n_components=6,covariance_type='full',random_state=42,n_init=50,max_iter=500).fit(Z[:,:2])
lab=g.predict(Z[:,:2])
# map main labels 0(A),5(B),2(C)
main=[0,5,2]
means=np.array([X[:,lab==c].mean(1) for c in range(6)])
# Regress each small cluster mean on main means (intercept)
for c in [1,3,4]:
 y=means[c]
 Xm=means[main].T
 beta,res,rank,s=np.linalg.lstsq(np.column_stack([np.ones(1000),Xm]),y,rcond=None)
 yhat=np.column_stack([np.ones(1000),Xm])@beta
 ss_res=np.sum((y-yhat)**2); ss_tot=np.sum((y-y.mean())**2)
 print('small',c,'n',np.sum(lab==c),'coefs',np.round(beta,3),'R2',1-ss_res/ss_tot)
PY"
