/bin/bash -lc "python - <<'PY'
import scipy.io,gzip,numpy as np
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f:A=scipy.io.mmread(f)
A=np.asarray(A)
lib=A.sum(0); genes=(A>0).sum(0)
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
fig,ax=plt.subplots(1,3,figsize=(12,3))
ax[0].hist(lib,bins=60);ax[0].set_title('libsize')
ax[1].hist(genes,bins=60);ax[1].set_title('ngenes')
# plot top max gene expression per cell perhaps outliers
maxexpr=A.max(0)
ax[2].hist(maxexpr,bins=60);ax[2].set_title('max expr')
plt.savefig('qc.png',dpi=100)
print('max expr percentiles',np.percentile(maxexpr,[50,90,95,99,99.5,100]))
# identify any weird cells
print('max>200',(maxexpr>200).sum(),'max>100',(maxexpr>100).sum())
PY"
