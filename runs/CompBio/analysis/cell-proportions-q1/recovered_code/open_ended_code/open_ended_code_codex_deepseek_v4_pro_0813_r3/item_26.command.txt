/bin/bash -lc "python - <<'PY'
import numpy as np, time
from sklearn.mixture import GaussianMixture
Z=np.load('pca_standardized.npy')
for n in [2,3,4,5,10,20,50]:
    data=Z[:,:n]
    print('\\nnpc',n)
    for k in range(2,9):
        g=GaussianMixture(n_components=k, covariance_type='full', random_state=0, n_init=10, max_iter=200)
        try:
            g.fit(data)
            print(k, 'BIC %.1f AIC %.1f'%(g.bic(data),g.aic(data)), end='; ')
        except Exception as e: print(k,'err',e,end='; ')
    print()
PY"
