/bin/bash -lc "python - <<'PY'
import numpy as np
from sklearn.cluster import KMeans
Z=np.load('pca_standardized.npy')
data=Z[:,:2]
for k in [3,4,5,6,7]:
    km=KMeans(n_clusters=k,n_init=100,random_state=42).fit(data)
    labels=km.labels_
    print('\\nk',k)
    counts=np.bincount(labels)
    order=np.argsort(-counts)
    for c in order:
        ctr=km.cluster_centers_[c]
        print(c,counts[c],f'{counts[c]/len(labels)*100:.1f}%','ctr',np.round(ctr,2))
PY"
