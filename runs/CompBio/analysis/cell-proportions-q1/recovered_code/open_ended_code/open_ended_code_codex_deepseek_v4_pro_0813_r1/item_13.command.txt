/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, pandas as pd, anndata as ad, scanpy as sc
import warnings; warnings.filterwarnings('ignore')
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrows,ncols=map(int,line.split()); break
    arr=np.loadtxt(f)
X=arr.reshape((nrows,ncols), order='F')
adata=ad.AnnData(X.T)
adata.var_names=[f'gene{i}' for i in range(X.shape[0])]
adata.obs_names=[f'cell{i}' for i in range(X.shape[1])]
sc.pp.calculate_qc_metrics(adata, inplace=True)
sc.pp.normalize_total(adata, target_sum=1e4); sc.pp.log1p(adata)
adata.raw=adata.copy()
sc.pp.highly_variable_genes(adata, n_top_genes=1000); sc.pp.scale(adata,max_value=10)
sc.tl.pca(adata, svd_solver='arpack', n_comps=50); sc.pp.neighbors(adata,n_neighbors=15,n_pcs=50)
sc.tl.leiden(adata,resolution=0.2,random_state=0,key_added='leiden')
sc.tl.rank_genes_groups(adata,'leiden',method='t-test',n_genes=10, use_raw=False)
cats=list(adata.obs['leiden'].cat.categories)
for group in cats:
    genes=list(adata.uns['rank_genes_groups']['names'][group][:10])
    scores=list(adata.uns['rank_genes_groups']['scores'][group][:10])
    print('cluster',group,'top genes',[(g,round(s,1)) for g,s in zip(genes,scores)])
# cluster mean raw normalized expressions
M=np.zeros((len(cats),adata.n_vars))
for i,c in enumerate(cats):
    M[i]=np.asarray(adata.raw[adata.obs['leiden']==c].X.mean(axis=0)).ravel()
print('\\nmean corr matrix')
print(pd.DataFrame(np.corrcoef(M),index=[f'c{c}' for c in cats],columns=[f'c{c}' for c in cats]).round(3).to_string())
# correlations of each small cluster mean to main means after subtract maybe
PY"
