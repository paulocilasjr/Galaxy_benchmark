/bin/bash -lc "set -euo pipefail
# show qValue around rank 1000 and number of ties at cutoff
gzip -cd data/ENCFF717TCQ.bed.gz | awk '"'$9!="."{print $9}'"' | sort -gr | awk 'NR==999||NR==1000||NR==1001{print NR,"'$1}'"'
cutoff="'$(gzip -cd data/ENCFF717TCQ.bed.gz | awk '"'"'$9!="."{print $9}'"' | sort -gr | awk 'NR==1000{print; exit}')
echo cutoff="'$cutoff
gzip -cd data/ENCFF717TCQ.bed.gz | awk -v c="$cutoff" '"'"'$9==c{n++} END{print "ties",n+0}'"'"
