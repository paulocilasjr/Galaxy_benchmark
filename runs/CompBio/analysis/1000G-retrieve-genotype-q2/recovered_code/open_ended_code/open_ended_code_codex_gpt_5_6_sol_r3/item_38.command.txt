/bin/bash -lc "bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\n' raw_candidates.vcf.gz > raw_candidate_sites.tsv
awk 'NR==FNR {p["'$1 FS $2]=1;next} length($3)==1 && $3~/''^[ACGT]$/ && length($4)==1 && $4~/''^[ACGT]$/ && (($1 FS $2) in p){print}'"' eligible_positions.tsv raw_candidate_sites.tsv > final_sites.tsv
wc -l raw_candidate_sites.tsv final_sites.tsv
awk 'NR==FNR {p["'$1 FS $2]=1;next} !(($1 FS $2) in p){print}'"' eligible_positions.tsv raw_candidate_sites.tsv
awk '{n["'$1 FS $2]++} END{for(k in n)if(n[k]>1)print k,n[k]}'"' raw_candidate_sites.tsv | sort -k2,2n"
