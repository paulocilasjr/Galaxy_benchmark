/bin/bash -lc "python3 - <<'PY'
import gzip
fn='data/ENCFF717TCQ.bed.gz'
peaks=[]
with gzip.open(fn,'rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        chrom=p[0]; start=int(p[1]); end=int(p[2]); name=p[3]; q=float(p[8])
        peaks.append((q,chrom,start,end,name))
peaks.sort(key=lambda x:x[0], reverse=True)
print('total',len(peaks))
print('top1-10:')
for i,p in enumerate(peaks[:10],1): print(i,p)
# Save top 1000 maybe
import collections
c=collections.Counter(x[1] for x in peaks[:1000]); print('top1000 chrom counts',c)
chr1_top=[x for x in peaks[:1000] if x[1]=='chr1']
print('chr1 top',len(chr1_top), 'coordinate range', min(x[2] for x in chr1_top), max(x[3] for x in chr1_top))
# sorted chr1 intervals
ints=sorted((x[2],x[3]) for x in chr1_top)
print('first/last', ints[:5], ints[-5:])
PY"
