/bin/bash -lc "python3 - <<'PY'
import requests, json
url='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main'
items=[]; page=0
while True:
    r=requests.get(url, params={'recursive':'true','expand':'false','limit':'1000','offset':str(page*1000)}, timeout=30)
    data=r.json()
    if not data: break
    items += data
    print('page',page,'got',len(data),'total',len(items))
    if len(data)<1000: break
    page+=1
json.dump(items, open('[REDACTED LOCAL PATH]','w'))
print('total items',len(items))
files=[x for x in items if x.get('type')=='file']
print('files',len(files))
for x in files[:200]: print(x.get('path'))
PY"
