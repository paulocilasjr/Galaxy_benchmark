/bin/bash -lc "python3 - <<'PY'
import gzip
peaks=[]
with gzip.open('data/ENCFF717TCQ.bed.gz','rt') as f:
    for line in f:
        p=line.rstrip().split('\\t')
        peaks.append((float(p[8]),p[0],int(p[1]),int(p[2])))
peaks.sort(key=lambda x:x[0], reverse=True)
ints=sorted((x[2],x[3]) for x in peaks[:1000] if x[1]=='chr1')
merged=[]
for a,b in ints:
    if merged and a <= merged[-1][1]:
        merged[-1]=(merged[-1][0],max(merged[-1][1],b))
    else:
        merged.append((a,b))
with open('data/top1000_chr1_regions.tsv','w') as f:
    for a,b in merged:
        f.write(f'chr1\\t{a+1}\\t{b}\\n')
print(len(merged), sum(b-a for a,b in merged))
PY
head -3 data/top1000_chr1_regions.tsv; tail -3 data/top1000_chr1_regions.tsv; wc -l data/top1000_chr1_regions.tsv"
