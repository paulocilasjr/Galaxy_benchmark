with open('data/output.tsv') as f:
    header=f.readline().rstrip('\\n').split('\\t')
print('header cols', len(header), 'sample cols', len(header)-4)

vals=set(); bad=[]

    if p[0]"'!='"'chr1': bad.append(('chrom',n,p[0]))

    if len(p[3])"'!=1 or p[3] not in '"'ACGT': bad.append(('alt',n,p[3]))
    pos=int(p[1])
    if pos < posprev: bad.append(('sort',n,pos,posprev))
    posprev=pos
    for x in p[4:]:
        vals.add(x)
        if x not in ('0/0','0/1','1/1','./.'):
