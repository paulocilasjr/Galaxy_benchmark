/bin/bash -lc "python3 - <<'PY'
import gzip, heapq, collections
peaks=[]
with gzip.open('data/ENCFF717TCQ.bed.gz','rt') as f:
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        chrom=parts[0]; start=int(parts[1]); end=int(parts[2]); q=float(parts[8])
        peaks.append((q,chrom,start,end,line.rstrip('\\n')))
# sort desc q; tie chrom start asc
peaks.sort(key=lambda x:(-x[0], x[1], x[2], x[3]))
top=peaks[:1000]
c=collections.Counter(p[1] for p in top)
print('top chr counts', c)
print('top q range', top[0][0], top[-1][0])
# write intervals
with open('data/top1000_peaks.bed','w') as f:
    for q,ch,st,en,line in top:
        f.write(line+'\\n')
print('wrote')
PY
head data/top1000_peaks.bed | head -5"
