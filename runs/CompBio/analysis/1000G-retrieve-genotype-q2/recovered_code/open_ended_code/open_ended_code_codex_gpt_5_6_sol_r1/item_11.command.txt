/bin/bash -lc "bcftools --version
vcfbase='https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.annotated.vcf.gz'
curl -sIL \""'$vcfbase" | tr -d '"'\\r' | tail -15
curl -L --fail --silent --show-error -o chr1.vcf.gz.tbi \""'$vcfbase.tbi"
ls -lh chr1.vcf.gz.tbi
bcftools view -h "$vcfbase" > header.txt 2>bcf.err
printf '"'status %s\\n' \""'$?"
wc -l header.txt
cat bcf.err
head -c 200 header.txt'
