/bin/bash -lc "python - <<'PY'
import gzip,csv
rows=[]
with gzip.open('gwas.ancestry.q1.tsv.gz','rt') as f:
    r=csv.DictReader(f,delimiter='\\t')
    for row in r:
        if row['P']=='NA': continue
        p=float(row['P']); pos=int(row['POS'])
        if p<1e-6:
            rows.append((p,row['RSID'],pos,row['SNPID']))
rows.sort()
clumps=[]
for p,rs,pos,snpid in rows:
    if all(abs(pos-c[2])>1_000_000 for c in clumps):
        clumps.append((p,rs,pos,snpid))
print(len(clumps))
for c in clumps[:30]: print(c)
PY"
