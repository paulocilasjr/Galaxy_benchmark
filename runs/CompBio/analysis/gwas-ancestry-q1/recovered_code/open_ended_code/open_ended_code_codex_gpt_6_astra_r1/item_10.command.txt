/bin/bash -lc "cat > score_ld.py <<'PY'
import numpy as np,pandas as pd
from scipy.stats import norm,spearmanr
x=np.load('ref_peak.npz'); g=pd.read_csv('gwas.ancestry.q1.tsv.gz',sep='\\t').set_index('POS').loc[x['pos']]
panel=pd.read_csv('integrated_call_samples_v3.20130502.ALL.panel',sep='\\t').set_index('sample').loc[x['samples']]
z=norm.isf(g.P.to_numpy()/2); lead=np.nanargmax(z)
print('Lead',x['pos'][lead],'Z',z[lead])
for pop in ['AFR','EUR','EAS','SAS','AMR']:
 gt=x['geno'][:,(panel.super_pop==pop).to_numpy()].astype(float)
 gt-=gt.mean(axis=1,keepdims=True); sd=np.linalg.norm(gt,axis=1); gt/=np.maximum(sd[:,None],1e-10)
 r=gt@gt[lead]; pred=np.abs(r)*z[lead]
 use=np.isfinite(z)&(abs(x['pos']-x['pos'][lead])<300000)&(sd>0)
 print(pop,'n',sum(use),'r2-fit',np.corrcoef(r[use]**2,z[use]**2)[0,1], 'Zerr',np.mean((pred[use]-z[use])**2),'top-r',spearmanr(abs(r[use]),z[use]).statistic)
print(g.iloc[np.argsort(-z)[:20]] [['RSID','P']])
PY
ls -lh ref_peak.npz"
