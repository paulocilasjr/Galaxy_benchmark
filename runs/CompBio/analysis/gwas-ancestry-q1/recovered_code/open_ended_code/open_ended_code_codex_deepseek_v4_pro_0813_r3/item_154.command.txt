/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, math
# file p and effect allele
pmap={}
alleles={}
for line in open('gwas.ancestry.q1.tsv'):
    f=line.rstrip().split('\\t')
    pmap[f[1]]=float(f[6]) if f[6]"'!='"'NA' else None
    alleles[f[1]]=(f[4],f[5])
rslist=['rs9435734','rs4920608','rs2871776','rs17361789','rs3738814','rs761423','rs926702','rs2076600','rs9435660','rs7531163','rs9435671','rs3738815','rs2871775','rs2311528','rs2209096']
ids={'rs9435734':'1_16984695_A_G','rs4920608':'1_16992792_C_T','rs2871776':'1_17001806_T_C','rs17361789':'1_172153461_T_G','rs3738814':'1_17005181_A_G','rs761423':'1_16975177_T_C','rs926702':'1_17010026_G_T','rs2076600':'1_17010896_G_A','rs9435660':'1_16987578_A_G','rs7531163':'1_16986959_G_A','rs9435671':'1_17013444_T_C','rs3738815':'1_16988207_C_T','rs2871775':'1_17019410_A_G','rs2311528':'1_16983045_G_A','rs2209096':'1_172169864_T_C'}
query='{ credibleSets(variantIds: ['+','.join(json.dumps(ids[r]) for r in rslist)+'], page: {index:0, size:1000}) { count rows { studyId pValueMantissa pValueExponent position variant { id rsIds } study { id traitFromSource initialSampleSize hasSumstats } } } }'
req=urllib.request.Request('https://api.platform.opentargets.org/api/v4/graphql', data=json.dumps({'query':query}).encode(), headers={'Content-Type':'application/json'})
d=json.load(urllib.request.urlopen(req, timeout=60))
rows=d['data']['credibleSets']['rows']
print('rows',len(rows))
# Build matches per study among rows whose variant.rsIds includes a top rs
for target_rs in rslist:
    fp=pmap[target_rs]
    print('\\nTARGET',target_rs,'file p',fp)
    candidates=[]
    for r in rows:
        if target_rs in r['variant']['rsIds'] and r['pValueMantissa'] is not None:
            p=r['pValueMantissa']*10**r['pValueExponent']
            if fp and abs(math.log10(p/fp))<0.001: # exact or close
                candidates.append((r['studyId'], p, r['study']['traitFromSource'], r['study']['initialSampleSize'], r['study']['hasSumstats']))
    for c in candidates[:20]: print('  MATCH',c)
    # also print closest 3
    closest=[]
    for r in rows:
        if target_rs in r['variant']['rsIds'] and r['pValueMantissa'] is not None:
            p=r['pValueMantissa']*10**r['pValueExponent']
            if fp and p>0:
                closest.append((abs(math.log10(p/fp)), r['studyId'], p, r['study']['traitFromSource'], r['study']['initialSampleSize'], r['study']['hasSumstats']))
    closest.sort(key=lambda x:x[0])
    print(' closest:')
    for c in closest[:5]: print('  ',c)
PY"
