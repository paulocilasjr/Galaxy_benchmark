/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, time
for rs in ['rs9435734','rs4920608','rs2871776','rs17361789','rs3738814','rs761423','rs926702','rs2076600']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?projection=associationBySnp'
    try:
        d=json.load(urllib.request.urlopen(url, timeout=30))
        print('\\n',rs)
        for a in d['_embedded']['associations']:
            p=a.get('pvalue'); man=a.get('pvalueMantissa'); exp=a.get('pvalueExponent')
            traits=[e.get('trait') for e in a.get('efoTraits',[])]
            print(p, man, exp, traits[:2], a.get('riskFrequency'))
    except Exception as e: print(rs,e)
    time.sleep(.2)
PY"
