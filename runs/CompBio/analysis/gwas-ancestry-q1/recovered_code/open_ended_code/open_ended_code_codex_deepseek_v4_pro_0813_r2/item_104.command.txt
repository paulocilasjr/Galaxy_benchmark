/bin/bash -lc "python3 - <<'PY'
import requests, json
snp_list=['rs9435734','rs4920608','rs2871776','rs17361789','rs10429950','rs3766763','rs11164665','rs2744666','rs1892425','rs11205303','rs1014719']
for rs in snp_list:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations'
    r=requests.get(url)
    print('\\n###',rs,r.status_code)
    if not r.ok: continue
    data=r.json().get('_embedded',{}).get('associations',[])
    for a in data:
        # extract fields
        fields={k:a.get(k) for k in ['pvalueMantissa','pvalueExponent','pValue','riskFrequency','orPerCopyNum','betaNum','betaUnit','range','studyAccession','lastUpdateDate']}
        print(json.dumps(fields))
        traits=[]
        if 'traitNames' in a:
            traits=[t.get('trait') for t in a.get('traitNames',[])]
        # embedded traits may be nested
        emb=a.get('_embedded',{})
        if 'traitNames' in emb:
            traits=[t.get('trait') for t in emb['traitNames']]
        print('traits',traits[:10])
        # study
        print('study', a.get('studyAccession'), emb.keys() if emb else None)
        if 'study' in emb: print('study info', {k:emb['study'].get(k) for k in ['accessionId','pubmedId','publicationDate','initialSampleSize','replicateSampleSize','ancestryLinks']})
        # parent trait? print raw keys maybe
        print('keys', list(a.keys()))
PY"
