/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# Fetch associations for study GCST011873, filter top SNP names
for snp in ['rs9435734','rs761423','rs10429950','rs17361789','rs4920608','rs2871776']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{snp}/associations?projection=associationBySnp'
    try:
        data=json.load(urllib.request.urlopen(url, timeout=30))
        assocs=data.get('_embedded',{}).get('associations',[])
        print('\\nSNP',snp,'total',len(assocs))
        for a in assocs:
            # get study link accession
            sid=a.get('_links',{}).get('study',{}).get('href','')
            print(' p',a.get('pvalue'), 'trait',[t.get('trait') for t in a.get('efoTraits',[])], 'study',sid)
    except Exception as e: print(snp,e)
PY"
