/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# search studies by trait
for trait in ['BMI-adjusted waist circumference','waist-hip ratio','body mass index']:
    q=urllib.parse.quote(trait)
    url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByEfoTrait?efoTrait={q}&size=100'
    try:
        data=json.load(urllib.request.urlopen(url, timeout=30))
        emb=data.get('_embedded',{}).get('studies',[])
        print('\\nTRAIT',trait,'count',data.get('page',{}).get('totalElements'))
        for s in emb:
            print(s.get('accessionId'), s.get('initialSampleSize'), s.get('publicationDate'))
            # maybe link to summary stats
    except Exception as e: print(e)
PY"
