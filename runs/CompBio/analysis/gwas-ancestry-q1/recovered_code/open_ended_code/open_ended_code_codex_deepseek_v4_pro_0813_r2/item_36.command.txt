/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
snps=['rs17361789','rs3738814','rs761423','rs2209096','rs17277372','rs10429950']
for snp in snps:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{snp}/associations?projection=associationBySnp'
    try:
        data=json.load(urllib.request.urlopen(url, timeout=30))
        assocs=data.get('_embedded',{}).get('associations',[])
        print('\\n',snp, 'n', len(assocs))
        for a in assocs[:5]:
            print(a.get('pvalue'), a.get('efoTraits'), a.get('riskFrequency'), a.get('study',{}))
    except Exception as e:
        print(snp,e)
PY"
