/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for url in ['https://genetics-api.opentargets.io/graphql','https://api.platform.opentargets.org/api/v4/graphql']:
    query='''query { variantInfo(variantId: \"1_17311190_A_G\") { id rsId chromosome position } }'''
    data=json.dumps({'query':query}).encode()
    req=urllib.request.Request(url,data=data,headers={'Content-Type':'application/json','User-Agent':'Mozilla/5.0'})
    print('URL',url)
    try:
        resp=urllib.request.urlopen(req,timeout=30).read().decode()
        print(resp[:3000])
    except Exception as e: print(type(e).__name__,e)
PY"
