/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://gnomad.broadinstitute.org/api'
query='''
query V("'$rsid: String!, $dataset: DatasetId!) {
 variant(rsid: $rsid, dataset: $dataset) {
  variant_id rsids chrom pos ref alt
  exome { ac an af populations { id ac an } }
  genome { ac an af populations { id ac an } }
  joint { ac an populations { id ac an } }
 }
}
'"'''
for ds in ['gnomad_r4','gnomad_r2_1','gnomad_v2_1','gnomad_r3']:
 r=requests.post(url,json={'query':query,'variables':{'rsid':'rs2235934','dataset':ds}},timeout=30)
 print(ds,r.status_code, r.text[:1000])
PY"
