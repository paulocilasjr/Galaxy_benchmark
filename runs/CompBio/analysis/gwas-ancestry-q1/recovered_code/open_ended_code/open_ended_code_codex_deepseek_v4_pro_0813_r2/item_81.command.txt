/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
acc='GCST011875'
# Get associations by study, page size 100, iterate
url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/{acc}/associations?projection=associationByStudy&size=100'
page=0
all_assoc=[]
while True:
    u=url+f'&page={page}'
    try:
        data=json.load(urllib.request.urlopen(u,timeout=60))
    except Exception as e:
        print('err',e); break
    emb=data.get('_embedded',{}).get('associations',[])
    if not emb: break
    all_assoc += emb
    total=data.get('page',{}).get('totalElements',0)
    print('page',page,'got',len(emb),'total',total)
    if len(all_assoc)>=total: break
    page+=1
print('total fetched',len(all_assoc))
# Extract snps and p
rows=[]
for a in all_assoc:
    snps=a.get('snps',[])
    rsids=[s.get('rsId') for s in snps]
    rows.append((a.get('pvalue'), rsids, [t.get('trait') for t in a.get('efoTraits',[])], a.get('riskFrequency')))
for r in rows:
    print(r)
PY"
