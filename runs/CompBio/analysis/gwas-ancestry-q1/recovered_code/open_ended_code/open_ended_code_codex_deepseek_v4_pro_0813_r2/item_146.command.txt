/bin/bash -lc "python3 - <<'PY'
import requests,json
# Query GWAS Catalog study associations for specific studies and SNP rs9435734
studies=['GCST90245847','GCST90245844','GCST90245845','GCST90095033','GCST90245843','GCST90245848','GCST90468178','GCST90435412','GCST90565843','GCST90662911','GCST90662910']
for study in studies:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/{study}/associations'
    r=requests.get(url)
    if not r.ok:
        print(study,'err',r.status_code); continue
    data=r.json().get('_embedded',{}).get('associations',[])
    print('\\n',study,'n',len(data))
    for a in data:
        # check rs via links or snps
        snps=a.get('_embedded',{}).get('snps',[])
        if not snps:
            # maybe loci
            snps=[]
            for loc in a.get('loci',[]):
                for sa in loc.get('strongestRiskAlleles',[]):
                    name=sa.get('riskAlleleName','')
                    if name.startswith('rs'):
                        snps.append(name.split('-')[0])
        if any('rs9435734' in s for s in snps):
            print('  match rs9435734', a.get('pvalueMantissa'),a.get('pvalueExponent'),a.get('riskFrequency'),a.get('betaNum'),a.get('betaUnit'),a.get('_links',{}).get('self',{}))
            print('  traits', [t.get('trait') for t in a.get('_embedded',{}).get('efoTraits',[])])
PY"
