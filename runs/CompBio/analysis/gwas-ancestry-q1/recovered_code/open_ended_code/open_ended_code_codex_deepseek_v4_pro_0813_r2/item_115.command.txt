import pandas as pd, pysam, time
p='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-0bb2941c0318233f7b766bc0/workspace/gwas.ancestry.q1.tsv'
df=pd.read_csv(p,sep='\\t')
na=df[df['P'].isna()].copy()
url='https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz'
vf=pysam.VariantFile(url)
rows=[]
for _,r in na.iterrows():
    pos=r['POS']; eff=r['EFFECT_ALLELE']; other=r['OTHER_ALLELE']
    recs=list(vf.fetch('1',pos-1,pos+1))
    found=[]
    for rec in recs:
        if rec.pos==pos:
            found.append(rec)
    if not found:
        rows.append((pos,eff,other,'ABSENT','','','','','',''))
    else:
        for rec in found:
            alts=','.join(rec.alts or [])
            info=dict(rec.info)
            rows.append((pos,eff,other,'PRESENT',rec.ref,alts,
                round(info.get('EAS_AF',(float('nan'),))[0],4) if info.get('EAS_AF') else '',
                round(info.get('AMR_AF',(float('nan'),))[0],4) if info.get('AMR_AF') else '',
                round(info.get('AFR_AF',(float('nan'),))[0],4) if info.get('AFR_AF') else '',
                round(info.get('EUR_AF',(float('nan'),))[0],4) if info.get('EUR_AF') else '',
                round(info.get('SAS_AF',(float('nan'),))[0],4) if info.get('SAS_AF') else ''))
for row in rows:
    print('\\t'.join(map(str,row)))
PY"
