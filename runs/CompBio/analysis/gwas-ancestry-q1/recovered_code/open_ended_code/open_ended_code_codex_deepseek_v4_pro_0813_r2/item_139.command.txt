/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://gnomad.broadinstitute.org/api'
# find Variant type and its exome/genome/pop fields
q='''query { __type(name:\"Variant\") { fields { name type { kind name ofType { kind name ofType { kind name } } } } } }'''
r=requests.post(url,json={'query':q},timeout=30)
d=r.json()['data']['__type']
print('Variant fields count', len(d['fields']))
for f in d['fields']:
 print(f['name'], f['type'])
PY"
