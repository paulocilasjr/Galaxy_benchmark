/bin/bash -lc "python3 - <<'PY'
import requests
url='https://gnomad.broadinstitute.org/api'
for typ in ['VariantSequencingTypeData','VariantPopulation','VariantJointSequencingTypeData','VariantFilteringAlleleFrequency']:
 q='query { __type(name:\"%s\") { fields { name type { kind name ofType { kind name ofType { kind name } } } } } }'%typ
 r=requests.post(url,json={'query':q},timeout=30)
 d=r.json()['data']['__type']
 print('\\n###',typ,'fields')
 for f in d['fields']:
  print(f['name'], f['type'])
PY"
