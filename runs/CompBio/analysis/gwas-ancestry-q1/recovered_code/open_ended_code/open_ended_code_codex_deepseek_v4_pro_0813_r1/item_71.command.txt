/bin/bash -lc "python3 - <<'PY'
import requests,json,gzip
ids=[]
with gzip.open('gwas.ancestry.q1.tsv.gz','rt') as f:
 f.readline()
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  if p[6]=='NA': ids.append(p[1])
r=requests.post('https://myvariant.info/v1/variant', json={'ids':ids,'fields':'gnomad_genome.af'}, timeout=60)
d=r.json()
by={}
for x in d:
    by.setdefault(x.get('query'),[]).append(x)
for rs in ids:
    xs=by.get(rs,[])
    if not xs:
        print(rs, 'notfound')
        continue
    for x in xs:
        af=(x.get('gnomad_genome') or {}).get('af') or {}
        print(f\"{rs}\\t_id={x.get('_id')}\\tAFR={af.get('af_afr')}\\tAMR={af.get('af_amr')}\\tASJ={af.get('af_asj')}\\tEAS={af.get('af_eas')}\\tNFE={af.get('af_nfe')}\\tOTH={af.get('af_oth')}\")
PY"
