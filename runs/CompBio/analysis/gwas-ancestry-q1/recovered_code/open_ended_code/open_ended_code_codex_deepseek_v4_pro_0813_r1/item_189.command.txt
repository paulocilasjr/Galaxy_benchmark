/bin/bash -lc "python3 - <<'PY'
from playwright.sync_api import sync_playwright
import json,time
with sync_playwright() as p:
    browser=p.chromium.launch(headless=True)
    page=browser.new_page()
    page.goto('https://dataverse.harvard.edu/api/datasets/:persistentId/?persistentId=doi:10.7910/DVN/FAWEQK', wait_until='domcontentloaded', timeout=60000)
    time.sleep(12)
    data=json.loads(page.evaluate('() => document.body.innerText'))
    files=data['data']['latestVersion']['files']
    for f in files:
        print(f.get('dataFile',{}).get('id'), f.get('label'), f.get('dataFile',{}).get('filesize'))
    browser.close()
PY"
