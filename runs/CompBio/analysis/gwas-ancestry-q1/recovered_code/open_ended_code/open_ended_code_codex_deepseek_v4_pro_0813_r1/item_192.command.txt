/bin/bash -lc "python3 - <<'PY'
from playwright.sync_api import sync_playwright
import time, os
name='AMR_height.txt'; fid=7172695
out=f'[REDACTED LOCAL PATH]'
with sync_playwright() as p:
    browser=p.chromium.launch(headless=True)
    ctx=browser.new_context(accept_downloads=True)
    page=ctx.new_page()
    url=f'https://dataverse.harvard.edu/api/access/datafile/{fid}'
    try:
        page.goto(url, wait_until='domcontentloaded', timeout=60000)
    except Exception as e:
        print('first',repr(e)[:100])
    # wait for challenge auto-reload
    for i in range(8):
        time.sleep(5)
        # if challenge token solved, content may become download? 
    try:
        with page.expect_download(timeout=120000) as dl_info:
            try:
                page.goto(url, wait_until='domcontentloaded', timeout=120000)
            except Exception as e:
                print('goto2',repr(e)[:100])
        dl=dl_info.value
        print('filename',dl.suggested_filename)
        dl.save_as(out)
        print('saved',os.path.getsize(out))
    except Exception as e:
        print('dl err',repr(e))
        print(page.content()[:500])
    browser.close()
PY"
