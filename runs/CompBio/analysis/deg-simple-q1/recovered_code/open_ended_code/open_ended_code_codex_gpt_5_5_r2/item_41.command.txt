/bin/bash -lc "python3 - <<'PY'
import math
from pathlib import Path

def load(p):
    d={}
    for line in Path(p).read_text().splitlines()[1:]:
        if not line: continue
        g,c=line.split('\\t')
        d[g]=int(c)
    return d
A=load('results/sampleA.mapq20.gene_counts.tsv')
B=load('results/sampleB.mapq20.gene_counts.tsv')
rows=[]
for g in set(A)|set(B):
    a=A.get(g,0); b=B.get(g,0)
    rows.append((abs(b-a), abs(math.log2((b+1)/(a+1))), g,a,b,b-a,math.log2((b+1)/(a+1))))
for _,_,g,a,b,d,l2 in sorted(rows, reverse=True)[:30]:
    print(f'{g}\\tA={a}\\tB={b}\\tdiff={d}\\tlog2FC={l2:.3f}')
PY"
