/bin/bash -lc "python3 - <<'PY'
import pysam, collections
for sample in ['A','B']:
    mq=collections.Counter(); nm=collections.Counter(); n=0; multi=0; flags=collections.Counter()
    with pysam.AlignmentFile(f'sample{sample}.bam','rb') as bam:
        for read in bam.fetch(until_eof=True):
            name=read.reference_name or ''
            if '|KLF4|' not in name: continue
            n+=1
            mq[read.mapping_quality]+=1
            try: nm[read.get_tag('NM')]+=1
            except: nm[-1]+=1
            flags[read.flag]+=1
    print(sample, 'n', n, 'mq', mq.most_common(20), 'nm', nm.most_common(10), 'flags', flags.most_common(10))
PY"
