/bin/bash -lc "Rscript - <<'RS'
suppressMessages(library(DESeq2))
x <- read.delim('gene_counts_abundance.tsv', stringsAsFactors=FALSE, check.names=FALSE)
x <- x[is.finite(x"'$count_A) & is.finite(x$count_B),]
cts <- round(as.matrix(x[,c('"'count_A','count_B')]))
rownames(cts) <- x"'$gene
colnames(cts) <- c('"'A','B')
cold <- data.frame(condition=factor(c('A','B')), row.names=colnames(cts))
dds <- DESeqDataSetFromMatrix(countData=cts, colData=cold, design=~condition)
dds <- DESeq(dds, quiet=TRUE)
res <- results(dds, contrast=c('condition','B','A'))
res <- as.data.frame(res)
res"'$gene <- rownames(res)
res <- res[order(res$pvalue),]
print(head(res[,c('"'gene','baseMean','log2FoldChange','lfcSE','pvalue','padj')], 30))
RS"
