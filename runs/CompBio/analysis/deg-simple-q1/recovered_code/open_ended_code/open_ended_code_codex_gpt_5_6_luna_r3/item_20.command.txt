/bin/bash -lc "awk 'BEGIN{OFS=\"\\t\"} NR==FNR{b["'$1]=$2; next} {a[$1]=$2; seen[$1]=1} END{for(g in b) if(!(g in seen)) print g,0,b[g]; for(g in a) print g,a[g],(g in b?b[g]:0)}'"' counts.B.tsv counts.A.tsv | awk 'BEGIN{OFS=\"\\t\"}{a="'$2;b=$3; l=log((a+1)/(b+1))/log(2); d=a-b; ad=(d<0?-d:d); print $1,a,b,l,ad}'"' > joined.tsv
printf '%s\\n' 'largest log2 A/B' && sort -k4,4nr joined.tsv | head -n 30
printf '%s\\n' 'smallest log2 A/B' && sort -k4,4n joined.tsv | head -n 30
printf '%s\\n' 'largest absolute count differences' && sort -k5,5nr joined.tsv | head -n 30"
