/bin/bash -lc "awk 'BEGIN{OFS=\"\\t\"} {split("'$6,a,/[|]/); c[a[6]]++} END{for(g in c) print g,c[g]}'"' sampleA.primary.paf | sort -k1,1 > sampleA.gene_counts.tsv
awk 'BEGIN{OFS=\"\\t\"} {split("'$6,a,/[|]/); c[a[6]]++} END{for(g in c) print g,c[g]}'"' sampleB.primary.paf | sort -k1,1 > sampleB.gene_counts.tsv
join -a1 -a2 -e 0 -o 0,1.2,2.2 sampleA.gene_counts.tsv sampleB.gene_counts.tsv > gene_counts.joined.tsv
awk 'BEGIN{OFS=\"\\t\"; print \"gene\",\"A\",\"B\",\"log2ratio\",\"difference\"} {lr=log(("'$2+10)/($3+10))/log(2); d=$2-$3; if(d<0)d=-d; print $1,$2,$3,lr,d}'"' gene_counts.joined.tsv | sort -t "'$'"'\\t' -k5,5nr | head -n 30"
