/bin/bash -lc "awk 'BEGIN{FS=\"\\t\"} {split("'$6,a,"|"); g=a[6]; if(g!="") c[g]++} END{for(g in c) print g,c[g]}'"' results/sampleA.paf | sort > results/sampleA.gene_counts.tsv
awk 'BEGIN{FS=\"\\t\"} {split("'$6,a,"|"); g=a[6]; if(g!="") c[g]++} END{for(g in c) print g,c[g]}'"' results/sampleB.paf | sort > results/sampleB.gene_counts.tsv
join -a1 -a2 -e 0 -o 0,1.2,2.2 results/sampleA.gene_counts.tsv results/sampleB.gene_counts.tsv | awk 'BEGIN{OFS=\"\\t\"; totalA=969407; totalB=968593; print \"gene\",\"A\",\"B\",\"log2ratio\",\"abs_norm_diff\"} NR>1{a="'$2+0; b=$3+0; na=a/totalA*1e6; nb=b/totalB*1e6; lr=log((na+1)/(nb+1))/log(2); d=na-nb; if(d<0)d=-d; print $1,a,b,lr,d}'"' > results/gene_compare.tsv
sort -k5,5nr results/gene_compare.tsv | head -n 30"
