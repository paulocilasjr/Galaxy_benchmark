/bin/bash -lc "python3 - <<'PY'
import gzip, collections
for fn in ['deg.simple.q1.sampleA.fq.gz','deg.simple.q1.sampleB.fq.gz']:
    c=collections.Counter()
    with gzip.open(fn,'rt') as f:
        for i,line in enumerate(f):
            if i%4==1:
                c[line.strip()]+=1
    print(fn, 'unique', len(c), 'total', sum(c.values()), 'top', c.most_common(5))
PY"
