/bin/bash -c "python3 - <<'PY'
from collections import Counter
BIN=10000
bins=Counter()
with open('chr22_alns.tsv') as f:
    next(f)
    for line in f:
        p=line.rstrip().split('\\t')
        pos=int(p[2])
        bins[pos//BIN]+=1
maxbin=(max(bins.keys())+1)*BIN if bins else 0
print('maxbin',maxbin,'len bins',len(bins))
for b in [10510000, 10600000, 20000000, 38270000, 38280000]:
    print(b, bins.get(b//BIN,0), b//BIN)
# print first 20 lines output
cnt=0
for b in range(0,maxbin,BIN):
    c=bins.get(b,0)
    if c:
        print(b,c)
        cnt+=1
        if cnt>=10:break
PY"
