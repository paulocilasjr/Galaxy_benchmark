/bin/bash -c "python3 - <<'PY'
from collections import defaultdict, Counter
pairs=defaultdict(list)
with open('chr22_alns.tsv') as f:
    next(f)
    for line in f:
        p=line.rstrip().split('\\t')
        q=p[0]; flag=int(p[1]); pos=int(p[2]); mapq=int(p[3]); cigar=p[4]; mate=p[5]; mpos=int(p[6]); tlen=int(p[7])
        pairs[q].append((flag,pos,mapq,cigar,mate,mpos,tlen))
print('pairs',len(pairs),'aligns',sum(len(v) for v in pairs.values()))
# for qnames with exactly 2 alignments, compute abs tlen, positions
normal=[]; big=[]; crossing=[]
for q, als in pairs.items():
    if len(als)"'!=2: continue
    a,b=als
    # primary pair should have tlen opposite maybe same abs
    if abs(a[6])<2000 and abs(b[6])<2000:
        normal.append((q,min(a[1],b[1]),max(a[1],b[1]),abs(a[6])))
    elif abs(a[6])>=100000 or abs(b[6])>=100000:
        big.append((q,a[1],b[1],a[6],b[6]))
    # mates on opposite sides of region 19.5-21.5
    lo=min(a[1],b[1]); hi=max(a[1],b[1])
    if lo < 20500000 and hi > 20500000 and hi-lo>200000:
        crossing.append((q,lo,hi,hi-lo,a[6],b[6]))
print('"'normal small tlen pairs',len(normal))
print('big tlen pairs (>=100kb)',len(big))
for x in big[:30]: print('BIG',x)
print('crossing wide pairs count',len(crossing))
for x in sorted(crossing,key=lambda z:z[3],reverse=True)[:50]: print('CROSS',x)
# tlen distribution for chr22 all pairs exact two
tlens=[]
for q,als in pairs.items():
    if len(als)==2:
        a,b=als
        tlens.append(abs(a[6]))
print('tlen stats: total',len(tlens),'max',max(tlens),'99.9',sorted(tlens)[int(len(tlens)*0.999)] if tlens else None)
# find max tlen pairs
bigs=sorted([(q,als[0][1],als[1][1],abs(als[0][6]),abs(als[1][6])) for q,als in pairs.items() if len(als)==2], key=lambda x:max(x[3],x[4]), reverse=True)
for x in bigs[:30]: print('MAXTLEN',x)
PY"
