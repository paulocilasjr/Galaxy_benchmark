/bin/bash -c "python3 - <<'PY'
from collections import Counter
bins=Counter()
with open('chr22_depth_10kb.tsv') as f:
    for line in f:
        if line.startswith('BIN'): continue
        b,c=line.split()
        bins[int(b)]=int(c)
maxb=max(bins)
arr=[bins.get(b,0) for b in range(0,maxb+1,10000)]
print('max bin',maxb,'nonzero',sum(c>0 for c in arr),'mean',sum(arr)/len(arr),'max',max(arr))
# zero runs
runs=[]
in_zero=False;start=0
for i,c in enumerate(arr):
    if c==0 and not in_zero: start=i; in_zero=True
    elif c>0 and in_zero: runs.append((start,i-1,i-start)); in_zero=False
if in_zero: runs.append((start,len(arr)-1,len(arr)-start))
for s,e,n in runs:
    if n>=5:
        print(f'zero {s*10000:>10}-{(e+1)*10000-1:>10} len {n*10000} bins {n}')
        lo=max(0,s-5); hi=min(len(arr)-1,e+5)
        print(' around:')
        for i in range(lo,hi+1): print(f'  {i*10000:>10}: {arr[i]}')
PY"
