/bin/bash -c "python3 - <<'PY'
from collections import Counter
bins=Counter()
with open('chr22_alns.tsv') as f:
    next(f)
    for line in f:
        p=line.split('\\t')
        pos=int(p[2])
        bins[pos//10000]+=1
print(len(bins), sum(bins.values()), max(bins.values()), sorted(bins.items())[:3])
PY"
