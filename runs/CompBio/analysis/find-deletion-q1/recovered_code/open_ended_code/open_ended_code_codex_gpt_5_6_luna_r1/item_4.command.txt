/bin/bash -lc "printf 'R1 records: '; gzip -dc find.deletion.r1.fq.gz | awk 'END{print NR/4}'
printf 'R2 records: '; gzip -dc find.deletion.r2.fq.gz | awk 'END{print NR/4}'
printf 'R1 first reads:\\n'; gzip -dc find.deletion.r1.fq.gz | head -n 12
printf 'R2 first reads:\\n'; gzip -dc find.deletion.r2.fq.gz | head -n 12
printf 'R1 length distribution (first 10000):\\n'; gzip -dc find.deletion.r1.fq.gz | awk 'NR%4==2{print length("'$0)}'"' | head -n 10000 | sort | uniq -c
printf 'R2 length distribution (first 10000):\\n'; gzip -dc find.deletion.r2.fq.gz | awk 'NR%4==2{print length("'$0)}'"' | head -n 10000 | sort | uniq -c"
