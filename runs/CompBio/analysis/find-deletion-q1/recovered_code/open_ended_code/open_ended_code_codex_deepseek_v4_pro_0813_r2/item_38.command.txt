/bin/bash -lc "bedtools makewindows -g chr22.genome -w 1000 > chr22.w1k.bed; bedtools coverage -a chr22.w1k.bed -b aligned.bam -mean > chr22.cov1k.tsv; python3 - <<'PY'
import numpy as np
# parse coverage
cov=[]
with open('chr22.cov1k.tsv') as f:
    for line in f:
        p=line.split(); cov.append(float(p[3]))
for region in [(12000000,16000000),(18000000,22000000),(22000000,26000000),(40000000,50000000)]:
    s,e=region
    vals=cov[s//1000:e//1000]
    print('region',region,'nz',sum(1 for v in vals if v>0),'zero',sum(1 for v in vals if v==0),'max',max(vals),'mean',np.mean(vals))
    # runs > 20kb
    i=0
    while i<len(vals):
        if vals[i]==0:
            j=i
            while j<len(vals) and vals[j]==0: j+=1
            length=(j-i)*1000
            if length>=20000:
                print('  zero run',s+i*1000,s+j*1000,length)
            i=j
        else:i+=1
PY"
