import sys,gzip,collections
sys.path.insert(0,'python_packages')
import ahocorasick
A=ahocorasick.Automaton()
comp=str.maketrans('ACGT','TGCA')
with gzip.open('find.deletion.r1.fq.gz','rt') as f:
 for i in range(200):
  f.readline();seq=f.readline().strip()[25:125];f.readline();f.readline()
  A.add_word(seq,i)
  A.add_word(seq.translate(comp)[::-1],i)
A.make_automaton()
counts=collections.Counter()
chrom=None
buf=''
def scan(chrom,buf):
 if not chrom:return
 hits=set(i for end,i in A.iter(buf))
 counts[chrom]=len(hits)
 print(chrom,len(buf),len(hits),flush=True)
with gzip.open('hg38.fa.gz','rt') as f:
 parts=[]
 for line in f:
  if line.startswith('>'):
   if chrom:
    seq=''.join(parts).upper();scan(chrom,seq)
    if counts[chrom]>150:
     with open('source.fa','w') as out:
      out.write('>'+chrom+'\\n')
      for j in range(0,len(seq),60):out.write(seq[j:j+60]+'\\n')
     open('source_chromosome.txt','w').write(chrom+'\\n')
     break
   chrom=line[1:].split()[0];parts=[]
  else:parts.append(line.strip())
 else:scan(chrom,''.join(parts).upper())
print(counts.most_common(10))
PY"
