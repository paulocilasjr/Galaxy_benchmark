/bin/bash -lc "cat > scripts/make_windows25.py <<'PY'
import gzip
positions=list(range(0,83,8))
L=25
with gzip.open('gene.fusion.q2.fq.gz','rt') as f, open('tmp/read_windows25.fa','w') as out:
    while True:
        name=f.readline().strip()
        if not name: break
        seq=f.readline().strip().upper(); f.readline(); f.readline()
        rid=name[1:].split()[0]
        for p in positions:
            s=seq[p:p+L]
            if len(s)==L:
                out.write(f'>{rid}|{p}\\n{s}\\n')
PY
python3 scripts/make_windows25.py
minimap2 -t 4 -x sr -k11 -w3 -N 20 -p 0.8 refs/gencode.v49.transcripts.fa.gz tmp/read_windows25.fa > results/windows25_to_gencode.paf
wc -l results/windows25_to_gencode.paf"
