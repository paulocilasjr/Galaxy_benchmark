/bin/bash -lc "cat > scripts/extract_gene_transcripts.py <<'PY'
import gzip
want={'MYC','SP1','CATSPER4','ZIC2'}
with gzip.open('refs/gencode.v49.transcripts.fa.gz','rt') as f, open('refs/candidate_gene_transcripts.fa','w') as out:
    keep=False
    for line in f:
        if line.startswith('>'):
            parts=line[1:].strip().split('|')
            gene=parts[5] if len(parts)>5 else ''
            keep=gene in want
        if keep:
            out.write(line)
PY
python3 scripts/extract_gene_transcripts.py
awk '/"'^>/{print}'"' refs/candidate_gene_transcripts.fa | head -40
minimap2 -t 2 -x sr -k11 -w3 -N 50 -p 0.1 refs/candidate_gene_transcripts.fa results/candidate_reads.fq > results/candidate_reads_to_implicated.paf
wc -l results/candidate_reads_to_implicated.paf
head -20 results/candidate_reads_to_implicated.paf"
