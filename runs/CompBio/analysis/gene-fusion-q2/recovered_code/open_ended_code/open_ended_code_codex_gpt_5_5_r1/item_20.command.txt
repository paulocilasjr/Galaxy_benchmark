/bin/bash -lc "cat > scripts/make_windows.py <<'PY'
import gzip
positions=[0,12,24,36,48,60,72]
L=35
with gzip.open('gene.fusion.q2.fq.gz','rt') as f, open('tmp/read_windows35.fa','w') as out:
    while True:
        name=f.readline().strip()
        if not name: break
        seq=f.readline().strip().upper(); f.readline(); f.readline()
        rid=name[1:].split()[0]
        for p in positions:
            s=seq[p:p+L]
            if len(s)==L:
                out.write(f'>{rid}|{p}\\n{s}\\n')
PY
python3 scripts/make_windows.py
wc -l tmp/read_windows35.fa
minimap2 -t 4 -c -k15 -w5 -N 20 -p 0.8 refs/gencode.v49.transcripts.fa.gz tmp/read_windows35.fa > results/windows35_to_gencode.paf
wc -l results/windows35_to_gencode.paf
head -3 results/windows35_to_gencode.paf"
