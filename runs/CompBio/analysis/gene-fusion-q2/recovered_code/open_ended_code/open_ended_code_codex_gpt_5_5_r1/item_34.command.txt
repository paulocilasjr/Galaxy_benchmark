/bin/bash -lc "cat > scripts/all_pairs_table.py <<'PY'
# emit all pair counts from existing parser, threshold any, involving candidates and zero-rev pairs
from collections import defaultdict, Counter
hits=defaultdict(list)
with open('results/windows35_to_gencode.paf') as f:
    for line in f:
        fields=line.rstrip('\\n').split('\\t')
        q=fields[0]; strand=fields[4]; t=fields[5]
        matches=int(fields[9]); blen=int(fields[10]); qs=int(fields[2]); qe=int(fields[3])
        if blen<28 or matches<28 or matches/blen<0.90: continue
        gene=t.split('|')[5]
        hits[q].append((matches, qe-qs, gene, strand))
assign={}
for q,hs in hits.items():
    best=max((h[0],h[1]) for h in hs)
    top=[h for h in hs if (h[0],h[1])==best]
    genes={h[2] for h in top}; strands={h[3] for h in top}
    if len(genes)==1 and len(strands)==1:
        assign[q]=(next(iter(genes)), next(iter(strands)))
byread=defaultdict(dict)
for q,(g,s) in assign.items():
    rid,pos=q.rsplit('|',1); byread[rid][int(pos)]=(g,s)
pair=Counter()
for rid,d in byread.items():
    blocks=[]
    for pos,(g,s) in sorted(d.items()):
        if not blocks or blocks[-1][1]"'!=g or blocks[-1][2]!=s:
            blocks.append((pos,g,s))
    seen=set()
    for a,b in zip(blocks,blocks[1:]):
        _,g1,s1=a; _,g2,s2=b
        if g1==g2 or s1!=s2: continue
        p=(g1,g2) if s1=='"'+' else (g2,g1)
        seen.add(p)
    for p in seen: pair[p]+=1
with open('results/all_pairs.tsv','w') as out:
    for (a,b),c in pair.most_common():
        out.write(f'{a}\\t{b}\\t{c}\\t{pair.get((b,a),0)}\\n')
for gene in ['MYC','SP1','CATSPER4','ZIC2']:
    print('---',gene)
    rows=[(c,a,b,r) for (a,b),c in pair.items() if a==gene or b==gene for r in [pair.get((b,a),0)]]
    for c,a,b,r in sorted(rows, reverse=True)[:100]:
        print(c,r,a,b)
print('--- zero reverse >=8')
zeros=[(c,a,b) for (a,b),c in pair.items() if c>=8 and pair.get((b,a),0)==0]
for c,a,b in sorted(zeros, reverse=True)[:200]: print(c,a,b)
PY
python3 scripts/all_pairs_table.py > results/all_pairs_summary.txt
sed -n '1,220p' results/all_pairs_summary.txt"
