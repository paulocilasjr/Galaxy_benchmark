/bin/bash -lc "python3 - <<'PY'
import subprocess,re,collections
pairs=collections.Counter(); clusters=collections.defaultdict(set); examples={}
for line in subprocess.Popen(['samtools','view','reads_candchrs.bam'],stdout=subprocess.PIPE,text=True).stdout:
 p=line.rstrip().split('\\t'); flag=int(p[1]); rname=p[2]; pos=int(p[3]); cigar=p[5]; tags={}
 for f in p[11:]:
  if ':' in f:
   k,t,v=f.split(':',2);tags[k]=v
 sa=tags.get('SA')
 if not sa:continue
 for entry in sa.rstrip(';').split(';'):
  if not entry:continue
  parts=entry.split(',')
  if len(parts)<6:continue
  rt=parts[0]; pairs[(rname,rt)]+=1
  # rough cluster by chrom pair and positions rounded 100
  clusters[(rname,pos//100,rt,int(parts[1])//100)].add(p[0])
  examples.setdefault((rname,rt),(p[0],rname,pos,cigar,rt,parts[1],parts[2],parts[3]))
print('top chrom pairs:')
for k,c in pairs.most_common(30):print(c,k)
print('\\ntop clusters:')
for k,reads in sorted(clusters.items(),key=lambda kv:-len(kv[1]))[:30]:print(len(reads),k,next(iter(reads)))
PY"
