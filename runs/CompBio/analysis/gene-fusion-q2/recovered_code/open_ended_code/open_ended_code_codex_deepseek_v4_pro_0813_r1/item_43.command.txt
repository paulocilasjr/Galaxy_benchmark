# transcript gene map and lengths
trans_gene={}
with open('hs_cdna.fa') as f:

        if line.startswith('>'):
            h=line[1:].strip(); tid=h.split()[0]

                if part.startswith('gene_symbol:'): trans_gene[tid]=part.split(':',1)[1];break

def cigar_bounds(cigar,pos):
    m=re.findall(r'(\\d+)([MIDNSHP=X])',cigar)
    ref=pos; covered=[]
    for ln,op in m:
        ln=int(ln)
        if op in 'MDN=X': covered.append((ref,ref+ln-1)); ref+=ln
        elif op in 'ISHP': pass
    if not covered:return None
    return (covered[0][0],covered[-1][1])

def qint(cigar):
    m=re.findall(r'(\\d+)([MIDNSHP=X])',cigar)
    q=0;qs=None;qe=None
    for ln,op in m:
        ln=int(ln)
        if op in 'SH': q+=ln
        elif op in 'M=X':
            if qs is None:qs=q
            q+=ln;qe=q
        elif op=='I':q+=ln
    return (qs,qe)

def bp_for(cigar,pos):
    b=cigar_bounds(cigar,pos)
    m=re.findall(r'(\\d+)([MIDNSHP=X])',cigar)
    if not b:return None
    if m and m[0][1] in 'SH': return b[0] # right segment, start
    if m and m[-1][1] in 'SH': return b[1] # left segment, end
    return None

def exon_of(tid,pos):
    base=tid.split('.')[0]
    bounds=tr_exons.get(base)
    if not bounds:return None,None
    for i,(s,e) in enumerate(bounds,1):
        if s<=pos<=e:return i,(s,e)
    return None,None

        else:
            left_g,right_g=g2,g1; left_strand,right_strand=strand2,strand1
