/bin/bash -lc "python3 - <<'PY'
import requests, json
for gene in ['SP1','MYC','ZIC2','CATSPER4']:
    url=f'https://rest.ensembl.org/lookup/symbol/homo_sapiens/{gene}?expand=1'
    r=requests.get(url,headers={'Content-Type':'application/json'})
    data=r.json()
    print('\\nGENE',gene)
    for tr in data.get('Transcript',[]):
        ex=tr.get('Exon',[])
        total=sum((e['end']-e['start']+1) for e in ex)
        print(tr['id'],'biotype',tr.get('biotype'),'strand',tr.get('strand'),'length',tr.get('length'),'exons',len(ex),'cDNA_exons_sum',total)
        # print cumulative starts
        cum=0; bounds=[]
        for i,e in enumerate(ex,1):
            ln=e['end']-e['start']+1
            bounds.append((i,cum+1,cum+ln,e['start'],e['end']))
            cum+=ln
        print('  ',bounds[:10],'... last',bounds[-1] if bounds else None)
PY"
