/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict
import subprocess
trans_gene={}
with open('hs_cdna.fa') as f:
    for line in f:
        if line.startswith('>'):
            h=line[1:].strip(); tid=h.split()[0]
            for part in h.split():
                if part.startswith('gene_symbol:'): trans_gene[tid]=part.split(':',1)[1];break
cnt=Counter()
proc=subprocess.Popen(['samtools','view','-F','0x100','reads_cdna.bam'],stdout=subprocess.PIPE,text=True)
for line in proc.stdout:
    p=line.split('\\t'); flag=int(p[1])
    if flag&4: continue
    g=trans_gene.get(p[2])
    if g:cnt[g]+=1
proc.wait()
for g in ['SP1','MYC','ZIC2','CATSPER4','EEF1A1','ACTB','IGKV1-33','IGKJ3','IGKC','IGHG1']:
    print(g,cnt.get(g,0))
PY"
