/bin/bash -lc "python - <<'PY'
from collections import defaultdict,Counter
D=defaultdict(list)
for l in open('reads.partial.paf'):
 f=l.rstrip().split('\\t'); gene=f[5].split('|')[5]
 D[f[0]].append((int(f[2]),int(f[3]),f[4],gene,int(f[9]),int(f[10]),int(f[11]),f[5].split('|')[0]))
C=Counter(); examples={}
for q,a in D.items():
 # ignore alternate alignments to same query segment; --secondary=no leaves split primaries
 a=sorted(a)
 for i,x in enumerate(a):
  for y in a[i+1:]:
   if x[3]"'!=y[3] and x[1]-x[0]>=20 and y[1]-y[0]>=20:
    pair=(x[3],y[3]) if x[0]<=y[0] else (y[3],x[3])
    C[pair]+=1; examples.setdefault(pair,(q,x,y))
for p,n in C.most_common(50): print(n,p,examples[p])
print('"'mapped_reads',len(D),'chimeric_pairs',sum(C.values()))
PY"
