/bin/bash -lc "python - <<'PY'
from collections import defaultdict,Counter
D=defaultdict(list)
for l in open('reads.partial.paf'):
 f=l.rstrip().split('\\t'); gene=f[5].split('|')[5]
 D[f[0]].append((int(f[2]),int(f[3]),f[4],gene,int(f[9]),int(f[10]),int(f[11])))
C=Counter(); ex={}
for q,a in D.items():
 a=sorted(a)
 for i,x in enumerate(a):
  for y in a[i+1:]:
   if x[3]==y[3] or x[2]"'!=y[2] or min(x[1]-x[0],y[1]-y[0])<20: continue
   pair=(x[3],y[3]) if x[2]=='"'+' else (y[3],x[3])
   C[pair]+=1; ex.setdefault(pair,(q,x,y))
for p,n in C.most_common(60): print(f'{n:3}', '-'.join(p), ex[p])
PY"
