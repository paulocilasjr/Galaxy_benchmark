/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
rec=SeqIO.read('NODE_63_rc.fa','fasta')
s=str(rec.seq)
with open('gap.fa','w') as f: f.write('>gap\\n'+s[393:721]+'\\n')
with open('SP1_seg.fa','w') as f: f.write('>SP1_seg\\n'+s[721:927]+'\\n')
with open('MYC_seg.fa','w') as f: f.write('>MYC_seg\\n'+s[0:393]+'\\n')
with open('CATSPER4_seg.fa','w') as f: f.write('>CATSPER4_seg\\n'+s[923:1396]+'\\n')
PY
cat gap.fa SP1_seg.fa MYC_seg.fa CATSPER4_seg.fa > segments.fa
minimap2 -t 4 -a -x asm5 gencode.mmi segments.fa > segments.sam
grep -v '"'^@'"' segments.sam | cut -f1-6,14 | head -n 100"
