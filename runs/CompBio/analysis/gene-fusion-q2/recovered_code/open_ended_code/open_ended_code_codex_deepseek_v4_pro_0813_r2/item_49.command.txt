/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
bam=pysam.AlignmentFile('node63_reads.sorted.bam','rb')
left=[]; right=[]; stats=[]
for aln in bam.fetch():
    if aln.is_unmapped or aln.is_secondary or aln.is_supplementary: continue
    # Get aligned pairs / cigar
    seq=aln.query_sequence
    cigar=aln.cigartuples
    # find leading/trailing S and positions
    qpos=0
    rpos=aln.reference_start
    ops=[]
    for op,n in cigar:
        if op==4: # S
            if not ops or ops[-1][1]"'!=0:
                if rpos==aln.reference_start: left.append((seq[qpos:qpos+n], rpos))
                elif rpos>=1395: right.append((seq[qpos:qpos+n], rpos))
            qpos+=n
        elif op==0:
            qpos+=n; rpos+=n
        elif op==1:
            qpos+=n
        elif op==2:
            rpos+=n
        ops.append((op,n))
print('"'left clips',len(left),'right clips',len(right))
print('left clip lengths',Counter(len(s) for s,r in left))
print('right clip lengths',Counter(len(s) for s,r in right))
print('long left clips')
for s,r in left:
    if len(s)>=20: print(r,len(s),s)
print('long right clips')
for s,r in right:
    if len(s)>=20: print(r,len(s),s)
PY"
