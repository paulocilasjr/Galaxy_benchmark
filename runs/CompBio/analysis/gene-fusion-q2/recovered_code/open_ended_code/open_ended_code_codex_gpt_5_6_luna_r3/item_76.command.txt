/bin/bash -lc "python - <<'PY'
import gzip, subprocess
wanted={'READ_97800','READ_26857','READ_52960','READ_13910'}
fastq={}
with gzip.open('gene.fusion.q2.fq.gz','rt') as f:
    while True:
        h=f.readline().rstrip()
        if not h: break
        s=f.readline().rstrip(); f.readline(); f.readline()
        if h[1:] in wanted: fastq[h[1:]]=s
sam=subprocess.check_output(['samtools','view','reads.cdna.bam'],text=True)
for line in sam.splitlines():
    a=line.split('\\t')
    if a[0] not in wanted or int(a[1])&0x100 or int(a[1])&0x800: continue
    s=a[9]
    print(a[0], 'flag',a[1], 'fastq==sam',fastq[a[0]]==s, 'fastq==rcsam',fastq[a[0]]==s.translate(str.maketrans('ACGT','TGCA'))[::-1])
PY"
