/bin/bash -lc "samtools faidx spades_out/transcripts.fasta 'NODE_58_length_1396_cov_39.263941_g56_i0:676-1003' | sed '1d' | tr -d '\\n' | tr 'ACGTacgt' 'TGCAtgca' | rev | awk 'BEGIN{print \">gap\"} {print}' > gap.fa
blastn -task blastn -query gap.fa -subject <(gzip -dc gencode.v47.transcripts.fa.gz) -outfmt '6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore' -evalue 1e-5 -max_target_seqs 30 -max_hsps 5 -num_threads 8 > gap_vs_gencode.tsv 2> gap_blast.log
wc -l gap_vs_gencode.tsv; sort -k12,12nr gap_vs_gencode.tsv | sed -n '1,40p'; tail -n 20 gap_blast.log"
