/bin/bash -lc "python3 - <<'PY'
import pysam,re
sam=pysam.AlignmentFile('contigs2trans.sam','r')
targets={'MYC','SP1','ZIC2','CATSPER4'}
rows=[]
for al in sam:
    if al.is_unmapped or al.is_secondary or al.is_supplementary: continue
    gene=al.reference_name.split('|')[5]
    if gene in targets:
        m=re.search(r'cov_([0-9.]+)',al.query_name); cov=float(m.group(1)) if m else 0
        rows.append((al.query_name,gene,cov,al.reference_start,al.reference_end,al.query_alignment_start,al.query_alignment_end))
for r in sorted(rows,key=lambda x:(x[1],-x[2])):
    print(r)
PY"
