/bin/bash -lc "python3 - <<'PY'
import pysam
for f in ['NODE_2_length_3643_cov_4.687570','NODE_83_length_1396_cov_36.266219']:
    bam=pysam.AlignmentFile(f+'.bam','rb')
    lens={ref:bam.get_reference_length(ref) for ref in bam.references}
    depth={r:[0]*l for r,l in lens.items()}
    for al in bam.fetch():
        if al.is_unmapped: continue
        r=al.reference_name
        for pos in al.get_reference_positions(full_length=False):
            depth[r][pos]+=1
    print('===',f)
    for r,l in lens.items():
        # print min/mean by 100bp windows
        vals=[]
        for s in range(0,l,100):
            seg=depth[r][s:min(s+100,l)]
            vals.append((s,min(seg),sum(seg)/len(seg),max(seg)))
        for s,mn,av,mx in vals:
            if mn<5 or mx>40:
                print(s, 'min',mn,'avg',round(av,1),'max',mx)
PY"
