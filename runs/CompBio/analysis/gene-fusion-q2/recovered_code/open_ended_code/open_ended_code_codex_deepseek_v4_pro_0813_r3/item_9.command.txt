def reads(n):

def overlap(a,b,minlen=20):
    best=(0,0)
    for k in range(minlen,min(len(a),len(b))+1):
        if a[-k:]==b[:k]: best=(k,0)
        if b[-k:]==a[:k]: best=(k,1)
    return best
