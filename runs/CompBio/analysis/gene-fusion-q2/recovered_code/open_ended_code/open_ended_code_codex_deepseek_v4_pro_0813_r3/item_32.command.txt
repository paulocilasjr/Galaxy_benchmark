/bin/bash -lc "python3 - <<'PY'
import pysam
from Bio.Seq import Seq
from Bio.SeqRecord import SeqRecord
from Bio import SeqIO
bam=pysam.AlignmentFile('NODE_83_length_1396_cov_36.266219.bam','rb')
recs=[]
for al in bam.fetch():
    if al.is_unmapped or al.is_secondary or al.is_supplementary: continue
    recs.append(SeqRecord(Seq(al.query_sequence), id=al.query_name, description=''))
SeqIO.write(recs,'node83_reads.fa','fasta')
print(len(recs))
PY
makeblastdb -in candidates.fa -dbtype nucl -title cand -out cand_blast >/dev/null
blastn -query node83_reads.fa -db cand_blast -outfmt '6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore' -evalue 1e-4 -max_target_seqs 5 -num_threads 4 > node83_reads_blast.tsv
wc -l node83_reads_blast.tsv"
