/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
r=SeqIO.read('NODE_83_length_1396_cov_36.266219.fa','fasta'); s=str(r.seq)
# write segments as fasta
segs={'seg_mid':s[388:927], 'seg_sp1first':s[388:564], 'seg_zic2mid':s[559:726], 'seg_sp1second':s[721:927]}
with open('segments.fa','w') as f:
    for n,seq in segs.items(): f.write(f'>{n}\\n{seq}\\n')
PY
blastn -query segments.fa -db cand_blast -outfmt '6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore' -evalue 1e-4 -max_target_seqs 20 -num_threads 4"
