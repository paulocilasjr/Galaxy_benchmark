/bin/bash -lc "python3 - <<'PY'
import pysam
# get node83 read names
bam=pysam.AlignmentFile('NODE_83_length_1396_cov_36.266219.bam','rb')
node=set()
for al in bam.fetch():
    if not al.is_unmapped and not al.is_secondary and not al.is_supplementary:
        node.add(al.query_name)
# get candidate gene read names from map.sam
sam=pysam.AlignmentFile('map.sam','r')
cand=set(); genes={}
for al in sam.fetch():
    if al.is_unmapped or al.is_secondary or al.is_supplementary: continue
    g=al.reference_name.split('|')[5]
    if g in {'MYC','SP1','ZIC2','CATSPER4'}:
        cand.add(al.query_name); genes[al.query_name]=g
print('node',len(node),'cand',len(cand),'overlap',len(node&cand))
# candidate reads not in node counts
from collections import Counter
print('cand-not-node',Counter(genes[n] for n in cand-node))
print('node-not-cand',len(node-cand))
# distribution of node reads among candidate genes via map.sam
print('node overlap genes',Counter(genes[n] for n in (node&cand)))
PY"
