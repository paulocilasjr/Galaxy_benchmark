/bin/bash -lc "python3 - <<'PY'
import pysam, re
sam=pysam.AlignmentFile('contigs2trans.sam','r')
from collections import defaultdict
contig_hits=defaultdict(list)
for al in sam:
    if al.is_unmapped: continue
    rna=al.query_name
    target=al.reference_name
    gene=target.split('|')[5]
    contig_hits[rna].append((al.reference_start, al.reference_end, gene, al.query_alignment_start, al.query_alignment_end, al.flag, al.mapping_quality, al.is_supplementary, al.is_secondary))
sam.close()
rows=[]
for rna,hits in contig_hits.items():
    primary=[h for h in hits if not h[8] and not h[7]]
    # distinct genes from all
    genes=[]
    for h in sorted(hits, key=lambda x:(x[3],x[4])):
        if not h[2] in genes: genes.append(h[2])
    if len(genes)>=3:
        rows.append((rna,len(set(genes)),genes, len(hits), [h[2] for h in hits if not h[7]][:10]))
print('contigs with >=3 distinct genes:',len(rows))
for r in sorted(rows,key=lambda x:-x[1])[:50]:
    print(r[0], 'distinct',r[1], r[2])
PY"
