/bin/bash -lc "printf 'Alignment summary:\\n'
samtools flagstat pc_align.bam | sed -n '1,12p'
printf '\\nSAM records with split-alignment tags:\\n'
grep -c 'SA:Z:' pc_align.sam || true
printf 'Records with soft clipping (first 10):\\n'
awk 'BEGIN{FS=\"\\t\"} "'$6 ~ /S/ {print $1,$2,$3,$4,$5,$6; n++; if(n==10)exit}'"' pc_align.sam
printf '\\nFirst SAM alignments:\\n'
samtools view pc_align.bam | sed -n '1,8p'"
