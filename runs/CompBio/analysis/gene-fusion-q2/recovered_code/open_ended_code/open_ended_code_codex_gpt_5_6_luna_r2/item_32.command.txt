/bin/bash -lc "python3 - <<'PY'
import collections
c=collections.Counter(); primary=collections.Counter(); supp=collections.Counter(); flagc=collections.Counter()
with open('pc_align.sam') as f:
 for line in f:
  if line[0]=='@': continue
  a=line.rstrip().split('\\t'); q=a[0]; fl=int(a[1]); rn=a[2]
  if rn=='*': continue
  x=rn.split('|'); g=x[5] if len(x)>5 else rn
  flagc[fl&0xfff]+=1
  if fl&2048: supp[g]+=1
  elif not fl&256: primary[g]+=1
print('primary total',sum(primary.values()),'genes',len(primary))
for g,n in primary.most_common(50): print(n,g)
print('candidate primary')
for g in ['ZIC2','SP1','MYC','CATSPER4']: print(g,primary[g])
print('supp candidate', {g:supp[g] for g in ['ZIC2','SP1','MYC','CATSPER4']})
PY"
