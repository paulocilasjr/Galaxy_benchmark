import gzip
from Bio.Seq import Seq
def g(h):
 x=h.split('|'); return x[5] if len(x)>5 else x[0]
refs={}
with gzip.open('gencode.v49.pc_transcripts.fa.gz','rt') as f:
 h=None; c=[]
 for line in f:
  if line.startswith('>'):
   if h and g(h) in {'ZIC2','SP1','MYC','CATSPER4'}: refs.setdefault(g(h),[]).append((h,''.join(c)))
   h=line[1:].strip(); c=[]
  else:c.append(line.strip())
 if h and g(h) in {'ZIC2','SP1','MYC','CATSPER4'}: refs.setdefault(g(h),[]).append((h,''.join(c)))
for gene in refs:
 print(gene,[(h.split('|')[0],len(s)) for h,s in refs[gene]])
# print exact relevant sequences per transcript from SAM-dominant first records
for gene,coord in [('ZIC2',(1330,1600)),('SP1',(450,750)),('SP1',(1650,2150)),('MYC',(300,550)),('CATSPER4',(1320,1500))]:
 print('\\n',gene,coord)
 for h,s in refs[gene]:
  print(h.split('|')[0],s[coord[0]-1:coord[1]])
PY"
