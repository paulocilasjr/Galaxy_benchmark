/bin/bash -lc "set -o pipefail
printf 'FASTQ records: '
gzip -cd gene.fusion.q2.fq.gz | awk 'END{print NR/4}'
printf 'First read records:\\n'
gzip -cd gene.fusion.q2.fq.gz | sed -n '1,20p'
printf '\\nSequence length distribution:\\n'
gzip -cd gene.fusion.q2.fq.gz | awk 'NR%4==2{print length("'$0)}'"' | sort | uniq -c | head -30
printf '\\nAvailable relevant tools:\\n'
for t in seqkit minimap2 bwa bowtie2 blastn samtools bedtools diamond kallisto salmon STAR hisat2; do command -v \""'$t" || true; done'
