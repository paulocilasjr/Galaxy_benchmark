/bin/bash -lc "curl -sS -H 'Content-Type: application/json' 'https://rest.ensembl.org/variation/human/rs241311?content-type=application/json' -o ensembl_rs241311.json
curl -sS -H 'Content-Type: application/json' 'https://rest.ensembl.org/variation/human/rs200533370?content-type=application/json' -o ensembl_rs200533370.json
python3 - <<'PY'
import json
for fn in ['ensembl_rs241311.json','ensembl_rs200533370.json']:
 d=json.load(open(fn)); print(fn)
 for m in d.get('mappings',[]): print(m.get('assembly_name'),m.get('seq_region_name'),m.get('start'),m.get('end'),m.get('allele_string'))
PY"
