/bin/bash -lc "curl -sL 'https://www.bing.com/search?q=%22CA548798891%22' | python3 - <<'PY'
import sys, re, html
s=sys.stdin.read()
for m in re.finditer(r'<li class=\"b_algo\".*?</li>', s, re.S):
    frag=m.group(0)
    title=re.sub('<.*?>',' ',frag)
    title=html.unescape(re.sub(r'\\s+',' ',title))
    print(title[:1000])
PY"
