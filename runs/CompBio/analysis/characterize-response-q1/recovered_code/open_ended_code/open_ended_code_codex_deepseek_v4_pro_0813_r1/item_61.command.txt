/bin/bash -lc "python3 - <<'PY'
import GEOparse, numpy as np, pandas as pd, re, traceback
from collections import defaultdict

def get_data(gse):
    g=GEOparse.get_GEO(geo=gse,destdir='./geo',silent=True)
    mats=[]
    for gsm_name,gsm in g.gsms.items():
        t=gsm.table.copy()
        valcol='VALUE' if 'VALUE' in t.columns else [c for c in t.columns if c not in ['ID_REF']][0]
        t=t[['ID_REF',valcol]].rename(columns={valcol:'value'})
        t['sample']=gsm_name
        mats.append(t)
    df=pd.concat(mats,ignore_index=True)
    expr=df.pivot(index='ID_REF',columns='sample',values='value').astype(float)
    sym=None
    for gpl in g.gpls.values():
        ann=gpl.table
        symcol=next((c for c in ann.columns if c.lower() in ['gene symbol','genesymbol','symbol','gene_symbol']),None)
        if symcol:
            sym=ann.set_index('ID')[symcol]
            break
    return g,expr,sym

def compare(gse, cond_pat, ctrl_pat, label):
    g,expr,sym=get_data(gse)
    names=list(expr.columns)
    cond=[];ctrl=[]
    for name in names:
        gsm=g.gsms[name]
        title=gsm.metadata.get('title')
        if isinstance(title,list): title=title[0]
        source=gsm.metadata.get('source_name_ch1')
        if isinstance(source,list): source=source[0]
        s=str(title)+' '+str(source)
        if re.search(cond_pat,s,re.I): cond.append(name)
        elif re.search(ctrl_pat,s,re.I): ctrl.append(name)
    print('\\n===',gse,label,'cond',cond,'ctrl',ctrl)
    if not cond or not ctrl: return
    m0=expr[cond].mean(axis=1); m1=expr[ctrl].mean(axis=1)
    logfc=np.log2((m0+1)/(m1+1))
    best=defaultdict(lambda:(0,0,''))
    for probe in expr.index:
        if sym is None or probe not in sym.index: continue
        gene=str(sym.loc[probe]).split('///')[0].strip().upper()
        if gene and abs(logfc[probe])>best[gene][0]: best[gene]=(abs(logfc[probe]),logfc[probe],probe)
    ranked=sorted(best.items(),key=lambda kv:-kv[1][0])
    top=[g for g,v in ranked[:112]]
    qlist=open('../characterize.response.q1.txt').read().split()
    q=set(x.upper() for x in qlist)
    inter=set(top)&q
    print('top112 overlap',len(inter), sorted(inter,key=lambda x:qlist.index(x.capitalize()) if x.capitalize() in qlist else 999))
    rank={g:i+1 for i,(g,v) in enumerate(ranked)}
    rr=[(g.capitalize(),rank.get(g.upper())) for g in qlist if g.upper() in rank]
    print('query rank first 40:',rr[:40])
    return ranked

compare('GSE17721', r'"'^Lps,'"\\s*24_hours', r'"'^Control,'"\\s*24_hours', 'LPS24 vs Ctrl24')
compare('GSE17721', r'"'^Lps,'"\\s*24_hours', r'"'^Control,'"\\s*0_hours', 'LPS24 vs Ctrl0')
compare('GSE17721', r'"'^Lps,'"\\s*6_hours', r'"'^Control,'"\\s*6_hours', 'LPS6 vs Ctrl6')
compare('GSE69076', r'LPS 24hr', r'Unstimulated', 'LPS24 vs Unstim')
compare('GSE69076', r'LPS 6hr', r'Unstimulated', 'LPS6 vs Unstim')
PY"
