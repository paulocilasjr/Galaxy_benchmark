import requests, json, time
q='(lipopolysaccharide OR LPS) AND (monocyte OR monocytes OR macrophage) AND \"Mus musculus\"[Organism] AND \"Expression profiling by array\"[DataSet Type]'
r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':q,'retmax':100,'retmode':'json'})
d=r.json(); ids=d['esearchresult']['idlist']; print('count',d['esearchresult']['count'],'ids',len(ids))
r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':','.join(ids[:100]),'retmode':'json'})
s=r.json()['result']
for uid in ids[:100]:
    x=s[uid]
    print(uid,x.get('accession'), x.get('title','')[:140], x.get('summary','')[:160].replace('\\n',' '), 'gpl',x.get('gpl'))
PY"
