/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
qset=set(g.upper() for g in open('characterize.response.q1.txt').read().split())
def calc(df,ctrl,lps,label):
 lib=df.sum(axis=0); cpm=df/lib*1e6
 A=cpm[ctrl].mean(axis=1);B=cpm[lps].mean(axis=1);lfc=np.log2((B+0.5)/(A+0.5))
 rows=[];seen=set()
 for g,x in lfc.items():
  su=str(g).upper()
  if su in seen: continue
  seen.add(su);rows.append((str(g),x))
 d=pd.DataFrame(rows,columns=['g','x']).drop_duplicates('g').sort_values('x',ascending=False)
 print('\\n',label,'genes',len(d))
 for name,dd in [('up',d.head(112)),('down',d.tail(112)),('abspos',d[d.x>0].assign(al=d[d.x>0].x.abs()).sort_values('al',ascending=False).head(112)),('absneg',d[d.x<0].assign(al=d[d.x<0].x.abs()).sort_values('al',ascending=False).head(112))]:
  print(name,len(set(dd.g.str.upper())&qset),[x for x in dd.g.str.upper().tolist() if x in qset][:40])
# GSE204955
df=pd.read_csv('geo2/GSE204955_counts.tsv.gz',sep='\\t',index_col=0)
df=df.drop(columns=['ext_gene']).set_index('ext_gene')
calc(df,['R1A','R1B','R1C'],['R2A','R2B','R2C'],'GSE204955 LPS')
# GSE134869
df=pd.read_csv('geo2/GSE134869_counts.tsv.gz',sep='\\t',index_col=0)
ctrl=[c for c in df.columns if c.startswith('Ctrl_')]; lps=[c for c in df.columns if c.startswith('LPS_')]
calc(df,ctrl,lps,'GSE134869 LPS')
PY"
