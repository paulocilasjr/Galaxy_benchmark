/bin/bash -lc "python3 - <<'PY'
import requests,os
for gse in ['GSE78209','GSE64027','GSE75246']:
 nnn=gse[:-3]+'nnn'; base=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{nnn}/{gse}/'
 for kind in ['matrix','soft']:
  fn=f'{gse}_{\"series_matrix.txt.gz\" if kind==\"matrix\" else \"family.soft.gz\"}'
  url=base+kind+'/'+fn; path='geo2/'+gse+'_'+fn
  if os.path.exists(path) and os.path.getsize(path)>1000: continue
  r=requests.get(url,timeout=120); print(gse,kind,r.status_code,len(r.content)); open(path,'wb').write(r.content)
PY
ls -lh geo2/GSE78209* geo2/GSE64027* geo2/GSE75246*"
