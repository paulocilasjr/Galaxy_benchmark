/bin/bash -lc "mkdir -p enrichr_libs && python3 - <<'PY'
import requests, os, time
libs=['RNA-Seq_Disease_Gene_and_Drug_Signatures_from_GEO','Disease_Perturbations_from_GEO_up','Disease_Perturbations_from_GEO_down','Drug_Perturbations_from_GEO_up','Drug_Perturbations_from_GEO_down','Microbe_Perturbations_from_GEO_up','Microbe_Perturbations_from_GEO_down','Ligand_Perturbations_from_GEO_up','Ligand_Perturbations_from_GEO_down','Aging_Perturbations_from_GEO_up','Aging_Perturbations_from_GEO_down','Mouse_Gene_Atlas','CellMarker_Augmented_2021']
for lib in libs:
 path='enrichr_libs/'+lib+'.txt'
 if os.path.exists(path) and os.path.getsize(path)>1000:
  print('exists',path,os.path.getsize(path)); continue
 r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'mode':'text','libraryName':lib},timeout=120)
 print(lib, r.status_code, len(r.content))
 if r.status_code==200:
  open(path,'w').write(r.text)
  time.sleep(.2)
 else:
  print(r.text[:200])
PY
ls -lh enrichr_libs"
