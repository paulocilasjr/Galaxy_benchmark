/bin/bash -lc "python3 - <<'PY'
import requests, os
gses=['GSE134869','GSE120807','GSE120808','GSE62746','GSE66178','GSE66315','GSE96628','GSE56929','GSE55068','GSE204955','GSE204957','GSE181134','GSE163975']
for gse in gses:
 nnn=gse[:-3]+'nnn'
 url=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{nnn}/{gse}/soft/{gse}_family.soft.gz'
 path='geo2/'+gse+'_family.soft.gz'
 if os.path.exists(path) and os.path.getsize(path)>1000: continue
 r=requests.get(url,timeout=180)
 print(gse,r.status_code,len(r.content))
 if r.status_code==200: open(path,'wb').write(r.content)
PY
ls -lh geo2/*_family.soft.gz"
