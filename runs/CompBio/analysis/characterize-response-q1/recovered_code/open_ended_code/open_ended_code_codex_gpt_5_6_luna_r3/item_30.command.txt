/bin/bash -lc "python - <<'PY'
import pandas as pd
p='immune_dictionary_table3.xlsx'
target=[x.strip() for x in open('characterize.response.q1.txt') if x.strip()]
print('target',len(target),target[:12],target[-12:])
allrows=[]
for s in pd.ExcelFile(p).sheet_names:
 d=pd.read_excel(p,sheet_name=s)
 for (ct,cy,cs),g in d.groupby(['Celltype_Str','Cytokine','Cytokine_Str'],sort=False):
  genes=g.Gene.astype(str).tolist()
  # overlap and target order; compare target as subsequence/prefix
  overlap=len(set(target)&set(genes))
  prefix=0
  for a,b in zip(target,genes):
   if a==b: prefix+=1
   else: break
  # rank positions for first 10
  pos=[genes.index(x) if x in genes else None for x in target[:10]]
  in_order=sum(pos[i] is not None and (i==0 or (pos[i-1] is not None and pos[i]>=pos[i-1])) for i in range(10))
  allrows.append((overlap,prefix,in_order,ct,cy,cs,len(genes),genes[:15],pos))
for r in sorted(allrows,reverse=True)[:30]:
 print(r[:8], 'pos',r[8])
PY"
