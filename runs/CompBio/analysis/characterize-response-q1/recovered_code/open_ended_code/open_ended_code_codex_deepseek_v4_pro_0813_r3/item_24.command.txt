/bin/bash -c "python3 - <<'PY'
import requests, gzip, io
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE2nnn/GSE2002/matrix/GSE2002_series_matrix.txt.gz'
r=requests.get(url,timeout=60)
print(r.status_code, len(r.content), r.url)
if r.status_code==200:
    open('GSE2002_series_matrix.txt.gz','wb').write(r.content)
    import gzip, shutil
    with gzip.open('GSE2002_series_matrix.txt.gz','rt') as f, open('GSE2002_series_matrix.txt','w') as out:
        out.write(f.read())
    print('saved')
PY"
