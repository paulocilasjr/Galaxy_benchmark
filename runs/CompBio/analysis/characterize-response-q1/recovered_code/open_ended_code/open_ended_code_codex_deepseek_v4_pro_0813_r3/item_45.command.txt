/bin/bash -c "python3 - <<'PY'
import glob, os, math, re
query=[g.strip() for g in open('characterize.response.q1.txt')]
qset=set(query); qrank={g:i for i,g in enumerate(query)}
# parse GMT lines (tab-separated, gene values maybe gene,score)
def parse_gmt(path):
 terms=[]
 for line in open(path):
  parts=line.rstrip('\\n').split('\\t')
  if len(parts)<3: continue
  name=parts[0]
  genes=[]
  for x in parts[2:]:
   if x:
    # strip score if comma
    g=x.split(',')[0].upper()
    genes.append(g)
  terms.append((name,genes))
 return terms
conds={
 'LPS':r'\\bLPS\\b|lipopolysaccharide',
 'exhaustion':r'exhaust',
 'LIF':r'\\bLIF\\b|leukemia inhibitory factor',
 'IL-39':r'IL-?39|interleukin-?39',
 'COVID-19':r'COVID|SARS-CoV|coronavirus',
 'sepsis':r'sepsis|septic',
 'Med12':r'\\bMed12\\b|MED12',
 'SLE':r'lupus|SLE|systemic lupus',
 'BMS':r'BMS|34554|345541',
 'aldesleukin':r'aldesleukin|interleukin-2|\\bIL-2\\b',
 'aging':r'ag(e|ing)|aging',
 'RA':r'rheumatoid|arthritis',
}
cells={
 'monocytes':r'monocyte|mono',
 'CD8':r'CD8|cytotoxic T',
 'CD4':r'CD4|T helper|Th cell|T-cell|T cell|Tlymphocyte|Tcell',
 'B':r'\\bB[- ]?cell|\\bB[- ]?lymphocyte|follicular B',
 'DC':r'dendritic|\\bDC\\b|pDC',
 'NK':r'\\bNK\\b|natural killer',
 'Neutrophils':r'neutrophil|granulocyte',
}
rows=[]
for path in glob.glob('libs/*.gmt'):
 lib=os.path.basename(path)[:-4]
 for name,genes in parse_gmt(path):
  gset=set(genes)
  inter=qset & gset
  if not inter: continue
  # jaccard
  j=len(inter)/len(qset | gset)
  # overlap coefficient query
  oc=len(inter)/len(qset)
  # hypergeom p approximate using R? skip
  # rank sum for query genes in term order: lower better if same order
  posmap={g:i for i,g in enumerate(genes) if g in qset}
  # mean query rank of overlap
  meanq=sum(qrank[g] for g in inter)/len(inter)
  # spearman between term positions and query ranks
  xs=[posmap[g] for g in inter]; ys=[qrank[g] for g in inter]
  if len(xs)>2:
   # pearson
   mx=sum(xs)/len(xs); my=sum(ys)/len(ys)
   num=sum((x-mx)*(y-my) for x,y in zip(xs,ys)); den=(sum((x-mx)**2 for x in xs)*sum((y-my)**2 for y in ys))**.5
   corr=num/den if den else 0
  else: corr=0
  cond_hit=[c for c,pat in conds.items() if re.search(pat,name,re.I)]
  cell_hit=[c for c,pat in cells.items() if re.search(pat,name,re.I)]
  rows.append((len(inter), j, oc, corr, lib, name, cond_hit, cell_hit, len(genes), list(inter)))
# sort by intersection desc then jaccard
rows.sort(key=lambda x:(-x[0], -x[1]))
print('top by overlap/jaccard:')
for r in rows[:100]:
 print(f\"{r[0]:3d} int j={r[1]:.3f} oc={r[2]:.3f} corr={r[3]:.3f} size={r[8]:4d} {r[4]:35s} cond={r[6]} cell={r[7]} {r[5][:120]}\")
print('\\nfiltered to any condition+cell, sorted by intersection:')
f=[r for r in rows if r[6] and r[7]]
for r in f[:80]:
 print(f\"{r[0]:3d} j={r[1]:.3f} corr={r[3]:.3f} {r[4]:35s} cond={r[6]} cell={r[7]} :: {r[5][:160]}\")
PY"
