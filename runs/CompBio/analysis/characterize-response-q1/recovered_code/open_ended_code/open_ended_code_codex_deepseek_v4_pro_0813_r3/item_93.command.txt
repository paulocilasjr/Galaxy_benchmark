/bin/bash -c "python3 - <<'PY'
import glob, os, re, time
query=[g.strip().upper() for g in open('characterize.response.q1.txt')]
qset=set(query)
# Parse each library streaming, keep top by Jaccard
rows=[]
for path in glob.glob('all_libs/*.gmt'):
 lib=os.path.basename(path)[:-4]
 for line in open(path, errors='ignore'):
  p=line.rstrip('\\n').split('\\t')
  if len(p)<3: continue
  name=p[0]
  genes=[]
  for x in p[2:]:
   if x:
    g=x.split(',')[0].upper().strip()
    if g: genes.append(g)
  gset=set(genes)
  if not gset: continue
  inter=qset & gset
  if not inter: continue
  j=len(inter)/len(qset|gset)
  oc=len(inter)/len(qset)
  # overlap of term
  ot=len(inter)/len(gset)
  rows.append((j, oc, ot, len(inter), len(genes), lib, name, inter))
# sort by jaccard desc, overlap coeff desc, term coverage
rows.sort(key=lambda x:(-x[0],-x[1],-x[3]))
print('top 200 exact-ish:')
for r in rows[:200]:
 print(f\"j={r[0]:.3f} oc={r[1]:.3f} ot={r[2]:.3f} inter={r[3]}/{len(query)} term={r[4]} lib={r[5]:40s} name={r[6][:150]}\")
print('\\nterms with j>0.4 or oc>0.5:')
for r in rows:
 if r[0]>0.4 or r[1]>0.5:
  print(r)
PY"
