/bin/bash -lc "python3 - <<'PY'
import json
from scipy.stats import spearmanr
mp=json.load(open('mouse_to_human3.json'))
q=list(dict.fromkeys(mp.values())); qset=set(q); qmap={g:i for i,g in enumerate(q)}
terms=[]
for line in open('c7.immunesigdb.v2024.1.Hs.symbols.gmt'):
 p=line.rstrip('\\n').split('\\t')
 if len(p)<3: continue
 name,desc,genes=p[0],p[1],p[2:]
 s=set(genes); ov=qset&s
 if not ov: continue
 seen=set(); og=[]
 for g in genes:
  if g in qset and g not in seen: seen.add(g); og.append(g)
 rho=None
 if len(og)>=3:
  rho,_=spearmanr([qmap[g] for g in og],list(range(len(og))))
 jac=len(ov)/(len(s)+112-len(ov))
 terms.append((len(ov),jac,rho,name,desc,og,s))
# keyword filter condition/cell
keys=['LPS','DC','DENDRITIC','MONO','NEUTRO','CD8','EXHAUST','IL2','SLE','LUPUS','AGE','RHEUM','SEPSIS','LIF','COVID','MACROPHAGE','MYELOID','BMDC','BMDM']
for kw in keys:
 print('\\n###',kw)
 hits=[t for t in terms if kw.lower() in (t[3]+' '+t[4]).lower()]
 hits.sort(key=lambda t:(-t[0],-t[1]))
 for x,jac,rho,name,desc,og,s in hits[:25]:
  print(f'{x:2d} jac={jac:.3f} rho={rho if rho is None else round(rho,2)} term={name}')
PY"
