/bin/bash -lc "python3 - <<'PY'
import geo_de, json, re
# patch groups function dynamically
def groups_from_titles(titles):
    groups={}
    for i,t in enumerate(titles):
        g=t.split(' [')[0].strip()
        g=re.sub(r'-\\d+"'$'"','',g)
        groups.setdefault(g,[]).append(i)
    return groups
geo_de.groups_from_titles=groups_from_titles
mp=json.load(open('mouse_to_human3.json'))
q=list(dict.fromkeys(mp.values()))
path='GSE22886-GPL96_series_matrix.txt'
meta, ids, titles, mat=geo_de.parse_geo(path)
groups=geo_de.groups_from_titles(titles)
print(groups.keys())
gA='DendriticCell-Control'; gB='DendriticCell-LPSstimulated'
res=geo_de.de_contrast(mat, groups[gB], groups[gA])
probes=list(mat.index)
hmap=geo_de.get_human_map(probes)
print('mapped',len(hmap),'/',len(probes))
agg=geo_de.aggregate_gene(res,hmap,set(q))
df,rho,p,rho_abs,p2=geo_de.compare_query_order(q,agg)
print('genes found',len(df),'rho logFC',rho,'rho_abs',rho_abs)
print(df.sort_values('qrank')[['gene','logFC','t','qrank']].to_string(index=False))
print('top20')
print(agg.nlargest(20,'logFC')[['gene','logFC','t']].to_string(index=False))
print('bottom20')
print(agg.nsmallest(20,'logFC')[['gene','logFC','t']].to_string(index=False))
PY"
