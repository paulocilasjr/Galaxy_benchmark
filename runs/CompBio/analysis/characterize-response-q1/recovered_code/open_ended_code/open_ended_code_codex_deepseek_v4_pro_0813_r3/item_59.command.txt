/bin/bash -c "python3 - <<'PY'
import requests, gzip
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/matrix/GSE22886_series_matrix.txt.gz'
r=requests.get(url,timeout=120)
print(r.status_code,len(r.content),r.url)
if r.status_code==200:
 open('GSE22886_series_matrix.txt.gz','wb').write(r.content)
 with gzip.open('GSE22886_series_matrix.txt.gz','rt') as f, open('GSE22886_series_matrix.txt','w') as out: out.write(f.read())
 print('saved')
PY
head -5 GSE22886_series_matrix.txt | cut -c1-500"
