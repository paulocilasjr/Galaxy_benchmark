/bin/bash -c "python3 - <<'PY'
# quick recompute with mapping and rank our genes in GSE2002 sorted abs logfc up and down
import numpy as np, mygene
lines=open('GSE2002_series_matrix.txt').read().splitlines()
for i,l in enumerate(lines):
 if l.startswith('"'!series_matrix_table_begin'"'): start=i+1
 if l.startswith('"'!series_matrix_table_end'"'): end=i
header=lines[start].split('\\t'); data=[l.split('\\t') for l in lines[start+1:end]]
cols={h.strip('\"'):j for j,h in enumerate(header)}
cc=[cols['GSM35899'],cols['GSM35900'],cols['GSM35901']]; lc=[cols['GSM35902'],cols['GSM35903'],cols['GSM35904']]
probes=[]; logfc=[]
for row in data:
 p=row[0].strip('\"'); probes.append(p)
 try:
  c=np.mean([float(row[j]) for j in cc]); l=np.mean([float(row[j]) for j in lc])
 except: logfc.append(np.nan); continue
 logfc.append(np.log2((l+1)/(c+1)))
logfc=np.array(logfc)
mg=mygene.MyGeneInfo(); res=mg.querymany(probes, scopes='reporter', species='mouse', fields='symbol', returnall=True, verbose=False)
sym={o['query']:o.get('symbol') for o in res['out'] if o.get('symbol')}
genes=open('characterize.response.q1.txt').read().split()
# build ranks for positive logfc only and negative only
pos=np.argsort(-logfc); neg=np.argsort(logfc)
pos_rank={probes[i]:r+1 for r,i in enumerate(pos) if logfc[i]>0}
neg_rank={probes[i]:r+1 for r,i in enumerate(neg) if logfc[i]<0}
# map symbol to probes (first)
probe_by_sym={}
for p,s in sym.items(): probe_by_sym.setdefault(s,p)
for g in genes:
 p=probe_by_sym.get(g)
 if not p: print(g,'not mapped'); continue
 print(g,'logfc',f'{logfc[probes.index(p)]:+.2f}','pos_rank',pos_rank.get(p),'neg_rank',neg_rank.get(p))
PY"
