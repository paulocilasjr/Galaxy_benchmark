import os, glob, math, json
q=[l.strip() for l in open('characterize.response.q1.txt') if l.strip()]
qs=set(q)
# map human aliases for some query to enrich direct? Direct only for mouse libs.
rows=[]
files=glob.glob('all_libs/*.gmt')
for f in files:
    lib=os.path.basename(f)[:-4]
    with open(f, errors='ignore') as fh:
      for line in fh:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<3: continue
        term=p[0]
        # sometimes desc included? GMT standard term desc genes...
        genes=[g for g in p[1:] if g] # Enrichr GMT perhaps p[1] desc and p[2:] genes? inspect later
        # Actually files maybe term, gene list (some are 2-col). handle desc by gene length?
        s=set(genes)
        ov=qs & s
        if not ov: continue
        # If p[1] looks description but not gene, it will be counted as gene unlikely in q; harmless except desc could match? no.
        # rank positions
        pos=[q.index(g) for g in ov if g in q]
        mean_pos=sum(pos)/len(pos)
        # enrichment score rank correlation
        rows.append((len(ov), -mean_pos, lib, term, sorted(pos), len(genes)))
rows.sort(reverse=True)
print('top 120 by overlap')
for x,negmean,lib,term,pos,ntot in rows[:120]:
    print(f'{x:2d}/{ntot:<5d} meanpos={-negmean:6.1f} {lib:40s} | {term}')
print('\\nTOP by rank mean pos among overlap>=8')
f=[r for r in rows if r[0]>=8]
f.sort(key=lambda r:r[1])
for x,negmean,lib,term,pos,ntot in f[:100]:
    print(f'{x:2d}/{ntot:<5d} meanpos={-negmean:6.1f} {lib:40s} | {term}')
PY
