import requests,re,json

 print('\\n',name)
 # rough regex
 for m in re.finditer(r'\\{(?:["'^{}]|'"\\{["'^{}]*'"\\})*\\}', r):
  s=m.group(0)
  if 'tooltip' in s or 'title' in s:
   pass
# instead search for keywords
for kw in ['LPS','Med12','aging','Age','Sepsis','exhaust','LIF','IL39','SLE','lupus','rheumatoid','COVID','BMS','aldesleukin']:
 hits=[m.start() for m in re.finditer(kw, r, re.I)]
 if hits:
  print(kw,len(hits),'first',hits[:5])
  for h in hits[:2]: print(r[max(0,h-200):h+400].replace('\\n',' '))
PY"
