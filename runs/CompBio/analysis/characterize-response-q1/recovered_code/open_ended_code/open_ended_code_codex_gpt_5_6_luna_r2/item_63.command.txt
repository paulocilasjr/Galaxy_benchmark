/bin/bash -lc "python - <<'PY'
import io, urllib.request, openpyxl
url='https://media.springernature.com/original/springer-static/esm/art%3A10.1038%2Fs41586-023-06816-9/MediaObjects/41586_2023_6816_MOESM5_ESM.xlsx'
data=urllib.request.urlopen(url).read()
print('bytes',len(data))
wb=openpyxl.load_workbook(io.BytesIO(data),read_only=True,data_only=True)
print('sheets',wb.sheetnames)
for ws in wb.worksheets:
    print(ws.title,ws.max_row,ws.max_column)
    rows=ws.iter_rows(values_only=True)
    for idx,row in enumerate(rows):
        vals=[str(x) if x is not None else '' for x in row]
        if any('LIF' in x.upper() for x in vals):
            print('LIF row',idx+1,vals[:12])
            break
PY"
