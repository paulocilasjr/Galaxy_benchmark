/bin/bash -lc "python - <<'PY'
import io, urllib.request, zipfile, xml.etree.ElementTree as ET, re
url='https://media.springernature.com/original/springer-static/esm/art%3A10.1038%2Fs41586-023-06816-9/MediaObjects/41586_2023_6816_MOESM5_ESM.xlsx'
z=zipfile.ZipFile(io.BytesIO(urllib.request.urlopen(url).read()))
ns={'m':'http://schemas.openxmlformats.org/spreadsheetml/2006/main','r':'http://schemas.openxmlformats.org/officeDocument/2006/relationships','pr':'http://schemas.openxmlformats.org/package/2006/relationships'}
wb=ET.fromstring(z.read('xl/workbook.xml'))
rels=ET.fromstring(z.read('xl/_rels/workbook.xml.rels'))
relmap={x.attrib['Id']:x.attrib['Target'] for x in rels}
for sh in wb.find('m:sheets',ns):
    print(sh.attrib['name'], sh.attrib['{'+ns['r']+'}id'], relmap[sh.attrib['{'+ns['r']+'}id']])
PY"
