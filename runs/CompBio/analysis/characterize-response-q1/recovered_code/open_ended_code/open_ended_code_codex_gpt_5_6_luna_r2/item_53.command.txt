/bin/bash -lc "python3 - <<'PY'
import concurrent.futures, re, urllib.request
q = {x.strip() for x in open('characterize.response.q1.txt') if x.strip()}
html = urllib.request.urlopen('https://www.gsea-msigdb.org/gsea/msigdb/mouse/genesets.jsp?collection=M7').read().decode('latin1')
names = sorted(set(re.findall(r'CUI_[A-Z0-9_]*', html)))
names = [n for n in names if '_LIF_' in n]
pat = re.compile(r'<tr["'^>]*>'"\\s*<td valign=\"top\">(["'^<]+)</td>'"')
def get(n):
    try:
        s=urllib.request.urlopen(f'https://www.gsea-msigdb.org/gsea/msigdb/mouse/geneset/{n}.html', timeout=30).read().decode('latin1')
        g=set(pat.findall(s))
        return n, len(g), len(q&g), sorted(q&g), None
    except Exception as e:
        return n,0,0,[],repr(e)
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as ex:
    rows=list(ex.map(get,names))
for n,ng,no,ov,e in sorted(rows,key=lambda x:(-x[2],-x[1],x[0])):
    print(n, 'set',ng,'overlap',no, ' '.join(ov), e or '')
PY"
