/bin/bash -lc "python3 - <<'PY'
import requests, json, time, re, csv
stats=requests.get('https://maayanlab.cloud/Enrichr/datasetStatistics',timeout=30).json()['statistics']
libs=[s['libraryName'] for s in stats]
print(len(libs))
# save names
open('enrichr_libraries.txt','w').write('\\n'.join(libs))
# Filter likely libraries containing GEO/Disease/Perturb/Signature/Cell/Immune
pat=re.compile(r'GEO|Disease|Perturb|Signature|Cell|Immune|MSigDB|Human|Mouse|Aging|Drug', re.I)
selected=[l for l in libs if pat.search(l)]
print(len(selected))
open('enrichr_selected_libraries.txt','w').write('\\n'.join(selected))
print('\\n'.join(selected[:200]))
PY"
