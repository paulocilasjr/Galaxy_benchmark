/bin/bash -lc "Rscript - <<'RS'
library(msigdbr)
# check collections and species
m <- msigdbr(species='Mus musculus', category='C7')
cat('rows', nrow(m), 'sets', length(unique(m"'$gs_name)), '"'\\n')
# save all gene sets
write.csv(m, 'msigdb_mouse_c7.csv', row.names=FALSE)
h <- msigdbr(species='Homo sapiens', category='C7')
write.csv(h, 'msigdb_human_c7.csv', row.names=FALSE)
cat('human rows',nrow(h),'sets',length(unique(h"'$gs_name)),'"'\\n')
RS
ls -lh msigdb*"
