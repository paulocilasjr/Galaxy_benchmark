/bin/bash -c "Rscript - <<'RS'
suppressMessages(library(msigdbr))
genes <- toupper(readLines('characterize.response.q1.txt'))
m <- msigdbr(species='Mus musculus', collection='C7')
sets <- split(m"'$gene_symbol, m$gs_name)
keys <- c('"'LPS','SEPSIS','LIF','IL39','IL-39','COVID','SARS','MED12','LUPUS','SLE','BMS','ALDESLEUKIN','IL2','IL-2','AGING','AGE','RHEUMATOID','ARTHRITIS','EXHAUST','MONOCYTE','CD8','CD4','B_CELL','BCELL','DENDRITIC','NK_CELL','NEUTROPHIL')
# collect sets with key
idx <- unique(unlist(lapply(keys, function(k) grep(k, names(sets), ignore.case=TRUE))))
cat('matching sets', length(idx), '\\n')
for(i in idx){
 s<-toupper(sets[[i]])
 inter<-intersect(genes,s)
 if(length(inter)>=8){
  score<-sum(1/(seq_along(genes)[genes %in% s]))
  cat(sprintf('%3d score %.2f | %s\\n',length(inter),score,names(sets)[i]))
  cat('   ',paste(inter,collapse=','),'\\n')
 }
}
RS"
