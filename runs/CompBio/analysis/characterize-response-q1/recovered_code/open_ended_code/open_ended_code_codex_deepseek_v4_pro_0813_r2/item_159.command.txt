/bin/bash -lc "Rscript - <<'RS'
library(data.table)
library(moe430a.db)
expr <- fread('GSE14769_expr.csv')
probes <- gsub('\"','',expr[[1]])
map <- as.data.table(select(moe430a.db, keys=probes, columns='SYMBOL', keytype='PROBEID'))
setnames(map,'PROBEID','probe')
expr[, probe := probes]
numcols <- grep('"'^""GSM'"', names(expr), value=TRUE)
for(j in numcols) set(expr, j=j, value=as.numeric(expr[[j]]))
m <- merge(expr, map, by='probe', allow.cartesian=TRUE)
m <- m["'!is.na(SYMBOL)]
m[, SYMBOL := toupper(SYMBOL)]
out <- m[, lapply(.SD, mean), by=SYMBOL, .SDcols=numcols]
setnames(out, names(out), gsub('"'\"','',names(out)))
fwrite(out, 'GSE14769_symbol_expr.csv')
cat('symbols',nrow(out),'cols',ncol(out),'\\n')
print(names(out)[1:5])
RS
python3 - <<'PY'
import gzip,json,csv
with gzip.open('GSE14769_series_matrix.txt.gz','rt') as f: lines=f.read().splitlines()
meta={}
for l in lines:
 if l.startswith('"'!Sample_title'"'):
  titles=[x.strip('\"') for x in l.split('\\t')[1:]]
 if l.startswith('"'!Sample_geo_accession'"'):
  gsms=[x.strip('\"') for x in l.split('\\t')[1:]]
for g,t in zip(gsms,titles): meta[g]=t
json.dump(meta,open('GSE14769_meta.json','w'))
print(list(meta.items())[:5])
PY"
