/bin/bash -lc "python3 - <<'PY'
import json, re
input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
ri={g:i+1 for i,g in enumerate(input)}
results=[]
for fn in ['creeds_disease.json','creeds_gene.json']:
 data=json.load(open(fn))
 print('processing',fn,len(data))
 for rec in data:
  for direction in ['up_genes','down_genes']:
   arr=rec.get(direction,[])
   genes=[g.upper() for g,v in arr]
   # exact prefix overlap length
   m=0
   for g in genes:
    if m<len(input) and g==input[m]: m+=1
    else: break
   if m>=5:
    results.append((m,fn,rec.get('id'),rec.get('geo_id'),rec.get('cell_type'),rec.get('do_id'),rec.get('umls_cui'),direction,rec.get('mm_gene_symbol'),rec.get('pert_type'),len(genes)))
# also full jaccard/weighted for top
rows=[]
for rec in (json.load(open('creeds_disease.json'))+json.load(open('creeds_gene.json'))):
 for d in ['up_genes','down_genes']:
  arr=rec.get(d,[])
  genes=[g.upper() for g,v in arr]
  s=set(genes); ov=s&set(input); jac=len(ov)/len(s|set(input))
  if len(ov)>=20:
   rows.append((len(ov),jac,rec.get('id'),rec.get('geo_id'),rec.get('cell_type'),rec.get('do_id'),rec.get('umls_cui'),d,rec.get('mm_gene_symbol'),rec.get('pert_type'),len(genes)))
rows.sort(reverse=True)
print('\\nTop by overlap>=20:')
for r in rows[:50]: print(r)
print('\\nTop exact prefix:')
for r in sorted(results,reverse=True)[:50]: print(r)
PY"
