/bin/bash -c "Rscript - <<'RS'
suppressMessages(library(moe430a.db))
lines <- readLines(gzfile('GSE2002_series_matrix.txt.gz'))
start <- which(lines=='"'!series_matrix_table_begin'"')+1; end <- which(lines=='"'!series_matrix_table_end'"')-1
con <- textConnection(lines[start:end]); tab <- read.delim(con, header=TRUE, row.names=1, check.names=FALSE, stringsAsFactors=FALSE, quote=''); close(con)
colnames(tab)<-gsub('\"','',colnames(tab)); rownames(tab)<-gsub('\"','',rownames(tab)); expr<-as.matrix(tab); mode(expr)<-'numeric'
unt <- rowMeans(expr[,1:3]); lps <- rowMeans(expr[,4:6]); logfc <- log2((lps+0.5)/(unt+0.5))
sym <- mapIds(moe430a.db, keys=rownames(expr), column='SYMBOL', keytype='PROBEID', multiVals='first')
df<-data.frame(sym=as.character(sym), logfc=as.numeric(logfc), stringsAsFactors=FALSE); df<-df["'!is.na(df$sym)&df$sym!='"'NA'&"'!grepl('"'/',df"'$sym),]
agg<-aggregate(logfc~sym, data=df, FUN=mean); agg<-agg[order(-agg$logfc),]
our<-toupper(readLines('"'characterize.response.q1.txt')); m<-setNames(agg"'$logfc,toupper(agg$sym)); found<-our%in%names(m)
cat('"'found',sum(found),'\\n'); for(i in seq_along(our)) if(found[i]) cat(our[i],m[[our[i]]],'\\n')
cat('overlap top up112',sum(our %in% toupper(head(agg"'$sym,112))),'"'down112',sum(our %in% toupper(tail(agg"'$sym,112))),'"'\\n')
print(head(agg,150)); print(tail(agg,150))
RS"
