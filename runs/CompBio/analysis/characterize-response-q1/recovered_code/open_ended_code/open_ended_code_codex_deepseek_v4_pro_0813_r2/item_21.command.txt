/bin/bash -c "python3 - <<'PY'
import requests, json
libs=['Disease_Perturbations_from_GEO_up','Disease_Perturbations_from_GEO_down','Ligand_Perturbations_from_GEO_up','Ligand_Perturbations_from_GEO_down','Gene_Perturbations_from_GEO_up','Gene_Perturbations_from_GEO_down']
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
gs=set(genes)
keys=['lipopolysaccharide','LPS','endotoxin','exhaust','LIF','IL-39','IL39','COVID','SARS-CoV','coronavirus','sepsis','septic','Med12','MED12','lupus','SLE','BMS-34554','BMS 34554','BMS','aldesleukin','interleukin-2','IL-2','aging','aged','senesc','rheumatoid','arthritis','IL-4','IL-6','IL-10','IL-15','IL-21']
for lib in libs:
 r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
 terms=r.json()[lib]['terms']
 rows=[]
 for term,vals in terms.items():
  if any(k.lower() in term.lower() for k in keys):
   s=set(v.upper() for v in vals)
   inter=gs & s
   if len(inter)>=3:
    score=sum(1/(i+1) for i,g in enumerate(genes) if g in s)
    rows.append((len(inter),score,term,[g for g in genes if g in s]))
 rows.sort(key=lambda x:x[1], reverse=True)
 print('\\n###',lib,'matches',len(rows))
 for row in rows[:25]: print(row[0],round(row[1],2),row[2],','.join(row[3][:25]))
PY"
