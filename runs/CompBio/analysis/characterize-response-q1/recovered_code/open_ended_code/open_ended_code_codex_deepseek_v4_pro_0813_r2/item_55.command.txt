/bin/bash -c "python3 - <<'PY'
import requests, os
# get series matrix file URL
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE4nnn/GSE4479/matrix/GSE4479_series_matrix.txt.gz'
r=requests.get(url,timeout=60)
print(r.status_code, len(r.content), r.url)
if r.status_code==200:
 open('GSE4479_series_matrix.txt.gz','wb').write(r.content)
 os.system('gzip -dc GSE4479_series_matrix.txt.gz | head -40')
PY"
