/bin/bash -lc "python3 - <<'PY'
import requests, json, math
uid='135528001'
libs=['RNA-Seq_Disease_Gene_and_Drug_Signatures_from_GEO','Disease_Perturbations_from_GEO_up','Disease_Perturbations_from_GEO_down','Drug_Perturbations_from_GEO_up','Drug_Perturbations_from_GEO_down','Ligand_Perturbations_from_GEO_up','Ligand_Perturbations_from_GEO_down','Aging_Perturbations_from_GEO_up','Aging_Perturbations_from_GEO_down','Gene_Perturbations_from_GEO_up','Gene_Perturbations_from_GEO_down','Microbe_Perturbations_from_GEO_up','Microbe_Perturbations_from_GEO_down']
input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
pos={g:i+1 for i,g in enumerate(input)}
for lib in libs:
 r=requests.get(f'https://maayanlab.cloud/Enrichr/enrich?userListId={uid}&backgroundType={lib}',timeout=120)
 terms=r.json().get(lib,[])
 rows=[]
 for t in terms:
  name=t[1]; ov=t[5]
  # weighted reciprocal rank sum; only overlapping genes
  score=sum(1/pos[g] for g in ov)
  rows.append((score,len(ov),float(t[2]),name,ov))
 rows.sort(key=lambda x:(-x[0],-x[1]))
 print('\\n',lib)
 for score,n,p,name,ov in rows[:15]:
  print(f'{score:.3f} {n:2} p={p:.1e} {name} {ov[:15]}')
PY"
