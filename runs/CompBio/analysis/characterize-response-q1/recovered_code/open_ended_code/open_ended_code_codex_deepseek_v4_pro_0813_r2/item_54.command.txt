/bin/bash -c "Rscript - <<'RS'
suppressMessages(library(msigdbr))
genes <- toupper(readLines('characterize.response.q1.txt'))
m <- msigdbr(species='Mus musculus', collection='C7')
sets <- split(m"'$gene_symbol, m$gs_name)
keys <- c('"'EXHAUST','DYSFUNCTION','PD1','PD-1','LAG3','TIM3','TOX','TEX')
idx <- unique(unlist(lapply(keys, function(k) grep(k, names(sets), ignore.case=TRUE))))
for(i in idx){ s<-toupper(sets[[i]]); inter<-intersect(genes,s); if(length(inter)>=5) cat(sprintf('%d | %s\\n  %s\\n',length(inter),names(sets)[i],paste(inter,collapse=',')))}
RS"
