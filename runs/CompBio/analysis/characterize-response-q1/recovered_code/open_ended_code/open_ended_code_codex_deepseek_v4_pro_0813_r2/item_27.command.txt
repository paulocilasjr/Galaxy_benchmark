/bin/bash -c "python3 - <<'PY'
import requests, json
lib='RNA-Seq_Disease_Gene_and_Drug_Signatures_from_GEO'
r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
terms=r.json()[lib]['terms']
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
N=len(genes)
rank={g:i+1 for i,g in enumerate(genes)}
results=[]
for term,vals0 in terms.items():
 vals=[v.upper() for v in vals0]
 # positional score over first N
 prefix=0
 for j,v in enumerate(vals[:N]):
  if v in rank: prefix += 1/(abs(j+1-rank[v])+1)
 # Also check exact first N intersection
 inter=set(vals[:N])&set(genes)
 # if our genes are subset of vals first N? count
 subset=sum(1 for g in genes if g in set(vals[:N]))
 # exact first 10/20/50
 c10=sum(1 for j,v in enumerate(vals[:10]) if v in rank and j==rank[v]-1)
 c20=sum(1 for j,v in enumerate(vals[:20]) if v in rank and j==rank[v]-1)
 c50=sum(1 for j,v in enumerate(vals[:50]) if v in rank and j==rank[v]-1)
 # average index of our genes in full vals (top rank position)
 present=[g for g in genes if g in vals]
 avg=sum(vals.index(g)+1 for g in present)/len(present) if present else 9999
 results.append((prefix,subset,inter.__len__(),c10,c20,c50,avg,len(vals),term))
results.sort(key=lambda x:x[0], reverse=True)
print('top positional in RNA-seq lib')
for row in results[:40]:
 print(round(row[0],2),'subset_firstN',row[1],'ov',row[2],'exact10/20/50',row[3],row[4],row[5],'avg',round(row[6],1),'len',row[7],row[8])
PY"
