/bin/bash -lc "python3 - <<'PY'
import requests, urllib.parse
queries=[
 '\"Ifitm1\" AND \"Tspo\" AND \"Lgmn\" AND \"Mafb\"',
 'Ifitm1 Tspo Lgmn Mafb Ifitm2 Fabp5',
 'Ifitm1 Tspo Lgmn Mafb macrophage',
 'Ifitm1 Clec4n Wfdc17 Mcemp1',
]
for q in queries:
    u='https://www.ebi.ac.uk/europepmc/webservices/rest/search?query='+urllib.parse.quote(q)+'&format=json&pageSize=5'
    r=requests.get(u,timeout=30)
    try:
      data=r.json()
      print('\\nQUERY',q,'hits',data.get('hitCount'))
      for x in data.get('resultList',{}).get('result',[])[:5]: print(x.get('id'),x.get('source'),x.get('title'))
    except Exception as e: print('ERR',e,r.text[:200])
PY"
