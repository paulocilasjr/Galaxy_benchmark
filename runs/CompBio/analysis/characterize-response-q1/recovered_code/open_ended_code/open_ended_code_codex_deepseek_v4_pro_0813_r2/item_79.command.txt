/bin/bash -c "python3 - <<'PY'
import requests, json
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils'
queries=[
 'aging[All Fields] AND monocyte[All Fields] AND Mus musculus[Organism] AND expression profiling by high throughput sequencing[DataSet Type]',
 'aging[All Fields] AND monocytes[All Fields] AND Mus musculus[Organism]',
 'old[Title] AND monocyte[Title] AND Mus musculus[Organism]',
 'young[Title] AND aged[Title] AND monocyte[Title] AND Mus musculus[Organism]',
 'age[Title] AND monocyte[Title] AND Mus musculus[Organism]',
 'age[Title] AND macrophage[Title] AND Mus musculus[Organism]',
 'age[Title] AND dendritic cells[Title] AND Mus musculus[Organism]',
]
for q in queries:
 r=requests.get(base+'/esearch.fcgi',params={'db':'gds','term':q,'retmax':50,'retmode':'json'},timeout=20)
 js=r.json().get('esearchresult',{})
 print('\\nQ',q,'count',js.get('count'))
 ids=js.get('idlist',[])[:20]
 if ids:
  r2=requests.get(base+'/esummary.fcgi',params={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=20)
  res=r2.json().get('result',{})
  for i in ids:
   it=res.get(i,{})
   print(i,'|',it.get('accession'),'|',it.get('title'),'|',(it.get('summary') or '')[:160])
PY"
