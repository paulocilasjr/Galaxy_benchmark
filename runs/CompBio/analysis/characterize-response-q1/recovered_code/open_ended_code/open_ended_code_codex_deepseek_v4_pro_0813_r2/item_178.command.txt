/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.stats as st
# file is tab, first col ID. read
raw=pd.read_csv('GSE139913_norm_data.txt.gz',sep='\\t',index_col=0)
# normalize maybe log2(x+1)
expr=np.log2(raw+1)
# groups columns numbers
sepsis=[str(x) for x in [1070,1071,1072,1074]]; non=[str(x) for x in [1075,1076,1077,1079]]; healthy=[str(x) for x in [1080,1081,1082,1083,1084]]
# some column names maybe strings
input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
ri={g:i+1 for i,g in enumerate(input)}
for name, a, b in [('sepsis_vs_healthy',sepsis,healthy),('sepsis_vs_nonsepsis',sepsis,non)]:
    s=expr[a].mean(axis=1)-expr[b].mean(axis=1)
    # index uppercase
    s.index=s.index.str.upper()
    # collapse duplicate gene names (already)
    got=s[s.index.isin(set(input))]
    print(name,'mapped',len(got))
    aa=[ri[g] for g in input if g in s.index]; bb=[s[g] for g in input if g in s.index]
    print('spearman',st.spearmanr(aa,bb))
    ranked=s.sort_values(ascending=False); top=ranked.head(112).index.tolist()
    print('overlap top112',len(set(top)&set(input)), [g for g in input if g in set(top)][:50])
    print('top input values', [(g,round(s.get(g,float('nan')),2)) for g in input[:30] if g in s.index])
PY"
