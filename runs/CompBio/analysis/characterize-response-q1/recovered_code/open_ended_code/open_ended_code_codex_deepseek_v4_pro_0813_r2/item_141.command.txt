/bin/bash -lc "python3 - <<'PY'
import requests, xml.etree.ElementTree as ET
ids='200287785,200287517,200287516,200277567,200228934,200216991,200178953,200213043,200137429,200155855,200139944,200024945'
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=gds&id='+ids
r=requests.get(url,timeout=30)
root=ET.fromstring(r.text)
for doc in root.findall('.//DocSum'):
    uid=doc.find('Id').text
    title=doc.find(\".//Item[@Name='title']\").text if doc.find(\".//Item[@Name='title']\") is not None else ''
    acc=doc.find(\".//Item[@Name='GDS']\") or doc.find(\".//Item[@Name='Accession']\")
    acc=acc.text if acc is not None else ''
    print(uid, acc, title[:200])
PY"
