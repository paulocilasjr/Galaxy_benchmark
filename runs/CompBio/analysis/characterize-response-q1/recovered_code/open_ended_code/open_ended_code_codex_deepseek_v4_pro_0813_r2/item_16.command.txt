/bin/bash -c "python3 - <<'PY'
import requests, json
lib='Disease_Perturbations_from_GEO_up'
r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
data=r.json()[lib]['terms']
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
gs=set(genes)
for term,vals in data.items():
    if any(k in term for k in ['Sepsis','sepsis','Lupus','lupus','COVID','SARS','rheumatoid','Med12','BMS','aldesleukin','IL-39','LIF','aging','Acute Lung','GSE4479','GSE36700','GSE1871']):
        s=set(v.upper() for v in vals)
        inter=gs & s
        if len(inter)>=3:
            print(len(inter), term, ','.join([g for g in genes if g in s][:40]))
PY"
