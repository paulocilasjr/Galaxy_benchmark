/bin/bash -c "python3 - <<'PY'
import requests
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
gs=set(genes)
libs=['Drug_Perturbations_from_GEO_up','Drug_Perturbations_from_GEO_down','LINCS_L1000_Chem_Pert_up','LINCS_L1000_Chem_Pert_down','LINCS_L1000_Ligand_Perturbations_up','LINCS_L1000_Ligand_Perturbations_down','DrugMatrix','DSigDB','HDSigDB_Human_2021','HDSigDB_Mouse_2021','NIBR_DRUGseq_2025_up','NIBR_DRUGseq_2025_down','Sciplex_Drug_Perturbation_Signatures_2025']
keys=['BMS','aldesleukin','IL-2','interleukin-2','proleukin','LIF','leukemia inhibitory factor','IL-39','lipopolysaccharide','LPS','BMS-34554']
for lib in libs:
 try:
  r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
  if r.status_code"'!=200: print('"'bad',lib,r.status_code); continue
  terms=r.json()[lib]['terms']
 except Exception as e: print('err',lib,e); continue
 rows=[]
 for term,vals in terms.items():
  if any(k.lower() in term.lower() for k in keys):
   s=set(v.upper() for v in vals)
   inter=gs & s
   if len(inter)>=3:
    score=sum(1/(i+1) for i,g in enumerate(genes) if g in s)
    rows.append((len(inter),score,term,[g for g in genes if g in s]))
 rows.sort(key=lambda x:x[1], reverse=True)
 if rows:
  print('\\n###',lib,'matches',len(rows))
  for row in rows[:20]: print(row[0],round(row[1],2),row[2],','.join(row[3][:30]))
PY"
