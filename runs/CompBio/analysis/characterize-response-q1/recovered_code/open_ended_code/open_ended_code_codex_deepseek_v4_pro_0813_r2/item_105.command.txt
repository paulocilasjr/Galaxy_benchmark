/bin/bash -lc "python3 - <<'PY'
import csv, collections
genes=[x.strip().upper() for x in open('characterize.response.q1.txt')]
gset=set(genes)
pos={g:i for i,g in enumerate(genes)}
# read mouse c7, gene_symbol column
rows=csv.DictReader(open('msigdb_mouse_c7.csv'))
terms=collections.defaultdict(set)
for r in rows:
    terms[r['gs_name']].add(r['gene_symbol'].upper())
out=[]
for name, tg in terms.items():
    ov=sorted(gset & tg, key=lambda g:pos[g])
    # jaccard and rank sum
    jac=len(ov)/len(gset|tg)
    # weighted early overlap: sum 1/(rank+1) / sum max? 
    score=sum(1/(pos[g]+1) for g in ov)
    out.append((len(ov),jac,score,name,ov))
out.sort(key=lambda x:(-x[0],-x[2]))
print('top by count:')
for n,j,s,name,ov in out[:40]:
    print(f'{n:3} jac={j:.3f} score={s:.3f} {name} {ov[:20]}')
print('\\ntop by weighted score:')
out2=sorted(out,key=lambda x:-x[2])
for n,j,s,name,ov in out2[:60]:
    if n>=8:
        print(f'{n:3} jac={j:.3f} score={s:.3f} {name} {ov[:20]}')
PY"
