/bin/bash -c "python3 - <<'PY'
import requests, time
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils'
queries=[
 'Ifitm1[Gene Symbol] AND Tspo[Gene Symbol] AND Lgmn[Gene Symbol] AND Mafb[Gene Symbol] AND Ifitm2[Gene Symbol]',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" \"Mafb\"',
 'Clec4n Wfdc17 Cdk2ap2 Tgfbi',
]
for q in queries:
 r=requests.get(base+'/esearch.fcgi',params={'db':'gds','term':q,'retmax':20,'retmode':'json'},timeout=20)
 print(q, r.status_code, r.text[:1000])
 time.sleep(0.4)
PY"
