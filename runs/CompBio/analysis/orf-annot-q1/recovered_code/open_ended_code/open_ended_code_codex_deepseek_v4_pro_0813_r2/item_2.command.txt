/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
s=''.join(l.strip() for l in Path('orf.annot.q1.fa').read_text().splitlines()[1:]).upper()
code = {
'UUU':'F','UUC':'F','UUA':'L','UUG':'L','UCU':'S','UCC':'S','UCA':'S','UCG':'S','UAU':'Y','UAC':'Y','UAA':'*','UAG':'*','UGU':'C','UGC':'C','UGA':'*','UGG':'W',
'CUU':'L','CUC':'L','CUA':'L','CUG':'L','CCU':'P','CCC':'P','CCA':'P','CCG':'P','CAU':'H','CAC':'H','CAA':'Q','CAG':'Q','CGU':'R','CGC':'R','CGA':'R','CGG':'R',
'AUU':'I','AUC':'I','AUA':'I','AUG':'M','ACU':'T','ACC':'T','ACA':'T','ACG':'T','AAU':'N','AAC':'N','AAA':'K','AAG':'K','AGU':'S','AGC':'S','AGA':'R','AGG':'R',
'GUU':'V','GUC':'V','GUA':'V','GUG':'V','GCU':'A','GCC':'A','GCA':'A','GCG':'A','GAU':'D','GAC':'D','GAA':'E','GAG':'E','GGU':'G','GGC':'G','GGA':'G','GGG':'G'}
stops={'UAA','UAG','UGA'}
orfs=[]
for i in range(len(s)-2):
    if s[i:i+3]=='AUG':
        aa=[]
        j=i
        while j+3<=len(s):
            c=s[j:j+3]
            aa.append(code[c])
            if c in stops:
                break
            j+=3
        if aa and aa[-1]=='*':
            orfs.append((i,j+3,len(aa)-1,''.join(aa[:-1])))
orfs.sort(key=lambda x:(-x[2],x[0]))
for o in orfs[:20]:
    print(o[0], o[1], o[2], o[3][:50])
print('num',len(orfs))
# longest among all ATG
print('max',max(orfs,key=lambda x:x[2]))
# start of predicted: find first in longest, maybe upstream starts
# all ORFs near beginning
for o in orfs:
    if o[0]<300:
        print('early', o[0],o[2],o[1],o[3][:40])
PY"
