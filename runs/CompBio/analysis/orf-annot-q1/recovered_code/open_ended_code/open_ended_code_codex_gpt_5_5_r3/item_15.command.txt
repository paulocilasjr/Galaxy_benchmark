/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
from Bio.Blast import NCBIWWW, NCBIXML
seq=''.join(l.strip() for l in Path('orf.annot.q1.fa').read_text().splitlines() if not l.startswith('>')).upper().replace('U','T')
print('submitting blastn megablast refseq_rna len', len(seq), flush=True)
result = NCBIWWW.qblast('blastn','refseq_rna',seq, entrez_query='Homo sapiens[Organism]', megablast=True, hitlist_size=10, expect=1e-20, format_type='XML')
xml=result.read()
Path('blast_refseq_rna.xml').write_text(xml)
print('xml bytes', len(xml))
rec=NCBIXML.read(open('blast_refseq_rna.xml'))
for aln in rec.alignments[:5]:
    print('ALIGN', aln.title[:200], aln.length)
    for hsp in aln.hsps[:2]:
        print(' hsp', hsp.score, hsp.expect, hsp.query_start, hsp.query_end, hsp.sbjct_start, hsp.sbjct_end, hsp.identities, '/', hsp.align_length)
PY"
