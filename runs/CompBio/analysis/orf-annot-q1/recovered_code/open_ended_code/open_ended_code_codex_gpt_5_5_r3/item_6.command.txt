/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
seq=''.join(l.strip() for l in Path('orf.annot.q1.fa').read_text().splitlines() if not l.startswith('>')).upper().replace('T','U')
stops={'UAA','UAG','UGA'}
orfs=[]
for i in range(len(seq)-2):
    if seq[i:i+3]=='AUG':
        for j in range(i+3,len(seq)-2,3):
            if seq[j:j+3] in stops:
                orfs.append((i,j,(j-i)//3,seq[j:j+3],seq[max(0,i-10):i+10])) # aa excluding stop
                break
        else:
            orfs.append((i,None,(len(seq)-i)//3,None,seq[max(0,i-10):i+10]))
for o in sorted(orfs,key=lambda x:(x[2]), reverse=True):
    print(o)
PY"
