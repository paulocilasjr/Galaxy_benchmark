/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
s=''.join(l.strip() for l in Path('orf.annot.q1.fa').read_text().splitlines() if not l.startswith('>')).upper()
print('len',len(s))
stops={'UAA','UAG','UGA'}
# all ATG/AUG to first in-frame stop, by frame; track all ORFs
orfs=[]
for i in range(len(s)-2):
    if s[i:i+3]=='AUG':
        for j in range(i+3,len(s)-2,3):
            if s[j:j+3] in stops:
                aa=(j-i)//3
                orfs.append((aa,i,j,s[j:j+3],s[i-15:i+15]))
                break
        else:
            orfs.append(((len(s)-i)//3,i,None,None,s[i-15:i+15]))
print('num AUG orfs',len(orfs))
for x in sorted(orfs, reverse=True)[:20]:
    print(x[:4], x[4])
# longest open intervals between stops in each frame (not requiring start)
for frame in range(3):
    last=frame
    print('frame',frame)
    for j in range(frame,len(s)-2,3):
        if s[j:j+3] in stops:
            print(' segment',last,j,(j-last)//3,'stop',j,s[j:j+3])
            last=j+3
    print(' tail',last,len(s),(len(s)-last)//3)
PY"
