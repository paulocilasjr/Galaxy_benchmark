/bin/bash -c "python3 - <<'PY'
from pathlib import Path
seq=''.join(l.strip() for l in Path('orf.annot.q1.fa').read_text().splitlines() if not l.startswith('>'))
print(len(seq))
# codons
stops={'UAA','UAG','UGA'}
# find all ORFs starting AUG to first stop
orfs=[]
for i in range(len(seq)-2):
    if seq[i:i+3]=='AUG':
        aa=[]
        j=i
        while j+3<=len(seq):
            c=seq[j:j+3]
            if c in stops:
                aa.append('*')
                break
            # translate standard
            aa.append({'UUU':'F','UUC':'F','UUA':'L','UUG':'L','CUU':'L','CUC':'L','CUA':'L','CUG':'L','AUU':'I','AUC':'I','AUA':'I','AUG':'M','GUU':'V','GUC':'V','GUA':'V','GUG':'V','UCU':'S','UCC':'S','UCA':'S','UCG':'S','AGU':'S','AGC':'S','CCU':'P','CCC':'P','CCA':'P','CCG':'P','ACU':'T','ACC':'T','ACA':'T','ACG':'T','GCU':'A','GCC':'A','GCA':'A','GCG':'A','UAU':'Y','UAC':'Y','UAA':'*','UAG':'*','UGA':'*','CAU':'H','CAC':'H','CAA':'Q','CAG':'Q','AAU':'N','AAC':'N','AAA':'K','AAG':'K','GAU':'D','GAC':'D','GAA':'E','GAG':'E','UGU':'C','UGC':'C','UGG':'W','CGU':'R','CGC':'R','CGA':'R','CGG':'R','AGA':'R','AGG':'R','GGU':'G','GGC':'G','GGA':'G','GGG':'G'}[c])
            j+=3
        if aa and aa[-1]=='*':
            orfs.append((i,j, ''.join(aa)))
print('num start ORFs complete',len(orfs))
for i,end,prot in sorted(orfs,key=lambda x:(len(x[2])-1), reverse=True)[:20]:
    print(i, end, 'protlen', len(prot)-1, 'frame', i%3, 'aa head', prot[:30], 'tail', prot[-30:])
PY"
