/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict

peaks = defaultdict(list)
with gzip.open('ENCFF333TAT.bed.gz','rt') as f:
    for line in f:
        if not line.strip() or line.startswith('#'):
            continue
        fields = line.rstrip('\\n').split('\\t')
        summit = int(fields[1]) + int(fields[9])
        peaks[fields[0]].append((summit - 250, summit + 250))

merged = defaultdict(list)
for chrom, ivs in peaks.items():
    ivs.sort()
    for s, e in ivs:
        if not merged[chrom] or s > merged[chrom][-1][1]:
            merged[chrom].append([s, e])
        elif e > merged[chrom][-1][1]:
            merged[chrom][-1][1] = e

black = defaultdict(list)
with gzip.open('ENCFF356LFX.bed.gz','rt') as f:
    for line in f:
        if not line.strip() or line.startswith('#'):
            continue
        fields = line.rstrip('\\n').split('\\t')
        black[fields[0]].append((int(fields[1]), int(fields[2])))
for chrom in black:
    black[chrom].sort()

count = 0
bases = 0
for chrom, ivs in merged.items():
    b = black.get(chrom, [])
    j = 0
    for s, e in ivs:
        while j < len(b) and b[j][1] <= s:
            j += 1
        overlap = j < len(b) and b[j][0] < e and b[j][1] > s
        if not overlap:
            count += 1
            bases += e - s
print(count, bases)
PY"
