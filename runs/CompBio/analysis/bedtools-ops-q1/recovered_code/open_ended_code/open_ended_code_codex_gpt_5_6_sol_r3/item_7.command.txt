/bin/bash -lc "set -o pipefail
make_merged() { LC_ALL=C gzip -dc ENCFF333TAT.bed.gz | awk 'BEGIN{OFS=\"\\t\"} {s="'$2+$10; print $1,s-250,s+250}'"' | LC_ALL=C sort -k1,1 -k2,2n | bedtools merge -i -; }
printf 'all '
make_merged | awk '{n++;b+="'$3-$2} END{print n,b}'"'
printf 'kept '
make_merged | bedtools intersect -v -a stdin -b ENCFF356LFX.bed.gz | awk '{n++;b+="'$3-$2} END{print n,b}'"'
printf 'rejected '
make_merged | bedtools intersect -u -a stdin -b ENCFF356LFX.bed.gz | awk '{n++;b+="'$3-$2} END{print n,b}'"'"
