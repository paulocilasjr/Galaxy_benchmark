/bin/bash -lc "set -euo pipefail
mkdir -p work_intermediate
LC_ALL=C gzip -cd ENCFF333TAT.bed.gz \\
  | awk 'BEGIN{OFS=\"\\t\"} "'$10 >= 0 { summit=$2+$10; s=summit-250; e=summit+250; if (s<0) s=0; print $1,s,e }'"' \\
  | LC_ALL=C sort -k1,1 -k2,2n > work_intermediate/centered_500.sorted.bed
bedtools merge -i work_intermediate/centered_500.sorted.bed > work_intermediate/centered_500.merged.bed
bedtools intersect -v -a work_intermediate/centered_500.merged.bed -b ENCFF356LFX.bed.gz > work_intermediate/final.no_blacklist_overlap.bed
awk '{sum += "'$3-$2} END{printf "%d'"\\n\", sum}' work_intermediate/final.no_blacklist_overlap.bed
wc -l work_intermediate/centered_500.sorted.bed work_intermediate/centered_500.merged.bed work_intermediate/final.no_blacklist_overlap.bed"
