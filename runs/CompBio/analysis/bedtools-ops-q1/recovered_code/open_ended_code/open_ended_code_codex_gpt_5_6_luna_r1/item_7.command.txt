/bin/bash -lc "set -e
printf '%s\\n' '--- centered coordinate bounds ---'
awk 'BEGIN{FS=\"\\t\"; min=1e18; max=0; neg=0; bad=0} {if("'$2<min)min=$2; if($3>max)max=$3; if($2<0)neg++; if($3-$2!=500)bad++} END{print "min_start="min,"max_end="max,"negative_starts="neg,"non500="bad}'"' centered.bed
printf '%s\\n' '--- independent blacklist filter of merged ---'
awk 'BEGIN{FS=OFS=\"\\t\"} FNR==NR {bc[++n]="'$1; bs[n]=$2; be[n]=$3; next} {keep=1; for(i=1;i<=n;i++) if($1==bc[i] && $2<be[i] && bs[i]<$3) {keep=0; break} if(keep){cnt++; bases += $3-$2}} END{print "intervals="cnt,"bases="bases}'"' blacklist.sorted.bed merged.bed
printf '%s\\n' '--- sum using bedtools stats ---'
bedtools coverage -a kept.intersect-v.bed -b kept.intersect-v.bed | awk '{bases += "'$3-$2} END{print bases+0}'"'
printf '%s\\n' '--- sortedness ---'
awk 'BEGIN{FS=\"\\t\"; prevc=\"\"; prevs=-1; bad=0} {if("'$1==prevc && $2<prevs)bad++; prevc=$1; prevs=$2} END{print "bad_within_chrom="bad+0}'"' kept.intersect-v.bed"
