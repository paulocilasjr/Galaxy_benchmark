/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict

peaks=defaultdict(list)
with gzip.open('ENCFF333TAT.bed.gz','rt') as f:
    for line in f:
        if not line.strip() or line.startswith(('#','track','browser')):
            continue
        p=line.rstrip('\\n').split('\\t')
        summit=int(p[1])+int(p[9])
        peaks[p[0]].append((summit-250, summit+250))

merged=[]
for chrom, vals in peaks.items():
    vals.sort()
    cs=ce=None
    for s,e in vals:
        if cs is None:
            cs,ce=s,e
        elif s <= ce:
            if e > ce: ce=e
        else:
            merged.append((chrom,cs,ce))
            cs,ce=s,e
    if cs is not None:
        merged.append((chrom,cs,ce))

black=defaultdict(list)
with gzip.open('ENCFF356LFX.bed.gz','rt') as f:
    for line in f:
        if not line.strip() or line.startswith(('#','track','browser')):
            continue
        p=line.rstrip('\\n').split('\\t')
        black[p[0]].append((int(p[1]), int(p[2])))
for chrom in black:
    black[chrom].sort()

keep=[]
for chrom,s,e in merged:
    ov=False
    for bs,be in black.get(chrom,[]):
        if be <= s:
            continue
        if bs >= e:
            break
        ov=True
        break
    if not ov:
        keep.append((chrom,s,e))
print(len(merged), len(keep), sum(e-s for _,s,e in keep))
PY"
