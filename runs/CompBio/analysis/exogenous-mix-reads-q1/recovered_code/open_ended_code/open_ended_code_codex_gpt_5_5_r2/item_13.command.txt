/bin/bash -lc "python3 - <<'PY'
from collections import Counter
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s.translate(comp)[::-1]
def can(s):
    r=rc(s); return min(s,r)
def readseqs(fn):
    with open(fn) as f:
        while True:
            h=f.readline()
            if not h: break
            s=f.readline().strip(); f.readline(); f.readline()
            yield s
seqs={n:list(readseqs(f)) for n,f in {'mix':'exogenous.mix.reads.q1.mix.fq','exo':'exogenous.mix.reads.q1.exo.fq','endo':'exogenous.mix.reads.q1.endo.fq'}.items()}
def kcnt(ss,k):
    c=Counter()
    for s in ss:
        for i in range(len(s)-k+1):
            mer=s[i:i+k]
            c[can(mer)] += 1
    return c
for k in [10,12,15,20,25,31,40,50]:
    C={n:kcnt(ss,k) for n,ss in seqs.items()}
    keys=set().union(*[set(c) for c in C.values()])
    totals={n:sum(c.values()) for n,c in C.items()}
    P={n:{x:C[n][x]/totals[n] for x in keys} for n in C}
    num=sum((P['mix'][x]-P['endo'][x])*(P['exo'][x]-P['endo'][x]) for x in keys)
    den=sum((P['exo'][x]-P['endo'][x])**2 for x in keys)
    p=num/den
    # with centered log? print unique exo-specific fraction in mix
    exospec=set(C['exo'])-set(C['endo']); endospec=set(C['endo'])-set(C['exo']); common=set(C['exo'])&set(C['endo'])
    exosig=sum(C['mix'][x] for x in exospec)/totals['mix']; endosig=sum(C['mix'][x] for x in endospec)/totals['mix']
    print('k',k,'LS',p,'exo-spec mix tokens',exosig,'endo-spec mix tokens',endosig,'common',len(common),'uniq', {n:len(C[n]) for n in C})
PY"
