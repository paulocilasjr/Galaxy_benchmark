from collections import defaultdict,Counter
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s.translate(comp)[::-1]
def readseqs(fn):
    with open(fn) as f:
        while True:
            h=f.readline()
            if not h: break
            s=f.readline().strip(); f.readline(); f.readline()
            yield s
for name,fn in [('exo','exogenous.mix.reads.q1.exo.fq'),('endo','exogenous.mix.reads.q1.endo.fq'),('mix','exogenous.mix.reads.q1.mix.fq')]:
    seqs=list(readseqs(fn))
    for k in [21,31]:
        k2r=defaultdict(list)
        parent=list(range(len(seqs)))
        def find(x):
            while parent[x]"'!=x:
                parent[x]=parent[parent[x]]; x=parent[x]
            return x
        def union(a,b):
            ra,rb=find(a),find(b)
            if ra!=rb: parent[rb]=ra
        for idx,s in enumerate(seqs):
            for i in range(len(s)-k+1):
                mer=s[i:i+k]; c=min(mer,rc(mer))
                k2r[c].append(idx)
        for rs in k2r.values():
            first=rs[0]
            for r in rs[1:]: union(first,r)
        compc=Counter(find(i) for i in range(len(seqs)))
        print(name,'"'canon k',k,'components',len(compc),'top sizes',compc.most_common(10))
    print()
PY"
