from collections import Counter

def readseqs(fn):
    with open(fn) as f:
        while True:
            h=f.readline().strip()
            if not h: break
            s=f.readline().strip(); plus=f.readline().strip(); q=f.readline().strip()
            yield s,q
mix=Counter(s for s,q in readseqs('exogenous.mix.reads.q1.mix.fq'))
exo=Counter(s for s,q in readseqs('exogenous.mix.reads.q1.exo.fq'))
endo=Counter(s for s,q in readseqs('exogenous.mix.reads.q1.endo.fq'))
for name,C in [('exo',exo),('endo',endo)]:
    vals=[]; missing=0; less=0; eq=0; gt=0; total_src=0; total_mix=0
    for s,c in C.items():
        m=mix.get(s,0)
        if m==0: missing += 1
        if m<c: less += 1
        elif m==c: eq += 1
        else: gt += 1
        total_src+=c; total_mix+=m
    print(name,'missing uniq',missing,'less',less,'eq',eq,'gt',gt,'src reads',total_src,'mix count on source uniq',total_mix)
    print('diff hist top', Counter(mix.get(s,0)-c for s,c in C.items()).most_common(20))
# sequence+quality exact overlap
mixsq=Counter(readseqs('exogenous.mix.reads.q1.mix.fq'))
for name,fn in [('exo','exogenous.mix.reads.q1.exo.fq'),('endo','exogenous.mix.reads.q1.endo.fq')]:
    C=Counter(readseqs(fn))
    print('seq+qual',name,'unique overlap',len(set(mixsq)&set(C)),'reads overlap count in mix',sum(mixsq[x] for x in set(mixsq)&set(C)),'src unique',len(C))
PY"
