/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def readseqs(path):
    with open(path) as f:
        return [line.strip().upper() for i,line in enumerate(f) if i%4==1]
seqs={k:readseqs(f'exogenous.mix.reads.q1.{k}.fq') for k in ['exo','endo','mix']}
for k in range(30,46):
    km={}
    for cls in ['exo','endo']:
        S=set()
        for s in seqs[cls]:
            for i in range(len(s)-k+1): S.add(s[i:i+k])
        km[cls]=S
    exo_only=km['exo']-km['endo']; endo_only=km['endo']-km['exo']
    def cat(s):
        eo=any(s[i:i+k] in exo_only for i in range(len(s)-k+1))
        no=any(s[i:i+k] in endo_only for i in range(len(s)-k+1))
        if eo and not no: return 'exo'
        if no and not eo: return 'endo'
        if eo and no: return 'both'
        return 'unknown'
    cats=['exo','endo','both','unknown']
    P={cls:Counter(cat(s) for s in seqs[cls]) for cls in ['exo','endo','mix']}
    n={cls:len(seqs[cls]) for cls in P}
    # frequencies
    pe=[P['exo'][c]/n['exo'] for c in cats]; pn=[P['endo'][c]/n['endo'] for c in cats]; pm=[P['mix'][c]/n['mix'] for c in cats]
    num=den=0
    for a,b,m in zip(pe,pn,pm):
        d=a-b; num+=d*(m-b); den+=d*d
    alpha=num/den
    print(k, round(alpha,3), P['mix'])
PY"
