/bin/bash -lc "python3 - <<'PY'
from collections import Counter
import math

def readseqs(path):
    with open(path) as f:
        return [line.strip().upper() for i,line in enumerate(f) if i%4==1]
seqs={k:readseqs(f'exogenous.mix.reads.q1.{k}.fq') for k in ['exo','endo','mix']}
for k in range(4,41):
    prof={}
    for cls in ['exo','endo','mix']:
        c=Counter()
        total=0
        for s in seqs[cls]:
            for i in range(len(s)-k+1):
                c[s[i:i+k]]+=1; total+=1
        prof[cls]={m:v/total for m,v in c.items()}
    keys=set(prof['exo'])|set(prof['endo'])|set(prof['mix'])
    # minimize sum (m - (a e +(1-a)n))"'^2 over keys
    num=den=0
    for x in keys:
        e=prof['"'exo'].get(x,0); n=prof['endo'].get(x,0); m=prof['mix'].get(x,0)
        d=e-n
        num += d*(m-n)
        den += d*d
    a=num/den if den else float('nan')
    a_clip=max(0,min(1,a))
    # L1 grid as robust
    best=None
    for j in range(101):
        aa=j/100
        l1=sum(abs(prof['mix'].get(x,0)-(aa*prof['exo'].get(x,0)+(1-aa)*prof['endo'].get(x,0))) for x in keys)
        if best is None or l1<best[0]: best=(l1,aa)
    print(k, round(a,4), round(best[1],2))
PY"
