/bin/bash -lc "python3 - <<'PY'
from collections import Counter
files=['exo','endo','mix']
paths={k:f'exogenous.mix.reads.q1.{k}.fq' for k in files}
def reads(path):
    with open(path) as f:
        while True:
            h=f.readline().strip()
            if not h: break
            s=f.readline().strip().upper(); f.readline(); q=f.readline()
            yield s
cs={k:Counter(reads(p)) for k,p in paths.items()}
for k,c in cs.items():
    print(k, 'n', sum(c.values()), 'unique', len(c), 'dupmax', max(c.values()))
print('set overlaps')
for a in files:
    for b in files:
        if a<b:
            print(a,b,len(set(cs[a])&set(cs[b])))
# classify mix exact
exo=set(cs['exo']); endo=set(cs['endo'])
classes=Counter()
for s,n in cs['mix'].items():
    key=('E' if s in exo else '-')+('N' if s in endo else '-')
    classes[key]+=n
print(classes)
# unique mix counts
classesu=Counter((('E' if s in exo else '-')+('N' if s in endo else '-')) for s in cs['mix'])
print('unique',classesu)
PY"
