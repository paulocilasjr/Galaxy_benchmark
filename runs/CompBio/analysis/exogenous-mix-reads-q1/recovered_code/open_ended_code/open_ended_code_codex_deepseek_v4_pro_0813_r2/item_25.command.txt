/bin/bash -lc "python3 - <<'PY'
import pysam
L=2903; rl=50; cds_start=548; cds_stop=1987
start_min=cds_start; start_max=cds_stop-rl  # 1937
possible=set(range(0,L-rl+1))
outside=sum(1 for s in possible if s<start_min or s>start_max)
p_out=outside/len(possible)

def getpos(fn):
    out=[]
    with pysam.AlignmentFile(fn,'r') as f:
        for r in f:
            if r.is_unmapped or r.is_secondary or r.is_supplementary:
                continue
            out.append(r.reference_start)
    return out
mix=getpos('exogenous.mix.reads.q1.mix.sam')
endo=getpos('exogenous.mix.reads.q1.endo.sam')
exo=getpos('exogenous.mix.reads.q1.exo.sam')
O=sum(s<start_min or s>start_max for s in mix)
Nend=O/p_out
alpha_mapped=1-Nend/len(mix)
# include 13 unmapped as endogenous; BWA total mapped=1952, all reads=1965
Nend_all=Nend+(1965-len(mix))
alpha_all=1-Nend_all/1965
print('possible starts',len(possible),'outside',outside,'p_out',p_out)
print('mix mapped',len(mix),'outside',O,'Nend',Nend,'exo mapped est',len(mix)-Nend,'alpha_mapped',alpha_mapped)
print('all reads alpha (unmapped endo)',alpha_all)
print('nearest10',round(alpha_all*100/10)*10)
PY"
