/bin/bash -lc "python3 - <<'PY'
import pysam
for fn in ['exogenous.mix.reads.q1.exo.sam','exogenous.mix.reads.q1.endo.sam','exogenous.mix.reads.q1.mix.sam']:
    positions=[]; flags=[]; mapq=[]; nm=[]
    unmapped=0
    with pysam.AlignmentFile(fn,'r') as f:
      for r in f:
        if r.is_unmapped: unmapped+=1; continue
        if r.is_secondary or r.is_supplementary: continue
        positions.append(r.reference_start)
        flags.append((r.flag,r.is_reverse))
        mapq.append(r.mapping_quality)
        try: nm.append(r.get_tag('NM'))
        except: nm.append(None)
    print(fn,'unmapped',unmapped,'primary mapped',len(positions))
    from collections import Counter
    print('pos min max',min(positions),max(positions),'median',sorted(positions)[len(positions)//2])
    print('mapq',Counter(mapq).most_common(3))
    print('NM',Counter(nm).most_common(10))
PY"
