/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
def getpos(fn):
 a=[]
 with pysam.AlignmentFile(fn,'r') as f:
  for r in f:
   if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
   a.append(r.reference_start)
 return np.array(a)
mix=getpos('exogenous.mix.reads.q1.mix.sam')
endo=getpos('exogenous.mix.reads.q1.endo.sam')
L=2903; rl=50; a=548; b=1937
p=(a + (L-rl-b))/(L-rl+1) # outside possible starts: 0..a-1 count a, b+1..L-rl count L-rl-b
print('p',p)
# empirical endo p
pe=((endo<a)|(endo>b)).mean()
print('pe',pe)
# bootstrap alpha using uniform and empirical
rng=np.random.default_rng(0)
for pe_name, pval in [('uniform',p),('emp',pe)]:
 vals=[]
 for _ in range(2000):
  mb=rng.choice(mix,len(mix),replace=True)
  O=((mb<a)|(mb>b)).sum()
  Nend=O/pval
  alpha=1-Nend/len(mix)
  vals.append(alpha)
 vals=np.array(vals)
 print(pe_name, 'mean',vals.mean(),'CI',np.percentile(vals,[2.5,97.5]))
PY"
