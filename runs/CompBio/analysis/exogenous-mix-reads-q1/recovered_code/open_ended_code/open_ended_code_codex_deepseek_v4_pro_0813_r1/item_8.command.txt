/bin/bash -lc "python3 - <<'PY'
import pysam, collections
# generate bams
PY
for f in exogenous.mix.reads.q1.mix.fq exogenous.mix.reads.q1.exo.fq exogenous.mix.reads.q1.endo.fq; do b="'${f%.fq}.bam; bwa mem -t 2 klf4_mrna.fa "$f" 2>/dev/null | samtools sort -o "$b" -; samtools index "$b"; done
python3 - <<'"'PY'
import pysam, numpy as np
for f in ['exogenous.mix.reads.q1.mix.bam','exogenous.mix.reads.q1.exo.bam','exogenous.mix.reads.q1.endo.bam']:
    b=pysam.AlignmentFile(f)
    starts=[]; ends=[]; mapqs=[]; flags=[]
    for r in b:
        if r.is_unmapped: continue
        starts.append(r.reference_start); ends.append(r.reference_end); mapqs.append(r.mapping_quality); flags.append(r.flag)
    print('\\n',f, 'mapped',len(starts))
    print('start range',min(starts),max(starts),'mean',np.mean(starts),'median',np.median(starts))
    print('mapq dist', collections.Counter(mapqs).most_common())
    print('flag dist', collections.Counter(flags).most_common(5))
    print('cigar first', [r.cigarstring for r in b if not r.is_unmapped][:10])
    b.close()
PY"
