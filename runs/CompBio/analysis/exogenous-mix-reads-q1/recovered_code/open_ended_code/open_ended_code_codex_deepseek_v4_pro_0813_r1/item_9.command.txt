/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np, collections
for f in ['exogenous.mix.reads.q1.mix.bam','exogenous.mix.reads.q1.exo.bam','exogenous.mix.reads.q1.endo.bam']:
    b=pysam.AlignmentFile(f)
    starts=[]; ends=[]; mapqs=[]; flags=[]; cigars=[]
    for r in b:
        if r.is_unmapped: continue
        starts.append(r.reference_start); ends.append(r.reference_end); mapqs.append(r.mapping_quality); flags.append(r.flag); cigars.append(r.cigarstring)
    print('\\n',f, 'mapped',len(starts))
    print('start range',min(starts),max(starts),'mean %.1f median %.0f'%(np.mean(starts),np.median(starts)))
    print('mapq dist', collections.Counter(mapqs).most_common())
    print('flag dist', collections.Counter(flags).most_common(8))
    print('cigar dist', collections.Counter(cigars).most_common(8))
    b.close()
PY"
