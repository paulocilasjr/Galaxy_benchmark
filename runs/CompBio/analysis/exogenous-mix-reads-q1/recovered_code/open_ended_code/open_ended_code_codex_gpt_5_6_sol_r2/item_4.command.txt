/bin/bash -lc "python3 - <<'PY'
from collections import Counter
D={}
for n in ['exo','endo','mix']:
 with open(f'exogenous.mix.reads.q1.{n}.fq') as f: z=[x.rstrip() for x in f]
 D[n]=(z[0::4],z[1::4],Counter(z[1::4]))
 print(n,'headers',z[0],z[-4])
for n in ['exo','endo']:
 A=D[n][2]; M=D['mix'][2]
 shared=sum((A&M).values())
 missing=sum((A-M).values())
 surplus=sum((M-A).values())
 print(n,'multiset shared',shared,'missing',missing,'mix_not_source',surplus)
# exact seq source-exclusive classes in mixture counts
E=set(D['exo'][1]); N=set(D['endo'][1]); M=D['mix'][2]
for label,S in [('exo_only',E-N),('endo_only',N-E),('both',E&N),('neither',set(M)-E-N)]:
 print(label,'unique',len(S),'mix reads',sum(M[x] for x in S))
PY"
