from collections import Counter,defaultdict
files=['exo','endo','mix']
paths={'exo':'exogenous.mix.reads.q1.exo.fq','endo':'exogenous.mix.reads.q1.endo.fq','mix':'exogenous.mix.reads.q1.mix.fq'}
def reads(path):
    with open(path) as f:
        for i,line in enumerate(f):
            if i%4==1: yield line.strip()
R={k:list(reads(v)) for k,v in paths.items()}
print({k:(len(v),min(map(len,v)),max(map(len,v)),len(set(v))) for k,v in R.items()})
for k in [15,21,27,31,41,54]:
    km={}
    for name,rs in R.items():
        c=Counter()
        for r in rs:
            if k==54:
                c[r]+=1
            else:
                for i in range(len(r)-k+1): c[r[i:i+k]]+=1
        km[name]=c
    exo_set=set(km['exo']); endo_set=set(km['endo'])
    def classify(r):
        if k==54: toks=[r]
        else: toks=[r[i:i+k] for i in range(len(r)-k+1)]
        e=sum(t in exo_set for t in toks); d=sum(t in endo_set for t in toks)
        eu=sum(t in exo_set and t not in endo_set for t in toks)
        du=sum(t in endo_set and t not in exo_set for t in toks)
        return e,d,eu,du
    cats=Counter()
    scores=[]
    for r in R['mix']:
        e,d,eu,du=classify(r)
        if eu and not du: cats['exo_unique_only']+=1
        elif du and not eu: cats['endo_unique_only']+=1
        elif eu and du: cats['both_unique_conflict']+=1
        elif e and not d: cats['exo_seen_only']+=1
        elif d and not e: cats['endo_seen_only']+=1
        elif e and d: cats['both_seen_no_unique']+=1
        else: cats['neither']+=1
        scores.append((eu,du,e,d,r))
    print('k',k,'exo kmers',len(exo_set),'endo kmers',len(endo_set),'inter',len(exo_set&endo_set),'cats',cats)
