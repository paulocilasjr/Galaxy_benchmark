/bin/bash -lc "python - <<'PY'
from urllib.request import urlopen
from collections import Counter
import math
ref=''.join(urlopen('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.4&rettype=fasta&retmode=text',timeout=30).read().decode().splitlines()[1:]).upper()
tr=str.maketrans('ACGT','TGCA')
def get(path):
  with open(path) as f: ls=[x.strip() for x in f]
  return [ls[i] for i in range(1,len(ls),4)]
def coord(s):
  p=ref.find(s)
  if p>=0:return p,0
  p=ref.find(s.translate(tr)[::-1])
  if p>=0:return p,1
  return None
for n in ('exo','endo','mix'):
  cs=[coord(s) for s in get(f'exogenous.mix.reads.q1.{n}.fq')]
  exact=[p for p in cs if p]
  print('\\n',n,'total',len(cs),'exact',len(exact),'not',len(cs)-len(exact))
  print('orientation',Counter(o for p,o in exact))
  print('bins 0-600,600-1983,1983+',Counter('5UTR' if p<600 else 'CDS' if p<=1983 else '3UTR' for p,o in exact))
  print('start quartiles', [sorted(p for p,o in exact)[int(len(exact)*q)] for q in (0,.1,.25,.5,.75,.9,.99)])
  bins=Counter((p//100)*100 for p,o in exact)
  print('bins:', ' '.join(f'{b}:{bins[b]}' for b in sorted(bins)))
PY"
