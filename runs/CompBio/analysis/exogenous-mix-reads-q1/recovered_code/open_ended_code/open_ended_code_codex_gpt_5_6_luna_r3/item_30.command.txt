/bin/bash -lc "python - <<'PY'
import subprocess,collections,math

def positions(x):
 q=f\"awk 'NR%4==1{{print \\\">\\\"substr("'$0,2)}} NR%4==2{{print}}'"' exogenous.mix.reads.q1.{x}.fq\"
 cmd=f\"blastn -task blastn-short -query <({q}) -subject <(curl -L --fail --silent 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.4&rettype=fasta&retmode=text') -outfmt '6 qseqid sstart send length pident mismatch gaps' -max_target_seqs 1 -max_hsps 1 -evalue 100\"
 out=subprocess.run(cmd,shell=True,executable='/bin/bash',capture_output=True,text=True,check=True).stdout.strip().splitlines()
 return [min(int(f.split('\\t')[1]),int(f.split('\\t')[2])) for f in out]
P={x:positions(x) for x in ('exo','endo','mix')}
print({x:len(v) for x,v in P.items()})
for width in (25,50,100,200,300,500):
 cats={x:collections.Counter((p-1)//width for p in ps) for x,ps in P.items()}
 keys=set().union(*(c.keys() for c in cats.values()))
 ns={x:sum(cats[x].values()) for x in cats}
 best=(-1e99,None)
 for i in range(10001):
  p=i/10000
  ll=0
  for k in keys:
   a=cats['exo'][k]/ns['exo']; b=cats['endo'][k]/ns['endo']; prob=p*a+(1-p)*b
   if prob>0: ll+=cats['mix'][k]*math.log(prob)
   elif cats['mix'][k]: ll=-1e300; break
  if ll>best[0]:best=(ll,p)
 print('width',width,'ML p',best[1], 'round10',round(best[1]*10)*10)
# Outside ORF cutoff from GenBank, and observed exo all inside
for st,en in [(595,2034),(601,2033)]:
 print('cutoff',st,en)
 for x in ('exo','endo','mix'):
  out=sum(p<st or p>en-49 for p in P[x])
  print(x,out,len(P[x]),out/len(P[x]))
PY"
