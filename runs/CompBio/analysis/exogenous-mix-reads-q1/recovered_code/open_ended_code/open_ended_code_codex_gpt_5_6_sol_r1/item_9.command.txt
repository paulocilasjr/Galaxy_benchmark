/bin/bash -lc "python3 - <<'PY'
from collections import Counter
ref=''.join(x.strip() for x in open('KLF4.NM_004235.6.fa') if not x.startswith('>'))
tr=str.maketrans('ACGT','TGCA')
for name in ('exo','endo','mix'):
 pos=[]; no=[]
 with open('exogenous.mix.reads.q1.'+name+'.fq') as f:
  reads=[s.strip() for i,s in enumerate(f) if i%4==1]
 for r in reads:
  p=ref.find(r)
  if p<0:p=ref.find(r.translate(tr)[::-1])
  (pos if p>=0 else no).append(p if p>=0 else r)
 print(name,'map',len(pos),'unmap',len(no),'range',min(pos) if pos else None,max(pos) if pos else None)
 print('unmap sample',no[:3])
 # bins 200
 print(sorted(Counter(p//200 for p in pos).items()))
PY"
