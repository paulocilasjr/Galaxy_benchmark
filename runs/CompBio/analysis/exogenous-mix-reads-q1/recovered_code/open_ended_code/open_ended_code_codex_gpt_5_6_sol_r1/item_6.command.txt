/bin/bash -lc "python3 - <<'PY'
from collections import Counter
files={x:f'exogenous.mix.reads.q1.{x}.fq' for x in ('exo','endo','mix')}
def reads(p):
 with open(p) as f:
  return [s.strip() for i,s in enumerate(f) if i%4==1]
R={k:reads(v) for k,v in files.items()}
comp=str.maketrans('ACGT','TGCA')
def canon(s): return min(s,s.translate(comp)[::-1])
print({k:len(v) for k,v in R.items()})
for k in [10,15,20,25,30,35,40,45,50]:
 sets={x:{canon(r[i:i+k]) for r in R[x] for i in range(51-k)} for x in ('exo','endo')} 
 cnt=Counter()
 scores=[]
 for r in R['mix']:
  km={canon(r[i:i+k]) for i in range(51-k)}
  ne=len(km & (sets['exo']-sets['endo'])); nd=len(km & (sets['endo']-sets['exo'])); nb=len(km&sets['exo']&sets['endo'])
  typ='exo_only' if ne and not nd else 'endo_only' if nd and not ne else 'both_unique' if ne and nd else 'shared_only' if nb else 'unseen'
  cnt[typ]+=1; scores.append((ne,nd))
 print(k, len(sets['exo']),len(sets['endo']),len(sets['exo']&sets['endo']),dict(cnt))
PY"
