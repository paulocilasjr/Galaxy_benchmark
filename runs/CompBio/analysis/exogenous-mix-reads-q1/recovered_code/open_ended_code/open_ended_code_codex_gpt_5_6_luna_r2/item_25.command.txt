/bin/bash -lc "python3 - <<'PY'
# Recompute mapped start positions, then mixture likelihood at multiple bins/smoothing values
import subprocess,numpy as np
cmd=\"curl -L --fail --silent 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.5&rettype=fasta&retmode=text' | minimap2 -a -x sr - {fq} 2>/dev/null\"
ps={}
for n in ['exo','endo','mix']:
 out=subprocess.check_output(cmd.format(fq=f'exogenous.mix.reads.q1.{n}.fq'),shell=True,text=True)
 ps[n]=np.array([int(x.split('\\t')[3]) for x in out.splitlines() if x and not x.startswith('@') and not (int(x.split('\\t')[1])&4)])
for width in [10,15,20,25,30,40,50,60,75,100,125,150,200,250,300,400,500]:
 edges=np.arange(1,2904+width,width)
 h={n:np.histogram(ps[n],bins=edges)[0].astype(float) for n in ps}
 for alpha in [0,0.1,0.5,1,2,5,10]:
  p={n:(h[n]+alpha)/(h[n].sum()+alpha*len(h[n])) for n in h}
  fgrid=np.linspace(0,1,10001)
  vals=np.array([np.sum(h['mix']*np.log(np.maximum(f*p['exo']+(1-f)*p['endo'],1e-300))) for f in fgrid])
  f=fgrid[vals.argmax()]
  if alpha in [0,1,5]: print(width,alpha,round(float(f),4),end='; ')
 print()
PY"
