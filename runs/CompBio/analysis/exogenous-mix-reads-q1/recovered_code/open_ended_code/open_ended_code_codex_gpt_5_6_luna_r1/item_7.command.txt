/bin/bash -lc "python -u - <<'PY'
from pathlib import Path
from collections import Counter

def reads(p): return Path(p).read_text().splitlines()[1::4]
exo=reads('exogenous.mix.reads.q1.exo.fq'); endo=reads('exogenous.mix.reads.q1.endo.fq'); mix=reads('exogenous.mix.reads.q1.mix.fq')
for name,a in [('exo',exo),('endo',endo),('mix',mix)]:
 c=Counter(a)
 print(name,'reads',len(a),'unique',len(c),'duplicates',sum(v-1 for v in c.values() if v>1),'maxdup',max(c.values()),'seqs>=2',sum(v>=2 for v in c.values()))
se=set(exo); sd=set(endo); sm=set(mix)
print('overlap exo/endo exact',len(se&sd))
print('mix reads exact exo',sum(s in se for s in mix),'endo',sum(s in sd for s in mix),'both',sum(s in se&sd for s in mix))
print('mix unique exact exo',len(sm&se),'endo',len(sm&sd),'both',len(sm&se&sd))
PY"
