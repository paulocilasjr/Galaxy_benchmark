/bin/bash -lc "python -u - <<'PY'
from pathlib import Path
from collections import Counter

def reads(p): return Path(p).read_text().splitlines()[1::4]
exo=reads('exogenous.mix.reads.q1.exo.fq'); endo=reads('exogenous.mix.reads.q1.endo.fq'); mix=reads('exogenous.mix.reads.q1.mix.fq')
print('sizes',len(exo),len(endo),len(mix))
for k in [10,12,15,18,20,25,30,35,40]:
  def km(a): return {s[i:i+k] for s in a for i in range(51-k)}
  xe=km(exo); xd=km(endo)
  c=Counter(); diffs=[]
  for s in mix:
    me=sum(s[i:i+k] in xe for i in range(51-k))
    md=sum(s[i:i+k] in xd for i in range(51-k))
    # distinct sets not count repeated (not relevant)
    c[(me,md)]+=1
    diffs.append(me-md)
  print('\\nk',k,'kmers',len(xe),len(xd),'class',sum(d>0 for d in diffs),sum(d<0 for d in diffs),sum(d==0 for d in diffs),'mean diff',sum(diffs)/len(diffs))
  print('top',c.most_common(12))
PY"
