/bin/bash -lc "python -u - <<'PY'
from urllib.request import urlopen
from pathlib import Path
from collections import Counter,defaultdict
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text'
txt=urlopen(url,timeout=30).read().decode()
ref=''.join(txt.splitlines()[1:]).upper()
print('ref length',len(ref),txt.splitlines()[0])
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
def reads(p): return Path(p).read_text().splitlines()[1::4]
for p in ['exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q1.mix.fq']:
 a=reads(p); hits=[]; noh=[]
 for s in a:
  f=ref.find(s); r=ref.find(rc(s))
  if f>=0: hits.append((f,'+'))
  elif r>=0: hits.append((r,'-'))
  else: noh.append(s)
 print(Path(p).name,'exact 50mer hits',len(hits),'nohit',len(noh),'pos range', (min(x[0] for x in hits),max(x[0] for x in hits)) if hits else None,'orient',Counter(o for _,o in hits))
 print('top positions',Counter(x[0] for x in hits).most_common(10))
 print('nohit examples',noh[:2])
PY"
