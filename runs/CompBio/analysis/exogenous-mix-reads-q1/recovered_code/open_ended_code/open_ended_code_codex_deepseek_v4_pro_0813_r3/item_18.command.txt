/bin/bash -lc "python3 - <<'PY'
import subprocess
from collections import Counter

def poscounts(bam):
 out=subprocess.check_output(f'samtools view {bam}',shell=True,text=True).splitlines()
 c=Counter()
 for line in out:
  parts=line.split('\\t'); pos=int(parts[3])-1; c[pos]+=1
 return c
for name in ['exo','endo','mix']:
 c=poscounts(f'exogenous.mix.reads.q1.{name}.mrna.bam')
 print(name,'n mapped positions',sum(c.values()),'unique pos',len(c),'min/max',min(c),max(c))
 # distribution by thirds maybe
 L=2903
 thirds=[sum(v for k,v in c.items() if k < L*i/5) for i in range(1,6)]
 # cumulative sections
 print('  sections by 1/5:', [sum(v for k,v in c.items() if L*i/5 <= k < L*(i+1)/5) for i in range(5)])
PY"
