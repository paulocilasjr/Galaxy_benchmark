/bin/bash -lc "python3 - <<'PY'
import subprocess
# check unmapped mRNA reads in genomic bams
for fn,bam_m,bam_g in [('endo','exogenous.mix.reads.q1.endo.mrna.bam','exogenous.mix.reads.q1.endo.genomic.bam'),('mix','exogenous.mix.reads.q1.mix.mrna.bam','exogenous.mix.reads.q1.mix.genomic.bam')]:
 mapped_m=set(subprocess.check_output(f'samtools view {bam_m} | cut -f1',shell=True,text=True).split())
 mapped_g=set(subprocess.check_output(f'samtools view {bam_g} | cut -f1',shell=True,text=True).split())
 print(fn,'mRNA mapped',len(mapped_m),'genomic mapped',len(mapped_g),'in genomic not mRNA',len(mapped_g-mapped_m))
 # print rows of genomic-only
 for name in list(mapped_g-mapped_m)[:10]:
  row=subprocess.check_output(f'samtools view {bam_g} | awk -F\"\\t\" \\'"'$1=="{name}"'"\\'',shell=True,text=True).strip()
  print(row)
PY"
