/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('klf4_genomic.txt'))
dna=j['dna']
open('klf4_genomic.fa','w').write('>KLF4_genomic_hg38\\n'+dna+'\\n')
print(len(dna))
# also mRNA fasta already
print(open('klf4_mrna.fa').read().split('\\n',1)[1].replace('\\n','').__len__())
PY
# build bwa index for both
bwa index klf4_mrna.fa
bwa index klf4_genomic.fa
# map reads to both
for f in exogenous.mix.reads.q1.exo.fq exogenous.mix.reads.q1.endo.fq exogenous.mix.reads.q1.mix.fq; do
  base="'${f%.fq}
  bwa mem -t 2 -k 15 klf4_mrna.fa $f | samtools view -bS -F 4 - > ${base}.mrna.bam
  bwa mem -t 2 -k 15 klf4_genomic.fa $f | samtools view -bS -F 4 - > ${base}.genomic.bam
  echo $base mrna $(samtools view -c ${base}.mrna.bam) genomic $(samtools view -c ${base}.genomic.bam)
done'
