/bin/bash -lc "python3 - <<'PY'
import subprocess, math
from collections import Counter

def positions(bam):
 out=subprocess.check_output(f'samtools view {bam}',shell=True,text=True).splitlines()
 return [int(l.split('\\t')[3])-1 for l in out]
ex=positions('exogenous.mix.reads.q1.exo.mrna.bam')
en=positions('exogenous.mix.reads.q1.endo.mrna.bam')
mx=positions('exogenous.mix.reads.q1.mix.mrna.bam')
L=2903
# E support variations; include positions inclusive [lo,hi]
variants=[(555,1937),(548,1937),(548,1987),(555,1987),(550,1940),(575,1937)]
for lo,hi in variants:
 Le=hi-lo+1
 # MLE q,p via grid; exo likelihood constant
 best=None
 for qi in range(0,101):
  q=qi/100
  # p grid
  for pi in range(0,101):
   p=pi/100
   ll=0
   # endo
   for pos in en:
    pr=(q/Le if lo<=pos<=hi else 0) + (1-q)/L
    ll += math.log(pr)
   # mix
   for pos in mx:
    pr=(p/Le if lo<=pos<=hi else 0) + (1-p)/L
    ll += math.log(pr)
   if best is None or ll>best[0]: best=(ll,q,p)
 print('E',lo,hi,'len',Le,'MLE q,p',best[1],best[2],'ll',best[0])
PY"
