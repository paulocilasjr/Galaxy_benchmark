/bin/bash -lc "python3 - <<'PY'
import math

def solve_L(n,u):
 lo=1; hi=100000
 # ratio = (1-exp(-n/L))/(n/L), solve L
 for _ in range(100):
  mid=(lo+hi)/2
  x=n/mid
  val=(1-math.exp(-x))/x if x else 1
  if val > u/n: # need larger x -> smaller L
   hi=mid
  else:
   lo=mid
 return (lo+hi)/2
for n,u,name in [(777,685,'exo'),(395,385,'endo'),(1965,1684,'mix')]:
 L=solve_L(n,u)
 print(name,'n',n,'u',u,'L estimate',L,'coverage',n/L)
PY"
