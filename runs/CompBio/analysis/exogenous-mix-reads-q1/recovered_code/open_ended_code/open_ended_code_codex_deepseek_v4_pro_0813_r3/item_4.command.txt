/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def reads(fn):
    lines=open(fn).read().splitlines(); return [lines[i+1] for i in range(0,len(lines),4)]
exo=reads('exogenous.mix.reads.q1.exo.fq'); endo=reads('exogenous.mix.reads.q1.endo.fq'); mix=reads('exogenous.mix.reads.q1.mix.fq')
E=set(exo); N=set(endo)
print('sizes exo unique',len(E),'endo unique',len(N),'mix unique',len(set(mix)))
print('exo-endo exact overlap',len(E&N))
# count overlap duplicates
ce=Counter(exo); cn=Counter(endo); cm=Counter(mix)
shared=E&N
print('overlap reads',len(shared),'exo counts total shared',sum(ce[s] for s in shared),'endo counts total',sum(cn[s] for s in shared),'mix count shared',sum(cm[s] for s in shared))
# exact in one only
eo_only=E-N; en_only=N-E
print('mix in exo only',sum(cm[s] for s in eo_only),'in endo only',sum(cm[s] for s in en_only),'in shared',sum(cm[s] for s in shared),'in neither',sum(cm[s] for s in set(mix)-E-N))
# For exo samples, how many reads in endo set etc
print('exo reads in endo',sum(s in N for s in exo),'endo reads in exo',sum(s in E for s in endo))
PY"
