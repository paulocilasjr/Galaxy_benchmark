/bin/bash -lc "python3 - <<'PY'
import urllib.request
# UCSC hg38 sequence via API
chrom='chr9'; start=107484851; end=107489768
url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={start};end={end}'
print(url)
try:
 data=urllib.request.urlopen(url,timeout=30).read().decode()
 print(data[:1000])
 open('klf4_genomic.txt','w').write(data)
except Exception as e: print('ERR',e)
PY"
