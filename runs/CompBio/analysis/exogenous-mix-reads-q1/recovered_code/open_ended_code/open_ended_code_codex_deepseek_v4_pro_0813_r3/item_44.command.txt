import subprocess
from collections import Counter
L=2936
def counts(bam):
 c=Counter()
 for l in subprocess.check_output(f'samtools view {bam}',shell=True,text=True).splitlines():
  p=l.split('\\t'); c[int(p[3])-1]+=1
 return c
en=counts('exogenous.mix.reads.q1.endo.v2.bam'); mx=counts('exogenous.mix.reads.q1.mix.v2.bam')
Nn=sum(en.values());Nm=sum(mx.values())
for lo,hi in [(602,1988),(603,1987),(604,1986),(605,1985),(600,1990),(595,1995)]:
 r=(hi-lo+1)/L; oe=sum(v for p,v in en.items() if p<lo or p>=hi); om=sum(v for p,v in mx.items() if p<lo or p>=hi)
 print(lo,hi,'p',1-om/(Nm*(1-r)),'q',1-oe/(Nn*(1-r)))
PY"
