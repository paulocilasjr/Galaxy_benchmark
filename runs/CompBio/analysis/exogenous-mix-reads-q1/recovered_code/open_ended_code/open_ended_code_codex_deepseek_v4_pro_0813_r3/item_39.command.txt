/bin/bash -lc 'bwa index klf4_v2.fa
bwa index klf4_v1.fa
for f in exogenous.mix.reads.q1.exo.fq exogenous.mix.reads.q1.endo.fq exogenous.mix.reads.q1.mix.fq; do
  base=${f%.fq}
  bwa mem -t 2 -k 15 klf4_v2.fa $f | samtools view -bS -F 4 - > ${base}.v2.bam
  bwa mem -t 2 -k 15 klf4_v1.fa $f | samtools view -bS -F 4 - > ${base}.v1.bam
  echo $base v2 $(samtools view -c ${base}.v2.bam) v1 $(samtools view -c ${base}.v1.bam)
done'
