/bin/bash -lc "python - <<'PY'
import json,collections
from count_parameters import independent_count
v=json.load(open('checkpoint_variables.json'))
train=[x for x in v if '/moving_mean/' not in x['name'] and '/moving_variance/' not in x['name'] and x['name'].startswith('module/')]
print('Trainable variable types:', collections.Counter(x['name'].split('/')[-3] for x in train))
print('Trainable tensors:',len(train))
total=sum(x['count'] for x in train)
print('Checkpoint trainable parameters:',total)
assert total==independent_count()
json.dump(train,open('trainable_variables.json','w'),indent=2)
open('answer.txt','w').write(str(total)+'\\n')
assert open('answer.txt').read()==str(total)+'\\n'
PY"
