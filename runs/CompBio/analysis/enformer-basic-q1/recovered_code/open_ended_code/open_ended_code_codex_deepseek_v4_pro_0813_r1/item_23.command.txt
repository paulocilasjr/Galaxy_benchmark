/bin/bash -lc "python3 - <<'PY'
C=1536
filters=[768,896,1024,1152,1280,1536]

def conv(i,o,w,b=True): return i*o*w+(o if b else 0)
def bn(ch): return 2*ch
def pool(ch): return ch*ch
def conv_block(in_,out,w): return bn(in_)+conv(in_,out,w)

stem=conv(4,768,15)+conv_block(768,768,1)+pool(768)
tower=0; prev=768
for f in filters:
    tower += conv_block(prev,f,5)+conv_block(f,f,1)+pool(f)
    prev=f
mha=conv(C,512,1,False)*2 + conv(C,C,1,False)+conv(C,C,1,True)+conv(192,512,1,False)+512*2
mlp=bn(C)+conv(C,2*C,1)+conv(2*C,C,1)
trans=11*(bn(C)+mha+mlp)
final=bn(C)+conv(C,2*C,1)
human=conv(2*C,5313,1)
mouse=conv(2*C,1643,1)
trunk=stem+tower+trans+final
print(stem,tower,trans,final,human,mouse)
print('trunk', trunk, 'human-only', trunk+human, 'both', trunk+human+mouse)
PY"
