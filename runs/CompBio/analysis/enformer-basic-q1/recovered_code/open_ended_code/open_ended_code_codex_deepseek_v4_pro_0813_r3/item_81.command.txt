/bin/bash -c "cd workspace && python - <<'PY'
import struct, io
path='enformer_saved/variables/variables.index'
data=open(path,'rb').read()
print('len', len(data), hex(len(data)))
footer=data[-48:]
magic=footer[-8:]
print('magic', magic.hex(), hex(struct.unpack('<Q', magic)[0]))
# parse varints from footer first 40
def read_varint(buf,pos):
    shift=0; result=0
    while True:
        b=buf[pos]; pos+=1
        result |= (b & 0x7f) << shift
        if not (b & 0x80): return result,pos
        shift+=7
pos=0
meta_off,pos=read_varint(footer,pos)
meta_size,pos=read_varint(footer,pos)
idx_off,pos=read_varint(footer,pos)
idx_size,pos=read_varint(footer,pos)
print('meta',meta_off,meta_size,'idx',idx_off,idx_size,'endpos',pos)
print('idx bytes', data[idx_off:idx_off+idx_size].hex()[:100])
print('idx size mod', idx_size%4096)
PY"
