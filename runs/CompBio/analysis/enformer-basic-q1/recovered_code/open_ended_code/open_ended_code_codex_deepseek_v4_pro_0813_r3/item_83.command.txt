/bin/bash -c "python - <<'PY'
import struct
path='enformer_saved/variables/variables.index'; data=open(path,'rb').read()
idx_off,idx_size=5825,15
block=data[idx_off:idx_off+idx_size]
print(block.hex())
# parse entries, strip trailer
content=block[:-5]
num_restarts=struct.unpack('<I',content[-4:])[0]
restart_base=len(content)-4-4*num_restarts
print('content len',len(content),'num_restarts',num_restarts,'restart_base',restart_base,'restarts',[struct.unpack('<I',content[restart_base+4+i*4:restart_base+8+i*4])[0] for i in range(num_restarts)])
def read_varint(buf,pos):
    shift=0; result=0
    while True:
        b=buf[pos]; pos+=1
        result |= (b & 0x7f) << shift
        if not (b & 0x80): return result,pos
        shift+=7
pos=0; prev=b''
while pos < restart_base:
    shared,pos=read_varint(content,pos)
    nonshared,pos=read_varint(content,pos)
    vlen,pos=read_varint(content,pos)
    key=prev[:shared]+content[pos:pos+nonshared]; pos+=nonshared
    val=content[pos:pos+vlen]; pos+=vlen
    prev=key
    print('entry', pos, shared,nonshared,vlen,key,val.hex())
print('endpos',pos)
PY"
