/bin/bash -lc "./venv/bin/python - <<'PY'
import inspect
if not hasattr(inspect,'ArgSpec'):
    from collections import namedtuple
    inspect.ArgSpec=namedtuple('ArgSpec','args varargs keywords defaults')
if not hasattr(inspect,'getargspec'):
    inspect.getargspec=lambda f: inspect.ArgSpec(*inspect.getfullargspec(f)[:4])
from tensorflow_checkpoint_reader.py_checkpoint_reader import NewCheckpointReader
r=NewCheckpointReader('enformer_saved/variables/variables')
shapes=r.get_variable_to_shape_map()
print('num vars',len(shapes))
total=0
for name,shape in shapes.items():
    n=1
    for s in shape: n*=s
    total+=n
print('total params sum shapes',total)
print(f'{total:,}')
for name in list(shapes)[:120]: print(name, shapes[name])
print('---tail---')
for name in list(shapes)[-120:]: print(name, shapes[name])
PY"
