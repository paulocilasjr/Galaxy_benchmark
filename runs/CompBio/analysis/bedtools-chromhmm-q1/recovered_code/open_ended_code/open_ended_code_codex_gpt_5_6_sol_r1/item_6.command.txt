/bin/bash -lc "bedtools intersect -sorted -a peaks.sorted.bed -b states.sorted.bed -wo | awk '{den+="'$NF; if ($(NF-1)=="14_ReprPCWk") num+=$NF} END{printf "record_num=%0.f record_den=%0.f pct=%.9f rounded=%.0f'"\\n\",num,den,100*num/den,100*num/den}'
bedtools intersect -sorted -a peaks.union.bed -b states.sorted.bed -wo | awk '{den+="'$NF; if ($(NF-1)=="14_ReprPCWk") num+=$NF} END{printf "union_num=%0.f union_den=%0.f pct=%.9f rounded=%.0f'"\\n\",num,den,100*num/den,100*num/den}'
bedtools intersect -sorted -a peaks.sorted.bed -b states.sorted.bed -wo | awk '{by["'$(NF-1)]+=$NF; den+=$NF} END{for(k in by) printf "%s'"\\t%.0f\\t%.6f\\n\",k,by[k],100*by[k]/den}' | sort -k2,2nr | head -n 15"
