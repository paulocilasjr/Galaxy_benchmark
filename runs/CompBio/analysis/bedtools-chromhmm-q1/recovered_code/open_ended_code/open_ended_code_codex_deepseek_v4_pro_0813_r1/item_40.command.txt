/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict
# annotation end per chrom from ann_all sorted
ends={}
with open('ann_all.bed') as f:
    for line in f:
        c,s,e,name=line.rstrip().split('\\t')
        ends[c]=max(ends.get(c,0),int(e))
print('ends sample', {k:ends[k] for k in ['chr1','chr11','chr17']})
# raw row annotated bases
row_total=0; row_annot=0
with open('peaks_raw.bed') as f:
    for line in f:
        c,s,e=line.rstrip().split('\\t'); s=int(s); e=int(e)
        row_total += e-s
        end=ends.get(c)
        if end is not None:
            row_annot += max(0, min(e,end)-max(s,0))
print('row_total',row_total,'row_annot',row_annot,'diff',row_total-row_annot)
# union annotated
u_total=0; u_annot=0
with open('peaks_union.bed') as f:
    for line in f:
        c,s,e=line.rstrip().split('\\t'); s=int(s); e=int(e)
        u_total += e-s
        end=ends.get(c)
        if end is not None:
            u_annot += max(0, min(e,end)-max(s,0))
print('union_total',u_total,'union_annot',u_annot,'diff',u_total-u_annot)
PY"
