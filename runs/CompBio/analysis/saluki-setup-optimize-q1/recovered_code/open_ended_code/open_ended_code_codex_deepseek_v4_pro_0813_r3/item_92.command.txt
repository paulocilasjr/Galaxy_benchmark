import tensorflow as tf
import numpy as np
files=['train20.tfr']
ds=tf.data.TFRecordDataset(files, compression_type='ZLIB')
features={'lengths':tf.io.FixedLenFeature((1,),tf.int64),'sequence':tf.io.FixedLenFeature([],tf.string),'coding':tf.io.FixedLenFeature([],tf.string),'splice':tf.io.FixedLenFeature([],tf.string),'targets':tf.io.FixedLenFeature([],tf.string)}
def parse(x):
 f=tf.io.parse_single_example(x,features); l=f['lengths'][0]; seq=tf.io.decode_raw(f['sequence'],tf.uint8)[:l]; cod=tf.io.decode_raw(f['coding'],tf.uint8)[:l]; spl=tf.io.decode_raw(f['splice'],tf.uint8)[:l]; return seq,cod,spl,l
found=0
for i,raw in enumerate(ds):
 seq,cod,spl,l=parse(raw); seq=seq.numpy(); cod=cod.numpy(); spl=spl.numpy(); l=int(l.numpy()); n=int(spl.sum())
 if n==1:
  print('idx',i,'len',l,'splice_pos',np.where(spl)[0].tolist(),'lastidx',l-1)

  if found>=5: break
print('found',found)
