/bin/bash -lc "cat > eval_one.py <<'PY'
import os, json, numpy as np
import tensorflow as tf
from basenji import rnann

seq='GCCGCCACCATGGTGAGCAAGGGCGAGTAGTGTACATAATAAGGACT'
L=12288
def encode(seq, mode='paper'):
    x=np.zeros((L,6), dtype='float32')
    for i,nt in enumerate(seq):
        d='ACGT'.index(nt)
        x[i,d]=1
    if mode=='paper':
        cds_start=seq.index('ATG')
        cds_end=seq.index('TAG')+3
        for i in range(cds_start,cds_end,3):
            x[i,4]=1
    elif mode=='archive':
        cds_start=next(i for i,c in enumerate(seq) if c in 'ACGT')
        for i in range(cds_start,len(seq),3):
            x[i,4]=1
    return x[None,...]
params={'model':{'activation':'relu','rnn_type':'gru','seq_length':12288,'augment_shift':3,'num_targets':1,'heads':2,'filters':64,'kernel_size':5,'dropout':0.3,'l2_scale':0.001,'ln_epsilon':0.007,'num_layers':6,'bn_momentum':0.90}}
model=rnann.RnaNN(params['model'])
model.restore('models_h5/f0_c0_model0_best.h5',0)
for mode in ['paper','archive']:
    x=encode(seq,mode)
    p=model.predict(x)
    print(mode,p, p.shape, float(p[0,0]))
PY
PYTHONPATH=basenji_repo ./tfenv/bin/python eval_one.py"
