b=open('zip_central.bin','rb').read(); i=0; found=[]
while i<len(b):
 vals=struct.unpack_from('<4s6H3I5H2I',b,i)
 sig,vm,vn,flag,method,tm,dt,crc,cs,us,nl,xl,cl,disk,internal,external,off=vals
 name=b[i+46:i+46+nl].decode('utf-8','replace')
 extra=b[i+46+nl:i+46+nl+xl]
 if name.startswith('datasets/deeplearning/train_gru/') and (name.endswith('model0_best.h5') or name.endswith('model1_best.h5') or name.endswith('/params.json')):
  ex={}; j=0
  while j+4<=len(extra):
   eid,sz=struct.unpack_from('<HH',extra,j); dat=extra[j+4:j+4+sz]; ex[eid]=dat; j+=4+sz
  realoff=off
  if off==0xffffffff and 1 in ex:
   z=ex[1]; k=0
   # zip64 extra values only for saturated fields in order usize,csize,offset,disk
   if us==0xffffffff: k+=8
   if cs==0xffffffff: k+=8
   if off==0xffffffff: realoff=struct.unpack_from('<Q',z,k)[0]
  found.append((name,cs,us,realoff,method,flag,extra.hex()))
 i+=46+nl+xl+cl
print('count',len(found))
for x in sorted(found): print('\\t'.join(map(str,x[:-1])))
