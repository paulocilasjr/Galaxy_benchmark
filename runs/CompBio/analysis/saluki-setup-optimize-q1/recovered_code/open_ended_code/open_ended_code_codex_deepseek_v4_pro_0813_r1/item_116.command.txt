torch.set_num_threads(1)
from saluki_torch import Saluki, load_weights

MODEL_PATHS = [f'models/train_gru/f{fi}_c0/train/model0_best.h5' for fi in range(10)]
UPSTREAM = 'GCCGCCACCATGGTGAGCAAGGGCGAGTAG'
UTR0 = 'TGTACATAATAAGGACT'
BASES = 'ACGT'
MAXLEN = 12288

def encode_sequence(full_dna):
    # full_dna length 47
    L = len(full_dna)
    seq = np.frombuffer(full_dna.encode(), dtype=np.uint8)
    # map A,C,G,T to 0,1,2,3
    code = np.full(L, -1, dtype=np.int64)
    for i,ch in enumerate(full_dna):
        code[i] = BASES.index(ch)
    one = np.eye(4, dtype=np.float32)[code]
    x = np.zeros((MAXLEN,6), dtype=np.float32)
    x[:L,:4] = one
    # coding frame marks first base of each codon in CDS including stop.
    # 5'UTR length 9; coding starts at 9 and length 21.
    cds_start = 9

    for pos in range(cds_start, cds_start+cds_len, 3):
        x[pos,4] = 1.0
    # no splice sites (single-exon synthetic mRNA)
    return x

def build_batch(seqs):
    arr = np.stack([encode_sequence(s) for s in seqs], axis=0)
    return torch.from_numpy(arr).permute(0,2,1).contiguous()

def predict_all_models(seqs):
    \"\"\"Average prediction across 10 fold models (human output head0).\"\"\"
    X = build_batch(seqs)
    total = np.zeros(len(seqs), dtype=np.float32)

        model = Saluki()
        load_weights(model, mp)
        model.eval()
