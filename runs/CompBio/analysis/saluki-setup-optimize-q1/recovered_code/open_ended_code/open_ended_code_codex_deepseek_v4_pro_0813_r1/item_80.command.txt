/bin/bash -lc "python - <<'PY'
import json, urllib.request, struct, zlib
entries={e['name']:e for e in json.load(open('central_entries.json'))}
for name in ['datasets/deeplearning/train_gru/f0_c0/test0.out','datasets/deeplearning/train_gru/f0_c0/test0.err','datasets/deeplearning/train_gru/f0_c0/train.out']:
 e=entries[name]; dest='models/train_gru/f0_c0/'+name.split('/')[-1]
 start=e['local_off']; end=start+e['comp_size']+65536-1
 url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
 req=urllib.request.Request(url, headers={'Range':f'bytes={start}-{end}'})
 with urllib.request.urlopen(req, timeout=60) as r: data=r.read()
 rec=struct.unpack('<IHHHHHIIIHH', data[:30]); method=rec[3]; nl,el=rec[9],rec[10]
 comp=data[30+nl+el:30+nl+el+e['comp_size']]
 out=zlib.decompress(comp,-15) if method==8 else comp
 open(dest,'wb').write(out)
 print(dest,len(out))
PY
for f in models/train_gru/f0_c0/*.out models/train_gru/f0_c0/*.err; do echo '---' "'$f; head -30 $f; done'
