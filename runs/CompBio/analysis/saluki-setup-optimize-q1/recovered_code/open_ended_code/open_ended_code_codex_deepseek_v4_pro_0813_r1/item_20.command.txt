/bin/bash -lc "python - <<'PY'
import urllib.request
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'

with urllib.request.urlopen(req, timeout=60) as r:
    data=r.read()
open('tail.zip','wb').write(data)
print(len(data), r.headers.get('Content-Range'))
PY"
