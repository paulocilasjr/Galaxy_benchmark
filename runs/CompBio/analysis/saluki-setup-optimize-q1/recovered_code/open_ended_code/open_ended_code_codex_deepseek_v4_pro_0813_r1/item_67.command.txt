/bin/bash -lc "python - <<'PY'
import json, urllib.request, struct, zlib, os
entries={e['name']:e for e in json.load(open('central_entries.json'))}
base='datasets/deeplearning/train_gru/f0_c0/test0/'
for fn in ['preds.h5','targets.h5','acc.txt']:
 name=base+fn; e=entries[name]; dest='models/train_gru/f0_c0/test0_'+fn
 start=e['local_off']; end=start+e['comp_size']+65536-1
 url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
 req=urllib.request.Request(url, headers={'Range':f'bytes={start}-{end}'})
 with urllib.request.urlopen(req, timeout=60) as r: data=r.read()
 rec=struct.unpack('<IHHHHHIIIHH', data[:30]); method=rec[3]
 nl,el=rec[9],rec[10]
 comp=data[30+nl+el:30+nl+el+e['comp_size']]
 out=zlib.decompress(comp,-15) if method==8 else comp
 open(dest,'wb').write(out)
 print(dest,len(out))
PY
python - <<'PY'
import h5py
for fn in ['models/train_gru/f0_c0/test0_preds.h5','models/train_gru/f0_c0/test0_targets.h5']:

 for k in f.keys(): print(k, f[k].shape, f[k].dtype)
 print('pred first',f['preds'][:5].ravel())
 f.close()
