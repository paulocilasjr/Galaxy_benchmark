/bin/bash -lc "python - <<'PY'
import struct, sys, numpy as np
sys.path.insert(0,'proto')
from example_pb2 import Example
fn='models/train_gru/f0_c0/test-0.tfr.dec'
data=open(fn,'rb').read(); pos=0; n=0
while pos+16<=len(data) and n<3:
    length=struct.unpack('<Q', data[pos:pos+8])[0]
    payload=data[pos+12:pos+12+length]
    ex=Example(); ex.ParseFromString(payload)
    feats=ex.features.feature
    seq=feats['sequence'].bytes_list.value[0]
    coding=feats['coding'].bytes_list.value[0]
    splice=feats['splice'].bytes_list.value[0]
    targets=feats['targets'].bytes_list.value[0]
    print('record',n,'len',list(feats['lengths'].int64_list.value),'raw',len(seq),'coding sum',sum(coding),'splice sum',sum(splice),'target',np.frombuffer(targets,dtype='<f2'))
    print('seq bytes first',list(seq[:15]))
    pos += 12+length+4
    n+=1
PY"
