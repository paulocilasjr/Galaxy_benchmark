/bin/bash -lc "python - <<'PY'
import urllib.request, struct
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
off=8391375635
req=urllib.request.Request(url, headers={'Range':f'bytes={off}-{off+99}'})
r=urllib.request.urlopen(req, timeout=60)
b=r.read()
print(r.status, len(b), b[:30])
print(b[:4], b[:4].hex())
# local header parse if PK
if b[:4]==b'PK\\x03\\x04':
    ver,flags,method,modt,modd,crc,compsz,uncsz,nlen,xlen=struct.unpack('<IHHHHHIIIHH', b[:30])
    print(ver,flags,method,crc,compsz,uncsz,nlen,xlen)
    print(b[30:30+nlen])
PY"
