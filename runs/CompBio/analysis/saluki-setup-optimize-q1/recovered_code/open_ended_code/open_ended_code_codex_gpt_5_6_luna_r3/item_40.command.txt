/bin/bash -lc "pwd; python - <<'PY'
import h5py, json
p='saluki_models/train_gru/f0_c0/train/model0_best.h5'
with h5py.File(p,'r') as f:
 cfg=json.loads(f.attrs['model_config'])
 print('layers')
 for l in cfg['config']['layers']:
  c=l['config']
  print(l['name'],l['class_name'],'eps=',c.get('epsilon'),'mom=',c.get('momentum'),'units=',c.get('units'),'filters=',c.get('filters'),'kernel=',c.get('kernel_size'))
 print('weights tree')
 def walk(g,prefix=''):
  for k in g.keys():
   o=g[k]
   if isinstance(o,h5py.Dataset): print(prefix+k,o.shape,o.dtype)
   else: walk(o,prefix+k+'/')
 walk(f['model_weights'])
PY"
