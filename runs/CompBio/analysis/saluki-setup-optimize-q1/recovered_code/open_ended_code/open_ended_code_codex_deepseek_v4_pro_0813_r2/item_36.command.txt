/bin/bash -lc "python - <<'PY'
from remotezip import RemoteZip
import time, os
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
out='models/test_model0.h5'
with RemoteZip(url) as z:
    name='datasets/deeplearning/train_gru/f0_c0/train/model0_best.h5'
    t=time.time()
    data=z.read(name)
    print('downloaded',len(data),'time',time.time()-t)
    open(out,'wb').write(data)
PY
ls -lh models/test_model0.h5"
