/bin/bash -lc "PYTHONPATH=./pydeps python3 - <<'PY'
from remotezip import RemoteZip
from pathlib import Path
import shutil, os
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
outroot=Path('.')
with RemoteZip(url) as z:
    names=['datasets/deeplearning/train_gru/params.json']
    for f in range(10):
        for c in range(5):
            names.append(f'datasets/deeplearning/train_gru/f{f}_c{c}/train/params.json')
            names.append(f'datasets/deeplearning/train_gru/f{f}_c{c}/train/model0_best.h5')
    for idx,n in enumerate(names,1):
        target=outroot/n
        if target.exists() and target.stat().st_size > 0:
            continue
        target.parent.mkdir(parents=True, exist_ok=True)
        with z.open(n) as src, open(target, 'wb') as dst:
            shutil.copyfileobj(src, dst)
        if idx % 10 == 0 or n.endswith('params.json'):
            print(idx, n)
print('done')
PY"
