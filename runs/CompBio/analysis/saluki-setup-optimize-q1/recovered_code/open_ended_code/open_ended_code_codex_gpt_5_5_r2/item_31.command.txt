/bin/bash -lc "python3 - <<'PY'
from remotezip import RemoteZip
from pathlib import Path
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
out=Path('saluki_models')
out.mkdir(exist_ok=True)
needed=['datasets/deeplearning/train_gru/params.json']
for f in range(10):
 for c in range(5):
  needed.append(f'datasets/deeplearning/train_gru/f{f}_c{c}/train/params.json')
  needed.append(f'datasets/deeplearning/train_gru/f{f}_c{c}/train/model0_best.h5')
with RemoteZip(url) as z:
 names=set(z.namelist())
 missing=[n for n in needed if n not in names]
 print('missing', missing[:5], 'count', len(missing))
 for idx,n in enumerate(needed,1):
  dest=out/n
  if dest.exists() and dest.stat().st_size>0:
   continue
  dest.parent.mkdir(parents=True, exist_ok=True)
  data=z.read(n)
  dest.write_bytes(data)
  print(idx, n, len(data), flush=True)
PY"
