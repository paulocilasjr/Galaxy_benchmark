import urllib.request,concurrent.futures
urls=['https://resources.altius.org/~jvierstra/projects/DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz','https://resources.altius.org/~jvierstra/projects/','https://meuleman.org/research/dhsindex/','http://www.meuleman.org/DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz','https://hgdownload.soe.ucsc.edu/gbdb/hg38/encode3/']
def run(t):
 i,u=t
 try:
  req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
  with urllib.request.urlopen(req,timeout=20) as r:
   b=r.read(100000);open(f'url_probe_{i}.txt','wb').write(b);return u,r.status,len(b),b[:100]
 except Exception as e:return u,str(e)
with concurrent.futures.ThreadPoolExecutor(5) as ex:
