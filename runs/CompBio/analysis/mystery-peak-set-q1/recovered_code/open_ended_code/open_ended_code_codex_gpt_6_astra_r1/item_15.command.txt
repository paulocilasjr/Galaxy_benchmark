/bin/bash -lc "python - <<'PY'
import urllib.request
u='http://www.meuleman.org/DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz'
req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
with urllib.request.urlopen(req,timeout=60) as r,open('DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz','wb') as f:
 print(dict(r.headers),flush=True)
 while True:
  b=r.read(1024*1024)
  if not b:break
  f.write(b)
print('Downloaded DHS reference')
PY"
