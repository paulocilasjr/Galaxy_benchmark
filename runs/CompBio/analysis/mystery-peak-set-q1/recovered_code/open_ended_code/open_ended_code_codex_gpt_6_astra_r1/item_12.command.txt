/bin/bash -lc "cat > compare_ccre.py <<'PY'
import pyBigWig,gzip,collections,json,numpy as np
b=pyBigWig.open('ccre.bb')
peaks=collections.defaultdict(list)
for l in gzip.open('mystery.peak.set.q1.bed.gz','rt'):
 c,a,z,s=l.split();peaks[c].append((int(a),int(z),s))
n=collections.Counter();hits=collections.Counter();centers=collections.Counter()
for c,ps in peaks.items():
 es=b.entries(c,0,b.chroms()[c]) or []
 starts=np.array([e[0] for e in es]); ends=np.maximum.accumulate([e[1] for e in es]) if es else np.array([])
 for a,z,s in ps:
  n[s]+=1
  i=np.searchsorted(starts,z,side='left')-1
  if i>=0 and ends[i]>a:hits[s]+=1
  m=(a+z)//2;i=np.searchsorted(starts,m,side='right')-1
  if i>=0 and ends[i]>m:centers[s]+=1
results={s:{'n':n[s],'overlap':hits[s],'fraction_overlap':hits[s]/n[s],'fraction_center_overlap':centers[s]/n[s]} for s in sorted(n)}
print(json.dumps(results,indent=2));json.dump(results,open('ccre_overlap_results.json','w'),indent=2)
a=json.load(open('encode_annotation.json'))
for f in a['files']:print(f.get('accession'),f.get('file_format'),f.get('output_type'),f.get('href'),f.get('file_size'))
PY
PYTHONPATH=./python_packages python compare_ccre.py"
