/bin/bash -lc "gzip -dc mystery.peak.set.q1.bed.gz > mystery.peak.set.q1.bed
python3 - <<'PY'
from collections import Counter,defaultdict
c=Counter(); lens=defaultdict(Counter); chroms=defaultdict(Counter)
for line in open('mystery.peak.set.q1.bed'):
    f=line.strip().split('\\t')
    s=f[3]; c[s]+=1; lens[s][int(f[2])-int(f[1])]+=1; chroms[s][f[0]]+=1
print('counts',c)
for s in sorted(c):
    print(s, 'n=',c[s], 'lens', lens[s].most_common(5), 'chroms', chroms[s].most_common(10))
PY"
