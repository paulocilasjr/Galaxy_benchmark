/bin/bash -lc "gzip -cd mystery.peak.set.q1.bed.gz | bedtools intersect -a - -b rmsk.bed -wo | awk 'BEGIN{FS=\"\\t\";OFS=\"\\t\"} {key="'$1 FS $2 FS $3 FS $4; cls=$9; base[key FS cls]+=$NF; any[key]+= $NF; if(!(key in seen)){n[$4]++;seen[key]=1}; bases[$4 FS cls]+=$NF; fam[key FS $10]+=$NF} END{for(k in any){split(k,a,FS); total[a[4]]+=any[k];} for(k in bases){split(k,a,FS); print a[1],a[2],bases[k]} for(k in total) print "TOTAL",k,total[k]}'"' | sort > repeat_bases.tsv; echo '---base totals---'; grep '"'^TOTAL'"' repeat_bases.tsv; echo '---top class bases percent---'; awk '"'$1!="TOTAL"{sum[$1]+=$3; c[$1 FS $2]+=$3} END{for(k in c){split(k,a,FS); print a[1],a[2],c[k],sprintf("%.1f%%",100*c[k]/sum[a[1]])}}'"' repeat_bases.tsv | sort -k1,1 -k3,3nr | head -30"
