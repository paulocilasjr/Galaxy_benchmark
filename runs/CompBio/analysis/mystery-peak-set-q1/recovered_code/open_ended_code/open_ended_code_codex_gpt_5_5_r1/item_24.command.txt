/bin/bash -c "gzip -dc refs/DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz | awk 'NR>1{print "'$1"'"\\t\""'$2"'"\\t\""'$3"'"\\t\""'$4"'"\\t\""'$6"'"\\t\""'$10}'"' | sort -k1,1 -k2,2n > refs/dhs_index.sorted.bed
wc -l refs/dhs_index.sorted.bed
for s in S1 S2 S3 S4 S5 S6; do 
  n="'$(awk '"'END{print NR}' work/"'${s}.bed); 
  ov=$(bedtools intersect -u -a work/${s}.bed -b refs/dhs_index.sorted.bed | awk '"'END{print NR+0}'); 
  bp="'$(bedtools coverage -a work/${s}.bed -b refs/dhs_index.sorted.bed | awk '"'{sum+="'$7} END{print sum+0}'"'); 
  awk -v s=\""'$s" -v n="$n" -v ov="$ov" -v bp="$bp" '"'BEGIN{printf \"%s\\t%d\\t%d\\t%.5f\\t%d\\t%.5f\\n\",s,n,ov,ov/n,bp,bp/(500*n)}'; 
done | tee results/dhs_index_overlap.tsv"
