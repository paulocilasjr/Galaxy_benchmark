/bin/bash -c '# Best-overlap DHS sample-count summary per interval.
for s in S1 S2 S3 S4 S5 S6; do
  bedtools intersect -wao -a work/${s}.bed -b refs/dhs_index.sorted.bed '"\\
    | awk -v s=\""'$s" '"'BEGIN{OFS=\"\\t\"} {key="'$1":"$2"-"$3; if($NF>best[key]){best[key]=$NF; ns[key]=($(NF-2)=="."?0:$(NF-2)); comp[key]=$(NF-1)}} END{n=0; sum=0; zero=0; for(k in best){n++; sum+=ns[k]; if(ns[k]==0)zero++; bins[comp[k]]++} printf "%s'"\\tn=%d\\tmean_numsamples=%.3f\\tzero=%d\\tzero_frac=%.5f\\n\",s,n,sum/n,zero,zero/n; for(c in bins) print s,c,bins[c] > \"results/dhs_components_\"s\".tsv\"}'
done | tee results/dhs_numsamples_summary.tsv
curl -L --fail --silent --show-error https://hgdownload.soe.ucsc.edu/goldenPath/hg38/database/wgEncodeRegDnaseClustered.txt.gz -o refs/wgEncodeRegDnaseClustered.txt.gz
gzip -dc refs/wgEncodeRegDnaseClustered.txt.gz | awk 'BEGIN{OFS=\"\\t\"}{print "'$2,$3,$4,$1,$5,$6}'"' | sort -k1,1 -k2,2n > refs/encode_dnase_clustered.sorted.bed
for s in S1 S2 S3 S4 S5 S6; do n="'$(awk '"'END{print NR}' work/"'${s}.bed); ov=$(bedtools intersect -u -a work/${s}.bed -b refs/encode_dnase_clustered.sorted.bed | awk '"'END{print NR+0}'); awk -v s=\""'$s" -v n="$n" -v ov="$ov" '"'BEGIN{printf \"%s\\t%d\\t%d\\t%.5f\\n\",s,n,ov,ov/n}'; done | tee results/encode_dnase_clustered_overlap.tsv"
