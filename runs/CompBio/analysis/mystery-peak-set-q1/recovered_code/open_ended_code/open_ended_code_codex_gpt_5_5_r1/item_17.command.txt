/bin/bash -c 'sort -k1,1 -k2,2n refs/GRCh38-cCREs.bed > refs/GRCh38-cCREs.sorted.bed
for s in S1 S2 S3 S4 S5 S6; do 
  n=$(wc -l < work/${s}.bed); 
  ov=$(bedtools intersect -u -a work/${s}.bed -b refs/GRCh38-cCREs.sorted.bed | wc -l); 
  bp=$(bedtools coverage -a work/${s}.bed -b refs/GRCh38-cCREs.sorted.bed | awk '"'{sum+="'$NF} END{print sum+0}'"'); 
  printf \"%s\\t%s\\t%s\\t%.4f\\t%s\\t%.4f\\n\" \""'$s" "$n" "$ov" "$(awk -v o=$ov -v n=$n '"'BEGIN{print o/n}')\" \""'$bp" "$(awk -v b=$bp -v n=$n '"'BEGIN{print b/(500*n)}')\"; 
done | tee results/ccre_overlap.tsv"
