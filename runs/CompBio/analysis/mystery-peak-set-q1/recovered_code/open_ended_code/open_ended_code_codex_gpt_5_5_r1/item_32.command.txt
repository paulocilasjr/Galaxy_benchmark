/bin/bash -c 'for url in https://downloads.wenglab.org/V4/GRCh38-cCREs.bed https://downloads.wenglab.org/V4/GRCh38-cCREs.bed.gz https://downloads.wenglab.org/V4/hg38-cCREs.bed; do echo "$url"; curl -L --silent --show-error --head "$url" | head -5; done
for s in S1 S2 S3 S4 S5 S6; do 
  n=$(awk '"'END{print NR}' work/"'${s}.bed); 
  ov=$(bedtools intersect -u -a work/${s}.bed -b refs/GRCh38-cCREs.sorted.bed | awk '"'END{print NR+0}'); 
  cov="'$(bedtools coverage -a work/${s}.bed -b refs/GRCh38-cCREs.sorted.bed | awk '"'{sum+="'$7} END{print sum+0}'"'); 
  awk -v s=\""'$s" -v n="$n" -v ov="$ov" -v cov="$cov" '"'BEGIN{printf \"%s\\t%d\\t%d\\t%.5f\\t%d\\t%.5f\\n\",s,n,ov,ov/n,cov,cov/(500*n)}'; 
done | tee results/ccre_overlap_clean.tsv"
