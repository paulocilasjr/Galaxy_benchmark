/bin/bash -lc "python3 -u - <<'PY'
try:
 import fitz; print('fitz')
 d=fitz.open('dissertation.pdf'); print('pages',len(d));
 with open('dissertation.txt','w') as f:
  for i,p in enumerate(d):
   if i%20==0: print(i,flush=True)
   f.write('\\n---PAGE %d---\\n'% (i+1)); f.write(p.get_text())
except Exception as e:
 print(type(e),e)
PY
ls -lh dissertation.txt 2>&1"
