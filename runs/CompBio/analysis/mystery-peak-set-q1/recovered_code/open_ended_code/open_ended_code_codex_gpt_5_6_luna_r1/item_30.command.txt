/bin/bash -lc "python3 -m pip install --quiet pypdf && python3 - <<'PY'
from pypdf import PdfReader
reader=PdfReader('compbiobench.paper.pdf')
with open('compbiobench.paper.txt','w') as f:
 for i,p in enumerate(reader.pages):
  f.write(f'\\n---PAGE {i+1}---\\n')
  f.write(p.extract_text() or '')
print('pages',len(reader.pages))
PY
grep -in -C 4 -E 'mystery|deplet|peak set|unperturbed' compbiobench.paper.txt | head -300"
