/bin/bash -lc "python3 - <<'PY'
import requests
base='https://www.encodeproject.org/search/'
for region in ['chr1:29102-29602','chr1:450495-450995','chr1:9937-10437']:
 for typ in ['File','Experiment']:
  p={'type':typ,'assembly':'GRCh38','region':region,'assay_title':'ATAC-seq','status':'released','limit':100,'format':'json'}
  r=requests.get(base,params=p); j=r.json(); print(region,typ,r.url,'n',len(j.get('@graph',[])),'total',j.get('total'),[(x.get('accession'),x.get('file_format'),x.get('output_type')) for x in j.get('@graph',[])[:3]])
PY"
