/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
hic=pd.read_csv('ep.interactions.q1.hic.csv')
expr=pd.read_csv('ep.interactions.q1.expr.csv')
# Hic means
hic['mean_count']=hic[['count_rep1','count_rep2','count_rep3']].mean(axis=1)
print('Candidates HIC')
print(hic[hic['set']=='candidate'][['pair_id','distance_bp','mean_count','count_rep1','count_rep2','count_rep3']])
# expr means and log2fc all observations
print('\\nExpr summary all')
for pid,g in expr.groupby('pair_id'):
    c=g[g.condition=='control'].rna_count
    p=g[g.condition=='perturbed'].rna_count
    print(pid, 'ctrl', c.mean(), 'pert', p.mean(), 'log2fc', np.log2((p.mean()+1)/(c.mean()+1)), 'diff',p.mean()-c.mean())
print('\\nExpr per guide log2fc')
for pid,g in expr.groupby('pair_id'):
    print(pid)
    for guide,gg in g.groupby('guide_id'):
        c=gg[gg.condition=='control'].rna_count.mean(); p=gg[gg.condition=='perturbed'].rna_count.mean()
        print(' ',guide, round(np.log2((p+1)/(c+1)),3), round(c,1), round(p,1))
PY"
