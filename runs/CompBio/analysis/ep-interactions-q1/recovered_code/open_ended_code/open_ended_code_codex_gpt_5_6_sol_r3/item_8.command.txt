/bin/bash -lc "python3 -S -u - <<'PY'
import csv, math, statistics as st
with open('ep.interactions.q1.hic.csv') as f: h=list(csv.DictReader(f))
for r in h:
 r['d']=int(r['distance_bp']); r['s']=sum(int(r[x]) for x in ('count_rep1','count_rep2','count_rep3'))
print('HIC nearest distance')
for r in [x for x in h if x['set']=='candidate']:
 bg=[x['s'] for x in h if x['set']=='background' and .75*r['d']<=x['d']<=1.25*r['d']]
 print(r['pair_id'],'sum',r['s'],'n',len(bg),'bgmean',round(st.mean(bg),2),'upperp',round((1+sum(x>=r['s'] for x in bg))/(len(bg)+1),3),'percentile',round(sum(x<=r['s'] for x in bg)/len(bg),3))
with open('ep.interactions.q1.expr.csv') as f: e=list(csv.DictReader(f))
d={}
for r in e: d[(r['pair_id'],r['guide_id'],r['replicate'],r['condition'])]=int(r['rna_count'])
print('EXPR')
for ep in sorted({k[0] for k in d}):
 fcs=[]; gm=[]
 for g in sorted({k[1] for k in d if k[0]==ep}):
  xs=[]
  for rep in ('1','2','3'):
   xs.append(math.log2(d[(ep,g,rep,'perturbed')]/d[(ep,g,rep,'control')]))
  fcs += xs; gm.append(st.mean(xs))
 t=st.mean(fcs)/(st.stdev(fcs)/math.sqrt(len(fcs)))
 print(ep,'log2fc',round(st.mean(fcs),3),'t',round(t,2),'guide',*[round(x,3) for x in gm])
PY"
