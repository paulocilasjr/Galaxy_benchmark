/bin/bash -lc "python3 - <<'PY'
import math,statistics
cands=[
('EP1',50,-0.37), # use negative? For enhancer effect = -logFC
('EP2',27,1.225),
('EP3',58,0.206),
('EP4',20,0.807),
('EP5',5,0.469),
('EP6',13,0.249),
('EP7',7,0.534),
]
# correlate sum counts with negative logFC (enhancer effect)
x=[c[1] for c in cands]; y=[c[2] for c in cands]
r=statistics.correlation(x,y)
print('corr raw',r)
# rank corr
rx=[sorted(x).index(v)+1 for v in x]; ry=[sorted(y).index(v)+1 for v in y]
print('rank corr',statistics.correlation(rx,ry))
# linear residuals y ~ x (all)
mx=statistics.mean(x);my=statistics.mean(y)
b=sum((a-mx)*(d-my) for a,d in zip(x,y))/sum((a-mx)**2 for a in x)
a=my-b*mx
print('slope',b,'intercept',a)
for (pid,xi,yi) in cands:
 pred=a+b*xi
 print(pid,'resid',yi-pred)
# exclude EP1 (direction opposite maybe)
PY"
