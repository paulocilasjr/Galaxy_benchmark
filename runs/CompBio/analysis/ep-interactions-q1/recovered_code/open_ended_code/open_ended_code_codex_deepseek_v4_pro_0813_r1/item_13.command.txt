/bin/bash -lc "python3 - <<'PY'
# quick z from previous p approximations using scipy
from scipy import stats
pairs={'EP1':(None,2.36e-5,0.37),'EP2':(None,9e-11,-1.225),'EP3':(None,0.086,-.206),'EP4':(None,4.3e-9,-.807),'EP5':(None,3.5e-8,-.469),'EP6':(None,.00129,-.249),'EP7':(None,6.8e-7,-.534)}
z_hic={'EP1':4.92,'EP2':2.81,'EP3':7.72,'EP4':4.41,'EP5':-1.62,'EP6':-1.45,'EP7':-1.54}
for pid,(_,pabs,fc) in pairs.items():
    zabs=stats.norm.ppf(1-pabs/2)
    # zdown from one-sided t alternative less using abs p for negative? pdown=pabs/2 for negative, =1-pabs/2 for positive
    if fc<0: zdown=stats.norm.ppf(pabs/2)  # negative
    else: zdown=stats.norm.ppf(1-pabs/2) # positive
    print(pid,'zhic',z_hic[pid],'zabs',round(zabs,2),'zdown',round(zdown,2),'diff abs',round(abs(z_hic[pid]-zabs),2),'diff down',round(abs(z_hic[pid]-zdown),2))
PY"
