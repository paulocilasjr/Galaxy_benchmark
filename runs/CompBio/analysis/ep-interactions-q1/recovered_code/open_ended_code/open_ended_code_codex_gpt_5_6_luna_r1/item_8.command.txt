/bin/bash -lc "python - <<'PY'
import csv, math, statistics
with open('ep.interactions.q1.hic.csv') as f: h=list(csv.DictReader(f))
bg=[r for r in h if r['set']=='background']
for r in h:
 if r['set']"'!='"'candidate': continue
 d=int(r['distance_bp']); m=sum(int(r[f'count_rep{i}']) for i in range(1,4))/3
 near=sorted(bg,key=lambda x:abs(math.log(int(x['distance_bp'])/d)))[:20]
 nm=[sum(int(x[f'count_rep{i}']) for i in range(1,4))/3 for x in near]
 print(r['pair_id'], 'd',d,'cand',round(m,2),'near20 mean',round(statistics.mean(nm),2),'med',statistics.median(nm),'sd',round(statistics.pstdev(nm),2),'z',round((m-statistics.mean(nm))/(statistics.pstdev(nm) or 1),2))
with open('ep.interactions.q1.expr.csv') as f:e=list(csv.DictReader(f))
for p in sorted({r['pair_id'] for r in e},key=lambda x:int(x[2:])):
 ds=[]
 for g in sorted({r['guide_id'] for r in e if r['pair_id']==p}):
  x=[r for r in e if r['pair_id']==p and r['guide_id']==g]
  c=[int(r['rna_count']) for r in x if r['condition']=='control']; q=[int(r['rna_count']) for r in x if r['condition']=='perturbed']
  ds.append(math.log2(statistics.mean(q)/statistics.mean(c)))
 print(p,'log effects', [round(x,3) for x in ds], 'mean',round(statistics.mean(ds),3),'sd',round(statistics.stdev(ds),3))
PY"
