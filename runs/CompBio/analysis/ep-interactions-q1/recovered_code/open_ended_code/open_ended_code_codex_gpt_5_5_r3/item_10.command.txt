/bin/bash -lc "python3 - <<'PY'
import csv, statistics, math
from collections import defaultdict
try:
    from scipy import stats
except Exception as e:
    print('no scipy',e); stats=None
vals=defaultdict(lambda: defaultdict(list))
g=defaultdict(lambda: defaultdict(lambda: defaultdict(list)))
with open('ep.interactions.q1.expr.csv') as f:
    for r in csv.DictReader(f):
        vals[r['pair_id']][r['condition']].append(int(r['rna_count']))
        g[r['pair_id']][r['guide_id']][r['condition']].append(int(r['rna_count']))
for pid in sorted(vals):
    c=vals[pid]['control']; p=vals[pid]['perturbed']
    if stats:
        print(pid, 'welch p', stats.ttest_ind(p,c,equal_var=False).pvalue, 'mann', stats.mannwhitneyu(p,c,alternative='two-sided').pvalue)
        diffs=[]
        for gid in sorted(g[pid]):
            diffs.append(statistics.mean(g[pid][gid]['perturbed'])-statistics.mean(g[pid][gid]['control']))
        print(' guide diffs',diffs,'one-sample p',stats.ttest_1samp(diffs,0).pvalue)
PY"
