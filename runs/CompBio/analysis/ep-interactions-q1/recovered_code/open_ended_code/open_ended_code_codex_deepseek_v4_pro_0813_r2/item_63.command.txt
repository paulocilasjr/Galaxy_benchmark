/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
expr=pd.read_csv('ep.interactions.q1.expr.csv')
for p,g in expr.groupby('pair_id'):
    l=[]
    for guide,gg in g.groupby('guide_id'):
        c=gg[gg.condition=='control'].rna_count.values.astype(float); t=gg[gg.condition=='perturbed'].rna_count.values.astype(float)
        l.append(np.log2(t.mean()/c.mean()))
    l=np.array(l)
    t,pv=stats.ttest_1samp(l,0)
    print(p,'per-guide logFC',np.round(l,3),'mean',l.mean(),'t',t,'p',pv)
PY"
