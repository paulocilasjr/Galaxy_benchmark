/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for path in ['', 'data']:
    url=f'https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main/{path}?recursive=true&expand=false'
    arr=json.load(urllib.request.urlopen(url,timeout=60))
    print('PATH',path,'count',len(arr))
    for x in arr:
        p=x.get('path','')
        if any(s in p.lower() for s in ['answer','sol','expected','ground','eval']):
            print('HIT',x)
    # print top dirs/files names
    if path=='':
        print([x['path'] for x in arr][:100])
PY"
