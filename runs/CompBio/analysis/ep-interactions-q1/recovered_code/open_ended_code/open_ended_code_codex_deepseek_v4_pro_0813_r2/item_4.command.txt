/bin/bash -lc "python3 - <<'PY'
import pandas as pd
expr=pd.read_csv('ep.interactions.q1.expr.csv')
# per pair, guide mean diff, t
from scipy import stats
for p,g in expr.groupby('pair_id'):
    print('\\n',p)
    for guide, gg in g.groupby('guide_id'):
        c=gg[gg.condition=='control'].rna_count.values
        t=gg[gg.condition=='perturbed'].rna_count.values
        fc=t.mean()/c.mean()
        try:
            pv=stats.ttest_ind(t,c,equal_var=False).pvalue
        except: pv=float('nan')
        print(guide,'ctl',c.mean(),'pert',t.mean(),'ratio',round(fc,3),'p',pv)
PY"
