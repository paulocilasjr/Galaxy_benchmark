import urllib.request

# print printable strings containing likely words
import re
strings=re.findall(rb'[ -~]{8,}', data)

    t=s.decode('latin1')
    if any(k in t.lower() for k in ['answer','solution','ground','expected','eval','question','tsv','csv']):
        print(t[:300])
