/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
import statsmodels.api as sm
hic=pd.read_csv('ep.interactions.q1.hic.csv'); expr=pd.read_csv('ep.interactions.q1.expr.csv')
bg=hic[hic.set=='background']; cand=hic[hic.set=='candidate'].copy(); cand['total']=cand[['count_rep1','count_rep2','count_rep3']].sum(axis=1)
long=bg.melt(id_vars=['pair_id','distance_bp'], value_vars=['count_rep1','count_rep2','count_rep3'], var_name='rep', value_name='count'); long['logd']=np.log(long.distance_bp)
pois=sm.GLM(long['count'], sm.add_constant(long[['logd']]), family=sm.families.Poisson()).fit()
mu=pois.predict(sm.add_constant(np.log(cand.distance_bp)))
cand['enrich']=np.log2((cand['total']/3)/mu)
cand['p_up']=[1-stats.poisson.cdf(k-1,lam) for k,lam in zip(cand['total'].astype(int),mu*3)]
rows=[]
for pair in expr.pair_id.unique():
    d=expr[expr.pair_id==pair]; c=d[d.condition=='control'].sort_values(['guide_id','replicate']).rna_count.values; p=d[d.condition=='perturbed'].sort_values(['guide_id','replicate']).rna_count.values
    fc=np.mean(np.log2(p/c)); _,pv=stats.ttest_rel(np.log2(p),np.log2(c)); rows.append((pair,abs(fc),pv,fc))
ex=pd.DataFrame(rows,columns=['pair_id','absfc','p_expr','logFC'])
cand=cand.merge(ex,on='pair_id')
# z scores via -log10 p directional: contact z = -log10(p_up), high contact -> high. effect z = -log10(p_expr).
# normalize ranks maybe min-max
cand['cont_z']=-np.log10(cand['p_up']); cand['expr_z']=-np.log10(cand['p_expr'])
cand['enrich_z']=(cand['enrich']-cand['enrich'].mean())/cand['enrich'].std()
cand['effect_z']=(cand['absfc']-cand['absfc'].mean())/cand['absfc'].std()
# ideal = max each
for method in ['sum_z','min_z','product_enrich_effect','rank_sum']:
    pass
# Compute composite true-positive score using normalized log evidence and effect size
cand['sum_norm']=(cand['enrich']-cand['enrich'].min())/(cand['enrich'].max()-cand['enrich'].min()) + (cand['absfc']-cand['absfc'].min())/(cand['absfc'].max()-cand['absfc'].min())
cand['prod_rank']=(cand['cont_z'].rank() + cand['expr_z'].rank())/2
cand['enrich_rank']=cand['enrich'].rank(); cand['effect_rank']=cand['absfc'].rank()
cand['rank_sum']=cand['enrich_rank']+cand['effect_rank']
print(cand[['pair_id','total','enrich','p_up','cont_z','absfc','p_expr','expr_z','sum_norm','rank_sum']].sort_values('sum_norm').to_string(index=False))
print('\\nSort rank_sum:')
print(cand[['pair_id','enrich_rank','effect_rank','rank_sum']].sort_values('rank_sum').to_string(index=False))
# discordance = abs(rank contact - rank effect)
cand['discord']=abs(cand['enrich_rank']-cand['effect_rank'])
print('\\nDiscord:')
print(cand[['pair_id','enrich_rank','effect_rank','discord']].sort_values('discord',ascending=False).to_string(index=False))
PY"
