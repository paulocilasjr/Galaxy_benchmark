/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
expr=pd.read_csv('ep.interactions.q1.expr.csv')
for pair in sorted(expr.pair_id.unique()):
    d=expr[expr.pair_id==pair]
    print('\\nPAIR',pair)
    vals=[]
    for g,dd in d.groupby('guide_id'):
        c=dd[dd.condition=='control'].rna_count
        p=dd[dd.condition=='perturbed'].rna_count
        fc=np.log2(p.mean()/c.mean())
        t,pv=stats.ttest_ind(np.log2(p), np.log2(c), equal_var=True)
        # maybe Welch
        tw,pvw=stats.ttest_ind(np.log2(p), np.log2(c), equal_var=False)
        vals.append((g,fc,pv,pvw,c.mean(),p.mean()))
        print(g, 'log2FC %.3f'%fc, 'p %.4g'%pv, 'welch %.4g'%pvw)
    fcs=[v[1] for v in vals]; ps=[v[3] for v in vals]
    print('meanFC',np.mean(fcs),'medianFC',np.median(fcs),'comb fisher',stats.combine_pvalues(ps, method='fisher')[1])
    # paired by replicate across guides? each replicate paired. paired t log counts over 12 pairs
    c=d[d.condition=='control'].sort_values(['guide_id','replicate']).rna_count.values
    p=d[d.condition=='perturbed'].sort_values(['guide_id','replicate']).rna_count.values
    t,pv=stats.ttest_rel(np.log2(p), np.log2(c))
    print('paired overall log2',np.mean(np.log2(p/c)),'p',pv)
PY"
