/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, statsmodels.api as sm
import statsmodels.formula.api as smf
expr=pd.read_csv('ep.interactions.q1.expr.csv')
for pair in expr.pair_id.unique():
 d=expr[expr.pair_id==pair].copy()
 # condition effect with guide main effect NB
 # NegativeBinomial default alpha? Use Poisson first
 print('\\n',pair)
 for famname, fam in [('poisson',sm.families.Poisson()),('nb',sm.families.NegativeBinomial(alpha=1.0))]:
  try:
   model=smf.glm('rna_count ~ condition + C(guide_id)', data=d, family=fam).fit()
   print(famname,'cond p',model.pvalues['condition[T.perturbed]'],'coef',model.params['condition[T.perturbed]'])
  except Exception as e: print(famname,'ERR',e)
 # interaction model likelihood ratio? 
 try:
  m1=smf.glm('rna_count ~ condition + C(guide_id)', data=d, family=sm.families.Poisson()).fit()
  m2=smf.glm('rna_count ~ condition * C(guide_id)', data=d, family=sm.families.Poisson()).fit()
  from scipy import stats
  lr=2*(m2.llf-m1.llf); p=stats.chi2.sf(lr,3)
  print('interaction LR p',p)
 except Exception as e: print(e)
PY"
