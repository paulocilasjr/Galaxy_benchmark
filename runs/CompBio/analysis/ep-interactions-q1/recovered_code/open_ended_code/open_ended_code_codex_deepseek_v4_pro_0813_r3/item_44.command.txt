/bin/bash -lc "python3 - <<'PY'
# quick ranks p expr
import pandas as pd, numpy as np
from scipy import stats
expr=pd.read_csv('ep.interactions.q1.expr.csv')
rows=[]
for pair in expr.pair_id.unique():
 d=expr[expr.pair_id==pair]; c=d[d.condition=='control'].sort_values(['guide_id','replicate']).rna_count.values; p=d[d.condition=='perturbed'].sort_values(['guide_id','replicate']).rna_count.values
 fc=np.mean(np.log2(p/c)); _,pv=stats.ttest_rel(np.log2(p),np.log2(c)); rows.append((pair,pv,fc))
ex=pd.DataFrame(rows,columns=['pair','p','fc'])
ex['rank_p']=ex.p.rank(ascending=False) # high rank = least significant? ascending=False gives rank 1 largest p (least sig)
print(ex)
PY"
