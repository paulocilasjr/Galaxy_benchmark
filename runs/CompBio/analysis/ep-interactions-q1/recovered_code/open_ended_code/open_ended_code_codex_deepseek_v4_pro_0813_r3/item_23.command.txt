/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for repo in ['Genentech/compbiobench-results-v1','Genentech/compbiobench-submissions-v1']:
    url=f'https://huggingface.co/api/datasets/{repo}/tree/main?recursive=true&expand=false'
    print('REPO',repo)
    try:
        data=json.load(urllib.request.urlopen(url, timeout=30))
        for x in data:
            print(x.get('path'),x.get('type'),x.get('size'))
    except Exception as e: print('ERR',e)
PY"
