/bin/bash -lc "blastn -query top_unique_200.fa -db blastdb/16S_ribosomal_RNA -task blastn-short -max_target_seqs 20 -outfmt '6 qseqid saccver staxids sscinames pident length evalue bitscore stitle' -evalue 1000 > local_top200_16S.tsv
blastn -query top_unique_200.fa -db blastdb/LSU_prokaryote_rRNA -task blastn-short -max_target_seqs 20 -outfmt '6 qseqid saccver staxids sscinames pident length evalue bitscore stitle' -evalue 1000 > local_top200_LSU.tsv
printf '16S lines '; wc -l < local_top200_16S.tsv
printf 'LSU lines '; wc -l < local_top200_LSU.tsv
head -n 20 local_top200_16S.tsv
printf '\\nLSU\\n'
head -n 20 local_top200_LSU.tsv"
