/bin/bash -lc "awk 'BEGIN{mapped=unmapped=0} /"'^@/{next} {if(int($2/4)%2) unmapped++; else mapped++} END{print mapped, unmapped, mapped+unmapped}'"' human_cdna.sam
# Extract reads unmapped to human cDNA for a quick look at how many sampled Hydra BLAST reads are in that set
awk 'BEGIN{FS=\"\\t\"} /"'^@/{next} {if(int($2/4)%2) print $1}'"' human_cdna.sam > human_cdna_unmapped.ids
wc -l human_cdna_unmapped.ids
cut -f1 blast_api_random200_summary.tsv | grep -Fxf human_cdna_unmapped.ids | wc -l
grep -i 'Hydra vulgaris' blast_api_random200_summary.tsv | cut -f1 | grep -Fxf human_cdna_unmapped.ids | wc -l"
