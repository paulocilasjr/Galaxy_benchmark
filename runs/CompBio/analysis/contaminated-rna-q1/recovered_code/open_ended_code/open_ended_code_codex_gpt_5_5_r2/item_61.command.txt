/bin/bash -lc "python3 - <<'PY'
import requests, re, time
seq='CTGCCAGTAGCATATGCTTGTCTCAAAGATTAAGCCATGCATGTCTAAGTACGCACGGCCGGTACAGTGAAACTG'
base='https://blast.ncbi.nlm.nih.gov/Blast.cgi'
params={
 'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,
 'MEGABLAST':'on','SHORT_QUERY_ADJUST':'on','EXPECT':'1000','HITLIST_SIZE':'50','FORMAT_TYPE':'Tabular'
}
r=requests.post(base,data=params,timeout=60)
print(r.status_code)
print(r.text[:1000])
rid=re.search(r'RID = (\\S+)', r.text)
rtoe=re.search(r'RTOE = (\\d+)', r.text)
print('RID', rid.group(1) if rid else None, 'RTOE', rtoe.group(1) if rtoe else None)
if rid:
    rid=rid.group(1)
    for i in range(12):
        time.sleep(10 if i else int(rtoe.group(1) if rtoe else 5))
        q={'CMD':'Get','RID':rid,'FORMAT_TYPE':'Tabular','ALIGNMENT_VIEW':'Tabular','DESCRIPTIONS':'50','ALIGNMENTS':'50'}
        g=requests.get(base,params=q,timeout=60)
        txt=g.text
        print('poll',i,'len',len(txt), txt[:120].replace('\\n',' | '))
        if 'Status=WAITING' in txt: continue
        open('blast_api_top1.txt','w').write(txt)
        break
PY"
