/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
import gzip
seen={}
n=0
for r in SeqIO.parse(gzip.open('q1_unmapped.unclass.fq.gz','rt'),'fastq'):
    n+=1
    seen[str(r.seq)]=seen.get(str(r.seq),0)+1
print('reads', n, 'unique', len(seen))
from collections import Counter
c=Counter(seen.values())
print('duplicate counts top', c.most_common(10))
# output unique fasta
with open('q1_unmapped_unclass_unique.fa','w') as f:
    for i,(s,cnt) in enumerate(seen.items()):
        f.write(f'>r{i}_x{cnt}\\n{s}\\n')
print('wrote unique')
PY
ls -lh q1_unmapped_unclass_unique.fa"
