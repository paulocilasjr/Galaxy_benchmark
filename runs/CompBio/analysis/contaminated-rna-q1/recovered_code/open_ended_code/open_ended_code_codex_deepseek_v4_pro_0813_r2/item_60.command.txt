/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
import gzip, random
recs=list(SeqIO.parse(gzip.open('q1_unmapped_nohydra.fq.gz','rt'),'fastq'))
random.seed(123)
random.shuffle(recs)
# deduplicate roughly
seen=set(); out=[]
for r in recs:
    s=str(r.seq)
    if s not in seen:
        seen.add(s); out.append(r)
    if len(out)>=20: break
SeqIO.write(out,'sample_nohydra20.fa','fasta')
print('selected', len(out))
PY
blastn -remote -db nt -query sample_nohydra20.fa -out sample_nohydra20.out -outfmt '6 qseqid sseqid staxids pident length mismatch gapopen qstart qend sstart send evalue bitscore' -max_target_seqs 3 -evalue 1e-5 -task megablast 2>&1 | tail -5"
