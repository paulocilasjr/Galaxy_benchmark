/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]); mapq=int(p[4]);
            if flag&4: continue
            nm=999
            for x in p[11:]:
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            # store max mapq, min nm for mapq
            if name not in d or mapq>d[name][0] or (mapq==d[name][0] and nm<d[name][1]):
                d[name]=(mapq,nm)
    return d
h=best('human.sam'); m=best('mouse.sam')
cand=[]
for n,(mq,nm) in m.items():
    if mq>=60:
        hm=h.get(n)
        if hm is None:
            cand.append((n,nm,'no_human'))
        elif hm[0]<60:
            cand.append((n,nm,'human_low',hm))
print('no human',sum(1 for c in cand if c[2]=='no_human'))
print('human_low',sum(1 for c in cand if c[2]=='human_low'))
from collections import Counter
print(Counter(c[1] for c in cand if c[2]=='no_human'))
print(Counter((c[3][1]) for c in cand if c[2]=='human_low'))
for c in cand:
    if c[2]=='no_human': print(c)
PY"
