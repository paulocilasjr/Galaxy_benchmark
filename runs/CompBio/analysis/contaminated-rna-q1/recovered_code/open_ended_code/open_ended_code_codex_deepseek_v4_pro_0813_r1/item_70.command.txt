/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None; mapq=int(p[4]); cg=p[5]
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm,mapq,cg)
    return d
h=best('hg38.sam'); m=best('mouse.sam')
for diff in [0,5,10,20,30,40,50]:
    c=Counter()
    for n,mv in m.items():
        mas,mnm,mmapq,mcg=mv
        if mas is None or mas<80: continue
        hv=h.get(n)
        if hv is None:
            c['mouse_only']+=1
        else:
            has=hv[0]
            if has is None or mas-has>diff:
                c[f'mouse_win>{diff}']+=1
    print('diff',diff,c)

# For mouse winners diff20, look NM distribution and human NM
names=[]
for n,mv in m.items():
    mas,mnm,mmapq,mcg=mv
    if mas is None or mas<80: continue
    hv=h.get(n)
    if hv is not None and hv[0] is not None and mas-hv[0]>20 and (mnm is None or mnm<=4):
        names.append((n,mas,mnm,hv[0],hv[1]))
print('count diff20 nm<=4',len(names))
from collections import Counter as C
print('mouse nm',C(x[2] for x in names))
print('human nm',C(x[4] for x in names))
# examples
for x in names[:30]: print(x)
PY"
