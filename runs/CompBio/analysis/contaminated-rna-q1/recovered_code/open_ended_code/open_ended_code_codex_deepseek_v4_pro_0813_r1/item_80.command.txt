/bin/bash -lc "python3 - <<'PY'
# get mouse transcript IDs for top 20 mouse_confirm names
names=set()
with open('mouse_confirm.fa') as f:
    for line in f:
        if line.startswith('>'): names.add(line[1:].strip())
# print best mouse sam line for these names
for line in open('mouse.sam'):
    if line.startswith('@'): continue
    p=line.split('\\t')
    if p[0] in names:
        print(p[0], p[2], p[5], p[9][:30])
        names.discard(p[0])
    if not names: break
PY"
