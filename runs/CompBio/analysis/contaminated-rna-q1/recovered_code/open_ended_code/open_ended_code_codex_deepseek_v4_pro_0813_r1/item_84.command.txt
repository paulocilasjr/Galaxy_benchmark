from collections import Counter

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm)
    return d
h=best('hg38.sam'); m=best('mouse.sam')
for min_as in [120,130,140,145,150]:
    c=Counter()
    for n,mv in m.items():
        if mv[0] is None or mv[0]<min_as: continue
        hv=h.get(n)
        if hv is None or hv[0] is None:
            c['no_human']+=1
        elif mv[0]-hv[0]>=20:
            c['human_worse20']+=1
    print('min_as',min_as,c)
# count mouse full 150 and nm0
c=0
for n,mv in m.items():
    if mv[0]==150 and mv[1]==0:
        hv=h.get(n)
        if hv is None or hv[0] is None or mv[0]-hv[0]>=20: c+=1
print('mouse AS150 NM0 human worse20/no',c)
PY"
