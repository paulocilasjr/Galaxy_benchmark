/bin/bash -lc "python3 - <<'PY'
# names assigned taxids
for tax in ['10376','3050299','1396','29382','1404']:
    names=set()
    for line in open('all.kraken'):
        p=line.rstrip().split('\\t')
        if p[2]==tax: names.add(p[1])
    # human mapped names
    hmapped=set()
    with open('human.sam') as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.split('\\t')
            if not (int(p[1])&4): hmapped.add(p[0])
    print(tax,len(names),'mapped_to_human_cdna',len(names&hmapped),'not',len(names-hmapped))
PY"
