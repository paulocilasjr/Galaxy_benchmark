/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            mapq=int(p[4]); as_=None; nm=None
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d:
                d[name]=(as_,nm,mapq)
            else:
                old=d[name]
                # choose higher AS, tie lower NM/higher mapq
                if as_ is not None and old[0] is not None:
                    better = as_>old[0] or (as_==old[0] and nm is not None and old[1] is not None and nm<old[1])
                else:
                    better = as_ is not None
                if better: d[name]=(as_,nm,mapq)
    return d
h=best('human.sam'); m=best('mouse.sam')
from collections import Counter
cat=Counter(); details=[]
for n,mv in m.items():
    hv=h.get(n)
    if hv is None:
        cat['mouse_only']+=1; details.append((n,mv,hv))
    elif mv[0] is not None and (hv[0] is None or mv[0]>hv[0]):
        cat['mouse_better_AS']+=1; details.append((n,mv,hv))
print(cat)
for n,mv,hv in details[:100]: print(n,'mouse',mv,'human',hv)
PY"
