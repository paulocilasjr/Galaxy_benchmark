/bin/bash -lc "python3 - <<'PY'
import random
random.seed(123)
reads=[]
with open('unmapped.fq') as f:
    while True:
        h=f.readline().strip(); s=f.readline().strip(); p=f.readline().strip(); q=f.readline().strip()
        if not h: break
        reads.append((h,s))
idx=random.sample(range(len(reads)), 200)
with open('unmapped_sample200.fa','w') as out:
    for i in idx:
        out.write('>'+reads[i][0][1:]+'\\n'+reads[i][1]+'\\n')
PY
time blastn -remote -db nt -task megablast -query unmapped_sample200.fa -max_target_seqs 1 -outfmt '6 qseqid sseqid staxids pident length qcovs evalue bitscore' -out unmapped_sample200.remote.tsv"
