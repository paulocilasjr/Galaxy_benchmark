/bin/bash -lc "python3 - <<'PY'
import gzip, os

def write_file(src, outpath, prefix, taxid, gz=False):
    out=open(outpath,'w'); n=0; seq=[]
    opener=gzip.open(src,'rt') if gz else open(src)
    with opener as f:
        for line in f:
            if line.startswith('>'):
                if seq:
                    n+=1
                    out.write(f'>{prefix}|{n}|kraken:taxid|{taxid}\\n'+''.join(seq)+'\\n')
                    seq=[]
            else:
                seq.append(line.strip())
        if seq:
            n+=1
            out.write(f'>{prefix}|{n}|kraken:taxid|{taxid}\\n'+''.join(seq)+'\\n')
    out.close(); return n
files=[
 ('Homo_sapiens.GRCh38.cdna.all.fa.gz','customdb/library/custom_human.fa','human',9606,True),
 ('Mus_musculus.GRCm39.cdna.all.fa.gz','customdb/library/custom_mouse.fa','mouse',10090,True),
 ('Rattus_norvegicus.mRatBN7.2.cdna.all.fa.gz','customdb/library/custom_rat.fa','rat',10116,True),
 ('ebv.fa','customdb/library/custom_ebv.fa','ebv',3050299,False),
]
for src,outp,prefix,taxid,gz in files:
    n=write_file(src,outp,prefix,taxid,gz); print(outp,n)
PY
head -1 customdb/library/custom_human.fa; head -1 customdb/library/custom_ebv.fa"
