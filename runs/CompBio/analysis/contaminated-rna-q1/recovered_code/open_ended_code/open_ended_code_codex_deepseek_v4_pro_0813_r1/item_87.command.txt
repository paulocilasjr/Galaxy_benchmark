/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm)
    return d
h=best('hg38.sam'); m=best('mouse.sam'); r=best('rat.sam')
# candidate reads where mouse AS >=100 and (no human or mouse>human+20)
cand=[]
for n,mv in m.items():
    if mv[0] is None or mv[0]<100: continue
    hv=h.get(n)
    if hv is None or hv[0] is None or mv[0]-hv[0]>=20:
        cand.append(n)
print('candidate mouse>human',len(cand))
cat=Counter()
for n in cand:
    mv=m[n]; rv=r.get(n)
    if rv is None:
        cat['rat_no_aln']+=1
    elif mv[0]-rv[0]>=20:
        cat['mouse_better_rat20']+=1
    elif rv[0]-mv[0]>=20:
        cat['rat_better_mouse20']+=1
    else:
        cat['ambiguous']+=1
print(cat)
# print rat-better examples if any
for n in cand:
    mv=m[n]; rv=r.get(n)
    if rv and rv[0]-mv[0]>=20:
        print(n,'mouse',mv,'rat',rv)
PY"
