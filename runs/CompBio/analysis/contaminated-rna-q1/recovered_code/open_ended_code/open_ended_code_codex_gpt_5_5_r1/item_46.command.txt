/bin/bash -lc "{ gzip -cd refs/mus_musculus_cdna.fa.gz | awk '/"'^>/{print ">mus_musculus|" substr($0,2); next} {print}'"'; gzip -cd refs/mus_caroli_cdna.fa.gz | awk '/"'^>/{print ">mus_caroli|" substr($0,2); next} {print}'"'; } > refs/mus_musculus_caroli_cdna.fa
makeblastdb -in refs/mus_musculus_caroli_cdna.fa -dbtype nucl -out refs/mus_musculus_caroli_cdna_db >/dev/null
blastn -query results/mus_mapped_candidates.fa -db refs/mus_musculus_caroli_cdna_db -task blastn-short -max_target_seqs 5 -evalue 1e-8 -outfmt '6 qseqid sseqid pident length mismatch gapopen evalue bitscore' > results/mus_mapped_local_mus_blast.tsv
awk -F '\\t' '"'!seen[$1]++{split($2,a,"|"); print a[1]}'"' results/mus_mapped_local_mus_blast.tsv | sort | uniq -c | sort -nr"
