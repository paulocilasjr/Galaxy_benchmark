/bin/bash -c "python3 - <<'PY'
from collections import Counter, defaultdict
names={}
with open('names.dmp') as f:
    for line in f:
        line=line.rstrip('\\n').rstrip('\\t|')
        fields=line.split('\\t|\\t')
        if len(fields)>=4:
            tid=fields[0].strip(); name=fields[1].strip(); cls=fields[3].strip()
            if cls=='scientific name': names[tid]=name
print('names sample', list(names.items())[:3], names.get('9606'), names.get('6087'), names.get('9565'))
best=[]
for line in open('blast_sample_100.tsv'):
    parts=line.rstrip('\\n').split('\\t')
    q=parts[0]; taxid=parts[12].split(';')[0]; name=names.get(taxid, taxid)
    best.append((q,taxid,name,parts[2],parts[3],parts[12],parts[1]))
seen=set(); c=Counter(); per=defaultdict(list)
for q,taxid,name,pident,length,staxids,sseqid in best:
    if q not in seen:
        seen.add(q); c[(taxid,name)]+=1; per[(taxid,name)].append((q,pident,length,sseqid))
print('top-hit taxid counts:')
for (taxid,name),cnt in c.most_common(30):
    print(cnt, taxid, name)
    for item in per[(taxid,name)][:5]: print('   ',item)
PY"
