/bin/bash -c "python3 - <<'PY'
from pathlib import Path
ids=set()
with open('unmapped_kraken.out') as f:
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        if parts[0] in ('U',):
            ids.add(parts[1])
print('unclassified ids', len(ids))
# extract first N ids preserving fastq order
N=1000
out=[]
count=0
with open('unmapped.fq') as f:
    while True:
        header=f.readline()
        if not header: break
        seq=f.readline(); plus=f.readline(); qual=f.readline()
        rid=header[1:].split()[0]
        if rid in ids:
            out.append((rid,seq.rstrip('\\n'),qual.rstrip('\\n')))
            count+=1
            if count>=N: break
with open('sample_unclassified_1000.fq','w') as f:
    for rid,seq,qual in out:
        f.write(f'@{rid}\\n{seq}\\n+\\n{qual}\\n')
print('wrote', len(out))
# also create fasta first 100 for blast
with open('sample_unclassified_100.fa','w') as f:
    for rid,seq,qual in out[:100]:
        f.write(f'>{rid}\\n{seq}\\n')
PY
wc -l sample_unclassified_1000.fq sample_unclassified_100.fa; head -n 8 sample_unclassified_100.fa"
