/bin/bash -lc "python3 - <<'PY'
import re, xml.etree.ElementTree as ET
from collections import Counter,defaultdict
root=ET.parse('unmapped_sample.nt.xml').getroot()
pat=re.compile(r'\\b([A-Z][a-z]+ [a-z]{3,})\\b')
counts=Counter(); stats=defaultdict(list)
for it in root.findall('.//Iteration'):
    hits=it.findall('./Iteration_hits/Hit')
    if not hits: continue
    h=hits[0]
    d=h.findtext('Hit_def') or ''
    m=pat.search(d)
    species=m.group(1) if m else 'NO_SPECIES'
    hs=h.find('./Hit_hsps/Hsp')
    ident=int(hs.findtext('Hsp_identity')) if hs is not None else 0
    alen=int(hs.findtext('Hsp_align-len')) if hs is not None else 0
    bits=float(hs.findtext('Hsp_bit-score')) if hs is not None else 0
    counts[species]+=1
    stats[species].append((ident,alen,bits,d,it.findtext('Iteration_query-def')))
for sp,n in counts.most_common(50):
    a=stats[sp]
    print(f'{n:4d} {sp:30s} ident={sum(x[0] for x in a)/n:.1f}/{sum(x[1] for x in a)/n:.1f} full={sum(x[0]==x[1] for x in a)}')
PY"
