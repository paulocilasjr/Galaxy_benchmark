/bin/bash -lc "samtools view -F 2308 hydra_alignments.bam | awk '"'$6=="75M"{nm=999; for(i=12;i<=NF;i++) if($i~/''^NM:i:/){split($i,a,":"); nm=a[3]} if(nm==0)print $1}'"' > hydra_exact_full_ids.txt
samtools view -b -N hydra_exact_full_ids.txt hydra_alignments.bam | samtools fastq -n -0 hydra_exact_full_reads.fq.gz - 2>/dev/null
minimap2 -t 6 -ax sr Homo_sapiens.GRCh38.cdna.all.fa.gz hydra_exact_full_reads.fq.gz | samtools view -c -F 2308 -"
