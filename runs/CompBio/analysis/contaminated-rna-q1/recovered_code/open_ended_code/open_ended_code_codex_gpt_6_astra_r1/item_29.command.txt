import collections
human=set()
for fn in ['human.paf','human.ncrna.paf','human.rrna.paf','human.mt.paf']:
 human.update(line.split()[0] for line in open(fn))
hydra=set();genes=set();exact=set()

 r=line.split();hydra.add(r[0]);genes.add(r[5])
 if int(r[9])==75 and int(r[10])==75:exact.add(r[0])
print('Hydra matches:',len(hydra),'transcripts:',len(genes),'full exact:',len(exact),'without human match:',len(hydra-human),'exact without human:',len(exact-human))
print('reads weighted',sum(int(x.split('_n')[1]) for x in hydra))
