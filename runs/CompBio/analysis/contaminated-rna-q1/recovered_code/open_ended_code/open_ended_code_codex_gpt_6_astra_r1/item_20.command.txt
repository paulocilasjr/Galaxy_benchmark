/bin/bash -lc "cat > summarize_blast.py <<'PY'
import xml.etree.ElementTree as ET,sys,collections
root=ET.parse(sys.argv[1]).getroot()
counts=collections.Counter()
for it in root.findall('.//Iteration'):
 query=it.findtext('Iteration_query-def'); hits=it.findall('./Iteration_hits/Hit')
 print('\\n'+query)
 for hit in hits[:5]:
  hsp=hit.find('./Hit_hsps/Hsp'); title=hit.findtext('Hit_def'); acc=hit.findtext('Hit_accession')
  print(acc,hsp.findtext('Hsp_identity')+'/'+hsp.findtext('Hsp_align-len'),hsp.findtext('Hsp_bit-score'),title[:230])
 if hits:counts[hits[0].findtext('Hit_def').split()[0]]+=1
print(counts)
PY
ls -lh *.xml *.tsv"
