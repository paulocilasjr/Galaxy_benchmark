/bin/bash -lc "cat > remote_blast.py <<'PY'
import urllib.request,urllib.parse,time,re,sys,json
query=open(sys.argv[1]).read();prefix=sys.argv[2]
data=urllib.parse.urlencode(dict(CMD='Put',PROGRAM='blastn',DATABASE='nt',QUERY=query,MEGABLAST='on',HITLIST_SIZE='5',EXPECT='0.0001',WORD_SIZE='28')).encode()
req=urllib.request.Request('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=data)
response=urllib.request.urlopen(req,timeout=120).read().decode();open(prefix+'.submit','w').write(response)
rid=re.search(r'RID = (\\S+)',response)
if not rid: print(response[:2000]);sys.exit(1)
rid=rid.group(1);print('RID',rid,flush=True)
while True:
 time.sleep(30)
 url='https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode(dict(CMD='Get',RID=rid,FORMAT_TYPE='XML'))
 try:response=urllib.request.urlopen(url,timeout=120).read().decode()
 except Exception as e:print(e,flush=True);continue
 if '<BlastOutput' in response:open(prefix+'.xml','w').write(response);print('DONE',flush=True);break
 print(re.findall(r'Status=\\w+',response) or response[:100],flush=True)
 if 'Status=FAILED' in response or 'Status=UNKNOWN' in response:break
PY
python3 remote_blast.py unmapped.sample.fa unmapped.sample"
