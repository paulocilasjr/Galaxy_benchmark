/bin/bash -lc "awk '{print "'$1}'"' work/to_human_tx.paf | sort -u > work/human_tx_mapped.ids
# Extract unmapped reads as FASTA and count duplicate sequences
gzip -dc contaminated.rna.q1.fq.gz | awk 'NR==FNR{m["'$1]=1; next} NR%4==1{h=substr($0,2)} NR%4==2{if(!(h in m)) print ">"h"'"\\n\""'$0}'"' work/human_tx_mapped.ids - > work/unmapped_human_tx.fa
awk 'BEGIN{RS=\">\"; FS=\"\\n\"} NR>1{seq=\"\"; for(i=2;i<=NF;i++) seq=seq "'$i; c[seq]++} END{for(s in c) print c[s],s}'"' work/unmapped_human_tx.fa | sort -nr > work/unmapped_seq_counts.txt
wc -l work/human_tx_mapped.ids work/unmapped_human_tx.fa work/unmapped_seq_counts.txt && head -30 work/unmapped_seq_counts.txt"
