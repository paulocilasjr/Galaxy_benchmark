/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
x<-readRDS('tissue.fibroblast.q1.rds'); genes<-rownames(x); cells<-colnames(x)
sel<-order(as.numeric(x['Cxcl12',]),decreasing=T)[1:100]
mark<-intersect(c('Cxcl12','Cxcl14','Ccl19','Ccl21a','Il7','Cxcl13','Pdpn','Tmem176a','Tmem176b','Lepr','Serpina3g','Spp1','Col1a1','Col1a2','Col11a1','Mmp13','Runx2','Clec11a','Tnc','Ibsp','Adipoq','Esm1','Grem1','Kitl','Pappa','Hp','Csf1','Igfbp4','Igfbp5','Gpx3','Mgp','Pdgfrb','Cdh11','Gja1'),genes)
y<-x[mark,sel]
cat(paste(c('cell',mark),collapse='\\t'),'\\n',sep='')
for(k in seq_along(sel)) cat(c(cells[sel[k]],as.numeric(y[,k])),sep='\\t',fill=TRUE)
cat('\\nTarget rank by Cxcl12',rank(-as.numeric(x['Cxcl12',]))[match('Cell_10369_WXV',cells)],'of',length(cells),'\\n')
RS"
