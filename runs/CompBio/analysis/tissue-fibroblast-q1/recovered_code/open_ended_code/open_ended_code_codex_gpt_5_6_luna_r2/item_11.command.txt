/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
x <- readRDS(\"tissue.fibroblast.q1.rds\")
cs <- Matrix::colSums(x)
# tf-idf weights downweight ubiquitous genes
ng <- ncol(x)
df <- Matrix::rowSums(x > 0)
w <- log1p(ng / pmax(df, 1))
z <- x * w
z <- z %*% Diagonal(x=1/sqrt(Matrix::colSums(z"'^2)))
for (bc in c("Cell_11249_MCI", "Cell_10369_WXV")) {
  j <- match(bc, colnames(x))
  sim <- as.numeric(crossprod(z[,j], z))
  sim[j] <- -Inf
  ii <- order(sim, decreasing=TRUE)[seq_len(50)]
  cat("'"\\nTARGET\", bc, \"\\n\")
  print(data.frame(rank=seq_along(ii), barcode=colnames(x)[ii], sim=round(sim[ii],4), total=cs[ii], detected=Matrix::colSums(x[,ii]>0)), row.names=FALSE)
}
RS"
