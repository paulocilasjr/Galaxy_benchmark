/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
query <- readRDS('tissue.fibroblast.q1.rds')
source_obj <- readRDS('mouse_steady_atlas.rds')
att <- attributes(source_obj)
counts <- attributes(att"'$assays[[1]])$counts
md <- att$meta.data
q_genes <- attr(query,'"'Dimnames')[[1]]
s_genes <- attr(counts,'Dimnames')[[1]]
cat('same genes:', identical(q_genes, s_genes), '\\n')
q_cells <- attr(query,'Dimnames')[[2]]
targets <- c('Cell_11249_MCI','Cell_10369_WXV')
# source metadata rownames should equal colnames
cat('metadata aligns:', identical(rownames(md), attr(counts,'Dimnames')[[2]]), '\\n')
# precompute nCount and nFeature from metadata if available
s_sum <- md"'$nCount_RNA
s_nnz <- md$nFeature_RNA
for (tar in targets) {
  qi <- match(tar, q_cells)
  qv <- query[, qi]
  qsum <- sum(qv)
  qnnz <- length(qv@i)
  cand <- which(s_sum == qsum & s_nnz == qnnz)
  cat('"'\\nTARGET', tar, 'qsum', qsum, 'qnnz', qnnz, 'candidates', length(cand), '\\n')
  matches <- integer()
  # compare slot pattern and counts for candidates
  q_i <- qv@i; q_x <- qv@x
  for (j in cand) {
    st <- counts@p[j] + 1L; en <- counts@p[j+1L]
    if ((en - st + 1L) "'!= qnnz) next
    if (qnnz == 0) next
    if (identical(counts@i[st:en], q_i) && identical(counts@x[st:en], q_x)) matches <- c(matches, j)
  }
  cat('"'matches', length(matches), '\\n')
  if (length(matches)>0) {
    print(data.frame(source_cell=colnames(counts)[matches], Tissue=md"'$Tissue[matches], ClustName=md$ClustName[matches], nCount=md$nCount_RNA[matches], nFeature=md$nFeature_RNA[matches]), row.names=FALSE)
  } else {
    # if no exact, show closest candidates by overlap
    cat('"'no exact matches\\n')
  }
}
RS"
