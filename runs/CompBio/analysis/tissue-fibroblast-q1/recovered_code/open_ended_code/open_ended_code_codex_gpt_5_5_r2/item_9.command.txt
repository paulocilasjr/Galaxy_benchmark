/bin/bash -lc "Rscript - <<'RS'
obj <- readRDS('tissue.fibroblast.q1.rds')
dn <- attr(obj,'Dimnames')
cat(length(dn), length(dn[[1]]), length(dn[[2]]), '\\n')
cat('genes:', paste(head(dn[[1]],20), collapse=', '), '\\n')
cat('cells:', paste(head(dn[[2]],5), collapse=', '), '\\n')
# Check key tissue marker genes for targets
markers <- c('Pdgfra','Col1a1','Col1a2','Dcn','Lum','Pi16','Postn','Ccl19','Ccl21a','Ccl21b','Ccl21c','Ccl19','Cxcl13','Pdpn','Lyve1','Rgs5','Acta2','Myh11','Tagln','Tnmd','Scx','Mkx','Col22a1','Tnxb','Myod1','Myog','Myl1','Myl2','Tnnt3','Des','Pax7','Alb','Apoa1','Krt14','Krt5','Krt10','Krt17','Epcam','Muc2','Reg3g','Fabp1','Lyz1','Sftpc','Sftpa1','Scgb1a1','Cyp2e1','Hbb-bs','Hba-a1','Rbp4','Amy2a5','Prss2','Pnlip','Ins1','Ins2','Adipoq','Plin1','Lpl','Fabp4','Pparg','Msln','Wt1','Upk3b','C3','Ccl2','Vcam1','Icam1','Cd74','H2-Aa','Clec3b','Comp','Sparc','Mgp','Cox7a1','Tnnt2','Myh6','Myh7','Nppa','Nppb','Myl7','Myl2','Dpt','Coch','Wif1','Efemp1','Sost','Bglap','Ibsp','Col10a1','Mmp13','Spp1','Cd52','Ptprc','Cd79a','Ms4a1','Cd3d','Fcer2a','Ccl5')
markers <- unique(markers[markers %in% dn[[1]]])
targets <- c('Cell_11249_MCI','Cell_10369_WXV')
mat <- obj[match(markers,dn[[1]]), match(targets,dn[[2]]), drop=FALSE]
rownames(mat) <- markers; colnames(mat) <- targets
nz <- which(as.matrix(mat)>0, arr.ind=TRUE)
if (nrow(nz)>0) {
  df <- data.frame(gene=rownames(mat)[nz[,1]], cell=colnames(mat)[nz[,2]], count=as.matrix(mat)[nz])
  print(df[order(df"'$cell, -df$count),], row.names=FALSE)
} else cat('"'no marker expression\\n')
RS"
