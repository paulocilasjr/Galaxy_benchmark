/bin/bash -lc "Rscript - <<'RS'
cat('file size GB:', file.info('mouse_steady_atlas.rds')"'$size/1024''^3, '"'\\n')
obj <- readRDS('mouse_steady_atlas.rds')
cat('class:', paste(class(obj), collapse=','), '\\n')

  md <- slot(obj, 'meta.data')
  cat('metadata dim:', paste(dim(md), collapse=' x '), '\\n')
  print(colnames(md))
  print(head(md[,seq_len(min(ncol(md),20)),drop=FALSE]))
}
cat('object names:', paste(names(obj), collapse=','), '\\n')
RS"
