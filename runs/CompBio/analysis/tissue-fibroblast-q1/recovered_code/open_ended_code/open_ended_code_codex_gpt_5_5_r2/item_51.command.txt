/bin/bash -lc "Rscript - <<'RS'
obj <- readRDS('mouse_steady_atlas.rds')
cat('isS4', isS4(obj), '\\n')
cat('attributes:', paste(names(attributes(obj)), collapse=','), '\\n')
print(str(attributes(obj), max.level=1))
# try low-level slots via attributes
att <- attributes(obj)
for (nm in names(att)) {
  cat('ATTR', nm, 'class', paste(class(att[[nm]]), collapse=','), 'length', length(att[[nm]]), '\\n')
}
RS"
