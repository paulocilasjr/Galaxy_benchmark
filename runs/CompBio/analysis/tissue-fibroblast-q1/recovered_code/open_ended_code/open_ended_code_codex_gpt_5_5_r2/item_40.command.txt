/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
obj <- readRDS('tissue.fibroblast.q1.rds')
dn <- attr(obj,'Dimnames')
nums <- as.integer(sub('Cell_([0-9]+)_.*','\\\\1', dn[[2]]))
cat('range', range(nums), 'unique', length(unique(nums)), 'n', length(nums), 'missing', sum("'!(1:length(nums) %in% nums)), '"'\\n')
cat('cor order col index vs num', cor(seq_along(nums), nums), '\\n')
# Aggregate expression into numeric bins of 500 for marker genes
markers <- c('Adipoq','Lpl','Fabp4','Plin1','Ccl19','Ccl21a','Cxcl13','Lyve1','Tnmd','Scx','Mkx','Myod1','Myog','Myl1','Tnnt3','Tnnt2','Myh6','Myh7','Nppa','Alb','Apoa1','Cyp2e1','Sftpc','Sftpa1','Scgb1a1','Krt14','Krt5','Krt10','Epcam','Muc2','Reg3g','Fabp1','Amy2a5','Prss2','Pnlip','Ins1','Ins2','Sost','Bglap','Ibsp','Mmp13','Mgp','Vcam1','Icam1','Msln','Wt1','Upk3b','Pdgfra','Dcn','Col1a1','Col1a2','Lum')
markers <- markers[markers %in% dn[[1]]]
ord <- order(nums)
X <- obj[match(markers,dn[[1]]), ord]
# normalize by colsum CPM/log? do raw detection per bin
cs <- Matrix::colSums(obj[,ord])
bins <- cut(nums[ord], breaks=seq(0, ceiling(max(nums)/500)*500, by=500), include.lowest=TRUE)
for (g in markers) {
  vals <- as.numeric(X[match(g,markers),])
  mean_by <- tapply(vals>0, bins, mean)
  maxbin <- names(which.max(mean_by))
  if (max(mean_by) > 0.02) cat(g, 'max', round(max(mean_by),3), maxbin, '\\n')
}
RS"
