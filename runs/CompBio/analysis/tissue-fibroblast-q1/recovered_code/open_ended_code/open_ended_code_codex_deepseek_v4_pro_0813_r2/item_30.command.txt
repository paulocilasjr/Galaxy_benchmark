/bin/bash -lc "curl -sL 'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/compbiobench.v1.tsv' -o compbiobench.v1.tsv && wc -c compbiobench.v1.tsv && python3 - <<'PY'
import csv
fn='compbiobench.v1.tsv'
with open(fn, encoding='utf-8') as f:
    for i,row in enumerate(csv.reader(f, delimiter='\\t')):
        if i==0:
            print(row)
        if i<5: print(row)
        if 'tissue.fibroblast' in '\\t'.join(row).lower() or 'fibroblast' in '\\t'.join(row).lower():
            print('MATCH',i,row)
PY"
