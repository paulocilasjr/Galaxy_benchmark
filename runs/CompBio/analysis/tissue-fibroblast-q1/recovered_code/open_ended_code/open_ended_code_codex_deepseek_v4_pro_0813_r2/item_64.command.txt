/bin/bash -lc "Rscript -e '
library(Seurat)
full<-readRDS(\"Mouse_SS_Fibro.RDS\")
q<-readRDS(\"tissue.fibroblast.q1.rds\")
cat(\"loaded\\n\")
targets<-c(\"Cell_11249_MCI\",\"Cell_10369_WXV\")
qc<-GetAssayData(full,slot=\"counts\")
# ensure same genes and order
cat(\"genes identical\", identical(rownames(q), rownames(qc)), \"query dim\", dim(q), \"full dim\", dim(qc),\"\\n\")
for(cc in targets){
  v<-q[,cc]
  s<-sum(v); nz<-sum(v>0)
  cat(\"\\nTarget\",cc,\"sum\",s,\"nz\",nz,\"\\n\")
  # candidate by colSums may be memory; compute full sums once
}
fs<-Matrix::colSums(qc)
cat(\"full sums computed\\n\")
for(cc in targets){
  v<-q[,cc]; s<-sum(v)
  idx<-which(fs==s)
  cat(cc,\"candidates total\",length(idx),\"\\n\")
  if(length(idx)>0 && length(idx)<=5000){
    # compare sparse vector equality for candidates
    found<-which(vapply(idx, function(j) isTRUE(all.equal(v, qc[,j], check.attributes=FALSE)), logical(1)))
    cat(\"exact matches\",length(found),\"\\n\")
    if(length(found)>0) print(data.frame(idx=idx[found], cell=colnames(qc)[idx[found]], tissue=full"'$Tissue[idx[found]], clust=full$ClustName[idx[found]]))
  } else if(length(idx)>5000){cat("too many candidates; refining by nz'"\\n\")}
}
'"
