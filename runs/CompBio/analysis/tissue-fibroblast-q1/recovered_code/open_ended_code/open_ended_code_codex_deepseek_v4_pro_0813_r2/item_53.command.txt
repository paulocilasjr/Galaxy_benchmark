/bin/bash -lc "curl -L -s 'https://www.fibroxplorer.com/download' -o fibrox_download.html && grep -o -i 'href=\"["'^"]*"'"\\|data-url=\"["'^"]*"'"\\|drive["'^<]*'"\\|download["'^<]*'"' fibrox_download.html | head -200; python3 - <<'PY'
import re,html
s=open('fibrox_download.html',encoding='utf-8',errors='ignore').read()
# print text
txt=re.sub(r'<script.*?</script>',' ',s,flags=re.S)
txt=re.sub(r'<style.*?</style>',' ',txt,flags=re.S)
txt=re.sub(r'<["'^>]+>'"',' ',txt)
txt=html.unescape(txt)
txt=re.sub(r'\\s+',' ',txt)
print(txt[:5000])
PY"
