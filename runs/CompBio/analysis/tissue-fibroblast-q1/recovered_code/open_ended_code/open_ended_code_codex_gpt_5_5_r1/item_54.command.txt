/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
local <- readRDS('tissue.fibroblast.q1.rds')
full <- readRDS('Mouse_SS_Fibro.RDS')
md <- attr(full,'meta.data')
counts <- attr(attr(full,'assays')"'$RNA,'"'counts')
stopifnot(identical(rownames(local), rownames(counts)))
targets <- c('Cell_11249_MCI','Cell_10369_WXV')
for (target in targets) {
  tv <- local[,target,drop=FALSE]
  nCount <- Matrix::colSums(tv)[1]
  nFeature <- length(tv@x)
  cand <- which(md"'$nCount_RNA == nCount & md$nFeature_RNA == nFeature)
  cat('"'\\nTARGET', target, 'nCount', nCount, 'nFeature', nFeature, 'candidates', length(cand), '\\n')
  exact <- c()
  if (length(cand)) {
    for (idx in cand) {
      fv <- counts[,idx,drop=FALSE]
      if (identical(tv@i, fv@i) && identical(as.numeric(tv@x), as.numeric(fv@x))) exact <- c(exact, idx)
    }
  }
  cat('exact matches', length(exact), '\\n')
  if (length(exact)) {
    print(data.frame(cell=rownames(md)[exact], md[exact,], row.names=NULL))
  } else if (length(cand) > 0) {
    print(head(data.frame(cell=rownames(md)[cand], md[cand,], row.names=NULL),20))
  }
}
RS"
