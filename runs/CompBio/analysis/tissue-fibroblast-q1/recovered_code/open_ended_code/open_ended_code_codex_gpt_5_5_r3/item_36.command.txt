/bin/bash -lc "Rscript -e '
obj <- readRDS(\"Mouse_SS_Fibro.RDS\")
assays <- attr(obj, \"assays\")
cat(\"assays class\", class(assays), \"names\", names(assays), \"\\n\")
rna <- assays"'$RNA
cat("rna class", paste(class(rna), collapse=","), " isS4", isS4(rna), "'"\\n\")
cat(\"rna attrs\", paste(names(attributes(rna)), collapse=\",\"), \"\\n\")
print(try(slotNames(rna), silent=TRUE))
for (nm in names(attributes(rna))) {
  val <- attr(rna, nm)
  cat(\"ATTR\", nm, \"class\", paste(class(val),collapse=\",\"), \"dim\", paste(dim(val), collapse=\"x\"), \"len\", length(val), \"\\n\")
}
'"
