/bin/bash -lc "Rscript -e '
library(Matrix)
local <- readRDS(\"tissue.fibroblast.q1.rds\")
obj <- readRDS(\"Mouse_SS_Fibro.RDS\")
md <- attr(obj, \"meta.data\")
ref <- attr(attr(obj, \"assays\")"'$RNA, "counts")
cat("local dim", paste(dim(local), collapse="x"), "ref dim", paste(dim(ref), collapse="x"), "'"\\n\")
cat(\"rownames identical\", identical(rownames(local), rownames(ref)), \"\\n\")
cat(\"target rownames missing\", sum("'!(rownames(local) %in% rownames(ref))), "'"\\n\")
targets <- c(\"Cell_11249_MCI\",\"Cell_10369_WXV\")
for (t in targets) {
  v <- local[, t]
  nc <- sum(v); nf <- sum(v "'!= 0)
  cand <- rownames(md)[md$nCount_RNA == nc & md$nFeature_RNA == nf]
  cat("TARGET", t, "nCount", nc, "nFeature", nf, "candidates", length(cand), "'"\\n\")
  exact <- character()
  if (length(cand) > 0) {
    for (cc in cand) {
      rv <- ref[, cc]
      if (identical(as.numeric(rv@x), as.numeric(v@x)) && identical(rv@i, v@i) && identical(rv@p, v@p) && identical(dim(rv), dim(v))) exact <- c(exact, cc)
      # fallback if sparse internals differ
      else if (length(rv@x) == length(v@x) && all(rv@i == v@i) && all(rv@x == v@x)) exact <- c(exact, cc)
    }
  }
  cat(\"exact\", length(exact), paste(exact, collapse=\",\"), \"\\n\")
  if (length(exact)) print(md[exact,,drop=FALSE])
}
'"
