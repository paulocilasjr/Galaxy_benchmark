/bin/bash -lc "python - <<'PY'
from pgenlib import PgenReader
import numpy as np
r=PgenReader(b'match.genotypes.1.3.pgen')
for sm in [0,1]:
 for shape in [(3,175),(175,3),(3*175,)]:
  a=np.empty(shape,dtype=np.int8)
  try:
   ret=r.read_range(0,3,a,sample_maj=sm); print('sm',sm,'shape',shape,'ret',ret,'first',a.reshape(-1)[:10])
  except Exception as e: print('sm',sm,'shape',shape,'ERR',type(e).__name__,e)
r.close()
PY"
