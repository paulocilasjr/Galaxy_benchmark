/bin/bash -c "python3 - <<'PY'
import pgenlib, numpy as np
r=pgenlib.PgenReader(b'match.genotypes.1.1.pgen')
print(r.get_raw_sample_ct(), r.get_variant_ct())
for dt in [np.uint32, np.int32, np.uint64, np.int64]:
    idx=np.array([0],dtype=dt)
    buf=np.empty([4,1],dtype=np.uint32)
    try:
        r.read_list(idx,buf)
        print('ok',dt,buf)
    except Exception as e:
        print('fail',dt,e)
PY"
