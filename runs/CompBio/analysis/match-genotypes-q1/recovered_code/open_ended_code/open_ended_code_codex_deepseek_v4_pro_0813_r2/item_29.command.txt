import pgenlib, pandas as pd, numpy as np, itertools

def read_pvar(pre):
    p=pd.read_csv(pre+'.pvar', sep='\\t', comment='#', names=['CHROM','POS','ID','REF','ALT'], dtype={'CHROM':str,'POS':int})
    return p

def read_genos(pre, variant_subset):
    r=pgenlib.PgenReader(pre.encode()+b'.pgen')
    ids=[]
    with open(pre+'.psam') as f:
        for line in f:
            line=line.strip()
            if line.startswith('#') or not line: continue
            ids.append(line)
    ns=r.get_raw_sample_ct()
    idx=np.array(variant_subset,dtype=np.uint32)
    buf=np.empty([idx.size, ns], dtype=np.int32)
    r.read_list(idx, buf)
    return ids,buf

p1=read_pvar('match.genotypes.1.1')
p3=read_pvar('match.genotypes.1.3')
lo,hi=133250000,133262000
idx3=p3.index[(p3.POS>=lo)&(p3.POS<=hi)].to_numpy()
ids3,g3=read_genos('match.genotypes.1.3',idx3)
sub3=p3.iloc[idx3].reset_index(drop=True)
idx1=p1.index[(p1.POS>=lo)&(p1.POS<=hi)].to_numpy()
ids1,g1=read_genos('match.genotypes.1.1',idx1)
sub1=p1.iloc[idx1].reset_index(drop=True)
print('reference',g3.shape,'target',g1.shape)
