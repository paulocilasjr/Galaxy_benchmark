/bin/bash -lc "python3 - <<'PY'
import numpy as np, pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
D=np.load('abo_aligned.npz',allow_pickle=True)
rX=D['rX']; tX=D['tX']; rcols=[tuple(x) for x in D['rcols']]; tcols=[tuple(x) for x in D['tcols']]; ts=D['ts']; rs=D['rs']
L=np.load('abo_ref_labels.npz',allow_pickle=True); labels=L['labels']; Ocnt=L['Ocnt']; Bcnt=L['Bcnt']
# use features with MAF >0.01
mask=np.apply_along_axis(lambda x: len(set(x))>1,0,rX)
# standardize? tree no need
rf=RandomForestClassifier(n_estimators=500,random_state=0,class_weight='balanced').fit(rX[:,mask],labels)
print('RF train acc',rf.score(rX[:,mask],labels))
print('RF target preds',rf.predict(tX[:,mask]))
print('RF target proba')
print(pd.DataFrame(rf.predict_proba(tX[:,mask]),index=ts,columns=rf.classes_))
# O classifier and B classifier probabilities
clfO=RandomForestClassifier(n_estimators=500,random_state=1).fit(rX[:,mask],Ocnt)
clfB=RandomForestClassifier(n_estimators=500,random_state=2).fit(rX[:,mask],Bcnt)
print('Ocnt target',clfO.predict(tX[:,mask]),clfO.predict_proba(tX[:,mask]))
print('Bcnt target',clfB.predict(tX[:,mask]),clfB.predict_proba(tX[:,mask]))
# Logistic multinomial
lr=LogisticRegression(max_iter=1000, multi_class='multinomial', penalty=None).fit(StandardScaler().fit_transform(rX[:,mask]),labels)
print('LR train',lr.score(StandardScaler().fit_transform(rX[:,mask]),labels))
print('LR target',lr.predict(StandardScaler().fit_transform(tX[:,mask])))
print(pd.DataFrame(lr.predict_proba(StandardScaler().fit_transform(tX[:,mask])),index=ts,columns=lr.classes_))
PY"
