import csv, sys, collections
import numpy as np

def read_pvar_pos(fn, ids=False):
    # returns mapping rawcolname -> (pos, ref_allele, id/pos)
    # rawcolname = id_ref or pos_ref
    m=collections.OrderedDict()
    with open(fn) as f:
        for line in f:
            if line.startswith('#'): continue
            parts=line.rstrip('\\n').split('\\t')
            chrom,pos,vid,ref,alt=parts[0],parts[1],parts[2],parts[3],parts[4]
            key=vid+'_'+ref
            # For target IDs can duplicate; for ref pos unique maybe also duplicates with same ref
            if key not in m:
                m[key]=(pos,ref,vid)
    return m

def read_raw(fn, pvar_map):

        header=f.readline().rstrip('\\n').split('\\t')
        # first six are FID IID PAT MAT SEX PHENOTYPE
        fixed=header[:6]
        colnames=header[6:]
