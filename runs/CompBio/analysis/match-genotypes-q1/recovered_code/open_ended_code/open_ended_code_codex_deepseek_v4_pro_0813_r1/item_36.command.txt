/bin/bash -lc "python3 - <<'PY'
import numpy as np,pandas as pd
D=np.load('abo_aligned.npz',allow_pickle=True)
rX=D['rX']; tX=D['tX']; rcols=[tuple(x) for x in D['rcols']]; tcols=[tuple(x) for x in D['tcols']]; ts=D['ts']; rs=D['rs']
L=np.load('abo_ref_labels.npz',allow_pickle=True); Ocnt=L['Ocnt']; Bcnt=L['Bcnt']
# find all tags with r2=1 for B and top O r2 >0.9
btags=[]; otags=[]
for j,c in enumerate(rcols):
 x=rX[:,j].astype(float)
 if np.std(x)==0: continue
 rb=np.corrcoef(x,Bcnt)[0,1]**2
 ro=np.corrcoef(x,Ocnt)[0,1]**2
 if rb>0.999: btags.append((c,x,rb))
 if ro>0.9: otags.append((c,x,ro))
print('B perfect tags',len(btags))
for c,x,rb in btags[:30]:
 ti=next(i for i,cc in enumerate(tcols) if cc[0]==c[0])
 print(c[0],c[3], 'ref vals', sorted(set(rX[:,j])) if False else '', 'target',tX[:,ti])
print('O tags >0.9',len(otags))
for c,x,ro in otags:
 ti=next(i for i,cc in enumerate(tcols) if cc[0]==c[0])
 print(c[0],c[3],'r2',ro,'target',tX[:,ti],'crosstab')
 print(pd.crosstab(x,Ocnt).to_dict())
PY"
