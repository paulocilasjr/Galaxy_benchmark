/bin/bash -lc "python - <<'PY'
import json
p=44907187
hits=[]
with open('GRCh38-cCREs-v4.bed') as f:
 for line in f:
  fields=line.rstrip('\\n').split('\\t')
  if fields[0]=='chr19' and int(fields[1]) < p and int(fields[2]) >= p:
   print(line.rstrip())
   hits.append({'bed_record':fields,'one_based_inclusive_start':int(fields[1])+1,'one_based_inclusive_end':int(fields[2])})
with open('overlap-evidence.json','w') as f:
 json.dump({'source':'https://downloads.wenglab.org/Registry-V4/GRCh38-cCREs.bed','assembly':'GRCh38','variant_reference_span_1based_inclusive':['chr19',p,p],'hits':hits},f,indent=2)
print('Number of overlaps:',len(hits))
PY"
