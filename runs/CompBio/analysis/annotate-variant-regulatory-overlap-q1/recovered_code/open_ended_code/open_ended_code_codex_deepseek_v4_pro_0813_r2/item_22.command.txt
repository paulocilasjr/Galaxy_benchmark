/bin/bash -lc "python3 - <<'PY'
import pyBigWig, collections
b=pyBigWig.open('encodeCcreRegistry.bb')
classes=collections.Counter()
count=0
for chrom in ['chr1','chr2','chr19']:
    c=b.chroms(chrom)
    for e in b.entries(chrom,0,c):
        parts=e[2].split('\\t')
        if len(parts)>=7:
            classes[parts[6]] += 1
        count+=1
        if count>100000: break
    if count>100000: break
print(classes)
PY"
