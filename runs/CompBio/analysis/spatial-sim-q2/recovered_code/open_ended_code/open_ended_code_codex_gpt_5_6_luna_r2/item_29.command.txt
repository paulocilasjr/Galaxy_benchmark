import numpy as np,csv
from scipy.io import mmread
from scipy.optimize import nnls
from scipy.stats import spearmanr,pearsonr
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001))
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels)); S=np.array([sc[labels==t].mean(0) for t in types]); sp=mmread('data/matrix.mtx.gz').toarray(); blocks=[(0,50),(50,100),(100,150),(150,200),(200,250),(250,300)]; X=np.array([[S[i,a:b].sum() for i in range(6)] for a,b in blocks]);Y=np.array([sp[a:b].sum(0) for a,b in blocks]).T; C=np.array([nnls(X,Y[j])[0] for j in range(900)]); ti=types.index('Tumor_Core');
print(types)
for i,t in enumerate(types):
 print(t,'Pearson with tumor',pearsonr(C[:,ti],C[:,i]),'Spearman',spearmanr(C[:,ti],C[:,i]))
print('conditional means other types in tumor spots vs non')
tum=C[:,ti]>.5
for i,t in enumerate(types): print(t,C[tum,i].mean(),C[~tum,i].mean(), 'frac present',np.mean(C[tum,i]>.5),np.mean(C[~tum,i]>.5))
PY"
