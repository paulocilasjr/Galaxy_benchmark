/bin/bash -lc "python3 -u - <<'PY'
import numpy as np,csv
from scipy.io import mmread
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001))
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels)); S=np.array([sc[labels==t].mean(0) for t in types]);
sp=mmread('data/matrix.mtx.gz').toarray();
blocks={'Tumor_Core':(0,50),'Fibroblast_Stroma':(50,100),'T_Cell':(100,150),'B_Cell':(150,200),'Macrophage':(200,250),'Endothelial':(250,300)}
print('raw block avg counts per cell')
for t,(a,b) in blocks.items(): print(t,round(S[types.index(t),a:b].mean(),2), [round(S[i,a:b].mean(),2) for i in range(6)])
print('spatial block sum quantiles')
for t,(a,b) in blocks.items():
 x=sp[a:b].sum(0)
 print(t, np.quantile(x,[0,.1,.25,.5,.75,.9,.99,1]).round(1).tolist())
PY"
