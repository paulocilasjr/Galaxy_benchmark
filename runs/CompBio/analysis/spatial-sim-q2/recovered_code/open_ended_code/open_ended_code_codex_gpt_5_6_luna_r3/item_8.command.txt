/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, gzip, scipy.io as sio
from scipy import sparse
print('loading')
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
print('sc',sc.shape,'totals',sc.sum(axis=1).describe().to_dict())
with gzip.open('matrix.mtx.gz','rt') as f: sm=sio.mmread(f).tocsr()
print('spatial',sm.shape, 'nnz',sm.nnz,'sum',sm.sum(), 'spot totals min/max/mean',np.asarray(sm.sum(axis=0)).ravel().min(),np.asarray(sm.sum(axis=0)).ravel().max(),np.asarray(sm.sum(axis=0)).ravel().mean())
# marker stats mean normalized expression
X=sc.values.astype(float)
lib=X.sum(1)
Xn=X/lib[:,None]*1e4
for typ in meta.cell_type.unique():
 ix=meta.cell_type.values==typ
 mean=Xn[ix].mean(0)
 others=Xn[~ix].mean(0)
 # log2 ratio with small
 lfc=np.log2((mean+0.1)/(others+0.1))
 top=np.argsort(lfc)[-15:][::-1]
 print('\\n',typ,'n',ix.sum(),'top',[(f'Gene_{j}',round(lfc[j],2),round(mean[j],1),round(others[j],1)) for j in top])
# show spatial profiles of marker genes maybe gene indices top marker means
PY"
