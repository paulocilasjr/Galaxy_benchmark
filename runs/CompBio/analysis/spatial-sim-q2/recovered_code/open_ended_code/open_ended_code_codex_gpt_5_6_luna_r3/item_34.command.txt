/bin/bash -lc "python - <<'PY'
# quickly calculate rounded counts using prior code and summarize discrete composition and total k
import pandas as pd,numpy as np,gzip,scipy.io as sio
from scipy.optimize import nnls
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0);meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0);S=sio.mmread(gzip.open('matrix.mtx.gz','rt')).tocsr().toarray().T.astype(float)
types=list(meta.cell_type.unique());X=sc.values.astype(float);means=np.array([X[meta.cell_type.values==t].mean(0) for t in types]);sets=[]
for j in range(6):
 o=np.max(np.delete(means,j,axis=0),0);ss=np.log2((means[j]+.1)/(o+.1));sets.append(np.argsort(ss)[-50:])
for use,name in [(np.arange(1000),'all'),(np.unique(np.concatenate(sets)),'mark')]:
 co=np.array([nnls(means[:,use].T,S[i,use])[0] for i in range(900)]);r=np.rint(co).astype(int)
 print(name,'rounded counts total vs spot total k',np.bincount(r.sum(1)), 'tumor',np.sum(r[:,0]>0),'noninteger maxdiff',np.max(np.abs(co-r)), 'sum mismatch counts',np.sum(r.sum(1)"'!=np.rint(S.sum(1)/1240)))
 print('"'unique comps',len(np.unique(r,axis=0)))
 # list counts each type by true cells
 print(dict(zip(types,r.sum(0))))
PY"
