/bin/bash -lc "python - <<'PY'
import pandas as pd,numpy as np,gzip,scipy.io as sio
from scipy.optimize import nnls
from scipy.spatial.distance import cdist
from scipy.stats import mannwhitneyu, permutation_test, fisher_exact
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0); meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0); pos=pd.read_csv('tissue_positions.csv')
S=sio.mmread(gzip.open('matrix.mtx.gz','rt')).tocsr().toarray().T.astype(float)
types=list(meta.cell_type.unique()); X=sc.values.astype(float); means=np.array([X[meta.cell_type.values==t].mean(0) for t in types])
# marker sets raw top 50 each
sets=[]
for j in range(6):
 other=np.max(np.delete(means,j,axis=0),axis=0); score=np.log2((means[j]+0.1)/(other+0.1)); sets.append(np.argsort(score)[-50:])
use=np.unique(np.concatenate(sets)); co=np.array([nnls(means[:,use].T,S[i,use])[0] for i in range(900)])
# direct presence by coeff > .25 maybe infer robust
coords=pos[['array_row','array_col']].to_numpy(); D=cdist(coords,coords)
for th in [.1,.25,.5,.75]:
 present=co>th; tumor=present[:,0]
 d=D[:,tumor].min(1)
 print('\\n=== presence coeff >',th,'tumor n',tumor.sum(),'===')
 print('tumor coords bbox',coords[tumor].min(0),coords[tumor].max(0))
 print('tumor d quantiles all',np.quantile(d,[0,.1,.25,.5,.75,.9,1]))
 for j,t in enumerate(types[1:],1):
  m=present[:,j]
  # focal include tumor spots or exclude to avoid d=0 co-presence
  for excl,label in [(np.ones(900,bool),'all'),(~tumor,'nonTumor')]:
   mm=m&excl; comp=(~m)&excl
   if mm.sum()==0 or comp.sum()==0: continue
   u,p=mannwhitneyu(d[mm],d[comp],alternative='less')
   print(t,label,'n',mm.sum(),'median/mean',round(np.median(d[mm]),2),round(d[mm].mean(),2),'not',round(np.median(d[comp]),2),round(d[comp].mean(),2),'p',p,'within1',int((d[mm]<=1.01).sum()),'within2',int((d[mm]<=2.01).sum()),'within3',int((d[mm]<=3.01).sum()))
 # contact at distances, compare observed type presence near target vs prevalence using count/expected
 for rad in [1,1.5,2,3,4,5]:
  near=(d<=rad)
  print('rad',rad,'near n',near.sum(),[(t,int((present[:,j]&near).sum()),round((present[:,j]&near).sum()/max(near.sum(),1),3),round(present[:,j].mean(),3)) for j,t in enumerate(types[1:],1)])
PY"
