/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
from scipy.stats import spearmanr, mannwhitneyu, pointbiserialr
comp=pd.read_csv('deconv_composition.csv')
ctypes=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
coords=comp[['array_row','array_col']].values
tm=comp['Tumor_Core']>0
D=cdist(coords,coords[tm]); nearest=D.min(1)
for t in ctypes:
    if t=='Tumor_Core': continue
    x=comp[t].values
    rho,p=spearmanr(nearest,x)
    rpb,pb=pointbiserialr(nearest, x>0)
    print(t, 'spearman nearest-vs-count', rho, p, 'point-biserial nearest vs presence', rpb, pb)
PY"
