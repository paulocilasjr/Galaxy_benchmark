import pandas as pd, numpy as np
from scipy.io import mmread
from scipy.spatial.distance import cdist
from scipy.stats import spearmanr
sc=pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
genes=sc.columns.tolist()
X=mmread('matrix.mtx.gz').T.toarray()
comp=pd.read_csv('deconv_composition.csv')
coords=comp[['array_row','array_col']].values
tm=comp['Tumor_Core']>0; D=cdist(coords,coords[tm]); nearest=D.min(1)
# blocks from marker specificity top? define indices
blocks={
'B_Cell': list(range(140,180)),
'Endothelial': list(range(270,330)),
'Fibroblast_Stroma': list(range(50,110)),
'Macrophage': list(range(200,260)),
'T_Cell': list(range(100,140)),
'Tumor_Core': list(range(0,50)),
}
# better exact top genes? Let's use all genes in blocks? Compute correlations with nearest for each gene individually maybe.
for t, idxs in blocks.items():
    score=X[:,idxs].sum(axis=1)
    # spearman of score vs nearest
    r,p=spearmanr(nearest, score)
    print(t, 'block sum corr nearest', r,p, 'mean score near/far')
    # score normalized
    score_z=(score-score.mean())/score.std()
    r2,p2=spearmanr(nearest, score_z)
    print(' ',r2,p2)
# gene-level correlations of marker top genes with nearest
print('\\nGene-level top corr (most negative = near tumor)')
corrs=[]
for g in genes:
    r,_=spearmanr(nearest, X[:,genes.index(g)])
    corrs.append((r,g))
corrs_sorted=sorted(corrs)[:30]
for r,g in corrs_sorted:
    # block association
    pass
    print(r,g)
