/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
from scipy.optimize import nnls
sc=pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
ctypes=meta['cell_type'].astype('category').cat.categories.tolist()
sig=np.array([sc.values[meta['cell_type']==t].mean(axis=0) for t in ctypes])
X=mmread('matrix.mtx.gz').T.toarray()
W=np.array([nnls(sig.T,x)[0] for x in X])
R=np.rint(W)
d=np.abs(W-R)
print('max frac dist', d.max(), 'mean', d.mean())
print('quantiles', np.quantile(d,[.5,.75,.9,.95,.99,1]))
# count weights with frac >0.4
print('num weights frac>0.4', (d>0.4).sum(), '>0.3', (d>0.3).sum(), '>0.2',(d>0.2).sum())
# confusion if threshold soft
PY"
