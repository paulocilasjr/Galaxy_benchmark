/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
comp=pd.read_csv('deconv_composition.csv')
ctypes=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
for t in ctypes:
    mat=np.zeros((30,30),int)
    for _,r in comp.iterrows():
        mat[int(r.array_row), int(r.array_col)]=r[t]
    print('\\n',t, 'spots', (mat>0).sum())
    # print compressed presence and counts
    for i in range(30):
        row=''.join(str(mat[i,j]) if mat[i,j]>0 else '.' for j in range(30))
        if row.strip('.'): print(f'{i:2d} {row}')
PY"
