/bin/bash -lc "OPENBLAS_NUM_THREADS=1 python -u - <<'PY'
import pandas as pd,numpy as np
from scipy.io import mmread
from scipy.optimize import nnls
from scipy.spatial.distance import cdist
from scipy.stats import mannwhitneyu
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0); m=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0).iloc[:,0]; X=mmread('matrix.mtx.gz').toarray().T; p=pd.read_csv('tissue_positions.csv'); types=sorted(m.unique()); A=np.column_stack([sc.loc[m.eq(c)].mean().values for c in types]); co=np.rint(np.array([nnls(A,x)[0] for x in X])).astype(int); xy=p[['array_row','array_col']].values; T=co[:,types.index('Tumor_Core')]>0; d=cdist(xy,xy[T]).min(1); rng=np.random.default_rng(8); base=~T
print('Exclude tumor spots:')
for j,c in enumerate(types[:-1]):
 P=(co[:,j]>0)&base; obs=d[P].mean(); pool=np.flatnonzero(base); rr=np.array([d[rng.choice(pool,P.sum(),False)].mean() for _ in range(5000)]); print(c,P.sum(),'mean',round(obs,3),'null',round(rr.mean(),3),'z',round((obs-rr.mean())/rr.std(),2),'p', (sum(rr<=obs)+1)/5001, 'MW',mannwhitneyu(d[P],d[base & ~P],alternative='less').pvalue)
print('Floor-bin presence n/N excluding tumor')
for j,c in enumerate(types[:-1]):
 P=co[:,j]>0
 print(c,[(b,int(np.sum(P&base&(d>=b)&(d<b+1))),int(np.sum(base&(d>=b)&(d<b+1)))) for b in range(1,15)])
PY"
