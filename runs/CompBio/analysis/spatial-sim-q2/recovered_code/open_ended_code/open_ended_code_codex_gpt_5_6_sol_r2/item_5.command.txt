/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
from scipy import sparse
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
X=mmread('matrix.mtx.gz').tocsr() # genes x spots
pos=pd.read_csv('tissue_positions.csv')
print('sc',sc.shape,'meta',meta.shape,'sp',X.shape,'pos',pos.shape)
print('\\ncell counts:\\n',meta.cell_type.value_counts().sort_index())
print('\\nsc library by type:')
print(pd.DataFrame({'lib':sc.sum(1),'type':meta.cell_type}).groupby('type').lib.agg(['count','mean','std','min','max']).sort_index())
print('\\nspot library',np.percentile(np.asarray(X.sum(0)).ravel(),[0,25,50,75,100]))
# Means normalized vs raw and best genes
cts=sorted(meta.cell_type.unique())
raw=np.column_stack([sc.loc[meta.cell_type.eq(c)].mean(0).values for c in cts])
norm=sc.div(sc.sum(1),axis=0)*1000
means=np.column_stack([norm.loc[meta.cell_type.eq(c)].mean(0).values for c in cts])
print('\\ntypes',cts)
for j,c in enumerate(cts):
    ratio=(means[:,j]+.1)/(np.max(np.delete(means,j,axis=1),axis=1)+.1)
    ix=np.argsort(ratio)[-12:][::-1]
    print(c,[(sc.columns[i],round(means[i,j],2),round(ratio[i],1)) for i in ix])
print('\\ncoords rows/cols',pos.array_row.min(),pos.array_row.max(),pos.array_col.min(),pos.array_col.max(), 'counts',pos.array_row.nunique(),pos.array_col.nunique())
PY"
