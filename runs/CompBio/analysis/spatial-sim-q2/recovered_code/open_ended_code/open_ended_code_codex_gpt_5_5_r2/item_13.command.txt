/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
np.random.seed(1)
fr=pd.read_csv('nnls_spot_fractions.csv', index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode').loc[fr.index]
coords=pos[['array_row','array_col']].astype(float).values
cts=list(fr.columns)
for thr in [0.05,0.1,0.2,0.3,0.5]:
    tumor=fr['Tumor_Core'].values>thr
    if tumor.sum()<2: continue
    d=cdist(coords, coords[tumor]).min(1)
    print('\\nTumor thr',thr,'n',tumor.sum())
    for ct in cts:
        if ct=='Tumor_Core': continue
        w=fr[ct].values
        obs=np.average(d, weights=w+1e-12)
        null=[]
        for _ in range(2000):
            wp=np.random.permutation(w)
            null.append(np.average(d, weights=wp+1e-12))
        null=np.array(null)
        p=(np.sum(null<=obs)+1)/(len(null)+1) # closer/smaller than random
        z=(obs-null.mean())/null.std()
        print(f'{ct:18s} obs {obs:.3f} null {null.mean():.3f} z {z:.2f} p {p:.4g}')
PY"
