/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
C=pd.read_csv('mle_counts.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[C.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
# dominant type per spot using max count, tie-break? np argmax first (alphabetical?) Need maybe random? check ties
dom=C.idxmax(axis=1)
print(dom.value_counts())
tidx=np.where(dom.values=='Tumor_Core')[0]
print('tumor-dominant spots',len(tidx))
d=cdist(coords,coords[tidx]).min(axis=1)
for ct in [c for c in C.columns if c"'!='"'Tumor_Core']:
    idx=np.where(dom.values==ct)[0]
    print(ct,'n',len(idx),'mean dist',d[idx].mean() if len(idx) else None, 'median',np.median(d[idx]) if len(idx) else None)
PY"
