/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
R=pd.read_csv('deconv_raw.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[R.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
tumor_idx=np.where(R['Tumor_Core'].values>0.85)[0]
non_tumor_idx=np.setdiff1d(np.arange(len(R)),tumor_idx)
dist_all=cdist(coords, coords[tumor_idx]).min(axis=1)
for ct in [c for c in R.columns if c"'!='"'Tumor_Core']:
    # present spots outside tumor spots
    pres=(R[ct].values>0.5)&np.isin(np.arange(len(R)),non_tumor_idx)
    vals=dist_all[pres]
    # compare to all non-tumor spots (same set)
    all_non=dist_all[non_tumor_idx]
    rng=np.random.default_rng(2)
    sims=np.array([dist_all[rng.choice(non_tumor_idx,size=pres.sum(),replace=False)].mean() for _ in range(2000)])
    print(ct,'n outside',pres.sum(),'mean',vals.mean(),'all',all_non.mean(),'z',(vals.mean()-sims.mean())/sims.std(),'p',(np.sum(sims<=vals.mean())+1)/(len(sims)+1))
PY"
