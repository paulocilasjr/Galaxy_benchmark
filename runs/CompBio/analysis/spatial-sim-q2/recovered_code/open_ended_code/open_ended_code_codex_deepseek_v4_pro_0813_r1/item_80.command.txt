/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
repo='amanutej/trustworthy-biology-agents-traces'
for path in ['traces/compbiobench','derived','']:
    url=f'https://huggingface.co/api/datasets/{repo}/tree/main/{path}?recursive=true&expand=true&limit=100'
    try:
        data=json.load(urllib.request.urlopen(url,timeout=30))
        print('\\nPATH',path,'entries',len(data))
        for f in data:
            p=f.get('path')
            if p and ('result' in p.lower() or 'score' in p.lower() or 'metric' in p.lower() or 'answer' in p.lower() or p.endswith('.json') or p.endswith('.csv')):
                print(f['type'],p,f.get('size'))
    except Exception as e: print('ERR',path,e)
PY"
