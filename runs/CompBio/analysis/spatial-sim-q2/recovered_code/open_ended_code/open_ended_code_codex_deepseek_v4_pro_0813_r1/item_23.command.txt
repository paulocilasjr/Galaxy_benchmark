/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
R=pd.read_csv('deconv_raw.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[R.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
tumor_idx=np.where(R['Tumor_Core'].values>0.85)[0]
dist=cdist(coords, coords[tumor_idx]).min(axis=1)
# define bins distance to nearest tumor in pixels
bins=[0,100,150,200,250,300,400,600,10000]
labels=['0-100','100-150','150-200','200-250','250-300','300-400','400-600','>600']
df=pd.DataFrame({'dist':dist})
for ct in [c for c in R.columns if c"'!='"'Tumor_Core']:
    df[ct]=R[ct].values
df['Tumor']=np.isin(np.arange(len(R)),tumor_idx)
print('mean raw count by distance bin')
for ct in [c for c in R.columns if c"'!='"'Tumor_Core']:
    print('\\n',ct)
    print(df.groupby(pd.cut(df['dist'],bins=bins,labels=labels))[ct].mean().round(3))
print('\\nProportions by bins')
P=R.div(R.sum(axis=1),axis=0)
for ct in [c for c in P.columns if c"'!='"'Tumor_Core']:
    print(ct, pd.Series(P[ct].values,index=df.index).groupby(pd.cut(df['dist'],bins=bins,labels=labels)).mean().round(3).to_dict())
PY"
