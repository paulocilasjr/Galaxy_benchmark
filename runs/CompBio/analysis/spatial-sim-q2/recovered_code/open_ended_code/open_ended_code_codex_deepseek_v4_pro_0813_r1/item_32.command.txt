/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
R=pd.read_csv('deconv_raw.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[R.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
# tumor centroid
tidx=np.where(R['Tumor_Core'].values>0.85)[0]
cent=coords[tidx].mean(axis=0)
print('tumor centroid',cent)
# rounded counts
C=pd.DataFrame(index=R.index)
for ct in R.columns:
    C[ct]=np.where(R[ct].values>0.5, np.round(R[ct].values),0).astype(int)
for ct in R.columns:
    # weighted mean coords of cells
    n=C[ct].sum()
    wm=(C[ct].values[:,None]*coords).sum(axis=0)/n
    # nearest? distance centroid
    d=np.sqrt(((wm-cent)**2).sum())
    print(ct,'cells',n,'centroid',wm.round(0),'dist_to_tumor_cent',d.round(0))
PY"
