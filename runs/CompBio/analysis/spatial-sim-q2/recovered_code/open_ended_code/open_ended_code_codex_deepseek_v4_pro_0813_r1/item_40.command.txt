/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
from scipy.stats import spearmanr, pearsonr
C=pd.read_csv('mle_counts.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[C.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
tidx=np.where(C['Tumor_Core'].values>0)[0]
d=cdist(coords,coords[tidx]).min(axis=1)
# include all spots, proportions based on MLE counts normalized
P=C.div(C.sum(axis=1),axis=0)
print('All spots: corr proportions with distance')
for ct in [c for c in C.columns if c"'!='"'Tumor_Core']:
    print(ct,'pearson',pearsonr(P[ct],d)[0].round(4),'spearman',spearmanr(P[ct],d)[0].round(4))
# Exclude tumor spots
idx=np.setdiff1d(np.arange(len(C)),tidx)
print('\\nExcluding tumor-containing spots')
for ct in [c for c in C.columns if c"'!='"'Tumor_Core']:
    print(ct,'pearson',pearsonr(P[ct].values[idx],d[idx])[0].round(4),'spearman',spearmanr(P[ct].values[idx],d[idx])[0].round(4))
# Exclude tumor spots and use raw counts
print('\\nRaw counts excluding tumor spots corr')
for ct in [c for c in C.columns if c"'!='"'Tumor_Core']:
    print(ct,'pearson',pearsonr(C[ct].values[idx],d[idx])[0].round(4),'spearman',spearmanr(C[ct].values[idx],d[idx])[0].round(4))
PY"
