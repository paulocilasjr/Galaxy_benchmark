/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, itertools
from scipy.io import mmread
sp=mmread('matrix.mtx.gz').toarray() # genes x spots
features=pd.read_csv('features.tsv.gz',sep='\\t',header=None,names=['ens','gene','type'])
sc=pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
means=sc.groupby(meta['cell_type']).mean()
cts=means.index.tolist()
X=means.loc[:,features['gene']].values.T # genes x types
combos=[c for c in itertools.product(range(6), repeat=6) if 0 < sum(c) <=5]
combos=np.array(combos)
print('n combos',len(combos), X.shape)
lam = X @ combos.T  # genes x combos
Y=sp
log_lam=np.log(np.maximum(lam,1e-12))
score = Y.T @ log_lam - lam.sum(axis=0)[None,:]
best=combos[score.argmax(axis=1)]
bestdf=pd.DataFrame(best,index=pd.read_csv('barcodes.tsv.gz',sep='\\t',header=None)[0],columns=cts)
print('best counts total stats')
print(pd.Series(best.sum(axis=1)).describe())
print(bestdf.sum(axis=0).to_dict())
print('tumor spots', (bestdf['Tumor_Core']>0).sum(), 'tumor cells', bestdf['Tumor_Core'].sum())
bestdf.to_csv('mle_counts.csv')
R=pd.read_csv('deconv_raw.csv',index_col=0)
for ct in cts:
    nn=(R[ct]>0.5).astype(int)
    bb=(bestdf[ct]>0).astype(int)
    print(ct,'mismatch presence', (nn"'!=bb).sum(), '"'corr', np.corrcoef(R[ct],bestdf[ct])[0,1])
print('tumor presence mismatch', ((R['Tumor_Core']>0.85)"'!=(bestdf['"'Tumor_Core']>0)).sum())
PY"
