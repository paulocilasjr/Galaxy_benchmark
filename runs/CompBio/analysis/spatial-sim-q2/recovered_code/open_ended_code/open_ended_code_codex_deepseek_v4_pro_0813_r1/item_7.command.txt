/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
# Load full sc
sc=pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
print(sc.shape)
# gene columns are strings Gene_0...
cts=meta['cell_type'].unique()
means=sc.groupby(meta['cell_type']).mean()
print(means.shape)
# top markers by std across types or ratio
# Save for future? just print correlation matrix
print(means.corr())
# Print top 20 genes per type by ratio to others
for ct in cts:
    # rank ratio of mean / mean of all other
    other=means.drop(index=ct).mean(axis=0)
    r=(means.loc[ct]+1)/(other+1)
    print('\\n',ct, r.sort_values(ascending=False).head(15).index.tolist(), r.sort_values(ascending=False).head(10).values.round(2))
PY"
