/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for repo in ['Genentech/compbiobench-results-v1','Genentech/compbiobench-submissions-v1']:
 url=f'https://huggingface.co/api/datasets/{repo}/tree/main?recursive=true&expand=true&limit=100'
 try:
  data=json.load(urllib.request.urlopen(url,timeout=30))
  print(repo,'entries',len(data))
  for f in data[:20]: print(f['type'],f['path'],f.get('size'))
 except Exception as e: print('ERR',repo,e)
PY"
