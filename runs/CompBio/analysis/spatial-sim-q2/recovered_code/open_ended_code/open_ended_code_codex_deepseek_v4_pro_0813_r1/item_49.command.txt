/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true&expand=true'
data=json.load(urllib.request.urlopen(url))
print('num entries',len(data))
for f in data:
    p=f['path']
    if 'spatial' in p.lower() or p.lower().startswith('data/spatial') or 'tar.gz' in p.lower() and 'spatial' in p.lower():
        print(f['type'],p,f.get('size'),f.get('lfs',{}).get('size'))
# print all data paths containing sim
for f in data:
    if 'sim' in f['path'].lower(): print('SIM',f['path'],f['type'])
PY"
