# dominant tumor target
import pandas as pd,numpy as np
from scipy.io import mmread
from scipy.optimize import nnls
from scipy.spatial.distance import cdist

mask=np.argmax(cnt,1)==types.index('Tumor_Core');d=D[:,mask].min(1);print('n',mask.sum(), 'mean',d.mean(),' '.join(f'{t}:{np.average(d,weights=cnt[:,j]):.2f}' for j,t in enumerate(types)))
PY"
