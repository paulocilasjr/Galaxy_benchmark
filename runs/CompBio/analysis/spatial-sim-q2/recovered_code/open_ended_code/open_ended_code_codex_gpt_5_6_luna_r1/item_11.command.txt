/bin/bash -lc "python - <<'PY'
import pandas as pd,numpy as np
sc=pd.read_csv('extracted/spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('extracted/spatial_q_sc_metadata.csv',index_col=0)
means=meta.join(sc).groupby('cell_type')[sc.columns].mean()
norm=sc.div(sc.sum(1),axis=0)*1e4
nmeans=meta.join(norm).groupby('cell_type')[sc.columns].mean()
print('mean totals by type',means.sum(axis=1).to_string())
for t in means.index:
 print('\\nTYPE',t)
 other=nmeans.drop(t).mean()
 score=nmeans.loc[t]/(other+1e-9)
 for direction in [True,False]:
  ix=score.sort_values(ascending=direction).head(20).index
  print(('top' if direction else 'low'), [(g,round(float(nmeans.loc[t,g]),1),round(float(other[g]),1),round(float(score[g]),1)) for g in ix[:12]])
print('\\nTop by max/min nmean ratio')
mx=nmeans.max(axis=0); mn=nmeans.min(axis=0)
ix=(mx/(mn+1e-6)).sort_values(ascending=False).head(50)
for g in ix.index: print(g,round(float(ix[g]),1),nmeans[g].round(1).to_dict())
PY"
