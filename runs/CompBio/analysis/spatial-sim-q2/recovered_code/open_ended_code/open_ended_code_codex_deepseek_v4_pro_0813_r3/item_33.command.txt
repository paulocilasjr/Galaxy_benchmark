/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
bar=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0]
P=pd.read_csv('nnls_weights.csv'); P.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']; P.index=bar
pres=P>0.2
tumor=pres['Tumor_Core']; tcoords=coords[tumor.values]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
for ct in P.columns[:-1]:
 sel=pres[ct].values
 others=~sel
 # compare D[sel] to D[others] (all not ct), with sel possibly includes tumor spots
 u,p=scipy.stats.mannwhitneyu(D[sel],D[others],alternative='less')
 print(ct,'n',sel.sum(),'mean',D[sel].mean(),'other_mean',D[others].mean(),'p_less',p,'u',u)
 # exclude tumor spots from sel? compare ct non-tumor to all other non-tumor not ct
 sel2=sel & ~tumor.values
 others2=(~sel)&~tumor.values
 u2,p2=scipy.stats.mannwhitneyu(D[sel2],D[others2],alternative='less')
 print('  exclT p',p2,'mean',D[sel2].mean(),'other',D[others2].mean())
PY"
