/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.stats
C=pd.read_csv('map_compositions.csv'); C.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
tumor=C['Tumor_Core']>0
for ct in C.columns[:-1]:
 p=C[ct]>0
 a=p[tumor].sum(); b=p[~tumor].sum(); c=tumor.sum()-a; d=(~tumor).sum()-b
 odds,pval=scipy.stats.fisher_exact([[a,b],[c,d]],alternative='greater')
 print(f'{ct:20s} tumor {a}/{tumor.sum()} nonT {b}/{(~tumor).sum()} odds={odds:.3f} p={pval:.4g}')
PY"
