/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
# tumor center
w=C['Tumor_Core'].values
center=np.average(coords,weights=w,axis=0)
print('tumor center',center,'total tumor cells',w.sum())
# centroid each type weighted by counts
for ct in cts:
 w=C[ct].values
 c=np.average(coords,weights=w,axis=0)
 d=np.linalg.norm(c-center)
 print(f'{ct:20s} centroid={c} dist={d:.3f} total={w.sum()}')
PY"
