/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
bar=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0]
P=pd.read_csv('nnls_weights.csv'); P.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']; P.index=bar
S=pd.read_csv('marker_scores.csv',index_col=0); S.index=bar
for q in [0.8,0.9,0.93,0.95]:
 thr=np.quantile(S['Tumor_Core'],q); mask=S['Tumor_Core']>thr
 print('marker q',q,'n',mask.sum(),'overlap', (mask & (P['Tumor_Core']>0.2)).sum())
print(pd.DataFrame({'marker_tumor':S['Tumor_Core'],'nnls_tumor':P['Tumor_Core']>0.2}).groupby('nnls_tumor')['marker_tumor'].describe())
print('corr',np.corrcoef(S['Tumor_Core'],P['Tumor_Core'])[0,1])
PY"
