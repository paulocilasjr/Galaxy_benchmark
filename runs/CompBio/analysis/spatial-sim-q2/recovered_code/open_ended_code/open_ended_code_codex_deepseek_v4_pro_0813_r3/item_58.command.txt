/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
tumor=C['Tumor_Core']>0; tcoords=coords[tumor]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
# logistic regression presence ~ -D or D for each type on nonT spots maybe all
from sklearn.linear_model import LogisticRegression
for ct in cts[:-1]:
 y=(C[ct]>0).astype(int)
 for subset,mask in [('nonT',~tumor),('all',np.ones(900,dtype=bool))]:
  X=D[mask].reshape(-1,1)
  lr=LogisticRegression(penalty='none',solver='lbfgs').fit(X,y[mask])
  # coefficient negative means closer increases probability
  coef=lr.coef_[0][0]; p=1/(1+np.exp(-lr.intercept_[0]-coef*0))
  print(f'{ct:20s} {subset:5s} coef(D)={coef:.4f} p_at0={p:.3f} accuracy={lr.score(X,y[mask]):.3f}')
 print()
PY"
