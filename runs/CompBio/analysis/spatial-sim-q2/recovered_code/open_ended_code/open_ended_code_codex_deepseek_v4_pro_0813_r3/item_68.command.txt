/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
C=pd.read_csv('map_compositions.csv'); C.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
# tumor sets variants
variants={}
variants['any tumor']=C['Tumor_Core']>0
variants['tumor count >=2']=C['Tumor_Core']>=2
# pure tumor = total cells==tumor count? or only Tumor and no others
pure=(C['Tumor_Core']>0)&(C[C.columns[:-1]].sum(axis=1)==0)
variants['pure tumor']=pure
# dominant tumor (argmax and tumor >0)
idx=C.values.argmax(axis=1)
dominant=(C['Tumor_Core']>0)&(idx==5)
variants['dominant tumor']=dominant
for name,tumor in variants.items():
 tcoords=coords[tumor]; D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
 print('\\n',name,'n',tumor.sum())
 for ct in C.columns[:-1]:
  sel=(C[ct]>0)&~tumor
  if sel.sum()<5: continue
  obs=D[sel].mean(); base=D[~tumor].mean()
  u,p=scipy.stats.mannwhitneyu(D[sel],D[(~tumor)&(~sel)],alternative='less')
  print(f'{ct:20s} mean={obs:.3f} base={base:.3f} p={p:.4g}')
PY"
