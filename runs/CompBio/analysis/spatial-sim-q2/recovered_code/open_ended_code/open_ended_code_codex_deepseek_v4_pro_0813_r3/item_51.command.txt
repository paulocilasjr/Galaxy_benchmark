/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
# radius 1 queen neighbors excluding self
rng=np.random.default_rng(0)
tumor=(C['Tumor_Core']>0).values
# compute neighbor pairs among spots within dist<=1.5 (queen+diag) excluding same
dist=scipy.spatial.distance.cdist(coords,coords)
adj=(dist<=1.5)&(dist>0)
for ct in cts[:-1]:
 pres=(C[ct]>0).values
 obs_pairs=((pres[:,None]&tumor[None,:])&adj).sum()
 # also count tumor-type ordered? symmetric same
 # permutation of ct labels across all spots
 perms=[]
 for _ in range(5000):
  idx=rng.permutation(900)
  p=pres[idx]
  perms.append(((p[:,None]&tumor[None,:])&adj).sum())
 perms=np.array(perms); p=(perms>=obs_pairs).mean(); z=(obs_pairs-perms.mean())/perms.std()
 print(f'{ct:20s} observed_pairs={obs_pairs:4d} expected={perms.mean():.1f} p_enrich={p:.5f} z={z:.2f}')
PY"
