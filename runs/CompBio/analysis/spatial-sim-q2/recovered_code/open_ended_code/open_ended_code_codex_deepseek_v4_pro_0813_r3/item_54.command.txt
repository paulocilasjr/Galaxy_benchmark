/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
C=pd.read_csv('map_compositions.csv'); C.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
tumor=(C['Tumor_Core']>0).values
fib=(C['Fibroblast_Stroma']>0).values
dist=scipy.spatial.distance.cdist(coords,coords); adj=(dist<=1.5)&(dist>0)
obs=(fib[:,None]&tumor[None,:]&adj).sum()
# expected under independent labels preserving degrees: sum over tumor spots adj_degree * p_fib
p_fib=fib.mean(); exp=sum(adj[i,:].sum()*p_fib for i in range(900) if tumor[i])
# simpler sum over ordered edges with one endpoint tumor, other any * p_fib
edges_inc_tumor=sum(adj[i,:].sum() for i in range(900) if tumor[i])
exp2=edges_inc_tumor*p_fib
print('p_fib',p_fib,'obs',obs,'edges inc tumor',edges_inc_tumor,'exp',exp2)
print('fib near nonT unique rate', ((fib&~tumor)&(dist[tumor].min(axis=1)<=1.5)).sum()/ ((~tumor)&(dist[tumor].min(axis=1)<=1.5)).sum())
PY"
