/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
P=pd.read_csv('nnls_weights.csv'); P.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
pres=(P>0.2).values
tumor=pres[:,5]; tcoords=coords[tumor]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
for i,ct in enumerate(P.columns[:-1]):
 sel=pres[:,i] & ~tumor
 n=sel.sum(); obs=D[sel].mean(); obsmed=np.median(D[sel])
 base=D[~tumor].mean()
 rng=np.random.default_rng(42); perm=[]
 nonidx=np.where(~tumor)[0]
 for _ in range(10000):
  idx=rng.choice(nonidx,size=n,replace=False); perm.append(D[idx].mean())
 p=(np.array(perm)<=obs).mean(); z=(obs-np.mean(perm))/np.std(perm)
 print(f'{ct:20s} n_nonTumor={n:3d} obs_mean={obs:.3f} med={obsmed:.3f} baseline={base:.3f} p={p:.5f} z={z:.2f}')
PY"
