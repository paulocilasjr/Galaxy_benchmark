/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, time, re, sys
seq='GTCGGCATGTATTAGCTCTAGAATTACCACAGTTATCCAAGTAGGAGAGGAGCGAGCGACCAAAGGAACCATAACTGATTTAATGAGCCATTCGCAGTTTCACTGTACCGGCCGTGCGTACTTAGACATGCATGGCTTAATCTTTGAGACAAGCATATGCTACTGGCAG'
params={
 'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,'MEGABLAST':'on','HITLIST_SIZE':'20','EXPECT':'1000','FILTER':'L','WORD_SIZE':'16','FORMAT_TYPE':'XML'
}
url='https://blast.ncbi.nlm.nih.gov/Blast.cgi'
data=urllib.parse.urlencode(params).encode()
resp=urllib.request.urlopen(url,data=data,timeout=60).read().decode('utf-8','replace')
print(resp[:1000])
rid=re.search(r'RID = (\\S+)', resp)
rtoe=re.search(r'RTOE = (\\d+)', resp)
if not rid:
    sys.exit('no RID')
rid=rid.group(1); wait=int(rtoe.group(1)) if rtoe else 10
print('RID',rid,'wait',wait)
for i in range(30):
    time.sleep(max(5, min(wait,20)))
    q=urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML','HITLIST_SIZE':'20'})
    txt=urllib.request.urlopen(url+'?'+q,timeout=60).read().decode('utf-8','replace')
    print('poll',i,'len',len(txt),txt[:120].replace('\\n',' '))
    if 'Status=WAITING' in txt:
        continue
    open('qblast_one.xml','w').write(txt)
    print(txt[:4000])
    break
PY"
