/bin/bash -lc "python3 - <<'PY'
# Parse saved BLAST XML quickly and count top-hit labels from the 200 sampled reads.
import xml.etree.ElementTree as ET, re
from collections import Counter
root=ET.parse('genome_unmapped_200_blast.xml').getroot()
c=Counter(); examples={}
for it in root.findall('.//Iteration'):
    hit=it.find('.//Hit')
    if hit is None:
        c['nohit']+=1; continue
    d=hit.findtext('Hit_def') or ''
    label='other'
    for name in ['Sus scrofa','Homo sapiens','Human gammaherpesvirus 4','Lymphocryptovirus humangamma4','Epstein-Barr virus','Synthetic construct','Staphylococcus cohnii','Megaleporinus elongatus']:
        if name in d:
            label=name; break
    c[label]+=1; examples.setdefault(label,d[:120])
print(c)
for k,v in examples.items(): print(k,':',v)
PY"
