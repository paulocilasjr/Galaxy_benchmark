import collections
n=0;exact=0;genes=set();ids=set()
with open('pig.sam') as f:
 for l in f:
  if l.startswith('@'):continue
  p=l.rstrip().split('\\t')
  if int(p[1])&260:continue
  n+=1;ids.add(p[0]);genes.add(p[2])
  if p[5]=='75M' and 'NM:i:0' in p[11:]:exact+=1
print('Pig mapped reads',len(ids),'distinct transcripts',len(genes),'full-length exact alignments',exact)
PY"
