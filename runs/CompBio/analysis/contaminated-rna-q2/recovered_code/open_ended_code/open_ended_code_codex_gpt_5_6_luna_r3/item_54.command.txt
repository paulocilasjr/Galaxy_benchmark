/bin/bash -lc "python3 - <<'PY'
import re, xml.etree.ElementTree as ET
root=ET.parse('tmp/blast_get100.xml').getroot()
for n in [1,3,5,10]:
 c={}
 for it in root.findall('.//Iteration'):
  hs=it.findall('./Iteration_hits/Hit')[:n]
  if not hs: continue
  d=hs[0].findtext('Hit_def','')
  # coarse genus species from common forms
  if re.search(r'Homo sapiens|Human DNA|Human ',d): k='human'
  elif 'Sus scrofa' in d: k='sus scrofa'
  elif 'Callithrix jacchus' in d: k='callithrix jacchus'
  elif 'Halichoerus grypus' in d: k='halichoerus grypus'
  elif 'Microtus pennsylvanicus' in d: k='microtus pennsylvanicus'
  elif '18S rRNA' in d or '28S rRNA' in d or 'ribosomal RNA' in d or 'rRNA' in d: k='other rRNA'
  elif not d: k='no hit'
  else: k=d.split(':')[0][:40]
  c[k]=c.get(k,0)+1
 print('n',n,sorted(c.items(),key=lambda x:-x[1])[:20])
PY"
