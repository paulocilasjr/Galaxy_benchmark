/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET,re,collections
root=ET.parse('tmp/blast_pig.xml').getroot()
print('iterations',len(root.findall('.//Iteration')))
counts=collections.Counter()
for it in root.findall('.//Iteration'):
 q=it.findtext('Iteration_query-def')
 hits=it.findall('./Iteration_hits/Hit')
 if not hits:
  print(q,'NOHIT'); continue
 print('\\n'+q)
 for h in hits[:5]:
  hs=h.find('./Hit_hsps/Hsp')
  d=h.findtext('Hit_def','')
  print(d[:180], 'id',hs.findtext('Hsp_identity'),'aln',hs.findtext('Hsp_align-len'),'q',hs.findtext('Hsp_query-from')+'-'+hs.findtext('Hsp_query-to'),'e',hs.findtext('Hsp_evalue'))
 # species crude first hit
 d=hits[0].findtext('Hit_def','')
 m=re.search(r'(?:(?:PREDICTED: )?)([A-Z][a-z]+ [a-z]+)',d)
 counts[m.group(1) if m else d[:30]]+=1
print('\\nfirst-hit species',counts)
PY"
