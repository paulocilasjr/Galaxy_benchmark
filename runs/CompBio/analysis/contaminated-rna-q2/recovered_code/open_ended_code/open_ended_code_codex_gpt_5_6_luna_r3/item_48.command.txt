/bin/bash -lc "awk 'NR%4==1{h=substr("'$0,2)} NR%4==2{print ">" h "'"\\n\" "'$0}'"' results/unmapped.fq | head -400 > tmp/unmapped.100.fa
curl -sS --fail --max-time 30 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data-urlencode CMD=Put --data-urlencode PROGRAM=blastn --data-urlencode DATABASE=nt --data-urlencode QUERY@tmp/unmapped.100.fa --data-urlencode HITLIST_SIZE=10 --data-urlencode MEGABLAST=on > tmp/blast_put100.txt
grep -A2 -B2 'RID =' tmp/blast_put100.txt | tail -8"
