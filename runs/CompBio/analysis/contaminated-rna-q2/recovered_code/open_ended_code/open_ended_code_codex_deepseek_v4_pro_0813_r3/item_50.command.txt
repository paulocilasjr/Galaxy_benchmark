/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict,Counter
best=defaultdict(lambda: {'HUMAN':(-1,0),'MOUSE':(-1,0),'RAT':(-1,0)})
for line in open('hmr_all.paf'):
    f=line.rstrip().split('\\t')
    if len(f)<12: continue
    sp=f[5].split('|',1)[0]
    if sp not in best[f[0]]: continue
    matches=int(f[9]); block=int(f[10])
    if (matches,block)>best[f[0]][sp]:
        best[f[0]][sp]=(matches,block)
counts=Counter()
for q,d in best.items():
    vals={sp:d[sp][0] for sp in d}
    ordered=sorted(vals.items(), key=lambda kv: kv[1], reverse=True)
    counts[(ordered[0][0], ordered[0][1]-ordered[1][1] >= 5)] += 1
print(counts)
# print total best per sp
simple=Counter()
for q,d in best.items():
    simple[max(d, key=lambda sp:d[sp])] +=1
print(simple)
PY"
