/bin/bash -lc "curl -L --fail --retry 2 -o bat_refseq_rna.fna.gz https://ftp.ncbi.nlm.nih.gov/genomes/all/GCF/951/640/355/GCF_951640355.1_mEptNil1.1/GCF_951640355.1_mEptNil1.1_rna.fna.gz
ls -lh bat_refseq_rna.fna.gz
minimap2 -x sr --secondary=no -t 8 bat_refseq_rna.fna.gz contaminated.rna.q2.fq.gz > all_bat.paf
awk '{print "'$1}'"' all_bat.paf | sort -u | wc -l
awk '{x+="'$10;y+=$11;if($10/$11>=0.95)h++} END{print "alignments="NR,"identity="x/y,"ge95="h}'"' all_bat.paf"
