/bin/bash -lc "blastdbcmd -db mito_refseq_db -entry NC_012920.1 -out human_mito.fa
minimap2 -x sr --secondary=no -t 8 human_mito.fa contaminated.rna.q2.fq.gz > human_mito.paf
awk '{print "'$1}'"' human_mito.paf | sort -u > human_mito.ids
sort -u human_mapped.ids human_mito.ids > all_human_rna_mito.ids
wc -l human_mito.ids all_human_rna_mito.ids
awk 'NR==FNR{m["'$1]=1;next} NR%4==1{h=$0; id=$0; sub(/''^@/,"",id); sub(/ .*/,"",id); keep=!(id in m)} keep{if(NR%4==1){sub(/''^@/ , ">", h); print h}else if(NR%4==2)print}'"' all_human_rna_mito.ids <(gzip -cd contaminated.rna.q2.fq.gz) > nonhuman_candidate.fa
awk 'END{print \"candidate_reads=\" NR/2}' nonhuman_candidate.fa
head -6 nonhuman_candidate.fa"
