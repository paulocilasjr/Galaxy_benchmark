/bin/bash -lc "curl -L --fail --retry 2 -o mouse_refseq_rna.fna.gz https://ftp.ncbi.nlm.nih.gov/genomes/refseq/vertebrate_mammalian/Mus_musculus/all_assembly_versions/GCF_000001635.27_GRCm39/GCF_000001635.27_GRCm39_rna.fna.gz
ls -lh mouse_refseq_rna.fna.gz
minimap2 -x sr --secondary=no -t 8 mouse_refseq_rna.fna.gz nonhuman_candidate.fa > candidate_mouse.paf
awk '{print "'$1}'"' candidate_mouse.paf | sort -u | wc -l
awk '{x+="'$10;y+=$11} END{print "aligned_bases="x,"block_bases="y,"identity="x/y}'"' candidate_mouse.paf"
