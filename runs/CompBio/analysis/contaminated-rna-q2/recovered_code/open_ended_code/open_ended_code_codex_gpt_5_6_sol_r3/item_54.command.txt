/bin/bash -lc "blastn -query genome_unmapped.fa -db mito_refseq_db -task megablast -outfmt '6 qseqid saccver pident length qcovs evalue bitscore stitle' -max_target_seqs 5 -evalue 1e-8 -num_threads 8 > genome_unmapped.mito_blast.tsv
