/bin/bash -lc "curl -L --fail --retry 2 -o mitochondrion_refseq.fna.gz https://ftp.ncbi.nlm.nih.gov/refseq/release/mitochondrion/mitochondrion.1.1.genomic.fna.gz
ls -lh mitochondrion_refseq.fna.gz
gzip -cd mitochondrion_refseq.fna.gz > mitochondrion_refseq.fna
makeblastdb -in mitochondrion_refseq.fna -dbtype nucl -parse_seqids -out mito_refseq_db > makeblastdb.log
blastn -query unitigs_top200.fa -db mito_refseq_db -task megablast -outfmt '6 qseqid sseqid pident length qcovs evalue bitscore stitle' -max_target_seqs 10 -evalue 1e-15 > unitigs.mito.tsv
head -30 unitigs.mito.tsv"
