# parse ebv alignments with strong
from collections import Counter

def parse(sam):
    d={}
    for line in open(sam):
        if line.startswith('@'): continue
        p=line.rstrip().split('\\t')
        flag=int(p[1]); rid=p[0]
        if flag & 0x100: continue
        if p[2]=='*': d[rid]=None; continue
        as_=None; nm=None
        for tag in p[11:]:
            if tag.startswith('AS:i:'): as_=int(tag[5:])
            if tag.startswith('NM:i:'): nm=int(tag[5:])
        if rid not in d or (as_ or 0)>(d[rid][0] or 0): d[rid]=(as_,nm,p[2])
    return d
h=parse('human_all.sam'); m=parse('mouse_all.sam'); e=parse('ebv_all.sam')
ebv_strong=[]
for rid,ev in e.items():
    if ev and ev[1] is not None and ev[1]<=2:
        hs=h.get(rid); hv=hs[0] if hs else -1
        ms=m.get(rid); mv=ms[0] if ms else -1
        ebv_strong.append((rid,ev[0],ev[1],hv,mv,ev[2]))
print('ebv strong',len(ebv_strong))
# how many also map to human/mouse
print('also human >= ebv as',sum(1 for x in ebv_strong if x[3]>=x[1]))
print('also mouse >= ebv as',sum(1 for x in ebv_strong if x[4]>=x[1]))
for x in ebv_strong[:20]: print(x)
PY"
