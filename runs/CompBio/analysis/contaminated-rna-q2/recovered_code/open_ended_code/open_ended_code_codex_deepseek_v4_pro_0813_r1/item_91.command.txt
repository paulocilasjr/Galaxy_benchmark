/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def parse(sam):
    d={}
    for line in open(sam):
        if line.startswith('@'): continue
        p=line.rstrip().split('\\t')
        flag=int(p[1]); rid=p[0]
        if flag & 0x100: continue
        if p[2]=='*': d[rid]=None; continue
        as_=None; nm=None
        for tag in p[11:]:
            if tag.startswith('AS:i:'): as_=int(tag[5:])
            if tag.startswith('NM:i:'): nm=int(tag[5:])
        if rid not in d or (as_ or 0)>(d[rid][0] or 0): d[rid]=(as_,nm,p[2])
    return d
h=parse('human_all.sam'); m=parse('mouse_all.sam'); r=parse('rat_all.sam')
cnt=Counter(); examples=[]
for rid in set(h)|set(m)|set(r):
    hv=h.get(rid); mv=m.get(rid); rv=r.get(rid)
    hs=hv[0] if hv else -1; ms=mv[0] if mv else -1; rs=rv[0] if rv else -1
    best=max(hs,ms,rs)
    if rs>hs and rs>ms: cat='rat_best'
    elif rs>=0 and rs==best and (hs==rs or ms==rs): cat='tie_with_rat'
    elif ms>hs and ms>=rs: cat='mouse_best_or_tie'
    elif hs>=ms and hs>=rs: cat='human_best_or_tie'
    else: cat='other'
    cnt[cat]+=1
    if cat=='rat_best' and len(examples)<30:
        examples.append((rid,hs,ms,rs,hv[2] if hv else None,mv[2] if mv else None,rv[2] if rv else None,hv[1] if hv else None,mv[1] if mv else None,rv[1] if rv else None))
print(cnt)
print('rat_best',cnt['rat_best'])
for e in examples: print(e)
PY"
