/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def parse(sam):
    d={}
    for line in open(sam):
        if line.startswith('@'): continue
        p=line.rstrip().split('\\t')
        flag=int(p[1]); rid=p[0]
        if flag & 0x100: continue
        if p[2]=='*':
            d[rid]=None; continue
        as_=None; nm=None
        for tag in p[11:]:
            if tag.startswith('AS:i:'): as_=int(tag[5:])
            if tag.startswith('NM:i:'): nm=int(tag[5:])
        # keep best alignment for read (primary should be best)
        if rid not in d or (as_ or 0)>(d[rid][0] or 0):
            d[rid]=(as_,nm,p[2])
    return d
h=parse('human_all.sam')
m=parse('mouse_all.sam')
all_ids=set(h)|set(m)
cnt=Counter()
examples=[]
for rid in all_ids:
    hv=h.get(rid); mv=m.get(rid)
    hs=hv[0] if hv else -1
    ms=mv[0] if mv else -1
    if hs<0 and ms<0: cat='neither'
    elif hs<0 and ms>=0: cat='mouse_only'
    elif ms<0 and hs>=0: cat='human_only'
    elif hs>ms: cat='human_better'
    elif ms>hs: cat='mouse_better'
    else: cat='tie'
    cnt[cat]+=1
    if cat in ('mouse_only','mouse_better') and len(examples)<20:
        examples.append((rid,cat,hs,ms,hv[2] if hv else None,mv[2] if mv else None,hv[1] if hv else None,mv[1] if mv else None))
print(cnt)
print('mouse_only+mouse_better',cnt['mouse_only']+cnt['mouse_better'])
for e in examples: print(e)
PY"
