/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict, Counter
best={}
for line in open('remaining_rrna.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    flag=int(p[1]); rid=p[0]; ref=p[2]
    if flag & 0x900: # supplementary/secondary? skip secondary but keep primary
        pass
    if flag & 0x100: # secondary
        continue
    if ref=='*': continue
    as_=None; nm=None
    for tag in p[11:]:
        if tag.startswith('AS:i:'): as_=int(tag[5:])
        if tag.startswith('NM:i:'): nm=int(tag[5:])
    key=(ref, as_ if as_ is not None else -1, -(nm or 999))
    if rid not in best or key>best[rid][0]:
        best[rid]=(key,ref,as_,nm)
counts=Counter(v[1] for v in best.values())
print('best ref counts:', counts)
for ref in ['NR_003286.4','NR_003278.3','NR_003287.4','NR_003279.1']:
    c=counts[ref]; print(ref,c)
# among reads that map to rRNA, count those better to human vs mouse
h=counts['NR_003286.4']+counts['NR_003287.4']
m=counts['NR_003278.3']+counts['NR_003279.1']
print('human_ref',h,'mouse_ref',m)
PY"
