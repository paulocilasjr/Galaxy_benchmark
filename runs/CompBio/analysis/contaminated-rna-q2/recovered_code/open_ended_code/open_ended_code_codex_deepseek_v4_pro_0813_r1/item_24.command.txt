/bin/bash -lc "python3 - <<'PY'
import gzip
# parse kraken output taxids
wanted = {}
for line in open('kraken_output.txt'):
    p=line.rstrip('\\n').split('\\t')
    if p[0]=='C' and p[2] not in ('9606','0','131567','1','2'):
        wanted.setdefault(p[2], []).append(p[1])
print({k:len(v) for k,v in wanted.items()})
# extract reads for top nonhuman taxids from fastq
ids=set()
for tax, arr in wanted.items():
    if len(arr)>=10: # EBV, staph, bacillus, velezensis? 
        ids.update(arr)
with gzip.open('contaminated.rna.q2.fq.gz','rt') as fin:
    with open('nonhuman_top.fq','w') as out:
        while True:
            h=fin.readline().rstrip('\\n')
            if not h: break
            s=fin.readline().rstrip('\\n')
            plus=fin.readline().rstrip('\\n')
            q=fin.readline().rstrip('\\n')
            rid=h[1:].split()[0]
            if rid in ids:
                out.write(f'{h}\\n{s}\\n{plus}\\n{q}\\n')
print('written')
PY
ls -lh nonhuman_top.fq"
