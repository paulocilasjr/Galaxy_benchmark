/bin/bash -lc "python3 - <<'PY'
import gzip
# generate fasta for 10 mouse_only ids from earlier list plus some EBV/Staph
ids=['READ_112286','READ_170270','READ_11515','READ_177800','READ_181730','READ_64394','READ_78380','READ_133144','READ_209586','READ_244']
with gzip.open('contaminated.rna.q2.fq.gz','rt') as f, open('blast_check.fa','w') as out:
    while True:
        h=f.readline().rstrip('\\n')
        if not h: break
        s=f.readline().rstrip('\\n'); plus=f.readline().rstrip('\\n'); q=f.readline().rstrip('\\n')
        rid=h[1:].split()[0]
        if rid in ids:
            out.write(f'>{rid}\\n{s}\\n')
PY
wc -l blast_check.fa; head -4 blast_check.fa"
