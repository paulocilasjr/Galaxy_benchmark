/bin/bash -lc 'for spec in pig ebv sars staph_cohnii staph_haemolyticus; do bam=$([ "$spec" = pig ] && echo pig_map.bam || echo "$spec".bam); echo "=== $spec"; samtools view -F 4 "$bam" | awk '"'{n="'$1;sub(/''^READ_/,"",n); print n}'"' | sort -n | awk 'NR==1{first="'$1} {last=$1; if($1<100000)low++; else high++} END{print "n",NR,"first",first,"last",last,"low",low,"high",high}'"'; done
printf '\\n10k bins exact mapping counts:\\n'; for spec in pig ebv sars staph_cohnii staph_haemolyticus; do bam="'$([ "$spec" = pig ] && echo pig_map.bam || echo "$spec".bam); samtools view -F 4 "$bam" | awk -v s="$spec" '"'{n="'$1;sub(/''^READ_/,"",n); print (int(n/10000)*10000+1),s}'"' ; done | sort -n -k1,1 | awk '{a["'$1]=a[$1]" "$2} END{for(k in a)print k,a[k]}'"' | sed -n '1,30p'"
