/bin/bash -lc "cut -f2 unique_best_available.tsv | sed 's/\\.[0-9]*"'$//'"' | sort -u > avail_acc.txt
split -l 50 -a 2 avail_acc.txt avail_acc_
: > avail_metadata.tsv
for f in avail_acc_*; do ids="'$(paste -sd, "$f"); curl -sS --max-time 60 -X POST '"'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi' --data-urlencode db=nuccore --data-urlencode \"id="'${ids}" -o tmp_avail.xml; awk -F'"'[<>]' '/Item Name=\"Caption\"/{c="'$3} /Item Name="Title"/{t=$3} /Item Name="TaxId"/{tax=$3} /Item Name="AccessionVersion"/{print c"'"\\t\"tax\"\\t\"t; c=t=tax=\"\"}' tmp_avail.xml >> avail_metadata.tsv; done
sort -u avail_metadata.tsv > avail_metadata_unique.tsv
awk 'NR==FNR{count[\"U\" NR]="'$1; next}{if($1 in count) print count[$1]"'"\\t\""'$2"'"\\t\""'$3"'"\\t\""'$4"'"\\t\""'$1}'"' unmapped2_seq_counts.tsv unique_best_available.tsv | sort -nr > weighted_available.tsv
awk 'NR==FNR{tax["'$1]=$2;title[$1]=$3;next}{a=$2;sub(/'"\\.[0-9]+"'$/,"",a);n[tax[a]]+=$1;q[tax[a]]++;hi[tax[a]]+=($3>=99?$1:0)}END{for(t in n)print n[t]"'"\\t\"q[t]\"\\t\"hi[t]\"\\t\"t\"\\t\"title[t]}' avail_metadata_unique.tsv weighted_available.tsv | sort -nr | sed -n '1,80p'"
