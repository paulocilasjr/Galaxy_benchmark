/bin/bash -lc "samtools fastq -f 4 -0 unmapped.fq -n human_map.bam 2> /dev/null && gzip -c unmapped.fq > unmapped.fq.gz
printf 'unmapped records: '; awk 'END{print NR/4}' unmapped.fq
printf 'unique sequences: '; awk 'NR%4==2{print}' unmapped.fq | sort -u | wc -l
printf 'top exact sequences:\\n'; awk 'NR%4==2{print}' unmapped.fq | sort | uniq -c | sort -nr | head -30
printf 'lengths:\\n'; awk 'NR%4==2{n[length("'$0)]++}END{for(l in n)print l,n[l]}'"' unmapped.fq | sort -n
ls -lh unmapped.fq.gz"
