/bin/bash -lc "curl -sS --max-time 60 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' \\
  --data-urlencode 'CMD=Put' \\
  --data-urlencode 'PROGRAM=blastn' \\
  --data-urlencode 'DATABASE=nt' \\
  --data-urlencode 'MEGABLAST=on' \\
  --data-urlencode 'QUERY@tiny.fa' \\
  -o blast_put.txt
wc -c blast_put.txt; sed -n '1,40p' blast_put.txt"
