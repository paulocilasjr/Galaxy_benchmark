/bin/bash -lc "curl -sS --max-time 30 -X POST 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi' --data-urlencode db=taxonomy --data-urlencode 'id=3050299,10376,29382,9823,2697049' -o taxa.xml
awk -F'[<>]' '/<ScientificName>/{s="'$3} /<TaxId>/{id=$3} /<Rank>/{r=$3} /<Lineage>/{print id"'"\\t\"r\"\\t\"s\"\\t\""'$3;id=r=s=""}'"' taxa.xml | sed -n '1,30p'
sed -n '1,160p' taxa.xml | grep -E '<(TaxId|ScientificName|Rank|Lineage)' | head -40"
