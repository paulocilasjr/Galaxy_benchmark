/bin/bash -lc '# Build per-read metrics for candidate-reference mappings
for spec in ebv sars staph_cohnii pig_map; do bam=$([ "$spec" = pig_map ] && echo pig_map.bam || echo "$spec.bam"); samtools view -F 4 "$bam" | awk -v s="$spec" '"'{q=\"\";for(i=12;i<=NF;i++)if("'$i~/''^NM:/)q=$i; print $1"'"\\t\"s\"\\t\"q}' ; done > map_labels.tsv
awk 'NR%4==1{h="'$0;sub(/''^@/,"",h)} NR%4==2{seq=$0} NR%4==4{qual=$0; q=0;for(i=1;i<=length(qual);i++){x=ord=0; c=substr(qual,i,1); # approximate phred by ASCII via awk unavailable
} gc=gsub(/[GCgc]/,"",seq); print h"'"\\t\"length(seq)\"\\t\"gc}' unmapped2.fq > read_gc.tsv
for spec in ebv sars staph_cohnii pig_map; do echo -n \""'$spec "; awk -v s="$spec" '"'"'$2==s{print $1}'"' map_labels.tsv | sort -u | while read id; do awk -v id=\""'$id" '"'"'$1==id{print $3}'"' read_gc.tsv; done | awk '{n++;sum+="'$1}END{print n,sum/n}'"'; done"
