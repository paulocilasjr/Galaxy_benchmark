/bin/bash -lc "gzip -cd contaminated.rna.q2.fq.gz | awk 'NR%4==1{h="'$0} NR%4==2{if((++n%100)==0){print h; print $0}}'"' > sample_every100.fa
wc -l sample_every100.fa; du -h sample_every100.fa
blastn -remote -db nt -query sample_every100.fa -task blastn -evalue 1e-10 -max_target_seqs 10 -outfmt '6 qseqid saccver pident length mismatch gapopen qstart qend sstart send evalue bitscore staxids sscinames sskingdoms' -out sample_every100.nt.tsv"
