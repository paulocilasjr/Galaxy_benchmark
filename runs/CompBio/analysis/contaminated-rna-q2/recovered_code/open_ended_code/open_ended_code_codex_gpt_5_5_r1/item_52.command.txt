/bin/bash -lc "samtools view -F 256 results/unmapped_to_mus_caroli.bam | awk '{nm=\"\"; for(i=12;i<=NF;i++) if("'$i~/''^NM:i:/) nm=substr($i,6); if(nm==0) print $1}'"' | head -n 100 > results/mus_caroli_perfect_100.ids
awk 'NR==FNR{ids["'$1]=1; next} NR%4==1{h=substr($0,2); keep=(h in ids)} keep && NR%4==1{print ">"h} keep && NR%4==2{print}'"' results/mus_caroli_perfect_100.ids results/unmapped_human_tx.fq > results/mus_caroli_perfect_100.fa
awk 'BEGIN{c=0}/"'^>/{c++}END{print c}'"' results/mus_caroli_perfect_100.fa
blastn -db nt -query results/mus_caroli_perfect_100.fa -remote -task megablast -max_target_seqs 5 -perc_identity 95 -qcov_hsp_perc 90 -outfmt '6 qseqid sacc staxids pident length evalue bitscore' -out results/mus_caroli_perfect_100_nt.tsv"
