/bin/bash -lc "curl -L --retry 3 -o refs/rat_norvegicus_rna.fna.gz https://ftp.ncbi.nlm.nih.gov/genomes/all/GCF/015/227/675/GCF_015227675.2_mRatBN7.2/GCF_015227675.2_mRatBN7.2_rna.fna.gz
ls -lh refs/rat_norvegicus_rna.fna.gz
minimap2 -d refs/rat_norvegicus_rna.mmi refs/rat_norvegicus_rna.fna.gz
minimap2 -t 4 -ax sr refs/rat_norvegicus_rna.mmi results/unmapped_human_genome.fq 2> results/map_rat_unmapped_humangenome.log | samtools view -b -F 4 - > results/humangenome_unmapped_to_rat.bam
printf 'rat '; samtools view -c results/humangenome_unmapped_to_rat.bam
printf 'mouse '; samtools view -c results/humangenome_unmapped_to_mus_musculus.bam
printf 'caroli '; samtools view -c results/humangenome_unmapped_to_mus_caroli.bam
for sp in rat mus_musculus mus_caroli; do case "'$sp in rat) bam=results/humangenome_unmapped_to_rat.bam;; *) bam=results/humangenome_unmapped_to_${sp}.bam;; esac; echo $sp; samtools view -F 256 $bam | awk '"'BEGIN{tot=0;p=0;hi=0}{tot++; nm=\"\"; for(i=12;i<=NF;i++) if("'$i~/''^NM:i:/) nm=substr($i,6); if(nm==0)p++; if(nm!="" && nm<=2)hi++} END{print "primary",tot,"NM0",p,"NM<=2",hi}'"'; done"
