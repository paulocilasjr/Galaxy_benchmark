/bin/bash -lc 'for sp in mus_caroli mus_musculus; do
  bam=results/unmapped_to_${sp}.bam
  echo $sp
  samtools view -F 256 "$bam" | awk '"'BEGIN{tot=0; perfect=0; high=0; as150=0} {tot++; nm=\"\"; as=\"\"; for(i=12;i<=NF;i++){if("'$i~/''^NM:i:/) nm=substr($i,6); if($i~/''^AS:i:/) as=substr($i,6)} if(nm==0) perfect++; if(nm<=2 && nm!="") high++; if(as>=145) as150++} END{print "primary",tot,"NM0",perfect,"NM<=2",high,"AS>=145",as150}'"'
done"
