/bin/bash -lc "{
  gzip -dc refs/rat_norvegicus_rna.fna.gz | awk '/"'^>/{sub(/''^>/,">rn|"); print; next}{print}'"'
  gzip -dc refs/mus_musculus_rna.fna.gz | awk '/"'^>/{sub(/''^>/,">mm|"); print; next}{print}'"'
  gzip -dc refs/mus_caroli_rna.fna.gz | awk '/"'^>/{sub(/''^>/,">mc|"); print; next}{print}'"'
} > refs/rodents_rna.fa
minimap2 -d refs/rodents_rna.mmi refs/rodents_rna.fa
minimap2 -t 4 -ax sr refs/rodents_rna.mmi results/unmapped_human_genome.fq 2> results/map_rodents_unmapped_humangenome.log | samtools view -F 4 - > results/humangenome_unmapped_to_rodents.sam
awk '"'$0 !~ /''^@/ && and($2,256)==0 {split($3,a,"|"); pref=a[1]; nm=""; for(i=12;i<=NF;i++) if($i~/''^NM:i:/) nm=substr($i,6); total[pref]++; if(nm==0) perfect[pref]++; if(nm!="" && nm<=2) hi[pref]++} END{for(p in total) print p,total[p],perfect[p]+0,hi[p]+0}'"' results/humangenome_unmapped_to_rodents.sam | sort"
