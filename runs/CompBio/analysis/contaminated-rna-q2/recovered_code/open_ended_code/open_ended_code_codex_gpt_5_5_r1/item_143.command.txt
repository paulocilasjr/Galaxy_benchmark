/bin/bash -lc "perl -lane 'next if /"'^@/; next if ($F[1] & 256) || ($F[1] & 4); @a=split(/'"\\|/,"'$F[2]); next unless $a[0] eq "mm"; $nm=""; for($i=11;$i<@F;$i++){if($F[$i]=~/''^NM:i:('"\\d+)/){"'$nm=$1}} print "$F[0]'"\\t"'$F[2]" if $nm eq "0"'"' results/humangenome_unmapped_to_rodents.sam | head -n 50 > results/mouse_perfect_combined_50.tsv
cut -f2 results/mouse_perfect_combined_50.tsv | sed 's/"'^mm|//'"' | cut -d. -f1 | sort -u | head -n 20 > results/mouse_perfect_accessions.txt
cat results/mouse_perfect_combined_50.tsv | head -n 20
ACC="'$(paste -sd, results/mouse_perfect_accessions.txt)
curl -s "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=nuccore&id=$ACC&retmode=json" -o results/mouse_perfect_accession_summary.json
python - <<'"'PY'
import json
j=json.load(open('results/mouse_perfect_accession_summary.json'))
for uid in j['result'].get('uids',[])[:20]:
    r=j['result'][uid]
    print(r.get('accessionversion'), r.get('title'), 'taxid='+str(r.get('taxid')))
PY"
