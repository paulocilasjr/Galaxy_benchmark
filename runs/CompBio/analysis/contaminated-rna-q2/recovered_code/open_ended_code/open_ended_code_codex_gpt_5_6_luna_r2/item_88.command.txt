/bin/bash -lc 'minimap2 -d chimp_cdna.mmi Pan_troglodytes.cdna.fa.gz > /dev/null 2> chimp_cdna.index.log & p1=$!
minimap2 -d bonobo_cdna.mmi Pan_paniscus.cdna.fa.gz > /dev/null 2> bonobo_cdna.index.log & p2=$!
minimap2 -d gorilla_cdna.mmi Gorilla_gorilla.cdna.fa.gz > /dev/null 2> gorilla_cdna.index.log & p3=$!
wait "$p1" "$p2" "$p3"
minimap2 -ax sr --secondary=no -t 4 chimp_cdna.mmi candidate_nonhuman.fq > candidate_chimp.sam 2> candidate_chimp.log & p1=$!
minimap2 -ax sr --secondary=no -t 4 bonobo_cdna.mmi candidate_nonhuman.fq > candidate_bonobo.sam 2> candidate_bonobo.log & p2=$!
minimap2 -ax sr --secondary=no -t 4 gorilla_cdna.mmi candidate_nonhuman.fq > candidate_gorilla.sam 2> candidate_gorilla.log & p3=$!
minimap2 -ax sr --secondary=no -t 4 human_cdna.mmi candidate_nonhuman.fq > candidate_human.sam 2> candidate_human.log & p4=$!
wait "$p1" "$p2" "$p3" "$p4"
for f in candidate_chimp.sam candidate_bonobo.sam candidate_gorilla.sam candidate_human.sam; do printf '"'%s ' \""'$f"; samtools view -c -F 4 "$f"; done
ls -lh candidate_*.sam'
