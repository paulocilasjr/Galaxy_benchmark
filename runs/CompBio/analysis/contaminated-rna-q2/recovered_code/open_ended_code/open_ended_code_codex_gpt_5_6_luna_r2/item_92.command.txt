/bin/bash -lc "for pair in 'chimp candidate_chimp.sam' 'bonobo candidate_bonobo.sam' 'gorilla candidate_gorilla.sam'; do set -- "'$pair; name=$1; f=$2; samtools view -F 4 "$f" | awk -v sp="$name" '"'BEGIN{best=-1} {q="'$1; as=-1; nm=""; for(i=12;i<=NF;i++){if($i ~ /''^AS:i:/){as=$i; sub(/''^AS:i:/,"",as)} if($i ~ /''^NM:i:/){nm=$i; sub(/''^NM:i:/,"",nm)}} if(as>=0 && (!(q in b) || as>b[q])){b[q]=as; n[q]=nm; mapq[q]=$5; cigar[q]=$6}} END{for(q in b) print q,sp,b[q],n[q],mapq[q],cigar[q]}'"' | sort -k1,1 > \""'$name.best.tsv"; done
printf '"'best rows/counts:\\n'; for f in chimp.best.tsv bonobo.best.tsv gorilla.best.tsv; do printf '%s ' \""'$f"; wc -l < "$f"; awk '"'{if("'$3>=140)n140++;if($3>=130)n130++;if($3>=100)n100++} END{print " >=140",n140," >=130",n130," >=100",n100}'"' \""'$f"; done
printf '"'\\nHighest-scoring representative rows:\\n'; for f in chimp.best.tsv bonobo.best.tsv gorilla.best.tsv; do echo ==="'$f; sort -k3,3nr "$f" | head -n 10; done'
