/bin/bash -lc "printf 'candidate EBV alignments total: '; samtools view -c -F 4 candidate_ebv.bam
printf 'primary exact full-length: '; samtools view -F 260 candidate_ebv.bam | awk '{nm=99;for(i=12;i<=NF;i++){if("'$i~/''^NM:i:/){nm=$i;sub(/''^NM:i:/,"",nm)}}; if($6=="75M"&&nm==0)n++} END{print n+0}'"'
printf 'primary by NM/full length:\\n'; samtools view -F 260 candidate_ebv.bam | awk '{nm=99;for(i=12;i<=NF;i++){if("'$i~/''^NM:i:/){nm=$i;sub(/''^NM:i:/,"",nm)}}; print (nm==0?"NM0":nm),$6}'"' | sort | uniq -c | sort -nr | head -n 20
printf 'EBV read ID bins (original):\\n'; samtools view -F 260 candidate_ebv.bam | awk '{print "'$1}'"' | sed 's/READ_//' | awk '{print int("'$1/10000)*10000}'"' | sort -n | uniq -c
printf 'unique EBV sequences: '; samtools fastq -F 4 candidate_ebv.bam 2>/dev/null | awk 'NR%4==2{print}' | sort -u | wc -l
printf 'EBV genomic span: '; samtools view -F 260 candidate_ebv.bam | awk '{print "'$4}'"' | sort -n | awk 'NR==1{a="'$1} {b=$1} END{print a,b,b-a+75}'"'"
