/bin/bash -lc "grep -a -c '<Iteration_query-def>' candidate_sample.xml; grep -a -c '<Hit_def>' candidate_sample.xml; awk '
/<Iteration_query-def>/{q="'$0; sub(/.*<Iteration_query-def>/,"",q); sub(/<.*/,"",q); hit=""; ident=""; alen=""}
/<Hit_def>/ && hit==""{hit=$0; sub(/.*<Hit_def>/,"",hit); sub(/<.*/,"",hit)}
/<Hsp_identity>/ && ident==""{ident=$0; sub(/.*<Hsp_identity>/,"",ident); sub(/<.*/,"",ident)}
/<Hsp_align-len>/ && alen==""{alen=$0; sub(/.*<Hsp_align-len>/,"",alen); sub(/<.*/,"",alen)}
/<'"\\/Iteration>/{if(q"'!="") print q "'"\\t\" ident \"/\" alen \"\\t\" hit; q=\"\"}
' candidate_sample.xml > candidate_sample_top.tsv
wc -l candidate_sample_top.tsv
cat candidate_sample_top.tsv | sed -n '1,40p'
printf '\\nTop-hit species normalized:\\n'; awk -F '\\t' '{print "'$3}'"' candidate_sample_top.tsv | sed -E 's/"'^PREDICTED: //'"' | sed -E 's/([A-Z][a-z]+ [a-z]+).*/\\1/' | sort | uniq -c | sort -nr | head -n 50"
