/bin/bash -lc "curl -L --fail --max-time 30 -sS 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_007605&rettype=fasta&retmode=text' > ebv_b958.fa
awk '/"'^>/{print;next}{printf "%s",$0} END{print ""}'"' ebv_b958.fa | head -c 300; echo
awk 'BEGIN{s=0}/"'^>/{if(s)print n,length(s);n=$0;s="";next}{s=s $0}END{print n,length(s)}'"' ebv_b958.fa
minimap2 -ax sr --secondary=no ebv_b958.fa candidate_nonhuman.fq 2> candidate_ebv.log | samtools view -b -o candidate_ebv.bam -
printf 'EBV mapped: '; samtools view -c -F 4 candidate_ebv.bam
printf 'EBV high identity/length reads:\\n'; samtools view -F 4 candidate_ebv.bam | awk '{nm=\"\";for(i=12;i<=NF;i++){if("'$i~/''^NM:i:/){nm=$i;sub(/''^NM:i:/,"",nm)}}; if(nm<=2 && $6~/''^[0-9]+M$/) print $1,nm,$6,$5}'"' | head -n 30
cat candidate_ebv.log"
