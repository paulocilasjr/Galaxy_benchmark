/bin/bash -lc "awk '{sc["'$1,$2]=$3; seen[$1]=1} END{for(q in seen){c=sc[q,"chimp"]+0;b=sc[q,"bonobo"]+0;g=sc[q,"gorilla"]+0; mx=c;if(b>mx)mx=b;if(g>mx)mx=g; win=(c==mx?"chimp,":"") (b==mx?"bonobo,":"") (g==mx?"gorilla":""); print q,c,b,g,mx,win}}'"' chimp.best.tsv bonobo.best.tsv gorilla.best.tsv | sort -k1,1 > primate_scores.tsv
printf 'queries with scores: '; wc -l < primate_scores.tsv
printf 'winner patterns:\\n'; awk '{print "'$6}'"' primate_scores.tsv | sort | uniq -c | sort -nr
printf 'strict max chimp examples:\\n'; awk '"'$6=="chimp,"{print}'"' primate_scores.tsv | head -n 40
printf 'strict max bonobo examples:\\n'; awk '"'$6=="bonobo,"{print}'"' primate_scores.tsv | head -n 40
printf 'strict max gorilla examples:\\n'; awk '"'$6=="gorilla"{print}'"' primate_scores.tsv | head -n 40"
