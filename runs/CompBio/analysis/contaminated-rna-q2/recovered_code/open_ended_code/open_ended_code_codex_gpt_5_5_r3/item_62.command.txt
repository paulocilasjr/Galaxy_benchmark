/bin/bash -lc 'mkdir -p refs/genomes
rat_url=$(curl -fsSL https://ftp.ensembl.org/pub/current_fasta/rattus_norvegicus/dna/ | sed -n '"'s/.*href=\"\\(["'^"]*dna'"\\.primary_assembly\\.fa\\.gz\\)\".*/\\1/p' | head -1)
echo \""'$rat_url"
curl -L --retry 3 -o refs/genomes/rattus_norvegicus.dna.primary_assembly.fa.gz "https://ftp.ensembl.org/pub/current_fasta/rattus_norvegicus/dna/${rat_url}"
ls -lh refs/genomes/rattus_norvegicus.dna.primary_assembly.fa.gz
minimap2 -t 4 -x sr refs/genomes/rattus_norvegicus.dna.primary_assembly.fa.gz nonhuman_enriched.fa > rattus_genome.paf
awk '"'{all["'$1]=1; if($10>=60 && $11>=60 && $10/$11>=0.95) hi[$1]=1; if($11>=70 && $10/$11>=0.99) vhi[$1]=1} END{print "rat_genome all", length(all), "hi95", length(hi), "hi99", length(vhi), "alns", NR}'"' rattus_genome.paf"
