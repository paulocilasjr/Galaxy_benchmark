/bin/bash -lc "curl -sS 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_007605.1&rettype=fasta&retmode=text' -o ebv.fa
curl -sS 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_045512.2&rettype=fasta&retmode=text' -o sars2.fa
wc -c ebv.fa sars2.fa
minimap2 -x sr -t 4 --secondary=no ebv.fa contaminated.rna.q2.fq.gz > ebv.paf 2> ebv.log
minimap2 -x sr -t 4 --secondary=no sars2.fa contaminated.rna.q2.fq.gz > sars2.paf 2> sars2.log
printf 'EBV '; cut -f1 ebv.paf | sort -u | wc -l
printf 'SARS2 '; cut -f1 sars2.paf | sort -u | wc -l
awk '{m="'$10/$11; if(m>=.9 && $11>=60)n++} END{print "EBV high",n}'"' ebv.paf
awk '{m="'$10/$11; if(m>=.9 && $11>=60)n++} END{print "SARS2 high",n}'"' sars2.paf"
