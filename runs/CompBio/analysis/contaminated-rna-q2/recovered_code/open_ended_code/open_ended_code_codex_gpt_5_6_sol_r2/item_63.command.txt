/bin/bash -lc "python -c 'import xml.etree.ElementTree as E, collections, re
r=E.parse(\"genome_unmapped_blast.xml\").getroot(); out=[]
for it in r.findall(\".//Iteration\"):
 q=it.findtext(\"Iteration_query-def\"); hit=it.find(\"./Iteration_hits/Hit\")
 if hit is None: continue
 d=hit.findtext(\"Hit_def\"); h=hit.find(\"./Hit_hsps/Hsp\"); ident=int(h.findtext(\"Hsp_identity\")); al=int(h.findtext(\"Hsp_align-len\")); ev=h.findtext(\"Hsp_evalue\")
 out.append((q,d,ident,al,ev))
for x in out: print(*x,sep=\"\\t\")' > genome_unmapped_firsthits.tsv
wc -l genome_unmapped_firsthits.tsv
cut -f2 genome_unmapped_firsthits.tsv | sed -E 's/"'^(PREDICTED: |TPA: )//; s/''^(Homo sapiens) .*/'"\\1/; s/"'^(Human DNA|Human chromosome|Human BAC|Human gammaherpesvirus 4|Human ORFeome) .*/'"\\1/; s/"'^(Mus musculus) .*/'"\\1/; s/"'^(Pan troglodytes) .*/'"\\1/; s/"'^(Gorilla gorilla gorilla) .*/'"\\1/; s/"'^(Severe acute respiratory syndrome coronavirus 2) .*/'"\\1/' | sort | uniq -c | sort -nr | head -n 60
awk -F'\\t' '"'$2 !~ /Homo sapiens|Human DNA|Human chromosome|Human BAC|Human ORFeome/ {print}'"' genome_unmapped_firsthits.tsv | head -n 80"
