/bin/bash -lc "awk 'BEGIN{srand(7);n=0} /"'^>/{h=$0;getline s;if(rand()<0.4 && n<200){print h; print s;n++}} END{print n > "/dev/stderr"}'"' nonhuman_sample.fa > blast200.fa
