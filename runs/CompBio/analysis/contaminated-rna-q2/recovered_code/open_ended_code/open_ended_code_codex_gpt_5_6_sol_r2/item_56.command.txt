/bin/bash -lc "awk 'BEGIN{srand(99);n=0} FNR%4==1{h=substr("'$0,2)} FNR%4==2{if(rand()<0.08&&n<300){print ">"h;print;n++}} END{print n > "/dev/stderr"}'"' genome_unmapped.fq > genome_unmapped_sample.fa
