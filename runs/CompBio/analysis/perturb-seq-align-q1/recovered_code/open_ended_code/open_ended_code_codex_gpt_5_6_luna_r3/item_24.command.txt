import h5py, numpy as np
from scipy.sparse import csc_matrix
from scipy.stats import rankdata
from sklearn.metrics import pairwise_distances
base='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-19bbba4fe4429f9f5da451cf/workspace/'

def read(fn):
 with h5py.File(base+fn,'r') as f:
  X=csc_matrix((f['X/data'][...],f['X/indices'][...],f['X/indptr'][...]),shape=tuple(f['X'].attrs['shape']))
  genes=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in f['var/_index'][...]],dtype=str)
  g=f['obs']['guide']; cats=[x.decode() if isinstance(x,bytes) else str(x) for x in g['categories'][...]]
  groups=np.array([cats[i] for i in g['codes'][...]],dtype=str)
 return X,genes,groups

def pseudobulk(X,genes,groups):
 names=np.unique(groups)
 out=np.zeros((len(names),X.shape[1]),dtype=float)
 for i,n in enumerate(names):
  v=np.asarray(X[groups==n,:].sum(axis=0)).ravel()
  out[i]=np.log1p(v/max(v.sum(),1)*1e6)
 return names,out

rnames,R=pseudobulk(rX,rgenes,rg); qnames,Q=pseudobulk(qX,qgenes,qg)
common=sorted(set(rgenes)&set(qgenes)); ri=np.array([np.where(rgenes==x)[0][0] for x in common]); qi=np.array([np.where(qgenes==x)[0][0] for x in common]);
