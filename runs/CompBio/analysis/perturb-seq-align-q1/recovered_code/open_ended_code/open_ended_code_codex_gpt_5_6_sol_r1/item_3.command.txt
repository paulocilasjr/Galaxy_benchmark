/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
f='perturb.seq.align.q1.ref.h5ad'
with h5py.File(f) as h:
 def dec(a): return [x.decode() if isinstance(x,bytes) else str(x) for x in a]
 guides=dec(h['obs/guide/categories'][:]); gc=h['obs/guide/codes'][:]
 targets=dec(h['obs/target_gene/categories'][:]); tc=h['obs/target_gene/codes'][:]
 print('guides:', guides)
 print('targets:', targets)
 for target in ['PABPC1','NUDT21','LEO1']:
  ti=targets.index(target)
  vals,cnt=np.unique(gc[tc==ti],return_counts=True)
  print(target, [(guides[v],int(c)) for v,c in zip(vals,cnt)])
PY"
