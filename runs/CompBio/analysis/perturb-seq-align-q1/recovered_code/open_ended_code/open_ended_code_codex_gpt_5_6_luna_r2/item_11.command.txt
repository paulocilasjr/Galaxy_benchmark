/bin/bash -lc "python -u - <<'PY'
import h5py, numpy as np
from scipy.sparse import csc_matrix
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 with h5py.File(fn,'r') as f:
  nobs=len(f['obs/_index']); nvar=len(f['var/_index'])
  X=csc_matrix((f['X/data'][...],f['X/indices'][...],f['X/indptr'][...]),shape=(nobs,nvar))
  genes=f['var/_index'][...].astype(str)
  guidecats=f['obs/guide/categories'][...].astype(str)
  guidecodes=f['obs/guide/codes'][...]
  print(fn, 'shape',X.shape,'nnz',X.nnz,'sum',X.sum(),'median cell',np.median(np.asarray(X.sum(1)).ravel()),'median gene',np.median(np.asarray(X.sum(0)).ravel()))
  print('genes target', [(g, int((genes==g).sum())) for g in ['PABPC1','NUDT21','LEO1']])
  print('guide cells', dict(zip(guidecats, np.bincount(guidecodes,minlength=len(guidecats)))))
PY"
