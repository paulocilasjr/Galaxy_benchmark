/bin/bash -lc "python -u - <<'PY'
import h5py,numpy as np
from scipy.sparse import csc_matrix
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 with h5py.File(fn,'r') as f:
  nobs,nvar=len(f['obs/_index']),len(f['var/_index']); X=csc_matrix((f['X/data'][...],f['X/indices'][...],f['X/indptr'][...]),shape=(nobs,nvar)).tocsr(); genes=f['var/_index'][...].astype(str)
  s=np.asarray(X.sum(0)).ravel(); top=np.argsort(s)[::-1][:30]
  print('\\n',fn,'top genes by counts')
  print([(genes[i],int(s[i])) for i in top])
  for g in ['HBB','HBA1','HBA2','PPBP','PF4','NKG7','GNLY','IL7R','LTB','MALAT1','GAPDH','ACTB','KRT10','KRT19','VIM']:
   if g in genes: print(g,int(s[np.where(genes==g)[0][0]]))
PY"
