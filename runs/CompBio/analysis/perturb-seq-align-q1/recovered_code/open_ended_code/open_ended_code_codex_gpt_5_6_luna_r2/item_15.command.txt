import h5py, numpy as np
from scipy.sparse import csc_matrix, csr_matrix
from scipy.stats import pearsonr, spearmanr

def load(fn):
 with h5py.File(fn,'r') as f:
  nobs,nvar=len(f['obs/_index']),len(f['var/_index'])
  X=csc_matrix((f['X/data'][...],f['X/indices'][...],f['X/indptr'][...]),shape=(nobs,nvar)).tocsr()
  genes=f['var/_index'][...].astype(str)
  gc=f['obs/guide/categories'][...].astype(str)
  codes=f['obs/guide/codes'][...]
  return X,genes,gc,codes

def normlog(X):
 lib=np.asarray(X.sum(axis=1)).ravel().astype(float)
 Y=X.multiply(1e4/np.maximum(lib,1)[:,None]).tocsr()
 Y.data=np.log1p(Y.data)
 return Y,lib

def group_means(Y,gc,codes):
 out=[]
 for i,g in enumerate(gc):
  out.append(np.asarray(Y[codes==i].mean(axis=0)).ravel())
 return np.array(out)

R,lr=normlog(R); Q,lq=normlog(Q)

cq=np.isin(qc,[i for i,g in enumerate(qg) if g.startswith('NT-')])
