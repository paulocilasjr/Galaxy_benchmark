print('start')
import h5py
print('h5py')
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 print('\\n'+fn)
 with h5py.File(fn,'r') as f:
  def rec(g,p='',d=0):
   if d>2:return

    pp=p+'/'+k
    if hasattr(v,'shape'): print(pp, 'dataset',v.shape,v.dtype)
    else: print(pp, 'group')
    if hasattr(v,'items'):rec(v,pp,d+1)
  rec(f)
  print('attrs',dict(f.attrs))
