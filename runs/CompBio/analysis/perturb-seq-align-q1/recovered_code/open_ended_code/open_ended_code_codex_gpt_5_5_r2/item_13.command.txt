/bin/bash -lc "python - <<'PY'
import h5py, collections

def read_cat(f, path):
    cats=[x.decode() if isinstance(x, bytes) else str(x) for x in f[path+'/categories'][()]]
    codes=f[path+'/codes'][()]
    return cats, codes, [cats[c] if c>=0 else None for c in codes]
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as f:
    g_cats,g_codes,g=read_cat(f,'obs/guide')
    t_cats,t_codes,t=read_cat(f,'obs/target_gene')
    print('guide cats:', g_cats)
    print('target cats:', t_cats)
    counts=collections.Counter(zip(g,t))
    # show mapping per guide with target distribution
    for guide in g_cats:
        dist=collections.Counter(ti for gi,ti in zip(g,t) if gi==guide)
        print(guide, dist.most_common())
    print('targets requested')
    for target in ['PABPC1','NUDT21','LEO1']:
        gs=collections.Counter(gi for gi,ti in zip(g,t) if ti==target)
        print(target, gs.most_common())

with h5py.File('perturb.seq.align.q1.query.h5ad','r') as f:
    qcats=[x.decode() if isinstance(x,bytes) else str(x) for x in f['obs/guide/categories'][()]]
    print('query guide cats:', qcats)
PY"
