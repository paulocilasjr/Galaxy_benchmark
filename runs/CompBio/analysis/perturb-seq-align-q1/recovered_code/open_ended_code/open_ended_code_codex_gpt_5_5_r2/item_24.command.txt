import h5py, numpy as np
from scipy.sparse import csc_matrix, csr_matrix

def decode_arr(a): return np.array([x.decode() if isinstance(x,bytes) else str(x) for x in a])
def load(fn):
    with h5py.File(fn,'r') as f:
        shape=tuple(int(x) for x in f['X'].attrs['shape'])
        X=csc_matrix((f['X/data'][()], f['X/indices'][()], f['X/indptr'][()]), shape=shape).tocsr().astype(np.float32)
        guides=decode_arr(f['obs/guide/categories'][()]); codes=f['obs/guide/codes'][()].astype(np.int32)
        var=decode_arr(f['var/_index'][()])
    return X,guides,codes,var
def lognorm(X):
    X=X.copy().tocsr().astype(np.float32)
    lib=np.asarray(X.sum(axis=1)).ravel().astype(np.float32)
    scale=np.divide(1e4, lib, out=np.zeros_like(lib), where=lib>0)
    X.data *= np.repeat(scale, np.diff(X.indptr))
    np.log1p(X.data, out=X.data)
    return X
def means(X,codes,ng):
    G=csr_matrix((np.ones(codes.size,dtype=np.float32),(np.arange(codes.size),codes)), shape=(codes.size,ng))
    sums=(G.T@X).toarray().astype(np.float32)
    n=np.bincount(codes,minlength=ng).astype(np.float32)
    return sums/n[:,None],n
refX,refg,refc,refv=load('perturb.seq.align.q1.ref.h5ad')
qX,qg,qc,qv=load('perturb.seq.align.q1.query.h5ad')
ref_order=[]; q_order=[]
qpos={g:i for i,g in enumerate(qv)}
for i,g in enumerate(refv):

refM,refN=means(lognorm(refX),refc,len(refg)); qM,qN=means(lognorm(qX),qc,len(qg))

mask_defs.append(('ctrl>0.1', (ctrl_ref>0.1)&(ctrl_q>0.1)))

def corr(A,B):
    A=A.copy(); B=B.copy()
    A-=A.mean(1,keepdims=True); B-=B.mean(1,keepdims=True)
    A/=np.linalg.norm(A,axis=1,keepdims=True)+1e-9; B/=np.linalg.norm(B,axis=1,keepdims=True)+1e-9
