/bin/bash -lc "python - <<'PY'
# Print full Hungarian assignment for raw common signatures K=1000 and top conflicts around guide14
# Reuse compact script
import h5py, numpy as np, collections
from scipy import sparse
from scipy.optimize import linear_sum_assignment

def dec(a): return np.array([x.decode() if isinstance(x,(bytes,bytearray)) else str(x) for x in a])
def read(fn):
    with h5py.File(fn,'r') as f:
        X=sparse.csc_matrix((f['X/data'][()].astype(np.float32), f['X/indices'][()], f['X/indptr'][()]), shape=tuple(f['X'].attrs['shape']))
        var=dec(f['var/_index'][()]); gc=dec(f['obs/guide/categories'][()]); lab=gc[f['obs/guide/codes'][()]]
        tar=None
        if 'target_gene' in f['obs']:
            tc=dec(f['obs/target_gene/categories'][()]); tar=tc[f['obs/target_gene/codes'][()]]
        return X,var,gc,lab,tar
def lognorm(X):
    tot=np.asarray(X.sum(1)).ravel().astype(np.float32); tot[tot==0]=1; coo=X.tocoo(copy=True); coo.data=np.log1p(coo.data/tot[coo.row]*1e4).astype(np.float32); return coo.tocsr()
def means(X,lab,groups): return np.vstack([np.asarray(X[lab==g].mean(0)).ravel() for g in groups]).astype(np.float32)
def corr(A,B):
    A=A.astype(float); B=B.astype(float); A-=A.mean(1,keepdims=True); B-=B.mean(1,keepdims=True); return A@B.T/(np.linalg.norm(A,axis=1)[:,None]*np.linalg.norm(B,axis=1)[None,:]+1e-12)
Xr,vr,grc,gr,tr=read('perturb.seq.align.q1.ref.h5ad'); Xq,vq,gqc,gq,_=read('perturb.seq.align.q1.query.h5ad')
refmap={g:collections.Counter(tr[gr==g]).most_common(1)[0][0] for g in grc}
qpos={g:i for i,g in enumerate(vq)}; ridx=[]; qidx=[]
for i,g in enumerate(vr):
    if g in qpos: ridx.append(i); qidx.append(qpos[g])
Mr=means(lognorm(Xr)[:,ridx],gr,grc); Mq=means(lognorm(Xq)[:,qidx],gq,gqc)
ref_nt=[i for i,g in enumerate(grc) if g.startswith('NT-')]; q_nt=[i for i,g in enumerate(gqc) if g.startswith('NT-')]
ref_non=[i for i,g in enumerate(grc) if not g.startswith('NT-')]; q_non=[i for i,g in enumerate(gqc) if not g.startswith('NT-')]
refnames=[grc[i] for i in ref_non]; qnames=[gqc[i] for i in q_non]
Er=Mr[ref_non]-Mr[ref_nt].mean(0); Eq=Mq[q_non]-Mq[q_nt].mean(0); score=np.maximum(abs(Er).max(0),abs(Eq).max(0))
for k in [500,1000,2000,5000,11858]:
    idx=np.argsort(score)[-k:] if k<score.size else np.arange(score.size); C=corr(Er[:,idx],Eq[:,idx]); rows,cols=linear_sum_assignment(-C)
    assigned={refnames[r]:qnames[c] for r,c in zip(rows,cols)}; un=[g for g in refnames if g not in assigned]
    print('\\nK',k,'unassigned',[(g,refmap[g]) for g in un])
    for rg,qg in sorted(assigned.items(), key=lambda x:int(x[1].replace('guide',''))):
        if qg in ['guide10','guide14','guide15'] or refmap[rg] in ['LEO1','PAF1','CTR9','CDC73']:
            ri=refnames.index(rg); qi=qnames.index(qg)
            print(qg, rg, refmap[rg], round(float(C[ri,qi]),4),'rowbest',qnames[C[ri].argmax()],round(float(C[ri].max()),4),'colbest',refnames[C[:,qi].argmax()],refmap[refnames[C[:,qi].argmax()]],round(float(C[:,qi].max()),4))
PY"
