/bin/bash -lc "python - <<'PY'
import h5py

def dec(ds): return [x.decode() if isinstance(x,bytes) else str(x) for x in ds[:]]
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as fr, h5py.File('perturb.seq.align.q1.query.h5ad','r') as fq:
 r=dec(fr['var/_index']); q=dec(fq['var/_index'])
 print('same order?', r==q)
 print('only ref count', len(set(r)-set(q)))
 print(sorted(set(r)-set(q))[:100])
 print('only query count', len(set(q)-set(r)))
 print(sorted(set(q)-set(r))[:100])
 for gene in ['PABPC1','NUDT21','LEO1']:
  print(gene, 'ref index', r.index(gene) if gene in r else None, 'query at ref idx', q[r.index(gene)] if gene in r else None)
PY"
