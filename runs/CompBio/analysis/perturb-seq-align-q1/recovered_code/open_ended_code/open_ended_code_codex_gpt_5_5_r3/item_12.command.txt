/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from collections import Counter, defaultdict

def decode_arr(ds):
    return [x.decode() if isinstance(x, bytes) else str(x) for x in ds[:]]
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as f:
    guide_cats=decode_arr(f['obs/guide/categories'])
    target_cats=decode_arr(f['obs/target_gene/categories'])
    guide_codes=f['obs/guide/codes'][:]
    target_codes=f['obs/target_gene/codes'][:]
    print('guide categories')
    print(guide_cats)
    print('target categories')
    print(target_cats)
    by_target=defaultdict(Counter)
    for gc,tc in zip(guide_codes,target_codes):
        if gc >= 0 and tc >= 0:
            by_target[target_cats[int(tc)]][guide_cats[int(gc)]] += 1
    for t in ['PABPC1','NUDT21','LEO1']:
        print('REQUEST', t, dict(by_target[t].most_common()))
    print('ALL')
    for t in sorted(by_target):
        print(t, dict(by_target[t].most_common()))
with h5py.File('perturb.seq.align.q1.query.h5ad','r') as f:
    print('query guide categories')
    print(decode_arr(f['obs/guide/categories']))
PY"
