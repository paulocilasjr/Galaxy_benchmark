/bin/bash -lc "python - <<'PY'
import h5py
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 with h5py.File(fn,'r') as f:
  print('\\n',fn)
  print('root attrs', dict(f.attrs))
  for key in ['obs/_index','obs/guide/categories']:
   arr=f[key][:10]
   print(key, [x.decode() if isinstance(x,bytes) else str(x) for x in arr])
  def print_uns(name,obj):
   if name.startswith('uns'):
    if isinstance(obj,h5py.Dataset): print('D', name, obj.shape, obj.dtype, obj[:5] if obj.shape and obj.shape[0]<10 else '')
    else: print('G', name, dict(obj.attrs))
  f.visititems(print_uns)
PY"
