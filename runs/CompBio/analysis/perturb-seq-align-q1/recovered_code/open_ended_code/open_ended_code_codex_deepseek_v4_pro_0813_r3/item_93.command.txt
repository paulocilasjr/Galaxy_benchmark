/bin/bash -lc "python - <<'PY'
import gzip,time
fn='GSE269596_CPA_K562.Rds.gz'
offset=9312500000
count=3000000
chunk=16*1024*1024
print('skip to',offset)
t=time.time()
with gzip.open(fn,'rb') as f1, gzip.GzipFile(fileobj=f1, mode='rb') as f2:
    skipped=0
    while skipped<offset:
        data=f2.read(min(chunk, offset-skipped))
        if not data: raise EOFError
        skipped+=len(data)
    seg=f2.read(count)
open('k562_meta2_segment.bin','wb').write(seg)
print('wrote',len(seg),'time',time.time()-t)
PY
strings -n 3 k562_meta2_segment.bin | grep -E 'Singlet|Doublet|Negative|R1-L1|R1-L2|R2-L|rep[0-9]|Lane|lane|GUIDE|HTO|TRUE|FALSE|target_gene|guide' | head -200"
