/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np
from scipy import sparse
ref=ad.read_h5ad('perturb.seq.align.q1.ref.h5ad'); q=ad.read_h5ad('perturb.seq.align.q1.query.h5ad')
targets=set(ref.obs['target_gene'].astype(str).unique())-{'NT'}
common=[g for g in ref.var_names if g in set(q.var_names) and g not in targets]
r=ref[:,common].copy(); q=q[:,common].copy()
def norm_log(a):
 X=a.X.astype(float).tocsr(); lib=np.asarray(X.sum(1)).ravel(); sf=1e4/np.maximum(lib,1); X=sparse.diags(sf).dot(X); X.data=np.log1p(X.data); return X.toarray()
R=norm_log(r); Q=norm_log(q)
rguide=r.obs.guide.astype(str).values; qguide=q.obs.guide.astype(str).values
rnt=R[np.array([s.startswith('NT') for s in rguide])]; qnt=Q[np.array([s.startswith('NT') for s in qguide])]
rg=[g for g in r.obs.guide.cat.categories if not g.startswith('NT-')]; qg=[g for g in q.obs.guide.cat.categories if not g.startswith('NT-')]
def tsig(X,guide,nt,g):
    a=X[guide==g]; ma=a.mean(0); mb=nt.mean(0); va=a.var(0,ddof=1); vb=nt.var(0,ddof=1)
    se=np.sqrt(va/a.shape[0]+vb/nt.shape[0]); return (ma-mb)/(se+1e-6)
Tr=np.vstack([tsig(R,rguide,rnt,g) for g in rg]); Tq=np.vstack([tsig(Q,qguide,qnt,g) for g in qg])
def corr(A,B):
 A=A-A.mean(1,keepdims=True); B=B-B.mean(1,keepdims=True); A/=np.linalg.norm(A,axis=1,keepdims=True)+1e-9; B/=np.linalg.norm(B,axis=1,keepdims=True)+1e-9; return A@B.T
for k in [1000,3000,11857]:
 v=Tr.var(0)+Tq.var(0); idx=np.argsort(v)[-k:] if k<11857 else np.arange(Tr.shape[1])
 C=corr(Tr[:,idx],Tq[:,idx])
 print('\\nk',k)
 for target in ['PABPC1-I1','NUDT21-I3','LEO1-I2']:
  i=rg.index(target); top=np.argsort(C[i])[::-1][:10]
  print(target,[(qg[j],round(C[i,j],3)) for j in top])
PY"
