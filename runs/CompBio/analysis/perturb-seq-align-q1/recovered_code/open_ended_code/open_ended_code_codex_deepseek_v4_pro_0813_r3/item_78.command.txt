/bin/bash -lc "python - <<'PY'
import gzip, time, re, subprocess, os
fn='GSE269596_CPA_K562.Rds.gz'
offset=2712800000
count=20000000
chunk=16*1024*1024
print('skipping to',offset)
t=time.time()
with gzip.open(fn,'rb') as f1:
    with gzip.GzipFile(fileobj=f1, mode='rb') as f2:
        skipped=0
        while skipped<offset:
            data=f2.read(min(chunk, offset-skipped))
            if not data: raise EOFError((skipped,offset))
            skipped += len(data)
        seg=f2.read(count)
open('k562_guides_segment.bin','wb').write(seg)
print('wrote',len(seg),'time',time.time()-t)
# extract strings guide-like
out=subprocess.check_output(['strings','-n','4','k562_guides_segment.bin'], text=True)
pat=re.compile(r'"'^(?:[A-Z0-9]+-I'"\\d+|NT-I\\d+)"'$'"')
labels=[x.strip() for x in out.splitlines() if pat.match(x)]
open('k562_guide_like_labels.txt','w').write('\\n'.join(labels)+'\\n')
print('labels',len(labels))
from collections import Counter
print(Counter(labels).most_common(20))
PY"
