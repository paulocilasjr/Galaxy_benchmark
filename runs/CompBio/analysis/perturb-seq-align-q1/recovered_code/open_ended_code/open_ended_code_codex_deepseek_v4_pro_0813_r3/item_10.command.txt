/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np, pandas as pd
from scipy import sparse
ref=ad.read_h5ad('perturb.seq.align.q1.ref.h5ad')
q=ad.read_h5ad('perturb.seq.align.q1.query.h5ad')

def prep(a):
    X=a.X
    if sparse.issparse(X): X=X.toarray()
    lib=X.sum(axis=1,keepdims=True)
    cpm=X/lib*1e4
    return np.log1p(cpm), a.obs.copy(), np.array(a.var_names)

for name,a in [('ref',ref),('query',q)]:
    X,obs,genes=prep(a)
    guide=obs.guide.astype(str).values
    nt_mask=np.array([g.startswith('NT') for g in guide])
    nt_mean=X[nt_mask].mean(axis=0)
    print('\\n',name)
    uniq=sorted(set(guide))
    for g in uniq:
        if g.startswith('NT'): continue
        mask=guide==g
        m=X[mask].mean(axis=0)-nt_mean
        order=np.argsort(m)
        print(g, 'down:', ', '.join(genes[order[:8]]), 'scores', np.round(m[order[:3]],3))
PY"
