/bin/bash -lc "python - <<'PY'
import gzip,time
fn='GSE269596_CPA_K562.Rds.gz'
patterns=[b'Singlet',b'Doublet',b'Negative',b'R1-L2',b'R1-L1',b'rep1',b'rep2',b'GUIDE_classification',b'HTO_classification',b'Replicate']
chunk=1024*1024
buf=b''; pos=0; found={p.decode():[] for p in patterns}
t=time.time()
with gzip.open(fn,'rb') as f1, gzip.GzipFile(fileobj=f1, mode='rb') as f2:
    while True:
        data=f2.read(chunk)
        if not data: break
        block=buf+data
        base=pos-len(buf)
        for p in patterns:
            start=0
            while True:
                idx=block.find(p,start)
                if idx<0: break
                off=base+idx
                if len(found[p.decode()])<10:
                    found[p.decode()].append(off)
                start=idx+1
        pos+=len(data)
        buf=block[-200:]
        if time.time()-t>120:
            print('timeout at',pos); break
print({k:v for k,v in found.items() if v})
print('pos',pos)
PY"
