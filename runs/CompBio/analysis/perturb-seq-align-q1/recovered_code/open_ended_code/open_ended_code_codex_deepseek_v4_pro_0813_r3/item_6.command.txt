print('ref NT counts by guide', ref.obs[ref.obs.guide.str.startswith('NT')].guide.value_counts())
print('q NT counts by guide', q.obs[q.obs.guide.str.startswith('NT')].guide.value_counts())

qq=qv[~qv.index.str.startswith('NT')].sort_values()
