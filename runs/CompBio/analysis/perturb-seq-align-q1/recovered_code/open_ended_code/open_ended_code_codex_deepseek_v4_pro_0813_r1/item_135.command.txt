/bin/bash -lc "cat > inspect_meta.R <<'EOF'
suppressMessages({library(Seurat); library(Matrix)})
con<-pipe('gzip -dc CPA_K562.Rds.gz | gzip -dc','rb'); obj<-readRDS(con); close(con)
for(cc in c('guide','target_gene','cellline','replicate')) {cat(cc,'class',class(obj@meta.data[[cc]]),'head',paste(head(as.character(obj@meta.data[[cc]]),10),collapse=','),'nuniq',length(unique(as.character(obj@meta.data[[cc]]))),'\\n')}
cat('guide uniques\\n'); writeLines(sort(unique(as.character(obj"'$guide))))
cat('"'target uniques\\n'); writeLines(sort(unique(as.character(obj"'$target_gene))))
EOF
Rscript inspect_meta.R > inspect_meta.out 2>&1'
