/bin/bash -lc "python3 - <<'PY'
import h5py,numpy as np
for f in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 with h5py.File(f,'r') as h:
  arr=h['var/_index'][:]
  genes=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in arr])
 print(f,genes.shape,genes[:5])
 for target in ['PABPC1','NUDT21','LEO1']:
  idx=np.where(genes==target)[0]
  print(' ',target,idx, genes[idx] if len(idx) else '')
  # grep case insensitive
  matches=[(i,g) for i,g in enumerate(genes) if target.lower() in g.lower()]
  print(' contains',matches[:5])
PY"
