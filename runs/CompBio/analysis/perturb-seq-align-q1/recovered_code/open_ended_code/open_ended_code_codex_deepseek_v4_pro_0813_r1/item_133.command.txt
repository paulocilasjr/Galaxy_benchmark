/bin/bash -lc "cat > inspect_layers.R <<'EOF'
suppressMessages({library(Seurat); library(Matrix)})
con<-pipe('gzip -dc CPA_K562.Rds.gz | gzip -dc','rb'); obj<-readRDS(con); close(con)
rna<-obj@assays"'$RNA
cat('"'layers names', paste(names(rna@layers), collapse=', '), '\\n')
for(nm in names(rna@layers)) { cat(nm, class(rna@layers[[nm]]), dim(rna@layers[[nm]]), '\\n') }
cat('cells len', length(rna@cells), 'head', head(rna@cells,3), '\\n')
cat('features len', length(rna@features), 'head', head(rna@features,3), '\\n')
cat('meta guide table\\n'); print(table(obj"'$guide))
cat('"'target table\\n'); print(table(obj"'$target_gene))
EOF
Rscript inspect_layers.R > inspect_layers.out 2>&1'
