/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np, scipy.sparse as sp

def load(f):
    with h5py.File(f,'r') as h:
        X=sp.csr_matrix((h['X/data'][:], h['X/indices'][:], h['X/indptr'][:]), shape=(len(h['var/_index']), len(h['obs/_index']))).T.tocsr()
        guides=h['obs/guide/categories'][:].astype(str)
        codes=h['obs/guide/codes'][:]
        if 'obs/target_gene/categories' in h:
            targets=h['obs/target_gene/categories'][:].astype(str)
            tcodes=h['obs/target_gene/codes'][:]
        else:
            targets=None; tcodes=None
    return X, guides, codes, targets, tcodes

Xr, gr, cr, tr, tcr = load('perturb.seq.align.q1.ref.h5ad')
Xq, gq, cq, tq, tcq = load('perturb.seq.align.q1.query.h5ad')
# map guide idx -> target label
guide2target={}
for i,g in enumerate(gr):
    vals, counts=np.unique(tr[tcr[cr==i]], return_counts=True) if tr is not None else ([],[])
    if len(vals): guide2target[g]=vals[np.argmax(counts)]
print({g:guide2target[g] for g in gr if not g.startswith('NT-')})

def norm_log(X):
    sums=np.asarray(X.sum(1)).ravel().astype(float); sums[sums==0]=1
    return np.log1p(X.multiply(1/sums[:,None]*1e6).tocsr())

Lr=norm_log(Xr).toarray(); Lq=norm_log(Xq).toarray()
ur=np.array([Lr[cr==i].mean(0) for i in range(len(gr))])
uq=np.array([Lq[cq==i].mean(0) for i in range(len(gq))])
ntr=np.where(np.char.startswith(gr,'NT-'))[0]; ntq=np.where(np.char.startswith(gq,'NT-'))[0]
ref_ctrl=ur[ntr].mean(0); q_ctrl=uq[ntq].mean(0)
gene_var=ur.var(0)
ngenes=2000
top=np.argsort(gene_var)[::-1][:ngenes]
Er=ur-ref_ctrl; Eq=uq-q_ctrl
pr=[i for i,g in enumerate(gr) if not g.startswith('NT-')]
pq=[i for i,g in enumerate(gq) if not g.startswith('NT-')]
def corr_rows(A,B):
    A=A-A.mean(1,keepdims=True); B=B-B.mean(1,keepdims=True)
    denom=np.sqrt((A*A).sum(1,keepdims=True)) @ np.sqrt((B*B).sum(1,keepdims=True)).T
    denom[denom==0]=1e-12
    return (A@B.T)/denom
C=corr_rows(Eq[pq][:,top], Er[pr][:,top])
print('\\npairwise top matches:')
for ii,q in enumerate(pq):
    j=C[ii].argmax()
    print(f'{gq[q]:10s} -> {gr[pr[j]]:12s} {guide2target[gr[pr[j]]]:8s} {C[ii,j]:.3f}')
print('\\nDesired:')
for target in ['PABPC1','NUDT21','LEO1']:
    rinds=[ri for ri in pr if guide2target[gr[ri]]==target]
    print('target',target,'ref', [gr[ri] for ri in rinds])
    for ri in rinds:
        j=pr.index(ri); scores=C[:,j]
        order=np.argsort(scores)[::-1][:15]
        for k in order:
            print(' ',gq[pq[k]], f'{scores[k]:.3f}')
PY"
