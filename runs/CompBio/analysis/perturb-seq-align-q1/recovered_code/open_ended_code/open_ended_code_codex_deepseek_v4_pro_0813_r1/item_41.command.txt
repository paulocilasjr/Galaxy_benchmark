/bin/bash -lc "python3 - <<'PY'
import requests,re,html
u='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.supplementary-material'
text=requests.get(u,headers={'User-Agent':'Mozilla/5.0'},timeout=30).text
plain=html.unescape(re.sub('<["'^>]+>'"',' ',text))
print(plain[:5000])
print('hrefs')
for m in re.finditer(r'href=\"(["'^"]+)"'"',text):
 h=m.group(1)
 if 'supp' in h.lower() or 'xlsx' in h.lower() or 'zip' in h.lower() or 'csv' in h.lower() or 'tsv' in h.lower(): print(h)
PY"
