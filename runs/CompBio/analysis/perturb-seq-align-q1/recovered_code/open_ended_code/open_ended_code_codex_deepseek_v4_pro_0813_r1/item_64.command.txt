/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np, scipy.sparse as sp

def load(f):
    with h5py.File(f,'r') as h:
        X=sp.csr_matrix((h['X/data'][:], h['X/indices'][:], h['X/indptr'][:]), shape=(len(h['var/_index']), len(h['obs/_index']))).T.tocsr()
        guides=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['obs/guide/categories'][:]])
        codes=h['obs/guide/codes'][:]
        genes=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['var/_index'][:]])
        targets=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['obs/target_gene/categories'][:]]) if 'obs/target_gene/categories' in h else None
        tcodes=h['obs/target_gene/codes'][:] if targets is not None else None
    return X, guides, codes, genes, targets, tcodes
Xr,gr,cr,gr_genes,tr,tcr=load('perturb.seq.align.q1.ref.h5ad')
Xq,gq,cq,gq_genes,tq,tcq=load('perturb.seq.align.q1.query.h5ad')
common, ri, qi = np.intersect1d(gr_genes,gq_genes,return_indices=True)
Xrc=Xr[:,ri].tocsc(); Xqc=Xq[:,qi].tocsc()
def norm_log(X):
    sums=np.asarray(X.sum(1)).ravel().astype(float); sums[sums==0]=1
    return np.log1p(X.multiply(1/sums[:,None]*1e6).tocsr())
Lr=norm_log(Xrc).toarray(); Lq=norm_log(Xqc).toarray()
ur=np.array([Lr[cr==i].mean(0) for i in range(len(gr))])
uq=np.array([Lq[cq==i].mean(0) for i in range(len(gq))])
ntr=[i for i,g in enumerate(gr) if g.startswith('NT-')]; ntq=[i for i,g in enumerate(gq) if g.startswith('NT-')]
Er=ur-ur[ntr].mean(0); Eq=uq-uq[ntq].mean(0)
guide2target={}
for i,g in enumerate(gr):
    if g.startswith('NT-'): continue
    vals,counts=np.unique(tr[tcr[cr==i]],return_counts=True)
    guide2target[g]=vals[np.argmax(counts)]
pr=[i for i,g in enumerate(gr) if not g.startswith('NT-')]
pq=[i for i,g in enumerate(gq) if not g.startswith('NT-')]
# Try different gene selections and metrics
for method in ['ref_var','query_var','mean_var','common_hvg']:
  if method=='ref_var': scores_var=ur.var(0)
  elif method=='query_var': scores_var=uq.var(0)
  elif method=='mean_var': scores_var=(ur.var(0)+uq.var(0))/2
  else: scores_var=(ur.mean(0)+uq.mean(0))
  for ngenes in [500,1000,2000,5000,11858]:
    top=np.argsort(scores_var)[::-1][:ngenes]
    A=Eq[np.ix_(pq,[top])] if False else Eq[pq][:,top]
    B=Er[pr][:,top]
    A=A-A.mean(1,keepdims=True); B=B-B.mean(1,keepdims=True)
    denom=np.sqrt((A*A).sum(1,keepdims=True)) @ np.sqrt((B*B).sum(1,keepdims=True)).T
    denom[denom==0]=1e-12
    C=A@B.T/denom
    print('\\n',method,ngenes)
    for target in ['PABPC1','NUDT21','LEO1']:
      for refidx in [ri for ri in pr if guide2target[gr[ri]]==target]:
       j=pr.index(refidx); scores=C[:,j]
       order=np.argsort(scores)[::-1]
       print(target, gr[refidx], [(gq[pq[k]],round(scores[k],3)) for k in order[:5]])
PY"
