def lognorm(X, scale=1e4):
    X=X.astype(np.float64).tocsr()
    lib=np.asarray(X.sum(axis=1)).ravel()
    lib[lib==0]=1
    X=X.multiply(1.0/lib[:,None]).tocsr()
    X.data=np.log1p(X.data*scale)
    return X.tocsr()
