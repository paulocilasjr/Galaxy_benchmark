import h5py, difflib
vs=[]
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 with h5py.File(fn,'r') as f:
  vs.append([x.decode() if isinstance(x,bytes) else x for x in f['var/_index'][:]])
a,b=vs
sm=difflib.SequenceMatcher(a=a,b=b,autojunk=False)
deletes=[]; inserts=[]
for tag,i1,i2,j1,j2 in sm.get_opcodes():
 if tag=='delete':
  deletes.extend(a[i1:i2])
 elif tag=='insert':
  inserts.extend(b[j1:j2])
print(len(deletes),len(inserts))
for i,(d,ins) in enumerate(zip(deletes,inserts),1):
 print(i,d,'<-',ins)
PY"
