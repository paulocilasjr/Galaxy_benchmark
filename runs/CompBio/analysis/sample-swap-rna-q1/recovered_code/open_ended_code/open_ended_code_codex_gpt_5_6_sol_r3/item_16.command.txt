/bin/bash -lc "curl -L --fail --silent --show-error 'https://www.ebi.ac.uk/europepmc/webservices/rest/PMC11136973/fullTextXML' -o article.xml
python - <<'PY'
from pathlib import Path
import re
x=Path('article.xml').read_text()
for pat in ['Data availability','GEO','BLAG','epidermal_neural','putative','Accession']:
 print('\\n##',pat)
 for m in re.finditer(pat,x,re.I): print(re.sub('<["'^>]+>'"',' ',x[max(0,m.start()-300):m.start()+500]).replace('\\n',' ')[:900])
PY"
