/bin/bash -lc "python -u - <<'PY'
import numpy as np, pandas as pd
z=np.load('reference_pseudobulk.npz')
r=pd.DataFrame(z['counts'],index=z['genes'],columns=z['labels'])
r=1e6*r/r.sum()
x=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
r=r.loc[x.index]
A=np.log1p(x.values); B=np.log1p(r.values)
c=np.corrcoef(A.T,B.T)[:12,12:]
print(pd.DataFrame(c,index=x.columns,columns=r.columns).round(4).to_string(),flush=True)
for i,a in enumerate(x):
 j=c[i].argmax(); b=r.columns[j]; d=np.abs(x[a].values-r[b].values)
 print(a,'=>',b,'corr',c[i,j],'maxerr',d.max(),'meanerr',d.mean(),flush=True)
PY"
