import requests,concurrent.futures
base='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE255nnn/GSE255742/suppl/'
files=['GSE255742_Amphioxus_21hpf_sc_UMI_counts.tsv.gz','GSE255742_Amphioxus_21hpf_sc_annotations.tsv.gz','GSE255742_Amphioxus_21hpf_metacell_gene_expression.tsv.gz']
def run(f):
 r=requests.get(base+f,timeout=180);r.raise_for_status();open(f,'wb').write(r.content);print(f,len(r.content),flush=True)
with concurrent.futures.ThreadPoolExecutor() as e:list(e.map(run,files))
PY
python -u download_reference.py"
