/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
ids=['2187408513','2182809125','2187430154']
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=nuccore&retmode=json&id='+','.join(ids)
with urllib.request.urlopen(url, timeout=30) as r: data=json.load(r)
for uid in data['result']['uids']:
    it=data['result'][uid]
    for k in ['caption','title','extra','organism','subname','slen','taxid']:
        print(uid,k,it.get(k))
    print('---')
PY"
