import struct, json
b=open('[REDACTED LOCAL PATH]','rb').read(); base=13350000000

while p < len(b)-46 and b[p:p+4] == b'PK\\x01\\x02':

        ep += 4+sz

print('count',len(files))
for f in files:

print('rds')
