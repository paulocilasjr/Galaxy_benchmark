/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
g=pd.read_csv('gene_pseudobulk.tsv',sep='\\t',index_col=0)
cols=g.columns.tolist()
# log1p, zscore across cols
log=np.log1p(g)
z=(log.sub(log.mean(axis=1),axis=0)).div(log.std(axis=1),axis=0)
# specificity: expression / sum
props=g.div(g.sum(axis=1),axis=0)
for c in cols:
    # score = z * props perhaps
    score=z[c]*props[c]
    top=score.sort_values(ascending=False).head(40)
    print('\\n###',c)
    for gene,val in top.items():
        if g.loc[gene,c] < 1: continue
        print(f'{gene:15s} expr={g.loc[gene,c]:8.1f} prop={props.loc[gene,c]:.2f} z={z.loc[gene,c]:.2f} score={val:.3f}')
PY"
