/bin/bash -lc "python3 - <<'PY'
import openpyxl
wb=openpyxl.load_workbook('[REDACTED LOCAL PATH]', read_only=True, data_only=True)
terms=['endoderm','endostyle_csg','epidermal_neural','epidermis','mesodermal_notochord','muscular_somites','neural','neural_like','putative_neuro_endo','somite','tailbud','tailbud_neural']
for ws in wb.worksheets:
    hits=[]
    for row in ws.iter_rows(values_only=True):
        for cell in row:
            if cell is not None and any(term.lower() in str(cell).lower() for term in terms):
                hits.append(row)
                break
    if hits:
        print('\\n',ws.title,len(hits))
        for h in hits[:10]: print(h)
PY"
