def find_genes(m):
    # normalize marker: split slash, parentheses, dots
    tokens=[]
    for part in re.split(r'[/,()]', m):
        part=part.strip().rstrip('+')
        if part:
            tokens.append(part)

            # regex with boundaries; avoid too broad

            if len(sub)>20:
                sub=[x for x in sub if tok.lower() in x.lower() and (x.upper().startswith(tok.upper()) or x.upper()==tok.upper())][:5]
