/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.cluster.hierarchy import linkage, dendrogram
from scipy.spatial.distance import squareform
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
g=pd.read_csv('gene_pseudobulk.tsv',sep='\\t',index_col=0)
X=np.log1p(g)
corr=X.corr()
# cluster columns by correlation distance
Z=linkage(squareform(1-corr), method='average')
plt.figure(figsize=(10,6))
dn=dendrogram(Z, labels=corr.columns, leaf_rotation=45)
plt.title('Hierarchical clustering of pseudobulk columns')
plt.tight_layout()
plt.savefig('dendro.png', dpi=100)
print(Z)
print('order', dn['ivl'])
PY"
