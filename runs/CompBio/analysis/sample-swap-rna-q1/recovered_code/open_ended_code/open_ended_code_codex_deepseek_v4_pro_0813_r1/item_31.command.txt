import urllib.request, urllib.parse, json, time
ids=['BLAG01000002','BLAG02000002','BLAG03000002','BLAG04000002','BLAG05000002','BLAG06000002','BLAG07000002','BLAG08000002','BLAG09000002','BLAG10000002','BLAG19000002','BLAG90000002']

with urllib.request.urlopen(url, timeout=30) as r: data=json.load(r)
res=data.get('result',{})
for uid in res.get('uids',[]):
    it=res[uid]
    print(uid, it.get('caption'), it.get('title'), it.get('organism'), it.get('subname'), it.get('slen'))
PY"
