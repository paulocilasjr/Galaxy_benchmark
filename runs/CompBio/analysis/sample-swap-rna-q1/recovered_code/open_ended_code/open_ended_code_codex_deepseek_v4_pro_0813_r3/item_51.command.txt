/bin/bash -lc "python3 - <<'PY'
import gzip,csv,re,json,numpy as np
id2name=json.load(open('id2name.json'))
with gzip.open('sample.swap.rna.q1.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t'); h=next(r); cols=h[1:]
    rows=[]
    for row in r: rows.append((row[0],[float(x) for x in row[1:]]))
A=np.array([v for _,v in rows]); ids=[g for g,_ in rows]
# helper exact/substring
def show(pattern):
    pat=re.compile(pattern,re.I)
    matches=[]
    for i,g in enumerate(ids):
        name=id2name.get(g,g)
        if pat.search(name):
            matches.append((name,g,A[i]))
    # aggregate duplicates by name maybe max
    print('\\nPATTERN',pattern,'count',len(matches))
    for name,g,vals in matches[:30]:
        print(name, g, ['%.1f'%v for v in vals])
for p in [r'"'^T$'"',r'BRACHYURY',r'FOXA2',r'SOX17',r'GATA4',r'GATA6',r'NOTO',r'COL2A1',r'COL8A1',r'NKX2-1',r'NKX2\\.1',r'PAX8',r'FOXE1',r'TPO"'$'"',r'TG"'$'"',r'SLC5A5',r'MYOD1',r'MYF5',r'MYOG',r'ACTC1',r'TNNC2',r'TNNT3',r'MYL1',r'MYL12B',r'KRT',r'TP63',r'LAMB3',r'COL17A1',r'ELAVL',r'NEUROD',r'NEUROG',r'TUBB',r'SYN1',r'CHAT',r'SOX2',r'PAX6',r'OTX',r'FOXJ1',r'SHH',r'HNF4',r'CDX',r'PDX1',r'XLOX']:
    show(p)
PY"
