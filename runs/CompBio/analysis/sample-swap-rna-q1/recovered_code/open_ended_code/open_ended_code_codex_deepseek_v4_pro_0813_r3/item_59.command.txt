/bin/bash -lc "python3 - <<'PY'
import gzip,csv,json,re,numpy as np
id2name=json.load(open('id2name.json'))
with gzip.open('sample.swap.rna.q1.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t'); h=next(r); cols=h[1:]; rows=[]
    for row in r: rows.append((row[0],[float(x) for x in row[1:]]))
A=np.array([v for _,v in rows]); ids=[g for g,_ in rows]
print('cols',cols)
patterns=['POU4','POU','SIX1','SIX2','ISL1','ISL2','NEUROG','ASCL','ATOH','DLL','NOTCH','HES','INSM','ELAVL','TUBA','TUBB','SNAP25','SYT','PRDM12','NKX6','ONECUT','MNX','HB9','SOX4','SOX11','FOXQ','TP63','KRT13','KRT75','KRT5','COL1A','COL2A','COL8A','COL9A','MYOD','MYF5','MYF6','MRF4','MYOG','ACTC','MYL','TNNC','TNNT','TPM','TTN','NEB','SMYD','PAX3','PAX7','MEOX','TCF15','PARAXIS','TBX6','FOXC','ALX','FOXA','GATA','HHEX','NKX2','NKX1','PDX','CDX','XLOX','SOX17','HNF4','HNF1','RBP','CP','ELA','CELA','CPA','AMY','INS','GCG','SST','GHRL','CHGA','CHGB','TPH','DDC','SLC18','SCG']
for p in patterns:
    rr=re.compile('"'^'"'+p,re.I)
    matches=[]
    for i,g in enumerate(ids):
        n=id2name.get(g,g)
        if rr.search(n):
            matches.append((n,g,A[i]))
    if matches:
        print('\\n##',p,len(matches))
        for n,g,v in matches[:20]:
            print(f'{n:15s} {g:15s} ' + ' '.join(f'{x:7.1f}' for x in v))
PY"
