/bin/bash -lc "python3 - <<'PY'
import csv, collections
fn='meta_extract/metadata_merged.csv'
cnt=collections.Counter(); cnt2=collections.Counter(); cnt3=collections.Counter()
with open(fn) as f:
 r=csv.DictReader(f)
 for row in r:
  cnt[row['stage_name']]+=1
  cnt2[row['refined_group']]+=1
  cnt3[row['lineage']]+=1
print('stage_name',cnt.most_common(50))
print('refined_group',cnt2.most_common(100))
print('lineage',cnt3.most_common(100))
PY"
