/bin/bash -lc "python3 - <<'PY'
import gzip,csv,json,re,numpy as np
id2name=json.load(open('id2name.json'))
with gzip.open('sample.swap.rna.q1.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t'); h=next(r); cols=h[1:]; rows=[]
    for row in r: rows.append((row[0],[float(x) for x in row[1:]]))
A=np.array([v for _,v in rows]); ids=[g for g,_ in rows]
names=[id2name.get(g,g) for g in ids]
# use log2(x+1) z
X=np.log2(A+1); mean=X.mean(1,keepdims=True); std=X.std(1,keepdims=True); std[std==0]=1
Z=(X-mean)/std
for j,c in enumerate(cols):
    # specificity score: z col plus uniqueness? top 50 by col z with expr >10 maybe
    order=np.argsort(-Z[:,j])
    print('\\n### %s'%c)
    k=0
    for idx in order:
        if A[idx,j] > 20 and Z[idx,j] > 1.5:
            print(f'{names[idx]:20s} {A[idx,j]:9.2f} {Z[idx,j]:5.2f}')
            k+=1
        if k>=50: break
PY"
