/bin/bash -lc "python3 -u - <<'PY'
import gzip, time, numpy as np, pandas as pd
D=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
A=pd.read_csv('annotations.tsv.gz',sep='\\t')
with gzip.open('umi.tsv.gz','rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
# exact annotation cell order is the UMI column order
labels=A.cell_type.to_numpy()
uniq=pd.unique(labels)
li={x:i for i,x in enumerate(uniq)}
g=np.array([li[x] for x in labels],dtype=np.int16)
keep=set(D.index)
out=np.zeros((len(keep),len(uniq)),dtype=np.float64)
rowids=[]
tot=np.zeros(len(uniq),dtype=np.float64)
start=time.time(); n=0; j=0
with gzip.open('umi.tsv.gz','rt') as f:
    f.readline()
    for line in f:
        n+=1
        parts=line.rstrip('\\n').split('\\t')
        gene=parts[0]
        vals=np.fromstring('\\t'.join(parts[1:]),sep='\\t')
        sums=np.bincount(g,weights=vals,minlength=len(uniq))
        tot += sums
        if gene in keep:
            rowids.append(gene)
            out[j,:]=sums
            j+=1
        if n%5000==0: print('rows',n,'kept',j,'sec',round(time.time()-start,1),flush=True)
print('done',n,j,'sec',round(time.time()-start,1),flush=True)
# preserve D order
R=pd.DataFrame(out[:j,:],index=rowids,columns=uniq)
R=R.reindex(D.index)
# cpm of pooled cells (within exact groups)
Rc=R.div(R.sum(axis=0),axis=1)*1e6
# correlations and errors after matching normalized profiles
# sample might be each col a CPM of some groups; compare log and Pearson/Spearman
print('group cell counts')
print(pd.Series(labels).value_counts().to_string())
print('total umi')
print(pd.Series(tot,index=uniq).round(0).to_string())
L=np.log1p(D); LR=np.log1p(Rc)
C=pd.DataFrame(index=D.columns,columns=uniq,dtype=float)
MAE=pd.DataFrame(index=D.columns,columns=uniq,dtype=float)
for a in D.columns:
 for b in uniq:
  C.loc[a,b]=L[a].corr(LR[b])
  MAE.loc[a,b]=np.mean(np.abs(D[a]-Rc[b]))
print('corr')
print(C.round(4).to_string())
print('best')
for a in D.columns:
 print(a,'corr',C.loc[a].nlargest(5).round(4).to_dict(),'mae',MAE.loc[a].nsmallest(5).round(1).to_dict())
# write compact ref for further analysis
R.to_csv('umi_pseudobulk_exact.tsv.gz',sep='\\t',compression='gzip')
PY"
