/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
D=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
for c in D.columns:
 x=D[c]; rest=D.drop(columns=c).mean(axis=1)
 score=np.log2((x+1)/(rest+1))
 ix=score.nlargest(15).index
 print(c+'\\t'+'; '.join(f'{g}:{x[g]:.1f}/{rest[g]:.1f}/{score[g]:.2f}' for g in ix),flush=True)
PY"
