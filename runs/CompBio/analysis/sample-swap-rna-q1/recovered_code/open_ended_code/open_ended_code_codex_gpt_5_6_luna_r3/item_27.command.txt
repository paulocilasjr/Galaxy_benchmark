/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
print('reading annotations',flush=True)
a=pd.read_csv('GSE255742_Amphioxus_21hpf_sc_annotations.tsv.gz',sep='\\t')
print(a.shape,a.columns.tolist(),a['cell_type'].value_counts().to_string(),flush=True)
print('reading counts',flush=True)
c=pd.read_csv('GSE255742_Amphioxus_21hpf_sc_UMI_counts.tsv.gz',sep='\\t',index_col=0)
print('counts shape',c.shape,flush=True)
print(c.index[:3].tolist(),c.columns[:3].tolist(),flush=True)
print('common',len(set(c.columns)&set(a.cell)))
labs=a.set_index('cell').loc[c.columns,'cell_type']
print(labs.value_counts().to_string(),flush=True)
D=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
print('D',D.shape,'common genes',len(set(D.index)&set(c.index)),flush=True)

for lab in labs.unique():

 x=c[cols].sum(axis=1)
 agg[lab]=x/x.sum()*1e6
A=pd.DataFrame(agg).reindex(D.index)
print('A sums',A.sum().round(2).to_dict(),flush=True)
print('D sums',D.sum().round(2).to_dict(),flush=True)
print('cor D vs A')
for d in D:
 vals=[]
 for a0 in A:
  vals.append((a0,np.corrcoef(D[d].values,A[a0].values)[0,1],np.corrcoef(np.log1p(D[d].values),np.log1p(A[a0].values))[0,1],np.abs(D[d].values-A[a0].values).mean()))
 print(d,sorted(vals,key=lambda z:z[1],reverse=True)[:4])
