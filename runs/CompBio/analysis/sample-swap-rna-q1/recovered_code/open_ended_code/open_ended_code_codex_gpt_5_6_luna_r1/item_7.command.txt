/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
p='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t',index_col=0)
print('shape',df.shape)
for x in [('sum',df.sum()),('mean',df.mean()),('median',df.median()),('nonzero',(df>0).sum())]:
 print(x[0]); print(x[1].to_string())
print('max',df.max().max())
# correlations raw/log/ ranks
for name,x in [('raw',df),('log',np.log1p(df)),('z', (df-df.mean())/df.std())]:
 print('\\n',name)
 print(x.corr().round(3).to_string())
PY"
