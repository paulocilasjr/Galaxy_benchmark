/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
our=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
ref=pd.read_csv('reference_pseudobulk.tsv.gz',sep='\\t',index_col=0,compression='gzip')
for oc,rc in [('epidermal_neural','tailbud_neural'),('tailbud_neural','epidermal_neural'),('epidermal_neural','epidermal_neural'),('tailbud_neural','tailbud_neural')]:
    a=our[oc]; b=ref[rc]
    print(oc,'vs ref',rc)
    print('pearson raw',np.corrcoef(a,b)[0,1], 'spearman?')
    print('sum our/ref',a.sum(),b.sum())
    # max abs diff scaled
    scale=b.sum()/a.sum()
    print('scale',scale)
    # if proportional, log1p pearson
    print('log1p pearson',np.corrcoef(np.log1p(a),np.log1p(b))[0,1])
    print('max relative diff for nonzero', np.max(np.abs(a*scale-b)/(b+1e-9)))
    print('head')
    print(pd.DataFrame({'our':a,'ref':b}).head(10))
PY"
