/bin/bash -lc "cat > parse_gbff.py <<'PY'
import re, json, sys
path='ref3/ncbi_dataset/data/GCA_927797965.1/genomic.gbff'
out='gbff_id_gene.json'
pat_id=re.compile(r'\\bID:(BLAG\\d+)(?:[_;]|"'$)'"')
pat_gene=re.compile(r'/gene=\"(["'^"]+)"'"')
pat_locus=re.compile(r'/locus_tag=\"(["'^"]+)"'"')
mapping={}
current_gene=None
current_locus=None
count=0
with open(path, 'r', errors='ignore') as f:
    for line in f:
        stripped=line.strip()
        m=pat_gene.search(stripped)
        if m:
            current_gene=m.group(1)
        m=pat_locus.search(stripped)
        if m:
            current_locus=m.group(1)
        # A new feature? current gene persists; if ID line then map
        for m in pat_id.finditer(stripped):
            gid=m.group(1)
            if gid not in mapping:
                mapping[gid]={'gene':current_gene,'locus':current_locus}
                count+=1
            elif current_gene and mapping[gid]['gene'] is None:
                mapping[gid]['gene']=current_gene
                mapping[gid]['locus']=current_locus
print('mapped',count,'ids')
json.dump(mapping, open(out,'w'))
print('saved',out)
PY
python3 parse_gbff.py"
