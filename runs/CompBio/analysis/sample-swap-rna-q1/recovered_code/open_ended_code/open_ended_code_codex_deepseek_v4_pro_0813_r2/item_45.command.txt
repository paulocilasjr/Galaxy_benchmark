/bin/bash -lc "python3 - <<'PY'
from Bio import Entrez, SeqIO
Entrez.email='a@b.com'
id='BLAG06001064'
handle=Entrez.esearch(db='nuccore', term=f'{id}[All Fields]')
rec=Entrez.read(handle)
print(rec)
uid=rec['IdList'][0]
handle=Entrez.efetch(db='nuccore', id=uid, rettype='gb', retmode='text')
raw=handle.read()
print(raw[:500])
# find gene around ID
lines=raw.splitlines()
for i,l in enumerate(lines):
    if id in l:
        for j in range(max(0,i-8), min(len(lines), i+12)):
            print(lines[j])
        break
PY"
