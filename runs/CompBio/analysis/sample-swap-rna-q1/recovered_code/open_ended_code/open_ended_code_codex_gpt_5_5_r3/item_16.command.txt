/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
fn='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(fn,sep='\\t',index_col=0)
print(df.shape)
print(df.columns.tolist())
# basic column sums/nonzero
print('sums')
print(df.sum().sort_values())
# correlations log1p all genes
x=np.log1p(df)
for method in ['pearson','spearman']:
    c=x.corr(method=method)
    print('\\n',method)
    print(c.round(3).to_string())
    print('nearest')
    for col in c.columns:
        s=c[col].drop(col).sort_values(ascending=False)
        print(col, s.index[0], round(s.iloc[0],3), '2nd', s.index[1], round(s.iloc[1],3))
PY"
