/bin/bash -lc "python3 - <<'PY'
import pysam, collections
b=pysam.AlignmentFile('mt.sorted.bam','rb')
flags=collections.Counter()
lens=collections.Counter()
for r in b.fetch(until_eof=True):
    flags[r.flag & 0x1]+=1
    lens[len(r.query_sequence)]+=1
print('paired flag counts', flags)
print('top lengths', lens.most_common(10))
print('n total', sum(flags.values()))
PY"
