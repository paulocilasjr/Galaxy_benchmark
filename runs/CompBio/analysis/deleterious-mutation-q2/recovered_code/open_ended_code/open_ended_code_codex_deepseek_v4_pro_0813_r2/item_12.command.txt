/bin/bash -c "python3 - <<'PY'
import gzip,re
fn='raw.vcf'
rows=[]
with open(fn) as f:
    for line in f:
        if line.startswith('#'): continue
        parts=line.rstrip().split('\\t')
        chrom,pos,id,ref,alts,qual,filter,info,fmt,sample=parts[:10]
        if len(ref)"'!=1: continue
        if '"',' in alts: continue
        if len(alts)"'!=1: continue
        fmtkeys=fmt.split('"':')
        sv=sample.split(':')
        d=dict(zip(fmtkeys,sv))
        dp=int(d.get('DP','0'))
        ad=d.get('AD','')
        adparts=ad.split(',') if ad else []
        altad=int(adparts[1]) if len(adparts)>1 else 0
        refad=int(adparts[0]) if adparts else 0
        if dp==0: continue
        af=altad/dp
        # raw high quality
        try: q=float(qual)
        except: q=0
        if dp>=5 and altad>=2:
            rows.append((q,chrom,int(pos),ref,alts,dp,refad,altad,af,info))
rows.sort(reverse=True)
print('count',len(rows))
for r in rows[:200]:
    print(r)
PY"
