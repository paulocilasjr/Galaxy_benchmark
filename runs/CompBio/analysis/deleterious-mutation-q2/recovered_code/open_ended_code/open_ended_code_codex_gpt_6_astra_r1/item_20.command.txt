/bin/bash -lc "cat work/flagstat.txt; samtools idxstats work/aligned.hg38.bam; python3 - <<'PY'
import pysam,collections
b=pysam.AlignmentFile('work/aligned.hg38.bam'); bins=collections.Counter();nm=collections.Counter()
for r in b.fetch():
 if r.is_secondary or r.is_supplementary:continue
 bins[r.reference_start//10000000]+=1
 if r.mapping_quality>=30:nm[r.get_tag('NM')]+=1
print('Reads per 10Mb bin:',bins)
print('Edit distances for MAPQ >=30:',nm.most_common(15))
PY"
