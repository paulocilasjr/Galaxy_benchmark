/bin/bash -lc "printf 'mapped pos range/depth summary\\n'; samtools view -F 4 work/reads.hg38.bam | awk 'BEGIN{min=1e12;max=0} {if("'$4<min)min=$4;if($4>max)max=$4;n++;} END{print min,max,n}'"'
printf 'window counts (10Mb)\\n'; samtools view -F 4 work/reads.hg38.bam | awk '{print int("'$4/10000000)*10,1}'"' | sort -n | uniq -c | head -30
printf 'high depth sites\\n'; samtools depth -aa -Q 20 -q 20 work/reads.hg38.bam | awk '"'$3>=5'"' | head -20"
