import pysam, collections, heapq
bam=pysam.AlignmentFile('reads.chr9.bam','rb')
ref=pysam.FastaFile('chr9.fa')
# Keep candidate observations where a non-reference allele has at least 2 high-quality bases.
cands=[]
ndp=collections.Counter(); nsites=0
for col in bam.pileup('chr9',0,138394717,stepper='samtools',min_mapping_quality=20,min_base_quality=20,max_depth=100000,ignore_overlaps=True,compute_baq=True):
    nsites+=1
    pos=col.reference_pos
    rb=ref.fetch('chr9',pos,pos+1).upper()
    counts=collections.Counter()
    for pr in col.pileups:
        if pr.is_del or pr.is_refskip or pr.query_position is None: continue
        q=pr.alignment.query_qualities[pr.query_position]
        if q is None or q<20: continue
        b=pr.alignment.query_sequence[pr.query_position].upper()
        if b in 'ACGT': counts[b]+=1
    dp=sum(counts.values())
    if dp: ndp[min(dp,100)]+=1
    if rb not in 'ACGT': continue
    for alt,n in counts.items():
        if alt"'!=rb and n>=2 and dp>=5:
            af=n/dp
            if af<=0.35:
                cands.append((af,n,dp,pos+1,rb,alt,dict(counts)))
print('"'covered sites',nsites,'candidate count',len(cands),'depth bins',ndp.most_common(15))
for x in sorted(cands, reverse=True)[:100]:
    print(f'{x[3]} {x[4]}>{x[5]} {x[2]} {x[1]} {x[0]:.3f} {x[6]}')
PY"
