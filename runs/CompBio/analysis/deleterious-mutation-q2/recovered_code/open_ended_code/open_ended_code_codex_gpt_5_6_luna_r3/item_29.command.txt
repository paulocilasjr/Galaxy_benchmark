/bin/bash -lc "set -o pipefail
samtools mpileup -B -q 20 -Q 20 -d 100000 -f chr9.fa reads.chr9.bam | perl -lane '
  my ("'$chr,$pos,$ref,$dp,$s)=@F[0..4];
  1 while $s =~ s/'"\\"'^.[''^'"\\"'^]*?//;
  $s =~ s/'"\\"'$//g;
  while ($s =~ s/([+-])('"\\d+)//) { my "'$n=$2; substr($s,$-[0],$n)=""; }
  my %c=(A=>0,C=>0,G=>0,T=>0); my $r=uc($ref); my $rd=0;
  for my $b (split //,$s) { if ($b eq "." || $b eq ",") {$rd++} elsif (exists $c{uc($b)}) {$c{uc($b)}++} }
  my $tot=$rd+$c{A}+$c{C}+$c{G}+$c{T}; next if $tot<10 || $r !~ /''^[ACGT]$/;
  for my $a (qw(A C G T)) { next if $a eq $r || !$c{$a}; my $af=$c{$a}/$tot; next unless $af>=0.02 && $af<=0.30; print join("'"\\t\","'$chr,$pos,$r,$a,$tot,$c{$a},sprintf("%.4f",$af),$rd,$c{A},$c{C},$c{G},$c{T}); }
'"' | sort -t "'$'"'\\t' -k6,6nr -k5,5nr | head -300"
