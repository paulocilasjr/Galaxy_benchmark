/bin/bash -lc "bcftools mpileup -f chr9.hg38.fa -q 20 -Q 20 -a FORMAT/AD,FORMAT/DP,FORMAT/SP,INFO/DP -Ou reads.hg38.sorted.bam | bcftools call -mv -f GQ -Oz -o calls.hg38.vcf.gz && tabix -p vcf calls.hg38.vcf.gz && bcftools stats calls.hg38.vcf.gz | sed -n '1,80p'; echo '--- variants'; bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t%DP[\\t%AD]\\n' calls.hg38.vcf.gz | head -n 30"
