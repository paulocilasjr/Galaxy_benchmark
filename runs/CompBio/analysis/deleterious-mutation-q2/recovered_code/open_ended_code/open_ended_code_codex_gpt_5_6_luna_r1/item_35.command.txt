/bin/bash -lc "python3 - <<'PY'
import json
rows = [x.rstrip().split('\\t') for x in open('candidates.lowaf.tsv')]
by = {(x[0][3:], x[1], x[2], x[3]): x for x in rows}
variants = json.load(open('vep_candidates.json'))
for record in variants:
    consequences = [c for c in record.get('transcript_consequences', []) if 'stop_gained' in c.get('consequence_terms', [])]
    if consequences:
        ref, alt = record['allele_string'].split('/')
        key = (record['seq_region_name'], str(record['start']), ref, alt)
        print('VAR', key, 'ROW', by.get(key))
        for consequence in consequences:
            print(consequence)
PY"
