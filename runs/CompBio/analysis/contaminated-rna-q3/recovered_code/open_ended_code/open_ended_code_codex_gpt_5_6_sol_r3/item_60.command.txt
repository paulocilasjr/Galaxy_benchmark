/bin/bash -lc "curl -L --fail --retry 3 -o pan_rna.zip 'https://api.ncbi.nlm.nih.gov/datasets/v2/genome/accession/GCF_028858775.2/download?include_annotation_type=RNA_FASTA' && unzip -l pan_rna.zip | head -10 && unzip -p pan_rna.zip 'ncbi_dataset/data/GCF_028858775.2/rna.fna' | minimap2 -t 6 -ax sr - contaminated.rna.q3.fq.gz | samtools view -@ 2 -b -o pan_tx.bam - && samtools flagstat -@ 2 pan_tx.bam && samtools view -F 2304 pan_tx.bam | awk '{nm=\"\";for(i=12;i<=NF;i++)if("'$i~/''^NM:i:/){split($i,a,":");nm=a[3]} c[nm]++}END{for(k in c)print k,c[k]}'"' | sort -nk1"
