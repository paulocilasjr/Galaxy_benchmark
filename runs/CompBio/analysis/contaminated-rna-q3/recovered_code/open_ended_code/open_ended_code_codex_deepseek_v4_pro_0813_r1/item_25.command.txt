/bin/bash -lc "python3 - <<'PY'
import random
N=1000
random.seed(123)
res=[]; n=0
with open('unmapped.fq') as f:
    while True:
        h=f.readline()
        if not h: break
        seq=f.readline().strip(); plus=f.readline(); qual=f.readline()
        n+=1
        name=h.rstrip()[1:]
        if len(res)<N:
            res.append((name,seq))
        else:
            j=random.randrange(n)
            if j<N: res[j]=(name,seq)
print('unmapped reads',n,'sampled',len(res))
with open('unmapped_sample1000.fa','w') as out:
    for name,seq in res: out.write(f'>{name}\\n{seq}\\n')
PY
ls -lh unmapped_sample1000.fa"
