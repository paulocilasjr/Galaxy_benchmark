/bin/bash -lc "python3 - <<'PY'
import json, os, glob
from collections import Counter,defaultdict
cnt=Counter(); taxcnt=Counter(); genus_counter=Counter(); total=0; nohits=0
species_top = []
for path in sorted(glob.glob('blast_out/chunk_*.json')):
    data=json.load(open(path))
    for rep in data['BlastOutput2']:
        search=rep['report']['results']['search']; hits=search.get('hits',[]); total+=1
        if not hits:
            nohits+=1; continue
        desc=hits[0].get('description',[{}])[0]
        sc=desc.get('sciname',''); tax=desc.get('taxid','')
        cnt[sc]+=1; taxcnt[tax]+=1
        genus=sc.split()[0].lower() if sc else 'none'
        genus_counter[genus]+=1
        species_top.append((search.get('query_title'),sc,tax))
print('total',total,'nohits',nohits)
print('\\nTop 80 species top-hit counts:')
for sc,n in cnt.most_common(80): print(n,sc)
print('\\nTop 50 genus top-hit counts:')
for g,n in genus_counter.most_common(50): print(n,g)
print('\\nTop taxid counts:')
for t,n in taxcnt.most_common(50): print(n,t)
PY"
