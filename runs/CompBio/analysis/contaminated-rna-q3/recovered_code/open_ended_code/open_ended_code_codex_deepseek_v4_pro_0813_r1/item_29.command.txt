/bin/bash -lc "python3 - <<'PY'
import os
os.makedirs('unmapped_chunks',exist_ok=True)
chunk=1000
out=None; n=0; idx=0; total=0
with open('unmapped.fq') as f:
    while True:
        h=f.readline()
        if not h: break
        seq=f.readline().strip(); plus=f.readline(); qual=f.readline()
        if n%chunk==0:
            if out: out.close()
            idx+=1
            out=open(f'unmapped_chunks/chunk_{idx:02d}.fa','w')
        out.write(f'>{h.rstrip()[1:]}\\n{seq}\\n')
        n+=1; total+=1
if out: out.close()
print('chunks',idx,'reads',total)
PY
ls unmapped_chunks | head; ls unmapped_chunks | wc -l; wc -l unmapped_chunks/*.fa | tail -1"
