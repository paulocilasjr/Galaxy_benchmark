/bin/bash -lc "python3 - <<'PY'
import time
t=time.time(); names={}
with open('names.dmp') as f:
    for line in f:
        parts=line.rstrip('\\n').split('\\t|\\t')
        if len(parts)>=4 and parts[3].split('\\t|')[0]=='scientific name':
            names[int(parts[0])]=parts[1]
print('names',len(names),'secs',time.time()-t)
print(list(names.items())[:5])
PY"
