/bin/bash -lc "python3 - <<'PY'
import time
t=time.time()
parent={}; rank={}
with open('nodes.dmp') as f:
    for line in f:
        parts=line.rstrip('\\n').split('\\t|\\t')
        tid=int(parts[0]); parent[tid]=int(parts[1]); rank[tid]=parts[2]
print('nodes',len(parent),'secs',time.time()-t)
t=time.time(); names={}
with open('names.dmp') as f:
    for line in f:
        parts=line.rstrip('\\n').split('\\t|\\t')
        if len(parts)>=4 and parts[3]=='scientific name':
            names[int(parts[0])]=parts[1]
print('names',len(names),'secs',time.time()-t)
PY"
