/bin/bash -lc 'seq=$(awk '"'NR==1{print "'$2}'"' top_duplicated_reads.tsv)
curl -s 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data-urlencode CMD=Put --data-urlencode PROGRAM=blastn --data-urlencode DATABASE=nt --data-urlencode MEGABLAST=on --data-urlencode EXPECT=1e-5 --data-urlencode HITLIST_SIZE=20 --data-urlencode QUERY=\""'$seq" > qblast_put1.txt
cat qblast_put1.txt | sed -n '"'1,80p'"
