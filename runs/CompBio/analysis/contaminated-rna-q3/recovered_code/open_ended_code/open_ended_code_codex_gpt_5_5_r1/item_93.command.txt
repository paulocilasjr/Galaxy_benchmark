/bin/bash -lc '# Generate best-distance files for non-primate panel where not already generated, then compare using combined human baseline.
for ref in refs/panel/Caenorhabditis_elegans*.fa.gz refs/panel/Danio_rerio*.fa.gz refs/panel/Drosophila_melanogaster*.fa.gz refs/panel/Mus_musculus*.fa.gz refs/panel/Rattus_norvegicus*.fa.gz refs/panel/Saccharomyces_cerevisiae*.fa.gz; do
  name=$(basename "$ref" .fa.gz)
  out="best_${name}.tsv"
  if [ ! -s "$out" ]; then
    minimap2 -t 4 -x sr --secondary=no -a "$ref" contaminated.rna.q3.fq.gz 2> "${name}.finalbest.map.log" | awk '"'BEGIN{OFS=\"\\t\"} /"'^@/{next} {unmapped=int($2/4)%2; if(!unmapped){nm=999; for(i=12;i<=NF;i++) if($i ~ /''^NM:i:/){nm=substr($i,6)+0}; if(!($1 in best) || nm<best[$1]) best[$1]=nm}} END{for(k in best) print k,best[k]}'"' > \""'$out"
  fi
done
python3 - <<'"'PY'
from glob import glob
files={'human':'human_combined_best_nm.tsv'}
for fn in glob('best_*.tsv'):
    key=fn[5:-4]
    files[key]=fn
# include only one record per file; compare all nonhuman refs
human={}
for line in open(files['human']):
    r,m=line.split()[:2]; human[r]=int(m)
rows=[]
for key,fn in files.items():
    if key=='human': continue
    better=better3=better5=exactbad=0
    for line in open(fn):
        r,m=line.split()[:2]; v=int(m); h=human.get(r,99)
        if v<h: better+=1
        if h-v>=3: better3+=1
        if h-v>=5: better5+=1
        if v==0 and h>=8: exactbad+=1
    rows.append((better5,better3,better,exactbad,key))
print('better_h5\\tbetter_h3\\tbetter_h\\texact_hbad\\treference')
for row in sorted(rows, reverse=True)[:20]:
    print(*row, sep='\\t')
PY"
