/bin/bash -lc "awk 'NR%4==1 {id=substr("'$0,2)} NR%4==2 {i++; if(i%31==1) print ">" id "'"\\n\" "'$0}'"' work/unmapped.fq > work/unmapped_sample.fa
printf 'sample records: '; awk 'END{print NR/2}' work/unmapped_sample.fa
curl --max-time 120 -sS -A 'Mozilla/5.0' -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' \\
  --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' \\
  --data-urlencode 'MEGABLAST=on' --data-urlencode 'HITLIST_SIZE=20' --data-urlencode 'QUERY@work/unmapped_sample.fa' > work/unmapped_put.html
awk 'BEGIN{IGNORECASE=1} /RID =|RTOE =|NUM_QUERIES/ {print}' work/unmapped_put.html | sed -n '1,20p'"
