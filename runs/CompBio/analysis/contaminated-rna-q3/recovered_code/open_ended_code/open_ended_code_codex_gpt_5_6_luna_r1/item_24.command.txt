/bin/bash -lc "awk '
/<Iteration_query-def>/ {q="'$0; sub(/''^.*<Iteration_query-def>/,"",q); sub(/<'"\\\\/Iteration_query-def>.*"'$/, "",q);}
/<Hit_num>1<'"\\\\/Hit_num>/ {in1=1; gotdef=0; gotid=0; gotlen=0; def=\"\"}
in1 && /<Hit_def>/ && "'!gotdef {def=$0; sub(/''^.*<Hit_def>/,"",def); sub(/<'"\\\\/Hit_def>.*"'$/, "",def); gotdef=1}
in1 && /<Hsp_identity>/ && !gotid {id=$0; sub(/''^.*<Hsp_identity>/,"",id); sub(/<'"\\\\/Hsp_identity>.*"'$/, "",id); gotid=1}

/<'"\\\\/Iteration>/ {if(q"'!="") print q "'"\\t\" id \"/\" al \"\\t\" def; q=\"\"; in1=0}
