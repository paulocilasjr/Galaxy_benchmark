/bin/bash -lc "samtools view -F 2308 work/human.bam | awk '{as=\"\"; for(i=12;i<=NF;i++) if("'$i ~ /''^AS:i:/){as=$i; sub(/''^AS:i:/,"",as)}; if(as!="") print $1"'"\\t\"as\"\\t\""'$5}'"' | sort -k1,1 > work/human_scores.tsv
samtools view -F 2308 work/mouse.bam | awk '{as=\"\"; for(i=12;i<=NF;i++) if("'$i ~ /''^AS:i:/){as=$i; sub(/''^AS:i:/,"",as)}; if(as!="") print $1"'"\\t\"as\"\\t\""'$5}'"' | sort -k1,1 > work/mouse_scores.tsv
join -t '\\t' -j 1 work/human_scores.tsv work/mouse_scores.tsv > work/human_mouse_scores.tsv
awk -F '\\t' '{h="'$2;m=$4; if(m>h) mouse++; else if(h>m) human++; else tie++; if(m>=130 && h<130) print}'"' work/human_mouse_scores.tsv | tee work/mouse_prefer.tsv | head -50
printf '\\ncomparison counts: '; awk -F '\\t' '{if("'$2>$4)h++; else if($2<$4)m++; else t++} END{print "human="h,"mouse="m,"tie="t}'"' work/human_mouse_scores.tsv
printf 'mouse-prefer total: '; wc -l < work/mouse_prefer.tsv
printf '\\nstrong mouse-prefer examples:\\n'; sed -n '1,30p' work/mouse_prefer.tsv"
