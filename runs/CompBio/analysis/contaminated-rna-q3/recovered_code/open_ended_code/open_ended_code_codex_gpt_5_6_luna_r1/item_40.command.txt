printf '\\nfirst alignments:\\n'; samtools view work/meso.bam | sed -n '1,25p'
printf '\\nread ids:\\n'; samtools view work/meso.bam | cut -f1 | sort -n -t_ -k2,2 | sed -n '1,160p'"
