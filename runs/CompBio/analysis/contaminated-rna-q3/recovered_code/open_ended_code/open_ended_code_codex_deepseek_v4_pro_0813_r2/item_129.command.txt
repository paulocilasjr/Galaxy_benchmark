import pysam

def best(fn):
 d={}
 with pysam.AlignmentFile(fn,'r') as sam:
  for rec in sam.fetch(until_eof=True):
   if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
   q=rec.query_name; asv=rec.get_tag('AS')
   if q not in d or asv>d[q]: d[q]=asv
 return d
h=best('q3_candidates_humanmt.sam')
p=best('q3_candidates_panmt.sam')
# all candidates? compare
pan=set(subprocess if False else [])
better=0; worse=0; onlypan=0; onlyhuman=0; equal=0
for q in set(h)|set(p):
 if q not in h: onlypan+=1
 elif q not in p: onlyhuman+=1
 else:
  if p[q]>h[q]: better+=1
  elif p[q]<h[q]: worse+=1
  else: equal+=1
print('onlypan',onlypan,'onlyhuman',onlyhuman,'better_pan',better,'better_human',worse,'equal',equal)
PY"
