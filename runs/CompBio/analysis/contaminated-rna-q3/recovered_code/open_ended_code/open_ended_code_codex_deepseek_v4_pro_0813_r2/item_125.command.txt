/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob, os, re
accs=set()
for fn in glob.glob('qblast_cand_*.xml'):
    if os.path.getsize(fn)==0: continue
    try: recs=list(NCBIXML.parse(open(fn)))
    except: continue
    for rec in recs:
        for al in rec.alignments[:3]:
            if 'Pan paniscus' in al.hit_def or 'Pan troglodytes' in al.hit_def:
                if 'mitochondrion' in al.hit_def:
                    accs.add(al.hit_id)
                    print(al.hit_id, al.hit_def[:120])
print('accessions',accs)
PY"
