/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIWWW
from Bio.Blast import NCBIXML
from Bio import SeqIO
import time
seqs=list(SeqIO.parse('sample20.fa','fasta'))[:5]
for rec in seqs:
    try:
        handle=NCBIWWW.qblast('blastn','nt',rec.seq,entrez_query='',hitlist_size=5,expect=0.001)
        with open(f'blast_{rec.id}.xml','w') as f:
            f.write(handle.read())
        handle.close()
        print(rec.id, 'done')
    except Exception as e:
        print(rec.id, 'error', e)
    time.sleep(1)
PY
ls -lh blast_*.xml 2>/dev/null | head"
