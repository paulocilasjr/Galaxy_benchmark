/bin/bash -c "python3 - <<'PY'
# read names mapping to XR_007021210.2 in q3_unmapped_hydra.sam
names=set()
import subprocess
out=subprocess.check_output(\"samtools view -F4 q3_unmapped_hydra.sam | awk '"'$3=='"\\\"XR_007021210.2\\\" {print "'$1}'"'\",shell=True,text=True)
names=set(out.split())
# read names in candidates fastq
cand=set()
from Bio.SeqIO.QualityIO import FastqGeneralIterator
for title,seq,qual in FastqGeneralIterator(open('q3_unmapped_hg38_unmapped.fq')):
    cand.add(title)
print('18S mapped count',len(names),'in candidates',len(names&cand))
print('not in candidates',len(names-cand))
PY"
