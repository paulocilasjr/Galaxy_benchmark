/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob, os, re
ids=[]
for fn in glob.glob('qblast_cand_*.xml'):
    try:
        recs=list(NCBIXML.parse(open(fn)))
    except: continue
    for rec in recs:
        for al in rec.alignments[:5]:
            if 'Conus episcopatus' in al.hit_def:
                m=re.search(r'gi\\|\\d+\\|(?:gb|emb|dbj|ref)\\|([A-Z0-9_.]+)\\|', al.hit_def)
                if not m:
                    # hit_id may be accession.version
                    print('NOID', al.hit_id, al.hit_def[:100])
                else:
                    ids.append(m.group(1))
                break
print(len(set(ids)))
print(sorted(set(ids))[:20])
PY"
