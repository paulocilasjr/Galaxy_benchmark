/bin/bash -c "python3 - <<'PY'
from Bio.SeqIO.QualityIO import FastqGeneralIterator
# load candidate reads by name
seqs={}
for title,seq,qual in FastqGeneralIterator(open('q3_unmapped_hg38_unmapped.fq')):
    seqs[title]=seq
import subprocess
meso=set(subprocess.check_output(\"samtools view -F4 q3_candidates_meso.sam | cut -f1\",shell=True,text=True).split())
ad=sum(1 for n in meso if 'AGATCGGAAGAGC' in seqs[n] or 'GATCGGAAGAGC' in seqs[n] or 'CTGTCTCTTATACACATCT' in seqs[n])
print('meso',len(meso),'adapter',ad)
# print first 20 meso sequences and if adapter
for n in list(meso)[:20]: print(n, seqs[n][:75])
PY"
