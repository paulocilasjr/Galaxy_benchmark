/bin/bash -c "python3 - <<'PY'
import gzip
sets=[]
for f in ['contaminated.rna.q1.fq.gz','contaminated.rna.q2.fq.gz','contaminated.rna.q3.fq.gz']:
    s=set()
    with gzip.open(f,'rt') as fh:
        i=0
        for line in fh:
            if i%4==1:
                s.add(line.rstrip())
            i+=1
    sets.append(s)
    print(f,len(s))
print('q1&q2',len(sets[0]&sets[1]))
print('q1&q3',len(sets[0]&sets[2]))
print('q2&q3',len(sets[1]&sets[2]))
print('all3',len(sets[0]&sets[1]&sets[2]))
# common all sequences to fasta
common=sets[0]&sets[1]&sets[2]
with open('common_all.fa','w') as out:
    for i,seq in enumerate(common):
        out.write(f'>common_{i}\\n{seq}\\n')
print('written',len(common))
PY"
