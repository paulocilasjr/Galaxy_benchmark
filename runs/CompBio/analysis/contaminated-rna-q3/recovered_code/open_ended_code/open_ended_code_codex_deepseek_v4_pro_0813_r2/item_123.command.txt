/bin/bash -c "python3 - <<'PY'
import pysam

def best_alignments(fn):
    best={}
    with pysam.AlignmentFile(fn,'r') as sam:
        for rec in sam.fetch(until_eof=True):
            if rec.is_secondary or rec.is_supplementary or rec.is_unmapped: continue
            q=rec.query_name
            asv=rec.get_tag('AS')
            if q not in best or asv>best[q]: best[q]=asv
    return best
h=best_alignments('minimap_q3.sam')
p=best_alignments('q3_all_panpan.sam')
print('reads in h',len(h),'p',len(p))
# reads where pan best AS > human best AS by thresholds, or pan mapped and human unmapped
for th in [0,5,10,15,20,30,50]:
    n=0; pan_only=0; pan_better=0
    for q,pasv in p.items():
        hasv=h.get(q)
        if hasv is None:
            pan_only+=1
        elif pasv>hasv+th:
            pan_better+=1
    print(th, 'pan_better', pan_better, 'pan_only', pan_only, 'sum', pan_better+pan_only)
PY"
