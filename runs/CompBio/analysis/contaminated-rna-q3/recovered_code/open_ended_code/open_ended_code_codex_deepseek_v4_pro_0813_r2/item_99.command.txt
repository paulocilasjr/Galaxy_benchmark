/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob, os, re
counts={}
for fn in glob.glob('qblast_cand_*.xml'):
    if os.path.getsize(fn)==0: continue
    try: recs=list(NCBIXML.parse(open(fn)))
    except Exception as e: continue
    for rec in recs:
        if not rec.alignments: continue
        al=rec.alignments[0]
        h=al.hsps[0]
        hit=al.hit_def
        # crude organism from title first token(s)
        print(os.path.basename(fn), '=>', hit[:120], h.expect, f'{h.identities}/{h.align_length}')
PY"
