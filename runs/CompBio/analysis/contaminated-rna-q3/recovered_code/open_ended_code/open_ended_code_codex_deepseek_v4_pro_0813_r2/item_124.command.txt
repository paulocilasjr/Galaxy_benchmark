/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob, os, re, collections
cat=collections.Counter()
rows=[]
for fn in glob.glob('qblast_cand_*.xml'):
    if os.path.getsize(fn)==0: continue
    try: recs=list(NCBIXML.parse(open(fn)))
    except: continue
    for rec in recs:
        if not rec.alignments: cat['no_hit']+=1; continue
        t=rec.alignments[0].hit_def
        # classify
        if 'Pan paniscus' in t or 'Pan troglodytes' in t: c='Pan'
        elif 'Mesomycoplasma' in t or 'Mycoplasma' in t: c='Mycoplasma/Mesomycoplasma'
        elif 'Conus' in t: c='Conus(adapter)'
        elif 'Homo sapiens' in t or t.startswith('Human'): c='Human'
        elif 'PREDICTED:' in t:
            # first species binomial
            m=re.search(r'PREDICTED: ([A-Z][a-z]+ [a-z]+)', t)
            c='other_'+ (m.group(1) if m else 'PRED')
        elif 'Vector' in t or 'Cloning Vector' in t or 'synthetic construct' in t: c='Vector/synth'
        else: c='other'
        cat[c]+=1
print(cat)
PY"
