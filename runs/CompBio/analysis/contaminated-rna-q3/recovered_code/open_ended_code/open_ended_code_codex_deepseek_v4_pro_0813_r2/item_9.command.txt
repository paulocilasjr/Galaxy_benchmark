/bin/bash -c "python3 - <<'PY'
import gzip, random
path='contaminated.rna.q3.fq.gz'
# decompress to fasta
reads=[]
with gzip.open(path,'rt') as f:
    i=0
    for line in f:
        if i%4==0:
            name=line.rstrip()
        elif i%4==1:
            seq=line.rstrip()
            reads.append((name,seq))
        i+=1
random.seed(42)
s=random.sample(reads,20)
with open('sample20.fa','w') as out:
    for name,seq in s:
        out.write(f'>{name[1:]}\\n{seq}\\n')
print(len(reads))
PY
head sample20.fa"
