/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob
for fn in sorted(glob.glob('blast_*.xml')):
    recs=list(NCBIXML.parse(open(fn)))
    for rec in recs:
        print('\\n',rec.query)
        for align in rec.alignments[:5]:
            hsp=align.hsps[0]
            print(' ', align.hit_def[:150], hsp.expect, f'{hsp.identities}/{hsp.align_length}', hsp.sbjct)
PY"
