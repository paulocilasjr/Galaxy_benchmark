/bin/bash -c "python3 - <<'PY'
from Bio.SeqIO.QualityIO import FastqGeneralIterator
import random
random.seed(5)
reads=[]
with open('q3_unmapped.fq') as f:
    for title,seq,qual in FastqGeneralIterator(f):
        reads.append((title,seq))
s=random.sample(reads,12)
with open('q3_unmapped12.fa','w') as out:
    for t,s in s:
        out.write(f'>{t}\\n{s}\\n')
print(len(reads))
PY
head q3_unmapped12.fa"
