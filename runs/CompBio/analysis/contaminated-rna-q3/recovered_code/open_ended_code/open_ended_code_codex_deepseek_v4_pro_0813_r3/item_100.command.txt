/bin/bash -c "python3 - <<'PY'
import json, collections
j=json.load(open('blast_q1_lympho.json'))
c=collections.Counter(); ex={}
for rep in j['BlastOutput2']:
 r=rep['report']['results']['search']; hits=r.get('hits',[])
 if not hits: continue
 d=hits[0]['description'][0]; sc=d.get('sciname',''); acc=d.get('accession',''); title=d.get('title','')
 c[(sc,acc)]+=1; ex[(sc,acc)]=title
for (sc,acc),n in c.most_common(20): print(n,sc,acc,ex[(sc,acc)][:100])
PY"
