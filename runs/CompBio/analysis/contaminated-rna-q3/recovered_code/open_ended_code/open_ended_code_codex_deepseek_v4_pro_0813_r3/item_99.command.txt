/bin/bash -c "python3 - <<'PY'
import gzip
for q in ['q1','q2']:
 want=set(line.strip() for line in open(f'{q}_lympho_ids.txt'))
 out=open(f'{q}_lympho.fa','w')
 count=0
 with gzip.open(f'contaminated.rna.{q}.fq.gz','rt') as f:
  while True:
   h=f.readline().strip()
   if not h: break
   s=f.readline().strip(); f.readline(); f.readline()
   rid=h[1:].split()[0]
   if rid in want:
    out.write(f'>{rid}\\n{s}\\n'); count+=1
 out.close(); print(q, count)
PY
python3 ncbi_blast.py q1_lympho.fa blast_q1_lympho.json"
