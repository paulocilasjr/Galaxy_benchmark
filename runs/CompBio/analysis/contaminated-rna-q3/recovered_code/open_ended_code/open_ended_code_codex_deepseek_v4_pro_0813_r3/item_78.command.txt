import json
j=json.load(open('blast_top_nonhuman.json'))
for rep in j['BlastOutput2']:

    if d.get('taxid')==2100 or 'Mesomycoplasma' in d.get('sciname','') or 'Mycoplasma' in d.get('sciname',''):
        print(rid, d.get('accession'), d.get('title'), d.get('taxid'), d.get('sciname'), hits[0]['hsps'][0].get('identity'), hits[0]['hsps'][0].get('align_len'))
PY"
