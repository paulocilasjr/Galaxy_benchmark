/bin/bash -c "python3 - <<'PY'
import json, collections
j=json.load(open('blast_top_nonhuman.json'))
wanted=set()
# get readids assigned mycoplasma from top_nonhuman_readids
for line in open('top_nonhuman_readids.txt'):
    if line.startswith('Mycoplasma'):
        wanted.update(line.rstrip().split('\\t')[1].split(','))
        break
print('wanted', len(wanted))
c=collections.Counter()
for rep in j['BlastOutput2']:
    r=rep['report']['results']['search']; rid=r.get('query_title')
    if rid not in wanted: continue
    for h in r.get('hits',[]):
        d=h['description'][0]
        c[(d.get('sciname'), d.get('accession'), d.get('title'))]+=1
for k,n in c.most_common(30): print(n,k)
PY"
