import urllib.parse, urllib.request, time
seq='>READ_16\\nCTGGGGCGGAGGCTTGACTCAGGCCTCAGTCAGCAGAGCCGTCACCGGGAGTGCGGACGTCACCGTCTGTCCACT\\n'
data={
'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,
