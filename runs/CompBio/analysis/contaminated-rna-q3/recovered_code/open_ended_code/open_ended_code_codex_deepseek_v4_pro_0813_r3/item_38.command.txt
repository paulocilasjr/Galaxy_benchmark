/bin/bash -c "python3 - <<'PY'
import urllib.parse, urllib.request, re
seq='>READ_16\\nCTGGGGCGGAGGCTTGACTCAGGCCTCAGTCAGCAGAGCCGTCACCGGGAGTGCGGACGTCACCGTCTGTCCACT\\n'
params={
'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,
'FORMAT_TYPE':'JSON2_S','MEGABLAST':'on','HITLIST_SIZE':'5','DESCRIPTIONS':'5','ALIGNMENTS':'0','NCBI_GI':'off','EXPECT':'1e-5','FILTER':'F','GAPCOSTS':'0 0','MATCH_SCORES':'1,-2'
}
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi?'+urllib.parse.urlencode(params)
print('URL len',len(url))
req=urllib.request.Request(url, method='GET', headers={'User-Agent':'Mozilla/5.0'})
r=urllib.request.urlopen(req, timeout=60)
text=r.read().decode('utf-8','ignore')
print(r.status, r.url)
for m in re.findall(r'(?:RID|Request ID|QBlastInfo|BlastRequest)["'^<]{0,200}'"', text, re.I)[:20]: print(m)
print(text[:1000])
PY"
