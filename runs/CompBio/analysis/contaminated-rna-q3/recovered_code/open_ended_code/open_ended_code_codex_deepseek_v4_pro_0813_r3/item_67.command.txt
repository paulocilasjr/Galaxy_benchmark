/bin/bash -c "python3 - <<'PY'
import collections
# parse taxonomy
parent={}; rank={}; name={}
for line in open('kraken_db/ktaxonomy.tsv'):
    parts=line.rstrip('\\n').split('\\t|\\t')
    if len(parts)<5: continue
    taxid=int(parts[0].strip()); par=int(parts[1].strip()); r=parts[2].strip(); nm=parts[4].strip()
    parent[taxid]=par; rank[taxid]=r; name[taxid]=nm

def genus_of(t):
    seen=set()
    while t in parent and t not in seen:
        seen.add(t)
        if rank.get(t)=='G':
            return t,name[t]
        t=parent[t]
    return None,None

counts=collections.Counter(); readids=collections.defaultdict(list)
for line in open('q3_kraken.out'):
    p=line.rstrip('\\n').split('\\t')
    if len(p)<5 or p[0]"'!='"'C': continue
    taxid=int(p[2])
    if taxid==0: continue
    gid,gn=genus_of(taxid)
    if gn and gn"'!='"'Homo':
        counts[(gid,gn)]+=1
        readids[(gid,gn)].append(p[1])
for (gid,gn),c in counts.most_common(30):
    print(c,gid,gn)
# write read ids for top non-human
with open('top_nonhuman_readids.txt','w') as f:
    for (gid,gn),c in counts.most_common(20):
        f.write(f'{gn}\\t{\",\".join(readids[(gid,gn)])}\\n')
PY
head -20 top_nonhuman_readids.txt"
