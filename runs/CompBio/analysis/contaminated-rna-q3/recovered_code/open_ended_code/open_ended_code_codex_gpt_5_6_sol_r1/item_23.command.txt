/bin/bash -lc "awk '"'$4>90000 && $4<650000 && $6=="75M" {print ">"$1"'"\\n\""'$10; n++; if(n==5) exit}'"' mycoplasma_hyorhinis.mapped.sam > myco_unique5.fa
