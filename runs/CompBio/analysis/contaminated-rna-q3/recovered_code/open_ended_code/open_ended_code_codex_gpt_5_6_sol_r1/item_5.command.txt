/bin/bash -lc "gzip -cd contaminated.rna.q3.fq.gz | awk 'NR%4==1 {i=(NR+3)/4; h="'$0} NR%4==2 && i%2400==0 {sub(/''^@/,">",h); print h; print $0}'"' > sampled_100.fa
blastn -remote -db nt -task blastn-short -query sampled_100.fa -max_target_seqs 3 -evalue 1e-5 -outfmt '6 qseqid sacc staxids sscinames pident length evalue bitscore' -out sampled_100.blast.tsv
wc -l sampled_100.blast.tsv
head -20 sampled_100.blast.tsv"
