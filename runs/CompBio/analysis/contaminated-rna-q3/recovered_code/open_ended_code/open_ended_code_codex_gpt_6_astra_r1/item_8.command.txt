/bin/bash -lc "cat > blast_submit.py <<'PY'
import urllib.request,urllib.parse,re,sys
p=sys.argv[1]
data=urllib.parse.urlencode(dict(CMD='Put',PROGRAM='blastn',DATABASE='nt',MEGABLAST='on',QUERY=open(p).read(),HITLIST_SIZE=5,FORMAT_TYPE='XML')).encode()
r=urllib.request.urlopen('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data,timeout=90).read().decode()
open(p+'.submit','w').write(r)
print(re.findall(r'RID = (\\S+)|RTOE = (\\S+)',r))
PY
python3 blast_submit.py sample.fa"
