/bin/bash -lc "curl -L --max-time 60 -s 'https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&RID=A4BW34NY014&FORMAT_TYPE=XML' -o candidate.reads.xml
python3 - <<'PY'
import xml.etree.ElementTree as E,collections
s=open('candidate.reads.xml').read()
if '<BlastOutput' not in s:print(s[-600:]);raise SystemExit
r=E.fromstring(s);c=collections.Counter()
for it in r.findall('.//Iteration'):
 hs=it.findall('.//Hit');q=it.findtext('Iteration_query-def')
 if not hs:continue
 h=hs[0];desc=h.findtext('Hit_def'); c[desc.split()[0]]+=1
 if 'hyorhinis' not in desc:print(q,[(h.findtext('Hit_def')[:150],h.findtext('.//Hsp_identity'),h.findtext('.//Hsp_align-len')) for h in hs])
print(c)
PY"
