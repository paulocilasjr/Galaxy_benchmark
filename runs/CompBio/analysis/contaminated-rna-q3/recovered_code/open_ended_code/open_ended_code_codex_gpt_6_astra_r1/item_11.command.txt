/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as E,collections
r=E.parse('sample.xml'); c=collections.Counter()
with open('sample.summary.tsv','w') as f:
 for it in r.findall('.//Iteration'):
  q=it.findtext('Iteration_query-def');hits=it.findall('.//Hit')
  if not hits: print(q,'NO_HIT');continue
  hs=[]
  for hit in hits:
   h=hit.find('.//Hsp'); desc=hit.findtext('Hit_def');acc=hit.findtext('Hit_accession')
   hs.append((float(h.findtext('Hsp_bit-score')),int(h.findtext('Hsp_identity')),int(h.findtext('Hsp_align-len')),desc,acc))
  top=hs[0];c[top[3].split()[0]]+=1
  row=q+'\\t'+'\\t'.join(map(str,top));f.write(row+'\\n');print(row[:350])
print(c)
PY"
