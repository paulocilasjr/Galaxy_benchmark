/bin/bash -lc "awk 'NR%4==1{h="'$0} NR%4==2{n++; s=$0; if(n<=100){print ">" substr(h,2); print s}}'"' human.unmapped.fq > unmapped100.fa
blastn -remote -db nt -query unmapped100.fa -task blastn-short -outfmt '6 qseqid saccver staxids sscinames pident length evalue bitscore' -max_target_seqs 5 -max_hsps 1 -evalue 1e-5 -out unmapped100.nt.tsv 2> unmapped100.nt.stderr
printf '%s\\n' '--- nt results ---'; wc -l unmapped100.nt.tsv; sed -n '1,80p' unmapped100.nt.tsv; printf '%s\\n' '--- stderr ---'; cat unmapped100.nt.stderr"
