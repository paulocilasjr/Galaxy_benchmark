/bin/bash -lc "gzip -cd contaminated.rna.q3.fq.gz | awk 'NR%4==1{h="'$0} NR%4==2{n++;s=$0;if(n%4800==0 && n<=240000){print ">"substr(h,2);print s}}'"' > full.distributed50.fa
wc -l full.distributed50.fa
query="'$(awk '"'{print}' full.distributed50.fa)
curl -L --fail --silent --show-error --retry 2 --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' --data-urlencode 'MEGABLAST=on' --data-urlencode 'HITLIST_SIZE=5' --data-urlencode 'FORMAT_TYPE=XML' --data-urlencode \"QUERY="'$query" '"'https://blast.ncbi.nlm.nih.gov/Blast.cgi' > full.blast50.put.html
grep -E 'RID =|RTOE =|value=\"[A-Z0-9]+\" id=\"rid\"' full.blast50.put.html | head -10"
