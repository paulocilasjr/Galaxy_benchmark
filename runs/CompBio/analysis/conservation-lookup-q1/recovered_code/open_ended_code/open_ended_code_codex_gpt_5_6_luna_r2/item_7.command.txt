import json,glob
for f in glob.glob('*_lookup.json'):
 x=json.load(open(f)); print('\\n###',f,x.get('species'),x.get('canonical_transcript'))
 for t in x.get('Transcript',[]):
  tr=t.get('Translation') or {}
  print(t['id'],t.get('version'),t.get('biotype'),t.get('is_canonical'),t.get('length'),'coords',t.get('start'),t.get('end'),'translation',tr.get('id'),tr.get('start'),tr.get('end'), 'exons',len(t.get('Exon',[])))
  for e in t.get('Exon',[]): print(' ',e['id'],e['start'],e['end'],e.get('rank'))
PY"
