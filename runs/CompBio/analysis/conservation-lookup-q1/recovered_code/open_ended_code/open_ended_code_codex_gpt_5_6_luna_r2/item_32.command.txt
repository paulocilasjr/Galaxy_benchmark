import json,glob,re
for f in glob.glob('*_lookup.json'):

 seq=open(t['id']+'_cdna.fa').read().splitlines()[1].strip().upper()

 for e in exons:
  lens.append(e['end']-e['start']+1)

  off=sum(lens[:0]) + tr['start']-exons[0]['start']
 else:
  off=exons[0]['end']-tr['end']

 print(f, t['id'],strand,'seq',len(seq),'exon lens',lens,'cds_off',off,'seglen',len(seg),'start',seg[:12],'human?',t['id'].startswith('ENST'))
