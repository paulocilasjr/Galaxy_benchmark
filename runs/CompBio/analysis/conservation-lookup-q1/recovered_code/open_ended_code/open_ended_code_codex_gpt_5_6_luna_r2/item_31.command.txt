/bin/bash -lc "python3 - <<'PY'
import json,urllib.request,urllib.parse,glob,os,time
for f in glob.glob('*_lookup.json'):
 x=json.load(open(f));
 for t in x.get('Transcript',[]):
  if t.get('is_canonical')==1:
   tid=t['id']
   u='https://rest.ensembl.org/sequence/id/'+tid+'?type=cdna'
   req=urllib.request.Request(u,headers={'Content-Type':'text/plain'})
   try:
    seq=urllib.request.urlopen(req).read().decode().strip()
    open(tid+'_cdna.fa','w').write('>'+tid+'\\n'+seq+'\\n')
    print(f,tid,len(seq),seq[:12])
   except Exception as e: print('ERR',f,tid,e)
   time.sleep(.2)
PY
ls -1 *_cdna.fa"
