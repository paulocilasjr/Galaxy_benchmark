/bin/bash -lc "python3 - <<'PY'
import requests
term='ortholog_gene_6469[group]'
j=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gene','term':term,'retmode':'json','retmax':1000}).json(); ids=j['esearchresult']['idlist']; print('n',len(ids))
# batch summaries
out=[]
for k in range(0,len(ids),200):
 x=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gene','id':','.join(ids[k:k+200]),'retmode':'json'}).json()['result']
 for i in ids[k:k+200]:
  d=x.get(i,{}); org=d.get('organism',{}); out.append((i,org.get('scientificname',''),d.get('name',''),d.get('description','')))
for row in out:
 if any(q in ' '.join(row).lower() for q in ['pan troglodytes','macaca mulatta','pan paniscus','indri indri','johnii','trachypithecus','ailuropoda melanoleuca','delphinapterus leucas']): print('\\t'.join(row))
PY"
