import urllib.request,json
for sp in ['pan_troglodytes','macaca_mulatta','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']:

 req=urllib.request.Request(u,headers={'Content-Type':'application/json'})

 print(sp)
 for h in x['data'][0]['homologies']:
  print(h['type'],h['id'],h.get('protein_id'),h.get('taxonomy_level'))
PY"
