/bin/bash -lc "cat > parse_maf.py <<'PY'
import json
from distances import lev
human=json.load(open('human_cds.json'))['seq'][:300]
spp=['Pan_troglodytes','Macaca_mulatta','Pan_paniscus','Indri_indri','Semnopithecus_johnii','Ailuropoda_melanoleuca','Delphinapterus_leucas']
x=json.load(open('shh_cactus447.json'))
out={s:'' for s in spp};out['hg38']=''
for b in sorted(x['cactus447way'],key=lambda b:b['chromStart']):
 rows=[r.split() for r in b['mafBlock'].split(';') if r.startswith('s ')]
 h=rows[0]; start=int(h[2]); seq=h[-1]
 # Include inserts internal to the human-defined interval, exclude exterior gaps.
 cols=[i for i,c in enumerate(seq) if c"'!='"'-']
 left=cols[max(155811822-start,0)]; right=cols[min(155812122-start,len(cols))-1]+1
 for s in out:
  rr=[r for r in rows if r[1].startswith(s+'.')]
  if len(rr)"'!=1:print(s,'"'rows',len(rr))
  for r in rr:out[s]+=r[-1][left:right].replace('-','').upper()
for s in out:out[s]=out[s].translate(str.maketrans('ACGT','TGCA'))[::-1]
assert out['hg38']==human
for s in spp:
 print(s,len(out[s]),lev(human,out[s]),out[s])
json.dump(out,open('cactus_segments.json','w'),indent=2)
PY
python parse_maf.py"
