import urllib.request,urllib.parse,json,concurrent.futures,time
from pathlib import Path

def get(url,path):
 p=Path(path)
 if p.exists(): return json.loads(p.read_text())
 for i in range(4):
  try:
   with urllib.request.urlopen(url,timeout=60) as r: b=r.read()
   x=json.loads(b); p.write_bytes(b); return x
  except Exception as e:
   print(path,e,flush=True); time.sleep(2+i)
 raise RuntimeError(url)

def ensembl(s,h):
 x=get('https://rest.ensembl.org/lookup/id/'+h['target']['id']+'?expand=1;content-type=application/json',s+'_gene.json')
 ts=[t for t in x['Transcript'] if 'Translation' in t]
 t=max(ts,key=lambda t:(t.get('is_canonical',0),t['Translation']['length']))
 print(s,x['id'],t['id'],t['Translation'],t['Exon'],flush=True)
 for typ in ['cds','cdna']:
  get('https://rest.ensembl.org/sequence/id/'+t['id']+'?type='+typ+';content-type=application/json',s+'_'+typ+'.json')
 return s
