/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, xml.etree.ElementTree as ET
terms=[
 'SHH[Gene] AND Indri indri[Organism]',
 'sonic hedgehog[Title] AND Indri indri[Organism]',
 'SHH[Gene] AND Semnopithecus johnii[Organism]',
 'sonic hedgehog[Title] AND Semnopithecus johnii[Organism]',
]
for db in ['gene','nucleotide','protein']:
 print('DB',db)
 for term in terms:
  url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':db,'term':term,'retmode':'json','retmax':10})
  try:
   data=json.load(urllib.request.urlopen(url,timeout=30))
   print(term, data['esearchresult']['count'], data['esearchresult'].get('idlist'))
  except Exception as e:
   print('ERR', term, e)
 print()
PY"
