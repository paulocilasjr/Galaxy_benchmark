/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
base='https://rest.ensembl.org'
url=base+'/lookup/symbol/homo_sapiens/SHH?expand=1;content-type=application/json'
data=json.load(urllib.request.urlopen(url,timeout=30))
print(data['id'], data['seq_region_name'], data['start'], data['end'], data['strand'])
for tr in data.get('Transcript',[]):
    print(tr.get('id'), tr.get('version'), tr.get('is_canonical'), tr.get('biotype'), tr.get('display_name'), tr.get('MANE'), tr.get('Translation',{}).get('id'), tr.get('Translation',{}).get('start'), tr.get('Translation',{}).get('end'), len(tr.get('Exon',[])))
PY"
