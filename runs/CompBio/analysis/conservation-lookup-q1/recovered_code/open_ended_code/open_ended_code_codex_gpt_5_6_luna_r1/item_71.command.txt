/bin/bash -lc "curl -L --fail --silent --show-error 'https://api.ncbi.nlm.nih.gov/datasets/v2/genome/taxon/34827/dataset_report?page_size=10' -o indri_assembly_report.json; curl -L --fail --silent --show-error 'https://api.ncbi.nlm.nih.gov/datasets/v2/genome/taxon/66063/dataset_report?page_size=10' -o semno_assembly_report.json; python3 - <<'PY'
import json
for f in ['indri_assembly_report.json','semno_assembly_report.json']:
 x=json.load(open(f)); print('\\n',f,'n',len(x['reports']))
 for r in x['reports']:
  print(r['accession'],r['organism'],r.get('assembly_info',{}).keys())
  print('stats',r.get('assembly_stats'))
  print('links',r.get('assembly_info',{}).get('ftp_path'),r.get('assembly_info',{}).get('refseq_category'))
  print('all keys',r.keys())
PY"
