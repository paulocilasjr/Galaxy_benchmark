/bin/bash -lc "curl -L --fail --silent --show-error -H 'Content-type: application/json' 'https://rest.ensembl.org/info/species' -o species.json && python3 - <<'PY'
import json
x=json.load(open('species.json'))
for s in x['species']:
 if any(q in (s.get('name','')+' '+s.get('display_name','')).lower() for q in ['indri','semnopithecus','johnii','pan_paniscus']): print(s)
print('count',len(x['species']))
PY"
