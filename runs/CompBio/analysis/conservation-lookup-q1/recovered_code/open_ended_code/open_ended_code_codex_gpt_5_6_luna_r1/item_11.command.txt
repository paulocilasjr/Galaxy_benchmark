/bin/bash -lc "curl -L --fail --silent --show-error -H 'Content-Type: application/json' 'https://rest.ensembl.org/lookup/id/ENSG00000164690?expand=1' -o ensembl_human_gene.json && python3 - <<'PY'
import json
x=json.load(open('ensembl_human_gene.json'))
print(x.keys())
print('gene',x.get('id'),x.get('display_name'))
for t in x.get('Transcript',[]):
 print(t['id'],t.get('is_canonical'),t.get('biotype'),t.get('length'),t.get('Translation',{}).get('id'),t.get('external_name'),[(e.get('id'),e.get('start'),e.get('end'),e.get('rank')) for e in t.get('Exon',[])])
PY"
