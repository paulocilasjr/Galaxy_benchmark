/bin/bash -lc "curl -L --fail --silent --show-error -H 'Content-type: application/json' 'https://rest.ensembl.org/homology/id/human/ENSG00000164690?type=orthologues;sequence=cdna;format=full' -o homology.json && python3 - <<'PY'
import json
x=json.load(open('homology.json'))
print(x.keys())
for d in x['data']:
 print('source',d.get('id'),d.get('type'))
 for h in d.get('homologies',[]):
  t=h['target']; print(h['type'],t['species'],t['id'],t.get('perc_id'),t.get('perc_pos'),t.keys())
PY"
