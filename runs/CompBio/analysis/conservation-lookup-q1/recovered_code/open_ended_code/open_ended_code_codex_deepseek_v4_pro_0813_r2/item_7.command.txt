/bin/bash -c "python3 - <<'PY'
import requests, json
species_list=['pan_troglodytes','macaca_mulatta','pan_paniscus','indri_indri','semnopithecus_johnii','ailuropoda_melanoleuca','delphinapterus_leucas']
for sp in species_list:
    url=f'https://rest.ensembl.org/homology/symbol/homo_sapiens/SHH'
    r=requests.get(url,params={'target_species':sp,'type':'orthologues','sequence':'none','format':'condensed'},headers={'Content-Type':'application/json'},timeout=60)
    print('\\n###',sp,r.status_code)
    if r.status_code==200:
        data=r.json()
        s=json.dumps(data,indent=2)
        print(s[:2500])
    else:
        print(r.text[:500])
PY"
