/bin/bash -c "python3 - <<'PY'
import requests,time
for db in ['nucleotide','protein','gene']:
  for term in [
    'Semnopithecus johnii[orgn] AND hedgehog',
    'Semnopithecus johnii[orgn] AND \"sonic hedgehog\"',
    'Semnopithecus johnii[orgn] AND SHH',
    'Trachypithecus johnii[orgn] AND SHH',
  ]:
    r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':db,'term':term,'retmode':'json','retmax':20},timeout=30)
    j=r.json().get('esearchresult',{})
    print(db,term,j.get('count'),j.get('idlist'))
    time.sleep(0.25)
PY"
