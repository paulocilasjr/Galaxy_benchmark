/bin/bash -c "python3 - <<'PY'
import requests, json, time
# NCBI gene SHH human
queries = {
 'human':'Homo sapiens[orgn]+SHH[gene]+alive[prop]',
 'chimpanzee':'Pan troglodytes[orgn]+SHH[gene]+alive[prop]',
 'rhesus':'Macaca mulatta[orgn]+SHH[gene]+alive[prop]',
 'bonobo':'Pan paniscus[orgn]+SHH[gene]+alive[prop]',
 'indri':'Indri indri[orgn]+SHH[gene]+alive[prop]',
 'langur':'Semnopithecus johnii[orgn]+SHH[gene]+alive[prop]',
 'panda':'Ailuropoda melanoleuca[orgn]+SHH[gene]+alive[prop]',
 'beluga':'Delphinapterus leucas[orgn]+SHH[gene]+alive[prop]',
}
for key,q in queries.items():
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi'
    r=requests.get(url,params={'db':'gene','term':q,'retmode':'json','retmax':5},timeout=30)
    print(key, r.status_code, r.text[:300])
    if r.status_code==200:
      print(r.json().get('esearchresult',{}).get('idlist'))
    time.sleep(0.34)
PY"
