/bin/bash -c "python3 - <<'PY'
import requests,time
terms={
'indri':'Indri indri[orgn] AND (SHH[gene] OR sonic hedgehog[title] OR \"sonic hedgehog\"[title])',
'langur':'Semnopithecus johnii[orgn] OR Trachypithecus johnii[orgn] AND (SHH[gene] OR sonic hedgehog[title] OR \"sonic hedgehog\"[title])',
}
for db in ['nucleotide','protein']:
  for key,term in terms.items():
    r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':db,'term':term,'retmode':'json','retmax':20},timeout=30)
    print(db,key,r.status_code,r.json().get('esearchresult',{}).get('idlist'))
    time.sleep(0.34)
PY"
