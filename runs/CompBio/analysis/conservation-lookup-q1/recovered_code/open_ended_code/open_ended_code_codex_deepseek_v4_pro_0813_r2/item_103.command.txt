/bin/bash -c "python3 - <<'PY'
import requests,time
for gid in ['105883225','105818206']:
 # elink gene to nucleotide refseq
 for db in ['nucleotide','protein']:
  r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/elink.fcgi',params={'dbfrom':'gene','db':db,'id':gid,'linkname':f'gene_{db}_refseq','retmode':'json'},timeout=30)
  print('elink',gid,db,r.status_code,r.text[:500])
  if r.status_code==200:
   j=r.json(); print(j)
  time.sleep(0.4)
PY"
