import gzip
from Bio.Seq import Seq

def readseq(fn,target):
 with gzip.open(fn,'rt') as f:
  name=None; parts=[]
  for line in f:
   if line.startswith('>'):
    if name==target: break
    name=line[1:].split()[0]; parts=[]
   elif name==target:
    parts.append(line.strip())
  return ''.join(parts)

def upstream_indri(k=60):
 s=readseq('indri.fna.gz','CAUYTU010176261.1')

 return str(Seq(sub).reverse_complement())
