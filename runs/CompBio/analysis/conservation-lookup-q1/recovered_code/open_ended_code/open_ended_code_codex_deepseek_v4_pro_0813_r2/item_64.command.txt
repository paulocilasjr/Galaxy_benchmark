/bin/bash -c "python3 - <<'PY'
import requests
for taxon in ['Indri indri','Semnopithecus johnii','Trachypithecus johnii']:
  # uniprot taxonomy search
  r=requests.get('https://rest.uniprot.org/taxonomy/search',params={'query':taxon,'format':'json'},timeout=30)
  print(taxon,r.status_code)
  if r.status_code==200:
    j=r.json(); print(j.get('results',[])[:3])
PY"
