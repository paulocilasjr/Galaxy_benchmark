/bin/bash -c "python3 - <<'PY'
import requests
urls=['https://rapid.ensembl.org/rest/info/species','https://rapid.ensembl.org/info/species','https://rapid.ensembl.org/rest/info/species?division=EnsemblVertebrates']
for u in urls:
 try:
  r=requests.get(u,headers={'Content-Type':'application/json'},timeout=30)
  print(u,r.status_code,r.url,len(r.text),r.text[:300])
 except Exception as e: print(e)
PY"
