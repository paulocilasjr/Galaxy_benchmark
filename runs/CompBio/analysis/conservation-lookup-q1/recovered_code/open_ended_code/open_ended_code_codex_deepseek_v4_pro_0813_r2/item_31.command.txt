/bin/bash -c "python3 - <<'PY'
import requests,json,time
sp='macaca_mulatta'
for i in range(8):
    r=requests.get('https://rest.ensembl.org/homology/symbol/homo_sapiens/SHH',params={'target_species':sp,'type':'orthologues','format':'condensed'},headers={'Content-Type':'application/json'},timeout=60)
    print(i,r.status_code, r.text[:200])
    if r.status_code==200:
        print(json.dumps(r.json(),indent=2)[:2000])
        break
    time.sleep(5)
PY"
