import gzip
from Bio.Seq import Seq

def get_sem_up(k=300):
 target='CAUYPG010105159.1'; human_start_high=30507
 with gzip.open('semjoh.fna.gz','rt') as f:
  name=None; parts=[]
  for line in f:
   if line.startswith('>'):
    if name==target: break
    name=line[1:].split()[0]; parts=[]
   elif name==target:
    parts.append(line.strip())
  s=''.join(parts)
 # upstream cDNA high coords human_start_high+1 .. +k
 sub=s[human_start_high:human_start_high+k]
 return str(Seq(sub).reverse_complement())
up=get_sem_up(300)
print(up[-150:])
print('ATG offsets:')
for i in range(len(up)):
 if up[i:i+3]=='ATG': print('offset',i-len(up),'mod',(i-len(up))%3)
PY"
