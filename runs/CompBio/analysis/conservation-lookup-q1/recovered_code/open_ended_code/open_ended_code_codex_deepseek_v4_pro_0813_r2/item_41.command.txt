/bin/bash -c "python3 - <<'PY'
import json
p='indri_blast_result.txt'
data=json.load(open(p))
reports=data['BlastOutput2']
print('reports',len(reports))
for rep in reports:
    rep=rep.get('report') or rep
    print('program',rep.get('program'))
    res=rep.get('results')
    if isinstance(res,dict):
      search=res.get('search')
      print('query len',search.get('query_len'))
      hits=res.get('hits',[])
      print('hits',len(hits))
      for h in hits[:10]:
        desc=h.get('description',[])[0] if h.get('description') else {}
        print('hit num',h.get('num'),'id',desc.get('id'),'title',desc.get('title'),'len',h.get('len'))
        for hs in h.get('hsps',[])[:2]:
          print('  hsp',hs.get('qseq'),hs.get('hseq'),hs.get('qcov'),hs.get('hsp_hseq'))
PY"
