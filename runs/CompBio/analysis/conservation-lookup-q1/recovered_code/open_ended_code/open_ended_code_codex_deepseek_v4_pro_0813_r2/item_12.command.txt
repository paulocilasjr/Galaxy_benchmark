/bin/bash -c "python3 - <<'PY'
import requests
url='https://rest.ensembl.org/info/species'
r=requests.get(url,params={'division':'EnsemblVertebrates'},headers={'Content-Type':'application/json'},timeout=120)
print(r.status_code, len(r.text))
if r.status_code==200:
    data=r.json()
    print(type(data), len(data.get('species',[])))
    for s in data.get('species',[]):
        name=s.get('name','')
        if any(x in name.lower() for x in ['indri','semnopithecus','trachypithecus','johnii']):
            print(s)
PY"
