from Bio import SeqIO
# use pyfaidx? Biopython may not installed. Let's parse manually specific accessions.
# Instead use samtools faidx? Check. Use grep sequences with custom parser streaming.
def extract(path, acc, start, end, strand):
    # 1-based inclusive coords
    name=None; seq=[]
    found=False
    for line in open(path):
        if line.startswith('>'):
            if found: break
            name=line[1:].strip().split()[0]
            if name==acc:
                found=True; seq=[]
            else:
                found=False; seq=[]
        elif found:
            seq.append(line.strip())
    s=''.join(seq).upper()
    sub=s[start-1:end]
    if strand=='-':
        comp=str.maketrans('ACGT','TGCA')
        sub=sub.translate(comp)[::-1]
    return sub
# Indri hit CAUYTU010176261.1 17146..17445 plus

print('indri extract len',len(s)); print(s)

print('sem extract len',len(s2)); print(s2)
