/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, time
ids={'ENSPTRT00000109799':'pan_troglodytes','ENSMMUT00000090541':'macaca_mulatta','ENSPPAT00000043352':'pan_paniscus','ENSAMET00000046595':'ailuropoda_melanoleuca','ENSDLET00000022460':'delphinapterus_leucas'}
for tid,sp in ids.items():
 try:
  d=json.loads(urllib.request.urlopen(f'https://rest.ensembl.org/lookup/id/{tid}?content-type=application/json;expand=1').read().decode())
  print('\\n',sp,tid)
  print('strand',d.get('strand'),'length',d.get('length'),'biotype',d.get('biotype'))
  print('exons',[(e['start'],e['end']) for e in d.get('Exon',[])])
  print('translation',d.get('Translation'))
 except Exception as e: print('ERROR',tid,e)
 time.sleep(0.2)
PY"
