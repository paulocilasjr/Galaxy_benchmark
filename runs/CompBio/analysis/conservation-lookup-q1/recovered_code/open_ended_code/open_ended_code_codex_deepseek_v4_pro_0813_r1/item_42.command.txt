/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, time
for sp in ['Indri indri','Semnopithecus johnii']:
 u='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':'assembly','term':f'{sp}[Organism] AND latest[filter]','retmode':'json'})
 j=json.loads(urllib.request.urlopen(u).read().decode())
 ids=j['esearchresult']['idlist']
 print('\\n',sp,'assembly ids',ids,'count',j['esearchresult']['count'])
 for aid in ids[:5]:
  u2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urllib.parse.urlencode({'db':'assembly','id':aid,'retmode':'json'})
  s=json.loads(urllib.request.urlopen(u2).read().decode())['result'][aid]
  print(aid, s.get('assemblyname'), s.get('assemblystatus'), s.get('refseq_category'), s.get('ftp_path_refseq'), s.get('ftp_path_genbank'))
  time.sleep(0.34)
 time.sleep(0.34)
PY"
