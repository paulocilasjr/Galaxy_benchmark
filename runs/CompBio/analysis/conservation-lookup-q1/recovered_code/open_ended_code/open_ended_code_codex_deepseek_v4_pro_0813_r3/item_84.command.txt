/bin/bash -lc "curl -s \"https://rest.ensembl.org/lookup/symbol/macaca_mulatta/SHH?expand=1\" -H \"Content-Type: application/json\" -o ens_mmul.json; python3 - <<'PY'
import json
j=json.load(open('ens_mmul.json'))
print(j.keys() if isinstance(j,dict) else j)
if 'id' in j:
 print('gene',j['id'],j.get('canonical_transcript'))
 for t in j.get('Transcript',[]): print(t['id'],t['biotype'],t['is_canonical'],t['length'])
else:
 print(j)
PY"
