/bin/bash -lc "python3 - <<'PY'
import json, subprocess
ids=['743371','716553','100995872','100481788','111176269']
url='https://api.ncbi.nlm.nih.gov/datasets/v2/gene/id/'+','.join(ids)+'/product_report?page_size=100'
out='ncbi_products.json'
subprocess.run(['curl','-s',url,'-o',out])
j=json.load(open(out))
print(j.keys(), j.get('total_count'))
for rep in j.get('reports',[]):
    print('\\nREPORT KEYS',rep.keys())
    print(json.dumps(rep, indent=2)[:3000])
    break
PY"
