/bin/bash -lc "curl -s \"https://omabrowser.org/api/genome/?page=1&per_page=10000\" -H \"Content-Type: application/json\" -o oma_genomes.json; wc -c oma_genomes.json; python3 - <<'PY'
import json
j=json.load(open('oma_genomes.json'))
print(type(j), len(j) if isinstance(j,list) else j.keys())
if isinstance(j,dict):
    data=j.get('data') or j.get('results') or j.get('genomes') or []
else: data=j
print('n',len(data))
for target in ['Indri indri','Semnopithecus johnii','Pan troglodytes','Delphinapterus leucas']:
    hits=[g for g in data if target.lower() in (g.get('species','') or '').lower()]
    print(target, [(g.get('code'),g.get('species'),g.get('taxon_id')) for g in hits[:10]])
PY"
