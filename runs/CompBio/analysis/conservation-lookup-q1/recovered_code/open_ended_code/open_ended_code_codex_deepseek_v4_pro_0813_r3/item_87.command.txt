/bin/bash -lc 'for sp in pan_troglodytes pan_paniscus ailuropoda_melanoleuca delphinapterus_leucas; do out="ens_${sp}.json"; code=$(curl -s -o "$out" -w "%{http_code}" "https://rest.ensembl.org/lookup/symbol/${sp}/SHH?expand=1" -H "Content-Type: application/json"); echo $sp $code $(wc -c <$out); done; python3 - <<'"'PY'
import json,os
for sp in ['pan_troglodytes','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']:
 f=f'ens_{sp}.json'
 try:
  j=json.load(open(f))
  print('\\n',sp,j.get('id'),'canonical',j.get('canonical_transcript'))
  for t in j.get('Transcript',[]): print(t['id'],t.get('biotype'),t.get('is_canonical'),t.get('length'),t.get('Translation',{}).get('length'))
 except Exception as e: print(sp,'ERR',e, open(f).read()[:200])
PY"
