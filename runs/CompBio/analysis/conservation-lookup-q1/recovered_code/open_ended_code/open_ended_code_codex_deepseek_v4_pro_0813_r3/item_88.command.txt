/bin/bash -lc "python3 - <<'PY'
import subprocess,json,time,os
for sp in ['pan_troglodytes','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']:
 out=f'ens_{sp}.json'
 ok=False
 for i in range(8):
  code=subprocess.run(['curl','-s','-o',out,'-w','%{http_code}',f'https://rest.ensembl.org/lookup/symbol/{sp}/SHH?expand=1','-H','Content-Type: application/json'],capture_output=True,text=True).stdout
  if code=='200':
   try:
    j=json.load(open(out)); ok=True
    print(sp,j.get('id'),'canonical',j.get('canonical_transcript'))
    for t in j.get('Transcript',[]): print(t['id'],t.get('biotype'),t.get('is_canonical'),t.get('length'),t.get('Translation',{}).get('length'))
    break
   except Exception as e:
    print(sp,'parse error',code,e,open(out).read()[:100])
  else:
   print(sp,'try',i,code, end='; ')
  time.sleep(2)
 if not ok: print(sp,'FAIL')
PY"
