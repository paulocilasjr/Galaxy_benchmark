/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET
root=ET.parse('workspace_gene.xml').getroot()
# traverse all Gene-commentary_accession and labels
for gc in root.iter('Gene-commentary'):
    acc=gc.find('Gene-commentary_accession')
    heading=gc.findtext('Gene-commentary_heading')
    label=gc.findtext('Gene-commentary_label')
    typev=gc.findtext('Gene-commentary_type')
    if acc is not None and (label=='MANE Select' or heading=='mRNA Sequence'):
        print(acc.text, heading, label, typev)
PY"
