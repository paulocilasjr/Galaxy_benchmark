/bin/bash -lc "python3 - <<'PY'
import subprocess,json
id='117073036'
url=f'https://api.ncbi.nlm.nih.gov/datasets/v2/gene/id/{id}/product_report?page_size=100'
out='trachy_prod.json'; subprocess.run(['curl','-s',url,'-o',out]); j=json.load(open(out))
for rep in j['reports']:
 p=rep['product']; print(p['taxname'],p['gene_id'])
 for tr in p['transcripts']:
  print(' ',tr['accession_version'],'len',tr['length'],'protein',tr.get('protein',{}).get('length'),'cds',tr['cds']['range'])
  gl=tr['genomic_locations'][0]; ex1=[e for e in gl['exons'] if e.get('order')==1][0]; print('  ex1 len',int(ex1['end'])-int(ex1['begin'])+1,'cds begin',tr['cds']['range'][0]['begin'])
PY"
