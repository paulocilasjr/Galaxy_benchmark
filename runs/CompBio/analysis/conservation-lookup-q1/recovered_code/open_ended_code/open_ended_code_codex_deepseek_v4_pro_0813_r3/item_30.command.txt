/bin/bash -lc "curl -s \"https://rest.ensembl.org/lookup/symbol/homo_sapiens/SHH?expand=1\" -H \"Content-Type: application/json\" -o ens_shh.json; python3 - <<'PY'
import json
j=json.load(open('ens_shh.json'))
print(j.keys())
print('id',j.get('id'),'seq_region',j.get('seq_region_name'))
print('transcripts',len(j.get('Transcript',[])))
for t in j.get('Transcript',[]):
    print(t.get('id'), t.get('display_name'), t.get('biotype'), t.get('is_canonical'), t.get('length'), t.get('Translation',{}).get('id'))
PY"
