/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET
root=ET.parse('workspace_gene.xml').getroot()
for gc in root.iter('Gene-commentary'):
    if gc.findtext('Gene-commentary_label')=='MANE Select':
        # print ancestors/accessions
        print('label found')
        for anc in root.iter():
            pass
        # parent chain manually
        p=gc
        chain=[]
        while p is not None:
            tag=p.tag
            acc=p.findtext('Gene-commentary_accession')
            heading=p.findtext('Gene-commentary_heading')
            chain.append((tag, acc, heading))
            # find parent
            p_parent=None
            for q in root.iter():
                if p in list(q):
                    p_parent=q; break
            p=p_parent
        print(chain)
        print(ET.tostring(gc, encoding='unicode')[:1000])
        break
PY"
