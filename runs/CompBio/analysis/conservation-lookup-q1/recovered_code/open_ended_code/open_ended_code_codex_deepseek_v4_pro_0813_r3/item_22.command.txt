/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('orthologs.json'))
for t in ['Pan troglodytes','Macaca mulatta','Pan paniscus','Indri indri','Semnopithecus johnii','Ailuropoda melanoleuca','Delphinapterus leucas']:
    hits=[]
    for r in j['reports']:
        blob=json.dumps(r)
        if t.lower() in blob.lower(): hits.append((r.get('gene_id'),r.get('taxname'),r.get('common_name')))
    print(t, hits[:10])
PY"
