/bin/bash -lc "curl -s \"https://api.ncbi.nlm.nih.gov/datasets/v2/gene/id/6469/orthologs?returned_content=COMPLETE&page_size=1000\" -o orthologs.json; wc -c orthologs.json; python3 - <<'PY'
import json
j=json.load(open('orthologs.json'))
print(j.keys())
print(j.get('total_count'))
reports=j.get('reports') or j.get('genes') or []
print(len(reports))
for r in reports:
    print(r.get('gene_id'), r.get('gene',''), r.get('taxname'), r.get('common_name'))
PY"
