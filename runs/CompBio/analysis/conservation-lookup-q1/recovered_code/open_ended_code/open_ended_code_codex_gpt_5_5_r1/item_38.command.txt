/bin/bash -lc "python3 - <<'PY'
import json,urllib.parse,urllib.request,re,time
human=json.load(open('ensembl_segments.json'))['homo_sapiens']['seq']
for db in ['wgs','nt','core_nt']:
 for org in ['Indri indri','Semnopithecus johnii']:
  params={\"CMD\":\"Put\",\"PROGRAM\":\"blastn\",\"MEGABLAST\":\"on\",\"DATABASE\":db,\"ENTREZ_QUERY\":f\"{org}[Organism]\",\"QUERY\":human,\"WORD_SIZE\":\"16\"}
  text=urllib.request.urlopen(urllib.request.Request('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=urllib.parse.urlencode(params).encode()),timeout=60).read().decode()
  m=re.search(r'RID = ([A-Z0-9-]+).*?RTOE = (\\d+)',text,re.S)
  print(db,org,'rid',m.groups() if m else 'NO', 'err?', 'Message ID#' in text)
  open(f'blast_put_{db}_{org.replace(\" \",\"_\")}.html','w').write(text)
  time.sleep(5)
PY"
