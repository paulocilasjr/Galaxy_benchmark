import json
out=json.load(open('ensembl_segments.json'))
h=out['homo_sapiens']['seq']

 s=rec['seq']

 print('dist full?', end=' ')
 # simple levenshtein
 dp=list(range(len(s)+1))
 for i,ch in enumerate(h,1):
  nd=[i]+[0]*len(s)
  for j,c in enumerate(s,1):
   nd[j]=min(dp[j]+1, nd[j-1]+1, dp[j-1]+(ch"'!=c))
  dp=nd
 print(dp[-1])
 print('"'human in seq index',s.find(h[:30]), 'prefix', s[:15])
 if len(s)==309:
  ss=s[9:]
  dp=list(range(len(ss)+1))

   nd=[i]+[0]*len(ss)
   for j,c in enumerate(ss,1): nd[j]=min(dp[j]+1,nd[j-1]+1,dp[j-1]+(ch"'!=c))
   dp=nd
  print('"'dist minus9', dp[-1], ss[:30])
PY"
