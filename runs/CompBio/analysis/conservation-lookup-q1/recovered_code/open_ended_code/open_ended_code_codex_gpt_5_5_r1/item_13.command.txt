/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, json, time
queries=['SHH AND Indri indri[Organism]','sonic hedgehog AND Indri indri[Organism]','SHH AND Semnopithecus johnii[Organism]','sonic hedgehog AND Semnopithecus johnii[Organism]']
for q in queries:
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gene&term='+urllib.parse.quote(q)+'&retmode=json&retmax=10'
    j=json.load(urllib.request.urlopen(url))
    print(q, j['esearchresult']['count'], j['esearchresult']['idlist'])
    time.sleep(.35)
PY"
