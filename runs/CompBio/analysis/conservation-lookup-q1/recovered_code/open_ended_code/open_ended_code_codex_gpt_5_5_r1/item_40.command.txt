/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, time
runs={'indri_wgs':'6NRC5WT9014','sem_wgs':'6NRCB6AR014','indri_nt':'6NRCH9E5014','sem_nt':'6NRCR9RC014'}
for name,rid in runs.items():
 print('poll',name,rid,flush=True)
 for k in range(20):
  url='https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_OBJECT':'SearchInfo'})
  text=urllib.request.urlopen(url,timeout=60).read().decode()
  print(text.strip().replace('\\n',' | ')[:200],flush=True)
  if 'Status=WAITING' in text:
   time.sleep(10); continue
  if 'Status=READY' in text and 'ThereAreHits=yes' in text:
   xml=urllib.request.urlopen('https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML'}),timeout=120).read().decode()
   open(name+'.xml','w').write(xml)
   print('saved',name,len(xml),flush=True)
  break
PY"
