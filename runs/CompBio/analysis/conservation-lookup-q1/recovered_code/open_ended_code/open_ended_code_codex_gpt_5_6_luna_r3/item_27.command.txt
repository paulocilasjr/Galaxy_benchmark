/bin/bash -lc "for x in 'pan_troglodytes ENSPTRT00000109799' 'macaca_mulatta ENSMMUT00000090541' 'pan_paniscus ENSPPAT00000043352' 'ailuropoda_melanoleuca ENSAMET00000046595' 'delphinapterus_leucas ENSDLET00000022460'; do set -- "'$x; sp=$1; tr=$2; curl -L --fail --silent --show-error -H '"'Content-Type: application/json' \"https://rest.ensembl.org/lookup/id/"'$tr?expand=1" -o "${sp}.json"; curl -L --fail --silent --show-error -H '"'Content-Type: text/plain' \"https://rest.ensembl.org/sequence/id/"'$tr?type=cdna" -o "${sp}.cdna.fa"; done
python - <<'"'PY'
import json,glob
for f in glob.glob('*.json'):
 if f in ['human_ens.json','species.json']: continue
 x=json.load(open(f)); print('\\n',f,x['id'],x.get('species'),x.get('strand'),x.get('start'),x.get('end'),x.get('Translation'))
 print([(e['id'],e['start'],e['end'],e.get('strand')) for e in x.get('Exon',[])])
 seq=open(f.replace('.json','.cdna.fa')).read().replace('\\n','')
 print('seqhead',seq[:60],'len',len(seq))
PY"
