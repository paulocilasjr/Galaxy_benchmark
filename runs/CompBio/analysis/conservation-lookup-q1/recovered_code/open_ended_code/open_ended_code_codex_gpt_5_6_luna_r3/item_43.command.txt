/bin/bash -lc "python - <<'PY'
import gzip
name='CAUYTU010176261.1'; lo,hi=16816,17445
found=False; pos=0; out=[]
with gzip.open('indri_genomic.fna.gz','rt') as f:
 for line in f:
  if line.startswith('>'):
   found=line[1:].split()[0]==name; pos=0
   continue
  if found:
   s=line.strip()
   a=max(0,lo-pos-1); b=min(len(s),hi-pos)
   if a<b: out.append(s[a:b])
   pos+=len(s)
   if pos>=hi: break
seq=''.join(out)
print('length',len(seq)); print(seq)
PY"
