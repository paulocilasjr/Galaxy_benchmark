/bin/bash -lc "python3 - <<'PY'
import requests,re
seq=''.join(x.strip() for x in open('human_target.fa') if not x.startswith('>'))
for asm in ['GCA_004363605.1','GCA_963574185.1']:
 p={'CMD':'Put','PROGRAM':'blastn','DATABASE':asm,'QUERY':seq,'MEGABLAST':'on','HITLIST_SIZE':10,'BLAST_SPEC':'GDH_'+asm,'PAGE_TYPE':'BlastSearch','PROG_DEF':'blastn'}
 r=requests.post('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=p,timeout=60)
 rid=re.search(r'RID = ([A-Z0-9-]+)',r.text); err=re.findall(r'(?i)(?:error|message)["'^<]{0,200}'"',r.text)
 print(asm,r.status_code,len(r.text),rid.group(1) if rid else None,err[-5:])
 open('submit_'+asm+'.html','w').write(r.text)
PY"
