/bin/bash -lc "python3 - <<'PY'
from rapidfuzz.distance import Levenshtein
import json
h=''.join(x.strip() for x in open('human_target.fa') if not x.startswith('>'))
seqs=[]
# annotated transcript intervals (last 300 nt of orthologous first exon)
def readfa(p):
 d={};k=None
 for l in open(p):
  if l.startswith('>'): k=l[1:].split()[0];d[k]=''
  else:d[k]+=l.strip().upper()
 return d
for n,a,e in [('chimp','XM_001147185.7',641),('macaque','XM_001106515.5',641),('bonobo','XM_034965340.4',1850)]: seqs.append(readfa(f'ncbi_{n}/ncbi_dataset/data/rna.fna')[a][e-300:e])
for f in ['blast2/759P8D7G014_1.json','blast2/759P9DG1014_1.json']:
 x=json.load(open(f))['BlastOutput2']['report']['results']['search']['hits'][0]['hsps'][0]
 seqs.append(x['hseq'].replace('-',''))
for n,a,e in [('panda','XM_034650041.1',609),('beluga','XM_022576607.2',444)]: seqs.append(readfa(f'ncbi_{n}/ncbi_dataset/data/rna.fna')[a][e-300:e])
ans=[Levenshtein.distance(h,s) for s in seqs]
assert all(len(s)==300 for s in seqs)
assert ans==[0,7,0,20,5,24,27]
print(','.join(map(str,ans)))
PY"
