/bin/bash -lc "curl -s 'https://api.ncbi.nlm.nih.gov/datasets/v2alpha/gene/id/6469/orthologs' -o shh_orthologs.json && python3 - <<'PY'
import json
x=json.load(open('shh_orthologs.json'))
wanted={'Pan troglodytes','Macaca mulatta','Pan paniscus','Indri indri','Semnopithecus johnii','Ailuropoda melanoleuca','Delphinapterus leucas'}
for r in x['reports']:
 g=r['gene']
 if g.get('taxname') in wanted:
  print(json.dumps(g,indent=2))
PY"
