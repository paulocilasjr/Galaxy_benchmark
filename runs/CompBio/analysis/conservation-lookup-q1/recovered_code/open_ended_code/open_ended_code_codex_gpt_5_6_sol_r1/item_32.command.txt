/bin/bash -lc "python3 - <<'PY'
import requests,time
rids={'indri':'759KUGAR016','johnii':'759KV35E016'}
time.sleep(30)
for name,rid in rids.items():
 r=requests.get('https://blast.ncbi.nlm.nih.gov/Blast.cgi',params={'CMD':'Get','RID':rid,'FORMAT_TYPE':'JSON2'},timeout=120)
 open('blast_'+name+'_genome.bin','wb').write(r.content)
 print(name,r.status_code,len(r.content),r.headers.get('Content-Type'))
PY"
