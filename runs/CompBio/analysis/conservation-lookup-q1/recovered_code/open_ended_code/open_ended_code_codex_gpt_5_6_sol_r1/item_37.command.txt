/bin/bash -lc "sleep 30
python3 - <<'PY'
import requests
for name,rid in [('indri','759P8D7G014'),('johnii','759P9DG1014')]:
 r=requests.get('https://blast.ncbi.nlm.nih.gov/Blast.cgi',params={'CMD':'Get','RID':rid,'FORMAT_TYPE':'JSON2'},timeout=120)
 open('blast2_'+name+'.bin','wb').write(r.content)
 print(name,len(r.content),r.headers.get('Content-Type'),r.text[:80])
PY"
