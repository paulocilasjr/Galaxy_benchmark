/bin/bash -lc 'for pair in chimp:743371 macaque:716553 bonobo:100995872 panda:100481788 beluga:111176269; do name=${pair%:*}; id=${pair#*:}; curl -s -L "https://api.ncbi.nlm.nih.gov/datasets/v2/gene/id/$id/download?include_annotation_type=FASTA_RNA" -o "ncbi_${name}.zip" & done; wait
for f in ncbi_*.zip; do echo -n "$f "; file "$f"; done'
