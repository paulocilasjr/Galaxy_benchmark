/bin/bash -lc "command -v blastn || true
python3 - <<'PY'
import requests
seq=''.join(x.strip() for x in open('human_target.fa') if not x.startswith('>'))
params={'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,'MEGABLAST':'on','HITLIST_SIZE':100,'ENTREZ_QUERY':'Indri indri[Organism]'}
r=requests.post('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=params,timeout=60)
print(r.status_code,r.text[:1000])
open('blast_submit_indri.txt','w').write(r.text)
PY"
