/bin/bash -lc "curl -sS 'https://rest.ensembl.org/info/species?content-type=application/json' -o species.json
python3 - <<'PY'
import json
x=json.load(open('species.json'))['species']
terms=['troglodytes','mulatta','paniscus','indri','johnii','melanoleuca','leucas']
for t in terms:
 print('\\n',t)
 for z in x:
  if t in (z.get('name','')+' '+z.get('display_name','')+' '+','.join(z.get('aliases',[]))).lower(): print(z)
PY"
