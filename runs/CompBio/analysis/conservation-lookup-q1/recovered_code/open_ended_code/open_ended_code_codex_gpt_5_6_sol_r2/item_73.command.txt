/bin/bash -lc "curl --max-time 60 -sS 'https://rest.ensembl.org/sequence/id/ENSMMUT00000090541?type=cdna;content-type=text/plain' -o mmu_ensembl_cdna.fa &
curl --max-time 60 -sS 'https://rest.ensembl.org/overlap/id/ENSMMUT00000090541?feature=exon;content-type=application/json' -o mmu_exons.json &
wait
python3 - <<'PY'
import json
s=''.join(open('mmu_ensembl_cdna.fa').read().split())
ex=[e for e in json.load(open('mmu_exons.json')) if e['Parent']=='ENSMMUT00000090541']
print(len(s),[(e['rank'],e['start'],e['end']) for e in ex])
for m in ['ATGGGCGAGATG','ATGCTGCTGCT']: print(m,s.find(m))
PY"
