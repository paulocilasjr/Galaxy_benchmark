/bin/bash -lc "curl -sS 'https://rest.ensembl.org/sequence/id/ENSP00000297261?type=protein;content-type=text/x-fasta' -o human_shh_protein.fa
gzip -dk johnii_tiberius.aa.gz
gzip -dk johnii_tiberius.cds.gz
gzip -dk johnii_tiberius.gtf.gz
makeblastdb -in johnii_tiberius.aa -dbtype prot -out johnii_protdb >/dev/null
blastp -query human_shh_protein.fa -db johnii_protdb -outfmt '6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore' -max_target_seqs 5"
