/bin/bash -lc "curl --max-time 60 -sS 'https://rest.ensembl.org/sequence/id/ENSDLET00000022460?type=cdna;content-type=text/plain' -o beluga_ensembl_cdna.fa &
curl --max-time 60 -sS 'https://rest.ensembl.org/overlap/id/ENSDLET00000022460?feature=exon;content-type=application/json' -o beluga_exons.json &
wait
python3 - <<'PY'
import json
s=''.join(open('beluga_ensembl_cdna.fa').read().split()); ex=[e for e in json.load(open('beluga_exons.json')) if e['Parent']=='ENSDLET00000022460']
print(len(s),[(e['rank'],e['start'],e['end']) for e in ex]);
for motif in ['ATGGACGAGATG','ATGCTACTG','ATGCTG']: print(motif,s.find(motif))
print(s[:180])
PY"
