/bin/bash -lc "curl --max-time 60 -sS 'https://rest.ensembl.org/sequence/id/ENSPPAT00000043352?type=cdna;content-type=text/plain' -o ppan_ensembl_cdna.fa &
curl --max-time 60 -sS 'https://rest.ensembl.org/overlap/id/ENSPPAT00000043352?feature=exon;content-type=application/json' -o ppan_exons.json &

import json
s=''.join(open('ppan_ensembl_cdna.fa').read().split()); ex=[e for e in json.load(open('ppan_exons.json')) if e['Parent']=='ENSPPAT00000043352']

PY"
