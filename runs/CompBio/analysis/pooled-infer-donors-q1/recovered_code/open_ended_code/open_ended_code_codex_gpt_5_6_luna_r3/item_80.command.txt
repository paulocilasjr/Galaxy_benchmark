/bin/bash -lc "python - <<'PY'
import numpy as np
from sklearn.metrics import adjusted_rand_score
A=np.load('cell_site_counts.npz')['counts']; R=A[:,:,0].T.astype(float); ALT=A[:,:,1].T.astype(float)
Z=np.load('em_all_K3.npz'); L=Z['labels']; ix=Z['site_indices']; G=Z['genotypes']; lines=open('validated_candidates.tsv').read().splitlines(); chrom=np.array([lines[i].split('\\t')[0] for i in ix])
p=np.array([.03,.5,.97])
print('global',np.bincount(L))
for ch in sorted(set(chrom),key=lambda x:(len(x),x)):
 j=np.where(chrom==ch)[0]; # G indexed cluster, j local
 ll=[]
 for k in range(3):
  pp=p[G[k,j]]; ll.append(R[:,ix[j]]@np.log1p(-pp)+ALT[:,ix[j]]@np.log(pp))
 ll=np.array(ll).T
 pred=ll.argmax(1); n=(R[:,ix[j]]+ALT[:,ix[j]]).sum(1); ok=n>=3
 print(ch,'sites',len(j),'reads',int(n.sum()),'ok',int(ok.sum()),'ARI',round(adjusted_rand_score(L[ok],pred[ok]),3),'conf',np.bincount(pred[ok],minlength=3),'marg',round(np.mean(np.sort(ll[ok],axis=1)[:,-1]-np.sort(ll[ok],axis=1)[:,-2]),3))
PY"
