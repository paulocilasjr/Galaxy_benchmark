/bin/bash -lc "python - <<'PY'
import numpy as np, os
A=np.load('cell_site_counts.npz')['counts']; R=A[:,:,0].T.astype(float); Alt=A[:,:,1].T.astype(float); lines=open('validated_candidates.tsv').read().splitlines()
p=np.array([.03,.5,.97]); base_ix=np.load('em_all_K3.npz')['site_indices']; chrom=np.array([lines[i].split('\\t')[0] for i in base_ix])
for K in [2,3,4,5]:
 z=np.load(f'em_all_K{K}.npz'); L=z['labels']; ix=z['site_indices']; assert np.array_equal(ix,base_ix)
 vals=[]
 for ch in sorted(set(chrom), key=lambda x:(len(x),x)):
  tr=np.where(chrom"'!=ch)[0]; te=np.where(chrom==ch)[0]
  g=np.zeros((K,len(tr)),int)
  for k in range(K):
   sel=L==k; rr=R[:,ix[tr]][sel].sum(0); aa=Alt[:,ix[tr]][sel].sum(0)
   g[k]=np.argmax(rr[:,None]*np.log1p(-p)[None,:]+aa[:,None]*np.log(p)[None,:],1)
  scores=[]
  for k in range(K):
   pp=p[g[k]]; scores.append(R[:,ix[te]]@np.log1p(-pp)+Alt[:,ix[te]]@np.log(pp))
  sc=np.array(scores).T[np.arange(len(L)),L].sum()
  # max-score heldout as an upper envelope
  mx=np.max(np.array(scores),axis=0).sum()
  n=(R[:,ix[te]]+Alt[:,ix[te]]).sum()
  vals.append((ch,sc,mx,n))
 print('"'K',K,'fixed',round(sum(v[1] for v in vals),1),'max',round(sum(v[2] for v in vals),1),'perread',round(sum(v[1] for v in vals)/sum(v[3] for v in vals),4))
 print(' '.join(f'{ch}:{sc:.0f}/{n:.0f}' for ch,sc,mx,n in vals))
PY"
