/bin/bash -lc "python - <<'PY'
import numpy as np
from sklearn.metrics import adjusted_rand_score,accuracy_score
A=np.load('cell_site_counts_full.npz')['counts']; R=A[:,:,0].T.astype(float); Alt=A[:,:,1].T.astype(float); p=np.array([.03,.5,.97]); C=R.shape[0]
for support in [5,10,20,30]:
 print('support',support)
 for K in [2,3,4,5,6]:
  fn=f'em_full{support}_independent_K{K}.npz'; z=np.load(fn); L=z['labels']; ix=z['site_indices']; r=R[:,ix]; a=Alt[:,ix]; ar=[]; margins=[]
  for fold in range(5):
   rng=np.random.default_rng(9000+support*11+K*3+fold); tr=np.zeros(C,bool)
   for k in range(K):
    ii=np.where(L==k)[0]; rng.shuffle(ii); tr[ii[:int(.8*len(ii))]]=1
   te=~tr; g=np.zeros((K,len(ix)),int)
   for k in range(K):
    rr=r[tr&(L==k)].sum(0); aa=a[tr&(L==k)].sum(0); g[k]=np.argmax(rr[:,None]*np.log1p(-p)[None,:]+aa[:,None]*np.log(p)[None,:],1)
   ll=np.zeros((C,K))
   for k in range(K):
    pp=p[g[k]]; ll[:,k]=r@np.log1p(-pp)+a@np.log(pp)
   pred=ll[te].argmax(1); ar.append(adjusted_rand_score(L[te],pred)); margins.append(np.mean(np.sort(ll[te],1)[:,-1]-np.sort(ll[te],1)[:,-2]))
  print(K,'ARI',round(float(np.mean(ar)),3),'margin',round(float(np.mean(margins)),2))
PY"
