/bin/bash -lc 'mkdir -p vireo_models
for donor_n in $(seq 2 12); do
  PYTHONPATH=workspace_pkgs workspace_pkgs/bin/vireo -c cellsnp_common -N "$donor_n" -o "vireo_models/N${donor_n}" -M 10 -p 1 --noPlot --noDoublet --randSeed=20260805 > "vireo_models/N${donor_n}.log" 2>&1 &

for donor_n in $(seq 2 12); do

  grep -E '"'lower bound ranges|donor size before' \"vireo_models/N"'${donor_n}.log" | tr '"'\\n' '\\t'
  tail -n 8 \"vireo_models/N"'${donor_n}.log" | tr '"'\\n' ' '
  printf '\\n'
done"
