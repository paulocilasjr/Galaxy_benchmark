/bin/bash -lc "python - <<'PY'
import pysam, collections
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
cb_bc=collections.defaultdict(collections.Counter); n=0
for r in b.fetch(until_eof=True):
    cb=r.get_tag('CB'); bc=r.get_tag('BC')
    cb_bc[cb][bc]+=1; n+=1
print('cells',len(cb_bc),'reads',n)
# print first 5 cb distributions and totals
for cb,d in list(cb_bc.items())[:10]: print(cb, dict(d), sum(d.values()))
# summary of per-cell per-bc counts
mins=collections.defaultdict(lambda:10**9); maxs=collections.defaultdict(lambda:0)
for cb,d in cb_bc.items():
    for bc,ct in d.items():
        mins[bc]=min(mins[bc],ct); maxs[bc]=max(maxs[bc],ct)
print('perBC min/max')
for bc in sorted(mins): print(bc,mins[bc],maxs[bc])
PY"
