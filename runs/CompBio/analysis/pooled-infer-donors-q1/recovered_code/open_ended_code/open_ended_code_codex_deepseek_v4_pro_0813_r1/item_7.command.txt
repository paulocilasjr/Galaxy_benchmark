/bin/bash -lc "python3 - <<'PY'
import pysam, collections, sys
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb',check_sq=False)
bc=collections.Counter(); cell_bc=collections.defaultdict(set); bc_cells=collections.defaultdict(set); cell_bcs=collections.Counter()
n=0
for r in b:
    n+=1
    bcv=r.get_tag('BC')
    cbv=r.get_tag('CB')
    bc[bcv]+=1
    cell_bc[cbv].add(bcv)
    bc_cells[bcv].add(cbv)
    cell_bcs[cbv]+=1
print('total reads',n,'distinct BC',len(bc),'distinct CB',len(cell_bcs))
for k,v in bc.most_common(50):
    print(k,v,'cells',len(bc_cells[k]))
print('cells with >1 BC:')
cnt=0
for cbv,s in cell_bc.items():
    if len(s)>1:
        cnt+=1
        if cnt<=20: print(cbv,s)
print('count cells >1 BC',cnt)
# BC per cell sequence sets sizes
sizes=collections.Counter(len(s) for s in cell_bc.values())
print('BC set sizes per cell',sizes)
# CB read counts distribution
print('CB readcount minmax', min(cell_bcs.values()), max(cell_bcs.values()))
PY"
