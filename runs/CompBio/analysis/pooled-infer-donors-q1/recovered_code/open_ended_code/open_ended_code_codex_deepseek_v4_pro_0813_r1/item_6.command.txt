/bin/bash -lc "python3 - <<'PY'
import pysam, collections
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb',check_sq=False)
cb=collections.Counter(); cr=collections.Counter(); names=collections.Counter(); readnames_prefix=collections.Counter()
n=0
for r in b:
    n+=1
    if n<=10000000:
      try: cb[r.get_tag('CB')]+=1
      except: pass
      try: cr[r.get_tag('CR')]+=1
      except: pass
      names[r.query_name.split()[0].split(':')[0]]+=1
    if n%1000000==0: print('n',n,file=__import__('sys').stderr)
print('total scanned',n)
print('unique CB first 10M',len(cb))
print('top CB',cb.most_common(10))
print('unique CR',len(cr))
print('run prefixes',names.most_common(20))
PY"
