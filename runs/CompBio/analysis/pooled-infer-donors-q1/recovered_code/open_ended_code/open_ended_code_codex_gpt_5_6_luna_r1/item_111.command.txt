/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.metrics import adjusted_rand_score, normalized_mutual_info_score
ks=range(2,13)
lab={}
for k in ks:
 x=pd.read_csv(f'vireo_{k}/donor_ids.tsv',sep='\\t')
 y=x.donor_id.astype(str).to_numpy()
 # include only assigned singlets; categories donor0...
 lab[k]=y
 print(k,dict(zip(*np.unique(y,return_counts=True))))
print('ARI same cells (all non-unassigned pairwise)')
for k in ks:
 print(k, [round(adjusted_rand_score(lab[k],lab[j]),3) for j in ks])
print('NMI')
for k in ks:
 print(k, [round(normalized_mutual_info_score(lab[k],lab[j]),3) for j in ks])
# contingency K10 x K8 on cells assigned both
for a,b in [(6,8),(8,10),(10,11),(10,12),(6,10)]:
 m=np.char.startswith(lab[a],'donor') & np.char.startswith(lab[b],'donor')
 tab=pd.crosstab(lab[a][m],lab[b][m])
 print('\\n',a,b,'n',m.sum()); print(tab.to_string())
PY"
