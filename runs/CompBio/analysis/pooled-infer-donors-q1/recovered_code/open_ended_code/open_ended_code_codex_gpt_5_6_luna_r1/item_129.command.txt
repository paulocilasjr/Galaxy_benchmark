/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.metrics import adjusted_rand_score
for a,b in [('vknown_k4','vknown_k4_s22')]:
 x=pd.read_csv(a+'/donor_ids.tsv',sep='\\t').donor_id.to_numpy(); y=pd.read_csv(b+'/donor_ids.tsv',sep='\\t').donor_id.to_numpy()
 print('all ARI',adjusted_rand_score(x,y))
 m=np.array([str(z).startswith('donor') for z in x])&np.array([str(z).startswith('donor') for z in y])
 print('singlet ARI',adjusted_rand_score(x[m],y[m]),'n',m.sum())
 print(pd.crosstab(x[m],y[m]).to_string())
PY"
