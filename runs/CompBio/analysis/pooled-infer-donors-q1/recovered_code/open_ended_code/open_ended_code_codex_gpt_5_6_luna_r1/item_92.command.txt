/bin/bash -lc "python3 - <<'PY'
from scipy.io import mmread
import numpy as np, time
for tag in ['AD','DP','OTH']:
 t=time.time(); m=mmread(f'cellsnp/cellSNP.tag.{tag}.mtx').tocsc()
 print(tag,m.shape,m.nnz,m.dtype,'load',round(time.time()-t,1),'data quant',np.quantile(m.data,[0,.1,.5,.9,.99,1]) if m.nnz else None)
ad=mmread('cellsnp/cellSNP.tag.AD.mtx').tocsr()
dp=mmread('cellsnp/cellSNP.tag.DP.mtx').tocsr()
# row stats in chunks through sparse CSR
alt_cells=np.diff(ad.indptr); cov_cells=np.diff(dp.indptr)
with np.errstate(divide='ignore',invalid='ignore'):
 totals=np.asarray(dp.sum(axis=1)).ravel()
 alts=np.asarray(ad.sum(axis=1)).ravel()
 af=alts/totals
print('rows',len(af),'DP nnz',dp.nnz,'AD nnz',ad.nnz)
print('total DP quant',np.quantile(totals,[0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]))
print('total AD quant',np.quantile(alts,[0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]))
print('AF quant finite',np.quantile(af[np.isfinite(af)],[0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]))
for thr in [0.005,.01,.02,.03,.05,.08,.1,.15,.2,.3,.4]:
 print('af>=',thr,'n',np.sum(af>=thr),'sumDP>=20',np.sum((af>=thr)&(totals>=20)),'cellsALT>=3 med',np.median(alt_cells[af>=thr]) if np.any(af>=thr) else 0)
for n in [1,2,3,5,10,20,50,100]: print('alt_cells>=',n,np.sum(alt_cells>=n),'dp_cells>=',n,np.sum(cov_cells>=n))
PY"
