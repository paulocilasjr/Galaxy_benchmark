/bin/bash -lc "python3 - <<'PY'
import pysam, collections, time, numpy as np
p='pooled.infer.donors.q1.bam'; t=time.time()
per=collections.defaultdict(collections.Counter)
n=0
with pysam.AlignmentFile(p,'rb') as bam:
 for a in bam.fetch(until_eof=True):
  n+=1
  cb=a.get_tag('CB') if a.has_tag('CB') else 'NO_CB'
  pref=a.query_name.split(':',1)[0]
  per[cb][pref]+=1
print('reads',n,'sec',round(time.time()-t,1),'prefixes',[(x,len(y),sum(y.values())) for x,y in per.items() if x=='NO_CB'])
prefs=sorted({p for c in per.values() for p in c})
print('all prefixes',prefs)
rows=[]
for cb,c in per.items():
 vals=[c.get(x,0) for x in prefs]; total=sum(vals); rows.append((cb,total,*vals))
print('cells',len(rows),'counts by prefix tuple',collections.Counter(tuple(r[2:]) for r in rows).most_common(20))
for pr in prefs:
 fr=np.array([r[rows[0][2:].index(r[2:]) if False else 0] for r in []])
print('first rows',rows[:20])
for pref in prefs:
 x=np.array([r[2+prefs.index(pref)]/r[1] for r in rows if r[1]])
 print(pref,'frac quant',np.quantile(x,[0,.01,.1,.5,.9,.99,1]),'near0',sum(x<.01),'near1',sum(x>.99))
PY"
