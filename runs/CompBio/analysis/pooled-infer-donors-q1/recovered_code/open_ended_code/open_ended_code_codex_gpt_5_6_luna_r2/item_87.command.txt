/bin/bash -lc "python - <<'PY'
import gzip, glob, re
import numpy as np
for n in range(2,9):
 path=f'vireo_50_N{n}/GT_donors.vireo.vcf.gz'
 rows=[]; names=None
 with gzip.open(path,'rt') as fh:
  for line in fh:
   if line.startswith('#CHROM'):
    f=line.rstrip().split('\\t'); names=f[9:]
   elif line.startswith('#'): continue
   else:
    f=line.rstrip().split('\\t'); vals=[]
    for x in f[9:]:
     gt=x.split(':')[0]
     if gt in ('0/0','0|0'): vals.append(0.)
     elif gt in ('1/1','1|1'): vals.append(1.)
     elif gt in ('0/1','1/0','0|1','1|0'): vals.append(.5)
     else: vals.append(np.nan)
    rows.append(vals)
 G=np.array(rows)
 # genotype distances: fraction of loci differing, treating hom/hom 1, het vs hom .5
 D=np.nanmean(np.abs(G[:,:,None]-G[:,None,:]),axis=0)
 print('\\nN',n,'sites',G.shape[0])
 print('sizes',open(f'vireo_50_N{n}/summary.tsv').read().strip().splitlines()[-1])
 print('dist\\n',np.round(D,3))
PY"
