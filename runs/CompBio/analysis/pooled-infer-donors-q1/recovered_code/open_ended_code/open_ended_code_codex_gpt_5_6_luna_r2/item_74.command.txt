/bin/bash -lc "python - <<'PY'
import numpy as np
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score, adjusted_rand_score
from sklearn.decomposition import PCA
z=np.load('allele_counts.npz'); keys=z['keys']; dp=z['dp'].astype(float); ad=z['ad'].astype(float); sites=z['sites']; ncell=len(z['barcodes']); rows=keys>>16; cells=keys&65535; ns=len(sites)
sp=np.bincount(rows,minlength=ns); sd=np.bincount(rows,weights=dp,minlength=ns); sa=np.bincount(rows,weights=ad,minlength=ns); f=sa/np.maximum(sd,1)
for mincells in [20,30,50,100,200]:
 sel=(sp>=mincells)&(sd>=50)&(f>=0.05)&(f<=0.5)
 # remove candidate loci with > 70% of cells? keep
 ii=np.flatnonzero(sel); X=np.zeros((ncell,len(ii)),dtype='float32')
 mp={r:j for j,r in enumerate(ii)}
 # centered, quality-weighted signed allele evidence
 for k,r in enumerate(rows):
  j=mp.get(r)
  if j is not None:
   ff=f[r]; X[cells[k],j]=(ad[k]-dp[k]*ff)/np.sqrt(dp[k]+1)
 # use PCA, standardized? 
 p=PCA(n_components=min(20,X.shape[1],ncell),random_state=1).fit_transform(X)
 print('\\nmincells',mincells,'sites',len(ii),'explained',PCA(n_components=min(5,X.shape[1],ncell),random_state=1).fit(X).explained_variance_ratio_[:5])
 for k in range(2,9):
  lab=KMeans(n_clusters=k,n_init=20,random_state=1).fit_predict(p[:,:min(10,p.shape[1])])
  sil=silhouette_score(p[:,:min(10,p.shape[1])],lab)
  print(k,round(sil,4),np.bincount(lab).tolist())
PY"
