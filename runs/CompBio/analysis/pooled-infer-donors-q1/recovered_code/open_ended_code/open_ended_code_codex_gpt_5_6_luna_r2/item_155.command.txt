/bin/bash -lc "bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\n' common_candidate_snps.vcf.gz > common_candidate_snps.tsv
python - <<'PY'
import numpy as np
z=np.load('allele_counts.npz');sites=z['sites'];known={}
for line in open('common_candidate_snps.tsv'):
 c,p,r,a=line.split();known[(c,int(p))]=(r,a)
for mode in ['set','exact']:
 ok=[];flip=[];bad=[]
 for i,(c,p,r,a) in enumerate(sites):
  x=known.get((c,int(p)))
  if x and set(x)==set([r,a]):
   ok.append(i)
   if a==x[1]:flip.append(False)
   else:flip.append(True)
  elif x:bad.append((i,c,p,r,a,x))
 print(mode,'knownpos',sum((c,int(p)) in known for c,p,_,_ in sites),'allele_match',len(ok),'flip',sum(flip),'mismatch',len(bad))
 if mode=='set':
  print('examples mismatch',bad[:10])
PY"
