/bin/bash -lc "python - <<'PY'
import numpy as np
z=np.load('allele_counts.npz'); keys=z['keys'];dp=z['dp'].astype(float);ad=z['ad'].astype(float);sites=z['sites'];rows=keys>>16
sp=np.bincount(rows,minlength=len(sites)); sd=np.bincount(rows,weights=dp,minlength=len(sites)); sa=np.bincount(rows,weights=ad,minlength=len(sites));f=sa/np.maximum(sd,1)
for thr in [20,30,50,100]:
 keep=np.flatnonzero((sp>=thr)&(sd>=50)&(f>=.05)&(f<=.5))
 print('thr',thr,'orig',len(keep))
 for win in [100,1000,10000,100000]:
  sel=[];last={}
  for r in keep:
   c,p=sites[r,0],int(sites[r,1])
   if c not in last or p-last[c]>=win:
    sel.append(r);last[c]=p
  print(win,len(sel))
PY"
