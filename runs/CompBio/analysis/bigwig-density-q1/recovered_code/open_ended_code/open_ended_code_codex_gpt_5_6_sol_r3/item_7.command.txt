/bin/bash -lc "python3 -u - <<'PY'
import pyBigWig, multiprocessing as mp
PATH='ENCFF822FDB.bigWig'

def calc(c):
    chrom=f'chr{c}'
    with pyBigWig.open(PATH) as bw:
        size=bw.chroms(chrom)
        n=size//1_000_000
        vals=bw.stats(chrom,0,n*1_000_000,nBins=n,type='sum',exact=True)
    rows=[(float(v or 0.0),chrom,i*1_000_000,(i+1)*1_000_000) for i,v in enumerate(vals)]
    return c, rows

allrows=[]
with mp.Pool(4) as pool:
    for c,rows in pool.imap_unordered(calc, range(1,20)):
        allrows.extend(rows)
        print('finished',c,'bins',len(rows),flush=True)
allrows.sort(reverse=True)
print('TOP20')
for row in allrows[:20]: print(row)
PY"
