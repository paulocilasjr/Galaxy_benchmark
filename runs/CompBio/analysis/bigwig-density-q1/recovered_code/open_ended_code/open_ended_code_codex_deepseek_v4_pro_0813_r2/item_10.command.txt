/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
for chrom in ['chr1','chr2','chr10','chr19']:
 size=bw.chroms(chrom)
 for exact in [False, True]:
  try:
   print(chrom, exact, bw.stats(chrom,0,size,type='mean',exact=exact), bw.stats(chrom,0,size,type='max',exact=exact), bw.stats(chrom,0,size,type='sum',exact=exact))
  except Exception as e: print(chrom, exact, 'ERR', e)
bw.close()
PY"
