import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')

print(r)
# compare individual queries exact
for i in range(n):
 print(i, bw.stats(chrom,i*1000000,(i+1)*1000000,type='sum',exact=True))
bw.close()
PY"
