/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
M=1000000
methods=['exact_sum','default_sum','exact_mean_times_M','default_mean_times_M']
data={m:[] for m in methods}
for chrom in [f'chr{i}' for i in range(1,20)]:
    size=bw.chroms(chrom); n=size//M
    if not n: continue
    for i in range(n):
        st=i*M; en=st+M
        vals=[]
        es=bw.stats(chrom,st,en,type='sum',exact=True)[0]
        ds=bw.stats(chrom,st,en,type='sum',exact=False)[0]
        em=bw.stats(chrom,st,en,type='mean',exact=True)[0]
        dm=bw.stats(chrom,st,en,type='mean',exact=False)[0]
        rec=(chrom,st)
        data['exact_sum'].append((es if es is not None else 0.0,rec))
        data['default_sum'].append((ds if ds is not None else 0.0,rec))
        data['exact_mean_times_M'].append((em*M if em is not None else 0.0,rec))
        data['default_mean_times_M'].append((dm*M if dm is not None else 0.0,rec))
for m in methods:
    arr=sorted(data[m], key=lambda x:x[0], reverse=True)
    print('\\n',m)
    for rank in range(5):
        val,(chrom,st)=arr[rank]
        print(rank+1, f'{chrom}:{st}-{st+M}', val)
bw.close()
PY"
