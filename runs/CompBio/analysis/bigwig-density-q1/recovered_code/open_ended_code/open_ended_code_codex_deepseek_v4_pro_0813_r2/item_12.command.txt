/bin/bash -lc "python3 - <<'PY'
import pyBigWig, time, heapq
bw=pyBigWig.open('ENCFF822FDB.bigWig')
M=1_000_000
bins=[]
for chrom in [f'chr{i}' for i in range(1,20)]:
    size=bw.chroms(chrom)
    n=size//M
    if n==0: continue
    sums=bw.stats(chrom,0,n*M,type='sum',nBins=n,exact=True)
    for i,s in enumerate(sums):
        if s is None:
            s=0.0
        bins.append((s, chrom, i*M, (i+1)*M))
bins.sort(key=lambda x:x[0], reverse=True)
print('total full bins',len(bins))
for rank in range(min(10,len(bins))):
    s,chrom,st,en=bins[rank]
    print(rank+1, f'{chrom}:{st}-{en}', s)
# also maybe aggregate by chromosome
from collections import defaultdict
chrs=defaultdict(lambda:[0,0.0])
for s,chrom,st,en in bins:
    chrs[chrom][0]+=1; chrs[chrom][1]+=s
print('chr bin counts', {c:chrs[c][0] for c in chrs})
bw.close()
PY"
