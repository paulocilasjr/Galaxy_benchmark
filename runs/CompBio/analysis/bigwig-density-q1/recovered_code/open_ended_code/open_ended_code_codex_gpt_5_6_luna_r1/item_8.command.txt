/bin/bash -lc "python -u - <<'PY'
import pyBigWig
b=pyBigWig.open('ENCFF822FDB.bigWig')
for c,s in [('chr10',80_000_000),('chr11',69_000_000),('chr8',70_000_000),('chr7',45_000_000)]:
    total=0.0; n=0; first=[]
    for a,z,v in (b.intervals(c,s,s+1_000_000) or ()):
        ov=max(0,min(z,s+1_000_000)-max(a,s))
        total += ov*v
        n+=1
        if len(first)<3: first.append((a,z,v,ov))
    stats=b.stats(c,s,s+1_000_000,type='sum',exact=True)[0]
    print(c,s,'direct',repr(total),'stats',repr(stats),'entries',n,'first',first)
b.close()
PY"
