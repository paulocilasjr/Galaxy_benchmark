/bin/bash -lc "python -u - <<'PY'
import pyBigWig, time, math
p='ENCFF822FDB.bigWig'
b=pyBigWig.open(p)
rows=[]
for i in range(1,20):
    c=f'chr{i}'; n=b.chroms()[c]
    m=n//1_000_000
    print(c,n,m,flush=True)
    for k in range(m):
        s=k*1_000_000; e=s+1_000_000
        val=b.stats(c,s,e,type='sum',exact=True)[0]
        if val is None: val=0.0
        rows.append((val,c,s,e))
b.close()
rows.sort(reverse=True)
print('TOP')
for rank,r in enumerate(rows[:20],1): print(rank,repr(r[0]),r[1],r[2],r[3])
PY"
