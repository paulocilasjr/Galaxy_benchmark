/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
for chrom,start,end in [('chr10',80000000,81000000),('chr11',69000000,70000000),('chr8',70000000,71000000),('chr7',45000000,46000000)]:
    ints=bw.intervals(chrom,start,end)
    manual=sum(v*(min(e,end)-max(s,start)) for s,e,v in ints if min(e,end)>max(s,start))
    stats=bw.stats(chrom,start,end,type='sum',exact=True)[0]
    print(chrom,start, manual, stats, manual-stats, len(ints))
PY"
