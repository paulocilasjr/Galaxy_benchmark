/bin/bash -lc "python3 - <<'PY'
import pyBigWig, math, time
bw=pyBigWig.open('ENCFF822FDB.bigWig')
rank=[]
for i in range(1,20):
 c=f'chr{i}'; n=bw.chroms(c)//1_000_000; end=n*1_000_000
 t=time.time()
 vals=bw.stats(c,0,end,type='sum',nBins=n,exact=True)
 for j,v in enumerate(vals):
  rank.append((0.0 if v is None or math.isnan(v) else v,c,j*1_000_000,(j+1)*1_000_000))
 print(c,n,'done',round(time.time()-t,2),flush=True)
bw.close()
rank.sort(key=lambda x:x[0],reverse=True)
print('TOP20')
for k,x in enumerate(rank[:20],1): print(k,*x)
PY"
