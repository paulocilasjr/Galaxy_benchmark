/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
regions=[('chr10',80_000_000,81_000_000),('chr11',69_000_000,70_000_000),('chr8',70_000_000,71_000_000),('chr7',45_000_000,46_000_000),('chr8',69_000_000,70_000_000),('chr8',71_000_000,72_000_000)]
for c,s,e in regions:
 iv=bw.intervals(c,s,e) or ()
 direct=sum((min(b,e)-max(a,s))*v for a,b,v in iv if min(b,e)>max(a,s))
 exact=bw.stats(c,s,e,type='sum',exact=True)[0]
 mean=bw.stats(c,s,e,type='mean',exact=True)[0]
 covered=sum(min(b,e)-max(a,s) for a,b,v in iv if min(b,e)>max(a,s))
 print(f'{c}:{s}-{e}\\tintervals={len(iv)}\\tcovered={covered}\\tdirect={direct:.12f}\\texact={exact:.12f}\\tmean={mean:.12f}')
bw.close()
PY"
