/bin/bash -lc "python3 - <<'PY'
import pyBigWig, json
bw=pyBigWig.open('ENCFF822FDB.bigWig')
wanted={f'chr{i}' for i in range(1,20)}
results=[]
for c,l in bw.chroms().items():
    if c not in wanted: continue
    n=int(l//1000000)
    if n<=0: continue
    means=bw.stats(c,0,n*1000000,type='mean',nBins=n)
    print(c, 'len',l,'n',n,'means_type',type(means),'means_len',len(means) if means else None)
    if means is None:
        print('None means for',c)
        continue
    for i,m in enumerate(means):
        if m is None:
            print('None bin',c,i)
            continue
        total=m*1000000.0
        results.append((total,c,i*1000000,i*1000000+1000000))
bw.close()
print('total bins',len(results))
results_sorted=sorted(results,key=lambda x:-x[0])
for rank,(total,c,st,en) in enumerate(results_sorted[:20],1):
    print(rank, f'{c}:{st}-{en}', repr(total), 'mean', total/1e6)
print('last bins')
for rank,(total,c,st,en) in enumerate(results_sorted[-5:],len(results_sorted)-4):
    print(rank, f'{c}:{st}-{en}', repr(total))
with open('bin_totals.json','w') as f:
    json.dump([{'total':t,'chrom':c,'start':st,'end':en} for t,c,st,en in results_sorted], f)
PY"
