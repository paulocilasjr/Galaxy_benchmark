/bin/bash -lc "python3 - <<'PY'
import pyBigWig, json
bw=pyBigWig.open('ENCFF822FDB.bigWig')
wanted={f'chr{i}' for i in range(1,20)}
results=[]
for c,l in bw.chroms().items():
    if c not in wanted: continue
    n=int(l//1000000)
    if n<=0: continue
    sums=bw.stats(c,0,n*1000000,type='sum',exact=True,nBins=n)
    print(c,'len',l,'n',n,'sums_len',len(sums) if sums else None)
    if not sums: continue
    for i,total in enumerate(sums):
        if total is None:
            print('None bin',c,i)
            continue
        results.append((float(total),c,i*1000000,i*1000000+1000000))
bw.close()
print('total bins',len(results))
results_sorted=sorted(results,key=lambda x:-x[0])
for rank,(total,c,st,en) in enumerate(results_sorted[:25],1):
    print(rank, f'{c}:{st}-{en}', repr(total))
print('ties around 3?')
for i in range(1,6):
 print(i, results_sorted[i-1])
print('last bins')
for rank,(total,c,st,en) in enumerate(results_sorted[-5:],len(results_sorted)-4):
    print(rank, f'{c}:{st}-{en}', repr(total))
with open('bin_totals_exact.json','w') as f:
    json.dump([{'total':t,'chrom':c,'start':st,'end':en} for t,c,st,en in results_sorted], f)
PY"
