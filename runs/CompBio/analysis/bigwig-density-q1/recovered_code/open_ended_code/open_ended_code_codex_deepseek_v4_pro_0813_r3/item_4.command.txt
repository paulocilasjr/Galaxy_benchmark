/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
for c,st,en in [('chr1',0,1000000),('chr1',1000000,2000000),('chr19',60000000,61431566)]:
    its=bw.intervals(c,st,en)
    print(c,st,en,'n',len(its) if its else 0, 'firstlast', (its[:3] if its else None),(its[-3:] if its else None))
    if its:
      s=sum(v*(min(e,en)-max(s,st)) for s,e,v in its)
      print('sum intervals',s)
      print('mean',bw.stats(c,st,en,type='mean')[0])
bw.close()
PY"
