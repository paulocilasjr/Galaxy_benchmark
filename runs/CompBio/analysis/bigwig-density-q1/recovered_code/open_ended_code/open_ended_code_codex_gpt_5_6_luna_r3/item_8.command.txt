/bin/bash -lc "python - <<'PY'
import pyBigWig, math
bw=pyBigWig.open('ENCFF822FDB.bigWig')
checks=[('chr10',80_000_000,81_000_000),('chr11',69_000_000,70_000_000),('chr8',70_000_000,71_000_000),('chr7',45_000_000,46_000_000)]
for c,s,e in checks:
 stat=bw.stats(c,s,e,type='sum',exact=True)[0]
 iv=bw.intervals(c,s,e)
 manual=sum((min(x,e)-max(a,s))*v for a,b,v in iv for x in [a] if False) if False else None
 # intervals returned clipped to query in pyBigWig; use overlap explicitly either way
 manual=0.0; covered=0
 if iv:
  for a,b,v in iv:
   aa=max(a,s); bb=min(b,e)
   if bb>aa:
    manual+=(bb-aa)*v; covered += bb-aa
 print(c,s,e,'stats=',repr(stat),'manual=',repr(manual),'covered=',covered,'records=',len(iv) if iv else 0)
print('direct neighboring bins for chr8')
for s in [68_000_000,69_000_000,70_000_000,71_000_000,72_000_000]:
 print(s, repr(bw.stats('chr8',s,s+1_000_000,type='sum',exact=True)[0]))
PY"
