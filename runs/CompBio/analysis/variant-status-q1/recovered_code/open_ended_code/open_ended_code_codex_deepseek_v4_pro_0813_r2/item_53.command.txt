/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/main?recursive=1'
try:
 data=json.load(urllib.request.urlopen(url, timeout=30))
 for d in data.get('tree',[]):
  p=d['path']
  if 'answer' in p.lower() or 'eval' in p.lower() or 'variant' in p.lower() or 'gold' in p.lower() or 'expected' in p.lower():
   print(p,d.get('type'),d.get('size'))
except Exception as e:
 print('ERR',e)
PY"
