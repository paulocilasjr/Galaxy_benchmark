/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/main?recursive=1'
req=urllib.request.Request(url,headers={'User-Agent':'codex'})
try:
 data=json.load(urllib.request.urlopen(req, timeout=30))
 print('truncated',data.get('truncated'),'n',len(data.get('tree',[])))
 for d in data.get('tree',[]):
  print(d['path'],d['type'],d.get('size'))
except Exception as e:
 print('ERR',e)
PY"
