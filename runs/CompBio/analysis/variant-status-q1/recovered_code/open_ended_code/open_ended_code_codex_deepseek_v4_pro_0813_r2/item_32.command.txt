/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for track in ['knownGene','refGene','ncbiRefSeq','omimGene2']:
 url=f'https://api.genome.ucsc.edu/getData/track?genome=hg38;track={track};chrom=chrX;start=154397000;end=154400000'
 try:
  data=json.load(urllib.request.urlopen(url, timeout=30))
  print('TRACK',track,'keys',list(data.keys()))
  # print first few relevant entries
  items=data.get(track,[])
  print('n',len(items))
  for d in items[:10]:
   print(d)
 except Exception as e:
  print('ERR',track,e)
PY"
