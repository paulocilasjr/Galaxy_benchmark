/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://api.genome.ucsc.edu/getData/track?genome=hg38;track=dbSnp155;chrom=chrX;start=154398495;end=154398505'
data=json.load(urllib.request.urlopen(url, timeout=30))
for d in data.get('dbSnp155',[]):
    print(d.get('chromStart')+1, d.get('chromEnd'), d.get('name'), 'ref',d.get('ref'),'alts',d.get('alts'),'class',d.get('class'))
PY"
