import pysam,json,urllib.request
ref=json.load(urllib.request.urlopen('https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chrX;start=154398450;end=154398700'))['dna']
def rb(pos1): return ref[pos1-154398451]
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
for name,k in [('READ_144249',2),('READ_75703',2)]:
 for read in bam.fetch('chrX',154398450,154398700):
  if read.query_name==name:
   start=read.reference_start+1; refpos=start+k; qidx=k
   mism=[]
   for op,l in read.cigartuples:
    if op==0:
     for j in range(l):
      if qidx>=read.query_length: break
      if read.query_sequence[qidx]"'!=rb(refpos): mism.append((refpos,qidx,read.query_sequence[qidx],rb(refpos)))
      refpos+=1; qidx+=1
    elif op==1:qidx+=l
    elif op==2:refpos+=l
    elif op==3:break
    if qidx>=read.query_length: break
   print(name,'"'trim',k,'start',start,'mism_after',mism[:20])
PY"
