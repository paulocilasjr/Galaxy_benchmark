/bin/bash -lc "python3 - <<'PY'
import pysam, collections
ref=pysam.FastaFile('chrX.fa')
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
# Tally mismatch bases for first 5 query positions in first M block for chrX reads
pos_counts=[collections.Counter() for _ in range(5)]
n=0
for read in bam.fetch('chrX'):
    if read.cigartuples is None or not read.cigartuples: continue
    op,l=read.cigartuples[0]
    if op"'!=0: continue
    start0=read.reference_start
    qlen=read.query_length
    upto=min(5,l,qlen)
    seq=read.query_sequence[:upto]
    refseq=ref.fetch('"'chrX',start0,start0+upto).upper()
    for i,(q,r) in enumerate(zip(seq,refseq)):
        if q"'!=r:
            pos_counts[i][q]+=1
    n+=1
print('"'chrX reads',n)
for i,c in enumerate(pos_counts):
    print('qpos',i,c)
PY"
