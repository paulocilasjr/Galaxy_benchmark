/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter, defaultdict
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
pos0=154398500-1
c=Counter()
quals=defaultdict(list)
reads=[]
for pileupcolumn in bam.pileup('chrX', pos0, pos0+1, truncate=True, stepper='all', min_mapping_quality=0, min_base_quality=0):
    if pileupcolumn.pos "'!= pos0: continue
    print('"'col', pileupcolumn.pos+1, 'n', pileupcolumn.n, 'segments', len(pileupcolumn.pileups))
    for p in pileupcolumn.pileups:
        if p.is_del or p.is_refskip:
            key='DEL' if p.is_del else 'REF_SKIP'
            c[key]+=1
            quals[key].append(None)
            continue
        q=p.alignment.query_sequence[p.query_position] if p.query_position is not None and p.query_position < p.alignment.query_length else 'N'
        bq=p.alignment.query_qualities[p.query_position] if p.query_position is not None and p.query_position < p.alignment.query_length else None
        c[q]+=1
        quals[q].append(bq)
        reads.append((p.alignment.query_name, p.alignment.reference_start+1, p.alignment.cigarstring, p.alignment.flag, p.query_position, q, bq))
print(c)
for base, qlist in quals.items():
    print(base, len(qlist), sum(x for x in qlist if x is not None)/max(1,len([x for x in qlist if x is not None])), qlist)
for r in reads:
    print(r)
PY"
