/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
for base in ['C','T']:
    seqs=[]
    for read in bam.fetch('chrX',154398300,154398600):
        pairs=read.get_aligned_pairs(matches_only=False)
        qpos=None
        for q,r in pairs:
            if r==154398499:
                qpos=q; break
        if qpos is None: continue
        b=read.query_sequence[qpos]
        if b==base:
            seqs.append(read.query_sequence)
    print(base, len(seqs), 'unique exact', len(set(seqs)))
    for s,c in collections.Counter(seqs).most_common(10): print(' ',c,s[:60])
PY"
