/bin/bash -lc "python3 - <<'PY'
import urllib.request
base='https://huggingface.co/spaces/Genentech/compbiobench-evaluator-v1/resolve/main/'

  print(fn,r.status,len(data),data[:100])
 except Exception as e:
  print(fn,'ERR',e)
PY"
