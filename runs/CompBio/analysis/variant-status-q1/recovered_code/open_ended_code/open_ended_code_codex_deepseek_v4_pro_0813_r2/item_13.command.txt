/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
pos=154398500
rows=[]
for read in bam.fetch('chrX', pos-200, pos+200):
    # check covers pos
    pairs=read.get_aligned_pairs(matches_only=False)
    qpos=None
    for q,r in pairs:
        if r==pos-1:
            qpos=q
            break
    if qpos is not None:
        base=read.query_sequence[qpos] if qpos is not None and qpos<read.query_length else '?'
        qual=read.query_qualities[qpos] if qpos is not None and qpos<read.query_length else None
        rows.append((read.reference_start+1, read.reference_end, read.cigarstring, read.query_name, read.flag, read.query_length, base, qual, read.get_tag('NM') if read.has_tag('NM') else read.get_tag('nM'), read.get_tag('AS') if read.has_tag('AS') else None))
print('count',len(rows))
for r in sorted(rows):
    print(r)
PY"
