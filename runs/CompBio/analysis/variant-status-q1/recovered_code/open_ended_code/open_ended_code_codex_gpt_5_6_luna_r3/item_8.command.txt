/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam='variant.status.q1.bam'; chrom='chrX'; p0=154398470; p1=154398520
b=pysam.AlignmentFile(bam,'rb')
rows=[]
for rec in b.fetch(until_eof=True):
    if rec.is_unmapped or rec.reference_name"'!=chrom: continue
    if rec.reference_end<=p0 or rec.reference_start>=p1: continue
    # map reference positions to bases in aligned orientation; query_sequence is alignment orientation
    mp={r:(rec.query_sequence[q] if q is not None else '"'-') for q,r in rec.get_aligned_pairs(matches_only=False) if r is not None}
    rows.append((rec,mp))
print('reads intersecting window',len(rows))
print('pos\\t'+'\\t'.join(str(x) for x in range(p0,p1)))
for rec,mp in rows:
    print(rec.query_name, rec.flag, rec.reference_start+1, rec.reference_end, rec.cigarstring, ''.join(mp.get(x,' ') for x in range(p0,p1)))
print('\\ncounts')
for x in range(p0,p1):
 c=collections.Counter(mp.get(x,'-') for rec,mp in rows if x in mp)
 if c: print(x, c)
print('\\ntags first')
for rec,mp in rows[:20]: print(rec.query_name, rec.get_tags())
PY"
