/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam='variant.status.q1.bam'; chrom='chrX'; pos1=154_398_500; pos0=pos1-1
b=pysam.AlignmentFile(bam,'rb')
reads=[]; total=0; chromn=0
for rec in b.fetch(until_eof=True):
    total+=1
    if rec.reference_name "'!= chrom: continue
    chromn+=1
    if rec.is_unmapped: continue
    if rec.reference_start <= pos0 < rec.reference_end:
        qpos=None; base=None; qual=None
        for q,r in rec.get_aligned_pairs(matches_only=False, with_seq=False):
            if r==pos0:
                qpos=q
                if q is not None:
                    base=rec.query_sequence[q]
                    qual=rec.query_qualities[q] if rec.query_qualities else None
                break
        reads.append((rec, qpos, base, qual))
print('"'total records',total,'chrX records',chromn,'overlap',len(reads))
print('counts all bases',collections.Counter(x[2] if x[2] is not None else 'DEL' for x in reads))
print('mapq',collections.Counter(x[0].mapping_quality for x in reads))
print('flags',collections.Counter(x[0].flag for x in reads))
for rec,qpos,base,qual in reads:
    print('\\t'.join(map(str,[rec.query_name, rec.reference_start+1, rec.reference_end, rec.cigarstring, rec.flag, rec.mapping_quality, qpos, base, qual, rec.query_sequence[max(0,(qpos or 0)-10): (qpos or 0)+11] if qpos is not None else ''])))
PY"
