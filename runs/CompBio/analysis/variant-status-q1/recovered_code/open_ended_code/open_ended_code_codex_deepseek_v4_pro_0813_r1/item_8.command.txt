/bin/bash -c "python - <<'PY'
import pysam
bam='variant.status.q1.bam'
pos=154398500
af=pysam.AlignmentFile(bam,'rb')
counts={}
rows=[]
for r in af.fetch('chrX', pos-1, pos):
    pairs=r.get_aligned_pairs(matches_only=False, with_seq=True)
    for qpos,rpos,base in pairs:
        if rpos==pos-1 and base is not None:
            counts[base.upper()]=counts.get(base.upper(),0)+1
            rows.append((r.query_name, r.flag, r.reference_start+1, r.cigarstring, qpos+1, rpos+1, base, r.mapping_quality, r.query_qualities[qpos] if qpos is not None and qpos < len(r.query_qualities) else None))
            break
print('counts', counts)
print('total rows', len(rows))
for row in rows:
    print('\\t'.join(map(str,row)))
af.close()
PY"
