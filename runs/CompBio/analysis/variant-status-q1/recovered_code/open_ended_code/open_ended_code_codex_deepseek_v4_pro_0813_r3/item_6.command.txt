/bin/bash -lc "python3 - <<'PY'
import pysam
fn='variant.status.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
pos=154398500
counts={}
for pc in b.pileup('chrX', pos-1, pos, truncate=True, stepper='nofilter'):
    if pc.reference_pos == pos-1:
        for pr in pc.pileups:
            if pr.is_del or pr.is_refskip:
                continue
            q=pr.alignment.query_sequence[pr.query_position]
            counts[q]=counts.get(q,0)+1
print(counts)
print('sum',sum(counts.values()))
PY"
