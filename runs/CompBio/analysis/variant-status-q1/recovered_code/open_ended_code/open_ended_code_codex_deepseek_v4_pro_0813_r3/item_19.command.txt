/bin/bash -lc "python3 - <<'PY'
import pysam
b=pysam.AlignmentFile('variant.status.q1.bam','rb')
for pos in [154398498,154398499]:
    for pc in b.pileup('chrX',pos-1,pos,truncate=True,stepper='nofilter'):
        if pc.reference_pos==pos-1:
            for pr in pc.pileups:
                a=pr.alignment
                if pr.is_del or pr.is_refskip: continue
                q=pr.query_position
                base=a.query_sequence[q]
                if (pos==154398498 and base=='T') or (pos==154398499 and base=='T'):
                    print(pos,a.query_name,base,a.query_qualities[q],a.reference_start+1,a.cigarstring,q)
PY"
