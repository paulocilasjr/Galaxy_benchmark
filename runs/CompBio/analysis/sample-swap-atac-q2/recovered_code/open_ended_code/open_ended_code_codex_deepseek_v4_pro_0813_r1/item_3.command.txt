/bin/bash -lc "python3 - <<'PY'
import gzip
fn='sample.swap.atac.q2.tsv.gz'
with gzip.open(fn,'rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    n=0
    sums=[0]*(len(header)-3)
    maxes=[0]*(len(header)-3)
    nonzero=[0]*(len(header)-3)
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        n+=1
        for i,v in enumerate(p[3:]):
            x=int(v)
            sums[i]+=x
            if x>maxes[i]: maxes[i]=x
            if x: nonzero[i]+=1
print('header',header)
print('rows',n)
for h,s,m,nz in zip(header[3:],sums,maxes,nonzero):
    print(h,s,m,nz)
PY"
