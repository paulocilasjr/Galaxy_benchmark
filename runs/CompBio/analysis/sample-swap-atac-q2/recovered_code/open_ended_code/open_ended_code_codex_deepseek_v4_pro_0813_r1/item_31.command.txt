/bin/bash -lc "python3 - <<'PY'
import gzip
fn='sample.swap.atac.q2.tsv.gz'
from collections import Counter
chr_counts=Counter(); last={}
with gzip.open(fn,'rt') as f:
    f.readline()
    for line in f:
        p=line.rstrip().split('\\t')
        c=p[0]; chr_counts[c]+=1; last[c]=(p[1],p[2])
print(chr_counts)
for c,(s,e) in last.items(): print(c,s,e)
PY"
