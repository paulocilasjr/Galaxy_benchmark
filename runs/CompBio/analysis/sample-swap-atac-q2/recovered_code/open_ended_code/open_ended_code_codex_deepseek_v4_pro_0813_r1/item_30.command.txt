/bin/bash -lc "python3 - <<'PY'
import requests, json, os, time
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE196nnn/GSE196235/suppl/GSE196235_RAW.tar'
files=json.load(open('tar_index.json'))
target=[x for x in files if x[0]=='GSM5866073_AIIamacrine_frags.tsv.gz'][0]
name,off,size,typ=target
out='ref_AIIamacrine_frags.tsv.gz'
print(target, out)
# stream chunks with requests to file, maybe range entire file in one response (size 105MB)
headers={'Range':f'bytes={off}-{off+size-1}'}
start=time.time()
with requests.get(url,headers=headers,stream=True,timeout=120) as r:
    r.raise_for_status()
    with open(out,'wb') as f:
        got=0
        for chunk in r.iter_content(chunk_size=1<<20):
            f.write(chunk); got+=len(chunk)
print('got',got,'time',time.time()-start,'size',os.path.getsize(out))
PY
gunzip -c ref_AIIamacrine_frags.tsv.gz | head -5"
