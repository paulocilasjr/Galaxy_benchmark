/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np
names=[]; coords=[]; vals=[]
with gzip.open('sample.swap.atac.q2.tsv.gz','rt') as f:
 names=f.readline().rstrip().split('\\t')[3:]
 for line in f:
  a=line.split('\\t'); coords.append((a[0],int(a[1]),int(a[2]))); vals.append([int(x) for x in a[3:]])
x=np.asarray(vals,dtype=float); lib=x.sum(0); y=x/lib*1e6
print('library',dict(zip(names,lib.astype(int))))
# log fold vs mean of other normalized profiles and specificity
for i,n in enumerate(names):
 other=np.delete(y,i,1); score=np.log2((y[:,i]+0.01)/(np.median(other,axis=1)+0.01))
 # top by score with meaningful signal (top 30 among values > perhaps 5 cpm)
 ix=np.argsort(score)[::-1]
 print('\\nTOP',n)
 k=0
 for j in ix:
  if y[j,i]>1 and score[j]>1:
   print(coords[j], 'raw',int(x[j,i]), 'cpm',round(y[j,i],1), 'score',round(score[j],2), 'all',list(x[j].astype(int)))
   k+=1
   if k>=20: break
PY"
