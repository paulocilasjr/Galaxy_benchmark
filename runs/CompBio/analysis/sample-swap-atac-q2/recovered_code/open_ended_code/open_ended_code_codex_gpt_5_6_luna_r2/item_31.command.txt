/bin/bash -lc "python - <<'PY'
import json,subprocess
markers='GJD2 NFIA DNER GAD1 GAD2 SLC6A9 TFAP2A TFAP2B PROX1 ONECUT1 GFAP C1QA RLBP1 ARR3 PDE6A NEFL RBPMS'.split()
cmd=['curl','-L','--max-time','60','-sS','-X','POST','-H','Content-Type: application/json','-H','Accept: application/json','https://rest.ensembl.org/lookup/symbol/homo_sapiens','--data',json.dumps({'symbols':markers})]
d=json.loads(subprocess.check_output(cmd,text=True))
for g in markers:
 x=d[g]; print(f'{g}\\t{x[\"seq_region_name\"]}\\t{x[\"start\"]}\\t{x[\"end\"]}')
PY"
