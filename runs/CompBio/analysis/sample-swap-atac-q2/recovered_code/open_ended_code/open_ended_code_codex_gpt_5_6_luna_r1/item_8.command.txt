/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np
p='sample.swap.atac.q2.tsv.gz'
with gzip.open(p,'rt') as f:
 rd=csv.reader(f,delimiter='\\t'); h=next(rd); names=h[3:]; coords=[]; dat=[]
 for row in rd:
  coords.append(tuple(row[:3])); dat.append([int(x) for x in row[3:]])
a=np.asarray(dat,dtype=float); totals=a.sum(0); lib=a/totals*1e6
# log fold vs all other columns, with pseudocount normalized counts
for i,n in enumerate(names):
 others=np.delete(lib,i,axis=1)
 # difference log normalized with modest pseudocount, require count support
 score=np.log2((lib[:,i]+0.1)/(others.mean(1)+0.1))
 # rank by score while requiring own >= 2 counts and top 500 bins
 idx=np.where(a[:,i]>=2)[0]
 ix=idx[np.argsort(score[idx])[-30:][::-1]]
 print('\\n',n,'top specific bins')
 for j in ix[:20]: print(coords[j][0]+':'+coords[j][1]+'-'+coords[j][2], 'count',int(a[j,i]), 'norm',round(lib[j,i],2),'lfc',round(score[j],2))
PY"
