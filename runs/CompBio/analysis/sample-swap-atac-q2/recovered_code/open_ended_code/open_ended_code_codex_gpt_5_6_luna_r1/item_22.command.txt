import gzip,csv,numpy as np
p='sample.swap.atac.q2.tsv.gz'
with gzip.open(p,'rt') as f:
 rd=csv.reader(f,delimiter='\\t'); h=next(rd); names=h[3:]; coords=[]; vals=[]
 for row in rd: coords.append((row[0],int(row[1]),int(row[2]))); vals.append([int(x) for x in row[3:]])
a=np.asarray(vals,dtype=float); lib=a/a.sum(0)*1e6
# gene: chrom,start,end,strand from GRCh38 Ensembl
G={
'GJD2':('15',34751032,34755003,-1),'SLC6A9':('1',43989353,44031767,-1),'GAD1':('2',170813213,170861151,1),'GAD2':('10',26216251,26304558,1),
'GFAP':('17',44903159,44916937,-1),'AQP4':('18',26851707,26865820,-1),'SLC1A3':('5',36596584,36688335,1),'RLBP1':('15',89209869,89222577,-1),'GLUL':('1',182378098,182392206,-1),
'ARR3':('X',70268305,70281840,1),'GNAT2':('1',109601407,109619929,-1),'PDE6C':('10',93612533,93666010,1),'RHO':('3',129528637,129535344,1),'PDE6A':('5',149857953,149945287,-1),
'ONECUT1':('15',52755053,52791078,-1),'LHX1':('17',36936784,36944612,1),'NEFL':('8',24950955,24956721,-1),'RBPMS':('8',30383636,30572261,1),
'C1QA':('1',22635057,22640201,1),'C1QB':('1',22652713,22661637,1),'AIF1':('6',31615214,31617025,1),'CX3CR1':('3',39263495,39281735,-1),
}
# indexes by chrom; promoter centered at TSS and gene +/-100kb
for g,(c,s,e,strand) in G.items():
 t=s if strand==1 else e
 idx=np.array([j for j,(cc,st,en) in enumerate(coords) if cc=='chr'+c and en>t-2000 and st<t+2000])
 ix2=np.array([j for j,(cc,st,en) in enumerate(coords) if cc=='chr'+c and en>s-50000 and st<e+50000])
 v=lib[idx].sum(0) if len(idx) else np.zeros(9)
 w=lib[ix2].sum(0) if len(ix2) else np.zeros(9)
 ranks=np.argsort(-v)
 print('\\n'+g+' promoter rows',len(idx),'gene window rows',len(ix2))
 print(' promoter '+' '.join(f'{names[i]}={v[i]:.1f}' for i in ranks))
 print(' window   '+' '.join(f'{names[i]}={w[i]:.1f}' for i in np.argsort(-w)))
PY"
