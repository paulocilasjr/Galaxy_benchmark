print('nonzero',dict(zip(names,(a>0).sum(0))))

# correlations
for transform,x in [('raw',a),('log',np.log1p(a)),('binary',(a>0).astype(float))]:
 c=np.corrcoef(x.T)
 print('\\n'+transform+' correlations')
 for i,n in enumerate(names):
  vals=[(c[i,j],names[j]) for j in range(len(names)) if j"'!=i]
  vals.sort(reverse=True)
  print(n, '"' '.join(f'{m}:{v:.3f}' for v,m in vals))
