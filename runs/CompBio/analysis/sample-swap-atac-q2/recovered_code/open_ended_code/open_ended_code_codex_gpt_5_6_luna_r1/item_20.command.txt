/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,time
G='GJD2 SLC6A9 GAD1 GAD2 GFAP AQP4 SLC1A3 VIM RLBP1 GLUL KCNJ10 ARR3 GNAT2 PDE6C OPN1SW OPN1MW OPN1LW GUCA1C RHO PDE6A PDE6B GNAT1 SAG NRL NR2E3 ONECUT1 LHX1 CALB1 NEFL RBPMS SNCG POU4F1 POU4F2 THY1 C1QA C1QB C1QC AIF1 CX3CR1 TYROBP CSF1R P2RY12 SPI1 TMEM119 APOE CST3 RAX NFIA SOX9 SLC6A9 GAD1 GAD2 TFAP2A TFAP2B MEIS1 MEIS2 PROX1'.split()
for g in G:
 u='https://rest.ensembl.org/lookup/symbol/homo_sapiens/'+g+'?expand=1'
 req=urllib.request.Request(u,headers={'Content-Type':'application/json','User-Agent':'codex'})
 try:
  x=json.load(urllib.request.urlopen(req,timeout=20))
  print('\\t'.join(map(str,[g,x.get('seq_region_name'),x.get('start'),x.get('end'),x.get('strand'),x.get('assembly_name')])))
 except Exception as e: print(g,'ERR',e)
 time.sleep(0.1)
PY"
