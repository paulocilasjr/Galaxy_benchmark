/bin/bash -lc "python3 -u - <<'PY'
import urllib.request,json
for g in 'GJD2 SLC6A9 GAD1 GAD2 GFAP AQP4 SLC1A3 RLBP1 GLUL ARR3 GNAT2 PDE6C RHO PDE6A ONECUT1 LHX1 NEFL RBPMS C1QA C1QB AIF1 CX3CR1'.split():
 u='https://rest.ensembl.org/lookup/symbol/homo_sapiens/'+g+'?content-type=application/json'
 req=urllib.request.Request(u,headers={'User-Agent':'codex'})
 try:
  x=json.load(urllib.request.urlopen(req,timeout=10))
  print(g,x.get('seq_region_name'),x.get('start'),x.get('end'),x.get('strand'),flush=True)
 except Exception as e: print(g,'ERR',repr(e),flush=True)
PY"
