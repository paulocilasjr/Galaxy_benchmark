/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np,time
path='sample.swap.atac.q2.tsv.gz'
cols=None; ours={}
with gzip.open(path,'rt') as f:
    r=csv.reader(f,delimiter='\\t'); header=next(r); cols=header[3:]
    for row in r:
        if len(row)<12: continue
        ours[(row[0],int(row[1]))]=np.array([float(x) for x in row[3:]],dtype=float)
idx=cols.index('AIIamacrine')
bins={}
with gzip.open('original_frags/AIIamacrine_frags.tsv.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<5: continue
        chrom=p[0]; s=int(p[1]); e=int(p[2]); c=int(p[4])
        b1=s//10000; b2=(e-1)//10000  # assuming end exclusive; if end inclusive, e//10000?
        for b in range(b1,b2+1):
            key=(chrom,b*10000)
            bins[key]=bins.get(key,0)+1  # per row overlap, not count column
print('sum overlap row',sum(bins.values()),'nbins',len(bins))
keys=set(ours)&set(bins)
o=np.array([bins[k] for k in keys],dtype=float); u=np.array([ours[k][idx] for k in keys],dtype=float)
print('common',len(keys),'corr',np.corrcoef(o,u)[0,1], 'sums',o.sum(),u.sum())
# check equal
print('exact matches',np.sum(o==u),'total',len(o))
print('max diff',np.max(np.abs(o-u)), 'mean diff',np.mean(u-o))
PY"
