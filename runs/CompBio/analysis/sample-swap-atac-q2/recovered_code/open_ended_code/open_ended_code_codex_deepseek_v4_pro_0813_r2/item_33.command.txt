/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np, time
# load our matrix
path='sample.swap.atac.q2.tsv.gz'
cols=None; ours={}
with gzip.open(path,'rt') as f:
    r=csv.reader(f,delimiter='\\t'); header=next(r); cols=header[3:]
    for row in r:
        if len(row)<12: continue
        ours[(row[0],int(row[1]))]=np.array([float(x) for x in row[3:]],dtype=float)
idx=cols.index('AIIamacrine')
our=ours.copy() # not needed
# compute original bins for AII
bins={}
start=time.time()
with gzip.open('original_frags/AIIamacrine_frags.tsv.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<5: continue
        chrom, s, e, bc, c = p[0], int(p[1]), int(p[2]), p[3], int(p[4])
        key=(chrom,(s//10000)*10000)
        bins[key]=bins.get(key,0)+1  # count each row
        #bins[key]=bins.get(key,0)+c
print('orig bins',len(bins),'sum rowcount',sum(bins.values()),'time',time.time()-start)
# compare only keys present in ours maybe
keys=set(ours)&set(bins)
o=np.array([bins[k] for k in keys],dtype=float)
u=np.array([ours[k][idx] for k in keys],dtype=float)
print('common',len(keys),'corr',np.corrcoef(o,u)[0,1])
print('sum ours common',u.sum(),'orig',o.sum())
# nonzero stats
print('ours nonzero',np.count_nonzero(u),'orig nonzero',np.count_nonzero(o))
# print first mismatches
for k in list(keys)[:10]:
    print(k,ours[k][idx],bins[k])
PY"
