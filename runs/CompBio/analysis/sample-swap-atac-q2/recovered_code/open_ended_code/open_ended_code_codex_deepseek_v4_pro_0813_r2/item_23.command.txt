/bin/bash -lc "python3 - <<'PY'
import requests
urls=[
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE196nnn/GSE196235/suppl/GSE196235_RAW.tar',
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE196nnn/GSE196235/suppl/',
]
for u in urls:
    r=requests.head(u,timeout=30,allow_redirects=True)
    print(u, r.status_code, r.headers.get('content-length'), r.headers.get('content-type'))
    if r.status_code==405:
        r=requests.get(u,timeout=30,stream=True)
        print(r.status_code, r.headers.get('content-length'))
        print(r.text[:2000])
PY"
