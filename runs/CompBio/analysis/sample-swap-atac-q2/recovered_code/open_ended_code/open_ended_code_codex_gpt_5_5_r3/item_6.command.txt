/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
fn='sample.swap.atac.q2.tsv.gz'
df=pd.read_csv(fn, sep='\\t', compression='gzip')
print(df.shape)
print(df.columns.tolist())
cols=df.columns[3:]
print(df[cols].sum().sort_values())
# correlations raw log1p cpm maybe
X=df[cols].to_numpy(dtype=float)
# filter nonzero rows
print('nonzero bins', (X.sum(axis=1)>0).sum())
for transform in ['raw','log1p','cpm_log1p','tfidf']:
    Y=X.copy()
    if transform=='log1p': Y=np.log1p(Y)
    elif transform=='cpm_log1p':
        Y=Y/Y.sum(axis=0)*1e6
        Y=np.log1p(Y)
    elif transform=='tfidf':
        # term frequency per col * idf by row (# columns nonzero)
        tf=Y/Y.sum(axis=0)
        nz=(Y>0).sum(axis=1)
        idf=np.log(1+Y.shape[1]/(1+nz))
        Y=tf*idf[:,None]
        Y=np.log1p(Y*1e6)
    c=np.corrcoef(Y.T)
    print('\\n',transform)
    print(pd.DataFrame(c,index=cols,columns=cols).round(3))
PY"
