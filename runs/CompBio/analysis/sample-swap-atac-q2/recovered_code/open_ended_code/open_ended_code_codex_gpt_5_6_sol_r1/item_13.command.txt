/bin/bash -lc "python -u - <<'PY'
import gzip,csv,numpy as np
T=np.array([11836803,7530916,48403523,37015795,31640818,8155368,94395242,21014655,413545691.],float)
rows=[]
with gzip.open('sample.swap.atac.q2.tsv.gz','rt') as f:
 r=csv.reader(f,delimiter='\\t'); H=next(r)[3:]
 for q in r:
  x=np.array(q[3:],float)
  if x.sum()>0: rows.append(np.log1p(x/T*1e6))
X=np.stack(rows)
C=np.corrcoef(X.T)
print('bins',X.shape)
print(''.join(f'{h[:7]:>9}' for h in ['']+H))
for h,row in zip(H,C): print(f'{h[:7]:>9}'+''.join(f'{v:9.3f}' for v in row))
PY"
