/bin/bash -lc "python3 - <<'PY'
import gzip,re,math
import pandas as pd
markers = {
'Rod':['RHO','NRL','GNAT1','PDE6A','PDE6B','SAG','CNGA1','RCVRN'],
'Cone':['OPN1LW','OPN1MW','OPN1SW','ARR3','GNAT2','PDE6C','CNGB3','GNGT2'],
'Mullerglia':['RLBP1','GLUL','SOX9','VIM','AQP4','CRYAB','CLU','SLC1A3'],
'Astrocyte':['GFAP','ALDH1L1','S100B','AQP4','SOX9','GJA1'],
'Microglia':['CX3CR1','P2RY12','AIF1','TYROBP','C1QA','C1QB','CSF1R','SPI1','PTPRC'],
'Retinalganglioncell':['RBPMS','SNCG','THY1','POU4F1','POU4F2','NEFL','NEFM'],
'Horizontal':['LHX1','ONECUT1','ONECUT2','PROX1','CALB1','LIM2'],
'GABAamacrine':['GAD1','GAD2','SLC6A1','SLC32A1','DLX1','DLX2','TFAP2A','TFAP2B'],
'AIIamacrine':['GRIK1','SLC1A6','PCP2','CACNA2D3','MEIS2','TFAP2A','TFAP2B']
}
need={g for v in markers.values() for g in v}
coords={}
pat=re.compile(r'gene_name \"(["'^"]+)"'"')
with gzip.open('gencode.v44.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        fields=line.rstrip('\\n').split('\\t')
        if len(fields)<9 or fields[2] "'!= '"'gene': continue
        m=pat.search(fields[8])
        if not m: continue
        name=m.group(1)
        if name in need and fields[0] in ['chr'+str(i) for i in range(1,23)]+['chrX','chrY']:
            coords[name]=(fields[0], int(fields[3])-1, int(fields[4]), fields[6])
print('coord found count', len(coords))
# load matrix
df=pd.read_csv('sample.swap.atac.q2.tsv.gz', sep='\\t')
cts=list(df.columns[3:])
tot=df[cts].sum()
# CPM per bin, log not for sums
for window in [10000,50000,100000,250000]:
    print('\\nWINDOW',window)
    scores=pd.DataFrame(0.0,index=markers.keys(),columns=cts)
    gene_details=[]
    for group,gs in markers.items():
        used=0
        for g in gs:
            if g not in coords: continue
            chrom,start,end,strand=coords[g]
            lo=max(0,start-window); hi=end+window
            m=(df['chrom'].eq(chrom) & (df['end']>lo) & (df['start']<hi))
            vals=df.loc[m,cts].sum()/tot*1e6
            # maybe log fold vs median? Use CPM raw across group average
            scores.loc[group]+=vals
            used+=1
        if used: scores.loc[group]/=used
    # zscore row by cell types
    z=scores.copy()
    for idx in z.index:
        vals=z.loc[idx].astype(float)
        z.loc[idx]=(vals-vals.mean())/(vals.std(ddof=0) if vals.std(ddof=0)>0 else 1)
    print('top per marker group (raw CPM avg):')
    for group in markers:
        vals=scores.loc[group].sort_values(ascending=False)
        print(group, 'top3', ', '.join(f'{k}:{v:.1f}' for k,v in vals.head(3).items()), ' expected_rank', list(vals.index).index(group)+1 if group in vals.index else None)
    print('\\nconfusion z matrix rows marker groups, columns labels:')
    print(z.round(2).to_string())
PY"
