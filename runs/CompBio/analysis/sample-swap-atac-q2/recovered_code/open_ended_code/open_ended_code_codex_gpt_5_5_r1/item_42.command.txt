/bin/bash -lc "python3 - <<'PY'
import gzip,re,pandas as pd,numpy as np
markers=['GRIK1','SLC1A6','PCP2','CACNA2D3','MEIS2','TFAP2A','TFAP2B','SLC6A9','GLRA2','GLRA3','GAD1','GAD2','SLC6A1','SLC32A1','DLX1','DLX2','PAX6','PROX1','LHX1','ONECUT1','ONECUT2','CALB1','CALB2','RELN']
need=set(markers)
pat=re.compile(r'gene_name \"(["'^"]+)"'"')
coords={}
with gzip.open('gencode.v44.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        flds=line.rstrip('\\n').split('\\t')
        if len(flds)<9 or flds[2]"'!='"'gene': continue
        m=pat.search(flds[8])
        if m and m.group(1) in need:
            coords[m.group(1)]=(flds[0], int(flds[3])-1, int(flds[4]), flds[6])
df=pd.read_csv('sample.swap.atac.q2.tsv.gz',sep='\\t')
cts=list(df.columns[3:]); tot=df[cts].sum()
for window in [10000,50000,100000]:
    print('\\nWINDOW',window)
    for g in markers:
        if g not in coords:
            print(g,'missing'); continue
        chrom,start,end,strand=coords[g]
        lo=max(0,start-window); hi=end+window
        mask=df['chrom'].eq(chrom)&(df['end']>lo)&(df['start']<hi)
        vals=(df.loc[mask,cts].sum()/tot*1e6).sort_values(ascending=False)
        print(g, chrom,start,end, 'top', ', '.join(f'{k}:{v:.1f}' for k,v in vals.head(5).items()))
PY"
