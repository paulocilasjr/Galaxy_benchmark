/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
df=pd.read_csv('data.tsv',sep='\\t')
cols=list(df.columns[3:])
gt=pd.read_csv('gene_tss.tsv',sep='\\t')
bins=df[['chrom','start','end']].copy()
genes=['RHO','NRL','NR2E3','PDE6B','GNAT1','SAG','ARR3','PDE6C','PDE6H','GNAT2','OPN1SW','OPN1MW','OPN1LW','RLBP1','GLUL','SLC1A3','CLU','SOX9','DIO3','VIM','GFAP','AQP4','S100B','ALDH1L1','SLC1A2','GJA1','P2RY12','CSF1R','CX3CR1','TYROBP','ITGAM','TMEM119','SPI1','AIF1','C1QA','RBPMS','POU4F1','POU4F2','SLC17A6','THY1','NEFL','NEFM','ISL1','SNCG','GAD1','GAD2','SLC6A1','SLC32A1','CALB2','GJD2','SLC12A7','DAB1','NEUROD1','DPYSL3','GRM7','ONECUT1','ONECUT2','LHX1','CALB1']
res={}
for g in genes:
    r=gt[gt.gene==g]
    if r.empty: continue
    r=r.iloc[0]
    s=r.tss-2000; e=r.tss+2000
    mask=(bins.chrom==r.chrom)&(bins.start<e)&(bins.end>s)
    res[g]=df.loc[mask,cols].sum(axis=0)
raw=pd.DataFrame(res, index=cols)  # rows cell types, columns genes
cpm=raw.div(df[cols].sum(axis=0), axis=0)*1e6
pd.set_option('display.max_columns',100); pd.set_option('display.width',200)
print('CPM marker table (rows are columns/cell labels):')
print(cpm.round(2).to_string())
# Also rank normalized by max for each marker
print('\\nFraction of max per gene (normalized):')
print((cpm/cpm.max(axis=0)).round(2).to_string())
PY"
