/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, gzip, sys
# load matrix bins and astro counts
df = pd.read_csv('data.tsv', sep='\\t', usecols=['chrom','start','end','Astrocyte'])
# Create dict for bins keyed (chrom,start)
bins=df[['chrom','start','end']]
# Initialize counts
counts=np.zeros(len(df), dtype=np.int64)
# stream fragments and count by 10kb bin start
with gzip.open('Astrocyte_rep3_frags.tsv.gz','rt') as f:
    for i,line in enumerate(f):
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<3: continue
        chrom=parts[0]; start=int(parts[1]); end=int(parts[2])
        # fragment midpoint
        mid=(start+end)//2
        bin_start=(mid//10000)*10000
        # find index
        idx=df.index[(df.chrom==chrom)&(df.start==bin_start)]
        if len(idx)==1:
            counts[idx[0]] += 1
print('reference total', counts.sum(), 'matrix total', df.Astrocyte.sum())
# correlation pearson/spearman between vector
print('Pearson', np.corrcoef(counts, df.Astrocyte.values)[0,1])
from scipy.stats import spearmanr
print('Spearman', spearmanr(counts, df.Astrocyte.values).correlation)
# MAD and relative differences at nonzero
print('max diff', np.max(np.abs(counts-df.Astrocyte.values)))
PY"
