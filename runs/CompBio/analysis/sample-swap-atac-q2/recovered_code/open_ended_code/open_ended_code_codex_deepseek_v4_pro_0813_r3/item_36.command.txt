/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, gzip, time
# load bins
bins=pd.read_csv('data.tsv', sep='\\t', usecols=['chrom','start'])
idx_map = {}
for row in bins.itertuples():
    idx_map[(row.chrom, row.start)] = row.Index
print('map size',len(idx_map))
counts=np.zeros(len(bins), dtype=np.int64)
t=time.time()
with gzip.open('Astrocyte_rep3_frags.tsv.gz','rt') as f:
    for i,line in enumerate(f):
        parts=line.split('\\t')
        if len(parts)<3: continue
        chrom=parts[0]
        start=int(parts[1]); end=int(parts[2])
        mid=(start+end)//2
        key=(chrom, (mid//10000)*10000)
        idx=idx_map.get(key)
        if idx is not None:
            counts[idx]+=1
        if i % 1000000 == 0 and i:
            print(i, time.time()-t)
print('done lines',i+1,'time',time.time()-t,'total',counts.sum())
np.save('astro_ref_counts.npy', counts)
# load matrix astro
mat=pd.read_csv('data.tsv',sep='\\t',usecols=['Astrocyte']).Astrocyte.values
print('matrix total',mat.sum())
print('Pearson',np.corrcoef(counts,mat)[0,1])
from scipy.stats import spearmanr
print('Spearman',spearmanr(counts,mat).correlation)
print('max diff',np.max(np.abs(counts-mat)))
print('mean diff nonzero',np.mean(np.abs(counts-mat)[mat>0]))
PY"
