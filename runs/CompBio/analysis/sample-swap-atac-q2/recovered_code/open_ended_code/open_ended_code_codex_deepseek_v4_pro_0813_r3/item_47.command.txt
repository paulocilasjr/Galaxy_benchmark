/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, os
from scipy.stats import spearmanr
cols = ['AIIamacrine','Astrocyte','Cone','GABAamacrine','Horizontal','Microglia','Mullerglia','Retinalganglioncell','Rod']
df = pd.read_csv('data.tsv', sep='\\t', usecols=cols)
refs={}
for c in cols:
    path='Astrocyte_ref_counts.npy' if c=='Astrocyte' else f'{c}_ref_counts.npy'
    if c=='Astrocyte' and not os.path.exists(path): path='astro_ref_counts.npy'
    refs[c]=np.load(path)
print('Totals ref vs matrix:')
for c in cols:
    print(c, refs[c].sum(), df[c].sum())
# Pearson full
print('\\nPearson:')
for r in cols:
    scores=[]
    for c in cols:
        scores.append(np.corrcoef(refs[r], df[c].values)[0,1])
    scores=np.array(scores)
    order=np.argsort(-scores)
    print(f'{r:22s}', '  '.join(f'{cols[j]}={scores[j]:.5f}' for j in order[:3]))
# Spearman sample 100k
rng=np.random.default_rng(0); idx=rng.choice(len(df),100000,replace=False)
print('\\nSpearman sample:')
for r in cols:
    scores=[]
    for c in cols:
        scores.append(spearmanr(refs[r][idx], df[c].values[idx]).correlation)
    scores=np.array(scores); order=np.argsort(-scores)
    print(f'{r:22s}', '  '.join(f'{cols[j]}={scores[j]:.5f}' for j in order[:3]))
# For each matrix column, best reference
print('\\nMatrix column best ref:')
for c in cols:
    scores=[]
    for r in cols:
        scores.append(np.corrcoef(refs[r], df[c].values)[0,1])
    scores=np.array(scores); order=np.argsort(-scores)
    print(f'{c:22s} -> {cols[order[0]]} {scores[order[0]]:.5f} | 2nd {cols[order[1]]} {scores[order[1]]:.5f}')
PY"
