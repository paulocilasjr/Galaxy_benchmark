/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
# reproduce score matrix for window 10k and print max, z-scores row-wise
# We'll paste scores from previous? Better recompute succinctly.
df = pd.read_csv('data.tsv', sep='\\t')
cols = df.columns[3:]
gt = pd.read_csv('gene_tss.tsv', sep='\\t')
bins = df[['chrom','start','end']].copy()
markers = {
 'Rod': ['RHO','NRL','NR2E3','PDE6B','PDE6A','GNAT1','CNGA1','CNGB1','SAG','ROM1','ABCA4','RBP3'],
 'Cone': ['ARR3','PDE6C','PDE6H','GNAT2','CNGB3','OPN1SW','OPN1MW','OPN1LW','RXRG','SALL3','GUCA1A'],
 'Mullerglia': ['RLBP1','GLUL','SLC1A3','CLU','SOX9','DIO3','VIM','KCNJ10','HES1'],
 'Astrocyte': ['GFAP','AQP4','S100B','ALDH1L1','SLC1A2','GJA1','SOX2'],
 'Microglia': ['P2RY12','CSF1R','CX3CR1','TYROBP','ITGAM','TMEM119','SPI1','AIF1','C1QA','C1QB','PTPRC'],
 'Retinalganglioncell': ['RBPMS','POU4F1','POU4F2','SLC17A6','THY1','NEFL','NEFM','ISL1','SNCG'],
 'GABAamacrine': ['GAD1','GAD2','SLC6A1','SLC32A1','CALB2','TFAP2A'],
 'AIIamacrine': ['GJD2','SLC12A7','DAB1','NEUROD1','DPYSL3','GRM7'],
 'Horizontal': ['ONECUT1','ONECUT2','LHX1','CALB1','SEMA5A'],
}
# collect all gene windows: TSS +/- 2000
scores={}
for ct, gl in markers.items():
    vals=[]
    for g in gl:
        r=gt[gt.gene==g]
        if r.empty: continue
        r=r.iloc[0]
        s=r.tss-2000; e=r.tss+2000
        mask=(bins.chrom==r.chrom)&(bins.start<e)&(bins.end>s)
        c=df.loc[mask, cols].sum(axis=0)
        vals.append(c)
    if vals:
        mat=pd.DataFrame(vals).T
        cpm=mat.div(df[cols].sum(axis=0),axis=0)*1e6
        scores[ct]=cpm.mean(axis=1)
score=pd.DataFrame(scores)
print('Raw score matrix (CPM +/-2kb):')
print(score.round(1).to_string())
print('\\nRow zscore:')
z=score.sub(score.mean(axis=1),axis=0).div(score.std(axis=1),axis=0)
print(z.round(2).to_string())
print('\\nBest class per row:')
for r in z.index:
    s=z.loc[r].sort_values(ascending=False)
    print(r, '->', s.index[0], s.iloc[0], 'second', s.index[1], s.iloc[1])
PY"
