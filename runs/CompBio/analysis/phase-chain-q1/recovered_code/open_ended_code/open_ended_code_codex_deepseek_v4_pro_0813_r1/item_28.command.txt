/bin/bash -lc "python3 - <<'PY'
import pysam
bam=pysam.AlignmentFile('phase.bam','rb')
start=27080260; end=27080295
for read in bam.fetch('chr22', start, end):
    if read.is_unmapped: continue
    # print aligned pairs for range compact
    pairs=read.get_aligned_pairs(matches_only=False)
    seg=[]
    for q,r in pairs:
        if r is not None and start <= r+1 <= end:
            if q is None: seg.append(f'{r+1}:D')
            else: seg.append(f'{r+1}:{read.query_sequence[q]}')
    # only print reads that include positions 27080272-27080282 and some key
    if read.reference_start <= 27080279 and read.reference_end >= 27080275:
        print(read.query_name, read.reference_start, read.cigarstring, ' '.join(seg))
PY"
