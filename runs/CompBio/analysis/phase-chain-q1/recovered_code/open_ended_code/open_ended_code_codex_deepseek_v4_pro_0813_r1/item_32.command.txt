/bin/bash -lc "python3 - <<'PY'
import pysam, collections
fasta=pysam.FastaFile('hg38.fa')
bam=pysam.AlignmentFile('phase.bam','rb')
start=27079000; end=27080700
refseq=fasta.fetch('chr22', start-1, end).upper()
for col in bam.pileup('chr22', start, end, truncate=True, min_base_quality=0, stepper='nofilter', max_depth=1000000):
    pos=col.reference_pos+1
    c=collections.Counter()
    for pr in col.pileups:
        if pr.is_refskip: continue
        if pr.is_del: c['D']+=1
        else: c[pr.alignment.query_sequence[pr.query_position]]+=1
    tot=sum(c.values())
    if tot==0: continue
    ref=refseq[pos-start]
    top=c.most_common()
    # candidate if any two non-noise; print if second count >=2 or refcount <= tot*0.5 or D present
    if len(c)>=2 and top[1][1]>=2:
        print(pos,'ref',ref,'tot',tot,c.most_common(5))
PY"
