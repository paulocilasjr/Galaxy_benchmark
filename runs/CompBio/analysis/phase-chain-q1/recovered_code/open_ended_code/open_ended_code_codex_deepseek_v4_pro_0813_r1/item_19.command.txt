/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('phase.bam','rb')
start=27079100; end=27080650
counts={}
for col in bam.pileup('chr22', start, end, truncate=True, min_base_quality=0, stepper='nofilter', max_depth=1000000):
    pos=col.reference_pos+1
    c=collections.Counter()
    for pr in col.pileups:
        if pr.is_del or pr.is_refskip: continue
        c[pr.alignment.query_sequence[pr.query_position]]+=1
    counts[pos]=c
for pos,c in counts.items():
    tot=sum(c.values())
    if tot<20: continue
    top=c.most_common()
    # alt is all nonref? Need reference per pos, but just look top2
    if len(c)==1: continue
    top1n=top[0][1]; top2n=top[1][1]
    frac2=top2n/tot
    if frac2>=0.15 or top1n/tot<=0.85:
        print(pos, tot, top, 'top2frac', round(top2n/tot,3), 'majorfrac', round(top1n/tot,3))
PY"
