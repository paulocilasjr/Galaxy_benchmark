/bin/bash -lc "python3 - <<'PY'
import pysam, collections
fasta=pysam.FastaFile('hg38.fa')
bam=pysam.AlignmentFile('phase.bam','rb')
for rng in [(27079120,27079170),(27080590,27080625)]:
 print('RANGE',rng)
 for col in bam.pileup('chr22', rng[0], rng[1], truncate=True, min_base_quality=0, stepper='nofilter', max_depth=1000000):
    pos=col.reference_pos+1; c=collections.Counter()
    for pr in col.pileups:
        if pr.is_refskip: continue
        if pr.is_del: c['D']+=1
        else: c[pr.alignment.query_sequence[pr.query_position]]+=1
    tot=sum(c.values())
    print(pos,'ref',fasta.fetch('chr22',pos-1,pos).upper(),'tot',tot,c.most_common())
PY"
