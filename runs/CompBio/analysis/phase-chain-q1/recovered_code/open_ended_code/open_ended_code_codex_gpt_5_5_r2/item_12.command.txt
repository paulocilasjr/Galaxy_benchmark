/bin/bash -lc "python3 - <<'PY'
import gzip, collections
seqs=[]
for fn in ['phase.chain.q1.R1.fq.gz','phase.chain.q1.R2.fq.gz']:
 with gzip.open(fn,'rt') as f:
  for i,l in enumerate(f):
   if i%4==1: seqs.append(l.strip())
print('reads',len(seqs),'unique',len(set(seqs)))
print('top duplicate counts',collections.Counter(seqs).most_common(10))
for k in [31,51,75,101,125]:
 c=collections.Counter()
 for s in seqs:
  for i in range(len(s)-k+1): c[s[i:i+k]]+=1
 vals=collections.Counter(c.values())
 print('k',k,'distinct',len(c),'count bins <=5',sum(v for cnt,v in vals.items() if cnt<=5),'min',min(c.values()),'max',max(c.values()),'common counts',vals.most_common(8))
PY"
