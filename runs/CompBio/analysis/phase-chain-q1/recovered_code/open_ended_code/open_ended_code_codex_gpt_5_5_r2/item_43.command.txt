/bin/bash -lc "python3 - <<'PY'
import pysam, sys
bam=pysam.AlignmentFile('aln.hg38.bam')
mins={}; maxs={}
for r in bam.fetch():
    if r.is_unmapped: continue
    c=r.reference_name
    mins[c]=min(mins.get(c,10**18), r.reference_start+1)
    maxs[c]=max(maxs.get(c,0), r.reference_end)
print(mins,maxs)
PY"
