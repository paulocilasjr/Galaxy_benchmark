/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, xml.etree.ElementTree as ET
rid='7HSSA32P014'
url='https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML'})
out=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read()
root=ET.fromstring(out)
for hit in root.findall('.//Hit'):
 print(hit.findtext('Hit_num'),hit.findtext('Hit_id'),hit.findtext('Hit_def'),hit.findtext('Hit_accession'))
 for hsp in hit.findall('./Hit_hsps/Hsp'):
  print(' ',hsp.findtext('Hsp_identity'),hsp.findtext('Hsp_align-len'),hsp.findtext('Hsp_query-from'),hsp.findtext('Hsp_query-to'),hsp.findtext('Hsp_hit-from'),hsp.findtext('Hsp_hit-to'),hsp.findtext('Hsp_qseq'),hsp.findtext('Hsp_hseq'))
PY"
