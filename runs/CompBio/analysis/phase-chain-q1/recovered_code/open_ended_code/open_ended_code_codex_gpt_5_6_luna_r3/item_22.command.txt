/bin/bash -lc "python3 - <<'PY'
import gzip,urllib.parse,urllib.request,re,time,xml.etree.ElementTree as ET
from Bio.Seq import Seq

def reads(p):
 with gzip.open(p,'rt') as f:
  while True:
   h=f.readline().strip()
   if not h: return
   s=f.readline().strip(); f.readline(); f.readline()
   yield h,s
r1=dict(reads('phase.chain.q1.R1.fq.gz')); r2=dict(reads('phase.chain.q1.R2.fq.gz'))
ids=['@READ_2','@READ_3','@READ_6','@READ_10','@READ_100']
fa=''.join(f'>{h[1:]}\\n{s}\\n' for h,s in r1.items() if h in ids)
print(fa)
data=urllib.parse.urlencode({'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':fa,'MEGABLAST':'on','HITLIST_SIZE':'5','FORMAT_TYPE':'XML','EXPECT':'1e-10','FILTER':'L'}).encode()
out=urllib.request.urlopen(urllib.request.Request('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=data,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read().decode('utf8','replace')
rids=re.findall(r'name=\"RID\"["'^>]*value="([A-Z0-9]+)"'"',out)
print('RID candidates',rids[:3])
if not rids:
 m=re.search(r'<input name=\"RID\"["'^>]*value="([''^"]+)"'"',out); rids=[m.group(1)] if m else []
print('RID',rids)
if not rids: raise SystemExit
rid=rids[0]
for i in range(20):
 time.sleep(4)
 url='https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML'})
 out2=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read()
 print('poll',i,len(out2),out2[:50])
 if b'<Hit>' in out2:
  root=ET.fromstring(out2)
  for it in root.findall('.//Iteration'):
   print('QUERY',it.findtext('Iteration_query-def'))
   for hit in it.findall('./Iteration_hits/Hit')[:3]:
    print(' ',hit.findtext('Hit_def'))
    for hsp in hit.findall('./Hit_hsps/Hsp')[:2]: print('   ',hsp.findtext('Hsp_identity'),hsp.findtext('Hsp_align-len'),hsp.findtext('Hsp_hit-from'),hsp.findtext('Hsp_hit-to'),hsp.findtext('Hsp_qseq'),hsp.findtext('Hsp_hseq'))
  break
PY"
