/bin/bash -lc "samtools stats reads_vs_contigs.bam | grep -E 'insert size|average length|maximum length|SN\\s+average|SN\\s+insert|SN\\s+maximum' | head -20
printf 'CIGAR distribution\\n'; samtools view reads_vs_contigs.bam | awk '{print "'$6}'"' | sort | uniq -c | sort -nr | head -30
printf 'alignment starts\\n'; samtools view -F 0x900 reads_vs_contigs.bam | awk '{print "'$2,$4,$5,$6,$9}'"' | sort -k2,2n | head -30
printf 'indel relevant alignments\\n'; samtools view reads_vs_contigs.bam | awk '"'$6 ~ /[ID]/ {print}'"' | head -30"
