/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('reads_vs_contigs.bam','rb'); fa=pysam.FastaFile('spades_out/contigs.fasta'); ref=fa.fetch(fa.references[0])
for col in bam.pileup(fa.references[0],0,len(ref),truncate=True,stepper='all',min_base_quality=0,min_mapping_quality=0,max_depth=1000,ignore_overlaps=False):
 p=col.reference_pos
 if p>=len(ref): continue
 c=collections.Counter(); ins=collections.Counter(); dele=collections.Counter()
 for pr in col.pileups:
  if pr.is_del or pr.is_refskip: dele[pr.indel]+=1 if pr.is_del else 0
  else:
   b=pr.alignment.query_sequence[pr.query_position]; c[b]+=1
   if pr.indel>0: ins[pr.alignment.query_sequence[pr.query_position+1:pr.query_position+1+pr.indel]]+=1
 # any nonref count >=4 or indel >=4
 alt=sum(v for b,v in c.items() if b"'!=ref[p])
 if alt>=4 or sum(ins.values())>=4:
  print(f'"'{p+1:4d} ref={ref[p]} cov={col.nsegments:3d} bases={dict(c)} ins={dict(ins)}')
PY"
