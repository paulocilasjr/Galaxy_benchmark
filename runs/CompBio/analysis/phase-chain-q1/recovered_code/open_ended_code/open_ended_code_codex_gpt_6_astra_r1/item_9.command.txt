/bin/bash -lc "python3 -u - <<'PY'
import urllib.request
for label,u in [('faq','https://genome.ucsc.edu/FAQ/FAQdownloads.html#CAPTCHA'),('api','https://api.genome.ucsc.edu/')]:
 try:
  s=urllib.request.urlopen(u,timeout=30).read().decode();open('work/'+label+'.html','w').write(s)
  from bs4 import BeautifulSoup
  t=BeautifulSoup(s,'html.parser').get_text();open('work/'+label+'.txt','w').write(t)
  for line in t.splitlines():
   if any(k in line.lower() for k in ['blat','captcha','bot','api','programmatic']):print(label,line[:500])
 except Exception as e:print(e)
PY"
