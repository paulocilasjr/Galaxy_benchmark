/bin/bash -lc "python3 - <<'PY'
import pysam,collections
b=pysam.AlignmentFile('work/reads.bam');f=pysam.FastaFile('work/region.fa');ref=f.fetch(b.references[0])
for start,end in [(3085,3140),(2640,2690)]:
 c=collections.Counter()
 for r in b.fetch():
  d={p:q for q,p in r.get_aligned_pairs() if q is not None and p is not None}
  if start in d and end in d:c[r.query_sequence[d[start]:d[end]+1]]+=1
 print('REF',start+1,ref[start:end+1])
 for s,n in c.most_common(10):print(n,s)
PY"
