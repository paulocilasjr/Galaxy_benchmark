/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
ins=[]
for rec in b:
 if rec.is_read1 and rec.is_proper_pair:
  ins.append(abs(rec.tlen))
print('insert sizes',Counter(ins).most_common(20), 'min',min(ins),'max',max(ins))
# coverage per pos for range 27079120..27080630
cov=Counter()
for rec in b.fetch('chr22',27079120,27080631):
 if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
 for qpos,rpos in rec.get_aligned_pairs(matches_only=False):
  if rpos is not None: cov[rpos+1]+=1
print('cov min region 27079120-27080630')
for p in range(27079120,27080631):
 if p in [27079120,27079125,27079126,27079127,27080620,27080624,27080625,27080626,27080630]:
  print(p,cov[p])
print('nonzero range',min(p for p,c in cov.items() if c),max(p for p,c in cov.items() if c))
PY"
