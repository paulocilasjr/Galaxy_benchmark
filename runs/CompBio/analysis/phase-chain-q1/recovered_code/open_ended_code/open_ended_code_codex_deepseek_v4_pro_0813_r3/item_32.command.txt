/bin/bash -lc "python3 - <<'PY'
import pysam
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
def print_group(start,end,del_):
 n=0
 for rec in b.fetch('chr22',start-1,end):
    if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
    seq=rec.query_sequence
    chars=[]
    for qpos,rpos in rec.get_aligned_pairs(matches_only=False):
        if rpos is None: continue
        if start<=rpos+1<=end:
            ch='-' if qpos is None else seq[qpos]
            chars.append(ch)
    # require covers full window length
    if len(chars)<end-start+1: continue
    has='-' in chars
    if has==del_:
        print(rec.qname, ''.join(chars))
        n+=1
        if n>=8: break
print('del region1'); print_group(27079818,27079832,True)
print('nondel region1'); print_group(27079818,27079832,False)
print('del region2'); print_group(27080270,27080282,True)
print('nondel region2'); print_group(27080270,27080282,False)
PY"
