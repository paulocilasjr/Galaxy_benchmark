/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
start=27079126; end=27080624; L=end-start+1
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
pos_counts=[Counter() for _ in range(L)]
for rec in b.fetch('chr22',start-1,end+1):
 if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
 seq=rec.query_sequence
 for qpos,rpos in rec.get_aligned_pairs(matches_only=False):
  if rpos is None: continue
  idx=rpos-(start-1)
  if 0<=idx<L:
   pos_counts[idx]['-' if qpos is None else seq[qpos]]+=1
print('all positions with two alleles and balanced-ish')
for i,c in enumerate(pos_counts):
 total=sum(c.values())
 if total==0: continue
 top=c.most_common(2)
 if len(top)>=2:
  f=top[1][1]/total
  # include if minor freq >=0.2, or count>=3 and freq>=0.3
  if (f>=0.2 and top[1][1]>=3) or (top[1][1]>=5 and f>=0.15):
   print(start+i, 'depth',total, top)
PY"
