/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
ref_start=27079126; ref_end=27080625; L=ref_end-ref_start+1
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
pos_counts=[Counter() for _ in range(L)]
for rec in b.fetch('chr22',ref_start-1,ref_end):
    if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
    seq=rec.query_sequence
    for qpos,rpos in rec.get_aligned_pairs(matches_only=False):
        if rpos is None: continue
        idx=rpos-(ref_start-1)
        if idx<0 or idx>=L: continue
        ch='-' if qpos is None else seq[qpos]
        pos_counts[idx][ch]+=1
print('high-confidence variables')
for i,c in enumerate(pos_counts):
    total=sum(c.values())
    if total<10: continue
    top=c.most_common(2)
    # true SNP/indel if second freq >=0.15 and second count >=10, and top two bases (including '-') not rare
    if len(top)>=2 and top[1][1]>=10 and top[1][1]/total>=0.15:
        print(ref_start+i, i, 'total',total, top)
PY"
