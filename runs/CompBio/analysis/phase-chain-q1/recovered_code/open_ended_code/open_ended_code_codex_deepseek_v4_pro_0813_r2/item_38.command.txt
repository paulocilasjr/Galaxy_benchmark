/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter

def get_aligned_window(r, start, end):
    '''return aligned query bases for ref interval [start,end) (0-based), or None if not fully covered'''
    out=[]
    refpos=r.reference_start; qpos=0
    for op,length in r.cigartuples:
        if op==0:
            for i in range(length):
                if start<=refpos+i<end:
                    out.append(r.query_sequence[qpos+i])
                refpos+=1; qpos+=1
        elif op==1:
            # insertions before current refpos: only include if previous refpos in window maybe as lowercase? skip for simplicity
            qpos+=length
        elif op==2:
            for i in range(length):
                if start<=refpos+i<end:
                    out.append('-')
                refpos+=1
        elif op==3: refpos+=length
        elif op==4: qpos+=length
        else: raise
    # ensure full window covered by alignment: reference_start <= start and reference_end >= end (approx)
    if r.reference_start<=start and r.reference_end>=end and len(out)==(end-start):
        return ''.join(out)
    return None

bam=pysam.AlignmentFile('aln.sorted.bam','rb')
windows=[(27079810,27079845),(27080265,27080300),(27079300,27079312),(27079660,27079672),(27080070,27080082)]
for ws,we in windows:
    print('\\nWINDOW 0-based',ws,we,'1-based',ws+1,we)
    variants=Counter()
    for r in bam.fetch('chr22',ws,we):
        if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
        s=get_aligned_window(r,ws,we)
        if s is not None:
            variants[s]+=1
    for s,n in variants.most_common(30):
        print(n, s)
bam.close()
PY"
