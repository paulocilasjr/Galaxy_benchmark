/bin/bash -lc "python3 - <<'PY'
# create a VCF with only non-ref alt lines (interesting) for bcftools norm
import sys
with open('variants.vcf') as f, open('vars_only.vcf','w') as out:
    for line in f:
        if line.startswith('#'):
            out.write(line)
            continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<5: continue
        alts=p[4].split(',')
        if any(a not in ('.','<*>') and a"'!=p[3] for a in alts):
            out.write(line)
PY
bcftools norm -m -both vars_only.vcf -Ov -o vars_norm.vcf 2>norm.err || true
cat vars_norm.vcf | grep -v '"'"'^#'"' | head -100
echo '---err'; cat norm.err"
