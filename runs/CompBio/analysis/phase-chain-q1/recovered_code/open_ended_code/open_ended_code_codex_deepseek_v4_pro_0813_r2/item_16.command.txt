/bin/bash -lc "python3 - <<'PY'
import gzip
from itertools import islice

def read_fastq(fn):
    with gzip.open(fn,'rt') as f:
        while True:
            name=f.readline().strip()
            if not name: break
            seq=f.readline().strip(); plus=f.readline().strip(); qual=f.readline().strip()
            yield name,seq,qual

def rc(s):
    return s.translate(str.maketrans('ACGT','TGCA'))[::-1]
r1=list(islice(read_fastq('phase.chain.q1.R1.fq.gz'),5))
r2=list(islice(read_fastq('phase.chain.q1.R2.fq.gz'),5))
for (n1,s1,q1),(n2,s2,q2) in zip(r1,r2):
    s2rc=rc(s2)
    print(n1,'R1',s1)
    print(n2,'R2rc',s2rc)
PY"
