/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter

def get_aligned_window(r, start, end):
    # Build per-ref columns for window
    out=['*']*(end-start)
    pairs=r.get_aligned_pairs(matches_only=False)
    for qp,rp in pairs:
        if rp is None: continue
        if start<=rp<end:
            out[rp-start]= '-' if qp is None else r.query_sequence[qp]
    # check full coverage: reference_start <= start and reference_end >= end
    if r.reference_start<=start and r.reference_end>=end and all(c"'!='"'*' for c in out):
        return ''.join(out)
    return None

bam=pysam.AlignmentFile('aln.sorted.bam','rb')
windows=[(27079810,27079845),(27080265,27080300),(27079300,27079312),(27079660,27079672),(27080070,27080082),(27080520,27080532)]
for ws,we in windows:
    print('\\nWINDOW 0-based',ws,we,'1-based',ws+1,we)
    variants=Counter()
    for r in bam.fetch('chr22',ws,we):
        if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
        s=get_aligned_window(r,ws,we)
        if s is not None:
            variants[s]+=1
    for s,n in variants.most_common(20):
        print(f'{n:3d} {s}')
bam.close()
PY"
