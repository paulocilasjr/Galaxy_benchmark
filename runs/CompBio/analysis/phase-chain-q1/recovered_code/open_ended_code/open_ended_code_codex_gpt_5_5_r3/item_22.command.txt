/bin/bash -lc "command -v bcftools || true
minimap2 -ax sr AL008638.1.fa phase.chain.q1.R1.fq.gz phase.chain.q1.R2.fq.gz > aln.sam 2> minimap2.log
samtools view -bS aln.sam | samtools sort -o aln.bam
samtools index aln.bam
samtools flagstat aln.bam
samtools depth -a aln.bam | awk '"'$3>0{if(!s)s=$2;e=$2; if(min==0||$3<min)min=$3; if($3>max)max=$3; sum+=$3; n++}END{print s,e,n,min,max,sum/n}'"'"
