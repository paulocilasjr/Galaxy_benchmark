/bin/bash -lc "python3 - <<'PY'
import os, tarfile, urllib.request
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE242nnn/GSE242423/suppl/GSE242423_RAW.tar'
wanted={'GSM7763420_D2.barcodes.tsv.gz','GSM7763420_D2.matrix.mtx.gz'}
class ProgressFile:
    def __init__(self, fp):
        self.fp=fp; self.n=0; self.next=50*1024*1024
    def read(self, size=-1):
        b=self.fp.read(size)
        self.n += len(b)
        if self.n >= self.next:
            print(f'read {self.n/1024/1024:.0f} MiB', flush=True)
            self.next += 50*1024*1024
        return b
with urllib.request.urlopen(url) as resp:
    pf=ProgressFile(resp)
    with tarfile.open(fileobj=pf, mode='r|') as tf:
        found=set()
        for m in tf:
            name=os.path.basename(m.name)
            if name in wanted:
                print('extracting', name, m.size, flush=True)
                src=tf.extractfile(m)
                with open(name, 'wb') as out:
                    while True:
                        chunk=src.read(1024*1024)
                        if not chunk: break
                        out.write(chunk)
                found.add(name)
                if found == wanted:
                    print('done', flush=True)
                    break
print('found', sorted(found))
PY
ls -lh GSM7763420_D2.*.gz"
