/bin/bash -lc "archive_url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE242nnn/GSE242423/suppl/GSE242423_RAW.tar'
d2_barcodes_header=244328960
d2_barcodes_size=7391073
d2_barcodes_payload_start="'$((d2_barcodes_header+512))
d2_barcodes_payload_end=$((d2_barcodes_payload_start+d2_barcodes_size-1))
d2_matrix_header=251720704
d2_matrix_size=125124004
d2_matrix_payload_start=$((d2_matrix_header+512))
d2_matrix_payload_end=$((d2_matrix_payload_start+d2_matrix_size-1))
curl -L --fail --silent --show-error --range "$d2_barcodes_payload_start-$d2_barcodes_payload_end" -o GSM7763420_D2.barcodes.tsv.gz "$archive_url"
curl -L --fail --silent --show-error --range "$d2_matrix_payload_start-$d2_matrix_payload_end" -o GSM7763420_D2.matrix.mtx.gz "$archive_url"
ls -l GSM7763420_D2.barcodes.tsv.gz GSM7763420_D2.matrix.mtx.gz
printf '"'barcode gzip: '; gzip -t GSM7763420_D2.barcodes.tsv.gz && echo OK
printf 'matrix gzip: '; gzip -t GSM7763420_D2.matrix.mtx.gz && echo OK
printf 'barcode first: '; gzip -cd GSM7763420_D2.barcodes.tsv.gz | head -1
printf 'matrix header: '; gzip -cd GSM7763420_D2.matrix.mtx.gz | head -3"
