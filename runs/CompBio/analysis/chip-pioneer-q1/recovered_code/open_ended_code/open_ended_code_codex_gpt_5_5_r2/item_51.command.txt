/bin/bash -lc "cat > scripts/count_open_overlap.sh <<'SH'
#"'!/usr/bin/env bash
set -euo pipefail
out=results/open_fragment_overlap.tsv
printf '"'open_set\\tfile\\ttotal_lines\\ttotal_weight\\topen_lines\\topen_weight\\n' > \""'$out"
for b in data/open_sets/*.bed; do
  setname=$(basename "$b" .bed)
  for f in chip.pioneer.q1/*.bed.gz; do
    name=$(basename "$f")
    total=$(gzip -cd "$f" | awk '"'BEGIN{l=0;w=0} "'$1 !~ /_/ && $1!="chrM" {l++; w+=(NF>=7?$7:1)} END{print l"'"\\t\"w}')
    open="'$(gzip -cd "$f" | bedtools intersect -u -a stdin -b "$b" | awk '"'BEGIN{l=0;w=0} {l++; w+=(NF>=7?"'$7:1)} END{print l"'"\\t\"w}')
    printf '%s\\t%s\\t%s\\t%s\\n' \""'$setname" "$name" "$total" "$open" >> "$out"
  done
done
SH
bash scripts/count_open_overlap.sh'
