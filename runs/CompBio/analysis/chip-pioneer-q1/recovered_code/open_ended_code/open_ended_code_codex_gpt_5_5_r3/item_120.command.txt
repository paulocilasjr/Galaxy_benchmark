import subprocess, os

def primary_total(pattern):
    p=subprocess.run(f\"gzip -cd {pattern} | awk '"'$1 ~ /''^chr([0-9]+|X|Y)$/ {{c++}} END{{print c+0}}'"'\", shell=True, check=True, text=True, capture_output=True)
    return float(p.stdout.strip())

                for idx,mask in [(3,'narrow'),(4,'hotspot')]:
                    state='open' if a[idx]=='1' else 'closed'
