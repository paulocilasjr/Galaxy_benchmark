/bin/bash -lc "chmod +x work/count_dnase_overlap.sh
./work/count_dnase_overlap.sh work/BJ_DNase_preferred_union.bed work/fragment_overlap_narrow.tsv
./work/count_dnase_overlap.sh work/BJ_DNase_hotspot_union.bed work/fragment_overlap_hotspot.tsv
python3 - <<'PY'
import collections, pathlib
for label,path in [('narrow','work/fragment_overlap_narrow.tsv'),('hotspot','work/fragment_overlap_hotspot.tsv')]:
    d=collections.defaultdict(lambda: {'rep':[0,0,0], 'ctl':[0,0,0]})
    for line in open(path):
        f,total,op,cl=line.rstrip().split('\\t')
        tf=pathlib.Path(f).name.split('.')[0]
        typ='ctl' if 'control' in f else 'rep'
        vals=list(map(int,[total,op,cl]))
        for i,v in enumerate(vals):
            d[tf][typ][i] += v
    print(label)
    print('TF\\trep_open_frac\\tctl_open_frac\\tclosed_enrich_vs_ctl\\topen_enrich_vs_ctl')
    for tf in sorted(d):
        rt,ro,rc=d[tf]['rep']; ct,co,cc=d[tf]['ctl']
        print(tf, f'{ro/rt:.6f}', f'{co/ct:.6f}', f'{(rc/rt)/(cc/ct):.6f}', f'{(ro/rt)/(co/ct):.6f}', sep='\\t')
PY"
