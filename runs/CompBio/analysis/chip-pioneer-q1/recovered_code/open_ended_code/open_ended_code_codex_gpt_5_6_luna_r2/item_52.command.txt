/bin/bash -lc "python - <<'PY'
from analyze_pioneer import read_intervals, union, intersection
p1=read_intervals('dnase_rep1_peaks.bed.gz'); p2=read_intervals('dnase_rep2_peaks.bed.gz')
h1=read_intervals('dnase_rep1_hotspots.bed.gz'); h2=read_intervals('dnase_rep2_hotspots.bed.gz')
regions={'peak_r1':p1,'peak_r2':p2,'peak_union':union(p1,p2),'peak_both':intersection(p1,p2),'hot_r1':h1,'hot_r2':h2,'hot_union':union(h1,h2),'hot_both':intersection(h1,h2)}
with open('dnase_regions.tsv','w') as o:
 for name,d in regions.items():
  for chrom, arr in d.items():
   for s,e in arr: o.write(f'{name}\\t{chrom}\\t{s}\\t{e}\\n')
for k,v in regions.items(): print(k,sum(map(len,v.values())),sum(e-s for a in v.values() for s,e in a))
PY
wc -l dnase_regions.tsv; sed -n '1,3p' dnase_regions.tsv"
