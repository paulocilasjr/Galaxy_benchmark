/bin/bash -lc "command -v chromap || true; python - <<'PY'
import gzip,glob,collections,os
for f in sorted(glob.glob('chip.pioneer.q1/*.bed.gz')):
 c=collections.Counter(); lens=[]; score=collections.Counter(); n=0
 with gzip.open(f,'rt') as h:
  for line in h:
   p=line.rstrip().split('\\t'); n+=1; c[len(p)]+=1; c['last_'+p[-1]]+=1; score[p[4]]+=1
   if n<=100000: lens.append(int(p[2])-int(p[1]))
 print(os.path.basename(f), n, 'fields', {k:v for k,v in c.items() if isinstance(k,int)}, 'last',sum(v for k,v in c.items() if isinstance(k,str) and k.startswith('last_')), 'lastvals', {k[5:]:v for k,v in c.items() if isinstance(k,str) and k.startswith('last_')}, 'len_first100k', min(lens),max(lens))
PY"
