/bin/bash -lc "python - <<'PY'
import gzip,collections
for f in ['dnase_rep1_peaks.bed.gz','dnase_rep2_peaks.bed.gz','dnase_rep1_hotspots.bed.gz','dnase_rep2_hotspots.bed.gz']:
 n=bp=0; cs=collections.Counter()
 with gzip.open(f,'rt') as h:
  for l in h:
   p=l.split('\\t'); n+=1; bp+=int(p[2])-int(p[1]); cs[p[0]]+=1
 print(f,n,bp,len(cs),cs.most_common(3))
PY"
