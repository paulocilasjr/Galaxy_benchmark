/bin/bash -lc "python - <<'PY'
import os,math
regs=['peak_r1','peak_r2','peak_union','peak_both','hot_r1','hot_r2','hot_union','hot_both']
D={}
for l in open('fast_counts.tsv'):
 p=l.rstrip().split('\\t'); b=os.path.basename(p[0]); tf=b.split('.')[0]; typ='c' if 'control' in b else 'r'
 D.setdefault((tf,typ),[]).append((int(p[1]),int(p[2]),{k:tuple(map(int,v.split('/'))) for k,v in zip(regs,p[3:])}))
for typenum in [1,2]:
 print('WEIGHTED' if typenum==2 else 'ROWS')
 for tf in [f'TF{i}' for i in range(1,7)]:
  rn=sum(x[typenum] for x in D[(tf,'r')]); cn=sum(x[typenum] for x in D[(tf,'c')])
  vals=[]
  for reg in ['peak_r1','peak_r2','peak_union','peak_both','hot_union','hot_both']:
   ra=sum(x[2][reg][typenum-1] for x in D[(tf,'r')]); ca=sum(x[2][reg][typenum-1] for x in D[(tf,'c')])
   fa=ra/rn; fc=ca/cn
   vals.append(f'{reg}:A{math.log2(fa/fc):.3f},O{math.log2((1-fa)/(1-fc)):.3f},od{math.log2(fa/(1-fa)*((1-fc)/fc)):.3f}')
  print(tf,'; '.join(vals))
PY"
