/bin/bash -lc "python - <<'PY'
import re,os,math,glob,statistics
regions=['peak_r1','peak_r2','peak_union','peak_both','hot_r1','hot_r2','hot_union','hot_both']
rows={}
for l in open('fast_counts.tsv'):
 p=l.rstrip().split('\\t'); name=os.path.basename(p[0]); parts=name.split('.')
 tf=parts[0]; typ='control' if 'control' in name else 'rep'
 total=int(p[1]); wtotal=int(p[2]); vals={}
 for k,x in zip(regions,p[3:]): vals[k]=tuple(map(int,x.split('/')))
 rows.setdefault((tf,typ),[]).append((name,total,wtotal,vals))
# combine replicates weighted by total rows, and separately print per replicate fraction and fold vs control
for tf in [f'TF{i}' for i in range(1,7)]:
 cs=rows[(tf,'control')]
 rs=rows[(tf,'rep')]
 print('\\n',tf)
 print('samples',[(x[0],x[1]) for x in rs],[(x[0],x[1]) for x in cs])
 for reg in ['peak_r1','peak_r2','peak_union','peak_both','hot_union','hot_both']:
  cN=sum(x[1] for x in cs); rN=sum(x[1] for x in rs)
  ca=sum(x[3][reg][0] for x in cs); ra=sum(x[3][reg][0] for x in rs)
  # per-row normalized fraction; odds/fraction enrichment
  fa=ra/rN; fc=ca/cN
  fo=1-fa; fco=1-fc
  print(reg, f'chip_A={fa:.5f}',f'ctrl_A={fc:.5f}',f'log2A={math.log2(fa/fc):.3f}',f'log2O={math.log2(fo/fco):.3f}',f'odds={math.log2((fa/fo)/(fc/fco)):.3f}',f'ratioA/O={fa/fo:.3f}')
PY"
