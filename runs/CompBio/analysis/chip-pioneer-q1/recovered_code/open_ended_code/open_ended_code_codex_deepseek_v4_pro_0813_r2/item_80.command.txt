/bin/bash -lc "mkdir -p geo_peaks && python3 - <<'PY'
import re, subprocess
text=open('gse_gsm.txt').read()
samples=text.split('"'^SAMPLE = '"')[1:]
# map factor -> first peak file containing 'peaks_hg19'
want=['FOXA1wt','FOXA1dbd','FOXA1nhaa','FOXA1rraa','HNF4A']
for s in samples:
    title=''
    files=[]
    for line in s.splitlines():
        if line.startswith('"'!Sample_title = '"'): title=line.split('= ',1)[1]
        if line.startswith('"'!Sample_supplementary_file_'"'): files.append(line.split('= ',1)[1])
    if not title: continue
    for factor in want:
        if title.startswith(factor) and 'input' not in title:
            peak=[f for f in files if 'peaks_hg19' in f]
            if peak:
                out=f'geo_peaks/{factor}.bed.gz'
                print(factor, title, peak[0], '->', out)
                subprocess.run(['curl','-L','-sS','-o',out,peak[0]],check=True)
PY
ls -lh geo_peaks"
