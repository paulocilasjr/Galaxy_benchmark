/bin/bash -lc "python3 - <<'PY'
import subprocess, os
# create broad hotspot3 merged from dnase_rep2_hotspot... if not exists
if not os.path.exists('processed/dnase_hotspot3_broad.bed'):
    import gzip
    raw='processed/tmp_hotspot3_hg38.bed'
    with gzip.open('dnase_rep2_hotspot_ENCFF034CWP.bed.gz','rt') as f, open(raw,'w') as o:
        for line in f:
            if line.startswith('#'): continue
            p=line.split('\\t'); o.write(f'{p[0]}\\t{p[1]}\\t{p[2]}\\n')
    subprocess.run('bedtools sort -i '+raw+' | bedtools merge -i - > processed/dnase_hotspot3_broad.bed',shell=True,check=True)
sets={
 'narrow_union':'processed/dnase_access_union.bed',
 'hotspot_union_150':'processed/dnase_hotspot_union.bed',
 'hotspot3_broad':'processed/dnase_hotspot3_broad.bed',
}
for name,dnase in sets.items():
    print('\\nDNASE SET',name)
    for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
        peaks=f'processed/union_peaks/{tf}.bed'
        n=int(subprocess.check_output(['wc','-l','<',peaks]).decode().strip())
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {peaks} -b {dnase} | wc -l']).decode().strip())
        print(f'{tf}\\tn={n}\\tacc={a}\\tfrac_acc={a/n:.3f}\\tfrac_res={(n-a)/n:.3f}')
PY"
