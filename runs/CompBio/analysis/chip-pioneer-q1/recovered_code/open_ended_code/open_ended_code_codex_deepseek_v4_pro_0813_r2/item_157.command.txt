/bin/bash -lc "cat processed/dnase_access_union.bed processed/dnase_hotspot_union.bed processed/dnase_hotspot3_broad.bed | bedtools sort -i - | bedtools merge -i - > processed/dnase_all_union.bed && wc -l processed/dnase_all_union.bed && python3 - <<'PY'
import subprocess
for fac in ['FOXA1wt','SOX2wt','HNF4A','FOXA1nhaa','FOXA1rraa','FOXA1dbd','SOX2dbd']:
    geo=f'geo_peaks_hg38/{fac}.bed'
    n=sum(1 for _ in open(geo))
    a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {geo} -b processed/dnase_all_union.bed | wc -l']).decode().strip())
    print(f'{fac}: acc={a/n:.3f} res={(n-a)/n:.3f}')
PY"
