/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('dnase_rep1_ENCFF768KOM.bigWig')
print(bw.chroms())
# compute mean over geo peaks, handling chroms
import numpy as np
for fac in ['FOXA1wt','SOX2wt','HNF4A','FOXA1dbd','FOXA1nhaa','FOXA1rraa','SOX2dbd']:
    vals=[]
    with open(f'geo_peaks_hg38/{fac}.bed') as f:
        for line in f:
            p=line.split('\\t'); chrom=p[0]; start=int(p[1]); end=int(p[2])
            try: v=bw.stats(chrom,start,end,type='mean',exact=True)[0]
            except: v=None
            if v is not None: vals.append(v)
    arr=np.array(vals)
    print(f'{fac} n={len(vals)} mean={arr.mean():.4f} median={np.median(arr):.4f} p25={np.percentile(arr,25):.4f} p75={np.percentile(arr,75):.4f}')
bw.close()
PY"
