/bin/bash -lc "python3 - <<'PY'
import pandas as pd
D=pd.read_csv('processed/frag_access_counts.tsv',sep='\\t',header=None,names=['sample','acc','inacc','total'])
D['tf']=D['sample'].str.extract(r'(TF\\d)')
D['type']=D['sample'].str.extract(r'\\.(rep|control)')
D['acc_frac']=D.acc/D.total
print(D[['sample','acc','inacc','total','acc_frac']].sort_values('sample').to_string(index=False))
for tf in sorted(D.tf.unique()):
    sub=D[D.tf==tf]
    reps=sub[sub.type=='rep']
    ctrls=sub[sub.type=='control']
    print('\\n'+tf)
    print(reps[['sample','acc_frac']].to_string(index=False))
    print('control mean acc_frac',ctrls.acc_frac.mean())
    print('rep mean acc_frac',reps.acc_frac.mean())
    print('ratio rep/ctrl',reps.acc_frac.mean()/ctrls.acc_frac.mean())
PY"
