/bin/bash -lc "python3 - <<'PY'
import os
from Bio import SeqIO
# If Bio not installed? Try
try:
    from Bio import SeqIO
except Exception as e:
    print('biopython not installed',e); raise SystemExit
motifs={
 'FOXA1_AAAGTAAACA':['AAAGTAAACA','TGTTTACTTT'],
 'FOXA1_core_TAAACA':['GTAAACA','TGTTTAC'],
 'HNF4A_CAAAGTCCA':['CAAAGTCCA','TGGACTTTG'],
 'HNF4A_DR1':['CAAAGGTCA','TGACCTTTG'],
 'SOX2_CATTGTT':['CATTGTT','AACAATG'],
 'SOX2_core_ACAAT':['AACAAT','ATTGTT'],
}
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    fa=f'processed/motif_fa/{tf}_top10k.fa'
    seqs=[]
    for rec in SeqIO.parse(fa,'fasta'):
        seqs.append(str(rec.seq).upper())
    n=len(seqs)
    counts={}
    for m,words in motifs.items():
        c=sum(1 for s in seqs if any(w in s for w in words))
        counts[m]=c
    print(tf, 'n',n, ' '.join(f'{k}:{v/n:.3f}' for k,v in counts.items()))
PY"
