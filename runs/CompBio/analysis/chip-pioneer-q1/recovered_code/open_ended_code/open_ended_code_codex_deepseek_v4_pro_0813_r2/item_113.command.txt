/bin/bash -lc "python3 - <<'PY'
import pandas as pd
D=pd.read_csv('processed/frag_access_counts.tsv',sep='\\t',header=None,names=['sample','acc','inacc','total'])
D['tf']=D['sample'].str.extract(r'(TF\\d)')
D['type']=D['sample'].str.extract(r'\\.(rep|control)')
print('\\nPioneering odds ratio by sample')
rows=[]
for tf in sorted(D.tf.unique()):
    sub=D[D.tf==tf]
    reps=sub[sub.type=='rep']
    ctrls=sub[sub.type=='control']
    rep_acc=reps.acc.sum(); rep_in=reps.inacc.sum(); rep_tot=reps.total.sum()
    ctrl_acc=ctrls.acc.sum(); ctrl_in=ctrls.inacc.sum(); ctrl_tot=ctrls.total.sum()
    # if multiple reps and controls pooled
    ratio_acc=(rep_acc/rep_tot)/(ctrl_acc/ctrl_tot)
    ratio_in=(rep_in/rep_tot)/(ctrl_in/ctrl_tot)
    index=ratio_in/ratio_acc
    print(f'{tf}: acc_ratio={ratio_acc:.4f} inacc_ratio={ratio_in:.4f} pioneer_index={index:.6f} rep_tot={rep_tot} ctrl_tot={ctrl_tot}')
    for _,r in reps.iterrows():
        c=ctrls
        # use pooled ctrl
        ra=(r.acc/r.total)/(ctrl_acc/ctrl_tot)
        ri=(r.inacc/r.total)/(ctrl_in/ctrl_tot)
        print(f'  {r[\"sample\"]}: pioneer_index={ri/ra:.6f} acc_ratio={ra:.4f} inacc_ratio={ri:.4f}')
PY"
