/bin/bash -lc "python3 - <<'PY'
import subprocess
factors=['FOXA1wt','FOXA1dbd','FOXA1nhaa','FOXA1rraa','HNF4A','SOX2wt','SOX2dbd']
print('TF\\t'+'\\t'.join(factors)+'\\tn')
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    peaks=f'processed/union_peaks/{tf}.bed'
    n=sum(1 for _ in open(peaks))
    vals=[]
    for fac in factors:
        geo=f'geo_peaks_hg38/{fac}.bed'
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {peaks} -b {geo} | wc -l']).decode().strip())
        vals.append(f'{a/n:.3f}')
    print(tf+'\\t'+'\\t'.join(vals)+f'\\t{n}')
PY"
