/bin/bash -lc "python3 - <<'PY'
import subprocess
sets={
 'narrow_union':'processed/dnase_access_union.bed',
 'hotspot_union_150':'processed/dnase_hotspot_union.bed',
 'hotspot3_broad':'processed/dnase_hotspot3_broad.bed',
}
for name,dnase in sets.items():
    print('\\nDNASE',name)
    for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
        peaks=f'processed/union_peaks_q001/{tf}.bed'
        n=sum(1 for _ in open(peaks))
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {peaks} -b {dnase} | wc -l']).decode().strip())
        print(f'{tf}\\tn={n}\\tacc={a}\\tfrac_acc={a/n:.3f}\\tfrac_res={(n-a)/n:.3f}')
PY"
