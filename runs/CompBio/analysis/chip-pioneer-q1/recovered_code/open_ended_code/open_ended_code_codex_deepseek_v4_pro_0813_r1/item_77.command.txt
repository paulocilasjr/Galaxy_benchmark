/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os
for peaksfile in ['dnase/dnase_hotspots.merged.bed','dnase/dnase_peaks.merged.bed']:
    print('USING', peaksfile)
    for f in sorted(glob.glob('macs_peaks_q001/*_union.narrowPeak')):
        total=sum(1 for _ in open(f))
        p=subprocess.Popen(['bedtools','intersect','-a',f,'-b',peaksfile,'-wa','-u'], stdout=subprocess.PIPE)
        open_count=sum(1 for _ in p.stdout); p.wait()
        name=os.path.basename(f).replace('_union.narrowPeak','')
        print(f'{name:8s} peaks={total:6d} open={open_count:6d} open_frac={open_count/total:.3f} resistant={1-open_count/total:.3f}')
PY"
