/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os, concurrent.futures
files=sorted(glob.glob('chip.pioneer.q1/*.bed.gz'))
peaks=os.path.abspath('dnase/dnase_peaks.merged.bed')
out='midpoint_open_counts.tsv'
def run(f):
    p1=subprocess.Popen(['gzip','-cd',f], stdout=subprocess.PIPE)
    p2=subprocess.Popen(['awk','BEGIN{OFS=\"\\t\"}{mid=int(("'$2+$3)/2); print $1, mid, mid+1}'"'], stdin=p1.stdout, stdout=subprocess.PIPE)
    p1.stdout.close()
    p3=subprocess.Popen(['bedtools','intersect','-a','-','-b',peaks,'-wa','-u'], stdin=p2.stdout, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
    p2.stdout.close()
    count=0
    for _ in p3.stdout: count+=1
    p3.wait()
    return os.path.basename(f), count
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
    futs=[ex.submit(run,f) for f in files]
    results={}
    for fut in concurrent.futures.as_completed(futs):
        name,c=fut.result(); results[name]=c; print(name,c,flush=True)
with open(out,'w') as o:
    o.write('file\\topen_midpoints\\n')
    for f in files:
        name=os.path.basename(f); o.write(f'{name}\\t{results[name]}\\n')
print('done')
PY"
