/bin/bash -lc "python3 - <<'PY'
import pyBigWig, glob, os, statistics
bw=pyBigWig.open('dnase/ENCFF723WAJ.bigWig')
print('chroms sample', list(bw.chroms().items())[:3], 'len', len(bw.chroms()))
# check chr prefix
print('has chr1', 'chr1' in bw.chroms())
for f in sorted(glob.glob('macs_peaks_q001/*_union.narrowPeak')):
    vals=[]
    with open(f) as fh:
        for line in fh:
            p=line.rstrip().split('\\t')
            chrom=p[0]; start=int(p[1]); end=int(p[2])
            try:
                s=bw.stats(chrom,start,end,type='mean',exact=True)[0]
            except Exception as e:
                continue
            if s is not None: vals.append(s)
    if not vals: continue
    mean=statistics.mean(vals); med=statistics.median(vals)
    frac0=sum(v==0 for v in vals)/len(vals)
    frac_low=sum(v<1 for v in vals)/len(vals)
    name=os.path.basename(f).replace('_union.narrowPeak','')
    print(f'{name:8s} n={len(vals):6d} dnase_mean={mean:.4f} median={med:.4f} frac0={frac0:.3f} frac<1={frac_low:.3f}')
bw.close()
PY"
