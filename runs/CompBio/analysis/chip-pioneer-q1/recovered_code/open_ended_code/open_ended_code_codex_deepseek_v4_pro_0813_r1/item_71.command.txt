/bin/bash -lc "curl -s -L 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=sra&term=PRJNA910201&retmax=100' -o esearch2.xml && python3 - <<'PY'
import xml.etree.ElementTree as ET, requests, csv
root=ET.parse('esearch2.xml').getroot()
ids=[e.text for e in root.findall('.//Id')]
r=requests.post('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi',data={'db':'sra','id':','.join(ids),'rettype':'runinfo','retmode':'text'})
open('sra_runinfo_super.csv','w').write(r.text)
print(len(ids), r.text[:100])
PY
python3 - <<'PY'
import csv
rows=list(csv.DictReader(open('sra_runinfo_super.csv')))
for r in rows:
    if r['LibraryStrategy'] in ['ChIP-Seq','OTHER','MNase-seq','ATAC-seq','OTHER']:
        print(r['Run'], r['Sample'], r['spots'], r['LibraryStrategy'], r['LibraryName'], r['LibraryLayout'])
PY"
