rows=list(csv.DictReader(open('sra_runinfo.csv')))
spot={r['SampleName']:int(r['spots']) for r in rows}
constructs={

'FOXA1dbd': {'rep1':'GSM6806963','rep2':'GSM6806965','ctrl1':'GSM6806964','ctrl2':'GSM6806966'},
'FOXA1nhaa': {'rep1':'GSM6806967','rep2':'GSM6806969','ctrl1':'GSM6806968','ctrl2':'GSM6806970'},
'FOXA1rraa': {'rep1':'GSM6806971','rep2':'GSM6806973','ctrl1':'GSM6806972','ctrl2':'GSM6806974'},
'HNF4A': {'rep1':'GSM6806975','rep2':'GSM6806977','ctrl1':'GSM6806976','ctrl2':'GSM6806978'},
'H2B_t0': {'rep1':'GSM6806979','rep2':'GSM6806980','ctrl1':'GSM6806981','ctrl2':None},
'H2B_t6': {'rep1':'GSM6806982','rep2':'GSM6806983','ctrl1':'GSM6806984','ctrl2':None},
}
vec={
'TF1':[30262780,34212511,37039899,45007567],

'TF3':[15298277,17189525,23274203,13216643],

'TF5':[16135021,14755327,16452140,None],
'TF6':[8156838,5302061,16833124,None],
}
def ratios(a,b):
    vals=[]
    for x,y in zip(a,b):
        if x is None or y is None: continue
        vals.append(x/y)
    return vals
for tfi, vi in vec.items():
    best=[]
    for con, names in constructs.items():
        si=[spot[names['rep1']], spot[names['rep2']], spot[names['ctrl1']], spot[names.get('ctrl2')] if names.get('ctrl2') else None]
        r=ratios(vi,si)
        if not r: continue
        mean=sum(r)/len(r); cv=math.sqrt(sum((x-mean)**2 for x in r)/len(r))/mean if mean else 0
        best.append((cv,mean,con,r))
    best.sort()
    print(tfi)
    for cv,mean,con,r in best[:3]:
        print(' ',con,'cv',round(cv,3),'mean',round(mean,3),'ratios',[round(x,3) for x in r])
PY"
