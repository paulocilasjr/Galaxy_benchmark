/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os
peaks='dnase/dnase_peaks.merged.bed'
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    f1=f'macs_peaks_q001/{tf}_rep1_peaks.narrowPeak'; f2=f'macs_peaks_q001/{tf}_rep2_peaks.narrowPeak'
    # intersect rep1 with rep2, any overlap; output rep1 intervals
    p=subprocess.Popen(['bedtools','intersect','-a',f1,'-b',f2,'-wa','-u'],stdout=subprocess.PIPE)
    lines=p.stdout.readlines(); p.wait(); shared=len(lines)
    # write temp shared
    tmp=f'[REDACTED LOCAL PATH]'
    with open(tmp,'wb') as o: o.writelines(lines)
    # classify open
    p=subprocess.Popen(['bedtools','intersect','-a',tmp,'-b',peaks,'-wa','-u'],stdout=subprocess.PIPE)
    open_count=sum(1 for _ in p.stdout); p.wait()
    print(f'{tf:4s} shared={shared:6d} open={open_count:6d} open_frac={open_count/shared:.3f} resistant={1-open_count/shared:.3f}')
PY"
