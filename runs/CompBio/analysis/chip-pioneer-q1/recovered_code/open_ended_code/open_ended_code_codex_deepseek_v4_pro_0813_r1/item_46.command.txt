/bin/bash -lc "python3 - <<'PY'
import re, html
text=open('pmc_article.html',encoding='utf-8',errors='ignore').read()
# Extract text around positions where mutants and DNase-resistant occur
for m in re.finditer(r'(?i)(NHAA|RRAA|FOXA1-WT|DBD)', text):
    s=text[max(0,m.start()-1200):m.end()+2500]
    clean=re.sub('<["'^>]+>'"',' ',s); clean=html.unescape(clean); clean=re.sub(r'\\s+',' ',clean)
    if 'DNase' in clean or 'resistant' in clean or 'open chromatin' in clean or 'ChIP-seq' in clean:
        print('---', m.group(0), '---')
        print(clean[:3500])
        print()
PY"
