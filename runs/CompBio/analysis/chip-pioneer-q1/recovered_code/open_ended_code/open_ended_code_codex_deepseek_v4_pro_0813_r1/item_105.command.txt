/bin/bash -lc "python3 - <<'PY'
data={}
for line in open('fragment_dnase_means.tsv'):
 p=line.rstrip().split('\\t')
 if p[0]=='file': continue
 data[p[0]]=(int(p[1]),float(p[2]))
controls={'TF1':['TF1.control1.bed.gz','TF1.control2.bed.gz'],'TF2':['TF2.control1.bed.gz','TF2.control2.bed.gz'],'TF3':['TF3.control1.bed.gz','TF3.control2.bed.gz'],'TF4':['TF4.control1.bed.gz','TF4.control2.bed.gz'],'TF5':['TF5.control.bed.gz'],'TF6':['TF6.control.bed.gz']}
import math
print('TF\\tctrl_mean\\trep1\\tmean\\trep2\\tmean\\tavg_chip\\tlog2FC')
rows=[]
for tf,cs in controls.items():
    ctrl_mean=sum(data[c][1] for c in cs)/len(cs)
    r1=data[f'{tf}.rep1.bed.gz'][1]; r2=data[f'{tf}.rep2.bed.gz'][1]
    avg=(r1+r2)/2
    fc=math.log2(avg/ctrl_mean) if avg and ctrl_mean else float('nan')
    rows.append((tf,ctrl_mean,r1,r2,avg,fc))
for row in sorted(rows,key=lambda x:x[4]):
    tf,cm,r1,r2,avg,fc=row
    print(f'{tf:4s}\\t{cm:.5f}\\t{r1:.5f}\\t{r2:.5f}\\t{avg:.5f}\\t{fc:+.3f}')
PY"
