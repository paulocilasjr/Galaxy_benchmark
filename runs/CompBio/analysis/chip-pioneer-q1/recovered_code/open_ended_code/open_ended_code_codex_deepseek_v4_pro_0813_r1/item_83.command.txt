import glob, os, statistics
for f in sorted(glob.glob('macs_peaks_q001/*_rep*_peaks.narrowPeak')):
    scores=[]; qs=[]; fc=[]
    with open(f) as fh:
        for line in fh:
            p=line.rstrip().split('\\t')
            if len(p)<10: continue
            scores.append(float(p[6])); fc.append(float(p[8])); qs.append(float(p[6]))
    name=os.path.basename(f).replace('_peaks.narrowPeak','')
    print(f'{name:12s} n={len(scores):6d} sig_mean={statistics.mean(scores):.2f} sig_med={statistics.median(scores):.1f} fc_med={statistics.median(fc):.2f}')
PY"
