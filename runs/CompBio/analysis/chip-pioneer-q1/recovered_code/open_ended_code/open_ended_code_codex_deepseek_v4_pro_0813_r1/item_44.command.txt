/bin/bash -lc "python3 - <<'PY'
import re, html
text=open('pmc_article.html',encoding='utf-8',errors='ignore').read()
# strip tags around keywords snippets
for kw in ['DBD','NHAA','RRAA','64.5','67%','16.7']:
    print('\\n---',kw,'---')
    for m in re.finditer(re.escape(kw), text, re.I):
        s=text[max(0,m.start()-500):m.end()+800]
        s=re.sub('<["'^>]+>'"',' ',s)
        s=html.unescape(s)
        s=re.sub(r'\\s+',' ',s)
        print(s[:1000])
        break
PY"
