/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os, sys
files=sorted(glob.glob('chip.pioneer.q1/*.bed.gz'))
peaks='dnase/dnase_peaks.merged.bed'
out=open('open_counts.tsv','w')
out.write('file\\tlines\\topen_lines\\n')
for f in files:
    p1=subprocess.Popen(['gzip','-cd',f], stdout=subprocess.PIPE)
    p2=subprocess.Popen(['bedtools','intersect','-a','-','-b',peaks,'-wa','-u'], stdin=p1.stdout, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
    p1.stdout.close()
    count=0
    for line in p2.stdout:
        count+=1
    p2.wait()
    total=sum(1 for _ in subprocess.Popen(['gzip','-cd',f], stdout=subprocess.PIPE).stdout)
    out.write(f'{os.path.basename(f)}\\t{total}\\t{count}\\n')
    out.flush()
    print(os.path.basename(f), total, count, flush=True)
out.close()
PY
cat open_counts.tsv"
