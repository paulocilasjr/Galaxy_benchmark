/bin/bash -lc "python3 - <<'PY'
import subprocess, os
for peaksfile in ['dnase/dnase_peaks.merged.bed','dnase/dnase_hotspots.merged.bed']:
 print('USING', peaksfile)
 for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    f1=f'macs_peaks_q001/{tf}_rep1_peaks.narrowPeak'; f2=f'macs_peaks_q001/{tf}_rep2_peaks.narrowPeak'
    p=subprocess.Popen(['bedtools','intersect','-a',f1,'-b',f2,'-wa','-u'],stdout=subprocess.PIPE)
    lines=p.stdout.readlines(); p.wait(); shared=len(lines)
    tmp=f'[REDACTED LOCAL PATH]'; open(tmp,'wb').writelines(lines)
    p=subprocess.Popen(['bedtools','intersect','-a',tmp,'-b',peaksfile,'-wa','-u'],stdout=subprocess.PIPE)
    op=sum(1 for _ in p.stdout); p.wait()
    print(f'{tf:4s} shared={shared:6d} open={op:6d} resistant={1-op/shared:.3f}')
PY"
