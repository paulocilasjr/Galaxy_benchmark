/bin/bash -lc "python - <<'PY'
# derive differences from previous 1kb results by quick parse, using threshold average
from collections import defaultdict
S={'TF1':(['TF1.rep1','TF1.rep2'],['TF1.control1','TF1.control2']),'TF2':(['TF2.rep1','TF2.rep2'],['TF2.control1','TF2.control2']),'TF3':(['TF3.rep1','TF3.rep2'],['TF3.control1','TF3.control2']),'TF4':(['TF4.rep1','TF4.rep2'],['TF4.control1','TF4.control2']),'TF5':(['TF5.rep1','TF5.rep2'],['TF5.control']),'TF6':(['TF6.rep1','TF6.rep2'],['TF6.control'])}
D={}
with open('data/dnase_1kb.tsv') as f:
 next(f)
 for l in f:
  c,s,e,x,y=l.rstrip().split('\\t');D[(c,int(s))]=((float(x)+float(y))/2)
low={k for k,v in D.items() if v<=.05}
for tf,(cs,bs) in S.items():
 def load(xs):
  d=defaultdict(int);n=0
  for x in xs:
   for l in open('data/'+x+'.1kb.tsv'):
    c,s,e,v=l.rstrip().split('\\t');d[(c,int(s))]+=int(v);n+=int(v)
  return d,n
 c,cn=load(cs);b,bn=load(bs)
 cf=sum(c[k] for k in low)/cn;bf=sum(b[k] for k in low)/bn
 print(tf,cf,bf,'difference',cf-bf,'ratio',cf/bf)
PY"
