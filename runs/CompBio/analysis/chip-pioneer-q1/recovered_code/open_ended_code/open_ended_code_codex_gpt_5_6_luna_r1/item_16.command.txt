/bin/bash -lc 'set -e
for s in current001 current005 old; do
 case $s in
 current001) a=ENCFF812QOW.bed.gz; b=ENCFF407NXV.bed.gz;;
 current005) a=ENCFF530GZW.bed.gz; b=ENCFF350XAH.bed.gz;;
 old) a=ENCFF876DOW.bed.gz; b=ENCFF764MOY.bed.gz;;
 esac
 for x in a b; do eval f='"\\"'$$x; gzip -cd "$f" | awk '"'BEGIN{OFS=\"\\t\"} "'$1 !~ /''^#/ {print $1,$2,$3}'"' | sort -k1,1 -k2,2n > \""'${s}.${x}.bed"; done
 bedtools merge -i "${s}.a.bed" > "${s}.rep1.union.bed"
 bedtools merge -i "${s}.b.bed" > "${s}.rep2.union.bed"
 cat "${s}.a.bed" "${s}.b.bed" | sort -k1,1 -k2,2n | bedtools merge -i - > "${s}.union.bed"
 bedtools intersect -a "${s}.a.bed" -b "${s}.b.bed" -u | bedtools merge -i - > "${s}.intersection.bed"
 echo $s; wc -l "${s}"*.bed
 done'
