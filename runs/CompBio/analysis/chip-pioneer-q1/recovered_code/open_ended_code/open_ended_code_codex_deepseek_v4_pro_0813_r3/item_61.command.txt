/bin/bash -lc "python3 - <<'PY'
import glob, os, subprocess, math
reg='dnase/hotspots_union.bed'
for cutoff in [3,5,10,20,30,50,100]:
    print('q>=',cutoff)
    for f in sorted(glob.glob('macs/TF*_peaks.narrowPeak')):
        tf=os.path.basename(f).split('_')[0]
        lines=[]
        with open(f) as fh:
            for line in fh:
                p=line.rstrip().split('\\t')
                if len(p)<9: continue
                if not p[0].startswith('chr'): continue
                # main only quick
                chrom=p[0]
                if not (chrom in ['chr'+str(i) for i in range(1,23)]+['chrX','chrY','chrM']): continue
                q=float(p[8]) if p[8] not in ('.','-1') else 0
                if q>=cutoff:
                    lines.append('\\t'.join(p[:3]))
        if not lines:
            print(tf,'no peaks'); continue
        with open('[REDACTED LOCAL PATH]','w') as out: out.write('\\n'.join(lines)+'\\n')
        total=len(lines)
        acc=int(subprocess.check_output(f\"bedtools intersect -u -a [REDACTED LOCAL PATH] -b {reg} | wc -l\",shell=True,text=True).strip())
        print(f'{tf} n={total} closed={total-acc} frac={(total-acc)/total:.3f}', end=' | ')
    print()
PY"
