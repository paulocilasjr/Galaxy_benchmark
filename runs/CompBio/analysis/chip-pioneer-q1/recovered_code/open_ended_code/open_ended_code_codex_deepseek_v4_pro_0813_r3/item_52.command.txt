/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os, math
files=sorted(glob.glob('macs/TF*_peaks.narrowPeak'))
for regfile,regname in [('dnase/hotspots_union.bed','hotspots'),('dnase/peaks_union.bed','peaks')]:
    print('\\n',regname)
    for f in files:
        total=int(subprocess.check_output(f\"wc -l < '{f}'\",shell=True,text=True).strip())
        acc=int(subprocess.check_output(f\"bedtools intersect -u -a '{f}' -b '{regfile}' | wc -l\",shell=True,text=True).strip())
        closed=total-acc
        print(os.path.basename(f), 'total',total,'acc',acc,'closed',closed,'acc_frac',round(acc/total,4),'closed_frac',round(closed/total,4))
PY"
