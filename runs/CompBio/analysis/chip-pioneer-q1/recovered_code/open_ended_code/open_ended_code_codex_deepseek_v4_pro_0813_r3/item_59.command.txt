/bin/bash -lc "python3 - <<'PY'
import glob, os, subprocess
main_regex=\"chr([1-9]|1[0-9]|2[0-2]|X|Y|M)"'$"
for regfile,regname in [('"'dnase/hotspots_union.bed','hotspots'),('dnase/peaks_union.bed','peaks')]:
    print('\\nMain chromosomes only', regname)
    for f in sorted(glob.glob('macs/TF*_peaks.narrowPeak')):
        tf=os.path.basename(f).split('_')[0]
        # filter main using awk, temp
        subprocess.run(f\"awk -F '\\\\t' '"'$1 ~ /''^{main_regex}/'"' '{f}' > [REDACTED LOCAL PATH]", shell=True, check=True)
        total=int(subprocess.check_output(f\"wc -l < [REDACTED LOCAL PATH]",shell=True,text=True).strip())
        acc=int(subprocess.check_output(f\"bedtools intersect -u -a [REDACTED LOCAL PATH] -b '{regfile}' | wc -l\",shell=True,text=True).strip())
        closed=total-acc
        print(tf,'mainN',total,'acc',acc,'closed',closed,'closed_frac',round(closed/total,4),'acc_frac',round(acc/total,4))
PY"
