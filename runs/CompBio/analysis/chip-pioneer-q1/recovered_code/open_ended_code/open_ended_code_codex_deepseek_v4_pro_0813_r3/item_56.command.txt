/bin/bash -lc "python3 - <<'PY'
import glob, os, subprocess
for regfile,regname in [('dnase/hotspots_union.bed','hotspots'),('dnase/peaks_union.bed','peaks')]:
    print('\\nWeighted unique',regname)
    for f in sorted(glob.glob('macs/TF*_peaks.narrowPeak')):
        tf=os.path.basename(f).split('_')[0]
        acc_lines=subprocess.check_output(f\"bedtools intersect -wa -u -a '{f}' -b '{regfile}'\",shell=True,text=True).strip().splitlines()
        all_lines=[l for l in open(f) if l.strip()]
        def sums(lines):
            tot=len(lines); sig=0; q=0; p=0
            for line in lines:
                x=line.split('\\t')
                try: sig+=float(x[6])
                except: pass
                try: p+=float(x[7])
                except: pass
                try:
                    qv=float(x[8])
                    if qv>0: q+=qv
                except: pass
            return tot,sig,p,q
        atot,asig,ap,aq=sums(all_lines)
        btot,bsig,bp,bq=sums(acc_lines)
        ct=atot-btot; cs=asig-bsig; cp=ap-bp; cq=aq-bq
        print(tf, 'N',atot,'closedN',ct,'fracN',round(ct/atot,3),'sigClosed',round(cs/asig,3) if asig else 0,'pClosed',round(cp/ap,3) if ap else 0,'qClosed',round(cq/aq,3) if aq else 0)
PY"
