/bin/bash -lc 'mkdir -p macs/logs
run_tf() {
  tf=$1; shift
  echo "start $tf $(date)"
  macs3 callpeak -t "$@" -c "$2" -f BEDPE -g hs --keep-dup all -q 0.05 --outdir macs -n $tf --call-summits > macs/logs/${tf}.log 2>&1
  echo "done $tf $(date)"
}
run_tf TF1 chip.pioneer.q1/TF1.rep1.bed.gz chip.pioneer.q1/TF1.rep2.bed.gz chip.pioneer.q1/TF1.control1.bed.gz chip.pioneer.q1/TF1.control2.bed.gz &
pid1=$!
run_tf TF2 chip.pioneer.q1/TF2.rep1.bed.gz chip.pioneer.q1/TF2.rep2.bed.gz chip.pioneer.q1/TF2.control1.bed.gz chip.pioneer.q1/TF2.control2.bed.gz &
pid2=$!
run_tf TF3 chip.pioneer.q1/TF3.rep1.bed.gz chip.pioneer.q1/TF3.rep2.bed.gz chip.pioneer.q1/TF3.control1.bed.gz chip.pioneer.q1/TF3.control2.bed.gz &
pid3=$!
run_tf TF4 chip.pioneer.q1/TF4.rep1.bed.gz chip.pioneer.q1/TF4.rep2.bed.gz chip.pioneer.q1/TF4.control1.bed.gz chip.pioneer.q1/TF4.control2.bed.gz &
pid4=$!
run_tf TF5 chip.pioneer.q1/TF5.rep1.bed.gz chip.pioneer.q1/TF5.rep2.bed.gz chip.pioneer.q1/TF5.control.bed.gz &
pid5=$!
wait $pid1 $pid2 $pid3 $pid4 $pid5
echo ALLDONE'
