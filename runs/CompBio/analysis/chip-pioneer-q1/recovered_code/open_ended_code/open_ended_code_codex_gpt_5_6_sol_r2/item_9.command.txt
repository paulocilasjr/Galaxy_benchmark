/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('ENCSR000EME.json'))
for f in j['files']:
 if f.get('status')=='released' and f.get('assembly')=='GRCh38' and f.get('file_format') in ('bed','bigWig'):
  av=f.get('analysis_step_version') or {}
  step=av.get('analysis_step',{}) if isinstance(av,dict) else {}
  pipes=step.get('pipelines',[])
  pipe=pipes[0].get('accession') if pipes and isinstance(pipes[0],dict) else ''
  print(f\"{f.get('accession')}\\t{f.get('file_format')}\\t{f.get('file_format_type')}\\t{f.get('output_type')}\\trep={','.join(map(str,f.get('biological_replicates',[])))}\\tdefault={f.get('preferred_default')}\\tdate={f.get('date_created','')[:10]}\\tpipe={pipe}\\tsize={f.get('file_size')}\")
PY"
