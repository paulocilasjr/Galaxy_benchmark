/bin/bash -lc "python3 - <<'PY'
import subprocess
from collections import Counter, defaultdict
from pathlib import Path

def mapreads(fn):
    p=subprocess.run(['minimap2','-a','-x','sr','--secondary=no','klf4_ref.fa',fn],capture_output=True,text=True,check=True)
    out=[]
    for line in p.stdout.splitlines():
        if not line or line[0]=='@': continue
        f=line.split('\\t'); q=int(f[0].split('_')[1]); flag=int(f[1]); pos=None
        if f[2]"'!='"'*': pos=int(f[3]);
        nm=None
        for x in f[11:]:
            if x.startswith('NM:i:'): nm=int(x[5:])
        out.append((q,pos,'-' if flag&16 else '+',nm,f[5] if pos else '*'))
    return out
for fn,ranges in [('exogenous.mix.reads.q2.exo.fq',[(1,395,'common'),(396,592,'suffix')]),('exogenous.mix.reads.q2.mix.fq',[(1,592,'common'),(593,691,'suffix')])]:
    a=mapreads(fn); print('\\n',fn,len(a))
    for lo,hi,label in ranges:
        x=[r for r in a if lo<=r[0]<=hi]; mapped=[r for r in x if r[1] is not None]
        bins=Counter((r[1]-1)//100*100+1 for r in mapped)
        nm=Counter(r[3] for r in mapped)
        print(label,'n',len(x),'mapped',len(mapped),'unmapped',len(x)-len(mapped),'positions',min([r[1] for r in mapped],default=None),max([r[1] for r in mapped],default=None),'NM',nm)
        print(' bins',' '.join(f'{k}:{bins[k]}' for k in sorted(bins)))
PY"
