import subprocess
from collections import Counter, defaultdict
from pathlib import Path
ref=''.join(Path('klf4_ref.fa').read_text().splitlines()[1:])
def call(fn):

 groups=defaultdict(Counter); sub=defaultdict(Counter); total=Counter()
 for line in p.stdout.splitlines():
  if line.startswith('@'): continue
  f=line.split('\\t'); q=int(f[0].split('_')[1]);
  if f[2]=='*': continue
  pos=int(f[3])-1; seq=f[9]
  g='common' if q<=395 else 'suffix'
  for i,(a,b) in enumerate(zip(seq,ref[pos:pos+len(seq)])):
   total[g]+=1
   if a"'!=b: sub[g][f'"'{b}>{a}@{pos+i+1}']+=1
 return total,sub
for fn in ['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']:
 total,sub=call(fn)
 print('\\n',fn)
 for g in total:
  print(g,'bases',total[g],'mismatches',sum(sub[g].values()),'top',sub[g].most_common(30))
PY"
