/bin/bash -lc "for f in exogenous.mix.reads.q2.exo.fq exogenous.mix.reads.q2.endo.fq exogenous.mix.reads.q2.mix.fq; do
  awk 'NR%4==2{print}' \""'$f" | sort | uniq -c | sort -nr > "$f.seqcounts.txt"
  awk '"'NR%4==2{n++; lens[length("'$0)]++} END{printf "%s reads=%d unique_lengths=", FILENAME,n; for(k in lens) printf "%s:%s ",k,lens[k]; print ""}'"' \""'$f"
done
join -j 2 <(awk '"'{print "'$1,$2}'"' exogenous.mix.reads.q2.exo.fq.seqcounts.txt | sort -k2,2) <(awk '{print "'$1,$2}'"' exogenous.mix.reads.q2.endo.fq.seqcounts.txt | sort -k2,2) | awk '{shared++; exo+="'$2; endo+=$3} END{print "exo-endo shared unique",shared,"occurrences",exo,endo}'"'
join -j 2 <(awk '{print "'$1,$2}'"' exogenous.mix.reads.q2.exo.fq.seqcounts.txt | sort -k2,2) <(awk '{print "'$1,$2}'"' exogenous.mix.reads.q2.mix.fq.seqcounts.txt | sort -k2,2) | awk '{shared++; exo+="'$2; mix+=$3} END{print "exo-mix shared unique",shared,"occurrences",exo,mix}'"'
join -j 2 <(awk '{print "'$1,$2}'"' exogenous.mix.reads.q2.endo.fq.seqcounts.txt | sort -k2,2) <(awk '{print "'$1,$2}'"' exogenous.mix.reads.q2.mix.fq.seqcounts.txt | sort -k2,2) | awk '{shared++; endo+="'$2; mix+=$3} END{print "endo-mix shared unique",shared,"occurrences",endo,mix}'"'
md5 exogenous.mix.reads.q2.*.fq"
