import sys,subprocess
from collections import Counter
import numpy as np
from scipy.stats import chi2_contingency
p=sys.argv[1]
def get(f,lo=0,hi=10**9):
 out=subprocess.check_output(['samtools','view',f],text=True); c=Counter()
 for x in out.splitlines():
  n=int(x.split('\\t')[0].split('_')[1])
  if lo<=n<=hi:c[(int(x.split('\\t')[3])-1)//100]+=1
 return c
endo=get(f'{p}/exogenous.mix.reads.q2.endo.sorted.bam')
exo_tail=get(f'{p}/exogenous.mix.reads.q2.exo.sorted.bam',396,592)
mix_tail=get(f'{p}/exogenous.mix.reads.q2.mix.sorted.bam',593,691)
keys=sorted(set(endo)|set(exo_tail)|set(mix_tail))
for name,c in [('exo_tail',exo_tail),('mix_tail',mix_tail)]:
 # compare normalized profiles 2-row contingency
 tab=np.array([[endo[k] for k in keys],[c[k] for k in keys]])
 print(name,'total',sum(c.values()),'chi2 p',chi2_contingency(tab)[1], 'ratios',[(k,endo[k],c[k]) for k in keys])
