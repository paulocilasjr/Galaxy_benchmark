/bin/bash -lc 'tmpdir=$(mktemp -d ./refcheck.XXXXXX)
curl -fsSL '"'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text' -o \""'$tmpdir/KLF4_NM_004235.6.fa"
head -n 2 "$tmpdir/KLF4_NM_004235.6.fa"; awk '"'NR>1{n+=length("'$0)}END{print "length="n}'"' \""'$tmpdir/KLF4_NM_004235.6.fa"
# use minimap2 to map q2 references and report map stats
for f in exogenous.mix.reads.q2.endo.fq exogenous.mix.reads.q2.exo.fq exogenous.mix.reads.q2.mix.fq; do minimap2 -a -x sr "$tmpdir/KLF4_NM_004235.6.fa" "$f" 2>/dev/null | samtools view -F 4 -c | awk -v f="$f" '"'{print f,\"mapped=\" "'$1}'"'; done"
