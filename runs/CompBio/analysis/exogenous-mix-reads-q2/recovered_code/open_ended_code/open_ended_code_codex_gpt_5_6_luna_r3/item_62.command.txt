/bin/bash -lc 'tmpdir=$(find . -maxdepth 1 -type d -name '"'hfcheck.*' | head -n 1); for pair in 'exo.fq endo.fq' 'exo.fq mix.fq' 'endo.fq mix.fq'; do set -- "'$pair; a="$tmpdir/$1";b="$tmpdir/$2"; echo "$(basename "$a") vs $(basename "$b")"; comm -12 <(awk '"'NR%4==2' \""'$a" | sort -u) <(awk '"'NR%4==2' \""'$b" | sort -u) | wc -l; done
# q1 exo unique sequence positions distribution versus endo overlap; q2 too
for k in exo endo mix; do echo -n "q1 $k "; awk '"'NR%4==2' \""'$tmpdir/$k.fq" | sort -u | wc -l; done'
