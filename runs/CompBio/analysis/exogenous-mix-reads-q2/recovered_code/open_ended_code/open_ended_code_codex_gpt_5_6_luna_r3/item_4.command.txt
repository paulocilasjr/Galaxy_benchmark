/bin/bash -lc 'for f in exogenous.mix.reads.q2.*.fq; do b=${f%.fq}; awk '"'NR%4==2' \""'$f" | sort > "$b.seq.sorted"; done
for f in exogenous.mix.reads.q2.*.seq.sorted; do echo "$(basename "$f") $(wc -l < "$f") lines, $(uniq "$f" | wc -l) unique"; done
comm -12 exogenous.mix.reads.q2.exo.seq.sorted exogenous.mix.reads.q2.endo.seq.sorted | wc -l | awk '"'{print \"exo∩endo unique shared=\" "'$1}'"'
comm -12 exogenous.mix.reads.q2.exo.seq.sorted exogenous.mix.reads.q2.mix.seq.sorted | wc -l | awk '{print \"exo∩mix unique shared=\" "'$1}'"'
comm -12 exogenous.mix.reads.q2.endo.seq.sorted exogenous.mix.reads.q2.mix.seq.sorted | wc -l | awk '{print \"endo∩mix unique shared=\" "'$1}'"'
echo 'exo-only sequence counts:'; comm -23 exogenous.mix.reads.q2.exo.seq.sorted exogenous.mix.reads.q2.endo.seq.sorted | uniq -c | sort -nr | head
echo 'endo-only sequence counts:'; comm -13 exogenous.mix.reads.q2.exo.seq.sorted exogenous.mix.reads.q2.endo.seq.sorted | uniq -c | sort -nr | head"
