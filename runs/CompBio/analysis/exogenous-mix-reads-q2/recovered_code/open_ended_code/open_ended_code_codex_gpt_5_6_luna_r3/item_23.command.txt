/bin/bash -lc 'tmpdir=$(find . -maxdepth 1 -type d -name '"'hfcheck.*' | head -n 1); for pair in 'exo.fq mix.fq' 'endo.fq mix.fq'; do set -- "'$pair; a="$tmpdir/$1"; b="$tmpdir/$2"; n=$(awk '"'END{print NR/4}' \""'$a"); echo "$(basename "$a") vs $(basename "$b")"; diff -q "$a" <(head -n $((n*4)) "$b") >/dev/null; echo "exact prefix status=$?"; done
# tail membership of mix beyond exo using line extraction
awk '"'NR%4==2' \""'$tmpdir/exo.fq" | sort -u > "$tmpdir/exoset"
awk '"'NR%4==2' \""'$tmpdir/endo.fq" | sort -u > "$tmpdir/endoset"
exo_n=$(awk '"'END{print NR/4}' \""'$tmpdir/exo.fq"); tail_n=$((1965-exo_n)); awk '"'NR%4==2' \""'$tmpdir/mix.fq" | tail -n "$tail_n" > "$tmpdir/mixtail"
for s in exoset endoset; do awk '"'NR==FNR{a["'$0]=1;next}{if($0 in a) i++;else o++}END{print FILENAME,"in="i+0,"out="o+0}'"' \""'$tmpdir/$s" "$tmpdir/mixtail"; done'
