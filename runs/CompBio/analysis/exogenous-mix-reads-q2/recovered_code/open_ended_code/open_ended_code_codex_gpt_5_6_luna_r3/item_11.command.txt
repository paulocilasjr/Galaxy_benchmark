/bin/bash -lc 'tmpdir=$(mktemp -d)
awk '"'NR%4==2' exogenous.mix.reads.q2.exo.fq | sort -u > \""'$tmpdir/exo"
awk '"'NR%4==2' exogenous.mix.reads.q2.endo.fq | sort -u > \""'$tmpdir/endo"
awk '"'NR%4==2' exogenous.mix.reads.q2.mix.fq | tail -n 99 > \""'$tmpdir/mix_tail"
echo '"'mix tail 99 membership'
awk 'NR==FNR{a["'$0]=1;next} {if($0 in a) inx++; else out++} END{print "in exo=" inx+0, "not in exo=" out+0}'"' \""'$tmpdir/exo" "$tmpdir/mix_tail"
awk '"'NR==FNR{a["'$0]=1;next} {if($0 in a) inx++; else out++} END{print "in endo=" inx+0, "not in endo=" out+0}'"' \""'$tmpdir/endo" "$tmpdir/mix_tail"
echo '"'tail first/last'; head -n 5 \""'$tmpdir/mix_tail"; tail -n 5 "$tmpdir/mix_tail"
rm -r "$tmpdir"'
