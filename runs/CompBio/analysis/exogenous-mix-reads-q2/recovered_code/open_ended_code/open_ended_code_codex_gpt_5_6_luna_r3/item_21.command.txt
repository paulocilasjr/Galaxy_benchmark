/bin/bash -lc 'tmpdir=$(mktemp -d ./hfcheck.XXXXXX)
for kind in exo endo mix; do url="https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/data/exogenous.mix.reads.q1.${kind}.fq?download=true"; curl -fsSL "$url" -o "$tmpdir/${kind}.fq" || echo "download failed $kind"; done
ls -lh "$tmpdir"
for f in "$tmpdir"/*.fq; do echo "$(basename "$f") reads=$(awk '"'END{print NR/4}' \""'$f")"; done'
