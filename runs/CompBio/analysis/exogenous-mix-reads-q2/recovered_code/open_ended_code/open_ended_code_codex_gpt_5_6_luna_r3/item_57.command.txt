/bin/bash -lc 'tmpdir=$(find . -maxdepth 1 -type d -name '"'refcheck.*' | head -n 1); python - \""'$tmpdir" <<'"'PY'
import sys, subprocess
from collections import Counter
import numpy as np
p=sys.argv[1]
for prefix in ('exogenous.mix.reads.q2','q1'):
 names={k:(f'{p}/{prefix}.{k}.sorted.bam' if prefix.startswith('exogenous') else f'{p}/q1_{k}.sorted.bam') for k in ('endo','exo','mix')}
 cs={}
 for k,f in names.items():
  out=subprocess.check_output(['samtools','view',f],text=True)
  cs[k]=Counter((int(x.split('\\t')[3])-1)//100 for x in out.splitlines())
 keys=sorted(set().union(*[set(c) for c in cs.values()]))
 D=np.array([[cs['exo'][x],cs['endo'][x]] for x in keys],float);M=np.array([cs['mix'][x] for x in keys],float)
 coef,resid,rank,s=np.linalg.lstsq(D,M,rcond=None)
 pred=D@coef
 print(prefix,'coeff exo,endo',coef,'sum',coef.sum(),'fraction',coef[0]/coef.sum(),'resid norm',np.linalg.norm(M-pred),'rank',rank)
 print('M total',M.sum(),'pred total',pred.sum())
PY"
