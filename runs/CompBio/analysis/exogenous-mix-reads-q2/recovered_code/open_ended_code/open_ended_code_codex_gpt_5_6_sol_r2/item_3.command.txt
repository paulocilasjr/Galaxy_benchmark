/bin/bash -lc 'for f in exogenous.mix.reads.q2.exo.fq exogenous.mix.reads.q2.endo.fq exogenous.mix.reads.q2.mix.fq; do echo "$f"; awk '"'NR%4==1{first=("'$0); last=$0; n++} END{print n, first, last}'"' \""'$f"; done
awk '"'NR%4==2' exogenous.mix.reads.q2.exo.fq | sort > exo.seq
awk 'NR%4==2' exogenous.mix.reads.q2.endo.fq | sort > endo.seq
awk 'NR%4==2' exogenous.mix.reads.q2.mix.fq | sort > mix.seq
printf 'unique counts\\n'; uniq exo.seq | wc -l; uniq endo.seq | wc -l; uniq mix.seq | wc -l
printf 'exact intersection line counts via comm\\n'; comm -12 exo.seq endo.seq | wc -l; comm -12 exo.seq mix.seq | wc -l; comm -12 endo.seq mix.seq | wc -l
printf 'subset/differences unique\\n'; comm -23 mix.seq exo.seq | wc -l; comm -23 mix.seq endo.seq | wc -l
md5 exogenous.mix.reads.q2.*.fq *.seq"
