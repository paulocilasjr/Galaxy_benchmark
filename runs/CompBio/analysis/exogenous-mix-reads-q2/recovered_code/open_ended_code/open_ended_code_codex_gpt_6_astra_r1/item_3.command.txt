/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter
D={}
for p in Path('.').glob('*.fq'):
 ls=p.read_text().splitlines(); reads=[ls[i+1] for i in range(0,len(ls),4)]; c=Counter(reads);D[p.name.split('.')[-2]]=c
 print(p.name,'reads',len(reads),'unique',len(c),'lengths',Counter(map(len,reads)),'multiplicities',Counter(c.values()))
 print('top',c.most_common(5))
for a in D:
 for b in D:
  if a<b:print(a,b,'shared seqs',len(D[a].keys()&D[b].keys()),'shared counts',sum((D[a]&D[b]).values()))
for s in set().union(*D.values()):
 print(*(D[k][s] for k in ['endo','exo','mix']),s)
PY"
