/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter
D={}
for k in ['endo','exo','mix']:
 l=Path(f'exogenous.mix.reads.q2.{k}.fq').read_text().splitlines();D[k]=[tuple(l[i:i+4]) for i in range(0,len(l),4)]
for a,b in [('endo','exo'),('exo','mix')]:
 print(a,b,'exact prefix',D[b][:len(D[a])]==D[a], 'same seq order',all(x[1]==y[1] for x,y in zip(D[a],D[b])))
segments=[D['endo'],D['exo'][len(D['endo']):],D['mix'][len(D['exo']):]]
rc=lambda s:s.translate(str.maketrans('ACGT','TGCA'))[::-1]
for i,seg in enumerate(segments):
 km=Counter(min(s[j:j+21],rc(s[j:j+21])) for r in seg for s in [r[1]] for j in range(30))
 print('block',i,'reads',len(seg),'GC',sum(r[1].count('G')+r[1].count('C') for r in seg)/(50*len(seg)), 'unique kmers',len(km))
PY"
