/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict
# load fasta
seq=''.join(l.strip() for l in open('NM_004235_KLF4.fasta') if not l.startswith('>')).upper()
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
# read mix seqs
seqs=[]
with open('exogenous.mix.reads.q2.mix.fq') as fh:
    for i,l in enumerate(fh):
        if i%4==1: seqs.append(l.strip().upper())

def bestpos(r):
    p=seq.find(r)
    if p>=0: return ('+',p,50)
    rr=rc(r); p=seq.find(rr)
    if p>=0: return ('-',p,50)
    # hamming best window both strands
    best=(None,None,999)
    for strand,x in [('+',r),('-',rr)]:
        for i in range(len(seq)-50+1):
            d=sum(a"'!=b for a,b in zip(x,seq[i:i+50]))
            if d<best[2]: best=(strand,i,d)
    return best
for name,lo,hi in [('"'common',0,395),('exo_tail',395,592),('mix_tail',592,691)]:
    hits=[]; mism=[]
    for r in seqs[lo:hi]:
        strand,p,d=bestpos(r)
        hits.append(p); mism.append(d if d"'!=50 else 0)
    print(name, len(hits), '"'exact', sum(d==50 for d in mism), 'mism_count', Counter(mism).most_common())
    print('pos min max', min(hits), max(hits), 'median', sorted(hits)[len(hits)//2])
    bins=Counter((p//100)*100 for p in hits)
    print('bins', sorted(bins.items())[:10], '...', sorted(bins.items())[-10:])
# Actually d variable is 50 for exact returned above; fix detailed exact positions
print('first few positions by block')
for lo,hi in [(0,5),(390,400),(590,600),(686,691)]:
    for i in range(lo,hi):
        print(i+1,bestpos(seqs[i]),seqs[i][:10])
PY"
