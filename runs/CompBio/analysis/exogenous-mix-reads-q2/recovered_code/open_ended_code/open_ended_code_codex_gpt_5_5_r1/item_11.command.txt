/bin/bash -lc "python3 - <<'PY'
# For consecutive reads in mix, find max suffix-prefix overlap and orientation overlap.
seqs=[]
with open('exogenous.mix.reads.q2.mix.fq') as fh:
    for i,line in enumerate(fh):
        if i%4==1: seqs.append(line.strip())

def ov(a,b):
    m=0
    maxl=min(len(a),len(b))
    for k in range(1,maxl+1):
        if a[-k:]==b[:k]: m=k
    return m
for start,end in [(0,20),(385,405),(585,605),(670,690)]:
    print('range',start+1,end)
    print([ov(seqs[i],seqs[i+1]) for i in range(start,end-1)])
# find if consecutive reads overlap with shifts likely positions
from collections import Counter
allov=[ov(seqs[i],seqs[i+1]) for i in range(len(seqs)-1)]
print('overlap counts top', Counter(allov).most_common(20))
print('zero examples', [i+1 for i,o in enumerate(allov) if o==0][:20])
PY"
