/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict
ref=''.join(l.strip() for l in open('NM_004235_KLF4.fasta') if not l.startswith('>')).upper()
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
seqs=[]
with open('exogenous.mix.reads.q2.mix.fq') as fh:
    for i,l in enumerate(fh):
        if i%4==1: seqs.append(l.strip().upper())
def align(r):
    candidates=[]
    for strand,x in [('+',r),('-',rc(r))]:
        p=ref.find(x)
        if p>=0: return strand,p,x,0
        # collect best approx
        for i in range(len(ref)-50+1):
            d=sum(a"'!=b for a,b in zip(x,ref[i:i+50]))
            candidates.append((d,strand,i,x))
    d,strand,p,x=min(candidates)
    return strand,p,x,d
for name,lo,hi in [('"'common',0,395),('exo_tail',395,592),('mix_tail',592,691)]:
    var=Counter(); total=0; mismatch_reads=0
    for r in seqs[lo:hi]:
        strand,p,x,d=align(r)
        if d:
            mismatch_reads+=1
            for j,(a,b) in enumerate(zip(x,ref[p:p+50])):
                if a"'!=b:
                    var[(p+j,b,a)]+=1; total+=1
    print('"'\\n',name,'mismatch_reads',mismatch_reads,'mismatches',total)
    for k,c in var.most_common(20): print(k,c)
PY"
