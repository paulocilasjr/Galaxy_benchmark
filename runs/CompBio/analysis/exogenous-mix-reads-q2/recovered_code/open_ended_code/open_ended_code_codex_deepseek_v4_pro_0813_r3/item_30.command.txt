/bin/bash -c "python3 - <<'PY'
from pathlib import Path
import statistics
f='exogenous.mix.reads.q2.mix.fq'
lines=Path(f).read_text().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
gcs=[(s.count('G')+s.count('C'))/50 for s in seqs]
# print average gc in blocks of 50/100
for block in range(0,len(gcs),50):
    chunk=gcs[block:block+50]
    if chunk: print(block+1, block+len(chunk), round(statistics.mean(chunk),3))
PY"
