/bin/bash -c "python3 - <<'PY'
from pathlib import Path
mrna=''.join(Path('klf4_mrna.fa').read_text().splitlines()[1:]).upper()
print('mrna len',len(mrna))
def rc(s):
 return s.translate(str.maketrans('ACGT','TGCA'))[::-1]
for f in sorted(Path('.').glob('exogenous*.fq')):
    seqs=[l for l in f.read_text().splitlines()[1::4]]
    pos=[]
    no=[]
    for s in seqs:
        p=mrna.find(s); q=mrna.find(rc(s))
        if p>=0: pos.append((p,'+'))
        elif q>=0: pos.append((q,'-'))
        else: no.append(s)
    print(f.name,'mapped exact',len(pos),'unmapped',len(no))
    if no[:3]: print(no[:3])
    from collections import Counter
    pc=Counter(p for p,strand in pos)
    print('range',min(p for p,_ in pos) if pos else None,max(p for p,_ in pos) if pos else None, 'top positions',pc.most_common(10))
PY"
