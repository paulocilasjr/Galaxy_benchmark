/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
import statistics

def positions(f,start=0,end=None):
    lines=Path(f).read_text().splitlines(); pos=[]
    for line in lines:
        if line.startswith('@'): continue
        p=line.split('\\t'); pos.append(int(p[3]))
    return pos[start:end]
exo=positions('exogenous.mix.reads.q2.exo.sam')
mix=positions('exogenous.mix.reads.q2.mix.sam')
endo=positions('exogenous.mix.reads.q2.endo.sam')
groups={'endo(1-395)':endo,'exo_extra(396-592)':exo[395:],'mix_extra(593-691)':mix[592:],'exo_all':exo,'mix_all':mix}
for name,ps in groups.items():
    hist=Counter((x-1)//100 for x in ps)
    print('\\n',name,'n',len(ps),'mean',statistics.mean(ps),'minmax',min(ps),max(ps))
    print([(k*100+1,hist.get(k,0)) for k in range(29)])
PY"
