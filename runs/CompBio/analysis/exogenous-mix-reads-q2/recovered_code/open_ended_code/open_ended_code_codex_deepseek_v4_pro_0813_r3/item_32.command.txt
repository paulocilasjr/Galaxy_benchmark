/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
from scipy.stats import chisquare, fisher_exact
# parse pos groups
def posgroups():
    f='exogenous.mix.reads.q2.mix.sam'; mix=[int(l.split('\\t')[3]) for l in Path(f).read_text().splitlines() if not l.startswith('@')]
    e=Path('exogenous.mix.reads.q2.endo.sam'); endo=[int(l.split('\\t')[3]) for l in e.read_text().splitlines() if not l.startswith('@')]
    exo=[int(l.split('\\t')[3]) for l in Path('exogenous.mix.reads.q2.exo.sam').read_text().splitlines() if not l.startswith('@')]
    return endo, exo[395:], mix[592:]
endo,exo_extra,mix_extra=posgroups()
for name,a,b in [('endo vs exo_extra',endo,exo_extra),('endo vs mix_extra',endo,mix_extra),('exo_extra vs mix_extra',exo_extra,mix_extra)]:
    ca=Counter((x-1)//100 for x in a); cb=Counter((x-1)//100 for x in b)
    bins=sorted(set(ca)|set(cb))
    fa=[ca.get(b,0) for b in bins]; fb=[cb.get(b,0) for b in bins]
    # chi square with normalization: compare proportions
    totala=sum(fa); totalb=sum(fb)
    expected=[totala*(fa[i]+fb[i])/(totala+totalb) for i in range(len(bins))]
    # too sparse but ok
    chi,p=chisquare(fa,f_exp=expected)
    print(name,'chi2',round(chi,2),'p',p)
PY"
