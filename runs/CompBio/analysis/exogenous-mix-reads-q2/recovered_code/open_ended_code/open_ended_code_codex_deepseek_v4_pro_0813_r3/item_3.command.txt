/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
files=['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']
sets={}
counts={}
for f in files:
    lines=Path(f).read_text().splitlines()
    seqs=[lines[i+1] for i in range(0,len(lines),4) if lines[i].startswith('@')]
    sets[f]=set(seqs); counts[f]=Counter(seqs)
    print(f, len(seqs), len(set(seqs)), min(map(len,seqs)), max(map(len,seqs)))
exo,endo,mix=[sets[f] for f in files]
print('exo&endo',len(exo&endo),'exo&mix',len(exo&mix),'endo&mix',len(endo&mix),'all',len(exo&endo&mix))
print('mix only',len(mix-exo-endo))
# counts of read occurrences by origin categories
mixc=counts[files[2]]
onlymix=sum(mixc[s] for s in mix-exo-endo)
both=sum(mixc[s] for s in exo&endo)
exoonly=sum(mixc[s] for s in exo-endo)
endoonly=sum(mixc[s] for s in endo-exo)
print('mix read counts categories: exo-only',exoonly,'endo-only',endoonly,'both',both,'neither',onlymix)
# duplicates?
for f in files:
 print(f, Counter(counts[f].values()))
PY"
