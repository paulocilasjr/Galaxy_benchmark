/bin/bash -c "python3 - <<'PY'
from pathlib import Path
import statistics
lines=Path('exogenous.mix.reads.q2.mix.fq').read_text().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
endo=seqs[:395]; exo_extra=seqs[395:592]; mix_extra=seqs[592:691]

def ham(a,b): return sum(x"'!=y for x,y in zip(a,b))
# compute min hamming distance from each mix_extra to endo and exo_extra sets (brute 99*592*50 ~3M ok)
for group_name,group in [('"'endo',endo),('exo_extra',exo_extra)]:
    dists=[]
    for s in mix_extra:
        d=min(ham(s,t) for t in group)
        dists.append(d)
    print(group_name,'mean min dist',statistics.mean(dists),'hist', {d:dists.count(d) for d in sorted(set(dists))})
# distances among groups
for name,group in [('endo',endo),('exo_extra',exo_extra),('mix_extra',mix_extra)]:
    d=min(ham(s,t) for i,s in enumerate(group) for j,t in enumerate(group) if i<j) if len(group)>1 else None
    print(name,'min internal',d)
PY"
