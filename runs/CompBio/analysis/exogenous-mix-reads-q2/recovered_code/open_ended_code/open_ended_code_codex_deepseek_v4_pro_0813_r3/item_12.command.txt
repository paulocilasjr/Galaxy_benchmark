/bin/bash -c "python3 - <<'PY'
import re
from pathlib import Path
from collections import Counter,defaultdict
for f in ['exogenous.mix.reads.q2.exo.sam','exogenous.mix.reads.q2.endo.sam','exogenous.mix.reads.q2.mix.sam']:
    mism=[]
    for line in Path(f).read_text().splitlines():
        if line.startswith('@'): continue
        p=line.split('\\t'); flag=int(p[1])
        if flag & 0x900: continue
        seq=p[9]; pos=int(p[3]); cigar=p[5]
        nm=int(re.search(r'NM:i:(\\d+)',line).group(1))
        md=re.search(r'MD:Z:(\\S+)',line).group(1) if nm else ''
        if cigar=='50M' and nm:
            refpos=pos; seqpos=0; j=0
            while j<len(md):
                c=md[j]
                if c.isdigit():
                    n=''
                    while j<len(md) and md[j].isdigit(): n+=md[j]; j+=1
                    n=int(n); refpos+=n; seqpos+=n
                elif c=='"'^'"':
                    j+=1
                    while j<len(md) and md[j].isalpha(): j+=1
                else:
                    mism.append((refpos,c,seq[seqpos] if seqpos<len(seq) else '?',flag))
                    refpos+=1; seqpos+=1; j+=1
    print('\\n',f)
    print('total mismatches',len(mism))
    print('positions',Counter([m[0] for m in mism]).most_common(30))
    for m in mism[:20]: print(m)
PY"
