/bin/bash -c "python3 - <<'PY'
from pathlib import Path
import statistics
f='exogenous.mix.reads.q2.mix.fq'; lines=Path(f).read_text().splitlines(); quals=[lines[i+3] for i in range(0,len(lines),4)]
for block in range(0,len(quals),50):
    chunk=quals[block:block+50]
    if chunk:
        mean=statistics.mean(ord(c)-33 for q in chunk for c in q)
        print(block+1,block+len(chunk),round(mean,2))
PY"
