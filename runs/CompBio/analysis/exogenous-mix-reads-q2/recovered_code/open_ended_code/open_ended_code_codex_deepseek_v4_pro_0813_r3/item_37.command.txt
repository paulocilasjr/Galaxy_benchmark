/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
lines=Path('exogenous.mix.reads.q2.mix.fq').read_text().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
groups={'endo':seqs[:395], 'exo_extra':seqs[395:592], 'mix_extra':seqs[592:691]}
for k in [10,15,20,25,30]:
    print('k',k)
    for name,ss in groups.items():
        c=Counter()
        for s in ss:
            for i in range(0,50-k+1): c[s[i:i+k]]+=1
        # k-mers present only in this group (among mix groups)
        others=Counter()
        for n2,s2 in groups.items():
            if n2==name: continue
            for s in s2:
                for i in range(0,50-k+1): others[s[i:i+k]]+=1
        uniq=[(km,count) for km,count in c.items() if km not in others]
        print(name,'total',len(c),'uniq',len(uniq),'top',uniq[:5])
PY"
