/bin/bash -c "python3 - <<'PY'
import urllib.request
ids=['NM_001314052.2','NM_004235.6']
seqs={}
for id in ids:
    url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id={id}&rettype=fasta&retmode=text'
    text=urllib.request.urlopen(url,timeout=30).read().decode()
    seqs[id]=''.join(text.splitlines()[1:]).upper()
    open(id+'.fa','w').write(text)
    print(id,len(seqs[id]),text.splitlines()[0])
# compare, find shared
a,b=seqs.values()
print('len',len(a),len(b))
# longest common prefix/suffix? rough
import difflib
# exact common substrings length 50 unique
PY"
