/bin/bash -c "python3 - <<'PY'
from pathlib import Path
mrna=''.join(Path('klf4_mrna.fa').read_text().splitlines()[1:]).upper()
# find start codon ATG and stop in CDS? RefSeq annotation via GenBank maybe.
# Use efetch genbank for features.
import urllib.request,re
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=gb&retmode=text'
text=urllib.request.urlopen(url).read().decode()
open('klf4_mrna.gb','w').write(text)
# print features CDS and exon
for line in text.splitlines():
    if line.startswith('     CDS') or line.startswith('     exon') or line.startswith('     mRNA'):
        print(line)
# parse CDS location
for m in re.finditer(r'CDS\\s+(\\d+)\\.\\.(\\d+)',text): print(m.group(0))
PY"
