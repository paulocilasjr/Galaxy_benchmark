/bin/bash -c "python3 - <<'PY'
import urllib.request,re
# esearch gene KLF4 human refseq mRNA
query='Homo sapiens[orgn] KLF4[gene] AND biomol_mrna[properties]'
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=nuccore&term='+urllib.parse.quote(query)+'&retmax=100&retmode=json'
import urllib.parse,json
data=json.load(urllib.request.urlopen(url))
ids=data['esearchresult']['idlist']
print(ids)
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=nuccore&id='+','.join(ids)+'&retmode=json'
summ=json.load(urllib.request.urlopen(url))
for uid in ids:
    r=summ['result'][uid]
    print(uid,r.get('caption'),r.get('title'))
PY"
