/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
seqs={}
for f in ['NM_001314052.2.fa','NM_004235.6.fa']:
    txt=Path(f).read_text().splitlines(); seqs[f]=''.join(txt[1:]).upper()
def rc(s): return s.translate(str.maketrans('ACGT','TGCA'))[::-1]
def class_read(s):
    # exact match original or rc in each
    m1=seqs['NM_001314052.2.fa'].find(s)>=0 or seqs['NM_001314052.2.fa'].find(rc(s))>=0
    m2=seqs['NM_004235.6.fa'].find(s)>=0 or seqs['NM_004235.6.fa'].find(rc(s))>=0
    if m1 and m2: return 'common'
    if m1: return 'var1'
    if m2: return 'var2'
    return 'neither'
for f in ['exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.mix.fq']:
    lines=Path(f).read_text().splitlines(); seqlist=[lines[i+1] for i in range(0,len(lines),4)]
    c=Counter(class_read(s) for s in seqlist)
    print(f, c)
# group within mix by prefix
lines=Path('exogenous.mix.reads.q2.mix.fq').read_text().splitlines(); seqlist=[lines[i+1] for i in range(0,len(lines),4)]
for name,inds in [('endo',range(0,395)),('exo_extra',range(395,592)),('mix_extra',range(592,691))]:
    c=Counter(class_read(seqlist[i]) for i in inds)
    print(name,c)
PY"
