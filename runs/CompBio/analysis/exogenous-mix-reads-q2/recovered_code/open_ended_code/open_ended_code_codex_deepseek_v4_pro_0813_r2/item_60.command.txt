/bin/bash -lc "python3 - <<'PY'
def load(f):
 lines=open(f).read().splitlines(); return set(lines[i+1] for i in range(0,len(lines),4))
sets={}
for q in ['q1','q2']:
 for t in ['exo','endo','mix']:
  sets[f'{q}{t}']=load(f'exogenous.mix.reads.{q}.{t}.fq')
pairs=[('q2exo','q1exo'),('q2exo','q1mix'),('q2mix','q1mix'),('q2mix','q1exo'),('q2exo','q1endo')]
for a,b in pairs:
 A=sets[a];B=sets[b]
 print(a,b,'A',len(A),'B',len(B),'intersection',len(A&B),'A in B',len(A-B))
# Is q2exo subset of q1endo plus some q1exo? 
print('q2exo overlap q1endo',len(sets['q2exo']&sets['q1endo']))
print('q2exo overlap q1exo',len(sets['q2exo']&sets['q1exo']))
print('q2mix extra not q2exo overlap q1endo/q1exo/q1mix')
extra=sets['q2mix']-sets['q2exo']
print('extra',len(extra),'in q1endo',len(extra&sets['q1endo']),'in q1exo',len(extra&sets['q1exo']),'in q1mix',len(extra&sets['q1mix']))
PY"
