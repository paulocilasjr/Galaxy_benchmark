            bestham=h; bestpos=i; strand='"'+'; mism=[(i+j,a,b) for j,(a,b) in enumerate(zip(s,t)) if a"'!=b]

    for i in range(len(ref)-50+1):
        t=ref[i:i+50]
        h=sum(a!=b for a,b in zip(r,t))
        if h<bestham:
            bestham=h; bestpos=i; strand='"'-'; mism=[(i+j,a,b) for j,(a,b) in enumerate(zip(r,t)) if a"'!=b]
