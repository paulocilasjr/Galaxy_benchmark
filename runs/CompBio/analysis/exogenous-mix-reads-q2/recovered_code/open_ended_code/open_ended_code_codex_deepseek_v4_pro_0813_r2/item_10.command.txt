/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict
ref=''.join(open('klf4_nm.fa').read().splitlines()[1:])
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s[::-1].translate(comp)
files=['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']
# precompute all ref 50-mers
kmers={}
for i in range(len(ref)-50+1):
    kmers.setdefault(ref[i:i+50],[]).append(i)
# unique sequences all
allseq=set()
for f in files:
    lines=open(f).read().splitlines(); allseq.update(lines[i+1] for i in range(0,len(lines),4))
best={}
for s in allseq:
    # fwd
    bestham=99; bestpos=None; strand='+'
    for i in range(len(ref)-50+1):
        t=ref[i:i+50]
        h=sum(a"'!=b for a,b in zip(s,t))
        if h<bestham:
            bestham=h; bestpos=i; strand='"'+'
        if bestham==0: break
    # rc
    r=rc(s)
    for i in range(len(ref)-50+1):
        t=ref[i:i+50]
        h=sum(a"'!=b for a,b in zip(r,t))
        if h<bestham:
            bestham=h; bestpos=i; strand='"'-'
        if bestham==0: break
    best[s]=(bestham,bestpos,strand)
# output mismatch patterns by file
for f in files:
    lines=open(f).read().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
    c=Counter(seqs)
    byham=Counter()
    for s,n in c.items():
        byham[best[s][0]]+=n
    print(f, 'hamming dist read counts',byham)
    # print reads hamming>0 with positions
    print('mismatch examples')
    shown=0
    for s in sorted(c, key=lambda x: best[x][0], reverse=True):
        ham,pos,strand=best[s]
        if ham>0 and shown<20:
            print(ham,pos,strand,s, 'rc' if strand=='-' else '')
            shown+=1
PY"
