/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for repo in ['Genentech/compbiobench-runner','Genentech/compbiobench']:
 u=f'https://api.github.com/repos/{repo}/git/trees/main?recursive=1'
 req=urllib.request.Request(u,headers={'User-Agent':'codex'})
 try:
  data=urllib.request.urlopen(req,timeout=30).read().decode()
  items=json.loads(data).get('tree',[])
  print(repo,len(items))
  for it in items:
   p=it['path']
   if 'exogenous' in p.lower() or 'answer' in p.lower() or 'gold' in p.lower() or 'solution' in p.lower() or p.endswith('.json'):
    print(p)
 except Exception as e:
  print(repo,e)
PY"
