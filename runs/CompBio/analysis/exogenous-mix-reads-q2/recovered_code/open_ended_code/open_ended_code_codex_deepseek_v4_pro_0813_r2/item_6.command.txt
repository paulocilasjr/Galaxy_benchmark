/bin/bash -lc "python3 - <<'PY'
from collections import Counter
ref=''.join(open('klf4_nm.fa').read().splitlines()[1:])
files=['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']
for f in files:
    seqs=[open(f).read().splitlines()[i+1] for i in range(0,len(open(f).read().splitlines()),4)]
    exact=0; rc=0; miss=0; ex_miss=[]
    for s in seqs:
        if s in ref: exact+=1
        elif s[::-1].translate(str.maketrans('ACGT','TGCA')) in ref: rc+=1
        else:
            miss+=1
            if len(ex_miss)<5: ex_miss.append(s)
    print(f,'exact fwd',exact,'revcomp',rc,'miss',miss,'sample misses',ex_miss)
    # positions of exact matches
    positions=[]
    for s in seqs:
        if s in ref:
            positions.append(ref.find(s))
    print('positions range',min(positions),max(positions),'n unique',len(set(positions)))
PY"
