/bin/bash -lc "python3 - <<'PY'
from collections import Counter
files=['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']
sets={}; counts={}
for f in files:
    lines=open(f).read().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
    counts[f]=Counter(seqs); sets[f]=set(seqs)
exo,endo,mix=files
print('read counts', {f:sum(counts[f].values()) for f in files})
print('uniq', {f:len(sets[f]) for f in files})
print('exo unique uniq in endo', len(sets[exo]&sets[endo]))
print('exo uniq not endo', len(sets[exo]-sets[endo]))
print('endo uniq not exo', len(sets[endo]-sets[exo]))
print('mix uniq in exo only', len(sets[mix]&(sets[exo]-sets[endo])))
print('mix uniq in endo only', len(sets[mix]&(sets[endo]-sets[exo])))
print('mix uniq in both', len(sets[mix]&sets[exo]&sets[endo]))
print('mix uniq neither', len(sets[mix]-sets[exo]-sets[endo]))
# reads counts classification
mix_c=counts[mix]
exo_only_reads=sum(mix_c[s] for s in sets[exo]-sets[endo])
endo_only_reads=sum(mix_c[s] for s in sets[endo]-sets[exo])
both_reads=sum(mix_c[s] for s in sets[exo]&sets[endo])
neither=sum(mix_c[s] for s in sets[mix]-sets[exo]-sets[endo])
print('mix read classification exo_only, endo_only, both, neither',exo_only_reads,endo_only_reads,both_reads,neither)
# sequence content nucleotide maybe exo-specific mismatches
for s in list(sets[exo]-sets[endo])[:20]: print('EXOONLY',s)
for s in list(sets[endo]-sets[exo])[:20]: print('ENDOONLY',s)
PY"
