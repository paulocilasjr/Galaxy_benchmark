/bin/bash -lc "python3 - <<'PY'
from collections import Counter
ref=''.join(open('klf4_nm.fa').read().splitlines()[1:])
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s[::-1].translate(comp)
def get(s):
 if s in ref: return ref.find(s),'+'
 r=rc(s)
 if r in ref: return ref.find(r),'-'
 # approx
 for i in range(len(ref)-50+1):
  if sum(a"'!=b for a,b in zip(s,ref[i:i+50]))<=1:return i,'"'+*'
 for i in range(len(ref)-50+1):
  if sum(a"'!=b for a,b in zip(r,ref[i:i+50]))<=1:return i,'"'-*'
 return None
def load(f):
 lines=open(f).read().splitlines(); return [lines[i+1] for i in range(0,len(lines),4)]
exo=load('exogenous.mix.reads.q2.exo.fq'); mix=load('exogenous.mix.reads.q2.mix.fq'); extra=mix[592:]
for label,arr in [('exo',exo),('extra',extra)]:
 ps=[get(s) for s in arr]
 strands=Counter(p[1] if p else None for p in ps)
 print(label,'strand counts',strands)
 # distribution of positions in bins
 bins=Counter((p[0]//200 if p else None) for p in ps)
 print(label,'pos bins',sorted(bins.items()))
PY"
