from pathlib import Path
from collections import defaultdict, Counter
import numpy as np
seq=''.join(l.strip() for l in open('klf4_nm.fa') if not l.startswith('>'))

 for j,orig in enumerate(s):
  for ch in 'ACGT':
   if ch==orig:continue
   t=s[:j]+ch+s[j+1:]
   if t in pos:
    for lab,i in pos[t]:

 return best

def region(i):
 if i+50<=cds[0]: return '5UTR'

 if i<cds[0] and i+50>cds[0]: return 'span5'
 if i<cds[1] and i+50>cds[1]: return 'span3'

 c=Counter(); strand=Counter(); coords=[]

   if not h:break
   s=f.readline().strip(); f.readline(); f.readline()

 print(name,dict(c),'strand',dict(strand),'coord mean',np.mean(coords),'min/max',min(coords),max(coords))

 c=Counter(region(i) for lab,i in records[sl]); print(grp,dict(c))
