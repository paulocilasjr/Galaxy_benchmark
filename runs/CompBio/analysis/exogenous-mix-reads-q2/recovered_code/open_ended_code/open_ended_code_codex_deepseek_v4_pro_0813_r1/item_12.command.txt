from collections import Counter, defaultdict

 b=None
 for lab,t in [('+',seq),('-',rc)]:
  for i in range(len(t)-len(s)+1):
   mism=sum(a"'!=b for a,b in zip(s,t[i:i+len(s)]))
   if mism<=1:
    if b is None or mism<b[0]: b=(mism,lab,i)

C={}; S={}
for name in ['"'exo','endo','mix']:
 arr=[]
 c=Counter()
 with open(f'exogenous.mix.reads.q2.{name}.fq') as f:
  while True:
   h=f.readline()
   if not h: break
   s=f.readline().strip(); f.readline(); f.readline(); arr.append(s); c[s]+=1
 C[name]=c; S[name]=set(c)
 # position distribution unique
 posc=Counter()
 for s in S[name]:
  b=best(s)
  if b: posc[(b[1],b[2])]+=1

   print(lab,'unique positions n',len(xs),'min',min(xs),'max',max(xs),'median',sorted(xs)[len(xs)//2])

 b=best(s)
 if not b: continue
 cat='exo_only' if s in S['exo'] and s not in S['endo'] else 'shared' if s in S['exo'] and s in S['endo'] else 'neither'

for lab in ['+','-']:
