/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
sets={}
counts={}
for name in ['exo','endo','mix']:
    p=Path(f'exogenous.mix.reads.q2.{name}.fq')
    c=Counter()
    with p.open() as f:
        while True:
            h=f.readline()
            if not h: break
            s=f.readline().strip(); f.readline(); f.readline()
            c[s]+=1
    counts[name]=c
    sets[name]=set(c)
    print(name, 'total',sum(c.values()),'unique',len(c),'topdup',c.most_common(3))
for a in ['exo','endo','mix']:
  for b in ['exo','endo','mix']:
    if a<b:
      inter=len(sets[a]&sets[b])
      reads_a=sum(v for k,v in counts[a].items() if k in sets[b])
      reads_b=sum(v for k,v in counts[b].items() if k in sets[a])
      print(a,b,'unique_intersection',inter,'a_reads_in_b',reads_a,'b_reads_in_a',reads_b)
PY"
