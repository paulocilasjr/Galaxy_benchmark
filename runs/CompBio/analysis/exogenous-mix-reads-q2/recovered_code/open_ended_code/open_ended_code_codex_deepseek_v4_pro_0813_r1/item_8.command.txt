/bin/bash -c "python3 - <<'PY'
import urllib.request
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text'
try:
    data=urllib.request.urlopen(url, timeout=30).read().decode()
    print(data[:1000])
    open('klf4_nm.fa','w').write(data)
except Exception as e:
    print('ERR',e)
PY
head -n 2 klf4_nm.fa; wc -c klf4_nm.fa"
