/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import defaultdict
seq=''.join(l.strip() for l in open('klf4_nm.fa') if not l.startswith('>'))
comp=str.maketrans('ACGT','TGCA'); rc=seq.translate(comp)[::-1]
pos=defaultdict(list)
for lab,t in [('+',seq),('-',rc)]:
 for i in range(len(t)-50+1): pos[t[i:i+50]].append((lab,i))
def bestpos(s):
 if s in pos:return min(pos[s])
 best=None
 for j,orig in enumerate(s):
  for ch in 'ACGT':
   if ch==orig:continue
   t=s[:j]+ch+s[j+1:]
   if t in pos:
    for lab,i in pos[t]:
     if best is None or i<best[1] or (i==best[1] and lab<best[0]):best=(lab,i)
 return best
cds=(597,2037)
def reg(i):
 if i+50<=cds[0]:return '5UTR'
 if i>=cds[1]:return '3UTR'
 if i<cds[0] and i+50>cds[0]:return 'span5'
 if i<cds[1] and i+50>cds[1]:return 'span3'
 return 'CDS'
for name in ['exo','endo','mix']:
 print('\\n---',name)
 n=0
 with open(f'exogenous.mix.reads.q2.{name}.fq') as f:
  while True:
   h=f.readline()
   if not h:break
   s=f.readline().strip(); f.readline(); f.readline()
   lab,i=bestpos(s); r=reg(i)
   if r in ('5UTR','3UTR') and n<8:
    n+=1
    print(r,'read',s,'align',lab,i,'NMseg',seq[i:i+50] if lab=='+' else rc[i:i+50])
PY"
