/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
# read files
files=['exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q1.mix.fq','exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.mix.fq']
C={}
for f in files:
 c=Counter()
 with open(f) as fh:
  while True:
   h=fh.readline()
   if not h:break
   s=fh.readline().strip(); fh.readline(); fh.readline(); c[s]+=1
 C[f]=c
S_exo=set(C['exogenous.mix.reads.q1.exo.fq']); S_endo=set(C['exogenous.mix.reads.q1.endo.fq'])
def marker_counts(name):
 c=C[name]
 exo_only=sum(n for s,n in c.items() if s in S_exo and s not in S_endo)
 shared=sum(n for s,n in c.items() if s in S_exo and s in S_endo)
 endo_only=sum(n for s,n in c.items() if s in S_endo and s not in S_exo)
 neither=sum(n for s,n in c.items() if s not in S_exo and s not in S_endo)
 total=sum(c.values())
 return total,exo_only,shared,endo_only,neither
for name in files:
 total,eo,sh,en,ne=marker_counts(name)
 print(name,total,'exo_only',eo,eo/total,'shared',sh,sh/total,'endo_only',en,en/total,'neither',ne,ne/total)
PY"
