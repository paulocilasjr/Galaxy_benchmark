/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter, defaultdict
C={}
S={}
for name in ['exo','endo','mix']:
 c=Counter()
 with open(f'exogenous.mix.reads.q2.{name}.fq') as f:
  while True:
   h=f.readline()
   if not h: break
   s=f.readline().strip(); f.readline(); f.readline()
   c[s]+=1
 C[name]=c; S[name]=set(c)
print('exo unique total',len(S['exo']),'endo',len(S['endo']),'mix',len(S['mix']))
print('endo subset exo?',S['endo']<=S['exo'])
print('mix has all exo?',S['exo']<=S['mix'],'all endo?',S['endo']<=S['mix'])
print('exo only unique',len(S['exo']-S['endo']),'mix only',len(S['mix']-S['exo']-S['endo']))
for name in ['exo','endo','mix']:
 d=defaultdict(int)
 for s,n in C[name].items():
  cat='exo_only' if s in S['exo'] and s not in S['endo'] else 'shared' if s in S['exo'] and s in S['endo'] else 'endo_only' if s in S['endo'] else 'neither'
  d[cat]+=n
 print(name,dict(d))
# counts by exact match in mix to reference
print('mix category based on seq membership')
d=defaultdict(int)
for s,n in C['mix'].items():
 cat='exo_only' if s in S['exo'] and s not in S['endo'] else 'shared' if s in S['exo'] and s in S['endo'] else 'endo_only' if s in S['endo'] and s not in S['exo'] else 'neither'
 d[cat]+=n
print(dict(d))
PY"
