/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
host='https://sep2025.rest.ensembl.org'
for (name,start,end) in [('naive',7686990,7687589),('strand_aware',7687391,7687990)]:
    url=f'{host}/sequence/region/human/17:{start}..{end}:1?'
    req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
    try:
        resp=json.load(urllib.request.urlopen(req, timeout=30))
    except Exception as e:
        print(name, 'ERR', e); continue
    seq=resp['seq']
    print(name, start, end, len(seq), 'gc', sum(1 for c in seq.upper() if c in 'GC'), 'fraction', round(sum(1 for c in seq.upper() if c in 'GC')/len(seq),2), seq[:50])
PY"
