/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for repo in ['Genentech/compbiobench-results-v1','Genentech/compbiobench-submissions-v1']:
    url=f'https://huggingface.co/api/datasets/{repo}/tree/main'
    req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    try:
        data=json.load(urllib.request.urlopen(req, timeout=30))
        print('REPO',repo, 'entries',len(data))
        for x in data[:20]: print(x)
    except Exception as e: print('ERR', repo, e)
PY"
