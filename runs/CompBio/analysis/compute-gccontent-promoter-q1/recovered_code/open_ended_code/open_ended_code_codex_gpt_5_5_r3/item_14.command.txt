/bin/bash -lc "python3 - <<'PY'
import gzip
fa='Homo_sapiens.GRCh38.dna.chromosome.17.fa.gz'
seq=[]
with gzip.open(fa,'rt') as f:
    for line in f:
        if line.startswith('>'):
            continue
        seq.append(line.strip().upper())
s=''.join(seq)
# ENST00000269305 REST coords: chr17 7668421-7687490 strand -; TSS=end for minus.
tss=7687490
up=500
down=100
start=tss-down+1
end=tss+up
prom=s[start-1:end]
print(len(s), start, end, len(prom), prom.count('G')+prom.count('C'), (prom.count('G')+prom.count('C'))/len(prom))
print(prom[:20], prom[-20:])
PY"
