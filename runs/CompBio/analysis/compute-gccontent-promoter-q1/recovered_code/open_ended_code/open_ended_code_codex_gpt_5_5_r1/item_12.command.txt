/bin/bash -lc "curl -L --fail --silent --show-error -o Homo_sapiens.GRCh38.dna.chromosome.17.fa.gz https://ftp.ensembl.org/pub/release-115/fasta/homo_sapiens/dna/Homo_sapiens.GRCh38.dna.chromosome.17.fa.gz
python3 - <<'PY'
import gzip
fa='Homo_sapiens.GRCh38.dna.chromosome.17.fa.gz'
start=7687391
end=7687990
seq_parts=[]
pos=0
with gzip.open(fa,'rt') as fh:
    for line in fh:
        if line.startswith('>'):
            continue
        s=line.strip()
        if not s:
            continue
        line_start=pos+1
        line_end=pos+len(s)
        if line_end >= start and line_start <= end:
            a=max(start,line_start)-line_start
            b=min(end,line_end)-line_start+1
            seq_parts.append(s[a:b])
        pos=line_end
seq=''.join(seq_parts).upper()
gc=sum(1 for b in seq if b in 'GC')
print(len(seq), gc, gc/len(seq), f'{gc/len(seq):.2f}')
print(seq[:20], seq[-20:])
PY"
