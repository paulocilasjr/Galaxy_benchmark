/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
server='https://sep2025.rest.ensembl.org'
# lookup transcript
lookup=json.load(urllib.request.urlopen(server+'/lookup/id/ENST00000269305?content-type=application/json;expand=0'))
print(lookup)
chrom=lookup['seq_region_name']; strand=lookup['strand']; tss=lookup['end'] if strand==-1 else lookup['start']
up=500; down=100
if strand==1:
    start=tss-up; end=tss+down-1
else:
    start=tss-down+1; end=tss+up
region=f'{chrom}:{start}..{end}:{strand}'
print('tss',tss,'region',region,'len',end-start+1)
# fetch fasta
req=urllib.request.Request(server+f'/sequence/region/human/{region}?content-type=text/plain')
seq=urllib.request.urlopen(req).read().decode().strip().upper()
print(len(seq), seq[:50], seq[-50:])
gc=sum(1 for b in seq if b in 'GC')
print(chrom,start,end,'-' if strand==-1 else '+',len(seq),gc,round(gc/len(seq),2), f'{gc/len(seq):.2f}')
PY"
