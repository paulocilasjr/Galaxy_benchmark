/bin/bash -lc 'set -euo pipefail
cd refs/refseq
for f in $(curl -Ls https://ftp.ncbi.nlm.nih.gov/refseq/H_sapiens/mRNA_Prot/ | sed -n '"'s/.*href=\"\\(human\\.[0-9][0-9]*\\.rna\\.fna\\.gz\\)\".*/\\1/p' | sort -V); do curl -L --fail --retry 3 -O https://ftp.ncbi.nlm.nih.gov/refseq/H_sapiens/mRNA_Prot/"'$f; done
for f in $(curl -Ls https://ftp.ncbi.nlm.nih.gov/refseq/M_musculus/mRNA_Prot/ | sed -n '"'s/.*href=\"\\(mouse\\.[0-9][0-9]*\\.rna\\.fna\\.gz\\)\".*/\\1/p' | sort -V); do curl -L --fail --retry 3 -O https://ftp.ncbi.nlm.nih.gov/refseq/M_musculus/mRNA_Prot/"'$f; done
for f in $(curl -Ls https://ftp.ncbi.nlm.nih.gov/refseq/S_scrofa/mRNA_Prot/ | sed -n '"'s/.*href=\"\\(pig\\.[0-9][0-9]*\\.rna\\.fna\\.gz\\)\".*/\\1/p' | sort -V); do curl -L --fail --retry 3 -O https://ftp.ncbi.nlm.nih.gov/refseq/S_scrofa/mRNA_Prot/"'$f; done
ls -lh'
