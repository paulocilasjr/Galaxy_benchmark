/bin/bash -lc 'set -euo pipefail
out=refs/refseq/three_species_refseq_rna.v1.fa.gz
if [ -e "$out" ]; then echo "combined reference already exists: $out" >&2; exit 1; fi
for sp in human mouse pig; do
  for f in refs/refseq/${sp}.*.rna.fna.gz; do
    SP=$sp gzip -cd "$f" | SP=$sp perl -pe '"'if(/"'^>/){s/''^>('"\\S+)/>"'$ENV{SP}|$1/}'"' | gzip -c >> \""'$out"
  done
done
gzip -t "$out"
ls -lh "$out"'
