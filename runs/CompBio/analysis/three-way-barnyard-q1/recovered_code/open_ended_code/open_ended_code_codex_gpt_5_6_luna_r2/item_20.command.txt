/bin/bash -lc "awk 'BEGIN{FS=\"\\t\"} {c["'$2,$3]++; total[$2]++; if($3=="human") h[$2]++; else if($3=="mouse") m[$2]++; else if($3=="pig") p[$2]++} END{for(b in total){a=h[b]+0; z=m[b]+0; y=p[b]+0; if(a>=z && a>=y){sp="human"; mx=a; sec=(z>y?z:y)} else if(z>=a && z>=y){sp="mouse"; mx=z; sec=(a>y?a:y)} else {sp="pig"; mx=y; sec=(a>z?a:z)} print b,a,z,y,total[b],sp,mx,sec,mx-sec}}'"' read_calls.tsv | sort -k6,6 -k9,9nr > cell_calls.tsv; echo '--- cell species counts ---'; cut -f6 cell_calls.tsv | sort | uniq -c; echo '--- calls total ---'; awk '{a+="'$2;m+=$3;p+=$4;t+=$5} END{print a,m,p,t}'"' cell_calls.tsv; echo '--- top ambiguous ---'; sort -k9,9n cell_calls.tsv | head -20; echo '--- barcode count ---'; wc -l cell_calls.tsv"
