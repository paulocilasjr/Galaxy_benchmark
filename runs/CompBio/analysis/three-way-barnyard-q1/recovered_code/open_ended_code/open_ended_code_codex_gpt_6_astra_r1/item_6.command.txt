/bin/bash -lc "mkdir -p references; python3 - <<'PY'
import concurrent.futures,urllib.request,time
refs={'human':('homo_sapiens','Homo_sapiens.GRCh38'),'mouse':('mus_musculus','Mus_musculus.GRCm39'),'pig':('sus_scrofa','Sus_scrofa.Sscrofa11.1')}
def get(item):
 s,(d,f)=item;url=f'https://ftp.ensembl.org/pub/release-115/fasta/{d}/cdna/{f}.cdna.all.fa.gz';t=time.time()
 urllib.request.urlretrieve(url,f'references/{s}.fa.gz');print(s,'downloaded',round(time.time()-t,1),flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as e:list(e.map(get,refs.items()))
PY"
