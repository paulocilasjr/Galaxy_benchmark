/bin/bash -lc "python3 - <<'PY'
import urllib.request,re
for d in ['homo_sapiens','mus_musculus','sus_scrofa']:
 url=f'https://ftp.ensembl.org/pub/release-113/fasta/{d}/cdna/'
 try:
  data=urllib.request.urlopen(url,timeout=20).read().decode('utf-8','ignore')
  print('\\n',d)
  for line in data.splitlines():
   if 'href' in line and '.fa.gz' in line:
    print(re.sub('<["'^>]+>'"','',line).strip())
 except Exception as e: print(d,'ERR',e)
PY"
