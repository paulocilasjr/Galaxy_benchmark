for q,(asv,sp) in best.items():
    b=bc.get(q)
    if b and sp in ('hg','mm','ss'):
        cc[b][sp]+=1
assign=[]
for b,c in cc.items():
    sp=c.most_common(1)[0][0]
    assign.append(sp)
