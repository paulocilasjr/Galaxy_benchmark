/bin/bash -lc "python3 - <<'PY'
import collections, sys
# load bc
bc={}
for line in open('work/id_bc.tsv'):
    p=line.rstrip().split('\\t'); bc[p[0]]=p[1]
cc=collections.defaultdict(lambda: collections.Counter())
# primary only, mapq>=0
for line in open('work/aln.sam'):
    if line.startswith('@'): continue
    f=line.rstrip().split('\\t')
    if len(f)<12: continue
    flag=int(f[1])
    if flag & 0x4: continue
    if flag & 0x800: continue
    if flag & 0x100: continue
    sp=f[2].split('|')[0]
    b=bc.get(f[0])
    if b and sp in ('hg','mm','ss'): cc[b][sp]+=1
# assignment and margins
assign=[]
for b,c in cc.items():
    total=sum(c.values())
    tops=c.most_common(3)
    first=tops[0] if tops else ('none',0)
    second=tops[1] if len(tops)>1 else ('none',0)
    margin=first[1]-second[1]
    frac=first[1]/total if total else 0
    assign.append((b, first[0], first[1], second[0], second[1], margin, frac, total, dict(c)))
# sort by margin
assign_sorted=sorted(assign, key=lambda x:x[5])
print('n cells', len(assign))
print('counts by assigned', collections.Counter(a[1] for a in assign))
print('worst margins:')
for a in assign_sorted[:30]:
    print(a)
print('best margins ambiguous maybe')
PY"
