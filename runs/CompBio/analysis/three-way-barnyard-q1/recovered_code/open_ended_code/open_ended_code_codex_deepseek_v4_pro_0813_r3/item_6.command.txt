/bin/bash -c "python3 - <<'PY'
import gzip, collections
cnt=collections.Counter(); n=0
with gzip.open('three.way.barnyard.q1.R1.fq.gz','rt') as f:
    for i,line in enumerate(f):
        if i%4==1:
            seq=line.strip(); b=seq[:16]
            cnt[b]+=1; n+=1
print('reads',n,'unique_barcodes',len(cnt),'max',cnt.most_common(5),'median',sorted(cnt.values())[len(cnt)//2])
PY"
