/bin/bash -c "python3 - <<'PY'
rows=[]
for line in open('cell_species_counts.tsv'):
    if line.startswith('barcode'): continue
    b,h,m,p=line.split(); rows.append((b,int(h),int(m),int(p)))
import numpy as np
arr=np.array([r[1:] for r in rows],float)
frac=arr/arr.sum(axis=1,keepdims=True)
# max fraction
mx=frac.max(axis=1)
print('min maxfrac',mx.min(), 'mean maxfrac',mx.mean(), 'p10',np.percentile(mx,10), 'median',np.median(mx))
# fraction cells with maxfrac>0.8, >0.9, >0.95
for t in [0.5,0.7,0.8,0.9,0.95,0.99]:
    print(t,(mx>t).sum())
# show species purity per assigned group
for sp,idx in [('human',0),('mouse',1),('pig',2)]:
    sub=arr[arr.argmax(axis=1)==idx]
    fr=sub/sub.sum(axis=1,keepdims=True)
    print(sp,'cells',len(sub),'mean fracs',fr.mean(axis=0),'min self',fr[:,idx].min())
PY"
