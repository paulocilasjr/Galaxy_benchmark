/bin/bash -lc "python3 - <<'PY'
import numpy as np
for name in ['dist_1kg','dist_hgdp']:
    with open(name+'.mdist.id') as f: lines=[line.split() for line in f]
    iids=[x[1] for x in lines]
    rows=[]
    with open(name+'.mdist') as f:
        for line in f: rows.append([float(x) for x in line.split()])
    print(name,'ids',len(iids),'rows',len(rows))
    for sid in ['Sample1','Sample2','Sample3','Sample4']:
        i=iids.index(sid)
        # row i lower
        dist=[]
        for j,v in enumerate(rows[i]): dist.append((iids[j],v))
        for j in range(i+1,len(iids)):
            dist.append((iids[j], rows[j][i]))
        dist=np.array([d[1] for d in dist]); idx=np.argsort(dist)
        print('\\n',sid, 'top 10')
        for k in idx[:10]: print(iids[k], round(dist[k],6))
PY"
