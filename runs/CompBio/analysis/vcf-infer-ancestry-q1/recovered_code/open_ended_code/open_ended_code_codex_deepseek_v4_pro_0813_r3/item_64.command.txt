/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
q=defaultdict(list)
with open('q1_rsid.bim') as f:
    for line in f:
        c=line.strip().split()
        if len(c)<6: continue
        chrom, sid, cm, pos, a1, a2=c[:6]
        q[sid].append((chrom,pos,a1,a2))
h=defaultdict(list)
with open('hgdp_subset_rsid.bim') as f:
    for line in f:
        c=line.strip().split()
        if len(c)<6: continue
        chrom, sid, cm, pos, a1, a2=c[:6]
        h[sid].append((chrom,pos,a1,a2))
def alleles_match(qq,hh):
    qa={qq[2],qq[3]}; ha={hh[2],hh[3]}
    # exact same or exact complement (for all non-palindromic; even palindromic same sets)
    return qa==ha
keep=[]
for sid,qrows in q.items():
    if sid not in h: continue
    hrows=h[sid]
    if len(qrows)==1 and len(hrows)==1 and alleles_match(qrows[0], hrows[0]):
        keep.append(sid)
print('keep',len(keep))
with open('common_rsids.txt','w') as f:
    for s in keep:
        f.write(s+'\\n')
PY
wc -l common_rsids.txt"
