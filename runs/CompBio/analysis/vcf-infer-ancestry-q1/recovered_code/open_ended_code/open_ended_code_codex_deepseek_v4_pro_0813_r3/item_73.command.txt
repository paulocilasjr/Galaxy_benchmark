/bin/bash -lc "wc -l dist_1kg.mdist.id dist_hgdp.mdist.id; head -3 dist_1kg.mdist | cut -c1-300; python3 - <<'PY'
import numpy as np
for name in ['dist_1kg','dist_hgdp']:
    ids=np.loadtxt(name+'.mdist.id',dtype=str)
    M=np.loadtxt(name+'.mdist')
    print(name, M.shape, ids.shape)
    for sid in ['Sample1','Sample2','Sample3','Sample4']:
        if sid in ids:
            i=np.where(ids==sid)[0][0]
            vals=M[i].copy()
            vals[i]=np.inf
            idx=np.argsort(vals)[:10]
            print(sid, [(ids[j], round(vals[j],6)) for j in idx])
PY"
