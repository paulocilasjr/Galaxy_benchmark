/bin/bash -lc "python3 - <<'PY'
import gzip, random
from pyliftover import LiftOver
lo=LiftOver('refs/hg19ToHg38.over.chain.gz')
records=[]
last=-10**9
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line[0]=='#': continue
        p=line.rstrip().split('\\t')
        chrom,pos,ref,alt=p[0],int(p[1]),p[3],p[4]
        if chrom"'!='"'1' or len(ref)"'!=1 or len(alt)!=1 or '"',' in alt: continue
        g=[]
        ok=True
        for field in p[9:]:
            gt=field.split(':',1)[0].replace('|','/')
            if '.' in gt: ok=False; break
            c=sum(1 for a in gt.split('/') if a"'!='"'0')
            if c==0: ok=False; break
            g.append(c)
        if not ok or len(set(g))==1: continue
        # choose spaced positions to avoid LD and repeated chain overhead
        if pos-last < 20000: continue
        res=lo.convert_coordinate('chr1', pos-1)
        if len(res)"'!=1 or res[0][0] != '"'chr1' or res[0][2] "'!= '"'+': continue
        hg38pos=res[0][1]+1
        records.append((chrom,pos,ref,alt,hg38pos,g))
        last=pos
# take up to 500 spread naturally
records=records[:500]
with open('selected_lifted.tsv','w') as out, open('selected_hg38.bed','w') as bed:
    for r in records:
        chrom,pos,ref,alt,hg38pos,g=r
        out.write('\\t'.join(map(str,[chrom,pos,ref,alt,hg38pos]+g))+'\\n')
        bed.write(f'chr1\\t{hg38pos-1}\\t{hg38pos}\\n')
print(len(records), records[:5])
PY
bcftools query -l https://storage.googleapis.com/gcp-public-data--gnomad/release/3.1.2/vcf/genomes/gnomad.genomes.v3.1.2.hgdp_tgp.chr1.vcf.bgz | wc -l"
