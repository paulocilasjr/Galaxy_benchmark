# collect matched biallelic-like gnomAD positions from existing AF records
sel={}
for line in open('selected_lifted.tsv'):
    chrom,pos,ref,alt,hg38pos,*gs=line.strip().split('\\t')
    sel[int(hg38pos)]=(ref,alt,[int(x) for x in gs],pos)
records={}
for fname in ['refs/gnomad_selected_af.tsv','refs/gnomad_selected_af_batched.tsv','refs/gnomad_single_site_af.tsv','refs/gnomad_single_site_af_more.tsv']:
    try: fh=open(fname)
    except: continue
    for line in fh:
        p=line.strip().split('\\t')
        if len(p)<14: continue
        records[tuple(p[:4])]=p
matched=[]
for key,p in records.items():
    chrom,pos,ref,alts=key; pos=int(pos); altlist=alts.split(',')
    if pos in sel:
        oref,oalt,ugs,hg19pos=sel[pos]
        if ref==oref and oalt in altlist and len(ref)==1 and len(oalt)==1:
            matched.append((pos,ref,oalt,ugs))
matched=matched[:120]
with open('refs/gnomad_matched_positions.tsv','w') as out:
    for pos,ref,alt,ugs in matched:
        out.write('\\t'.join(map(str,[pos,ref,alt]+ugs))+'\\n')
print(len(matched))

        r=f'chr1:{pos}-{pos}'
