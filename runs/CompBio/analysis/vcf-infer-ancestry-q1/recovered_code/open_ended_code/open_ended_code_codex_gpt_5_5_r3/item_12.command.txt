/bin/bash -lc "python3 - <<'PY'
import gzip, collections, math
samples=[]
stats=[]
ac_patterns=collections.Counter()
private_alt=[0]*4; private_hom=[0]*4; het=[0]*4; homalt=[0]*4; nonref=[0]*4; called=[0]*4
sites=0; bial=0; snp=0; multi=0
for line in gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt'):
    if line.startswith('#CHROM'):
        samples=line.rstrip().split('\\t')[9:]
        n=len(samples)
        continue
    if line.startswith('#'): continue
    parts=line.rstrip().split('\\t')
    ref,alts=parts[3],parts[4].split(',')
    sites+=1
    if len(alts)>1: multi+=1
    else: bial+=1
    if len(ref)==1 and all(len(a)==1 for a in alts): snp+=1
    g=[]
    for i,field in enumerate(parts[9:]):
        gt=field.split(':',1)[0].replace('|','/')
        if '.' in gt:
            g.append(None); continue
        alleles=[int(x) for x in gt.split('/')]
        alt_count=sum(1 for a in alleles if a>0)
        g.append(alt_count)
        called[i]+=1
        if alt_count==1: het[i]+=1
        if alt_count==2: homalt[i]+=1
        if alt_count>0: nonref[i]+=1
    altcar=[i for i,x in enumerate(g) if x and x>0]
    if len(altcar)==1:
        private_alt[altcar[0]]+=1
        if g[altcar[0]]==2: private_hom[altcar[0]]+=1
print('samples',samples)
print('sites',sites,'bial',bial,'multi',multi,'snp',snp)
for i,s in enumerate(samples):
    print(s,'called',called[i],'het',het[i],'homalt',homalt[i],'nonref',nonref[i],'private_alt',private_alt[i],'private_hom',private_hom[i], 'het/hom', het[i]/max(1,homalt[i]))
PY"
