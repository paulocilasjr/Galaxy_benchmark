/bin/bash -lc "python3 - <<'PY'
import gzip
n=0
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f, open('sites100.bed','w') as o:
    for line in f:
        if line[0]=='#': continue
        p=line.split('\\t')
        if len(p[3])==1 and ',' not in p[4] and len(p[4])==1:
            pos=int(p[1]); o.write(f'{p[0]}\\t{pos-1}\\t{pos}\\n')
            n+=1
            if n>=100: break
print(n)
PY
bcftools view -H -R sites100.bed https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz | head -n 3"
