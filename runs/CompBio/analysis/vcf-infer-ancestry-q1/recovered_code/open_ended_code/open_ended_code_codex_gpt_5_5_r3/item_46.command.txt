/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for prefix in ['release/3.1.2/','resources/hgdp_1kg/','resources/grch38/hgdp_1kg/','release/3.1.2/ht/','release/3.1.2/vcf/']:
    url='https://storage.googleapis.com/storage/v1/b/gcp-public-data--gnomad/o?maxResults=20&prefix='+urllib.parse.quote(prefix)
    try:
        data=json.load(urllib.request.urlopen(url, timeout=20))
        print('PREFIX',prefix,'count',len(data.get('items',[])))
        for item in data.get('items',[])[:20]: print(item['name'], item.get('size'))
    except Exception as e: print('ERR', prefix, e)
PY"
