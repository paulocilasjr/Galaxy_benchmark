import numpy as np
z=np.genfromtxt('gnomad_matched.tsv',names=True,delimiter='\\t',dtype=None,encoding=None)

 print(s,sorted(zip(pops,np.round(np.array(scores)-max(scores),2)),key=lambda x:-x[1]))
