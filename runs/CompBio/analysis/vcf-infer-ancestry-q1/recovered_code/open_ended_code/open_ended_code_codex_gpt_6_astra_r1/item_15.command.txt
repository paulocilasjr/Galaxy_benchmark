/bin/bash -lc "cat > infer_hgdp.py <<'PY'
import gzip, numpy as np, pgenlib, pyliftover, collections
lo=pyliftover.LiftOver('hg19ToHg38.over.chain.gz')
query={}
comp=str.maketrans('ACGT','TGCA')
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
 for l in f:
  if l[0]=='#':continue
  a=l.split();ref,alt=a[3:5]
  if len(ref)"'!=1 or len(alt)!=1:continue
  pos=int(a[1]);lift=lo.convert_coordinate('"'chr1',pos-1)
  if len(lift)"'!=1 or lift[0][0]!='"'chr1':continue
  if lift[0][2]=='-':ref=ref.translate(comp);alt=alt.translate(comp)
  g=[sum(map(int,x.split(':')[0].split('/'))) for x in a[9:13]]
  q=[int(x.split(':')[3]) if x.split(':')[3]"'!='"'.' else 0 for x in a[9:13]]
  query[(lift[0][1]+1,ref,alt)]=(pos,g,q)
print('Lifted SNPs',len(query),flush=True)
indices=[];pos=[];gs=[];qs=[]
with open('hgdp_chr1.pvar') as f:
 i=0
 for l in f:
  if l[0]=='#':continue
  a=l.split();k=(int(a[1]),a[3],a[4])
  if k in query:
   p,g,q=query[k];indices.append(i);pos.append(p);gs.append(g);qs.append(q)
  i+=1
print('Matched',len(indices),'of',i,flush=True)
meta=[l.split() for l in open('hgdp_psam.txt') if not l.startswith('#')]
n=len(meta)
reader=pgenlib.PgenReader(b'hgdp_chr1.pgen',raw_sample_ct=n)
refg=np.empty((len(indices),n),dtype=np.int8)
reader.read_list(np.array(indices,dtype=np.uint32),refg)
np.savez_compressed('matched.npz',pos=pos,g=gs,q=qs,refg=refg,meta=np.array(meta))
PY
PYTHONPATH=./packages python3 infer_hgdp.py"
