import gzip
ours=set()
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')
        ours.add((p[0],int(p[1])))
# parse graf bed format? inspect first
from collections import defaultdict
graf=[]
for line in open('graf_biallelic.bed'):
    if not line.strip(): continue
    c=line.rstrip('\\n').split('\\t')
    if len(c)>=2:
        try:
            chrom=c[0]; start=int(c[1])
            graf.append((chrom,start))
        except: pass
print('graf',len(graf))
gset={(x[0].replace('chr',''),x[1]) for x in graf}
print('overlap with our VCF', len(ours & gset))
print(list(ours & gset)[:20])
PY"
