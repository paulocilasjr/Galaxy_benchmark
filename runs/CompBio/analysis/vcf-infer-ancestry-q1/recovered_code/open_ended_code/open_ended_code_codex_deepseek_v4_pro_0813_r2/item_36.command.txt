/bin/bash -lc "python3 - <<'PY'
import gzip
counts=[0]*4; n=0
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.strip().split('\\t')
        for i,x in enumerate(p[9:]):
            gt=x.split(':')[0]
            if '.' in gt: continue
            g=gt.replace('|','/').split('/')[:2]
            counts[i]+=sum(int(a) for a in g)
        n+=1
print(n)
for i in range(4): print('Sample',i+1,counts[i],counts[i]/(2*n))
PY"
