    key=(int(p[1]),p[3],p[4]); ds=qdict.get(key)

    for field in p[7].split(';'):

            k,v=field.split('=',1)
            if k in vals:
                try: v=float(v)
                except: continue
                for qi,d in enumerate(ds):

for pop,dicts in vals.items():
