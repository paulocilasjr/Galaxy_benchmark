/bin/bash -lc "python3 - <<'PY'
import gzip,collections
c=collections.Counter(); multi=0
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')
        alts=p[4].split(',')
        c[len(alts)]+=1
        if len(alts)>1: multi+=1
print(c, 'multi', multi)
PY"
