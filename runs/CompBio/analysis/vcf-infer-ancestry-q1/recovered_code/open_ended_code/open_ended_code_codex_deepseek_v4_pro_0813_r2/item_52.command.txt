/bin/bash -lc "python3 - <<'PY'
import numpy as np, collections
panel={}
for line in open('1kg.panel'):
    if line.startswith('sample'): continue
    s,pop,sup,g=line.strip().split('\\t')[:4]
    panel[s]=(pop,sup)
samples=open('1kg.samples.txt').read().splitlines()
sum_ibs=np.load('1kg_match_ibs.npy'); exact=np.load('1kg_match_exact.npy'); count=np.load('1kg_match_count.npy')//4
sups=['AFR','AMR','EAS','EUR','SAS']
for qi in range(4):
    avg=sum_ibs[qi]/(2*count)
    print('\\nSample',qi+1)
    # mean and top per superpop
    for sup in sups:
        idx=[j for j,s in enumerate(samples) if panel[s][1]==sup]
        vals=avg[idx]
        print(f'  {sup:3s} n={len(idx):4d} mean={vals.mean():.6f} median={np.median(vals):.6f} max={vals.max():.6f} top3={[(samples[idx[k]],round(vals[k],5)) for k in np.argsort(-vals)[:3]]}')
PY"
