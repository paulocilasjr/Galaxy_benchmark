/bin/bash -lc "python3 - <<'PY'
import collections
# parse ref
ref=[]
for line in open('HGDP_1kGP.exon_UTRS.bim'):
    p=line.rstrip('\\n').split('\\t')
    chrom=p[0]; pos=int(p[3]); a1=p[4]; a2=p[5]
    key=(chrom,pos,frozenset([a1,a2]))
    ref.append((key,p[1],a1,a2))
# parse query
qmap={}
for line in open('query.bim'):
    p=line.rstrip('\\n').split('\\t')
    chrom=p[0]; pos=int(p[3]); a1=p[4]; a2=p[5]
    key=(chrom,pos,frozenset([a1,a2]))
    qmap[key]=p[1]
# match
ren=[]
common=[]
for key,rid,a1,a2 in ref:
    qid=qmap.get(key)
    if qid:
        ren.append((qid,rid)); common.append(rid)
print('mapped',len(ren))
with open('query.rename','w') as f:
    for qid,rid in ren:
        f.write(f'{qid}\\t{rid}\\n')
with open('common.ids','w') as f:
    for rid in common: f.write(rid+'\\n')
PY
wc -l query.rename common.ids
head query.rename
head common.ids"
