/bin/bash -lc "python3 - <<'PY'
import shutil
# mapping old simple id -> ref id
ren={}
for line in open('query_simple.rename'):
    a,b=line.rstrip('\\n').split('\\t'); ren[a]=b
with open('query_simple.bim') as fin, open('query_mapped.bim','w') as fout:
    for line in fin:
        p=line.rstrip('\\n').split('\\t')
        if p[1] in ren: p[1]=ren[p[1]]
        fout.write('\\t'.join(p)+'\\n')
for ext in ['bed','fam']:
    shutil.copy(f'query_simple.{ext}', f'query_mapped.{ext}')
print('mapped BIM rows', sum(1 for l in open('query_mapped.bim') if l.split('\\t')[1].startswith('1_')))
PY
./plink --bfile query_mapped --extract common.ids --make-bed --out query_common --allow-extra-chr 2>&1 | tail -n 30"
