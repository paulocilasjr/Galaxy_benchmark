/bin/bash -lc "PYTHONPATH=.pkg python3 - <<'PY' > hgdp.query.hg38.ranges
from pyliftover import LiftOver
import gzip
lo=LiftOver('hg19ToHg38.over.chain.gz')
seen=set(); n=ok=0
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
  for line in f:
    if line.startswith('#'): continue
    n+=1
    x=line.split('\\t'); p=int(x[1])
    hits=lo.convert_coordinate('chr1',p-1)
    if not hits: continue
    # use the primary chr1 mapping; all query loci are single-base and should map uniquely
    h=[z for z in hits if z[0] in ('chr1','1')]
    if not h: continue
    q=h[0][1]+1
    if q not in seen:
      seen.add(q); ok+=1; print('1',q,q,sep='\\t')
print('mapped',ok,'of',n,file=__import__('sys').stderr)
PY
wc -l hgdp.query.hg38.ranges; sed -n '1,4p' hgdp.query.hg38.ranges"
