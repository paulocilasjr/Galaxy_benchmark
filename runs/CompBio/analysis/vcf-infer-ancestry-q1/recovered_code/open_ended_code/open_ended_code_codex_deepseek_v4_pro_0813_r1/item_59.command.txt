/bin/bash -c "python3 - <<'PY'
import pickle, numpy as np
with open('sites.pkl','rb') as f: sites=pickle.load(f)
G=np.load('samples.G.npy')
# create target first 5000 positions
with open('first5000.tsv','w') as out:
 for c,pos,ref,alts in sites[:5000]: out.write(f'{c}\\t{pos}\\n')
PY
/usr/bin/time -p bcftools query -T first5000.tsv -f '%POS[\\t%GT]\\n' 1000G.chr1.v5a.vcf.gz > first5000.1000g.gt.tsv
wc -l first5000.1000g.gt.tsv; head -1 first5000.1000g.gt.tsv | cut -c1-200"
