/bin/bash -c "python3 - <<'PY'
import gzip, numpy as np, pickle
fn='vcf.infer.ancestry.q1.vcf.gz'
sites=[]; G=np.empty((119566,4), dtype=np.int8)
n=0
with gzip.open(fn,'rt') as f:
 for line in f:
  if line.startswith('#'): continue
  parts=line.rstrip().split('\\t')
  chrom,pos,id,ref,alt=parts[:5]
  fmt=parts[8].split(':'); gi=fmt.index('GT')
  gts=[]
  for i in range(4):
   gt=parts[9+i].split(':')[gi]
   if gt in ('0/0','0|0'): d=0
   elif gt in ('0/1','1/0','0|1','1|0'): d=1
   elif gt in ('1/1','1|1'): d=2
   else:
    # multi-allelic, store as 3 for now; maybe use dosage if alt index 0?
    d=3
   G[n,i]=d
  alts=alt.split(',')
  sites.append((chrom,int(pos),ref,alts))
  n+=1
np.save('samples.G.npy', G)
with open('sites.pkl','wb') as f: pickle.dump(sites,f)
print('sites',len(sites), 'multiallelic', int((G==3).sum()), 'total cells',G.size)
PY
ls -lh samples.G.npy sites.pkl"
