/bin/bash -c "python3 - <<'PY'
import requests, json, pickle, time
with open('sites.pkl','rb') as f: sites=pickle.load(f)
# first 25 biallelic sites with first alt simple, position/ref/alts
variants=[]
for c,pos,ref,alts in sites:
 if len(alts)>1: continue
 if any(len(a)>50 for a in alts): continue
 variants.append(f'{c}-{pos}-{ref}-{alts[0]}')
 if len(variants)>=25: break
print(variants)
# build aliases
q='query {\\n'
for i,vid in enumerate(variants):
 q += f'  v{i}: variant(variantId: \"{vid}\", dataset: gnomad_r2_1) {{ variant_id genome {{ ac an populations {{ id ac an homozygote_count }} }} }}\\n'
q += '}'
r=requests.post('https://gnomad.broadinstitute.org/api',json={'query':q},timeout=60)
print('status',r.status_code)
print(r.text[:1000])
PY"
