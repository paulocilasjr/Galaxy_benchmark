/bin/bash -c "python3 - <<'PY'
import gzip
fn='vcf.infer.ancestry.q1.vcf.gz'
pos=[]; n=0
with gzip.open(fn,'rt') as f:
 for line in f:
  if line.startswith('#'): continue
  p=line.split('\\t',2)
  pos.append(int(p[1])); n+=1
  if n>=1000: break
print(pos[0], pos[-1], min(pos), max(pos))
# distribution of first 1000 intervals
import numpy as np
print(np.mean(np.diff(pos)), np.median(np.diff(pos)), np.max(np.diff(pos)))
PY"
