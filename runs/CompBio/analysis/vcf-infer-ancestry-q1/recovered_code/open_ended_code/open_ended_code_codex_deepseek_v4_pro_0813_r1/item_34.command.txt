/bin/bash -c "python3 - <<'PY'
import gzip
fn='vcf.infer.ancestry.q1.vcf.gz'
with gzip.open(fn,'rt') as f, open('first1000.targets.bed','w') as out:
 n=0
 for line in f:
  if line.startswith('#'): continue
  p=line.split('\\t')[1]
  out.write(f'1\\t{p}\\t{p}\\n')
  n+=1
  if n>=1000: break
print(n)
PY
/usr/bin/time -p bcftools query -R first1000.targets.bed -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%INFO/AFR_AF\\t%INFO/EUR_AF\\t%INFO/EAS_AF\\t%INFO/SAS_AF\\n' https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz > first1000.1000g.af.tsv
wc -l first1000.1000g.af.tsv; head first1000.1000g.af.tsv"
