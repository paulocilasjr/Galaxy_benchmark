/bin/bash -c "python3 - <<'PY'
import gzip, sys, collections, numpy as np
fn='vcf.infer.ancestry.q1.vcf.gz'
hets=collections.Counter(); homalt=collections.Counter(); homref=collections.Counter(); total=0; afs=[]
with gzip.open(fn,'rt') as f:
 for line in f:
  if line.startswith('#'): continue
  parts=line.rstrip().split('\\t')
  fmt=parts[8].split(':'); gi=fmt.index('GT')
  gts=[parts[9+i].split(':')[gi] for i in range(4)]
  total+=1
  for i,gt in enumerate(gts):
   if gt=='0/0' or gt=='0|0': homref[i]+=1
   elif gt=='0/1' or gt=='1/0' or gt=='0|1' or gt=='1|0': hets[i]+=1
   elif gt=='1/1' or gt=='1|1': homalt[i]+=1
   else:
    # multi allelic etc
    if '/' in gt:
     a=gt.split('/')
     if a[0]"'!=a[1]: hets[i]+=1
     elif a[0]=='"'0': homref[i]+=1
     else: homalt[i]+=1
for i in range(4):
 print(i, total, 'het',hets[i], 'het rate',hets[i]/total,'homref',homref[i],'homalt',homalt[i], 'alt allele freq approx',(hets[i]+2*homalt[i])/(2*total))
PY"
