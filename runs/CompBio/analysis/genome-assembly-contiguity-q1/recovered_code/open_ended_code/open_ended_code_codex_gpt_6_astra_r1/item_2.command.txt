/bin/bash -lc "python - <<'PY'
from pathlib import Path
path = Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
lengths = []
length = None
with path.open() as handle:
    for line in handle:
        if line.startswith('>'):
            if length is not None:
                lengths.append(length)
            length = 0
        else:
            seq = ''.join(line.split())
            if seq:
                assert length is not None
                length += len(seq)
if length is not None:
    lengths.append(length)
lengths.sort(reverse=True)
total = sum(lengths)
cumulative = 0
for length in lengths:
    cumulative += length
    if cumulative * 2 >= total:
        print(length)
        break
PY"
