/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
p=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
lengths=[]
cur=[]
name=None
with p.open('rt') as f:
    for line in f:
        if line.startswith('>'):
            if name is not None:
                lengths.append(sum(cur))
            name=line[1:].strip()
            cur=[]
        else:
            s=line.strip()
            if s:
                cur.append(len(s))
if name is not None:
    lengths.append(sum(cur))
total=sum(lengths)
print('num contigs', len(lengths))
print('total', total)
print('min', min(lengths), 'max', max(lengths))
lengths.sort(reverse=True)
cum=0
n50=None
for l in lengths:
    cum+=l
    if cum >= total*0.5:
        n50=l
        break
print('N50', n50)
print('sorted top', lengths[:20])
PY"
