import collections
vals=[]
zero=[]
with open('cov100k.tsv') as f:
    for line in f:
        c,s,e,v=line.split(); s=int(s); e=int(e); v=float(v); vals.append(v)
        if v==0: zero.append((s,e))
print('zero count',len(zero),'ranges',zero[:20],zero[-20:])
# counts of windows by ranges
hist=collections.Counter()
for v in vals:
    hist[round(v)] += 1
print(hist)
# print first non-zero and last
for i,v in enumerate(vals):
    if v>0:
        print('first nonzero idx',i,'start',i*100000); break
for i,v in enumerate(vals):
    if v>0:
        last=i
print('last nonzero idx',last,'start',last*100000)
PY"
