def load_reg(s,e):
    vals=[]

            if a>=s and b<=e:
                vals.append(v)
    return np.array(vals)

amp=load_reg(36000000,36400000)
for name,s,e in [('imm',35500000,36000000),('imm2',36400000,36900000),('wide',35000000,36000000),('wide2',36400000,37000000)]:
