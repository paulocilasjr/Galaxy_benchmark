/bin/bash -lc ': > chrom_sample_counts.tsv
for c in $(cat primary_chroms.txt); do
  (
    n=$(minimap2 -x sr -t 1 chroms/${c}.fa.gz sample_R1_100.fa 2>/dev/null | awk '"'"'$12>=20{n++} END{print n+0}'"')
    printf \"%s\\t%s\\n\" \""'$c" "$n"
  ) >> chrom_sample_counts.tsv &
  while [ $(jobs -r | wc -l | tr -d '"' ') -ge 4 ]; do sleep 1; done
done
wait
sort -k2,2nr chrom_sample_counts.tsv | head -24"
