/bin/bash -lc "python3 - <<'PY'
import pyfaidx
fa=pyfaidx.Fasta('ref/hg38.fa')
chrm='chr22'; seq=fa[chrm][:].seq.upper()
import csv
# windows 10k
with open('gc10k.tsv','w') as out:
 for start in range(0, len(seq), 10000):
  s=start+1; e=min(start+10000,len(seq)); sub=seq[start:e]
  gc=(sub.count('G')+sub.count('C'))/len(sub)
  out.write(f'{chrm}\\t{s}\\t{e}\\t{gc:.6f}\\n')
print('done',len(seq))
PY
python3 - <<'PY'
import pandas as pd, numpy as np
cov=pd.read_csv('cov10k.tsv',sep='\\t',header=None,names=['c','s','e','depth'])
gc=pd.read_csv('gc10k.tsv',sep='\\t',header=None,names=['c','s','e','gc'])
m=cov.merge(gc,on=['c','s','e'])
print(m.head())
print('corr all',m['depth'].corr(m['gc']))
nonzero=m[m.depth>0]
print('corr nonzero',nonzero['depth'].corr(nonzero['gc']))
# high region GC
for a,b in [(36000001,36400000),(35000001,35600000),(36400001,36700000),(30000000,36000000)]:
 sub=m[(m.s>=a)&(m.e<=b)]
 print(a,b,'mean depth',sub.depth.mean(),'mean gc',sub.gc.mean())
PY"
