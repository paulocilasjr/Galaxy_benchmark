/bin/bash -lc "python3 - <<'PY'
import subprocess
p=subprocess.Popen(['samtools','depth','-a','-Q','20','aligned.bam'],stdout=subprocess.PIPE,text=True,bufsize=1000000)
regions={
'inside':(36000001,36400000),
'left':(34000001,36000000),
'right':(36400001,38000000),
}
sums={k:0 for k in regions}; counts={k:0 for k in regions}
for line in p.stdout:
    chrom,pos,dep=line.split(); pos=int(pos); dep=int(dep)
    for k,(a,b) in regions.items():
        if a<=pos<=b:
            sums[k]+=dep; counts[k]+=1
p.wait()
for k in regions:
    print(k,sums[k],counts[k],sums[k]/counts[k])
print('ratio left',sums['inside']/counts['inside']/(sums['left']/counts['left']))
print('ratio right',sums['inside']/counts['inside']/(sums['right']/counts['right']))
print('ratio avg flanks',sums['inside']/counts['inside']/((sums['left']+sums['right'])/(counts['left']+counts['right'])))
PY"
