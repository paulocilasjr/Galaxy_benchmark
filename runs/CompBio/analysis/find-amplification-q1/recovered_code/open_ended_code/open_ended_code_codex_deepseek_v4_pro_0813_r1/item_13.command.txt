/bin/bash -lc "python3 - <<'PY'
import requests,json
rid='7X03DXY9016'
r=requests.get('https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi',params={'CMD':'Get','FORMAT_TYPE':'JSON2_S','RID':rid},timeout=60)
data=r.json()
rep=data['BlastOutput2'][0]['report']
print('query',rep['results']['search']['query_title'], 'db',rep['results']['search']['query_db'])
hits=rep['results']['search']['hits']
print('num hits',len(hits))
for h in hits[:10]:
    print(h['description'][0]['id'], h['description'][0]['title'], 'hsps',len(h['hsps']))
    if h['hsps']:
        hs=h['hsps'][0]
        print('  q',hs['query_from'],hs['query_to'],'h',hs['hit_from'],hs['hit_to'],'identity',hs['identity'],'align_len',hs['align_len'],'bits',hs['bit_score'])
PY"
