/bin/bash -lc "python3 - <<'PY'
import csv,statistics
rows=[]
with open('depth_1kb.csv') as f:
    for r in csv.DictReader(f): rows.append((int(r['start']),float(r['mean'])))
for b in [5000,10000,20000,50000]:
    bins={}
    for s,m in rows:
        i=(s-1)//b
        bins.setdefault(i,[]).append(m)
    print('\\nbin',b)
    for i in sorted(bins):
        s=i*b+1; e=(i+1)*b
        if 34000000<=s<=37500000:
            vals=bins[i]
            avg=statistics.mean(vals)
            # print with mark if 35500000<=s<=37000000
            print(f'{s}-{e} {avg:.3f}')
PY"
