/bin/bash -lc "python3 - <<'PY'
import subprocess, csv, statistics
bam='aligned.bam'
bin=1000
size=50818468
nb=size//bin+2
counts=[0]*nb; sums=[0]*nb
p=subprocess.Popen(['samtools','depth','-a','-Q','20',bam],stdout=subprocess.PIPE,text=True,bufsize=1000000)
for line in p.stdout:
    chrom,pos,dep=line.split(); pos=int(pos); dep=int(dep)
    i=(pos-1)//bin; counts[i]+=1; sums[i]+=dep
p.wait()
means=[s/c if c else 0 for c,s in zip(counts,sums)]
with open('depth_1kb.csv','w') as f:
    w=csv.writer(f); w.writerow(['start','end','count','sum','mean'])
    for i,(c,s,m) in enumerate(zip(counts,sums,means)):
        w.writerow([i*bin+1,(i+1)*bin,c,s,round(m,4)])
# compute global and mappable median (excluding zero)
nz=[m for m in means if m>0]
print('global',sum(sums)/sum(counts),'median nz',statistics.median(nz),'mean nz',statistics.mean(nz),'bins nz',len(nz))
# contiguous segments mean > 1.25*median_nz and length >= 50kb, with transitions.
thr=statistics.median(nz)*1.25
start=None; seg=[]
for i,m in enumerate(means):
    if m>thr:
        if start is None: start=i
        end=i
    else:
        if start is not None:
            seg.append((start,end)); start=None
if start is not None: seg.append((start,end))
print('threshold',thr,'segments')
for a,b in seg:
    print(a*bin+1,(b+1)*bin,'len',(b-a+1)*bin,'mean',statistics.mean(means[a:b+1]),'nbins',b-a+1)
PY"
