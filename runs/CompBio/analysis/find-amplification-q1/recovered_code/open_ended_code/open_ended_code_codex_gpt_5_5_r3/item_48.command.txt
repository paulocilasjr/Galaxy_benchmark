/bin/bash -lc "awk '"'$1=="chr22" {for (s=0; s<$2; s+=100000) {e=s+100000; if (e>$2) e=$2; print $1"'"\\t\"s\"\\t\"e}}' ref/hg38.primary.fa.fai > work/chr22.100kb.bed
samtools bedcov -Q 20 work/chr22.100kb.bed align/find.amplification.q1.hg38.primary.minimap2.bam > work/chr22.100kb.bedcov
awk 'BEGIN{OFS=\"\\t\"}{len="'$3-$2; print $1,$2,$3,$4/len}'"' work/chr22.100kb.bedcov > work/chr22.100kb.depth.tsv
awk '"'$4>0{print}'"' work/chr22.100kb.depth.tsv | sort -k4,4nr | head -40
awk '"'$2>=35000000 && $2<37500000{print}'"' work/chr22.100kb.depth.tsv"
