/bin/bash -lc 'if [ -e ref/hg38.primary.fa ]; then mv ref/hg38.primary.fa ref/hg38.primary.fa.previous; fi
cp ref/primary/chroms/chr1.fa ref/hg38.primary.fa
for c in $(seq 2 22) X Y M; do cat ref/primary/chroms/chr${c}.fa >> ref/hg38.primary.fa; done
samtools faidx ref/hg38.primary.fa
bwa index ref/hg38.primary.fa'
