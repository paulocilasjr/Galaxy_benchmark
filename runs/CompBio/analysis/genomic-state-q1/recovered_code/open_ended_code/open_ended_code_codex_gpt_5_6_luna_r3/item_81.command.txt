/bin/bash -lc "python3 - <<'PY'
import pyBigWig,sys
base='https://epigenome.wustl.edu/epimap/data/averagetracks_pergroup/'
for mark in ['ATAC-seq','DNase-seq','H3K27ac','H3K4me1','H3K4me3','H3K27me3','H3K9me3','H3K36me3','EP300']:
 print('\\n'+mark,flush=True)
 for mode in ['observed','imputed']:
  vals=[]
  for group in ['Liver','Brain','ESC']:
   u=f'{base}average_{mark}_{mode}_{group}.bigWig'
   try:
    bw=pyBigWig.open(u)
    x=bw.stats('chr11',124738680,124738772,type='mean')[0]
    vals.append(f'{group}:{x}')
    bw.close()
   except Exception as e: vals.append(f'{group}:ERR({type(e).__name__})')
  print(mode, ' '.join(vals),flush=True)
PY"
