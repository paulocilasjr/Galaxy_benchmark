 print('\\n###',top)
 def rec(k,v,depth=0):
  if isinstance(v,dict):
   if k in (top,) or depth<=1 or ('bigDataUrl' in v):
    print('  '*depth,k,'|',v.get('shortLabel'), '|',v.get('type'),'|',v.get('bigDataUrl',''),'|',v.get('cellType',''),v.get('tissue',''),v.get('biosample',''),v.get('sample',''))
   for kk,vv in v.items():
    if kk not in ['bigDataUrl','html','filterValues','filterLimits','filterLabels']:
     rec(kk,vv,depth+1)
