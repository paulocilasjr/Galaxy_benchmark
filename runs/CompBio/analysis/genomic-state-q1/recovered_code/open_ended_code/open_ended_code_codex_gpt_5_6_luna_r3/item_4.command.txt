/bin/bash -lc "python - <<'PY'
import json, urllib.request
u='https://api.genome.ucsc.edu/list/tracks?genome=hg38'
data=json.load(urllib.request.urlopen(u,timeout=90))['hg38']
terms=('chrom','state','roadmap','enh','reg','histone','h3k','cpg','fantom','encode','tss','polycomb','seg')
for k,v in sorted(data.items()):
 s=(k+' '+json.dumps(v,separators=(',',':'))).lower()
 if any(t in s for t in terms):
  print(k, '|', v.get('shortLabel',''), '|', v.get('type',''), '|', v.get('bigDataUrl',''))
PY"
