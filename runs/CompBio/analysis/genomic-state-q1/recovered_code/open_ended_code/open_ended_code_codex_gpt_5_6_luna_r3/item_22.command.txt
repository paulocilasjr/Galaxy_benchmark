import json, urllib.request
base='https://api.genome.ucsc.edu/getData/track?genome=hg38;chrom=chr11;start=124738681;end=124738772;track='
tracks=['wgEncodeReg4AtacAllLiver','wgEncodeReg4AtacAllBrain','wgEncodeReg4AtacLiver','wgEncodeReg4AtacBrain','wgEncodeReg4DnaseAllLiver','wgEncodeReg4DnaseAllBrain','wgEncodeReg4DnaseLiver','wgEncodeReg4DnaseBrain','wgEncodeReg4MarkH3k27acAllLiver','wgEncodeReg4MarkH3k27acAllBrain','wgEncodeReg4MarkH3k27acLiver','wgEncodeReg4MarkH3k27acBrain','wgEncodeReg4MarkH3k4me1AllLiver','wgEncodeReg4MarkH3k4me1AllBrain','wgEncodeReg4MarkH3k4me1Liver','wgEncodeReg4MarkH3k4me1Brain','wgEncodeReg4MarkH3k4me3AllLiver','wgEncodeReg4MarkH3k4me3AllBrain','wgEncodeReg4MarkH3k4me3Liver','wgEncodeReg4MarkH3k4me3Brain','wgEncodeReg4TxnAllLiverPlus','wgEncodeReg4TxnAllLiverMinus','wgEncodeReg4TxnAllBrainPlus','wgEncodeReg4TxnAllBrainMinus','wgEncodeReg4TxnLiverPlus','wgEncodeReg4TxnBrainPlus']
for tr in tracks:
 try:
  d=json.load(urllib.request.urlopen(base+tr,timeout=60))
  print('\\n###',tr,d.get('trackType'),d.get('dataTime'), 'itemsReturned',d.get('itemsReturned'))
  for k,v in d.items():
   if k not in ('downloadTime','downloadTimeStamp','dataTime','dataTimeStamp','genome','trackType','track','chrom','chromSize','start','end','itemsReturned','bigDataUrl'):
    print(k,json.dumps(v)[:1500])
 except Exception as e: print('\\n###',tr,type(e).__name__,e)
PY"
