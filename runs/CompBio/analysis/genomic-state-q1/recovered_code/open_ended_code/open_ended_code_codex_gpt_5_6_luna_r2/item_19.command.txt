import urllib.request,gzip,io
base='https://personal.broadinstitute.org/cboix/epimap/ChromHMM/observed_aux_18_hg38/CALLS/'
ids=['BSS00478','BSS00483','BSS00484','BSS00315','BSS01159','BSS01168','BSS01169','BSS01170','BSS01271','BSS01272','BSS01273','BSS00369','BSS00371']
for bid in ids:
 u=base+bid+'_18_CALLS_segments.bed.gz'
 try:
  with urllib.request.urlopen(u,timeout=60) as f:
   raw=f.read()
  lines=[]
  with gzip.GzipFile(fileobj=io.BytesIO(raw)) as g:
   for line in g:
    if line.startswith(b'chr11\\t'):
     a=line.split(b'\\t')
     if int(a[1])<124739000 and int(a[2])>124738000:
      lines.append(line.decode().strip())
  print('\\n',bid, 'bytes',len(raw))
  print('\\n'.join(lines))
 except Exception as e: print('\\n',bid,'ERR',e)
PY"
