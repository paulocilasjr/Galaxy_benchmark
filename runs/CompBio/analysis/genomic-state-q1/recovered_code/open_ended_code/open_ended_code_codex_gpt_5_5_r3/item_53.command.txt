/bin/bash -lc "curl -s 'https://api.genome.ucsc.edu/list/tracks?genome=hg19' > ucsc_tracks_hg19.json && python3 - <<'PY'
import json
j=json.load(open('ucsc_tracks_hg19.json'))['hg19']
for name,meta in j.items():
 s=(name+' '+str(meta.get('shortLabel',''))+' '+str(meta.get('longLabel',''))+' '+str(meta.get('type',''))).lower()
 if any(x in s for x in ['chromhmm','roadmap','segway','hmm','state']): print(name,'|',meta.get('shortLabel'),'|',meta.get('longLabel'),'|',meta.get('type'))
PY"
