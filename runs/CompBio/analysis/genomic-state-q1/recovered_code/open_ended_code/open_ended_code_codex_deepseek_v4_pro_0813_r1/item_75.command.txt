/bin/bash -c "python3 - <<'PY'
import json, urllib.parse, urllib.request, statistics
# track names from JSON
j=json.load(open('[REDACTED LOCAL PATH]'))['hg38']['coreCcres']
# build map sample-> view tracks
views=['H3K27ac_view','H3K4me3_view','DNase_view']
sel={}
for view in views:
  v=j[view]
  for k,val in v.items():
    if isinstance(val,dict) and val.get('type')=='bigWig' and val.get('subGroups',''):
      sg=val['subGroups']
      if any(tok in sg for tok in ['simpleBiosample=middle_frontal_area_46-_male_adult__84_years_','simpleBiosample=right_lobe_of_liver-_female_adult__53_years_','simpleBiosample=H1','simpleBiosample=H9']):
        label=(sg.split('view=')[1].split(' ')[0], sg.split('simpleBiosample=')[1].split(' ')[0])
        sel.setdefault((label[1],view),[]).append(k)
for key,ks in sel.items():
  # take first
  k=ks[0]
  url='https://api.genome.ucsc.edu/getData/track?'+urllib.parse.urlencode({'genome':'hg38','track':k,'chrom':'chr11','start':124738681,'end':124738772})
  try:
    with urllib.request.urlopen(url,timeout=20) as r: d=json.load(r)
    arr=d.get(k,[])
    vals=[x['value'] for x in arr if 'value' in x]
    print(key, 'track',k,'mean',sum(vals)/len(vals) if vals else None,'max',max(vals) if vals else None,'vals',vals)
  except Exception as e: print(key,e)
PY"
