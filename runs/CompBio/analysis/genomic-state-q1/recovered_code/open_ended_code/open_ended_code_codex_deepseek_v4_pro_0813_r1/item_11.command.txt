/bin/bash -c "curl -s 'https://api.genome.ucsc.edu/list/tracks?genome=hg38' -o [REDACTED LOCAL PATH] && python3 - <<'PY'
import json
j=json.load(open('[REDACTED LOCAL PATH]'))
hg=j['hg38']
for k,v in hg.items():
    s=json.dumps(v)
    if any(term in s.lower() for term in ['chromhmm','roadmap','epigenome','candidate cis','candidatecis','encodecre','regulatory','enhancer','15-state','chromatin state','segmentation']):
        print(k, v.get('shortLabel',''), v.get('longLabel','')[:120], v.get('type',''))
PY"
