/bin/bash -lc 'set -e
for id in E003 E066 E067 E068 E069 E070 E071 E072 E073 E074 E081 E082; do
  f=data/roadmap/${id}_15_coreMarks_mnemonics.bed.gz
  [ -s "$f" ] || curl -sS -L -o "$f" "https://egg2.wustl.edu/roadmap/data/byFileType/chromhmmSegmentations/ChmmModels/coreMarks/jointModel/final/${id}_15_coreMarks_mnemonics.bed.gz"
done
python3 - <<'"'PY'
import gzip, glob, os
start=124608576; end=124608668
labels={
'E003':'ESC/H1', 'E066':'Liver', 'E067':'Brain angular gyrus', 'E068':'Brain anterior caudate', 'E069':'Brain cingulate gyrus', 'E070':'Brain germinal matrix', 'E071':'Brain hippocampus middle', 'E072':'Brain inferior temporal lobe', 'E073':'Brain dorsolateral prefrontal cortex', 'E074':'Brain substantia nigra', 'E081':'Fetal brain male', 'E082':'Fetal brain female'}
for path in sorted(glob.glob('data/roadmap/E*_15_coreMarks_mnemonics.bed.gz')):
    eid=os.path.basename(path).split('_')[0]
    hits=[]
    with gzip.open(path,'rt') as f:
        for line in f:
            c,s,e,state=line.rstrip().split('\\t')[:4]
            if c=='chr11' and int(e)>start and int(s)<end:
                hits.append((int(s),int(e),state))
    print(eid, labels.get(eid,''), hits)
PY"
