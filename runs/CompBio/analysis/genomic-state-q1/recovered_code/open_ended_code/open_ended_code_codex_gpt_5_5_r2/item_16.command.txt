/bin/bash -lc "curl -sS 'https://api.genome.ucsc.edu/list/tracks?genome=hg38' > tracks_hg38.json && python3 - <<'PY'
import json
j=json.load(open('tracks_hg38.json'))
for k,v in j.get('hg38',{}).items():
    name=(k+' '+str(v.get('shortLabel',''))+' '+str(v.get('longLabel',''))).lower()
    if any(term in name for term in ['roadmap','chromhmm','state','brain','liver']):
        print(k, v.get('type'), v.get('shortLabel'), v.get('longLabel'))
PY"
