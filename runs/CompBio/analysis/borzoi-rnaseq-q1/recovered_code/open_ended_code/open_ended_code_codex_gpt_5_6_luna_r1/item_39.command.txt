/bin/bash -lc "python -u - <<'PY'
import json,time,numpy as np,torch
from borzoi_pytorch import Borzoi
print('load sequence',flush=True)
d=json.load(open('hg38_chr1_69993520_70517808.json'))
seq=d['dna'].upper()
if len(seq)"'!=524288: raise ValueError(len(seq))
# Borzoi/Baskerville nucleotide order is A,C,G,T.
x=np.zeros((1,len(seq),4),dtype=np.float32)
for i,nt in enumerate(seq):
    if nt in '"'ACGT': x[0,i,'ACGT'.index(nt)]=1.0
x=torch.from_numpy(x)
print('load model',flush=True)
model=Borzoi.from_pretrained('johahi/borzoi-replicate-0',cache_dir='./model_cache',local_files_only=True)
device='mps' if torch.backends.mps.is_available() else 'cpu'
model=model.to(device).eval(); x=x.to(device)
print('device',device,'input',tuple(x.shape),flush=True)
t0=time.time()
with torch.inference_mode():
    y=model(x,is_human=True)
print('predicted',time.time()-t0, type(y), tuple(y.shape), y.dtype,flush=True)
y_np=y.detach().to('cpu').numpy()
np.save('borzoi_rep0_prediction_all.npy',y_np)
target=y_np[0,:,6068]
np.save('borzoi_rep0_target_6068.npy',target)
print('target stats',target.shape,float(target.min()),float(target.max()),float(target.mean()),float(target.sum()),flush=True)
print('first',target[:10], 'last',target[-10:],flush=True)
PY"
