/bin/bash -lc "PYTHONPATH=vendor python - <<'PY'
from pypdf import PdfReader
r=PdfReader('borzoi.rnaseq.q1.pdf')
terms=['squash','squashed','transform','coverage transform','RNA-seq preprocessing','RNA-seq coverage','log2','log 2','quantile','ENCODE','replicate','reverse transform','preprocess','scale']
for i,p in enumerate(r.pages):
 t=p.extract_text() or ''
 hits=[]
 for term in terms:
  pos=0
  while True:
   j=t.lower().find(term.lower(),pos)
   if j<0: break
   hits.append((j,term)); pos=j+len(term)
 if hits:
  print('\\n===== PAGE',i+1,'=====')
  # merge contexts around hits
  ranges=[]
  for j,_ in hits: ranges.append((max(0,j-280),min(len(t),j+520)))
  ranges.sort(); merged=[]
  for a,b in ranges:
   if merged and a<=merged[-1][1]: merged[-1]=(merged[-1][0],max(merged[-1][1],b))
   else: merged.append((a,b))
  for a,b in merged[:30]: print(t[a:b].replace('\\n',' '))
PY"
