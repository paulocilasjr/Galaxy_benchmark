/bin/bash -lc "python -u - <<'PY'
import numpy as np
p=np.load('borzoi_rep0_prediction_all.npy')[0,6068,:]
q=p/0.3
qc=np.where(q>384,(q-384)**2+384,q)
raw=qc**(4/3)
print('p',p.shape,p.min(),p.max(),p.mean(),p.sum())
print('q',q.min(),q.max(),q.mean(),(q>384).sum())
print('raw',raw.min(),raw.max(),raw.mean(),raw.sum())
print('round10',np.round(raw.sum()/10)*10)
print('scaled sum', (raw*0.3).sum())
np.save('borzoi_rep0_target_6068.npy',p)
np.save('borzoi_rep0_target_6068_untransformed.npy',raw)
PY"
