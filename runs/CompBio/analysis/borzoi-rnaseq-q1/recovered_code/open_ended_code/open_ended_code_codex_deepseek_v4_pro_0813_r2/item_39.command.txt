import numpy as np

print('len',len(z),'sum',z.sum(),'max',z.max())
# inverse squash: if z<=384: y=z**(4/3); else r=384+(z-384)"'^2; y=r**(4/3)

print('"'y sum', y.sum(), 'max', y.max(), 'min', y.min())
# round nearest 10
s=float(y.sum())
print('s',s,'round10', round(s/10)*10)
# number above 384
print('above384',(z>384).sum(),'zmax',z.max())
# top bins
for i in np.argsort(z)[-10:]: print(i,z[i],y[i])
PY"
