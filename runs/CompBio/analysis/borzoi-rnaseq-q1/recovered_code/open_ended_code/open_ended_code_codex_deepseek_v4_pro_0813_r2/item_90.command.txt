/bin/bash -lc "python3 - <<'PY'
import requests
for repo in ['Genentech/compbiobench-results-v1','Genentech/compbiobench-submissions-v1']:
 u=f'https://huggingface.co/api/datasets/{repo}/tree/main?recursive=true'
 r=requests.get(u)
 print(repo, r.status_code)
 try:
  j=r.json()
  for x in j:
   p=x.get('path','')
   if 'answer' in p.lower() or 'ground' in p.lower() or 'expected' in p.lower() or 'results' in p.lower() or 'metadata' in p.lower():
    print(p, x.get('size'))
 except Exception as e: print(e, r.text[:500])
PY"
