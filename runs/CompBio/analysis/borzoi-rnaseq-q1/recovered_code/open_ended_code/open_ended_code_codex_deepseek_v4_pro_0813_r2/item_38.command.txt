/bin/bash -lc "python3 - <<'PY'
import torch, time, json, numpy as np
from borzoi_pytorch import Borzoi
seq=open('input.fa').read().split('\\n',1)[1].replace('\\n','')
print('seqlen', len(seq))
# one hot mapping
mapping={'A':0,'C':1,'G':2,'T':3}
idx=[mapping.get(c,0) for c in seq]
x=np.zeros((4,len(seq)), dtype=np.float32)
x[idx,np.arange(len(seq))]=1
xt=torch.from_numpy(x)[None]
print('x', xt.shape)
model=Borzoi.from_pretrained('johahi/borzoi-replicate-0')
model.eval()
# Try MPS
dev=torch.device('mps')
model.to(dev)
xt=xt.to(dev)
t=time.time()
with torch.no_grad():
    y=model(xt, is_human=True)
print('output', y.shape, 'time', time.time()-t)
# save track 6068 maybe CPU
ycpu=y[0,6068,:].float().cpu().numpy()
print(ycpu.min(), ycpu.max(), ycpu.sum(), ycpu[:10])
np.save('track_6068_mps.npy', ycpu)
PY"
