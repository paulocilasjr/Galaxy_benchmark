/bin/bash -lc "PYTORCH_ENABLE_MPS_FALLBACK=1 native_venv/bin/python - <<'PY'
import sys, time
sys.path.insert(0,'borzoi_pytorch_repo')
from borzoi_pytorch import Borzoi
from borzoi_pytorch.config_borzoi import BorzoiConfig
from safetensors.torch import load_file
import torch
print('torch',torch.__version__,'mps',torch.backends.mps.is_available(),flush=True)
t=time.time()
model=Borzoi(BorzoiConfig(return_center_bins_only=True,enable_mouse_head=False))
print('built',time.time()-t,flush=True)
t=time.time()
sd=load_file('assets/borzoi_rep0_model.safetensors',device='cpu')
print('loaded sd',len(sd),time.time()-t,flush=True)
missing,unexpected=model.load_state_dict(sd,strict=False)
print('state',len(missing),len(unexpected),missing[:5],unexpected[:5],flush=True)
print('params',sum(p.numel() for p in model.parameters()),flush=True)
PY"
