/bin/bash -c "python3 - <<'PY'
import numpy as np, pandas as pd
targets=pd.read_csv('borzoi_pytorch_repo/borzoi_pytorch/precomputed/targets.txt',sep='\\t')
print(targets.iloc[6068][['identifier','clip','clip_soft','scale','sum_stat']])

scale=float(targets.loc[6068,'scale']); clip=float(targets.loc[6068,'clip_soft'])

total=float(c.sum(dtype=np.float64))

print('x sum/max', x.sum(), x.max())
print('counts sum/max', c.sum(), c.max())
print('total', total)
print('nearest10', int(round(total/10)*10))
print('using Python round', round(total))
PY"
