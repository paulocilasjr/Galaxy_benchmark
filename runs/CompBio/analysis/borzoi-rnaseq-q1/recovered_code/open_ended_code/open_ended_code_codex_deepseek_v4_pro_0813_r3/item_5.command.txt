/bin/bash -c "python3 - <<'PY'
from pypdf import PdfReader
r=PdfReader('borzoi.rnaseq.q1.pdf')
with open('borzoi_full.txt','w') as f:
    for i,p in enumerate(r.pages):
        try: t=p.extract_text() or ''
        except Exception as e: t=f'[ERROR {e}]'
        f.write(f'\\n===== PAGE {i+1} =====\\n{t}\\n')
print('done')
PY
wc -l borzoi_full.txt; grep -n -i -E 'RNA-seq coverage|coverage target|target|log|transform|replicate|strand|ENCFF|experiment' borzoi_full.txt | head -200"
