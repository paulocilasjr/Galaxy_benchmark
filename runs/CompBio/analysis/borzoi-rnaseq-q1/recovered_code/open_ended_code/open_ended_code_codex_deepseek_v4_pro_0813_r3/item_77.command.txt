/bin/bash -c "python3 - <<'PY'
import gzip, time
start=69993520; end=70517808; length=end-start
pos=0; out=[]; got=False
with gzip.open('hg38.ml.fa.gz','rt') as f:
    header=f.readline().rstrip('\\n')
    print('header',header)
    assert header.startswith('>chr1')
    while len(out)<length:
        line=f.readline().rstrip('\\n')
        if not line: break
        if line.startswith('>'):
            print('next header before enough', line, pos, len(out)); break
        # collect relevant overlap
        line_start=pos; line_end=pos+len(line); pos=line_end
        if line_end <= start: continue
        if line_start >= end: break
        s=max(start,line_start); e=min(end,line_end)
        out.append(line[s-line_start:e-line_start])
seq=''.join(out)
print('pos',pos,'len',len(seq), 'first last', seq[:20], seq[-20:])
with open('ml_region_seq.txt','w') as f: f.write(seq)
PY"
