/bin/bash -lc "python3 - <<'PY'
import requests, json
base='https://jaspar.elixir.no/api/v1/matrix/'
# try search by name
for name in ['FOXA1','GATA3','MAX','STAT1','POU5F1','TP53','NANOG','HNF4A','TBP','FOS','MYC','JUN','FOXO1','CTCF','TCF7','SOX2','E2F1','CREB1','GATA1','KLF3']:
    try:
        r=requests.get(base, params={'search':name,'tax_group':'vertebrates','collection':'CORE','page_size':20}, timeout=10)
        data=r.json()
        mats=data.get('results',[])
        print('\\n',name,len(mats))
        for m in mats[:10]:
            print(m.get('matrix_id'), m.get('name'), m.get('collection'), m.get('version'), m.get('tax_id'))
    except Exception as e: print(name,e)
PY"
