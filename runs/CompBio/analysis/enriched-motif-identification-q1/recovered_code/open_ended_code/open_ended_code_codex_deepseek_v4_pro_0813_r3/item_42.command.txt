/bin/bash -lc "python3 - <<'PY'
import json
from pathlib import Path
from Bio.Seq import Seq
allm=json.loads(Path('motifs.json').read_text())
# add MA0139
import requests
for mid in ['MA0139.1','MA0139.2']:
 d=requests.get(f'https://jaspar.elixir.no/api/v1/matrix/{mid}/',timeout=10).json(); allm.append(d)
IUPAC={'AC':'M','AG':'R','AT':'W','CG':'S','CT':'Y','GT':'K','ACG':'V','ACT':'H','AGT':'D','CGT':'B','ACGT':'N'}
def cons(pfm):
 out=[]
 for col in zip(*(pfm[a] for a in 'ACGT')):
  best=max(col); letters=[a for a,n in zip('ACGT',col) if n==best]
  out.append(letters[0] if len(letters)==1 else IUPAC.get(''.join(sorted(letters)),'N'))
 return ''.join(out)
pat='CCACCAGGGGGCG'
print('motifs containing exact substring or rc')
for m in allm:
 c=cons(m['pfm']); rc=str(Seq(c).reverse_complement())
 if pat in c or pat in rc:
  print(m['matrix_id'],m['name'],m['version'],c,rc)
PY"
