/bin/bash -lc "python3 - <<'PY'
import requests, json, time
ids='MA0148.5 MA0037.5 MA0058.4 MA0059.2 MA0137.4 MA0517.2 MA1115.2 MA0142.1 MA0106.3 MA0525.2 MA0861.2 MA2339.1 MA0114.5 MA1494.2 MA0108.3 MA0476.2 MA0099.4 MA0147.4 MA0480.3 MA0139.2 MA1929.2 MA1930.2 MA0769.3 MA0143.5 MA0024.3 MA0018.5 MA0035.5 MA1516.2'.split()
out={}
s=requests.Session()
for mid in ids:
    url=f'https://jaspar.elixir.no/api/v1/matrix/{mid}/'
    for attempt in range(3):
        try:
            r=s.get(url, timeout=(5,20)); r.raise_for_status(); out[mid]=r.json(); break
        except Exception as e:
            if attempt==2: print('ERR', mid, repr(e))
            time.sleep(2)
    time.sleep(.1)
Path('matrices.json').write_text(json.dumps(out))
for mid,j in out.items():
    p=j['pfm']; bases=list(p)
    n=len(p['A'])
    cons=''.join(max(bases,key=lambda b:p[b][i]) for i in range(n))
    print(mid, j.get('name'), n, cons)
PY"
