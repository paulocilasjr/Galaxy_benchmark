/bin/bash -c "python3 - <<'PY'
import urllib.request,json
ids=['MA1516.1','MA1516.2','MA0147.3','MA0147.4','MA0058.4','MA0139.2','MA0018.5','MA0037.5','MA0114.5','MA0024.3','MA0480.3']
for mid in ids:
    url=f'https://jaspar.elixir.no/api/v1/matrix/{mid}/'
    req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0','Accept':'application/json'})
    try:
        js=json.loads(urllib.request.urlopen(req).read())
        print('\\nID',mid,'name',js.get('name'),'base_id',js.get('base_id'))
        pfm=js.get('pfm') or js.get('pwm')
        if pfm:
            print('len',len(pfm))
            for row in pfm[:30]: print(row)
        else: print('keys',js.keys())
    except Exception as e: print('ERR',mid,e)
PY"
