/bin/bash -c "python3 - <<'PY'
import urllib.request,json
names=['FOXA1','GATA3','MAX','STAT1','OCT4','P53','NANOG','HNF4A','TBP','FOS','MYC','JUN','FOXO1','CTCF','TCF7','SOX2','E2F1','CREB1','GATA1','KLF3']
url='https://jaspar.elixir.no/api/v1/matrix/?collection=CORE&tax_group=vertebrates&page_size=1000&format=json'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
js=json.loads(urllib.request.urlopen(req).read())
res=js['results']
for n in names:
    matches=[r for r in res if r['name'].upper()==n or r['name'].upper().startswith(n+'.') or n in r['name'].upper()]
    print('\\n',n)
    for r in matches:
        print(r['matrix_id'],r['name'])
PY"
