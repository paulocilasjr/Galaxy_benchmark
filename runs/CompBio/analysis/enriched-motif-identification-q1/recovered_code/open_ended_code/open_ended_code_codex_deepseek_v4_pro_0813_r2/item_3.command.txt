/bin/bash -c "python3 - <<'PY'
import urllib.request, json
url='https://jaspar.elixir.no/api/v1/matrix/?collection=CORE&tax_group=vertebrates&page_size=1000&format=json'
try:
    req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    data=urllib.request.urlopen(req, timeout=20).read()
    print(len(data))
    js=json.loads(data)
    print(js.keys())
    print('count', js.get('count'))
    for x in js.get('results',[])[:3]: print(x.keys())
except Exception as e:
    print('ERR',e)
PY"
