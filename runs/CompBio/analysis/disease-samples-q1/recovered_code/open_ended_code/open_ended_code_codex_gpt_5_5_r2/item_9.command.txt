/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
fn='disease.samples.q1.tsv'
df=pd.read_csv(fn, sep='\\t', index_col=0)
samps=df.columns.tolist()
genes=['XIST','TSIX','DDX3Y','EIF1AY','KDM5D','UTY','RPS4Y1','RPS4Y2','ZFY','SRY','KDM6A','USP9Y']
print(df.loc[[g for g in genes if g in df.index]].to_string())
# DE between clusters
A=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
B=['Sample_2','Sample_3','Sample_5','Sample_7']
X=np.log2(df+1)
res=[]
for g,row in X.iterrows():
    ma=row[A].mean(); mb=row[B].mean(); diff=ma-mb
    # original means
    res.append((g,diff,df.loc[g,A].mean(),df.loc[g,B].mean(),row[A].mean(),row[B].mean()))
res=sorted(res, key=lambda x:x[1], reverse=True)
print('\\nTop A>B')
for r in res[:60]: print(r[0], f'logdiff={r[1]:.2f}', f'TPM_A={r[2]:.1f}', f'TPM_B={r[3]:.1f}')
print('\\nTop B>A')
for r in res[-60:][::-1]: print(r[0], f'logdiff={r[1]:.2f}', f'TPM_A={r[2]:.1f}', f'TPM_B={r[3]:.1f}')
PY"
