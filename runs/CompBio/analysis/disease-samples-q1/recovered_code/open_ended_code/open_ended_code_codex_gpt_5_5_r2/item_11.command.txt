/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
fn='disease.samples.q1.tsv'
df=pd.read_csv(fn, sep='\\t', index_col=0)
markers=['S100A8','S100A9','S100A7','SERPINB3','SERPINB4','KRT6A','KRT16','IL36G','IL36A','DEFB4A','PI3','LCN2','CXCL8','NOS2','IFI27','GJB2']
healthy=['FLG2','KRT77','SCARA5','H19','TNXB','IGFBP5','FBLN1','TIMP3','ADH1B','CFD']
print('Disease marker TPM')
print(df.loc[[g for g in markers if g in df.index]].round(2).to_string())
print('\\nHealthy/normal-high marker TPM')
print(df.loc[[g for g in healthy if g in df.index]].round(2).to_string())
X=np.log2(df+1)
for name, genes in [('disease',markers),('normal_high',healthy)]:
    present=[g for g in genes if g in X.index]
    z=X.loc[present].T
    z=(z-z.mean())/z.std(ddof=0)
    score=z.mean(axis=1)
    print('\\n',name,'score')
    print(score.sort_values(ascending=False).round(3).to_string())
PY"
