/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
p='disease.samples.q1.tsv'
df=pd.read_csv(p,sep='\\t',index_col=0)
print(df.shape)
print(df.columns.tolist())
print(df.isna().sum().sum())
print((df<0).sum().sum())
print(df.sum(axis=0).head())
# log
X=np.log2(df.values+1)
from sklearn.decomposition import PCA
pca=PCA(n_components=min(9,X.shape[0]))
scores=pca.fit_transform(X.T)
print('explained',pca.explained_variance_ratio_[:6])
print('scores')
for c,s in zip(df.columns,scores):
 print(c, ' '.join(f'{x:.3f}' for x in s[:6]))
# correlations
C=pd.DataFrame(X.T, columns=df.columns).corr()
print(C.round(3))
PY"
