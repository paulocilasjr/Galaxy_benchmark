/bin/bash -lc "python -u - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
from scipy.cluster.hierarchy import linkage, leaves_list, fcluster
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
y=np.log2(x+1)
v=y.var(axis=1).sort_values(ascending=False)
sel=y.loc[v.head(3000).index]
print('TOPVAR')
for g in v.head(40).index: print(g, *(f'{z:.2f}' for z in y.loc[g]))
print('CORR')
print(sel.corr().round(2).to_string())
Z=linkage(sel.T,method='ward')
print('LEAVES',','.join(sel.columns[leaves_list(Z)]))
print('CLUSTER2',dict(zip(sel.columns,fcluster(Z,2,criterion='maxclust'))))
pca=PCA(5); p=pca.fit_transform(sel.T)
print('EVR',pca.explained_variance_ratio_)
print(pd.DataFrame(p,index=x.columns).round(2).to_string())
PY"
