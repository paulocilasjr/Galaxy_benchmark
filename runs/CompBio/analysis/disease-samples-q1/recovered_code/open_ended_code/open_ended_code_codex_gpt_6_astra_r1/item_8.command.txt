/bin/bash -lc "python -u - <<'PY'
import numpy as np,pandas as pd
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
log=np.log2(x+1)
a=log.values.T;a=a-a.mean(axis=0)
u,s,v=np.linalg.svd(a,full_matrices=False)
print('PCA variance', s*s/(s*s).sum())
print(pd.DataFrame(u[:,:3]*s[:3],index=x.columns).round(2))
for k in [0,1]:
 z=pd.Series(v[k],index=x.index)
 print('PC',k+1,'positive',z.nlargest(25).round(3).to_dict(),'negative',z.nsmallest(25).round(3).to_dict())
genes=['KRT1','KRT2','KRT10','KRT5','KRT14','KRT6A','KRT16','KRT17','FLG','FLG2','LOR','IVL','S100A7','S100A8','S100A9','S100A12','DEFB4A','DEFB4B','PI3','IL36A','IL36G','IL17A','IL17F','IL22','IL23A','TNF','IFNG','CXCL1','CXCL8','CXCL9','CXCL10','CCL20','IL4','IL13','CCL17','CCL22','KRT15','MKI67','CD3D','CD8A','CD74']
print(x.reindex(genes).round(1).to_string())
print(log.corr().round(3))
PY"
