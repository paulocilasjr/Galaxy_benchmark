/bin/bash -lc "cat > analyze.py <<'PY'
import pandas as pd,numpy as np
from sklearn.decomposition import PCA
from scipy.cluster.hierarchy import linkage,leaves_list
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
l=np.log2(x+1)
print('shape',x.shape,'sums',x.sum().round().to_dict())
print('top overall',x.mean(axis=1).nlargest(45).round(1).to_string())
print('PCA')
p=PCA(n_components=5).fit(l.T)
print(p.explained_variance_ratio_)
print(pd.DataFrame(p.transform(l.T),index=x.columns).round(3))
print('correlation',l.corr().round(3).to_string())
for k in range(3):
 z=pd.Series(p.components_[k],index=x.index)
 print('PC',k+1,'positive',z.nlargest(35).to_string(),'negative',z.nsmallest(35).to_string(),sep='\\n')
print('top per sample')
for c in x:
 print(c, x[c].nlargest(20).round(1).to_dict())
l.to_pickle('log_tpm.pkl')
PY
OPENBLAS_NUM_THREADS=1 python analyze.py > analysis.txt
cat analysis.txt"
