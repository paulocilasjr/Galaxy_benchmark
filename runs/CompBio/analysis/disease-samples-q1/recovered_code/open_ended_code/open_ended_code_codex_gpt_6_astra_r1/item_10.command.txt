import numpy as np
import pandas as pd
x = pd.read_csv('disease.samples.q1.tsv', sep='\\t', index_col=0)

a -= a.mean(axis=0)

print('PCA explained variance:', s**2 / np.sum(s**2))
print(pd.DataFrame(u[:, :3]*s[:3], index=x.columns))
markers = ['S100A7', 'S100A8', 'S100A9', 'DEFB4A', 'PI3', 'KRT16', 'IL36G', 'FLG2', 'LOR']

print(','.join(diseased))
PY
