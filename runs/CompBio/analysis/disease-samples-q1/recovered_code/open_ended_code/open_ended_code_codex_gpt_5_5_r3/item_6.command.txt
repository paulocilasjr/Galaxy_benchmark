/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from scipy.cluster.hierarchy import linkage, leaves_list, fcluster
from scipy.spatial.distance import squareform

fn='disease.samples.q1.tsv'
df=pd.read_csv(fn,sep='\\t',index_col=0)
print(df.shape)
print(df.columns.tolist())
print(df.describe().loc[['mean','std','50%','75%','max']])
log=np.log2(df+1)
# sample stats
print('\\nSample totals TPM')
print(df.sum().sort_values())
print('\\nDetected genes >1 TPM')
print((df>1).sum().sort_values())
# correlations
corr=log.corr(method='pearson')
print('\\nCorrelation')
print(corr.round(3).to_string())
# distance clustering
D=1-corr
Z=linkage(squareform(D.values, checks=False), method='average')
print('leaves', [df.columns[i] for i in leaves_list(Z)])
for k in [2,3,4]:
    labs=fcluster(Z,k,criterion='maxclust')
    print('k',k,{c:[] for c in set(labs)})
    groups={}
    for s,l in zip(df.columns,labs): groups.setdefault(l,[]).append(s)
    print(groups)
# PCA on genes variable; samples rows
X=log.T.values
# center genes not scale to avoid noisy low genes maybe
pca=PCA(n_components=5)
pcs=pca.fit_transform(X-X.mean(axis=0))
print('\\nPCA var',pca.explained_variance_ratio_)
for s, coords in zip(df.columns, pcs):
    print(s, ' '.join(f'{x:.3f}' for x in coords[:5]))
# PCA with gene scaling maybe top variable genes only
var=log.var(axis=1)
top=var.sort_values(ascending=False).head(5000).index
X=log.loc[top].T.values
pca=PCA(n_components=5)
pcs=pca.fit_transform(StandardScaler(with_std=True).fit_transform(X))
print('\\nPCA scaled top var',pca.explained_variance_ratio_)
for s, coords in zip(df.columns, pcs): print(s, ' '.join(f'{x:.3f}' for x in coords[:5]))
PY"
