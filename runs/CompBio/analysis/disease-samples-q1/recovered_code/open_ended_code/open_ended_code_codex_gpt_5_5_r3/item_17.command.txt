/bin/bash -lc "python3 -u - <<'PY'
import math, statistics
fn='disease.samples.q1.tsv'
G1={'Sample_2','Sample_3','Sample_5','Sample_7'}
with open(fn) as f:
    header=f.readline().rstrip('\\n').split('\\t')[1:]
    idx1=[i for i,s in enumerate(header) if s in G1]
    idx2=[i for i,s in enumerate(header) if s not in G1]
    rows=[]
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        gene=p[0]; vals=[float(x) for x in p[1:]]
        m1=sum(vals[i] for i in idx1)/len(idx1); m2=sum(vals[i] for i in idx2)/len(idx2)
        l1=sum(math.log2(vals[i]+1) for i in idx1)/len(idx1); l2=sum(math.log2(vals[i]+1) for i in idx2)/len(idx2)
        # variance within log
        s1=statistics.pstdev([math.log2(vals[i]+1) for i in idx1]) if len(idx1)>1 else 0
        s2=statistics.pstdev([math.log2(vals[i]+1) for i in idx2]) if len(idx2)>1 else 0
        rows.append((gene,m1,m2,l1-l2,m1/(m2+1e-9),m2/(m1+1e-9),s1,s2,vals))
print('G1', [header[i] for i in idx1], 'G2', [header[i] for i in idx2])
print('\\nTop G1 by logdiff with mean TPM>2')
for gene,m1,m2,ld,fc,_,s1,s2,vals in sorted([r for r in rows if r[1]>2 or r[2]>2], key=lambda r:r[3], reverse=True)[:80]:
    print(f'{gene}\\tld={ld:.2f}\\tm1={m1:.1f}\\tm2={m2:.1f}\\tfc={fc:.1f}\\tsd={s1:.2f}/{s2:.2f}')
print('\\nTop G2 by logdiff with mean TPM>2')
for gene,m1,m2,ld,_,fc2,s1,s2,vals in sorted([r for r in rows if r[1]>2 or r[2]>2], key=lambda r:r[3])[:80]:
    print(f'{gene}\\tld={-ld:.2f}\\tm2={m2:.1f}\\tm1={m1:.1f}\\tfc={fc2:.1f}\\tsd={s2:.2f}/{s1:.2f}')
# selected markers
markers='GFAP AQP4 SLC1A2 ALDH1L1 MBP PLP1 MOG MOBP C1QA C1QB C1QC CX3CR1 P2RY12 AIF1 CD68 LYZ TYROBP TREM2 APOE SPP1 CD3D CD79A MS4A1 PECAM1 VWF COL1A1 ACTA2 DES MYH11 TNNT2 MYH6 MYH7 NPPA NPPB ACTC1 CKM MYH2 MYH1 MYLPF'.split()
print('\\nMarkers')
d={r[0]:r for r in rows}
for g in markers:
    if g in d:
        gene,m1,m2,ld,fc,fc2,s1,s2,vals=d[g]
        print(f'{g}\\tG1={m1:.2f}\\tG2={m2:.2f}\\tlogdiff={ld:.2f}\\t'+'\\t'.join(f'{header[i]}:{vals[i]:.1f}' for i in range(len(header))))
PY"
