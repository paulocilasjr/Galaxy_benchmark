/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import AgglomerativeClustering, KMeans
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, leaves_list, fcluster

fn='disease.samples.q1.tsv'
df=pd.read_csv(fn, sep='\\t', index_col=0)
print(df.shape)
print(df.columns.tolist())
X=np.log2(df+1)
# filter variable expressed genes
var=X.var(axis=1)
for n in [500,1000,2000,5000,10000]:
    top=X.loc[var.sort_values(ascending=False).head(n).index]
    M=top.T
    Z=StandardScaler(with_mean=True, with_std=True).fit_transform(M)
    p=PCA(n_components=4).fit_transform(Z)
    print('\\nN',n,'PCA')
    for s,coords in zip(df.columns,p):
        print(s, ['%.3f'%c for c in coords[:3]])
    print('explained', PCA(n_components=4).fit(Z).explained_variance_ratio_[:4])
    for k in [2,3,4]:
        km=KMeans(n_clusters=k, random_state=1, n_init=20).fit(Z)
        print('kmeans k',k, dict(zip(df.columns, km.labels_)))
    D=pdist(Z, metric='euclidean')
    L=linkage(D, method='ward')
    print('ward order', [df.columns[i] for i in leaves_list(L)])
    print('ward k2', dict(zip(df.columns, fcluster(L,2,criterion='maxclust'))))
    print('ward k3', dict(zip(df.columns, fcluster(L,3,criterion='maxclust'))))
PY"
