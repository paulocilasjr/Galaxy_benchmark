/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import ttest_ind
fn='disease.samples.q1.tsv'
df=pd.read_csv(fn, sep='\\t', index_col=0)
A=['Sample_2','Sample_3','Sample_5','Sample_7']
B=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
X=np.log2(df+1)
meanA=X[A].mean(axis=1); meanB=X[B].mean(axis=1)
lfc=meanA-meanB
# expression threshold and tstat
p=[]
for g,row in X.iterrows():
    try:
        p.append(ttest_ind(row[A], row[B], equal_var=False).pvalue)
    except Exception:
        p.append(np.nan)
res=pd.DataFrame({'meanA':meanA,'meanB':meanB,'lfc_A_vs_B':lfc,'p':p,'raw_meanA':df[A].mean(axis=1),'raw_meanB':df[B].mean(axis=1)})
res['abs']=res.lfc_A_vs_B.abs()
res=res[(res.raw_meanA+res.raw_meanB)>1]
print('Top A high (2,3,5,7):')
print(res.sort_values('lfc_A_vs_B', ascending=False).head(100).to_string())
print('\\nTop B high (1,4,6,8,9):')
print(res.sort_values('lfc_A_vs_B', ascending=True).head(100).to_string())
PY"
