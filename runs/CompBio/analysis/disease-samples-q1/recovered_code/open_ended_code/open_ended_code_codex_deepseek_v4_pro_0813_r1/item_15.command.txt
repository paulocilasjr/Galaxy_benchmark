/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.cluster.hierarchy import linkage, fcluster
from scipy.spatial.distance import pdist
p='disease.samples.q1.tsv'; df=pd.read_csv(p,sep='\\t',index_col=0)
X=np.log1p(df.values.T)
# var top 2000
idx=np.argsort(X.var(axis=0))[::-1][:2000]
Z=linkage(X[:,idx], method='ward')
for k in [2,3]:
    labels=fcluster(Z,k,criterion='maxclust')
    print(k, {c: df.columns[labels==c].tolist() for c in np.unique(labels)})
# correlations
C=np.corrcoef(X)
print('corr matrix sample labels', df.columns.tolist())
print(np.round(C,3))
PY"
