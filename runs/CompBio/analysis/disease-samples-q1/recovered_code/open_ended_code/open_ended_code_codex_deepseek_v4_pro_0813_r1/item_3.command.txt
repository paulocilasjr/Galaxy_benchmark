/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
p='disease.samples.q1.tsv'
df=pd.read_csv(p,sep='\\t',index_col=0)
A=['Sample_2','Sample_3','Sample_5','Sample_7']
B=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
X=df.values
meanA=np.log1p(X[:, [df.columns.get_loc(c) for c in A]]).mean(axis=1)
meanB=np.log1p(X[:, [df.columns.get_loc(c) for c in B]]).mean(axis=1)
lfc=meanA-meanB
# avg exp
avg=(meanA+meanB)/2
# t-test
from scipy import stats
pv=[]
for i in range(X.shape[0]):
    a=np.log1p(X[i,[df.columns.get_loc(c) for c in A]])
    b=np.log1p(X[i,[df.columns.get_loc(c) for c in B]])
    _,p=stats.ttest_ind(a,b,equal_var=False)
    pv.append(p if np.isfinite(p) else 1)
pv=np.array(pv)
q=pd.Series(pv).rank()/len(pv)*1 # pseudo
out=pd.DataFrame({'gene':df.index,'meanA':meanA,'meanB':meanB,'lfc':lfc,'p':pv})
print('Up in A (putative disease if A):')
print(out.nsmallest(50,'p').sort_values('lfc',ascending=False).head(50)[['gene','meanA','meanB','lfc','p']].to_string(index=False))
print('\\nUp in B:')
print(out.nsmallest(50,'p').sort_values('lfc',ascending=True).head(50)[['gene','meanA','meanB','lfc','p']].to_string(index=False))
PY"
