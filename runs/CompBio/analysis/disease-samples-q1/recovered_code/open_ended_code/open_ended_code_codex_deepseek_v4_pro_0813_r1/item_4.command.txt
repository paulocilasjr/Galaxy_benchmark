import pandas as pd, numpy as np
p='disease.samples.q1.tsv'; df=pd.read_csv(p,sep='\\t',index_col=0)
groups={'A':['Sample_2','Sample_3','Sample_5','Sample_7'],'B':['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']}
markers=['KRT16','S100A7','S100A8','S100A9','DEFB4A','DEFB4B','PI3','SERPINB3','SERPINB4','SPRR2A','SPRR2B','SPRR1B','LCN2','CXCL1','IL36G','IL17A','IL22','OASL','CHAC1','GJB2','HEPHL1','TYMP','ZC3H12A','LCE3A','LCE5A']

for g in groups:
    vals={}
    for m in markers:
        idx=np.where(df.index.str.upper()==m.upper())[0]
        if len(idx):
