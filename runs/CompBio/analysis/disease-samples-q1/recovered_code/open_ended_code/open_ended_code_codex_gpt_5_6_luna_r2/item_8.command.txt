/bin/bash -lc "python -u - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, leaves_list, dendrogram
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0).values
ids=[f'Sample_{i}' for i in range(1,10)]
for name,a in [('log1p',np.log1p(x)),('log2',np.log2(x+0.1))]:
 print('\\n',name)
 cor=np.corrcoef(a.T)
 print('corr')
 print(pd.DataFrame(cor,index=ids,columns=ids).round(4).to_string())
 d=squareform(pdist(a.T,metric='correlation'))
 print('corrdist')
 print(pd.DataFrame(d,index=ids,columns=ids).round(4).to_string())
 z=PCA().fit_transform(StandardScaler().fit_transform(a))
 p=PCA().fit(StandardScaler().fit_transform(a))
 print('PCA var',p.explained_variance_ratio_[:5].round(4))
 print(pd.DataFrame(z[:,:3],index=ids,columns=['PC1','PC2','PC3']).round(3).to_string())
 print('linkage avg order', [ids[i] for i in leaves_list(linkage(pdist(a.T,metric='correlation'),method='average'))])
 print('linkage ward order', [ids[i] for i in leaves_list(linkage(a.T,method='ward'))])
PY"
