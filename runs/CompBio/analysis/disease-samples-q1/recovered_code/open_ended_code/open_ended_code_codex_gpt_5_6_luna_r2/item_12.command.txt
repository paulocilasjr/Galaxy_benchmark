/bin/bash -lc "python -u - <<'PY'
import pandas as pd,numpy as np
from scipy.stats import ttest_ind
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
A=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
B=['Sample_2','Sample_3','Sample_5','Sample_7']
y=np.log2(x+1)
ma=y[A].mean(axis=1); mb=y[B].mean(axis=1)
lfc=ma-mb
res=pd.DataFrame({'Amean_log2':ma,'Bmean_log2':mb,'log2FC_A-B':lfc,'Amean_TPM':x[A].mean(axis=1),'Bmean_TPM':x[B].mean(axis=1)})
res['p']=ttest_ind(y[A],y[B],axis=1,equal_var=False).pvalue
# rank by effects with expression in either group
for label, sub in [('A_up',res[(res['Amean_log2']>1)|(res['Bmean_log2']>1)].sort_values('log2FC_A-B',ascending=False).head(80)),('B_up',res[(res['Amean_log2']>1)|(res['Bmean_log2']>1)].sort_values('log2FC_A-B').head(80))]:
 print('\\n###',label)
 print(sub.round(3).to_string())
print('\\nTop tstat')
res['t']=lfc/np.sqrt(y[A].var(axis=1,ddof=1)/len(A)+y[B].var(axis=1,ddof=1)/len(B))
print(res.sort_values('t',ascending=False).head(40).round(3).to_string())
print('\\nB t')
print(res.sort_values('t').head(40).round(3).to_string())
PY"
