/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
X=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
# clusters by kmeans 2 on log2
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
log=np.log2(X+1)
km=KMeans(n_clusters=2,n_init=10,random_state=0).fit(log.T)
for s,l in zip(log.columns,km.labels_): print(s,l)
# marker score using psoriatic upregulated list
up=['SERPINB4','S100A7A','S100A9','SPRR2A','S100A8','S100A7','PI3','DEFB4A','SPRR2B','SPRR2F','TCN1','KRT6C','HEPHL1','RHCG','SERPINB3','KRT16','SPRR2D','TMPRSS11D','KRT6A','LCE3E','AKR1B10','IFI27','LCE3A','LCN2','ATP12A','LCE3D','SPRR2G','GDA']
down=['BTC']
score=log.loc[[g for g in up if g in X.index]].mean()-log.loc[[g for g in down if g in X.index]].mean()
print(score.sort_values())
# differential mean logFC
group_d=log.columns[km.labels_==1]
group_h=log.columns[km.labels_==0]
print('d',group_d.tolist(),'h',group_h.tolist())
if log[group_d].mean(axis=1).loc[up].mean() > log[group_h].mean(axis=1).loc[up].mean():
    print('kmeans group1 has high up')
else:
    print('kmeans group0 has high up')
PY"
