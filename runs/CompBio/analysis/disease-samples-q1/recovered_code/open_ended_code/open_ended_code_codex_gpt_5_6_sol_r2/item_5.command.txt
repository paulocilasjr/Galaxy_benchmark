/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
from scipy.cluster.hierarchy import linkage, leaves_list
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
y=np.log2(x+1)
# Filter expressive and variable genes
v=y.var(axis=1)
sel=v.nlargest(5000).index
z=y.loc[sel].T
p=PCA().fit_transform(z)
print('PCA variance and scores')
print(PCA().fit(z).explained_variance_ratio_[:5])
print(pd.DataFrame(p[:,:5],index=x.columns).round(2))
print('\\npearson log correlations')
print(y.corr().round(3))
print('\\ncluster order')
print(list(y.columns[leaves_list(linkage(y.T,method='ward'))]))
print('\\nTop variable genes')
print(pd.DataFrame({'var':v.loc[sel[:80]], 'min':y.loc[sel[:80]].min(axis=1),'max':y.loc[sel[:80]].max(axis=1)}).head(80).round(2).to_string())
PY"
