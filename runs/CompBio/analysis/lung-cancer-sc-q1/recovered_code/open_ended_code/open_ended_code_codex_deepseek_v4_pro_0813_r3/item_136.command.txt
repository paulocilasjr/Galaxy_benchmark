/bin/bash -c "python - <<'PY'
import pandas as pd
res=pd.read_csv('centroid_pred_celltype_tumor.csv')
sub=pd.read_csv('sample_subtype.csv').rename(columns={'sample':'sampleID'})
res=res.merge(sub,on='sampleID',how='left')
immune_prefixes=('T cell','NK cell','B cell','Plasma cell','Macrophage','Monocyte','Mast cell','Neutrophils','cDC','pDC','DC mature','myeloid dividing')
non=~res.pred.str.startswith(immune_prefixes)
for lusc_patients in [True,False]:
 r=res[(res.subtype=='LUSC')] if lusc_patients else res
 num_all=r.pred.isin(['Tumor cells LUSC','Tumor cells LUSC mitotic']).sum()
 num_nonmit=(r.pred=='Tumor cells LUSC').sum()
 num_mit=(r.pred=='Tumor cells LUSC mitotic').sum()
 den=non[res.index.isin(r.index)].sum() if lusc_patients else non.sum()
 print('LUSC patients only',lusc_patients,'den nonimmune all samples?',den,'all',num_all,'nonmit',num_nonmit,'mit',num_mit,'pct all',100*num_all/den,'pct nonmit',100*num_nonmit/den)
# denominator using nonimmune all samples always; numerator LUSC patient predictions
den=non.sum()
r=res[res.subtype=='LUSC']
print('numerator LUSC patients among all nonimmune')
for kind,n in [('all',r.pred.isin(['Tumor cells LUSC','Tumor cells LUSC mitotic']).sum()),('nonmit',(r.pred=='Tumor cells LUSC').sum()),('mit',(r.pred=='Tumor cells LUSC mitotic').sum())]: print(kind,n,100*n/den)
PY"
