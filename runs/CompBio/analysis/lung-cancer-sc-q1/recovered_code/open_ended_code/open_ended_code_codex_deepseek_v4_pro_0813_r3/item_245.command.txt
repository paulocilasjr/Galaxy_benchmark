/bin/bash -lc "python - <<'PY'
import pandas as pd
cs=pd.read_pickle('cell_scores.pkl')
for s in ['Patient_018','Patient_040']:
 d=cs[cs.sampleID==s]
 # dendritic cells: any specific DC score positive maybe DC_general or cDC1/cDC2/pDC/DC_mature
 for name, sel in [('any_specific', (d[['cDC1','cDC2','pDC','DC_mature']]>0).any(axis=1)),
                   ('DC_general>0',d.DC_general>0),
                   ('DC_general>0.1',d.DC_general>0.1),
                   ('max_specific>0',d[['cDC1','cDC2','pDC','DC_mature']].max(axis=1)>0)]:
  print(s,name,sel.sum(),len(d),round(sel.mean()*100,2))
PY"
