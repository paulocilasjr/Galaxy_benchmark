/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
S=pd.read_pickle('cell_scores.pkl')
sub=pd.read_csv('sample_subtype.csv')
lusc=sub.loc[sub.subtype=='LUSC','sample'].values
mask=(S.LUSC_basal>0.3)&(S.epithelial>0.1)&(S.sampleID.isin(lusc))
for t_prolif in [0,0.05,0.1,0.2,0.3,0.5,1.0]:
 print('prolif<',t_prolif,'count',(mask&(S.prolif<t_prolif)).sum())
# denominator using nonimmune simple CD45 threshold maybe
for t_cd45 in [0.05,0.1,0.2,0.3]:
 non=((S.immune_CD45<t_cd45)&((S.epithelial>0.05)|(S.endothelial>0.05)|(S.fibroblast>0.05)|(S.smooth_muscle>0.05)|(S.LUSC_basal>0.1)|(S.LUAD>0.1)))
 print('cd45<',t_cd45,'non',non.sum())
 for t_prolif in [0,0.2,0.3]:
  num=(mask&(S.prolif<t_prolif)).sum()
  print('  num',num,'pct',100*num/non.sum())
PY"
