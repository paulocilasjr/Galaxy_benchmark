/bin/bash -lc "python - <<'PY'
import requests, xml.etree.ElementTree as ET
# fetch ids for Patient_018
r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':'Patient_018','retmax':100},timeout=30)
root=ET.fromstring(r.text)
ids=[e.text for e in root.findall('.//Id')]
print('ids',ids)
rr=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':','.join(ids)},timeout=30)
rroot=ET.fromstring(rr.text)
for doc in rroot.findall('DocSum'):
    uid=doc.find('Id').text
    title=doc.find(\"Item[@Name='title']\").text if doc.find(\"Item[@Name='title']\") is not None else ''
    summary=doc.find(\"Item[@Name='summary']\").text if doc.find(\"Item[@Name='summary']\") is not None else ''
    gse=doc.find(\"Item[@Name='GSE']\").text if doc.find(\"Item[@Name='GSE']\") is not None else ''
    print(uid, gse, title[:250], summary[:300])
PY"
