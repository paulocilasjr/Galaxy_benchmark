/bin/bash -lc "python - <<'PY'
import anndata as ad, pandas as pd, numpy as np, scipy.sparse as sp
a=ad.read_h5ad('lung.cancer.sc.h5ad',backed='r')
# marker genes
markers={'CD1C':None,'CLEC10A':None,'CLEC9A':None,'XCR1':None,'IL3RA':None,'CLEC4C':None,'CCR7':None,'LAMP3':None,'FCER1A':None}
var_sym=a.var['symbol'].astype(str).str.upper().to_numpy()
for g in list(markers):
    idx=np.where(var_sym==g)[0]
    markers[g]=idx[0] if len(idx) else None
    print(g,markers[g])
# per sample count cells expressing >=1 of these markers
samples=a.obs['sampleID'].to_numpy()
codes=pd.Categorical(samples).codes
for s in ['Patient_018','Patient_040']:
    sel=samples==s
    # load X for selected rows and marker columns
    idx_cols=np.array([v for v in markers.values() if v is not None])
    X=a.X[sel,:][:,idx_cols]
    if sp.issparse(X): X=X.toarray()
    pos=(X>0).any(axis=1)
    print(s,'cells',sel.sum(),'any DC marker',pos.sum(),'pct',pos.mean()*100)
    # average expression per marker
    means=X.mean(axis=0)
    for g,i in zip(markers.keys(),range(len(markers))):
        if markers[g] is not None: print(g, round(means[i],4), end=' ')
    print()
PY"
