/bin/bash -c "python - <<'PY'
import h5py, time
p='lung.cancer.sc.h5ad'
t=time.time()
with h5py.File(p,'r') as f:
    print('keys', list(f.keys()))
    for k in f.keys():
        o=f[k]
        print(k, type(o).__name__, dict(o.attrs) if hasattr(o,'attrs') else '')
    print('obs keys', list(f['obs'].keys()))
    print('var keys', list(f['var'].keys()))
    print('obsm keys', list(f['obsm'].keys()) if 'obsm' in f else 'none')
    print('varm keys', list(f['varm'].keys()) if 'varm' in f else 'none')
    print('obsp keys', list(f['obsp'].keys()) if 'obsp' in f else 'none')
    print('uns keys', list(f['uns'].keys()) if 'uns' in f else 'none')
    obs=f['obs']
    for k in obs.keys():
        print('obs',k, type(obs[k]).__name__, getattr(obs[k],'shape',None), dict(obs[k].attrs) if hasattr(obs[k],'attrs') else '')
    var=f['var']
    for k in var.keys():
        print('var',k, type(var[k]).__name__, getattr(var[k],'shape',None), dict(var[k].attrs) if hasattr(var[k],'attrs') else '')
    X=f['X']
    print('X',type(X).__name__, X.shape, dict(X.attrs) if hasattr(X,'attrs') else '')
    if isinstance(X,h5py.Group):
        print('X keys',list(X.keys()))
        for k in X.keys():
            print('X',k,type(X[k]).__name__,X[k].shape, dict(X[k].attrs) if hasattr(X[k],'attrs') else '')
    print('elapsed',time.time()-t)
PY"
