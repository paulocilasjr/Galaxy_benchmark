/bin/bash -c "python - <<'PY'
import numpy as np, pandas as pd
Mo=np.load('Mo.npy'); Ml=np.load('Ml.npy')
ours=pd.Index(np.load('our_samples.npy',allow_pickle=True))
luca=pd.Index(np.load('luca_samples.npy',allow_pickle=True))
# center rows
mo=Mo-Mo.mean(1,keepdims=True)
ml=Ml-Ml.mean(1,keepdims=True)
mo/=np.linalg.norm(mo,axis=1,keepdims=True)
ml/=np.linalg.norm(ml,axis=1,keepdims=True)
C=mo @ ml.T
print('C shape',C.shape)
top=np.argsort(C,axis=1)[:,::-1][:,:10]
for i,s in enumerate(ours):
    vals=[(luca[j],round(C[i,j],4)) for j in top[i]]
    print(s, vals)
# best per target
for s in ['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']:
    i=np.where(ours==s)[0][0]
    j=C[i].argmax()
    print('TARGET',s,'best',luca[j],C[i,j])
PY"
