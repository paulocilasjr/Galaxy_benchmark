/bin/bash -lc "python - <<'PY'
import h5py
p='lung.cancer.sc.h5ad'
with h5py.File(p,'r') as f:
 print('top',list(f.keys()))
 def show(g,path='',depth=0):
  if depth>2:return
  for k,v in g.items():
   pp=f'{path}/{k}'
   if isinstance(v,h5py.Dataset): print(pp, v.shape, v.dtype, dict(v.attrs))
   else:
    print(pp,'GROUP',dict(v.attrs))
    if path in ('','/obs','/uns'): show(v,pp,depth+1)
 show(f)
PY"
