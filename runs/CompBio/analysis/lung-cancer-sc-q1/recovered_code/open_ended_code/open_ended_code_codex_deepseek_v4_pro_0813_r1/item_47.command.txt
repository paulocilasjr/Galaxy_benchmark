/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
s=pd.read_parquet('scores.parquet')
# Marker-based broad classification
# immune if any immune score > 0.5? Define broad types max score among sets.
immune_scores=['Tcell','NK','Bcell','Plasma','Myeloid','DC','Mast']
# nonimmune scores
nonimmune_scores=['Epithelial','Endothelial','Fibroblast']
s['immune_max']=s[immune_scores].max(axis=1)
s['nonimmune_max']=s[nonimmune_scores].max(axis=1)
# classify immune if immune_max > nonimmune_max maybe
s['pred_immune']=s.immune_max>s.nonimmune_max
print('immune pred',s.pred_immune.sum(),'nonimmune',(~s.pred_immune).sum())
# LUSC patients list as above
lusc_list=['Patient_013','Patient_014','Patient_018','Patient_026','Patient_027','Patient_028','Patient_029','Patient_030','Patient_031','Patient_040','Patient_063','Patient_064','Patient_065','Patient_066','Patient_067']
# malignant basal = nonimmune & LUSC_basal > thresholds
for th in [0.5,1,1.5,2]:
 num=((~s.pred_immune)&s.sampleID.isin(lusc_list)&(s.LUSC_basal>th)).sum()
 den=(~s.pred_immune).sum()
 print('th',th,'num',num,'den',den,'ratio',num/den,'nearest20',round(num/den*100/20)*20)
# nonimmune as EPCAM>0.5 or endo/fibro score >1
ni=(s.EPCAM>0.5)|(s.Endothelial>0.5)|(s.Fibroblast>0.5)
print('marker ni',ni.sum())
for th in [0.5,1,1.5,2]:
 num=(ni&s.sampleID.isin(lusc_list)&(s.LUSC_basal>th)).sum()
 print('ni2 th',th,num,num/ni.sum(),round(num/ni.sum()*100/20)*20)
PY"
