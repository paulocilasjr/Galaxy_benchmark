import anndata as ad
import h5py, time
f='lung.cancer.sc.h5ad'
t=time.time()

    print('obs keys', list(h['obs'].keys()))
    for k in h['obs'].keys():

    print('shape', h['obs']['_index'].shape, h['var']['_index'].shape)
    print('obsm', list(h['obsm'].keys()) if 'obsm' in h else None)
    print('uns', list(h['uns'].keys())[:30] if 'uns' in h else None)
print('h5py time',time.time()-t)
PY"
