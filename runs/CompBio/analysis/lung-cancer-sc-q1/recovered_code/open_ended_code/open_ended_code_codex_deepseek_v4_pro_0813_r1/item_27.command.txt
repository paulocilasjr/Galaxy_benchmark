import h5py, numpy as np
f1='lung.cancer.sc.h5ad'; f2='remote_adata.h5ad'
with h5py.File(f1,'r') as h: obs1=np.array([x.decode() for x in h['obs/_index'][:]])
with h5py.File(f2,'r') as h:
    obs2=np.array([x.decode() for x in h['obs/_index'][:]])
    cats=np.array([x.decode() for x in h['obs/dataset']['categories'][:]])
    codes=h['obs/dataset']['codes'][:]
    dsarr=cats[codes]
    patient_cats=np.array([x.decode() for x in h['obs/patient']['categories'][:]])
    pcodes=h['obs/patient']['codes'][:]
    patarr=patient_cats[pcodes]
    cond_cats=np.array([x.decode() for x in h['obs/condition']['categories'][:]])
    ccodes=h['obs/condition']['codes'][:]
    condarr=cond_cats[ccodes]
    origin_cats=np.array([x.decode() for x in h['obs/origin']['categories'][:]])
    ocodes=h['obs/origin']['codes'][:]
    originarr=origin_cats[ocodes]
def barcode(x):
    return x.rsplit('_',1)[-1].rsplit('-',1)[0]
bc1=np.array([barcode(x) for x in obs1]); bc2=np.array([barcode(x) for x in obs2])
interset=set(bc1)
for ds in np.unique(dsarr):
    mask=dsarr==ds
    bc_sub=np.array([barcode(x) for x in obs2[mask]])
    inter=len(interset & set(bc_sub))
    if inter:
        print(ds, 'n', mask.sum(), 'inter', inter)
PY"
