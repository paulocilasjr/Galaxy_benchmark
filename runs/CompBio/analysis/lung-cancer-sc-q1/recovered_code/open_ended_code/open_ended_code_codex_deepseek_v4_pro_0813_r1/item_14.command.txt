/bin/bash -c "python - <<'PY'
import h5py, numpy as np
f='remote_adata.h5ad'
with h5py.File(f,'r') as h:
    print('keys', list(h.keys()))
    print('obs keys', list(h['obs'].keys()))
    for k in h['obs'].keys():
        obj=h['obs'][k]
        if isinstance(obj,h5py.Dataset):
            print('obs', k, obj.shape, obj.dtype, obj[:5])
        else:
            print('obs cat', k, 'cats', len(obj['categories'][:]), obj['categories'][:5] if 'categories' in obj else '')
            print('codes', obj['codes'][:10])
    print('uns', list(h['uns'].keys()) if 'uns' in h else None)
    print('X attrs', dict(h['X'].attrs))
    print('obs _index first', h['obs']['_index'][:5])
PY"
