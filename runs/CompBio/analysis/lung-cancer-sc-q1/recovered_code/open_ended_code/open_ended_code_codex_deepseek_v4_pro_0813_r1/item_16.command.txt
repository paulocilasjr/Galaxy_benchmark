/bin/bash -c "python - <<'PY'
import h5py, numpy as np
f='remote_adata.h5ad'
with h5py.File(f,'r') as h:
 def cat(h,k):
  cats=np.array([x.decode() for x in h[k]['categories'][:]]); codes=h[k]['codes'][:]; return cats,codes
 for k in ['obs/origin','obs/origin_fine','obs/tissue','obs/condition','obs/dataset']:
  cats,codes=cat(h,k); print(k, cats.tolist(), 'counts', np.bincount(codes, minlength=len(cats)).tolist())
PY"
