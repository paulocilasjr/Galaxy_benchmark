/bin/bash -lc "python -u - <<'PY'
from urllib.request import urlopen
from io import BytesIO
import pandas as pd
print('start',flush=True)
u='https://media.springernature.com/original/springer-static/esm/art%3A10.1038%2Fs41392-021-00824-9/MediaObjects/41392_2021_824_MOESM2_ESM.xlsx'
b=urlopen(u).read(); print('got',len(b),flush=True); x=pd.ExcelFile(BytesIO(b)); print(x.sheet_names,flush=True)
for s in x.sheet_names:
 d=pd.read_excel(BytesIO(b),sheet_name=s); print('\\n',s,d.shape,list(d.columns),flush=True); print(d.head(5).to_string(index=False)[:4000],flush=True)
PY"
