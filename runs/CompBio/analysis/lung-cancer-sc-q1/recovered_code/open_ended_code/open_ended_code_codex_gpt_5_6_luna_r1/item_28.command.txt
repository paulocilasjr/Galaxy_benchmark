/bin/bash -lc "python -u - <<'PY'
import anndata as ad,numpy as np,h5py,pandas as pd
A=ad.read_h5ad('lung.cancer.sc.h5ad',backed='r'); sy=np.asarray(A.var['symbol'].astype(str))
sets={'basal':['KRT5','KRT14','KRT17','TP63','SOX2','KRT6A','KRT13','KRT15','KRT16','S100A2','S100A7','SFN','CLCA2','TRIM29','DSC3','DSG3','SERPINB13','GPC1','FSCN1','GPNMB','HOXA1','FGFBP1','PERP','SLC6A8'],'luad':['NAPSA','NKX2-1','SFTPB','SFTPA1','SFTPA2','SFTPC','SFTPD','SCGB1A1','SCGB3A2','AQP5','MUC1','WFDC2','FOLR1','SLC34A2','EPCAM','KRT8','KRT18','KRT19'],'epi':['EPCAM','KRT8','KRT18','KRT19','KRT5','KRT14','KRT17','KRT6A','KRT13','KRT15','KRT16','MUC1','SFN'],'immune':['PTPRC'],'myeloid':['LYZ','LST1','TYROBP','FCER1G','CTSS','AIF1','LGALS3','CSTA','FCGR3A','S100A8','S100A9','S100A10','CTSD','MS4A7'],'dc':['FCER1A','CD1C','CLEC10A','CLEC9A','CST3','HLA-DRA','HLA-DPA1','HLA-DPB1','HLA-DQB1','CD74','COTL1','FCER1G','IRF8','BATF3']}
allgs=list(dict.fromkeys(g for x in sets.values() for g in x)); present=[g for g in allgs if np.any(sy==g)]; idx=[int(np.flatnonzero(sy==g)[0]) for g in present];mi={g:i for i,g in enumerate(present)}
X=A.X[:,idx].toarray().astype(np.float32)
with h5py.File('lung.cancer.sc.h5ad','r') as f:
 ip=np.asarray(f['X/indptr']); data=f['X/data']; rows=np.diff(ip); totals=np.zeros(len(rows),dtype=np.int64)
 for st in range(0,len(rows),10000):
  en=min(len(rows),st+10000); a=int(ip[st]); b=int(ip[en]); nz=np.flatnonzero(rows[st:en]);
  if b>a: totals[st:en][nz]=np.add.reduceat(np.asarray(data[a:b],dtype=np.int64),ip[st:en][nz]-a,dtype=np.int64)
Z=np.log1p(X/totals[:,None]*1e4)
scores={k:Z[:,[mi[g] for g in gs if g in mi]].mean(axis=1) for k,gs in sets.items()}
samples=np.asarray(A.obs['sampleID'].astype(str))
print('global score quantiles')
for k,v in scores.items(): print(k,np.round(np.quantile(v,[0,.5,.75,.9,.95,.975,.99,.995,.999,1]),3))
print('\\ncounts by sample')
rows=[]
for s in sorted(np.unique(samples),key=lambda z:int(z.split('_')[1])):
 m=samples==s; row={'sample':s,'n':m.sum()}
 for k,v in scores.items():
  row[k+'mean']=v[m].mean(); row[k+'>.3']=(v[m]>.3).sum(); row[k+'>.5']=(v[m]>.5).sum(); row[k+'>.7']=(v[m]>.7).sum(); row[k+'>1']=(v[m]>1).sum()
 rows.append(row)
df=pd.DataFrame(rows).set_index('sample'); pd.set_option('display.max_rows',100);pd.set_option('display.width',240);pd.set_option('display.max_columns',100)
print(df[['n','basalmean','basal>.3','basal>.5','basal>.7','basal>1','luadmean','luad>.5','luad>1','epi>.3','immune>.3']].round(3).to_string())
print('\\nall basal > .5',int((scores['basal']>.5).sum()),'basal >1',int((scores['basal']>1).sum()),'PTPRC+',(X[:,mi['PTPRC']]>0).sum())
PY"
