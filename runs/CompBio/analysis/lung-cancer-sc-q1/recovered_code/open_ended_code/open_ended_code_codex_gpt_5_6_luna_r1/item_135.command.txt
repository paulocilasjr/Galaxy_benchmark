import anndata as ad, numpy as np
p='lung.cancer.sc.h5ad'
a=ad.read_h5ad(p,backed='r')
syms=np.asarray(a.var['symbol'].values).astype(str)
genes=['PTPRC','KRT5','KRT14','KRT17','TP63','KRT6A','KRT13','S100A2','SFN','KRT16','SFTPC','SFTPB','SFTPA1','NAPSA','NKX2-1','SCGB1A1','EPCAM','KRT8','KRT18','KRT19','FCER1A','CD1C','CLEC10A','HLA-DRA','CD74']
idx=[int(np.where(syms==g)[0][0]) for g in genes]
print('selecting',len(idx))
X=a[:,idx].X.to_memory().toarray()
s=a.obs['sampleID'].astype(str).values
for patient in ['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']:
 ix=s==patient; non=ix & (X[:,0]==0)
 print('\\n',patient,'all',ix.sum(),'non',non.sum())
 print('all med/mean',np.median(X[ix],axis=0).round(2).tolist(),np.mean(X[ix],axis=0).round(2).tolist())
 print('non >0',[(g,int((non&(X[:,j]>0)).sum())) for j,g in enumerate(genes[1:])])
PY"
