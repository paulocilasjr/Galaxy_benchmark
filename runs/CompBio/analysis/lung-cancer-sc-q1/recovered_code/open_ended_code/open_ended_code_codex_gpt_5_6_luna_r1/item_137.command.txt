/bin/bash -lc "python - <<'PY'
import h5py,numpy as np
p='lung.cancer.sc.h5ad'
with h5py.File(p,'r') as f:
 syms=np.array([x.decode() for x in f['var']['symbol'][:]])
 genes=['PTPRC','KRT5','KRT14','KRT17','TP63','KRT6A','KRT13','S100A2','SFN','KRT16','SFTPC','FCER1A','CD1C','CLEC10A','HLA-DRA','CD74']
 idx=[np.where(syms==g)[0][0] for g in genes]
 cats=np.array([x.decode() for x in f['obs']['sampleID']['categories'][:]]); codes=f['obs']['sampleID']['codes'][:]
 indptr=f['X']['indptr'][:]; inds=f['X']['indices']; data=f['X']['data']
 for patient in ['Patient_018','Patient_040','Patient_005']:
  ixs=np.where(cats[codes]==patient)[0]
  print('\\n',patient)
  # print rows with KRT5 count > 0, max 5, and rows with KRT17
  for i in ixs:
   a,b=indptr[i],indptr[i+1]
   d=dict(zip(inds[a:b],data[a:b]))
   if d.get(idx[1],0)>0 or d.get(idx[3],0)>0:
    print('row',i,'sel',[(g,int(d.get(j,0))) for g,j in zip(genes,idx)],'nnz',b-a)
    if i>ixs[0]+10000: break
PY"
