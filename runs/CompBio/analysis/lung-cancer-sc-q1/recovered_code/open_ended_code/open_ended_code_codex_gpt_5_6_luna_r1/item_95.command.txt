/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
with ZipFile('immune_supplement.docx') as z:
 root=ET.fromstring(z.read('word/document.xml'))
 ns={'w':'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
 tbls=root.findall('.//w:tbl',ns)
 print('tables',len(tbls))
 for ti,tbl in enumerate(tbls[:3]):
  rows=tbl.findall('./w:tr',ns)
  print('TABLE',ti,'rows',len(rows))
  for row in rows[:5]:
   cells=row.findall('./w:tc',ns)
   print([ ''.join(x.text or '' for x in c.findall('.//w:t',ns)) for c in cells])
PY"
