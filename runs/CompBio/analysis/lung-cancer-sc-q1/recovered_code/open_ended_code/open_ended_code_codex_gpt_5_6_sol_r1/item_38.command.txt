/bin/bash -lc "python -u - <<'PY'
import csv
from collections import Counter
p='original_meta.csv'
cs={}
with open(p,newline='') as f:
 r=csv.DictReader(f)
 print(r.fieldnames)
 n=0
 for row in r:
  n+=1
  for c in r.fieldnames:
   cs.setdefault(c,Counter())[row[c]]+=1
print('n',n)
for c,v in cs.items():
 print('\\n',c,len(v),v.most_common(30))
PY"
