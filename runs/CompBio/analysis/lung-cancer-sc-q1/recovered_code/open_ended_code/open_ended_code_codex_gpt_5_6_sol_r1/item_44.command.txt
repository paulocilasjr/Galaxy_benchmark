/bin/bash -lc "python -u - <<'PY'
import csv
from collections import Counter
immune={'B','CD4','CD8','DC','Gran','Mast','Mo','NK','Tregs'}
c=Counter()
with open('original_meta.csv',newline='') as f:
 for r in csv.DictReader(f):
  if r['Disease'] not in ('LUAD','LUSC'):continue
  nonimmune=r['CellName'] not in immune
  c['all']+=1
  c['nonimmune']+=nonimmune
  c['basal_lusc']+=(r['CellName']=='Basal' and r['Disease']=='LUSC')
  c['basal_lusc_tumor']+=(r['CellName']=='Basal' and r['Disease']=='LUSC' and r['Tissue']=='Tumor')
  c[(r['Disease'],r['CellName'],r['Tissue'])]+=1
print(c['all'],c['nonimmune'],c['basal_lusc'],c['basal_lusc_tumor'])
for num in ['basal_lusc','basal_lusc_tumor']:
 print(num,100*c[num]/c['nonimmune'])
for k,v in sorted(c.items(),key=lambda x:str(x[0])):
 if isinstance(k,tuple) and k[1] in ('Basal','ATII','Epi','Cilia','Club','EC','Fib','NE'):print(k,v)
PY"
