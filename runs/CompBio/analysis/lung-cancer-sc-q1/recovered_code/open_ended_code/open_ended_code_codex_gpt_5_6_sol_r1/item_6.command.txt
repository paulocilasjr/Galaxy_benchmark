/bin/bash -lc "python -u - <<'PY'
import h5py
f=h5py.File('lung.cancer.sc.h5ad','r')
def walk(g,p='',depth=0,maxdepth=3):
 for k,v in g.items():
  q=f'{p}/{k}'
  if isinstance(v,h5py.Group):
   print(q,'GROUP',dict(v.attrs))
   if depth<maxdepth: walk(v,q,depth+1,maxdepth)
  else: print(q,'DATA',v.shape,v.dtype,dict(v.attrs))
walk(f['uns'],'/uns',0,5)
print('obsm'); walk(f['obsm'],'/obsm',0,2)
print('X'); x=f['X']; print(type(x), list(x.keys()) if isinstance(x,h5py.Group) else x.shape,dict(x.attrs))
print('layers'); walk(f['layers'],'/layers',0,2)
print('var'); print(list(f['var'].keys()))
PY"
