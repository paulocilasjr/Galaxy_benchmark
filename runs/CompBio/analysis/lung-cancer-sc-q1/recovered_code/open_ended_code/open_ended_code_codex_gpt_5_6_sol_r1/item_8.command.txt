import h5py
f=h5py.File('lung.cancer.sc.h5ad','r')
for k in f['var']:
 o=f['var'][k]
 if isinstance(o,h5py.Group):
  print(k,'group',list(o.keys()),dict(o.attrs))
  for z in o: print(' ',z,o[z].shape,o[z].dtype)
 else: print(k,'dataset',o.shape,o.dtype,dict(o.attrs))
for key in ['symbol','ID']:
 o=f['var'][key]
 if isinstance(o,h5py.Group):
  cats=o['categories'][:20]
  print(key,[x.decode() if isinstance(x,bytes) else str(x) for x in cats])
PY"
