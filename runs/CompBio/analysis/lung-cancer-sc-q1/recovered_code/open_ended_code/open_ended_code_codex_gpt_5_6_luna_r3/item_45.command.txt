/bin/bash -lc "python -u - <<'PY'
import urllib.request, io, pandas as pd
print('fetch',flush=True)
b=urllib.request.urlopen('https://lungcancer.chenlulab.com/api/download/?name=meta.csv',timeout=120).read(); print('bytes',len(b),flush=True)
m=pd.read_csv(io.BytesIO(b)); print('shape',m.shape,flush=True); print(m.columns.tolist(),flush=True); print(m.head().to_string(index=False),flush=True)
print('samples',m.SampleID.nunique(), 'patients',m.PatientID.nunique(), 'diseases',m.Disease.value_counts().to_dict(),flush=True)
print(m.groupby(['SampleID','PatientID','Disease','Stage']).size().to_string(),flush=True)
PY"
