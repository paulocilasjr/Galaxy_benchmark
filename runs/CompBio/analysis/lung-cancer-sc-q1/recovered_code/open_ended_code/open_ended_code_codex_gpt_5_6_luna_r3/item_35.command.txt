/bin/bash -lc "python -u - <<'PY'
import urllib.request, zipfile, io, re, xml.etree.ElementTree as ET
url='https://media.springernature.com/original/springer-static/esm/art%3A10.1038%2Fs41392-021-00824-9/MediaObjects/41392_2021_824_MOESM1_ESM.docx'
b=urllib.request.urlopen(url).read(); print('bytes',len(b))
z=zipfile.ZipFile(io.BytesIO(b)); xml=z.read('word/document.xml')
root=ET.fromstring(xml); ns={'w':'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
text='\\n'.join(''.join(t.text or '' for t in p.findall('.//w:t',ns)) for p in root.findall('.//w:p',ns))
print(text[:30000])
PY"
