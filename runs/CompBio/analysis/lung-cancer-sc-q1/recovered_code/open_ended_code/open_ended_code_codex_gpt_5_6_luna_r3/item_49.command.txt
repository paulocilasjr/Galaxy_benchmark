/bin/bash -lc "python -u - <<'PY'
import pandas as pd, anndata as ad, numpy as np
m=pd.read_csv('source_meta.csv')
print('meta',m.shape,'samples',m.SampleID.nunique(),'patients',m.PatientID.nunique(),'disease',m.Disease.value_counts().to_dict(),flush=True)
print('sample counts top',m.groupby('SampleID').size().sort_values(ascending=False).head(10).to_string(),flush=True)
A=ad.read_h5ad('lung.cancer.sc.h5ad',backed='r')
idx=pd.Index(A.obs_names.astype(str))
task=pd.DataFrame({'taskid':idx,'sample':A.obs['sampleID'].astype(str).to_numpy()})
task['barcode']=task.taskid.str.rsplit('_',n=1).str[-1]
print('task',task.shape,'samples',task.sample.nunique(),flush=True)
# source barcode sets and index maps
m['barcode']=m['Cells'].astype(str).str.rsplit('_',n=1).str[-1]
src={s:set(g.barcode) for s,g in m.groupby('SampleID')}
rows=[]
for s,g in task.groupby('sample',sort=True):
 ts=set(g.barcode)
 scores=sorted([(len(ts&bs),ss,len(bs)) for ss,bs in src.items()],reverse=True)
 rows.append((s,len(g),scores[0],scores[1] if len(scores)>1 else None))
print('mapping top:')
for row in rows: print(row,flush=True)
PY"
