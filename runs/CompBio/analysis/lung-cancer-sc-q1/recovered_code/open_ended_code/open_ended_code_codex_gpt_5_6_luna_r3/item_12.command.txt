/bin/bash -lc "python -u - <<'PY'
import anndata as ad, numpy as np, pandas as pd
p='lung.cancer.sc.h5ad'
a=ad.read_h5ad(p, backed='r')
print('genes head',a.var[['symbol','ID']].head(20).to_string(index=False))
print('gene symbols tail',a.var['symbol'].tail(20).to_list())
print('unique symbols',a.var['symbol'].nunique(), 'n',a.n_vars)
# backed sparse chunks per rows; compute total counts and detected genes
samples=a.obs['sampleID'].astype(str).to_numpy(); uniq=sorted(set(samples))
# Access X per sample: likely backed sparse element with slicing
rows=[]
for s in uniq:
 idx=np.flatnonzero(samples==s)
 x=a.X[idx,:]
 # x may csr
 tc=np.asarray(x.sum(axis=1)).ravel()
 ng=np.asarray((x>0).sum(axis=1)).ravel()
 rows.append((s,len(idx),tc.mean(),np.median(tc),tc.max(),ng.mean(),np.median(ng)))
print(pd.DataFrame(rows,columns=['sample','cells','mean_counts','median_counts','max_counts','mean_genes','median_genes']).to_string(index=False))
PY"
