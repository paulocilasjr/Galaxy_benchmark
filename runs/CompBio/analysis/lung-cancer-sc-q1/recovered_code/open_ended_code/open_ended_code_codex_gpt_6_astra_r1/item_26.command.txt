/bin/bash -lc "python - <<'PY'
import pandas as pd
x=pd.read_csv('results/initial_cell_scores.csv',index_col=0)
x['epi_score']=x[['Epi','Basal','AT2','Club','Ciliated']].max(axis=1)
x['nonimm_score']=x[['Fibro','Endo','Pericyte','Smooth','Epi','Basal','AT2','Club','Ciliated']].max(axis=1)
x['imm_score']=x[['T','NK','B','Plasma','Myeloid','Mast']].max(axis=1)
x['nonimm']=(x.nonimm_score>x.imm_score)&(x.Immune<1)
x['basal']=(x.Basal>0.6)&x.nonimm&(x.Basal>x.AT2)
print(x.groupby('sampleID').agg(n=('sampleID','size'),nonimm=('nonimm','sum'),basal=('basal','sum')).to_string())
print('Overall',x.nonimm.sum(),x.basal.sum(),x.basal.sum()/x.nonimm.sum())
PY"
