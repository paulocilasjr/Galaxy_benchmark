import urllib.request,urllib.parse,json
u='https://www.ebi.ac.uk/biostudies/api/v1/search?collection=arrayexpress&query='+urllib.parse.quote('lung AND single cell')+'&pageSize=100'

for x in d['hits']:print(x['accession'],x['title'])
PY"
