/bin/bash -lc "file results/AllSamplesCelllview.R && python - <<'PY'
import pandas as pd,json
s=pd.read_csv('results/cell_scores.csv',index_col=0)
candidates=['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']
nonimmune=[25,26,27,30,31,32,33,34,35,37,38,39,40,41,42,43,44,46,48,50]
basal=[26,27,39,42,43,44,46]
s['dendritic']=s.cluster.isin([11,14,36]); s['nonimmune']=s.cluster.isin(nonimmune);s['basal']=s.cluster.isin(basal)
r=s[s.sampleID.isin(candidates)].groupby('sampleID').agg(total=('sampleID','size'),dendritic=('dendritic','sum'))
r['pct_dc']=100*r.dendritic/r.total;r.to_csv('results/candidate_dc_proportions.csv')
summary={'candidate_dendritic_counts':r.to_dict('index'),'basal_count':int(s.basal.sum()),'nonimmune_count':int(s.nonimmune.sum()),'basal_percent':100*s.basal.sum()/s.nonimmune.sum()}
json.dump(summary,open('results/calculated_proportions.json','w'),indent=2)
print(r.to_string());print(summary['basal_percent'])
PY"
