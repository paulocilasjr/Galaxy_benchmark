/bin/bash -c "python - <<'PY'
import anndata as ad, numpy as np, scipy.sparse as sp, pandas as pd, time
A=ad.read_h5ad('lung.cancer.sc.h5ad', backed='r')
X=A.X[:].tocsr()
obs=A.obs['sampleID'].values
symbols=A.var['symbol'].astype(str).values
lib=np.asarray(X.sum(1)).ravel()
# normalize
norm=X.copy().astype(np.float64)
norm.data=np.log1p(norm.data / lib[norm.indices] * 10000)
markers=['PTPRC','EPCAM','KRT5','KRT6A','KRT14','TP63','SOX2','DSC3','NKX2-1','NAPSA','SFTPB','FCER1A','CD1C','CLEC9A','LILRA4','CD68','CD3D','CD79A','VWF','PECAM1','DCN','COL1A1','ACTA2']
# get vals
for g in markers:
 idx=np.where(symbols==g)[0]
 if len(idx)==0: continue
 v=np.asarray(norm[:,idx[0]].todense()).ravel() if sp.issparse(norm[:,idx[0]]) else np.asarray(norm[:,idx[0]]).ravel()
 print('\\n',g, 'mean', v.mean(), 'q', np.percentile(v,[50,75,90,95,99,99.5,99.9]))
 # positive count >0.5 or 1
 for th in [0.5,1,1.5,2]:
  print(' >',th, (v>th).sum())
PY"
