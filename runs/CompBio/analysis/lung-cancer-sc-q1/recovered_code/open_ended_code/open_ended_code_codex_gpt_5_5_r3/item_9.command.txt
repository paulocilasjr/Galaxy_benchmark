/bin/bash -lc "python - <<'PY'
import h5py
p='lung.cancer.sc.h5ad'
with h5py.File(p,'r') as f:
    def visit(name, obj):
        if isinstance(obj, h5py.Dataset):
            print('DATA', name, obj.shape, obj.dtype)
            for k,v in obj.attrs.items():
                print('  attr', k, repr(v)[:200])
        else:
            print('GRP ', name)
            for k,v in obj.attrs.items():
                print('  attr', k, repr(v)[:200])
    f.visititems(visit)
PY"
