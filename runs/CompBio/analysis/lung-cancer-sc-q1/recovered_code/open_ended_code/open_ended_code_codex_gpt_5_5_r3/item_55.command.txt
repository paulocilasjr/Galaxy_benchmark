/bin/bash -lc "python - <<'PY'
import pandas as pd, requests
j=pd.read_csv('annotated_benchmark_cells.csv')
# benchmark source cell IDs match source format: source_sample + '_' + barcode portion after Patient_xxx_
j['barcode']=j['bench_cell'].str.replace(r'"'^Patient_'"\\d+_', '', regex=True)
j['source_cell']=j['source_sample']+'_'+j['barcode']
nonimmune=['Epi','ATII','Basal','Cilia','Club','EC','Fib','NE']
den=len(j[j.CellName.isin(nonimmune)])
r=requests.get('https://lungcancer.chenlulab.com/api/coord', params={'cell':'Basal','disease':'LUSC','coord':'umap','group':'cluster'}, timeout=60)
data=r.json()['data']
mal=set()
cluster_counts={}
for cl,arr in data.items():
    cluster_counts[cl]=len(arr)
    for row in arr:
        mal.add(row[3])
print('api malignant basal LUSC cells', len(mal), cluster_counts)
inter=j[j.source_cell.isin(mal)]
print('intersection', len(inter), 'den', den, 'pct', 100*len(inter)/den)
print('inter cell labels/disease')
print(pd.crosstab(inter.Disease, inter.CellName).to_string())
print('missing malignant not in benchmark', len(mal-set(j.source_cell)))
# if denominator all nonimmune source full/process? compute with all source meta tumor and all source for reference intersection all source denominator
meta=pd.read_csv('source_meta.csv', low_memory=False)
print('source meta nonimmune all', len(meta[meta.CellName.isin(nonimmune)]), 100*len(mal)/len(meta[meta.CellName.isin(nonimmune)]))
print('source meta tumor nonimmune', len(meta[(meta.Tissue=='Tumor') & meta.CellName.isin(nonimmune)]), 100*len(mal)/len(meta[(meta.Tissue=='Tumor') & meta.CellName.isin(nonimmune)]))
# selected sample LUSC dc highest among list
sel=['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']
print('selected LUSC only')
for s in sel:
    x=j[j.bench_sample==s]
    if x.Disease.iloc[0]=='LUSC':
        print(s, len(x), (x.CellName=='DC').sum(), 100*(x.CellName=='DC').mean())
PY"
