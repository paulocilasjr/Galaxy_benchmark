/bin/bash -lc "python - <<'PY'
import pandas as pd, anndata as ad
from collections import defaultdict
adata=ad.read_h5ad('lung.cancer.sc.h5ad', backed='r')
obs=adata.obs.copy()
obs['bench_cell']=obs.index.astype(str)
obs['bench_sample']=obs['sampleID'].astype(str)
obs['barcode']=obs['bench_cell'].str.replace(r'"'^Patient_'"\\d+_', '', regex=True)
meta=pd.read_csv('source_meta.csv', usecols=['SampleID','Cells','PatientID','Disease','CellName','Tissue'])
meta['barcode']=meta['Cells'].str.replace(r'"'^[''^_]+_'"', '', regex=True)
# sizes
print('meta rows', len(meta), 'samples', meta.SampleID.nunique(), 'tissues', meta.Tissue.value_counts().to_dict())
print('benchmark rows', len(obs), 'samples', obs.bench_sample.nunique())
# For each benchmark sample, compute intersections with source sample barcode sets. optimize by barcode maps
source_sets={sid:set(g['barcode']) for sid,g in meta.groupby('SampleID')}
bench_sets={sid:set(g['barcode']) for sid,g in obs.groupby('bench_sample')}
rows=[]
for bs,bset in bench_sets.items():
    best=[]
    for ss,sset in source_sets.items():
        inter=len(bset & sset)
        if inter:
            best.append((inter, ss, len(bset), len(sset)))
    best=sorted(best, reverse=True)[:5]
    for inter,ss,lb,ls in best:
        rows.append((bs, ss, inter, lb, ls, inter/lb, inter/ls))
print('mapping top:')
for r in rows[:100]:
    print(r)
# Write full top mapping CSV
pd.DataFrame(rows, columns=['bench','source','inter','bench_n','source_n','frac_bench','frac_source']).to_csv('sample_mapping_top.csv', index=False)
PY"
