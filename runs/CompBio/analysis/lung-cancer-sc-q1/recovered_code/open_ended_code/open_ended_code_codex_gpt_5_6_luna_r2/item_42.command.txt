/bin/bash -lc "python - <<'PY'
import h5py, pandas as pd, numpy as np
m=pd.read_csv('site_meta.csv',low_memory=False)
# map site sample to set bare barcode
site_sets={s:set(sub['Cells'].astype(str).str.rsplit('_',n=1).str[-1]) for s,sub in m.groupby('SampleID',sort=False)}
with h5py.File('lung.cancer.sc.h5ad','r') as f:
 idx=f['obs/_index'][:].astype(str); codes=f['obs/sampleID/codes'][:]; cats=f['obs/sampleID/categories'][:].astype(str)
 for i,cat in enumerate(cats):
  ix=np.where(codes==i)[0]; vals=idx[ix]
  suffix={v.split('_',2)[2] for v in vals}
  candidates=[s for s,ss in site_sets.items() if len(suffix & ss)==len(suffix) and len(ss)==len(suffix)]
  if len(candidates)"'!=1:
   # match intersection max
   scores=sorted(((len(suffix&ss),len(ss),s) for s,ss in site_sets.items()),reverse=True)[:3]
   print(cat,len(ix),'"'candidates',candidates,'top',scores)
  else:
   print(cat,len(ix),candidates[0])
PY"
