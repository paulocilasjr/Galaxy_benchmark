/bin/bash -lc "python - <<'PY'
import urllib.request, re
base='https://lungcancer.chenlulab.com'
for path in ['/static/js/app.17ef3077.js','/static/js/chunk-vendors.e66fc360.js']:
    txt=urllib.request.urlopen(base+path,timeout=30).read().decode('utf-8','ignore')
    print('\\nPATH', path, 'len', len(txt))
    for pat in ['api','download','data','gene','cell','sample','patient','annotation','txt','csv','xlsx']:
        print(pat, txt.lower().find(pat))
    # print snippets around axios urls
    for m in re.finditer(r'(?:url:|baseURL:|axios|get\\(|post\\(|/api|/static|download|data)["'^'"\\n]{0,200}', txt):
        s=m.group(0)
        if any(x in s.lower() for x in ['api','download','data','cell','sample','gene','csv','txt','xlsx']):
            print(s[:500])
PY"
