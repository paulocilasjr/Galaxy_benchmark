/bin/bash -lc "python - <<'PY'
import csv,h5py,collections
src_sets=collections.defaultdict(set)
src_info={}
with open('source_meta.csv',newline='') as f:
 for r in csv.DictReader(f):
  sid=r['SampleID']; cell=r['Cells']; pref=sid+'_'
  bc=cell[len(pref):] if cell.startswith(pref) else cell.rsplit('_',1)[-1]
  src_sets[sid].add(bc)
  src_info[sid]=(r['Tissue'],r['Disease'],r['PatientID'])
with h5py.File('lung.cancer.sc.h5ad','r') as f:
 idx=[x.decode() for x in f['obs/_index'][:]]
 cats=[x.decode() for x in f['obs/sampleID/categories'][:]]
 codes=f['obs/sampleID/codes'][:]
our=collections.defaultdict(set)
for cell,code in zip(idx,codes):
 p=cats[int(code)]
 our[p].add(cell[len(p)+1:])
for p in cats:
 a=our[p]
 scores=[]
 for sid,b in src_sets.items():
  inter=len(a&b)
  if inter: scores.append((inter, inter/len(a), inter/len(b),sid,len(a),len(b),src_info[sid]))
 scores.sort(reverse=True)
 print(p, scores[:3])
PY"
