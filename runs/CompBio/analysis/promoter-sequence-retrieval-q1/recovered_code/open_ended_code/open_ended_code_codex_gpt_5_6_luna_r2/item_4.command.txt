/bin/bash -lc "curl -L --fail --silent --show-error 'https://rest.ensembl.org/lookup/symbol/danio_rerio/shha?expand=1' -H 'Content-Type: application/json' > evidence/shha_lookup.json
printf 'length='; wc -c < evidence/shha_promoter.txt
printf 'alphabet='; tr -d '\\n' < evidence/shha_promoter.txt | sed 's/[ACGT]//g' | wc -c
printf 'sha256='; sha256sum evidence/shha_promoter.txt | awk '{print "'$1}'"'
printf 'first_last='; sed 's/\\n//g' evidence/shha_promoter.txt | cut -c1-20; printf ' ... '; sed 's/\\n//g' evidence/shha_promoter.txt | tail -c 20; printf '\\n'
# Verify the first 100 bases against the canonical transcript cDNA sequence.
curl -L --fail --silent --show-error 'https://rest.ensembl.org/sequence/id/ENSDART00000149395?type=cdna' -H 'Content-Type: text/plain' > evidence/shha_cdna.txt
printf 'cdna_length='; wc -c < evidence/shha_cdna.txt
printf 'promoter_downstream_100_matches_cdna='; cmp -n 100 <(tail -c +501 evidence/shha_promoter.txt) <(head -c 100 evidence/shha_cdna.txt) && echo yes || echo no"
