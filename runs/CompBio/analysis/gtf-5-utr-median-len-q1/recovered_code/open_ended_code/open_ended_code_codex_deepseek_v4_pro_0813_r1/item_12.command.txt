/bin/bash -lc "python3 - <<'PY'
import gzip,re
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'; std=re.compile(r'"'^chr(?:[1-9]|1[0-9]|2[0-2]|X|Y)$'"'); ar=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
sel=set(); stops=set(); cds=set()
with gzip.open(path,'rt') as f:
 for line in f:
  if line.startswith('#'): continue
  p=line.rstrip().split('\\t')
  if len(p)"'!=9 or not std.match(p[0]): continue
  a={k:v for k,v in ar.findall(p[8])}; tid=a.get('"'transcript_id','')
  if p[2]=='transcript' and a.get('tag')=='MANE Select': sel.add(tid)
  elif tid in sel and p[2]=='stop_codon': stops.add(tid)
  elif tid in sel and p[2]=='CDS': cds.add(tid)
print(len(sel), len(stops), len(cds), 'selected without stop', len(sel-stops), list(sel-stops)[:5])
PY"
