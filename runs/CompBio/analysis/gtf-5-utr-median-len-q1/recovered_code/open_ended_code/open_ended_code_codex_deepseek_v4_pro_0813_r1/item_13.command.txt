/bin/bash -lc "python3 - <<'PY'
import gzip,re,numpy as np
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'; std=re.compile(r'"'^chr(?:[1-9]|1[0-9]|2[0-2]|X|Y)$'"'); ar=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
def pa(s): return {k:v for k,v in ar.findall(s)}
sel=set(); ex={}; cd={}; st={}; strand={}
with gzip.open(path,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip().split('\\t')
        if len(p)"'!=9 or not std.match(p[0]): continue
        a=pa(p[8]); tid=a.get('"'transcript_id','')
        if not tid: continue
        if p[2]=='transcript':
            if a.get('tag')=='MANE Select':
                sel.add(tid); strand[tid]=p[6]
            continue
        if tid not in sel: continue
        s,e=int(p[3]),int(p[4])
        if p[2]=='exon': ex.setdefault(tid,[]).append((s,e))
        elif p[2]=='CDS': cd.setdefault(tid,[]).append((s,e))
        elif p[2]=='stop_codon': st.setdefault(tid,[]).append((s,e))
def merge(x):
    x=sorted(x); out=[list(x[0])]
    for s,e in x[1:]:
        if s<=out[-1][1]+1: out[-1][1]=max(out[-1][1],e)
        else: out.append([s,e])
    return out
def il(x): return sum(e-s+1 for s,e in x)
def inter(x,lo,hi): return sum(max(0,min(e,hi)-max(s,lo)+1) for s,e in x) if lo<=hi else 0
bad=[]; f=[]; t=[]; c=[]
for tid in sel:
    exm=merge(ex.get(tid,[])); cod=merge(cd.get(tid,[])+st.get(tid,[])); tx0=exm[0][0]; tx1=exm[-1][1]; cs=cod[0][0]; ce=cod[-1][1]
    if strand[tid]=='+':
        a=inter(exm,tx0,cs-1); b=inter(exm,ce+1,tx1)
    else:
        a=inter(exm,ce+1,tx1); b=inter(exm,tx0,cs-1)
    cl=il(cod)
    if il(exm)"'!=a+cl+b: bad.append((tid,il(exm),a,cl,b))
    f.append(a); t.append(b); c.append(cl)
print('"'bad',len(bad),bad[:3])
print('medians',np.median(f),np.median(c),np.median(t))
print('answer',int(np.floor(np.median(f))),int(np.floor(np.median(c))),int(np.floor(np.median(t))))
PY"
