/bin/bash -lc "python3 - <<'PY'
import gzip,re,collections
p='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}
alltx=[]; f=collections.Counter(); seq=collections.Counter(); txseq={}; txtag={}
with gzip.open(p,'rt') as h:
    for line in h:
        if line.startswith('#'): continue
        a=line.rstrip('\\n').split('\\t'); attr=a[8]
        m=re.search(r'transcript_id \"(["'^"]*)"'"',attr); tid=m.group(1) if m else None
        if 'tag \"MANE Select\"' in attr:
            f[(a[0],a[2])]+=1
            seq[a[0]]+=1
            if a[2]=='transcript':
                alltx.append(tid); txseq[tid]=a[0]; txtag[tid]=attr
print('MANE transcript lines',len(alltx),'unique',len(set(alltx)))
print('MANE standard transcripts',sum(txseq[t] in std for t in alltx))
print('MANE feature counts standard')
for k,v in sorted(f.items()):
    if k[0] in std: print(k[1],v)
print('MANE feature counts all')
for k,v in sorted(f.items()): print(k,v)
print('MANE seq alt transcript count',collections.Counter(txseq.values()))
PY"
