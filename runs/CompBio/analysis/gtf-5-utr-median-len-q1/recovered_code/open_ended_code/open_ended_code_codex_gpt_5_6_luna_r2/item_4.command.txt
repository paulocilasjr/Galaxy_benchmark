/bin/bash -lc "python3 - <<'PY'
import gzip, re, collections
p='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}
counts=collections.Counter(); tx=collections.defaultdict(collections.Counter)
for line in gzip.open(p,'rt'):
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    chrom,typ,attrs=f[0],f[2],f[8]
    if chrom not in std: continue
    m=re.search(r'transcript_id \"(["'^"]*)"'"',attrs); tid=m.group(1) if m else ''
    mane='tag \"MANE Select\"' in attrs
    if typ=='transcript': counts[('transcript',mane)]+=1
    if mane: counts[(typ,'mane')]+=1; tx[tid][typ]+=1
print('standard feature counts:',counts)
print('standard MANE tx:',len(tx))
print('feature tx count ranges')
for typ in ['transcript','exon','CDS','start_codon','stop_codon']:
    vals=[c[typ] for c in tx.values()]
    print(typ,len(vals),sum(vals),min(vals) if vals else None,max(vals) if vals else None,collections.Counter(vals).most_common(8))
print('mane transcripts without CDS:',sum('CDS' not in c for c in tx.values()))
print('mane transcripts transcript-line:',sum(c['transcript'] for c in tx.values()))
PY"
