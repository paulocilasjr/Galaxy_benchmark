/bin/bash -lc "python3 - <<'PY'
import numpy as np, gzip, re, collections
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std=set([f'chr{i}' for i in list(range(1,23))+['X','Y']])
tx_exons=collections.defaultdict(list)
tx_cds=collections.defaultdict(list)
tx_strand={}
with gzip.open(fn,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        chrom=p[0]; typ=p[2]; start=int(p[3]); end=int(p[4]); strand=p[6]; attrs=p[8]
        if chrom not in std: continue
        m=re.search(r'transcript_id \"(["'^"]+)"'"', attrs)
        if not m: continue
        tid=m.group(1)
        if 'tag \"MANE Select\"' not in attrs: continue
        tx_strand[tid]=strand
        if typ=='exon':
            tx_exons[tid].append((start,end))
        elif typ in ('CDS','stop_codon','start_codon'):
            tx_cds[tid].append((start,end))
def union(intervals):
    if not intervals: return []
    intervals=sorted(intervals)
    out=[]; a,b=intervals[0]
    for c,d in intervals[1:]:
        if c<=b+1: b=max(b,d)
        else: out.append((a,b)); a,b=c,d
    out.append((a,b)); return out
def cl(intervals): return sum(b-a+1 for a,b in union(intervals))
bad=[]
for tid, exs in tx_exons.items():
    exu=union(exs); cdu=union(tx_cds.get(tid,[]))
    strand=tx_strand[tid]; cds_len=cl(cdu)
    cs=min(a for a,b in cdu); ce=max(b for a,b in cdu)
    five=three=0
    for a,b in exu:
        if strand=='+':
            five += max(0,min(b,cs-1)-a+1)
            three += max(0,b-max(a,ce+1)+1)
        else:
            five += max(0,b-max(a,ce+1)+1)
            three += max(0,min(b,cs-1)-a+1)
    total=cl(exu)
    if total "'!= five+three+cds_len:
        bad.append((tid,total,five,three,cds_len,strand,exu,cdu))
print('"'bad count',len(bad))
for x in bad[:10]: print(x)
PY"
