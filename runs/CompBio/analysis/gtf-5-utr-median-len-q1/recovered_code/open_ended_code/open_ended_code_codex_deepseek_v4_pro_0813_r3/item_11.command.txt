/bin/bash -lc "python3 - <<'PY'
import gzip,re,collections
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std=set([f'chr{i}' for i in list(range(1,23))+['X','Y']])
sel=set(); allfeat=collections.defaultdict(set)
with gzip.open(fn,'rt') as f:
 for line in f:
  if line.startswith('#'): continue
  p=line.rstrip('\\n').split('\\t')
  if len(p)<9: continue
  chrom=p[0]; typ=p[2]; attrs=p[8]
  if chrom not in std: continue
  m=re.search(r'transcript_id \"(["'^"]+)"'"',attrs)
  if not m: continue
  tid=m.group(1)
  allfeat[tid].add((typ,'MANE Select' in attrs))
  if typ=='transcript' and 'tag \"MANE Select\"' in attrs:
   sel.add(tid)
print('sel',len(sel),'all tid',len(allfeat))
# For selected tids, find any features lacking tag
miss=0
for tid in sel:
 for typ,tag in allfeat[tid]:
  if not tag:
   miss+=1
   if miss<10: print('missing tag',tid,typ)
print('missing features in sel',miss)
# features of select count
PY"
