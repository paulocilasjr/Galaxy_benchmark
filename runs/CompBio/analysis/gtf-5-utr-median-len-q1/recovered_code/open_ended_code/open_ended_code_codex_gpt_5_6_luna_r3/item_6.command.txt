import gzip,re
from collections import defaultdict,Counter
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'; std={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}

def attrs(s): return dict(re.findall(r'(\\S+) \"(["'^"]*)"'"',s))
D=defaultdict(lambda: defaultdict(list)); tags={}; seqs={}
with gzip.open(path,'rt') as f:
 for l in f:
  if l.startswith('#'): continue
  p=l.rstrip().split('\\t'); a=attrs(p[8]); tid=a.get('transcript_id','')
  if p[0] in std and tid and a.get('tag')=='MANE Select': D[tid][p[2]].append((int(p[3]),int(p[4])))

over=[]; nostop=[]; notstart=[]; same=[]
for t,d in D.items():
 cds=d['CDS']; stop=d['stop_codon']; start=d['start_codon']
 if not stop: nostop.append(t)
 if not start: notstart.append(t)
 for a,b in cds:
  for c,e in stop:
   if max(a,c)<=min(b,e): over.append((t,(a,b),(c,e)))
print(len(D), 'no stop',len(nostop),'no start',len(notstart),'overlaps',len(over))
print(over[:10])
print('feature types',Counter(k for d in D.values() for k in d))
PY"
