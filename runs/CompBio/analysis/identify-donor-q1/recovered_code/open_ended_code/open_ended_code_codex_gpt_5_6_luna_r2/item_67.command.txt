/bin/bash -lc "python3 - <<'PY'
import pysam, math, sys
known=pysam.VariantFile('chr22.snps.vcf.gz')
query=pysam.VariantFile('reads.called.vcf.gz')
samples=list(known.header.samples); scores=[0.0]*len(samples); used=[0]*len(samples); counts=[0]*len(samples)
qit=iter(query); n=0; sites=0; dpdist={}
for kr in known:
    try: qr=next(qit)
    except StopIteration: print('query ended',kr.pos); break
    if kr.pos"'!=qr.pos or kr.ref!=qr.ref or kr.alts!=qr.alts:
        print('"'mismatch',kr.pos,qr.pos,kr.ref,qr.ref,kr.alts,qr.alts); sys.exit()
    qs=qr.samples[0]; dp=qs.get('DP') or 0; pl=qs.get('PL')
    dpdist[dp]=dpdist.get(dp,0)+1
    if dp<2 or pl is None or len(pl)<3: continue
    # PL array in order 00,01,11 for biallelic
    for i,samp in enumerate(samples):
        gt=kr.samples[samp].get('GT')
        if gt is None or len(gt)"'!=2 or gt[0] is None or gt[1] is None or gt[0]>1 or gt[1]>1: continue
        idx=gt[0]+gt[1]
        scores[i]+=float(pl[idx]); used[i]+=1
        counts[i]+=1
    sites+=1
print('"'samples',len(samples),'matched query sites',n,'scored_sites',sites,'dp>=2',sum(v for k,v in dpdist.items() if k>=2))
order=sorted(range(len(samples)),key=lambda i:scores[i])
for rank,i in enumerate(order[:20],1):
    print(rank,samples[i],scores[i],used[i], 'avg',scores[i]/used[i])
print('margin',scores[order[1]]-scores[order[0]])
PY"
