/bin/bash -lc "gzip -dc identify.donor.R1.fq.gz | awk 'NR%4==2 && ++n<=10 {print \">r\" n \"\\n\" "'$0}'"' > blast_query.fa
curl -sS --max-time 60 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' --data-urlencode 'MEGABLAST=on' --data-urlencode 'QUERY@blast_query.fa' -o blast_submit.txt
cat blast_submit.txt | sed -n '/RID =/p;/RTOE =/p'"
