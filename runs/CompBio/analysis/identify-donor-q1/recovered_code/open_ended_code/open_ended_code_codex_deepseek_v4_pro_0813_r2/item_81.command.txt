/bin/bash -lc "python3 - <<'PY'
import subprocess, gzip
# Get positions and ref/alt from panel and calls via bcftools query (fast)
panel=subprocess.check_output(\"bcftools query -f '%POS\\t%REF\\t%ALT\\n' panel_region.vcf.gz\",shell=True,text=True).splitlines()
calls=subprocess.check_output(\"bcftools query -f '%POS\\t%REF\\t%ALT\\n' calls.vcf.gz\",shell=True,text=True).splitlines()
print('panel',len(panel),'calls',len(calls))
pset=set(panel); cset=set(calls)
print('overlap by pos/ref/alt',len(pset&cset))
pdict={int(x.split('\\t')[0]):x for x in panel}
cdict={int(x.split('\\t')[0]):x for x in calls}
common=set(pdict)&set(cdict)
print('common positions',len(common))
for pos in sorted(common)[:20]: print(pos,pdict[pos],cdict[pos])
PY"
