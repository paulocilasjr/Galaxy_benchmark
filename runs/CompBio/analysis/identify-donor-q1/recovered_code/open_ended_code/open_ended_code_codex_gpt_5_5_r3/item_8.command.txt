/bin/bash -lc "python - <<'PY'
import gzip, urllib.parse, urllib.request, json
seqs=[]
with gzip.open('identify.donor.R1.fq.gz','rt') as f:
    for i in range(20):
        name=f.readline().strip(); seq=f.readline().strip(); f.readline(); f.readline()
        seqs.append((name[1:] if name.startswith('@') else name, seq))
for name,seq in seqs[:5]:
    url='https://genome.ucsc.edu/cgi-bin/hgBlat?'+urllib.parse.urlencode({'userSeq':seq,'type':'DNA','db':'hg38','output':'json'})
    print('QUERY', name, seq[:20], len(seq))
    with urllib.request.urlopen(url, timeout=30) as r:
        data=json.load(r)
    # print keys and first matches
    print(data.keys())
    for p in data.get('blat',[])[:5]:
        print(p)
PY"
