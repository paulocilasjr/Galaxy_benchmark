/bin/bash -lc "bcftools view -m2 -M2 -v snps -i 'QUAL>=30 && FORMAT/DP>=4' donor.calls.vcf.gz -Oz -o donor.hqsnps.vcf.gz && bcftools index -t donor.hqsnps.vcf.gz && bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT[\\t%GT\\t%DP]\\n' donor.hqsnps.vcf.gz > donor.hqsnps.tsv && wc -l donor.hqsnps.tsv && head -n 5 donor.hqsnps.tsv && awk '"'$5=="1/1"{a++} $5=="0/1"{h++} END{print "hets",h,"homalts",a}'"' donor.hqsnps.tsv"
