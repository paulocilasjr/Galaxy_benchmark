/bin/bash -lc "bcftools merge --force-samples -m none overlap.vcf.gz donor.hqsnps.vcf.gz -Ob -o merged.panel_target.bcf && bcftools index merged.panel_target.bcf && bcftools query -l merged.panel_target.bcf > merged.samples.txt && tail -n 3 merged.samples.txt && bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT[\\t%GT]\\n' merged.panel_target.bcf > merged.gt.tsv && wc -l merged.gt.tsv && awk -F '\\t' 'BEGIN{OFS=\"\\t\"} {q="'$NF; gsub(/[|]/,"/",q); split(q,qa,"/"); qv=qa[1]+qa[2]; for(i=5;i<NF;i++){g=$i; if(g=="./." || g==".|.") continue; gsub(/[|]/,"/",g); split(g,ga,"/"); gv=ga[1]+ga[2]; n[i]++; if(gv!=qv)m[i]++}} END{for(i=5;i<NF;i++) print m[i]+0,n[i],(n[i]-m[i])/n[i],i-4}'"' merged.gt.tsv | sort -k1,1n -k3,3nr > mismatch.ranks.tmp; awk 'NR==FNR{name[NR]="'$1;next}{print $1,$2,$3,name[$4]}'"' merged.samples.txt mismatch.ranks.tmp | head -n 30"
