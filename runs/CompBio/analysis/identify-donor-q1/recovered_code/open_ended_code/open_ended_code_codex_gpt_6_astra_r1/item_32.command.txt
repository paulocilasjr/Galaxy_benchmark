/bin/bash -lc "python3 - <<'PY'
import pysam,json,collections
r=json.load(open('region.json'));bins=collections.Counter()
with pysam.AlignmentFile('reads.bam') as f:
 for a in f:
  if not a.is_unmapped and not a.is_supplementary and a.mapping_quality>=30:bins[(a.reference_start+r['start'])//1000000]+=1
print('Reads by megabase:', sorted(bins.items()))
PY"
