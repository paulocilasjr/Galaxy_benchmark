/bin/bash -lc "python - <<'PY'
import gzip,csv,collections, numpy as np
from scipy.io import mmread
features=[line.rstrip('\\n').split('\\t')[0] for line in gzip.open('filtered_feature_bc_matrix/features.tsv.gz','rt')]
barcodes=[x.strip() for x in gzip.open('filtered_feature_bc_matrix/barcodes.tsv.gz','rt')]
M=mmread('filtered_feature_bc_matrix/matrix.mtx.gz').tocsr()
rows=list(csv.DictReader(open('protospacer_calls_per_cell.csv'))); by={r['cell_barcode']:r['feature_call'] for r in rows}; labels=np.array([by[b] for b in barcodes])
ref=list(csv.DictReader(open('feature_reference.csv'))); refd={r['id']:r for r in ref}
idx={x:i for i,x in enumerate(features)}
lib=np.asarray(M.sum(axis=0)).ravel(); X=M.multiply(1e4/lib).tocsr()
order=sorted(set(labels),key=lambda x:int(x.split('_')[1]))
names=['STAT1','ISG15','IFIT1','IFIT2','IFIT3','MX1','MX2','OAS1','OAS2','OAS3','CXCL10','CXCL11','GBP1','GBP2','B2M','GAPDH','CCL8','IL6','CD69','CD83','CD39','SLC2A1']
print('gene\\t'+'\\t'.join(refd[g]['target_gene_name'] for g in order))
for n in names:
  vals=[]
  for g in order:
    ix=np.where(labels==g)[0]; vals.append(np.asarray(X[idx[n],ix].todense()).ravel().mean() if n in idx else np.nan)
  print(n+'\\t'+'\\t'.join(f'{v:.2f}' for v in vals))
PY"
