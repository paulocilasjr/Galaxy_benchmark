/bin/bash -lc "python3 - <<'PY'
import scipy.io, numpy as np, pandas as pd
from scipy.stats import ttest_ind, mannwhitneyu
from scipy.sparse import csr_matrix
mat=scipy.io.mmread('filtered_feature_bc_matrix/matrix.mtx.gz').tocsr().astype(float)
feat=pd.read_csv('filtered_feature_bc_matrix/features.tsv.gz',sep='\\t',header=None,names=['id','name','type'])
bc=pd.read_csv('filtered_feature_bc_matrix/barcodes.tsv.gz',sep='\\t',header=None,names=['barcode'])
calls=pd.read_csv('protospacer_calls_per_cell.csv')
ref=pd.read_csv('feature_reference.csv'); g2g=dict(zip(ref.id,ref.target_gene_name))
bc['pos']=np.arange(len(bc)); calls=calls.merge(bc,left_on='cell_barcode',right_on='barcode')
totals=np.asarray(mat.sum(axis=0)).ravel()
cpm=mat.multiply(1.0/totals).tocsr()*10000
logcpm=cpm.copy(); logcpm.data=np.log1p(logcpm.data)
path_idx=np.arange(20); nonpath_idx=np.arange(20,1000)
nt=calls[calls.feature_call.str.startswith('guide_00')]
nt_path=np.asarray(logcpm[path_idx,:][:,nt.pos.values].mean(axis=0)).ravel()
nt_non=np.asarray(logcpm[nonpath_idx,:][:,nt.pos.values].mean(axis=0)).ravel()
nt_score=nt_path-nt_non
rows=[]
for guide,grp in calls.groupby('feature_call'):
    gene=g2g[guide]
    if gene=='Non-Targeting': continue
    path=np.asarray(logcpm[path_idx,:][:,grp.pos.values].mean(axis=0)).ravel()
    non=np.asarray(logcpm[nonpath_idx,:][:,grp.pos.values].mean(axis=0)).ravel()
    score=path-non
    u,p=mannwhitneyu(score,nt_score,alternative='two-sided')
    rows.append(dict(gene=gene,n=len(grp),path_mean=path.mean(),non_mean=non.mean(),score_mean=score.mean(),diff=score.mean()-nt_score.mean(),p=p))
res=pd.DataFrame(rows).set_index('gene').sort_values('diff')
print('AddModuleScore (path - all nonpath) rank:')
print(res.to_string())
PY"
