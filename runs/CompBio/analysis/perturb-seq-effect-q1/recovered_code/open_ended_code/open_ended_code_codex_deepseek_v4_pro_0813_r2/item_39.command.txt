/bin/bash -lc "python3 - <<'PY'
import scipy.io, numpy as np, pandas as pd
mat=scipy.io.mmread('filtered_feature_bc_matrix/matrix.mtx.gz').tocsr().astype(float)
feat=pd.read_csv('filtered_feature_bc_matrix/features.tsv.gz',sep='\\t',header=None,names=['id','name','type'])
bc=pd.read_csv('filtered_feature_bc_matrix/barcodes.tsv.gz',sep='\\t',header=None,names=['barcode'])
calls=pd.read_csv('protospacer_calls_per_cell.csv')
ref=pd.read_csv('feature_reference.csv'); g2g=dict(zip(ref.id,ref.target_gene_name))
bc['pos']=np.arange(len(bc)); calls=calls.merge(bc,left_on='cell_barcode',right_on='barcode')
nt=calls[calls.feature_call.str.startswith('guide_00')]
for guide in ['guide_011','guide_012']:
    grp=calls[calls.feature_call==guide]
    for idx in range(20):
        g=feat['name'][idx]
        a=np.asarray(mat[idx,grp.pos.values].sum(axis=1)).ravel(); b=np.asarray(mat[idx,nt.pos.values].sum(axis=1)).ravel()
        print(guide,g,'target' if g==g2g[guide] else '', 'grp mean raw',round(a.mean(),1),'NT',round(b.mean(),1),'diff',round(a.mean()-b.mean(),1))
    print()
PY"
