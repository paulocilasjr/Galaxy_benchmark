/bin/bash -lc "python - <<'PY'
import gzip, csv, numpy as np
from scipy.io import mmread
from scipy import stats, sparse

with gzip.open('filtered_feature_bc_matrix/matrix.mtx.gz','rb') as f: X=mmread(f)
X=X.tocsc()
feat=[]
with gzip.open('filtered_feature_bc_matrix/features.tsv.gz','rt') as f:
    for line in f: feat.append(line.rstrip('\\n').split('\\t')[0])
bc=[]
with gzip.open('filtered_feature_bc_matrix/barcodes.tsv.gz','rt') as f:
    for line in f: bc.append(line.rstrip('\\n'))
calls={}
with open('protospacer_calls_per_cell.csv','rt') as f:
    r=csv.reader(f); next(r)
    for row in r: calls[row[0]]=row[2]
ref={}
with open('feature_reference.csv','rt') as f:
    r=csv.reader(f); next(r)
    for row in r: ref[row[0]]=row[7]
guides=np.array([ref[calls[b]] for b in bc])
path_genes=feat[:20]
idx=np.array([feat.index(g) for g in path_genes])
# use log1p of counts? pathway score average of log1p gene counts across genes per cell
L=X[idx,:].toarray().astype(float)
print('L shape',L.shape)
# library normalize? counts similar, log1p cpm maybe
lib=np.asarray(X.sum(axis=0)).ravel().astype(float)
logcpm=np.log1p(L*10000/lib[None,:])
score=logcpm.mean(axis=0)
# per target
controls=guides=='Non-Targeting'
c_score=score[controls]
print('control n',controls.sum(),'mean score',c_score.mean(),'sd',c_score.std())
rows=[]
for t in np.unique(guides):
    m=guides==t
    if t=='Non-Targeting': continue
    s=score[m]
    d=s.mean()-c_score.mean()
    # Welch t
    tw,p=stats.ttest_ind(s,c_score,equal_var=False)
    # rank sum U
    u,p2=stats.mannwhitneyu(s,c_score,alternative='two-sided')
    # also median
    rows.append((t,m.sum(),s.mean(),c_score.mean(),d,s.std(),tw,p,p2))
rows_sorted=sorted(rows,key=lambda r:r[5])
print('target n mean ctrl_mean diff sd t p p_rank')
for r in rows_sorted:
    print(f'{r[0]:15s} {r[1]:3d} {r[2]:.4f} {r[3]:.4f} {r[5]: .4f} {r[6]:.3f} {r[7]:.3g} {r[8]:.3g}')
# top and bottom by t
print('\\nTop negative by t:')
for r in sorted(rows,key=lambda r:r[7])[:15]: print(r[0],r[1],r[5],r[7],r[8])
PY"
