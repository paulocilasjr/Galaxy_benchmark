from scipy.io import mmread

# gene sets
sets={

 'first18_no_STAT1_ISG15':['IFIT1','IFIT2','IFIT3','MX1','MX2','OAS1','OAS2','OAS3','CXCL10','CXCL11','IRF7','BST2','IFI6','STAT2','EIF2AK2','GBP1','GBP2','IDO1'],

}

for setname, genes in sets.items():
