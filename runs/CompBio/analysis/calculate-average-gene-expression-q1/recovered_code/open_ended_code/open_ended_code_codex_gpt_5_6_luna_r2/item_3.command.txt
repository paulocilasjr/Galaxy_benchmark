/bin/bash -lc "python - <<'PY'
import anndata as ad
import numpy as np

path = 'pbmc3k.h5ad'
adata = ad.read_h5ad(path)
print('shape', adata.shape)
print('obs columns', list(adata.obs.columns))
print('louvain dtype', adata.obs['louvain'].dtype)
print('louvain values', adata.obs['louvain'].value_counts().to_dict())
print('S100B in var', 'S100B' in adata.var_names)
print('X type', type(adata.X), 'dtype', getattr(adata.X, 'dtype', None))
idx = np.flatnonzero(adata.obs['louvain'].astype(str).to_numpy() == 'B cells')
gidx = adata.var_names.get_loc('S100B')
vals = adata.X[idx, gidx]
if hasattr(vals, 'toarray'):
    vals = vals.toarray().ravel()
else:
    vals = np.asarray(vals).ravel()
print('n B cells', len(idx))
print('values nonzero', np.count_nonzero(vals), 'sum', float(vals.sum()), 'mean raw', float(vals.mean()), 'round2', format(float(vals.mean()), '.2f'))
print('min max', float(vals.min()), float(vals.max()))
PY"
