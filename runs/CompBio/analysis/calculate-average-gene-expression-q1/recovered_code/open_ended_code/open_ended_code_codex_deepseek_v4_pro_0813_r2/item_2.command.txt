/bin/bash -lc "python - <<'PY'
import anndata as ad
import numpy as np
p='pbmc3k.h5ad'
a=ad.read_h5ad(p)
print(a)
print('obs cols', list(a.obs.columns))
print('var cols', list(a.var.columns))
print('louvain values', a.obs['louvain'].value_counts(dropna=False).to_dict())
print('S100B in var?', 'S100B' in a.var_names)
print('var names sample', list(a.var_names[:10]), '...', list(a.var_names[-10:]))
print('X type', type(a.X), a.X.shape)
if 'S100B' in a.var_names:
    i=a.var_names.get_loc('S100B')
    print('S100B index', i)
    mask=a.obs['louvain']=='B cells'
    print('n B cells', mask.sum(), 'total', a.n_obs)
    vals=a.X[mask.to_numpy(), i]
    print('vals type', type(vals), getattr(vals,'shape',None))
    arr = np.asarray(vals).ravel()
    print('min max mean', arr.min(), arr.max(), arr.mean())
    print('mean rounded', round(float(arr.mean()),2))
PY"
