/bin/bash -lc "python - <<'PY'
import anndata as ad
import numpy as np
adata = ad.read_h5ad('pbmc3k.h5ad')
mask = adata.obs['louvain'].eq('B cells').to_numpy()
gene = np.flatnonzero(adata.var_names == 'S100B')
assert mask.sum() > 0
assert len(gene) == 1
values = adata.X[mask, :][:, gene]
mean = float(values.mean())
print('cells:', mask.sum())
print('gene:', adata.var_names[gene[0]])
print('mean:', repr(mean))
print(format(mean, '.2f'))
PY"
