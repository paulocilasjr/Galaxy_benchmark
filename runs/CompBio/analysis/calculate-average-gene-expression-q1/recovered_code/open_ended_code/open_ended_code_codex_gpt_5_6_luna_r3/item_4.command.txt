/bin/bash -lc "python - <<'PY'
import anndata as ad
import numpy as np
p='pbmc3k.h5ad'
a=ad.read_h5ad(p)
print('shape', a.shape)
print('X type', type(a.X), 'dtype', a.X.dtype)
print('obs cols', list(a.obs.columns))
print('var index name', a.var_names.name, 'S100B count', int((a.var_names=='S100B').sum()))
print('louvain unique', a.obs['louvain'].unique().tolist())
mask=(a.obs['louvain'].astype(str)=='B cells').to_numpy()
print('B cells n', int(mask.sum()))
idx=np.flatnonzero(a.var_names=='S100B')
print('idx', idx.tolist())
x=a.X[mask, idx[0]]
if hasattr(x,'toarray'): x=x.toarray()
x=np.asarray(x).ravel()
print('values n', len(x), 'min', x.min(), 'max', x.max(), 'sum', x.sum(), 'mean', x.mean())
print('mean format', f'{x.mean():.2f}')
PY"
