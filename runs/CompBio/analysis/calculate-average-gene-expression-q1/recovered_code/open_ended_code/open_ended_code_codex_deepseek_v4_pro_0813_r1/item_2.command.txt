/bin/bash -lc "python - <<'PY'
import anndata as ad
import numpy as np
print('reading', flush=True)
adata = ad.read_h5ad('pbmc3k.h5ad')
print('shape', adata.shape, flush=True)
print('obs columns', list(adata.obs.columns), flush=True)
print('var columns', list(adata.var.columns), flush=True)
print('louvain categories', adata.obs['louvain'].cat.categories.tolist() if hasattr(adata.obs['louvain'], 'cat') else adata.obs['louvain'].unique().tolist(), flush=True)
print('S100B in var?', 'S100B' in adata.var_names, flush=True)
mask = adata.obs['louvain'] == 'B cells'
print('n B cells', mask.sum(), flush=True)
if 'S100B' in adata.var_names:
    idx = adata.var_names.get_loc('S100B')
    vals = adata.X[mask, idx]
    if hasattr(vals, 'toarray'):
        vals = vals.toarray()
    vals = np.asarray(vals).ravel()
    print('mean', vals.mean(), flush=True)
    print('count nonzero', np.count_nonzero(vals), flush=True)
PY"
