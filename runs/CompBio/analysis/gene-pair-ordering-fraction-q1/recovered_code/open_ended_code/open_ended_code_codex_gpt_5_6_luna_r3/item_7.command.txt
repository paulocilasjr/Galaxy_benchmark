/bin/bash -lc "python - <<'PY'
import h5py
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
 def visit(name,obj):
  if isinstance(obj,h5py.Dataset):
   print(name, 'dataset', obj.shape, obj.dtype, 'attrs=', {k:obj.attrs[k] for k in obj.attrs.keys()})
  else:
   print(name, 'group', 'attrs=', {k:obj.attrs[k] for k in obj.attrs.keys()})
 f.visititems(visit)
PY"
