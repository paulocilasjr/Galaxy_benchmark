/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
    cats=[x.decode() for x in f['uns/bulk_labels_categories'][:]]
    print('categories')
    for i,c in enumerate(cats): print(i, c)
    var=[x.decode() for x in f['var']['index'][:]]
    for g in ['SRM','MRPS21']:
        print(g, var.index(g) if g in var else None)
    obs=f['obs'][:]
    target=cats.index('CD19+ B')
    mask=(obs['bulk_labels']==target) & (obs['percent_mito'] < 0.04)
    print('n filter', int(mask.sum()))
    sidx=var.index('SRM'); midx=var.index('MRPS21')
    X=f['X'][:]
    diff=X[mask,sidx]-X[mask,midx]
    print('counts X', int(np.sum(diff>1.0)), int(np.sum(diff<-1.0)), int(np.sum((diff>=-1.0)&(diff<=1.0))))
    print('diff values', diff)
    # raw details
    rx=f['raw.X']
    print('raw.X type', type(rx), list(rx.keys()) if isinstance(rx,h5py.Group) else None, getattr(rx,'shape',None), dict(rx.attrs))
    rv=f['raw.var']
    print('raw.var type', type(rv), getattr(rv,'shape',None), getattr(rv,'dtype',None), list(rv.keys()) if isinstance(rv,h5py.Group) else None, dict(rv.attrs))
PY"
