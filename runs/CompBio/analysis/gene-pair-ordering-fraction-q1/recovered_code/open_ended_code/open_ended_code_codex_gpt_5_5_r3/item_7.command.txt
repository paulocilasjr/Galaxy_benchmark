/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
with h5py.File('10x_pbmc68k_reduced.h5ad','r') as f:
    obs=f['obs'][:]
    cats=[x.decode() for x in f['uns/bulk_labels_categories'][:]]
    print('bulk categories:')
    for i,c in enumerate(cats): print(i, repr(c))
    var=[x.decode() for x in f['var'][:]['index']]
    print('SRM idx', var.index('SRM'), 'MRPS21 idx', var.index('MRPS21'))
    mask=(obs['bulk_labels']==cats.index('CD19+ B')) & (obs['percent_mito'] < 0.04)
    print('n mask', int(mask.sum()))
    X=f['X'][:]
    s=X[mask, var.index('SRM')]
    m=X[mask, var.index('MRPS21')]
    d=s-m
    print('X counts', int((d>1.0).sum()), int((d<-1.0).sum()), int(((d>=-1.0)&(d<=1.0)).sum()))
    print('range X s', s.min(), s.max(), 'm', m.min(), m.max(), 'd unique sample', d[:20])
    # raw sparse columns extract by reconstructing rows only
    rv=[x.decode() for x in f['raw.var'][:]['index']]
    print('raw idx', rv.index('SRM'), rv.index('MRPS21'))
    data=f['raw.X/data'][:]; indices=f['raw.X/indices'][:]; indptr=f['raw.X/indptr'][:]
    sraw=[]; mraw=[]
    js=rv.index('SRM'); jm=rv.index('MRPS21')
    for row in np.where(mask)[0]:
        st,en=indptr[row], indptr[row+1]
        idx=indices[st:en]; vals=data[st:en]
        # row sorted likely
        ds=dict(zip(idx, vals))
        sraw.append(ds.get(js, 0.0)); mraw.append(ds.get(jm, 0.0))
    sraw=np.array(sraw); mraw=np.array(mraw); draw=sraw-mraw
    print('raw counts', int((draw>1.0).sum()), int((draw<-1.0).sum()), int(((draw>=-1.0)&(draw<=1.0)).sum()))
    print('range raw s', sraw.min(), sraw.max(), 'm', mraw.min(), mraw.max(), 'draw first', draw[:20])
PY"
