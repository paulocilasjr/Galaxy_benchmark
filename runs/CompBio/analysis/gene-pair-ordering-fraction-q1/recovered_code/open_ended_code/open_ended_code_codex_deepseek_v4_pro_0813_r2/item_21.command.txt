/bin/bash -lc "python - <<'PY'
import urllib.request
urls=[
 'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/compbiobench.v1.tsv',
 'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/raw/main/compbiobench.v1.tsv',
 'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/raw/main/data/compbiobench.v1.tsv',
]
for u in urls:
    print('URL',u)
    try:
        with urllib.request.urlopen(u, timeout=20) as r:
            data=r.read(1000)
            print(r.status, data[:200])
    except Exception as e:
        print('ERR',e)
PY"
