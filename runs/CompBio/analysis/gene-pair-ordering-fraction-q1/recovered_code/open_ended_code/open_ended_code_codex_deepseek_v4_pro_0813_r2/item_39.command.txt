/bin/bash -lc "python - <<'PY'
import urllib.request, csv, io
u='https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/compbiobench.v1.tsv'
data=urllib.request.urlopen(u, timeout=30).read().decode('utf-8')
reader=csv.DictReader(io.StringIO(data), delimiter='\\t')
for row in reader:
    q=row.get('question','')
    if 'expression' in q.lower() or 'adata' in q.lower() or 'h5ad' in q.lower():
        print(row['question_id'], '::', q[:400].replace('\\n',' '))
        print('---')
PY"
