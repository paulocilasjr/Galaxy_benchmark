/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('10x_pbmc68k_reduced.h5ad','r') as f:
    X=f['X'][:]
    mask=(f['obs']['bulk_labels'][:]==6)&(f['obs']['percent_mito'][:]<0.04)
    names=[x.decode() for x in f['var']['index'][:]]
    si=names.index('SRM'); mi=names.index('MRPS21')
    raw=csr_matrix((f['raw.X']['data'][:], f['raw.X']['indices'][:], f['raw.X']['indptr'][:]), shape=(700,765))
    sx=X[mask,si]; mx=X[mask,mi]
    rs=np.asarray(raw[mask,si].toarray()).ravel(); rm=np.asarray(raw[mask,mi].toarray()).ravel()
    for label,s,m in [
        ('X',sx,mx),
        ('raw(log1p)',rs,rm),
        ('expm1(raw)',np.expm1(rs),np.expm1(rm)),
        ('raw/10',rs/10,rm/10),
    ]:
        d=s-m
        h=(d>1.0).sum(); l=(d<-1.0).sum(); r=((d>=-1)&(d<=1)).sum()
        print(label, h,l,r, 'total',len(d))
PY"
