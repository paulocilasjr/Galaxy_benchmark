/bin/bash -lc "python - <<'PY'
import urllib.request, json
u='https://genentech-compbiobench-leaderboard-v1.hf.space/gradio_api/call/refresh'
req=urllib.request.Request(u, data=b'{\"data\":[]}', headers={'Content-Type':'application/json'})
resp=urllib.request.urlopen(req, timeout=60).read().decode()
event=json.loads(resp)['event_id']
u2=f'https://genentech-compbiobench-leaderboard-v1.hf.space/gradio_api/call/refresh/{event}'
# poll until complete with stream; use urllib and read maybe server sent events
text=urllib.request.urlopen(u2, timeout=120).read().decode()
# print event prefix and parse data lines
print(text[:100])
lines=text.split('\\n')
payloads=[]
for line in lines:
    if line.startswith('data: '):
        payloads.append(line[6:])
print('num payloads', len(payloads))
for i,p in enumerate(payloads):
    try:
        obj=json.loads(p)
        print('payload',i,'type',type(obj), 'len', len(obj))
        if isinstance(obj,list):
            for j,df in enumerate(obj):
                print(' df',j,'headers',df['headers'], 'nrows',len(df['data']))
                if j==1:
                    for row in df['data'][:5]: print(row)
    except Exception as e: print('parse err',e, p[:200])
PY"
