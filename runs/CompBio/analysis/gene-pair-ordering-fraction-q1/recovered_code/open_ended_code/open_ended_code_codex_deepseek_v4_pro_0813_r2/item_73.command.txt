/bin/bash -lc "python - <<'PY'
import urllib.request, re
u='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v1/supplementary-material'
req=urllib.request.Request(u, headers={'User-Agent':'Mozilla/5.0'})
data=urllib.request.urlopen(req, timeout=30).read().decode()
for m in re.finditer(r'href=\"(["'^"]*(?:suppl|file|media)[''^"]*)"'"', data, re.I):
    print(m.group(1))
print('links containing tsv')
for m in re.finditer(r'href=\"(["'^"]*tsv[''^"]*)"'"', data, re.I):
    print(m.group(1))
PY"
