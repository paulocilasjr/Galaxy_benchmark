/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('10x_pbmc68k_reduced.h5ad','r') as f:
    mask=(f['obs']['bulk_labels'][:]==6)&(f['obs']['percent_mito'][:]<=0.04)
    names=[x.decode() for x in f['var']['index'][:]]; si=names.index('SRM'); mi=names.index('MRPS21')
    X=f['X'][:]; sx=X[mask,si]; mx=X[mask,mi]
    raw=csr_matrix((f['raw.X']['data'][:], f['raw.X']['indices'][:], f['raw.X']['indptr'][:]), shape=(700,765))
    rs=np.asarray(raw[mask,si].toarray()).ravel(); rm=np.asarray(raw[mask,mi].toarray()).ravel()
    print('total',mask.sum())
    for label,s,m in [('X',sx,mx),('raw',rs,rm)]:
        d=s-m
        print(label,(d>1).sum(),(d<-1).sum(),((d>=-1)&(d<=1)).sum())
PY"
