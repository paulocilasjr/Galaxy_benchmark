/bin/bash -lc "python - <<'PY'
import urllib.request, urllib.parse, json
for q in ['MRPS21 SRM', 'n_srm_higher', 'gene-pair-ordering-fraction']:
    u='https://grep.app/api/search?q='+urllib.parse.quote(q)
    try:
        data=urllib.request.urlopen(u, timeout=30).read().decode()
        js=json.loads(data)
        print(q, js.get('hits',{}).get('total'))
        for h in js.get('hits',{}).get('hits',[])[:5]:
            print(h.get('repo'), h.get('path'))
    except Exception as e:
        print(q,'ERR',e)
PY"
