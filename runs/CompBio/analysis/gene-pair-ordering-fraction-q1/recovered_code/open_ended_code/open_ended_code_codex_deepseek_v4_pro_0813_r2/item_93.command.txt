/bin/bash -lc "python - <<'PY'
import urllib.request, urllib.parse, json, re
for val in ['\"37;20;37\"','\"29;23;42\"','\"0.39;0.21;0.39\"','\"0.31;0.24;0.45\"']:
    u='https://github.com/search?q='+urllib.parse.quote(val)+'&type=code'
    req=urllib.request.Request(u, headers={'User-Agent':'Mozilla/5.0'})
    try:
        data=urllib.request.urlopen(req, timeout=30).read().decode()
        # extract embedded result_count
        m=re.search(r'\"result_count\":(\\d+)', data)
        print(val, 'result_count', m.group(1) if m else 'no')
    except Exception as e: print(val,e)
PY"
