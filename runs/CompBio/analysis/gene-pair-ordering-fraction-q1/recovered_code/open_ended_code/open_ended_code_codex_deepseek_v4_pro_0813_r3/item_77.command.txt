/bin/bash -lc "python3 - <<'PY'
import urllib.request
urls=[
 'https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.full',
 'https://www.biorxiv.org/content/biorxiv/early/2026/05/02/2026.04.06.716850.full.pdf',
]
for u in urls:
 print('URL',u)
 req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
 try:
  with urllib.request.urlopen(req,timeout=30) as r:
   print(r.status,r.getheader('content-type'),r.read(500).decode('utf-8','replace'))
 except Exception as e: print('ERR',repr(e))
PY"
