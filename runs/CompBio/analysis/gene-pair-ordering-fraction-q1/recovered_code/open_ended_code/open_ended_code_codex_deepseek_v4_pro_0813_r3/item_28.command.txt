/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
f=h5py.File('10x_pbmc68k_reduced.h5ad','r')
obs=f['obs'][...]; labels=obs['bulk_labels']; p=obs['percent_mito']
# bulk label code 6 is CD19+ B
sel=(labels==6)&(p<0.04)
idx=np.where(sel)[0]
X=f['X'][...]
srm=X[idx,5].copy(); mrp=X[idx,48].copy(); d=srm-mrp
print('count',len(idx))
print('>1', np.sum(d>1.0), '<-1', np.sum(d < -1.0), 'within', np.sum((d<=1.0)&(d>=-1.0)))
print('sum',np.sum(d>1.0)+np.sum(d<-1.0)+np.sum((d<=1.0)&(d>=-1.0)))
print('min max',d.min(),d.max())
print('exact 1/-1 counts', np.sum(d==1.0), np.sum(d==-1.0))
# inspect sorted differences around boundaries
for val in np.sort(d):
    if val>-1.05 and val<1.05:
        pass
print('diffs near +/-1')
print([(float(v), int(i)) for i,v in enumerate(np.sort(d)) if abs(v-1)<0.02 or abs(v+1)<0.02][:20])
# raw too
raw=f['raw.X']; data=raw['data'][...]; inds=raw['indices'][...]; ptr=raw['indptr'][...]; shape=tuple(raw.attrs['h5sparse_shape'])
r=np.zeros(shape,dtype=np.float32)
for i in range(shape[0]):
    s,e=ptr[i],ptr[i+1]
    if s<e: r[i,inds[s:e]]=data[s:e]
rd=r[idx,5]-r[idx,48]
print('raw counts',np.sum(rd>1.0),np.sum(rd<-1.0),np.sum((rd<=1.0)&(rd>=-1.0)), 'sum', len(idx))
print('raw exact boundary',np.sum(rd==1.0),np.sum(rd==-1.0))
f.close()
PY"
