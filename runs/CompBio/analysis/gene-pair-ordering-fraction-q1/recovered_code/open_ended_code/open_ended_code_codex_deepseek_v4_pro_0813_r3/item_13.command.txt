/bin/bash -lc "python3 - <<'PY'
import urllib.request
urls=[
 'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/README.md',
 'https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1',
]
for u in urls:
    print('URL',u)
    try:
        with urllib.request.urlopen(u, timeout=20) as r:
            data=r.read(2000)
            print(r.status, data[:1000])
    except Exception as e:
        print('ERR',repr(e))
PY"
