/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for repo in ['Genentech/compbiobench-results-v1','Genentech/compbiobench-submissions-v1']:
 u=f'https://huggingface.co/api/datasets/{repo}'
 try:
  with urllib.request.urlopen(u, timeout=30) as r: d=json.load(r)
  print(repo, 'private', d.get('private'), 'gated', d.get('gated'))
  print('siblings', d.get('siblings'))
 except Exception as e: print(repo,e)
PY"
