/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
f=h5py.File('10x_pbmc68k_reduced.h5ad','r')
obs=f['obs'][...]; var=f['var'][...]
labels=obs['bulk_labels']; p=obs['percent_mito']; sel=(labels==6)&(p<0.04)
idx=np.where(sel)[0]
X=f['X'][...]
gi=[5,48]
# raw CSR reconstruct
raw=f['raw.X']; shape=tuple(raw.attrs['h5sparse_shape']); data=raw['data'][...]; indices=raw['indices'][...]; indptr=raw['indptr'][...]
rawmat=np.zeros(shape,dtype=data.dtype)
for i in range(shape[0]):
    s,e=indptr[i],indptr[i+1]
    if s<e: rawmat[i,indices[s:e]]=data[s:e]
print('sel count',len(idx),'first ids',idx[:10])
print('X selected')
print(X[np.ix_(idx,gi)][:10])
print('raw selected')
print(rawmat[np.ix_(idx,gi)][:10])
print('stats all X SRM MRPS21')
for j in gi:
    v=X[:,j]; print(j,var['index'][j], v.min(), v.max(), v.mean(), v.std())
print('stats all raw')
for j in gi:
    v=rawmat[:,j]; print(j,var['index'][j], v.min(), v.max(), v.mean(), np.count_nonzero(v))
# differences counts on X and raw
for matname, mat in [('X',X),('raw',rawmat)]:
    srm=mat[idx,5]; mrp=mat[idx,48]; d=srm-mrp
    n_srm=np.sum(d>1.0); n_mrp=np.sum(d<-1.0); n_within=np.sum((d<=1.0)&(d>=-1.0))
    print(matname, 'n_srm', n_srm, 'n_mrp', n_mrp, 'n_within', n_within, 'sum', n_srm+n_mrp+n_within)
    print('d range', d.min(), d.max())
f.close()
PY"
