/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
queries=['\"n_srm_higher\"','\"gene-pair-ordering-fraction\"','\"MRPS21\" \"SRM\" \"CD19\"','\"10x_pbmc68k_reduced.h5ad\" \"MRPS21\"']
for q in queries:
    url='https://api.github.com/search/code?q='+urllib.parse.quote(q)
    print('QUERY',q)
    req=urllib.request.Request(url, headers={'Accept':'application/vnd.github+json','User-Agent':'codex'})
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            d=json.load(r)
        print('total',d.get('total_count'))
        for item in d.get('items',[])[:5]: print(item.get('html_url'))
    except Exception as e:
        print('ERR',e)
PY"
