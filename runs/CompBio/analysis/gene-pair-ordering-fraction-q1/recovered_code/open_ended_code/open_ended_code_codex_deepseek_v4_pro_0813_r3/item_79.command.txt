/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
import re
txt=Path('paper.html').read_text()
for pat in ['Supplementary','supplement','media','zip','xlsx','csv','table']:
    idxs=[m.start() for m in re.finditer(pat,txt,re.I)][:10]
    print(pat, idxs)
# print all hrefs containing supp
for m in re.finditer(r'href=\"(["'^"]+)"'"',txt):
    href=m.group(1)
    if any(s in href.lower() for s in ['supp','media','zip','xlsx','csv']):
        print(href)
PY"
