/bin/bash -lc "python3 - <<'PY'
import urllib.request,re
u='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.supplementary-material'
req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
with urllib.request.urlopen(req,timeout=60) as r:
    txt=r.read().decode('utf-8','replace')
print('len',len(txt))
open('supp.html','w').write(txt)
# print links to files
for m in re.finditer(r'href=\"(["'^"]+)"'"',txt):
    href=m.group(1)
    if any(s in href.lower() for s in ['.zip','.xlsx','.csv','supp','media','.txt']):
        print(href)
PY"
