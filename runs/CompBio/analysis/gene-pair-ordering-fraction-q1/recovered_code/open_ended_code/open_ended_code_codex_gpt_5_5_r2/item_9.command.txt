import anndata as ad
import numpy as np
from scipy import sparse
adata = ad.read_h5ad('10x_pbmc68k_reduced.h5ad')

if sparse.issparse(sub):
    sub = sub.toarray()
else:
    sub = np.asarray(sub)
diff = sub[:,0] - sub[:,1]
print('diffs', diff)
print('n_srm_higher', int(np.sum(diff > 1.0)))
print('n_mrps21_higher', int(np.sum(diff < -1.0)))
print('n_within_range', int(np.sum((diff >= -1.0) & (diff <= 1.0))))
print('answer', f\"{int(np.sum(diff > 1.0))};{int(np.sum(diff < -1.0))};{int(np.sum((diff >= -1.0) & (diff <= 1.0)))}\")
print('raw exists', adata.raw is not None)
print('layers', list(adata.layers.keys()))
PY"
