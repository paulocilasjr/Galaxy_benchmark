/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
import gzip

# load sc
sc = pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
print('sc', sc.shape)
meta = pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
print(meta.head())
# spatial
mat = mmread('matrix.mtx.gz').tocsc()
print('mat', mat.shape, mat.nnz)
bar = pd.read_csv('barcodes.tsv.gz', compression='gzip', sep='\\t', header=None, names=['barcode'])
feat = pd.read_csv('features.tsv.gz', compression='gzip', sep='\\t', header=None, names=['ens','gene','type'])
# map genes to spatial rows? features likely same order Gene_0..999
print(feat.head(), feat.tail())
print('bar spot710 idx', bar.index[bar.barcode=='Spot_710-1'].tolist())
spot_idx = bar.index[bar.barcode=='Spot_710-1'][0]
spot_vec = mat[:,spot_idx].toarray().ravel()
print('spot sum', spot_vec.sum(), 'nonzero', (spot_vec>0).sum())
# top genes in spot
order=np.argsort(spot_vec)[::-1][:20]
print(list(zip(feat.gene.iloc[order], spot_vec[order])))
# sc means per cell type
ct = meta['cell_type']
cts = ct.unique()
print(cts)
# align genes same order
means = sc.groupby(ct).mean()
print(means.shape)
# correlations
for c in cts:
    corr=np.corrcoef(means.loc[c], spot_vec)[0,1]
    print(c, corr)
PY"
