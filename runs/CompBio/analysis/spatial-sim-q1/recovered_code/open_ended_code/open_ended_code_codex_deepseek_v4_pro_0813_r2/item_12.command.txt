/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
sc = pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
mat = mmread('matrix.mtx.gz').tocsc()
rows = sc.values.astype(np.int32)
spot = mat.toarray().T.astype(np.int32) # 900x1000, ~1.8M ints okay
print(spot.shape)
# check spots that exactly match any single cell by row bytes
row_bytes={r.tobytes(): i for i,r in enumerate(rows)}
match1=0; examples=[]
for j,s in enumerate(spot):
    if s.tobytes() in row_bytes:
        match1+=1; examples.append((j,row_bytes[s.tobytes()]))
print('exact single matches', match1, examples[:20])
# check spot total sums equal sc total
sc_tot=rows.sum(1)
spot_tot=spot.sum(1)
print('spot totals that equal sc total count', sum(np.isin(spot_tot, sc_tot)))
# check spot equals 2*row? etc
PY"
