/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
mat=mmread('matrix.mtx.gz').tocsc()
t=mat[:,710].toarray().ravel()
means=sc.groupby(meta.cell_type).mean()
cts=means.index
# z-score across types for each gene
z=(means-means.mean())/means.std()
# top marker genes by z per type
for c in cts:
    top=z.loc[c].sort_values(ascending=False).head(20)
    print('\\n',c)
    for g,v in top.items():
        print(g, 'mean', means.loc[c,g].round(1), 'z', round(v,2), 'spot', t[sc.columns.get_loc(g)])
# Also for spot correlation with marker sums etc
PY"
