import pandas as pd, numpy as np
from scipy.io import mmread
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)

types=meta.cell_type.values
labels=meta.index.tolist()
cts=['B_Cell','Endothelial','Macrophage']
groups={c:np.where(types==c)[0].tolist() for c in cts}
# pair dict among candidate 600 cells including both orders? need all unordered pairs within union and between groups, plus self
pair_dict={}; pair_info={}
cand=[]
for c in cts: cand += groups[c]
cand_arr=np.array(cand)
for ai_i, ai in enumerate(cand_arr):
    va=A[ai]
    for bj_i in range(ai_i, len(cand_arr)):
        bj=cand_arr[bj_i]
        s=va+A[bj]
        key=s.tobytes()
        if key not in pair_dict:
            pair_dict[key]=(ai,bj)
            pair_info[key]=(types[ai],types[bj])
print('pairs',len(pair_dict))
# third among all 1200
for k in range(A.shape[0]):
    rem=t-A[k]
    key=rem.tobytes()
    if key in pair_dict:
        ai,bj=pair_dict[key]
        print('EXACT', labels[k], types[k], 'pair', labels[ai],types[ai],labels[bj],types[bj])
        break
else:
    print('no exact among candidate pairs and all third')
