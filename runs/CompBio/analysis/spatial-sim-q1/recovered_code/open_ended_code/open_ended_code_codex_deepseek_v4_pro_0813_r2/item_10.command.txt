/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from collections import Counter
sc = pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta = pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
tot=sc.sum(axis=1).values
cnt=Counter(tot)
target=3693
# count triples by total via grouping values
vals=sorted(cnt)
triples=[]
# iterate values <= target
for i,a in enumerate(vals):
    if a>target: break
    for j in range(i,len(vals)):
        b=vals[j]
        if a+b>target: break
        c=target-a-b
        if c in cnt:
            # count combinations
            if a==b==c:
                n=cnt[a]*(cnt[a]-1)*(cnt[a]-2)//6
            elif a==b:
                n=cnt[a]*(cnt[a]-1)//2*cnt[c]
            elif b==c:
                n=cnt[a]*cnt[b]*(cnt[b]-1)//2
            else:
                n=cnt[a]*cnt[b]*cnt[c]
            if n:
                triples.append((a,b,c,n))
print('number triple combos by total', sum(n for *_,n in triples), 'distinct totals', triples[:20], len(triples))
# target spot sum from spatial
from scipy.io import mmread
mat=mmread('matrix.mtx.gz').tocsc()
print('spot total', mat[:,710].sum())
PY"
