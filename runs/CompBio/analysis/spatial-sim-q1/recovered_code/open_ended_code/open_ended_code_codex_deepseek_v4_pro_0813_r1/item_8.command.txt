/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.io
meta=pd.read_csv('data/spatial_q_sc_metadata.csv',index_col=0).squeeze('columns')
sc=pd.read_csv('data/spatial_q_sc_counts.csv',index_col=0)
print(sc.shape, meta.shape)
# align features? sc columns gene names; matrix features order same maybe.
feat=pd.read_csv('data/features.tsv.gz',header=None,sep='\\t',compression='gzip')
print('sc genes equal features order?', (sc.columns == feat[1]).all())
mat=scipy.io.mmread('data/matrix.mtx.gz').tocsc()
spot=pd.Series(np.asarray(mat[:,710].todense()).ravel(), index=feat[1])
# sc mean per type
means=sc.groupby(meta).mean()
print(means.head())
for ct in means.index:
    # pearson correlation of spot with mean profile
    corr=np.corrcoef(spot, means.loc[ct])[0,1]
    print(ct, round(corr,4), 'spot sum',spot.sum())
# maybe spot correlation with individual cells
corrs=sc.corrwith(spot, axis=1).sort_values(ascending=False)
print('top cells')
print(pd.DataFrame({'cell':corrs.index,'corr':corrs.values,'type':meta[corrs.index].values}).head(20))
PY"
