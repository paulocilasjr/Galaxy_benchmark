/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.io
meta=pd.read_csv('data/spatial_q_sc_metadata.csv',index_col=0).squeeze('columns')
sc=pd.read_csv('data/spatial_q_sc_counts.csv',index_col=0)
feat=pd.read_csv('data/features.tsv.gz',header=None,sep='\\t',compression='gzip')
means=sc.groupby(meta).mean()
# top genes per type by logFC vs rest
for ct in means.index:
    rest=sc[meta"'!=ct].mean()
    fc=(means.loc[ct]+0.1)/(rest+0.1)
    top=fc.nlargest(15).index.tolist()
    print('"'\\n',ct,'top markers')
    print([(g,round(fc[g],1),round(means.loc[ct,g],2)) for g in top])
# spot top 50
mat=scipy.io.mmread('data/matrix.mtx.gz').tocsc()
x=pd.Series(np.asarray(mat[:,710].todense()).ravel(), index=feat[1])
print('\\nspot top 60')
for g in x.nlargest(60).index:
    print(g, int(x[g]))
PY"
