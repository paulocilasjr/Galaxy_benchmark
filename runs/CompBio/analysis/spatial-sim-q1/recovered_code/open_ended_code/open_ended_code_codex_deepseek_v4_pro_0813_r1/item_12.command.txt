/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.io
sc=pd.read_csv('data/spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('data/spatial_q_sc_metadata.csv',index_col=0).squeeze('columns')
mat=scipy.io.mmread('data/matrix.mtx.gz').tocsc()
for si in [0,1,4,7,9]:
 x=np.asarray(mat[:,si].todense()).ravel()
 l1=np.abs(sc.values-x).sum(axis=1)
 order=np.argsort(l1)
 print('spot',si,'sum',x.sum(),'nnz',np.sum(x>0))
 for j in order[:3]: print(' ',sc.index[j],meta.iloc[j],'L1',l1[j],'scsum',sc.values[j].sum())
 # count identical?
 print('exact', (sc.values==x).all(axis=1).sum())
PY"
