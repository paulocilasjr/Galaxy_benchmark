import pandas as pd, numpy as np
from scipy.io import mmread
import gzip, time
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0).values
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
with gzip.open('matrix.mtx.gz','rt') as f:
 X=mmread(f).toarray().T
y=X[710]
groups={t:np.where(meta.cell_type==t)[0] for t in sorted(meta.cell_type.unique())}
B,E,M=groups['B_Cell'],groups['Endothelial'],groups['Macrophage']
# hash M rows by tuple -> idx
md={tuple(sc[m]):m for m in M}
found=[]
# maybe need allow B,E order? just B and E groups
for i in B:
    for j in E:
        need=y-sc[i]-sc[j]
        if (need>=0).all():
            k=md.get(tuple(need))
            if k is not None:
                found.append((i,j,k))
                print('found',i,j,k)
                break
    if found: break
print('found count',found)
