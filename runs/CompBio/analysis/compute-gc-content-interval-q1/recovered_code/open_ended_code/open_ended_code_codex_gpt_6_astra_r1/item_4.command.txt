/bin/bash -lc "curl -L --fail --silent --show-error 'https://ftp.ncbi.nlm.nih.gov/genomes/all/GCF/000/001/405/GCF_000001405.39_GRCh38.p13/GCF_000001405.39_GRCh38.p13_assembly_report.txt' -o GRCh38.p13_assembly_report.txt && python - <<'PY'
from pathlib import Path
report=Path('GRCh38.p13_assembly_report.txt').read_text()
assert '# Assembly name:  GRCh38.p13' in report
rows=[line for line in report.splitlines() if line.startswith('1\\t')]
assert len(rows)==1 and 'NC_000001.11' in rows[0]
print(rows[0])
answer=Path('answer.txt').read_text()
assert len(answer.splitlines())==1
print(answer,end='')
PY"
