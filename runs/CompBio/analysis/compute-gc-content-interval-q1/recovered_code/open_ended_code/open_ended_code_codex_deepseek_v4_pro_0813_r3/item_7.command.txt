/bin/bash -c "python3 - <<'PY'
import urllib.request, gzip, hashlib, time
url='https://ftp.ncbi.nlm.nih.gov/genomes/all/GCF/000/001/405/GCF_000001405.39_GRCh38.p13/GCF_000001405.39_GRCh38.p13_assembly_structure/Primary_Assembly/assembled_chromosomes/FASTA/chr1.fna.gz'
start=1_000_000
end=1_000_100
response=urllib.request.urlopen(url, timeout=120)
g=gzip.GzipFile(fileobj=response)
collected=[]; pos=0; first=True
for raw in g:
    if first:
        first=False; continue
    line=raw.decode().strip()
    if not line: continue
    ls=len(line)
    # current line covers 1-based positions pos+1 .. pos+ls
    # indices to extract within line: max(0, start-pos-1) .. min(ls, end-pos)
    lo=max(0, start-pos-1)
    hi=min(ls, end-pos)
    if hi>lo:
        collected.append(line[lo:hi])
    pos += ls
    if pos >= end:
        break
seq=''.join(collected)
print('len seq', len(seq), 'pos', pos)
print(seq)
up=seq.upper()
gc=sum(1 for c in up if c in 'GC')
n=up.count('N')
print('metrics', len(up), gc, round(gc/len(up)*100,2), n, hashlib.md5(up.encode()).hexdigest())
PY"
