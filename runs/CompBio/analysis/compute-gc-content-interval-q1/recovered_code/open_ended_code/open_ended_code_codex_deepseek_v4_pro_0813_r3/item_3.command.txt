/bin/bash -c "python3 - <<'PY'
import urllib.request
url='https://rest.ensembl.org/sequence/region/human/1:1000000..1000100:1?content-type=text/plain'
req=urllib.request.Request(url, headers={'Content-Type':'text/plain'})
try:
    data=urllib.request.urlopen(req, timeout=30).read().decode()
    print(data[:1000])
except Exception as e:
    print('ERR', repr(e))
PY"
