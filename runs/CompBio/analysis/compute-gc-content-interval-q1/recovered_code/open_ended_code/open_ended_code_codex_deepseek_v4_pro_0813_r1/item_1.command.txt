/bin/bash -c "python3 - <<'PY'
import urllib.request,json,hashlib
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr1;start=999999;end=1000100'
data=json.load(urllib.request.urlopen(url))
seq=data['dna']
print(len(seq))
print(seq)
up=seq.upper()
print('up len',len(up))
print(up)
gc=100*(up.count('G')+up.count('C'))/len(up)
n=up.count('N')
print(round(gc,2),len(up),n,hashlib.md5(up.encode()).hexdigest())
PY"
