/bin/bash -c "python3 - <<'PY'
import urllib.request
urls=[
'https://rest.ensembl.org/sequence/region/human/1:1000000..1000100:1?content-type=text/x-fasta',
'https://rest.ensembl.org/sequence/region/human/1:1000000..1000100?content-type=application/json',
]
for url in urls:
    print('URL',url)
    try:
        req=urllib.request.Request(url, headers={'Accept':'application/json'})
        with urllib.request.urlopen(req,timeout=30) as r:
            data=r.read().decode()
        print(data[:1000])
    except Exception as e:
        print('ERR',repr(e))
PY"
