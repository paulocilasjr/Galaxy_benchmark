/bin/bash -lc "curl -L --fail --silent --show-error 'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr1;start=999999;end=1000100' -o data/chr1_1000000_1000100_ucsc.json && sed -n '1,4p' data/chr1_1000000_1000100_ucsc.json && perl -MJSON::PP -MDigest::MD5=md5_hex -e 'local "'$/; $j=decode_json(<>); $s=uc($j->{dna}); $len=length($s); $gc=($s=~tr/GC/GC/); $n=($s=~tr/N/N/); printf "%s'"\\n%d %d %d %.2f %s\\n\", "'$s,$len,$gc,$n,100*$gc/$len,md5_hex($s)'"' data/chr1_1000000_1000100_ucsc.json"
