import gzip, random, math, argparse, sys
from collections import defaultdict
import numpy as np
from scipy import sparse
from sklearn.decomposition import TruncatedSVD
from sklearn.neighbors import NearestNeighbors
from sklearn.preprocessing import normalize

ap=argparse.ArgumentParser()
ap.add_argument('--binsize',type=int,default=10000)
ap.add_argument('--n_doublets',type=int,default=5000)
ap.add_argument('--seed',type=int,default=1)
ap.add_argument('--components',type=int,default=50)
ap.add_argument('--neighbors',type=int,default=50)
ap.add_argument('--min_bin_cells',type=int,default=5)
ap.add_argument('--max_bin_frac',type=float,default=0.5)
ap.add_argument('--autosomes',action='store_true')
args=ap.parse_args()
random.seed(args.seed); np.random.seed(args.seed)
chroms=[f'chr{i}' for i in range(1,23)] if args.autosomes else None
bc_to_i={}; bcs=[]
feat_to_j={}
rows=[]; cols=[]; data=[]
# aggregate counts per cell/bin to avoid duplicate COO bloat
counts=defaultdict(int)
with gzip.open('atac.doublet.q1.bed.gz','rt') as f:
    for line in f:
        chrom,s,e,bc,c=line.rstrip('\\n').split('\\t')
        if chroms is not None and chrom not in chroms: continue
        if bc not in bc_to_i:
            bc_to_i[bc]=len(bcs); bcs.append(bc)
        # midpoint bin
        mid=(int(s)+int(e))//2
        key=(chrom, mid//args.binsize)
        j=feat_to_j.get(key)
        if j is None:
            j=len(feat_to_j); feat_to_j[key]=j
        counts[(bc_to_i[bc],j)] += int(c)
for (i,j),v in counts.items():
    rows.append(i); cols.append(j); data.append(v)
X=sparse.csr_matrix((data,(rows,cols)), shape=(len(bcs),len(feat_to_j)), dtype=np.float32)
# binarize accessibility, common for ATAC tiles
X.data[:] = 1.0
n=X.shape[0]
df=np.asarray((X>0).sum(axis=0)).ravel()
keep=(df>=args.min_bin_cells) & (df <= args.max_bin_frac*n)
X=X[:,keep].tocsr()
# TF-IDF: term frequency per cell * log(1+n/df)
cell_sums=np.asarray(X.sum(axis=1)).ravel()
Xtf=X.multiply(1/np.maximum(cell_sums,1)[:,None])
df=np.asarray((X>0).sum(axis=0)).ravel()
idf=np.log1p(n/np.maximum(df,1))
Xtf=Xtf.multiply(idf).tocsr()
Xtf.data=np.log1p(Xtf.data*1e4)
# simulate doublets as sums of raw binary vectors, then same TF-IDF using observed idf
pairs=np.random.randint(0,n,size=(args.n_doublets,2))
same=pairs[:,0]==pairs[:,1]
while same.any():
    pairs[same,1]=np.random.randint(0,n,size=same.sum())
    same=pairs[:,0]==pairs[:,1]
Xd=(X[pairs[:,0]] + X[pairs[:,1]]).tocsr()
Xd.data[:] = 1.0
sums=np.asarray(Xd.sum(axis=1)).ravel()
Xdtf=Xd.multiply(1/np.maximum(sums,1)[:,None]).multiply(idf).tocsr()
Xdtf.data=np.log1p(Xdtf.data*1e4)
Xall=sparse.vstack([Xtf,Xdtf], format='csr')
# SVD; skip component 0 often tracks technical/global variation
ncomp=min(args.components, min(Xall.shape)-1)
svd=TruncatedSVD(n_components=ncomp, random_state=args.seed)
Z=svd.fit_transform(Xall)
# normalize LSI coordinates after excluding first component
if ncomp>2:
    Zuse=Z[:,1:]
else:
    Zuse=Z
Zuse=normalize(Zuse)
nn=NearestNeighbors(n_neighbors=args.neighbors+1, metric='euclidean', algorithm='auto')
nn.fit(Zuse)
inds=nn.kneighbors(Zuse[:n], return_distance=False)[:,1:]
is_doublet=np.zeros(Xall.shape[0], dtype=bool); is_doublet[n:]=True
scores=is_doublet[inds].mean(axis=1)
# Adjusted score relative to expected simulated fraction in joint pool
order=np.argsort(-scores)
print('params',vars(args),'cells',n,'features',X.shape[1], 'explained',float(svd.explained_variance_ratio_.sum()))
for rank,i in enumerate(order[:100],1):
    print(rank,bcs[i],scores[i],int(cell_sums[i]))
