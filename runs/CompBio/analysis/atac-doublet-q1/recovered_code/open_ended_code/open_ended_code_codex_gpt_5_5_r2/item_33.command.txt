/bin/bash -lc "cat > amulet_like.py <<'PY'
import gzip, heapq, collections, argparse
ap=argparse.ArgumentParser(); ap.add_argument('--max_frag',type=int,default=1000); ap.add_argument('--autosomes',action='store_true'); args=ap.parse_args()
chrom_ok={f'chr{i}' for i in range(1,23)}
# For each barcode, sweep intervals within each chromosome. Count start events where active overlapping fragments before adding current >=2.
# Also track max depth and total starts in loci with resulting depth >2.
stats=collections.defaultdict(lambda:[0,0,0])  # total, over2_events, maxdepth
cur_chr=None
active=collections.defaultdict(list)
with gzip.open('atac.doublet.q1.bed.gz','rt') as f:
    for line in f:
        chrom,s,e,bc,c=line.rstrip('\\n').split('\\t')
        if args.autosomes and chrom not in chrom_ok: continue
        s=int(s); e=int(e)
        if e-s > args.max_frag: continue
        if chrom "'!= cur_chr:
            active.clear(); cur_chr=chrom
        h=active[bc]
        while h and h[0] <= s:
            heapq.heappop(h)
        depth=len(h)+1
        stats[bc][0]+=1
        if depth>2: stats[bc][1]+=1
        if depth>stats[bc][2]: stats[bc][2]=depth
        heapq.heappush(h,e)
for bc,(tot,over,maxd) in sorted(stats.items(), key=lambda kv:(-kv[1][1]/kv[1][0], -kv[1][1]))[:100]:
    print(bc,tot,over,over/tot,maxd)
PY
python3 amulet_like.py --max_frag 1000 --autosomes > amulet_like.txt
head -60 amulet_like.txt'
