/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, snapatac2 as snap
U=pd.read_csv('umap_peak_coords.tsv',sep='\\t')
at=snap.read('atac.snap.h5ad'); names_at=np.array(at.obs_names); nf=np.array(at.obs['n_fragment']); ss=np.array(at.obs['doublet_score'])
p=np.loadtxt('scrublet_peaks_scores.txt',dtype=str); ps={b:float(s) for b,s in zip(p[:,0],p[:,1])}
t=np.loadtxt('scrublet_tile_scores.txt',dtype=str); ts={b:float(s) for b,s in zip(t[:,0],t[:,1])}
U['nf']=[nf[np.where(names_at==b)[0][0]] for b in U.barcode]
U['snap_s']=[ss[np.where(names_at==b)[0][0]] for b in U.barcode]
U['peak_scrub']=[ps.get(b,np.nan) for b in U.barcode]
U['tile_scrub']=[ts.get(b,np.nan) for b in U.barcode]
print(U[U.leiden==6].sort_values('snap_s',ascending=False)[['barcode','nf','snap_s','peak_scrub','tile_scrub']].head(30))
PY"
