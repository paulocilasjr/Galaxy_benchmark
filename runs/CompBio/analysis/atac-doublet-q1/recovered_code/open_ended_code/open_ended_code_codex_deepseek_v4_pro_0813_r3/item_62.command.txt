/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np, time
adata=snap.read('atac.snap.h5ad')
t=time.time()
snap.tl.spectral(adata, n_comps=30, inplace=True)
print('spectral done',time.time()-t, 'obsm keys', list(adata.obsm.keys()))
snap.pp.knn(adata, n_neighbors=30, use_dims=30, inplace=True)
print('knn done')
snap.tl.leiden(adata, resolution=1.0, inplace=True)
print('leiden done')
leiden=np.array(adata.obs['leiden'])
print('clusters', np.unique(leiden, return_counts=True))
score=np.array(adata.obs['doublet_score'])
for cl in np.unique(leiden):
 mask=leiden==cl
 print(cl, mask.sum(), round(score[mask].mean(),4), round(score[mask].max(),4))
# save names and clusters
names=np.array(adata.obs_names)
np.savetxt('snap_clusters.txt', np.c_[names, leiden, score], fmt='%s')
PY"
