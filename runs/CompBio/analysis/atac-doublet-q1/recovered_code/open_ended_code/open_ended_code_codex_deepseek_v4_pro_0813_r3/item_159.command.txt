/bin/bash -c "cat > snap_default_features.py <<'PY'
import snapatac2 as snap, numpy as np, time
frags='atac.doublet.q1.bed.gz'
adata=snap.pp.import_fragments(frags, chrom_sizes=snap.genome.GRCh38.chrom_sizes, is_paired=True, min_num_fragments=0, sorted_by_barcode=False, file='atac.snap.default.h5ad')
snap.pp.add_tile_matrix(adata, bin_size=500, inplace=True)
snap.pp.select_features(adata, inplace=True)  # default 500k
print('selected', np.array(adata.var['selected']).sum())
snap.pp.scrublet(adata, expected_doublet_rate=0.06, sim_doublet_ratio=2.0, n_comps=30, random_state=0)
p=np.array(adata.obs['doublet_probability']); s=np.array(adata.obs['doublet_score']); names=np.array(adata.obs_names)
print('max p',p.max(),'p>0.5',(p>0.5).sum(),'p>0.1',(p>0.1).sum(),'s max',s.max())
idx=np.argsort(-s)
for i in idx[:50]: print(names[i],p[i],s[i])
PY
python3 snap_default_features.py"
