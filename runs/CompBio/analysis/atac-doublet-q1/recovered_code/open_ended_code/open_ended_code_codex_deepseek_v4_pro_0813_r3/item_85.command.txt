/bin/bash -c "cat > snap_peaks.py <<'PY'
import snapatac2 as snap, numpy as np, time
adata=snap.read('atac.snap.h5ad')
t=time.time()
peaks=snap.tl.macs3(adata, groupby=None, qvalue=0.05, inplace=True)
print('macs done',time.time()-t)
print(type(peaks), peaks.keys() if isinstance(peaks,dict) else peaks)
if isinstance(peaks,dict):
 for k,v in peaks.items(): print(k, v.shape, v)
print('adata uns keys', list(adata.uns.keys()))
# make peak matrix
adata2=snap.pp.make_peak_matrix(adata, use_rep=None, peak_file=None, inplace=False, counting_strategy='fragment')
print('peak matrix', adata2.shape, type(adata2.X), getattr(adata2.X,'nnz',None))
# select features? peak matrix has fewer features
snap.pp.select_features(adata2, n_features=None, filter_lower_quantile=0.005, filter_upper_quantile=0.005)
print('after select', adata2.shape, 'n selected', np.array(adata2.var['selected']).sum() if 'selected' in adata2.var else '?')
# save
adata2.write('atac_peaks.h5ad')
print('saved')
PY
python3 snap_peaks.py"
