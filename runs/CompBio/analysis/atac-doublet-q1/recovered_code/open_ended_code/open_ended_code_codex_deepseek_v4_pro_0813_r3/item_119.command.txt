/bin/bash -c "python3 - <<'PY'
import numpy as np
# load top50 lists from files and peak top50 from earlier? save snap top lists.
bins={}
for b in [500,1000,5000,10000]:
    if b==500:
        # use atac.snap.h5ad scores
        import snapatac2 as snap
        adata=snap.read('atac.snap.h5ad')
        names=np.array(adata.obs_names); s=np.array(adata.obs['doublet_score'])
        top=set(names[np.argsort(-s)[:50]])
    else:
        top=set(open(f'snap_bin{b}_top50.txt').read().split())
    bins[b]=top
# peak top50
import snapatac2 as snap
ap=snap.read('atac_peaks_scrub.h5ad')
names=np.array(ap.obs_names); s=np.array(ap.obs['doublet_score'])
bins['peak']=set(names[np.argsort(-s)[:50]])
keys=list(bins)
for k in keys: print(k,len(bins[k]))
common=set.intersection(*bins.values()) if bins else set()
print('common all',common)
# cells in at least 4 of 5
from collections import Counter
cnt=Counter()
for top in bins.values(): cnt.update(top)
for b,c in cnt.most_common(30): print(b,c)
PY"
