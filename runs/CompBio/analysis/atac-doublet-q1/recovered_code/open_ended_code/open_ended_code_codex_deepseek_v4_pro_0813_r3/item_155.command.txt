/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
from scipy.sparse import csr_matrix
ap=snap.read('atac_peaks_scrub.h5ad'); X=ap.X[:].tocsr().astype(np.float32); X.data[:]=1; N=np.asarray(X.sum(axis=1)).ravel(); names=np.array(ap.obs_names)
at=snap.read('atac.snap.h5ad'); names2=np.array(at.obs_names); nf=np.array(at.obs['n_fragment']); idx=[np.where(names2==n)[0][0] for n in names]; nf=nf[idx]
# residual N ~ nf
coef=np.polyfit(nf,N,1); pred=np.polyval(coef,nf); res=N-pred
order=np.argsort(-res)
for i in order[:50]: print(names[i], 'nf',nf[i],'N',N[i],'res',round(res[i],1))
# candidate values
for b in ['GCGACACGCCGGGCAC-1','CGAGCAATAGAGGTGG-1','CGCACGCGTACTGCAG-1','GACTATAATGATTTTT-1','TGAGCCTCACGCTTCG-1','GGCGACCTCTACGTAG-1','CGTGTGGATCTGGCAG-1','TTCGAGACGTAGAAAA-1']:
 j=np.where(names==b)[0][0]; print('cand',b,'nf',nf[j],'N',N[j],'res',round(res[j],1))
PY"
