/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np, time
adata=snap.read('atac.snap.h5ad')
t=time.time()
adata2=snap.pp.make_peak_matrix(adata, use_rep='macs3_pseudobulk', inplace=False, counting_strategy='fragment')
print('peak mat', adata2.shape, type(adata2.X), getattr(adata2.X,'nnz',None), time.time()-t)
print('var keys?')
# run select_features? for peaks maybe n_features None all?
snap.pp.select_features(adata2, n_features=100000, filter_lower_quantile=0.005, filter_upper_quantile=0.005)
print('selected sum', np.array(adata2.var['selected']).sum() if 'selected' in adata2.var else '?')
# scrublet on peak matrix using selected features
snap.pp.scrublet(adata2, expected_doublet_rate=0.06, sim_doublet_ratio=2.0, n_comps=30, random_state=0)
print('scrub done')
p=np.array(adata2.obs['doublet_probability']); s=np.array(adata2.obs['doublet_score']); names=np.array(adata2.obs_names)
print('p max',p.max(),'p>0.5',(p>0.5).sum(),'p>0.1',(p>0.1).sum(),'s max',s.max())
idx=np.argsort(-s)
for i in idx[:40]: print(names[i],p[i],s[i])
adata2.write('atac_peaks_scrub.h5ad')
PY"
