/bin/bash -c "cat > snap_multibin.py <<'PY'
import snapatac2 as snap, numpy as np, time, gc
frags='atac.doublet.q1.bed.gz'
# import once to in-memory? Use backed file with copy? Simpler import each with file names
for bin_size in [1000,5000,10000]:
    t=time.time()
    file=f'atac.snap.bin{bin_size}.h5ad'
    adata=snap.pp.import_fragments(frags, chrom_sizes=snap.genome.GRCh38.chrom_sizes, is_paired=True, min_num_fragments=0, sorted_by_barcode=False, file=file)
    snap.pp.add_tile_matrix(adata, bin_size=bin_size, counting_strategy='fragment', inplace=True)
    snap.pp.select_features(adata, n_features=50000, filter_lower_quantile=0.005, filter_upper_quantile=0.005, inplace=True)
    snap.pp.scrublet(adata, expected_doublet_rate=0.06, sim_doublet_ratio=2.0, n_comps=30, random_state=0, inplace=True)
    names=np.array(adata.obs_names); s=np.array(adata.obs['doublet_score']); p=np.array(adata.obs['doublet_probability'])
    idx=np.argsort(-s)
    print('\\nBIN',bin_size,'max s',s.max(),'p>0.5',(p>0.5).sum(),'time',time.time()-t)
    top=[]
    for i in idx[:50]:
        top.append(names[i]); print(names[i], round(s[i],4))
    np.savetxt(f'snap_bin{bin_size}_top50.txt', np.array(top), fmt='%s')
    del adata; gc.collect()
PY
python3 snap_multibin.py"
