/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
from scipy.sparse import csr_matrix
adata=snap.read('atac_peaks_scrub.h5ad'); X=adata.X[:].tocsr().astype(np.float32); X.data[:]=1; names=np.array(adata.obs_names)
at=snap.read('atac.snap.h5ad'); names2=np.array(at.obs_names); idx=[np.where(names2==n)[0][0] for n in names]; leiden=np.array(at.obs['leiden'])[idx]; uniq,cl=np.unique(leiden,return_inverse=True)
means=np.vstack([np.asarray(X[cl==c].mean(axis=0)).ravel() for c in range(len(uniq))])
# cosine sim between cluster means
from sklearn.metrics.pairwise import cosine_similarity
S=cosine_similarity(means)
print('cluster similarity matrix')
print('uniq',uniq)
print(S.round(3))
# top similar pairs
for i in range(len(uniq)):
 for j in range(i+1,len(uniq)):
  print(uniq[i],uniq[j],round(S[i,j],3))
PY"
