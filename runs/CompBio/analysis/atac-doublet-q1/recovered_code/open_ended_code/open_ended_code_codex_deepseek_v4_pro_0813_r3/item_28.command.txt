/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
adata=snap.read('atac.snap.h5ad')
obs=adata.obs
# print available keys by trying known
for key in ['n_fragment','frac_dup','frac_mito','doublet_probability','doublet_score','doublet']:
    try:
        s=obs[key]
        print(key, s[:5].to_list())
    except Exception as e:
        print(key, 'no', e)
# names
print('obs names first', adata.obs_names[:5])
# top doublet prob
if 'doublet_probability' in obs:
    p=obs['doublet_probability'].to_numpy()
    names=np.array(adata.obs_names)
    idx=np.argsort(-p)
    for i in idx[:40]:
        print(names[i], p[i], obs['doublet_score'][i], obs['doublet'][i])
PY"
