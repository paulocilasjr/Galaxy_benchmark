/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
adata=snap.read('atac.snap.h5ad'); nf=np.array(adata.obs['n_fragment']); names=np.array(adata.obs_names)
print('nf quantiles',np.quantile(nf,[0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]))
# sorted top 100 and bottom 100
idx=np.argsort(nf)
print('lowest 10',[(names[i],nf[i]) for i in idx[:10]])
print('highest 10',[(names[i],nf[i]) for i in idx[-10:][::-1]])
# fit Gaussian mixture 2 comps
from sklearn.mixture import GaussianMixture
X=nf.reshape(-1,1)
for k in [2,3]:
 g=GaussianMixture(n_components=k,random_state=0).fit(X)
 print('k',k,'means',g.means_.ravel(),'weights',g.weights_,'bic',g.bic(X))
PY"
