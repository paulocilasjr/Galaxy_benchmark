/bin/bash -c "python3 - <<'PY'
import numpy as np, snapatac2 as snap
# snap tile scores from atac.snap.h5ad
at=snap.read('atac.snap.h5ad')
names=np.array(at.obs_names); st=np.array(at.obs['doublet_score']); pt=np.array(at.obs['doublet_probability'])
# snap peak scores from atac_peaks_scrub
ap=snap.read('atac_peaks_scrub.h5ad')
namesp=np.array(ap.obs_names); sp=np.array(ap.obs['doublet_score']); pp=np.array(ap.obs['doublet_probability'])
for b in ['CGCACGCGTACTGCAG-1','GACTATAATGATTTTT-1','TGAGCCTCACGCTTCG-1','GGCGACCTCTACGTAG-1','CGTGTGGATCTGGCAG-1','GCGACACGCCGGGCAC-1','TTCGAGACGTAGAAAA-1']:
    j=np.where(names==b)[0]
    if len(j): j=j[0]; print(b,'snap_tile score',st[j],'prob',pt[j])
    k=np.where(namesp==b)[0]
    if len(k): k=k[0]; print(b,'snap_peak score',sp[k],'prob',pp[k])
PY"
