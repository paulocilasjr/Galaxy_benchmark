/bin/bash -lc "python3 - <<'PY'
import gzip, collections, sys
bc_counts=collections.Counter()
n=0
with gzip.open('atac.doublet.q1.bed.gz','rt') as f:
    for line in f:
        parts=line.split()
        bc=parts[3]
        bc_counts[bc]+=1
        n+=1
        if n%1000000==0:
            print('read',n, 'uniq',len(bc_counts), file=sys.stderr)
print('total',n,'uniq',len(bc_counts))
vals=list(bc_counts.values())
print('sum',sum(vals))
import statistics
vals_sorted=sorted(vals)
print('min',vals_sorted[0], 'max',vals_sorted[-1])
for q in [0.01,0.05,0.1,0.25,0.5,0.75,0.9,0.95,0.99,0.995,0.999]:
    idx=min(len(vals_sorted)-1,int(q*len(vals_sorted)))
    print(q, vals_sorted[idx])
# print top 30
for bc,c in bc_counts.most_common(50):
    print(bc,c)
PY"
