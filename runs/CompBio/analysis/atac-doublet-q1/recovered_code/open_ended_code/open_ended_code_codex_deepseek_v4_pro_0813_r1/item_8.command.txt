/bin/bash -lc "cat > compute_basic2.py <<'PY'
import gzip, collections, sys, statistics, time
fn='atac.doublet.q1.bed.gz'
bcs=[]; first=set()
counts=collections.Counter()
frag_sets={}
bin5_sets={}
bin10_sets={}
with gzip.open(fn,'rt') as f:
    for line in f:
        parts=line.split()
        ch=parts[0]; start=int(parts[1]); end=int(parts[2]); bc=parts[3]
        if bc not in first:
            first.add(bc); bcs.append(bc)
            frag_sets[bc]=set(); bin5_sets[bc]=set(); bin10_sets[bc]=set()
        counts[bc]+=1
        frag_sets[bc].add((start,end))
        bin5_sets[bc].add((ch,start//5000))
        bin10_sets[bc].add((ch,start//10000))
rows=[]
for bc in bcs:
    c=counts[bc]; uf=len(frag_sets[bc]); ub5=len(bin5_sets[bc]); ub10=len(bin10_sets[bc])
    rows.append((bc,c,uf,ub5,ub10,c-uf,c/ub10 if ub10 else 0,c/ub5 if ub5 else 0))
print('n',len(rows))
for name,idx in [('total',1),('unique_frag',2),('bin5k',3),('bin10k',4),('dup',5),('frag_per10k',6),('frag_per5k',7)]:
    vals=sorted(r[idx] for r in rows)
    print('\\n',name,'min max mean median p95 p99 p999')
    print(vals[0],vals[-1],statistics.mean(vals),statistics.median(vals),vals[int(.95*(len(vals)-1))],vals[int(.99*(len(vals)-1))],vals[int(.999*(len(vals)-1))])
    top=sorted(rows,key=lambda r:r[idx], reverse=True)[:20]
    print([(r[0],r[idx]) for r in top])
print('\\nTop total')
for r in sorted(rows,key=lambda r:r[1], reverse=True)[:30]: print(r)
PY
time python3 compute_basic2.py"
