/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np
path='atac.doublet.q1.bed.gz'; bd={}; names=[]; n=[]; sl=[]; sl2=[]; bins=[]
with gzip.open(path,'rt') as h:
 for line in h:
  f=line.rstrip().split('\\t'); b=f[3];
  if b not in bd:
   bd[b]=len(names);names.append(b);n.append(0);sl.append(0);sl2.append(0);bins.append([0]*6)
  i=bd[b]; L=int(f[2])-int(f[1]);n[i]+=1;sl[i]+=L;sl2[i]+=L*L
  bins[i][0]+=L<100; bins[i][1]+=100<=L<180; bins[i][2]+=180<=L<300; bins[i][3]+=300<=L<500; bins[i][4]+=500<=L<1000; bins[i][5]+=L>=1000
n=np.array(n); sl=np.array(sl); sl2=np.array(sl2); bins=np.array(bins)/n[:,None]; mean=sl/n; sd=np.sqrt(sl2/n-mean**2)
print('mean range',mean.min(),mean.max(),'sd range',sd.min(),sd.max())
for metric,vals in [('mean',mean),('sd',sd)]+[(f'bin{i}',bins[:,i]) for i in range(6)]:
 print('\\n',metric,'high')
 for i in np.argsort(-vals)[:15]:print(names[i],round(float(vals[i]),4),n[i])
 print('low')
 for i in np.argsort(vals)[:10]:print(names[i],round(float(vals[i]),4),n[i])
PY"
