import gzip
from collections import Counter
path='atac.doublet.q1.bed.gz'; target='AACATGGAGCGCCGGG-1'
keys=set()
with gzip.open(path,'rt') as h:
 for l in h:
  f=l.rstrip().split('\\t')
  if f[3]==target: keys.add((f[0],int(f[1]),int(f[2])))
print('target unique',len(keys))
c=Counter(); n=Counter()
with gzip.open(path,'rt') as h:
 for l in h:
  f=l.rstrip().split('\\t'); b=f[3]; n[b]+=1
  if (f[0],int(f[1]),int(f[2])) in keys:c[b]+=1
for b,v in c.most_common(30):print(b,v,n[b],round(v/n[b],4))
PY"
