/bin/bash -lc "python3 - <<'PY'
import numpy as np
p=np.load('profile_1000.npz',allow_pickle=True); z=p['Z']; b=p['barcodes']; n=len(b)
s=z@z.T; np.fill_diagonal(s,-1)
best=np.max(s,axis=1); arg=np.full((n,2),-1,int)
# initialize args nearest
arg[:,0]=np.argmax(s,axis=1)
ii,jj=np.triu_indices(n,1); mask=s[ii,jj]<0.7; ii=ii[mask];jj=jj[mask]
for st in range(0,len(ii),1000):
 j=ii[st:st+1000]; k=jj[st:st+1000]
 a=s[:,j]; bb=s[:,k]; c=s[j,k][None,:]
 # optimum positive mixture weight r on v; if denominator is nonzero
 den=a-bb*c
 num=bb-a*c
 r=num/den
 valid=(den>1e-9)&(r>0)&np.isfinite(r)
 # cosine at optimum, otherwise boundary already represented by best a/b only if those are >=0
 val=(a+r*bb)/np.sqrt(1+2*r*c+r*r)
 val[~valid]=-1
 # exclude pairs containing target
 q=np.arange(n)[:,None]
 val[(q==j[None,:])|(q==k[None,:])]=-1
 mx=np.max(val,axis=1); ix=np.argmax(val,axis=1); m=mx>best
 best[m]=mx[m]; arg[m,0]=j[ix[m]];arg[m,1]=k[ix[m]]
score=best-np.max(s,axis=1)
print('candidate',b[369],best[369],np.max(s[369]),score[369],b[arg[369,0]],b[arg[369,1]])
for i in np.argsort(-score)[:50]: print(b[i],round(float(score[i]),4),round(float(best[i]),4),round(float(np.max(s[i])),4),b[arg[i,0]],b[arg[i,1]])
PY"
