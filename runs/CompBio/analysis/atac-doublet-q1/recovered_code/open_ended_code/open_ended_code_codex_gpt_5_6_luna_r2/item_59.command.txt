/bin/bash -lc "python3 - <<'PY'
import numpy as np
from sklearn.cluster import KMeans
p=np.load('profile_1000.npz',allow_pickle=True);z=p['Z']; b=p['barcodes'];f=p['fragments'];
for k in [5,8,10,15,20,30]:
 c=KMeans(k,n_init=5,random_state=1).fit_predict(z);s=z@z.T;np.fill_diagonal(s,-1);o=np.argsort(-s,axis=1)
 print('\\nK',k)
 for r in [10,20,50]:
  ent=[]
  for i in range(len(b)):
   h=np.bincount(c[o[i,:r]],minlength=k);q=h/h.sum();ent.append(-(q[q>0]*np.log2(q[q>0])).sum())
  ent=np.array(ent);print('r',r,'range',ent.min(),ent.max(),'median',np.median(ent))
  for i in np.argsort(-ent)[:10]: print(b[i],round(float(ent[i]),3),np.bincount(c[o[i,:r]],minlength=k).tolist(),f[i])
PY"
