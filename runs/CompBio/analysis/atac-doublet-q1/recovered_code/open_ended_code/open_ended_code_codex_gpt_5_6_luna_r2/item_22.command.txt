/bin/bash -lc "python3 - <<'PY'
import numpy as np
p=np.load('profile_10000.npz',allow_pickle=True)
cc=p['chrom_counts']; names=p['chrom_names']; b=p['barcodes']; f=p['fragments']
print(names)
# normalize chrom fractions; flag cells with largest Mahalanobis-ish residual after PCA
frac=cc[:,:len(names)]/f[:,None]
from sklearn.decomposition import PCA
z=PCA(10).fit_transform(frac)
from sklearn.neighbors import NearestNeighbors
D,ix=NearestNeighbors(n_neighbors=2).fit(z).kneighbors(z)
for i in np.argsort(-D[:,1])[:30]: print(i,b[i],f[i],D[i,1],np.round(frac[i],3))
print('var by chr')
for j,n in enumerate(names):print(n,frac[:,j].mean(),frac[:,j].std())
PY"
