import numpy as np
x=np.loadtxt('cell_metrics.tsv',dtype=str,delimiter='\\t'); b=x[:,0]; v=x[:,1:].astype(float); n=v[:,:24].sum(1); p=v[:,:24]/n[:,None]; med=np.median(p,0); mad=np.median(abs(p-med),0); z=(p-med)/(1.4826*mad); score=np.sqrt((z*z).sum(1));
for j in np.argsort(score)[-50:][::-1]: print(b[j],round(score[j],2),round(n[j]),np.round(z[j],1))
