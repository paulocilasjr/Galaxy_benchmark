/bin/bash -lc "python3 - <<'PY'
import numpy as np
x=np.loadtxt('cell_metrics.tsv',dtype=str,delimiter='\\t'); b=x[:,0]; v=x[:,1:].astype(float)
# aggregate
n=v[:,:24].sum(1); bins=v[:,24:29]; mean=v[:,29]
for name,a in [('n',n),('mean',mean),('short_frac',bins[:,0]/n),('mid_frac',bins[:,1]/n),('mono_frac',bins[:,2]/n),('long_frac',bins[:,3]/n),('xlong_frac',bins[:,4]/n)]:
 med=np.median(a); mad=np.median(np.abs(a-med)); z=(a-med)/(1.4826*mad) if mad else (a-a.mean())/a.std()
 print('\\n',name,'median',med,'mad',mad,'range',a.min(),a.max())
 for j in np.argsort(z)[-15:][::-1]: print(b[j],round(float(a[j]),5),round(float(z[j]),2))
 print('low')
 for j in np.argsort(z)[:5]: print(b[j],round(float(a[j]),5),round(float(z[j]),2))
# robust multivariate normalized QC score positive n, mean, xlong, maybe combination
A=np.column_stack([n,mean,bins/n])
med=np.median(A,axis=0); mad=np.median(abs(A-med),axis=0); Z=(A-med)/(1.4826*mad); score=np.sqrt(np.sum(np.maximum(Z,0)**2,axis=1))
print('\\ncombined')
for j in np.argsort(score)[-50:][::-1]: print(b[j],round(float(score[j]),3),np.round(Z[j],2),np.round(A[j],3))
PY"
