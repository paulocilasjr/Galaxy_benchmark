/bin/bash -lc "python3 - <<'PY'
import numpy as np
from scipy import sparse
z=np.load('doublet_profiles.npz',allow_pickle=True)
X=z['X'].item(); b=z['barcodes']; n=X.shape[0]
df=np.asarray((X>0).sum(0)).ravel(); idf=np.log1p(n/(1+df)); Y=X.multiply(idf); Y=Y.multiply(1/np.sqrt(np.asarray(Y.multiply(Y).sum(1)).ravel())[:,None])
G=(Y@Y.T).toarray(); G=(G+G.T)/2
w,u=np.linalg.eigh(G); ix=np.argsort(w)[::-1]; w=w[ix]; u=u[:,ix]
print('eigen',w[:30], 'ratio',w[:10]/w[0])
# nearest in PC 1..N (center? no)
for start in [0,1,2,3,5,10]:
 Z=u[:,start:start+20]*np.sqrt(np.maximum(w[start:start+20],0))[None,:]
 # normalize? Euclidean
 d=((Z[:,None,:]-Z[None,:,:])**2).sum(2); np.fill_diagonal(d,np.inf)
 kn=np.sort(d,axis=1)[:,:30]
 score=kn.mean(1)
 print('\\nstart',start,'range',score.min(),score.max())
 for j in np.argsort(score)[-20:][::-1]: print(b[j],round(score[j],5), 'pc',np.round(Z[j,:3],3))
PY"
