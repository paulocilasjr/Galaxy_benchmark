/bin/bash -lc "mkdir -p tmp_sort && gzip -dc atac.doublet.q1.bed.gz | awk '{print "'$4"'"\\t\""'$2; print $4"'"\\t\""'$3}'"' | LC_ALL=C sort -S 2G -T tmp_sort > tmp_sort/coords_sorted.tsv && python3 - <<'PY'
from collections import defaultdict
# aggregate positions with multiplicity >2 per barcode
prev=None; run=1; positions=defaultdict(int)
with open('tmp_sort/coords_sorted.tsv') as f:
 for line in f:
  b,pos=line.rstrip('\\n').split('\\t')
  key=(b,pos)
  if key==prev:
   run+=1
  else:
   if prev is not None and run>2:
    positions[prev[0]]+=1
   prev=key; run=1
 if prev is not None and run>2:
  positions[prev[0]]+=1
pairs=sorted(positions.items(), key=lambda x:-x[1])
print('barcodes with >2 loci',len(pairs))
for b,c in pairs[:100]: print(b,c)
print('top counts')
# load total counts from [REDACTED LOCAL PATH]
counts={}
for line in open('[REDACTED LOCAL PATH]'):
 c,b=line.split(); counts[b]=int(c)
for b,c in pairs[:30]: print(b,'loci_gt2',c,'fragments',counts.get(b))
PY"
