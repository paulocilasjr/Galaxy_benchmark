/bin/bash -lc "gzip -dc atac.doublet.q1.bed.gz | awk '"'$4=="AACATGGAGCGCCGGG-1"'"' > tmp_sort/barcode_frags.tsv; wc -l tmp_sort/barcode_frags.tsv; head -20 tmp_sort/barcode_frags.tsv; python3 - <<'PY'
from collections import Counter
starts=Counter();ends=Counter(); both=Counter()
with open('tmp_sort/barcode_frags.tsv') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  starts[p[1]]+=1; ends[p[2]]+=1; both[p[1]]+=1; both[p[2]]+=1
print('starts >2',sum(1 for v in starts.values() if v>2),'ends >2',sum(1 for v in ends.values() if v>2),'both >2',sum(1 for v in both.values() if v>2))
print('top start counts', starts.most_common(20))
print('top end counts', ends.most_common(20))
print('loci >2 both',[(k,v) for k,v in both.most_common(60) if v>2])
PY"
