/bin/bash -lc "python3 - <<'PY'
import snapatac2 as snap, numpy as np, time, gc, sys
from snapatac2.preprocessing._scrublet import scrub_doublets_core
adata=snap.read('atac.h5ad')
mask=np.array(adata.var['selected'].to_list())
X=adata.X[:,mask].tocsr()
print('X',X.shape,X.nnz)
for seed in [0,1,2]:
 print('\\nseed',seed, flush=True)
 t=time.time()
 scores_obs, scores_sim, *_ = scrub_doublets_core(X, 16, 2.0, 0.05, n_comps=15, random_state=seed, verbose=False)
 names=adata.obs_names
 pairs=sorted(zip(scores_obs,names), reverse=True)
 print('time',time.time()-t)
 for s,n in pairs[:15]: print(n,s)
 # save
 np.save(f'scores_obs_seed{seed}.npy', scores_obs)
 np.save(f'scores_sim_seed{seed}.npy', scores_sim)
 del scores_obs,scores_sim
 gc.collect()
PY"
