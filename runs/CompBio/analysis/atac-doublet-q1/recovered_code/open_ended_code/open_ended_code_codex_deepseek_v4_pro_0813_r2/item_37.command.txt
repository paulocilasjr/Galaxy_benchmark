/bin/bash -lc "python3 - <<'PY'
import snapatac2 as snap, time
adata=snap.read('atac.h5ad')
t=time.time()
snap.tl.spectral(adata, n_comps=30, features='selected', random_state=0, distance_metric='cosine', weighted_by_sd=True, inplace=True, num_threads=8)
print('spectral done', time.time()-t)
t=time.time()
snap.pp.knn(adata, n_neighbors=30, use_rep='X_spectral', method='kdtree', inplace=True)
print('knn done', time.time()-t)
t=time.time()
snap.tl.leiden(adata, resolution=1.0, key_added='leiden', inplace=True)
print('leiden done', time.time()-t)
t=time.time()
snap.tl.umap(adata, use_rep='X_spectral', key_added='umap', random_state=0, inplace=True)
print('umap done', time.time()-t)
print('keys', adata.obsm.keys(), adata.obsp.keys())
PY"
