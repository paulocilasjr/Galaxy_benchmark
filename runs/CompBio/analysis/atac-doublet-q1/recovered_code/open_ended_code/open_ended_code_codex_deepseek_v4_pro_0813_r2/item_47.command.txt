import snapatac2 as snap, numpy as np
adata=snap.read('atac.h5ad')
sim=adata.uns['scrublet_sim_doublet_score']
obs=np.array(adata.obs['doublet_score'].to_list())

print('obs len',len(obs),'min max mean',obs.min(),obs.max(),obs.mean())
for q in [0,0.01,0.05,0.1,0.25,0.5,0.75,0.9,0.95,0.99,1]:
 print(q, np.quantile(sim,q), np.quantile(obs,q))
# fraction sims <= obs max etc
print('sim <= obs max',(sim<=obs.max()).mean())
print('sim <= obs 0.139',(sim<=0.139158576).mean())
# top obs and sim percentile
for val in [0.139158576,0.12138728,0.1070496,0.095238,0.085339]:
 print(val, (sim<=val).mean())
