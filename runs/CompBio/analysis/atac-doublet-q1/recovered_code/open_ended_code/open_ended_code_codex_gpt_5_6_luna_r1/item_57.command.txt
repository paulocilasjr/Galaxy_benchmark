/bin/bash -lc "python3 -u - <<'PY'
import gzip,h5py,bisect,collections,time
peaks=collections.defaultdict(list)
with h5py.File('pbmc10k_v2_peaks.h5','r') as f:
 for x in f['matrix/features/name'][:]:
  s=x.decode(); ch,co=s.split(':'); st,en=map(int,co.split('-')); peaks[ch].append((st,en))
for ch in peaks: peaks[ch].sort()
print('peaks',sum(map(len,peaks.values())),{k:len(v) for k,v in list(peaks.items())[:3]},flush=True)
starts={ch:[x[0] for x in v] for ch,v in peaks.items()}
nh=0; nf=0
with gzip.open('atac.doublet.q1.bed.gz','rt') as f:
 for l in f:
  a=l.rstrip().split('\\t');ch=a[0]; st=int(a[1]); en=int(a[2]); nf+=1
  for pos in (st,en-1):
   j=bisect.bisect_right(starts[ch],pos)-1
   if j>=0 and peaks[ch][j][1]>pos:nh+=1
  if nf==1000000:break
print('frags',nf,'cuts in peaks',nh,'frac',nh/(2*nf),flush=True)
PY"
