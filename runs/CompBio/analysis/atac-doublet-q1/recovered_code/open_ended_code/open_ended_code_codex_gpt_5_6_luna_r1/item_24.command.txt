/bin/bash -lc "python3 -u - <<'PY'
from doublet_scan import load_fragments
import numpy as np
for bs,mode in [(50000,'mid'),(10000,'mid'),(5000,'mid'),(2000,'start'),(1000,'start')]:
 x,b,n=load_fragments('atac.doublet.q1.bed.gz',bs,mode)

 print('\\n',bs,mode,'nnz q',np.quantile(u,[0,.01,.05,.5,.95,.99,1]),flush=True)
 print('low',[(b[i],int(u[i])) for i in np.argsort(u)[:10]])
 print('high',[(b[i],int(u[i])) for i in np.argsort(u)[-10:][::-1]])
PY"
