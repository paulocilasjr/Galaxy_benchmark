/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
 cats=np.array([x.decode() for x in f['var/gene_symbols/categories'][:]])
 syms=cats[f['var/gene_symbols/codes'][:]]
 c=np.asarray(X.sum(1)).ravel(); keep=c>3000
 print('keep',keep.sum())
 total=np.asarray(X[keep].sum(0)).ravel(); nz=np.asarray((X[keep]>0).sum(0)).ravel()
 excluded={'Malat1'}
 for i in np.argsort(total)[-300:][::-1]:
  if not (syms[i].startswith(('Rpl','Rps','mt-')) or syms[i] in excluded): print(syms[i],total[i],nz[i])
PY"
