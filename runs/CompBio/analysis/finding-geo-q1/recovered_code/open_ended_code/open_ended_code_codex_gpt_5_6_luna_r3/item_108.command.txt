/bin/bash -lc "python - <<'PY'
import json,urllib.parse,urllib.request,time,sys
q='H5[All Fields] AND Mus musculus[Organism] AND GSE[Entry Type] AND 1900:2020[PDAT]'
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
u=base+'esearch.fcgi?'+urllib.parse.urlencode({'db':'gds','term':q,'retmode':'json','retmax':100000})
ids=json.load(urllib.request.urlopen(u))['esearchresult']['idlist']
print('ids',len(ids),file=sys.stderr)
out=[]
for i in range(0,len(ids),10):
 params={'db':'gds','id':','.join(ids[i:i+10]),'retmode':'json'}
 url=base+'esummary.fcgi?'+urllib.parse.urlencode(params)
 for attempt in range(4):
  try:
   raw=urllib.request.urlopen(url,timeout=60).read()
   x=json.loads(raw)['result']; break
  except Exception as e:
   if attempt==3: print('FAIL',i,e,file=sys.stderr); x={}
   else: time.sleep(1+attempt)
 for k,v in x.items():
  if k=='uids': continue
  # retain only useful metadata
  out.append({z:v.get(z) for z in ['uid','accession','title','summary','suppfile','ftplink','gse','n_samples','ssinfo','relations','extrelations','pdat','seriestitle','platformtitle','taxon']})
 if i%50==0: print('batch',i,file=sys.stderr)
 time.sleep(.12)
open('geo_h5_mouse_old_summary.json','w').write(json.dumps(out))
print('out',len(out),file=sys.stderr)
PY"
