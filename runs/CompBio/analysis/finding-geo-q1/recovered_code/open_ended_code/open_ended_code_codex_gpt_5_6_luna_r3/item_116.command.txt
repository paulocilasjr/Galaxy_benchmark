curl -L --fail --silent --show-error -o GSM3578982.h5 https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM3578nnn/GSM3578982/suppl/GSM3578982_raw_gene_bc_matrices_h5.h5
curl -L --fail --silent --show-error -o GSM3659648.h5 https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM3659nnn/GSM3659648/suppl/GSM3659648_cNSC-BAC_Cntrl_Rep1_abundance.h5
python - <<'PY'
import h5py,os
for fn in ['GSM3578982.h5','GSM3659648.h5']:
 print('\\n',fn,os.path.getsize(fn)/1e6)
 with h5py.File(fn) as f:
  def rec(g,p=''):
   for k,v in g.items():
    q=p+'/'+k
    if hasattr(v,'shape'): print(q,v.shape,v.dtype)
    else:rec(v,q)
  rec(f)
PY"
