/bin/bash -lc "python - <<'PY'
import json, urllib.parse, subprocess
ids=','.join(json.load(open('geo_ids2.json'))['esearchresult']['idlist'])
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=gds&id='+urllib.parse.quote(ids)+'&retmode=json'
open('geo_summary.json','wb').write(subprocess.check_output(['curl','-L','--fail','--silent','--show-error',url]))
PY
python - <<'PY'
import json
x=json.load(open('geo_summary.json'))['result']
rows=[]
for k,v in x.items():
 if k=='uids': continue
 acc=v.get('accession','')
 if acc.startswith('GSE'):
  rows.append((acc,v.get('title',''),v.get('entrytype',''),v.get('n_samples','')))
print('GSE',len(rows))
for r in rows[:200]: print('\\t'.join(map(str,r)))
PY"
