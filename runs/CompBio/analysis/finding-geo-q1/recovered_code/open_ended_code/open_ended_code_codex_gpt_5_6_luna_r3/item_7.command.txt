import h5py, json
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    print('root attrs', {k: f.attrs[k] for k in f.attrs})
    print('X shape attr', f['X'].attrs['shape'])
    for grp in ['obs','var','uns']:
        print(grp, list(f[grp].keys()), {k:f[grp].attrs[k] for k in f[grp].attrs})
    obs=f['obs']['_index'][:]
    var=f['var']['_index'][:]
    print('obs first', [x.decode() if isinstance(x,bytes) else x for x in obs[:20]])
    print('obs last', [x.decode() if isinstance(x,bytes) else x for x in obs[-20:]])
    print('var first', [x.decode() if isinstance(x,bytes) else x for x in var[:20]])
    print('var last', [x.decode() if isinstance(x,bytes) else x for x in var[-20:]])
    gs=f['var']['gene_symbols']
    print('gene_symbols keys',list(gs.keys()))
    cats=[x.decode() if isinstance(x,bytes) else x for x in gs['categories'][:]]
    codes=gs['codes'][:]
    print('gene symbols first', [cats[i] for i in codes[:30]])
    print('gene symbols last', [cats[i] for i in codes[-30:]])
PY"
