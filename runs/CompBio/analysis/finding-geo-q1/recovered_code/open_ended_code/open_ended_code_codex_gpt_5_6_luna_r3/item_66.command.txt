import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
 cats=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/gene_symbols/categories'][:]],object)
 genes=cats[f['var/gene_symbols/codes'][:]].astype(str)
# map exact, list markers
markers='Dcx Dclk1 Tbr1 Tbr2 Eomes Neurog1 Neurog2 Neurod1 Neurod2 Sox2 Sox3 Sox11 Sox4 Hes1 Hes5 Ascl1 Dll1 Notch1 Pax6 Emx1 Emx2 Foxg1 Bcl11b Ctip2 Satb2 Cux1 Cux2 Tle4 Fezf2 Btg1 Btg2 Ebf1 Ebf2 Onecut1 Onecut2 Tubb3 Tuba1a Stmn1 Stmn2 Dlx1 Dlx2 Dlx5 Dlx6 Gad1 Gad2 Slc6a1 Slc17a7 Slc17a6 Nkx2-1 Lhx6 Lhx1 Lhx2 Lhx5 Sp8 Sp9 Igfbp7 Olig1 Olig2 Pdgfra Pdgfrb Mbp Plp1 Cnp Aqp4 Aldoc Gfap S100b C1qa C1qb Cx3cr1 Tmem119 Itgam Pecam1 Kdr Col1a1 Dcn Reln Lhx1 Lhx9 Foxp2 Foxp1 Pcp4 Rorb Rprm Bmp3 Kcnc1 Gabra4'.split()
for g in markers:
 ix=np.flatnonzero(genes==g)
 if len(ix):
  i=ix[0]; vals=X[:,i].toarray().ravel(); tot=vals.sum(); n=(vals>0).sum();
  print(f'{g:10s} sum={tot:8d} cells={n:5d} medpos={np.median(vals[vals>0]) if n else 0:6.1f} max={vals.max():5d}')
# high quality marker expression cells > 2000
mask=np.asarray(X.sum(axis=1)).ravel()>2000
print('high cells',mask.sum())
for g in markers:
 ix=np.flatnonzero(genes==g)
 if len(ix):
  vals=X[mask,ix[0]].toarray().ravel(); n=(vals>0).sum();
  if n: print(f'{g:10s} {n:5d}/{mask.sum()} sum {int(vals.sum())}')
PY"
