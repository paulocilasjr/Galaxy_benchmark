/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
 cats=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/gene_symbols/categories'][:]],dtype=object)
 genes=cats[f['var/gene_symbols/codes'][:]]
# overall
s=np.asarray(X.sum(0)).ravel(); pct=np.asarray((X>0).sum(0)).ravel()
for metric,name in [(s,'sum'),(pct,'cells')]:
 ix=np.argsort(metric)[::-1][:150]
 print('\\nTOP',name)
 print(' | '.join(f'{genes[i]}:{int(metric[i])}' for i in ix))
# highly expressed among high-count cells > 1000
for threshold in [500,1000,2000,5000,10000]:
 mask=np.asarray(X.sum(1)).ravel()>=threshold
 ss=np.asarray(X[mask].sum(0)).ravel(); pp=np.asarray((X[mask]>0).sum(0)).ravel()
 ix=np.argsort(ss)[::-1][:50]
 print('\\nTHR',threshold,'n',mask.sum(),'top sum',' | '.join(f'{genes[i]}:{int(ss[i])}' for i in ix))
PY"
