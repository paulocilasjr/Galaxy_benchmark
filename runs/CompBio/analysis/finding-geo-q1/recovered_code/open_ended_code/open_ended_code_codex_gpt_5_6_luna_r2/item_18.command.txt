/bin/bash -lc "python - <<'PY'
import h5py,numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 x=csr_matrix((f['X/data'][...],f['X/indices'][...],f['X/indptr'][...]),shape=f['X'].attrs['shape'])
 print(x.shape,x.nnz)
 counts=np.asarray(x.sum(axis=1)).ravel(); genes=np.diff(x.indptr)
 print('counts min max',counts.min(),counts.max(),'quantiles',np.quantile(counts,[0,.001,.005,.01,.02,.05,.1,.25,.5,.75,.9,.95,.99,1]))
 print('genes min max',genes.min(),genes.max(),'quantiles',np.quantile(genes,[0,.001,.005,.01,.02,.05,.1,.25,.5,.75,.9,.95,.99,1]))
 print('unique count low')
 for t in [0,1,10,50,100,200,500,1000,1500,2000,2500,3000,5000,10000]: print(t,(counts<=t).sum())
 print('first 50 count/genes')
 print(list(zip(counts[:50],genes[:50])))
PY"
