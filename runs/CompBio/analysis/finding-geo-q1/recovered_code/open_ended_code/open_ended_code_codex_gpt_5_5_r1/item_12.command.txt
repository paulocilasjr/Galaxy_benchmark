/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
fn='finding.geo.q1.h5ad'
markers=['Snap25','Syt1','Rbfox3','Tubb3','Sox2','Mki67','Mbp','Plp1','Aldoc','Aqp4','Pecam1','C1qa','Lyz2','Col1a1','Krt8','Epcam','Alb','Ins1','Pdx1','Nkg7','Cd3d','Cd79a','Hbb-bs','Hba-a1','Pou5f1','Nanog','Pax6','Dcx','Gad1','Slc17a7','Thy1','Foxj1','Pdgfra','Nes','Fabp7','Hes1','Neurod1','Neurog2','Top2a','Pcna','Stmn2','Sox10','Dlx1','Dlx2','Tbr1','Eomes','Bcl11b','Satb2','Reln','Calb2','Mog','Olig1','Olig2','Dcn','Rgs5','Cldn5','Kdr']
with h5py.File(fn,'r') as f:
    cats=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/gene_symbols/categories'][:]]
    codes=f['var/gene_symbols/codes'][:]
    syms=np.array([cats[c] if c>=0 else '' for c in codes])
    data=f['X/data'][:]; indices=f['X/indices'][:]; indptr=f['X/indptr'][:]
    col_sum=np.zeros(len(syms), dtype=np.int64); col_nnz=np.zeros(len(syms), dtype=np.int64)
    np.add.at(col_sum, indices, data); np.add.at(col_nnz, indices, 1)
    for m in markers:
        wh=np.where(syms==m)[0]
        if len(wh):
            i=wh[0]
            print(f'{m}\\tidx={i}\\tsum={col_sum[i]}\\tnnz={col_nnz[i]}\\tfrac={col_nnz[i]/len(indptr[:-1]):.3f}')
        else:
            print(f'{m}\\tmissing')
    # top non-ribo/mito by cells expressed and counts
    mask=np.array([not (s.startswith('Rpl') or s.startswith('Rps') or s.startswith('mt-') or s in ('Malat1','Actb','Actg1','Tmsb4x','Tmsb10','Eef1a1','Ptma')) for s in syms])
    print('\\nTop count nonhouse')
    for i in np.argsort(np.where(mask,col_sum,-1))[-80:][::-1]: print(syms[i], int(col_sum[i]), int(col_nnz[i]))
    print('\\nTop nnz nonhouse')
    for i in np.argsort(np.where(mask,col_nnz,-1))[-80:][::-1]: print(syms[i], int(col_nnz[i]), int(col_sum[i]))
PY"
