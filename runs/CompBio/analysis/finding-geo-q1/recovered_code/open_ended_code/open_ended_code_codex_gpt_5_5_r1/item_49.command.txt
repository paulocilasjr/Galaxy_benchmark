/bin/bash -lc "python - <<'PY'
import urllib.request, gzip, io, h5py
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE125nnn/GSE125620/suppl/GSE125620_barcodes.tsv.gz'
data=urllib.request.urlopen(url, timeout=30).read()
bar=[line.decode().strip() for line in gzip.GzipFile(fileobj=io.BytesIO(data))]
with h5py.File('finding.geo.q1.h5ad','r') as f:
    obs=[x.decode() for x in f['obs/_index'][:]]
print('remote n',len(bar),'local n',len(obs))
print('remote first20',bar[:20])
print('local first20',obs[:20])
print('remote last20',bar[-20:])
print('local last20',obs[-20:])
print('equal?',bar==obs)
# compare set/intersection if not equal
print('intersection',len(set(bar)&set(obs)))
PY"
