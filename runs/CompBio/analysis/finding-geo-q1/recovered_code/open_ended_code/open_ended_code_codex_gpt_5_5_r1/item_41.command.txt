import urllib.request
for acc in ['GSE206869','GSE206870','GSE208199']:

    txt=urllib.request.urlopen(url, timeout=30).read().decode('utf-8','replace')
    for line in txt.splitlines():
        if any(key in line for key in ['"'^SERIES'"','"'!Series_title'"','"'!Series_status'"','"'!Series_summary'"','"'!Series_overall_design'"','"'!Series_sample_id'"','"'!Series_supplementary_file'"','"'!Series_platform_organism'"','"'!Series_sample_organism'"','"'!Series_relation'"','"'!Series_type'"']):
            print(line)
PY"
