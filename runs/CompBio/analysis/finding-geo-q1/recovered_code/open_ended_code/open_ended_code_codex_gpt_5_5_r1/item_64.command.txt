/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
with h5py.File('finding.geo.q1.h5ad','r') as lf, h5py.File('GSM3578982_raw_gene_bc_matrices_h5.h5','r') as rf:
    local_obs=lf['obs/_index'][:]
    local_var=lf['var/_index'][:]
    rg=rf['output_genome2']
    raw_barcodes=rg['barcodes'][:]
    raw_genes=rg['genes'][:]
    raw_names=rg['gene_names'][:]
    indptr=rg['indptr'][:]; data=rg['data'][:]
    cs=np.concatenate(([0], np.cumsum(data, dtype=np.int64)))
    sums=cs[indptr[1:]]-cs[indptr[:-1]]
    for threshold in [200,250,273,274,300,500,1000]:
        arr=raw_barcodes[sums>=threshold]
        print('>=',threshold,'n',len(arr),'equals',np.array_equal(arr, local_obs))
    # since raw barcodes sorted, local sorted, find positions via searchsorted
    pos=np.searchsorted(raw_barcodes, local_obs)
    print('all local barcodes in raw', bool(np.all(raw_barcodes[pos]==local_obs)))
    print('local raw sum min max', int(sums[pos].min()), int(sums[pos].max()))
    mask=np.ones(len(raw_barcodes), dtype=bool); mask[pos]=False
    print('nonlocal raw sum max', int(sums[mask].max()))
    raw_gene_set=set(raw_genes.tolist()); local_gene_set=set(local_var.tolist())
    miss=[(g.decode(),n.decode()) for g,n in zip(raw_genes,raw_names) if g not in local_gene_set]
    print('raw genes',len(raw_genes),'local genes',len(local_var),'missing in local',miss[:10])
PY"
