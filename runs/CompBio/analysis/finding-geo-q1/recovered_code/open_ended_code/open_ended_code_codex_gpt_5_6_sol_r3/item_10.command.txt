/bin/bash -lc "python - <<'PY'
import h5py,numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
 sy=f['var/gene_symbols/categories'][:].astype(str)[f['var/gene_symbols/codes'][:]]
s=np.asarray(X.sum(1)).ravel()
for th in [500,1000,2000,5000]:
 Y=X[s>th]; z=np.asarray(Y.sum(0)).ravel(); nc=np.asarray((Y>0).sum(0)).ravel()
 print('\\n## threshold',th,'cells',Y.shape[0])
 skip=('Rpl','Rps','mt-','Gm','Malat','Tms','Eef','Hsp','Hist','Tubb','Tuba','Act','Ubb','Fau','H3f','Ppia','Ptma')
 ct=0
 for j in np.argsort(z)[::-1]:
  if sy[j].startswith(skip): continue
  print(f'{sy[j]}:{z[j]}({nc[j]})',end='  ');ct+=1
  if ct%8==0: print()
  if ct>=80: break
 print()
PY"
