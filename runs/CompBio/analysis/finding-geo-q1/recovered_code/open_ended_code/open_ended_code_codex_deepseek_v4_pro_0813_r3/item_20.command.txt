/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
f=h5py.File('finding.geo.q1.h5ad','r')
var_idx=f['var/_index'][:].astype(str); cats=f['var/gene_symbols/categories'][:].astype(str); codes=f['var/gene_symbols/codes'][:]
sym=np.where(codes>=0,cats[codes],'NA'); sym2idx={}
for i,s in enumerate(sym):
    if s not in sym2idx: sym2idx[s]=i
idx=f['X/indices'][:]; data=f['X/data'][:]
gc=np.bincount(idx,minlength=len(sym)); gs=np.bincount(idx,weights=data.astype(float),minlength=len(sym))
# all genes starting Hox or with region markers
for m in sorted(sym2idx):
    if m.startswith('Hox') or m in ['Onecut2','Onecut1','Isl1','Isl2','Pou4f1','Lhx1','Lhx2','Lhx3','Lhx4','Phox2a','Phox2b','Tlx3','Gata2','Gata3','Olig3','Nkx6-2','Dbx1','Dbx2','Evx1','Evx2','Foxd3','Sox10']:
        i=sym2idx[m]
        if gc[i]>0:
            print(f'{m:10s}\\t{gc[i]:5d}\\t{int(gs[i]):7d}')
PY"
