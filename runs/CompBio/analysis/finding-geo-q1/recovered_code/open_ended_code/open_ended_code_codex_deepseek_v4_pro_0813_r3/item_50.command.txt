/bin/bash -lc "python - <<'PY'
import json
S=json.load(open('spinal_summaries.json'))
for uid,x in S.items():
    title=x.get('title','')
    summary=x.get('summary') or ''
    supp=(x.get('suppfile') or '').upper()
    type_=(x.get('gdstype') or '')
    low=(title+' '+summary).lower()
    if 'mtx' not in supp and 'tsv' not in supp:
        continue
    if 'single' not in low and 'scrna' not in low and 'single-cell' not in low:
        continue
    # exclude adult/disease/immune/injury/MS/ALS/microglia maybe but include develop/embryo/neural
    if not any(w in low for w in ['develop','embry','e9','e10','e11','e12','e13','neural tube','progenitor','temporal','spatial','spinal cord']):
        continue
    if any(w in low for w in ['injury','multiple sclerosis','als','amyotrophic','microglia','t cell','immune','astrocyte','oligodendrocyte lineage','opc','pdgfra','pain','axon regeneration','corticospinal tract','adult']):
        # some could still be spinal dev but skip disease
        continue
    print(uid, x.get('accession'), 'n=',x.get('n_samples'), 'type=',type_, 'supp=',x.get('suppfile'), 'pdat=',x.get('pdat'))
    print('  ',title)
    print('  ',summary[:300].replace('\\n',' '))
PY"
