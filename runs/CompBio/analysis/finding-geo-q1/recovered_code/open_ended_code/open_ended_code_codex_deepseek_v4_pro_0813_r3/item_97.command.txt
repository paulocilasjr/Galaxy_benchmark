/bin/bash -lc "python - <<'PY'
import json
S=json.load(open('spinal_summaries.json'))
for uid,x in S.items():
    supp=(x.get('suppfile') or '').upper()
    if 'H5' in supp or 'MTX' in supp or 'TSV' in supp:
        low=(x.get('title','')+' '+(x.get('summary') or '')).lower()
        if any(w in low for w in ['develop','embry','spinal cord','neural tube','motor neuron','progenitor','temporal','spatial']):
            print(uid,x.get('accession'),'n',x.get('n_samples'),'supp',x.get('suppfile'),'pdat',x.get('pdat'))
            print(' ',x.get('title'))
PY"
