/bin/bash -lc "python - <<'PY'
import requests, json, time
ids=[]
for retstart in [0,100,200,300]:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':'spinal cord AND single cell AND Mus musculus[Organism] AND gse[Entry Type]','retmax':100,'retstart':retstart,'retmode':'json'})
 ids += r.json()['esearchresult']['idlist']
 time.sleep(0.4)
print('got',len(ids),len(set(ids)))
open('spinal_ids.txt','w').write('\\n'.join(ids))
# fetch summaries in chunks of 200? esummary max maybe 200
summaries={}
for i in range(0,len(ids),200):
 chunk=ids[i:i+200]
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':','.join(chunk),'retmode':'json'})
 data=r.json()['result']
 for uid in data['uids']:
  summaries[uid]=data[uid]
 print('chunk',i,'got',len(data.get('uids',[])))
 time.sleep(0.5)
open('spinal_summaries.json','w').write(json.dumps(summaries,indent=2))
print('summaries',len(summaries))
PY"
