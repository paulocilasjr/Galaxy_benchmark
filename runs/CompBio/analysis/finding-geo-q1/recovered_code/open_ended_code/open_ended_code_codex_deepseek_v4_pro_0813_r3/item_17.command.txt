/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
f=h5py.File('finding.geo.q1.h5ad','r')
var_idx=f['var/_index'][:].astype(str)
cats=f['var/gene_symbols/categories'][:].astype(str)
codes=f['var/gene_symbols/codes'][:]
sym=np.where(codes>=0,cats[codes],'NA')
indices=f['X/indices'][:]; data=f['X/data'][:]
sym2idx={}
for i,s in enumerate(sym):
    if s not in sym2idx: sym2idx[s]=i
gc=np.bincount(indices,minlength=len(var_idx))
gs=np.bincount(indices,weights=data.astype(float),minlength=len(var_idx))
markers=['Sox2','Hes5','Pax6','Eomes','Tbr2','Neurog1','Neurog2','Ascl1','Dlx1','Dlx2','Dlx5','Lhx6','Nkx2-1','Emx1','Emx2','Foxg1','Otx1','Otx2','En1','En2','Lmx1a','Lmx1b','Wnt1','Shh','Gli1','Pax2','Lhx1','Hoxb4','Hoxa2','Gbx2','Fgf8','Zic1','Zic2','Atoh1','Ptf1a','Olig2','Nkx6-1','Nkx2-2','Irx3','Lhx2','Nr2e1','Nr2f1','Meis2','Foxp2','Rorb','Satb2','Cux1','Cux2','Bcl11b','Tbr1','Fezf2','Lmo4','Neurod2','Neurod6','Bhlhe22','Mef2c','Rbfox3','Dcx','Mki67','Pcna','Ccnd2','Cdk1','Top2a','Sox10','Pdgfra','Olig1','Mbp','Plp1','Mag','Mog','Cldn5','Pecam1','Aif1','Cx3cr1','Ttr','Emcn','Rgs5','Pdgfrb','Acta2','Lum','Dcn','Col1a1','Col1a2','Slc17a6','Slc17a7']
for m in markers:
    idx=sym2idx.get(m,-1)
    if idx>=0:
        print(f'{m:10s}\\t{gc[idx]:5d}\\t{int(gs[idx]):7d}')
    else:
        print(f'{m:10s}\\tMISSING')
PY"
