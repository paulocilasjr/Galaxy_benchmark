/bin/bash -lc "python - <<'PY'
import json
S=json.load(open('spinal_summaries.json'))
for uid,x in S.items():
    low=(x.get('title','')+' '+(x.get('summary') or '')).lower()
    supp=(x.get('suppfile') or '').upper()
    if not ('MTX' in supp or 'TSV' in supp): continue
    if any(w in low for w in ['develop','embryo','embryonic','e9.5','e10.5','e11.5','e12.5','e13.5','neural tube','progenitor','motor neuron']):
        print(uid,x.get('accession'),x.get('n_samples'),x.get('gdstype'),x.get('suppfile'),x.get('pdat'))
        print(' ',x.get('title'))
        print(' ',(x.get('summary') or '')[:250].replace('\\n',' '))
PY"
