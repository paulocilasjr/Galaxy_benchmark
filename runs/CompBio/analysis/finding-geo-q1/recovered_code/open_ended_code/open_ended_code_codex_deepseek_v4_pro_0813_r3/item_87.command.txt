/bin/bash -lc "python - <<'PY'
import json,re
L=json.load(open('candidate_filelists.json'))
matches=[]
for item in L:
    acc=item['acc']; text=item.get('text') or ''
    if not text: continue
    entries=[]
    for line in text.splitlines():
        if line.startswith('#') or not line.strip(): continue
        parts=line.split('\\t')
        if len(parts)>=4:
            entries.append({'name':parts[1], 'size':int(parts[3]), 'type':parts[4] if len(parts)>4 else ''})
    feat=[e for e in entries if re.search(r'(features|genes)',e['name'],re.I) and not re.search(r'(atac|peaks)',e['name'],re.I)]
    bar=[e for e in entries if re.search(r'barcodes',e['name'],re.I)]
    mtx=[e for e in entries if re.search(r'(matrix|counts).*mtx|\\.mtx',e['name'],re.I)]
    # filter any feature size in broad 180k-260k and barcode >60k
    good_feat=[e for e in feat if 150000 <= e['size'] <= 270000]
    good_bar=[e for e in bar if e['size'] >= 60000]
    if good_feat or (feat and any(150000<=e['size']<=270000 for e in feat)):
        matches.append((acc,item['status'],feat,bar,mtx,good_feat,good_bar))
print('matches with feature size range',len(matches))
for m in matches:
    acc,status,feat,bar,mtx,gf,gb=m
    print('\\n',acc,'status',status)
    print(' feats:',[(e['name'],e['size']) for e in feat[:10]])
    print(' goodfeat:',[(e['name'],e['size']) for e in gf])
    print(' bars:',[(e['name'],e['size']) for e in bar[:10]])
    print(' mtx:',[(e['name'],e['size']) for e in mtx[:5]])
PY"
