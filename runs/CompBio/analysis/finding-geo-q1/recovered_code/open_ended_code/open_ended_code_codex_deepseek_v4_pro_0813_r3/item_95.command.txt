/bin/bash -lc "python - <<'PY'
import requests,json,re
uids=['200160136','200190593','200245105','200153162','200247594','200169519']
for uid in uids:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':uid,'retmode':'json'})
 if uid not in r.json().get('result',{}): continue
 x=r.json()['result'][uid]; acc=x['accession']; prefix=acc[:6]+'nnn'
 print('\\n===',uid,acc,'n',x.get('n_samples'),'supp',x.get('suppfile'),'pdat',x.get('pdat'))
 print(x.get('title'))
 url=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{prefix}/{acc}/suppl/filelist.txt'
 rl=requests.get(url)
 print('filelist',rl.status_code)
 if rl.status_code==200: print(rl.text[:3000])
PY"
