/bin/bash -lc "python3 - <<'PY'
import requests, urllib.parse, time
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi'
for term in ['AAACCTGAGCCCAATT-1[All Fields]', 'AAACCTGAGCCCAATT[All Fields]', 'ENSMUSG00000051951[All Fields]']:
    url=base+'?'+urllib.parse.urlencode({'db':'gds','term':term,'retmax':20,'retmode':'json'})
    r=requests.get(url,timeout=30)
    print(term, r.status_code, r.text[:500])
    time.sleep(0.5)
PY"
