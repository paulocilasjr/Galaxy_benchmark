/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    sym=f['var/gene_symbols/categories'][:][f['var/gene_symbols/codes'][:]].astype(str)
    X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]), shape=(f['X'].attrs['shape'][0], f['X'].attrs['shape'][1]))
    sums=np.asarray(X.sum(0)).ravel(); nc=np.asarray((X>0).sum(0)).ravel()
    order=np.argsort(-sums)
    count=0
    for i in order:
        s=sym[i]
        if s.startswith(('mt-','Rpl','Rps','Gm','RP')) or s in ('Malat1','Tmsb4x','Tmsb10','Ptma','Eef1a1','Actb','Actg1','H3f3b','H3f3a','Ubb','Ppia','Fau'):
            continue
        print(s,int(sums[i]),int(nc[i]))
        count+=1
        if count>=120: break
PY"
