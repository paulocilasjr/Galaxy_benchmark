/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Sox2 Pax6 Eomes Tbr1 Bcl11b Cux1 Cux2 Satb2 Neurod2 Neurod6 Slc17a7 Gad1 Gad2 Gfap Aqp4 Olig1 Olig2 Pdgfra Mbp Plp1 P2ry12 Cx3cr1 C1qa Cldn5 Pecam1 Kdr Rbfox3 Snap25 Dcx Top2a Mki67 Mcm6 Pcna Hes1 Notch1 Ascl1 Neurog2 Tubb3 Stmn2 Map2 Reln Calb2 Prox1 Gjb6 Aldh1l1'.split()
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]
    codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
    sums=np.asarray(X.sum(axis=0)).ravel(); cells=np.asarray((X>0).sum(axis=0)).ravel()
    for m in markers:
        idx=np.where(genes==m)[0]
        if len(idx):
            i=idx[0]; print(f'{m}\\t{sums[i]:.0f}\\t{cells[i]}')
PY"
