/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, xml.etree.ElementTree as ET, time
terms=['cranial placode 10x Mus musculus', 'placode single cell RNA-seq Mus musculus GEO', 'preplacodal ectoderm single cell GEO mouse', 'neurogenic placode single cell RNA-seq mouse', 'neural plate border single-cell RNA-seq Mus musculus GEO', 'Foxi3-CreER single cell RNA-seq GEO', 'Foxi3 single cell RNA-seq mouse GEO']
for t in terms:
    time.sleep(0.5)
    print('\\nTERM', t)
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gds&retmax=15&term='+urllib.parse.quote(t)
    try:
        root=ET.fromstring(urllib.request.urlopen(url).read())
    except Exception as e:
        print('ERR search',e); continue
    ids=[x.text for x in root.findall('.//Id')]
    print('count', root.findtext('Count'), ids)
    if not ids: continue
    time.sleep(0.5)
    url2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=gds&id='+','.join(ids)
    try:
        r=ET.fromstring(urllib.request.urlopen(url2).read())
    except Exception as e:
        print('ERR sum',e); continue
    for doc in r.findall('DocSum'):
        def item(name):
            x=doc.find(f\"Item[@Name='{name}']\")
            return x.text if x is not None and x.text else ''
        print(item('Accession'), item('entryType'), 'GSE'+item('GSE'), item('taxon'), item('title')[:120])
PY"
