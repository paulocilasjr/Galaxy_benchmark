/bin/bash -lc "Rscript - <<'RS'
f <- gzcon(gzfile('GSM3890935_Hypothalamus_traject_E17_srt_annotated_wo_blood.rds.gz','rb'))
obj <- readRDS(f); close(f)
cat('isS4', isS4(obj), '\\n')
print(class(obj))
print(names(attributes(obj)))
print(attributes(obj)"'$names)
# Try direct slots via @ known from Seurat v3: assays, meta.data, active.assay
for (s in c('"'assays','meta.data','active.assay','reductions','project.name')) {
  cat('SLOT',s,'\\n')
  y <- try(slot(obj,s), silent=TRUE)
  if (inherits(y,'try-error')) print(y) else { print(class(y)); print(names(y)[1:min(10,length(names(y)))]) ; if (is.data.frame(y)) print(dim(y)) }
}
RS"
