/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
p='finding.geo.q1.h5ad'
exclude=lambda g: g.startswith(('mt-','Rpl','Rps')) or g in {'Malat1','Tuba1a','Tmsb4x','Eef1a1','Ptma','Tmsb10','Actg1','Actb','H3f3b','Tubb5','Gm10116','Gm8730','H3f3a','Ubb','Ppia','Tpt1','Hsp90ab1','Npm1','Gapdh','Eef2','Eef1g','Fau','Cfl1','Hspa8','Eif1','Eif4a1','Sumo2','Hnrnpa1'}
with h5py.File(p,'r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]; codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
    sums=np.asarray(X.sum(axis=0)).ravel(); nz=np.asarray((X>0).sum(axis=0)).ravel()
    order=[i for i in np.argsort(-sums) if not exclude(genes[i])]
    for i in order[:200]: print(f'{genes[i]}\\t{int(sums[i])}\\t{int(nz[i])}')
PY"
