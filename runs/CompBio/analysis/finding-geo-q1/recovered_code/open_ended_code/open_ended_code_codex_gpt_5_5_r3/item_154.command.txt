/bin/bash -lc "Rscript - <<'RS' > gse132730_e17_meta.txt
f <- gzcon(gzfile('GSM3890935_Hypothalamus_traject_E17_srt_annotated_wo_blood.rds.gz','rb'))
obj <- readRDS(f); close(f)
md <- slot(obj,'meta.data')
cat('dim', dim(md), '\\n')
cat('first', paste(head(rownames(md),20), collapse='\\n'), '\\n')
cat('last', paste(tail(rownames(md),20), collapse='\\n'), '\\n')
cat('nCount summary', paste(summary(md"'$nCount_RNA), collapse='"' '), '\\n')
assays <- slot(obj,'assays')
rna <- assays"'$RNA
cat('"'rna class', class(rna), '\\n')
print(names(attributes(rna)))
for (s in c('counts','data','scale.data','meta.features','var.features','key')) {
  cat('slot',s,'\\n')
  y<-try(slot(rna,s),silent=TRUE)
  if (inherits(y,'try-error')) {print(y)} else {print(class(y)); if (inherits(y,'Matrix')||is.matrix(y)) print(dim(y)); if(is.data.frame(y)) print(dim(y)); if(is.character(y)) print(head(y));}
}
RS
cat gse132730_e17_meta.txt"
