from urllib.request import urlopen
from urllib.parse import urlencode
ids='200342074,200330876,200320340'

data=json.load(urlopen(url))
for uid in data['result']['uids']:
    d=data['result'][uid]
    print(uid,d.keys())
    print(d.get('accession'),d.get('title'),d.get('gdsType'),d.get('taxon'))
PY"
