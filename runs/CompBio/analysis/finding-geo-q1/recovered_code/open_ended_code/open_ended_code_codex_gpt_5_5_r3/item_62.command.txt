/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Atoh1 Atoh7 Barhl1 Barhl2 Pax2 Ptf1a Zic1 Zic2 Zic3 Zic4 Math1 Neurod1 Neurod2 Neurod6 Meis2 Lmx1a Lmx1b En1 En2 Wnt1 Fgf8 Otx2 Gbx2 Pax5 Pax8 Eya1 Eya2 Six1 Shox2 Myt1 Casz1 Sall3 Maf Epha5 Gata3 Isl1 Isl2 Pou4f1 Pou4f2 Phox2b Tlx3'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
 X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
 cats=f['var/gene_symbols/categories'][:]; codes=f['var/gene_symbols/codes'][:]
 genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
 sums=np.asarray(X.sum(axis=0)).ravel(); cells=np.asarray((X>0).sum(axis=0)).ravel()
 for m in markers:
  idx=np.where(genes==m)[0]
  if len(idx): print(f'{m}\\t{int(sums[idx[0]])}\\t{int(cells[idx[0]])}')
PY"
