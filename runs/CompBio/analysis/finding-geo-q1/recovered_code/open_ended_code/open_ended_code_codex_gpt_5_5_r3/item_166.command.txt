/bin/bash -lc "python3 - <<'PY'
import csv, urllib.request, concurrent.futures, time, re
rows=list(csv.DictReader(open('geo_candidates_neural_10x_all.tsv'), delimiter='\\t'))

def filelist_url(acc):
    n=int(acc[3:]); prefix=f'GSE{n//1000:03d}nnn'
    return f'https://ftp.ncbi.nlm.nih.gov/geo/series/{prefix}/{acc}/suppl/filelist.txt'

def fetch(r):
    acc=r['accession']; url=filelist_url(acc)
    try:
        with urllib.request.urlopen(url, timeout=8) as fh:
            txt=fh.read(200000).decode('utf-8','ignore')
    except Exception as e:
        return None
    hits=[]
    for line in txt.splitlines():
        if line.startswith('#') or not line.strip(): continue
        if re.search(r'barcodes.*\\.tsv\\.gz|genes.*\\.tsv\\.gz|features.*\\.tsv\\.gz|matrix.*\\.mtx\\.gz|filtered_feature_bc_matrix\\.h5|raw_feature_bc_matrix\\.h5', line, re.I):
            hits.append(line)
    if hits:
        return (acc,r['title'],r['taxon'],r['n_samples'],r['pdat'],hits[:50],len(hits),txt)
    return None

out=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=12) as ex:
    futs=[ex.submit(fetch,r) for r in rows]
    for i,f in enumerate(concurrent.futures.as_completed(futs),1):
        res=f.result()
        if res: out.append(res)
        if i%500==0: print('done',i,'hits',len(out))
print('total hits',len(out))
with open('geo_filelist_barcode_hits.tsv','w') as f:
    for acc,title,taxon,ns,pdat,hits,nh,txt in out:
        f.write('\\t'.join([acc,taxon,str(ns),str(pdat),str(nh),title.replace('\\t',' ')])+'\\n')
        for h in hits[:20]: f.write('\\t'+h+'\\n')
for rec in out[:30]:
    print('\\n',rec[0],rec[3],rec[4],rec[1][:120]); print('\\n'.join(rec[5][:8]))
PY"
