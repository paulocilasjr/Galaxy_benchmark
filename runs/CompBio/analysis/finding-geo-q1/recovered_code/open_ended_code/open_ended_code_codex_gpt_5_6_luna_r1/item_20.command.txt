/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from collections import Counter
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
 x=f['X']; data=x['data'][:]; ind=x['indices'][:]; ptr=x['indptr'][:]
 genes=f['/var/_index'][:].astype(str)
 syms=f['/var/gene_symbols/categories'][:].astype(str)
 codes=f['/var/gene_symbols/codes'][:]
 symbols=syms[codes]
 totals=np.add.reduceat(data,ptr[:-1]) # ptr last? all nonempty rows likely
 nnz=np.diff(ptr)
 print('cells',len(nnz),'genes',len(genes),'nnz',len(data))
 for a in [totals,nnz]: print('stats min median mean p1 p5 p25 p75 p95 p99 max',*[float(np.percentile(a,q)) for q in [0,1,5,25,50,75,95,99,100]])
 print('zeros',sum(totals==0), 'total range')
 gs=np.bincount(ind,weights=data,minlength=len(genes))
 gn=np.bincount(ind,minlength=len(genes))
 for typ,arr in [('counts',gs),('cells',gn)]:
  ii=np.argsort(arr)[::-1][:50]
  print('\\nTOP',typ)
  for i in ii[:50]: print(i, genes[i],symbols[i],int(arr[i]))
 # first 10 rows details top genes
 for r in range(5):
  ii=ind[ptr[r]:ptr[r+1]]; dd=data[ptr[r]:ptr[r+1]]; oo=np.argsort(dd)[::-1][:15]
  print('cell',r,'total',totals[r],'nnz',nnz[r],[(symbols[ii[j]],int(dd[j])) for j in oo])
PY"
