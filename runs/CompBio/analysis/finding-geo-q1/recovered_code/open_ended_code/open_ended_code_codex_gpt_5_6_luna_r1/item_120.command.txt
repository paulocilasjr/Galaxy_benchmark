/bin/bash -lc "python - <<'PY'
import urllib.request
for g in ['GSE104156','GSE104157','GSE104158','GSE104159','GSE104276','GSE104323','GSE106678','GSE108761','GSE109796','GSE110823','GSE111527','GSE114000','GSE115746','GSE116470','GSE117891','GSE118020','GSE118257','GSE118403','GSE118918','GSE118948','GSE118953']:
 print('\\n'+g)
 for path in ['suppl/filelist.txt','soft/'+g+'_family.soft.gz']:
  u=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{g[:-3]}nnn/{g}/{path}'
  try:
   x=urllib.request.urlopen(u,timeout=20).read()
   if path.startswith('soft'):
    import gzip; x=gzip.decompress(x)
    lines=[l.decode(errors='replace').strip() for l in x.splitlines() if l.startswith(b'"'!Series_title'"') or l.startswith(b'"'!Series_relation'"')]
    print('\\n'.join(lines[:4]))
   else:
    print('\\n'.join(x.decode(errors='replace').splitlines()[:15]))
  except Exception as e: print('ERR',path,str(e)[:80])
PY"
