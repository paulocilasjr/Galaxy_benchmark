/bin/bash -lc "python - <<'PY'
import urllib.request, re
for acc in ['GSE102934','GSE118953']:
 txt=urllib.request.urlopen(f'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc={acc}',timeout=30).read().decode('utf-8','replace')
 print('\\n',acc)
 for term in ['SuperSeries','SubSeries','Series','SubSeries of','SuperSeries of','Status']:
  inds=[m.start() for m in re.finditer(term,txt,re.I)]
  print(term, len(inds), inds[:10])
  for i in inds[:3]:
   print(txt[max(0,i-200):i+500].replace('\\n',' ')[:700])
PY"
