/bin/bash -lc "python - <<'PY'
import urllib.request, json, urllib.parse
for acc in ['GSE118953','GSE102934','GSE275643']:
 term=acc+'[ACCN]'
 url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':'gds','term':term,'retmode':'json'})
 with urllib.request.urlopen(url) as r: d=json.load(r)
 uid=d['esearchresult']['idlist']
 print(acc, uid)
 if uid:
  u=uid[0]
  url2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urllib.parse.urlencode({'db':'gds','id':u,'retmode':'json'})
  with urllib.request.urlopen(url2) as r: d=json.load(r)
  rec=d['result'][u]
  print(rec.keys())
  print('ftp',rec.get('ftplink'))
  print('suppfile',rec.get('suppfile'))
  print('samples',rec.get('samples'))
PY"
