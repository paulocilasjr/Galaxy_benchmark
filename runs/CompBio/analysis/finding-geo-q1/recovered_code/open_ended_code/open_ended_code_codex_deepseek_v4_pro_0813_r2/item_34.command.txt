/bin/bash -lc "python - <<'PY'
import urllib.request, urllib.parse, json
for acc in ['GSE102934','GSE118953','GSE125788','GSE116140','GSE89798','GSE275643']:
 term=acc+'[ACCN]'
 url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':'gds','term':term,'retmode':'json'})
 try:
  with urllib.request.urlopen(url,timeout=30) as r: d=json.load(r)
  uid=d['esearchresult']['idlist']
  if uid:
   u=uid[0]
   url2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urllib.parse.urlencode({'db':'gds','id':u,'retmode':'json'})
   with urllib.request.urlopen(url2,timeout=30) as r: rec=json.load(r)['result'][u]
   print('\\n',acc,'UID',u)
   for key in ['title','entrytype','gds','gse','n_samples','relations','extrelations','pubmedids','seriestitle']:
    print(key,rec.get(key))
 except Exception as e: print(acc,'ERR',e)
PY"
