import h5py, os
p='finding.geo.q1.h5ad'
print(os.path.getsize(p))
with h5py.File(p,'r') as f:
    print(list(f.keys()))
    for k in f:
        print(k, f[k], getattr(f[k], 'shape', None))
    if 'X' in f:
        X=f['X']; print('X keys', list(X.keys()), X.attrs)
    for k in ['obs','var']:
        if k in f:
            print(k, list(f[k].keys()))
            for kk in f[k]:
                print(' ', kk, f[k][kk].shape, f[k][kk].dtype, f[k][kk][:5])
PY"
