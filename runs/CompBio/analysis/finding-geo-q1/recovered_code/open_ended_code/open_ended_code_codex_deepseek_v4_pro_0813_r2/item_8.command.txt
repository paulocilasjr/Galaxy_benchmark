/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np, pandas as pd
a=ad.read_h5ad('finding.geo.q1.h5ad', backed='r')
X=a.X
# gene-level total counts and expressed cell counts
genes=np.asarray(a.var_names)
syms=a.var['gene_symbols'].astype(str).values
n_genes=X.shape[1]
tot=np.zeros(n_genes,dtype=np.int64); nc=np.zeros(n_genes,dtype=np.int64)
for j in range(n_genes):
 col=X[:,j]
 tot[j]=col.sum(); nc[j]=col.nnz
idx=np.argsort(-tot)[:100]
for i in idx:
 print(f'{i:6d} {genes[i]:20s} {syms[i]:20s} total={tot[i]:10d} cells={nc[i]:5d}')
# also marker list
markers=['Pecam1','Cdh5','Kdr','Ptprc','Cd3d','Cd3e','Cd4','Cd8a','Cd19','Ms4a1','Cd68','Adgre1','Itgam','Nkg7','Ncr1','Col1a1','Col3a1','Pdgfra','Epcam','Krt8','Krt18','Acta2','Myh11','Mbp','Plp1','Sox10','Hbb-bs','Hba-a1','Hba-a2','Cst3','Apoe','S100a8','S100a9','Lyz2','Cd74','H2-Aa','H2-Ab1','Nkg7','Gzmb','Foxp3']
s2i={s:i for i,s in enumerate(syms)}
for m in markers:
 i=s2i.get(m)
 if i is not None: print(m, 'tot',tot[i],'cells',nc[i])
PY"
