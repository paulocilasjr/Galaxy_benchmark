/bin/bash -lc "python - <<'PY'
import urllib.request, json, urllib.parse
acc='GSE118953'
# use esummary for gds id and efetch? Let's fetch GEO series page html or use NCBI esearch for gsm supplementary
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urllib.parse.urlencode({'db':'gds','id':acc,'retmode':'json'})
with urllib.request.urlopen(url) as r: d=json.load(r)
uid=list(d['result']['uids'])[0]
rec=d['result'][uid]
print('accession',rec['accession'])
print('ftp',rec.get('ftplink'))
print('samples', rec.get('samples'))
print('suppfile',rec.get('suppfile'))
print('extrelations',rec.get('extrelations'))
print(rec.keys())
PY"
