def urls(field, kind):
    parts=re.split(r'[;\\t\\s]+', field.strip())
    out=[]
    for u in parts:
        if not u: continue
        lu=u.lower()
        if kind=='gene' and (lu.endswith('genes.tsv.gz') or lu.endswith('features.tsv.gz')): out.append(u)
        if kind=='barcode' and (lu.endswith('barcodes.tsv.gz') or lu.endswith('barcode.tsv.gz')): out.append(u)
    return out

def fetch(url, path):
    if path.exists() and path.stat().st_size>0: return True
    u=url.replace('ftp://ftp.ncbi.nlm.nih.gov','https://ftp.ncbi.nlm.nih.gov')
    try:
        with urllib.request.urlopen(u, timeout=30) as r, open(path,'wb') as f:
            while True:
                b=r.read(1<<20)
                if not b: break
                f.write(b)
        return True
    except Exception as e:
        print('FETCH_FAIL',u,e, file=sys.stderr)
        return False

def read_gz_lines(path, max_lines=None):
    out=[]
    try:
        with gzip.open(path,'rt',errors='replace') as f:
            for i,line in enumerate(f):
                if max_lines and i>=max_lines: break
                out.append(line.rstrip('\\n'))
        return out
    except Exception as e:
        return None
