/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    shape=tuple(f['X'].attrs['shape'])
    data=f['X/data'][:]
    indices=f['X/indices'][:]
    indptr=f['X/indptr'][:]
    X=csr_matrix((data,indices,indptr),shape=shape)
    gene_ids=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/_index'][:]]
    cats=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/gene_symbols/categories'][:]]
    codes=f['var/gene_symbols/codes'][:]
    symbols=[cats[c] if c>=0 else '' for c in codes]
    sums=np.asarray(X.sum(axis=0)).ravel()
    nnz=np.asarray((X>0).sum(axis=0)).ravel()
    top=np.argsort(-sums)[:100]
    for i in top:
        print(symbols[i], gene_ids[i], int(sums[i]), int(nnz[i]))
PY"
