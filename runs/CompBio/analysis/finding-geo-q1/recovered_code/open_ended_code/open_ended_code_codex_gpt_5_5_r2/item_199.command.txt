import h5py, numpy as np, scipy.sparse as sp
from pathlib import Path
rawfn='geo_check/gse114176/GSM3578982_raw_gene_bc_matrices_h5.h5'
localfn='finding.geo.q1.h5ad'

    Xloc=sp.csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    obs=[x.decode() for x in f['obs/_index'][:]]
    genes_loc=[x.decode() for x in f['var/_index'][:]]

    g=f['output_genome2']

    names=[x.decode() for x in g['gene_names'][:]]
    bars=[x.decode() for x in g['barcodes'][:]]

    # 10x H5 matrix is csc genes x barcodes

mask=cell_sums>=274
sel_idx=np.nonzero(mask)[0]
