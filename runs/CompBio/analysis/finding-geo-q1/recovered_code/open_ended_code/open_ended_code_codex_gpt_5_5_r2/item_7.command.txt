/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    indptr=f['X/indptr'][:]
    data=f['X/data'][:]
    n_counts=np.add.reduceat(data, indptr[:-1])
    n_genes=np.diff(indptr)
    print('cells', len(n_counts), 'genes', f['X'].attrs['shape'][1], 'nnz', len(data))
    for name, arr in [('n_counts',n_counts),('n_genes',n_genes)]:
        print(name, 'min', int(arr.min()), 'q', np.quantile(arr,[.01,.05,.1,.25,.5,.75,.9,.95,.99]).astype(int).tolist(), 'max', int(arr.max()), 'sum', int(arr.sum()) if name=='n_counts' else '')
PY"
