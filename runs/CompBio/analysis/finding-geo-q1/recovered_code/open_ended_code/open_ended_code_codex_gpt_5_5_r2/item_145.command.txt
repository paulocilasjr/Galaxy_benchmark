/bin/bash -lc "python - <<'PY'
import h5py, scipy.sparse as sp, numpy as np, gzip
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=sp.csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    genes=[x.decode() for x in f['var/_index'][:]]
# candidate gse132355 gene set
with gzip.open('geo_check/gse132355/GSM3860735_E12_genes.tsv.gz','rt') as f:
    cand=[line.split('\\t')[0] for line in f]
extra=[i for i,g in enumerate(genes) if g not in set(cand)]
missing=[g for g in cand if g not in set(genes)]
print('extra count', len(extra), 'missing', len(missing))
print('extra first idx/gene', [(i,genes[i]) for i in extra[:20]])
colsum=np.asarray(X[:,extra].sum(axis=0)).ravel() if extra else np.array([])
print('extra total counts', int(colsum.sum()), 'nonzero extra genes', int((colsum>0).sum()), 'top', sorted([(int(c),genes[extra[i]]) for i,c in enumerate(colsum)], reverse=True)[:20])
PY"
