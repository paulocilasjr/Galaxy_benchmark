/bin/bash -lc "python - <<'PY'
from urllib.parse import urlencode
from urllib.request import urlopen
import json
term='(\"Mus musculus\"[Organism] OR mouse[All Fields]) AND (10x[All Fields] OR \"10X Genomics\"[All Fields]) AND (brain[All Fields] OR cortex[All Fields] OR neural[All Fields] OR neuron[All Fields]) AND \"Expression profiling by high throughput sequencing\"[Filter]'
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urlencode({'db':'gds','term':term,'retmax':200,'retmode':'json'})
print(url)
data=json.load(urlopen(url))
ids=data['esearchresult']['idlist']
print('count', data['esearchresult']['count'], 'returned', len(ids))
# fetch summaries in batches
for i in range(0,len(ids),20):
    u='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urlencode({'db':'gds','id':','.join(ids[i:i+20]),'retmode':'json'})
    js=json.load(urlopen(u))
    for id in ids[i:i+20]:
        s=js['result'][id]
        print(s.get('accession'), '|', s.get('gdsType'), '|', s.get('title')[:120])
PY"
