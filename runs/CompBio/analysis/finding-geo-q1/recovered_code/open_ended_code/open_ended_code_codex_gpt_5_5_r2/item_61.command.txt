/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Omp Gnal Gng8 Cnga2 Cnga4 Adcy3 Lhx2 Ascl1 Neurog1 Gap43 Krt5 Krt14 Foxj1 Sox9 Krt18 Ncam1 Calb2 Calb1 Trpm5 Chat Dlx5 Lhx6 Nkx2-1 Sp8 Gsx2 Isl1 Tac1 Penk Ppp1r1b'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/gene_symbols/categories'][:]]
    syms=np.array([cats[c] if c>=0 else '' for c in f['var/gene_symbols/codes'][:]])
    for m in markers:
        idx=np.where(syms==m)[0]
        if len(idx):
            v=X[:,idx[0]].toarray().ravel()
            print(f'{m}\\t{int(v.sum())}\\t{int((v>0).sum())}')
PY"
