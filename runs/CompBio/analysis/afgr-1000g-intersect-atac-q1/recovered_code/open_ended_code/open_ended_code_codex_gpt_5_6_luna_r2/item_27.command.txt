/bin/bash -lc "python3 - <<'PY'
import json
exp=json.load(open('encode_afgr_experiments.json'))
exps={x['accession']:x for x in exp['@graph'] if x.get('assay_term_name')=='ATAC-seq'}
files=json.load(open('afgr_atac_filtered_bams.json'))
by_dataset={x['dataset'].rstrip('/').rsplit('/',1)[-1]:x for x in files['@graph']}
with open('1kGP.3202_samples.pedigree_info.txt') as f:
 next(f); kgp={line.split()[0] for line in f}
rows=[]
for eid,x in exps.items():
 sample=x['biosample_summary'].split()[-1]
 alias='NA'+sample[2:] if sample.startswith('GM') else sample
 if alias in kgp:
  f=by_dataset[eid]
  rows.append((sample,alias,eid,f['accession'],f['md5sum']))
assert len(rows)==83 and len({r[0] for r in rows})==83 and len({r[4] for r in rows})==83
print('rows',len(rows),'unique md5',len({r[4] for r in rows}))
print('nonoverlap',sorted(set(x['biosample_summary'].split()[-1] for x in exps.values())-{r[0] for r in rows}))
print('\\nDETAILS')
for r in sorted(rows): print('\\t'.join(r))
print('\\nSORTED_MD5')
for md5 in sorted(r[4] for r in rows): print(md5)
PY"
