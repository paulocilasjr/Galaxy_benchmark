/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, concurrent.futures
ex=json.load(open('encode_afgr_experiments.json'))['@graph']
kg={line.split()[0] for line in open('1kGP.3202_samples.pedigree_info.txt') if not line.startswith('sampleID')}
exp_ids={x['@id'] for x in ex if x.get('assay_title')=='ATAC-seq' and x['biosample_ontology']['term_name'] in kg}
files=json.load(open('encode_afgr_files.json'))['@graph']
selected=[x for x in files if x.get('dataset') in exp_ids and x.get('file_format')=='bam' and x.get('output_type')=='alignments' and x.get('assembly')=='GRCh38' and x.get('status')=='released']
assert len(selected)==50, len(selected)

def get(x):
    url='https://www.encodeproject.org'+x['@id']+'?format=json'
    with urllib.request.urlopen(url,timeout=60) as f:
        p=json.load(f)
    return {'sample':x['biosample_ontology']['term_name'],'file':p['accession'],'md5':p['md5sum'],'assembly':p['assembly'],'output_type':p['output_type'],'file_format':p['file_format'],'status':p['status'],'dataset':p['dataset']}
with concurrent.futures.ThreadPoolExecutor(max_workers=12) as pool:
    out=list(pool.map(get,selected))
out.sort(key=lambda x:x['sample'])
assert len(out)==50 and all(x['md5'] and len(x['md5'])==32 for x in out)
for x in out: print('\\t'.join(x.values()))
PY"
