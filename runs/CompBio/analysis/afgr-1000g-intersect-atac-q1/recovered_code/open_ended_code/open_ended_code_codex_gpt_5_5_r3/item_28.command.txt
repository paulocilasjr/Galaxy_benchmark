/bin/bash -lc "python3 - <<'PY'
import json, urllib.request
from urllib.request import Request, urlopen
# all AFGR ATAC names from details/non previous? Read from encode_afgr_search
enc=json.load(open('encode_afgr_search.json'))
afgr=[]
for x in enc['@graph']:
 if x.get('assay_title')=='ATAC-seq':
  name=x.get('biosample_ontology',{}).get('term_name') or x.get('biosample_summary','').split()[-1]
  afgr.append(name)
print('afgr', len(afgr))

def samples(title):
 s=set()
 for start in range(0,4000,500):
  req=Request('https://www.internationalgenome.org/api/beta/sample/_search', data=json.dumps({'from':start,'size':500,'_source':['name'], 'query': {'constant_score': {'filter': {'term': {'dataCollections.title':title}}}}}).encode(), headers={'Content-Type':'application/json','Accept':'application/json'})
  j=json.load(urlopen(req)); hits=j['hits']['hits']; s.update(h['_source']['name'] for h in hits)
  if start+len(hits)>=j['hits']['total']: break
 return s|{('GM'+n[2:]) for n in s if n.startswith('NA')}
for title in ['1000 Genomes phase 3 release','1000 Genomes 30x on GRCh38','1000 Genomes on GRCh38']:
 ali=samples(title); inc=sorted(set(afgr)&ali); print(title, len(inc), 'non', sorted(set(afgr)-ali))
PY"
