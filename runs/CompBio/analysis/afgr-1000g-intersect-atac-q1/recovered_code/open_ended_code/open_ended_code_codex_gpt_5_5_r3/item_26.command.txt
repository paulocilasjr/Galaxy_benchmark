/bin/bash -lc "python3 - <<'PY'
# inspect IGSR data collection title lists maybe 1000G phase 3 release count
import json, urllib.request
from urllib.request import Request, urlopen

def post(obj):
 req=Request('https://www.internationalgenome.org/api/beta/data-collection/_search', data=json.dumps(obj).encode(), headers={'Content-Type':'application/json','Accept':'application/json'}); return json.load(urlopen(req))
j=post({'size':-1,'query':{'multi_match':{'query':'1000 Genomes','fields':['title.std']}}})
for h in j['hits']['hits']:
 s=h['_source']; print(h['_id'], s.get('title'), s.get('samples',{}).get('count'))
PY"
