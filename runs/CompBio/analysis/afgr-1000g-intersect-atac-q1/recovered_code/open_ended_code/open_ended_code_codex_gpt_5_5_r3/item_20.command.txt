/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, urllib.parse, urllib.error, time, hashlib, os
from urllib.request import Request, urlopen

def post_json(url, obj):
    data=json.dumps(obj).encode()
    req=Request(url, data=data, headers={'Content-Type':'application/json','Accept':'application/json','User-Agent':'Mozilla/5.0'})
    with urlopen(req, timeout=60) as r:
        return json.load(r)

igsr=post_json('https://www.internationalgenome.org/api/beta/sample/_search', {
    'size': 5000,
    '_source': ['name'],
    'query': {'constant_score': {'filter': {'term': {'dataCollections.title': '1000 Genomes 30x on GRCh38'}}}}
})
samples={h['_source']['name'] for h in igsr['hits']['hits']}
print('igsr total', igsr['hits']['total'], len(samples))
# search ENCODE AFGR
url='https://www.encodeproject.org/search/?type=Experiment&searchTerm=AFGR&format=json&limit=all'
req=Request(url, headers={'Accept':'application/json','User-Agent':'Mozilla/5.0'})
with urlopen(req, timeout=60) as r:
    enc=json.load(r)
atac=[x for x in enc['@graph'] if x.get('assay_title')=='ATAC-seq']
print('atac', len(atac))
rows=[]; missing=[]; nonigsr=[]; multi=[]
for x in atac:
    acc=x['accession']
    # sample name robust
    name=x.get('biosample_ontology',{}).get('term_name')
    if not name:
        parts=x.get('biosample_summary','').split()
        name=parts[-1] if parts else None
    req=Request(f'https://www.encodeproject.org/experiments/{acc}/?format=json', headers={'Accept':'application/json','User-Agent':'Mozilla/5.0'})
    with urlopen(req, timeout=60) as r:
        exp=json.load(r)
    bams=[f for f in exp.get('files',[]) if f.get('file_format')=='bam' and f.get('assembly')=='GRCh38' and f.get('status')=='released' and f.get('output_type')=='alignments']
    if name in samples:
        if len(bams)"'!=1:
            multi.append((name,acc,[(f.get('"'accession'), f.get('output_type'), f.get('md5sum'), f.get('submitted_file_name')) for f in bams]))
        for f in bams:
            rows.append((name, acc, f['accession'], f['md5sum']))
    else:
        nonigsr.append((name,acc, len(bams)))
print('rows',len(rows),'unique samples',len({r[0] for r in rows}),'nonigsr',len(nonigsr),'multi issues',multi[:3])
print('nonigsr names', sorted(n for n,_,__ in nonigsr))
# duplicates
from collections import Counter
c=Counter(r[0] for r in rows)
print('dup samples', [x for x in c.items() if x[1]>1])
# save details and md5 list sorted alphabetically by md5
with open('afgr_1000g_filtered_bam_details.tsv','w') as out:
    out.write('sample\\texperiment\\tfile\\tmd5sum\\n')
    for r in sorted(rows): out.write('\\t'.join(r)+'\\n')
md5s=sorted(r[3] for r in rows)
with open('afgr_1000g_filtered_bam_md5sums.txt','w') as out:
    for m in md5s: out.write(m+'\\n')
h=hashlib.md5(open('afgr_1000g_filtered_bam_md5sums.txt','rb').read()).hexdigest()
print('ANSWER', len({r[0] for r in rows}), h)
print('file lines', len(md5s), 'same as unique?', len(md5s)==len({r[0] for r in rows}))
PY"
