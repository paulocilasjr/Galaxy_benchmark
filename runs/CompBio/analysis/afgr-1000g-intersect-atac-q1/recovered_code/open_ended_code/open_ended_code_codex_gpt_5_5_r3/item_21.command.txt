import json, urllib.request, time, hashlib
from urllib.request import Request, urlopen

def post_json(url, obj):
    req=Request(url, data=json.dumps(obj).encode(), headers={'Content-Type':'application/json','Accept':'application/json','User-Agent':'Mozilla/5.0'})
    with urlopen(req, timeout=60) as r: return json.load(r)

samples=set(); total=None
for start in range(0,4000,500):
    j=post_json('https://www.internationalgenome.org/api/beta/sample/_search', {
        'from': start, 'size': 500, '_source': ['name'],
        'query': {'constant_score': {'filter': {'term': {'dataCollections.title': '1000 Genomes 30x on GRCh38'}}}}
    })
    total=j['hits']['total']
    hits=j['hits']['hits']
    print(start, len(hits), hits[0]['_source']['name'] if hits else None)
    samples.update(h['_source']['name'] for h in hits)
    if start+len(hits)>=total: break
print('total', total, len(samples))
print('contains test', 'HG02763' in samples, 'GM18907' in samples, 'GM18498' in samples)
PY"
