/bin/bash -lc "curl -L -s -H 'Content-Type: application/json' -X POST 'https://www.internationalgenome.org/api/beta/sample/_search' -d '{\"size\":4000,\"_source\":[\"name\",\"biosampleId\",\"populations.code\"],\"sort\":[\"name\"],\"query\":{\"constant_score\":{\"filter\":{\"term\":{\"dataCollections.title\":\"1000 Genomes 30x on GRCh38\"}}}}}' > igsr_30x_samples.json
jq '.hits.total, (.hits.hits|length)' igsr_30x_samples.json
jq -r '.hits.hits[:5][] | ._source.name' igsr_30x_samples.json
jq -r '.hits.hits[-5:][] | ._source.name' igsr_30x_samples.json"
