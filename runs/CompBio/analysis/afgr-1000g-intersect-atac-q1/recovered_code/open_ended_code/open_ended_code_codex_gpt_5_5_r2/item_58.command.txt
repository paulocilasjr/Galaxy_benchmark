/bin/bash -lc "set -e
: > igsr_30x_names.txt
for from in 0 1000 2000 3000; do
  curl -L -s -H 'Content-Type: application/json' -X POST 'https://www.internationalgenome.org/api/beta/sample/_search' \\
    -d \"{\\\"from\\\":"'$from,'"\\\"size\\\":1000,\\\"_source\\\":[\\\"name\\\"],\\\"query\\\":{\\\"constant_score\\\":{\\\"filter\\\":{\\\"term\\\":{\\\"dataCollections.title\\\":\\\"1000 Genomes 30x on GRCh38\\\"}}}}}\" \\
    > \"igsr_page_"'$from.json"
  jq -r '"'.hits.hits[]._source.name' \"igsr_page_"'$from.json" >> igsr_30x_names.txt
done
sort -u igsr_30x_names.txt -o igsr_30x_names.txt
wc -l igsr_30x_names.txt
head -5 igsr_30x_names.txt
tail -5 igsr_30x_names.txt'
