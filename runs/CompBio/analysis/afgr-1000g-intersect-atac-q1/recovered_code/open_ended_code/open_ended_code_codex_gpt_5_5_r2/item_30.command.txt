/bin/bash -lc "curl -L -s 'https://www.encodeproject.org/search/?type=File&assay_title=ATAC-seq&lab.title=Stephen%20Montgomery,%20Stanford&file_format=bam&format=json&limit=all' > encode_mont_atac_bam_files.json
