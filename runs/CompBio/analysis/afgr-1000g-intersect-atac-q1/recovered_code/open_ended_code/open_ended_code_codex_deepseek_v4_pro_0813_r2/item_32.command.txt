/bin/bash -lc "python3 - <<'PY'
import json, hashlib, os
rows=json.load(open('afgr_atac_bams.json'))
samples=set()
with open('1000g_3202_samples.tsv') as f:
    next(f)
    for line in f:
        parts=line.split()
        if len(parts)>=2:
            samples.add(parts[1])
md5s=[r['md5sum'] for r in rows if r['sample'] in samples]
md5s_sorted=sorted(md5s)
with open('afgr_filtered_bam_md5s.txt','w') as f:
    for m in md5s_sorted:
        f.write(m+'\\n')
file_md5=hashlib.md5(open('afgr_filtered_bam_md5s.txt','rb').read()).hexdigest()
print('count', len(md5s_sorted))
print('file_md5', file_md5)
print('contents:')
print(open('afgr_filtered_bam_md5s.txt').read())
print('answer', f'{len(md5s_sorted)},{file_md5}')
PY"
