/bin/bash -lc "curl -L --fail --silent --show-error 'https://www.encodeproject.org/search/?searchTerm=AFGR&type=Experiment&format=json&limit=all' -H 'Accept: application/json' -o encode_afgr_experiments.json
wc -c encode_afgr_experiments.json
python3 - <<'PY'
import json
x=json.load(open('encode_afgr_experiments.json'))
print(x.keys())
print(len(x.get('@graph',[])))
from collections import Counter
print(Counter(y.get('assay_title') for y in x['@graph']))
for y in x['@graph'][:5]: print(y.get('accession'),y.get('assay_title'),y.get('description'),y.get('biosample_summary'))
PY"
