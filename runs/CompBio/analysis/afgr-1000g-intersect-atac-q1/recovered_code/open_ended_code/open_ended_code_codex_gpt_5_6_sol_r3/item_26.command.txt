/bin/bash -lc "python3 - <<'PY'
import csv,collections
kg={r['SampleID']:r for r in csv.DictReader(open('20130606_g1k_3202_samples_ped_population.txt'),delimiter=' ')}
rows=list(csv.DictReader(open('encode_afgr_atac_metadata.tsv'),delimiter='\\t'))
names=sorted({r['Biosample term name'] for r in rows})
matched=[]; unmatched=[]
for name in names:
 candidates=[name]
 if name.startswith('GM'): candidates.append('NA'+name[2:])
 hit=next((x for x in candidates if x in kg),None)
 (matched if hit else unmatched).append((name,hit,kg[hit]['Population'] if hit else None))
print('rows',len(rows),'names',len(names),'matched',len(matched),'unmatched',len(unmatched))
print('matched populations',collections.Counter(x[2] for x in matched))
print('unmatched',unmatched)
print('outputs',collections.Counter((r['File format'],r['Output type'],r['File assembly']) for r in rows))
PY"
