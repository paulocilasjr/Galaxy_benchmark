import xml.etree.ElementTree as ET
root=ET.parse('afgr_article.xml')
for p in root.iter('passage'):

    txt=p.findtext('text','')
    if 'availability' in txt.lower() or 'accession' in txt.lower() or 'download' in txt.lower() or 'repository' in txt.lower() or sec in ('DATA_AVAILABILITY','SUPPL'):
        print(sec, txt)
PY"
