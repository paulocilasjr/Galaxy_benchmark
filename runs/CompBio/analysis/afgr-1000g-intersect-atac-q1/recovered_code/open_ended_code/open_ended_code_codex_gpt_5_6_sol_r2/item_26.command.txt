/bin/bash -lc "python3 - <<'PY'
import json,collections,hashlib,re
xs=json.load(open('afgr_atac_experiments.json'))
panel={x.split()[0] for x in open('1kGP.3202_samples.pedigree_info.txt') if x.split()[0]"'!='"'sampleID'}
selected=[]
for e in xs:
    raw=e['biosample_summary'].rsplit(' ',1)[-1]
    sid=('NA'+raw[2:]) if raw.startswith('GM') else raw
    if sid not in panel: continue
    b=[f for f in e['files'] if f.get('file_type')=='bam' and f.get('output_type')=='alignments' and f.get('assembly')=='GRCh38' and f.get('status')=='released']
    assert len(b)==1
    assert re.fullmatch(r'[0-9a-f]{32}',b[0]['md5sum'])
    selected.append((sid,b[0]['accession'],b[0]['md5sum'],b[0].get('submitted_file_name')))
actual=open('filtered_bam_md5sums.txt','rb').read()
expected=(''.join(x[2]+'\\n' for x in sorted(selected,key=lambda x:x[2]))).encode()
print('experiments',len(xs),'selected',len(selected),'unique_samples',len({x[0] for x in selected}),'unique_files',len({x[1] for x in selected}),'unique_md5s',len({x[2] for x in selected}))
print('bytes_identical',actual==expected,'bytes',len(actual),'trailing_newline',actual.endswith(b'\\n'))
print('digest',hashlib.md5(actual).hexdigest())
print('submitted examples')
for row in selected[:5]: print(row)
PY"
