/bin/bash -lc "python - <<'PY'
import gzip,heapq,collections
f='reverse.search.gwas.q1.tsv.gz'
bychr=collections.defaultdict(list); alltop=[]
with gzip.open(f,'rt') as h:
 next(h)
 for line in h:
  x=line.rstrip().split('\\t'); p=float(x[7]); rec=(p,int(x[1]),x[0],x[2])
  if len(bychr[x[0]])<20: heapq.heappush(bychr[x[0]],(-p,rec))
  elif p < -bychr[x[0]][0][0]: heapq.heapreplace(bychr[x[0]],(-p,rec))
  if len(alltop)<100: heapq.heappush(alltop,(-p,rec))
  elif p < -alltop[0][0]: heapq.heapreplace(alltop,(-p,rec))
for c in sorted(bychr,key=lambda z:int(z)):
 print('\\nCHR',c)
 for _,r in sorted(bychr[c],key=lambda z:z[1][0]): print('%.8g'%r[0],r[1],r[3])
PY"
