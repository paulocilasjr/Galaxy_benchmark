/bin/bash -lc "python - <<'PY'
import urllib.request,struct,zlib
u='https://zenodo.org/api/records/11111186/files/afr47041.zip/content'; off=71
# fetch local header first 2KB
D=urllib.request.urlopen(urllib.request.Request(u,headers={'Range':f'bytes={off}-{off+2000}'})).read(); print('len',len(D),D[:4])
sig,ver,flag,method,tm,dt,crc,cs,us,fnl,exl=struct.unpack('<4s5H3L2H',D[:30]); print(sig,flag,method,cs,us,fnl,exl,D[30:30+fnl])
# use first 2MB compressed; local starts data at off+30+fnl+exl
start=off+30+fnl+exl; n=2_000_000
C=urllib.request.urlopen(urllib.request.Request(u,headers={'Range':f'bytes={start}-{start+n-1}'})).read(); print('compressed got',len(C),'start',C[:4])
obj=zlib.decompressobj(-15); out=obj.decompress(C); print('out',len(out),'unused',len(obj.unused_data),'eof',obj.eof);print(out[:2000].decode('utf-8','replace'))
PY"
