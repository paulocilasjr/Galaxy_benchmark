/bin/bash -lc "python - <<'PY'
import gzip, collections, math
f='reverse.search.gwas.q1.tsv.gz'
chrom=collections.Counter(); tests=collections.Counter();
minv=(1.0,None); top=[]; n=0; nonstd=[]
with gzip.open(f,'rt') as h:
    header=next(h).rstrip().split('\\t')
    for line in h:
        x=line.rstrip().split('\\t'); n+=1
        chrom[x[0]]+=1; tests[x[6]]+=1
        p=float(x[7])
        if p<minv[0]: minv=(p,x)
        if p < 5e-8: top.append((p,x))
        if p==0 or p<1e-300: nonstd.append((p,x))
print('header',header,'n',n,'chrom',chrom,'tests',tests)
print('min',minv)
print('n significant',len(top),'zero/extreme',len(nonstd))
for p,x in sorted(top)[:50]: print(p,'\\t'.join(x))
PY"
