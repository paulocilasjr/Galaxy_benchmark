/bin/bash -lc "python - <<'PY'
import urllib.request,json

def get(u):
 req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
 return json.load(urllib.request.urlopen(req,timeout=60))
# HF tree filter
x=get('https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true&expand=true')
for a in x:
 p=a.get('path','')
 if 'reverse' in p or p.endswith('README.md'):
  print('HF',p,a.get('size'),a.get('oid'),a.get('lastCommit'))
# GitHub repo tree
for repo in ['Genentech/compbiobench-runner']:
 info=get('https://api.github.com/repos/'+repo+'/git/trees/main?recursive=1')
 print('repo',repo,'truncated',info.get('truncated'))
 for a in info.get('tree',[]): print(a['path'],a['size'])
PY"
