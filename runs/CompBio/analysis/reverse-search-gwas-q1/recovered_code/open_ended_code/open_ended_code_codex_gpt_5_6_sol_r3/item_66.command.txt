/bin/bash -lc "python - <<'PY'
import json, urllib.request, math
variants=[
('rs34562254','17-16842991-G-A',1.4221e-16),
('rs72717048','1-161500796-C-G',9.31382e-23),
('rs7554873','1-161612233-T-C',7.85125e-21),
]
for rs,v,q in variants:
    print('\\n',rs,v,'q=',q)
    try:
      d=json.load(urllib.request.urlopen('https://pheweb.jp/api/variant/'+v))
    except Exception as e:
      print('ERROR',e); continue
    hits=[]
    for p in d.get('phenos',[]):
      pv=p.get('pval')
      if pv:
        hits.append((abs(math.log10(pv)-math.log10(q)),p.get('phenocode'),p.get('phenostring'),pv,p.get('beta'),p.get('sebeta')))
    for h in sorted(hits)[:15]: print(h)
PY"
