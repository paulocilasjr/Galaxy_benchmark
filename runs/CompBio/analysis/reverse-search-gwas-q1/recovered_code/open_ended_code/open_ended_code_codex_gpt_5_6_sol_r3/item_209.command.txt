/bin/bash -lc "curl -L --max-time 60 -s 'https://www.ebi.ac.uk/gwas/summary-statistics/api/chromosomes/17/associations/rs34562254?size=100&reveal=all' -o rs345_api.json; PYTHONPATH=./pydeps python - <<'PY'
import json,math
x=json.load(open('rs345_api.json'))['_embedded']['associations']
if isinstance(x,dict):x=x.values()
q=1.4221e-16;rows=[]
for a in x:
 p=a.get('p_value')
 try:d=abs(math.log10(float(p))-math.log10(q))
 except:continue
 rows.append((d,a.get('study_accession'),p,a.get('beta'),a.get('odds_ratio'),a.get('effect_allele'),a.get('other_allele'),a.get('hm_p_value'),a.get('hm_beta')))
for r in sorted(rows)[:40]:print(r)
print('count',len(rows))
PY"
