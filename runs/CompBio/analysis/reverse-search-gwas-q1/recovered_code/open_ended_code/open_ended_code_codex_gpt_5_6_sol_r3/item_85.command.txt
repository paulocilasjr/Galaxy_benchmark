/bin/bash -lc "python - <<'PY'
import csv,math,collections
p='q_catalog_position_hits.tsv'
by=collections.defaultdict(list)
with open(p) as f:
 for row in csv.reader(f,delimiter='\\t'):
  if len(row)<10: continue
  qp,pmid,title,trait,sra,snp,cp,ptxt,acc,pos=row[:10]
  try: d=abs(math.log10(float(qp))-math.log10(float(cp)))
  except: continue
  chrom=int(pos.split(':')[0]); bp=int(pos.split(':')[1])
  nuisance=(chrom==6 and 25_000_000<bp<35_000_000) or (chrom==1 and 160_000_000<bp<163_000_000) or pos=='2:27508073'
  by[(pmid,title)].append((pos,float(qp),float(cp),d,trait,acc,nuisance,snp))
sc=[]
for k,v in by.items():
 # best association per position by closeness
 best={}
 for x in v:
  if x[0] not in best or x[3]<best[x[0]][3]: best[x[0]]=x
 vv=list(best.values())
 close=[x for x in vv if x[3]<1]
 close2=[x for x in vv if x[3]<2]
 non=[x for x in vv if not x[6]]
 nonclose=[x for x in non if x[3]<2]
 score=sum(max(0,3-x[3]) for x in vv)
 sc.append((len(nonclose),len(close),len(close2),len(vv),score,k,vv))
for nc,c,c2,n,s,k,v in sorted(sc,reverse=True)[:80]:
 print('\\nRANK',nc,c,c2,n,round(s,2),k)
 for x in sorted(v,key=lambda z:z[3])[:12]: print(' ',x[:6], 'NUIS' if x[6] else '',x[7])
PY"
