/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
import re
M='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'; z=ZipFile('sharapov_MOESM5.xlsx')
rt=ET.fromstring(z.read('xl/sharedStrings.xml')); ss=[''.join(t.text or '' for t in x.iter(M+'t')) for x in rt.findall(M+'si')]
root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
for row in root.iter(M+'row'):
 d={}
 for c in row.findall(M+'c'):
  col=re.match('[A-Z]+',c.get('r')).group();v=c.find(M+'v');x='' if v is None else v.text
  if c.get('t')=='s' and x:x=ss[int(x)]
  d[col]=x
 if d.get('A','').startswith('rs'):
  print(*(d.get(x,'') for x in ['A','B','C','F','G','K','M']),sep='\\t')
PY"
