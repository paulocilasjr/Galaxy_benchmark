/bin/bash -c "python3 - <<'PY'
import itertools
rows=[]
for line in open('[REDACTED LOCAL PATH]'):
 p,chrom,pos,vid,ref,alt,a1=line.rstrip('\\n').split('\\t')
 p=float(p); pos=int(pos)
 if p>=5e-8: continue
 rows.append((p,chrom,pos,vid,ref,alt,a1))
rows.sort(key=lambda x:(x[1],x[2],x[0]))
# distance clump 500kb, keep lower p
loci=[]
for r in rows:
 p,chrom,pos,vid,ref,alt,a1=r
 found=False
 for i,l in enumerate(loci):
  if l['chrom']==chrom and abs(pos-l['lastpos'])<500000:
   l['lastpos']=pos
   found=True
   break
 if not found:
  loci.append({'chrom':chrom,'pos':pos,'p':p,'vid':vid,'ref':ref,'alt':alt,'a1':a1,'lastpos':pos})
print('nloci',len(loci))
for l in sorted(loci,key=lambda x:x['p']):
 print(l['p'],l['chrom'],l['pos'],l['vid'],l['ref'],l['alt'],l['a1'])
PY"
