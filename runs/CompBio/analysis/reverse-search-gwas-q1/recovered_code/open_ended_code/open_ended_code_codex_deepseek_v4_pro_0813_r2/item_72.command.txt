/bin/bash -lc "python3 - <<'PY'
import gzip, heapq
f='[REDACTED LOCAL PATH]'
rows=[]
with gzip.open(f,'rt') as fh:
    h=fh.readline()
    for line in fh:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<8: continue
        try: pv=float(p[6])
        except: continue
        if pv<1e-8:
            rows.append((pv,p[0],int(p[1]),p[2],p[3]))
rows.sort()
print('n p<1e-8',len(rows))
# clump 500kb
loci=[]
for pv,chrom,pos,ea,oa in rows:
    placed=False
    for l in loci:
        if l['chr']==chrom and abs(l['pos']-pos)<500000:
            if pv<l['p']:
                l.update(p=pv,pos=pos,ea=ea,oa=oa)
            placed=True;break
    if not placed:
        loci.append({'chr':chrom,'pos':pos,'p':pv,'ea':ea,'oa':oa})
loci.sort(key=lambda x:(x['chr'],x['pos']))
print('loci',len(loci))
for l in loci:
    print(l)
PY"
