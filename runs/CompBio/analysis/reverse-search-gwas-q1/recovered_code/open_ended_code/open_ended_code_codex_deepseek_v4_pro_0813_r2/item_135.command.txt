/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
# introspection for variant type fields
q='''query { __type(name: \"Variant\") { fields { name type { kind name ofType { kind name } } } } }'''
r=requests.post(url,json={'query':q})
print(json.dumps(r.json(),indent=2)[:8000])
PY"
