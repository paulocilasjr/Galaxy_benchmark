/bin/bash -lc "python3 - <<'PY'
import requests, json, time
rsids=['rs34562254','rs4690098','rs3117301','rs3095152','rs3130908','rs1150756','rs9274623','rs372779368']
for rs in rsids:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations'
    r=requests.get(url,headers={'Accept':'application/json'},params={'size':100})
    print('\\n',rs,r.status_code)
    if not r.ok: continue
    assoc=r.json()['_embedded']['associations']
    for a in assoc[:10]:
        p=a.get('pvalue'); beta=a.get('betaNum'); or_=a.get('orPerCopyNum')
        study_url=a['_links']['study']['href']; trait_url=a['_links']['efoTraits']['href']
        sr=requests.get(study_url,headers={'Accept':'application/json'})
        tr=requests.get(trait_url,headers={'Accept':'application/json'})
        study=sr.json() if sr.ok else {}
        traits=[]
        if tr.ok:
            emb=tr.json().get('_embedded',{})
            if 'efoTraits' in emb: traits=[t.get('trait') for t in emb['efoTraits']]
        print('  p=',p,'beta=',beta,'or=',or_,'study=',study.get('accessionId'),'pmid=',study.get('pubmedId'),'title=',study.get('title'),'traits=',traits)
        time.sleep(.05)
PY"
