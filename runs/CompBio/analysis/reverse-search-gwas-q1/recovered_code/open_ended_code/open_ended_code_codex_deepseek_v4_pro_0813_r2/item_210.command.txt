/bin/bash -c "python3 - <<'PY'
rows=[]
for line in open('[REDACTED LOCAL PATH]'):
 p,chrom,pos,vid,ref,alt,a1=line.rstrip('\\n').split('\\t')
 p=float(p); pos=int(pos)
 if p>=5e-8: continue
 rows.append((chrom,pos,p,vid,ref,alt,a1))
from collections import defaultdict
byd=defaultdict(list)
for r in rows: byd[r[0]].append(r)
for chrom in byd:
 byd[chrom].sort(key=lambda x:x[1])
# greedy: iterate sorted by pos; if within 500kb of current locus max pos, update max pos, retain min p; else new
all_loci=[]
for chrom,arr in byd.items():
 loci=[]
 for chrom,pos,p,vid,ref,alt,a1 in arr:
  if loci and pos-loci[-1]['maxpos']<500000:
   if p<loci[-1]['p']:
    loci[-1].update(chrom=chrom,pos=pos,p=p,vid=vid,ref=ref,alt=alt,a1=a1)
   loci[-1]['maxpos']=max(loci[-1]['maxpos'],pos)
  else:
   loci.append({'chrom':chrom,'pos':pos,'p':p,'vid':vid,'ref':ref,'alt':alt,'a1':a1,'maxpos':pos})
 all_loci += loci
print('nloci',len(all_loci))
for l in sorted(all_loci,key=lambda x:x['p']):
 print(l['p'],l['chrom'],l['pos'],l['vid'],l['ref'],l['alt'],l['a1'])
PY"
