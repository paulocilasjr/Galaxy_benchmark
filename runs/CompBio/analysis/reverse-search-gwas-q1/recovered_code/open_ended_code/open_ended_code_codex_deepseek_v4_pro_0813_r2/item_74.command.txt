import requests, json
positions=[('1',161530617),('17',16939677),('19',49525049),('4',3445429),('6',29686923),('6',30355154),('6',30931873),('6',31481775),('6',32067917),('6',32668221)]
for chrom,pos in positions:
    url=f'https://rest.ensembl.org/overlap/region/human/{chrom}:{pos}-{pos}?feature=gene;content-type=application/json'
    r=requests.get(url)
    if r.ok:
        data=r.json()
        genes=[]
        for x in data:
            if x.get('biotype')=='protein_coding' or x.get('gene_id','').startswith('ENSG'):
                genes.append((x.get('external_name'),x.get('gene_id'),x.get('biotype')))
        print(chrom,pos,genes)
    else:
        print(chrom,pos,r.status_code,r.text[:100])
