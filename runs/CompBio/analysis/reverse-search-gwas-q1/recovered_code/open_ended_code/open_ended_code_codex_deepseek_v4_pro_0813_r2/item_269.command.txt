import subprocess,json,sys,concurrent.futures,time

def curl_json(url, timeout=120):
    out=subprocess.check_output(['curl','--http1.1','-L','--max-time',str(timeout),'-sS',url],stderr=subprocess.DEVNULL,text=True,timeout=timeout+5)
    return json.loads(out)

def get_assoc(rs):
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?size=1000'
    d=curl_json(url)
    arr=(d.get('_embedded',{}) or {}).get('associations',[])
    out=[]
    for a in arr:
        assoc_id=a.get('_links',{}).get('self',{}).get('href','').rstrip('/').split('/')[-1]
        study_href=a.get('_links',{}).get('study',{}).get('href','')
        pval=a.get('pvalue')
        beta=a.get('betaNum'); se=a.get('standardError'); rf=a.get('riskFrequency')
        genes=[]
        for loc in a.get('loci',[]):
            for g in loc.get('authorReportedGenes',[]): genes.append(g.get('geneName'))
        out.append({'rs':rs,'assoc_id':assoc_id,'study_href':study_href,'pvalue':pval,'beta':beta,'se':se,'rf':rf,'genes':list(set(genes))})
    return out

def get_study(href):
    d=curl_json(href)
    return {
      'accession': d.get('accessionId'),
      'pubmed': d.get('publicationInfo',{}).get('pubmedId'),
      'trait': d.get('diseaseTrait',{}).get('trait'),
      'title': d.get('title','')[:300] if d.get('title') else d.get('publicationInfo',{}).get('title','')[:300],
      'initial': d.get('initialSampleSize'),
      'full': d.get('fullPvalueSet'),
      'snpCount': d.get('snpCount'),
      'date': d.get('publicationDate'),
      'cohort': d.get('cohort'),
    }
