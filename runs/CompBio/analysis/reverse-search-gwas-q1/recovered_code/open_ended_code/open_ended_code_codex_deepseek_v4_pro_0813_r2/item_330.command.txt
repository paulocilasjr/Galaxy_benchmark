/bin/bash -lc "python3 - <<'PY'
import requests
url='https://api.platform.opentargets.org/api/v4/graphql'
query='{ __type(name:\"Query\") { fields { name description } } }'
r=requests.post(url,json={'query':query},timeout=30)
print(r.status_code)
text=r.text
# print fields containing variant/gwas/study
import re
for m in re.finditer(r'\"name\":\"(["'^"]+)"'"', text):
 n=m.group(1)
 if any(x in n.lower() for x in ['variant','gwas','study','trait','assoc']): print(n)
PY"
