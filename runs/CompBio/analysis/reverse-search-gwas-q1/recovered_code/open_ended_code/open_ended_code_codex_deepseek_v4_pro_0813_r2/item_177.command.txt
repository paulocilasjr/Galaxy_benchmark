import urllib.request, json
for rs in ['rs34562254','rs4690098']:
 u=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}'
 print('\\nSNP',u)

  d=json.load(urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'}),timeout=30))
  print(json.dumps(d, indent=2)[:8000])
 except Exception as e: print(e)
PY"
