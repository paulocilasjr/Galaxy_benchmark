/bin/bash -c "python3 - <<'PY'
import urllib.request,json,time
positions=[('1','161530617'),('6','32067917'),('17','16939677'),('19','49525049'),('4','3445429'),('4','102486185'),('2','111471251'),('13','108308037'),('2','27508073'),('6','33546790')]
for chrom,pos in positions:
 u=f'https://rest.ensembl.org/overlap/region/human/{chrom}:{int(pos)-1}-{int(pos)+1}?feature=gene;content-type=application/json'
 try:
  d=json.load(urllib.request.urlopen(urllib.request.Request(u,headers={'Content-Type':'application/json'}),timeout=20))
  genes=[]
  for x in d:
   if x.get('biotype')=='protein_coding' or x.get('biotype')=='lincRNA':
    genes.append((x.get('external_name'),x.get('biotype'),x.get('strand')))
  print(chrom,pos,genes)
 except Exception as e: print(chrom,pos,'ERR',e)
PY"
