/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict
# read all? 12.7m okay memory maybe big. stream and keep p<1e-8
rows=[]
with gzip.open('reverse.search.gwas.q1.tsv.gz','rt') as f:
    header=f.readline()
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<8: continue
        try: p=float(parts[7])
        except: continue
        if p<1e-8:
            rows.append((p,parts[0],int(parts[1]),parts[2]))
rows.sort()
print('num p<1e-8',len(rows))
# greedy clump by chr and pos within 1Mb and take min p
loci=[]
for p,chrom,pos,vid in rows:
    placed=False
    for locus in loci:
        if locus['chr']==chrom and abs(locus['pos']-pos)<500000:
            if p<locus['p']:
                locus['p']=p;locus['pos']=pos;locus['vid']=vid
            placed=True;break
    if not placed:
        loci.append({'chr':chrom,'pos':pos,'p':p,'vid':vid})
loci.sort(key=lambda x:(x['chr'],x['pos']))
print('loci',len(loci))
for l in loci:
    print(l)
PY"
