/bin/bash -c "python3 - <<'PY'
import gzip
f='[REDACTED LOCAL PATH]'
best=[]
with gzip.open(f,'rt') as fh:
    h=next(fh)
    for line in fh:
        parts=line.rstrip().split('\\t')
        if len(parts)<7: continue
        try: p=float(parts[6])
        except: continue
        if len(best)<20 or p<best[-1][0]:
            best.append((p,line.rstrip()))
            best.sort(key=lambda x:x[0])
            best=best[:20]
for x in best: print(x)
PY"
