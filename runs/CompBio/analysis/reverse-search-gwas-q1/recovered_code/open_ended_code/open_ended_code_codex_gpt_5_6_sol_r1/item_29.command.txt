/bin/bash -lc "jq -r '._embedded.associations[] | [.pvalue,(._links.study.href|split(\"/\")[-2])] | @tsv' assoc_rs345.json | sort -u > rs345_study_ids.tsv
cut -f2 rs345_study_ids.tsv | xargs -P 8 -I{} sh -c 'curl -sS \"https://www.ebi.ac.uk/gwas/rest/api/associations/{}/study\" | jq -r --arg id \"{}\" \"[\\"'$id,.accessionId,.diseaseTrait.trait,.publicationInfo.pubmedId,.publicationInfo.title] | @tsv"'"' > rs345_studies.tsv
join -t "'$'"'\\t' -1 2 -2 1 <(sort -k2,2 rs345_study_ids.tsv) <(sort -k1,1 rs345_studies.tsv) | sort -t "'$'"'\\t' -k2,2g | head -100"
