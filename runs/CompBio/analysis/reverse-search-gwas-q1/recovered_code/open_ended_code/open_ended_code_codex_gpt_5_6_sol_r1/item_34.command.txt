/bin/bash -lc 'for rs in rs372779368 rs1150756 rs3132451 rs34562254 rs3095152 rs10525267 rs4690098 rs3117301 rs4849748 rs374039502 rs1260326 rs6918046; do curl -sS "https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/${rs}/associations?size=1000" | jq -r --arg rs "$rs" '"'._embedded.associations[]? | ["'$rs,.pvalue,(._links.study.href|split("/")[-2])] | @tsv'"'; done > candidate_assoc.tsv
cut -f3 candidate_assoc.tsv | sort -u | xargs -P 12 -I{} sh -c 'a="'$(curl -sS "https://www.ebi.ac.uk/gwas/rest/api/associations/{}/study" | jq -r .accessionId); test "$a" = GCST90671983 && echo {}'"' > glycan_assoc_ids.txt
awk 'NR==FNR{x["'$1]=1;next} x[$3]'"' glycan_assoc_ids.txt candidate_assoc.tsv | sort -k1,1 -k2,2g"
