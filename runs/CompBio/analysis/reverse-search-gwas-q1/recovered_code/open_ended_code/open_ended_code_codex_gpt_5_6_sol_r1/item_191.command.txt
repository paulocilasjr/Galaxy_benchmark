/bin/bash -lc "curl -L --fail --silent --show-error https://www.ebi.ac.uk/europepmc/webservices/rest/PMC12218978/fullTextXML -o nglycan.xml && python - <<'PY'
import xml.etree.ElementTree as ET
r=ET.parse('nglycan.xml')
for s in r.iter():
 if s.tag.split('}')[-1]=='supplementary-material':
  href='';
  for e in s.iter():
   if e.tag.split('}')[-1]=='media': href=next((v for k,v in e.attrib.items() if k.split('}')[-1]=='href'),'')
  print(href,' :: ',' '.join(''.join(s.itertext()).split())[:300])
PY"
