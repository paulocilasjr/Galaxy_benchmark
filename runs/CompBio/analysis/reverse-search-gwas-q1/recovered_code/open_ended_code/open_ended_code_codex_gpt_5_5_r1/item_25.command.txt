/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,re,urllib.parse
rsids=['rs34562254','rs144433536','rs1150756']
for rs in rsids:
    print('\\nRS',rs)
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?projection=associationBySnp&size=200'
    data=json.load(urllib.request.urlopen(url,timeout=30))
    ass=data.get('_embedded',{}).get('associations',[])
    print('n',len(ass),'page',data.get('page'))
    for a in ass[:200]:
        pv=(a.get('pvalueMantissa'),a.get('pvalueExponent'),a.get('pvalue'))
        desc=a.get('pvalueDescription')
        traits=[t.get('trait') for t in a.get('efoTraits',[])]
        # study link later
        print('p',pv,'desc',desc,'traits',traits,'beta',a.get('betaNum'),a.get('betaUnit'),a.get('orPerCopyNum'))
        print('snps',[(s.get('rsId'),s.get('riskAlleleName')) for loc in a.get('loci',[]) for s in loc.get('strongestRiskAlleles',[])])
        print('study', a.get('_links',{}).get('study',{}).get('href'))
PY"
