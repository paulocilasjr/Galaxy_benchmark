/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
studies=[]
for page in range(4):
 url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByPublicationIdPubmedId?pubmedId=39789286&size=1000&page={page}'
 d=json.load(urllib.request.urlopen(url,timeout=60))
 studies += d.get('_embedded',{}).get('studies',[])
for s in studies:
 trait=s.get('diseaseTrait',{}).get('trait','')
 if any(x.lower() in trait.lower() for x in ['neutro','monocyte','lymphocyte','platelet','white blood','red blood','basophil','eosinophil','haemat','hemoglobin','haemoglobin']):
  print(s.get('accessionId'), trait)
PY"
