/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for gcst in ['GCST008571','GCST008572','GCST008573','GCST008574','GCST008575','GCST008576']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/{gcst}/associations?projection=associationByStudy&size=100'
    print('\\n',gcst)
    data=json.load(urllib.request.urlopen(url,timeout=30))
    ass=data.get('_embedded',{}).get('associations',[])
    print('n',len(ass))
    # print loci sorted by p roughly
    rows=[]
    for a in ass:
        pv=a.get('pvalue') or (a.get('pvalueMantissa') or 1)*10**(a.get('pvalueExponent') or 0)
        sn=[]
        for loc in a.get('loci',[]):
            for s in loc.get('strongestRiskAlleles',[]):
                sn.append(s.get('riskAlleleName') or s.get('rsId'))
        rows.append((pv,sn,[t.get('trait') for t in a.get('efoTraits',[])]))
    for pv,sn,tr in sorted(rows,key=lambda x:x[0])[:20]:
        print(pv,sn,tr)
PY"
