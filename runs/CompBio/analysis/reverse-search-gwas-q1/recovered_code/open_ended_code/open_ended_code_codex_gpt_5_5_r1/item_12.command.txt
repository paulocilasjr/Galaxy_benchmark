/bin/bash -lc "python3 - <<'PY'
import gzip, heapq, math
path='reverse.search.gwas.q1.tsv.gz'
N=200
heap=[]
with gzip.open(path,'rt') as f:
    header=next(f)
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<8: continue
        try: p=float(parts[7])
        except: continue
        item=(-p, p, parts)
        if len(heap)<N:
            heapq.heappush(heap,item)
        elif -heap[0][0] > p:
            heapq.heapreplace(heap,item)
for _,p,parts in sorted(heap, key=lambda x:x[1]):
    print('\\t'.join([f'{p:.6g}', parts[2], parts[0], parts[1], parts[3], parts[4], parts[5]]))
PY"
