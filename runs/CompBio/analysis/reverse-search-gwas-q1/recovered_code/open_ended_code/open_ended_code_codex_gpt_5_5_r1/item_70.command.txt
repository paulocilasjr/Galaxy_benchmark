import urllib.request,json
for gcst in ['GCST90469200','GCST90469201','GCST90469202','GCST90469203']:
 print('\\n',gcst)
 url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/{gcst}/associations?projection=associationByStudy&size=200'
 data=json.load(urllib.request.urlopen(url,timeout=30))
 ass=data.get('_embedded',{}).get('associations',[])
 rows=[]
 for a in ass:
  pv=a.get('pvalue') or (a.get('pvalueMantissa') or 1)*10**(a.get('pvalueExponent') or 0)
  locs=[]
  for loc in a.get('loci',[]):
   for s in loc.get('strongestRiskAlleles',[]): locs.append(s.get('riskAlleleName'))
  rows.append((pv,locs,[t.get('trait') for t in a.get('efoTraits',[])]))
 print('n',len(rows))
 for r in sorted(rows,key=lambda x:x[0])[:20]: print(r)
PY"
