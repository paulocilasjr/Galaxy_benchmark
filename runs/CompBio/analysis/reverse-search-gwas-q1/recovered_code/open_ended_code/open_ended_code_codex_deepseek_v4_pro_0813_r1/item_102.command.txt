/bin/bash -lc "python3 - <<'PY'
import requests,json
sid='GCST90671983'; base='https://www.ebi.ac.uk/gwas/rest/api'
r=requests.get(f'{base}/studies/{sid}/associations',params={'size':100,'projection':'associationBySnp'},timeout=30)
d=r.json(); rows=d.get('_embedded',{}).get('associations',[])
for a in rows:
    snps=a.get('snps',[])
    rs=','.join(s.get('rsId','') for s in snps)
    pos=','.join(str(s.get('locations',[{}])[0].get('chromosomePosition')) for s in snps)
    chr=','.join(str(s.get('locations',[{}])[0].get('chromosomeName')) for s in snps)
    print(a.get('pvalue'), rs, chr,pos, a.get('range'), a.get('orPerCopyNum'), a.get('betaNum'))
PY"
