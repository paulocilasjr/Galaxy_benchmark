import urllib.request, json
url='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByPublicationIdPubmedId?pubmedId=40593539&size=1000'
with urllib.request.urlopen(url, timeout=60) as r: data=json.load(r)
studies=data['_embedded']['studies']
print('total',len(studies))
for s in studies:
    n=s.get('snpCount')
    if n and abs(n-12774703)<500000:
        print(s['accessionId'], n, s['fullPvalueSet'], s['diseaseTrait']['trait'])
print('null snpCount:')
for s in studies:
    if s.get('snpCount') is None:
        print(s['accessionId'], s['fullPvalueSet'], s['diseaseTrait']['trait'])
PY"
