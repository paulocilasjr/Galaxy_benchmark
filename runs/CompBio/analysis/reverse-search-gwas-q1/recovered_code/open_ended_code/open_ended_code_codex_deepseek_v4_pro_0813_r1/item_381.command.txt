/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,math
vid='1_161530617_CTT_C'
q={\"query\":f'{{ variant(variantId: \"{vid}\") {{ rsIds credibleSets {{ rows {{ study {{ id pubmedId publicationTitle traitFromSource }} studyType pValueMantissa pValueExponent }} }} }} }}'}
req=urllib.request.Request('https://api.platform.opentargets.org/api/v4/graphql', data=json.dumps(q).encode(), headers={'Content-Type':'application/json'})
with urllib.request.urlopen(req,timeout=60) as r:d=json.load(r)
rows=d['data']['variant']['credibleSets']['rows']
def key(r):
    m=r.get('pValueMantissa') or 0
    e=r.get('pValueExponent') or 0
    return e + math.log10(max(m,1e-300))
for row in sorted(rows,key=key):
    st=row['study']; m=row.get('pValueMantissa') or 0; e=row.get('pValueExponent') or 0
    print(f'{m:g}e{e}\\t{st[\"id\"]}\\t{st[\"pubmedId\"]}\\t{repr(st[\"traitFromSource\"])}\\t{(st[\"publicationTitle\"] or \"\")[:70]}')
PY"
