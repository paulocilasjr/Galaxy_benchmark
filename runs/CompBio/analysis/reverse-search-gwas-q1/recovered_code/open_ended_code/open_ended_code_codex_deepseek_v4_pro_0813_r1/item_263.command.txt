/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for url in [
'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByPublicationIdPubmedId?pubmedId=33710309&size=1000',
'https://www.ebi.ac.uk/gwas/rest/api/associations/search/findByPubmedId?pubmedId=33710309&projection=associationByStudy']:
 print('URL',url)
 try:
  with urllib.request.urlopen(url,timeout=60) as r: data=json.load(r)
  print(data.keys(), data.get('page'))
  if 'studies' in data.get('_embedded',{}):
   for s in data['_embedded']['studies'][:50]: print(s['accessionId'],s['snpCount'],s['fullPvalueSet'],s['diseaseTrait']['trait'])
  if 'associations' in data.get('_embedded',{}):
   ass=data['_embedded']['associations']; print('assoc',len(ass))
   for a in ass[:100]: print(a['study']['accessionId'],a['study']['diseaseTrait']['trait'],a.get('pvalue'))
 except Exception as e: print('ERR',repr(e))
PY"
