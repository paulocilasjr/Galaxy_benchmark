/bin/bash -lc "python3 - <<'PY'
import csv, urllib.request, re, concurrent.futures, struct
path='summary_statistics_table_export.tsv'
urls=[]
with open(path, encoding='utf-8', newline='') as f:
    r=csv.DictReader(f, delimiter='\\t')
    for row in r:
        if row.get('pubmedId')=='40593539':
            u=row['summaryStatistics']
            if u and u not in urls: urls.append(u)

def list_harm(base):
    try:
        html=urllib.request.urlopen(base+'/harmonised/', timeout=30).read().decode('utf-8','replace')
        files=re.findall(r'href=\"(["'^"]+'"\\.h\\.tsv\\.gz)\"',html)
        return [(base.rstrip('/')+'/harmonised/'+f, base.split('/')[-1]) for f in files]
    except Exception: return []

harm=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=20) as ex:
    for res in ex.map(list_harm,urls): harm.extend(res)
print('harm',len(harm))
target=653224897

def isize(u):
    req=urllib.request.Request(u, headers={'User-Agent':'curl/8','Range':'bytes=-4'})
    try:
        with urllib.request.urlopen(req, timeout=30) as resp:
            data=resp.read()
            return struct.unpack('<I', data[-4:])[0]
    except Exception as e: return None

found=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=20) as ex:
    futs=[ex.submit(isize,u) for u,_ in harm]
    for (u,acc),fut in zip(harm,futs):
        size=fut.result()
        if size is not None and size==target:
            print('FOUND',acc,u,size); found.append((acc,u,size))
print('found',len(found))
PY"
