import urllib.request,json
variants=['1_161530617_CTT_C','6_32067917_C_T','17_16939677_G_A','4_3445429_C_T','19_49525049_CCTGCTG_C']
for vid in variants:
    q={\"query\":f'{{ variant(variantId: \"{vid}\") {{ id rsIds position credibleSets {{ count rows {{ study {{ id pubmedId publicationTitle traitFromSource }} studyType pValueMantissa pValueExponent }} }} }} }}'}
    req=urllib.request.Request('https://api.platform.opentargets.org/api/v4/graphql', data=json.dumps(q).encode(), headers={'Content-Type':'application/json'})
    try:
        with urllib.request.urlopen(req, timeout=60) as r: d=json.load(r)
    except Exception as e:
        print('ERR',vid,e); continue
    v=d['data']['variant']
    rows=v.get('credibleSets',{}).get('rows',[])
    print('\\n',vid,'rs',v.get('rsIds'),'count',len(rows))
    for row in rows:
        st=row['study']; pmid=st.get('pubmedId'); trait=st.get('traitFromSource')
        if pmid=='40593539' or (trait and any(t in trait.lower() for t in ['glycan','glycosylation','neutral glycan'])):
            print('  MATCH',st.get('id'),pmid,repr(trait),row['pValueMantissa'],row['pValueExponent'])
    if not any(st.get('pubmedId')=='40593539' for st in [r['study'] for r in rows]):
        print('  no 40593539 in rows; first glycan-ish:')

PY"
