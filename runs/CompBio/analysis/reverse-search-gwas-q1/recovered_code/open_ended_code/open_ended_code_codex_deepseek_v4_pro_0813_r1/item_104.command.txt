/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
vids=['1_161530617_CTT_C','6_30931873_G_A','6_32067917_C_T','17_16939677_G_A','19_49525049_CCTGCTG_C','4_3445429_C_T']
# some null; use existing rs variants
for vid in vids:
 q=f'''query {{ variant(variantId:\"{vid}\") {{ id credibleSets {{ count rows {{ studyId studyType pValueMantissa pValueExponent variant {{ id rsIds }} }} }} }} }}'''
 r=requests.post(url,json={'query':q},timeout=30)
 d=r.json().get('data',{}).get('variant')
 if not d:
  print('\\n',vid,'NO'); continue
 print('\\n###',vid,'rs',d.get('rsIds'))
 for row in d['credibleSets']['rows']:
  if row['studyType']=='gwas':
   print(row['studyId'], row['pValueMantissa'],row['pValueExponent'], row['variant']['id'], row['variant']['rsIds'])
PY"
