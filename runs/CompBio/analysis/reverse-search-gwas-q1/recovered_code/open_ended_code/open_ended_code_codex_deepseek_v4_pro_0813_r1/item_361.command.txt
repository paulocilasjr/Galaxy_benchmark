/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,concurrent.futures
ids=['22003670','22002978','22005469','127948001','127944128','127944618','127948252','127953953','127945517','91506062','215454904','191874879','191832935','214364412','93531521','106378682','106377337']
def get(i):
    url=f'https://www.ebi.ac.uk/gwas/rest/api/associations/{i}/study'
    try:
        with urllib.request.urlopen(url,timeout=30) as r: d=json.load(r)
        pi=d.get('publicationInfo',{}) if d else {}
        return i,d.get('accessionId'),d.get('reportedTrait'),pi.get('pubmedId'),pi.get('title')
    except Exception as e: return i,None,None,None,str(e)
with concurrent.futures.ThreadPoolExecutor(max_workers=10) as ex:
    for res in ex.map(get,ids): print(res)
PY"
