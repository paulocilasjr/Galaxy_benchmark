/bin/bash -lc "python3 - <<'PY'
import requests, json, sys, time
base='https://www.ebi.ac.uk/gwas/rest/api'
for rs in ['rs1801274','rs9273373','rs3130342','rs9272598']:
    print('\\n###',rs)
    r=requests.get(f'{base}/singleNucleotidePolymorphisms/{rs}/associations',params={'size':500,'projection':'associationByEfoTrait'},timeout=30)
    if not r.ok:
        print(r.status_code); continue
    d=r.json()
    page=d.get('page',{})
    total=d.get('page',{}).get('totalElements')
    assocs=d.get('_embedded',{}).get('associations',[])
    print('total', total)
    for a in assocs:
        st=a.get('study',{})
        traits=[t.get('trait') for t in a.get('efoTraits',[])]
        print(a.get('pvalue'), st.get('accessionId'), st.get('diseaseTrait',{}).get('trait'), '|', '; '.join(traits), '|', st.get('publicationInfo',{}).get('pubmedId'), '|', st.get('initialSampleSize'))
PY"
