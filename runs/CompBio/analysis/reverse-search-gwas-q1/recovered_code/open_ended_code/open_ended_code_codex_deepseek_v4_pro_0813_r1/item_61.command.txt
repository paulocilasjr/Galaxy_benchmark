/bin/bash -lc "python3 - <<'PY'
import requests
base='https://rest.ensembl.org'
regions=['1:161430000-161650000','4:3400000-3500000','6:29500000-31050000','6:31000000-32700000','17:16800000-17100000','19:49300000-49700000']
for reg in regions:
    url=f'{base}/overlap/region/human/{reg}?feature=gene;content-type=application/json'
    r=requests.get(url,timeout=30)
    print('\\nREGION',reg,'status',r.status_code)
    if r.ok:
        genes=[]
        for x in r.json():
            if x.get('biotype','')=='protein_coding':
                genes.append((x.get('start'),x.get('end'),x.get('external_name')))
        for g in sorted(set(genes))[:30]: print(g)
PY"
