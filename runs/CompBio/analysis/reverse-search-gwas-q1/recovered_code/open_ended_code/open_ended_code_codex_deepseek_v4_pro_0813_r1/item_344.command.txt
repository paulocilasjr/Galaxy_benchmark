/bin/bash -lc "python3 - <<'PY'
import urllib.request
url='https://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/GCST008001-GCST009000/GCST008108/GCST008108.tsv'
req=urllib.request.Request(url, method='HEAD', headers={'User-Agent':'Mozilla/5.0'})
try:
    with urllib.request.urlopen(req,timeout=30) as r: print(r.status, r.headers.get('Content-Length'))
except Exception as e: print(type(e).__name__,e)
req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0','Range':'bytes=0-2000'})
try:
    with urllib.request.urlopen(req,timeout=30) as r: print(r.read()[:1000].decode('utf-8','ignore'))
except Exception as e: print(e)
PY"
