/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
q='''{ __schema { queryType { fields { name description args { name } } } } }'''
r=requests.post(url,json={'query':q},timeout=30)
print(r.status_code)
data=r.json()
for f in data['data']['__schema']['queryType']['fields']:
    if any(s in f['name'].lower() for s in ['variant','study','gwas','assoc']):
        print(f['name'], [a['name'] for a in f.get('args',[])])
PY"
