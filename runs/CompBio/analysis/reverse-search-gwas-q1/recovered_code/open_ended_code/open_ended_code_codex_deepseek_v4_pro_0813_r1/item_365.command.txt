/bin/bash -lc "python3 - <<'PY'
import openpyxl, glob, os
terms=['144433536','1150756','32067917','32083300','rs144433536','rs1150756','HGFAC','4690098','49525049','161530617','3445429']
for f in glob.glob('supp/*.xlsx'):
    try:
        wb=openpyxl.load_workbook(f, read_only=True, data_only=True)
    except Exception as e: continue
    hits=[]
    for sn in wb.sheetnames:
        ws=wb[sn]
        for row in ws.iter_rows(values_only=True):
            for cell in row:
                if cell is not None and isinstance(cell,str):
                    if any(t.lower() in cell.lower() for t in terms):
                        hits.append((sn,row[:12]))
                        break
    if hits:
        print('\\n===',os.path.basename(f))
        for sn,row in hits[:20]:
            print(sn, '\\t'.join('' if x is None else str(x)[:80] for x in row))
    wb.close()
PY"
