/bin/bash -lc "python3 - <<'PY'
import requests,json
base='https://www.ebi.ac.uk/gwas/rest/api'
for rs in ['rs34562254','rs1150756','rs10525267','rs4690098']:
    print('\\n###',rs)
    r=requests.get(f'{base}/singleNucleotidePolymorphisms/{rs}/associations',params={'size':200,'projection':'associationByEfoTrait'},timeout=30)
    if not r.ok: print(r.status_code); continue
    d=r.json(); assocs=d.get('_embedded',{}).get('associations',[])
    print('n',len(assocs))
    for a in assocs[:100]:
        st=a.get('study',{})
        traits='; '.join(t.get('trait') for t in a.get('efoTraits',[]))
        print(a.get('pvalue'), st.get('accessionId'), st.get('diseaseTrait',{}).get('trait'),'|',traits,'|',st.get('publicationInfo',{}).get('pubmedId'),'|',st.get('initialSampleSize'))
PY"
