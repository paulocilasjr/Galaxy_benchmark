import openpyxl
wb=openpyxl.load_workbook('supp/41467_2025_60431_MOESM3_ESM.xlsx', read_only=True, data_only=True)
for sn in wb.sheetnames:
    ws=wb[sn]
    print('\\n===',sn)
    for i,row in enumerate(ws.iter_rows(values_only=True)):
        if i>=100: break
        print(i, '\\t'.join('' if x is None else str(x)[:120] for x in row[:12]))
wb.close()
PY"
