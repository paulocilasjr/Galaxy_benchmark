/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
rs='rs144433536'
url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?projection=associationByStudy'
with urllib.request.urlopen(url, timeout=60) as r: data=json.load(r)
for a in data['_embedded']['associations']:
    s=a['study']
    print(s['accessionId'], s['publicationInfo']['pubmedId'], s['diseaseTrait']['trait'], a.get('pvalue'), a.get('pvalueMantissa'), a.get('pvalueExponent'))
    for loc in a.get('loci',[]):
      for allele in loc.get('strongestRiskAlleles',[]): print(' ', allele.get('riskAlleleName'))
PY"
