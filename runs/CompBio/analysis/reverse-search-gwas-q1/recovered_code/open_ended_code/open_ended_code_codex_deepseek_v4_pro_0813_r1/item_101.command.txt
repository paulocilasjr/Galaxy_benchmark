/bin/bash -lc "python3 - <<'PY'
import requests,json
sid='GCST90671983'
base='https://www.ebi.ac.uk/gwas/rest/api'
url=f'{base}/studies/{sid}/associations'
params={'size':1000,'projection':'associationBySnp'}
rows=[]
page=0
while True:
 p=params.copy(); p['page']=page
 r=requests.get(url,params=p,timeout=30)
 if not r.ok: print('err',r.status_code,r.text[:200]); break
 d=r.json(); assocs=d.get('_embedded',{}).get('associations',[])
 if not assocs: break
 rows += assocs
 pageinfo=d.get('page',{})
 if pageinfo.get('number',0)+1 >= pageinfo.get('totalPages',1): break
 page+=1
print('n assoc',len(rows))
# inspect first
for a in rows[:10]:
 print(json.dumps(a,indent=2)[:2000])
PY"
