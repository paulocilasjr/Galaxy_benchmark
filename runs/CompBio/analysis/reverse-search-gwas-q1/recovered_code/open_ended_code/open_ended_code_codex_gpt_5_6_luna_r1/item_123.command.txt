/bin/bash -lc 'for rs in rs72717048 rs1150756 rs34562254 rs200748895 rs230523 rs143596860 rs7554873 rs150420714; do
  curl -LfsS --max-time 60 "https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/$rs/associations?size=100" |
    jq -r --arg rs "$rs" '"'._embedded.associations[] | ["'$rs,.pvalue,._links.study.href] | @tsv'"' 2>/dev/null |
    while IFS="'$'"'\\t' read -r v p u; do
      curl -LfsS --max-time 30 \""'$u" 2>/dev/null | jq -r --arg v "$v" --arg p "$p" '"'[ "'$v,$p,.accessionId,(.publicationInfo.pubmedId // ""),(.trait_uris[0] // "")] | @tsv'"' 2>/dev/null
    done
done | sort -k3,3 -k1,1 > catalog_immune_variants.tsv
wc -l catalog_immune_variants.tsv
cat catalog_immune_variants.tsv"
