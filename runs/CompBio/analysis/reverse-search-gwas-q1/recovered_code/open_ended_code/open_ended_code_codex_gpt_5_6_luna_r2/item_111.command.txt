/bin/bash -lc "jq -r '._embedded.associations[] | [.pvalue, (._links.self.href | split(\"/\") | last)] | @tsv' rs34562254.associations.json > rs34562254.assoc.tsv
: > rs34562254.studies.tsv
while IFS="'$'"'\\t' read -r p id; do
  study="'$(curl -L --fail --silent --show-error "https://www.ebi.ac.uk/gwas/rest/api/associations/$id/study") || continue
  printf '"'%s\\t%s\\t%s\\t%s\\n' \""'$p" "$id" "$(printf '"'%s' \""'$study" | jq -r '"'.accessionId // \"\"')\" \""'$(printf '"'%s' \""'$study" | jq -r '"'[.publicationInfo.pubmedId // \"\", .publicationInfo.title // \"\"] | @tsv')\" >> rs34562254.studies.tsv
done < rs34562254.assoc.tsv
sort -g rs34562254.studies.tsv | sed -n '1,100p'"
