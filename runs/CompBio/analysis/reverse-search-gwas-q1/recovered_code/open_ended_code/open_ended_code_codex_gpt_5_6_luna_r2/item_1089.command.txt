/bin/bash -lc "python3 - <<'PY'
# positions of selected GRCh38 variants, one generated from known mappings
pos={'rs379794':161530615,'rs4657041':161509069,'rs71632979':161556607,'rs55971447':161545536,'rs1801274':161509955,'rs56157533':161513866,'rs71632989':161563265,'rs9427401':161522385,'rs9427403':161524532,'rs10919033':161518227,'rs10919543':161538827,'rs3845548':161537658,'rs10919347':161531085,'rs9273902':32661939,'rs34562254':16939677,'rs4792800':16941853,'rs4561508':16945436,'rs4273077':16945825,'rs4985726':16960324}
# use coordinates only, print all variants at positions
import subprocess
cmd=['gzip','-dc','reverse.search.gwas.q1.tsv.gz']
p=subprocess.Popen(cmd,stdout=subprocess.PIPE,text=True)
for line in p.stdout:
 f=line.rstrip('\\n').split('\\t')
 if len(f)>=8 and f[0]=='#CHROM':
  print(line.rstrip()); continue
 if len(f)>=8 and f[0].isdigit() and int(f[1]) in pos.values():
  names=[k for k,v in pos.items() if v==int(f[1])]
  print(','.join(names),line.rstrip())
p.wait()
PY"
