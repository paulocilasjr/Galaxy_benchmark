print('start',flush=True)
import json,urllib.request,concurrent.futures
fs=['rs7554873','rs9273902','rs4273077','rs4561508','rs4792800','rs4985726','rs34562254','rs3752442','rs3803800','rs1801274']
ids=[]
for x in fs:
 d=json.load(open(x+'.associations.json'))
 for a in d.get('_embedded',{}).get('associations',[]):
  h=a.get('_links',{}).get('study',{}).get('href')
  if h: ids.append((x,a.get('pvalue'),h))
uniq={h:(x,p) for x,p,h in ids}
print('calls',len(uniq),flush=True)
def get(item):
 h,(x,p)=item
 try:
  req=urllib.request.Request(h,headers={'User-Agent':'Mozilla/5.0'})
  with urllib.request.urlopen(req,timeout=20) as f: d=json.load(f)
  return {'variant':x,'assoc_p':p,'url':h,'study':d}
 except Exception as e: return {'variant':x,'assoc_p':p,'url':h,'error':str(e)}
with concurrent.futures.ThreadPoolExecutor(max_workers=10) as ex:
 out=list(ex.map(get,uniq.items()))
print('mapped',flush=True)
json.dump(out,open('key-study-meta.json','w'))
print('ok',sum('study' in x for x in out),'err',sum('error' in x for x in out),flush=True)
EOF"
