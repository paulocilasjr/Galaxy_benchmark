/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET
root=ET.parse('glycan-clock.xml').getroot()
for tw in root.iter('table-wrap'):
 title=' '.join(tw.itertext()).split('\\n')[0:5]
 print('\\nTABLE',tw.attrib.get('id'), ' '.join(title))
 for tr in tw.iter('tr'):
  vals=[' '.join(''.join(td.itertext()).split()) for td in tr.findall('td')]
  if vals: print('\\t'.join(vals))
PY"
