/bin/bash -lc "python3 - <<'PY'
import glob,openpyxl,io,os
for p in glob.glob('sharapov-n[3-9]')+glob.glob('sharapov-n1[0-9]')+['sharapov-n22']:
 try:
  wb=openpyxl.load_workbook(io.BytesIO(open(p,'rb').read()),read_only=True,data_only=True)
  print(os.path.basename(p),wb.sheetnames)
  for ws in wb.worksheets: print(' ',ws.title,ws.max_row,ws.max_column)
 except Exception as e: print(p,e)
PY"
