import json,urllib.request,concurrent.futures
x=json.load(open('rs34562254-assoc.json'))
def f(a):
 aid=a['_links']['self']['href'].rstrip('/').split('/')[-1]
 try:
  d=json.load(urllib.request.urlopen(f'https://www.ebi.ac.uk/gwas/rest/api/associations/{aid}/study',timeout=15))
  pub=d.get('publicationInfo') or {}
  return aid,d.get('accessionId'),pub.get('pubmedId'),a.get('pvalue'),d.get('diseaseTrait',{}).get('trait')
 except Exception as e:return aid,'ERR',repr(e),'',''
with concurrent.futures.ThreadPoolExecutor(max_workers=12) as ex:
 for z in ex.map(f,x['_embedded']['associations']):
  # all, compact
  print('\\t'.join(map(str,z)),flush=True)
PY"
