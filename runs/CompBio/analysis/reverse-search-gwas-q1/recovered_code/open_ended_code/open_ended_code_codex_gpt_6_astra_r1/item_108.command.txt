/bin/bash -lc "cat > ot_all.py <<'PY'
import requests,json,concurrent.futures
u='https://api.platform.opentargets.org/api/v4/graphql'
ids=['1_161530617_CTT_C','1_161531006_C_G','17_16939677_G_A','4_3445429_C_T','6_32067917_C_T']
def f(i):
 q='query {variant(variantId:\"'+i+'\"){id rsIds credibleSets(page:{index:0,size:500},studyTypes:[gwas]) {count rows {studyId pValueMantissa pValueExponent variant{id} study{id traitFromSource pubmedId publicationTitle nSamples}}}}}'
 r=requests.post(u,json={'query':q},timeout=30);open('ot_all_'+i+'.json','w').write(r.text)
 j=r.json();v=j.get('data',{}).get('variant')
 if not v:print(j);return
 print('COUNT',i,v['credibleSets']['count'])
 for a in v['credibleSets']['rows']:
  if a['variant']['id']==i or (i.startswith('1_') and a['pValueExponent'] < -17):print(i,a,flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as e:list(e.map(f,ids))
PY
python -u ot_all.py"
