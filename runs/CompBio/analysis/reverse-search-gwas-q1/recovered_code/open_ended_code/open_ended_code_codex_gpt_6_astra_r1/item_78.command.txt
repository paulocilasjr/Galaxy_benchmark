/bin/bash -lc "python - <<'PY'
import requests,re,html,concurrent.futures
urls=['https://pmc.ncbi.nlm.nih.gov/articles/PMC7867639/','https://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/GCST90019001-GCST90020000/GCST90019522/']
def f(u):
 r=requests.get(u,timeout=30);open('sinnott.html' if 'PMC' in u else 'GCST90019522_dir.html','w').write(r.text)
 print(u,r.status_code)
 if 'PMC' in u:
  s=html.unescape(re.sub('<["'^>]*>'"',' ',r.text));s=re.sub(r'\\s+',' ',s);open('sinnott.txt','w').write(s)
  for q in ['Data availability','summary statistics','non-British','replication','PLINK']:
   p=s.find(q);print(q,s[max(0,p-100):p+1800] if p>=0 else '')
 else:print(r.text[:5500])
with concurrent.futures.ThreadPoolExecutor() as e:list(e.map(f,urls))
PY"
