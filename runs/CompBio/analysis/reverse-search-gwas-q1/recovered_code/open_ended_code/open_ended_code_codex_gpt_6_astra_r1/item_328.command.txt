/bin/bash -lc "python - <<'PY'
import requests,concurrent.futures,gzip
pops=['white_other','irish','south_asian','black','chinese']
def f(p):
 u=f'https://margoliash-et-al-2023.s3.amazonaws.com/associations/{p}_total_protein_snp_gwas_results.tab.gz'
 try:
  r=requests.get(u,stream=True,timeout=30);print(p,r.status_code,r.headers.get('Content-Length'),flush=True)
  if r.ok:
   z=gzip.GzipFile(fileobj=r.raw)
   for i in range(5):print(p,z.readline().decode().strip(),flush=True)
  r.close()
 except Exception as e:print(p,e,flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as e:list(e.map(f,pops))
PY"
