import zipfile,xml.etree.ElementTree as E,re,sys
for f in sys.argv[1:]:
 z=zipfile.ZipFile(f);root=E.fromstring(z.read('xl/sharedStrings.xml'));strings=[''.join(t.itertext()) for t in root];print(f,'STRINGS',len(strings),flush=True)
 targets={str(i):s for i,s in enumerate(strings) if any(t in s for t in ['rs34562254','rs7554873','rs72717048','rs10419198','rs4690098'])};print('TARGETS',targets,flush=True)
 for name in z.namelist():
  if name.startswith('xl/worksheets/sheet') and name.endswith('.xml'):
   s=z.read(name).decode()
   for pattern in [r'1\\.4221\\d*[Ee]-16',r'7\\.81024\\d*[Ee]-23',r'9\\.31382\\d*[Ee]-23']:
    for m in re.finditer(pattern,s):print('EXACT',name,s[m.start()-70:m.start()+100],flush=True)
   hits=0
   for row in re.finditer(r'<row\\b["'^>]*>.*?</row>'"',s):

    if not any('t=\"s\"><v>'+i+'</v>' in text or re.search(r't=\"s\"["'^>]*><v>'"'+i+'</v>',text) for i in targets):continue
    vals=[]
    for cell in re.finditer(r'<c\\b(["'^>]*)>(.*?)</c>'"',text):
     v=re.search(r'<v>(.*?)</v>',cell[2]);val=v[1] if v else ''
     if 't=\"s\"' in cell[1] and val:val=strings[int(val)]
     vals.append(val)
    print('HIT',name,vals,flush=True);hits+=1
   print('DONE',name,hits,flush=True)
PY
