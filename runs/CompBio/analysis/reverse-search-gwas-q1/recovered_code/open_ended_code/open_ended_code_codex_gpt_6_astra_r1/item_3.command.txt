/bin/bash -lc "python - <<'PY'
import gzip,heapq,json
h=[];n=0; sig=[]
with gzip.open('reverse.search.gwas.q1.tsv.gz','rt') as f:
 print(next(f).strip())
 for l in f:
  n+=1
  a=l.rstrip().split('\\t')
  try:p=float(a[-1])
  except:continue
  t=(-p,n,a)
  if len(h)<100:heapq.heappush(h,t)
  elif t>h[0]:heapq.heapreplace(h,t)
  if p<5e-8:sig.append(a)
print('N',n,'sig',len(sig))
for t in sorted(h,reverse=True)[:50]:print('\\t'.join(t[2]))
with open('significant.json','w') as f:json.dump(sig,f)
PY"
