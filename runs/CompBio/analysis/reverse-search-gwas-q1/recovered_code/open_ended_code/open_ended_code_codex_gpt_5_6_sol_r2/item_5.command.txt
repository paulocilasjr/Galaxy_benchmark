/bin/bash -lc "python - <<'PY'
rows=[]
with open('hits_1e-6.tsv') as f:
    for line in f:
        a=line.rstrip().split('\\t'); rows.append((int(a[0]),int(a[1]),float(a[7]),a[2]))
rows.sort()
clusters=[]
for x in rows:
    if not clusters or x[0]"'!=clusters[-1][-1][0] or x[1]-clusters[-1][-1][1]>1_000_000:
        clusters.append([x])
    else: clusters[-1].append(x)
mins=sorted((min(c,key=lambda x:x[2]),len(c),c[0][1],c[-1][1]) for c in clusters)
mins.sort(key=lambda z:z[0][2])
for lead,n,start,end in mins:
    print(*lead, '"'n='+str(n), f'range={start}-{end}', sep='\\t')
PY"
