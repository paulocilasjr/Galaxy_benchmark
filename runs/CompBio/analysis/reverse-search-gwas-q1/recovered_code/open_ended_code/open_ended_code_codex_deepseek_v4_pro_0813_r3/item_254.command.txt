/bin/bash -lc "cat > otg_phewas.py <<'PY'
import requests, json
url='https://api.genetics.opentargets.org/graphql'
variants=['1_161531006_C_G','1_161540726_C_G','1_161530615_T_A','6_32067917_C_T','17_16939677_G_A','4_3445429_C_T','2_27508073_T_C']
query='''query phe("'$id:String!){ pheWAS(variantId:$id){ totalGWASStudies associations{ studyId pval beta nTotal traitReported study{ traitReported pmid pubDate pubAuthor } } } }'"'''
for v in variants:
 payload={'query':query,'variables':{'id':v}}
 try:
  r=requests.post(url,json=payload,timeout=30)
  print('\\n###',v,r.status_code)
  data=r.json()
  if 'errors' in data: print(data['errors'])
  phe=data.get('data',{}).get('pheWAS') or {}
  print('total',phe.get('totalGWASStudies'))
  for a in phe.get('associations',[])[:30]:
   print(a.get('studyId'),a.get('pval'),a.get('traitReported'),(a.get('study') or {}).get('pmid'))
 except Exception as e: print('ERR',v,e)
PY
python3 otg_phewas.py"
