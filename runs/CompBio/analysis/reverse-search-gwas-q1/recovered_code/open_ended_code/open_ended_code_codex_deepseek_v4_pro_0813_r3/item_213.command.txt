  rs=[]

   if cid.startswith('rs'): rs.append(cid)
  print(f'{ch}:{pos}:{ref}:{alt}\\t{\",\".join(rs[:5])}')
 except Exception as e:
  print(f'{ch}:{pos}:{ref}:{alt}\\tERR {e}')
 time.sleep(.05)
