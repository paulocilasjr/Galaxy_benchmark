/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, urllib.parse, time
# VEP REST GET? Use POST JSON
server='https://rest.ensembl.org'
vars=['1 161530617 CTT C','1 161530615 T A','1 161642443 T C','6 32067917 C T','4 3445429 C T','17 16939677 G A','13 108308037 T A','2 111471251 T C','19 49525049 CCTGCTG C','4 102486185 C T','2 27508073 T C','6 29686923 C T','6 30355154 GT G','6 30844983 G GTTT','6 31347929 G A','6 32668221 G T','6 32932606 CTT C','6 33546790 T A']
# use hgvsp? Let's use VEP region format: chr:pos pos allele
for v in vars:
 ch,pos,ref,alt=v.split()
 # VEP POST can accept variant string in hgvs notation; build simple for SNV/indel: chr:g.posref>alt
 hgvs=f\"{ch}:g.{pos}{ref}>{alt}\"
 data={'hgvs_notations':[hgvs], 'variant_class':True,'transcript_id':[]}
 req=urllib.request.Request(server+'/vep/human/hgvs', data=json.dumps(data).encode(), headers={'Content-Type':'application/json','Accept':'application/json'})
 try:
  resp=urllib.request.urlopen(req,timeout=30).read().decode()
  j=json.loads(resp)[0]
  conseq=[]; genes=set()
  for tc in j.get('transcript_consequences',[]):
   if tc.get('gene_symbol'): genes.add(tc['gene_symbol'])
   if tc.get('consequence_terms'): conseq+=tc['consequence_terms']
  print(v, 'genes=',','.join(sorted(genes)[:8]), 'top_cons=', ','.join(dict.fromkeys(conseq[:4])))
 except Exception as e:
  print('ERR',v,e)
 time.sleep(.1)
PY"
