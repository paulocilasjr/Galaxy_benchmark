/bin/bash -lc "python3 - <<'PY'
import os, glob, numpy as np
for dirname in ['bins10000','bins100k']:
 print('\\n',dirname)
 for f in sorted(glob.glob(dirname+'/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0])):
  i=int(os.path.basename(f).split('.')[0]); vals=[]
  for line in open(f):
   k,v=line.rstrip('\\n').split('\\t')
   if '_' in k or 'random' in k: continue
   vals.append(int(v))
  a=np.array(vals,float); total=a.sum(); a.sort(); desc=a[::-1]
  def frac(n): return desc[:max(1,int(len(desc)*n))].sum()/total
  # occupied bins and high-count quantiles
  p=a/total
  entropy=-(p*np.log(p)).sum()/np.log(len(a))
  print(i, 'bins',len(a), 'top0.01%',f'{frac(0.0001):.3f}', 'top0.1%',f'{frac(0.001):.3f}', 'top1%',f'{frac(0.01):.3f}', 'q99',np.percentile(a,99), 'max',int(desc[0]), 'entropy',f'{entropy:.3f}')
PY"
