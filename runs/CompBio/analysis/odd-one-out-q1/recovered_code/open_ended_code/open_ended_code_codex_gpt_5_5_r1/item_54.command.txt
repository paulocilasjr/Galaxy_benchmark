/bin/bash -lc "python3 - <<'PY'
import gzip, os, bisect
from collections import Counter
MAXD=1000
for i in range(1,11):
    plus=Counter(); minus=Counter(); n=0
    with gzip.open(f'odd.one.out.q1/{i}.tagAlign.gz','rt') as fh:
        for line in fh:
            parts=line.split('\\t')
            if parts[0] "'!= '"'chr1':
                if n and parts[0] "'!= '"'chr1':
                    # files sorted; after chr1 ends, stop
                    if parts[0] "'!= '"'chr1': break
                continue
            n+=1
            pos=int(parts[1])
            if parts[5].strip()=='+': plus[pos]+=1
            else: minus[pos]+=1
    ppos=sorted(plus); mpos=sorted(minus)
    hist=[0]*(MAXD+1)
    for p in ppos:
        a=bisect.bisect_left(mpos,p)
        b=bisect.bisect_right(mpos,p+MAXD)
        pc=min(plus[p],3)
        for m in mpos[a:b]:
            hist[m-p]+=pc*min(minus[m],3)
    # smooth and find top local peaks excluding first few bp
    sm=[]
    for d in range(MAXD+1):
        lo=max(0,d-2); hi=min(MAXD,d+2)
        sm.append(sum(hist[lo:hi+1])/(hi-lo+1))
    peaks=sorted(range(5,MAXD+1), key=lambda d: sm[d], reverse=True)[:10]
    # selected distances
    selected=[0,5,9,20,38,50,80,100,150,200,250,300,400,500]
    print('SAMPLE',i,'chr1_reads',n,'plus_unique',len(ppos),'minus_unique',len(mpos))
    print('top', ' '.join(f'{d}:{sm[d]:.0f}' for d in peaks[:8]))
    print('sel', ' '.join(f'{d}:{hist[d]}' for d in selected))
PY"
