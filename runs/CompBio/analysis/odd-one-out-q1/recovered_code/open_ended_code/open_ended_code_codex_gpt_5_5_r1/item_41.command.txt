/bin/bash -lc "python3 - <<'PY'
import os, glob, re
import numpy as np
primary=re.compile(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y|M):'"')
for dirname in ['bins10000','bins100k','bins1000000']:
    files=sorted(glob.glob(dirname+'/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0]))
    keys=set(); vals=[]
    for f in files:
        d={}
        with open(f) as fh:
            for line in fh:
                parts=line.rstrip('\\n').split('\\t')
                if len(parts)"'!=2: continue
                k,v=parts
                if not primary.match(k): continue
                d[k]=int(v); keys.add(k)
        vals.append(d)
    keys=sorted(keys)
    X=np.array([[d.get(k,0) for k in keys] for d in vals], dtype=float)
    print('"'\\nDIR', dirname, 'shape', X.shape)
    transforms=[('raw',X),('log1p',np.log1p(X)),('sqrt',np.sqrt(X)),('binary',(X>0).astype(float))]
    cap=X.copy()
    for r in range(cap.shape[0]):
        nz=cap[r][cap[r]>0]
        q=np.percentile(nz,99.5)
        cap[r]=np.minimum(cap[r], q)
    transforms.append(('cap995_log1p',np.log1p(cap)))
    for name,Y in transforms:
        C=np.corrcoef(Y)
        avgs=[(row.sum()-1)/(len(row)-1) for row in C]
        order=np.argsort(avgs)+1
        print(name, 'avg_order', ' '.join(f'{i}:{avgs[i-1]:.4f}' for i in order))
PY"
