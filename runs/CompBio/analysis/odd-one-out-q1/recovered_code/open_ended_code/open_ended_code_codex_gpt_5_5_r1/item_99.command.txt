/bin/bash -lc "python3 - <<'PY'
import gzip, os, glob, bisect, re, numpy as np
from collections import defaultdict
classes=['PLS','pELS','dELS','DNase-H3K4me3','CTCF-only']
tmp={c:defaultdict(list) for c in classes}
with gzip.open('refs/ENCFF924IMH.bed.gz','rt') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  chrom,s,e=p[0],int(p[1]),int(p[2]); base=p[9].split(',')[0]
  if base in tmp and re.match(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y)$'"', chrom): tmp[base][chrom].append((s,e))
ints={}
for c,d in tmp.items():
 ints[c]={}
 for chrom,arr in d.items():
  arr=sorted(arr); m=[]
  for s,e in arr:
   if not m or s>m[-1][1]: m.append([s,e])
   elif e>m[-1][1]: m[-1][1]=e
  ints[c][chrom]=([x[0] for x in m],[x[1] for x in m])
def ov(c,chrom,s,e):
 if chrom not in ints[c]: return False
 starts,ends=ints[c][chrom]; i=bisect.bisect_right(ends,s); return i<len(starts) and starts[i]<e
primary=re.compile(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y|M):'"')
# precompute bin class masks per key encountered
for mode in ['raw','cap100','cap_q99','binary']:
 print('\\nMODE',mode)
 print('sample\\tPLS\\tpELS\\tdELS\\tDNase-H3K4me3\\tCTCF-only\\tany\\tCTCF_over_PLS')
 for bf in sorted(glob.glob('bins10000/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0])):
  rows=[]
  for line in open(bf):
   key,v=line.rstrip('\\n').split('\\t'); v=int(v)
   if primary.match(key): rows.append((key,v))
  vals=np.array([v for _,v in rows])
  if mode=='cap100': weights=np.minimum(vals,100)
  elif mode=='binary': weights=(vals>0).astype(int)
  elif mode=='cap_q99': weights=np.minimum(vals,np.percentile(vals,99))
  else: weights=vals
  totals={c:0.0 for c in classes}; anytot=0.0; total=float(weights.sum())
  for (key,_),w in zip(rows,weights):
   chrom,b=key.split(':'); s=int(b)*10000; e=s+10000; hit=False
   for c in classes:
    if ov(c,chrom,s,e): totals[c]+=w; hit=True
   if hit: anytot+=w
  sample=os.path.basename(bf).split('.')[0]
  pls=totals['PLS']/total; ctcf=totals['CTCF-only']/total
  print(sample, *(f'{totals[c]/total:.4f}' for c in classes), f'{anytot/total:.4f}', f'{ctcf/(pls+1e-9):.2f}', sep='\\t')
PY"
