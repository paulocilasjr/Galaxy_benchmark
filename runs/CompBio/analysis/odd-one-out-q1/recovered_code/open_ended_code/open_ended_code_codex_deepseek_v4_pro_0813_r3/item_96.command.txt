/bin/bash -lc "python3 - <<'PY'
import glob,os,collections,numpy as np
files=sorted(glob.glob('genewincov/*.tsv'), key=lambda x:int(os.path.basename(x).split('.')[0]))
profiles={}
for f in files:
    n=os.path.basename(f).split('.')[0]
    sums=collections.defaultdict(float); cnt=collections.defaultdict(int)
    with open(f) as fh:
        for line in fh:
            p=line.split(); key=p[3]; val=int(p[6]); sums[key]+=val; cnt[key]+=1
    up=np.array([sums[f'up_{i}']/cnt[f'up_{i}'] if cnt[f'up_{i}'] else 0 for i in range(50)])
    body=np.array([sums[f'body_{i}']/cnt[f'body_{i}'] if cnt[f'body_{i}'] else 0 for i in range(100)])
    down=np.array([sums[f'down_{i}']/cnt[f'down_{i}'] if cnt[f'down_{i}'] else 0 for i in range(50)])
    profiles[n]=(up,body,down)
    print(n,'up mean',round(up.mean(),3),'body mean',round(body.mean(),3),'down mean',round(down.mean(),3),
          'body5p',round(body[:5].mean(),3),'body3p',round(body[-5:].mean(),3),
          'prom/body',round(up[-5:].mean()/body.mean(),2))
np.savez('metagene_profiles.npz', **{n:{'up':profiles[n][0],'body':profiles[n][1],'down':profiles[n][2]} for n in profiles})
PY"
