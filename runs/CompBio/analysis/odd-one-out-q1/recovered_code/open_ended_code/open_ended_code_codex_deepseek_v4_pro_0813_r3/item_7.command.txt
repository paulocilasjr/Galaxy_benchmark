/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections, statistics
files=sorted(glob.glob('odd/odd.one.out.q1/*.tagAlign.gz'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    uniq=set(); counts=collections.Counter(); n=0
    # sample/use streaming unique starts maybe 10M ok; track duplicate occupancy only max? use all starts
    with gzip.open(f,'rt') as fh:
        for line in fh:
            p=line.rstrip('\\n').split('\\t')
            key=(p[0],int(p[1]),p[5])
            counts[key]+=1; uniq.add(key); n+=1
    dups=sum(c-1 for c in counts.values() if c>1)
    maxc=max(counts.values())
    # top duplicate coordinates
    top=counts.most_common(5)
    print(os.path.basename(f), 'n',n,'unique keys',len(uniq),'dups',dups,'max',maxc,'top',top)
PY"
