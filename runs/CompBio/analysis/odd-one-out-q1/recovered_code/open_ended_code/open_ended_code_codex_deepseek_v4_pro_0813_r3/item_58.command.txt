/bin/bash -lc "python3 - <<'PY'
import gzip, collections, os
for name in ['4','8','1','9']:
    f=f'odd/odd.one.out.q1/{name}.tagAlign.gz'
    c=collections.Counter(); hist=collections.Counter()
    with gzip.open(f,'rt') as fh:
        for line in fh:
            p=line.rstrip('\\n').split('\\t')
            c[(p[0],int(p[1]),p[5])]+=1
    for k,v in c.items(): hist[v]+=1
    print(name, 'unique keys',len(c), 'duplicate fraction', 1-len(c)/sum(c.values()))
    for mult in range(1,11):
        print(mult, hist.get(mult,0), end='; ')
    print('>10', sum(v for k,v in hist.items() if k>10))
PY"
