/bin/bash -lc "python3 - <<'PY'
import glob,os
files=sorted(glob.glob('genecov/*.tsv'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    name=os.path.basename(f).split('.')[0]
    tot=0; n=0
    with open(f) as fh:
        for line in fh:
            p=line.split(); count=int(p[6]); tot+=count; n+=1
    print(name,'reads_in_genes',tot,'mean_per_gene',round(tot/n,1),'frac_of_10M',round(tot/1e6,1))
PY"
