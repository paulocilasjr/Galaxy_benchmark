/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections
files=sorted(glob.glob('odd/odd.one.out.q1/*.tagAlign.gz'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    c=collections.Counter()
    with gzip.open(f,'rt') as fh:
        for line in fh:
            p=line.rstrip('\\n').split('\\t')
            # cluster by 100bp? use exact start for top
            c[(p[0],int(p[1]),p[5])]+=1
    print('\\n',os.path.basename(f))
    for (chrom,start,strand),cnt in c.most_common(15):
        print(chrom,start,strand,cnt)
PY"
