/bin/bash -lc "python3 - <<'PY'
import numpy as np, glob, os
files=sorted(glob.glob('tsscov/*.tsv'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    n=os.path.basename(f).split('.')[0]
    sums=np.zeros(100); cnt=np.zeros(100)
    with open(f) as fh:
        for line in fh:
            p=line.split(); rel=int(p[3]); val=int(p[4])
            idx=(rel+5000)//100
            if 0<=idx<100:
                sums[idx]+=val; cnt[idx]+=1
    profile=sums/cnt
    maxv=profile.max(); maxidx=profile.argmax()
    half=maxv/2
    # find contiguous width around peak where >half
    left=maxidx
    while left>0 and profile[left-1]>=half: left-=1
    right=maxidx
    while right<99 and profile[right+1]>=half: right+=1
    width=(right-left+1)*100
    # shoulder ratio: peak vs flank +/-1kb
    flank=(profile[:40].mean()+profile[60:].mean())/2
    print(n,'max',round(maxv,3),'peakbin',(maxidx-50)*100,'FWHM',width,'peak/flank',round(maxv/flank,2))
PY"
