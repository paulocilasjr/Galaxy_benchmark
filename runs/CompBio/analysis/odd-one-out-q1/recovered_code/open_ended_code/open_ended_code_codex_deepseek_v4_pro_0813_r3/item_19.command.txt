import numpy as np, glob, os
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import AgglomerativeClustering
from scipy.spatial.distance import pdist, squareform
files=sorted(glob.glob('cov/*.counts'), key=lambda x:int(os.path.basename(x).split('.')[0]))
M=[]; names=[]
for f in files:
    a=np.loadtxt(f,usecols=3)
    M.append(a); names.append(os.path.basename(f).split('.')[0])
M=np.array(M)
L=np.log2(M+1)
X=StandardScaler().fit_transform(L.T).T
pca=PCA(n_components=3).fit(X)
print('var',pca.explained_variance_ratio_)
coords=pca.transform(X)
for n,c in zip(names,coords): print(n,c)
# pairwise euclidean on scaled log
D=squareform(pdist(X))
print('distance matrix')
print(names)
np.set_printoptions(precision=1,suppress=True,linewidth=200)
print(D)
# avg dist
for i,n in enumerate(names):
 vals=np.delete(D[i],i); print(n, vals.mean(), vals.min())
PY"
