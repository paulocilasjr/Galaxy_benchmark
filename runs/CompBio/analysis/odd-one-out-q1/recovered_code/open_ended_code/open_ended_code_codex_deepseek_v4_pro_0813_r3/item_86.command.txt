/bin/bash -lc "python3 - <<'PY'
import pyBigWig, numpy as np, glob, os, time
bw=pyBigWig.open('public/gm12878_dnase_rep1.bigWig')
vals=[]
with open('bins_10kb.bed') as f:
    for line in f:
        p=line.split(); c=p[0]; s=int(p[1]); e=int(p[2])
        try:
            r=bw.stats(c,s,e,exact=True)[0]
        except RuntimeError:
            r=None
        vals.append(0.0 if r is None else r)
vals=np.array(vals)
print('done', vals.mean(), (vals>0).mean())
np.save('public/gm12878_dnase_rep1_bins.npy', vals)
names=[]; Ms=[]
for f in sorted(glob.glob('cov/*.counts'), key=lambda x:int(os.path.basename(x).split('.')[0])):
    a=np.loadtxt(f,usecols=3); names.append(os.path.basename(f).split('.')[0]); Ms.append(a)
Ms=np.array(Ms)
for mode, arr in [('raw',Ms),('log',np.log2(Ms+1)),('sqrt',np.sqrt(Ms))]:
    print(mode)
    for i,n in enumerate(names):
        r=np.corrcoef(arr[i],vals)[0,1]
        print(n, round(r,3))
PY"
