/bin/bash -lc "python3 - <<'PY'
import numpy as np, glob, os
files=sorted(glob.glob('cov/*.counts'), key=lambda x:int(os.path.basename(x).split('.')[0]))
M=[]
names=[]
for f in files:
    a=[]
    with open(f) as fh:
        for line in fh:
            p=line.split()
            a.append(float(p[3]))
    v=np.array(a)
    M.append(v); names.append(os.path.basename(f).split('.')[0])
M=np.array(M)
print(names)
print('sum mean sd per sample:')
for i,n in enumerate(names):
    print(n, M[i].sum(), M[i].mean(), M[i].std(), np.percentile(M[i],99), M[i].max())
# pearson raw
R=np.corrcoef(M)
np.set_printoptions(precision=3,suppress=True,linewidth=200)
print('pearson raw')
print(names)
print(R)
# spearman approximate by ranking? skip maybe use scipy
try:
    from scipy.stats import spearmanr
    S=np.zeros((len(names),len(names)))
    for i in range(len(names)):
        for j in range(i,len(names)):
            rho,_=spearmanr(M[i],M[j]); S[i,j]=S[j,i]=rho
    print('spearman')
    print(S)
except Exception as e: print(e)
# log2(1+x) pearson
L=np.log2(M+1)
print('pearson log2')
print(np.corrcoef(L))
# pairwise maybe show avg correlation excluding self
for i,n in enumerate(names):
    vals=np.delete(R[i],i)
    print('avg pearson',n, vals.mean(), 'min', vals.min())
PY"
