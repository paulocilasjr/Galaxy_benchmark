/bin/bash -lc "python - <<'PY'
import numpy as np
X=np.stack([np.load(f'features/{i}.npy') for i in range(1,11)])
# exclude bins extreme in any track (likely artifacts), chrY effectively irrelevant
mask=(X.max(0)<500)
for n in [1000,3000,10000,30000,100000]:
 B=[]
 for x in X:
  xx=x.copy(); xx[~mask]=0
  ix=np.argpartition(xx,-n)[-n:]
  B.append(set(ix))
 J=np.array([[len(B[i]&B[j])/len(B[i]|B[j]) for j in range(10)] for i in range(10)])
 print('\\nn',n)
 print(np.array2string(J,precision=2,suppress_small=True))
 print('means',np.round((J.sum(1)-1)/9,3))
PY"
