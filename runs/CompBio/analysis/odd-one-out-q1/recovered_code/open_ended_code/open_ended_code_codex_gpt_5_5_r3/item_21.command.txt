/bin/bash -lc "python3 - <<'PY'
import gzip, os, collections
for idx in [8,9,10]:
    path=f'odd.one.out.q1/{idx}.tagAlign.gz'
    canon=set([f'chr{i}' for i in range(1,23)]+['chrX','chrY'])
    for bs in [1000,10000]:
        bins=collections.Counter(); strand_bins=collections.Counter(); total=0
        with gzip.open(path,'rt') as fh:
            for line in fh:
                c,s,e,_,_,st=line.rstrip('\\n').split('\\t')[:6]
                if c not in canon: continue
                k=(c,int(s)//bs)
                bins[k]+=1; strand_bins[(k,st)]+=1; total+=1
        vals=sorted(bins.values(), reverse=True)
        den=num=nbin=0
        for k,n in bins.items():
            if n>=20:
                p=strand_bins[(k,'+')]; m=strand_bins[(k,'-')]
                num += abs(p-m); den += n; nbin += 1
        print(idx, 'bin', bs, 'top1%',round(100*vals[0]/total,3),'top10%',round(100*sum(vals[:10])/total,3),'top100%',round(100*sum(vals[:100])/total,3),'top1000%',round(100*sum(vals[:1000])/total,3),'localStrandImb%',round(100*num/den,2),'nbin>=20',nbin)
PY"
