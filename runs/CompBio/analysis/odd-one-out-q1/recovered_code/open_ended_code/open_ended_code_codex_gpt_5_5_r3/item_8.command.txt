/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections, statistics
for path in sorted(glob.glob('odd.one.out.q1/*.tagAlign.gz'), key=lambda p:int(os.path.basename(p).split('.')[0])):
    total=0; chrom=collections.Counter(); strand=collections.Counter(); lens=collections.Counter(); dup_sample=collections.Counter(); first=0
    prev=None; sorted_ok=True; dup_adj=0
    for line in gzip.open(path,'rt'):
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<6: continue
        c,s,e,_,score,st=parts[:6]; s=int(s); e=int(e)
        total+=1; chrom[c]+=1; strand[st]+=1; lens[e-s]+=1
        key=(c,s,e,st)
        if total<=200000:
            dup_sample[key]+=1
        if prev and (c,s,e,st) < prev: sorted_ok=False
        if prev==key: dup_adj+=1
        prev=key
    dup_reads=sum(v-1 for v in dup_sample.values())
    print(os.path.basename(path), 'total', total, 'chrM%', round(100*chrom['chrM']/total,3), 'chrY%', round(100*chrom['chrY']/total,3), 'plus%', round(100*strand['+']/total,2), 'len', lens.most_common(3), 'sampledup%', round(100*dup_reads/min(total,200000),2), 'adjdup%', round(100*dup_adj/total,2), 'nchrom', len(chrom), 'sorted', sorted_ok)
    print(' topchrom', chrom.most_common(8))
PY"
