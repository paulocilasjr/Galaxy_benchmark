/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections, math
paths=sorted(glob.glob('odd.one.out.q1/*.tagAlign.gz'), key=lambda p:int(os.path.basename(p).split('.')[0]))
canon=set([f'chr{i}' for i in range(1,23)]+['chrX','chrY'])
for bs in [1000,10000,100000]:
    print('BIN',bs)
    for path in paths:
        bins=collections.Counter(); strand_bins=collections.Counter(); total=0; total_c=0
        with gzip.open(path,'rt') as fh:
            for line in fh:
                c,s,e,_,_,st=line.rstrip('\\n').split('\\t')[:6]
                if c not in canon: continue
                k=(c,int(s)//bs)
                bins[k]+=1; strand_bins[(k,st)]+=1; total+=1
        vals=sorted(bins.values(), reverse=True)
        top1=vals[0]/total; top10=sum(vals[:10])/total; top100=sum(vals[:100])/total; top1000=sum(vals[:1000])/total
        # weighted local strand imbalance over bins with >=20 reads
        den=0; num=0; nbin=0
        for k,n in bins.items():
            if n>=20:
                p=strand_bins[(k,'+')]; m=strand_bins[(k,'-')]
                num += abs(p-m); den += n; nbin+=1
        print(os.path.basename(path).split('.')[0], 'top1%',round(100*top1,3),'top10%',round(100*top10,3),'top100%',round(100*top100,3),'top1000%',round(100*top1000,3),'localStrandImb%',round(100*num/den,2),'nbin>=20',nbin)
    print()
PY"
