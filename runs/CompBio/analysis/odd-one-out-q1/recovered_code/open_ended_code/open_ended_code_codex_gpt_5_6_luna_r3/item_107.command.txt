/bin/bash -lc "python3 - <<'PY'
import json,urllib.parse,urllib.request
base='https://www.encodeproject.org/search/?'
for assay in ['ChIP-seq','DNase-seq','ATAC-seq','RNA-seq','eCLIP-seq','MNase-seq','FAIRE-seq']:
 q={'type':'File','file_format':'tagAlign','assay_term_name':assay,'limit':5,'format':'json'}
 u=base+urllib.parse.urlencode(q)
 try:d=json.load(urllib.request.urlopen(u))
 except Exception as e: print(assay,e);continue
 print(assay,d.get('total'))
 for f in d.get('@graph',[])[:3]: print(' ',f.get('accession'),f.get('assembly'),f.get('assay_term_name'),f.get('file_size'),f.get('href'))
PY"
