/bin/bash -lc "python3 - <<'PY'
import numpy as np
from pathlib import Path
keys={}; fs=range(1,11)
for j in fs:
 for line in Path(f'bins10_{j}.tsv').open():
  c,k,n=line.split(); key=c+':'+k
  if key not in keys:keys[key]=np.zeros(10,dtype=float)
  keys[key][j-1]=int(n)
K=list(keys); X=np.stack(list(keys.values()))
# canonical only
canon=np.array([k.split(':')[0] in {f'chr{i}' for i in range(1,23)}|{'chrX','chrY'} for k in K])
X=X[canon]
# rank within each vector (simple argsort ranks)
def ranks(a):
 o=np.argsort(a,kind='stable'); r=np.empty(len(a)); r[o]=np.arange(len(a)); return r
for maskname, Y in [('all',X),('no_repeat_top',np.minimum(X,100)),('log',np.log1p(X))]:
 R=np.stack([ranks(Y[:,j]) for j in range(10)],1)
 C=np.corrcoef(R,rowvar=False)
 print(maskname,'mean',np.round((C.sum(0)-1)/9,3),'lowest',np.argmin((C.sum(0)-1)/9)+1)
 print(np.round(C,2))
# Distance to centroid in PCA / cosine
for trans in [lambda x:x,lambda x:np.log1p(x),lambda x:np.sqrt(x),lambda x:np.minimum(x,100)]:
 Y=trans(X); Y=(Y-Y.mean(0))/(Y.std(0)+1e-9); cen=Y.mean(1); dist=np.sqrt(((Y-cen[:,None])**2).mean(0)); print('dist',np.round(dist,3),'rank',np.argsort(dist)[::-1]+1)
PY"
