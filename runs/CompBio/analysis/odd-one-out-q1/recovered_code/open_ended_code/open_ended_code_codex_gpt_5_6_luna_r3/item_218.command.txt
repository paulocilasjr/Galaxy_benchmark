/bin/bash -lc "python3 - <<'PY'
import json,urllib.parse,urllib.request
for assay in ['TF ChIP-seq','Histone ChIP-seq']:
 q={'type':'File','file_format':'tagAlign','assay_title':assay,'limit':20,'format':'json'}
 d=json.load(urllib.request.urlopen('https://www.encodeproject.org/search/?'+urllib.parse.urlencode(q)))
 print(assay,d.get('total'))
 for f in d.get('@graph',[])[:10]:
  print(f.get('accession'),f.get('assembly'),f.get('file_size'),f.get('biosample_ontology',{}).get('term_name'),f.get('target'),f.get('href'))
PY"
