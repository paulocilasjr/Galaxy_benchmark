/bin/bash -lc "python3 - <<'PY'
import numpy as np
from pathlib import Path
files=[str(i) for i in range(1,11)]
keys={}
for j,b in enumerate(files):
 for line in Path(f'bins10_{b}.tsv').open():
  c,k,n=line.split(); key=c+':'+k
  if key not in keys: keys[key]=np.zeros(10)
  keys[key][j]=int(n)
X=np.stack(list(keys.values()))
# show pairwise Pearson for various per-chrom normalization? global log and rank
for name,Y in [('raw',X),('log',np.log1p(X)),('sqrt',np.sqrt(X))]:
 C=np.corrcoef(Y,rowvar=False)
 print(name)
 for i in range(10): print(i+1,' '.join(f'{C[i,j]:.3f}' for j in range(10)))
# chromosome-normalize: each chr total to same, log
chs=np.array([k.split(':')[0] for k in keys])
Y=np.log1p(X.copy())
for c in sorted(set(chs)):
 ix=chs==c
 # scale raw per chr total before log
 for j in range(10):
  s=X[ix,j].sum()
  Y[ix,j]=np.log1p(X[ix,j]/s*1e6)
C=np.corrcoef(Y,rowvar=False)
print('chr-normalized log')
for i in range(10):print(i+1,' '.join(f'{C[i,j]:.3f}' for j in range(10)))
PY"
