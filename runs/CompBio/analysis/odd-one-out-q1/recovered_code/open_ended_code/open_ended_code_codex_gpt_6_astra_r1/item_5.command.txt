/bin/bash -lc "cat > analyze.py <<'PY'
import gzip,collections,json,os
import numpy as np
from concurrent.futures import ProcessPoolExecutor

def run(i):
 bins={}; chroms=collections.Counter(); lens=collections.Counter(); strands=collections.Counter(); dup=0;prev=None
 with gzip.open(f'odd.one.out.q1/{i}.tagAlign.gz','rt') as h:
  for l in h:
   a=l.split();ch=a[0];p=int(a[1]);e=int(a[2]);st=a[5]
   if ch not in bins:bins[ch]=np.zeros(300000,dtype=np.int32)
   bins[ch][p//1000]+=1;chroms[ch]+=1;lens[e-p]+=1;strands[st]+=1
   key=(ch,p,e,st)
   if key==prev:dup+=1
   prev=key
 np.savez_compressed(f'bins_{i}.npz',**{k:v[:np.flatnonzero(v)[-1]+1] for k,v in bins.items()})
 stats=dict(index=i,n=sum(chroms.values()),chroms=dict(chroms),lens=dict(lens),strands=dict(strands),adjacent_duplicates=dup)
 with open(f'stats_{i}.json','w') as f:json.dump(stats,f)
 print(json.dumps(stats),flush=True)
if __name__=='__main__':
 with ProcessPoolExecutor(max_workers=4) as ex:list(ex.map(run,range(1,11)))
PY
OPENBLAS_NUM_THREADS=1 python3 analyze.py > statistics.log"
