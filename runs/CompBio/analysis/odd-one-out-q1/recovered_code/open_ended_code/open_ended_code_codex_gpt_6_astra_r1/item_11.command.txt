/bin/bash -lc "cat > compare.py <<'PY'
import numpy as np,gzip,json
from scipy.stats import spearmanr
chroms=['chr'+str(i) for i in range(1,23)]
data=[np.load(f'bins_{i}.npz') for i in range(1,11)]
sizes={c:max(len(d[c]) for d in data) for c in chroms}
arrays=[{c:np.pad(d[c],(0,sizes[c]-len(d[c]))) for c in chroms} for d in data]
x=np.array([np.concatenate([d[c] for c in chroms]) for d in arrays])
np.set_printoptions(precision=3,suppress=True,linewidth=200)
print('PEARSON log1p bin counts\\n',np.corrcoef(np.log1p(x)))
print('SPEARMAN\\n',spearmanr(x,axis=1).statistic)
print('concentration')
for i,v in enumerate(x):
 s=np.sort(v);n=len(s)
 print(i+1,'top 1% fraction',s[-n//100:].sum()/s.sum(),'nonzero',np.mean(v>0),'quantiles',np.quantile(v,[.5,.9,.99,.999]),'max',v.max())
tss={c:set() for c in chroms}; bodies={c:np.zeros(sizes[c],dtype=bool) for c in chroms};exons={c:np.zeros(sizes[c],dtype=bool) for c in chroms}
with gzip.open('refGene.txt.gz','rt') as f:
 for l in f:
  a=l.split();c=a[2]
  if c not in tss:continue
  st,en=int(a[4]),int(a[5]);t=st if a[3]=='+' else en
  tss[c].add((t//1000,1 if a[3]=='+' else -1))
  bodies[c][st//1000:en//1000+1]=True
  for st,en in zip(a[9].strip(',').split(','),a[10].strip(',').split(',')):exons[c][int(st)//1000:int(en)//1000+1]=True
print('TSS mean signal at offsets -10kb to 10kb')
for i,d in enumerate(arrays):
 profile=np.zeros(21);n=0
 for c in chroms:
  for p,st in tss[c]:
   idx=p+st*np.arange(-10,11)
   if idx.min()<0 or idx.max()>=sizes[c]:continue
   profile+=d[c][idx];n+=1
 print(i+1,profile/n,'gene_enrich',np.mean(np.concatenate([d[c][bodies[c]] for c in chroms]))/x[i].mean(),'exon_enrich',np.mean(np.concatenate([d[c][exons[c]] for c in chroms]))/x[i].mean())
PY"
