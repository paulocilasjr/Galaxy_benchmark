/bin/bash -c "python3 - <<'PY'
import requests
for assay in ['ATAC-seq','DNase-seq','Histone ChIP-seq','TF ChIP-seq']:
 url='https://www.encodeproject.org/search/?type=File&file_format=bed&file_type=bed%20narrowPeak&assembly=GRCh38&dataset.assay_title='+requests.utils.quote(assay)+'&format=json&limit=5'
 r=requests.get(url,timeout=60); print(assay,r.status_code,len(r.json().get('@graph',[])))
 data=r.json().get('@graph',[])
 for f in data[:3]: print(f.get('accession'),f.get('dataset'))
PY"
