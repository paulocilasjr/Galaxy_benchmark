import glob, os, numpy as np
for f in sorted(glob.glob('macs_peaks2/*_peaks.narrowPeak'), key=lambda x:int(os.path.basename(x).split('_')[0])):

 print(b,'peaks',len(a),'width_med',np.median(widths),'width_mean',np.mean(widths),'q25',np.quantile(widths,.25),'q75',np.quantile(widths,.75),'sig_med',np.median(sig),'summit_len',np.mean(a[:,3]))
PY"
