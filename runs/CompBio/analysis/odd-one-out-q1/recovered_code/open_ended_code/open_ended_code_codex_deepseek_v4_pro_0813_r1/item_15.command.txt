/bin/bash -c "python3 - <<'PY'
import numpy as np, glob, re, os
from scipy.stats import spearmanr
files=glob.glob('counts_*.bed')
files.sort(key=lambda x:int(re.search(r'counts_(\\d+)\\.bed',x).group(1)))
mats=[]
for f in files:
    a=np.loadtxt(f, delimiter='\\t', usecols=3, dtype=float)
    mats.append(a)
X=np.vstack(mats).T
print(X.shape, X.sum(axis=0))
# normalize per total and maybe log? Pearson raw counts; Spearman rank
n=len(files)
for idx in ['pearson','spearman','pearson_norm','cosine']:
 print('\\n',idx)
 C=np.zeros((n,n))
 for i in range(n):
  for j in range(n):
   if idx=='pearson': c=np.corrcoef(X[:,i],X[:,j])[0,1]
   elif idx=='spearman':
    # scipy rank per pair too slow? n 30k fine
    from scipy.stats import spearmanr
    c=spearmanr(X[:,i],X[:,j])[0]
   elif idx=='pearson_norm':
    c=np.corrcoef(X[:,i]/X[:,i].sum(), X[:,j]/X[:,j].sum())[0,1]
   else:
    a=X[:,i]; b=X[:,j]; c=np.dot(a,b)/np.linalg.norm(a)/np.linalg.norm(b)
   C[i,j]=c
 names=[os.path.basename(f).replace('counts_','').replace('.bed','') for f in files]
 print('\\t'.join(names))
 for i in range(n):
  print(names[i]+'\\t'+'\\t'.join(f'{C[i,j]:.3f}' for j in range(n)))
PY"
