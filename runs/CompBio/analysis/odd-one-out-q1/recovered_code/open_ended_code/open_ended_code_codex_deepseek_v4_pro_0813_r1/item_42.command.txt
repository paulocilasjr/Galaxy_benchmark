/bin/bash -c "python3 - <<'PY'
import numpy as np, glob, re, os
from scipy.stats import skew, kurtosis
# load cCRE and gene counts
D={}
for fname,anns in [('feature_counts.tsv',['genes','tss','exons','introns']),('ccre_counts.tsv',['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3'])]:
 with open(fname) as f:
  for line in f:
   cat,b,n=line.split(); D[(cat,int(b))]=int(n)
# lengths
lengths={}
for cat in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']:
 lengths[cat]=sum(int(l.split('\\t')[2])-int(l.split('\\t')[1]) for l in open('ccre_'+cat+'_lex.bed'))
genome=sum(int(line.split()[1]) for line in open('hg38.chrom.sizes') if not line.startswith('_'))
# tss profile moments from tss_counts files
features={}
for b in range(1,11):
 f=f'tss_counts_{b}.bed'
 a=np.loadtxt(f,usecols=6,dtype=float).reshape(-1,100)
 prof=a.mean(0)
 # normalized to mean 1
 p=prof/prof.mean()
 features[b]={
 'gene_frac':D[('genes',b)]/1e7,
 'tss_frac':D[('tss',b)]/1e7,
 'exon_frac':D[('exons',b)]/1e7,
 'intron_frac':D[('introns',b)]/1e7,
 'tss_over_gene':D[('tss',b)]/D[('genes',b)],
 'intron_over_gene':D[('introns',b)]/D[('genes',b)],
 'tss_peak':p[40:61].mean(),
 'tss_flank':(p[0:20].mean()+p[80:100].mean())/2,
 'tss_enrich':p[40:61].mean()/max(1e-9,(p[0:20].mean()+p[80:100].mean())/2),
 'tss_width':np.sum(p>2)/100,
 'tss_skew':skew(prof),
 }
 for cat in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']:
  features[b]['enr_'+cat]=(D[(cat,b)]/lengths[cat])/(1e7/genome)
# 100kb count moments
for b in range(1,11):
 a=np.loadtxt(f'counts_{b}.bed',usecols=3,dtype=float)
 p=a/a.sum()
 features[b]['bins_zero']=np.mean(a==0)
 features[b]['bins_max']=a.max()/a.sum()
 features[b]['bins_top1']=np.sort(p)[::-1][:int(len(p)*0.01)].sum()
 features[b]['bins_top5']=np.sort(p)[::-1][:int(len(p)*0.05)].sum()
 p2=p[p>0]; features[b]['entropy']=-np.sum(p2*np.log(p2))/np.log(len(p))
 features[b]['skew']=skew(a); features[b]['kurt']=kurtosis(a)
 # chromosome concentration HHI across chroms? skip
keys=['gene_frac','tss_frac','exon_frac','intron_frac','tss_over_gene','intron_over_gene','tss_peak','tss_flank','tss_enrich','tss_width','tss_skew'] + [f'enr_{c}' for c in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']] + ['bins_zero','bins_max','bins_top1','bins_top5','entropy','skew','kurt']
X=np.array([[features[b][k] for k in keys] for b in range(1,11)])
print('keys',keys)
# standardize
Z=(X-X.mean(0))/X.std(0)
from sklearn.decomposition import PCA
pca=PCA(n_components=min(6,len(keys)))
P=pca.fit_transform(Z)
print('explained',pca.explained_variance_ratio_)
print('PC scores')
for b in range(1,11): print(b,' '.join(f'{x:.2f}' for x in P[b-1]))
print('distances from centroid')
for b in range(1,11): print(b, np.linalg.norm(Z[b-1]))
# correlation of standardized features (just visual)
PY"
