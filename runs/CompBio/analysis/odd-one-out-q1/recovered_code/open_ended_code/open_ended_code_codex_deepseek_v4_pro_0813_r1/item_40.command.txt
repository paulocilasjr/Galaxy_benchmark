/bin/bash -c "python3 - <<'PY'
import numpy as np
D={}
with open('ccre_counts.tsv') as f:
 for line in f:
  cat,b,n=line.split(); D[(cat,int(b))]=int(n)
lengths={}
for cat in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']:
 fn='ccre_'+cat+'_lex.bed'; total=sum(int(line.split('\\t')[2])-int(line.split('\\t')[1]) for line in open(fn))
 lengths[cat]=total
genome=sum(int(line.split()[1]) for line in open('hg38.chrom.sizes') if not line.startswith('_'))
X=[]
for b in range(1,11):
 X.append([D[(cat,b)]/lengths[cat]/(1e7/genome) for cat in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']])
X=np.array(X)
# zscore log
Z=np.log(X); Z=(Z-Z.mean(0))/Z.std(0)
print('log enrich zscore')
print('\\t'.join(str(i) for i in range(1,11)))
for i in range(10): print(i+1,'\\t'.join(f'{x:.2f}' for x in Z[i]))
print('corr')
C=np.corrcoef(X)
for i in range(10): print(i+1,'\\t'.join(f'{x:.2f}' for x in C[i]))
print('corr log')
C=np.corrcoef(Z)
for i in range(10): print(i+1,'\\t'.join(f'{x:.2f}' for x in C[i]))
PY"
