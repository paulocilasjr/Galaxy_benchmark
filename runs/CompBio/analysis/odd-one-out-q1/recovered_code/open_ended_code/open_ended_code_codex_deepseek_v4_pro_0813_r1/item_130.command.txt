/bin/bash -c "python3 - <<'PY'
import subprocess, os, concurrent.futures, itertools, time
# load peak list
peaks=[]
with open('atac_peaks.tsv') as f:
 next(f)
 for line in f:
  p=line.rstrip().split('\\t'); peaks.append((p[0],p[1],p[2]))
samples=[str(i) for i in range(1,11)]
def work(item):
 b, (exp,cell,acc)=item
 cmd=['bedtools','intersect','-a',f'samples/{b}.sample.bed','-b',f'atac_peaks/{acc}.bed.gz','-u']
 r=subprocess.run(cmd,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True)
 n=len(r.stdout.splitlines())
 return b,exp,cell,acc,n
items=list(itertools.product(samples,peaks))
print('pairs',len(items))
out=open('atac_overlap.tsv','w'); out.write('sample\\texperiment\\tcell\\tfile\\toverlap\\n')
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:
 for i,res in enumerate(ex.map(work,items,chunksize=20)):
  out.write('\\t'.join(map(str,res))+'\\n')
  if (i+1)%500==0: print('done',i+1,flush=True)
out.close()
print('finished')
PY"
