/bin/bash -c "python3 - <<'PY'
D={}
with open('ccre_counts.tsv') as f:
 for line in f:
  cat,b,n=line.split(); D[(cat,int(b))]=int(n)
lengths={}
for cat in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']:
 fn='ccre_'+cat+'_lex.bed'; total=0
 with open(fn) as f:
  for line in f:
   p=line.split('\\t'); total += int(p[2])-int(p[1])
 lengths[cat]=total
genome=sum(int(line.split()[1]) for line in open('hg38.chrom.sizes') if not line.startswith('_'))
print('category lengths and genome fracs')
for cat,L in lengths.items(): print(cat,L,f'{L/genome:.6f}')
print('file enrichments PLS pELS dELS CTCF DNaseH3K4me3')
for b in range(1,11):
 vals=[]
 for cat in ['PLS','pELS','dELS','CTCF_only','DNase_H3K4me3']:
  count=D[(cat,b)]
  enrich=(count/lengths[cat])/(1e7/genome)
  vals.append(enrich)
 print(b, '\\t'.join(f'{x:.2f}' for x in vals))
PY"
