/bin/bash -c "python3 - <<'PY'
import gzip, numpy as np, glob, re, os
files=glob.glob('odd.one.out.q1/*.tagAlign.gz'); files.sort(key=lambda x:int(re.search(r'/(\\d+)\\.tagAlign',x).group(1)))
chrom='chr1'; size=248956422; bin=50; n=(size+bin-1)//bin
shifts=np.arange(0,501,10)
for f in files:
 plus=np.zeros(n,dtype=np.int32); minus=np.zeros(n,dtype=np.int32)
 with gzip.open(f,'rt') as fh:
  for line in fh:
   p=line.split('\\t')
   if p[0]"'!=chrom: continue
   pos=int(p[1])//bin
   if p[5]=='"'+': plus[pos]+=1
   else: minus[pos]+=1
 # cross covariance plus vs minus shifted; normalize by count
 c=np.array([np.dot(plus[:-s],minus[s:]) if s>0 else np.dot(plus,minus) for s in shifts],dtype=float)
 c/=max(1,np.sqrt(plus.sum()*minus.sum()))
 name=os.path.basename(f).replace('.tagAlign.gz','')
 peak=shifts[np.argmax(c)]
 print(name, 'peak',peak, 'corrs', ' '.join(f'{x:.3f}' for x in c[::5]))
PY"
