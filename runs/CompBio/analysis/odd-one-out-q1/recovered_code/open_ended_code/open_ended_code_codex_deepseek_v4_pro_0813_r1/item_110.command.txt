/bin/bash -c "python3 - <<'PY'
import requests, collections, json
assays=['ATAC-seq','DNase-seq','Histone ChIP-seq','TF ChIP-seq','Control ChIP-seq']
for assay in assays:
 url='https://www.encodeproject.org/search/?type=Experiment&assay_title='+requests.utils.quote(assay)+'&replicates.library.biosample.organism.scientific_name=Homo%20sapiens&assembly=GRCh38&format=json&limit=all&field=accession&field=biosample_ontology.term_name&field=biosample_ontology.classification&field=target.label'
 data=requests.get(url,timeout=120).json().get('@graph',[])
 cells=[f for f in data if f.get('biosample_ontology',{}).get('classification')=='cell line']
 print('\\n',assay,'total',len(data),'cell_line',len(cells))
 # unique cell lines
 names=collections.Counter(f['biosample_ontology']['term_name'] for f in cells)
 print('unique',len(names),names.most_common(15))
PY"
