/bin/bash -c "python3 - <<'PY'
import requests, json
assays=['ATAC-seq','DNase-seq','Histone ChIP-seq','TF ChIP-seq','Control ChIP-seq']
for assay in assays:
 url='https://www.encodeproject.org/search/?type=File&file_format=tagAlign&assembly=GRCh38&status=released&dataset.assay_title='+requests.utils.quote(assay)+'&format=json&limit=20&field=accession&field=href&field=dataset&field=output_type'
 r=requests.get(url,timeout=60)
 data=r.json().get('@graph',[])
 print('\\n',assay,len(data))
 for f in data[:5]: print(f.get('accession'),f.get('output_type'),f.get('dataset'),f.get('href'))
PY"
