/bin/bash -c "python3 - <<'PY'
import requests, collections
url='https://www.encodeproject.org/search/?type=Experiment&assay_title=ATAC-seq&replicates.library.biosample.organism.scientific_name=Homo%20sapiens&assembly=GRCh38&format=json&limit=all'
exps=requests.get(url,timeout=120).json()['@graph']
for exp in exps:
 if exp.get('biosample_ontology',{}).get('classification')=='cell line' and exp['biosample_ontology']['term_name']=='K562':
  print(exp['accession'], len(exp.get('files',[])))
  c=collections.Counter()
  for f in exp.get('files',[]):
   c[(f.get('file_type'),f.get('output_type'))]+=1
  for k,v in c.items(): print(k,v)
  for f in exp.get('files',[])[:20]: print(f.get('accession'),f.get('file_type'),f.get('output_type'),f.get('file_size'))
  break
PY"
