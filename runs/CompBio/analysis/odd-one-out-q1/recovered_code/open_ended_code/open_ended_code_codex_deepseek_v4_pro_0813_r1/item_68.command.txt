/bin/bash -c "python3 - <<'PY'
import requests
for assay in ['ATAC-seq','DNase-seq','Histone ChIP-seq','Transcription factor ChIP-seq','Control ChIP-seq','RNA Bind-n-Seq']:
 url='https://www.encodeproject.org/search/?type=Experiment&assay_title='+requests.utils.quote(assay)+'&replicates.library.biosample.organism.scientific_name=Homo%20sapiens&assembly=GRCh38&format=json&limit=all'
 r=requests.get(url,timeout=60)
 try:
  data=r.json(); print(assay,len(data.get('@graph',[])))
 except Exception as e: print(assay,'err',r.status_code)
PY"
