/bin/bash -c "python3 - <<'PY'
import requests
for repo in ['compbiobench-submissions-v1','compbiobench-results-v1']:
 url=f'https://huggingface.co/api/datasets/Genentech/{repo}/tree/main?recursive=true'
 r=requests.get(url,timeout=30)
 print(repo, r.status_code)
 try:
  data=r.json()
  print(len(data))
  for d in data[:20]: print(d)
 except Exception as e: print(r.text[:500])
PY"
