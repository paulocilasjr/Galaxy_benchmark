/bin/bash -c "python3 - <<'PY'
# split cCREs by main category, sort lex
cats=['PLS','pELS','dELS','CTCF-only','DNase-H3K4me3']
files={c:open(f'ccre_{c.replace(\"-\",\"_\")}.bed','w') for c in cats}
with open('GRCh38-cCREs.bed') as f:
 for line in f:
  p=line.rstrip().split('\\t')
  if len(p)<6: continue
  cat=p[5].split(',')[0]
  if cat in files:
   files[cat].write('\\t'.join(p[:3]+[cat])+'\\n')
for c in cats: files[c].close()
PY
for f in ccre_*.bed; do sort -k1,1 -k2,2n "'$f > ${f%.bed}_lex.bed; done
ls -lh ccre_*_lex.bed
wc -l ccre_*_lex.bed'
