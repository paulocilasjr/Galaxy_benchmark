/bin/bash -c "python3 - <<'PY'
import subprocess, itertools, concurrent.futures
peaks=[]
with open('dnase_peaks.tsv') as f:
 next(f)
 for line in f:
  p=line.rstrip().split('\\t'); peaks.append((p[0],p[1],p[2]))
samples=[str(i) for i in range(1,11)]
def work(item):
 b,(cell,exp,acc)=item
 r=subprocess.run(['bedtools','intersect','-a',f'samples/{b}.sample.bed','-b',f'dnase_peaks/{acc}.bed.gz','-u'],stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True)
 return b,exp,cell,acc,len(r.stdout.splitlines())
out=open('dnase_overlap.tsv','w'); out.write('sample\\texperiment\\tcell\\tfile\\toverlap\\n')
items=list(itertools.product(samples,peaks)); print('pairs',len(items))
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:
 for i,res in enumerate(ex.map(work,items,chunksize=20)):
  out.write('\\t'.join(map(str,res))+'\\n')
  if (i+1)%500==0: print('done',i+1,flush=True)
out.close(); print('finished')
PY"
