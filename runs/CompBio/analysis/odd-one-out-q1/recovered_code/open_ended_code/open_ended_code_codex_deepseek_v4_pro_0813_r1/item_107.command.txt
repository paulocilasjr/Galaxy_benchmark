/bin/bash -c "python3 - <<'PY'
import requests, collections
assays=['ATAC-seq','DNase-seq','Histone ChIP-seq','TF ChIP-seq','Control ChIP-seq']
for assay in assays:
 url='https://www.encodeproject.org/search/?type=Experiment&assay_title='+requests.utils.quote(assay)+'&replicates.library.biosample.organism.scientific_name=Homo%20sapiens&replicates.library.biosample.classification=cell%20line&assembly=GRCh38&format=json&limit=all&field=accession&field=assay_title&field=biosample_ontology.term_name&field=target.label'
 r=requests.get(url,timeout=120)
 data=r.json().get('@graph',[])
 print(assay,len(data))
 # print unique cell lines sample
 cells=collections.Counter(f.get('biosample_ontology',{}).get('term_name') for f in data)
 print(cells.most_common(20))
PY"
