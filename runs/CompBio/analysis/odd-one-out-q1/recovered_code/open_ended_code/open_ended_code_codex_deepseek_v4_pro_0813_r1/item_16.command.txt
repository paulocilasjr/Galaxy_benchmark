/bin/bash -c "python3 - <<'PY'
import numpy as np, glob, re, os
files=glob.glob('counts_*.bed'); files.sort(key=lambda x:int(re.search(r'counts_(\\d+)\\.bed',x).group(1)))
names=[os.path.basename(f).replace('counts_','').replace('.bed','') for f in files]
X=[]
for f in files:
 a=np.loadtxt(f,usecols=3,dtype=float); X.append(a)
print('file\\tmean\\tmedian\\tp90\\tp99\\tmax\\tzeros\\tentropy\\tf_top1\\tf_top5')
for name,a in zip(names,X):
 a=a/a.sum(); q=np.quantile(a,[.5,.9,.99]); mx=a.max(); zeros=np.mean(a==0)
 # entropy of distribution
 p=a[a>0]; ent=-np.sum(p*np.log(p))/np.log(len(a)) if p.sum()>0 else 0
 f1=np.sort(a)[::-1][:int(len(a)*0.01)].sum(); f5=np.sort(a)[::-1][:int(len(a)*0.05)].sum()
 print(f'{name}\\t{a.mean():.5f}\\t{q[0]:.5f}\\t{q[1]:.5f}\\t{q[2]:.5f}\\t{mx:.5f}\\t{zeros:.3f}\\t{ent:.3f}\\t{f1:.3f}\\t{f5:.3f}')
PY"
