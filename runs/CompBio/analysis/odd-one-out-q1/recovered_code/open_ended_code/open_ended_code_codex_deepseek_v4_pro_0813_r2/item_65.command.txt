/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, os
cols=[]
stats=[]
for n in range(1,11):
    a=np.loadtxt(f'kb_cov_{n}.txt',delimiter='\\t',usecols=3,dtype=np.int32)
    cols.append(a)
    stats.append((n, a.mean(), np.median(a), a.max(), (a>0).mean(), np.std(a), (a**2).sum()/a.sum() if a.sum() else 0))
print('file mean median max nonzero_frac std sum_sq/sum')
for s in stats: print(s)
M=np.vstack(cols).T
df=pd.DataFrame(M,columns=[str(i) for i in range(1,11)])
print('pearson corr')
print(df.corr().round(2).to_string())
print('spearman corr (sample?) maybe')
# spearman on full 3.2m x10 might be heavy; use sample random 50000
rng=np.random.default_rng(0); idx=rng.choice(len(df),50000,replace=False)
print(df.iloc[idx].corr(method='spearman').round(2).to_string())
PY"
