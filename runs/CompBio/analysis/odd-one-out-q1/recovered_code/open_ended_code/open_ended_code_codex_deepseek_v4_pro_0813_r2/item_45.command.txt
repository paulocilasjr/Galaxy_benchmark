/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for term in ['GRCh38','hg38','hg19']:
 url=f'https://www.encodeproject.org/search/?type=File&file_format=tagAlign&assembly={term}&format=json&limit=10'
 try:
  with urllib.request.urlopen(url,timeout=20) as r: j=json.load(r)
  print(term,'total',j.get('total'),'filters',j.get('filters'))
  for f in j.get('@graph',[])[:2]: print(' ',f.get('accession'),f.get('assembly'))
 except Exception as e: print(term,'ERR',e)
PY"
