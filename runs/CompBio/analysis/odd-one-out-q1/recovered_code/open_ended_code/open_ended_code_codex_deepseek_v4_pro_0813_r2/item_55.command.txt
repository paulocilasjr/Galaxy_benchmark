/bin/bash -lc "cat > features.py <<'PY'
import re, os, json, math
import numpy as np
import pandas as pd
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import IsolationForest
from sklearn.neighbors import NearestNeighbors

# Parse scc_unique
scc={}
for line in open('scc_unique.txt'):
    if line.startswith('FILE'):
        m=re.match(r'FILE (\\d+)\\.tagAlign\\.gz plus (\\d+) minus (\\d+)',line)
        idx=int(m.group(1)); scc[idx]={'plus':int(m.group(2)),'minus':int(m.group(3)),'vals':{}}
    elif ':' in line and scc:
        # line like 0:0.965817 25:...
        for tok in line.split():
            s,v=tok.split(':')
            idx=max(scc.keys()) # last file being parsed
            scc[idx]['vals'][int(s)]=float(v)

# cov profiles parse
cov={}
for line in open('cov_profiles.txt'):
    m=re.match(r'(\\d+)\\.tagAlign\\.gz',line)
    idx=int(m.group(1)); cov[idx]={}
    for tok in line.rstrip().split()[1:]:
        if tok.startswith('1-5='):
            val=tok.split('=')[1]; cov[idx]['cov_1_5']=float(val.split('(')[0])
        elif tok.startswith('6-20='):
            cov[idx]['cov_6_20']=float(tok.split('=')[1].split('(')[0])
        elif tok.startswith('21-100='):
            cov[idx]['cov_21_100']=float(tok.split('=')[1].split('(')[0])
        elif tok.startswith('100+='):
            cov[idx]['cov_100p']=float(tok.split('=')[1].split('(')[0])
        elif tok.startswith('total_covered='):
            cov[idx]['covered']=float(tok.split('=')[1])

peak={1:(121830,0.3949),2:(109344,0.4103),3:(118379,0.3474),4:(82037,0.2288),5:(93122,0.3353),6:(84373,0.4528),7:(99504,0.4665),8:(61625,0.1374),9:(64070,0.3253),10:(76553,0.3433)}
top={1:760,2:729,3:138,4:20,5:2439,6:876,7:551,8:226,9:2059132,10:141323}
chrX={1:319015,2:258163,3:200165,4:272201,5:176058,6:174525,7:389485,8:283983,9:182414,10:223669}
entropy={1:3.0356,2:3.0263,3:3.0515,4:3.0085,5:2.9772,6:3.0297,7:3.0505,8:3.0382,9:2.8235,10:3.0345}
# TSS sums
tss_df=pd.read_csv('tss_counts.tsv',sep='\\t',header=None)
tss_sums=tss_df.iloc[:,6:16].sum().values

rows=[]
for i in range(1,11):
    vals=scc[i]['vals']
    feat={
      'idx':i,
      'unique_ratio':(scc[i]['plus']+scc[i]['minus'])/10000000,
      'plus_minus_ratio':abs(scc[i]['plus']-scc[i]['minus'])/5000000,
      'cc0':vals[0], 'cc25':vals[25], 'cc50':vals[50], 'cc100':vals[100], 'cc200':vals[200], 'cc500':vals[500], 'cc1000':vals[1000],
      'cov_1_5':cov[i]['cov_1_5']/1e6,'cov_6_20':cov[i]['cov_6_20']/1e6,'cov_21_100':cov[i]['cov_21_100']/1e6,'cov_100p':cov[i]['cov_100p']/1e6,'covered':cov[i]['covered']/1e6,
      'peaks':peak[i][0],'frip':peak[i][1],
      'top1':top[i], 'top1_frac':top[i]/1e7,
      'chrX_frac':chrX[i]/1e7,
      'entropy':entropy[i],
      'tss_sum':tss_sums[i-1],
    }
    rows.append(feat)
X=pd.DataFrame(rows).set_index('idx')
print('Features:')
print(X.to_string())
# standardize
scaler=StandardScaler()
Z=scaler.fit_transform(X)
# distance to centroid
cent=Z.mean(axis=0)
dist=np.linalg.norm(Z-cent,axis=1)
print('distance to centroid', pd.Series(dist,index=X.index).round(2).to_dict())
# PCA first components
pca=PCA(n_components=2)
P=pca.fit_transform(Z)
print('PCA coords', pd.DataFrame(P,index=X.index,columns=['PC1','PC2']).round(3).to_string())
print('explained', pca.explained_variance_ratio_)
# Isolation forest
iso=IsolationForest(contamination=0.1, random_state=0).fit(Z)
print('if scores', pd.Series(iso.decision_function(Z),index=X.index).round(3).to_dict())
# nearest neighbor distances in standardized space
nbrs=NearestNeighbors(n_neighbors=2).fit(Z)
d,idx=nbrs.kneighbors(Z)
print('nearest neighbor dist', pd.Series(d[:,1],index=X.index).round(2).to_dict())
PY
python3 features.py"
