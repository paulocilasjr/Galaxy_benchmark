/bin/bash -lc "python3 - <<'PY'
import numpy as np
A=np.stack([np.load(f'{i}.bins1k.npy') for i in range(1,11)]).astype('float32')
# Remove extreme artifact bins and constitutionally unmappable bins; focus occupied autosomal bins.
keep=(A.sum(0)>0)&(A.max(0)<10000)
print('shape/keep',A.shape,keep.sum())
for name,B in [('raw',A[:,keep]),('sqrt',np.sqrt(A[:,keep])),('log',np.log1p(A[:,keep]))]:
 C=np.corrcoef(B)
 print(name,'mean',np.round((C.sum(1)-1)/9,3))
 print(np.array2string(C,precision=2,suppress_small=True))
for n in [10000,30000,100000,300000,1000000]:
 sets=[set(np.argpartition(a,-n)[-n:]) for a in A]
 C=np.array([[len(x&y)/n for y in sets] for x in sets])
 print('top',n,'mean overlap',np.round((C.sum(1)-1)/9,3))
 print(np.array2string(C,precision=2,suppress_small=True))
# Similarity of each sample top bins to number of other samples enriched there.
for n in [10000,30000,100000,300000]:
 top=np.zeros_like(A,dtype=bool)
 for i,a in enumerate(A): top[i,np.argpartition(a,-n)[-n:]]=1
 print('n',n,'consensus scores',np.round([(top[:,top[i]].sum(0).mean()-1)/9 for i in range(10)],3))
PY"
