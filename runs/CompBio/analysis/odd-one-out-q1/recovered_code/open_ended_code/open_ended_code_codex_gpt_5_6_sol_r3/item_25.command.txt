import json,numpy as np
D=sorted(json.load(open('endpoint_bias.json')),key=lambda x:int(x['file'].split('.')[0]))

# normalize each nucleotide by its own distal background frequency, then central window
base=A[:,np.r_[0:8,33:41]].mean(1)

C=np.corrcoef(B.reshape(10,-1))
d=np.sqrt(((B[:,None]-B[None,:])**2).sum((2,3)))
print('correlation\\n',np.round(C,2))
print('mean corr',np.round((C.sum(1)-1)/9,3))
print('dist\\n',np.round(d,2))
print('median dist',np.round(np.median(d+np.eye(10)*999,axis=1),2),'mean',np.round(d.sum(1)/9,2))
# each sample distance from median of other nine
print('LOO median distance',np.round([np.linalg.norm(B[i]-np.median(np.delete(B,i,0),axis=0)) for i in range(10)],2))
PY"
