import gzip, math
from collections import defaultdict, Counter
names=[]; seqs=[]; name=None; parts=[]
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if not line: continue
    if line.startswith('>'):
        if name is not None:
            names.append(name); seqs.append(''.join(parts).upper())
        name=line[1:].split()[0]; parts=[]

if name is not None: names.append(name); seqs.append(''.join(parts).upper())
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
L=36; k=9; offsets=[0,9,18,27]
refs=[(s,rc(s)) for s in seqs]
index=defaultdict(list)
for ci,(f,r) in enumerate(refs):
    for strand,g in enumerate((f,r)):
        for p in range(len(g)-k+1):
            index[g[p:p+k]].append((ci,strand,p))
reads=Counter()
with gzip.open('read.proportions.q1.fq.gz','rt') as fh:

        read=fh.readline().strip().upper(); fh.readline(); fh.readline()
        reads[read]+=1
print('unique reads',len(reads),'total',sum(reads.values()))

def hdist(a,b): return sum(x"'!=y for x,y in zip(a,b))
# store mismatch counts per chr for candidate alignments, aggregated by unique read count
read_dists=[]; fallback=0; zeros=0
for read,cnt in reads.items():
    cand=set()
    for off in offsets:
        seed=read[off:off+k]
        for ci,strand,p in index.get(seed,()):
            start=p-off
            if 0 <= start <= len(refs[ci][strand])-L:
                cand.add((ci,strand,start))

    if not cand:

        for ci,(f,r) in enumerate(refs):
            for strand,g in enumerate((f,r)):

def em_for_eps(eps):

    weights=[]
    for cnt,ds in read_dists:
        p=[]
        for arr in ds:
            if arr:
                p.append(sum(q**d for d in arr))
            else:
                p.append(0.0)
        # add tiny background for completely missed very-error alignments? not needed
        weights.append((cnt,p))

        for cnt,p in weights:
            den=sum(pi[i]*p[i] for i in range(4))
            if den<=0:

                continue
            ll += cnt*math.log(den)
            for i in range(4):
                new[i] += cnt*pi[i]*p[i]/den
        tot=sum(new)
        new=[x/tot for x in new]
        if max(abs(new[i]-pi[i]) for i in range(4))<1e-12:
            return new,ll,it+1
        pi=new
    return pi,ll,10000

    print('eps',eps,'pi', ['%.4f'%(100*x) for x in pi], 'it',it)
