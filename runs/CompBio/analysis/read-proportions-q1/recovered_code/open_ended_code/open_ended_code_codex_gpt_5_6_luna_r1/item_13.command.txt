/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict,Counter
import gzip
fa={}
for l in open('read.proportions.q1.fa'):
 l=l.strip()
 if l.startswith('>'): n=l[1:];fa[n]=''
 else:fa[n]+=l
comp=str.maketrans('ACGTN','TGCAN')
rc=lambda s:s.translate(comp)[::-1]
idx={c:defaultdict(list) for c,g in fa.items()}
for c,g in fa.items():
 for p in range(len(g)-35): idx[c][g[p:p+36]].append(p)
counts=Counter(); examples=[]
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
 while h:=f.readline():
  r=f.readline().strip();f.readline();f.readline()
  fwd=tuple(c for c in fa if r in idx[c]); rev=tuple(c for c in fa if rc(r) in idx[c])
  counts[('fwd',fwd)]+=1; counts[('rev',rev)]+=1
  if not fwd and rev and len(examples)<4: examples.append((r,rc(r),rev))
print('fwd/rev exact any')
print('forward',sum(v for k,v in counts.items() if k[0]=='fwd' and k[1]))
print('reverse',sum(v for k,v in counts.items() if k[0]=='rev' and k[1]))
print('neither fwd',sum(v for k,v in counts.items() if k[0]=='fwd' and not k[1]))
print('both orientation reads',sum(1 for k in []))
print('top forward',[(k,v) for k,v in counts.items() if k[0]=='fwd'][:5])
print('examples',examples)
PY
printf '\\nminimap2 align summary:\\n'
minimap2 -a -x sr read.proportions.q1.fa read.proportions.q1.fq.gz 2>/dev/null | samtools view -F 4 -c -; minimap2 -a -x sr read.proportions.q1.fa read.proportions.q1.fq.gz 2>/dev/null | samtools view -F 4 -c -"
