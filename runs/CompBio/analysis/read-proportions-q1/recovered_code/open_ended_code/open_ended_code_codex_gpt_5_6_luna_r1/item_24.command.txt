from collections import defaultdict,Counter
import gzip
fa={}

 print('classes',{'f'+''.join(str(x+1) for x in s):v for (o,s),v in b.items() if o=='f'}, {'r'+''.join(str(x+1) for x in s):v for (o,s),v in b.items() if o=='r'})
