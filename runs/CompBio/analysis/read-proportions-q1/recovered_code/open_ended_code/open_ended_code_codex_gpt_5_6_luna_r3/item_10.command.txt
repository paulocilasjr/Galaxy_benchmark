/bin/bash -lc "python3 - <<'PY'
from itertools import combinations
fa={}; n=None
for l in open('read.proportions.q1.fa'):
 l=l.strip()
 if l.startswith('>'): n=l[1:]; fa[n]=''
 else: fa[n]+=l
for a,b in combinations(fa,2):
 s,t=fa[a],fa[b]
 ident=sum(x==y for x,y in zip(s,t))/len(s)
 print(a,b,'same-position identity',ident,'same bases',sum(x==y for x,y in zip(s,t)))
 print('prefix',s[:80],t[:80])
PY"
