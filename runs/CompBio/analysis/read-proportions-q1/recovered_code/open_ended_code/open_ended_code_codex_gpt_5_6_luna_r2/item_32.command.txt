/bin/bash -lc "python - <<'PY'
from collections import defaultdict,Counter
D=defaultdict(lambda:[-999]*4); cnt=Counter(); nlines=0
with open('reads.bwa.all.sam') as f:
 for line in f:
  if line[0]=='@':continue
  a=line.rstrip().split('\\t'); rid=a[0]; r=a[2]
  if r=='*':continue
  j=int(r[-1])-1
  AS=next(int(x.split(':')[2]) for x in a[11:] if x.startswith('AS:i:'))
  if AS>D[rid][j]:D[rid][j]=AS
for vals in D.values():
 m=max(vals); w=tuple(i for i,x in enumerate(vals) if x==m)
 cnt[(m,w,tuple(x if x>-900 else None for x in vals))]+=1
# condensed winner patterns m, winners; but include values distribution
p=Counter((max(v),tuple(i for i,x in enumerate(v) if x==max(v))) for v in D.values())
print('reads with alignments',len(D))
for x,n in sorted(p.items()):print(x,n)
# per ref max count and tie unique
print('unique max',Counter(next((i for i,x in enumerate(v) if x==max(v)),None) for v in D.values() if sum(x==max(v) for x in v)==1))
print('missing per ref',Counter(sum(x is None for x in v) for v in D.values()))
PY"
