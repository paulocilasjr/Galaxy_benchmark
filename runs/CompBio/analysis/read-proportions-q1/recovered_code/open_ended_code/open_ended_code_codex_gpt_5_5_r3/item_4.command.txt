/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict,Counter
# parse fasta
contigs=[]
name=None; seq=[]
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if not line: continue
    if line.startswith('>'):
        if name: contigs.append((name,''.join(seq)))
        name=line[1:].split()[0]; seq=[]
    else: seq.append(line.upper())
if name: contigs.append((name,''.join(seq)))
L=36
idx=defaultdict(list)
idx_fw=defaultdict(list)
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
for ci,(name,s) in enumerate(contigs):
    for pos in range(len(s)-L+1):
        k=s[pos:pos+L]
        idx[k].append(ci); idx_fw[k].append(ci)
        idx[rc(k)].append(ci)
counts=Counter(); counts_fw=Counter(); n=0; exact=0; exactfw=0; multi=0
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline().strip()
        if not h: break
        r=f.readline().strip().upper(); f.readline(); f.readline(); n+=1
        if r in idx:
            exact+=1
            cs=set(idx[r])
            if len(cs)>1: multi+=1
            for c in cs: counts[c]+=1/len(cs)
        if r in idx_fw:
            exactfw+=1
            cs=set(idx_fw[r])
            for c in cs: counts_fw[c]+=1/len(cs)
print('n',n,'exact both',exact,'multi',multi,'counts', [counts[i] for i in range(4)], 'pct',[counts[i]/exact*100 for i in range(4)])
print('exact fw',exactfw,'counts_fw',[counts_fw[i] for i in range(4)], 'pct',[counts_fw[i]/exactfw*100 for i in range(4)])
PY"
