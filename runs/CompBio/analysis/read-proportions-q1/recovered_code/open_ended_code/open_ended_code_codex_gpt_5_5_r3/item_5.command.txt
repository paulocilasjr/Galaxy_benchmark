/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict, Counter
contigs=[]; name=None; seq=[]
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'):
        if name: contigs.append((name,''.join(seq)))
        name=line[1:].split()[0]; seq=[]
    else: seq.append(line.upper())
if name: contigs.append((name,''.join(seq)))
L=36
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s.translate(comp)[::-1]
allkm=defaultdict(set)
for ci,(name,s) in enumerate(contigs):
    for pos in range(len(s)-L+1):
        k=s[pos:pos+L]
        allkm[k].add(ci); allkm[rc(k)].add(ci)
unique=[0]*4; shared=[0]*4; total=[0]*4
for ci,(name,s) in enumerate(contigs):
    seen=[]
    for pos in range(len(s)-L+1):
        for k in [s[pos:pos+L], rc(s[pos:pos+L])]:
            total[ci]+=1
            if allkm[k]=={ci}: unique[ci]+=1
            else: shared[ci]+=1
print('unique states per contig',unique,'shared',shared,'total',total)
# similarities: each contig states whose kmer appears in other contig
mat=[[0]*4 for _ in range(4)]
for ci,(name,s) in enumerate(contigs):
    for pos in range(len(s)-L+1):
        for k in [s[pos:pos+L], rc(s[pos:pos+L])]:
            for cj in allkm[k]: mat[ci][cj]+=1
print('mat')
for row in mat: print(row)
# maybe contig seq equality blocks? print first 80
for name,s in contigs:
    print(name, s[:100])
PY"
