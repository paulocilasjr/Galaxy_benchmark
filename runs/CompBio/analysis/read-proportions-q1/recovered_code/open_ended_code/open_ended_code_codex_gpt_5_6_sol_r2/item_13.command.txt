/bin/bash -lc "samtools view -F 4 reads.bam | python -c '
import sys,math
seqs={}; n=None
for z in open(\"read.proportions.q1.fa\"):
 z=z.strip()
 if z.startswith(\">\"): n=z[1:].split()[0]; seqs[n]=\"\"
 else: seqs[n]+=z
names=list(seqs); ds=[]
for l in sys.stdin:
 f=l.rstrip().split(\"\\t\")
 if f[5]"'!="36M": continue
 p=int(f[3])-1; q=f[9]; ds.append([sum(a!=b for a,b in zip(q,seqs[n][p:p+36])) for n in names])
print("n",len(ds),flush=True)
from collections import Counter
print("min_mismatch",Counter(min(x) for x in ds),flush=True)
hard=[0.0]*4
for x in ds:
 w=[i for i,v in enumerate(x) if v==min(x)]
 for i in w: hard[i]+=1/len(w)
print("hard_fractional",hard,[100*x/len(ds) for x in hard],flush=True)
for eps in [0.001,0.01,0.03]:
 pi=[.25]*4; lr=math.log((eps/3)/(1-eps))
 for it in range(300):
  s=[0.0]*4
  for x in ds:
   a=[pi[i]*math.exp((x[i]-min(x))*lr) for i in range(4)]; t=sum(a)
   for i in range(4): s[i]+=a[i]/t
  new=[v/len(ds) for v in s]
  if max(abs(new[i]-pi[i]) for i in range(4))<1e-10: pi=new; break
  pi=new
 print("eps",eps,"pi",[round(100*v,4) for v in pi],"iter",it,flush=True)
'"'"
