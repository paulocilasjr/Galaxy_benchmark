/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
import gzip, collections
trans=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(trans)[::-1]
seqs=[]; name=None; chunks=[]
for line in Path('read.proportions.q1.fa').read_text().splitlines():
    if line.startswith('>'):
        if name: seqs.append((name,''.join(chunks)))
        name=line[1:].split()[0]; chunks=[]
    else: chunks.append(line.strip())
if name: seqs.append((name,''.join(chunks)))
L=36
idx=collections.defaultdict(set)
for ci,(n,s) in enumerate(seqs):
    for pos in range(len(s)-L+1):
        idx[s[pos:pos+L]].add(ci)
        idx[rc(s[pos:pos+L])].add(ci)
setcounts=collections.Counter()
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h: break
        r=f.readline().strip(); f.readline(); f.readline()
        setcounts[tuple(sorted(idx.get(r,())))]+=1
for k,v in setcounts.most_common():
    print(k,v)
PY"
