from collections import Counter
seqs={}
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'): h=line[1:].split()[0]; seqs[h]=''
    else: seqs[h]+=line.upper()
cs=list(seqs); L=len(seqs[cs[0]]); bases='ACGT'; bi={b:i for i,b in enumerate(bases)}
counts=np.zeros((L,4),dtype=int)
p=subprocess.Popen(['bwa','samse','-n','20','read.proportions.q1.fa','read.proportions.q1.sai','read.proportions.q1.fq.gz'],stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True)

for line in p.stdout:
    if line.startswith('@'): continue
    f=line.rstrip().split('\\t')
    if int(f[1])&4 or f[5]"'!='"'36M': continue
    pos=int(f[3])-1
