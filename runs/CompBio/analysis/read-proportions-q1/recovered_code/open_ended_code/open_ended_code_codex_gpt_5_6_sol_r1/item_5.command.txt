/bin/bash -lc "bwa samse read.proportions.q1.fa read.proportions.q1.sai read.proportions.q1.fq.gz 2>/dev/null | samtools view -F 4 - | awk '{key="'$3"'"\\t\""'$5; c[key]++} END{for(k in c) print k,c[k]}'"' | sort -k1,1 -k2,2n
python - <<'PY'
from itertools import combinations
seq={}
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'):
        h=line[1:]; seq[h]=''
    else: seq[h]+=line
for a,b in combinations(seq,2):
    matches=sum(x==y for x,y in zip(seq[a],seq[b]))
    print(a,b,'same-coordinate identity',matches/len(seq[a]))
PY"
