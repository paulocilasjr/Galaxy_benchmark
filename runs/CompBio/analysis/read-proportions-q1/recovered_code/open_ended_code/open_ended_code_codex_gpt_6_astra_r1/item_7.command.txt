/bin/bash -lc "cat > estimate.py <<'PY'
import numpy as np
from collections import Counter
rows=[]; nm=Counter()
for l in open('reads.all.sam'):
 if l.startswith('@'):continue
 f=l.rstrip().split('\\t')
 if f[2]=='*':continue
 d={x[:2]:x.split(':',2)[2] for x in f[11:]}
 hits=[(f[2],int(d['NM']))]
 nm[int(d['NM'])]+=1
 for x in d.get('XA','').split(';'):
  if x:
   chrom,pos,cigar,dist=x.split(',');hits.append((chrom,int(dist)))
 rows.append(hits)
print('Minimum mismatch distribution:',nm)
for rate in [0,0.003,0.01,0.02]:
 a=np.zeros((len(rows),4))
 for i,hits in enumerate(rows):
  best=min(h[1] for h in hits)
  for chrom,dist in hits:
   a[i,int(chrom[3:])-1]+=(1. if dist==best else 0.) if rate==0 else (rate/3/(1-rate))**(dist-best)
 p=np.ones(4)/4
 for it in range(10000):
  w=a*p;new=(w/w.sum(axis=1)[:,None]).mean(axis=0)
  if abs(new-p).max()<1e-11:break
  p=new
 r=[int(5*round(x*100/5)) for x in p];r[-1]=100-sum(r[:-1])
 print('rate',rate,'iterations',it,'percent',100*p,'rounded',r)
PY
python3 estimate.py"
