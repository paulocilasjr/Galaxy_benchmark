/bin/bash -lc "which bowtie2 bwa minimap2 samtools 2>/dev/null || true; python3 - <<'PY'
from Bio import SeqIO
import gzip
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
# count exact 36-mer forward/reverse
from collections import Counter,defaultdict
comp=str.maketrans('ACGT','TGCA')
index=defaultdict(list)
for name,s in seqs.items():
    for i in range(len(s)-36+1):
        index[s[i:i+36]].append((name,i,'f'))
    rs=s.translate(comp)[::-1]
    for i in range(len(rs)-36+1):
        index[rs[i:i+36]].append((name,i,'r'))
counts=Counter(); unassigned=0; amb=0; n=0
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h: break
        seq=f.readline().strip().upper(); plus=f.readline(); qual=f.readline().strip()
        n+=1
        hits=index.get(seq,[])
        if not hits:
            unassigned+=1
        elif len(set(x[0] for x in hits))>1:
            amb+=1
        else:
            counts[hits[0][0]]+=1
print(n, counts, 'unassigned',unassigned,'amb',amb)
PY"
