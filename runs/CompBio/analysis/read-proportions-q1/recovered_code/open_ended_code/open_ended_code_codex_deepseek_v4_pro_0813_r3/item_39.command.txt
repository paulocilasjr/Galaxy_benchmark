/bin/bash -lc "python3 - <<'PY'
import pickle
from collections import Counter,defaultdict
with open('counts_mins.pkl','rb') as f: data=pickle.load(f)
mins=data['mins']; names=data['names']
patterns=Counter(); uniq=Counter()
for row in mins:
    md=min(row); idx=tuple(i for i,d in enumerate(row) if d==md)
    patterns[idx]+=1
    if len(idx)==1: uniq[names[idx[0]]]+=1
print('unique',uniq)
for pat,count in patterns.most_common():
    print(pat,count)
# predicted BWA counts if ties broken uniformly among tied indices
pred=[0.0]*4
for pat,count in patterns.items():
    share=count/len(pat)
    for j in pat: pred[j]+=share
print('pred uniform tie',pred,[round(x) for x in pred])
print('observed bwa',[13198,8993,9047,8630])
# predicted with ties broken first in lexicographic order (chr1 preferred then chr2 etc)
pred2=[0.0]*4
for pat,count in patterns.items():
    pred2[min(pat)]+=count
print('pred first index',pred2,[round(x) for x in pred2])
# if ties broken last index
pred3=[0.0]*4
for pat,count in patterns.items():
    pred3[max(pat)]+=count
print('pred last index',pred3,[round(x) for x in pred3])
PY"
