/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
names=list(seqs)
s=seqs['chr1']
# all positions differing among any pair
diff=set()
for i in range(len(s)):
    vals={seqs[n][i] for n in names}
    if len(vals)>1:
        diff.add(i)
print('num variant positions among all',len(diff))
print(sorted(diff)[:200])
# print variants table first 100 diff positions
for i in sorted(diff)[:200]:
    print(i, ''.join(seqs[n][i] for n in names))
PY"
