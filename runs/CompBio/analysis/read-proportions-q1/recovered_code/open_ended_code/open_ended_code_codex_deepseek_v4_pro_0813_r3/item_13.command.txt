/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
import gzip
from collections import defaultdict
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
comp=str.maketrans('ACGT','TGCA'); L=36;k=9
r='TTTTAACATGCGGGTGTGCACGCCACTAAGACGAAA'
# find brute best per chr and position
for name,s in seqs.items():
    refs=[('f',s),('r',s.translate(comp)[::-1])]
    best=99; bp=None
    for strand,ref in refs:
        for p in range(len(ref)-L+1):
            d=sum(ref[p+j]"'!=r[j] for j in range(L))
            if d<best:
                best=d;bp=(strand,p)
    print(name,best,bp,'"'ref match',refs[0 if bp[0]=='f' else 1][1][bp[1]:bp[1]+L])
# index and check candidates for chr1
name='chr1'
idx={'f':defaultdict(list),'r':defaultdict(list)}
s=seqs[name]
for i in range(len(s)-L+1): idx['f'][s[i:i+k]].append(i)
rs=s.translate(comp)[::-1]
for i in range(len(rs)-L+1): idx['r'][rs[i:i+k]].append(i)
print('seeds')
for start in [0,9,18,27]:
    seed=r[start:start+k]
    print(start,seed,'f cand',idx['f'].get(seed),'r cand',idx['r'].get(seed))
PY"
