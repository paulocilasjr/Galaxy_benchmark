/bin/bash -lc "python3 - <<'PY'
import numpy as np
cnt=np.load('cnt.npy')
# For each read and chr, min distance with nonzero windows
mins=np.full((cnt.shape[0],4),99)
for d in range(37):
    mask=cnt[:,:,d]>0
    mins[mask & (mins==99)]=d
# Actually fill min d per read chr
mins.fill(99)
for d in range(37):
    m=cnt[:,:,d]>0
    mins[m]=np.where(m, np.minimum(mins[m],d), mins[m])
# assign unique min chr fraction
best=mins.min(axis=1,keepdims=True)
tie=(mins==best).sum(axis=1)
assigned=(mins==best)/tie[:,None]
print('hard counts frac %', assigned.sum(axis=0)/cnt.shape[0]*100)
# Only unique exact zero reads? count best distance 0 unique? etc
print('best min dist counts',np.bincount(best[:,0]))
PY"
