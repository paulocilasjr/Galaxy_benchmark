/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
seqs={}; name=None
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'): name=line[1:]; seqs[name]=''
    else: seqs[name]+=line.upper()
comp=str.maketrans('ACGT','TGCA')
def rc(x): return x.translate(comp)[::-1]
idx=defaultdict(list)
for c,s in seqs.items():
    for i in range(len(s)-36+1):
        k=s[i:i+36]
        idx[k].append((c,i))
import gzip
both=0;fwd_only=0;rc_only=0;none=0
counts_f={c:0 for c in seqs}; counts_r={c:0 for c in seqs}
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline().strip()
        if not h: break
        s=f.readline().strip().upper(); f.readline(); f.readline()
        fwd=s in idx
        rev=rc(s) in idx
        if fwd and rev: both+=1
        elif fwd: fwd_only+=1
        elif rev: rc_only+=1
        else: none+=1
print(both,fwd_only,rc_only,none)
PY"
