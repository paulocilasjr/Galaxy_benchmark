/bin/bash -lc "python3 - <<'PY'
import numpy as np
cnt=np.load('cnt.npy')
K=37
for eps in [0.005,0.01,0.015,0.02,0.025,0.03,0.05]:
 P=np.array([(1-eps)**(K-1-d)*(eps/3)**d for d in range(K)])
 pi=np.ones(4)/4
 for it in range(100):
  L=cnt.astype(float)@P
  ll=np.log(L)+np.log(pi)
  mx=ll.max(axis=1,keepdims=True);w=np.exp(ll-mx);gamma=w/w.sum(axis=1,keepdims=True)
  newpi=gamma.mean(axis=0)
  if np.max(abs(newpi-pi))<1e-8: pi=newpi;break
  pi=newpi
 print('eps',eps,'pi%',pi*100,'round',[int(round(x*100/5)*5) for x in pi], 'sum',pi.sum())
PY"
