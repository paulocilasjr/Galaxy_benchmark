/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
seqs={}
cur=None
for line in Path('read.proportions.q1.fa').read_text().splitlines():
    if line.startswith('>'):
        cur=line[1:].strip(); seqs[cur]=[]
    else: seqs[cur].append(line.strip())
for c in seqs: seqs[c]=''.join(seqs[c])
for c,s in seqs.items(): print(c,len(s),s[:50],s[-50:])
# identical positions?
import itertools
for a,b in itertools.combinations(seqs,2):
    same=sum(x==y for x,y in zip(seqs[a],seqs[b]))
    print(a,b,'same',same,'of',len(seqs[a]))
PY"
