/bin/bash -lc "python3 - <<'PY'
import pysam
sam=pysam.AlignmentFile('reads_all.sam','r')
unmapped=0; total=0; align_count={}
# collect per query alignments
from collections import defaultdict,Counter
d=defaultdict(list)
for a in sam:
    total+=1
    if a.is_unmapped:
        unmapped+=1
    d[a.query_name].append((a.reference_name,a.reference_start,a.is_reverse,a.get_tag('NM') if a.has_tag('NM') else None,a.get_tag('AS') if a.has_tag('AS') else None))
print('total lines',total,'unmapped lines',unmapped,'queries',len(d))
print('reads with no mapped alignments',sum(1 for q,als in d.items() if all(al[0] is None for al in als)))
print('reads with some mapped',sum(1 for q,als in d.items() if any(al[0] is not None for al in als)))
# distribution of #alignments mapped
cnt=Counter(sum(1 for al in als if al[0] is not None) for als in d.values())
print(cnt)
PY"
