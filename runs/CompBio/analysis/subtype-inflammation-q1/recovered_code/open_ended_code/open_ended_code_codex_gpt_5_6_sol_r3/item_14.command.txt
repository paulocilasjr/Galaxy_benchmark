/bin/bash -lc "OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 python -u - <<'PY'
import pandas as pd,numpy as np
from scipy.io import mmread
g=pd.read_csv('subtype.inflammation.q1.genes.txt.gz',sep='\\t').gene_symbols.astype(str).tolist(); gi={x:i for i,x in enumerate(g)}
gl=['FOXP2','WIF1','TF','GLUL','RLBP1','SLC1A3','RGR','AQP4','GFAP','ALDH1L1']
inf=['CCL5','CXCL10','CCL2','NFKBIA','IRF7','IFIT1','IFIT3','RSAD2','STAT1','B2M']
for eye in [1,2]:
 x=mmread(f'subtype.inflammation.q1.{eye}.mtx.gz').tocsr().T.tocsr()
 print('eye',eye)
 d={z:np.asarray(x[:,gi[z]].toarray()).ravel() for z in gl+inf}
 for rule,m in [('FOXP2+',d['FOXP2']>0),('WIF1+GLUL+', (d['WIF1']>0)&(d['GLUL']>0)),('RLBP1+SLC1A3+', (d['RLBP1']>0)&(d['SLC1A3']>0)),('FOXP2-WIF1-',(d['FOXP2']==0)&(d['WIF1']==0))]:
  print(rule,'n',m.sum(),end=' ')
  print(' '.join(f'{z}:{(d[z][m]>0).mean():.3f}/{d[z][m].mean():.2f}' for z in inf))
PY"
