import gzip,numpy as np,pandas as pd
from scipy.io import mmread
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f:g=pd.read_csv(f,sep='\\t').gene_symbols.astype(str).to_numpy()
with gzip.open('subtype.inflammation.q1.1.mtx.gz','rb') as f:X=mmread(f).tocsc()
lib=np.asarray(X.sum(0)).ravel()
for j in [0,1,5,11,13,21,22,25,28,31,38,100,1000]:
 a,b=X.indptr[j],X.indptr[j+1]; rows=X.indices[a:b]; vals=X.data[a:b]; ord=np.argsort(vals)[::-1][:50]
 print('\\nCELL',j,'lib',lib[j],'n',len(rows))
 print('; '.join(f'{g[rows[k]]}={vals[k]}' for k in ord))
PY"
