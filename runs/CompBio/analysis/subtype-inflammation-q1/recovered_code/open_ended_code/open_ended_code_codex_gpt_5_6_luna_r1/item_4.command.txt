/bin/bash -lc "python - <<'PY'
import gzip, scipy.io, time
for f in ['subtype.inflammation.q1.1.mtx.gz','subtype.inflammation.q1.2.mtx.gz']:
 t=time.time()
 with gzip.open(f,'rb') as h:
  x=scipy.io.mmread(h).tocsr()
 print(f, x.shape, x.nnz, x.dtype, 'time',round(time.time()-t,1), 'sum',int(x.sum()), 'mean cell',float(x.sum(axis=0).mean()))
PY"
