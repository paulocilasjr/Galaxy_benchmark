/bin/bash -lc "python - <<'PY'
import gzip,numpy as np,pandas as pd
from scipy.io import mmread
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f:g=pd.read_csv(f,sep='\\t').gene_symbols.astype(str).to_numpy()
mp={x.upper():i for i,x in enumerate(g)}
with gzip.open('subtype.inflammation.q1.1.mtx.gz','rb') as f:X=mmread(f).tocsr()
lib=np.asarray(X.sum(0)).ravel(); Xn=X.multiply(1e4/np.maximum(lib,1)).tocsr();Xn.data=np.log1p(Xn.data)
for label,ii in [('CD69+', X.getrow(mp['CD69']).toarray().ravel()>0),('CD69-',X.getrow(mp['CD69']).toarray().ravel()==0),('CCL5+',X.getrow(mp['CCL5']).toarray().ravel()>0),('CSF3R+',X.getrow(mp['CSF3R']).toarray().ravel()>0)]:
 print('\\nSUBSET',label,'n',ii.sum(),'lib',np.percentile(lib[ii],[0,25,50,75,100]))
 m=np.asarray(Xn[:,ii].mean(1)).ravel(); d=np.asarray((X[:,ii]>0).mean(1)).ravel();
 print('top genes:')
 out=[]
 for ix in np.argsort(m)[::-1]:
  if g[ix].upper() in ['MALAT1'] or len(out)>=80: continue
  out.append(f'{g[ix]}={m[ix]:.2f}/{d[ix]:.2f}')
 print('; '.join(out))
PY"
