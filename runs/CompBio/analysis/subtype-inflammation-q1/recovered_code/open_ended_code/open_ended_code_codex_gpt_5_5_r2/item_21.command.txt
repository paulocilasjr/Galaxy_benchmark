/bin/bash -lc "python - <<'PY'
import numpy as np, pandas as pd
from scipy import sparse
from analyze_inflammation import read_genes, read_matrix, log_norm, gene_cols, module_score, CELL_MARKERS
symbols,index=read_genes(); xs=[]; labels=[]
for sample,path in [('eye1','subtype.inflammation.q1.1.mtx.gz'),('eye2','subtype.inflammation.q1.2.mtx.gz')]:
 raw=read_matrix(path); xs.append(log_norm(raw)); labels += [sample]*raw.shape[0]
x=sparse.vstack(xs,format='csr'); labels=np.array(labels)
scores={ct:module_score(x,gene_cols(index,genes)[0]) for ct,genes in CELL_MARKERS.items()}
df=pd.DataFrame(scores); df['ct']=df[list(CELL_MARKERS)].idxmax(axis=1)
for gene in ['CCL5','NFKBIA','CXCL9','CXCL10','CXCL11','IL1B','IL6','CCL2','IRF7','GLUL','RLBP1','SLC1A3']:
 col=index[gene][0]
 vals=np.asarray(x[:,col].todense()).ravel()
 print('\\n',gene)
 for ct in ['muller glia cell','astrocyte','macrophage','rod cell','bipolar cell']:
  for sample in ['eye1','eye2']:
   mask=(df.ct.values==ct)&(labels==sample)
   if mask.sum()==0: continue
   v=vals[mask]
   print(ct,sample,'n',mask.sum(),'mean',round(v.mean(),3),'pct>0',round((v>0).mean()*100,1),'median',round(float(np.median(v)),3))
PY"
