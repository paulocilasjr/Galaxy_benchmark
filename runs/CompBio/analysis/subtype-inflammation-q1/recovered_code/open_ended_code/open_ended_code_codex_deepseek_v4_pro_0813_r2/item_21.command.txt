import scanpy as sc, pandas as pd, numpy as np, warnings; warnings.filterwarnings('ignore')
adata=sc.read('retina_combined.h5ad')
X=adata.raw.X
varn=list(adata.raw.var.index)
obs=adata.obs
markers={
'Rod':['RHO','NRL','PDE6A','PDE6B','CNGB1','SAG'],
'Cone':['ARR3','OPN1SW','OPN1MW','OPN1LW','PDE6H','GNAT2'],
'BC_ON':['VSX2','CABP5','GRM6','TRPM1','GPR179','ISL1'],
'BC_OFF':['GRIK1','SLC4A?'], # skip
'RGC':['POU4F1','POU4F2','RBPMS','THY1','SLC17A6','NEFL'],
'Amacrine':['GAD1','GAD2','TFAP2A','TFAP2B','SLC6A9','CALB2'],
'Horizontal':['ONECUT1','ONECUT2','LHX1','CALB1'],
'Muller':['RLBP1','GLUL','SLC1A3','SOX9','CLU','DKK3'],
'Astro':['GFAP','ALDH1L1','PAX2','S100B'],
'Microglia':['P2RY12','TMEM119','CX3CR1','CSF1R','C1QA','C1QB','AIF1','TYROBP'],
'Macrophage':['CD163','MRC1','F13A1','LYZ','CD14','S100A8','FCGR3A'],
'T':['CD3D','CD3E','CD3G','TRAC','CD2','LCK'],
'B':['CD79A','CD79B','MS4A1','CD19','BANK1','MZB1'],
'Endo':['PECAM1','VWF','CDH5','CLDN5','KDR','FLT1'],
'Pericyte':['PDGFRB','RGS5','NOTCH3','ACTA2'],
'Fibro':['DCN','LUM','COL1A1','COL1A2','COL3A1'],
'Schwann':['MPZ','PLP1','SOX10','NGFR'],
'RPE':['RPE65','BEST1','TYRP1','PMEL','MLANA','RGR','LRAT'],
}
# use expression matrix, print columns = clusters, rows = celltype: mean of means, or top genes
means=[]
for ct,genes in markers.items():
    present=[g for g in genes if g in varn]
    inds=[varn.index(g) for g in present]
    m=np.asarray(X[:,inds].mean(axis=1)).ravel()
    df=pd.DataFrame({'val':m,'leiden':obs.leiden.values,'gene':ct})
    means.append(df.groupby('leiden').val.mean().rename(ct))
M=pd.DataFrame(means).T
# display top celltype per cluster and matrix
pd.set_option('display.width',250)
print('Cell type score means by cluster (rows clusters, cols types)')
print(M.round(2).to_string())
# top genes per cluster from markers
print('\\nTop celltype per cluster:')
for cl in M.index:
 print(cl, ' '.join(f'{k}:{M.loc[cl,k]:.2f}' for k in M.loc[cl].sort_values(ascending=False)[:4].index))
PY
python3 dotplot.py > dotplot_out.txt
