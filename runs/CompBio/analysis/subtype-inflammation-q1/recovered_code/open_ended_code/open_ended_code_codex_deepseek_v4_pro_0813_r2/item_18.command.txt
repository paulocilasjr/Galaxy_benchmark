/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read('retina_combined.h5ad')
varn=list(adata.raw.var.index)
genes=['TF','GABRG3','NEAT1','CP','RLBP1','GLUL','AQP4','SLC1A3','VIM','SOX9','CLU','GFAP','ALDH1L1','P2RY12','C1QA','C1QB','AIF1','CD74','PECAM1','VWF','CDH5','PDGFRB','DCN','RPE65','BEST1','PMEL']
X=adata.raw.X
for cl in ['6','26','0','9','10','12','15','27']:
 mask=(adata.obs['leiden']==cl).values
 print('\\nCluster',cl,'n',mask.sum(),'mean genes',np.asarray((X[mask]>0).sum(1)).mean(),'mean counts',np.asarray(adata.layers['counts'][mask].sum(1)).mean())
 print(pd.Series({g:float(np.asarray(X[mask,varn.index(g)].mean())) for g in genes if g in varn}).round(2).to_string())
PY"
