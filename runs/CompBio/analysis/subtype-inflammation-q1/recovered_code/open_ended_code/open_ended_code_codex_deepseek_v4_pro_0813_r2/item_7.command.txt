/bin/bash -c "cat > explore.py <<'PY'
import gzip, scipy.io, scipy.sparse as sp, numpy as np, pandas as pd, anndata as ad
from collections import Counter
genes = pd.read_csv('subtype.inflammation.q1.genes.txt.gz', sep='\\t')
Xs=[]
for f in ['subtype.inflammation.q1.1.mtx.gz','subtype.inflammation.q1.2.mtx.gz']:
    with gzip.open(f,'rb') as fh:
        X=scipy.io.mmread(fh).tocsr()
    Xs.append(X)
A=sp.vstack(Xs, format='csr')
print(A.shape, A.sum())
counts=np.asarray(A.sum(axis=1)).ravel()
ncell=np.asarray((A>0).sum(axis=1)).ravel()
print('mean genes/cell', ncell.mean(), np.percentile(ncell,[1,10,50,90,99]))
print('mean umi/cell', counts.mean(), np.percentile(counts,[1,10,50,90,99]))
# top expressed genes
idx=np.argsort(counts)[::-1][:50]
for i in idx[:50]:
 print(i, genes.iloc[i]['gene_symbols'], counts[i], ncell[i])
PY
python3 explore.py"
