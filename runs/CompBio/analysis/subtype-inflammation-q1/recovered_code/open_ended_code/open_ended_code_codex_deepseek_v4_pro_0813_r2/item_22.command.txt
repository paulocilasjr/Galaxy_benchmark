/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np, warnings; warnings.filterwarnings('ignore')
adata=sc.read('retina_combined.h5ad')
X=adata.raw.X
varn=list(adata.raw.var.index)
obs=adata.obs
def vals(genes):
 return np.column_stack([np.asarray(X[:,varn.index(g)].todense()).ravel() for g in genes if g in varn])
# print counts above threshold for key markers
thresholds={'Microglia':(['P2RY12','TMEM119','C1QA','C1QB','AIF1','CX3CR1','CSF1R'],0.5),'Macrophage':(['CD163','MRC1','F13A1','LYZ','S100A8','S100A9'],0.5),'T':(['CD3D','CD3E','CD3G','TRAC','CD2'],0.5),'B':(['CD79A','CD79B','MS4A1','CD19'],0.5),'Endo':(['PECAM1','VWF','CDH5','CLDN5','KDR'],0.5),'Pericyte':(['PDGFRB','RGS5','NOTCH3'],0.5),'Fibro':(['DCN','LUM','COL1A1'],0.5),'Schwann':(['MPZ','PLP1','SOX10'],0.5),'RPE':(['RPE65','BEST1','PMEL','TYRP1'],0.5),'Muller':(['RLBP1','GLUL','SLC1A3'],0.5),'Astro':(['GFAP','ALDH1L1'],0.5)}
for name,(genes,t) in thresholds.items():
    present=[g for g in genes if g in varn]
    if not present: continue
    V=vals(present)
    mask=(V>t).any(axis=1)
    n=mask.sum(); n1=(mask & (obs.batch.values=='eye1')).sum(); n2=(mask & (obs.batch.values=='eye2')).sum()
    print(f'{name}: any > {t}: n={n} (eye1 {n1}, eye2 {n2})')
    if n>0:
        print('  cluster dist', obs.loc[mask,'leiden'].value_counts().head(10).to_dict())
        # top batch by clusters
        sub=obs.loc[mask,['leiden','batch']]
        print('  by cluster batch', pd.crosstab(sub.leiden, sub.batch).to_dict())
PY"
