/bin/bash -lc "python -u - <<'PY'
import analyze_retina as ar
import numpy as np
from scipy import sparse
print('load',flush=True)
genes=ar.load_genes('subtype.inflammation.q1.genes.txt.gz'); pos={g:i for i,g in enumerate(genes)}
a=ar.load_matrix('subtype.inflammation.q1.1.mtx.gz'); b=ar.load_matrix('subtype.inflammation.q1.2.mtx.gz')
la,_=ar.normalized_log(a); lb,_=ar.normalized_log(b); cmb=sparse.hstack([la,lb],format='csr'); clusters=np.load('cluster_labels.npy')
keys=['TF','CTNNA2','GABRG3','FOXP2','LSAMP','WIF1','PDE8B','TRPM3','GPM6A','SLC1A3','GLUL','RLBP1','VIM','CLU','AQP4','S100B','ALDOC','KCNJ10','GFAP','RHO','PDE6A','EYS','NRL','RCVRN','SAG','GNAT1','LHX1','ONECUT1','CALB1','PROX1','GAD1','GAD2','VSX2','GRM6','TRPM1','CABP5','PRKCA','RBPMS','THY1','NEFL','KDR','FLT1','PDGFRB','RGS5','MPZ','SOX10','APOE','CST3','MARCO','P2RY12','TYROBP','HLA-DRA','CD74']
for c in [0,4,12,11,18,20,1,8,9]:
 print('\\ncluster',c,flush=True); ids=np.flatnonzero(clusters==c)
 for eye,ii in [('e1',ids[ids<a.shape[1]]),('e2',ids[ids>=a.shape[1]])]:
  if not len(ii):continue
  m=np.asarray(cmb[:,ii].mean(1)).ravel()
  vals=' '.join(f'{g}={m[pos[g]]:.2f}' for g in keys if m[pos[g]]>.05)
  print(eye,'n',len(ii),vals,flush=True)
PY"
