/bin/bash -lc "python - <<'PY'
import analyze_retina as ar
import numpy as np
from scipy import sparse

genes=ar.load_genes('subtype.inflammation.q1.genes.txt.gz'); pos={g:i for i,g in enumerate(genes)}
a=ar.load_matrix('subtype.inflammation.q1.1.mtx.gz'); b=ar.load_matrix('subtype.inflammation.q1.2.mtx.gz')
la,_=ar.normalized_log(a); lb,_=ar.normalized_log(b); cmb=sparse.hstack([la,lb],format='csr'); cl=np.load('cluster_labels.npy')
inf=['IL1B','CCL2','CCL7','CXCL9','CXCL10','CXCL11','IL6','CSF3','NFKBIA','CCL20','CD69','TNFRSF9']
markers={
'muller':['SLC1A3','GLUL','RLBP1','VIM','CLU','AQP4','SOX9'],
'astro':['GFAP','AQP4','S100B','ALDH1L1','GJA1'],
'rod':['RHO','GNAT1','PDE6A','SAG','NRL','RCVRN','EYS'],
'immune':['AIF1','TYROBP','LST1','C1QA','CST3','LYZ','CD68']}
for name,cs in [('affected clusters',[0,4,12]),('all other clusters',None),('rod clusters',[1,5,8,9,13,15,17,21,23])]:
 ids=np.flatnonzero(np.isin(cl,cs)) if cs is not None else np.flatnonzero(~np.isin(cl,[0,4,12]))
 e1=ids[ids<a.shape[1]]; e2=ids[ids>=a.shape[1]]
 print('\\n',name,'n',len(e1),len(e2))
 for g in inf+sum(markers.values(),[]):
  m1=float(np.asarray(cmb[pos[g],e1].mean())) if len(e1) else 0
  m2=float(np.asarray(cmb[pos[g],e2].mean())) if len(e2) else 0
  if g in inf or max(m1,m2)>.15: print(g,round(m1,3),round(m2,3),'delta',round(m1-m2,3))
PY"
