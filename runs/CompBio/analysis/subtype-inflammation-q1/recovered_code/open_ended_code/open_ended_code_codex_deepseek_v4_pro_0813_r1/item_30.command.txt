/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('combined.h5ad')
genes=['P2RY12','CSF1R','CX3CR1','AIF1','C1QA','C1QB','C1QC','TYROBP','ITGAM','CD14','LYZ','MRC1','FCGR3A','CD68','GFAP','RLBP1','SLC1A3']
present=[g for g in genes if g in adata.var_names]
for cl in sorted(adata.obs['leiden'].unique(), key=lambda x:int(x)):
 sub=adata[adata.obs['leiden']==cl]
 vals={g:float(np.asarray(sub[:,g].X.mean())) for g in present}
 # score microglia first 8
 score=np.mean([vals[g] for g in ['P2RY12','CSF1R','CX3CR1','AIF1','C1QA','C1QB','C1QC','TYROBP'] if g in vals])
 if score>0.05 or any(vals.get(g,0)>0.5 for g in ['P2RY12','CSF1R','CX3CR1','AIF1','C1QA','C1QB','C1QC','TYROBP']):
  print(cl, sub.n_obs, 'micro_score',round(score,2), '; '.join([f'{g}={vals.get(g,0):.2f}' for g in ['P2RY12','CSF1R','CX3CR1','AIF1','C1QA','C1QB','C1QC','TYROBP','CD14','LYZ','GFAP','RLBP1','SLC1A3']]))
PY"
