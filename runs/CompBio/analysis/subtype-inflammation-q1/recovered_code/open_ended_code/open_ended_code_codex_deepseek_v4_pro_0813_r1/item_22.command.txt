/bin/bash -c "cat > annotate.py <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('combined.h5ad')
# Ensure log counts in X and raw counts in raw? raw assigned log1p; for rank use log.
# individual markers
genes=['RHO','NRL','PDE6A','GNAT1','ARR3','GNAT2','OPN1SW','OPN1MW','OPN1LW','PDE6H','RCVRN','CRX',
'RPE65','BEST1','SERPINF1','PMEL','MLANA','TYRP1',
'ONECUT1','LHX1','CALB1','GAD2','SLC17A6','PVALB','TAC1','TFAP2A',
'VSX2','OTX2','GRIK1','TRPM1','CABP5','ISL1','PRKCA',
'RBPMS','THY1','SNCG','NEFL','NEFM','POU4F1','POU4F2','SLC17A6',
'RLBP1','GLUL','SLC1A3','CLU','DKK3','VIM','APOE','GFAP','AQP4','S100B',
'P2RY12','TMEM119','CX3CR1','CSF1R','AIF1','C1QA','C1QB','ITGAM','TYROBP',
'LYZ','CD14','MRC1','CD68','FCGR3A','S100A8','S100A9',
'CD3D','CD3E','CD3G','TRAC','LCK','IL7R','CD2',
'CD79A','CD79B','MS4A1','CD19','BANK1','PAX5',
'MPZ','PMP22','SOX10','MBP','PLP1','NGFR',
'PDGFRB','RGS5','ACTA2','NOTCH3','CSPG4',
'DCN','COL1A1','COL1A2','LUM','FBLN1',
'PECAM1','VWF','CLDN5','CDH5','EGFL7','KDR','FLT1',
# inflammation
'C1QA','C1QB','C1QC','HLA-DRA','HLA-DRB1','HLA-DPA1','CD74','CXCL8','IL1B','TNF','NFKBIA','CXCL10','CCL2','CCL3','CCL4','STAT1','IRF1','ISG15','IFITM3','SOD2','FOS','JUN','B2M'
]
present=[g for g in genes if g in adata.var_names]
missing=[g for g in genes if g not in adata.var_names]
print('missing',missing)
# mean expression per cluster
cats=sorted(adata.obs['leiden'].unique(), key=lambda x:int(x))
rows=[]
for cl in cats:
    sub=adata[adata.obs['leiden']==cl]
    row={'cluster':cl,'n':sub.n_obs}
    for g in present:
        row[g]=np.asarray(sub[:,g].X.mean()).ravel()[0]
    rows.append(row)
df=pd.DataFrame(rows).set_index('cluster')
# show selected
sel=['RHO','NRL','ARR3','GNAT2','RPE65','BEST1','ONECUT1','VSX2','GRIK1','RBPMS','THY1','RLBP1','GFAP','P2RY12','TMEM119','C1QA','LYZ','CD14','CD3D','CD79A','MPZ','PDGFRB','DCN','PECAM1']
print('\\nSelected mean log expr by cluster:')
print(df[['n']+sel].round(2).to_string())
# inflammation score mean across genes
inf=['C1QA','C1QB','C1QC','HLA-DRA','HLA-DRB1','HLA-DPA1','CD74','CXCL8','IL1B','TNF','NFKBIA','CXCL10','CCL2','CCL3','CCL4','STAT1','IRF1','ISG15','IFITM3','SOD2','FOS','JUN','B2M']
inf=[g for g in inf if g in df.columns]
df['inflam_score']=df[inf].mean(axis=1)
print('\\nTop inflam_score clusters:')
print(df[['n','inflam_score']].sort_values('inflam_score',ascending=False).head(10).round(3).to_string())
# save marker table
df.to_csv('marker_means.csv')
print('saved marker_means.csv')
PY
python3 annotate.py"
