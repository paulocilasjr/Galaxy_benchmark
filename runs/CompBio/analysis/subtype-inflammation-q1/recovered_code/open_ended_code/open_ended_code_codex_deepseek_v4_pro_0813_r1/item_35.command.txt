/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('combined.h5ad')
for cl in ['10','11']:
 sub=adata[adata.obs['leiden']==cl]
 for gene in ['RLBP1','GLUL','SLC1A3','GFAP','CD14','IL1B','CXCL10','CCL2','NFKBIA','IRF1']:
  x=np.asarray(sub[:,gene].X.toarray()).ravel() if hasattr(sub[:,gene].X,'toarray') else np.asarray(sub[:,gene].X).ravel()
  print(cl, gene, 'pct>0',(x>0).mean()*100, 'mean',x.mean())
 print('coexpr RLBP1 & CD14', np.mean((np.asarray(sub[:,'RLBP1'].X.toarray()).ravel()>0)&(np.asarray(sub[:,'CD14'].X.toarray()).ravel()>0))*100)
 print('coexpr RLBP1 & IL1B', np.mean((np.asarray(sub[:,'RLBP1'].X.toarray()).ravel()>0)&(np.asarray(sub[:,'IL1B'].X.toarray()).ravel()>0))*100)
 print('coexpr GLUL & CD14', np.mean((np.asarray(sub[:,'GLUL'].X.toarray()).ravel()>0)&(np.asarray(sub[:,'CD14'].X.toarray()).ravel()>0))*100)
 print()
PY"
