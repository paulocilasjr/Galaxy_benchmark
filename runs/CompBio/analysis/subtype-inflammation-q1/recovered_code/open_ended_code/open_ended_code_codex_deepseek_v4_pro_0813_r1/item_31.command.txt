/bin/bash -c "cat > celltype_diff.py <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('combined.h5ad')
# cell type marker sets
sets={
'astrocyte':['GFAP','AQP4','S100B','SLC1A3','GJA1','SLC1A2'],
'microglial cell':['P2RY12','CSF1R','CX3CR1','AIF1','C1QA','C1QB','TYROBP'],
'RPE cell':['RPE65','BEST1','SERPINF1','TYRP1','PMEL','MLANA'],
'cone cell':['ARR3','GNAT2','PDE6H','OPN1SW','OPN1MW','OPN1LW','CRX','RCVRN'],
'rod cell':['RHO','NRL','PDE6A','GNAT1','NR2E3','SAG','CNGB1','RCVRN'],
'horizontal cell':['ONECUT1','LHX1','CALB1','GAD2','SYT1'],
'bipolar cell':['VSX2','OTX2','GRIK1','TRPM1','CABP5','ISL1'],
'RGC':['RBPMS','THY1','SNCG','NEFL','NEFM','POU4F2','SLC17A6'],
'T cells':['CD3G','TRAC','TRBC2','LCK','IL7R'],
'macrophage':['LYZ','CD14','MRC1','CD68','FCGR3A','S100A8','S100A9'],
'Schwann cell':['MPZ','PMP22','SOX10','MBP','PLP1','NGFR'],
'Pericyte':['PDGFRB','RGS5','ACTA2','NOTCH3','CSPG4'],
'B cell':['CD19','BANK1','PAX5'],
'fibroblast':['DCN','COL1A1','COL1A2','LUM','VIM','FBLN1'],
'muller glia cell':['RLBP1','GLUL','SLC1A3','CLU','DKK3','VIM','APOE'],
'endothelial cell':['PECAM1','VWF','CLDN5','CDH5','EGFL7','KDR','FLT1'],
}
inf=['C1QA','C1QB','C1QC','HLA-DRA','HLA-DRB1','HLA-DPA1','CD74','CXCL8','IL1B','TNF','NFKBIA','CXCL10','CCL2','CCL3','CCL4','STAT1','IRF1','ISG15','IFITM3','SOD2','FOS','JUN','B2M']
# Add scores for cell types and inflammation
for name, genes in sets.items():
    present=[g for g in genes if g in adata.var_names]
    if len(present)<1: continue
    sc.tl.score_genes(adata, present, score_name='score_'+name.replace(' ','_'), ctrl_size=min(50,len(present)))
    print('scored',name,len(present),flush=True)
present_inf=[g for g in inf if g in adata.var_names]
sc.tl.score_genes(adata, present_inf, score_name='inflam_score', ctrl_size=min(50,len(present_inf)))
print('scored inflammation',len(present_inf),flush=True)
# assign cell type max among type scores
score_cols=['score_'+n.replace(' ','_') for n in sets if 'score_'+n.replace(' ','_') in adata.obs.columns]
mat=adata.obs[score_cols].values
names=[c.replace('score_','').replace('_',' ') for c in score_cols]
adata.obs['pred_type']=np.array(names)[np.argmax(mat,axis=1)]
print(adata.obs.pred_type.value_counts())
# map underscores back to spaces manually for final
# compare inflam score by pred_type and sample
res=adata.obs.groupby(['pred_type','sample']).agg(n=('inflam_score','size'), inflam_mean=('inflam_score','mean')).reset_index()
piv=res.pivot(index='pred_type',columns='sample',values='inflam_mean')
piv['n1']=res[res.sample=='eye1'].set_index('pred_type')['n']
piv['n2']=res[res.sample=='eye2'].set_index('pred_type')['n']
piv['diff_eye1_minus_eye2']=piv['eye1']-piv['eye2']
piv=piv.sort_values('diff_eye1_minus_eye2',ascending=False)
print('\\nInflammation score by predicted type and sample:')
print(piv.round(4).to_string())
# Also raw mean log expression inflammation per pred_type/sample
raw=adata.obs.copy()
raw['raw_inf']=np.asarray(adata[:,present_inf].X.mean(axis=1)).ravel()
r=raw.groupby(['pred_type','sample']).agg(n=('raw_inf','size'), raw_inf_mean=('raw_inf','mean')).reset_index().pivot(index='pred_type',columns='sample',values='raw_inf_mean')
r['diff']=r['eye1']-r['eye2']
print('\\nRaw mean log expr inflammation by predicted type/sample:')
print(r.sort_values('diff',ascending=False).round(4).to_string())
adata.obs.to_csv('celltype_assignments.csv')
PY
python3 celltype_diff.py"
