import gzip,numpy as np,gc,warnings
from scipy.io import mmread
from scipy.sparse import coo_matrix,vstack
from sklearn.decomposition import TruncatedSVD

warnings.filterwarnings('ignore')

# marker panels; score mean expression of available markers, weighted down generic-ish
panels={
'astrocyte':['GFAP','AQP4','S100B','SLC1A3','ALDOC','KCNJ10','AGT','WIF1','TF','CP'],
'microglial cell':['P2RY12','TMEM119','CX3CR1','AIF1','C1QA','C1QB','C1QC','CSF1R','TYROBP','LST1','FCER1G','HLA-DRA','CTSS','CST3'],

'RGC':['RBPMS','SNCG','NEFL','NEFM','ISL1','THY1','RBP4','PVALB','SLC17A7','TUBB3'],
'T cells':['CD3D','CD3E','CD3G','TRBC1','TRBC2','IL7R','LTB','CCL5','NKG7','CD8A','CD8B','KLRD1','LCK','TRAC','IL32','RASGRP1','CD247'],
'macrophage':['CD68','LILRB1','FCGR3A','APOC1','MSR1','MRC1','MARCO','CTSB','CTSD','TYROBP','LST1','FCER1G','CST3','STAB1','FPR1','OLR1','CCL2','CCL7','CYBB'],
'Schwann cell':['S100B','SOX10','PLP1','MPZ','MBP','GFAP','NCAM1','PMP22','NGFR'],
'Pericyte':['RGS5','CSPG4','PDGFRB','MCAM','NOTCH3','ACTA2','COL4A1','COL4A2','DES','KCNJ8','ABCC9'],
'B cell':['CD79A','MS4A1','CD37','CD74','HLA-DRA','CD79B','CD83','CD22','CD19','CD52'],
'fibroblast':['COL1A1','COL1A2','COL3A1','DCN','LUM','COL6A1','PDGFRA','C7','COL5A1','COL6A2'],
'muller glia cell':['RLBP1','GLUL','SLC1A3','VIM','AQP4','KCNJ10','RAX','SOX9','CLU','SLC1A2','ALDOC','SLC4A4','WIF1','TF','CP','FOXP2'],
'endothelial cell':['EMCN','PECAM1','VWF','KDR','ESAM','ENG','RAMP2','PLVAP','CLDN5','CA4','RGCC'],
}
# map genes to selected positions; zscore each feature across cells for score
muX=np.asarray(X.mean(0)).ravel(); sdX=np.sqrt(np.maximum(np.asarray(X.multiply(X).mean(0)).ravel()-muX*muX,1e-4))
P={t:[np.where(hvg==idx[g])[0][0] for g in ms if g in idx and idx[g] in set(hvg)] for t,ms in panels.items()}

cpos=np.where(hvg==idx['CCL5'])[0][0]
for c in range(40):
 sel=lab==c; cmean=np.asarray(X[sel].mean(0)).ravel(); cs=(cmean[cpos] > 0) # log mean not prevalence
 if (np.asarray(X[sel,cpos].todense()).ravel()>0).sum()==0: continue
 scores=[]
 for t,pp in P.items(): scores.append((float(np.mean((cmean[pp]-muX[pp])/sdX[pp])) if pp else -99,t))
 print('\\ncluster',c,'n',sel.sum(),'eye1',sel[:ns[0]].sum(),'eye2',sel[ns[0]:].sum(),'ccl5+',int((np.asarray(X[sel,cpos].todense()).ravel()>0).sum()),'scores',sorted(scores,reverse=True))
 print('panel genes',end=' ')
 for g in ['SLC1A3','RLBP1','GLUL','VIM','AQP4','WIF1','TF','CP','FOXP2','GABRG3','P2RY12','ADGRE1','MARCO','STAB1','OLR1','CCL2','CCL7','CYBB','CCL5','IL7R','TRAC','RBPMS','SLC17A7','GFAP','ALDOC']:
  if g in idx and idx[g] in set(hvg): print(f'{g}:{cmean[np.where(hvg==idx[g])[0][0]]:.2f}',end=' ')
 print()
