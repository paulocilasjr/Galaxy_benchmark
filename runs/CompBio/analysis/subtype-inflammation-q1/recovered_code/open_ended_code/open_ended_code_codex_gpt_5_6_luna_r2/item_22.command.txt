/bin/bash -lc "python -u - <<'PY'
import gzip,numpy as np
from scipy.io import mmread
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as h:
 next(h);genes=[x.rstrip().split('\\t')[1] for x in h]
idx={g:i for i,g in enumerate(genes)}
markers=['GFAP','AQP4','S100B','SLC1A3','ALDOC','KCNJ10','AGT','WIF1','TF','CP','GJA1','CLDN11','ALDH1L1','C3','APOE','LGALS3','PAX2','PAX8','COL15A1','SPARCL1','SLC14A1','RLBP1','GLUL','VIM','RAX','SOX9','CLU','SLC1A2','SLC4A4','DIO2','FABP7','CPE','GABRG3','GABRG3-AS1','FOXP2','DACH1','CTNNA2','LSAMP','LRRC4C','PIP5K1B','TRPM3','CABP5','PRKCA','VSX2','GRM6','TRPM1','PCP4','RBPMS','SNCG','NEFL','ISL1','THY1','ONECUT1','LHX1','PROX1','CALB1']
for f in ['subtype.inflammation.q1.1.mtx.gz','subtype.inflammation.q1.2.mtx.gz']:
 X=mmread(f).tocsc().astype(np.float32);lib=np.asarray(X.sum(0)).ravel(); c=np.asarray(X[idx['CCL5'],:].todense()).ravel() if 'CCL5' in idx else np.zeros(X.shape[1]); cells=c>0
 print('\\n',f,'CCL5+ n',cells.sum())
 for typ,ms in [('astro',markers[:20]),('muller',markers[20:34]),('identity',markers[34:])]:
  print(typ)
  for m in ms:
   i=idx[m]; row=X[i,cells]; print(f'{m:12s} {row.nnz/cells.sum()*100:5.1f}% {float(row.multiply(1e4/lib[cells]).mean()):7.3f}',end=' | ')
  print()
PY"
