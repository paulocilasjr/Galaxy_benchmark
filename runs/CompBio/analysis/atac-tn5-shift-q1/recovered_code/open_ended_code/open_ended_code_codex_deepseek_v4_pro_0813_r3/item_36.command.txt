/bin/bash -lc "python - <<'PY'
import numpy as np
D=np.load('shift_counts_fast.npz'); A=D['Avals']; B=D['Bvals']; L=D['left_counts']; R=D['right_counts']
bases='ACGT'; w=10
for name,arr,vals,v in [('left A=2',L,A,2),('right B=-1',R,B,-1),('left standard A=4',L,A,4),('right standard B=-5',R,B,-5)]:
 i=np.where(vals==v)[0][0]; c=arr[i]
 print('\\n',name)
 print('pos', 'A','C','G','T','cons')
 for p in range(-8,9):
  j=p+w; tot=c[j].sum();
  fs=c[j]/tot
  cons=bases[np.argmax(fs)]
  print(f'{p:3d}', ' '.join(f'{x:.3f}' for x in fs), cons)
PY"
