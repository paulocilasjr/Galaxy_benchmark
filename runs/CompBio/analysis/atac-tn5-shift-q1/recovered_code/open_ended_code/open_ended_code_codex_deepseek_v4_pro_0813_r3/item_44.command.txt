/bin/bash -lc "python - <<'PY'
import urllib.request, json
url='https://api.github.com/repos/JWolpe/Tn5bias/git/trees/master?recursive=1'
req=urllib.request.Request(url,headers={'User-Agent':'curl'})
try:
 data=json.load(urllib.request.urlopen(req,timeout=30))
 for item in data['tree']:
  path=item['path']
  if 'bias' in path.lower() or 'pwm' in path.lower() or 'motif' in path.lower() or 'seqout' in path.lower() or path.endswith('.txt') or path.endswith('.csv') or path.endswith('.rds'):
   print(path)
except Exception as e: print('ERR',e)
PY"
