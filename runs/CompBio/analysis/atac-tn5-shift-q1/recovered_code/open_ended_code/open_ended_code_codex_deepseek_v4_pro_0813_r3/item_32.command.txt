/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
frag='unknown.shift.frag.bed.gz'
df=pd.read_csv(frag,sep='\\t',header=None,usecols=[1,2],names=['s','e'])
lens=df['e']-df['s']
# histogram bins
hist,bins=np.histogram(lens,bins=np.arange(0,1001,5))
for i in range(len(hist)):
 if hist[i]>500:
  print(bins[i],bins[i+1],hist[i])
# local maxima
h=hist.astype(float)
for i in range(1,len(h)-1):
 if h[i]>h[i-1] and h[i]>=h[i+1] and h[i]>1000:
  print('peak',bins[i],bins[i+1],h[i])
print('percentiles',np.percentile(lens,[1,5,10,25,50,75,90,95,99]))
PY"
