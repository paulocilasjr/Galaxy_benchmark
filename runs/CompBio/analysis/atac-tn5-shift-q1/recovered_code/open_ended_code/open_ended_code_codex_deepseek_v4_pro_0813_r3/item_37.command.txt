/bin/bash -lc "python - <<'PY'
import numpy as np
D=np.load('shift_counts_fast.npz'); A=D['Avals']; B=D['Bvals']; L=D['left_counts']; R=D['right_counts']
Lf=L/(L.sum(2,keepdims=True)+1e-12); Rf=R/(R.sum(2,keepdims=True)+1e-12)
Lc=Lf.reshape(Lf.shape[0],-1); Rc=Rf.reshape(Rf.shape[0],-1)
Lm=Lc-Lc.mean(1,keepdims=True); Rm=Rc-Rc.mean(1,keepdims=True)
corr=Lm@Rm.T/np.sqrt((Lm*Lm).sum(1,keepdims=True)@(Rm*Rm).sum(1,keepdims=True).T)
for a,b in [(2,-1),(4,-5),(1,0),(0,1),(3,-2)]:
 i=np.where(A==a)[0][0];j=np.where(B==b)[0][0]
 print(a,b,corr[i,j])
PY"
