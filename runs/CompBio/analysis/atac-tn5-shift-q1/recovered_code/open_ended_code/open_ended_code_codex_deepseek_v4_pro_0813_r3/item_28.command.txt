/bin/bash -lc "python - <<'PY'
import numpy as np
D=np.load('shift_counts_fast.npz')
Avals=D['Avals']; Bvals=D['Bvals']; lc=D['left_counts']; rc=D['right_counts']
bases='ACGT'
def cons(counts):
 totals=counts.sum(axis=1,keepdims=True); f=counts/np.maximum(totals,1)
 idx=np.argmax(f,axis=1)
 return ''.join(bases[int(i)] for i in idx)
for arr,vals,name in [(lc,Avals,'left'),(rc,Bvals,'right')]:
 print('---',name)
 for v in vals:
  i=int(np.where(vals==v)[0][0])
  c=arr[i]
  print('shift',v, cons(c))
PY"
