/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
# Use rows from overexpress 300k and unknown all, compare profiles for shift
from twobitreader import TwoBitFile
import gzip,numpy as np,random
f=TwoBitFile('hg38.2bit'); ix={b:i for i,b in enumerate('ACGT')}; trans=str.maketrans('ACGT','TGCA')
def profile(fn, mode, maxn=None):
 S=np.zeros((41,4));n=0
 with gzip.open(fn,'rt') as h:
  for line in h:
   c,s,e,*_=line.rstrip().split('\\t');s=int(s);e=int(e);seq=f[c]
   q=seq[s-20:s+21] if mode=='s' else seq[e-21:e+20].translate(trans)[::-1]
   for j,x in enumerate(q.upper()):
    if x in ix:S[j,ix[x]]+=1
   n+=1
   if maxn and n>=maxn:break
 return S/S.sum(1,keepdims=True)
U=profile('unknown.shift.frag.bed.gz','s'); O=profile('overexpress.ref.bed.gz','s',300000)
for d in range(-10,11):
 vals=[]
 for i in range(41):
  j=i+d
  if 0<=j<41:vals.append(np.sum((U[i]-O[j])**2))
 print(d,np.mean(vals),np.corrcoef(U[max(0,-d):min(41,41-d)].ravel(),O[max(0,d):min(41,41+d)].ravel())[0,1])
PY"
