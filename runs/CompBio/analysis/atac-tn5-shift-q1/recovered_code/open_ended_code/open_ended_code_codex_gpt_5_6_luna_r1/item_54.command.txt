/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip,numpy as np,time
f=TwoBitFile('hg38.2bit'); offs=range(-12,13)
cs=np.zeros((25,4),int); ce=np.zeros((25,4),int); ce2=np.zeros((25,4),int)
ix=np.full(256,-1,dtype=int); [ix.__setitem__(ord(b),i) for i,b in enumerate('ACGT')]
trans=bytes.maketrans(b'ACGT',b'TGCA'); n=0;t=time.time()
with gzip.open('unknown.shift.frag.bed.gz','rt') as h:
 for line in h:
  c,s,e,*_=line.rstrip().split('\\t'); s=int(s);e=int(e); seq=f[c]
  a=seq[s-12:s+13].upper().encode();
  # oriented end with k relative to e-1; fetch genomic [e-13,e+12]
  raw=seq[e-13:e+12].upper().encode() # indexes genomic e-13..e+11; k=-12 -> e+11, k=+12 -> e-13
  b=raw.translate(trans)[::-1]
  for j,x in enumerate(a):
   z=ix[x]
   if z>=0:cs[j,z]+=1
  for j,x in enumerate(b):
   z=ix[x]
   if z>=0:ce[j,z]+=1
  n+=1
print('n',n,'sec',time.time()-t)
print('offset start A C G T endrev A C G T')
for k,a,b in zip(offs,cs,ce):
 print(f'{k:3d}',*(f'{x/a.sum():.4f}' for x in a),*(f'{x/b.sum():.4f}' for x in b),f'AT {((a[0]+a[3])/a.sum()):.4f} {((b[0]+b[3])/b.sum()):.4f}')
PY"
