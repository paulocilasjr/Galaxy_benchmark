/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip,numpy as np
f=TwoBitFile('hg38.2bit'); offs=np.arange(-20,21); n=len(offs)
S=np.zeros((n,4),float); E=np.zeros((n,4),float); ix={b:i for i,b in enumerate('ACGT')}; comp=str.maketrans('ACGT','TGCA')
with gzip.open('unknown.shift.frag.bed.gz','rt') as h:
 for line in h:
  c,s,e,*_=line.rstrip().split('\\t');s=int(s);e=int(e);seq=f[c]
  a=seq[s-20:s+21].upper(); b=seq[e-21:e+20].upper().translate(comp)[::-1] # k=0 = e-1? raw index e-21..e+19, reverse: index20=e-1 yes
  for j,x in enumerate(a):
   if x in ix:S[j,ix[x]]+=1
  for j,x in enumerate(b):
   if x in ix:E[j,ix[x]]+=1
S/=S.sum(1,keepdims=True); E/=E.sum(1,keepdims=True)
# compare E[k+d] to S[k], use overlap within n
for d in range(-15,16):
 # end index j = start idx + d
 vals=[]
 for i in range(n):
  j=i+d
  if 0<=j<n:
   # cosine/pearson vectors; Euclidean
   vals.append(np.sum((S[i]-E[j])**2))
 print(d, np.mean(vals), np.corrcoef(S[max(0,-d):min(n,n-d)].ravel(),E[max(0,d):min(n,n+d)].ravel())[0,1])
PY"
