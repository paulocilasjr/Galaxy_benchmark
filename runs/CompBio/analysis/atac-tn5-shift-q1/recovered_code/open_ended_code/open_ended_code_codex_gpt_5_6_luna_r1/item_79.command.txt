/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip,numpy as np,random
f=TwoBitFile('hg38.2bit'); random.seed(1); rows=[]; n=0
with gzip.open('overexpress.ref.bed.gz','rt') as h:
 for line in h:
  n+=1
  if len(rows)<300000: rows.append(line)
  else:
   j=random.randrange(n)
   if j<len(rows): rows[j]=line
print('read',n)
for which in ['start','end']:
 S=np.zeros((41,4),float);ix={b:i for i,b in enumerate('ACGT')}; trans=str.maketrans('ACGT','TGCA')
 for line in rows:
  c,s,e,*_=line.rstrip().split('\\t');s=int(s);e=int(e);seq=f[c]
  if which=='start': a=seq[s-20:s+21].upper()
  else: a=seq[e-21:e+20].upper().translate(trans)[::-1]
  for j,x in enumerate(a):
   if x in ix:S[j,ix[x]]+=1
 S/=S.sum(1,keepdims=True)
 print(which)
 best=[]
 for c in range(-8,9):
  l=[];r=[]
  for d in range(1,9):l.append(S[c-d+20]);r.append(S[c+d+20][[3,2,1,0]])
  best.append((np.mean((np.array(l)-np.array(r))**2),c,np.corrcoef(np.array(l).ravel(),np.array(r).ravel())[0,1]))
 print(sorted(best)[:5])
 print(' '.join(f'{S[i,2]:.3f}' for i in range(8,33)))
PY"
