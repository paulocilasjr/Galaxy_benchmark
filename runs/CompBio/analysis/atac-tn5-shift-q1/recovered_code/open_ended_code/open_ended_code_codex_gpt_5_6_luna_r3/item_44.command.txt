comp={'a':'t','c':'g','g':'c','t':'a'}
profiles={}
for side in 'LR':
 cnt={p:collections.Counter() for p in range(-20,21)}
 for line in gzip.open('endpoint_contexts.tsv.gz','rt'):
  s,c,pos,w,seq=line.rstrip().split('\\t');
  if s"'!=side:continue

  if side=='"'R':seq=''.join(comp[x] for x in seq[::-1])
  for p in range(-20,21):
   i=24+p
   if 0<=i<len(seq):cnt[p][seq[i]]+=w
 profiles[side]={p:{b:cnt[p][b]/sum(cnt[p].values()) for b in 'acgt'} for p in cnt}
for side,P in profiles.items():
 print('\\n',side)
 for c in [x/2 for x in range(-12*2,13*2+1)]:
  # integer/reflection positions only; use pairs with coordinates integer around c
  score=0;n=0
  for p in range(-12,13):
   q=2*c-p
   if q not in P or q==p:continue
   score+=sum((P[p][b]-P[q][comp[b]])**2 for b in 'acgt');n+=1
  print(f'{c:5.1f} {score/(n or 1):.5f}')
