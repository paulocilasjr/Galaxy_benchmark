/bin/bash -lc "python3 - <<'PY'
import gzip
cands=[(0,0),(-2,4),(-2,3),(-1,4),(-3,5),(-2,5),(2,-1),(-2,2),(-1,3)]
sets=[set() for _ in cands]
with gzip.open('unknown.shift.frag.bed.gz','rt') as f:
 for line in f:
  p=line.rstrip().split('\\t');ch=p[0];s=int(p[1]);e=int(p[2])
  for idx,(ds,de) in enumerate(cands):sets[idx].add((ch,s+ds,e+de))
counts=[0]*len(cands)
processed=0
with gzip.open('std_part.gz','rt') as f:
 try:
  for line in f:
   if line.startswith('#'):continue
   p=line.rstrip().split('\\t')
   if len(p)<3:continue
   key=(p[0],int(p[1]),int(p[2]))
   processed+=1
   for idx,s in enumerate(sets):
    if key in s:counts[idx]+=1
 except EOFError:
  pass
print('processed',processed)
for c,n in zip(cands,counts):print(c,n)
PY"
