/bin/bash -lc "python3 - <<'PY'
import gzip, math
import pysam
fasta=pysam.FastaFile('hg38.fa')
chrs=[]
with gzip.open('unknown.shift.frag.bed.gz','rt') as f:
 for line in f:
  p=line.rstrip().split('\\t'); chrs.append((p[0],int(p[1]),int(p[2])))
# use all
def extract(ch,c,flank=45):
 lo=c-flank; hi=c+flank+1
 if lo<0:return None
 try: s=fasta.fetch(ch,lo,hi).upper()
 except:return None
 return s if len(s)==2*flank+1 else None
for pos_type,d in [('start',-2),('end',3)]:
 counts=[[0]*4 for _ in range(21)]
 for ch,s,e in chrs:
  c=s if pos_type=='start' else e-1
  seq=extract(ch,c)
  if seq is None:continue
  center=45-d; lo=center-10; hi=lo+21
  for j,b in enumerate(seq[lo:hi]):
   idx='ACGT'.find(b)
   if idx>=0: counts[j][idx]+=1
 print('\\n',pos_type,'d',d)
 print('pos\\tA\\tC\\tG\\tT\\tconsensus\\tmaxfreq')
 for j,pos in enumerate(counts):
  n=sum(pos)
  f=[x/n for x in pos]
  cons='ACGT'[max(range(4),key=lambda i:f[i])]
  print(f'{j-10}\\t'+'\\t'.join(f'{x:.3f}' for x in f)+f'\\t{cons}\\t{max(f):.3f}')
PY"
