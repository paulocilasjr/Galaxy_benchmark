/bin/bash -lc "python3 - <<'PY'
import requests
for branch in ['main','master']:
 url=f'https://api.github.com/repos/loosolab/TOBIAS/git/trees/{branch}?recursive=1'
 r=requests.get(url, timeout=30)
 print(branch,r.status_code)
 if r.ok:
  for item in r.json().get('tree',[]):
   p=item['path']
   if any(s in p.lower() for s in ['bias','pwm','motif','data']):
    print(p)
PY"
