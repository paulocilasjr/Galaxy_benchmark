/bin/bash -lc "python3 - <<'PY'
import gzip, math, pysam
fasta=pysam.FastaFile('hg38.fa')
# build standard PWM from std_part 200k sample
counts_s=[[0]*4 for _ in range(21)]; counts_e=[[0]*4 for _ in range(21)]
n=0;line_i=0
with gzip.open('std_part.gz','rt') as f:
 for line in f:
  if line.startswith('#'):continue
  line_i+=1
  if line_i%40"'!=1:continue
  p=line.rstrip().split('"'\\t');ch=p[0];s=int(p[1]);e=int(p[2])
  try:seqs=fasta.fetch(ch,s-10,s+11).upper(); seqe=fasta.fetch(ch,e-10,e+11).upper()
  except:continue
  if len(seqs)"'!=21 or len(seqe)!=21:continue
  bd={'"'A':0,'C':1,'G':2,'T':3}
  for j,b in enumerate(seqs):
   if b in bd:counts_s[j][bd[b]]+=1
  for j,b in enumerate(seqe):
   if b in bd:counts_e[j][bd[b]]+=1
  n+=1
  if n>=200000:break
print('std n',n)
# PFMs for unknown corrected
corrected_s=[[0]*4 for _ in range(21)]; corrected_e=[[0]*4 for _ in range(21)]
with gzip.open('unknown.shift.frag.bed.gz','rt') as f:
 for line in f:
  p=line.rstrip().split('\\t');ch=p[0];s=int(p[1]);e=int(p[2])
  cs=s+2; ce=e-4 # correction to standard
  try:seqs=fasta.fetch(ch,cs-10,cs+11).upper();seqe=fasta.fetch(ch,ce-10,ce+11).upper()
  except:continue
  if len(seqs)"'!=21 or len(seqe)!=21:continue
  bd={'"'A':0,'C':1,'G':2,'T':3}
  for j,b in enumerate(seqs):
   if b in bd:corrected_s[j][bd[b]]+=1
  for j,b in enumerate(seqe):
   if b in bd:corrected_e[j][bd[b]]+=1
for name,mat in [('std_start',counts_s),('corr_start',corrected_s),('std_end',counts_e),('corr_end',corrected_e)]:
 cons=''.join('ACGT'[max(range(4),key=lambda i:mat[j][i])] for j in range(21))
 print(name,cons)
PY"
