fasta=pysam.FastaFile('hg38.fa')
NSTD=200000
counts_start=[[0]*4 for _ in range(21)]; counts_end=[[0]*4 for _ in range(21)]
n=0; lines=0
with gzip.open('std_part.gz','rt') as f:
 for line in f:
  if line.startswith('#'):continue
  lines+=1
  if (lines-1)%40"'!=0:continue

   seqe=fasta.fetch(ch,e-10,e+11).upper()

 return [(d,sums[i]/cnt[i] if cnt[i] else -999) for i,d in enumerate(range(-20,21))]
