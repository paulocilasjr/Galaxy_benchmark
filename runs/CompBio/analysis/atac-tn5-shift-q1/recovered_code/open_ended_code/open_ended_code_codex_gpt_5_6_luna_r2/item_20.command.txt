/bin/bash -lc "python3 - <<'PY'
import gzip,collections
T=collections.defaultdict(set); n=0
with gzip.open('ncbiRefSeq.txt.gz','rt') as f:
 for l in f:
  a=l.rstrip('\\n').split('\\t')
  if len(a)>=6 and a[2] in {'chr'+str(i) for i in range(1,23)}|{'chrX','chrY'}:
   n+=1; T[a[2]].add(int(a[4]) if a[3]=='+' else int(a[5]))
print('transcripts',n,'unique tss',sum(map(len,T.values())))
print({c:len(x) for c,x in T.items()})
PY"
