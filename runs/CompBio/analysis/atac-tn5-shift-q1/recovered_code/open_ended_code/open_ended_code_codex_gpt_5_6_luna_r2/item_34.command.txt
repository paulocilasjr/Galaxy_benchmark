/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np,pysam
s=[];e=[]
with gzip.open('unknown.shift.frag.bed.gz','rt') as f:
 for l in f:
  a=l.split('\\t');
  if a[0]=='chr1':s.append(int(a[1]));e.append(int(a[2]))
f=pysam.FastaFile('chr1.fa.bgz'); comp=np.array([3,2,1,0]) # A,C,G,T complement indexes

def prof(pos,side,w=15):
 P=np.zeros((2*w+1,4),float)
 for x in pos:
  z=f.fetch('chr1',max(0,x-w-2),x+w+3).upper()
  if len(z)"'!=2*w+5:continue
  # if x internal index w+2
  if side=='"'R': z=z.translate(str.maketrans('ACGTN','TGCAN'))[::-1]
  for j,b in enumerate(z[w+2-w:w+2+w+1]):
   if b in 'ACGT':P[j, 'ACGT'.index(b)]+=1
 P/=P.sum(axis=1,keepdims=True)
 return P
for side,pos in [('L',s),('R',e)]:
 P=prof(pos,side); print(side)
 rows=[]
 # center c can integer, compare pos c-d and c+d, complement
 for c in np.arange(-6,6.1,.5):
  err=0;n=0
  for d in np.arange(.5,9.1,.5):
   a=c-d; b=c+d
   if a%1 or b%1:continue
   ia=int(a+15); ib=int(b+15)
   if 0<=ia<31 and 0<=ib<31:
    err+=np.sum((P[ia]-P[ib][comp])**2); n+=1
  rows.append((err/n,c))
 print(sorted(rows)[:12])
 for i in range(31): print(f'{i-15:+d}', ' '.join(f'{x:.3f}' for x in P[i]))
PY"
