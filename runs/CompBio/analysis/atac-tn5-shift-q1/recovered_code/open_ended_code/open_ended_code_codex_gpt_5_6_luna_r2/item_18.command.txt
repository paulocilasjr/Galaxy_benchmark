/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
s=[];e=[]
with gzip.open('unknown.shift.frag.bed.gz','rt') as f:
 for l in f:
  a=l.split('\\t')
  if a[0]=='chr1':s.append(int(a[1]));e.append(int(a[2]))
s=np.array(s);e=np.array(e)
T=set()
with gzip.open('ncbiRefSeq.txt.gz','rt') as f:
 for l in f:
  a=l.rstrip().split('\\t')
  if len(a)>=6 and a[2]=='chr1':T.add(int(a[4]) if a[3]=='+' else int(a[5]))
t=np.array(list(T))
for name,x in [('s',s),('e',e),('both',np.r_[s,e])]:
 h=np.zeros(101,dtype=int)
 for p in t:
  q=x-p; q=q[(q>=-50)&(q<=50)]
  h+=np.bincount(q+50,minlength=101)
 print(name,'sum',h.sum(),'max',h.max())
 print(' '.join(f'{i-50}:{v}' for i,v in enumerate(h)))
PY"
