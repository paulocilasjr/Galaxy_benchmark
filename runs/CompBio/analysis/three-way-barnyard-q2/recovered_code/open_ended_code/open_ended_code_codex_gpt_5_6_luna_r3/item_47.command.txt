/bin/bash -lc "python3 - <<'PY'
import csv
from collections import Counter,defaultdict
with open('cell_species_compare.tsv') as f:
    cr=csv.DictReader(f,delimiter='\\t')
    lab={}
    for r in cr:
        v={s:int(r[s+'_margin10']) for s in ('human','mouse','pig')}
        lab[r['barcode']]=max(v,key=v.get)
with open('cell_markers.tsv') as f:
    rd=csv.DictReader(f,delimiter='\\t')
    rows=list(rd)
for r in rows:
    h=sum(int(r.get('human_'+g,0)) for g in ('HBB','HBA1','HBA2','HBD','AHSP','ALAS2','CA1','SLC25A37','HBM'))
    m=sum(int(r.get('mouse_'+g,0)) for g in ('SCGB1A1','SFTPC','SFTPA1','SFTPB','SFTPA2','KRT8','KRT18','EPCAM'))
    p=sum(int(r.get('pig_'+g,0)) for g in ('PRM1','PRM2','TNP1','TNP2','DDX4','DAZL','SYCP3','CRISP2','AKAP4','SPAG6','DDX25'))
    r['hs']=h;r['ms']=m;r['ps']=p
print('current label counts',Counter(lab.values()))
print('marker dominant counts',Counter(max((r['hs'],'human'),(r['ms'],'mouse'),(r['ps'],'pig'))[1] for r in rows))
print('label vs marker')
co=Counter((lab[r['barcode']],max((r['hs'],'human'),(r['ms'],'mouse'),(r['ps'],'pig'))[1]) for r in rows)
for x in sorted(co):print(x,co[x])
for s in ('human','mouse','pig'):
    rs=[r for r in rows if lab[r['barcode']]==s]
    print('\\n',s,len(rs),'marker sums',sum(r['hs'] for r in rs),sum(r['ms'] for r in rs),sum(r['ps'] for r in rs))
    for r in sorted(rs,key=lambda x:(x['hs']+x['ms']+x['ps']),reverse=True)[:5]: print(r['barcode'],r['hs'],r['ms'],r['ps'],r['total_reads'])
print('marker low/dominance sorted')
for r in sorted(rows,key=lambda x:max(x['hs'],x['ms'],x['ps']))[:30]: print(lab[r['barcode']],r['barcode'],r['hs'],r['ms'],r['ps'],r['total_reads'])
PY"
