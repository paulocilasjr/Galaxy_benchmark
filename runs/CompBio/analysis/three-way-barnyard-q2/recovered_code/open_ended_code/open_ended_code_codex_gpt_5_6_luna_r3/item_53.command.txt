/bin/bash -lc "python3 - <<'PY'
import gzip,csv
from collections import defaultdict,Counter
bcs=[];umis=[]
with gzip.open('three.way.barnyard.q2.R1.fq.gz','rt') as f:
 while f.readline():
  s=f.readline().strip();f.readline();f.readline();bcs.append(s[:16]);umis.append(s[16:])
sets={c:defaultdict(set) for c in (1,5,10,20)}
with open('read_species_scores.tsv') as f:
 for r in csv.DictReader(f,delimiter='\\t'):
  m=int(r['margin']); sp=r['best']
  if sp=='none': continue
  i=int(r['read_index'])-1
  for c in (1,5,10,20):
   if m>=c: sets[c][(r['barcode'],sp)].add(umis[i])
cells=list(dict.fromkeys(bcs))
for c in (1,5,10,20):
 calls=Counter(); totals=Counter()
 for bc in cells:
  vals={sp:len(sets[c][(bc,sp)]) for sp in ('human','mouse','pig')}; totals.update(vals); calls[max(vals,key=vals.get)]+=1
 print('cutoff',c,'cells',dict(calls),'UMI totals',dict(totals))
PY"
