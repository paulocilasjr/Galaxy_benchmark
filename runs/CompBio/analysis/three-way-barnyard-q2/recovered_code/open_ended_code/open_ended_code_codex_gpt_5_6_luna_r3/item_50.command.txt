/bin/bash -lc "python3 - <<'PY'
import csv
cmp={}
with open('cell_species_compare.tsv') as f:
 for r in csv.DictReader(f,delimiter='\\t'):
  v={s:int(r[s+'_margin10']) for s in ('human','mouse','pig')}; lab=max(v,key=v.get); cmp[r['barcode']]=(lab,v)
with open('cell_markers.tsv') as f: mr={r['barcode']:r for r in csv.DictReader(f,delimiter='\\t')}
H=['HBB','HBA1','HBA2','HBD','AHSP','ALAS2','CA1','SLC25A37','HBM']; M=['SCGB1A1','SFTPC','SFTPA1','SFTPB','SFTPA2','KRT8','KRT18','EPCAM']; P=['PRM1','PRM2','TNP1','TNP2','DDX4','DAZL','SYCP3','CRISP2','AKAP4','SPAG6','DDX25']
for bc,(lab,v) in cmp.items():
 r=mr[bc]; h=sum(int(r['human_'+x]) for x in H);m=sum(int(r['mouse_'+x]) for x in M);p=sum(int(r['pig_'+x]) for x in P)
 if lab in ('human','mouse') and (h or m) and (h<5 or m<5 or abs(h-m)<5):
  print(lab,bc,'speciesreads',v,'markers',h,m,p,'total',r['total_reads'])
PY"
