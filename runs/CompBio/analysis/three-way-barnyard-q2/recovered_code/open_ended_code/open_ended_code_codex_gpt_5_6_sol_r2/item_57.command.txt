/bin/bash -lc "python - <<'PY'
import pandas as pd
from sklearn.mixture import GaussianMixture
from sklearn.cluster import KMeans
import numpy as np
D={}
for s in ['human','mouse','pig']:
 x=pd.read_csv(s+'.bc.ratio.tsv',sep='\\t',header=None,names=['bc','reads',s+'_n',s]).set_index('bc')
 D[s]=x[s]
d=pd.DataFrame(D).fillna(0)
for alg in ['km','gm']:
 X=d[['human','mouse','pig']].to_numpy()
 model=KMeans(n_clusters=3,random_state=1,n_init=20) if alg=='km' else GaussianMixture(n_components=3,random_state=1,n_init=10)
 z=model.fit_predict(X)
 print(alg,np.bincount(z))
 print(d.groupby(z).mean().round(3))
PY"
