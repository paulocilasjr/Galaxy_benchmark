print('reads', len(cbs)-1)
# target mapping
target={}
with open('refs/cds/t2gene.tsv') as f:
    for line in f:
        tid,sp,gene,sym,desc=line.rstrip('\\n').split('\\t')
        target[tid]=(sp,sym,desc)

sp_gene_reads=collections.defaultdict(collections.Counter)
sp_gene_umis=collections.defaultdict(set)
read_sp_total=collections.Counter()

        read_sp_total[sp]+=1
