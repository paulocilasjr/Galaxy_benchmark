/bin/bash -c "python3 - <<'PY'
import scipy.io, collections
M=scipy.io.mmread('kallisto_countoutput.mtx').tocsr(); genes=[l.strip() for l in open('kallisto_countoutput.genes.txt')]
for th in [0.8,0.9]:
 ctr=collections.Counter(); amb=0
 for i in range(M.shape[0]):
  row=M.getrow(i); c=collections.Counter()
  for j,v in zip(row.indices,row.data): c[genes[j].split('|')[0]] += int(v)
  tot=sum(c.values()); sp,n=c.most_common(1)[0]
  if tot and n/tot>=th: ctr[sp]+=1
  else: amb+=1
 print(th,ctr,amb,{k:round(v/sum(ctr.values())*1000)/10 for k,v in ctr.items()}, {k:round(v/sum(ctr.values())*100) for k,v in ctr.items()})
PY"
