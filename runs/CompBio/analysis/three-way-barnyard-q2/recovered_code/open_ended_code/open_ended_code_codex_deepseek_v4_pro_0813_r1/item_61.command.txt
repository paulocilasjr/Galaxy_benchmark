    seqs=[]
    with gzip.open(path,'rt') as f:
        cur=[]; keep=False
        for line in f:
            if line.startswith('>'):
                if keep: seqs.append(''.join(cur))
                keep = ('gene_symbol:'+sym) in line
                cur=[]
            elif keep:
                cur.append(line.strip())
        if keep: seqs.append(''.join(cur))
    return seqs

  best=(999,None)
