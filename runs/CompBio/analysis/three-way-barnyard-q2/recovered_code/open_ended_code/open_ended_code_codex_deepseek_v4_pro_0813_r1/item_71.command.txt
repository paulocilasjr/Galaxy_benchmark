/bin/bash -c "python3 - <<'PY'
import gzip,collections
raw=collections.defaultdict(collections.Counter)
with gzip.open('read_unique_assign.tsv.gz','rt') as f:
 for line in f:
  p=line.rstrip().split('\\t'); bc,u,sp=p[0],p[1],p[2]; raw[bc][sp]+=1
for thresh in [0.5,0.6,0.7,0.8,0.9,0.95]:
 counts=collections.Counter(); amb=0
 for bc,c in raw.items():
  tot=sum(c.values()); top_sp,top=c.most_common(1)[0]
  if top/tot>=thresh: counts[top_sp]+=1
  else: amb+=1
 print(thresh,counts,'amb',amb,'pct',{k:round(v/sum(counts.values())*100) for k,v in counts.items()})
PY"
