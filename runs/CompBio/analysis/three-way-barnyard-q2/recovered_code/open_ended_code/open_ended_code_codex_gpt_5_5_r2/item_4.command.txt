/bin/bash -lc "python3 - <<'PY'
import gzip, collections, itertools
r1='three.way.barnyard.q2.R1.fq.gz'
counts=collections.Counter()
n=0
lens=collections.Counter()
with gzip.open(r1,'rt') as f:
    while True:
        h=f.readline()
        if not h: break
        s=f.readline().strip(); f.readline(); f.readline()
        n+=1; lens[len(s)]+=1; counts[s[:16]]+=1
print('reads',n,'lens',lens,'barcodes',len(counts))
print('top',counts.most_common(20))
print('minmax',min(counts.values()),max(counts.values()))
print('hist top counts',collections.Counter(counts.values()).most_common(20))
PY"
