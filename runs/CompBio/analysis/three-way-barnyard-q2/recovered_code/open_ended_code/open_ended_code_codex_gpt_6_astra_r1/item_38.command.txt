/bin/bash -lc "python3 - <<'PY'
import csv,collections,gzip
assign={}
with open('results/barcode_species.tsv') as f:
 for d in csv.DictReader(f,delimiter='\\t'):
  a=[int(d[s]) for s in ['human','mouse','pig']]
  if max(a)/sum(a)>=.8:assign[d['barcode']]=['human','mouse','pig'][a.index(max(a))]
markers={'bone_marrow':'HBB HBA1 HBA2 AHSP ALAS2 SLC25A37 GYPA KLF1 GATA1 MPO ELANE PRTN3 CTSG','spleen':'MS4A1 CD79A CD79B CD3D CD3E NKG7 CCL5','lung':'SCGB1A1 SFTPA1 SFTPB SFTPC SFTPD SCGB3A2 AGER','testis':'PRM1 PRM2 TNP1 TNP2 DDX4 DAZL ACRV1 SYCP3 KIT STRA8','liver':'ALB APOA1 APOA2 TTR CPS1','kidney':'UMOD AQP2 SLC12A1 SLC34A1 LRP2','heart':'TNNT2 TNNI3 MYH6 MYH7 MYL2','cortex':'SLC17A7 CAMK2A SNAP25 GAD1 GAD2','retina':'RHO RCVRN SAG GNAT1 PDE6A','skin':'KRT1 KRT10 KRT14 FLG LOR'}
c=collections.Counter()
with gzip.open('results/cell_gene_umis.tsv.gz','rt') as f:
 for d in csv.DictReader(f,delimiter='\\t'):
  if assign.get(d['barcode'])==d['species']:c[d['species'],d['gene'].upper()]+=int(d['umis'])
with open('results/marker_report.tsv','w') as out:
 for sp in ['human','mouse','pig']:
  print(sp)
  for tissue,gs in markers.items():
   hits=[(g,c[sp,g]) for g in gs.split() if c[sp,g]>0]
   print(tissue,hits)
   for g,count in hits:out.write(f'{sp}\\t{tissue}\\t{g}\\t{count}\\n')
PY"
