/bin/bash -lc "python3 - <<'PY'
import gzip
specs=[('human','refs/human.cdna.fa.gz','work/human.cdna.fa'),('mouse','refs/mouse.cdna.fa.gz','work/mouse.cdna.fa'),('pig','refs/pig.cdna.fa.gz','work/pig.cdna.fa')]
for sp,gz,out in specs:
    with gzip.open(gz,'rt') as fi, open(out,'w') as fo:
        for line in fi:
            if line.startswith('>'):
                parts=line.strip().split()
                tid=parts[0][1:]
                gene=''
                for p in parts:
                    if p.startswith('gene_symbol:'):
                        gene=p.split(':',1)[1]
                        break
                if not gene:
                    for p in parts:
                        if p.startswith('gene:'):
                            gene=p.split(':',1)[1]
                            break
                fo.write(f'>{sp}|{gene}|{tid}\\n')
            else:
                fo.write(line)
    print(sp,'done')
PY
ls -lh work/*.cdna.fa"
