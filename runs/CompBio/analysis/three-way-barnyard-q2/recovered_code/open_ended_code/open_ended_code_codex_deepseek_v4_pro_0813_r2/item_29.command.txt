/bin/bash -lc "mkdir -p work && gzip -cd refs/pig.cdna.fa.gz | python3 - <<'PY'
import sys
# modify headers to pig|gene|transcript; write plain fasta
for line in sys.stdin:
    if line.startswith('>'):
        # parse gene_symbol if any, else gene id
        parts=line.strip().split()
        tid=parts[0][1:]
        gene=''
        for p in parts:
            if p.startswith('gene_symbol:'):
                gene=p.split(':',1)[1]
                break
        if not gene:
            for p in parts:
                if p.startswith('gene:'):
                    gene=p.split(':',1)[1]
                    break
        print(f'>{gene}|{tid}')
    else:
        print(line.strip())
PY"
