/bin/bash -lc "python3 - <<'PY'
import collections
spcnt=collections.defaultdict(collections.Counter)
with open('species_barcode_gene_umi.tsv') as f:
 for line in f:
  sp,bc,g,c=line.rstrip('\\n').split('\\t')
  spcnt[sp][g]+=int(c)
for sp in ['HUMAN','MOUSE','PIG']:
 print('\\n###',sp,'total genes',len(spcnt[sp]),'total umi',sum(spcnt[sp].values()))
 for g,c in spcnt[sp].most_common(50): print(g,c)
PY"
