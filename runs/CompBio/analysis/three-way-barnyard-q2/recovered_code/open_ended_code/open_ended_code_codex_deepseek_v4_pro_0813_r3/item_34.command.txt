/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
df=pd.read_csv('barcode_species.tsv',sep='\\t',header=None,names=['bc','total','best','best_count','frac','h','m','p'])
print(df.groupby('best').agg(n=('bc','size'), total_reads=('total','sum'), median_frac=('frac','median'), mean_frac=('frac','mean')))
print(df['frac'].describe())
bins=[0,.6,.7,.8,.9,.95,.99,1.01]
print('frac bins by best')
print(pd.crosstab(pd.cut(df.frac,bins),df.best))
# Species purity using counts ratios; inspect doublet-like frac < .8
print('frac<.8 count', (df.frac<.8).sum(), 'frac<.9', (df.frac<.9).sum())
print(df.sort_values('frac').head(100).to_string())
PY"
