/bin/bash -lc "python3 - <<'PY'
import collections
conf=set()
with open('barcode_species.tsv') as f:
 for l in f:
  p=l.rstrip('\\n').split('\\t')
  if float(p[4])>=0.8: conf.add((p[0],p[2]))
cnt=collections.Counter()
with open('species_barcode_gene_umi.tsv') as f:
 for line in f:
  sp,bc,g,c=line.rstrip('\\n').split('\\t')
  if (bc,sp) in conf and sp=='HUMAN': cnt[g]+=int(c)
for g,c in cnt.most_common(300): print(g,c)
PY"
