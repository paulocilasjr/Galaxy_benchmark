import collections
umi=collections.defaultdict(lambda: collections.Counter())
for line in open('species_barcode_gene_umi.tsv'):
 sp,bc,g,c=line.rstrip('\\n').split('\\t')
 umi[bc][sp]+=int(c)
print('barcodes with umi',len(umi))
with open('barcode_species_umi.tsv','w') as f:
 for bc,c in umi.items():
  total=sum(c.values())
  best,bestc=c.most_common(1)[0]
  frac=bestc/total if total else 0
  f.write('\\t'.join([bc,str(total),best,str(bestc),f'{frac:.4f}',str(c.get('HUMAN',0)),str(c.get('MOUSE',0)),str(c.get('PIG',0))])+'\\n')
# summary thresholds
for th in [0.5,0.6,0.7,0.75,0.8,0.9,0.95]:
 cc=collections.Counter(); ex=0
 for bc,c in umi.items():
  total=sum(c.values()); best,bestc=c.most_common(1)[0]; frac=bestc/total if total else 0
  if frac>=th: cc[best]+=1
  else: ex+=1
 print(th, dict(cc), 'tot',sum(cc.values()),'ex',ex)
PY
