/bin/bash -lc "python3 - <<'PY'
import gzip
barcodes={'MIX':'AACAAGACAAGTTCCA','PURE_H':'CTGGTCTAGGAAAGTG','PURE_M':'GAGCCTGGTGACACAG','PURE_P':'ACTTCGCCACTGTGAT'}
outs={k:open(f'{k}.R2.fq','w') for k in barcodes}
r1=gzip.open('three.way.barnyard.q2.R1.fq.gz','rt')
r2=gzip.open('three.way.barnyard.q2.R2.fq.gz','rt')
counts={k:0 for k in barcodes}
while True:
 h1=r1.readline()
 if not h1: break
 s1=r1.readline().rstrip('\\n'); p1=r1.readline(); q1=r1.readline()
 h2=r2.readline(); s2=r2.readline().rstrip('\\n'); p2=r2.readline(); q2=r2.readline()
 bc=s1[:16]
 for k,b in barcodes.items():
  if bc==b and counts[k]<500:
   # write read with original R2 header but maybe read name /1? For mapping no need barcode
   outs[k].write(f'@{h1[1:].strip().split()[0].replace(\"/1\",\"\")}/{k}_{counts[k]}\\n{s2}\\n+\\n{q2}')
   counts[k]+=1
for f in outs.values(): f.close()
print(counts)
PY
for f in MIX PURE_H PURE_M PURE_P; do minimap2 -ax sr -t 4 -N 10 combined.mmi "'$f.R2.fq 2>/dev/null | grep -v '"'"'^@'"' | awk -v f="'$f '"'{print f, "'$0}'"' | head -3; done"
