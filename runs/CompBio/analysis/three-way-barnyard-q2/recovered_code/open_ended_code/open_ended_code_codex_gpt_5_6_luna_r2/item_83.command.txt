/bin/bash -lc "python3 - <<'PY'
import collections,math
# robust pig cells labels from read_scores
lab=collections.defaultdict(collections.Counter)
with gzip.open('read_scores.tsv.gz','rt') as f:
 next(f)
 for line in f:
  x=line.rstrip().split('\\t'); bc=x[1]; v=[int(x[2]),int(x[3]),int(x[4])]; a=sorted(v,reverse=True); sp=['hs','mm','ss'][v.index(a[0])]
  if a[0]>a[1]: lab[bc][sp]+=1
labels={bc:max(c,key=c.get) for bc,c in lab.items() if max(c.values())/sum(c.values())>=.8}
G=collections.defaultdict(collections.Counter)
with open('gene_counts.tsv') as f:
 next(f)
 for line in f:
  bc,sp,g,n=line.rstrip().split('\\t');
  if labels.get(bc)=='ss' and sp=='ss': G[bc][g]+=int(n)
# top genes in groups; score mean log1p count
for group,bs in [('CD80high',[b for b,c in G.items() if c['CD80']>=5]),('CD80zero',[b for b,c in G.items() if c['CD80']==0]),('testishigh',[b for b,c in G.items() if c['PRM1']+c['TNP1']+c['TNP2']+c['PRM2']>=5])]:
 print('\\nGROUP',group,'n',len(bs))
 score=[]
 for g in set().union(*(G[b] for b in bs)):
  if g=='NA' or g.startswith('RPL') or g.startswith('RPS') or g.startswith('COX') or g.startswith('ND') or g in {'ATP6','CYTB','TMEM186','TMSB10','PHKB','ZCCHC24'}: continue
  val=sum(math.log1p(G[b][g]) for b in bs)/len(bs)
  det=sum(G[b][g]>0 for b in bs)
  score.append((val,det,g,sum(G[b][g] for b in bs)))
 for v,d,g,n in sorted(score,reverse=True)[:80]: print(f'{g}\\tavglog={v:.3f}\\tdet={d}/{len(bs)}\\tcount={n}')
PY"
