/bin/bash -lc "python3 - <<'PY'
import collections,statistics
markers=['CD80','PRM1','PRM2','TNP1','TNP2','SPO11','DDX4','HSPA4L','INSL3','TDRD1','HLA-DRA','CD74','CD3D','CD3E','PTPRC','TYROBP','LST1','MS4A1','CD79A','APOA1','CLU']
g=collections.defaultdict(collections.Counter)
with open('gene_counts.tsv') as f:
 next(f)
 for line in f:
  bc,sp,gn,n=line.rstrip().split('\\t');
  if sp=='ss': g[bc][gn]+=int(n)
print('cells',len(g))
for m in markers:
 vals=[c[m] for c in g.values()]
 print(f'{m:8s} nonzero {sum(v>0 for v in vals):3d} sum {sum(vals):5d} median {statistics.median(vals):4.1f} q90 {sorted(vals)[int(.9*len(vals))-1]:3d} max {max(vals):3d}')
# print cells with highest total and marker vector
for bc,c in sorted(g.items(),key=lambda kv:kv[1]['CD80'],reverse=True)[:30]:
 print(bc,'total',sum(c.values()),' '.join(f'{m}={c[m]}' for m in markers if c[m]))
PY"
