/bin/bash -lc "mkdir -p data && curl -L --fail --silent 'https://rest.ensembl.org/lookup/symbol/homo_sapiens/SHH?expand=1' -H 'Content-Type: application/json' -o data/shh_ensembl.json && jq '{id,assembly_name,start,end,strand,canonical_transcript,Transcript:(.Transcript[]|select(.is_canonical==1 or .id==\"ENST00000297261\")|{id,start,end,strand,biotype,Parent,Translation,Exon})}' data/shh_ensembl.json | head -c 30000"
