/bin/bash -lc "python - <<'PY'
import urllib.request,json,concurrent.futures
tracks=['mane','omimAvSnp','alphaMissense_A','alphaMissense_C','alphaMissense_G','alphaMissense_T']
def fetch(track):
 start,end=(155790000,155820000) if track=='mane' else (155811822,155812122)
 url=f'https://api.genome.ucsc.edu/getData/track?genome=hg38;track={track};chrom=chr7;start={start};end={end}'
 try:
  data=urllib.request.urlopen(url).read();open(track+'.json','wb').write(data)
  obj=json.loads(data);return track,obj if track in ['mane','omimAvSnp'] else {'itemsReturned':obj.get('itemsReturned'), 'sample':obj.get(track,[])[:3]}
 except Exception as e:return track,str(e)
for r in concurrent.futures.ThreadPoolExecutor().map(fetch,tracks):print(r)
PY"
