/bin/bash -lc "curl -sL 'https://api.genome.ucsc.edu/list/tracks?genome=hg38' -o data/ucsc_tracks_hg38.json && python3 - <<'PY'
import json
j=json.load(open('data/ucsc_tracks_hg38.json'))
tracks=j.get('hg38',{})
for k,v in tracks.items():
    if 'alpha' in k.lower() or 'missense' in k.lower() or 'AlphaMissense' in str(v):
        print(k, v if isinstance(v,str) else {kk:v.get(kk) for kk in ('shortLabel','longLabel','type','parent','subtracks') if kk in v})
PY"
