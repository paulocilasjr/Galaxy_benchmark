/bin/bash -lc "python3 - <<'PY'
import json, re
j=json.load(open('data/clinvar_shh_omim_esummary.json'))['result']
for uid in j['uids']:
    title=j[uid].get('title','')
    m=re.search(r'c\\.(\\d+)[ACGT]>[ACGT] \\(p\\.([A-Za-z]+)(\\d+)([A-Za-z]+|Ter)\\)', title)
    if m and int(m.group(1))<=300:
        loc=[l for l in j[uid]['variation_set'][0]['variation_loc'] if l['assembly_name']=='GRCh38'][0]
        print(uid,title,'chr'+loc['chr']+':'+loc['start'])
PY"
