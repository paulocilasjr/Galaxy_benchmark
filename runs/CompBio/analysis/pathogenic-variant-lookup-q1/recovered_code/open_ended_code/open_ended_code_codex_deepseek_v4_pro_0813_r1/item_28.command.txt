/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, re, time
base='https://api.genome.ucsc.edu/getData/track?genome=hg38;track=omimAvSnp;chrom=chr7;start=155799900;end=155812500'
data=json.load(urllib.request.urlopen(base))['omimAvSnp']
# exon1 coding 1-based positions 155811823..155812122
lo, hi=155811823,155812122
for item in data:
    name=item['name']; s=item['chromStart']; e=item['chromEnd']
    if e < lo or s+1 > hi: continue
    # query clinvar around item center padded
    qs=max(155811800,s-5); qe=min(155812150,e+5)
    url=f'https://api.genome.ucsc.edu/getData/track?genome=hg38;track=clinvarMain;chrom=chr7;start={qs};end={qe}'
    arr=json.load(urllib.request.urlopen(url))['clinvarMain']
    matches=[x for x in arr if name in x.get('otherIds','')]
    print('OMIM',name,'interval',s,e)
    for x in matches:
        print('  ',x['origName'],'molConseq',x.get('molConseq'),'vcfDesc',x.get('vcfDesc'))
    time.sleep(0.1)
PY"
