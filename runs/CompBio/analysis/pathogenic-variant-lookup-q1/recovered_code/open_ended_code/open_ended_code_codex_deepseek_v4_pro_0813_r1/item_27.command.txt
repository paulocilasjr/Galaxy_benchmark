/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
def get(url):
    req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
    return json.load(urllib.request.urlopen(req))
cDNA=get('https://rest.ensembl.org/sequence/id/ENST00000297261?type=cdna;multiple_sequences=1')[0]['seq']
cds=get('https://rest.ensembl.org/sequence/id/ENST00000297261?type=cds;multiple_sequences=1')[0]['seq']
idx=cDNA.find(cds)
print('cds start in cDNA (0-based)', idx, 'end', idx+len(cds))
print('5UTR len',idx,'3UTR len',len(cDNA)-(idx+len(cds)))
print('cDNA around cds start:',cDNA[idx-20:idx+30])
print('cDNA around cds end:',cDNA[idx+len(cds)-30:idx+len(cds)+20])
# get genomic sequence properly via sequence/region endpoint format
for name,start,end in [('A',155811823,155812463),('B',155806296,155806557),('C',155799980,155803726)]:
    url=f'https://rest.ensembl.org/sequence/region/human/{start}..{end}:1?coord_system_version=GRCh38'
    req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
    try:
        out=urllib.request.urlopen(req).read().decode()
        print(name, out[:200])
    except Exception as e:
        print(name, 'ERR', e)
PY"
