/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
# fetch transcript with exons and translation
url='https://rest.ensembl.org/lookup/id/ENST00000297261?expand=1'
req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
d=json.load(urllib.request.urlopen(req))
print(d.keys())
for e in d['Exon']:
    print(e)
print('translation', d['Translation'])
# fetch cds sequence
url2='https://rest.ensembl.org/sequence/id/ENST00000297261?type=cds;multiple_sequences=1'
d2=json.load(urllib.request.urlopen(urllib.request.Request(url2, headers={'Content-Type':'application/json'})))
cds=d2[0]['seq']; print('cds len',len(cds), 'first', cds[:90], 'last', cds[-180:])
PY"
