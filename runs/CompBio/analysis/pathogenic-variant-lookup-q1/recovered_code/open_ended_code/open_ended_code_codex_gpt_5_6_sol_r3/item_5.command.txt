/bin/bash -lc "curl -fsSL 'https://api.genome.ucsc.edu/list/tracks?genome=hg38' -o tracks.json
python3 - <<'PY'
import json
x=json.load(open('tracks.json'))
print(x.keys())
# recursive keys containing omim alpha

def walk(o,path=''):
 if isinstance(o,dict):
  for k,v in o.items():
   if 'omim' in k.lower() or 'alpha' in k.lower(): print(path+'/'+k, type(v).__name__)
   walk(v,path+'/'+k)
 elif isinstance(o,list):
  for i,v in enumerate(o):walk(v,path)
walk(x)
PY"
