/bin/bash -lc "python - <<'PY'
import csv,gzip,re,json,collections
with gzip.open('borzoi/data/targets_human.txt.gz','rt') as f:b=list(csv.DictReader(f,delimiter='\\t'))
acc=[]
for x in b:
 acc += re.findall(r'(?:GSM\\d+|ENCSR[A-Z0-9]+)','\\t'.join(x.values()))
bs=set(acc)
cs={x['id']:x for x in json.load(open('cistrome_100000.json'))['samples']}
rows=[]
for idx,s in enumerate(open('sei-framework/model/target.names')):
 m=re.search(r'ID:(\\d+)\\s*"'$'"',s)
 if not m:continue
 x=cs[int(m.group(1))]
 base=re.sub(r'_\\d+"'$'"','',x['external_id'])
 if base in bs: rows.append((idx,s.strip(),x['id'],x['external_id'],base))
print('matched',len(rows),'by source',collections.Counter('GEO' if r[3].startswith('GSM') else 'ENCODE' for r in rows))
print('duplicate same ext exact',[(k,v) for k,v in collections.Counter(r[3] for r in rows).items() if v>1][:10])
print('GEO matching range/examples')
geo=[r for r in rows if r[3].startswith('GSM')]
print(geo[:10]);print(geo[-10:])
print('GEO unique exact',len(set(r[3] for r in geo)))
# ENCODE count by replicate suffix
enc=[r for r in rows if r[3].startswith('ENCSR')]
print('ENC suffix distribution',collections.Counter(re.search(r'(_\\d+)?"'$'"',r[3]).group(0) for r in enc))
print('ENC unique bases',len(set(r[4] for r in enc)))
# Count only exact match of ext without stripping suffix
print('strict exact',sum(r[3] in bs for r in rows))
# Counts as we vary Borzoi categories
for assay in ['CAGE','DNASE','ATAC','CHIP','RNA']:
 vals=set()
 for x in b:
  if x['description'].startswith(assay+':'):
   vals.update(re.findall(r'(?:GSM\\d+|ENCSR[A-Z0-9]+)','\\t'.join(x.values())))
 mm=[r for r in rows if r[4] in vals]
 print(assay,len(vals),len(mm),collections.Counter('GEO' if r[3].startswith('GSM') else 'ENCODE' for r in mm))
PY"
