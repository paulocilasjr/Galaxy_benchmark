/bin/bash -lc "python - <<'PY'
import csv,gzip,re,json,collections
with gzip.open('borzoi/data/targets_human.txt.gz','rt') as f:b=list(csv.DictReader(f,delimiter='\\t'))
cs={x['id']:x for x in json.load(open('cistrome_100000.json'))['samples']}
sei=[]
for idx,s in enumerate(open('sei-framework/model/target.names')):
 m=re.search(r'ID:(\\d+)\\s*"'$'"',s)
 if m:
  x=cs[int(m.group(1))].copy(); x['sei_index']=idx; x['name']=s.strip(); sei.append(x)
patterns=['GSM\\\\d+','ENCSR[A-Z0-9]+','ENCFF[A-Z0-9]+','SRR\\\\d+','DRR\\\\d+','ERR\\\\d+']
for p in patterns:
 vals=set()
 tracks=[]
 for i,x in enumerate(b):
  ms=re.findall(p,'\\t'.join(x.values()))
  if ms: vals.update(ms); tracks.append(i)
 print(p,'unique',len(vals),'tracks',len(tracks),'sample',list(sorted(vals))[:5])
 # compare Sei normalized ext to these vals
 matched=[]
 for x in sei:
  ext=x['external_id']
  base=re.sub(r'_\\d+"'$'"','',ext)
  if ext in vals or base in vals:matched.append(x)
 print(' matched Sei',len(matched),collections.Counter(x['external_id_type'] for x in matched))
 print(' examples',[(x['id'],x['external_id'],x['name']) for x in matched[:8]])
# Union matching any accessions
allvals=set()
for x in b:
 allvals.update(re.findall(r'(?:GSM\\d+|ENCSR[A-Z0-9]+|ENCFF[A-Z0-9]+|SRR\\d+|DRR\\d+|ERR\\d+)','\\t'.join(x.values())))
matched=[x for x in sei if x['external_id'] in allvals or re.sub(r'_\\d+"'$'"','',x['external_id']) in allvals]
print('UNION',len(matched),'unique ext base',len(set(re.sub(r'_\\d+"'$'"','',x['external_id']) for x in matched)))
print('by type',collections.Counter(x['external_id_type'] for x in matched))
print('by ext base duplicates',collections.Counter(collections.Counter(re.sub(r'_\\d+"'$'"','',x['external_id']) for x in matched).values()))
PY"
