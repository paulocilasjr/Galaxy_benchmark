/bin/bash -lc "python overlap_count.py && python - <<'PY'
# Independent set-oriented cross-check.
import gzip,re,json
b=gzip.open('borzoi/data/targets_human.txt.gz','rt').read()
provenance=set(re.findall(r'GSM\\d+|ENCSR[A-Z0-9]+',b))
ids={int(m.group(1)) for line in open('sei-framework/model/target.names') if (m:=re.search(r'ID:(\\d+)"'$'"',line.rstrip()))}
samples=json.load(open('cistrome_100000.json'))['samples']
matched={s['id'] for s in samples if s['id'] in ids and re.sub(r'_\\d+"'$'"','',s['external_id']) in provenance}
assert len(ids)==19905
assert len(matched)==1264
print(len(ids),len(matched),((len(matched)+50)//100)*100)
PY"
