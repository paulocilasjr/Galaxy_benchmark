# Inspect matched track count sensitivity to experiment types / assays and identifier edge cases.
import json,re,csv,collections
samples={a['id']:a for a in json.load(open('cistrome_all.json'))['samples']}
sei=[]
for idx,s in enumerate(open('sei-framework/model/target.names')):
 m=re.search(r'ID:(\\d+)\\s*"'$'"',s)
 if m:
  a=samples[int(m.group(1))]; ext=a['external_id']; base=(re.match(r'(ENCSR[A-Z0-9]+)',ext).group(1) if ext.startswith('ENCSR') else ext)
  sei.append((idx,s.strip(),a,base))
B=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
BE=set(re.findall(r'ENCSR[A-Z0-9]+','\\n'.join(r['file'] for r in B)))
BG=set(re.findall(r'GSM\\d+','\\n'.join(r['file'] for r in B)))
EG={}
for r in csv.reader(open('encode_report.tsv',errors='ignore'),delimiter='\\t'):
 if r and r[0]"'!='"'Accession': EG[r[0]]=set(re.findall(r'GSM\\d+',r[1] if len(r)>1 else ''))
EG['ENCSR432SZT']=set(re.findall(r'GSM\\d+',','.join(json.load(open('encode_ENCSR432SZT.json')).get('dbxrefs',[]))))
PG=BG|set().union(*(EG.get(e,set()) for e in BE))
M=[z for z in sei if z[3] in BE or (z[3].startswith('GSM') and z[3] in PG) or (z[3].startswith('ENCSR') and EG.get(z[3],set())&BG)]
print('final exact',len(M),'round',((len(M)+50)//100)*100)
print('targets',collections.Counter(z[1].split(' | ')[1] for z in M).most_common(20))
print('index range',min(z[0] for z in M),max(z[0] for z in M))
PY"
