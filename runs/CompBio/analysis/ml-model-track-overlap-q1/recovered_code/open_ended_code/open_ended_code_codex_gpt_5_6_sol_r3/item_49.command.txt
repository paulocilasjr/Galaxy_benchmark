/bin/bash -lc "python - <<'PY'
import csv,re,collections,json
B=set(re.findall(r'ENCSR[A-Z0-9]+',open('borzoi_targets_human.tsv').read()))
E={}
for r in csv.reader(open('encode_report.tsv',errors='ignore'),delimiter='\\t'):
 if r and r[0]"'!='"'Accession': E[r[0]]=r[1] if len(r)>1 else ''
E['ENCSR432SZT']=','.join(json.load(open('encode_ENCSR432SZT.json')).get('dbxrefs',[]))
c=collections.Counter()
for e in B:
 s=E.get(e,'')
 c['GSM']+=bool(re.search(r'GSM\\d+',s))
 c['GSE only']+=bool(re.search(r'GSE\\d+',s)) and not bool(re.search(r'GSM\\d+',s))
 c['no GEO']+=not bool(re.search(r'GS[EM]\\d+',s))
print(c)
for cat in ['GSE only','no GEO']:
 print('\\n',cat)
 n=0
 for e in sorted(B):
  s=E.get(e,'')
  hit=(bool(re.search(r'GSE\\d+',s)) and not bool(re.search(r'GSM\\d+',s))) if cat=='GSE only' else not bool(re.search(r'GS[EM]\\d+',s))
  if hit and n<20: print(e,s); n+=1
PY"
