/bin/bash -lc "pwd && python - <<'PY'
import json,re,collections
from openpyxl import load_workbook
wb=load_workbook('sei_supp_tables.xlsx',read_only=True,data_only=True)
ids=[];profiles={}
for row in wb['Supp Table 1'].iter_rows(min_row=3,max_col=1,values_only=True):
 s=row[0]
 if s and 'ID:' in s:
  i=int(re.search(r'ID:(\\d+)"'$'"',s).group(1)); ids.append(i); profiles[i]=s
samples={s['id']:s for s in json.load(open('cistrome_all_samples.json'))['samples']}
print('ids',len(ids),'mapped',sum(i in samples for i in ids),'missing',len([i for i in ids if i not in samples]))
print('types',collections.Counter((samples[i].get('external_id_type'),samples[i].get('species')) for i in ids if i in samples))
print('source external prefixes',collections.Counter(re.match(r'[A-Za-z]+',samples[i].get('external_id','')).group(0) if re.match(r'[A-Za-z]+',samples[i].get('external_id','')) else '' for i in ids if i in samples).most_common(30))
print('missing first',[(i,profiles[i]) for i in ids if i not in samples][:10])
for typ in ['GEO','ENCODE']:
 print('\\n',typ)
 n=0
 for i in ids:
  if i in samples and samples[i].get('external_id_type')==typ:
   print(i,samples[i].get('external_id'),profiles[i]); n+=1
   if n>=10: break
PY"
