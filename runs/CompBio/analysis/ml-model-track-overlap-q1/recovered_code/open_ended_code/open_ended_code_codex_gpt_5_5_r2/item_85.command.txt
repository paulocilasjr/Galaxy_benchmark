/bin/bash -lc "python3 - <<'PY'
# Recompute expanded matching separately and report candidate type used.
import json,re,csv,collections
aliases=json.load(open('borzoi_encode_experiment_aliases.json'))
expanded=set()
text=open('borzoi_targets_human.txt').read()
for pat in [r'\\b(?:GSM|GSE|SRR|ERR|DRR)\\d+\\b',r'\\bENC(?:SR|FF)[0-9A-Z]{6}\\b',r'\\bkai\\d+\\b']:
    expanded.update(re.findall(pat,text))
for vals in aliases.values():
    for v in vals:
        expanded.add(str(v).split(':',1)[-1])
samples={json.loads(l)['id']:json.loads(l) for l in open('cistrome_samples_for_sei.jsonl')}
reason=collections.Counter(); examples=[]
for line in open('sei_target.names'):
    m=re.search(r'ID:(\\d+)',line)
    if not m: continue
    cid=int(m.group(1)); s=samples[cid]
    candidates=[]
    if s.get('external_id'): candidates.append(('sample',s['external_id']))
    for series in s.get('series',[]) or []:
        if series.get('external_id'): candidates.append(('series',series['external_id']))
    for kind,c in candidates:
        if c.split(':',1)[-1] in expanded:
            reason[(kind,c[:3])]+=1
            if len(examples)<20: examples.append((cid,line.strip(),kind,c,s.get('external_id_type')))
            break
print(reason)
for e in examples: print(e)
PY"
