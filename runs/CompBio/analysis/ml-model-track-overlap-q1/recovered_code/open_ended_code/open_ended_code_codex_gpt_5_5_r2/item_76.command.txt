/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, json
accs=['ENCSR000AKB','ENCSR000DZN']
params=[('type','Experiment'),('format','json'),('limit','all'),('field','accession'),('field','dbxrefs'),('field','description')]
for a in accs: params.append(('accession',a))
url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode(params)
print(url)
req=urllib.request.Request(url, headers={'Accept':'application/json'})
data=json.load(urllib.request.urlopen(req, timeout=30))
print(len(data.get('@graph',[])))
for g in data.get('@graph',[]): print(g)
PY"
