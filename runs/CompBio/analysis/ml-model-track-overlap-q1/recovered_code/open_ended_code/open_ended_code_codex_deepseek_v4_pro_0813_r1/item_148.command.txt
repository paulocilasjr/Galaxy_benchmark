/bin/bash -lc "python3 - <<'PY'
import urllib.request,re
# find pdf link
url='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.full'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read().decode('utf-8','ignore')
for m in re.finditer(r'href=\"(["'^"]+'"\\.pdf["'^"]*)"'"',data):
 print(m.group(1))
PY"
