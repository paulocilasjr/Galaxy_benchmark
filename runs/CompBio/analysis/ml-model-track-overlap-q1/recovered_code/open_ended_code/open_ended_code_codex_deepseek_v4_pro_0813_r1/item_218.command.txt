/bin/bash -lc "python3 - <<'PY'
import json,collections
files=[json.loads(l) for l in open('data/borzoi_encff_metadata_full.jsonl')]
exps={g['accession']:g for g in [json.loads(l) for l in open('data/borzoi_experiments_reps_metadata.jsonl')]}
total_matches=0; no_geo=0; examples=[]
for r in files:
 acc=r.get('dataset','').rstrip('/').split('/')[-1]; ex=exps.get(acc); tech=set(r.get('technical_replicates') or [])
 if not ex or not tech: continue
 for rep in ex.get('replicates',[]):
  key=f\"{rep.get('biological_replicate_number')}_{rep.get('technical_replicate_number')}\"
  if key in tech:
   total_matches+=1
   db=(rep.get('library') or {}).get('dbxrefs',[])
   geos=[d for d in db if d.startswith('GEO:GSM')]
   if not geos:
    no_geo+=1
    examples.append((r['accession'],acc,key,rep.get('library',{}).get('accession')))
print('matched file-rep keys',total_matches,'with no GEO dbxref',no_geo)
print('examples',examples[:20])
PY"
