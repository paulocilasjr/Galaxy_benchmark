/bin/bash -lc "python3 - <<'PY'
import urllib.request,re,html
url='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.full'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
data=urllib.request.urlopen(req,timeout=60).read().decode('utf-8','ignore')
print('len',len(data))
text=re.sub('<script.*?</script>','',data,flags=re.S); text=re.sub('<style.*?</style>','',text,flags=re.S); text=re.sub('<["'^>]+>'"',' ',text); text=html.unescape(text); text=re.sub(r'\\s+',' ',text)
for pat in ['ml-model-track-overlap','Borzoi','Cistrome','Sei','share provenance','nearest 100']:
 print('\\n###',pat)
 for m in re.finditer(pat,text,re.I):
  s=max(0,m.start()-500); e=min(len(text),m.end()+1000)
  print(text[s:e][:1500])
  break
PY"
