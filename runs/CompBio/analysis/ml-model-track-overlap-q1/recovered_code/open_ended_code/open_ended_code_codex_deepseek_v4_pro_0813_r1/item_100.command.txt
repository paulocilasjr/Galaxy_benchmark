workspace='.'
sets=json.load(open('data/borzoi_encff_datasets.json'))

if os.path.exists(out_path): os.remove(out_path)
batch_size=100; rows=[]

 for attempt in range(5):
  try:
   data=json.load(urllib.request.urlopen(url,timeout=120)); break
  except Exception as e:
   print('retry',start,attempt,e,flush=True); time.sleep(2*(attempt+1))
 if data is None: print('FAIL',start); sys.exit(1)
 graph=data.get('@graph',[])
 with open(out_path,'a') as f:
  for g in graph: f.write(json.dumps(g)+'\\n'); rows.append(g)
 got={g.get('accession') for g in graph}; missing=set(batch)-got
 if missing: print('WARN missing',start,missing)
 print('batch',start//batch_size+1,'got',len(graph),'total',len(rows),flush=True)
