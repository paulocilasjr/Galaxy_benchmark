/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
queries=['targets_human.txt borzoi','ENCFF991NDB','targets_mouse.txt borzoi']
for q in queries:
 url='https://grep.app/api/search?q='+urllib.parse.quote(q)
 try:
  data=json.load(urllib.request.urlopen(url,timeout=30))
  print(q,data.get('hits',{}).get('total'))
  for h in data.get('hits',{}).get('hits',[])[:5]:
   print(h.get('repo'),h.get('path'))
 except Exception as e: print('ERR',e)
PY"
