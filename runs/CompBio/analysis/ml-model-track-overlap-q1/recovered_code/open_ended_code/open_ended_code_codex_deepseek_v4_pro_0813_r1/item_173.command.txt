import urllib.request,re,html
url='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.supplementary-material'

for m in re.finditer(r'href=\"(["'^"]*(?:media|DC1|download)[''^"]*)"'"',data,re.I):
 print(html.unescape(m.group(1)))
PY"
