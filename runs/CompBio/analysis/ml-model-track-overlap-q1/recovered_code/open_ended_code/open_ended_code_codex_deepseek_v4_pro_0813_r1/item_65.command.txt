import urllib.request, json, re, time, sys, os

workspace='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-281652bfe651d812e2dd4d4b/workspace'
ids=[]
for line in open('[REDACTED LOCAL PATH]'):
    m=re.search(r'ID:(\\d+)"'$'"',line.rstrip())
    if m:
        ids.append(int(m.group(1)))
print('sei ids',len(ids),'unique',len(set(ids)))
seen=set(); out=[]
out_path=os.path.join(workspace,'data','sei_cistrome_samples.jsonl')
if os.path.exists(out_path): os.remove(out_path)
batch_size=200
for start in range(0,len(ids),batch_size):
    batch=ids[start:start+batch_size]
    url='https://db3.cistrome.org/cistrome/samples?ids='+','.join(map(str,batch))+'&limit='+str(len(batch))
    data=None
    for attempt in range(5):
        try:
            req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
            data=json.load(urllib.request.urlopen(req,timeout=120))
            break
        except Exception as e:
            print('retry',start,attempt,e,flush=True)
            time.sleep(2*(attempt+1))
    if data is None:
        print('FAILED batch',start,file=sys.stderr); sys.exit(1)
    samples=data.get('samples',[])
    with open(out_path,'a') as f:
        for s in samples:
            f.write(json.dumps(s)+'\\n')
            out.append(s)
    print('batch',start//batch_size+1,'returned',len(samples),'total',len(out),'time',time.strftime('%H:%M:%S'),flush=True)
    time.sleep(0.1)
print('done',len(out))
PY
python3 fetch_sei_cistrome.py"
