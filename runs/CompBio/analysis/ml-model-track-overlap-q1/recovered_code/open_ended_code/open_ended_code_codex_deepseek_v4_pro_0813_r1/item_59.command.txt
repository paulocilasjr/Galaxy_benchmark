/bin/bash -lc "python3 - <<'PY'
import gzip,urllib.request,json,re
# print first 20 CHIP ENCFF and query metadata
records=[]
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  ident=p[1]; desc=p[8]
  if desc.startswith('CHIP') and ident.startswith('ENCFF'):
   records.append((ident,desc))
   if len(records)>=20: break
 f.close()
 if len(records)>=20: break
for ident,desc in records:
 ident2=re.sub(r'[+-]"'$'"','',ident)
 url=f'https://www.encodeproject.org/search/?type=File&accession={ident2}&format=json&field=accession&field=dataset&field=biological_replicates&field=output_type&field=assay_term_name&limit=all'
 try:
  g=json.load(urllib.request.urlopen(url,timeout=30))['@graph'][0]
  print(ident,desc,'=>',g.get('dataset','').split('/')[-2],g.get('biological_replicates'),g.get('output_type'),g.get('assay_term_name'))
 except Exception as e: print(ident,'ERR',e)
PY"
