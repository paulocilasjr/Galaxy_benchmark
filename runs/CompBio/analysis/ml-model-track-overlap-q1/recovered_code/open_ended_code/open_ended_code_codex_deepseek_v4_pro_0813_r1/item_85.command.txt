/bin/bash -lc "python3 - <<'PY'
import gzip,re,json
# parse Borzoi ENCFF records from target file with file path
records={}
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  ident=p[1]
  norm=re.sub(r'[+-]"'$'"','',ident)
  if norm.startswith('ENCFF'):
   filepath=p[2]
   m=re.search(r'/(ENCSR\\d+)/',filepath)
   records[norm]=m.group(1) if m else None
print('parsed',len(records),'none dataset',sum(1 for v in records.values() if v is None))
# load api rows
api={r['accession']:r for r in (json.loads(l) for l in open('data/borzoi_encff_metadata.jsonl'))}
missing=set(records)-set(api)
print('missing from api',missing)
mismatch=[(a,records[a],api[a]['dataset'].rstrip('/').split('/')[-1]) for a in set(records)&set(api) if records[a] and api[a]['dataset'].rstrip('/').split('/')[-1]"'!=records[a]]
print('"'mismatch count',len(mismatch),'examples',mismatch[:10])
PY"
