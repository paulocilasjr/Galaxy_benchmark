/bin/bash -lc "python3 - <<'PY'
import gzip,collections,re
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 print('\\n',path)
 f=gzip.open(path,'rt'); f.readline()
 pats=collections.Counter(); examples=collections.Counter()
 for line in f:
  parts=line.rstrip('\\n').split('\\t')
  ident=parts[1]
  if re.fullmatch(r'\\d+',ident): pats['numeric']+=1
  elif re.match(r'\\d+[+-]?"'$'"',ident): pats['numeric_strand']+=1
  elif ident.startswith('ENCFF'): pats['ENCFF']+=1
  elif ident.startswith('ENC'): pats['ENCother']+=1
  elif ident.startswith('GSM'): pats['GSM']+=1
  elif ident.startswith('CNhs'): pats['CNhs']+=1
  elif ident.startswith('SRX') or ident.startswith('SRR'): pats['SRA']+=1
  else:
   pats['other']+=1; examples[ident[:40]]+=1
 f.close()
 print(pats)
 print('other examples',examples.most_common(30))
PY"
