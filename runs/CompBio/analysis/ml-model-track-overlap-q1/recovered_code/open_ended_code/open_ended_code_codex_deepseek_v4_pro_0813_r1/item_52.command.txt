import urllib.request,json,re,time
ids=[]
for line in open('[REDACTED LOCAL PATH]'):
 m=re.search(r'ID:(\\d+)"'$'"',line.rstrip())
 if m: ids.append(int(m.group(1)))
batch=ids[:200]
for limit in [100,200,500]:
 url=f'https://db3.cistrome.org/cistrome/samples?ids={\",\".join(map(str,batch))}&limit={limit}'
 t=time.time()
 try:
  data=json.load(urllib.request.urlopen(url,timeout=120))
  print('limit',limit,'time',time.time()-t,'returned',len(data['samples']))
 except Exception as e: print('ERR',limit,e,time.time()-t)
PY"
