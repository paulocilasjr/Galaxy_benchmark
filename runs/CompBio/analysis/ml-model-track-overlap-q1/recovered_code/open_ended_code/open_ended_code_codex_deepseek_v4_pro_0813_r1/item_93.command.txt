/bin/bash -lc "python3 - <<'PY'
import json,re,gzip,collections
# Build direct Borzoi GSM and file->experiment geos
direct=set()
borz_file_exp={}
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  p=line.rstrip('\\n').split('\\t'); ident=re.sub(r'[+-]"'$'"','',p[1]); desc=p[8]
  if ident.startswith('GSM'): direct.add(ident)
  if ident.startswith('ENCFF'):
   # parse dataset from file path using [0-9A-Z]
   m=re.search(r'/ENCSR[0-9A-Z]+/',p[2])
   if m: borz_file_exp[ident]=m.group(0).strip('/')
# map experiment -> geos
exps={g['accession']:g for g in [json.loads(l) for l in open('data/borzoi_experiments_metadata.jsonl')]}
exp_geo={}
for acc,g in exps.items():
 exp_geo[acc]=[d.split(':',1)[1] for d in g.get('dbxrefs',[]) if d.startswith('GEO:GSM')]
# source membership per GSM
gsm_sources={g:set() for g in direct}
for g in direct: gsm_sources[g].add('direct')
file_geo={}
for r in [json.loads(l) for l in open('data/borzoi_encff_metadata.jsonl')]:
 acc=r['dataset'].rstrip('/').split('/')[-1]
 gs=exp_geo.get(acc,[])
 file_geo[r['accession']]=gs
 for g in gs:
  gsm_sources.setdefault(g,set()).add('encff')
# matched Sei geo
sei=[json.loads(l) for l in open('data/sei_cistrome_samples.jsonl')]
matched_geo=[]
for r in sei:
 if r.get('external_id_type')=='GEO' and r.get('external_id') in gsm_sources:
  matched_geo.append((r['id'],r['external_id'],gsm_sources[r['external_id']]))
print('matched_geo',len(matched_geo))
c=collections.Counter(frozenset(s) for _,_,s in matched_geo)
print(c)
# of encff-sourced, experiments with multiple geos
multi=[]
for rid,gsm,sources in matched_geo:
 if 'direct' not in sources:
  # find experiments containing gsm
  accs=[a for a,gs in exp_geo.items() if gsm in gs]
  if any(len(exp_geo[a])>1 for a in accs): multi.append((rid,gsm,accs,[exp_geo[a] for a in accs]))
print('encff-only matched that came from multi-geo experiment',len(multi))
print(multi[:20])
# unique experiments and counts for all matched encff only
exp_hit=collections.Counter()
for rid,gsm,sources in matched_geo:
 if 'direct' not in sources:
  for a,gs in exp_geo.items():
   if gsm in gs: exp_hit[a]+=1
print('experiments involved',len(exp_hit),'total',sum(exp_hit.values()))
print('top',exp_hit.most_common(20))
PY"
