/bin/bash -lc "python3 - <<'PY'
import gzip, urllib.request, json, time, urllib.parse
# collect ENCFF from both Borzoi
ids=[]
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  parts=line.rstrip('\\n').split('\\t')
  if parts[1].startswith('ENCFF'):
   ids.append(parts[1])
ids=sorted(set(ids))
print('unique ENCFF',len(ids))
batch=ids[:100]
params=[('type','File'),('format','json'),('field','accession'),('field','dataset'),('field','biological_replicates'),('limit','all')]
params += [('accession',a) for a in batch]
url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode(params)
t=time.time()
try:
 data=json.load(urllib.request.urlopen(url,timeout=60))
 print('time',time.time()-t,'total',data.get('total'),'graph',len(data.get('@graph',[])))
except Exception as e:
 print('ERR',e,time.time()-t)
print('url len',len(url))
PY"
