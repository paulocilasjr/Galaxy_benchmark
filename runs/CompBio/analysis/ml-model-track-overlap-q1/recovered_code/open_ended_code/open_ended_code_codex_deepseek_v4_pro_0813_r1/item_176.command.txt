/bin/bash -lc "python3 - <<'PY'
import urllib.request,re,html
url='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.supplementary-material'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read().decode('utf-8','ignore')
for term in ['Selected','selected','media-','DC']:
 print('\\nTERM',term)
 for m in re.finditer(term,data,re.I):
  s=max(0,m.start()-500); e=min(len(data),m.end()+1500)
  print(data[s:e][:2000]); break
# all links to downloads
for m in re.finditer(r'href=\"(["'^"]+)"'"',data):
 h=html.unescape(m.group(1))
 if '/DC' in h or 'media' in h or 'download' in h: print(h)
PY"
