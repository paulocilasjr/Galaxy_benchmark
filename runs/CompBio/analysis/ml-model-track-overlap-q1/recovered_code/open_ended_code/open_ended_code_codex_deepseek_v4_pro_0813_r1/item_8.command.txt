/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for repo in ['calico/borzoi','FunctionLab/sei-framework']:
    url=f'https://api.github.com/repos/{repo}/git/trees/main?recursive=1'
    req=urllib.request.Request(url,headers={'User-Agent':'Mozilla'})
    data=json.load(urllib.request.urlopen(req))
    print('\\n###',repo,'truncated',data.get('truncated'),'count',len(data.get('tree',[])))
    paths=[t['path'] for t in data['tree']]
    for p in paths:
        if 'track' in p.lower() or 'target' in p.lower() or 'metadata' in p.lower() or 'cistrome' in p.lower() or p.lower().endswith('.tsv') or p.lower().endswith('.txt') or p.lower().endswith('.csv'):
            print(p)
PY"
