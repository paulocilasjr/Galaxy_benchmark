/bin/bash -lc "python3 - <<'PY'
import json,csv,re
mapping=json.load(open('sei_cistrome_map.json'))
# human only source tokens
src=set(); ids=set()
for r in csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'):
 m=re.search(r'/(?:encode|geo|fantom|recount3)/(["'^/]+)/'"',r['file'])
 src.add(m.group(1) if m else r['identifier'])
 ids.add(r['identifier'])
count=0; en=ge=0
for line in open('sei_repo/model/target.names'):
 line=line.rstrip('\\n'); m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if not m: continue
 s=mapping[m.group(1)]; ext=s['external_id']; typ=s['external_id_type']; norm=re.sub(r'_\\d+"'$'"','',ext) if typ=='ENCODE' else ext
 if norm in src:
  count+=1
  if typ=='ENCODE': en+=1
  else: ge+=1
print(count,en,ge)
PY"
