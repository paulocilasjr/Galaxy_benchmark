/bin/bash -lc "python3 - <<'PY'
import re
lines=open('sei_repo/model/target.names').read().splitlines()
print(len(lines))
ids=[]
for line in lines:
 m=re.search(r'ID:(\\d+)',line)
 if m: ids.append(int(m.group(1)))
 else: print('no id',line)
print('with id',len(ids),'unique',len(set(ids)),'minmax',min(ids),max(ids))
# duplicates distribution
from collections import Counter
c=Counter(ids)
print('dups',sum(1 for v in c.values() if v>1), 'maxdup',max(c.values()))
print(c.most_common(20))
PY"
