/bin/bash -lc "python3 - <<'PY'
import csv,re
from collections import defaultdict
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 d=defaultdict(set)
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  m=re.search(r'/(?:encode|geo)/(["'^/]+)/'"',r['file'])
  if m and r['identifier'].startswith('ENCFF'):
   d[m.group(1)].add(r['identifier'])
 print(fn,'ENCSR unique',len(d),'with >1 ENCFF',sum(len(v)>1 for v in d.values()),'max',max(map(len,d.values())))
 # print multi
 for k,v in list(d.items())[:5]:
  if len(v)>1: print(k,v)
PY"
