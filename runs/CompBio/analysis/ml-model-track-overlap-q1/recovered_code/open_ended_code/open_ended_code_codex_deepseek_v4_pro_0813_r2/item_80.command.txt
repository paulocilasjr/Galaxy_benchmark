/bin/bash -lc "python3 - <<'PY'
import json,urllib.request,random
mapping=json.load(open('sei_cistrome_map.json'))
ids=random.sample(sorted(mapping,key=int),5)
for sid in ids:
 s=mapping[sid]
 url=f'https://dc2.cistrome.org/api/inspector?id={sid}'
 try:
  j=json.loads(urllib.request.urlopen(url,timeout=20).read().decode())
  old=j['treats'][0]['unique_id']
  print(sid,s['external_id'],old,'OK' if old==s['external_id'] else 'DIFF')
 except Exception as e: print('ERR',sid,e)
PY"
