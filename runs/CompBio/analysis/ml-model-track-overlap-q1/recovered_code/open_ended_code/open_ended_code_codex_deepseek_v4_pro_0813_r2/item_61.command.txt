/bin/bash -lc "gzip -cd borzoi_repo/data/targets_mouse.txt.gz > borzoi_targets_mouse.tsv && python3 - <<'PY'
import csv,re,json
rows=list(csv.DictReader(open('borzoi_targets_mouse.tsv'),delimiter='\\t'))
print('mouse rows',len(rows))
from collections import Counter
print(Counter(r['description'].split(':')[0] for r in rows))
print(Counter(re.match(r'"'^[A-Za-z]+'"',r['identifier']).group(0) for r in rows if r['identifier']).most_common())
# provenance tokens
tokens=set()
for r in rows:
 p=r['file']; m=re.search(r'/(?:encode|geo|fantom|recount3)/(["'^/]+)/'"',p)
 if m: tokens.add(m.group(1))
 else: tokens.add(r['identifier'])
print('tokens',len(tokens),'ENCSR',sum(x.startswith('ENCSR') for x in tokens),'GSM',sum(x.startswith('GSM') for x in tokens))
mapping=json.load(open('sei_cistrome_map.json'))
mouse_ids={int(k) for k,s in mapping.items() if s.get('species')=='mus_musculus'}
for sid in sorted(mouse_ids):
 s=mapping[str(sid)]; ext=s['external_id']; typ=s['external_id_type']; norm=re.sub(r'_\\d+"'$'"','',ext) if typ=='ENCODE' else ext
 print(sid,ext,norm,norm in tokens)
PY"
