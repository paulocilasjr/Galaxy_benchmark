/bin/bash -lc "python3 - <<'PY'
import json,re
mapping=json.load(open('sei_cistrome_map.json'))
mouse_ids={int(k) for k,s in mapping.items() if s.get('species')=='mus_musculus'}
print('mouse ids',sorted(mouse_ids))
lines=open('sei_repo/model/target.names').read().splitlines()
for line in lines:
 m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if m and int(m.group(1)) in mouse_ids:
  print(line)
PY"
