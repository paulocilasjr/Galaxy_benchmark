/bin/bash -lc "cat > fetch_cistrome_map.py <<'PY'
import urllib.request, json, re, time, csv
from collections import defaultdict
lines=open('sei_repo/model/target.names').read().splitlines()
sei=[]
for i,line in enumerate(lines):
 m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if m: sei.append((i,int(m.group(1)),line))
ids=sorted(set(x[1] for x in sei))
mapping={}
errors=[]
for start in range(0,len(ids),500):
 chunk=ids[start:start+500]
 url='https://db3.cistrome.org/cistrome/samples?ids='+','.join(map(str,chunk))+'&limit=500'
 for attempt in range(3):
  try:
   req=urllib.request.Request(url,headers={'User-Agent':'curl/8'})
   data=urllib.request.urlopen(req,timeout=120).read().decode()
   j=json.loads(data)
   samps=j.get('samples',[])
   if len(samps)"'!=len(chunk):
    print('"'short chunk',start,len(samps),len(chunk))
   for s in samps:
    mapping[int(s['id'])] = s
   break
  except Exception as e:
   print('ERR',start,repr(e)[:300])
   if attempt==2: errors.append((start,repr(e)))
   time.sleep(2*(attempt+1))
 print('progress',start+len(chunk),'/',len(ids),'errors',len(errors),'mapped',len(mapping))
print('total mapped',len(mapping),'missing',len(set(ids)-set(mapping)))
with open('sei_cistrome_map.json','w') as f: json.dump(mapping,f)
with open('sei_cistrome_errors.json','w') as f: json.dump(errors,f)
PY
python3 fetch_cistrome_map.py"
