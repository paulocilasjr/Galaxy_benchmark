/bin/bash -lc "python3 - <<'PY'
import csv,re
from collections import defaultdict
for fn in ['borzoi_targets_human.tsv']:
 d=defaultdict(list)
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  m=re.search(r'/(?:encode|geo)/(["'^/]+)/'"',r['file'])
  if m and r['identifier'].startswith('ENCFF'):
   d[m.group(1)].append(r)
 for k,rs in d.items():
  if len(rs)>1:
   print(k)
   for r in rs: print(' ',r['identifier'],r['description'],r['file'])
   break
PY"
