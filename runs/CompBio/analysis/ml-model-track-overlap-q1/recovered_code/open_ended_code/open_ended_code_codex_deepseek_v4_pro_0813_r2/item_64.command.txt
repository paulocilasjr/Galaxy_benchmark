/bin/bash -lc "python3 - <<'PY'
import json,csv,re
mapping=json.load(open('sei_cistrome_map.json'))
# Build borzoi identifier set human/mouse
bids=set()
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  bids.add(r['identifier'])
# also strip +/-? for CAGE no.
count=0; by={}
for line in open('sei_repo/model/target.names'):
 line=line.rstrip('\\n'); m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if not m: continue
 sid=int(m.group(1)); s=mapping[str(sid)]; typ=s['external_id_type']; ext=s['external_id']; norm=re.sub(r'_\\d+"'$'"','',ext) if typ=='ENCODE' else ext
 if ext in bids or norm in bids:
  count+=1; by[typ]=by.get(typ,0)+1
print('identifier-only matches',count,by)
# file path provenance with exact external id not normalize ENCODE suffix? 
count2=0
for line in open('sei_repo/model/target.names'):
 line=line.rstrip('\\n'); m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if not m: continue
 sid=int(m.group(1)); s=mapping[str(sid)]; ext=s['external_id']
 if ext in bids: count2+=1
print('ext exact identifier matches',count2)
PY"
