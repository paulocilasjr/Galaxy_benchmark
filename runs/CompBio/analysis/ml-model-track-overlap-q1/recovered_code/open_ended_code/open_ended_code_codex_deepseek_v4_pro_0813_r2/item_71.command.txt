  borz.add(m.group(1) if m else r['identifier'])
counts={}
for norm_enc in [True,False]:
 c=0; by={}
 for line in open('sei_repo/model/target.names'):

  if not m: continue
  s=mapping[m.group(1)]; ext=s['external_id']; typ=s['external_id_type']
  key=re.sub(r'_\\d+"'$'"','',ext) if (norm_enc and typ=='ENCODE') else ext
  if key in borz: c+=1; by[typ]=by.get(typ,0)+1
 print('norm',norm_enc,c,by)
