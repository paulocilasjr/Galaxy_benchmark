/bin/bash -lc "python3 - <<'PY'
import json,csv,re
from collections import defaultdict,Counter
mapping=json.load(open('sei_cistrome_map.json'))
base_enc=defaultdict(set)
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  if not r['description'].startswith(('CHIP:','DNASE:','ATAC:')): continue
  m=re.search(r'/(?:encode|geo)/(["'^/]+)/'"',r['file'])
  if m and r['identifier'].startswith('ENCFF'):
   base_enc[m.group(1)].add(r['identifier'].rstrip('+-'))
rows=defaultdict(list)
for line in open('sei_repo/model/target.names'):
 line=line.rstrip('\\n'); m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if not m: continue
 s=mapping[m.group(1)]; ext=s['external_id']; typ=s['external_id_type']
 if typ=='ENCODE':
  mm=re.match(r'"'^(ENCSR'"\\d+[A-Z0-9]+)(?:_(\\d+))?"'$'"',ext)
  if mm and mm.group(1) in base_enc: rows[mm.group(1)].append(mm.group(2))
print('unique bases',len(rows),'total rows',sum(len(v) for v in rows.values()))
print('bases by number of matching Sei rows',Counter(len(v) for v in rows.values()))
print('suffix counts',Counter(s for v in rows.values() for s in v))
PY"
