/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for exp in ['ENCSR037PGI','ENCSR083STA','ENCSR097BWW']:
 print('---',exp)
 url=f'https://www.encodeproject.org/experiments/{exp}/?format=json'
 try:
  data=json.loads(urllib.request.urlopen(url,timeout=30).read().decode())
  for rep in data.get('replicates',[]):
   print('bio',rep.get('biological_replicate_number'),'tech',rep.get('technical_replicate_number'),'library',rep.get('library',{}).get('accession'),'status',rep.get('status'))
   files=rep.get('files',[])
   print(' files',[(f.get('accession'),f.get('output_type'),f.get('file_format')) for f in files])
 except Exception as e: print('ERR',e)
PY"
