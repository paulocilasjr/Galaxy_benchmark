/bin/bash -lc "python3 - <<'PY'
import csv,re
from collections import Counter
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 rows=list(csv.DictReader(open(fn),delimiter='\\t'))
 tokens=[]
 for r in rows:
  p=r['file']; m=re.search(r'/(?:encode|geo|fantom|recount3)/(["'^/]+)/'"',p)
  tok=m.group(1) if m else r['identifier']
  tokens.append(tok)
 print(fn,'source token prefixes',Counter(tok.split('_')[0][:6] if not tok.startswith(('GSM','CNhs','GTEX','ENCFF','ENCSR')) else tok[:5] for tok in tokens).most_common(20))
PY"
