/bin/bash -lc "python3 - <<'PY'
import json,csv,re
mapping=json.load(open('sei_cistrome_map.json'))
paths=[]
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  paths.append(r['file'])
# substring matches normalized Cistrome IDs in any Borzoi path
count=0; by={}
for line in open('sei_repo/model/target.names'):
 line=line.rstrip('\\n'); m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if not m: continue
 s=mapping[m.group(1)]; ext=s['external_id']; typ=s['external_id_type']; norm=re.sub(r'_\\d+"'$'"','',ext) if typ=='ENCODE' else ext
 if any(norm in p for p in paths):
  count+=1; by[typ]=by.get(typ,0)+1
print('substring path matches',count,by)
PY"
