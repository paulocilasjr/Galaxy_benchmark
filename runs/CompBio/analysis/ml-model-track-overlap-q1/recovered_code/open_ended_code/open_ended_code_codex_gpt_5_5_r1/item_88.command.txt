/bin/bash -lc "python - <<'PY'
import gzip,csv,re,json
keys=set()
for path in ['borzoi/data/targets_human.txt.gz','borzoi/data/targets_mouse.txt.gz']:
 with gzip.open(path,'rt') as f:
  r=csv.DictReader(f,delimiter='\\t')
  for row in r:
   for field in [row.get('identifier',''),row.get('file',''),row.get('description','')]:
    for token in re.findall(r'(ENCSR[A-Z0-9]+|ENCFF[A-Z0-9]+|GSM[0-9]+|GSE[0-9]+|SRX[0-9]+|SRR[0-9]+)', field): keys.add(token)
print('keys',len(keys))
matched=0; mousehits=[]
for line in open('matched_sei_tracks.jsonl'):
 pass
samples={json.loads(l)['id']:json.loads(l) for l in open('sei_cistrome_samples.jsonl')}
for line in open('sei-framework/model/target.names'):
 m=re.search(r'ID:(\\d+)',line)
 if not m: continue
 s=samples[int(m.group(1))]
 ext=s.get('external_id') or ''
 norm={ext}
 for pat in [r'"'^(ENCSR[A-Z0-9]+)'"',r'"'^(ENCFF[A-Z0-9]+)'"',r'"'^(GSM[0-9]+)'"']:
  mm=re.match(pat, ext)
  if mm: norm.add(mm.group(1))
 if norm & keys:
  matched += 1
  if s.get('species')=='mus_musculus': mousehits.append((line.strip(), ext, norm&keys))
print('matched human+mouse borzoi',matched,'rounded',round(matched/100)*100)
print('mousehits',mousehits)
PY"
