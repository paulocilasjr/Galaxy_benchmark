/bin/bash -lc "python - <<'PY'
import gzip, csv, collections
c=collections.Counter()
with gzip.open('borzoi/data/targets_human.txt.gz','rt') as f:
    r=csv.DictReader(f, delimiter='\\t')
    for row in r:
        parts=row['file'].split('/')
        try: idx=parts.index('human')
        except ValueError: continue
        key='/'.join(parts[idx+1:idx+3])
        c[key]+=1
for k,v in c.most_common(50): print(v,k)
PY"
