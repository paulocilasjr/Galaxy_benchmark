/bin/bash -lc "python - <<'PY'
import json,re
from collections import Counter,defaultdict
s=json.load(open('cistrome_samples.json'))
# Build source accession maps explicitly from target table.
borzoi_enc=defaultdict(set); borzoi_geo=defaultdict(set)
with open('borzoi_targets_human.txt') as f:
 h=f.readline().rstrip().split('\\t')
 for l in f:
  x=dict(zip(h,l.rstrip().split('\\t')))
  assay=x['description'].split(':',1)[0]
  for eid in re.findall(r'ENCSR[A-Z0-9]+',x['file']): borzoi_enc[eid].add(assay)
  for gsm in re.findall(r'GSM\\d+',x['identifier']): borzoi_geo[gsm].add(assay)
# count tracks assigned by source type / Borzoi assay
ct=Counter(); ids=defaultdict(list)
for x in s.values():
 eid=x['external_id']; norm=re.sub(r'_\\d+"'$'"','',eid)
 if norm in borzoi_enc:
  ct['ENCODE total']+=1
  for assay in borzoi_enc[norm]:ct['ENCODE '+assay]+=1
  ids[norm].append(x)
 elif eid in borzoi_geo:
  ct['GEO total']+=1
  for assay in borzoi_geo[eid]:ct['GEO '+assay]+=1
  ids[eid].append(x)
print(ct)
print('all',sum(v for k,v in ct.items() if k.endswith('total')))
print('unique provenance IDs',len(ids))
print('unique ENCSR/GSM',Counter('ENCODE' if k.startswith('ENCSR') else 'GEO' for k in ids))
# Weird: Cistrome matches with both? none likely.
# direct external-id matching stats
borzoi_ident=set(); borzoi_file_accessions=set()
with open('borzoi_targets_human.txt') as f:
 h=f.readline().rstrip().split('\\t')
 for l in f:
  x=dict(zip(h,l.rstrip().split('\\t')))
  borzoi_ident.add(x['identifier'])
  borzoi_file_accessions.update(re.findall(r'ENCFF[A-Z0-9]+|ENCSR[A-Z0-9]+|GSM\\d+',x['file']))
print('literal ext id overlap with identifier',sum(x['external_id'] in borzoi_ident for x in s.values()),'with any file accession',sum(x['external_id'] in borzoi_file_accessions for x in s.values()))
PY"
