/bin/bash -lc "python3 - <<'PY'
import json,re
ids=[]
for line in open('sei-framework/model/target.names'):
 m=re.search(r'ID:(\\d+)',line)
 if m: ids.append(int(m.group(1)))
samples=json.load(open('cistrome_sei_samples.json'))
miss=[i for i in ids if str(i) not in samples]
print('first missing', miss[:20])
print('last missing', miss[-20:])
print('returned minmax', min(map(int,samples)), max(map(int,samples)))
print('missing below 10000', sum(i<10000 for i in miss), 'below 30000', sum(i<30000 for i in miss), 'above 50000', sum(i>50000 for i in miss))
PY"
