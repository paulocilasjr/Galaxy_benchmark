/bin/bash -lc "python3 - <<'PY'
import gzip,csv,re,collections
rows=list(csv.DictReader(gzip.open('borzoi/data/targets_human.txt.gz','rt'), delimiter='\\t'))
patterns={
 'ENCFF': re.compile(r'ENCFF[A-Z0-9]+'),
 'ENCSR': re.compile(r'ENCSR[A-Z0-9]+'),
 'FANTOM_CNHS': re.compile(r'CNhs\\d+'),
 'GSM': re.compile(r'GSM\\d+'),
 'SRR': re.compile(r'SRR\\d+'),
 'DRR': re.compile(r'DRR\\d+'),
 'ERR': re.compile(r'ERR\\d+'),
}
counts=collections.Counter()
examples={}
for r in rows:
 text='\\t'.join(r.values())
 for k,p in patterns.items():
  ms=p.findall(text)
  if ms:
   counts[k]+=len(set(ms))
   examples.setdefault(k, ms[:5])
print('rows',len(rows))
print(counts)
print(examples)
for i,r in enumerate(rows):
 if 'catlas' in r['file'].lower(): print('catlas first', i, r); break
for i,r in enumerate(rows):
 if '/chip/' in r['file'].lower() or 'chip' in r['description'].lower(): print('chip first', i, r); break
PY"
