/bin/bash -lc "python3 - <<'PY'
import json,collections,re
m=json.load(open('provenance_overlap_matches.json'))
by=collections.Counter()
for x in m:
 ov=x['overlap']
 if any(t.startswith('GSM') for t in ov): by['has_GSM']+=1
 if any(t.startswith('GSE') for t in ov): by['has_GSE']+=1
 if any(t.startswith('ENCSR') for t in ov): by['has_ENCSR']+=1
 if any(t.startswith('ENCFF') for t in ov): by['has_ENCFF']+=1
print(by)
print('only GSE', sum(all(t.startswith('GSE') for t in x['overlap']) for x in m))
print('only ENCSR', sum(all(t.startswith('ENCSR') for t in x['overlap']) for x in m))
print('only GSM', sum(all(t.startswith('GSM') for t in x['overlap']) for x in m))
PY"
