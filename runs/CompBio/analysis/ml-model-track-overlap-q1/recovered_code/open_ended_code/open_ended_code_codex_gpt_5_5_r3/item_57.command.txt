import urllib.parse, urllib.request, json
accs=['ENCSR000AHD','ENCSR000EIJ','ENCSR000AAA']

for a in accs: qs.append(('accession',a))
url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode(qs)
print(url)
with urllib.request.urlopen(url, timeout=30) as r: d=json.load(r)
print(d.get('total'), len(d.get('@graph',[])))
for x in d.get('@graph',[]): print(x.get('accession'), x.get('dbxrefs'))
PY"
