/bin/bash -lc "python - <<'PY'
import urllib.request,urllib.parse,json,re
lines=open('borzoi/examples/targets_human.txt').read().splitlines()[1:]
by={}
for x in lines:
 f=x.split('\\t'); ident=f[1].rstrip('+-');
 if ident.startswith('GSM'): by.setdefault(ident,[]).append(f[8])
ids=sorted(by)
u='https://db3.cistrome.org/cistrome/samples?external_ids='+','.join(ids)+'&limit=1000'
xs=urllib.request.urlopen(u,timeout=60).read(); xs=json.loads(xs)['samples']
sei={int(m.group(1)):line for line in open('sei-framework/model/target.names') if (m:=re.search(r'\\| ID:(\\d+)"'$'"',line))}
hits=[x for x in xs if x['id'] in sei]
from collections import Counter
print('matches',len(hits),'Borzoi assay',Counter(y.split(':')[0] for x in hits for y in by[x['external_id']]))
print('by source',Counter(x['external_id'][:7] for x in hits))
for x in hits: print(x['id'], x['external_id'], by[x['external_id']], sei[x['id']].strip())
PY"
