/bin/bash -lc "python - <<'PY'
import re
from collections import Counter
ls=open('sei-framework/model/target.names').read().splitlines()
ids=[int(re.search(r'\\| ID:(\\d+)"'$'"',x).group(1)) for x in ls if re.search(r'\\| ID:(\\d+)"'$'"',x)]
print('cistrome rows',len(ids),'unique ids',len(set(ids)),'min/max',min(ids),max(ids))
print('duplicates',sum(v>1 for v in Counter(ids).values()),'max multiplicity',max(Counter(ids).values()))
bor=open('borzoi/examples/targets_human.txt').read().splitlines()[1:]
bi=[x.split('\\t')[1].rstrip('+-') for x in bor]
print('borzoi IDs',len(bi),'unique',len(set(bi)))
print('prefix',Counter(re.match(r'[A-Za-z]+',x).group(0) if re.match(r'[A-Za-z]+',x) else '' for x in bi))
print('GSM',len({x for x in bi if x.startswith('GSM')}),'ENCFF',len({x for x in bi if x.startswith('ENCFF')}))
print('first/last ids',ids[:5],ids[-5:])
PY"
