/bin/bash -lc "python - <<'PY'
import urllib.request,urllib.parse,json,re
lines=open('borzoi/examples/targets_human.txt').read().splitlines()[1:]
ids=sorted({x.split('\\t')[1].rstrip('+-') for x in lines if x.split('\\t')[1].startswith('GSM')})
print('query',len(ids))
u='https://db3.cistrome.org/cistrome/samples?external_ids='+urllib.parse.quote(','.join(ids),safe=',')
with urllib.request.urlopen(u,timeout=60) as f: data=json.load(f)
xs=data['samples']; print('returned',len(xs))
sei=set()
for line in open('sei-framework/model/target.names'):
 m=re.search(r'\\| ID:(\\d+)"'$'"',line)
 if m: sei.add(int(m.group(1)))
hits=[x for x in xs if x['id'] in sei]
print('hits in Sei Cistrome IDs',len(hits))
for x in hits: print(x['id'],x['external_id'],x['experiment_type'],x['title'])
print('returned IDs in sei?',len([x for x in xs if x['id'] in sei]))
print('returned external type', {x['external_id_type'] for x in xs})
PY"
