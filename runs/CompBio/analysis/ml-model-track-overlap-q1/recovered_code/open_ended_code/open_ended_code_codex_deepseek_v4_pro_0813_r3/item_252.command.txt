/bin/bash -lc "python3 - <<'PY'
import requests,re
url='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.supplementary-material'
r=requests.get(url,headers={'User-Agent':'Mozilla/5.0'},timeout=60)
print(r.status_code,len(r.text))
open('supp_material.html','w').write(r.text)
for m in re.finditer(r'href=\"(["'^"]+)"[''^>]*>([''^<]*(?:xlsx|csv|xls|tsv|zip)[''^<]*)'"',r.text,re.I):
 print(m.groups())
# print links with file extensions
for line in r.text.splitlines():
 if '.xlsx' in line or '.csv' in line or '.tsv' in line or '.zip' in line:
  print(line.strip()[:500])
PY"
