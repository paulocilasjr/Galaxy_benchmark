/bin/bash -lc "python3 - <<'PY'
import csv,re,collections,json
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
fm=json.load(open('borzoi_encff_meta.json'))
# map path ENCSR to rows and file metadata reps
m=collections.defaultdict(list)
for r in rows:
 ident=r['identifier'].strip()
 if re.match(r'ENCFF\\d{3}[A-Z]{3}',ident):
  f=r['file'].strip().split('/')
  i=f.index('encode')
  en=f[i+1]
  acc=re.match(r'(ENCFF\\d{3}[A-Z]{3})',ident).group(1)
  ds=fm.get(acc,{}).get('dataset',''); cur=ds.rstrip('/').split('/')[-1] if ds else ''
  reps=fm.get(acc,{}).get('biological_replicates',[])
  m[en].append((acc,cur,tuple(reps),r['description']))
# Show ENCSR with multiple unique files or same file +/-; sample
for en,vals in list(m.items())[:30]:
 files=set(v[0] for v in vals); reps=set(v[2] for v in vals)
 if len(files)>1 or len(reps)>1:
  print(en,'rows',len(vals),'files',len(files),'reps',reps)
  for v in vals[:5]: print(' ',v)
PY"
