/bin/bash -lc "python3 - <<'PY'
import json,re
sei=json.load(open('sei_cistrome_meta.json'))
borzoi_enc=set()
import csv
for r in csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'):
 m=re.search(r'/ENCSR[0-9A-Z]+/',r['file'])
 if m: borzoi_enc.add(m.group(0).strip('/'))
print('borzoi unique ENCSR',len(borzoi_enc))
enc_sei=[s for s in sei.values() if s.get('external_id_type')=='ENCODE']
print('sei ENCODE ext',len(enc_sei))
# normalize base before _ rep
match=[]
for s in enc_sei:
 e=s['external_id']
 base=e.split('_')[0]
 if base in borzoi_enc:
  match.append((s['id'],e,base))
print('direct ENCSR matches',len(match))
# check some
for m in match[:30]: print(m)
# How many unique Sei tracks with base matching; also what about base+replicate?

PY"
