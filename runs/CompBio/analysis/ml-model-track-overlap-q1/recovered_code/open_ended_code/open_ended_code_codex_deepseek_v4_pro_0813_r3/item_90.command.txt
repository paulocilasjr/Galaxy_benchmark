/bin/bash -lc "python3 - <<'PY'
import requests,json,re
# load first chunk from script logic
import csv
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
accs=[]
for r in rows:
 ident=r['identifier'].strip(); m=re.match(r'(ENCFF\\d{3}[A-Z]{3})[+-]?"'$'"',ident)
 if m: accs.append(m.group(1))
accs=list(dict.fromkeys(accs))
chunk=accs[:150]
params=[('type','File'),('format','json'),('field','accession'),('field','dataset'),('field','biological_replicates')]+[('accession',a) for a in chunk]
url=requests.Request('GET','https://www.encodeproject.org/search/',params=params).prepare().url
print('url length',len(url), 'n acc',len(chunk))
r=requests.get(url,timeout=120)
print(r.status_code, len(r.url), r.url[:500])
print('returned',len(r.json().get('@graph',[])))
print('first ids',[g.get('accession') for g in r.json().get('@graph',[])[:10]])
PY"
