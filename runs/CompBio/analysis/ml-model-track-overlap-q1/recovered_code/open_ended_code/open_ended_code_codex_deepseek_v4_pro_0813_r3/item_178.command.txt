import requests,re,json
# NCBI esummary for GSE90397 to get GSM ids
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi'

print(r.status_code)
print(r.text[:1000])
if r.ok:
 ids=r.json().get('esearchresult',{}).get('idlist',[])
 print('ids',ids)
 for id in ids[:5]:
  # esummary
  s=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':id,'retmode':'json'},timeout=60)
  print(s.text[:1000])
PY"
