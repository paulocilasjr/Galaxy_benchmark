/bin/bash -lc "python3 - <<'PY'
import json,csv,re
sei=json.load(open('sei_cistrome_meta.json'))
direct=set(); enc_gsms=set()
for r in csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'):
 if re.fullmatch(r'GSM\\d+',r['identifier']): direct.add(r['identifier'])
enc=json.load(open('borzoi_encode_experiments.json'))
for d in enc:
 if d['gsms']: enc_gsms.update(d['gsms'].split(','))
sei_gsm={s['external_id'] for s in sei.values() if s.get('external_id_type')"'!='"'ENCODE' and s.get('external_id')}
print('direct overlap',len(sei_gsm & direct))
print('enc gsm overlap',len(sei_gsm & enc_gsms))
print('union',len(sei_gsm & (direct|enc_gsms)))
print('intersection direct & enc',len(direct & enc_gsms))
PY"
