/bin/bash -lc "python3 - <<'PY'
import json,requests
meta=json.load(open('sei_cistrome_meta.json'))
sei=set(v['external_id'] for v in meta.values() if v.get('external_id_type')=='GEO')
for acc in ['ENCSR946DXB','ENCSR322JEO']:
 params=[('type','Experiment'),('format','json'),('limit','all'),('field','accession'),('field','dbxrefs'),('field','replicates.library.accession'),('field','replicates.biological_replicate_number'),('field','possible_controls'),('field','assay_title')]
 params.append(('accession',acc))
 r=requests.get('https://www.encodeproject.org/search/',params=params,timeout=120)
 print('\\n',acc)
 for g in r.json().get('@graph',[]):
  print('dbxrefs',g.get('dbxrefs'),'assay',g.get('assay_title'))
  print('reps',[(x.get('library',{}).get('accession'),x.get('biological_replicate_number')) for x in g.get('replicates',[])])
  print('controls')
  for pc in g.get('possible_controls',[]):
   print(' ',pc.get('accession'),pc.get('dbxrefs'))
 # check which gsms in sei
 e=next(x for x in json.load(open('borzoi_encode_experiments.json')) if x['accession']==acc)
 print('gsms',e['gsms'])
 for gsm in e['gsms'].split(','):
  if gsm in sei: print('  IN SEI',gsm)
PY"
