/bin/bash -lc "python3 - <<'PY'
import requests,re,json
# quick GEO accession esummary
for gse in ['GSE92156','GSE100996','GSE101150']:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':gse+'[ACCN]','retmax':100,'retmode':'json'},timeout=60)
 ids=r.json().get('esearchresult',{}).get('idlist',[])
 print(gse,'ids',ids)
 for id in ids[:3]:
  s=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':id,'retmode':'json'},timeout=60)
  # print sample accessions/titles
  data=s.json().get('result',{}).get(id,{})
  print(' ',data.get('accession'),'title',data.get('title'),'samples',[(x.get('accession'),x.get('title')) for x in data.get('samples',[])])
  break
PY"
