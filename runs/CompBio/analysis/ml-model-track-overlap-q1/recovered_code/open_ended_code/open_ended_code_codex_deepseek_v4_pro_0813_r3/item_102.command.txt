import json
enc=json.load(open('borzoi_encode_experiments.json'))
sei=json.load(open('sei_cistrome_meta.json'))

for d in enc:
 if d['gses']: bgse.update(d['gses'].split(','))
sgse=set()
for s in sei.values():
 if s.get('series'): sgse.update(s['series'].split(','))
print(len(bgse),len(sgse),len(bgse&sgse))
print('borzoi gse examples',list(bgse)[:20])
print('sei gse examples',list(sgse)[:20])
# common numeric after GSE?
bnum={x[3:] for x in bgse if x.startswith('GSE')}
snum={x[3:] for x in sgse if x.startswith('GSE')}
print('numeric intersect',len(bnum&snum), list(bnum&snum)[:20])
PY"
