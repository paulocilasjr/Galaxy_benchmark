/bin/bash -lc "python3 - <<'PY'
import json
meta=json.load(open('sei_cistrome_meta.json'))
sei=set(v['external_id'] for v in meta.values() if v.get('external_id_type')=='GEO')
exp=json.load(open('borzoi_encode_experiments.json'))
matched_gsms=set()
for e in exp:
 for g in e.get('gsms','').split(','):
  if g in sei: matched_gsms.add(g)
print(len(matched_gsms))
json.dump(sorted(matched_gsms),open('matched_sei_borzoi_gsms.json','w'))
PY"
