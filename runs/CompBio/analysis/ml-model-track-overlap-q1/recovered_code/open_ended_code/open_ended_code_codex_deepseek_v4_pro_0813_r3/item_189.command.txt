/bin/bash -lc "python3 - <<'PY'
import json,csv,re,collections
meta=json.load(open('sei_cistrome_meta.json'))
sei_geo=[]; sei_enc=[]
for cid,v in meta.items():
 et=v.get('external_id_type')
 if et=='GEO':
  gses=[s.strip() for s in v.get('series','').split(',') if s.strip()]
  sei_geo.append((cid,v.get('external_id'),set(gses)))
 elif et=='ENCODE':
  ext=v.get('external_id',''); sei_enc.append((cid,ext,ext.split('_')[0],ext.split('_')[1] if '_' in ext else None))
# Borzoi exp includes gsms/gses
exp=json.load(open('borzoi_encode_experiments.json'))
bor_gsm=set(); bor_gse=set()
for e in exp:
 for g in e.get('gsms','').split(','):
  if g.strip(): bor_gsm.add(g.strip())
 for g in e.get('gses','').split(','):
  if g.strip(): bor_gse.add(g.strip())
# direct geo GSM; also direct GSE maybe not in target
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
bor_direct_gsm=set()
for r in rows:
 if re.match(r'GSM\\d+"'$'"',r['identifier'].strip()): bor_direct_gsm.add(r['identifier'].strip())
all_gsm=bor_gsm|bor_direct_gsm
print('sei geo tracks',len(sei_geo),'gse overlap tracks',sum(1 for cid,gsm,gses in sei_geo if gses & bor_gse))
print('sei geo gsm overlap tracks',sum(1 for cid,gsm,gses in sei_geo if gsm in all_gsm))
# unique matching by gse maybe count all Sei tracks whose any gse in bor
# also match by gsm or gse
print('either gsm/gse',sum(1 for cid,gsm,gses in sei_geo if gsm in all_gsm or gses & bor_gse))
# Sei ENCODE base match
bor_encsr=set()
for r in rows:
 f=r['file'].strip().split('/')
 if 'encode' in f:
  i=f.index('encode'); bor_encsr.add(f[i+1])
print('sei enc base overlap',sum(1 for cid,ext,base,rep in sei_enc if base in bor_encsr))
# combined gse or encsr?
PY"
