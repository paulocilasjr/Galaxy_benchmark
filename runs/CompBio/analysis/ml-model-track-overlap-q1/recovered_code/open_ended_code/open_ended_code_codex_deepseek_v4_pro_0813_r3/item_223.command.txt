/bin/bash -lc "python3 - <<'PY'
import json
meta=json.load(open('sei_cistrome_meta.json'))
sei=set(v['external_id'] for v in meta.values() if v.get('external_id_type')=='GEO')
exp=json.load(open('borzoi_encode_experiments.json'))
matched=[]
for e in exp:
 gs=set(g for g in e.get('gsms','').split(',') if g)
 if gs & sei:
  matched.append(e)
print('matched experiments',len(matched))
gses=set()
for e in matched:
 for g in e.get('gses','').split(','):
  if g: gses.add(g)
print('gses',len(gses))
print('gsm matches total',sum(len(set(g for g in e.get('gsms','').split(',') if g)&sei) for e in matched))
PY"
