import csv,re,collections
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))

 if m: exp.add(m.group(0).strip('/'))

for r in rows:

 if m: c[(r['file'].split('/')[6], r['file'].split('/')[7])]+=1
print(c)
print(sorted(exp)[:20])
PY"
