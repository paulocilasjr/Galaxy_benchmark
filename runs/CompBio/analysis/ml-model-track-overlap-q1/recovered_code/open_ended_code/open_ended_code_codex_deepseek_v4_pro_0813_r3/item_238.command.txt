/bin/bash -lc "python3 - <<'PY'
import json
exp_all=json.load(open('borzoi_encode_experiments.json'))
broad=set()
for e in exp_all:
 for g in e.get('gsms','').split(','):
  if g: broad.add(g)
# build target replicate gsms from library map
exp_reps=json.load(open('borzoi_exp_reps.json'))
lib_map=json.load(open('borzoi_lib_gsm_map.json'))
target=set()
for e in exp_reps:
 for r in e.get('replicates',[]):
  lib=r.get('library')
  if lib and lib in lib_map:
   target.update(lib_map[lib])
print('broad',len(broad),'target',len(target))
print('broad-target',len(broad-target), sorted(broad-target)[:100])
print('target-broad',len(target-broad), sorted(target-broad)[:100])
PY"
