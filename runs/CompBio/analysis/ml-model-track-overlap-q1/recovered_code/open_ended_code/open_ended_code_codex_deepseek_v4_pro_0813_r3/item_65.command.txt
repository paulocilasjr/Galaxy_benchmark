import csv,re,collections
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
c=collections.Counter()
for r in rows:
 p=r['file'].split('/')
 c[(p[6],p[7])]+=1
print(c)
# unique exps by category
for cat in [('chip','encode'),('rna','encode'),('dnase','encode')]:
 exps=[]
 for r in rows:
  p=r['file'].split('/')
  if p[6]==cat[0] and p[7]==cat[1]:
   m=re.search(r'/ENCSR[0-9A-Z]+/',r['file'])
   exps.append(m.group(0).strip('/'))
 print(cat,len(exps),len(set(exps)),collections.Counter(exps).most_common(5))
PY"
