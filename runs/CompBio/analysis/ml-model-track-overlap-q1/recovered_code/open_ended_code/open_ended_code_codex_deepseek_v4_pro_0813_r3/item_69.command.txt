/bin/bash -lc "python3 - <<'PY'
import json,re,csv
# load sei
sei=json.load(open('sei_cistrome_meta.json'))
# borzoi direct GSM identifiers
borzoi_gsm=set()
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
for r in rows:
 if re.fullmatch(r'GSM\\d+',r['identifier']): borzoi_gsm.add(r['identifier'])
# ENCODE library GSMs
enc=json.load(open('borzoi_encode_experiments.json'))
for d in enc:
 if d['gsms']: borzoi_gsm.update(d['gsms'].split(','))
# Borzoi ENCSR set from paths
borzoi_enc=set()
for r in rows:
 m=re.search(r'/ENCSR[0-9A-Z]+/',r['file'])
 if m: borzoi_enc.add(m.group(0).strip('/'))
# counts by type
geo_match=0; enc_match=0
for s in sei.values():
 if s.get('external_id_type')=='ENCODE':
  base=s['external_id'].split('_')[0]
  if base in borzoi_enc: enc_match+=1
 else:
  if s.get('external_id') in borzoi_gsm: geo_match+=1
print('geo_match',geo_match,'enc_match',enc_match,'total',geo_match+enc_match)
print('rounded', round((geo_match+enc_match)/100)*100)
# unique GSM matched and unique ENCSR matched
PY"
