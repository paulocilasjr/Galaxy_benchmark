/bin/bash -lc "python3 - <<'PY'
import json,random,requests,time
sei=json.load(open('sei_cistrome_meta.json'))
ids=random.sample(list(sei.keys()),20)
for i in ids:
 try:
  old=requests.get(f'http://dc2.cistrome.org/api/inspector?id={i}',timeout=30).json()
  oldid=old.get('treats',[{}])[0].get('unique_id')
  newid=sei[i]['external_id']
  print(i,oldid,newid,'OK' if oldid==newid else 'DIFF')
 except Exception as e:
  print(i,'ERR',e)
PY"
