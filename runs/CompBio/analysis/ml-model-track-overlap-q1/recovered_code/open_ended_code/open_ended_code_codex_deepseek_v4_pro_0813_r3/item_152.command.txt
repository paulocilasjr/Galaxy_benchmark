/bin/bash -lc "python3 - <<'PY'
import csv,re,collections,json
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
print('rows',len(rows))
# robust categories based on identifier and file
cnt=collections.Counter(); direct_gsm=set(); encff_set=set(); encff_rows=[]; fan=set(); catlas=set(); gtex=set(); other=[]
for r in rows:
    ident=r['identifier'].strip(); file=r['file'].strip()
    m=re.match(r'(ENCFF\\d{3}[A-Z]{3})', ident)
    if m:
      cnt['ENCFF']+=1; encff_set.add(m.group(1)); encff_rows.append(r); continue
    if re.match(r'GSM\\d+"'$'"', ident):
      cnt['GEO_GSM']+=1; direct_gsm.add(ident); continue
    if ident.startswith('CNhs'):
      cnt['FANTOM']+=1; fan.add(ident.rstrip('+-')); continue
    if ident.startswith('kai'):
      cnt['CATlas']+=1; catlas.add(ident); continue
    if re.match(r'GTEX-', ident):
      cnt['GTEX']+=1; gtex.add(ident); continue
    cnt['OTHER']+=1; other.append((ident,file))
print('cnt',cnt)
print('unique encff',len(encff_set),'direct gsm',len(direct_gsm),'fan unique',len(fan),'catlas',len(catlas),'gtex',len(gtex))
print('other',len(other),other[:10])
# Inspect encff rows path patterns unique ENCSR
p=collections.Counter(); encsr=[]
for r in encff_rows:
    file=r['file'].strip()
    parts=file.split('/')
    if 'encode' in file:
       idx=parts.index('encode') if 'encode' in parts else -1
       if idx>=0 and idx+1<len(parts): p[parts[idx+1]]+=1; encsr.append(parts[idx+1])
print('encsr unique',len(set(encsr)),'from rows',len(encsr))
print(p.most_common(5))
# check identifier +/- pairs for encff same
from collections import defaultdict
pairs=defaultdict(set)
for r in encff_rows:
    ident=r['identifier'].strip()
    base=ident.rstrip('+-')
    pairs[base].add(ident[-1:] if ident[-1:] in '+-' else 'none')
print('encff bases with >1 strand variant',sum(1 for k,v in pairs.items() if len(v)>1), 'max', max(map(len,pairs.values())))
PY"
