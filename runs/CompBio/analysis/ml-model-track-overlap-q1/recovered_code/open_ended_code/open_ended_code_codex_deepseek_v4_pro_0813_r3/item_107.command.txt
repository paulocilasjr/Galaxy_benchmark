/bin/bash -lc "python3 - <<'PY'
import json,csv,re,collections
sei=json.load(open('sei_cistrome_meta.json'))
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
borzoi_enc={}
for r in rows:
 m=re.search(r'/ENCSR[0-9A-Z]+/',r['file'])
 if m:
  en=m.group(0).strip('/')
  borzoi_enc.setdefault(en,r['description'].split(':')[0])
# factor/id map
id2track={}
for l in open('sei-framework/model/target.names'):
 p=[x.strip() for x in l.rstrip().split('|')]
 if len(p)>=3 and p[2].startswith('ID:'): id2track[p[2][3:]]=(p[0],p[1])
c=collections.Counter()
for s in sei.values():
 if s.get('external_id_type')=='ENCODE':
  base=s['external_id'].split('_')[0]
  if base in borzoi_enc:
   c[(id2track[s['id']][1],borzoi_enc[base])]+=1
print('total',sum(c.values()))
for k,v in c.most_common(30): print(k,v)
PY"
