/bin/bash -lc "python3 - <<'PY'
import json,csv,re
# Load Sei target tracks and meta, count exact
meta=json.load(open('sei_cistrome_meta.json'))
# meta is dict by cistrome id; target unique 19905. Build track records
rows=[l.rstrip('\\n').split('|') for l in open('sei-framework/model/target.names')]
records=[]
for p in rows:
 p=[x.strip() for x in p]
 if len(p)>=3 and p[2].startswith('ID:'):
  cid=p[2][3:]
  v=meta.get(cid)
  if v:
   records.append({'cid':cid,'ext':v.get('external_id'),'type':v.get('external_id_type')})
print('sei cistrome records',len(records))
from collections import Counter
print(Counter(r['type'] for r in records))
# Borzoi rows
bor=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
bor_encsr=set()
bor_direct_gsm=set()
for r in bor:
 ident=r['identifier'].strip()
 if re.match(r'GSM\\d+"'$'"',ident): bor_direct_gsm.add(ident)
 f=r['file'].strip().split('/')
 if 'encode' in f:
  i=f.index('encode')
  if i+1<len(f): bor_encsr.add(f[i+1])
# bor encode gsms from experiment meta
exp=json.load(open('borzoi_encode_experiments.json'))
bor_enc_gsm=set()
for e in exp:
 for g in e.get('gsms','').split(','):
  if g: bor_enc_gsm.add(g)
bor_all_gsm=bor_direct_gsm|bor_enc_gsm
# counts
geo_match=sum(1 for rec in records if rec['type']=='GEO' and rec['ext'] in bor_all_gsm)
enc_match=sum(1 for rec in records if rec['type']=='ENCODE' and rec['ext'].split('_')[0] in bor_encsr)
print('geo_match',geo_match,'enc_match',enc_match,'total',geo_match+enc_match)
print('rounded',round((geo_match+enc_match)/100)*100)
PY"
