/bin/bash -lc "python3 - <<'PY'
import json
exp=json.load(open('borzoi_encode_experiments.json'))
reps=json.load(open('borzoi_exp_reps.json'))
rep_by={e['accession']:e for e in reps}
issues=[]
for e in exp:
 if not e.get('gses'): continue
 gs=[g for g in e.get('gsms','').split(',') if g]
 nrep=sum(1 for r in rep_by.get(e['accession'],{}).get('replicates',[]) if r.get('bio') is not None)
 if len(gs)>nrep:
  issues.append((e['accession'],e['assay'],e['gses'],len(gs),nrep,e['gsms']))
print('issues',len(issues))
for x in issues[:50]: print(x)
PY"
