/bin/bash -lc "python3 - <<'PY'
import json,requests
reps=json.load(open('borzoi_exp_reps.json'))
for acc in ['ENCSR000AQK','ENCSR000AQJ','ENCSR000AUM','ENCSR000ATN']:
 e=next(x for x in reps if x['accession']==acc)
 print('\\n',acc,e)
 # query experiment dbxrefs, possible_controls
 params=[('type','Experiment'),('format','json'),('limit','all'),('field','accession'),('field','dbxrefs'),('field','possible_controls'),('field','assay_title'),('field','target.label')]
 params.append(('accession',acc))
 r=requests.get('https://www.encodeproject.org/search/',params=params,timeout=120)
 for g in r.json().get('@graph',[]):
  print('dbxrefs',g.get('dbxrefs'),'assay',g.get('assay_title'),'target',g.get('target'))
  print('possible_controls len',len(g.get('possible_controls',[])))
  for pc in g.get('possible_controls',[])[:2]: print(' pc',pc.get('accession'),pc.get('dbxrefs'),pc.get('description'))
PY"
