/bin/bash -lc "python -u - <<'PY'
import json,re,pandas as pd,collections
s=json.load(open('sei_cistrome_metadata.json'))['samples']
b=pd.read_csv('borzoi_targets.tsv',sep='\\t',index_col=0)
alltxt='\\n'.join(b.identifier.astype(str))+'\\n'+'\\n'.join(b.file.astype(str))
borzoi_gsm=set(re.findall(r'GSM\\d+', alltxt))
borzoi_encsr=set(re.findall(r'ENCSR[A-Z0-9]+', alltxt))
def encbase(v):
 m=re.search(r'ENCSR[A-Z0-9]+',v)
 return m.group() if m else v.split('_')[0]
encvals=[x['external_id'] for x in s if x['external_id_type']=='ENCODE']
print('odd', [v for v in encvals if not re.search(r'ENCSR[A-Z0-9]+',v)][:20])
match_gsm=[x for x in s if x['external_id'] in borzoi_gsm]
match_enc=[x for x in s if x['external_id_type']=='ENCODE' and encbase(x['external_id']) in borzoi_encsr]
matched={x['id']:x for x in match_gsm+match_enc}
print('Borzoi GSM',len(borzoi_gsm),'ENCSR',len(borzoi_encsr))
print('Sei GSM match',len(match_gsm),'unique ext',len({x['external_id'] for x in match_gsm}))
print('Sei ENCSR match',len(match_enc),'unique base',len({encbase(x['external_id']) for x in match_enc}), 'unique ext',len({x['external_id'] for x in match_enc}))
print('TOTAL tracks',len(matched))
print('rounded',round(len(matched),-2))
print('\\nGSM matches')
for x in match_gsm: print(x['id'],x['external_id'],x['title'])
print('\\nFirst ENCSR matches')
for x in match_enc[:30]: print(x['id'],x['external_id'],x['title'])
PY"
