/bin/bash -lc "grep 'ENCSR037PGI\\|GSM1208745' borzoi_targets.tsv; grep 'ENCSR037PGI' sei-framework/model/target.names; python -u - <<'PY'
import json,re,pandas as pd,collections
s=json.load(open('sei_cistrome_metadata.json'))['samples']
b=pd.read_csv('borzoi_targets.tsv',sep='\\t',index_col=0)
# classify Borzoi source experiments by modality
expmods=collections.defaultdict(set)
for _,r in b.iterrows():
 m=re.search(r'ENCSR[A-Z0-9]+',str(r.file))
 if m:
  mm=re.search(r'/human/(["'^/]+/[''^/]+)/'"',str(r.file))
  expmods[m.group()].add(mm.group(1) if mm else '?')
seimatch=[]
for x in s:
 m=re.search(r'ENCSR[A-Z0-9]+',x['external_id'])
 if m and m.group() in expmods: seimatch.append((x,m.group(),expmods[m.group()]))
print('matched Sei by Borzoi modality')
print(collections.Counter(tuple(sorted(mods)) for _,_,mods in seimatch))
print('suffix distribution',collections.Counter(re.sub(r'ENCSR[A-Z0-9]+','',x['external_id']) for x,_,_ in seimatch))
# number where Borzoi has more than one output track for originating experiment
nbyexp=collections.Counter(re.findall(r'ENCSR[A-Z0-9]+','\\n'.join(b.file.astype(str))))
print('Borzoi track multiplicity across matching experiments',collections.Counter(nbyexp[e] for _,e,_ in seimatch))
PY"
