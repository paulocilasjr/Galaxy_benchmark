/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np,itertools

def load(f):
 with gzip.open(f,'rt') as h:
  h.readline(); return np.loadtxt(h,delimiter='\\t',dtype=np.float64,usecols=range(1,9))
r=load('multiome.match.atac.rna.q1.rna.tsv.gz'); a=load('multiome.match.atac.rna.q1.atac.tsv.gz')
# per feature transforms and correlation variants
variants={}
for rn,x in [('raw',r),('log',np.log1p(r))]:
 for an,y in [('raw',a),('log',np.log1p(a))]:
  cr=np.corrcoef(x.T); ca=np.corrcoef(y.T)
  variants[rn+'/'+an]=(cr,ca)
# all mappings RNA -> ATAC
pairs=np.array([(i,j) for i in range(8) for j in range(i+1,8)])
for nm,(cr,ca) in variants.items():
 rv=np.array([cr[i,j] for i,j in pairs])
 out=[]
 for p in itertools.permutations(range(8)):
  av=np.array([ca[p[i],p[j]] for i,j in pairs])
  # mse and correlation and abs deviations
  out.append((np.mean((rv-av)**2),-np.corrcoef(rv,av)[0,1],p,np.mean(np.abs(rv-av))))
 out.sort()
 print('\\n',nm)
 for z in out[:10]: print('mse %.8f corr %.6f mae %.6f p %s'% (z[0],-z[1],z[3],z[2]))
PY"
