/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, itertools, gzip, os
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0)
atac=pd.read_csv('multiome.match.atac.rna.q1.atac.tsv.gz',sep='\\t',index_col=0)
print('shapes',rna.shape,atac.shape)
print('rna sums',rna.sum().round(2).tolist())
print('atac sums',atac.sum().round(2).tolist())

def report(name,a):
    for trans in ['raw','log','zrow','rankrow']:
        x=a.to_numpy(dtype=float)
        if trans=='log': x=np.log1p(x)
        elif trans=='zrow':
            x=np.log1p(x); x=(x-x.mean(axis=1,keepdims=True))/(x.std(axis=1,keepdims=True)+1e-12)
        elif trans=='rankrow':
            # rank each feature over samples, only 8; exact ties average
            from scipy.stats import rankdata
            x=np.apply_along_axis(rankdata,1,x)
        c=np.corrcoef(x.T)
        print(name,trans, np.round(c,4).tolist())
report('RNA',rna)
report('ATAC',atac)
PY"
