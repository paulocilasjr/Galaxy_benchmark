/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np,itertools,sys
from scipy.stats import rankdata

def load(f):
 with gzip.open(f,'rt') as h:
  h.readline(); return np.loadtxt(h,delimiter='\\t',dtype=np.float64,usecols=range(1,9))
r=load('multiome.match.atac.rna.q1.rna.tsv.gz'); a=load('multiome.match.atac.rna.q1.atac.tsv.gz')
perms=np.array(list(itertools.permutations(range(8))),dtype=np.int8)
pairs=[(i,j) for i in range(8) for j in range(i+1,8)]
def calc(x,kind):
 if kind=='raw': return x
 if kind=='log': return np.log1p(x)
 if kind=='sqrt': return np.sqrt(x)
 if kind=='center': return x-x.mean(1,keepdims=True)
 if kind=='z':
  y=x-x.mean(1,keepdims=True); sd=x.std(1,keepdims=True)
  return np.divide(y,sd,out=np.zeros_like(y),where=sd>1e-12)
 if kind=='rank': return np.apply_along_axis(rankdata,1,x)
 if kind=='sign': return np.sign(x-x.mean(1,keepdims=True))
def best(cr,ca,scorekind='mse'):
 rv=np.array([cr[i,j] for i,j in pairs])
 av=np.stack([ca[perms[:,i],perms[:,j]] for i,j in pairs],axis=1)
 if scorekind=='mse': score=np.mean((av-rv)**2,axis=1)
 elif scorekind=='corr':
  # negative correlation
  rr=rv-rv.mean(); vv=av-av.mean(0); score=-(vv@rr)/(np.sqrt((vv*vv).sum(1)*(rr@rr)))
 inds=np.argsort(score)[:5]
 return [(float(score[k]),tuple(perms[k])) for k in inds]
for kind in ['raw','log','sqrt','center','z','sign']:
 print('processing',kind,flush=True)
 cr=np.corrcoef(calc(r,kind).T); ca=np.corrcoef(calc(a,kind).T)
 print(kind,'mse',best(cr,ca,'mse'),'corr',best(cr,ca,'corr'),flush=True)
PY"
