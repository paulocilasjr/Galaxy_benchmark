/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0,compression='gzip')
X=np.log1p(rna.values)
# center? compute zscore across samples per gene and sum maybe
# top genes by specificity: log2(fold) vs mean others with pseudo
for j in range(8):
    vals=rna.iloc[:,j].values
    others=rna.drop(columns=rna.columns[j]).mean(axis=1).values
    fc=np.log2((vals+0.1)/(others+0.1))
    score=fc * (vals>0)
    idx=np.argsort(score)[::-1][:30]
    print('\\nRNA col',j)
    print(', '.join(f'{rna.index[i]}:{rna.iloc[i,j]:.1f}' for i in idx))
PY"
