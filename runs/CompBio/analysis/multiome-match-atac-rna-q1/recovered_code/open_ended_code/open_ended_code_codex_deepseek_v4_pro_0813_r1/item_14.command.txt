import gzip,bisect,numpy as np,pandas as pd
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0,compression='gzip')
atac=pd.read_csv('multiome.match.atac.rna.q1.atac.tsv.gz',sep='\\t',index_col=0,compression='gzip')
