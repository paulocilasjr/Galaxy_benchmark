/bin/bash -lc "python3 - <<'PY'
import gzip, csv, pandas as pd, re
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0,compression='gzip')
symbols=set(rna.index)
chroms=[]
pos=[]
gene_info={}
with gzip.open('gencode.v47.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<9 or parts[2]"'!='"'gene': continue
        attrs=parts[8]
        m=re.search(r'gene_name \"(["'^"]+)"'"',attrs)
        if not m: continue
        g=m.group(1)
        if g in symbols and g not in gene_info:
            gene_info[g]=(parts[0],int(parts[3]),int(parts[4]),parts[6])
print('matched',len(gene_info),'of',len(symbols))
miss=symbols-set(gene_info)
print('miss sample', list(miss)[:20], 'miss count',len(miss))
# save gene_info tsv
with open('gene_info.tsv','w') as f:
    for g,(c,s,e,strand) in gene_info.items():
        f.write(f'{g}\\t{c}\\t{s}\\t{e}\\t{strand}\\n')
PY"
