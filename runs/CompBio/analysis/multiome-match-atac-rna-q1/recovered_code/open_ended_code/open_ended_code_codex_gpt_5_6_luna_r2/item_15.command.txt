import gzip,numpy as np,collections

def load(fn):
 with gzip.open(fn,'rt') as f:
  next(f); n=[]; v=[]
  for l in f:
   z=l.rstrip().split('\\t'); n.append(z[0]); v.append([float(x) for x in z[1:]])
 return n,np.array(v,np.float32)
rn,R=load('multiome.match.atac.rna.q1.rna.tsv.gz'); an,A=load('multiome.match.atac.rna.q1.atac.tsv.gz')
amap={(n.rsplit('_',2)[0],int(n.rsplit('_',2)[1])):i for i,n in enumerate(an)}
chroms={f'chr{i}' for i in range(1,23)}|{'chrX'}; t=collections.defaultdict(list)
