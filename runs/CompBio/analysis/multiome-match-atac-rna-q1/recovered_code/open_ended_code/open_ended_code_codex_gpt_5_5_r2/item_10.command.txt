/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, itertools, gzip
R=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0)
A=pd.read_csv('multiome.match.atac.rna.q1.atac.tsv.gz',sep='\\t',index_col=0)

def corrmat(X, log=True, topvar=None):
    Y=np.log1p(X.to_numpy(dtype=float)) if log else X.to_numpy(dtype=float)
    if topvar:
        idx=np.argsort(Y.var(axis=1))[-topvar:]
        Y=Y[idx]
    return np.corrcoef(Y.T)

def rankoff(C):
    # vectorize offdiag row? keep full minus diag
    return C

def score_perm(Cr,Ca,p):
    B=Ca[np.ix_(p,p)]
    # off diag pearson across matrix upper tri
    iu=np.triu_indices(8,1)
    x=Cr[iu]; y=B[iu]
    return np.corrcoef(x,y)[0,1], -np.sum((x-y)**2)
for log in [False, True]:
  for top in [None,5000,10000,20000]:
    Cr=corrmat(R,log=log,topvar=top)
    Ca=corrmat(A,log=log,topvar=top)
    best=[]
    for p in itertools.permutations(range(8)):
        best.append((*score_perm(Cr,Ca,p),p))
    best.sort(reverse=True,key=lambda x:x[0])
    print('\\nlog',log,'top',top,'best corr')
    for b in best[:10]: print(round(b[0],4),round(b[1],4),b[2])
    best.sort(reverse=True,key=lambda x:x[1])
    print('best sse')
    for b in best[:5]: print(round(b[0],4),round(b[1],4),b[2])
PY"
