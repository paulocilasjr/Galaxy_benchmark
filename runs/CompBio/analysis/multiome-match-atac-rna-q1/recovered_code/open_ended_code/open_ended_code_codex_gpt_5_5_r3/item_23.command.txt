/bin/bash -lc "python3 - <<'PY'
import gzip, re, itertools
import numpy as np
from collections import defaultdict

rna_path='multiome.match.atac.rna.q1.rna.tsv.gz'; atac_path='multiome.match.atac.rna.q1.atac.tsv.gz'; gtf_path='gencode.v44.annotation.gtf.gz'
genes=[]; vals=[]
with gzip.open(rna_path,'rt') as f:
    next(f)
    for line in f:
        p=line.rstrip('\\n').split('\\t'); genes.append(p[0]); vals.append([float(x) for x in p[1:]])
rna=np.array(vals,float); gene_to_idx={g:i for i,g in enumerate(genes)}
attr_re=re.compile(r'(\\S+) \"(["'^"]+)";'"')
ann=[]
with gzip.open(gtf_path,'rt') as f:
    for line in f:
        if not line or line[0]=='#': continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9 or p[2]"'!='"'gene': continue
        attrs=dict(attr_re.findall(p[8])); name=attrs.get('gene_name')
        if name in gene_to_idx and '_' not in p[0] and p[0]"'!='"'chrM':
            tss=int(p[3]) if p[6]=='+' else int(p[4])
            ann.append((gene_to_idx[name],p[0],tss))

def promoter_activity(window):
    bmap=defaultdict(list)
    for i,chrom,tss in ann:
        lo=max(1,tss-window); hi=tss+window
        for b in range(((lo-1)//10000)*10000, ((hi-1)//10000)*10000+1, 10000):
            bmap[f'{chrom}_{b}_{b+10000}'].append(i)
    A=np.zeros((len(genes),8),float)
    with gzip.open(atac_path,'rt') as f:
        next(f)
        for line in f:
            p=line.rstrip('\\n').split('\\t')
            inds=bmap.get(p[0])
            if inds:
                x=np.array([float(v) for v in p[1:]],float)
                for i in inds:
                    A[i]+=x
    return A

def pear(x,y):
    x=x-x.mean(); y=y-y.mean(); return float(x@y/np.sqrt((x*x).sum()*(y*y).sum()))
for w in [0,10000,20000,50000]:
    A=promoter_activity(w)
    mask=(A.sum(axis=1)>0) & (rna.sum(axis=1)>0)
    R=np.log1p(rna); T=np.log1p(A)
    # LFC RNA 0 vs1; ATAC candidate 5 vs1. Positive means RNA0->ATAC5/RNA1->ATAC1.
    rx=R[:,0]-R[:,1]
    ax=T[:,5]-T[:,1]
    ay=T[:,1]-T[:,5]
    base=mask & (np.abs(rx)>0.2)
    print('\\nw',w,'n',base.sum(),'lfc corr candidate',pear(rx[base],ax[base]),'swap',pear(rx[base],ay[base]))
    # top RNA markers in either direction
    for thresh in [0.2,0.5,1.0]:
        m=mask & (np.abs(rx)>thresh) & (A[:,[1,5]].sum(axis=1)>0)
        print(' thresh',thresh,'n',m.sum(),'corr',pear(rx[m],ax[m]),'mean concord sign',np.mean(np.sign(rx[m])==np.sign(ax[m])))
    # names of top markers with ATAC diffs
    order=np.argsort(-np.abs(rx[base]))[:15]
    print(' top genes gene rna0-rna1 atac5-atac1')
    for k in order[:10]:
        print(genes[k], round(rx[k],3), round(ax[k],3), 'vals rna', np.round(R[k,[0,1]],2), 'atac', np.round(T[k,[5,1]],2))
PY"
