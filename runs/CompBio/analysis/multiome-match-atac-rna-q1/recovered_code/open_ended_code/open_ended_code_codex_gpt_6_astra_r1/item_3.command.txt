/bin/bash -lc "cat > match.py <<'PY'
import pandas as pd
import numpy as np
from scipy.optimize import linear_sum_assignment
r=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0)
a=pd.read_csv('multiome.match.atac.rna.q1.atac.tsv.gz',sep='\\t',index_col=0)
g=pd.read_csv('hg38.ncbiRefSeq.txt.gz',sep='\\t',header=None)
g=g[g[12].isin(r.index)&g[2].isin(['chr'+str(i) for i in range(1,23)]+['chrX','chrY'])].copy()
g['tss']=np.where(g[3]=='+',g[4],g[5]-1)
g['bin']=g.tss//10000*10000
g=g.drop_duplicates([12,2,'bin'])
def std(x):
 x=x-x.mean(axis=1,keepdims=True)
 return x/np.maximum(np.linalg.norm(x,axis=1,keepdims=True),1e-12)
for radius in [0,10000,20000]:
 vals=[]
 names=[]
 for name,grp in g.groupby(12):
  keys=set()
  for _,row in grp.iterrows():
   for shift in range(-radius,radius+1,10000):
    start=row['bin']+shift
    key=f'{row[2]}_{start}_{start+10000}'
    if key in a.index: keys.add(key)
  if keys:
   vals.append(a.loc[list(keys)].mean(axis=0).values)
   names.append(name)
 A=np.array(vals)
 R=r.loc[names].values
 for minexpr in [1,10,50]:
  mask=(R.max(axis=1)>=minexpr)&(A.max(axis=1)>0)
  X=std(np.log1p(R[mask])); Y=std(np.log1p(A[mask]))
  scores=X.T@Y / len(X)
  ri,ai=linear_sum_assignment(-scores)
  print('radius',radius,'minexpr',minexpr,'n',len(X),flush=True)
  print(np.round(scores,3),flush=True)
  print('MATCH',ai,'score',scores[ri,ai].sum(),flush=True)
PY"
