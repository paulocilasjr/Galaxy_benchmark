/bin/bash -lc "cat > pairs.py <<'PY'
import numpy as np
from itertools import permutations,combinations
for mode in ['tss','body']:
 d=np.load(mode+'_mapped.npz'); R=d['R']; A=d['A']; X=np.log1p(R); Y=np.log1p(A)
 rp=list(combinations(range(8),2)); score=np.zeros((28,8,8))
 for q,(i,j) in enumerate(rp):
  dx=X[:,i]-X[:,j]
  keep=(R[:,[i,j]].max(axis=1)>10)&(abs(dx)>.5)
  for k,l in combinations(range(8),2):
   v=np.corrcoef(dx[keep],(Y[:,k]-Y[:,l])[keep])[0,1]
   score[q,k,l]=v;score[q,l,k]=-v
 ps=np.array(list(permutations(range(8))))
 vals=np.zeros(len(ps))
 for q,(i,j) in enumerate(rp): vals+=score[q,ps[:,i],ps[:,j]]
 print(mode)
 for z in np.argsort(-vals)[:5]: print(ps[z],vals[z])
 best=ps[np.argmax(vals)]
 for q,(i,j) in enumerate(rp):
  if i in [0,1,3] and j in [0,1,3]: print('specific pair',i,j,'atac',best[i],best[j],'correlation',score[q,best[i],best[j]])
PY"
