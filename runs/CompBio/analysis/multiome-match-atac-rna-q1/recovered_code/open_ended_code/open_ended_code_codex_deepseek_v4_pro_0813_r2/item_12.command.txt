/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, pandas as pd
# read RNA
with gzip.open('multiome.match.atac.rna.q1.rna.tsv.gz','rt') as f:
    header=f.readline().rstrip().split('\\t')[1:]
    genes=[]; data=[]
    for line in f:
        p=line.rstrip().split('\\t')
        genes.append(p[0]); data.append(list(map(float,p[1:9])))
X=pd.DataFrame(data,index=genes,columns=header)
print(X.shape)
# log? TPM, identify specificity ratio col / mean other
Xp=X+0.1
# log fold change vs mean
logX=np.log2(Xp)
mean_other = np.log2((Xp.sum(axis=1).values[:,None]-Xp.values)/(7)+0.1)
lf=logX.values-mean_other
# top by mean expression too
for j in range(8):
    scores=lf[:,j]*np.log1p(X.values[:,j]) # product
    idx=np.argsort(-scores)[:30]
    print('\\nCOL',j)
    for i in idx:
        print(genes[i], 'expr',X.values[i,j], 'lf',lf[i,j], 'score',scores[i])
PY"
