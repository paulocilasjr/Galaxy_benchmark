/bin/bash -lc "python3 - <<'PY'
import requests, json
symbols=['CD19','CD3D','CD14','MS4A1','NKG7','FCGR3A','CD8A','IL7R']
for ep in ['https://mygene.info/v3/query','https://mygene.info/v3/querymany']:
    try:
        r=requests.post(ep, json={'q':symbols,'scopes':'symbol','fields':'symbol,genomic_pos','species':'human'}, timeout=20)
        print(ep, r.status_code)
        print(r.text[:1500])
    except Exception as e:
        print(ep, e)
PY"
