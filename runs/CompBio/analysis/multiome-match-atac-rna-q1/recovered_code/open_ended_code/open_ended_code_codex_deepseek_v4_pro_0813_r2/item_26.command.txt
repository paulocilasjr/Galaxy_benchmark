/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
from scipy.stats import spearmanr
from scipy.optimize import linear_sum_assignment
# load act tss, R
genes=[]; rna=[]
with gzip.open('multiome.match.atac.rna.q1.rna.tsv.gz','rt') as f:
 h=f.readline().rstrip().split('\\t')[1:]
 for line in f:
  p=line.rstrip().split('\\t'); genes.append(p[0]); rna.append([float(x) for x in p[1:9]])
R=pd.DataFrame(rna,index=genes,columns=h)
act=pd.read_pickle('act_tss.pkl')
markers=['TAGLN','DCN','LGALS1','COL1A1','CXCL8','IL1B','CCL2','MMP1','TIMP1','KRT14','S100A2','LY6D','KLF4','POU5F1','SOX2','NANOG','LIN28A','SUCNR1','OGN','POSTN','CD70','KRT8','KRT18','CD36','NLRP7','ALPG','DPPA3','LEFTY2','NODAL','CRABP1','CNTNAP2','PTPRZ1','SIX3','WFDC2','NLGN4X','COL3A1','COL6A2','HBA1','HBA2','CD19','MS4A1','CD3D','NKG7','CD14','FCGR3A','CD8A','IL7R']
markers=[m for m in markers if m in act.index and m in R.index]
X=np.log1p(R.loc[markers].values)
Y=np.log1p(act.loc[markers].values)
C=np.empty((8,8))
for i in range(8):
 for j in range(8):
  C[i,j]=spearmanr(X[:,i],Y[:,j])[0]
print('markers',len(markers))
print(np.round(C,3))
print('hung',linear_sum_assignment(-C)[1], 'greedy',C.argmax(axis=1))
# top variable 1000 genes from RNA and act combined
common=act.index.intersection(R.index)
Xl=np.log1p(R.loc[common].values); Yl=np.log1p(act.loc[common].values)
keep=(R.loc[common].values.sum(1)>0.1)&(act.loc[common].values.sum(1)>0)
idx=np.argsort(-(np.var(Xl[keep],axis=1)+np.var(Yl[keep],axis=1)))[:1000]
X2=Xl[keep][idx]; Y2=Yl[keep][idx]
C2=np.empty((8,8))
for i in range(8):
 for j in range(8): C2[i,j]=spearmanr(X2[:,i],Y2[:,j])[0]
print('top1000 hung',linear_sum_assignment(-C2)[1], C2.argmax(axis=1))
PY"
