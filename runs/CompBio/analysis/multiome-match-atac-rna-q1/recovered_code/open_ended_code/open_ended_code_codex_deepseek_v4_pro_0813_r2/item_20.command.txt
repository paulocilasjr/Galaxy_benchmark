/bin/bash -lc "python3 - <<'PY'
import requests, os
url='https://ftp.ensembl.org/pub/current/gtf/homo_sapiens/Homo_sapiens.GRCh38.116.gtf.gz'
out='Homo_sapiens.GRCh38.116.gtf.gz'
if not os.path.exists(out):
    with requests.get(url, stream=True, timeout=120) as r:
        r.raise_for_status()
        with open(out,'wb') as f:
            for chunk in r.iter_content(chunk_size=1<<20):
                f.write(chunk)
    print('downloaded',os.path.getsize(out))
else:
    print('exists',os.path.getsize(out))
PY"
