/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np
with gzip.open('multiome.match.atac.rna.q1.rna.tsv.gz','rt') as f:
 f.readline(); rows=[]; names=[]
 for l in f:
  q=l.rstrip().split('\\t'); names.append(q[0]); rows.append([float(x) for x in q[1:]])
a=np.asarray(rows); names=np.asarray(names)
# log fold to avg others, ranking log2 ratio but require expression > thresholds
meanother=(a.sum(1,keepdims=True)-a)/7
for j in range(8):
 logfc=np.log2((a[:,j]+0.05)/(meanother[:,j]+0.05))
 for thr in [1,0.2,0.05]:
  mask=a[:,j]>thr
  idx=np.where(mask)[0][np.argsort(logfc[mask])[::-1][:30]]
  print('\\nCOL',j,'expr>',thr)
  print(' '.join(f'{names[i]}={a[i,j]:.3g},lfc={logfc[i]:.2f}' for i in idx))
PY"
