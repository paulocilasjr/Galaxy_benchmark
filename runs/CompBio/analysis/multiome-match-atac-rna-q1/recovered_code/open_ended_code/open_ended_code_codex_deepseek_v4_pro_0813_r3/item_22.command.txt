/bin/bash -lc "python - <<'PY'
import gzip,csv,numpy as np
# load rna matrix
syms=[]; R=[]
with gzip.open('multiome.match.atac.rna.q1.rna.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t'); next(r)
    for row in r:
        syms.append(row[0]); R.append([float(x) for x in row[1:9]])
R=np.array(R)
logR=np.log1p(R)
# For each col, top genes by specificity score mean_col / (mean_others+epsilon) * expr maybe
for i in range(8):
    mean_all=logR.mean(axis=1)
    mean_col=logR[:,i]
    mean_others=(logR.sum(axis=1)-mean_col)/7
    specificity=mean_col - mean_others # or ratio
    # require expressed in col > threshold
    mask=R[:,i]>0.5
    scores=specificity.copy(); scores[~mask]=-999
    top=np.argsort(-scores)[:30]
    print('\\nRNA col',i)
    for gi in top:
        print(f'{syms[gi]:12s} expr={R[gi,i]:8.2f} spec={specificity[gi]:5.2f} others={R[gi].sum()-R[gi,i]:8.2f}')
PY"
