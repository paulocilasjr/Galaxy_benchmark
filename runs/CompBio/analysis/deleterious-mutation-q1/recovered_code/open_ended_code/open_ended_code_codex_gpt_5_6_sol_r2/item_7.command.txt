/bin/bash -lc "bcftools mpileup -f chr9.fa -q 20 -Q 20 -a FORMAT/DP,FORMAT/AD -Ou q1.hg38.bam | bcftools call -mv -f GQ -Oz -o q1.hg38.vcf.gz && bcftools index -f q1.hg38.vcf.gz && bcftools stats q1.hg38.vcf.gz | sed -n '/"'^SN/p'"' && bcftools query -i 'TYPE=\"snp\" && GT=\"AA\" && QUAL>=30 && FORMAT/DP>=10' -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%GT\\t%DP\\t%AD\\t%GQ]\\n' q1.hg38.vcf.gz | sort -k2,2n > hom_snvs.tsv && wc -l hom_snvs.tsv && sed -n '1,120p' hom_snvs.tsv"
