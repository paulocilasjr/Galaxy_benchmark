/bin/bash -lc "curl -L --fail --retry 3 -o chr9.hg19.fa.gz https://hgdownload.soe.ucsc.edu/goldenPath/hg19/chromosomes/chr9.fa.gz
gzip -cd chr9.hg19.fa.gz > chr9.hg19.fa
bwa index chr9.hg19.fa
samtools faidx chr9.hg19.fa
bwa mem -t 4 chr9.hg19.fa deleterious.mutation.q1.R1.fq.gz deleterious.mutation.q1.R2.fq.gz | samtools sort -@ 4 -o aligned.hg19.bam
samtools index aligned.hg19.bam
bcftools mpileup -Ou -f chr9.hg19.fa -q 20 -Q 20 -a FORMAT/DP,FORMAT/AD aligned.hg19.bam | bcftools call -mv -Ou | bcftools norm -f chr9.hg19.fa -Oz -o calls.hg19.vcf.gz
bcftools index -t calls.hg19.vcf.gz
bcftools stats calls.hg19.vcf.gz | awk -F '\\t' '"'$1=="SN" || $1=="TSTV" {print}'"'
bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%GT\\t%DP\\t%AD]\\n' calls.hg19.vcf.gz | awk '"'$6=="1/1" && $7>=10 {print}'"' | sort -k5,5nr > hom_calls.hg19.tsv
wc -l hom_calls.hg19.tsv
head -n 20 hom_calls.hg19.tsv"
