/bin/bash -lc "bcftools mpileup -Ou -f chr9.hg38.fa -q 20 -Q 20 -a FORMAT/DP,FORMAT/AD aligned.hg38.bam | bcftools call -mv -Ou | bcftools norm -f chr9.hg38.fa -Oz -o calls.hg38.vcf.gz
bcftools index -t calls.hg38.vcf.gz
bcftools stats calls.hg38.vcf.gz | sed -n '1,80p'
bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%GT\\t%DP\\t%AD]\\n' calls.hg38.vcf.gz | awk '"'$6=="1/1" && $7>=10 {print}'"' | sort -k5,5nr > hom_calls.tsv
wc -l hom_calls.tsv
head -n 30 hom_calls.tsv"
