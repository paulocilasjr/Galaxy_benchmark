/bin/bash -lc "bwa mem -t 8 -R '@RG\\tID:q1\\tSM:q1\\tPL:ILLUMINA' refs/hg38.chr9.fa deleterious.mutation.q1.R1.fq.gz deleterious.mutation.q1.R2.fq.gz | samtools sort -@ 4 -o align/hg38.sorted.bam
samtools index align/hg38.sorted.bam
samtools flagstat align/hg38.sorted.bam > align/hg38.flagstat.txt
samtools stats align/hg38.sorted.bam > align/hg38.stats.txt
sed -n '1,20p' align/hg38.flagstat.txt
awk -F'\\t' '"'$1 ~ /''^(SN|IS)/ && ($2 ~ /raw total sequences|reads mapped|reads properly paired|insert size average|insert size standard deviation/) {print}'"' align/hg38.stats.txt"
