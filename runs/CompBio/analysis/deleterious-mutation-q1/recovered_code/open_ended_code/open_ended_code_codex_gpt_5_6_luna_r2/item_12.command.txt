/bin/bash -lc 'for asm in hg38 hg19; do
  bcftools mpileup -f refs/${asm}.chr9.fa -q 20 -Q 20 -d 10000 -a FORMAT/AD,FORMAT/DP,FORMAT/SP,INFO/DP,INFO/AD -Ou align/${asm}.sorted.bam '"\\
    | bcftools call -m -v -f GQ -Oz -o calls/"'${asm}.vcf.gz
  tabix -f -p vcf calls/${asm}.vcf.gz
  bcftools stats calls/${asm}.vcf.gz > calls/${asm}.vcf.stats.txt
  echo ${asm}
  bcftools query -f '"'%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%GT\\t%DP\\t%AD\\t%GQ\\n]' calls/"'${asm}.vcf.gz | sed -n '"'1,10p'
done"
