/bin/bash -lc "python3 - <<'PY'
import pysam
v=pysam.VariantFile('q1.csq.vcf')
rows=[]
for rec in v:
    bcsq=rec.info.get('BCSQ')
    if not bcsq: continue
    dp=rec.info.get('DP',0)
    ad=rec.samples['q1'].get('AD',())
    gt=rec.samples['q1'].get('GT',(None,None))
    gtstr='/'.join(map(str,gt)) if gt and None not in gt else '.'
    for item in bcsq:
        for entry in item.split(','):
            f=entry.split('|')
            cons=f[0].split('&')
            if 'stop_gained' in cons:
                rows.append((dp, rec.qual or 0, rec.pos, rec.ref, rec.alts[0] if rec.alts else '.', gtstr, list(ad), f[1], f[2], f[3], f[4] if len(f)>4 else '', f[5] if len(f)>5 else '', rec.contig))
print('num stop entries',len(rows))
for r in sorted(rows, reverse=True)[:120]:
    print('\\t'.join(map(str,r)))
PY"
