/bin/bash -lc 'tools/salmon-cli-x86_64-apple-darwin/salmon quant -i reference/salmon_gencode49 -l A -r deg.simple.q2.sampleA.fq.gz -p 4 --validateMappings --seqBias --gcBias -o quant/sampleA > quant/sampleA.log 2>&1 &
pid_a=$!
tools/salmon-cli-x86_64-apple-darwin/salmon quant -i reference/salmon_gencode49 -l A -r deg.simple.q2.sampleB.fq.gz -p 4 --validateMappings --seqBias --gcBias -o quant/sampleB > quant/sampleB.log 2>&1 &
pid_b=$!
wait "$pid_a"; status_a=$?

printf '"'sampleA_status=%s sampleB_status=%s\\n' \""'$status_a" "$status_b"
tail -15 quant/sampleA.log
tail -15 quant/sampleB.log'
