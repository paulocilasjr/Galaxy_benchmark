/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def load(path):
    d={}
    with open(path) as f:
        next(f)
        for line in f:
            tid,length,eff,c,tpm=line.rstrip('\\n').split('\\t')
            parts=tid.split('|')
            gene_id=parts[1]
            gene_name=parts[5]
            tx_name=parts[4]
            d[tid]=(gene_id,gene_name,tx_name,float(c),float(tpm))
    return d
A=load('quant/sampleA/abundance.tsv')
B=load('quant/sampleB/abundance.tsv')
by=defaultdict(list)
for tid,(gid,gname,tx,c,tpm) in A.items():
    by[gid,gname].append(tid)
rows=[]
for (gid,gname),tids in by.items():
    if len(tids)<2: continue
    ca=[A[t][3] for t in tids]
    cb=[B[t][3] for t in tids]
    ta=sum(ca); tb=sum(cb)
    if ta < 20 or tb < 20: continue
    pa=[x/ta for x in ca]
    pb=[x/tb for x in cb]
    l1=sum(abs(x-y) for x,y in zip(pa,pb))/2
    maxdiff=max(abs(x-y) for x,y in zip(pa,pb))
    # gene expression fold ratio by count normalized for library size
    rows.append((l1,maxdiff,min(ta,tb),ta,tb,gname,gid,len(tids)))
rows.sort(reverse=True)
print('top by isoform proportion L1')
for r in rows[:50]:
    print('\\t'.join(map(str,r)))
PY"
