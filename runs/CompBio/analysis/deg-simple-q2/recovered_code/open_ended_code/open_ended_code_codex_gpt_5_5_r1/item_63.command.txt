/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def load(path):
    d={}
    with open(path) as f:
        next(f)
        for line in f:
            tid,length,eff,c,tpm=line.rstrip('\\n').split('\\t')
            parts=tid.split('|')
            d[tid]=(parts[5],parts[4],float(c),float(tpm),int(length))
    return d
A=load('quant/sampleA/abundance.tsv'); B=load('quant/sampleB/abundance.tsv')
by=defaultdict(list)
for t,v in A.items(): by[v[0]].append(t)
for gene in ['DEFB110','IKZF2','HUWE1','FLNA']:
    tids=by[gene]
    ta=sum(A[t][2] for t in tids); tb=sum(B[t][2] for t in tids)
    print('\\n',gene,ta,tb,'L1',sum(abs(A[t][2]/ta-B[t][2]/tb) for t in tids)/2)
    for t in tids:
        if A[t][2]+B[t][2] > 0.1:
            print(A[t][1],t.split('|')[0],A[t][4],A[t][2],B[t][2],A[t][3],B[t][3])
PY"
