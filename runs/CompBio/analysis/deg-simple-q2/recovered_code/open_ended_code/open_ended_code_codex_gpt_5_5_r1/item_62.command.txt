/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def load(path):
    d={}
    with open(path) as f:
        next(f)
        for line in f:
            name,l,el,tpm,num=line.rstrip('\\n').split('\\t')
            parts=name.split('|')
            d[name]={'gid':parts[1],'gname':parts[5],'tx':parts[4],'cnt':float(num),'tpm':float(tpm),'len':int(float(l))}
    return d
A=load('quant/salmon_sampleA/quant.sf'); B=load('quant/salmon_sampleB/quant.sf')
by=defaultdict(list)
for tid,v in A.items(): by[v['gname']].append(tid)
for gene in ['IKZF2','DEFB110','HUWE1','FLNA','RPL15','FNBP1','EIF4G2','STAT1','RPL4']:
    tids=by[gene]
    ta=sum(A[t]['cnt'] for t in tids); tb=sum(B[t]['cnt'] for t in tids)
    print('\\n',gene,'totalA',ta,'totalB',tb)
    rows=[]
    for t in tids:
        ca=A[t]['cnt']; cb=B[t]['cnt']
        if ca+cb>0.1:
            rows.append((abs(ca/ta-cb/tb),ca,cb,ca/ta if ta else 0,cb/tb if tb else 0,A[t]['tx'],t.split('|')[0],A[t]['len'],A[t]['tpm'],B[t]['tpm']))
    rows.sort(reverse=True)
    for r in rows[:12]: print('  ', '\\t'.join(map(lambda x: f'{x:.4g}' if isinstance(x,float) else str(x), r)))
PY"
