from collections import defaultdict
info={}
with open('gencode.v47.transcripts.fa') as f:
    for line in f:
        if line.startswith('>'):
            p=line[1:].rstrip().split('|'); info[p[0]]=p

def load(s):
 c=defaultdict(int)
 for line in open(f'sample{s}.transcript_counts.tsv'):
  line=line.rstrip()
  if line:
   p=line.rsplit(' ',1); c[p[0].split('|')[0]]=int(p[1])
 return c
A=load('A');B=load('B')
geneA=defaultdict(int);geneB=defaultdict(int)
for t,c in A.items():
 if t in info: geneA[info[t][1]]+=c
for t,c in B.items():
 if t in info: geneB[info[t][1]]+=c
rows=[]
for gid in set(geneA)|set(geneB):
 a=geneA[gid];b=geneB[gid]
 if a>=20 and b>=20:
  tsA={t for t in A if t in info and info[t][1]==gid}
  tsB={t for t in B if t in info and info[t][1]==gid}
  overlap=tsA & tsB
  # TV
  allt=tsA|tsB
  tv=sum(abs(A.get(t,0)/a-B.get(t,0)/b) for t in allt)/2
  if tv==1.0:
   rows.append((a,b,gid,info[next(iter(tsA|tsB))][5],len(tsA),len(tsB),sorted(tsA),sorted(tsB)))
print('TV=1 genes with counts>=20',len(rows))
for r in sorted(rows,reverse=True): print(r[:6],r[6:])
PY"
