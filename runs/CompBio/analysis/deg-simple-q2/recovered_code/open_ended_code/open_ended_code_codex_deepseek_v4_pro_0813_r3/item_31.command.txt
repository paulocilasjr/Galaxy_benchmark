/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
import math
info={}
with open('gencode.v47.transcripts.fa') as f:
    for line in f:
        if line.startswith('>'):
            parts=line[1:].rstrip().split('|')
            info[parts[0]]=parts

def load(sample):
    c=defaultdict(int)
    with open(f'sample{sample}.transcript_counts.tsv') as f:
        for line in f:
            line=line.rstrip()
            if not line: continue
            p=line.rsplit(' ',1)
            c[p[0].split('|')[0]]=int(p[1])
    return c
A=load('A');B=load('B')
# aggregate gene count, transcript matrix, types
genes=defaultdict(lambda: {'counts':defaultdict(int),'symbol':'','type':''})
for tid,c in A.items():
    if tid not in info: continue
    gid=info[tid][1]; sym=info[tid][5] if len(info[tid])>5 else ''; typ=info[tid][7] if len(info[tid])>7 else ''
    d=genes[gid]; d['counts'][tid]+=c; d['symbol']=sym; d['type']=typ
for tid,c in B.items():
    if tid not in info: continue
    gid=info[tid][1]; sym=info[tid][5] if len(info[tid])>5 else ''; typ=info[tid][7] if len(info[tid])>7 else ''
    d=genes.setdefault(gid,{'counts':defaultdict(int),'symbol':sym,'type':typ}); d['counts'][tid]+=c

rows=[]
for gid,d in genes.items():
    A_tot=sum(A.get(t,0) for t in d['counts'])
    B_tot=sum(B.get(t,0) for t in d['counts'])
    if A_tot<50 or B_tot<50: continue
    if len(d['counts'])<2: continue
    # only protein_coding? maybe all but separate
    # total variation distance between transcript distributions
    allt=sorted(set(d['counts']))
    tv=sum(abs(A.get(t,0)/A_tot-B.get(t,0)/B_tot) for t in allt)/2
    # max single transcript diff
    maxdiff=max(abs(A.get(t,0)/A_tot-B.get(t,0)/B_tot) for t in allt)
    rows.append((tv,maxdiff,A_tot,B_tot,gid,d['symbol'],d['type']))
print('Top by TV')
for r in sorted(rows,reverse=True)[:50]:
    tv,md,a,b,gid,sym,typ=r
    print(f'{sym:12} {gid:20} A={a:5} B={b:5} TV={tv:.3f} maxdiff={md:.3f} {typ}')
PY"
