/bin/bash -lc "samtools view sampleA.bam | awk '{c["'$3]++} END{for(t in c) print t,c[t]}'"' > sampleA.transcript_counts.tsv

wc -l sampleA.transcript_counts.tsv sampleB.transcript_counts.tsv
head sampleA.transcript_counts.tsv; head sampleB.transcript_counts.tsv"
