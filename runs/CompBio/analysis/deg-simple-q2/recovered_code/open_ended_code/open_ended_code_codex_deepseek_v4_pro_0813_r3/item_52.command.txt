/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
info={}
for line in open('gencode.v47.transcripts.fa'):
 if line.startswith('>'):
  p=line[1:].rstrip().split('|'); info[p[0]]=p

def load(s):
 c=defaultdict(int)
 for line in open(f'sample{s}.transcript_counts.tsv'):
  p=line.rstrip().rsplit(' ',1); c[p[0].split('|')[0]]=int(p[1])
 return c
A=load('A');B=load('B')
gA=defaultdict(lambda: defaultdict(int)); gB=defaultdict(lambda: defaultdict(int))
for t,c in A.items():
 if t in info: gA[info[t][1]][t]+=c
for t,c in B.items():
 if t in info: gB[info[t][1]][t]+=c
rows=[]
for gid in set(gA)|set(gB):
 a=gA[gid]; b=gB[gid]; ta=sum(a.values()); tb=sum(b.values())
 if ta<20 or tb<20: continue
 topA=max(a.items(),key=lambda kv:kv[1]) if a else (None,0)
 topB=max(b.items(),key=lambda kv:kv[1]) if b else (None,0)
 if topA[0] is None or topB[0] is None or topA[0]==topB[0]: continue
 fracA=topA[1]/ta; fracB=topB[1]/tb
 score=min(fracA,fracB)*(ta+tb)
 sym=info[topA[0]][5] if topA[0] in info else ''
 rows.append((fracA,fracB,ta,tb,topA[1],topB[1],gid,sym,topA[0],topB[0]))
rows.sort(key=lambda x:(min(x[0],x[1]),x[2]+x[3]), reverse=True)
print('Genes where top transcript differs and both top fractions high')
for r in rows[:50]:
 print(f'{r[7]:12} {r[6]:20} fracA={r[0]:.3f} fracB={r[1]:.3f} A={r[2]}({r[4]}) B={r[3]}({r[5]})')
PY"
