/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
info={}
with open('gencode.v47.transcripts.fa') as f:
    for line in f:
        if line.startswith('>'):
            p=line[1:].rstrip().split('|'); info[p[0]]=p

def load(s):
 c=defaultdict(int)
 for line in open(f'sample{s}.transcript_counts.tsv'):
  line=line.rstrip()
  if line:
   p=line.rsplit(' ',1); c[p[0].split('|')[0]]=int(p[1])
 return c
A=load('A');B=load('B')
bygene=defaultdict(lambda: [defaultdict(int),defaultdict(int)])
for t,c in A.items():
 if t in info: bygene[info[t][1]][0][t]+=c
for t,c in B.items():
 if t in info: bygene[info[t][1]][1][t]+=c
rows=[]
for gid,(ca,cb) in bygene.items():
 a=sum(ca.values());b=sum(cb.values())
 if a>=20 and b>=20:
  allt=set(ca)|set(cb)
  tv=sum(abs(ca[t]/a-cb[t]/b) for t in allt)/2
  if tv==1.0:
   sym=info[next(iter(allt))][5]
   rows.append((a,b,gid,sym,len(ca),len(cb),sorted(ca),sorted(cb)))
print('TV=1 genes with counts>=20',len(rows))
for r in sorted(rows,reverse=True):
 print(r[:6])
PY"
