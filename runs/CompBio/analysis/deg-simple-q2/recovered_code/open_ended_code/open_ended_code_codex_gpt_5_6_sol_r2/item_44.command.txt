/bin/bash -lc "python - <<'PY'
import gzip
want={'ENST00000393660.2','ENST00000371148.3'}
seq={}; hdr={}; cur=None
with gzip.open('reference/gencode.v49.transcripts.fa.gz','rt') as f:
 for line in f:
  if line.startswith('>'):
   cur=line[1:].strip().split('|')[0]
   if cur in want:
    hdr[cur]=line[1:].strip(); seq[cur]=[]
  elif cur in want:
   seq[cur].append(line.strip())
for tx in sorted(want):
 s=''.join(seq[tx]); fields=hdr[tx].split('|')
 print(tx, fields[1], fields[5], fields[7], len(s), s[:30], s[-30:])
print('identical', ''.join(seq[sorted(want)[0]]) == ''.join(seq[sorted(want)[1]]))
PY
ps aux | grep -E '[c]url.*gencode' || true
ls -lh reference/gencode.v49.annotation.gtf.gz 2>/dev/null || true"
