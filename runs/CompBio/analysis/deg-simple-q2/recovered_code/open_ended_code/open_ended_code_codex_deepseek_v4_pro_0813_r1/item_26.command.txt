/bin/bash -lc "python3 - <<'PY'
import gzip, re, csv, collections, math
hdr={}
with gzip.open('ref/Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as f:
    for line in f:
        if line.startswith('>'):
            tid=line.split()[0][1:]
            gs=re.search(r'gene_symbol:(\\S+)', line)
            gene=re.search(r'gene:(\\S+)', line)
            hdr[tid]=(gs.group(1) if gs else '', gene.group(1) if gene else '')
quants={}
for s in ['sampleA','sampleB']:
    with open(f'quant/{s}/quant.sf') as f:
        quants[s]={row['Name']: float(row['NumReads']) for row in csv.DictReader(f, delimiter='\\t')}
tids=set(hdr)&(set(quants['sampleA'])|set(quants['sampleB']))
data=[]
for t in tids:
    a=quants['sampleA'].get(t,0.0); b=quants['sampleB'].get(t,0.0)
    if a+b>=1:
        data.append((t,hdr[t][0],hdr[t][1],a,b))
genes=collections.defaultdict(list)
for t,gs,g,a,b in data:
    genes[g].append((gs,t,a,b))
print('genes with >=1 read transcript',len(genes))
from scipy.stats import chi2
results=[]
for g,lst in genes.items():
    rows=[(gs,t,a,b) for gs,t,a,b in lst if a+b>=1]
    # aggregate by transcript if duplicates names (unlikely)
    table=[]
    for gs,t,a,b in rows:
        table.append([int(round(a)),int(round(b))])
    # merge all rows by rounding? leave
    table=[r for r in table if sum(r)>0]
    if len(table)<2: continue
    totA=sum(r[0] for r in table); totB=sum(r[1] for r in table)
    if totA<1 or totB<1 or totA+totB<10: continue
    colsum=[totA,totB]; n=totA+totB
    G=0
    for r in table:
        rowsum=r[0]+r[1]
        if rowsum==0: continue
        for c in [0,1]:
            obs=r[c]; exp=rowsum*colsum[c]/n
            if obs>0 and exp>0:
                G += obs*math.log(obs/exp)
    G*=2
    df=len(table)-1
    p=chi2.sf(G,df) if df>0 and math.isfinite(G) else 1
    propA=[r[0]/totA for r in table]; propB=[r[1]/totB for r in table]
    l1=sum(abs(pa-pb) for pa,pb in zip(propA,propB))
    results.append((G,p,l1,totA,totB,g,rows))
results.sort(key=lambda x:-x[0])
print('top 50 G-test')
for G,p,l1,totA,totB,g,rows in results[:50]:
    print(f'G={G:.1f} p={p:.3g} L1={l1:.3f} counts={totA:.0f}/{totB:.0f} {rows[0][0]} {g}')
print('\\nTop 50 by L1 among total>=50 both')
res2=sorted([r for r in results if r[3]>=50 and r[4]>=50], key=lambda x:-x[2])[:50]
for G,p,l1,totA,totB,g,rows in res2:
    print(f'\\nG={G:.1f} p={p:.3g} L1={l1:.3f} counts={totA:.0f}/{totB:.0f} {rows[0][0]} {g}')
    diffs=[]
    for gs,t,a,b in rows:
        if totA and totB:
            diffs.append((abs(a/totA-b/totB), gs,t,a,b))
    for d,gs,t,a,b in sorted(diffs,reverse=True)[:6]:
        print(f'   d={d:.3f} {gs} {t} {a:.1f} -> {b:.1f}')
PY"
