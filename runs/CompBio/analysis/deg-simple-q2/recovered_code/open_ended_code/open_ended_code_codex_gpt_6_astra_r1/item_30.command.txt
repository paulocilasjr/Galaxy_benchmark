/bin/bash -lc "python3 - <<'PY'
import collections
for sample in 'AB':
 rs=collections.defaultdict(list)
 for line in open('results/DEFB110.'+sample+'.paf'):
  p=line.split('\\t');tags=dict(x.strip().split(':',2)[::2] for x in p[12:]);rs[p[0]].append((int(tags['AS']),p[5].split('|')[0],int(p[9]),int(p[10]),int(p[7]),int(p[8])))
 c=collections.Counter(); spans=collections.defaultdict(list)
 for read,hits in rs.items():
  best=max(h[0] for h in hits); top=[h for h in hits if h[0]==best]
  c[tuple(sorted(h[1] for h in top))]+=1
  for h in top:spans[h[1]].append((h[4],h[5]))
 print(sample,len(rs),'best alignments',dict(c))
 for t,v in spans.items():print(t,'target range',min(x[0] for x in v),max(x[1] for x in v))
PY"
