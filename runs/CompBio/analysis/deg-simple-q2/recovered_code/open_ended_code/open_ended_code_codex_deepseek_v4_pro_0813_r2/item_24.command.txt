/bin/bash -lc "python3 - <<'PY'
import pysam, collections
for bam in ['A.bam','B.bam']:
    f=pysam.AlignmentFile(bam,'rb')
    flags=collections.Counter(); mapped=0; unmapped=0; total=0
    for r in f.fetch(until_eof=True):
        total+=1
        if r.is_unmapped: unmapped+=1
        else: mapped+=1
        flags[(r.is_secondary,r.is_supplementary,r.is_qcfail,r.is_duplicate)]+=1
    print(bam, total, mapped, unmapped, flags)
PY"
