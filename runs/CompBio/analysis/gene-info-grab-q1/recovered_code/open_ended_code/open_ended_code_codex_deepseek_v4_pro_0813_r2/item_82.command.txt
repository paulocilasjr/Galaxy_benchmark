/bin/bash -lc "python3 - <<'PY'
import json
s=open('[REDACTED LOCAL PATH]').read().splitlines()
for line in s:
    if line.startswith('data: '):
        data=line[6:]
        try:
            obj=json.loads(data)
        except: continue
        if isinstance(obj,list):
            for m in obj:
                if m.get('type')=='content':
                    print('REPO',m.get('repository'),'PATH',m.get('path'))
                    for lm in m.get('lineMatches',[]):
                        print(' ',lm.get('line'))
                    print('---')
        elif isinstance(obj,dict) and obj.get('type')=='content':
            print('REPO',obj.get('repository'),'PATH',obj.get('path'))
            for lm in obj.get('lineMatches',[]):
                print(' ',lm.get('line'))
            print('---')
PY"
