/bin/bash -lc "curl -s 'https://sourcegraph.com/.api/search/stream?q=context:global+fork:yes+archived:yes+%22nearest+protein-coding+gene+upstream+of%22&v=V3' -o [REDACTED LOCAL PATH] && grep -n 'event: matches' [REDACTED LOCAL PATH] | head && python3 - <<'PY'
import json
s=open('[REDACTED LOCAL PATH]').read().splitlines()
for line in s:
 if line.startswith('data: '):
  try: obj=json.loads(line[6:])
  except: continue
  if isinstance(obj,list):
   for m in obj:
    if m.get('type')=='content':
     print('REPO',m.get('repository'),'PATH',m.get('path'))
     for lm in m.get('lineMatches',[]):
      print(' ',lm.get('line')[:500])
  elif isinstance(obj,dict) and obj.get('type')=='content':
   print('REPO',obj.get('repository'),'PATH',obj.get('path'))
   for lm in obj.get('lineMatches',[]):
    print(' ',lm.get('line')[:500])
PY"
