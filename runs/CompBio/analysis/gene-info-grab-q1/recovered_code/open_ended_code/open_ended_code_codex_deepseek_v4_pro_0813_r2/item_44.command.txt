/bin/bash -lc "curl -sL 'https://pubchem.ncbi.nlm.nih.gov/gene/15110' -o [REDACTED LOCAL PATH] && python3 - <<'PY'
import re,html
s=open('[REDACTED LOCAL PATH]').read()
for term in ['Sap30l','Larp1','upstream','downstream','Nearest','neighbor']:
 print(term, s.count(term))
# find context around gene names
for term in ['Sap30l','Larp1']:
 for m in re.finditer(term, s):
  seg=html.unescape(re.sub('<["'^>]+>'"',' ',s[max(0,m.start()-500):m.start()+1000]))
  seg=re.sub(r'\\s+',' ',seg)
  print(term, seg[:1000])
PY"
