/bin/bash -lc "curl -s 'https://rest.ensembl.org/overlap/region/mus_musculus/11:57600000-57900000?feature=gene;content-type=application/json' -o scratch/chr11_hand1_region_genes.json && node - <<'NODE'
const fs=require('fs');
const genes=JSON.parse(fs.readFileSync('scratch/chr11_hand1_region_genes.json','utf8'));
const pcs=genes.filter(g=>g.biotype==='protein_coding').sort((a,b)=>a.start-b.start);
for (const g of pcs) console.log("'`${g.seq_region_name}:${g.start}-${g.end} strand=${g.strand} ${g.id} ${g.external_name||g.description||'"''}"'`);
NODE'
