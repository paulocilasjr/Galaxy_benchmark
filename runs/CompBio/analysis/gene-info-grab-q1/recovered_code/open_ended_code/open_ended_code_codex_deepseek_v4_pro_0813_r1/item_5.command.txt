/bin/bash -c "python3 - <<'PY'
import requests
base='https://rest.ensembl.org'
for sym in ['Larp1','Hand1','Sap30l']:
    r=requests.get(base+'/lookup/symbol/mus_musculus/'+sym, headers={'Content-Type':'application/json'}, timeout=20)
    print(sym, r.json().get('id'), r.json().get('biotype'), r.json().get('strand'), r.json().get('start'), r.json().get('end'))
PY"
