/bin/bash -c "python3 - <<'PY'
import requests
base='https://rest.ensembl.org'
# Hand1 known
hand1={'id':'ENSMUSG00000037335','start':57719531,'end':57723644,'strand':-1}
r=requests.get(f'{base}/overlap/region/mus_musculus/11:57000000-58000000?feature=gene;content-type=application/json', headers={'Content-Type':'application/json'}, timeout=20)
genes=[g for g in r.json() if g.get('biotype')=='protein_coding' and g['id']"'!=hand1['"'id']]
# upstream relative to strand: 5' side coordinate = end for - strand, start for + strand
for g in genes:
    g['fiveprime']=g['end'] if g['strand']==-1 else g['start']
    if hand1['strand']==-1:
        # Hand1 5' is end=57723644; upstream genes have 3'? Actually distance from hand1 end to gene start (closest transcribed base)
        if g['start']>hand1['end']:
            g['dist']=g['start']-hand1['end']
            g['dir']='upstream'
        elif g['end']<hand1['start']:
            g['dist']=hand1['start']-g['end']
            g['dir']='downstream'
        else:
            g['dist']=0; g['dir']='overlap'
print('Candidate nearest protein-coding genes:')
for g in sorted(genes,key=lambda x:abs(x['dist'] if 'dist' in x else 1e9)):
    if 'dist' in g:
        print(g['dir'], g['dist'], g['id'], g['external_name'], g['strand'], g['start'], g['end'])
PY"
