/bin/bash -lc "python3 - <<'PY'
import gzip
for f in ['encode.atac.pipeline.q1.R1.fq.gz','encode.atac.pipeline.q1.R2.fq.gz']:
    n=0
    lengths=[]
    with gzip.open(f,'rt') as fh:
        for i,line in enumerate(fh):
            if i%4==1:
                n+=1; lengths.append(len(line.strip()))
                if n>=10000: break
    print(f, 'first_reads', n, 'min_len', min(lengths), 'max_len', max(lengths), 'example_len', lengths[0])
PY"
