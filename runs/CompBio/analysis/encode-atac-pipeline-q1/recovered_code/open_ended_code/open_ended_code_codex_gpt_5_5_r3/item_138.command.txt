/bin/bash -lc 'set -euo pipefail
ROOT="$PWD"
export PATH="$ROOT/envs/atac/bin:$ROOT/atac-seq-pipeline/src:$PATH"
export PYTHONPATH="$ROOT/atac-seq-pipeline/src"
CHRSZ="$ROOT/ref/GRCh38_EBV.chrom.sizes.tsv"
python3 "$ROOT/atac-seq-pipeline/src/encode_task_reproducibility.py" --peaks-pr "$ROOT/run/idr/rep1-pr1_vs_rep1-pr2.idr0.05.bfilt.narrowPeak.gz" --prefix idr --peak-type narrowPeak --chrsz "$CHRSZ" --out-dir "$ROOT/run/repro"
python3 - <<'"'PY'
import json, os
root = os.getcwd()
repro = os.path.join(root, 'run', 'repro', 'idr.reproducibility.qc')
with open(repro) as fh:
    header = fh.readline().rstrip('\\n').split('\\t')
    values = fh.readline().rstrip('\\n').split('\\t')
record = dict(zip(header, values))
qc = {'reproducibility': {'idr': {k: int(v) if k.startswith('N') else float(v) if k.endswith('_ratio') else v for k, v in record.items()}}}
with open(os.path.join(root, 'run', 'qc', 'qc.json'), 'w') as out:
    json.dump(qc, out, indent=4)
print(record['N_opt'])
PY"
