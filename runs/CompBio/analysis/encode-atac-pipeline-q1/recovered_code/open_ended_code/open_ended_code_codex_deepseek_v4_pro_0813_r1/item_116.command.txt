/bin/bash -lc 'QC=$(find atac_out_direct -name qc.json | head -1); echo $QC; python3 - <<'"'PY'
import json, glob
p=glob.glob('atac_out_direct/**/qc.json', recursive=True)[0]
print(p)
d=json.load(open(p))
# Print relevant keys
for k,v in d.items():
 if 'idr' in k.lower() or 'repro' in k.lower() or 'n_opt' in k.lower() or 'num_peak' in k.lower():
  print(k, v)
print('--- keys with N_opt')
print(json.dumps({k:v for k,v in d.items() if 'N_opt' in k or 'n_opt' in k}, indent=2))
print('--- full top keys', list(d.keys()))
PY"
