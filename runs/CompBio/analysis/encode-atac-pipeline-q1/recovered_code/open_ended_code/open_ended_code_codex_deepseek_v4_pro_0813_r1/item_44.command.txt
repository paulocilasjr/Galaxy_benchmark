version 1.0
workflow test2 { 
  input { String msg }

  input { String msg }

    set -e

    echo "java=$(java -version 2>&1 | head -1)" >> out.txt

  }
  output { File out = '"'out.txt' }
  runtime { memory: '2 GB' }
}
EOF
