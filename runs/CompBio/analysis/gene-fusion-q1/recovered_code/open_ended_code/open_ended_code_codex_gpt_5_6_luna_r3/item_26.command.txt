/bin/bash -lc "python3 - <<'PY'
import gzip
needle='GCCAAGGCTGTGGGGCAGCGCAGATGCAGCCCATGCACCGCT'
for fn in ['gene.fusion.q1.R1.fq.gz','gene.fusion.q1.R2.fq.gz']:
    hits=[]
    with gzip.open(fn,'rt') as f:
        while True:
            h=f.readline().rstrip()
            if not h: break
            s=f.readline().rstrip(); f.readline(); f.readline()
            if needle in s: hits.append((h,s))
    print(fn, len(hits))
    for h,s in hits[:20]: print(h, s)
PY"
