/bin/bash -c "python3 - <<'PY'
import pysam, json, collections, re
sym=json.load(open('sym.json'))
counts=collections.Counter(); pairs=collections.Counter(); total=0; mapped=0; supp=0
read_genes={}
infile=pysam.AlignmentFile('aln.sam','r')
for rec in infile:
    if rec.is_unmapped: continue
    mapped+=1
    gene=sym.get(rec.reference_name, rec.reference_name)
    if not rec.is_secondary and not rec.is_supplementary:
        counts[gene]+=1
        total+=1
    if rec.is_supplementary:
        supp+=1
print('primary alignments', total, 'all alignments', mapped, 'supp', supp)
for g,c in counts.most_common(30):
    print(f'{g}\\t{c}')
PY"
