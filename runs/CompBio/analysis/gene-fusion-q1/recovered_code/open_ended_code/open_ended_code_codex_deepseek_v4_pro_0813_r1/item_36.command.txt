/bin/bash -c "python3 - <<'PY'
import pysam, json, collections
sym=json.load(open('sym.json'))
meta=json.load(open('meta.json'))
def sym_of(ref): return meta.get(ref,(None,None,None))[0] or ref
def biotype_of(ref): return meta.get(ref,(None,None,None))[1]
def chrom_of(ref): return meta.get(ref,(None,None,None))[2]
infile=pysam.AlignmentFile('aln.sam','r')
sa=collections.Counter(); mate=collections.Counter(); mapq=collections.Counter()
for rec in infile:
    if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
    rg=sym_of(rec.reference_name)
    if rec.has_tag('SA'):
        for e in rec.get_tag('SA').split(';'):
            if not e: continue
            p=e.split(',')
            rg2=sym_of(p[0])
            if rg2"'!=rg:
                pair=tuple(sorted([rg,rg2]))
                sa[pair]+=1
                if rec.mapping_quality>=20: mapq[pair]+=1
    nr=rec.next_reference_name
    if nr and nr!='"'=' and nr"'!=rec.reference_name:
        ng=sym_of(nr)
        if ng!=rg:
            pair=tuple(sorted([rg,ng]))
            mate[pair]+=1
print('"'Candidates with mate>=2 or SA>=2, excluding mitochondrial/IG/TR/HLA? showing all:')
for pair,c in mate.most_common():
    if c>=2 or sa.get(pair,0)>=2:
        print(f\"{pair[0]}-{pair[1]}\\tmate={c}\\tSA={sa.get(pair,0)}\\thqSA={mapq.get(pair,0)}\")
PY"
