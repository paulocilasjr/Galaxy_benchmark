/bin/bash -c "python3 - <<'PY'
import gzip, re
f='Homo_sapiens.GRCh38.cdna.all.fa.gz'
sym={}
with gzip.open(f,'rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            m=re.search(r'gene_symbol:(["'^'"\\s]+)', line)
            if m:
                sym[tid]=m.group(1)
print(len(sym))
print(list(sym.items())[:5])
import json
json.dump(sym, open('sym.json','w'))
PY"
