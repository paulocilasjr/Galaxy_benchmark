/bin/bash -lc "python3 - <<'PY'
import pysam, re, gzip
sym={}
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            m=re.search(r'gene_symbol:(\\S+)',line)
            if m: sym[tid]=m.group(1)
sam=pysam.AlignmentFile('contigs_transcriptome.sam','r')
for a in sam.fetch(until_eof=True):
    if a.query_name=='NODE_11_length_2231_cov_18.585938' and not a.is_secondary:
        print('flag',a.flag,'ref',a.reference_name,'gene',sym.get(a.reference_name),'rstart',a.reference_start,'rend',a.reference_end,'qstart',a.query_alignment_start,'qend',a.query_alignment_end,'alen',a.query_alignment_length,'mapq',a.mapping_quality,'cigar',a.cigarstring,'strand', '-' if a.is_reverse else '+')
        print('seq?')
PY"
