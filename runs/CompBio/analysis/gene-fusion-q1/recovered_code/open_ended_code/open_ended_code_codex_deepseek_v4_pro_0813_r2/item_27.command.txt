/bin/bash -lc "python3 - <<'PY'
import pysam, re, gzip, collections
sym={}
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            m=re.search(r'gene_symbol:(\\S+)',line)
            if m: sym[tid]=m.group(1)
sam=pysam.AlignmentFile('raw_transcriptome.sam','r')
pair_counts=collections.Counter(); tri_counts=collections.Counter()
for a in sam.fetch(until_eof=True):
    if a.is_unmapped or a.is_secondary: continue
    genes=set([sym.get(a.reference_name,a.reference_name)])
    if a.has_tag('SA'):
        for item in a.get_tag('SA').split(';'):
            if not item: continue
            r=item.split(',')[0]
            genes.add(sym.get(r,r))
    g=sorted(genes)
    if len(g)==2:
        pair_counts[tuple(g)]+=1
    elif len(g)>2:
        tri_counts[tuple(g)]+=1
print('PAIRS')
for p,c in pair_counts.most_common(100):
    print(c,p)
print('TRIPLES')
for p,c in tri_counts.most_common(20):
    print(c,p)
PY"
