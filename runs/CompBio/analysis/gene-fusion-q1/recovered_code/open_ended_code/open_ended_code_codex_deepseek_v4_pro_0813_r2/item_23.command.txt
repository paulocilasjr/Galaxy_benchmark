/bin/bash -lc "python3 - <<'PY'
import pysam, re, gzip, collections
sym={}
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            m=re.search(r'gene_symbol:(\\S+)',line)
            if m: sym[tid]=m.group(1)
sam=pysam.AlignmentFile('contigs_transcriptome.sam','r')
perq=collections.defaultdict(dict)
for a in sam.fetch(until_eof=True):
    if a.is_unmapped: continue
    if a.is_secondary: continue
    q=a.query_name
    g=sym.get(a.reference_name, a.reference_name)
    alen=a.query_alignment_length
    if g not in perq[q] or perq[q][g]<alen: perq[q][g]=alen
for q,genes in perq.items():
    if len(genes)>=2:
        arr=sorted(genes.items(), key=lambda x:-x[1])
        if arr[0][1]>=100 and arr[1][1]>=100:
            print(q, arr[:8])
PY"
