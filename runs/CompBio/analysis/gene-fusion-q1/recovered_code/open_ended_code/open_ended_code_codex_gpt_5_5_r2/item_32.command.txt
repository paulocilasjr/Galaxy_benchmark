/bin/bash -lc "python - <<'PY'
from collections import defaultdict
# t2g
t2g={}
for line in open('refs/ensembl_t2g.tsv'):
    tid,g=line.rstrip('\\n').split('\\t')[:2]; t2g[tid]=g
hits=defaultdict(list)
for line in open('work/reads_vs_sox2_gapdh.tsv'):
    f=line.rstrip('\\n').split('\\t')
    q,s=f[0],f[1]
    g=t2g.get(s,'?')
    p=float(f[2]); length=int(f[3]); mm=int(f[4]); go=int(f[5])
    qs,qe=int(f[6]),int(f[7]); ss,se=int(f[8]),int(f[9]); bits=float(f[11]); strand=f[12]
    if length>=20 and p>=95:
        a,b=sorted((qs,qe))
        hits[q].append((g,s,p,length,a,b,ss,se,bits,strand,line.rstrip('\\n')))
count=0
for q,hs in hits.items():
    genes={h[0] for h in hs}
    if 'SOX2' in genes and 'GAPDH' in genes:
        # choose top per gene by bits
        tops=[]
        for gene in ('SOX2','GAPDH'):
            gh=sorted([h for h in hs if h[0]==gene], key=lambda x:(x[8],x[3]), reverse=True)[:3]
            tops+=gh
        print('\\n',q)
        for h in sorted(tops, key=lambda x:(x[4], -x[8])):
            print(' ',h[:10])
        count+=1
        if count>=50: break
print('split_candidate_reads', count)
PY"
