/bin/bash -lc "python - <<'PY' > results/discordant_gene_pairs.primary.tsv
from collections import defaultdict, Counter
# transcript -> gene symbol
t2g={}
with open('refs/ensembl_t2g.tsv') as f:
    for line in f:
        tid,g=line.rstrip('\\n').split('\\t')[:2]
        t2g[tid]=g
r1={}; r2={}; mapq1={}; mapq2={}
with open('work/primary_cdna.sam') as f:
    for line in f:
        if line.startswith('@'): continue
        fields=line.rstrip('\\n').split('\\t')
        if len(fields)<11: continue
        q,flag_s,rname,mapq_s=fields[0],fields[1],fields[2],fields[4]
        flag=int(flag_s)
        if flag & 4 or rname=='*': continue
        sym=t2g.get(rname, rname)
        mapq=int(mapq_s)
        if flag & 64:
            if q not in r1 or mapq>mapq1[q]: r1[q]=sym; mapq1[q]=mapq
        elif flag & 128:
            if q not in r2 or mapq>mapq2[q]: r2[q]=sym; mapq2[q]=mapq
c=Counter()
examples=defaultdict(list)
for q,a in r1.items():
    b=r2.get(q)
    if not b or a==b: continue
    c[(a,b)]+=1
    if len(examples[(a,b)])<5: examples[(a,b)].append(q)
for (a,b),n in c.most_common(100):
    print(n,a,b,','.join(examples[(a,b)]),sep='\\t')
PY
head -n 30 results/discordant_gene_pairs.primary.tsv"
