/bin/bash -c "python - <<'PY'
import gzip, csv, io
with gzip.open('GSE161918_allbatches.HTOandTimepointMetadata.csv.gz','rt') as f:
    rows=list(csv.DictReader(f))
print('rows',len(rows))
subjects={}
for r in rows:
    subjects.setdefault(r['Subject'], []).append(r['Class'])
print('unique subjects',len(subjects))
from collections import Counter
class_by_subj={s: Counter(v).most_common(1)[0][0] for s,v in subjects.items()}
print(Counter(class_by_subj.values()))
for s,c in sorted(class_by_subj.items()):
    print(s,c)
PY"
