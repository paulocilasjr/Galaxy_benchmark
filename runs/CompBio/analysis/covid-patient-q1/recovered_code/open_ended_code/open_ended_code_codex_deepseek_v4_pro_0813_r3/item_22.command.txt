/bin/bash -c "cat > score.py <<'PY'
import numpy as np
PB=np.load('pb_dense.npy')
donors=np.load('donor_names.npy', allow_pickle=True)
genes=np.array([g.decode() if isinstance(g,bytes) else str(g) for g in np.load('gene_names.npy', allow_pickle=True)])
lib=PB.sum(axis=0, keepdims=True)
log=np.log2(PB/lib*1e6+1)
sets={
'ISG':['IFI27','IFITM3','ISG15','MX1','OAS1','IFI44L','IFIT1','RSAD2','IFIT3','IFI6'],
'myeloid_inflam':['S100A8','S100A9','S100A12','LYZ','FCN1','CD14','VCAN','IL1B'],
'plasma':['IGHG1','IGHG3','IGHA1','MZB1','XBP1','CD38','TNFRSF17'],
'pDC':['CLEC4C','LILRA4','IRF7','GZMB','TCF4','PLD4'],
'naiveT':['CCR7','SELL','TCF7','LEF1','IL7R'],
'cDC2':['CD1C','FCER1A','CLEC10A'],
}
for name,gs in sets.items():
    idx=[np.where(genes==g)[0][0] for g in gs if g in genes]
    print('\\n',name, gs, 'found',len(idx))
    score=log[idx,:].mean(axis=0)
    order=np.argsort(score)[::-1]
    print('top donors')
    for j in order:
        print(donors[j], f'{score[j]:.2f}')
PY
python score.py"
