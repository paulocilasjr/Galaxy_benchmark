/bin/bash -c "python - <<'PY'
import requests, json
data=requests.get('https://api.cellxgene.cziscience.com/curation/v1/datasets',timeout=120).json()
for d in data:
    donors=set(d.get('donor_id') or [])
    n=len(donors)
    title=(d.get('title') or '')
    disease=d.get('disease') or []
    if 10<=n<=50:
        # check blood/tissue and covid
        tissues=[t.get('label','').lower() for t in (d.get('tissue') or [])]
        dis=[x.get('label','').lower() for x in disease]
        if 'blood' in tissues or any('covid' in x or 'normal' in x for x in dis):
            print(d.get('dataset_id'), 'cells',d.get('cell_count'),'donors',n,'title',title[:90], 'tissues',tissues[:2],'disease',dis[:3])
PY"
