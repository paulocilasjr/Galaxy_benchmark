/bin/bash -c "python - <<'PY'
import requests
base='https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/'
for f in ['README.md','compbiobench.v1.tsv']:
    r=requests.get(base+f,timeout=30)
    print('FILE',f,r.status_code,len(r.text))
    open(f.split('/')[-1],'wb').write(r.content)
PY
head -50 README.md
wc -c compbiobench.v1.tsv
grep -n \"covid\" compbiobench.v1.tsv | head -20"
