/bin/bash -c "python - <<'PY'
import h5py, numpy as np
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
    data=f['X/data'][:]
    indptr=f['X/indptr'][:]
    indices=f['X/indices'][:]
    var=f['var/_index'][:]
    ft=f['var/feature_type/categories'][:]
    ftc=f['var/feature_type/codes'][:]
    print('X data min max mean nonzero', data.min(), data.max(), data.mean(), len(data))
    print('indptr first last', indptr[:5], indptr[-3:])
    print('var names first/last', var[:10].tolist(), var[-10:].tolist())
    print('feature types', ft.tolist(), np.unique(ftc, return_counts=True))
    # sum each cell quickly with matvec? use array sum segments
    nnz=np.diff(indptr)
    print('nnz per cell min max mean', nnz.min(), nnz.max(), nnz.mean())
    # value distribution bincount sample
    print('bincount', np.bincount(data.astype(np.int64), minlength=10)[:10])
PY"
