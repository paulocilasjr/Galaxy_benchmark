/bin/bash -c "python - <<'PY'
import h5py
p='GSE180578_annotated.h5ad'
with h5py.File(p,'r') as f:
    print('keys', list(f.keys()))
    print('obs attrs', dict(f['obs'].attrs))
    def printkeys(group,indent=''):
        for k in group.keys():
            obj=group[k]
            if isinstance(obj,h5py.Group):
                print(indent,'G',k,list(obj.keys())[:20], 'nkeys',len(obj.keys()))
                printkeys(obj,indent+'  ')
            else:
                print(indent,'D',k,obj.shape,obj.dtype)
    printkeys(f['obs'])
    print('var attrs',dict(f['var'].attrs))
    printkeys(f['var'])
    print('X', dict(f['X'].attrs), f['X/data'].shape)
PY"
