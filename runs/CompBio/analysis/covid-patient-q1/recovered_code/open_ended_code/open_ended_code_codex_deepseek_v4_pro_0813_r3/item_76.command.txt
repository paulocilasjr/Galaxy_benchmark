/bin/bash -c "python - <<'PY'
import tarfile, io, csv, collections
tar=tarfile.open('GSE161918_demuxletResults.tar.gz','r:gz')
counts=collections.Counter(); counts_all=collections.Counter(); lane_counts=[]
for m in tar.getmembers():
    if not m.name.endswith('_demuxed.best'): continue
    f=tar.extractfile(m); text=io.TextIOWrapper(f, encoding='utf-8', newline='')
    reader=csv.DictReader(text, delimiter='\\t')
    c=collections.Counter(); ca=collections.Counter(); total=0
    for row in reader:
        total+=1
        guess=row['BEST.GUESS']
        # parse best guess first subject? 
        subj=guess.split(',')[0] if guess else row['SNG.BEST.GUESS']
        if subj:
            ca[subj]+=1
            if row['DROPLET.TYPE']=='SNG':
                c[subj]+=1
    for k,v in c.items(): counts[k]+=v
    for k,v in ca.items(): counts_all[k]+=v
    lane_counts.append((m.name,total))
    f.close()
print('total lanes',len(lane_counts),'total cells all',sum(x[1] for x in lane_counts))
print('SNG unique subjects',len(counts),'total SNG',sum(counts.values()))
for subj,n in counts.most_common():
    cls='HC' if subj.startswith(('AA','IM','CHI','SHD','HD')) else 'COVID'
    print(f'{subj:15s} {cls:5s} SNG {n:5d} ALL {counts_all[subj]:5d}')
print('--- by class totals')
from collections import Counter
ct=Counter(); ca=Counter()
for subj,n in counts.items():
    cls='HC' if subj.startswith(('AA','IM','CHI','SHD','HD')) else 'COVID'
    ct[cls]+=n; ca[cls]+=counts_all[subj]
print(ct, ca)
PY"
