/bin/bash -c "python - <<'PY'
import h5py, numpy as np
with h5py.File('covid.patients.q1.h5ad','r') as f:
    filt=f['var/feature_is_filtered'][:]
    ftype_cats=[x.decode() if isinstance(x,bytes) else x for x in f['var/feature_type/categories'][:]]
    ftype_codes=f['var/feature_type/codes'][:]
    biotype_cats=[x.decode() if isinstance(x,bytes) else x for x in f['var/feature_biotype/categories'][:]]
    ref_cats=[x.decode() if isinstance(x,bytes) else x for x in f['var/feature_reference/categories'][:]]
    ft_cats=[x.decode() if isinstance(x,bytes) else x for x in f['var/feature_types/categories'][:]]
    print('filtered true',filt.sum(),'false',(~filt).sum())
    print('ftype',np.unique(ftype_codes,return_counts=True))
    for c,count in zip(*np.unique(ftype_codes,return_counts=True)):
        print(ftype_cats[c],count)
    print('biotype cats',biotype_cats)
    print('ref cats',ref_cats)
    print('feature_types cats',ft_cats)
    print('feature_length codes range', f['var/feature_length/codes'][:].min(), f['var/feature_length/codes'][:].max())
PY"
