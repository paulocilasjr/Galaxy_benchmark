/bin/bash -c "python - <<'PY'
import requests
base='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE161nnn/GSE161918/suppl/'
files=['GSE161918_allbatches.HTOandTimepointMetadata.csv.gz','filelist.txt']
for fn in files:
    r=requests.get(base+fn,timeout=60)
    print(fn,r.status_code,len(r.content))
    open(fn,'wb').write(r.content)
PY
ls -lh GSE161918* filelist.txt"
