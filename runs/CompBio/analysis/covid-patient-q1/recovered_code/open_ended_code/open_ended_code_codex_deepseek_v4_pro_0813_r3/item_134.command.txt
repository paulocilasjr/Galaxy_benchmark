import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.mixture import GaussianMixture
from scipy.cluster.hierarchy import linkage, fcluster
PB=np.load('pb_dense.npy'); donors=np.load('donor_names.npy',allow_pickle=True)
genes=np.array([g.decode() if isinstance(g,bytes) else str(g) for g in np.load('gene_names.npy',allow_pickle=True)])
lib=PB.sum(axis=0,keepdims=True); log=np.log2(PB/lib*1e6+1)
healthy_genes=['FCER1A','CLEC4C','CD1C','CLEC10A','CCR7','SELL','TCF7','LEF1','IL7R','GIMAP7','S1PR1','LTB','PLD4','LILRA4','IRF7','MS4A1','IGHD','TCL1A','IL3RA','CST3']
covid_genes=['IFI27','IFITM3','ISG15','MX1','OAS1','IFI44L','IFIT1','RSAD2','S100A8','S100A9','S100A12','SOCS3','ADM','HP','MKI67','TOP2A','IGHG1','IGHG3','IGHA1','MZB1','XBP1','CD38','TNFRSF17','FCGR1A','CLU','RNASE1','RNASE2']
def get(gs): return log[[np.where(genes==g)[0][0] for g in gs if g in genes],:].mean(axis=0)
score=get(healthy_genes)-get(covid_genes)
order=np.argsort(score)[::-1]
print('ranked by marker diff')
for j in order:
    print(donors[j], round(score[j],2))
# define healthy as top 10 by score and also clusters from hvgs
mean=log.mean(axis=1); var=log.var(axis=1)
idx=np.where(mean>0.2)[0]; top=idx[np.argsort(var[idx])[::-1][:2000]]
X=log[top,:].T; Xs=StandardScaler().fit_transform(X)
for meth in ['ward','complete','average']:
    lab=fcluster(linkage(Xs,method=meth,metric='euclidean'),2,criterion='maxclust')
    groups={}
    for d,l in zip(donors,lab): groups.setdefault(l,[]).append(int(d[5:]))
    # identify high score group
    vals=[]
    for l,ids in groups.items():
        js=[np.where(donors==f'donor{i}')[0][0] for i in ids]
        vals.append((np.mean(score[js]),sorted(ids)))
    vals.sort(reverse=True)
    print(meth,'healthy group',vals[0][1],'score',round(vals[0][0],2),'other',vals[1][1])
# kmeans
for k in [2,3]:
    lab=KMeans(n_clusters=k,random_state=0,n_init=20).fit_predict(Xs)
    groups={}
    for d,l in zip(donors,lab): groups.setdefault(l,[]).append(int(d[5:]))
    vals=[]
    for l,ids in groups.items():
        js=[np.where(donors==f'donor{i}')[0][0] for i in ids]
        vals.append((np.mean(score[js]),sorted(ids)))
    vals.sort(reverse=True)
    print('kmeans',k,'healthy',vals[0][1],'score',round(vals[0][0],2))
PY"
