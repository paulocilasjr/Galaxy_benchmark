/bin/bash -c "python - <<'PY'
import h5py
p='GSE180578_annotated.h5ad'
with h5py.File(p,'r') as f:
    raw=f['raw']
    print('raw keys',list(raw.keys()))
    for k in raw.keys():
        obj=raw[k]
        if isinstance(obj,h5py.Group):
            print('group',k,list(obj.keys()))
            for kk in obj.keys(): print(' ',kk,obj[kk].shape,obj[kk].dtype)
        else: print('dataset',k,obj.shape,obj.dtype)
    print('raw var',list(raw['var'].keys()))
    for k in raw['var'].keys(): print(k,raw['var'][k].shape,raw['var'][k].dtype)
PY"
