import re
text=open('GSE157344_gsm.txt').read()

        clinic=[c.split(': ',1)[1] for c in chars if c.startswith('clinic status')][0]
        blood.append((gsm,title,clinic,chars))
print('n blood',len(blood))
from collections import Counter
print(Counter([x[2] for x in blood]))
