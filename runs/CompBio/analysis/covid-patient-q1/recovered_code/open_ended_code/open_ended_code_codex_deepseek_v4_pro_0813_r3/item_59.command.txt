/bin/bash -c "python - <<'PY'
import json, numpy as np, h5py
s=json.load(open('blood_summary.json'))
print('raw total cells', sum(x['cells'] for x in s), 'total umi', sum(x['total_umi'] for x in s))
for x in s:
    print(f\"{x['name'][-18:-11]:8s} cells {x['cells']:4d} umi {x['total_umi']:9.0f}\")
print('--- h5ad donors sorted by cells')
with h5py.File('covid.patients.q1.h5ad','r') as f:
    codes=f['obs/donor_id/codes'][:]; cats=[x.decode() if isinstance(x,bytes) else x for x in f['obs/donor_id/categories'][:]]
counts=np.bincount(codes,minlength=len(cats))
PB=np.load('pb_dense.npy'); lib=PB.sum(axis=0)
for j in np.argsort(counts)[::-1]:
    print(f'{cats[j]:8s} cells {counts[j]:4d} lib {lib[j]:9.0f}')
PY"
