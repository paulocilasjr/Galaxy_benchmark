/bin/bash -c "python - <<'PY'
import tarfile, io, numpy as np
outer=tarfile.open('GSE180578_RAW.tar','r')
name='GSM5465163_Healthy_1_Healthy_2_PBMC_GEX.tar.gz'
f=outer.extractfile(name); inner=tarfile.open(fileobj=f,mode='r:gz')
feat=None
for x in inner.getmembers():
    if x.name.endswith('/features.tsv'): feat=x
text=io.TextIOWrapper(inner.extractfile(feat),encoding='utf-8')
raw_syms=[]
for line in text:
    p=line.strip().split('\\t')
    if len(p)>=2: raw_syms.append(p[1])
text.detach(); inner.close(); f.close(); outer.close()
raw_syms=np.array(raw_syms)
print('raw syms',len(raw_syms), 'unique',len(set(raw_syms)), raw_syms[:10])
genes=np.array([g.decode() if isinstance(g,bytes) else str(g) for g in np.load('gene_names.npy',allow_pickle=True)])
print('our genes',len(genes),'unique',len(set(genes)))
common=np.intersect1d(np.array([g.upper() for g in genes]), np.array([g.upper() for g in raw_syms]))
print('common symbols',len(common), 'our coverage',len(common)/len(genes))
for g in ['FCER1A','CLEC4C','IFI27','S100A8','IGHG1']:
    print(g, g in raw_syms, g in set(genes))
PY"
