/bin/bash -lc "python - <<'PY'
import numpy as np
z=np.load('donor_aggregate.npz',allow_pickle=True)
g=np.array(z['gene_names'],dtype=str); S=z['sums']; lib=z['donor_umis']; names=np.array(z['donor_names'],dtype=str)
gm={x:i for i,x in enumerate(g)}
print(g.shape,S.shape,lib.shape,names[:3])
L=np.log1p(S/np.maximum(lib[:,None],1)*1e6)
for gs in [['ISG15','IFI6','IFI27','IFIT1','IFIT2','IFIT3','IFITM1','IFITM2','IFITM3','MX1','OAS1','OAS2','OAS3','OASL','RSAD2','IFI44','IFI44L','IRF7','STAT1'],['S100A8','S100A9','S100A12','FCN1','IL1B','CXCL8','CXCL10','CCL2','CCL3','CCL4','TNF','TNFAIP3','NFKBIA','JUN','FOS']]:
 ix=[gm[x] for x in gs if x in gm]
 score=L[:,ix].mean(1)
 print('genes',len(ix));
 for i in np.argsort(score): print(names[i],round(score[i],2),end='; ')
 print()
PY"
