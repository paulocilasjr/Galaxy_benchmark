/bin/bash -lc "python - <<'PY'
import urllib.request,re
s=urllib.request.urlopen('https://www.covid19cellatlas.org/index.patient.html').read().decode('utf8','replace')
urls=sorted(set(re.findall(r'https?://["'^" <>]+'"\\.h5ad',s)))
print('n',len(urls))
for u in urls:
 try:
  q=urllib.request.Request(u,method='HEAD')
  with urllib.request.urlopen(q,timeout=20) as r: print(r.headers.get('Content-Length'),u)
 except Exception as e: print('ERR',type(e).__name__,u)
PY"
