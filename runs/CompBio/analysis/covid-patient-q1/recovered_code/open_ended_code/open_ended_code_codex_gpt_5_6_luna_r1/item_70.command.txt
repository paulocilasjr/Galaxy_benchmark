/bin/bash -lc "python - <<'PY'
import numpy as np
z=np.load('donor_aggregate.npz',allow_pickle=True)
g=np.array(z['gene_names']).astype(str); S=z['sums']; lib=z['donor_umis']; names=np.array(z['donor_names']).astype(str)
# panel aliases found
panels={
'ISG_core':['ISG15','IFI6','IFI27','IFIT1','IFIT2','IFIT3','IFITM1','IFITM2','IFITM3','MX1','OAS1','OAS2','OAS3','OASL','RSAD2','IFI44','IFI44L','IRF7','STAT1','IFIH1','DDX58','SAMD9L','EPSTI1','GBP1','GBP2','GBP3','GBP4','GBP5','PARP9','PARP10'],
'Inflam':['S100A8','S100A9','S100A12','FCN1','IL1B','CXCL8','CXCL10','CCL2','CCL3','CCL4','TNF','TNFAIP3','NFKBIA','JUN','FOS','CCL5','LGALS3'],
'Mono':['LYZ','S100A8','S100A9','S100A12','FCN1','CTSS','TYROBP','LST1','FCGR3A','FCGR1A','MS4A7','CTSD','LGALS3','IFITM3'],
'Platelet':['PPBP','PF4','PF4V1','GNG11','NGRN','SDPR'],
'Lymph':['IL7R','CCR7','LTB','MAL','TRBC1','TRBC2','CD3D','CD3E','IL32','LTB','MALAT1'],
'NK':['NKG7','GNLY','GZMB','GZMH','GZMK','CCL5','KLRD1'],
'B':['MS4A1','CD79A','CD37','CD74','HLA-DRA','CD79B','CD83'],
'APC':['HLA-DRA','HLA-DPA1','HLA-DPB1','HLA-DQA1','HLA-DQB1','CD74','FCER1A','CLEC10A','CLEC4C','GZMB'],
'PDC':['GZMB','GZMB','JCHAIN','GZMB','IRF7','TCF4','CLEC4C','GZMB','LILRA4','SERPINF1'],
}
# map and logCPM
gm={x:i for i,x in enumerate(g)}
L=np.log1p(S/np.maximum(lib[:,None],1)*1e6)
print('panel scores logCPM gene counts')
for p,gs in panels.items():
 ix=[gm[x] for x in gs if x in gm]; X=L[:,ix]; score=X.mean(1)
 # standardize across donors and print sorted
 order=np.argsort(score)
 print('\\n'+p, 'genes', [gs[j] for j,x in enumerate(gs) if x in gm])
 print(' '.join(f'{names[i].replace(\"donor\",\"\")}:{score[i]:.2f}' for i in order))
# curated score: z panel scores, with inflam/ISG positive, healthy lymph maybe
scores=[]
for p,gs in panels.items():
 ix=[gm[x] for x in gs if x in gm]; scores.append(L[:,ix].mean(1))
P=np.array(scores).T
Z=(P-P.mean(0))/P.std(0)
# print key z combo
for combo in [('IFN',0),('infl',1),('mono',2),('plate',3),('lymph',4),('NK',5),('B',6),('APC',7),('pdc',8)]: pass
C=Z[:,0]+Z[:,1]+Z[:,3] - 0.2*Z[:,4] # high disease rough, but lymph not disease
print('\\nrough disease combo')
for i in np.argsort(C): print(names[i].replace('donor',''),f'{C[i]:.2f}',f'IFN{Z[i,0]:+.1f}',f'Inf{Z[i,1]:+.1f}',f'P{Z[i,3]:+.1f}')
PY"
