/bin/bash -lc "python - <<'PY'
import numpy as np,pandas as pd
z=np.load('donor_aggregate.npz',allow_pickle=True); G=z['gene_names']; S=z['gene_sum']; lib=z['donor_lib']; names=z['donor_names']; ids=np.array([int(x[5:]) for x in names]); cpm=S/lib[:,None]*1e6
sets={
'IFN':['ISG15','IFI6','IFI27','IFI44L','IFIT1','IFIT3','MX1','OAS1','RSAD2','XAF1','USP18'],
'Neut':['S100A8','S100A9','S100A12','FCGR3B','CSF3R','MNDA'],
'Mono':['LYZ','LST1','FCN1','CTSS','CTSD','TYMP','LGALS3'],
'HLA2':['HLA-DRA','HLA-DPA1','HLA-DPB1','CD74'],
'Plasma':['JCHAIN','MZB1','IGHG1','SDC1'],
'Platelet':['PPBP','PF4','GP9'],
'T':['CD3D','CD3E','TRAC','IL7R','CCR7','LTB'],
'NK':['NKG7','GNLY','KLRD1','PRF1','GZMB','GZMH'],
'Inflam':['IL1B','TNF','NFKBIA','FOS','JUN','DUSP1'],
}
df=pd.DataFrame({'id':ids})
for k,gs in sets.items():
 ix=[np.where(G==g)[0][0] for g in gs if np.any(G==g)]
 df[k]=cpm[:,ix].sum(1)
# log then z within score
for k in sets: df[k+'z']=(np.log1p(df[k])-np.log1p(df[k]).mean())/np.log1p(df[k]).std()
df=df.sort_values('id')
print(df[['id']+list(sets)].to_string(index=False,float_format=lambda x:f'{x:8.1f}'))
print('\\nz')
print(df[['id']+[x+'z' for x in sets]].to_string(index=False,float_format=lambda x:f'{x:6.2f}'))
PY"
