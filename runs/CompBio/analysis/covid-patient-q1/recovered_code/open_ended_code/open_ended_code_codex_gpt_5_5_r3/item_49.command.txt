import h5py, time
url=b'https://datasets.cellxgene.cziscience.com/fe2e847c-1602-4f1b-86a4-112e4dc7a8e3.h5ad'

    print('obs keys first', list(f['obs'].keys())[:50])
    for k in list(f['obs'].keys())[:20]:
        obj=f['obs'][k]
        print(k, type(obj).__name__, getattr(obj,'shape',''), dict(getattr(obj,'attrs',{})))
PY"
