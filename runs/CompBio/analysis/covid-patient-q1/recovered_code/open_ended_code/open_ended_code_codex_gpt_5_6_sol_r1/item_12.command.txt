/bin/bash -lc "python - <<'PY'
import numpy as np,pandas as pd
z=np.load('donor_aggregates.npz',allow_pickle=True)
don=z['donors']; genes=z['genes']; sums=z['sums']; det=z['detections']; cells=z['cells']; totals=z['total_umis']
# CPM pseudobulk log2
lcpm=np.log2(1+sums/totals[:,None]*1e6)
markers=['IFI27','IFI6','IFIT1','IFIT2','IFIT3','ISG15','MX1','OAS1','OAS2','OAS3','RSAD2','IFITM1','IFITM3','SIGLEC1','S100A8','S100A9','S100A10','IL1B','TNF','CXCL10','CCL2','CCL3','CCL4','NKG7','GNLY','GZMB','FCGR3A','LYZ','MS4A1','CD3D','CCR7','LTB','PPBP','PF4','HBB','HBA1']
idx={g:i for i,g in enumerate(genes)}
df=pd.DataFrame({'donor':don,'cells':cells,'umi/cell':totals/cells})
for g in markers:
 if g in idx: df[g]=lcpm[:,idx[g]]
# scores standardized across donors
def zscore(a): return (a-a.mean(0))/a.std(0)
isgs=[g for g in ['IFI27','IFI6','IFIT1','IFIT2','IFIT3','ISG15','MX1','OAS1','OAS2','OAS3','RSAD2','IFITM1'] if g in idx]
inf=[g for g in ['S100A8','S100A9','IL1B','SIGLEC1','CXCL10','CCL2'] if g in idx]
df['ISGz']=zscore(np.stack([lcpm[:,idx[g]] for g in isgs],1)).mean(1)
df['INFz']=zscore(np.stack([lcpm[:,idx[g]] for g in inf],1)).mean(1)
cols=['donor','cells','umi/cell','ISGz','INFz']+markers
print(df.sort_values('ISGz').to_string(index=False,float_format=lambda x:f'{x:.2f}'))
PY"
