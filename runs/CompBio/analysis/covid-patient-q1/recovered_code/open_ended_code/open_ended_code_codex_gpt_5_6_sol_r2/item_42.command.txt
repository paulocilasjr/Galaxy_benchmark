/bin/bash -lc "OPENBLAS_NUM_THREADS=1 python -u - <<'PY'
import numpy as np
z=np.load('donor_aggregates.npz');D=z['donor_names'];G=z['gene_names'];C=z['count_sum'];n=np.array([int(x[5:]) for x in D]);L=np.log2(1+C/C.sum(1)[:,None]*1e6)
H=np.isin(n,[1,5,11,12,17,21,24,27,29,32]); V=~H
muH=L[H].mean(0); sdH=L[H].std(0)+.2
for donor in [1,19,22,25,27,32]:
 i=np.flatnonzero(n==donor)[0]; Z=(L[i]-muH)/sdH
 # restrict genes at least moderate in donor or healthy
 keep=(L[i]>2)|(muH>2)
 ix=np.where(keep)[0][np.argsort(Z[keep])[-35:][::-1]]
 print('\\nDONOR',donor,'top vs healthy')
 print(' '.join(f'{G[j]}:{Z[j]:.1f}' for j in ix))
# global logCPM PCA optimal 2means on PC1 and score ordering
keep=(C>0).sum(0)>=20;X=L[:,keep];X-=X.mean(0); vv=X.var(0);X=X[:,vv>np.quantile(vv,.5)]
u,s,vt=np.linalg.svd(X,full_matrices=False); p=u[:,0]*s[0]
# orient healthy-looking negative based S100A8
if np.corrcoef(p,L[:,np.flatnonzero(G=='S100A8')[0]])[0,1]<0:p=-p
order=np.argsort(p); ss=p[order]
best=None
for k in range(2,len(n)-1):
 sse=((ss[:k]-ss[:k].mean())**2).sum()+((ss[k:]-ss[k:].mean())**2).sum()
 if best is None or sse<best[0]:best=(sse,k)
print('\\noptimal PC1 split k',best[1],sorted(n[order[:best[1]]].tolist()))
print('scores')
for i in np.argsort(n): print(n[i],round(p[i],2))
PY"
