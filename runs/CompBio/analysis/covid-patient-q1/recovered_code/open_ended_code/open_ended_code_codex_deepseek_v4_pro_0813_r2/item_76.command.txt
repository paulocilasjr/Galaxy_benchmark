/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, pandas as pd, time
url='https://cellxgene-census-public-us-west-2.s3.us-west-2.amazonaws.com/cell-census/2025-11-08/h5ads/ebc2e1ff-c8f9-466a-acf4-9d291afaf8b3.h5ad'
f=h5py.File(url,'r',driver='ros3',aws_region=b'us-west-2')
obs=f['obs']
dcodes=obs['donor_id/codes'][:]
dcats=obs['donor_id/categories'][:].astype(str)
ctcodes=obs['cell_type/codes'][:]
ctcats=obs['cell_type/categories'][:].astype(str)
discodes=obs['disease/codes'][:]
discats=obs['disease/categories'][:].astype(str)
print('loaded',len(dcodes))
df=pd.DataFrame({'donor_code':dcodes,'cell_code':ctcodes,'disease_code':discodes})
# donor-level disease mode and cell type counts
rows=[]
for i,d in enumerate(dcats):
    sub=df[df.donor_code==i]
    dis=discats[sub.disease_code.value_counts().idxmax()]
    cts=sub.cell_code.value_counts().sort_index()
    row={'donor':d,'disease':dis,'n':len(sub)}
    for ci,ct in enumerate(ctcats):
        row[ct]=int(cts.get(ci,0))
    rows.append(row)
meta=pd.DataFrame(rows)
meta.to_csv('ahern_donor_celltypes.csv',index=False)
print(meta[['donor','disease','n']].to_string())
print('cell types', ctcats)
print(meta[ctcats].sum())
PY"
