from scipy.spatial.distance import cdist
# our props from cell_scores
D=np.load('cell_scores.npz', allow_pickle=True)
cats=D['cats']; df={}
for k in D.keys():
 if k"'!='"'cats': df[k]=D[k]

 df[k+'_prop']=df[k]/df['total_cells']

 return np.sqrt(df)

for i,don in enumerate(cats):
