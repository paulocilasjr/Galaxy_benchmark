/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegressionCV, LogisticRegression
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.svm import SVC
from sklearn.pipeline import make_pipeline
from sklearn.model_selection import LeaveOneGroupOut, cross_val_predict, cross_val_score
from sklearn.metrics import roc_auc_score, classification_report

ref=pd.read_csv('ref_features_all.csv',index_col=0)
our=pd.read_csv('our_features.csv',index_col=0)
# feature columns: gene set scores and prop_ columns
base=['ISG','Inflam','Plasma','NaiveT','Cyto','cDC','pDC','Mono','B','T','NK']
prop=['prop_T','prop_NK','prop_B','prop_Mono','prop_DC','prop_Plasma','prop_Platelet','prop_RBC','prop_Neutrophil']
# Use only features present in both
features=base+prop
Xref=ref[features].values; yref=ref.y.values; groups=ref.dataset.values
Xour=our[features].values
# scale by reference
for name,model in [
 ('lr',make_pipeline(StandardScaler(),LogisticRegression(class_weight='balanced',max_iter=1000))),
 ('rf',make_pipeline(StandardScaler(),RandomForestClassifier(n_estimators=1000,class_weight='balanced',random_state=0))),
 ('svc',make_pipeline(StandardScaler(),SVC(kernel='linear',class_weight='balanced',probability=True,random_state=0))),
 ('gb',make_pipeline(StandardScaler(),GradientBoostingClassifier(random_state=0))),
]:
 logo=LeaveOneGroupOut()
 # cross-val predictions
 preds=cross_val_predict(model,Xref,yref,groups=groups,cv=logo,method='predict_proba')[:,1]
 print('\\n',name,'LOGO AUC',roc_auc_score(yref,preds))
 model.fit(Xref,yref)
 proba=model.predict_proba(Xour)[:,1]
 pred=model.predict(Xour)
 healthy=[int(our.index[i][5:]) for i in range(len(our)) if pred[i]==1]
 print('pred healthy',healthy)
 print(pd.Series(proba,index=[int(c[5:]) for c in our.index]).sort_values(ascending=False).round(3).to_string())
PY"
