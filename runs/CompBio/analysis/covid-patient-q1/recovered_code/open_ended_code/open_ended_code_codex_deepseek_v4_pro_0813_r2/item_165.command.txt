/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linear_sum_assignment
ref=pd.read_csv('ref_features_all.csv',index_col=0)
our=pd.read_csv('our_features.csv',index_col=0)
base=['ISG','Inflam','Plasma','NaiveT','Cyto','cDC','pDC','Mono','B','T','NK']
prop=['prop_T','prop_NK','prop_B','prop_Mono','prop_DC','prop_Plasma','prop_Platelet','prop_RBC','prop_Neutrophil']
features=base+prop
# Standardize across combined ref+our per feature
allX=pd.concat([ref[features],our[features]])
mean=allX.mean(); std=allX.std()
Xref=((ref[features]-mean)/std).values
Xour=((our[features]-mean)/std).values
for ds in ref.dataset.unique():
 sub=ref[ref.dataset==ds]; X=((sub[features]-mean)/std).values
 D=cdist(Xour,X,metric='correlation')
 row,col=linear_sum_assignment(D)
 assigned=sub.index[col]; dis=sub.y.values[col]
 # print for each our donor best match and disease
 print('\\nDataset',ds)
 for r,c in zip(row,col):
  ourdon=int(our.index[r][5:]); refdon=sub.index[c]; label=sub.y.values[c]
  if ourdon in [1,5,11,12,17,21,24,27,29,32,25]:
   print(ourdon,'->',refdon,label,'dist',round(D[r,c],3))
 print('assigned diseases counts',pd.Series(dis).value_counts().to_dict())
PY"
