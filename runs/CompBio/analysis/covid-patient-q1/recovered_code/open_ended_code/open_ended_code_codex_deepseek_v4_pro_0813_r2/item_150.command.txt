from sklearn.preprocessing import StandardScaler

# load/construct features similar codex but use our cell type props and expression scores from cell_scores/pseudobulk
D=np.load('cell_scores.npz',allow_pickle=True); cats=D['cats']; df={}
for k in D.keys():
 if k"'!='"'cats': df[k]=D[k]

for k in ['type_CD14_Mono','type_CD16_Mono','type_T','type_NK','type_B','type_Plasma','type_cDC','type_pDC','type_Platelet','type_RBC']:
 df[k+'_prop']=df[k]/df['total_cells']
df['ISG']=df['ISG_all_sum']/df['total_cells']
# pseudobulk marker features from previous script maybe compute quickly using saved pseudobulk and symbols
Dp=np.load('pseudobulk_raw.npz',allow_pickle=True); sums=Dp['sums']; totals=Dp['totals']; cpm=sums/totals[:,None]*1e6; logcpm=np.log1p(cpm)
import anndata as ad

      'Inflam':['S100A8','S100A9','S100A12','IL1B','FCN1','VCAN','LST1','CCL3','CCL4'],
      'cDC':['FCER1A','CLEC10A','CD1C','HLA-DRA','CST3'],
      'pDC':['LILRA4','CLEC4C','IL3RA','TCF4'],

 except Exception as e: print(link,e)
