/bin/bash -lc "python - <<'PY'
import requests, json, time
url='https://api.cellxgene.cziscience.com/curation/v1/datasets'
data=requests.get(url,timeout=60).json()
candidates=[]
for d in data:
    title=(d.get('title') or '')
    if ('COVID' in title.upper()) or ('PBMC' in title.upper()):
        candidates.append(d)
print('candidate datasets',len(candidates))
colls=set(d.get('collection_id') for d in candidates)
print('collections',len(colls))
# cache collection info
info={}
for cid in colls:
    r=requests.get(f'https://api.cellxgene.cziscience.com/curation/v1/collections/{cid}',timeout=30)
    if not r.ok:
        print('fail',cid,r.status_code); continue
    cd=r.json()
    for ds in cd.get('datasets',[]):
        donors=ds.get('donor_id',[])
        title=ds.get('title','')
        cc=ds.get('cell_count',0)
        # only if dataset candidate id or title covid/pbmc
        if ds.get('dataset_id') in set(x['dataset_id'] for x in candidates) or ('COVID' in title.upper() or 'PBMC' in title.upper()):
            print(cid[:8], ds['dataset_id'][:8], cc, len(set(donors)), title[:80])
            if len(set(donors))==33:
                print('FOUND 33 DONORS', ds['dataset_id'], title)
                print(sorted(donors))
PY"
