/bin/bash -lc "python - <<'PY'
import requests,json,re
# esearch gds
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi'
params={'db':'gds','term':'PBMC[All Fields] AND COVID-19[All Fields] AND Homo sapiens[Organism] AND gse[Filter]','retmax':'1000','retmode':'json'}
r=requests.get(url,params=params,timeout=30)
print(r.url, r.status_code)
data=r.json(); ids=data['esearchresult'].get('idlist',[]); print('count',len(ids),ids[:50])
# esummary all
if ids:
    resp=requests.post('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',data={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=60)
    res=resp.json()['result']
    for uid in ids:
        it=res.get(uid,{})
        samples=it.get('samples',[])
        print(it.get('accession'), len(samples), it.get('title','')[:100])
PY"
