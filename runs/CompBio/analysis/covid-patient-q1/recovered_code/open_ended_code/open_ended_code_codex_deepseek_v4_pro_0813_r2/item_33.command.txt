/bin/bash -lc "python - <<'PY'
import requests,re
text=requests.get('https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE163668&targ=gsm&form=text&view=brief',timeout=30).text
# split by "'^SAMPLE
parts=text.split('"'"'^SAMPLE = '"')[1:]
for part in parts:
    s=part.split('\\n',1)[0].strip()
    title=re.search(r'"'!Sample_title = ([''^'"\\n]+)',part)
    chars=re.findall(r'"'!Sample_characteristics_ch1 = ([''^'"\\n]+)',part)
    desc=re.search(r'"'!Sample_description = ([''^'"\\n]+)',part)
    # only GEX non pooled maybe
    if 'GEX' in (title.group(1) if title else '') or 'donor' in part.lower():
        print(s, title.group(1) if title else '', chars, desc.group(1) if desc else '')
PY"
