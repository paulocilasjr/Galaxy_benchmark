import anndata as ad, numpy as np, scipy.sparse as sp, pandas as pd, time, resource, os
f='covid.patients.q1.h5ad'

    j=min(n,i+chunk)
    rows=np.arange(i,j)
    c=codes[i:j]
    X=A.X[i:j]
    if hasattr(X,'toarray'):
        X=X.toarray()
    D=sp.csr_matrix((np.ones(j-i,dtype=np.float32), (np.arange(j-i), c)), shape=(j-i, nd))
    block=D.T @ X  # dense (nd,g)
    sums += block
    # ncells counts and total counts
    if sp.issparse(X):
        rowtot=np.asarray(X.sum(axis=1)).ravel()
    else:
        rowtot=X.sum(axis=1)
    np.add.at(ncells, c, 1)
    np.add.at(totals, c, rowtot)
    if i % 100000 < chunk:
