/bin/bash -lc "python - <<'PY'
# reproduce and print best matching with distance and disease
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
D=np.load('cell_scores.npz', allow_pickle=True)
cats=D['cats']; df={}
for k in D.keys():
 if k"'!='"'cats': df[k]=D[k]
df=pd.DataFrame(df,index=cats)
for k in ['type_CD14_Mono','type_CD16_Mono','type_T','type_NK','type_B','type_Plasma','type_cDC','type_pDC','type_Platelet','type_RBC']:
 df[k+'_prop']=df[k]/df['total_cells']
our=pd.DataFrame(index=cats)
our['T']=df['type_T_prop'].values; our['NK']=df['type_NK_prop'].values; our['B']=(df['type_B_prop']+df['type_Plasma_prop']).values; our['Mono']=(df['type_CD14_Mono_prop']+df['type_CD16_Mono_prop']).values; our['DC']=(df['type_cDC_prop']+df['type_pDC_prop']).values; our['Platelet']=df['type_Platelet_prop'].values; our['RBC']=df['type_RBC_prop'].values
feats=['T','NK','B','Mono','DC','Platelet','RBC']
our[feats]=our[feats].div(our[feats].sum(axis=1),axis=0)
meta=pd.read_csv('ahern_donor_celltypes.csv')
a=pd.DataFrame(index=meta['donor']); a['disease']=meta['disease'].values
a['T']=meta[['CD4-positive, alpha-beta T cell','CD8-positive, alpha-beta T cell','gamma-delta T cell','mucosal invariant T cell','double-positive, alpha-beta thymocyte','double negative T regulatory cell']].sum(axis=1).values
a['NK']=meta['natural killer cell'].values; a['B']=meta[['B cell','plasmablast']].sum(axis=1).values; a['Mono']=meta[['classical monocyte','non-classical monocyte']].sum(axis=1).values; a['DC']=meta['dendritic cell'].values; a['Platelet']=0; a['RBC']=meta[['enucleated reticulocyte','megakaryocyte-erythroid progenitor cell']].sum(axis=1).values
a[feats]=a[feats].div(a[feats].sum(axis=1),axis=0)
Dmat=cdist(np.sqrt(our[feats].values), np.sqrt(a[feats].values))
# best one-to-one using greedy/ Hungarian
from scipy.optimize import linear_sum_assignment
row_ind,col_ind=linear_sum_assignment(Dmat)
print('Hungarian matches')
for r,c in zip(row_ind,col_ind):
    print(cats[r],'->',a.index[c],a.iloc[c]['disease'],'dist',Dmat[r,c])
# print normal disease counts among assigned
print('assigned diseases', pd.Series([a.iloc[c]['disease'] for c in col_ind]).value_counts().to_dict())
# Also print our ISG mono for labels
df['ISG_mono']=df['ISG_mono_sum']/df['cnt_mono']
print('our assigned normal donors:', [cats[r] for r,c in zip(row_ind,col_ind) if a.iloc[c]['disease']=='normal'])
print('their ISG:', {cats[r]:round(df.loc[cats[r],'ISG_mono'],2) for r,c in zip(row_ind,col_ind) if a.iloc[c]['disease']=='normal'})
PY"
