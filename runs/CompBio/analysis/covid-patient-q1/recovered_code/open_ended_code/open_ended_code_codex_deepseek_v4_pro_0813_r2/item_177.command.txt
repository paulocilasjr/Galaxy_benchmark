/bin/bash -lc "python - <<'PY'
# quick compute control score top from our_features? We already have features, but codex control_score formula. Print first10 category and scores.
import pandas as pd, numpy as np
our=pd.read_csv('our_features.csv',index_col=0)
features=['ISG','Inflam','cDC','pDC','NaiveT','Cyto','Plasma']
for c in features: our[c+'_z']=(our[c]-our[c].mean())/our[c].std(ddof=0)
our['control_score']=our['cDC_z']+our['pDC_z']+0.5*our['NaiveT_z']+0.5*our['Cyto_z']-our['ISG_z']-our['Inflam_z']-0.5*our['Plasma_z']
top=our.sort_values('control_score',ascending=False)
print(top[['donor_num','control_score']].head(15).to_string(index=False))
print('top10 sorted',','.join(map(str,sorted(top['donor_num'].head(10).tolist()))))
PY"
