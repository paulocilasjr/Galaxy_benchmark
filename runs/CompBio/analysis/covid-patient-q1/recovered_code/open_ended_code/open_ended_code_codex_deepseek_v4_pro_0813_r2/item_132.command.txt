/bin/bash -lc "python - <<'PY'
import requests, json
url='https://api.cellxgene.cziscience.com/curation/v1/collections'
r=requests.get(url,timeout=60)
collections=r.json()
print('collections',len(collections))
candidates=[]
for coll in collections:
    for ds in coll.get('datasets',[]):
        donors=ds.get('donor_id',[]) or []
        nd=len(set(donors))
        title=ds.get('title','') or ''
        cname=coll.get('name','') or ''
        # find potential: 33 donors and (covid/pbmc in titles or cell count around ours)
        cc=ds.get('cell_count',0)
        if nd==33 or ('COVID' in title.upper() or 'PBMC' in title.upper() or 'COVID' in cname.upper() or 'PBMC' in cname.upper()):
            candidates.append((coll['collection_id'],coll.get('name',''),ds.get('dataset_id'),title,cc,nd,ds.get('disease','') if 'disease' in ds else ''))
print('candidates',len(candidates))
for c in candidates:
    print(c)
# Also specifically nd==33 all datasets
print('\\nAll 33-donor datasets:')
for coll in collections:
    for ds in coll.get('datasets',[]):
        donors=ds.get('donor_id',[]) or []
        if len(set(donors))==33:
            print(coll['collection_id'], coll.get('name',''), ds.get('dataset_id'), ds.get('title',''), ds.get('cell_count'))
PY"
