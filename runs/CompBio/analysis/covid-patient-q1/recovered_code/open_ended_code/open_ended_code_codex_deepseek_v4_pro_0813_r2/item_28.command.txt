/bin/bash -lc "python - <<'PY'
import requests,re
url='https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE166992&targ=gsm&form=text&view=brief'
text=requests.get(url,timeout=30).text
samples=re.findall(r'\\"'^SAMPLE = (GSM'"\\d+)',text)
print(len(samples))
for s in samples:
    # extract block
    m=re.search(r'\\"'^SAMPLE = '"'+s+r'\\n(.*?)(?=\\"'^SAMPLE|'"\\Z)', text, re.S)
    block=m.group(1)
    title=re.search(r'"'!Sample_title = (.*)'"',block).group(1)
    status=re.search(r'"'!Sample_characteristics_ch1 = (.*)'"',block).group(1)
    print(s,title,status)
PY"
