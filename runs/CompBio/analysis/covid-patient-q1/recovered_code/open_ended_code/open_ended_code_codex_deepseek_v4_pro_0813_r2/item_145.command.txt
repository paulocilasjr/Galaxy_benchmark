/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, pandas as pd, time
url='https://cellxgene-census-public-us-west-2.s3.us-west-2.amazonaws.com/cell-census/2025-11-08/h5ads/c7775e88-49bf-4ba2-a03b-93f00447c958.h5ad'
f=h5py.File(url,'r',driver='ros3',aws_region=b'us-west-2')
dcodes=f['obs/donor_id/codes'][:]; dcats=f['obs/donor_id/categories'][:].astype(str)
ctcodes=f['obs/cell_type/codes'][:]; ctcats=f['obs/cell_type/categories'][:].astype(str)
discodes=f['obs/disease/codes'][:]; discats=f['obs/disease/categories'][:].astype(str)
print(ctcats)
# aggregate coarse categories
meta=[]
for i,d in enumerate(dcats):
 sub=ctcodes[dcodes==i]; dis=discats[discodes[dcodes==i].min()]
 counts=pd.Series(sub).value_counts().sort_index()
 row={'donor':d,'disease':dis,'n':len(sub)}
 for ci,ct in enumerate(ctcats): row[ct]=int(counts.get(ci,0))
 meta.append(row)
meta=pd.DataFrame(meta)
meta.to_csv('stephenson_donor_celltypes.csv',index=False)
print(meta[['donor','disease','n']].head())
print('disease counts',meta.disease.value_counts())
PY"
