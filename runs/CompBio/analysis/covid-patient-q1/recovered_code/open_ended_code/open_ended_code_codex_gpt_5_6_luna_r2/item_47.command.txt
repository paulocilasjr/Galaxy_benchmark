/bin/bash -lc "python - <<'PY'
import h5py,re
with h5py.File('covid.patients.q1.h5ad','r') as f:
    donors=f['obs/donor_id/categories'].asstr()[:].tolist()
healthy=donors[:10]
assert healthy == ['donor12','donor24','donor29','donor5','donor17','donor32','donor1','donor27','donor21','donor11']
ids=sorted(int(re.fullmatch(r'donor(\\d+)',x).group(1)) for x in healthy)
answer=','.join(map(str,ids))
assert answer == '1,5,11,12,17,21,24,27,29,32'
print(answer)
PY"
