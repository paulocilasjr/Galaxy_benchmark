import h5py,numpy as np,scipy.sparse as sp
with h5py.File('covid.patients.q1.h5ad','r') as f:

 x=(mat.T@oh).T
 print(type(x),x.dtype,x.shape,x.format if hasattr(x,'format') else '')
 print(type(np.asarray(x)),np.asarray(x).dtype)
 print('data',mat.dtype,oh.dtype)
