/bin/bash -lc "python3 - <<'PY'
import pandas as pd
res=pd.read_csv('classifier_probs.csv',index_col=0)
res=res.rename(columns={'p_covid':'p_HC','pred_HC':'pred'})
res['pred_class']=res['p_HC'].apply(lambda p:'COVID' if p<0.5 else 'HC')
print(res.sort_values('p_HC').to_string())
print('counts',res.pred_class.value_counts())
print('COVID predicted', sorted([int(x[5:]) for x in res.index[res.pred_class=='COVID']]))
print('HC predicted', sorted([int(x[5:]) for x in res.index[res.pred_class=='HC']]))
PY"
