/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import cross_val_predict, StratifiedKFold
from sklearn.metrics import roc_auc_score, classification_report
odf=pd.read_csv('orig_signature_scores.csv',index_col=0)
wdf=pd.read_csv('working_signature_scores2.csv',index_col=0)
features=['up','down','diff','ISG','Mono','T','NK','B','pDC','Platelet','Erythroid','Neutro','myel_lymph_ratio','isg_lymph_ratio']
Xo=odf[features].values; yo=(odf.label.values) # label 1 HC? check
print(odf.label.value_counts())
Xw=wdf[features].values
scaler=StandardScaler().fit(Xo)
for C in [0.01,0.1,1,10]:
 lr=LogisticRegression(C=C,max_iter=10000).fit(scaler.transform(Xo),yo)
 print('\\nC',C,'coef')
 cv=cross_val_predict(lr,scaler.transform(Xo),yo,cv=StratifiedKFold(5,shuffle=True,random_state=0),method='predict_proba')[:,1]
 print('cv auc',roc_auc_score(yo,cv))
 proba=lr.predict_proba(scaler.transform(Xw))[:,1]
 res=pd.Series(proba,index=wdf.index).sort_values(ascending=False)
 print('pred HC>0.5',sorted([int(x[5:]) for x in res.index[res>0.5]]))
 print('top14',sorted([int(x[5:]) for x in res.index[:14]]))
 print(res.round(3).to_string())
PY"
