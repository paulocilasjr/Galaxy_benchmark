/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
m=pd.read_csv('seurat_meta.csv', index_col=0, low_memory=False)
# coarse mapping
def coarse(ct):
    ct=str(ct)
    if ct.startswith('CD4') or ct.startswith('CD8') or ct.startswith('T_') or ct in ['Tcell','Treg','MAIT','DNT','DPT_DoublePositiveT','gammadeltaT'] or 'T.' in ct: return 'T'
    if ct.startswith('Mono') or ct.startswith('Granul') or ct.startswith('cDC') or ct.startswith('DC') or ct in ['mDC']: return 'Myeloid'
    if ct.startswith('NK') or ct.startswith('Mono_Nonclassical') or 'NK' in ct: return 'NK' if ct.startswith('NK') else 'Myeloid'
    if ct.startswith('B_') or ct in ['Plasmablast','PB_Plasmablasts'] or 'B_' in ct: return 'B'
    if ct.startswith('pDC'): return 'pDC'
    if ct.startswith('Platelet') or ct.startswith('RBC') or ct.startswith('Eryth'): return 'Other'
    if ct.startswith('dim') or ct.startswith('dblt') or ct.startswith('Unk') or ct.startswith('Unknown') or ct.startswith('gated_out'): return 'Other'
    return 'Other'
m['coarse']=m['mergedcelltype'].apply(coarse)
freq=m.groupby(['Donor','Class'])['coarse'].value_counts(normalize=True).unstack(fill_value=0).reset_index()
freq.to_csv('orig_coarse_freq.csv', index=False)
print(freq.round(3).to_string())
# also class counts
print(m.groupby('Class').size())
PY"
