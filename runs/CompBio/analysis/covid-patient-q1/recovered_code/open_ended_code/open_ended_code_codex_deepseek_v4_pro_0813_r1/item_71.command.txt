/bin/bash -lc "python - <<'PY'
import pandas as pd
m=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False)
for maskname,mask in {
 'Class_COVID': m['Class']=='COVID',
 'COVID_specific_ungated': (m['Class']=='COVID')&(m['specific_gating']=='ungated'),
 'COVID_celltypeQC_true': (m['Class']=='COVID')&(m['celltypeQC'].astype(str)=='True'),
 'COVID_specific_ungated_QCtrue': (m['Class']=='COVID')&(m['specific_gating']=='ungated')&(m['celltypeQC'].astype(str)=='True'),
}.items():
 sub=m[mask]
 print(maskname, 'cells',len(sub),'donors',sub['Donor'].nunique(),'samples',sub['sample_id'].nunique())
 print(sub.groupby('Donor').size().sort_values().to_dict())
 print()
PY"
