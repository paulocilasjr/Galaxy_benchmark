/bin/bash -lc "cat > cluster_donors.py <<'PY'
import numpy as np, pandas as pd
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.mixture import GaussianMixture
import anndata as ad

adata=ad.read_h5ad('covid.patients.q1.h5ad')
donors=adata.obs['donor_id'].astype(str).values
cats=np.array(sorted(set(donors), key=lambda x:int(x[5:])))
cell_counts=np.array([(donors==d).sum() for d in cats])
D=np.load('donor_sum.npz')
sums=D['sum']; nnz=D['nnz']
assert list(D['cats'])==list(cats)
# cpm
libs=sums.sum(axis=0, keepdims=True)
cpm=sums/libs*1e6
logcpm=np.log1p(cpm)
# frac fix
frac=nnz/cell_counts.reshape(1,-1)
# filter genes: fraction expressed in at least 3 donors > 0.1 maybe
keep=(frac>0.05).sum(axis=1)>=3
print('genes kept', keep.sum())
X=logcpm[keep,:].T # donors x genes
# zscore genes
from sklearn.preprocessing import StandardScaler
Xz=StandardScaler().fit_transform(X)
# PCA
pca=PCA(n_components=min(20,X.shape[0]), random_state=0)
scores=pca.fit_transform(Xz)
print('explained variance', pca.explained_variance_ratio_[:10])
# donor PC table
pcdf=pd.DataFrame(scores[:,:10], index=cats, columns=[f'PC{i+1}' for i in range(10)])
pcdf['donor']=cats
pcdf.to_csv('donor_pca.csv', index=False)
# kmeans k=2 on first N PCs
for n in [2,3,5,10,20]:
    km=KMeans(n_clusters=2, n_init=20, random_state=0).fit(scores[:,:n])
    for lab in range(2):
        members=sorted([str(c) for c,l in zip(cats,km.labels_) if l==lab])
        print('kmeans nPC',n,'cluster',lab,len(members),members)
    # count high ISG in each cluster later
# print first 5 pcs for all donors
for c,s in zip(cats,scores):
    print(c, ' '.join(f'{v:.2f}' for v in s[:5]))
PY
python cluster_donors.py"
