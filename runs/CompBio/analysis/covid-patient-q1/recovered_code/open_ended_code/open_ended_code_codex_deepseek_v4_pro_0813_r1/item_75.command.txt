/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('group_sums.rds')
for(n in c('sums_Donor','sums_sample_id','sums_sample_name','sums_Subject','sums_sample')) {
  m <- as.matrix(x[[n]])
  out <- sub('"'^sums_'"','group_sums_', n)
  write.csv(m, paste0(out,'.csv'))
  cat(n, dim(m), '\\n')
}
write.csv(x"'$meta, '"'group_meta.csv')
RS
ls -lh group_sums_*.csv group_meta.csv"
