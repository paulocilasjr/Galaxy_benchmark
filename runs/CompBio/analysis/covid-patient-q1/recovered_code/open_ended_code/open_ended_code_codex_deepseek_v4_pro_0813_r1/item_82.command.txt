import pandas as pd, numpy as np
m=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False)
our=np.load('donor_codes.npy')
def canon(seq):
    mp={}; out=np.empty(len(seq),dtype=np.int32); nxt=0
    for i,v in enumerate(seq):
        if v not in mp: mp[v]=nxt; nxt+=1
        out[i]=mp[v]
    return out
our_canon=canon(our)
sl_codes,_=pd.factorize(m['sample_label'])
mask=(m['Class']=='COVID')&(m['specific_gating']=='ungated')
seq=sl_codes[mask.values]; seq=seq[seq>=0]
can=canon(seq)
print(len(seq),len(our_canon),len(can))
# Greedy subsequence
j=0; skips=0; matches=0
for i,v in enumerate(can):
    if j<len(our_canon) and v==our_canon[j]:
        j+=1; matches+=1
    else:
        skips+=1
print('greedy matches',matches,'of',len(our_canon),'skips',skips,'j',j)
# Try reverse direction? no.
# Show first mismatch context
j=0; mism=[]
for i,v in enumerate(can):
    if j>=len(our_canon): break
    if v==our_canon[j]:
        j+=1
    else:
        if len(mism)<20:
            mism.append((i,j,v,our_canon[j],m.loc[mask].index[i]))
print('mism contexts',mism)
PY"
