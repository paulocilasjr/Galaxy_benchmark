/bin/bash -lc "cat > assign_donors.py <<'PY'
import pandas as pd,numpy as np,anndata as ad
from sklearn.metrics.pairwise import cosine_similarity
from scipy.optimize import linear_sum_assignment
A=ad.read_h5ad('covid.patients.q1.h5ad',backed='r'); wnames=np.array([str(x) if x is not None else '' for x in A.var['feature_name']]); wl=np.char.lower(wnames.astype('U')); cats=np.load('donor_sum.npz')['cats']
meta=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False)
for f,colname,idtype in [('group_sums_Donor.csv','Donor','Donor'),('group_sums_sample_name.csv','sample_name','sample_name'),('group_sums_sample_id.csv','sample_id','sample_id'),('group_sums_sample.csv','sample','sample')]:
 orig=pd.read_csv(f,index_col=0); onames=np.array([str(x) for x in orig.index]); ocols=np.array(orig.columns.astype(str)); O=orig.values.astype(float); ol=np.char.lower(onames.astype('U')); common,wi,oi=np.intersect1d(wl,ol,return_indices=True)
 D=np.load('donor_sum.npz'); W=D['sum'][wi,:].astype(float); Or=O[oi,:].astype(float)
 def logcpm(X): return np.log1p(X/X.sum(axis=0,keepdims=True)*1e6)
 W=logcpm(W); Or=logcpm(Or); Wz=(W-W.mean(1,keepdims=True))/(W.std(1,keepdims=True)+1e-8); Oz=(Or-Or.mean(1,keepdims=True))/(Or.std(1,keepdims=True)+1e-8)
 def cls_for(c):
  sub=meta[meta[colname]==c]
  return sub['Class'].mode().iloc[0] if len(sub) else 'NA'
 labels=np.array([cls_for(c) for c in ocols])
 # Pearson and cosine cost matrices
 for metric in ['cos','pear']:
  if metric=='cos': sim=cosine_similarity(Wz.T,Oz.T)
  else: sim=np.corrcoef(Wz.T,Oz.T)[:Wz.shape[1],Wz.shape[1]:]
  cost=-sim
  # if more original than working, Hungarian rectangular (scipy supports)
  r,c=linear_sum_assignment(cost)
  print('\\n',f,idtype,metric,'total sim',sim[r,c].sum(),'nassign',len(r))
  assigned=[]
  for j,ocol in zip(r,c):
   d=int(cats[j][5:]); assigned.append((d,ocols[ocol],labels[ocol],round(sim[j,ocol],3)))
  print(assigned)
  hc=sorted([d for d,o,l,s in assigned if l=='HC'])
  print('HC count',len(hc),hc)
PY
python3 assign_donors.py"
