/bin/bash -lc "python3 - <<'PY'
import pandas as pd
m=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False)
sub=m.drop_duplicates('Donor')[['Donor','Class','Gender']]
counts=m.groupby('Donor').size()
sub['n']=sub['Donor'].map(counts)
for thr in [0,2000,4000,5000,6000]:
 sel=sub[sub.n>=thr]
 print('thr',thr,'n',len(sel),sel.groupby(['Class','Gender']).size().to_dict())
 print('donors',sel.Donor.tolist())
PY"
