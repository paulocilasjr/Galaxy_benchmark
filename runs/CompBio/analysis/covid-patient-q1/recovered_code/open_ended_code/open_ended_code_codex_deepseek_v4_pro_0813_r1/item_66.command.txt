/bin/bash -lc "cat > extract_seurat_meta.R <<'RS'
library(Seurat)
args <- commandArgs(trailingOnly=TRUE)
infile <- if(length(args)>=1) args[1] else 'GSE161918_AllBatches_SeuratObj.rds'
outfile <- if(length(args)>=2) args[2] else 'seurat_meta.csv'
cat('reading', infile, '\\n')
t0 <- Sys.time()
obj <- readRDS(infile)
cat('read in', format(difftime(Sys.time(), t0, units='mins')), '\\n')
cat('class', paste(class(obj), collapse=','), '\\n')
meta <- obj@meta.data
cat('meta dim', dim(meta), '\\n')
write.csv(meta, outfile, row.names=TRUE)
cat('wrote', outfile, '\\n')
cat('assay names', paste(names(obj@assays), collapse=','), '\\n')
for(n in names(obj@assays)) {
  cat('assay', n, 'class', paste(class(obj@assays[[n]]), collapse=','), 'dim', paste(dim(obj@assays[[n]]), collapse='x'), '\\n')
}
RS
Rscript extract_seurat_meta.R GSE161918_AllBatches_SeuratObj.rds seurat_meta.csv"
