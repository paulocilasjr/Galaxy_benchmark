/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('repo/A4/input/healthy_vs_covid_COVID-Healthy_module_score_gsva_filtered_samples_genes.rds')
# classify sample: HC vs COVID T0
x"'$group <- ifelse(grepl('"'_HC_', x"'$sample), '"'HC', ifelse(grepl('_T0_', x"'$sample), '"'COVID_T0', 'OTHER'))
y <- subset(x, group %in% c('HC','COVID_T0'))
# For each pathway celltype, mean score per group and diff
agg <- aggregate(module.score ~ pathway + celltype + group, data=y, FUN=mean)
wide <- reshape(agg, idvar=c('pathway','celltype'), timevar='group', direction='wide')
wide"'$diff <- wide$module.score.COVID_T0 - wide$module.score.HC
wide <- wide[order(-abs(wide$diff)),]
cat('"'n rows', nrow(wide), '\\n')
print(head(wide, 30))
# Overall pathway mean across celltypes
agg2 <- aggregate(diff ~ pathway, data=wide, FUN=mean)
agg2 <- agg2[order(-abs(agg2"'$diff)),]
print(head(agg2, 50))
saveRDS(wide, '"'pathway_celltype_diff.rds')
saveRDS(agg2, 'pathway_diff.rds')
RS"
