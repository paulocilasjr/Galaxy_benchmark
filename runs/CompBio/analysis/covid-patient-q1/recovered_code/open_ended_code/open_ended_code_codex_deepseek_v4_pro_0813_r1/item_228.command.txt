/bin/bash -lc "python3 - <<'PY'
import requests,json
q='\"33 samples\" AND \"PBMC\" AND \"single-cell\" AND \"GSE\"[ETYP]'
r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':q,'retmax':100,'retmode':'json'},timeout=30)
j=r.json()['esearchresult']; print(j.get('count'),j.get('idlist'))
ids=j.get('idlist',[])
r2=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=30)
for id in ids:
 it=r2.json()['result'].get(id)
 if it: print(id,it.get('accession'),it.get('title'))
PY"
