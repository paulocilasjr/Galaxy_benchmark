/bin/bash -lc "python3 - <<'PY'
import numpy as np, pandas as pd, anndata as ad
adata=ad.read_h5ad('covid.patients.q1.h5ad',backed='r')
names=np.array([x for x in adata.var['feature_name']])
D=np.load('donor_sum.npz'); sums=D['sum']; cc=D['cell_counts']; cats=D['cats']
lib=sums.sum(axis=0,keepdims=True); cpm=sums/lib*1e6; log=np.log1p(cpm)
mu=log.mean(axis=1,keepdims=True); sd=log.std(axis=1,keepdims=True); sd[sd==0]=1
Z=(log-mu)/sd
for g in ['XIST','RPS4Y1','DDX3Y','UTY','KDM5D','ZFY','USP9Y','EIF1AY','TXLNGY','NLGN4Y','TSIX','LINC00278']:
 arr=np.where(names==g)[0]
 if arr:
  vals=Z[arr[0]]
  print(g, np.array2string(vals, precision=2, max_line_width=1000))
# score ratio
def sc(g):
 arr=np.where(names==g)[0]; return Z[arr[0]] if len(arr) else np.zeros(len(cats))
X=sc('XIST'); Y=sc('RPS4Y1')+sc('DDX3Y')+sc('UTY')+sc('KDM5D')+sc('ZFY')+sc('USP9Y')+sc('EIF1AY')
score=X-Y
for d,x,y,s in zip(cats,X,Y,score):
 print(d,'X',round(x,2),'Ysum',round(y,2),'score',round(s,2))
PY"
