/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, anndata as ad
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import cross_val_predict, StratifiedKFold
from sklearn.metrics import roc_auc_score, accuracy_score, confusion_matrix

sums=pd.read_csv('group_sums_sample_id.csv',index_col=0)
meta=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False)
samples=sums.columns
smeta=meta.groupby('sample_id').agg(Class=('Class','first'), Timepoint=('Timepoint','first')).loc[samples]
genes=['ISG15','IFI27','IFI44L','IFIT1','IFIT2','IFIT3','IFIT5','MX1','MX2','OAS1','OAS2','OAS3','OASL','IFI6','IFIH1','IFI44','IFI35','IFITM1','IFITM2','IFITM3','ISG20','HLA-DRA','HLA-DRB1','CD14','LYZ','S100A8','S100A9','FCGR3A','CD3D','CD8A','MS4A1','IGHG1','JCHAIN','NKG7','GNLY','PPBP','PF4','IL1B','TNF','CXCL10','CCL2']
present=[g for g in genes if g in sums.index]
lib=sums.sum(axis=0)
logcpm=np.log1p(sums.loc[present]/lib*1e6).T
ct=meta.groupby(['sample_id','coursecelltype']).size().unstack(fill_value=0)
ct=ct.div(ct.sum(axis=1),axis=0).loc[samples]
X=pd.concat([logcpm.reset_index(drop=True), ct.reset_index(drop=True)],axis=1)
X.columns=list(present)+list(ct.columns)
X.index=samples
train_mask=((smeta.Class=='HC') | (smeta.Timepoint=='T0')).values
Xtr=X[train_mask]; ytr=(smeta.Class[train_mask]=='COVID').astype(int)
print('train n',len(Xtr),'pos',ytr.sum())
sc=StandardScaler().fit(Xtr); Xtr_s=sc.transform(Xtr)
cv=StratifiedKFold(n_splits=5,shuffle=True,random_state=0)
for clf,name in [(LogisticRegression(max_iter=2000,class_weight='balanced',C=1.0),'LR'),(RandomForestClassifier(n_estimators=500,random_state=0,class_weight='balanced'),'RF')]:
    pred=cross_val_predict(clf,Xtr_s,ytr,cv=cv,method='predict_proba')[:,1]
    predc=(pred>0.5).astype(int)
    print(name,'AUC',roc_auc_score(ytr,pred),'acc',accuracy_score(ytr,predc),'conf',confusion_matrix(ytr,predc))

adata=ad.read_h5ad('covid.patients.q1.h5ad')
our_names=np.array(adata.var['feature_name'].astype(str))
D=np.load('donor_sum.npz'); sums_o=D['sum']; cats=D['cats']; cc=D['cell_counts']
lib_o=sums_o.sum(axis=0); logcpm_o=np.log1p(sums_o/lib_o*1e6)
freq=pd.read_csv('celltype_freq.csv',index_col=0)
genes_o=[g for g in present if g in our_names]
Xo=pd.DataFrame(logcpm_o.T,index=cats,columns=our_names)[genes_o]
Xo_final=pd.DataFrame(index=cats)
for col in X.columns:
    if col in Xo.columns: Xo_final[col]=Xo[col]
    elif col in freq.columns: Xo_final[col]=freq[col].values
    else: Xo_final[col]=0
Xo_s=sc.transform(Xo_final)
for clf,name in [(LogisticRegression(max_iter=2000,class_weight='balanced',C=1.0),'LR'),(RandomForestClassifier(n_estimators=500,random_state=0,class_weight='balanced'),'RF')]:
    clf.fit(Xtr_s,ytr)
    prob=clf.predict_proba(Xo_s)[:,1]
    print('\\n',name,'our predictions')
    for c,p in zip(cats,prob):
        print(c,round(p,3), 'COVID' if p>=0.5 else 'HC')
PY"
