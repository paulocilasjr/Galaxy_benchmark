/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse
queries=['MIM:617658 gene','OMIM 617658 BEND5','BEN domain containing 5 OMIM']
for q in queries:
    u='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gene&term='+urllib.parse.quote(q)+'&retmode=json'
    try:
      data=urllib.request.urlopen(u,timeout=30).read().decode(); print(q,data)
    except Exception as e: print(q,'ERR',e)
PY"
