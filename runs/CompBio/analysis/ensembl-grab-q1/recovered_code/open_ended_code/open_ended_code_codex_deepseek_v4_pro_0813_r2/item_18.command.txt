/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
ids=[90527,145645,90525,9153,84419,58472,79656]
# use GET each query maybe rate
for id in ids:
 u=f'https://mygene.info/v3/query?q=entrezgene:{id}&fields=symbol,ensembl.gene,name'
 try:
  j=json.loads(urllib.request.urlopen(u,timeout=30).read().decode())
  print(id, [(h.get('symbol'),h.get('ensembl',{}).get('gene')) for h in j.get('hits',[])])
 except Exception as e: print(id,'ERR',e)
PY"
