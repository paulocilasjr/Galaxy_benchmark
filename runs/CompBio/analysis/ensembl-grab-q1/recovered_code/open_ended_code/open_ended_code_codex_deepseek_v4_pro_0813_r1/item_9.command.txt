/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
# verify several via Ensembl xref / lookup
items=['NM_001276266.2','TERB2','NP_922946.1','9153','617658']
for q in items:
    u='https://rest.ensembl.org/xrefs/symbol/homo_sapiens/'+urllib.parse.quote(q)+'?content-type=application/json'
    try:
        print(q, json.load(urllib.request.urlopen(u,timeout=20))[:5])
    except Exception as e:
        print('ERR',q,e)
PY"
