/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, time
ids=['NM_001276266.2','TERB2','ENST00000267814','chr15:45167214-45187966','GeneID:9153','NP_922946.1','MIM:617658','BEN domain containing 5, transcript variant 6']
for q in ids:
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gene&retmode=json&term='+urllib.parse.quote(q)
    try:
        with urllib.request.urlopen(url,timeout=20) as r:
            data=json.load(r)
        print('\\nQUERY',q)
        print(data.get('esearchresult',{}).get('idlist'))
    except Exception as e:
        print('ERR',q,e)
    time.sleep(.4)
PY"
