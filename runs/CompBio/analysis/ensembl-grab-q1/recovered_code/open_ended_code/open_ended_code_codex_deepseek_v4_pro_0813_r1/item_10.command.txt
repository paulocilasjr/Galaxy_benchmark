/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
# SHF lookup
u='https://rest.ensembl.org/lookup/symbol/homo_sapiens/SHF?expand=1;content-type=application/json'
print(json.load(urllib.request.urlopen(u,timeout=20)))
# BEND5 lookup
u='https://rest.ensembl.org/lookup/symbol/homo_sapiens/BEND5?expand=1;content-type=application/json'
d=json.load(urllib.request.urlopen(u,timeout=20))
print(d.get('id'), d.get('start'), d.get('end'))
PY"
