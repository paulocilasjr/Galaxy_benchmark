/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
for q in ['9153','chr15:45167214-45187966','NM_001276266','BEN domain containing 5 transcript variant 6']:
    url='https://mygene.info/v3/query?q='+urllib.parse.quote(q)+'&fields=symbol,name,ensembl.gene,taxid&species=human&size=10'
    try:
        data=json.load(urllib.request.urlopen(url,timeout=20))
        print('\\n',q)
        for h in data.get('hits',[]):
            print(h.get('_id'),h.get('symbol'),h.get('name'),h.get('ensembl'))
    except Exception as e: print('ERR',q,e)
PY"
