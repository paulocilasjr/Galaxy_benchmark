/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
queries=['NM_001276266.2','TERB2','ENST00000267814','GeneID:9153','NP_922946.1','MIM:617658','BEND5']
for q in queries:
    url='https://mygene.info/v3/query?q='+urllib.parse.quote(q)+'&fields=symbol,name,ensembl.gene,taxid&species=human&size=5'
    try:
        data=json.load(urllib.request.urlopen(url,timeout=20))
        print('\\n',q)
        for h in data.get('hits',[]):
            print(h.get('_id'),h.get('symbol'),h.get('name'),h.get('ensembl'))
    except Exception as e: print('ERR',q,e)
PY"
