/bin/bash -c "cat > ascii_xz_binary.py <<'PY'
import numpy as np
from scipy.ndimage import binary_dilation
coords=np.load('ca_coords.npy'); chains=np.load('ca_chains.npy'); coords-=coords.mean(0)
paths=[coords[chains==ch] for ch in np.unique(chains)]
# XZ projection, use x,z; y vertical z
uv=coords[:,[0,2]]
umin,umax=uv[:,0].min(),uv[:,0].max(); vmin,vmax=uv[:,1].min(),uv[:,1].max()
W=140;H=50
span=max(umax-umin,vmax-vmin)
scaleu=(W-1)/span; scalev=(H-1)/span
img=np.zeros((H,W),dtype=bool)
for p in paths:
    u=(p[:,0]-(umin+umax)/2)*scaleu+(W-1)/2
    v=(p[:,1]-(vmin+vmax)/2)*scalev+(H-1)/2
    for k in range(len(u)-1):
        steps=max(1,int(np.hypot(u[k+1]-u[k],v[k+1]-v[k])*2)+1)
        tt=np.linspace(0,1,steps); xs=np.rint(u[k]+tt*(u[k+1]-u[k])).astype(int); ys=np.rint(v[k]+tt*(v[k+1]-v[k])).astype(int)
        valid=(xs>=0)&(xs<W)&(ys>=0)&(ys<H); img[ys[valid],xs[valid]]=1
    xs=np.clip(np.rint(u).astype(int),0,W-1); ys=np.clip(np.rint(v).astype(int),0,H-1); img[ys,xs]=1
# print top row v=H-1 down to 0? We set v from z; row index = v. We'll print row 0..H; row0 corresponds zmax? Actually v=(z-center)*scale + 24.5, row index increases downward, row0 = z high, rowH z low.
# dilate for visibility
img=binary_dilation(img,iterations=1)
chars=' .:-=+*#%@'
for r in range(H):
    line=''
    for c in range(W):
        line += '#' if img[r,c] else ' '
    print(f'{r:2d} '+line)
# print labels endpoints/chain centers
print('z top is first row; x left to right')
for ch in np.unique(chains):
 p=coords[chains==ch]
 u0=(p[0,0]-(umin+umax)/2)*scaleu+(W-1)/2; v0=(p[0,2]-(vmin+vmax)/2)*scalev+(H-1)/2
 u1=(p[-1,0]-(umin+umax)/2)*scaleu+(W-1)/2; v1=(p[-1,2]-(vmin+vmax)/2)*scalev+(H-1)/2
 print(ch,'start row,col',round(v0,1),round(u0,1),'end',round(v1,1),round(u1,1))
PY
python ascii_xz_binary.py"
