/bin/bash -c "cat > chamfer_match.py <<'PY'
import numpy as np, time, glob
from PIL import Image, ImageDraw, ImageFont
from scipy.ndimage import rotate as ndrotate, distance_transform_edt
coords=np.load('ca_coords.npy'); chains=np.load('ca_chains.npy')
coords=coords-coords.mean(0)
paths=[coords[chains==ch] for ch in np.unique(chains)]
letters=list('BDFHJLNPRTVXZ')
N=64

def make_letter(ch, fontpath='/System/Library/Fonts/Supplemental/Arial.ttf'):
    font=ImageFont.truetype(fontpath, N*3)
    M=N*3
    im=Image.new('L',(M,M),0); d=ImageDraw.Draw(im)
    d.text((M//5,M//5),ch,fill=255,font=font)
    a=np.array(im)>127
    ys,xs=np.where(a)
    crop=Image.fromarray((a[ys.min():ys.max()+1,xs.min():xs.max()+1]*255).astype(np.uint8))
    span=max(xs.max()-xs.min()+1, ys.max()-ys.min()+1)
    scale=(N-2)/span
    w=max(1,int(round((xs.max()-xs.min()+1)*scale))); h=max(1,int(round((ys.max()-ys.min()+1)*scale)))
    crop=crop.resize((w,h),Image.Resampling.LANCZOS)
    out=np.zeros((N,N),dtype=np.uint8); y=(N-h)//2; x=(N-w)//2
    out[y:y+h,x:x+w]=(np.array(crop)>127).astype(np.uint8)
    return out

letter_masks={ch:make_letter(ch) for ch in letters}
# precompute rotations and dt
rot_step=15
angles=list(range(0,360,rot_step))
rot_masks={ch:[] for ch in letters}
rot_dt={ch:[] for ch in letters}
for ch,lm in letter_masks.items():
    for ang in angles:
        r=ndrotate(lm.astype(np.uint8),ang,reshape=False,order=0)>0.5
        # crop to bbox of rotated and re-normalize maybe
        if r.sum()==0: r=np.zeros_like(lm)
        rot_masks[ch].append(r)
        rot_dt[ch].append(distance_transform_edt(~r))

def basis_from_n(n):
    n=n/np.linalg.norm(n)
    if abs(n[0])<0.9:
        e1=np.cross(n,[1,0,0]); e1=e1/np.linalg.norm(e1)
    else:
        e1=np.cross(n,[0,1,0]); e1=e1/np.linalg.norm(e1)
    e2=np.cross(n,e1)
    return np.vstack([e1,e2])

def project_img(n):
    basis=basis_from_n(n); uv=coords@basis.T
    umin,umax=uv[:,0].min(),uv[:,0].max(); vmin,vmax=uv[:,1].min(),uv[:,1].max()
    span=max(umax-umin,vmax-vmin,1e-9)
    scale=(N-1)/span
    img=np.zeros((N,N),dtype=np.uint8)
    for p in paths:
        u=(p[:,0]-(umin+umax)/2)*scale + (N-1)/2
        v=(p[:,1]-(vmin+vmax)/2)*scale + (N-1)/2
        for k in range(len(u)-1):
            steps=max(1,int(np.hypot(u[k+1]-u[k],v[k+1]-v[k])*1.5)+1)
            tt=np.linspace(0,1,steps)
            xs=np.rint(u[k]+tt*(u[k+1]-u[k])).astype(int); ys=np.rint(v[k]+tt*(v[k+1]-v[k])).astype(int)
            valid=(xs>=0)&(xs<N)&(ys>=0)&(ys<N)
            img[ys[valid],xs[valid]]=1
        xs=np.clip(np.rint(u).astype(int),0,N-1); ys=np.clip(np.rint(v).astype(int),0,N-1)
        img[ys,xs]=1
    return img

def fib_sphere(n):
    i=np.arange(n); phi=np.arccos(1-2*(i+0.5)/n); theta=np.pi*(1+5**0.5)*i
    return np.stack([np.sin(phi)*np.cos(theta), np.sin(phi)*np.sin(theta), np.cos(phi)],axis=1)

dirs=fib_sphere(600)
best={ch:(1e9,None,None) for ch in letters}
# precompute protein dt for each dir? compute per dir
start=time.time()
for di,n in enumerate(dirs):
    img=project_img(n)
    if img.sum()==0: continue
    dtI=distance_transform_edt(~(img>0))
    areaI=img.sum()
    for ch in letters:
        for j,(rm,dtL) in enumerate(zip(rot_masks[ch],rot_dt[ch])):
            areaL=rm.sum()
            if areaL==0: continue
            # symmetric average chamfer; use mean distances
            d1=np.sum(img*dtL)/areaI
            d2=np.sum(rm*dtI)/areaL
            score=(d1+d2)/2
            if score<best[ch][0]:
                best[ch]=(score, img.copy(), angles[j])
    if (di+1)%100==0:
        print('dir',di+1,'elapsed',round(time.time()-start,1),{ch:round(v[0],3) for ch,v in best.items()},flush=True)
print('FINAL chamfer best', {ch:(round(v[0],3),v[2]) for ch,v in best.items()})
rank=sorted(best.items(), key=lambda kv:kv[1][0])
print('rank',[(ch,round(s,3),ang) for ch,(s,img,ang) in rank])
for ch,(s,img,ang) in best.items():
    Image.fromarray((img*255).astype(np.uint8)).save(f'chamfer_best_{ch}.png')
PY
python chamfer_match.py"
