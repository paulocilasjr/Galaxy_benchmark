/bin/bash -c "cat > ascii_proj.py <<'PY'
import numpy as np
coords=np.load('ca_coords.npy'); chains=np.load('ca_chains.npy')
# choose projection axes, center and grid
for name,(i,j) in [('XY',(0,1)),('XZ',(0,2)),('YZ',(1,2))]:
    X=coords[:,i]; Y=coords[:,j]
    # bin into 80x40 by data range
    W=100; H=40
    xmin,xmax=X.min(),X.max(); ymin,ymax=Y.min(),Y.max()
    grid=np.zeros((H,W),dtype=int)
    # scale
    xi=((X-xmin)/(xmax-xmin)*(W-1)).astype(int)
    yi=((ymax-Y)/(ymax-ymin)*(H-1)).astype(int)
    np.add.at(grid,(yi,xi),1)
    chars=' .:-=+*#%@'
    print('\\n===',name,'x',round(xmin),round(xmax),'y',round(ymin),round(ymax))
    for r in range(H):
        line=''
        for c in range(W):
            v=grid[r,c]
            idx=min(9,int(v*9/max(1,grid.max())))
            line+=chars[idx]
        print(line)
PY
python ascii_proj.py"
