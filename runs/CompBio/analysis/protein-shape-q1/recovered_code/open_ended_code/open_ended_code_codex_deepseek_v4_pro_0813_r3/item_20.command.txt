/bin/bash -c "cat > chain_geom.py <<'PY'
import numpy as np
coords=np.load('ca_coords.npy'); chains=np.load('ca_chains.npy')
coords=coords-coords.mean(0)
for ch in np.unique(chains):
 p=coords[chains==ch]
 c=p.mean(0)
 # PCA
 u,s,vt=np.linalg.svd(p-c, full_matrices=False)
 print('\\nChain',ch,'n',len(p),'center',np.round(c,1),'extent',np.round(p.max(0)-p.min(0),1))
 print('singular',np.round(s,1),'ratios',np.round(s/s[0],2))
 print('first',np.round(vt[0],2),'second',np.round(vt[1],2),'third',np.round(vt[2],2))
 print('start',np.round(p[0],1),'end',np.round(p[-1],1))
 print('min/max coords',np.round(p.min(0),1),np.round(p.max(0),1))
# pair distances between chain endpoints and closest
names=np.unique(chains)
print('\\nPairwise closest distances and mean between CA')
for i in range(len(names)):
 for j in range(i+1,len(names)):
  a=coords[chains==names[i]]; b=coords[chains==names[j]]
  # distance matrix via scipy cdist
  from scipy.spatial.distance import cdist
  d=cdist(a,b)
  print(names[i],names[j],'min',round(d.min(),1),'mean',round(d.mean(),1), 'closest idx', np.unravel_index(d.argmin(),d.shape), 'start-end',round(np.linalg.norm(a[0]-b[-1]),1),round(np.linalg.norm(a[-1]-b[0]),1))
PY
python chain_geom.py"
