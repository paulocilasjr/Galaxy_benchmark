import numpy as np
from collections import Counter
p='protein.shape.q1.pdb'
rows=[]
for line in open(p):
    if line.startswith('ATOM'):
        rows.append((line[21],line[12:16].strip(),*map(float,(line[30:38],line[38:46],line[46:54]))))
print('atoms',len(rows),'chains',Counter(x[0] for x in rows),'names',Counter(x[1] for x in rows).most_common(10))
for what,pts in [('all',np.array([x[2:] for x in rows])),('CA',np.array([x[2:] for x in rows if x[1]=='CA']))]:
 print(what,len(pts),'min',pts.min(0),'max',pts.max(0),'span',np.ptp(pts,0))
 c=pts-pts.mean(0); w,v=np.linalg.eigh(c.T@c/len(c)); print('eig',w,'vectors',v)
PY"
