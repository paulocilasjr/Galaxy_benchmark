/bin/bash -lc "python - <<'PY'
from collections import defaultdict
import numpy as np
f='protein.shape.q1.pdb'
chains=defaultdict(list)
res=defaultdict(dict)
for line in open(f):
    if line.startswith('ATOM'):
        chain=line[21]
        xyz=np.array([float(line[30:38]),float(line[38:46]),float(line[46:54])])
        chains[chain].append(xyz)
        rn=int(line[22:26])
        atom=line[12:16].strip()
        res[(chain,rn)][atom]=xyz
for c,pts in chains.items():
    a=np.array(pts)
    print(c,len(a),'res',len({k[1] for k in res if k[0]==c}),'range',a.min(0),a.max(0),'span',a.ptp(0),'centroid',a.mean(0))
print('all',sum(map(len,chains.values())))
PY"
