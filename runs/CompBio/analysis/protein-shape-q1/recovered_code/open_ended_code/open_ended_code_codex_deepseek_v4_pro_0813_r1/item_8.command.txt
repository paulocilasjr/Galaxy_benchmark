def ascii_art(x,y,w=72,h=32):
    xmin,xmax=x.min(),x.max(); ymin,ymax=y.min(),y.max()
    sx=(w-1)/(xmax-xmin); sy=(h-1)/(ymax-ymin)
    grid=np.zeros((h,w),bool)
    for xx,yy in zip(x,y):
        col=int((xx-xmin)*sx); row=int((yy-ymin)*sy)
        if 0<=row<h and 0<=col<w: grid[row,col]=True
    return grid
