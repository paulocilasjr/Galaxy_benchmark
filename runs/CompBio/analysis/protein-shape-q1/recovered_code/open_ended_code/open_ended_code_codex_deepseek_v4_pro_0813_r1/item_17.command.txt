/bin/bash -c "python3 - <<'PY'
import urllib.request
ids=['2qyc','4j3o','3j04','1xu9','1b3u','1ueb','1z85','3afc','2arp','3e98','3h90','1w3b','4bta']
for pid in ids:
    url=f'https://files.rcsb.org/download/{pid}.pdb'
    try:
        txt=urllib.request.urlopen(url,timeout=30).read().decode('ascii','ignore')
    except Exception as e:
        print(pid,'ERR',e); continue
    chains={}
    for line in txt.splitlines():
        if line.startswith('ATOM'):
            if line[12:16].strip()=='CA':
                chains[line[21]]=chains.get(line[21],0)+1
    # residue IDs set maybe
    print(pid, chains)
PY"
