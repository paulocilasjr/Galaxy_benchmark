/bin/bash -c "python3 - <<'PY'
from collections import Counter
p='protein.shape.q1.pdb'
counts=Counter(); chains=Counter(); residues=set(); ca=[]
for line in open(p):
    if line.startswith('ATOM') or line.startswith('HETATM'):
        rec=line[0:6].strip(); serial=line[6:11].strip(); name=line[12:16].strip(); alt=line[16]; res=line[17:20].strip(); chain=line[21]; resid=line[22:26].strip(); x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54])
        counts[rec]+=1; chains[chain]+=1; residues.add((chain,resid,res))
        if name=='CA': ca.append((chain,resid,res,x,y,z))
print(counts)
print('chains', chains)
print('residues', len(residues), 'CA', len(ca))
for c,n in chains.most_common(): print(c,n)
# bounding boxes
import numpy as np
for c in chains:
    pts=np.array([(x,y,z) for cc,_,_,x,y,z in ca if cc==c])
    if len(pts): print(c, pts.min(0), pts.max(0), pts.ptp(0), len(pts))
PY"
