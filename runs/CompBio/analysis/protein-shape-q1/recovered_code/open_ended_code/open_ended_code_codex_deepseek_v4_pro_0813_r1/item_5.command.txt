/bin/bash -c "python3 - <<'PY'
from PIL import Image
files=['proj_pc_0.png','proj_pc_1.png','proj_pc_2.png','proj_raw_01.png','proj_raw_02.png','proj_raw_12.png']
ims=[Image.open(f) for f in files]
w,h=ims[0].size
cols=3; rows=2; pad=10
out=Image.new('RGB',(cols*w+(cols+1)*pad, rows*h+(rows+1)*pad),'white')
for i,im in enumerate(ims):
    r,c=divmod(i,cols)
    out.paste(im,(pad+c*(w+pad),pad+r*(h+pad)))
out.save('montage.png')
PY
file montage.png"
