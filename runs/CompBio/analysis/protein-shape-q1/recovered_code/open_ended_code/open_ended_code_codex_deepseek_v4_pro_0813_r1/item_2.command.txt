/bin/bash -c "python3 - <<'PY'
import numpy as np
from collections import defaultdict
p='protein.shape.q1.pdb'
pts=[]
for line in open(p):
    if line.startswith('ATOM'):
        name=line[12:16].strip()
        if name=='CA':
            x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54])
            pts.append((x,y,z))
pts=np.array(pts)
print(pts.shape, pts.min(0), pts.max(0), pts.ptp(0))
# pca
c=pts.mean(0); U,S,Vt=np.linalg.svd(pts-c)
print('means',c); print('S',S); print('Vt',Vt)
# projections along principal axes (columns? coordinates in PC)
coords=(pts-c)@Vt.T
# save pngs using matplotlib
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
for i in range(3):
    j,k=[q for q in range(3) if q"'!=i]
    plt.figure(figsize=(6,6)); plt.scatter(coords[:,j],coords[:,k],s=.1,color='"'k',marker='o',linewidths=0)
    plt.gca().set_aspect('equal'); plt.title(f'PC j,k drop {i}'); plt.savefig(f'proj_pc_{i}.png',dpi=150); plt.close()
# also raw xy/xz/yz
pairs=[(0,1),(0,2),(1,2)]
for i,(a,b) in enumerate(pairs):
    plt.figure(figsize=(6,6)); plt.scatter(pts[:,a],pts[:,b],s=.1,color='k',marker='o',linewidths=0)
    plt.gca().set_aspect('equal'); plt.title(f'raw {a}{b}'); plt.savefig(f'proj_raw_{a}{b}.png',dpi=150); plt.close()
PY"
