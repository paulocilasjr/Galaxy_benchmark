/bin/bash -lc "python3 - <<'PY'
import numpy as np
import matplotlib.pyplot as plt
coords=[]
with open('protein.shape.q1.pdb') as f:
    for line in f:
        if line.startswith(('ATOM  ','HETATM')):
            coords.append([float(line[30:38]),float(line[38:46]),float(line[46:54])])
A=np.array(coords); cen=A.mean(0); X=A-cen
w,V=np.linalg.eigh(np.cov(X.T)); V=V[:,np.argsort(w)[::-1]]
# coordinates in PCA frame
Y=X@V
x=Y[:,0]; y0=Y[:,1]; z0=Y[:,2]
angles=np.linspace(0,180,18,endpoint=False)
fig,axs=plt.subplots(3,6,figsize=(18,8),dpi=130)
for ax,deg in zip(axs.ravel(),angles):
    th=np.deg2rad(deg)
    y=np.cos(th)*y0+np.sin(th)*z0
    ax.scatter(x,y,s=.12,c='black',alpha=.22)
    ax.set_aspect('equal','box'); ax.axis('off'); ax.set_title(f'{deg:.0f} deg',fontsize=8)
fig.savefig('outputs/long_axis_rotation_all.png',bbox_inches='tight',pad_inches=.05)
plt.close(fig)
print('wrote outputs/long_axis_rotation_all.png')
PY"
