from collections import defaultdict
import numpy as np
p='protein.shape.q1.pdb'
d=defaultdict(list)
allca=[]
with open(p) as f:
 for line in f:
  if line.startswith('ATOM') and line[12:16].strip()=='CA':
   xyz=[float(line[i:j]) for i,j in [(30,38),(38,46),(46,54)]]
   ch=line[21]
   d[ch].append(xyz); allca.append(xyz)
A=np.array(allca)
print('CA',len(A),'chains',list(d))
print('bounds',A.min(0),A.max(0),'span',np.ptp(A,axis=0))
w,v=np.linalg.eigh(np.cov(A.T)); ix=np.argsort(w)[::-1]
print('PCA eig',w[ix], 'vectors rows')
print(v[:,ix].T)
for ch,x in d.items():
 x=np.array(x)
 print(ch,len(x),'centroid',np.round(x.mean(0),1),'span',np.round(np.ptp(x,axis=0),1),'res endpoints',np.round(x[[0,-1]],1))
PY"
