/bin/bash -lc "python3 - <<'PY'
import numpy as np, matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from collections import defaultdict
chains=defaultdict(list)
res_ca=[]
for line in open('protein.shape.q1.pdb'):
    if line.startswith(('ATOM','HETATM')) and line[12:16].strip()=='CA':
        ch=line[21]; res=int(line[22:26]); x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54])
        chains[ch].append((res,x,y,z))
        res_ca.append((x,y,z))
print({k:len(v) for k,v in chains.items()})
for k,v in chains.items():
    arr=np.array([(x,y,z) for _,x,y,z in v])
    print(k,'min',arr.min(0),'max',arr.max(0),'range',arr.ptp(0),'center',arr.mean(0))
fig,axes=plt.subplots(2,3,figsize=(18,10),facecolor='white')
colors=dict(A='tab:blue',B='tab:orange',C='tab:green',D='tab:red',E='tab:purple',F='tab:brown')
for ax,(i,j,name) in zip(axes.flat,[(0,1,'XY'),(0,2,'XZ'),(1,2,'YZ'),(0,1,'XY lines'),(0,2,'XZ lines'),(1,2,'YZ lines')]):
    for ch,vals in sorted(chains.items()):
        arr=np.array([(x,y,z) for _,x,y,z in vals])
        if 'lines' in name:
            ax.plot(arr[:,i],arr[:,j],lw=.8,alpha=.75,color=colors.get(ch),label=ch)
        else:
            ax.scatter(arr[:,i],arr[:,j],s=2,color=colors.get(ch),alpha=.8,label=ch)
    ax.set_aspect('equal','box'); ax.axis('off'); ax.set_title(name)
    ax.legend(markerscale=3,fontsize=8,loc='upper right')
plt.tight_layout(); fig.savefig('chain_ca_projections.png',dpi=200)
print('wrote chain_ca_projections.png')
PY"
