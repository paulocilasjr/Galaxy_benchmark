/bin/bash -lc "python3 - <<'PY'
import numpy as np
coords=[]; cas=[]
chains={}
for line in open('protein.shape.q1.pdb'):
    if line.startswith(('ATOM','HETATM')):
        try:
            x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54])
        except ValueError: continue
        coords.append((x,y,z))
        ch=line[21]
        chains[ch]=chains.get(ch,0)+1
        if line[12:16].strip()=='CA': cas.append((x,y,z))
coords=np.array(coords); cas=np.array(cas)
print('atoms', coords.shape, 'CA', cas.shape, 'chains', chains)
print('min', coords.min(axis=0), 'max', coords.max(axis=0), 'range', coords.ptp(axis=0))
for name, arr in [('all',coords),('ca',cas)]:
    cen=arr.mean(axis=0); X=arr-cen
    w,v=np.linalg.eigh(np.cov(X.T)); idx=w.argsort()[::-1]
    print(name,'center',cen,'eigvals',w[idx])
    print('axes rows?')
    print(v[:,idx].T)
PY"
