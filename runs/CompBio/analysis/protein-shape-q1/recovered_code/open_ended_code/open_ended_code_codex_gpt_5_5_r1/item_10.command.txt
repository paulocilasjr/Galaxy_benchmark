import numpy as np, matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
coords=[]; ca=[]
for line in open('protein.shape.q1.pdb'):
    if line.startswith(('ATOM','HETATM')):
        x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54])
        coords.append((x,y,z))
        if line[12:16].strip()=='CA': ca.append((x,y,z))
coords=np.array(coords); ca=np.array(ca)
arr=coords
cen=arr.mean(axis=0); X=arr-cen
w,v=np.linalg.eigh(np.cov(X.T)); idx=w.argsort()[::-1]; pcs=v[:,idx]
# ensure horizontal pc1 left-to-right increasing roughly +x? flip so min original x on left maybe coords dot pc1; pc1 currently -x, flip
pcs[:,0]*=-1
# panel projections: xy,xz,yz plus PCA combos
fig,axes=plt.subplots(4,4,figsize=(16,16),facecolor='white')
angles=np.linspace(0,2*np.pi,16,endpoint=False)
for ax,theta in zip(axes.flat,angles):
    horiz=X@pcs[:,0]
    vert=X@(np.cos(theta)*pcs[:,1]+np.sin(theta)*pcs[:,2])
    ax.scatter(horiz,vert,s=0.08,c='black',alpha=.5,linewidths=0)
    ax.set_aspect('equal','box'); ax.axis('off'); ax.set_title(f'{theta*180/np.pi:.0f} deg',fontsize=8)
    # set common limits maybe
    ax.set_xlim(horiz.min()-5,horiz.max()+5); yr=max(abs(vert.min()),abs(vert.max()))+5; ax.set_ylim(-yr,yr)
plt.tight_layout(pad=.5)
fig.savefig('projections_pc1_rot.png',dpi=200)
# orthogonal original projections
fig,axes=plt.subplots(1,3,figsize=(15,5),facecolor='white')
for ax,(i,j,name) in zip(axes,[(0,1,'XY'),(0,2,'XZ'),(1,2,'YZ')]):
    ax.scatter(X[:,i],X[:,j],s=0.08,c='black',alpha=.5,linewidths=0)
    ax.set_aspect('equal','box'); ax.axis('off'); ax.set_title(name)
plt.tight_layout(); fig.savefig('projections_xyz.png',dpi=200)
print('wrote projections_pc1_rot.png projections_xyz.png')
PY"
