/bin/bash -lc "python - <<'PY'
import urllib.request
urls=['https://www.biorxiv.org/content/biorxiv/early/2026/04/06/2026.04.06.716850.full.pdf','https://www.biorxiv.org/content/10.64898/2026.04.06.716850v1.full.pdf','https://www.biorxiv.org/content/biorxiv/early/2026/04/06/2026.04.06.716850v1.full.pdf','https://www.biorxiv.org/content/10.64898/2026.04.06.716850v1.full.pdf?download=true']
for u in urls:
 try:
  r=urllib.request.urlopen(u,timeout=20); print(u,r.status,r.headers.get('Content-Type'),r.headers.get('Content-Length')); open('paper.pdf','wb').write(r.read()); break
 except Exception as e: print(u,type(e).__name__,e)
PY
ls -lh paper.pdf 2>/dev/null || true"
