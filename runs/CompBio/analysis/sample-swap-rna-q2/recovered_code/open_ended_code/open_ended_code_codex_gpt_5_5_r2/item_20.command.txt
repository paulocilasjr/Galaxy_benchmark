/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, math
expr=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
markers=pd.read_csv('wheatomics_markers.tsv',sep='\\t')
mapc={'Root cap':'root_cap','Cortex':'cortex','Epidermis':'epidermis','Provascular cells':'provascular','Phloem':'phloem','Endodermis':'endodermis','Xylem':'xylem'}
markers['expected']=markers['cluster'].map(mapc)
markers=markers.drop_duplicates(['gene','expected'])
markers=markers[markers.gene.isin(expr.index)]
print('markers present', markers.shape[0])
# Per cluster: mean log1p expression by column; zscore per gene across cols then average
log=np.log1p(expr)
z=log.sub(log.mean(axis=1),axis=0).div(log.std(axis=1).replace(0,np.nan),axis=0)
for exp, sub in markers.groupby('expected'):
    genes=sub.gene.tolist()
    print('\\nCLUSTER',exp,'n',len(genes))
    mean_expr=expr.loc[genes].mean(axis=0).sort_values(ascending=False)
    print('mean raw top', ', '.join(f'{k}:{v:.1f}' for k,v in mean_expr.items()))
    mean_z=z.loc[genes].mean(axis=0).sort_values(ascending=False)
    print('mean z   top', ', '.join(f'{k}:{v:.2f}' for k,v in mean_z.items()))
    # count marker genes whose max column is each
    maxcols=expr.loc[genes].idxmax(axis=1)
    print('max count', maxcols.value_counts().to_dict())
    # list genes not max in expected
    bad=[]
    for g in genes:
        row=expr.loc[g].sort_values(ascending=False)
        bad.append((g,row.index[0],row.iloc[0], expr.loc[g,exp], row.index[1], row.iloc[1]))
    for g,top,topv,ev,sec,secv in bad:
        flag='*' if top"'!=exp else '"' '
        print(flag,g,'top',top,topv,'expected',ev,'2nd',sec,secv)
PY"
