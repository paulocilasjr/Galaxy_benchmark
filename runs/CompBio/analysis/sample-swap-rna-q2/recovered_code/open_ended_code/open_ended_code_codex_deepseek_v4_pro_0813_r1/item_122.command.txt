/bin/bash -lc "python3 - <<'PY'
import re, html
s=open('article.html',encoding='utf-8').read()
# strip tags roughly
text=html.unescape(re.sub(r'<["'^>]+>'"',' ',s))
text=re.sub(r'\\s+',' ',text)
# find phrases around cluster N
for m in re.finditer(r'cluster[s]?\\s+(\\d{1,2})', text, flags=re.I):
    n=m.group(1)
    start=max(0,m.start()-250); end=min(len(text),m.end()+350)
    print('--- cluster',n,'---')
    print(text[start:end])
    print()
PY"
