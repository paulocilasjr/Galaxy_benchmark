/bin/bash -lc "python3 - <<'PY'
import gzip,json,collections,openpyxl
# load expr
expr={}; cols=None
with gzip.open('sample.swap.rna.q2.tsv.gz','rt') as f:
    raw=f.readline().rstrip('\\n'); cols=raw.split('\\t')[1:]
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)"'!=len(cols)+1: continue
        expr[p[0]]=[float(x) for x in p[1:]]
mp=json.load(open('"'genehub_old_to_03_fixed.json'))
wb=openpyxl.load_workbook('additional_file2.xlsx',read_only=True,data_only=True)
ws=wb['Table S2']
by=collections.defaultdict(list)
for r in ws.iter_rows(values_only=True):
    if isinstance(r[0],int): by[r[0]].append(str(r[1]))
def eval_clusters(cs,label):
    genes=set()
    for c in cs:
        for old in by[c]:
            for cur in mp.get(old,[]): genes.add(cur)
    found=[g for g in genes if g in expr]
    vals=collections.defaultdict(list); votes=collections.Counter()
    for g in found:
        arr=expr[g]; bi=max(range(len(arr)),key=lambda i:arr[i]); votes[cols[bi]]+=1
        for col,v in zip(cols,arr): vals[col].append(v)
    means={c:sum(vv)/len(vv) for c,vv in vals.items()}
    print('\\n',label,'clusters',cs,'n found',len(found),'genes',len(genes))
    print('votes',votes.most_common(5))
    print('means', ', '.join(f'{c}:{m:.2f}' for c,m in sorted(means.items(), key=lambda kv:kv[1], reverse=True)))
# combos
eval_clusters([2,3,7],'meristem-related')
eval_clusters([1,4,9],'pericycle paper')
eval_clusters([14,16,21],'vasculature phloem/pericycle?')
eval_clusters([6],'root hair')
eval_clusters([15,19],'root cap/border')
PY"
