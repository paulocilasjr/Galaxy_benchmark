/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np
genes=[]; data=[]
with gzip.open('sample.swap.rna.q2.tsv.gz','rt') as f:
 r=csv.reader(f,delimiter='\\t'); h=next(r)
 for row in r: genes.append(row[0]); data.append(list(map(float,row[1:])))
A=np.array(data); cols=h[1:]
# zscore each gene across columns? use ratio to row sum and max col
for ci,col in enumerate(cols):
 vals=A[:,ci]; rowsum=A.sum(1); ratio=vals/(rowsum+1e-9); ismax=(A.argmax(1)==ci)
 # score ratio * log? filter max and raw > 50
 candidates=np.where(ismax & (vals>50))[0]
 order=sorted(candidates,key=lambda i:ratio[i],reverse=True)
 print('\\n===',col,'n candidate',len(order))
 for i in order[:40]:
  print(f'{genes[i]}\\t{vals[i]:.1f}\\trowsum={rowsum[i]:.1f}\\tratio={ratio[i]:.3f}\\tothermax={np.sort(np.delete(A[i],ci))[-1]:.1f}')
PY"
