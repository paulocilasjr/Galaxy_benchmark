/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
obj <- readRDS('GSE270342_seuratObj_for_publication.rds')
md <- obj@meta.data
anns <- levels(md"'$annotation)
counts <- obj[['"'RNA']]"'$counts
data <- obj[['"'RNA']]"'$data
features <- rownames(counts)
sumc <- matrix(0, nrow=length(features), ncol=length(anns), dimnames=list(features, anns))
meanc <- sumc
meand <- sumc
for (j in seq_along(anns)) {
 cells <- rownames(md)[md$annotation == anns[j]]
 cat(anns[j], length(cells), '"'\\n')
 subc <- counts[, cells, drop=FALSE]
 subd <- data[, cells, drop=FALSE]
 sumc[,j] <- Matrix::rowSums(subc)
 meanc[,j] <- Matrix::rowMeans(subc)
 meand[,j] <- Matrix::rowMeans(subd)
}
write.table(sumc, 'seurat_ann_sum_counts.tsv', sep='\\t', quote=FALSE, col.names=NA)
write.table(meanc, 'seurat_ann_mean_counts.tsv', sep='\\t', quote=FALSE, col.names=NA)
write.table(meand, 'seurat_ann_mean_data.tsv', sep='\\t', quote=FALSE, col.names=NA)
EOF
Rscript [REDACTED LOCAL PATH]"
