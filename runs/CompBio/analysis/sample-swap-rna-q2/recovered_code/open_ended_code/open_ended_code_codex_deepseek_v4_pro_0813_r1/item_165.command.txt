/bin/bash -lc "python3 - <<'PY'
from PIL import Image, ImageOps, ImageEnhance
im=Image.open('Fig1_HTML.jpg').convert('RGB')
box=(210,100,430,500)
crop=im.crop(box)
crop=crop.resize(((box[2]-box[0])*8,(box[3]-box[1])*8), Image.LANCZOS)
g=crop.convert('L')
# adaptive threshold-ish: white bg dark text, but colored dots remain; threshold
bw=g.point(lambda x: 0 if x<180 else 255, '1')
bw.save('legend_left_bw.png')
PY
tesseract legend_left_bw.png stdout --psm 6 2>/dev/null"
