from PIL import Image, ImageOps, ImageEnhance, ImageFilter
im=Image.open('Fig1_HTML.jpg').convert('RGB')
box=(700,330,1250,1440)
crop=im.crop(box)

# threshold using white background, dark text
g=crop.convert('L')
bw=g.point(lambda x: 0 if x<180 else 255, '1')
bw.save('legend_bw.png')
PY
