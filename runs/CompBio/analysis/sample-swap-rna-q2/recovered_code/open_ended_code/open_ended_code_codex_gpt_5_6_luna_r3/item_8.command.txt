/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
D=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
# top specificity per column; avoid genes with tiny overall values by requiring max >= 2 and maybe col > 1
for c in D.columns:
    rest=D.drop(columns=c).mean(axis=1)
    score=np.log2((D[c]+1)/(rest+1))
    # high absolute expression and fold
    ix=score.sort_values(ascending=False).head(25).index
    print('\\n###',c)
    for g in ix:
        print(g, '%.2f'%D.loc[g,c], 'rest %.2f'%rest[g], 'lfc %.2f'%score[g])
PY"
