/bin/bash -lc "Rscript - <<'RS'
library(SeuratObject); library(Matrix)
z<-readRDS('gse270342_seurat.rds')
q<-read.delim(gzfile('sample.swap.rna.q2.tsv.gz'),row.names=1,check.names=FALSE)
cat('qdim',dim(q),'missing',sum("'!rownames(q)%in%rownames(z@assays$RNA@counts)),'"'\\n')
q<-as.matrix(q); m<-z@assays"'$RNA@counts; md<-z@meta.data
labs<-c('"'Cortex','Endodermis','Epidermis','Meristems','Pericycle','Phloem','Provascular cells','Root Cap','Root Hair','Xylem')
source<-matrix(NA_real_,nrow=nrow(q),ncol=length(labs),dimnames=list(rownames(q),labs))
for(j in seq_along(labs)){ix<-md"'$annotation==labs[j]; vv<-Matrix::rowSums(m[,ix,drop=FALSE])/sum(Matrix::colSums(m[,ix,drop=FALSE]))*1e6; source[,j]<-vv[rownames(q)]}
cat('"'max abs q-source diagonal',max(abs(q-source),na.rm=TRUE),'mean abs',mean(abs(q-source),na.rm=TRUE),'rounded equal fraction',mean(round(source,2)==q,na.rm=TRUE),'\\n')
corx<-cor(q,source,use='pairwise.complete.obs')
rmse<-sqrt(colMeans((q-source)"'^2))
cat('"'diagonal correlations\\n'); print(round(diag(corx),6)); cat('diagonal rmse\\n');print(round(rmse,6))
cat('best source per q\\n'); for(i in 1:ncol(q)){ord=order(corx[i,],decreasing=T);cat(colnames(q)[i],'->',labs[ord[1]],round(corx[i,ord[1]],6),'second',labs[ord[2]],round(corx[i,ord[2]],6),'\\n')}
cat('cor matrix\\n');print(round(corx,3))
RS"
