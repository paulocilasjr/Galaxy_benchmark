/bin/bash -lc "python - <<'PY'
import subprocess,re,html,time
for page in [1,2]:
 u=f'https://wheatomics.sdau.edu.cn/scRNA/?genome=Chinese_Spring&per_page=50&page={page}'
 p=subprocess.run(['curl','-L','--max-time','30','-sS',u],capture_output=True,text=True)
 s=p.stdout
 rows=re.findall(r'<tr>\\s*((?:<td.*?</td>\\s*)+)</tr>',s,re.S)
 print('page',page,'status',p.returncode,'rows',len(rows))
 for row in rows:
  cells=[html.unescape(re.sub('<.*?>','',c)).strip() for c in re.findall(r'<td.*?>(.*?)</td>',row,re.S)]
  if len(cells)==7: print(cells[4],cells[5])
PY"
