/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
# use log1p and contrasts: difference from mean of other cols; top 30
lx=np.log1p(x)
for c in x:
    others=[d for d in x if d"'!=c]
    score=lx[c]-lx[others].mean(axis=1)
    ix=score.sort_values(ascending=False).head(25).index
    print('"'\\n###',c)
    print(pd.DataFrame({'gene':ix,'value':x.loc[ix,c].round(2).values,'score':score.loc[ix].round(3).values,'rank':x.loc[ix].rank(ascending=False).loc[ix,c].astype(int).values}).to_string(index=False))
PY"
