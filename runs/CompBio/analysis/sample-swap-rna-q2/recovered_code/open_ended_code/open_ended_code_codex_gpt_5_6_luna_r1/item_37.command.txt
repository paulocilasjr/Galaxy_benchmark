/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
m=pd.read_csv('tae_markers.csv')
# top 10 markers per reference cluster by source avg log2fc, current values
for cl in sorted(m.cluster.unique()):
 g=m[m.cluster==cl].sort_values(['avg_log2FC','p_val_adj'],ascending=[False,True]).drop_duplicates('gene ID').head(10)
 genes=[z for z in g['gene ID'] if z in x.index]
 print('\\ncluster',cl,'top source markers')
 for gene in genes:
  vals=x.loc[gene].sort_values(ascending=False)
  print(gene, round(float(g[g['gene ID']==gene].iloc[0].avg_log2FC),2), vals.index[0],round(float(vals.iloc[0]),1), vals.round(1).to_dict())
PY"
