import pandas as pd,numpy as np
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
m=pd.read_csv('atlas_markers.csv')
y=np.log2(x+1)
z=y.sub(y.mean(axis=1),axis=0).div(y.std(axis=1),axis=0)
for cl,d in m.groupby('cluster'):
 genes=d.sort_values('avg_log2FC',ascending=False)['gene'].head(40)
 genes=[g for g in genes if g in x.index]
 score=z.loc[genes].mean().sort_values(ascending=False)
 print(cl,len(genes),' '.join(f'{c}:{v:.2f}' for c,v in score.items()))
PY"
