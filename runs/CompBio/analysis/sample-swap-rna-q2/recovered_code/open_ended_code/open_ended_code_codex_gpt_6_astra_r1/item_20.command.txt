/bin/bash -lc "python - <<'PY'
import zipfile,xml.etree.ElementTree as E
ns={'s':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
for f in ['reference/mmc3.xlsx','reference/mmc6.xlsx']:
 print(f)
 with zipfile.ZipFile(f) as z:
  ss=[''.join(t.itertext()) for t in E.fromstring(z.read('xl/sharedStrings.xml')).findall('s:si',ns)]
  for sheet in [n for n in z.namelist() if n.startswith('xl/worksheets/sheet') and n.endswith('.xml')]:
   for row in E.fromstring(z.read(sheet)).findall('.//s:row',ns):
    vals=[]
    for cell in row.findall('s:c',ns):
     v=cell.findtext('s:v','',ns)
     vals.append(ss[int(v)] if cell.get('t')=='s' and v else v)
    print('\\t'.join(vals))
PY"
