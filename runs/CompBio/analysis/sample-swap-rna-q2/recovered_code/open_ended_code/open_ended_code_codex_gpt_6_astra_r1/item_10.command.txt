/bin/bash -lc "curl -Ls 'https://www.ebi.ac.uk/europepmc/webservices/rest/PMC11860762/fullTextXML' -o PMC11860762.xml
python - <<'PY'
import xml.etree.ElementTree as E
r=E.parse('PMC11860762.xml').getroot()
for s in r.iter('sec'):
 t=' '.join(s.itertext())
 if ('marker' in t or 'availability' in t) and len(t)<22000: print(t)
for s in r.iter('supplementary-material'): print(E.tostring(s,encoding='unicode'))
PY"
