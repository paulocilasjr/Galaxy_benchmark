/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, urllib.parse, xml.etree.ElementTree as ET, time, os, collections
entrez=json.load(open('refs/v1_entrez_clean.json'))
# map unique geneid -> v2 ids
unique_ids=list(dict.fromkeys(entrez.values()))
outfile='refs/entrez_v2.json'
if os.path.exists(outfile): res=json.load(open(outfile))
else: res={}
todo=[i for i in unique_ids if i not in res]
print('unique entrez',len(unique_ids),'todo',len(todo), flush=True)
def fetch_batch(batch):
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=gene&id='+','.join(batch)+'&rettype=xml&retmode=xml'
    for attempt in range(4):
        try:
            data=urllib.request.urlopen(url, timeout=60).read()
            root=ET.fromstring(data)
            out={}
            for eg in root.findall('.//Entrezgene'):
                geneid_el=eg.find('.//Gene-track_geneid')
                gid=geneid_el.text if geneid_el is not None else None
                v2=[]
                for db in eg.findall('.//Gene-ref_db/Dbtag'):
                    dbname=db.findtext('Dbtag_db')
                    if dbname in ('EnsemblRapid','Ensembl'):
                        oid=db.find('.//Object-id_str')
                        if oid is not None and oid.text:
                            v2.append(oid.text)
                out[gid]=v2
            return out, None
        except Exception as e:
            err=str(e)
            if attempt==3: return {}, err
            time.sleep(2*(attempt+1))
for i in range(0,len(todo),200):
    batch=todo[i:i+200]
    out,err=fetch_batch(batch)
    res.update(out)
    print('batch',i,'err',err,'got',len(out),'total',len(res), flush=True)
    json.dump(res, open(outfile,'w'))
    time.sleep(0.35)
# summarize
v2count=sum(1 for v in res.values() if v)
print('entrez with v2',v2count,'of',len(res))
json.dump(res, open(outfile,'w'), indent=0)
# build v1->v2
v1v2={}
for g,gid in entrez.items():
    v=res.get(gid,[])
    if v:
        # prefer IDs present in matrix later; just first
        v1v2[g]=v[0]
json.dump(v1v2, open('refs/v1_v2.json','w'), indent=0)
print('v1 mapped',len(v1v2))
print(list(v1v2.items())[:5])
PY"
