/bin/bash -lc "cd refs && python3 - <<'PY'
import openpyxl
for fn in ['13059_2023_2908_MOESM2_ESM.xlsx','13059_2023_2908_MOESM5_ESM.xlsx','13059_2023_2908_MOESM6_ESM.xlsx']:
    wb=openpyxl.load_workbook(fn, read_only=True, data_only=True)
    print(fn, wb.sheetnames)
    for ws in wb.worksheets:
        print(' ', ws.title, ws.max_row, ws.max_column)
        for i,row in enumerate(ws.iter_rows(values_only=True)):
            print('   ', [str(x)[:40] if x is not None else '' for x in row][:8])
            if i>5: break
    wb.close()
PY"
