/bin/bash -lc "python -u - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
y=np.log1p(x)
for c in x:
 z=y[c]-y.drop(columns=c).max(axis=1)
 print('\\n###',c,'sum',x[c].sum(),'nz',(x[c]>0).sum())
 print('\\n'.join(f'{g}\\t{x.loc[g,c]}\\t{z[g]:.2f}' for g in z.nlargest(10).index))
PY"
