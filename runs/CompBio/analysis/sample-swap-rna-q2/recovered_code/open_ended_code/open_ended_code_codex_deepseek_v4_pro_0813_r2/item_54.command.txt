/bin/bash -lc "python3 - <<'PY'
import openpyxl, collections
wb=openpyxl.load_workbook('supp/mmc5.xlsx', read_only=True, data_only=True)
ws=wb['Table S4']
h=next(ws.iter_rows(values_only=True))
print(h)
cnt=collections.Counter(); samples={}
for r in ws.iter_rows(values_only=True):
    if r[0] is None: continue
    cl=r[7]
    cnt[cl]+=1
    if cl not in samples: samples[cl]=r
print(cnt)
for cl,r in samples.items():
    print(cl, r[:8])
wb.close()
PY"
