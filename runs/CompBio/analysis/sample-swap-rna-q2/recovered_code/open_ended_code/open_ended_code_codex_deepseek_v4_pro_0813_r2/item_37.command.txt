/bin/bash -lc "python3 - <<'PY'
import pandas as pd
idx=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0,nrows=1000).index
import re
from collections import Counter
p=Counter()
for g in idx:
    if g.startswith('TraesCS'):
        m=re.match(r'TraesCS([ABDU])0?([0-9])', g)
        p['cs'+str(m.groups()) if m else 'othercs'] +=1
    elif g.startswith('TraesCSU'):
        p['CSU'] +=1
    else: p['other'] +=1
print(p)
print([g for g in idx if not g.startswith('TraesCS')][:20])
PY"
