/bin/bash -lc "python3 - <<'PY'
import openpyxl
f='supp_asym/13059_2023_2908_MOESM2_ESM.xlsx'
wb=openpyxl.load_workbook(f, read_only=True, data_only=True)
for sn in wb.sheetnames:
    ws=wb[sn]
    print('\\nSHEET',sn)
    n=0
    for row in ws.iter_rows(values_only=True):
        print([str(c)[:60] if c is not None else None for c in row[:12]])
        n+=1
        if n>=4: break
wb.close()
PY"
