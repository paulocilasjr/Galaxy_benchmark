/bin/bash -lc "python3 - <<'PY'
import openpyxl, collections
wb=openpyxl.load_workbook('supp/mmc10.xlsx', read_only=True, data_only=True)
ws=wb['T. aestivum']
rows=ws.iter_rows(values_only=True)
h=next(rows)
print(h)
ct=collections.Counter(); best_by_ct={}
for r in rows:
    if r[4] is None: continue
    ct[r[4]] += 1
    if r[11] and r[4] not in best_by_ct:
        best_by_ct[r[4]]=r[11]
print('counts',ct)
for k,v in ct.items(): print(k, best_by_ct.get(k))
wb.close()
PY"
