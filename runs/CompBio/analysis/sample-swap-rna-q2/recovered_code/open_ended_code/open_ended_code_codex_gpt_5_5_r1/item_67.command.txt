from collections import defaultdict
# read local selected genes from fetched files

    with open(f) as fh:

# pearson and spearman-ish rank via simple ranks
def pearson(x,y):
    n=len(x)
    mx=sum(x)/n; my=sum(y)/n
    vx=sum((a-mx)**2 for a in x); vy=sum((b-my)**2 for b in y)
    if vx==0 or vy==0: return float('nan')
    return sum((a-mx)*(b-my) for a,b in zip(x,y))/math.sqrt(vx*vy)
def ranks(xs):
    order=sorted(range(len(xs)), key=lambda i: xs[i])
    r=[0]*len(xs)
    i=0
    while i<len(xs):
        j=i
        while j+1<len(xs) and xs[order[j+1]]==xs[order[i]]: j+=1
        val=(i+j)/2
        for k in range(i,j+1): r[order[k]]=val
        i=j+1
    return r

for j,h in enumerate(header):
    vals=[]

        genes=sorted(set(local)&set(site[cl]))
        x=[math.log1p(local[g][j]) for g in genes]
        y=[math.log1p(site[cl][g]) for g in genes]
        vals.append((cl,pearson(x,y),len(genes)))
    vals.sort(key=lambda x:x[1], reverse=True)
    print(f'{h:<12} ' + ' | '.join(f'{cl}:{r:.3f}' for cl,r,n in vals[:6]))
print('\\nSource cluster best local')
