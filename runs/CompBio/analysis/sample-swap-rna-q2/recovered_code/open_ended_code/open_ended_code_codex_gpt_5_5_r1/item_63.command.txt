/bin/bash -lc "python3 - <<'PY'
import gzip,csv,math,urllib.parse,urllib.request,time,os,http.client
rows=[]
with gzip.open('sample.swap.rna.q2.tsv.gz','rt') as f:
    r=csv.reader(f, delimiter='\\t')
    header=next(r)[1:]
    for row in r:
        vals=[float(x) for x in row[1:]]
        if max(vals) <= 0: continue
        logs=[math.log1p(v) for v in vals]
        mean=sum(logs)/len(logs)
        sd=(sum((x-mean)**2 for x in logs)/len(logs))**0.5
        rows.append((sd, max(vals), row[0]))
rows.sort(reverse=True)
sel=[g for sd,m,g in rows[:400]]
os.makedirs('dotplot_batches20', exist_ok=True)
base='https://wheatomics.sdau.edu.cn/scRNA/download_dotplot_data.php?genome=Chinese_Spring&folder=root&genes='
opener=urllib.request.build_opener()
opener.addheaders=[('User-Agent','Mozilla/5.0')]
for bi in range(0,len(sel),20):
    genes=sel[bi:bi+20]
    out=f'dotplot_batches20/batch_{bi//20:02d}.tsv'
    if os.path.exists(out) and os.path.getsize(out)>1000:
        continue
    url=base+urllib.parse.quote(','.join(genes), safe='')
    for attempt in range(4):
        try:
            print('fetch', out, len(genes), 'try', attempt+1, flush=True)
            with opener.open(url, timeout=60) as resp:
                data=resp.read()
            if len(data) < 100:
                raise RuntimeError('short response')
            open(out,'wb').write(data)
            break
        except Exception as e:
            print('  failed', repr(e), flush=True)
            time.sleep(1+attempt)
    else:
        raise SystemExit('failed '+out)
    time.sleep(0.3)
print('done')
PY"
