/bin/bash -lc "python3 - <<'PY'
import gzip, csv, math, statistics
path='sample.swap.rna.q2.tsv.gz'
with gzip.open(path,'rt') as f:
    r=csv.reader(f, delimiter='\\t')
    header=next(r); cols=header[1:]
    rows=[]
    for row in r:
        rows.append((row[0], [float(x) for x in row[1:]]))
print('shape', len(rows), len(cols))
print(cols)
for j,c in enumerate(cols):
    vals=[v[j] for _,v in rows]
    print(c, 'mean', round(sum(vals)/len(vals),3), 'max', max(vals))
# top specific markers per column: top value in column and ratio to second-highest
for j,c in enumerate(cols):
    scores=[]
    for g,v in rows:
        order=sorted(range(len(cols)), key=lambda k: v[k], reverse=True)
        if order[0]==j and v[j] > 1:
            ratio=(v[order[0]]+1)/(v[order[1]]+1)
            scores.append((ratio, v[j], g, cols[order[1]], v[order[1]]))
    scores.sort(reverse=True)
    print('\\n'+c)
    for rec in scores[:15]:
        print('\\t'.join([str(round(rec[0],3)), str(rec[1]), rec[2], rec[3], str(rec[4])]))
PY"
