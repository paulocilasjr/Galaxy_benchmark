/bin/bash -lc "python - <<'PY'
import urllib.request,csv,io,gzip,numpy as np
u='https://raw.githubusercontent.com/VIB-PSB/wheat_root_atlas/main/GRN_regulon_analysis/input/markers/tae_log2FC0.5_FDR0.05.csv'
rows=list(csv.DictReader(io.StringIO(urllib.request.urlopen(u).read().decode())))
f=gzip.open('sample.swap.rna.q2.tsv.gz','rt'); heads=f.readline().rstrip().split('\\t')[1:]
g=[]; dat=[]
for line in f:
 x=line.rstrip().split('\\t'); g.append(x[0]); dat.append(x[1:])
A=np.array(dat,float); ix={z:i for i,z in enumerate(g)}
for cl in ['3','4','5','6','8','12','13','14']:
 print('\\ncluster',cl)
 n=0
 for r in rows:
  if r['cluster']==cl and r['gene ID'] in ix:
   i=ix[r['gene ID']]; v=A[i]; o=np.argsort(v)[::-1]
   print(r['gene ID'],'FC',r['avg_log2FC'],'top',heads[o[0]],f'{v[o[0]]:.1f}','per',f'{v[heads.index(\"pericycle\")]:.1f}','phl',f'{v[heads.index(\"phloem\")]:.1f}','mer',f'{v[heads.index(\"meristems\")]:.1f}','hair',f'{v[heads.index(\"root_hair\")]:.1f}')
   n+=1
   if n>=15: break
PY"
