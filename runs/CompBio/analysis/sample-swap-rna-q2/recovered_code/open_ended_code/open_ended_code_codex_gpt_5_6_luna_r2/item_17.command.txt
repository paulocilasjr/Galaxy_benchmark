/bin/bash -lc "python - <<'PY'
import gzip,numpy as np
f=gzip.open('sample.swap.rna.q2.tsv.gz','rt')
heads=f.readline().rstrip().split('\\t')[1:]
g=[]; rows=[]
for line in f:
    x=line.rstrip().split('\\t'); g.append(x[0]); rows.append(x[1:])
A=np.array(rows,float); L=np.log1p(A)
# Print top by within-gene log specificity, plus values
print('shape',A.shape)
for j,h in enumerate(heads):
    # difference from mean other columns, log transformed; high raw and specificity
    others=np.delete(L,j,axis=1)
    score=L[:,j]-np.median(others,axis=1)
    # favor robust positive; top 30
    ix=np.argsort(score)[::-1][:30]
    print('\\n###',h)
    print('gene\\tscore\\t'+'\\t'.join(heads))
    for i in ix[:25]: print(g[i],f'{score[i]:.3f}',*([f'{v:.2f}' for v in A[i]]),sep='\\t')
PY"
