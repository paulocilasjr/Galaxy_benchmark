/bin/bash -lc "gzip -dc vcf.infer.build.q1.vcf.gz | awk 'BEGIN{OFS=\"\\t\"} "'!/''^#/ {print $2,$4}'"' > refs/variants.pos_ref.tsv
for build in hg18 hg19 hg38; do perl -e 'my ("'$fa,$v)=@ARGV; open my $F,"<",$fa or die $!; my $s=""; while(<$F>){next if /''^>/; chomp; $s.=$_} open my $V,"<",$v or die $!; my($n,$m)=(0,0); while(<$V>){chomp; my($p,$r)=split /'"\\t/; "'$n++; $m++ if uc(substr($s,$p-1,1)) eq uc($r)} print "$n $m ",($n-$m),"'"\\n\"' \"refs/"'${build}.chr20.fa" refs/variants.pos_ref.tsv | awk -v b="$build" '"'{print b,\"total=\" "'$1,"matches=" $2,"mismatches=" $3}'"'; done
wc -l refs/variants.pos_ref.tsv
ls -lh refs/*.chr20.fa"
