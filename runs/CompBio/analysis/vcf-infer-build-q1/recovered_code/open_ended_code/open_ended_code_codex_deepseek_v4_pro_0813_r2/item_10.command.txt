/bin/bash -c "python3 - <<'PY'
import urllib.request, json
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom=chr20;start=0;end=1000000'
try:
    data=json.load(urllib.request.urlopen(url, timeout=60))
    print(data.get('error'), len(data.get('dna','')), data.get('dna','')[:20])
except Exception as e: print('ERR',e)
PY"
