/bin/bash -c "python3 - <<'PY'
import gzip
seq=''
with open('refs/hg19.chr20.fa') as f:
    f.readline(); seq=f.read().replace('\\n','').upper()
bad=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<5: continue
        pos=int(p[1]); ref=p[3].upper(); alt=p[4].upper()
        if len(ref)==1 and pos<=len(seq) and seq[pos-1]"'!=ref:
            bad.append((pos,ref,seq[pos-1]))
print('"'mismatches among len1 ref',len(bad),bad[:20])
# multiallelic ref check
bad2=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<5: continue
        pos=int(p[1]); ref=p[3].upper(); alt=p[4].upper()
        if len(ref)"'!=1 and pos<=len(seq) and seq[pos-1]!=ref:
            bad2.append((pos,ref,seq[pos-1]))
print('"'multiallelic ref mismatches',len(bad2),bad2[:20])
PY"
