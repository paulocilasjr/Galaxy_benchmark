/bin/bash -c "python3 - <<'PY'
import gzip, time
refs={}
for name in ['hg18','hg19','hg38','hs1']:
    path=f'refs/{name}.chr20.fa'
    with open(path) as f:
        header=f.readline().strip()
        seq=f.read().replace('\\n','').replace(' ','').upper()
    refs[name]=seq
    print(name, len(seq), header)
print('parsing vcf')
counts={k:0 for k in refs}; total=0; mismatches={k:[] for k in refs}; bad=0
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<4: continue
        chrom,pos,ref,alt=parts[0],int(parts[1]),parts[3].upper(),parts[4].upper()
        if len(ref)"'!=1 or len(alt)!=1 or chrom!='"'20': 
            bad+=1; continue
        total+=1
        for name,seq in refs.items():
            if pos<=len(seq) and seq[pos-1]==ref:
                counts[name]+=1
            else:
                if len(mismatches[name])<10:
                    mismatches[name].append((pos,ref, seq[pos-1] if pos<=len(seq) else 'OOB'))
print('total',total,'bad',bad)
for name in refs:
    print(name, counts[name], 'match_rate', counts[name]/total if total else 0, 'example_mismatches', mismatches[name][:5])
PY"
