/bin/bash -lc "python3 - <<'PY'
import gzip, random, urllib.request, json, time
positions=[]; refs=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t'); positions.append(int(p[1])); refs.append(p[3])
idxs=random.Random(0).sample(range(len(positions)), 200)
matches=0; mism=0; ex=[]
for i,idx in enumerate(idxs):
    pos=positions[idx]; ref=refs[idx]
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hs1;chrom=chr20;start={pos-1};end={pos}'
    try:
        data=json.load(urllib.request.urlopen(url, timeout=10))
        base=data.get('dna','').upper()
    except Exception as e:
        print('err', e); continue
    if base==ref: matches+=1
    else:
        mism+=1
        if len(ex)<10: ex.append((pos,ref,base))
print('hs1 sample', matches, mism, 'matchpct', matches/(matches+mism)*100)
print(ex)
PY"
