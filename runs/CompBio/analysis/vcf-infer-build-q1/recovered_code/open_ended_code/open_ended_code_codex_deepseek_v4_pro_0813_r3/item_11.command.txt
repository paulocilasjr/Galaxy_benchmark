/bin/bash -lc "python3 - <<'PY'
import gzip, pysam
from collections import Counter
# Load variants positions/ref
positions=[]; refs=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')
        pos=int(p[1]); ref=p[3]
        positions.append(pos); refs.append(ref)
print('variants', len(positions), 'range', min(positions), max(positions))
for g in ['hg18','hg19','hg38']:
    fa=pysam.FastaFile(f'{g}_chr20.fa')
    matches=0; mismatches=0; n=0
    examples=[]
    for pos, ref in zip(positions, refs):
        try:
            base=fa.fetch('chr20', pos-1, pos).upper()
        except Exception as e:
            continue
        n+=1
        if base==ref: matches+=1
        else:
            mismatches+=1
            if len(examples)<5: examples.append((pos,ref,base))
    print(g, 'match', matches, 'mismatch', mismatches, 'tested', n, 'pct', f'{matches/n*100:.2f}')
    print(' mismatch examples', examples)
PY"
