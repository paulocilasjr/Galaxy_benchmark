/bin/bash -lc "python3 - <<'PY'
import gzip, sys, collections
# Load refs
refs={}
for g in ['hg18','hg19','hg38']:
    seq=[]
    with open(f'refs/{g}.chr20.fa') as f:
        f.readline()
        for line in f:
            seq.append(line.strip())
    refs[g]=''.join(seq).upper()
    print(g, len(refs[g]))
# Parse VCF positions/ref
pos_ref=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')
        pos_ref.append((int(p[1]), p[3].upper()))
print('variants', len(pos_ref))
for g in refs:
    matches=0; mism=0; examples=[]
    for p,r in pos_ref:
        # VCF 1-based; FASTA 0-based p-1. Sequence may shorter, handle
        b=refs[g][p-1] if p-1 < len(refs[g]) else 'N'
        if b==r:
            matches+=1
        else:
            mism+=1
            if len(examples)<10: examples.append((p,r,b))
    print(g, 'matches', matches, 'mismatches', mism, 'examples', examples[:10])
PY"
