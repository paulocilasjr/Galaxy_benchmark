/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, time, sys
out='refs/hs1.chr20.fa'
start=0
step=5_000_000
maxpos=70_000_000
seq_parts=[]
last=''
for s in range(0, maxpos, step):
    e=min(s+step, maxpos)
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hs1;chrom=chr20;start={s};end={e}'
    for attempt in range(5):
        try:
            data=urllib.request.urlopen(url, timeout=120).read().decode()
            j=json.loads(data)
            if 'dna' not in j:
                print('error response', j, file=sys.stderr)
                break
            dna=j['dna']
            print(s,e,len(dna), flush=True)
            seq_parts.append(dna)
            break
        except Exception as ex:
            print('retry',s,e,ex,file=sys.stderr)
            time.sleep(2)
    else:
        raise SystemExit('failed '+str(s))
# Important: chunk boundaries might have overlapping? API returns bases [s,e), so concat exactly.
with open(out,'w') as f:
    f.write('>chr20\\n')
    allseq=''.join(seq_parts)
    for i in range(0,len(allseq),60):
        f.write(allseq[i:i+60]+'\\n')
print('total', len(allseq))
PY"
