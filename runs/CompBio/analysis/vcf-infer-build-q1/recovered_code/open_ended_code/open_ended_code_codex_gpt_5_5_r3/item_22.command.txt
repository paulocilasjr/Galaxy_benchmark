/bin/bash -lc "python - <<'PY'
import gzip, glob
vcf='vcf.infer.build.q1.vcf.gz'
variants=[]
with gzip.open(vcf,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        fields=line.split('\\t')
        variants.append((int(fields[1]), fields[3].upper()))
print('variants', len(variants))
for path in sorted(glob.glob('refs/*.chr20.fa.gz')):
    if path.endswith('hs1.chr20.fa.gz'):
        continue
    seq=[]
    with gzip.open(path,'rt') as f:
        name=f.readline().strip()
        for line in f:
            if line.startswith('>'):
                break
            seq.append(line.strip())
    s=''.join(seq).upper()
    total=0; match=0; out=0; nbase=0; mismatch_examples=[]
    for pos,ref in variants:
        if pos<1 or pos>len(s):
            out += 1
            continue
        b=s[pos-1]
        if b=='N':
            nbase += 1
        total += 1
        if b==ref:
            match += 1
        elif len(mismatch_examples)<5:
            mismatch_examples.append((pos,ref,b))
    print(path, 'len', len(s), 'inrange', total, 'out', out, 'N', nbase, 'match', match, 'rate', match/total if total else None, 'examples', mismatch_examples)
PY"
