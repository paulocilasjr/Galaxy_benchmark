/bin/bash -lc "python3 - <<'PY'
import json,subprocess,base64,struct,math
uid='JfVtzAhFSCS-j6tNALioCQ'; z=2
for x in range(1516,1522):
 for y in range(1516,1522):
  key=f'{uid}.{z}.{x}.{y}'
  url=f'https://higlass.4dnucleome.org/api/v1/tiles/?d={key}'
  try:
   d=json.loads(subprocess.check_output(['curl','-L','--fail','--silent',url]))[key]
   b=base64.b64decode(d['dense']); vals=struct.unpack('<%df'%(len(b)//4),b)
   finite=[v for v in vals if math.isfinite(v)]
   print(x,y,'range',d.get('min_value'),d.get('max_value'),'finite',len(finite),'sample',finite[:2])
  except Exception as e: print('ERR',x,y,e)
PY"
