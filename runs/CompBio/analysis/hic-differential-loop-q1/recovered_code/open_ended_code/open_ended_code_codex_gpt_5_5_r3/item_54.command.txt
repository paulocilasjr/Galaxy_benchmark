/bin/bash -lc "PYTHONPATH=./pydeps python3 - <<'PY'
import numpy as np, matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
z=np.load('nanog_oe_10kb.npz')
H1=z['H1']; HFF=z['HFF']; start=int(z['start']); res=int(z['res'])
# zoom extents
lo,hi=7500000,8300000
i0=(lo-start)//res; i1=(hi-start)//res
x0=start+i0*res; x1=start+i1*res
H1z=H1[i0:i1,i0:i1]; HFFz=HFF[i0:i1,i0:i1]
D=np.log2((H1z+0.05)/(HFFz+0.05))
fig,axs=plt.subplots(1,3,figsize=(15,5),constrained_layout=True)
for ax,a,title,cmap,vmin,vmax in [
    (axs[0],np.log2(H1z),'H1 log2 OE','Reds',-1,2.2),
    (axs[1],np.log2(HFFz),'HFF log2 OE','Reds',-1,2.2),
    (axs[2],D,'H1-HFF log2 OE ratio','coolwarm',-2,2)]:
    im=ax.imshow(a.T,origin='lower',extent=[x0/1e6,x1/1e6,x0/1e6,x1/1e6],cmap=cmap,vmin=vmin,vmax=vmax)
    ax.axvspan(7629950/1e6,7809597/1e6,color='cyan',alpha=.15)
    ax.axhspan(7629950/1e6,7809597/1e6,color='cyan',alpha=.15)
    # mark candidate loops from lists
    for x,y,col,txt in [(7675000,7915000,'lime','H1 7.675-7.915'),(7645000,8075000,'yellow','HFF 7.645-8.075'),(7831000,8079000,'yellow','HFF 7.831-8.079'),(7797500,7907500,'yellow','HFF 7.797-7.907'),(7645000,7807000,'white','both')]:
        if x0<=x<=x1 and x0<=y<=x1:
            ax.plot([x/1e6],[y/1e6],marker='o',ms=5,mec=col,mfc='none',mew=1.4)
            ax.plot([y/1e6],[x/1e6],marker='o',ms=5,mec=col,mfc='none',mew=1.4)
    ax.set_title(title); ax.set_xlabel('chr12 Mb'); ax.set_ylabel('chr12 Mb')
    fig.colorbar(im,ax=ax,fraction=.046)
fig.savefig('nanog_zoom_candidates.png',dpi=200)
# print 10kb matrix values for candidate loops
cands=[('H1_microC',7675000,7915000),('HFF_microC_1',7645000,8075000),('HFF_microC_2',7831000,8079000),('HFF_microC_3',7797500,7907500),('both',7645000,7807000),('h1close',7800164,7905924)]
for name,x,y in cands:
    i=round((x-start)/res); j=round((y-start)/res)
    print(name, start+i*res, start+j*res, 'H1oe',H1[i,j], 'HFFoe',HFF[i,j], 'log2ratio',np.log2((H1[i,j]+.05)/(HFF[i,j]+.05)))
PY"
