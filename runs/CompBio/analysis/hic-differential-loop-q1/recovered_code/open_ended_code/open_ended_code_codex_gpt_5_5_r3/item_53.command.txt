/bin/bash -lc "python3 - <<'PY'
import gzip
nanog=(7629950,7809597)
region=(7000000,9000000)

def load(path):
    loops=[]
    with gzip.open(path,'rt') as f:
        for line in f:
            if line.startswith('#') or not line.strip(): continue
            parts=line.rstrip().split('\\t')
            c1,s1,e1,c2,s2,e2=parts[:6]
            s1,e1,s2,e2=map(int,[s1,e1,s2,e2])
            if c1=='chr12' and c2=='chr12' and region[0]<=s1<=region[1] and region[0]<=s2<=region[1]:
                loops.append((s1,e1,s2,e2,parts[6:] if len(parts)>6 else []))
    return loops
for name,path in [('H1','H1_union_loops.bedpe.gz'),('HFF','HFF_union_loops.bedpe.gz')]:
    loops=load(path)
    print('\\n',name,'all region',len(loops))
    # one anchor overlaps NANOG gene, or either anchor in broad subcompartment 7.6-7.82
    cand=[]
    for l in loops:
        s1,e1,s2,e2,rest=l
        ov1=max(s1,nanog[0])<min(e1,nanog[1]); ov2=max(s2,nanog[0])<min(e2,nanog[1])
        near1=nanog[0]-100000<=s1<=nanog[1]+100000 or nanog[0]-100000<=e1<=nanog[1]+100000
        near2=nanog[0]-100000<=s2<=nanog[1]+100000 or nanog[0]-100000<=e2<=nanog[1]+100000
        if ov1 or ov2 or near1 or near2:
            cand.append(l)
    print(name,'nanog/near',len(cand))
    for l in cand[:100]:
        s1,e1,s2,e2,rest=l
        print(s1,e1,s2,e2,'; mid',round((s1+e1)/2),round((s2+e2)/2), rest)
PY"
