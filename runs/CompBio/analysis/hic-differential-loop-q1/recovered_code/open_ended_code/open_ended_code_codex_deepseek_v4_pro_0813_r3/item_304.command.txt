/bin/bash -lc "python3 - <<'PY'
import cv2, numpy as np, os
from PIL import Image
img=cv2.imread('supp2_p08_dpi600.png',cv2.IMREAD_GRAYSCALE)
H,W=img.shape
# crop bottom right lower axis area
crop=img[int(H*0.94):int(H*0.99), int(W*0.42):W]
# binarize: dark text -> white
_,bw=cv2.threshold(crop,0,255,cv2.THRESH_BINARY_INV+cv2.THRESH_OTSU)
# remove small/large, find contours
n,labels,stats,cent=cv2.connectedComponentsWithStats(bw,8)
print('components',n,'crop shape',crop.shape)
# filter by area and aspect
chars=[]
for i in range(1,n):
 x,y,w,h,area=stats[i]
 if area<30 or area>5000: continue
 if h<8 or w<4: continue
 if h>200 or w>200: continue
 chars.append((x,y,w,h,area,cent[i]))
 print('comp',i,'x,y,w,h,area,cent',x,y,w,h,area,cent[i])
# save each with padding
os.makedirs('chars',exist_ok=True)
for idx,(x,y,w,h,area,c) in enumerate(chars):
 pad=8
 x0=max(0,x-pad); y0=max(0,y-pad); x1=min(crop.shape[1],x+w+pad); y1=min(crop.shape[0],y+h+pad)
 sub=crop[y0:y1,x0:x1]
 sub=cv2.resize(sub,None,fx=8,fy=8,interpolation=cv2.INTER_CUBIC)
 cv2.imwrite(f'chars/char_{idx}.png',sub)
 print('saved',idx,sub.shape)
PY
for f in chars/char_*.png; do echo -n \""'$f "; tesseract $f stdout --psm 10 -c tessedit_char_whitelist=0123456789. 2>/dev/null | tr '"'\\n' ' '; echo; done"
