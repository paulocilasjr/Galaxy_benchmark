/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# list HF dataset files
url='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true'
try:
 data=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=120).read())
 print('len',len(data))
 for x in data:
  p=x.get('path','')
  if 'hic' in p.lower() or 'answer' in p.lower() or 'question' in p.lower() or 'metadata' in p.lower() or 'README' in p:
   print(p,x.get('size'),x.get('type'))
except Exception as e: print('ERR',repr(e))
PY"
