/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/commits/main?limit=100'
try:
 data=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read())
 print(len(data))
 for c in data:
  print(c.get('id'), c.get('title'), c.get('date'))
except Exception as e: print('ERR',repr(e))
PY"
