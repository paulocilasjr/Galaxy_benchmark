/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for url in ['https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main?recursive=true&limit=100000','https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/derived/grades/biomnibench?recursive=true&limit=100000']:
 print('URL',url)
 try:
  data=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=120).read())
  print(len(data))
  for x in data:
   p=x.get('path','')
   if 'hic' in p.lower() or 'compbio' in p.lower() or 'grades' in p.lower():
    print(p,x.get('size'),x.get('type'))
 except Exception as e: print('ERR',repr(e))
PY"
