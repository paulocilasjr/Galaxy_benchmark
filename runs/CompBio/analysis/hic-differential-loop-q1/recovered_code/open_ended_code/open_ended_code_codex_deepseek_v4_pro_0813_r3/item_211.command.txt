/bin/bash -lc "python3 - <<'PY'
import requests
urls=[
'https://pmc.ncbi.nlm.nih.gov/articles/PMC7222625/pdf/',
'https://pmc.ncbi.nlm.nih.gov/articles/PMC7222625/pdf/main.pdf',
'https://www.ncbi.nlm.nih.gov/pmc/articles/PMC7222625/pdf/',
'https://www.ncbi.nlm.nih.gov/pmc/articles/PMC7222625/bin/NIHMS1574796-supplement-1.pdf',
]
for u in urls:
 r=requests.get(u,headers={'User-Agent':'Mozilla/5.0'},allow_redirects=True,timeout=30)
 print(u, r.status_code, r.headers.get('content-type'), len(r.content), r.url)
 if 'pdf' in (r.headers.get('content-type') or ''):
  fn=u.split('/')[-1]
  if not fn: fn='article.pdf'
  open(fn,'wb').write(r.content)
  print('saved',fn)
PY"
