/bin/bash -lc "python3 - <<'PY'
import requests
for url in [
'https://www.ncbi.nlm.nih.gov/pmc/articles/PMC7222625/?report=xml',
'https://www.ncbi.nlm.nih.gov/pmc/articles/PMC7222625/?report=xml&format=pmc',
'https://www.ncbi.nlm.nih.gov/research/bionlp/RESTful/pmcoa.cgi/BioC_xml/PMC7222625/unicode',
'https://pmc.ncbi.nlm.nih.gov/articles/PMC7222625/?report=reader',
]:
 print('URL',url)
 try:
  r=requests.get(url,timeout=30, headers={'User-Agent':'Mozilla/5.0'})
  print(r.status_code, r.headers.get('content-type'), len(r.content))
  print(r.text[:500].replace('\\n',' ')[:500])
 except Exception as e: print('err',e)
PY"
