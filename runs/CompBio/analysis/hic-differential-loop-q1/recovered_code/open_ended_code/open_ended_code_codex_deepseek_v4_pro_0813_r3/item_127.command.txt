/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for q in ['chr12:7629950-7809597','7629950','NANOG','chr12:7675000-7680000']:
 url='https://data.4dnucleome.org/search/?format=json&q='+urllib.parse.quote(q)+'&limit=20'
 print('\\n',q)
 try:
  d=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read())
  print('total',d.get('total'))
  for item in d.get('@graph',[])[:10]:
   print(item.get('@type'), item.get('accession'), item.get('display_title'), item.get('description'))
 except Exception as e: print('ERR',e)
PY"
