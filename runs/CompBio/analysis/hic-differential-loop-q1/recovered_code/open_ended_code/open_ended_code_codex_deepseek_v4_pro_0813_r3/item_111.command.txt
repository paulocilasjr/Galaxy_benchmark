/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for query in ['search/?type=File&format=json&q=differential+loop','search/?type=File&format=json&file_type=differential','search/?type=FileProcessed&format=json&q=loop']:
 url='https://data.4dnucleome.org/'+query
 print(url)
 try:
  d=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=120).read())
  print('graph',len(d.get('@graph',[])), 'total', d.get('total'))
  for item in d.get('@graph',[])[:20]:
   print(item.get('accession'), item.get('display_title'), item.get('file_type'), item.get('open_data_url'))
 except Exception as e: print('ERR',e)
PY"
