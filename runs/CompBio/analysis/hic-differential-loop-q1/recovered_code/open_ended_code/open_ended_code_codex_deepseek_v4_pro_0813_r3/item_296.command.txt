/bin/bash -lc "python3 - <<'PY'
from PIL import Image
img=Image.open('supp2_p08_dpi600.png')
W,H=img.size
print(W,H)
# split into quadrants/halves
boxes={
'full':(0,0,W,H),
'top':(0,0,W,H//2),
'bottom':(0,H//2,W,H),
'bottom_left':(0,H//2,W//2,H),
'bottom_right':(W//2,H//2,W,H),
'bottom_quarter_low':(0,int(H*0.7),W,H),
'bottom_right_low':(int(W*0.5),int(H*0.7),W,H),
'bottom_left_low':(0,int(H*0.7),W//2,H),
}
for name,box in boxes.items():
 img.crop(box).save(f'crop_{name}.png')
 print(name,box)
PY
python3 - <<'PY'
from rapidocr_onnxruntime import RapidOCR
from PIL import Image
ocr=RapidOCR()
import os
for f in sorted(os.listdir('.')):
 if f.startswith('crop_'):
  print('\\n====',f)
  result,_=ocr(f)
  if result:
   for box,text,conf in result:
    print(text, round(float(conf),2))
PY"
