/bin/bash -lc "python3 - <<'PY'
import urllib.request
for url in [
'https://www.ncbi.nlm.nih.gov/research/bionlp/RESTful/pmcoa.cgi/BioC_xml/PMC7222625/unicode',
'https://www.ncbi.nlm.nih.gov/research/bionlp/RESTful/pmcoa.cgi/BioC_json/PMC7222625/unicode',
'https://www.ncbi.nlm.nih.gov/pmc/articles/PMC7222625/?report=xml',
'https://www.ncbi.nlm.nih.gov/pmc/articles/PMC7222625/bin/NIHMS1573592-supplement-1.xlsx'
]:
 print('URL',url)
 try:
  req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
  d=urllib.request.urlopen(req,timeout=60).read()
  print('ok',len(d), d[:80])
 except Exception as e: print('ERR',repr(e))
PY"
