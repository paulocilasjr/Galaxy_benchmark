/bin/bash -lc "python3 - <<'PY'
# get paper html via sciencedirect maybe full text? Use cell PDF text
import urllib.request,re
url='https://www.sciencedirect.com/science/article/pii/S1097276520301519'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
try:
 data=urllib.request.urlopen(req,timeout=60).read().decode('utf-8','ignore')
 print(len(data))
 for term in ['NANOG','7629950','loop','differential']:
  print('\\n',term, data.count(term))
  for m in list(re.finditer(term,data))[:5]:
   s=max(0,m.start()-500); e=min(len(data),m.end()+1000)
   print(data[s:e][:1500].replace('\\n',' ')); print('---')
except Exception as e: print('ERR',e)
PY"
