/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# List space repo files
url='https://huggingface.co/api/spaces/Genentech/compbiobench-leaderboard-v1/tree/main?recursive=true'
try:
 data=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read())
 print(len(data))
 for x in data:
  p=x.get('path','')
  if any(s in p.lower() for s in ['answer','question','tsv','json','csv','app','eval','leader']):
   print(p,x.get('size'),x.get('type'))
except Exception as e: print('ERR',repr(e))
PY"
