/bin/bash -lc "python3 - <<'PY'
import cv2, os
img=cv2.imread('supp2_p08_dpi600.png',cv2.IMREAD_GRAYSCALE); H,W=img.shape
crop=img[int(H*0.93):int(H*0.99),0:W]
# groups x ranges
groups=[(308,400,'g1'),(966,1090,'g2'),(1640,1770,'g3'),(2316,2445,'g4'),(2531,2660,'g5'),(3624,3750,'g6')]
os.makedirs('groups',exist_ok=True)
for x0,x1,name in groups:
 sub=crop[:,x0:x1]
 # threshold original: dark text on light? use adaptive
 sub=cv2.adaptiveThreshold(sub,255,cv2.ADAPTIVE_THRESH_GAUSSIAN_C,cv2.THRESH_BINARY,51,15)
 # invert if needed maybe text white; tesseract expects dark text on white
 sub=255-sub
 sub=cv2.resize(sub,None,fx=8,fy=8,interpolation=cv2.INTER_CUBIC)
 cv2.imwrite(f'groups/{name}.png',sub)
 print(name,sub.shape)
PY
for f in groups/*.png; do echo \"==== "'$f"; tesseract $f stdout --psm 7 -c tessedit_char_whitelist=0123456789. 2>/dev/null; tesseract $f stdout --psm 8 -c tessedit_char_whitelist=0123456789. 2>/dev/null; tesseract $f stdout --psm 6 -c tessedit_char_whitelist=0123456789. 2>/dev/null; done'
