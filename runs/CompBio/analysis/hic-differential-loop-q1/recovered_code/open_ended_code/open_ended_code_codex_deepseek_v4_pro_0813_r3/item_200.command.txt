import pypdf
for f in ['paper.pdf','paper2.pdf','supp2.pdf']:
 print('FILE', f)

  r=pypdf.PdfReader(f)

  for i,p in enumerate(r.pages):

   if any(s in txt for s in ['NANOG','Micro-C','H1','HFF','loop','compartment','chr12']):
    print('page',i+1)
    print(txt[:3000])
    print('---')
 except Exception as e: print('ERR',e)
