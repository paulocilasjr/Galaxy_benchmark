/bin/bash -lc "cat > ocr.swift <<'SWIFT'
import Foundation
import Vision
import AppKit

let path = CommandLine.arguments[1]
guard let img = NSImage(contentsOfFile: path), let cg = img.cgImage(forProposedRect: nil, context: nil, hints: nil) else {
    print(\"no image\"); exit(1)
}
let req = VNRecognizeTextRequest { request, error in
    if let error = error { print(\"error\", error); return }
    guard let obs = request.results as? [VNRecognizedTextObservation] else { return }
    for o in obs {
        if let t = o.topCandidates(1).first {
            let box = o.boundingBox
            print(\"\\(t.string)\\t\\(box.origin.x),\\(box.origin.y),\\(box.size.width),\\(box.size.height)\")
        }
    }
}
req.recognitionLevel = .accurate
req.usesLanguageCorrection = false
let handler = VNImageRequestHandler(cgImage: cg, options: [:])
try? handler.perform([req])
SWIFT
swift ocr.swift crop_bottom_right.png"
