/bin/bash -lc "python3 - <<'PY'
import requests, hashlib, re, time
# get challenge page
s=requests.Session()
s.headers['User-Agent']='Mozilla/5.0'
u='https://pmc.ncbi.nlm.nih.gov/articles/instance/7222625/bin/NIHMS1574796-supplement-2.pdf'
r=s.get(u,timeout=30)
m=re.search(r'const POW_CHALLENGE = \"(["'^"]+)"'"',r.text)
difficulty=int(re.search(r'const POW_DIFFICULTY = \"(["'^"]+)"'"',r.text).group(1))
name=re.search(r'const POW_COOKIE_NAME = \"(["'^"]+)"'"',r.text).group(1)
print('challenge',m.group(1), 'diff',difficulty, 'name',name)
challenge=m.group(1)
nonce=0
target='0'*difficulty
while True:
 h=hashlib.sha256((challenge+str(nonce)).encode()).hexdigest()
 if h.startswith(target): break
 nonce+=1
 if nonce%100000==0: print('nonce',nonce)
print('nonce',nonce,h)
s.cookies.set(name, challenge+','+str(nonce), path='/')
r=s.get(u,timeout=60)
print('resp',r.status_code,r.headers.get('content-type'),len(r.content),r.content[:5])
open('supp2_real.pdf','wb').write(r.content)
PY"
