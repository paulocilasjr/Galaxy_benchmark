/bin/bash -lc "python - <<'PY'
import requests
for db in ['hg19','hg38']:
    url='https://api.genome.ucsc.edu/getData/track?genome='+db+';track=knownGene;chrom=chr12;start=7500000;end=8000000'
    try:
        r=requests.get(url,timeout=30)
        print(db,r.status_code,len(r.text))
        print(r.text[:500])
    except Exception as e: print(e)
PY"
