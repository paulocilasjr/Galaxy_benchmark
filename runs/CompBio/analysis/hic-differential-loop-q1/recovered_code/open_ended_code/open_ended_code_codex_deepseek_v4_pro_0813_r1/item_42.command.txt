/bin/bash -lc "python - <<'PY'
import requests
for url in [
'https://www.biorxiv.org/content/10.1101/639922v1.full.pdf',
'https://www.biorxiv.org/content/10.1101/639922v1.full.xml',
'https://www.biorxiv.org/content/biorxiv/early/2019/05/17/639922.full.pdf',
]:
 r=requests.get(url,headers={'User-Agent':'Mozilla/5.0'},timeout=30)
 print(url, r.status_code, len(r.content), r.headers.get('content-type'))
 if r.status_code==200 and 'pdf' in url:
  open('biorxiv.pdf','wb').write(r.content)
PY"
