/bin/bash -lc "python3 - <<'PY'
import gzip

def load(fn):
 d=[]
 for raw in gzip.open(fn,'rt'):
  if raw.startswith('chrom1'): continue
  p=raw.rstrip().split('\\t')
  if p[0]"'!='"'chr12' or p[3]"'!='"'chr12': continue
  s1,e1,s2,e2=map(int,[p[1],p[2],p[4],p[5]])
  if s1>s2: s1,e1,s2,e2=s2,e2,s1,e1
  d.append((s1,e1,s2,e2,p))
 return d
H1=load('H1_dots.bedpe.gz'); HFF=load('HFF_dots.bedpe.gz')
A,B=7629950,7809597
# overlap gene by at least one anchor (or both centers)
def inside_gene(s,e): return not (e<=A or s>=B)
for name,D in [('H1',H1),('HFF',HFF)]:
 print('\\n',name,'loops touching NANOG gene interval')
 for s1,e1,s2,e2,p in D:
  if inside_gene(s1,e1) and inside_gene(s2,e2):
   print(s1,e1,s2,e2,'count',p[9],'donut',p[12],'q',p[16], 'cstart',p[6],p[7])
print('\\nUnique H1 touching gene both anchors')
def key(x): return (x[0],x[1],x[2],x[3])
Fset={key(x) for x in HFF}; Hset={key(x) for x in H1}
for x in H1:
 s1,e1,s2,e2,p=x
 if inside_gene(s1,e1) and inside_gene(s2,e2) and key(x) not in Fset:
  print('H1 unique',x[:4])
for x in HFF:
 s1,e1,s2,e2,p=x
 if inside_gene(s1,e1) and inside_gene(s2,e2) and key(x) not in Hset:
  print('HFF unique',x[:4])
PY"
