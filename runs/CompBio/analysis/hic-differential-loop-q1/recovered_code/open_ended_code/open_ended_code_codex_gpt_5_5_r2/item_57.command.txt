/bin/bash -lc "python3 - <<'PY'
import requests, base64, numpy as np, math
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
H1='JfVtzAhFSCS-j6tNALioCQ'; HFF='3e8b4161-556e-4722-bcef-88265a3436a1'; server='https://higlass.4dnucleome.org/api/v1/tiles/'
offset=1943767673; res_for_z={11:2000,10:5000,9:10000}
cache={}
def fetch(uid,z,x,y):
 key=f'{uid}.{z}.{x}.{y}'
 if key not in cache:
  j=requests.get(server,params={'d':key},timeout=60).json()[key]
  dt='<f2' if j.get('dtype')=='float16' else '<f4'
  cache[key]=np.frombuffer(base64.b64decode(j['dense']),dtype=dt).astype('f4').reshape(256,256)
 return cache[key]
def mosaic(uid,z,win):
 res=res_for_z[z]; gs=offset+win[0]; ge=offset+win[1]; tw=256*res; x0=gs//tw; x1=(ge-1)//tw
 mat=np.full(((x1-x0+1)*256,(x1-x0+1)*256),np.nan,'f4')
 for x in range(x0,x1+1):
  for y in range(x0,x1+1): mat[(y-x0)*256:(y-x0+1)*256,(x-x0)*256:(x-x0+1)*256]=fetch(uid,z,x,y)
 a=(gs//res)-x0*256; b=math.ceil(ge/res)-x0*256
 return mat[a:b,a:b],res
win=(7550000,8000000); z=11
h1,res=mosaic(H1,z,win); hff,_=mosaic(HFF,z,win)
# mirror lower/upper for better visual, because tiles store one triangle sparsely in some off-diagonal orientation
# for diagonal local tile, keep max of transpose for symmetric display
h1s=np.fmax(h1,h1.T); hffs=np.fmax(hff,hff.T); diff=h1s-hffs
extent=[win[0]/1e6,win[1]/1e6,win[1]/1e6,win[0]/1e6]
fig,axs=plt.subplots(2,3,figsize=(15,10),constrained_layout=True)
for row,(A,B,D,label) in enumerate([(h1,hff,h1-hff,'raw'),(h1s,hffs,diff,'sym')]):
 vals=np.r_[A[np.isfinite(A)],B[np.isfinite(B)]]; vmax=np.nanpercentile(vals,98.8); md=np.nanpercentile(np.abs(D[np.isfinite(D)]),99)
 for ax,arr,title in zip(axs[row],[A,B,D],[f'H1 {label}',f'HFF {label}',f'H1-HFF {label}']):
  if 'H1-HFF' in title: im=ax.imshow(arr,origin='upper',extent=extent,cmap='coolwarm',vmin=-md,vmax=md,interpolation='nearest')
  else: im=ax.imshow(arr,origin='upper',extent=extent,cmap='Reds',vmin=0,vmax=vmax,interpolation='nearest')
  ax.set_title(title); ax.set_xlabel('chr12 Mb'); ax.set_ylabel('chr12 Mb')
  for p in [7.62995,7.809597,7.82,7.925,7.66]:
   ax.axvline(p,color='blue' if p in [7.62995,7.809597] else 'green',lw=.5)
   ax.axhline(p,color='blue' if p in [7.62995,7.809597] else 'green',lw=.5)
  fig.colorbar(im,ax=ax,fraction=.046)
fig.savefig('nanog_contrast_z11.png',dpi=180)
print('nanog_contrast_z11.png')
PY"
