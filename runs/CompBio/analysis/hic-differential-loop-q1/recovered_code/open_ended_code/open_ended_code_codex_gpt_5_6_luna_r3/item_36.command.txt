import numpy as np, matplotlib.pyplot as plt
h1=np.load('H1_1kb_raw.npy').astype(float); hff=np.load('HFF_1kb_raw.npy').astype(float)

for ax,m,title,cmap,vr in [(axs[0],m1,'H1','viridis',(-6,-3)),(axs[1],m2,'HFF','viridis',(-6,-3)),(axs[2],rat,'log2 H1/HFF','RdBu_r',(-4,4))]:
 im=ax.imshow(m,origin='lower',extent=[7620000,7820000,7620000,7820000],aspect='equal',cmap=cmap,vmin=vr[0],vmax=vr[1],interpolation='nearest')
 ax.set_title(title); ax.set_xlabel('chr12'); ax.set_ylabel('chr12')
 ax.axvspan(7629950,7809597,color='white',alpha=0.05)
 fig.colorbar(im,ax=ax,shrink=.7)
fig.savefig('nanog_maps.png',dpi=180)
PY
