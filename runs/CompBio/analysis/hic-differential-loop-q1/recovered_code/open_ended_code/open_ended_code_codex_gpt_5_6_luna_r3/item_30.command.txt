/bin/bash -lc "python -u - <<'PY'
import fsspec,h5py,cooler,numpy as np,time,sys
urls={'H1':'https://4dn-open-data-public.s3.amazonaws.com/fourfront-webprod/wfoutput/d13aa1ea-053c-4113-94fe-a1e7ab1dbbab/4DNFI9GMP2J8.mcool','HFF':'https://4dn-open-data-public.s3.amazonaws.com/fourfront-webprod/wfoutput/27bbfb40-0224-427c-be41-398b43d70c48/4DNFI9FVHJZQ.mcool'}
region='chr12:7620000-7820000'
for name,url in urls.items():
 t=time.time(); print(name,'start',flush=True)
 with fsspec.open(url,'rb',block_size=8*1024*1024,cache_type='readahead') as f:
  with h5py.File(f,'r') as h:
   c=cooler.Cooler(h['resolutions/1000'])
   raw=c.matrix(balance=False,sparse=False).fetch(region).astype(float)
   print(name,'raw fetched',raw.shape,time.time()-t,flush=True)
   bal=c.matrix(balance=True,sparse=False).fetch(region)
   print(name,'bal fetched',time.time()-t,flush=True)
   np.savez_compressed(name+'_1kb_region.npz',raw=raw,bal=bal)
   print(name,'saved',time.time()-t, 'raw sum',raw.sum(),'bal sum',np.nansum(bal),flush=True)
PY"
