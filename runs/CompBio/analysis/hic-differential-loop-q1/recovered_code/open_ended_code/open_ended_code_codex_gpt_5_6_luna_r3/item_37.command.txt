/bin/bash -lc "python - <<'PY'
import numpy as np, matplotlib.pyplot as plt
h1=np.load('H1_1kb_raw.npy').astype(float); hff=np.load('HFF_1kb_raw.npy').astype(float)
fig,axs=plt.subplots(2,2,figsize=(12,12),constrained_layout=True)
for ax,bs,name,mat,kind in [(axs[0,0],5,'H1',h1,'oe'),(axs[0,1],5,'HFF',hff,'oe'),(axs[1,0],5,'log H1/HFF',None,'ratio'),(axs[1,1],10,'ratio 10kb',None,'ratio')]:
 n=200//bs
 if kind=='ratio':
  a1=h1[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3));a2=hff[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3))
  m=np.log2((a1+.5)/(a2+.5)*a2.sum()/a1.sum()); cmap='RdBu_r'; v=(-3,3)
 else:
  a=mat[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3));
  d=np.abs(np.arange(n)[:,None]-np.arange(n)[None,:]); ex=np.array([np.nanmedian(np.diag(a,k)) for k in range(n)]); m=np.log2((a+.5)/(ex[d]+.5));cmap='viridis';v=(-2,2)
 im=ax.imshow(m,origin='lower',extent=[7620000,7820000,7620000,7820000],aspect='equal',cmap=cmap,vmin=v[0],vmax=v[1],interpolation='nearest')
 ax.set_title(name+' '+str(bs)+'kb');ax.set_xlabel('chr12');ax.set_ylabel('chr12');fig.colorbar(im,ax=ax,shrink=.7)
fig.savefig('nanog_aggregate.png',dpi=180)
PY"
