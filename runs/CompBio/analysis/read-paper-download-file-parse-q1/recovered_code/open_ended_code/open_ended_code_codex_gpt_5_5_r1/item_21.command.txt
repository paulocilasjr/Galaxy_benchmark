/bin/bash -lc "python3 - <<'PY'
import json
f='scATAC-reprog/src/analysis/20230814_exports/scATAC_data.ipynb'
nb=json.load(open(f))
for i in range(len(nb['cells'])):
    c=nb['cells'][i]
    s=''.join(c.get('source',[]))
    outs=''
    for o in c.get('outputs',[]):
        if 'text' in o: outs+=''.join(o['text'])
        if 'data' in o:
            for k,v in o['data'].items():
                if isinstance(v,list): outs+=''.join(map(str,v))
                else: outs+=str(v)
    if i>=10 or 'peak' in s.lower() or 'set' in s.lower():
        print('\\n--- cell',i,'type',c['cell_type'],'---')
        print(s[:2000])
        if outs:
            print('OUTPUT:', outs[:5000])
PY"
