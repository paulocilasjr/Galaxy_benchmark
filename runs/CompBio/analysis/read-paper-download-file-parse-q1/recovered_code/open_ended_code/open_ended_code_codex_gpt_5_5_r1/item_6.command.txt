/bin/bash -lc "PYTHONPATH=./pydeps python3 - <<'PY'
from pypdf import PdfReader
r=PdfReader('paper.pdf')
print(len(r.pages))
with open('paper.txt','w',encoding='utf-8') as f:
    for i,p in enumerate(r.pages,1):
        text=p.extract_text() or ''
        f.write(f'\\n\\n=== PAGE {i} ===\\n')
        f.write(text)
print('done')
PY
wc -l paper.txt
grep -i -n 'COC\\|L1\\|peak\\|cluster\\|data availability\\|github\\|supplement\\|link' paper.txt | head -250"
