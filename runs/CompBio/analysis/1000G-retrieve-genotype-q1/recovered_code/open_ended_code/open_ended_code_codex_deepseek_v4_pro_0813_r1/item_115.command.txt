/bin/bash -lc "curl -sL 'https://raw.githubusercontent.com/Eldergenix/Plato-Scientific-Research-Autonomous-Agent/b0a1fdce26567b9a01187d87c99ff81380527f8f/preprint/results/compbiobench_catalog/tasks.json' -o plato_tasks.json && wc -c plato_tasks.json && python3 - <<'PY'
import json
with open('plato_tasks.json') as f: d=json.load(f)
# find entry
def find(o, qid):
    if isinstance(o, dict):
        if o.get('question_id')==qid or o.get('id')==qid:
            return o
        for v in o.values():
            r=find(v,qid)
            if r: return r
    elif isinstance(o, list):
        for v in o:
            r=find(v,qid)
            if r: return r
    return None
r=find(d,'1000G-retrieve-genotype-q1')
print(json.dumps(r, indent=2)[:20000])
PY"
