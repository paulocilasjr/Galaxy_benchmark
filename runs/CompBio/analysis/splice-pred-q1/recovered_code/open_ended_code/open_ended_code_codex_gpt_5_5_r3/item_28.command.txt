/bin/bash -lc 'set -euo pipefail
for rs in 10 11 12 13 14; do
  url="https://github.com/Kuanhao-Chao/OpenSpliceAI/raw/v0.0.5/models/openspliceai-mane/10000nt/model_10000nt_rs${rs}.pt"
  out="models/model_10000nt_rs${rs}.pt"
  if [ ! -s "$out" ]; then
    curl -L --fail "$url" -o "$out"
  fi
  ls -lh "$out"
done'
