/bin/bash -lc ".venv/bin/python - <<'PY'
import json,sys,torch,numpy as np
sys.path.insert(0,'openspliceai_repo')
from openspliceai.predict.predict import create_datapoints,load_pytorch_models
j=json.load(open('inputs/chr12_120196699_120201111_ucsc.json'))
seq=j['dna'].upper().translate(str.maketrans('ACGTN','TGCAN'))[::-1]
X=torch.tensor(create_datapoints(seq,5000,10000),dtype=torch.float32).permute(0,2,1)
models,_=load_pytorch_models('openspliceai_repo/models/openspliceai-mane/10000nt',torch.device('cpu'),5000,10000)
with torch.no_grad(): y=torch.mean(torch.stack([m(X) for m in models]),axis=0).permute(0,2,1).contiguous().view(-1,3)[:len(seq)].numpy()
p=np.load('results/ensemble_predictions.npz')['ensemble'].T
print(y.shape, np.max(np.abs(y-p)), np.flatnonzero(y[:,1]>=.9), np.flatnonzero(y[:,2]>=.9))
PY"
