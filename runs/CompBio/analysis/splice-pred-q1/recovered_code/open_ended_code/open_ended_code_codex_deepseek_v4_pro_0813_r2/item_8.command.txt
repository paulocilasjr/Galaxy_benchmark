/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=120196698;end=120201111'
data=json.load(urllib.request.urlopen(url))
seq=data['dna'].upper()
print(len(seq))
comp=str.maketrans('ACGT','TGCA')
rc=seq.translate(comp)[::-1]
open('locus_minus.fa','w').write(f'>chr12:120196699-120201111_minus\\n{rc}\\n')
print(rc[:200], rc[-200:])
PY
wc -c locus_minus.fa && head -c 200 locus_minus.fa"
