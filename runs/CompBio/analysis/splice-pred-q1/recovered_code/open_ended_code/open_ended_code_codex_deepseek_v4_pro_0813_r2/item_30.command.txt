/bin/bash -lc "cat > run_predict.py <<'PY'
import torch, platform, time, os
import openspliceai.predict.predict as pp
import numpy as np

# force CPU to avoid MPS segfault
pp.setup_device = lambda: torch.device('cpu')

seq = open('locus_minus.fa').read().strip().split('\\n',1)[1].replace('\\n','')
print('len seq', len(seq), flush=True)
t=time.time()
y = pp.predict(seq, 'models', 10000)
print('pred shape', y.shape, 'time', time.time()-t, flush=True)
np.save('pred_avg.npy', y.numpy())
print(y[:10])
print('acceptor >=0.9 count', (y[:,1]>=0.9).sum().item(), 'donor >=0.9 count', (y[:,2]>=0.9).sum().item())
print('max acceptor', y[:,1].max().item(), 'max donor', y[:,2].max().item())
# Save positions and scores
acceptor_idx = (y[:,1]>=0.9).nonzero().flatten().tolist()
donor_idx = (y[:,2]>=0.9).nonzero().flatten().tolist()
with open('pred_sites.txt','w') as f:
    f.write('acceptor\\n')
    for i in acceptor_idx:
        f.write(f'{i}\\t{y[i,1].item():.6f}\\n')
    f.write('donor\\n')
    for i in donor_idx:
        f.write(f'{i}\\t{y[i,2].item():.6f}\\n')
print('acceptor idx', acceptor_idx)
print('donor idx', donor_idx)
PY
python3 run_predict.py"
