/bin/bash -lc "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 VECLIB_MAXIMUM_THREADS=1 python3 - <<'PY'
import json, numpy as np, torch
from openspliceai.predict.predict import create_datapoints, load_pytorch_models
torch.set_num_threads(1)
seq=json.load(open('downloads/hg38_chr12_120196699_120201111.json'))['dna'].upper()
x=create_datapoints(seq,5000,10000)
dna=torch.tensor(x.tolist(),dtype=torch.float32).permute(0,2,1)
models,_=load_pytorch_models('downloads/mane_10000nt_models',torch.device('cpu'),5000,10000)
with torch.no_grad(): s=torch.stack([m(dna) for m in models]).mean(0)[0].permute(1,0).numpy()[:len(seq)]
a=np.flatnonzero(s[:,1]>=.9); d=np.flatnonzero(s[:,2]>=.9)
print('forward calls',len(a),len(d),a.tolist(),d.tolist())
print('forward scores',[(int(p+1),float(s[p,1])) for p in a],[(int(p+1),float(s[p,2])) for p in d])
PY"
