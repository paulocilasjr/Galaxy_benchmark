/bin/bash -lc "python - <<'PY'
import urllib.request, json
start=120196699; end=120201111
# UCSC API half-open end-exclusive, so add 1
url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start={start};end={end+1}'
data=json.load(urllib.request.urlopen(url, timeout=60))
seq=data['dna']
print('len',len(seq), 'last', seq[-20:])
comp=str.maketrans('ACGTNacgtn','TGCANtgcan')
rc=seq.translate(comp)[::-1].upper()
print('rc len',len(rc))
open('locus_plus.fa','w').write(f'>chr12:{start}-{end}(+)\\n'+seq.upper()+'\\n')
open('locus_minus.fa','w').write(f'>chr12:{start}-{end}(-)\\n'+rc+'\\n')
PY
python - <<'PY'
from Bio import SeqIO
for f in ['locus_plus.fa','locus_minus.fa']:
 r=list(SeqIO.parse(f,'fasta'))[0]; print(f,r.id,len(r.seq),str(r.seq[:30]),str(r.seq[-30:]))
PY"
