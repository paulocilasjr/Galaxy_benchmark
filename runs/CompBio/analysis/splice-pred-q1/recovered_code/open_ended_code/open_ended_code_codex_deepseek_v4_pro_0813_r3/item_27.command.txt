/bin/bash -lc "python - <<'PY'
import urllib.request, json
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=120196699;end=120201111'
data=json.load(urllib.request.urlopen(url, timeout=60))
seq=data['dna']
print(len(seq), seq[:50], seq[-50:])
comp=str.maketrans('ACGTNacgtn','TGCANtgcan')
rc=seq.translate(comp)[::-1].upper()
print(len(rc), rc[:50], rc[-50:])
open('locus_plus.fa','w').write('>chr12:120196699-120201111(+)\\n'+seq.upper()+'\\n')
open('locus_minus.fa','w').write('>chr12:120196699-120201111(-)\\n'+rc+'\\n')
# also maybe transcript order should be original plus? output rc saved
PY
ls -lh locus_*.fa && head locus_minus.fa | cut -c1-100"
