/bin/bash -lc "cat > run_predict.py <<'PY'
import os, sys, time, json
import torch
import numpy as np
from openspliceai.predict import predict as predmod

start=time.time()
# read input FASTA sequence (already reverse complemented gene-strand)
from Bio import SeqIO
seq=str(next(SeqIO.parse('locus_minus.fa','fasta')).seq)
print('seq_len', len(seq), flush=True)
model_dir=os.path.abspath('repo/models/openspliceai-mane/10000nt')
y = predmod.predict(seq, model_dir, 10000)
print('elapsed', time.time()-start, 'shape', tuple(y.shape), flush=True)
np.save('predictions_avg.npy', y.numpy())
# Also save donor/acceptor scores
y_np=y.numpy()
np.save('acceptor_scores.npy', y_np[:,1])
np.save('donor_scores.npy', y_np[:,2])
print('done', flush=True)
PY
time python run_predict.py"
