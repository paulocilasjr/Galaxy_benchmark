/bin/bash -lc "python - <<'PY'
import numpy as np
L=4413
y=np.load('predictions_avg_cpu.npy')
acc=np.where(y[:,1]>=0.9)[0].tolist()
don=np.where(y[:,2]>=0.9)[0].tolist()
sites=sorted([(p,'D') for p in don]+[(p,'A') for p in acc])
# reconstruct exons: first terminal 0..first D inclusive; internal A..next D inclusive; last last A..L-1 inclusive
exons=[]
# first site should D
firstD=next(p for p,t in sites if t=='D')
exons.append((0,firstD))
for p,t in sites:
    if t=='A':
        # next donor with position > p
        nd=next((q for q,tt in sites if tt=='D' and q>p), None)
        if nd is not None:
            exons.append((p,nd))
        else:
            exons.append((p,L-1))
print('exons',exons)
print('lengths',[e-s+1 for s,e in exons])
print('count',len(exons))
# Ensure don positions are last bases of exons except last; acc positions first bases except first
PY"
