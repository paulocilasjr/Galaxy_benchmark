/bin/bash -lc ". .venv/bin/activate && python - <<'PY'
from pathlib import Path
import json
seq=json.loads(Path('data/ucsc_seq.json').read_text())['dna'].upper()
tr=str.maketrans('ACGTN','TGCAN')
g=seq.translate(tr)[::-1]
A=[1626,1906,2111,2372,3649,4177]
D=[381,1801,1993,2257,2557,3789]
print('gene length', len(g))
for p in D:
    print('D',p,'base',g[p], 'around', g[max(0,p-5):p+8])
for p in A:
    print('A',p,'base',g[p], 'around', g[max(0,p-8):p+5])
PY"
