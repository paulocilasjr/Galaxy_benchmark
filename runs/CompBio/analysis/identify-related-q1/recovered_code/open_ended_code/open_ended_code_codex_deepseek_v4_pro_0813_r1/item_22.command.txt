/bin/bash -lc "python3 - <<'PY'
import csv
rows=[]
with open('q1_genome.genome') as f:
    line=f.readline().split()
    print(line)
    idx={name:i for i,name in enumerate(line)}
    for raw in f:
        p=raw.split()
        d={name:p[i] for name,i in idx.items()}
        for k in ['PI_HAT','Z0','Z1','Z2']:
            d[k]=float(d[k])
        rows.append(d)
# sort pihat
rows.sort(key=lambda d:d['PI_HAT'], reverse=True)
print('top 50 pihat:')
for d in rows[:50]:
    print(d['IID1'], d['IID2'], 'PI_HAT', round(d['PI_HAT'],4), 'Z0',round(d['Z0'],4),'Z1',round(d['Z1'],4),'Z2',round(d['Z2'],4),'DST',d['DST'])
print('counts PI_HAT ranges')
ranges=[(0.5,2),(0.45,0.5),(0.4,0.45),(0.35,0.4),(0.3,0.35),(0.25,0.3),(0.2,0.25),(0.1,0.2),(0,0.1)]
for lo,hi in ranges:
    c=sum(1 for d in rows if lo<=d['PI_HAT']<hi)
    print(lo,hi,c)
print('pairs with pihat>0.4:', sum(1 for d in rows if d['PI_HAT']>0.4))
s=set([d['IID1'] for d in rows if d['PI_HAT']>0.4]+[d['IID2'] for d in rows if d['PI_HAT']>0.4])
print('samples involved >0.4:', len(s))
print('samples top?', sorted(s)[:100])
PY"
