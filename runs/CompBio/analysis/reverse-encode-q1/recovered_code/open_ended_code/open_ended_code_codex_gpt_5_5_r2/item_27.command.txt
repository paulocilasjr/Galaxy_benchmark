/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for acc in ['ENCAN040BSJ','ENCAN913JLY','ENCAN340GSB']:
    try:
        url=f'https://www.encodeproject.org/analyses/{acc}/?format=json'
        with urllib.request.urlopen(urllib.request.Request(url,headers={'accept':'application/json','user-agent':'curl/8'}),timeout=60) as r:data=json.load(r)
        print(acc, data.keys())
        print('title',data.get('title'),'pipeline',data.get('pipeline'), 'assembly', data.get('assembly'), 'dataset', data.get('dataset'))
        print('files',len(data.get('files',[])))
        for f in data.get('files',[])[:3]: print(f.get('accession'), f.get('file_format'), f.get('output_type'), f.get('assembly'), f.get('quality_metrics')[:1] if f.get('quality_metrics') else None)
    except Exception as e: print(acc,e)
PY"
