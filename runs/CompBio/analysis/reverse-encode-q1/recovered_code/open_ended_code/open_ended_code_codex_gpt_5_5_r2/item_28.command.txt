/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
queries=[
 {'type':'Analysis','pipelines':'/pipelines/ENCPL848KLD/','assembly':'GRCh38','limit':'all'},
 {'type':'Analysis','pipeline_version':'v3.0.0-alpha.2','assembly':'GRCh38','limit':'all'},
 {'type':'Analysis','searchTerm':'ENCODE4 v3.0.0-alpha.2 GRCh38','limit':'all'},
]
for params in queries:
    url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode({**params,'format':'json'})
    print('\\nURL',url)
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'accept':'application/json'}),timeout=120) as r:data=json.load(r)
        print('total',data.get('total'),'len',len(data.get('@graph',[])))
        for a in data.get('@graph',[])[:5]: print(a.get('accession'), a.get('title'), a.get('datasets')[:1] if a.get('datasets') else None, a.get('pipeline_version'), a.get('pipelines'))
    except Exception as e: print('ERR',e)
PY"
