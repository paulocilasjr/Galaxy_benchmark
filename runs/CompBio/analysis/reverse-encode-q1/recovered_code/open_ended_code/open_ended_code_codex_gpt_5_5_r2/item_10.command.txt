for acc in ['ENCFF321HCH']:
    url=f'https://www.encodeproject.org/files/{acc}/?format=json'

    print(json.dumps({k:data.get(k) for k in ['accession','dataset','assembly','file_format','output_type','file_size','biological_replicates','status','submitted_file_name']}, indent=2))
    print('analysis aliases', data.get('analysis_step_version',{}).get('aliases'))
    print('software_versions', [(sv.get('software',{}).get('name'),sv.get('version'),sv.get('downloaded_url')) for sv in data.get('analysis_step_version',{}).get('software_versions',[])])
    print('step_run', data.get('step_run',{}).get('@id'), data.get('step_run',{}).keys() if isinstance(data.get('step_run'),dict) else data.get('step_run'))
PY"
