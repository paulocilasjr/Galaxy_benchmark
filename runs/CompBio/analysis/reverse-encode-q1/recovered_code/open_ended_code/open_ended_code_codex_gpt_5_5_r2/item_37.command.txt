/bin/bash -lc "python3 - <<'PY'
import gzip, math
uniq=sorted(set(float(line.split()[3]) for line in gzip.open('reverse.encode.q1.tsv.gz','rt') if float(line.split()[3])>0))
lo,hi=1,10**12
for k,v in enumerate(uniq[:200], start=1):
    # v is rounded to 3: v-0.0005 <= k*1e6/N < v+0.0005 (approx half)
    a=(v-0.0005); b=(v+0.0005)
    if a<=0: continue
    # k*1e6/N >= a -> N <= k*1e6/a; < b -> N > k*1e6/b
    lo=max(lo, math.floor(k*1_000_000/b)+1)
    hi=min(hi, math.floor(k*1_000_000/a))
    if lo>hi:
        print('break at k',k,'v',v,'interval',lo,hi)
        break
    if k<=30: print(k,v,lo,hi)
print('final',lo,hi)
PY"
