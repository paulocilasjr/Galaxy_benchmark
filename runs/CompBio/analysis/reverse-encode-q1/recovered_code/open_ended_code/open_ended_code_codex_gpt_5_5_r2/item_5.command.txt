/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
url='https://www.encodeproject.org/search/?type=File&assay_title=DNase-seq&assembly=GRCh38&file_format=bam&output_type=alignments&limit=all&format=json'
print(url)
with urllib.request.urlopen(urllib.request.Request(url, headers={'accept':'application/json'}), timeout=60) as r:
    data=json.load(r)
print('total',data.get('total'))
print('keys', data['@graph'][0].keys() if data.get('@graph') else None)
for f in data.get('@graph',[])[:3]:
    print(json.dumps({k:f.get(k) for k in ['accession','output_type','file_format','assembly','dataset','biological_replicates','replicate','submitted_file_name','file_size','lab','award','status','analysis_step_version']}, indent=2)[:1000])
PY"
