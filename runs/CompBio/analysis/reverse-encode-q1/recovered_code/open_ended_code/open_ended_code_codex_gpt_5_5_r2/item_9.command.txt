/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for q in ['v3.0.0-alpha.2','3.0.0-alpha.2','dnase pipeline v3.0.0-alpha.2','ENCODE4 DNase v3.0.0-alpha.2']:
    url='https://www.encodeproject.org/search/?searchTerm='+urllib.parse.quote(q)+'&limit=5&format=json'
    with urllib.request.urlopen(urllib.request.Request(url,headers={'accept':'application/json'}),timeout=60) as r: data=json.load(r)
    print('\\nQUERY',q,'total',data.get('total'))
    for x in data.get('@graph',[])[:5]: print(x.get('@type'),x.get('@id'),x.get('title') or x.get('accession') or x.get('name'), x.get('version'))
PY"
