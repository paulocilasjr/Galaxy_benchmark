/bin/bash -lc "python3 - <<'PY'
import gzip, math
counts=[]
with open('ENCFF051GHB.q30.bin_counts.tsv') as f:
 for line in f:
  counts.append(int(line.split()[3]))
S=sum(counts)
print('candidate_sum',S,'scale',1e6/S)
mismatch=0; max_abs=0; sse=0; n=0; examples=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for c,line in zip(counts,f):
  tv=float(line.split()[3]); pv=round(c*1e6/S,3); d=abs(pv-tv)
  if d:
   mismatch+=1
   if len(examples)<5: examples.append((n,line.strip(),c,pv,d))
  max_abs=max(max_abs,d); sse+=d*d; n+=1
print('bins',n,'mismatch',mismatch,'max_abs',max_abs,'rmse',(sse/n)**0.5)
for e in examples: print(e)
PY"
