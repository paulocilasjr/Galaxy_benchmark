import gzip, math, statistics
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as fh:
 for line in fh:
  v=float(line.split()[3])
  if v>0: vals.append(v)
# Sample unique values <= maybe 20 and high too
uniq=sorted(set(vals))
print('max',max(vals),'nuniq',len(uniq))
# Coarse/fine search over N via s; residual to nearest integer counts in normalized units
# Use unique values weighted maybe not.
use=uniq[:5000]  # small values useful
best=[]
for N in range(70000000,83000001,1000):
 s=1e6/N
 err=0.0; bad=0
 for v in use:
  c=round(v/s)
  r=abs(v-c*s)
  if r>0.00051: bad+=1
  err+=r*r
 best.append((bad,err,N,s))
best=sorted(best)[:10]
print('coarse best')
for b in best: print(b)
lo=best[0][2]-5000; hi=best[0][2]+5000
best2=[]
for N in range(lo,hi+1):
 s=1e6/N
 err=0.0; maxr=0; bad=0
 for v in use:
  c=round(v/s)
  r=abs(v-c*s); maxr=max(maxr,r)
  if r>0.0005000001: bad+=1
  err+=r*r
 best2.append((bad,err,maxr,N,s))
for b in sorted(best2)[:20]: print('fine',b)
PY"
