/bin/bash -lc "python3 - <<'PY'
import json,gzip,collections,math,statistics,pyBigWig,sys,time
N=76275859
# load files and group totals from prior metadata
D=json.load(open('encode_dnase_alpha2_bams.json'))['@graph']
files=[f for f in D if f.get('output_type')=='alignments']
def metric(f, typ, key):
 for qm in f.get('quality_metrics',[]):
  if isinstance(qm,dict) and typ in qm.get('@type',[]) and key in qm: return qm[key]
 return None
by=collections.defaultdict(list)
for f in files: by[f['dataset']].append(f)
rows=[]
for ds,fs in by.items():
 total=sum((metric(f,'SamtoolsFlagstatsQualityMetric','total') or 0) for f in fs)
 if total>=N and total-N<20000000:
  rows.append((total-N,ds,total))
rows=sorted(rows)[:20]
# bws
B=json.load(open('encode_dnase_alpha2_signal_bws.json'))['@graph']
bw_by=collections.defaultdict(list)
for f in B: bw_by[f['dataset']].append(f)
# target top bins, avoid ultra repetitive? just top 80
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as fh:
 for line in fh:
  c,s,e,v=line.split(); v=float(v)
  if c in {'chr1','chr2','chr3','chr4','chr5','chr6','chr7','chr8','chr9','chr10','chr11','chr12','chr13','chr14','chr15','chr16','chr17','chr18','chr19','chr20','chr21','chr22','chrX','chrY'}:
   vals.append((v,c,int(s),int(e)))
# top and random-ish moderate bins
sample=sorted(vals, reverse=True)[:80]
y=[math.log1p(v) for v,_,_,_ in sample]
def pear(a,b):
 ma=sum(a)/len(a); mb=sum(b)/len(b)
 va=sum((x-ma)**2 for x in a); vb=sum((x-mb)**2 for x in b)
 return sum((x-ma)*(z-mb) for x,z in zip(a,b))/math.sqrt(va*vb) if va and vb else float('nan')
for diff,ds,total in rows:
 bws=bw_by.get(ds,[])
 if not bws:
  print(ds,total,'no_bw'); continue
 # choose bw derived from alignments if possible; first
 best=None
 for f in bws[:3]:
  url=f.get('cloud_metadata',{}).get('url')
  if not url: continue
  try:
   bw=pyBigWig.open(url)
   xs=[]
   for v,c,s,e in sample:
    z=bw.stats(c,s,e,type='sum')[0]
    xs.append(math.log1p(z or 0.0))
   bw.close()
   r=pear(y,xs)
   best=(r,f['accession'],url)
   break
  except Exception as e:
   best=('ERR '+str(e),f.get('accession'),url)
 print(ds.replace('/experiments/','').strip('/'), total, 'diff',diff,'bw',best[1] if best else None,'corr',best[0] if best else None)
 sys.stdout.flush()
PY"
