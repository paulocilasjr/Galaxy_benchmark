/bin/bash -c "python3 - <<'PY'
import pyBigWig, math
urls={
'rep1':'https://encode-public.s3.amazonaws.com/2020/09/16/780343e3-c4ed-446c-9e57-ddb9b5d4910a/ENCFF995SHL.bigWig',
}
# rep2 need URL
import json
# get rep2 cloud url quickly from ENCFF524RHM file
import urllib.request
with urllib.request.urlopen('https://www.encodeproject.org/files/ENCFF524RHM/?format=json') as f:
 j=json.load(f)
urls['rep2']=j['cloud_metadata']['url']
print(urls)
# read track values for chr10
vals=[]
with open('track.tsv') as f:
 for line in f:
  c,s,e,v=line.split()
  if c=='chr10': vals.append(float(v))
print(len(vals), vals[:20])
for name,url in urls.items():
 bw=pyBigWig.open(url)
 stats=bw.stats('chr10',0,bw.chroms('chr10'),type='mean',nBins=len(vals))
 # replace None 0
 stats=[0 if x is None else x for x in stats]
 # correlation with vals
 meanv=sum(vals)/len(vals); means=sum(stats)/len(stats)
 cov=sum((a-meanv)*(b-means) for a,b in zip(vals,stats))
 sdv=math.sqrt(sum((a-meanv)**2 for a in vals)); sds=math.sqrt(sum((b-means)**2 for b in stats))
 print(name,'corr',cov/(sdv*sds),'first stats',stats[:20])
PY"
