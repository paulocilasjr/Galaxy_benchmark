/bin/bash -c "python3 - <<'PY'
import pyBigWig,time,random
url='https://encode-public.s3.amazonaws.com/2021/02/08/13616ba1-4739-4665-99fd-772c9bd2985a/ENCFF424KBD.bigWig'
bw=pyBigWig.open(url)
for n in [5,10,20,50]:
 random.seed(1)
 starts=sorted(random.sample(range(0,133000000,10000),n))
 t=time.time()
 for s in starts:
  bw.stats('chr10',s,s+10000,type='mean')
 print(n,time.time()-t)
PY"
