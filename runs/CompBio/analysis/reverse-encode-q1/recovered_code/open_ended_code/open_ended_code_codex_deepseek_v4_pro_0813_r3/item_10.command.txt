/bin/bash -c "curl -s 'https://www.encodeproject.org/search/?type=File&file_format=bam&output_type=alignments&assembly=GRCh38&assay_title=DNase-seq&status=released&format=json&limit=all&field=accession&field=dataset&field=analysis_step_version.name&field=analysis_step_version.minor_version&field=analysis_step_version.aliases&field=biological_replicates&field=technical_replicates&field=read_count&field=file_size&field=href&field=preferred_default' -o dnase_bams_compact.json && python3 - <<'PY'
import json, collections
j=json.load(open('dnase_bams_compact.json'))
g=j['@graph']; print('total', j['total'], 'len', len(g))
c=collections.Counter()
for x in g:
    av=x.get('analysis_step_version') or {}
    c[(av.get('name'), tuple(av.get('aliases',[])))] +=1
for k,v in c.most_common(): print(v,k)
print('sample rows')
for x in g[:20]: print(x.get('accession'), x.get('dataset'), x.get('analysis_step_version'), x.get('biological_replicates'), x.get('technical_replicates'), x.get('preferred_default'))
PY"
