/bin/bash -c "python3 - <<'PY'
# create bins.bed and genome sizes from track.tsv
chroms=[]
with open('track.tsv') as fin, open('bins.bed','w') as fout:
 for line in fin:
  c,s,e,v=line.split()
  fout.write(f'{c}\\t{s}\\t{e}\\n')
# derive genome from last ends preserving order
import collections
last=collections.OrderedDict()
with open('track.tsv') as f:
 for line in f:
  c,s,e,v=line.split(); last[c]=int(e)
with open('genome.txt','w') as f:
 for c,e in last.items(): f.write(f'{c}\\t{e}\\n')
print(last)
PY
ls -lh bins.bed genome.txt"
