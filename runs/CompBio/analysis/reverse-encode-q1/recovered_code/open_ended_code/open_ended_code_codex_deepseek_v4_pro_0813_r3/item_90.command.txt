import sys, json, math, pyBigWig
from scipy.stats import spearmanr, pearsonr
acc,ds,url=sys.argv[1],sys.argv[2],sys.argv[3]
tr=json.load(open('track_chr10_1000.json'))
track=tr['vals']; n=len(track)

 bw=pyBigWig.open(url)
 st=bw.stats(tr['chrom'],0,tr['length'],type='mean',nBins=n)
 st=[0.0 if x is None else x for x in st]
 pear=pearsonr(track,st)[0]

 print(json.dumps({'acc':acc,'ds':ds,'pear':pear,'spear':spear}))

 print(json.dumps({'acc':acc,'ds':ds,'error':repr(e)}))

cat > run_score_subprocess.py <<'PY'
import json, subprocess, concurrent.futures, time, sys, os
j=json.load(open('dnase_v1_bigwigs.json'))['@graph']
cand=[x for x in j if (x.get('analysis_step_version') or {}).get('name')=='dnase-peak-calling-step-v-1-0']
if len(sys.argv)>1: cand=cand[:int(sys.argv[1])]
print('num',len(cand),flush=True)

def run(item):
    cmd=['python3','score_bw_worker.py',item['accession'],item['dataset'],item['cloud_metadata']['url']]
    try:
        p=subprocess.run(cmd,capture_output=True,text=True,timeout=20)
        if p.returncode==0 and p.stdout.strip():
            return json.loads(p.stdout.strip().splitlines()[-1])
        else:
            return {'acc':item['accession'],'ds':item['dataset'],'error':'rc'+str(p.returncode)+' '+p.stderr[:100]}
    except subprocess.TimeoutExpired:
        return {'acc':item['accession'],'ds':item['dataset'],'error':'timeout'}
    except Exception as e:
        return {'acc':item['accession'],'ds':item['dataset'],'error':repr(e)}

t=time.time(); results=[]; done=0
with concurrent.futures.ThreadPoolExecutor(max_workers=20) as ex:
    futs=[ex.submit(run,x) for x in cand]
    for fut in concurrent.futures.as_completed(futs):
        results.append(fut.result()); done+=1
        if done%100==0: print('done',done,'elapsed',round(time.time()-t,1),flush=True)
json.dump(results, open('scores_v1_bw_chr10.json','w'))
good=[r for r in results if 'pear' in r]
good.sort(key=lambda r:r.get('spear',-999), reverse=True)
print('good',len(good),'bad',len(results)-len(good))
for r in good[:50]: print(r['ds'],r['acc'],'pear',round(r['pear'],3),'spear',round(r['spear'],3))
PY
python3 run_score_subprocess.py 100"
