/bin/bash -lc "python3 - <<'PY'
import csv
rank=[]
for r in csv.DictReader(open('encode_bigwig_rank.tsv'),delimiter='\\t'):
 if r['exp']=='exp':continue
 rank.append(r)
rank.sort(key=lambda r:float(r['correlation']),reverse=True)
b={}
seen=set()
for r in csv.DictReader(open('encode_v3_grch38_bam.tsv'),delimiter='\\t'):
 k=(r['exp'],r['bam'])
 if k in seen:continue
 seen.add(k)
 b.setdefault(r['exp'],[]).append(r)
for r in rank[:50]:
 print(r['exp'],r['correlation'],'mapped',sum(int(x['mapped']) for x in b.get(r['exp'],[]) if x['mapped']),[(x['bam'],x['file_size']) for x in b.get(r['exp'],[])])
PY"
