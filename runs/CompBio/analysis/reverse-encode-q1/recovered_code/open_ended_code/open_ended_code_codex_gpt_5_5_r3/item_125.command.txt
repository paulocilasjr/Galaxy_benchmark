/bin/bash -lc "python3 - <<'PY'
import json, collections
N=76275859
# exp total BAM alignments across alpha2 released alignments bams
j=json.load(open('data/alpha2_bam_file_qc.json'))
tot=collections.defaultdict(int)
for f in j['@graph']:
 exp=f['dataset'].split('/')[-2]
 for qm in f.get('quality_metrics',[]):
  if qm.get('@type',[None])[0]=='SamtoolsFlagstatsQualityMetric':
   tot[exp]+=qm.get('total',0)
# read top40 scores
rows=[]
for line in open('data/bw_scores_top40.tsv'):
 f=line.rstrip('\\n').split('\\t')
 if len(f)>=5:
  try: rows.append((float(f[0]), f[4], f[5], f[8], tot.get(f[4],0)))
  except: pass
for r in sorted(rows, reverse=True):
 if r[4]>=N:
  print(r)
PY"
