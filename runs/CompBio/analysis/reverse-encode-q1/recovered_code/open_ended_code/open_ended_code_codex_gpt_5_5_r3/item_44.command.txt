/bin/bash -lc "python3 - <<'PY'
import gzip, math
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for line in f:
  v=float(line.split('\\t')[3])
  if v>0: vals.append(v)
uniq=sorted(set(vals))[:300]
# brute force N around scale [0.0125,0.0135]
best=[]
for N in range(74000000,80500001,1000):
 s=1e6/N
 ok=0; err=0
 for i,v in enumerate(uniq):
  k=round(v/s)
  pred=round(k*s+1e-12,3)
  if pred==v: ok+=1
  else: err+=abs(pred-v)
 best.append((ok,-err,N,s))
best=sorted(best, reverse=True)[:20]
for ok,ne,N,s in best: print(ok, -ne, N, s)
PY"
