/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np
vs=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  x=float(l.split()[3])
  if x>0: vs.append(x)
vs=np.array(vs)
# sample / scan total 74-80m integer at 10 read increments, values scaled N/1m should integer
sample=vs[::20]
best=[]
for N in range(74000000,80000001,100):
 x=sample*N/1e6
 score=np.mean(np.minimum(abs(x-np.rint(x)),.1)**2)
 best.append((score,N))
print(sorted(best)[:10])
# refine best regions exact ints
for _,Nc in sorted(best)[:3]:
 bb=[]
 for N in range(Nc-200,Nc+201):
  x=sample*N/1e6
  score=np.mean((x-np.rint(x))**2)
  bb.append((score,N))
 print(min(bb))
PY"
