/bin/bash -lc "python3 - <<'PY'
import pyBigWig,gzip,random,numpy as np,time
url='https://www.encodeproject.org/files/ENCFF907ITJ/@@download/ENCFF907ITJ.bigWig'
print('open',url)
b=pyBigWig.open(url);print('chroms',list(b.chroms().items())[:3])
rows=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  c,a,z,v=l.split();v=float(v)
  if c in ('chr1','chr7','chr19','chrX') and int(a)%1000000==0: rows.append((c,int(a),int(z),v))
random.seed(1);rows=random.sample(rows,200)
x=[];y=[]
for i,(c,a,z,v) in enumerate(rows):
 try:q=b.stats(c,a,z,type='sum',exact=True)[0]
 except RuntimeError:q=None
 if q is not None:x.append(v);y.append(q)
print('n',len(x),'corr',np.corrcoef(x,y)[0,1], 'logcorr',np.corrcoef(np.log1p(x),np.log1p(y))[0,1])
print(list(zip(x,y))[:10])
PY"
