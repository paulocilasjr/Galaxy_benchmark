files=json.load(open('files_bam.json'))['@graph']

 if acc in cand and any('filtering' in a for a in asv):
  groups[acc].append(f)
print('groups',len(groups),'files',sum(map(len,groups.values())))
print(collections.Counter(len(v) for v in groups.values()))
for acc,fs in list(groups.items())[:10]:
 print(acc, [(f['accession'],f['biological_replicates'],f['file_size']) for f in fs])
