/bin/bash -lc "python3 - <<'PY'
import pyBigWig,time
lines=open('counts.tsv').read().splitlines(); vals=[]
for line in lines:
 c,s,e,v=line.split(); vals.append((float(v),c,int(s),int(e)))
vals.sort(reverse=True); top=vals[:20]
url='https://encode-public.s3.amazonaws.com/2020/11/06/9569425d-04bc-4802-86a2-8068b5f427ff/ENCFF450AHB.bigWig'
bw=pyBigWig.open(url)
for nBins in [0,1,2,10]:
 t=time.time(); res=[]
 for v,c,s,e in top:
  x=bw.stats(c,s,e,nBins=nBins,type='sum',exact=True)
  res.append(x[0] if x else 0)
 print('nBins',nBins,'time',time.time()-t,res[0])
PY"
