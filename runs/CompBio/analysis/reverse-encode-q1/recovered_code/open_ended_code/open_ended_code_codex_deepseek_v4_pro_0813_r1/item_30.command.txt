/bin/bash -lc "python3 - <<'PY'
import json, collections
files=json.load(open('files_bam.json'))['@graph']
exps=json.load(open('experiments_analyses.json'))['@graph']
cand={e['accession'] for e in exps for a in e.get('analyses',[]) if a.get('pipeline_version')=='3.0.0-alpha.2'}
groups=collections.defaultdict(list)
for f in files:
 ds=f.get('dataset',''); acc=ds.split('/')[-2] if ds else None
 asv=f.get('analysis_step_version',{}).get('aliases',[])
 if acc in cand and any('filtering' in a for a in asv):
  groups[acc].append({'accession':f['accession'],'rep':f['biological_replicates'],'size':f['file_size'],'href':f['href']})
json.dump(groups, open('candidate_groups.json','w'), indent=2)
for f in groups[list(groups)[0]]: print(f)
PY"
