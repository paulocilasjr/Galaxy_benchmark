/bin/bash -lc "python3 -u - <<'PY'
import json, pyBigWig, time, math
sel=json.load(open('selected_bins_test.json')); bins=sel['bins']; obs=sel['obs']
cands=json.load(open('candidate_bw.json'))[:5]

def corr(a,b):
 n=len(a); ma=sum(a)/n; mb=sum(b)/n
 cov=sum((x-ma)*(y-mb) for x,y in zip(a,b)); va=sum((x-ma)**2 for x in a); vb=sum((y-mb)**2 for y in b)
 return cov/math.sqrt(va*vb) if va and vb else 0

def work(c):
 vec=[0.0]*len(bins)
 for f in c['files']:
  try:
   bw=pyBigWig.open(f['url'])
   for i,(ch,s,e) in enumerate(bins):
    x=bw.stats(ch,s,e,nBins=1,type='sum',exact=True)
    if x and x[0] is not None: vec[i]+=x[0]
   bw.close()
  except Exception as e:
   return c['exp'],None,repr(e)
 return c['exp'],corr(obs,vec),None
t=time.time()
for c in cands:
 print(work(c), flush=True)
print('elapsed',time.time()-t)
PY"
