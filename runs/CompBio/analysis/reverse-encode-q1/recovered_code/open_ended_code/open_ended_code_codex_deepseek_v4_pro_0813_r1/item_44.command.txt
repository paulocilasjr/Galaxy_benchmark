/bin/bash -lc "python3 - <<'PY'
import json, collections
exps=json.load(open('experiments_analyses.json'))['@graph']
bw=json.load(open('files_bigwig.json'))['@graph']
bwmap=collections.defaultdict(list)
for f in bw:
 ds=f.get('dataset',''); acc=ds.split('/')[-2] if ds else None
 bwmap[acc].append(f)
cands=[]
for e in exps:
 acc=e['accession']
 anas=[a for a in e.get('analyses',[]) if a.get('pipeline_version')=='3.0.0-alpha.2']
 if not anas: continue
 afiles=set()
 for a in anas: afiles.update(x.split('/')[-2] for x in a.get('files',[]))
 cfs=[f for f in bwmap.get(acc,[]) if f['accession'] in afiles and f.get('output_type')=='read-depth normalized signal']
 if cfs:
  cands.append({'exp':acc,'files':[{'accession':f['accession'],'rep':f.get('biological_replicates'),'size':f.get('file_size'),'href':f.get('href'),'output':f.get('output_type')} for f in cfs]})
print('candidates with signal bw',len(cands))
print('file count dist',collections.Counter(len(c['files']) for c in cands))
for c in cands[:10]: print(c)
PY"
