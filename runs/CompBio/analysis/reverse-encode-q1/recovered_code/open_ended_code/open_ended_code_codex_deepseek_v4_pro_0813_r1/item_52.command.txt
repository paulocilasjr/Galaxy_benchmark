/bin/bash -lc "python3 - <<'PY'
import pyBigWig, statistics
lines=open('counts.tsv').read().splitlines()
vals=[]
for line in lines:
 c,s,e,v=line.split(); vals.append((float(v),c,int(s),int(e)))
vals.sort(reverse=True); top=vals[:200]
url='https://encode-public.s3.amazonaws.com/2020/11/06/9569425d-04bc-4802-86a2-8068b5f427ff/ENCFF450AHB.bigWig'
bw=pyBigWig.open(url); print('opened')
res=[]
for v,c,s,e in top:
 x=bw.stats(c,s,e,type='sum',exact=True)[0]
 res.append(0.0 if x is None else x)
pairs=[(a,b) for a,b in zip(top,res)]
print(pairs[:10])
n=len(pairs); ma=sum(a for a,b in pairs)/n; mb=sum(b for a,b in pairs)/n
cov=sum((a-ma)*(b-mb) for a,b in pairs); va=sum((a-ma)**2 for a,b in pairs); vb=sum((b-mb)**2 for a,b in pairs)
print('corr',cov/(va*vb)**0.5 if va and vb else None)
print('ratio median',statistics.median([b/a for a,b in pairs if a>0]))
PY"
