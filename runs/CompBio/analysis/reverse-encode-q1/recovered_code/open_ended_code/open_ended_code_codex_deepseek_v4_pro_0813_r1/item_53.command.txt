/bin/bash -lc "python3 - <<'PY'
import pyBigWig, math, random
lines=open('counts.tsv').read().splitlines()
vals=[]
for line in lines:
 c,s,e,v=line.split(); vals.append((float(v),c,int(s),int(e)))
vals.sort(reverse=True); top=vals[:200]
url='https://encode-public.s3.amazonaws.com/2020/11/06/9569425d-04bc-4802-86a2-8068b5f427ff/ENCFF450AHB.bigWig'
bw=pyBigWig.open(url)
bins=[(c,s,e) for v,c,s,e in top]
obs=[float(v) for v,c,s,e in top]
res=[]
for c,s,e in bins:
 x=bw.stats(c,s,e,type='sum')[0]; res.append(0.0 if x is None else x)
def corr(a,b):
 n=len(a); ma=sum(a)/n; mb=sum(b)/n
 cov=sum((x-ma)*(y-mb) for x,y in zip(a,b)); va=sum((x-ma)**2 for x in a); vb=sum((y-mb)**2 for y in b)
 return cov/(va*vb)**0.5 if va and vb else 0
print('corr',corr(obs,res))
print('mean obs/res',sum(obs)/sum(res))
print('pairs', list(zip(obs,res))[:5])
PY"
