/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, pyBigWig
from scipy.stats import pearsonr,spearmanr
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for line in f:
  c,s,e,v=line.split()
  if c=='chr1': vals.append(float(v))
L=248956422;n=249
# aggregate fixed 10kb inputs to equal-width pybigwig bins
x=np.zeros(n)
for i,v in enumerate(vals): x[min(n-1,int((i*10000)*n/L))]+=v
u='https://encode-public.s3.amazonaws.com/2020/11/12/31cec132-6555-4b4e-a906-18ba1b587b94/ENCFF389PZY.bigWig'
b=pyBigWig.open(u); y=np.array(b.stats('chr1',0,L,nBins=n,type='mean'),dtype=float)
print('pearson',pearsonr(x,y).statistic,'spearman',spearmanr(x,y).statistic)
print('topx',np.argsort(x)[-10:][::-1],x[np.argsort(x)[-10:][::-1]])
print('topy',np.argsort(y)[-10:][::-1],y[np.argsort(y)[-10:][::-1]])
PY"
