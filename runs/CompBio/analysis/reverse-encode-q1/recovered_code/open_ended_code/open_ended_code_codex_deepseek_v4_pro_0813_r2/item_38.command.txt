/bin/bash -lc "python3 - <<'PY'
from collections import Counter
vals=[]
with open('track.tsv') as f:
 for line in f:
  v=float(line.split()[3])
  if v>0: vals.append(v)
u=min(vals)
c=Counter()
for v in vals:
 q=v/u
 # nearest half/third?
 c[(round(q*100))]+=1
for k,n in sorted(c.items()):
 print(k/100,n)
PY"
