/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('dnase_bams.json'))
g=j['@graph']
from collections import Counter
cnt=Counter(); exps=set()
for f in g:
    av=f.get('analysis_step_version')
    name=av.get('name') if isinstance(av,dict) else None
    cnt[name]+=1
    if name and ('v3.0.0-alpha.2' in name or 'v3' in name):
        exps.add((f.get('dataset'),name,f.get('accession')))
print(cnt)
print('matching files/exp')
for x in sorted(exps): print(x)
PY"
