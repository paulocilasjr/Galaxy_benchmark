/bin/bash -lc "python3 - <<'PY'
import json
from collections import defaultdict
bwj=json.load(open('dnase_v3_bigwigs.json'))
bwmap=defaultdict(list)
for f in bwj['@graph']:
 ds=f.get('dataset'); acc=f.get('accession'); size=f.get('file_size'); br=f.get('biological_replicates')
 if acc: bwmap[ds].append((acc,size,br))
bamj=json.load(open('dnase_v3_bams_api.json'))
exp=defaultdict(lambda:{'total':0,'files':[]})
for f in bamj['@graph']:
 ds=f['dataset']; mapped=None
 for qm in f.get('quality_metrics',[]):
  if 'reads mapped' in qm: mapped=qm['reads mapped']
 e=exp[ds]
 if mapped is not None:e['total']+=mapped
 e['files'].append((f['accession'],f.get('mapped_run_type'),mapped,f.get('file_size')))
for T,N in [(76275859,15),(152551718,15)]:
 rows=[(abs(e['total']-T),e['total'],ds,e['files']) for ds,e in exp.items() if e['total']>=T]
 rows.sort()
 print('\\nTARGET',T)
 for diff,total,ds,files in rows[:N]:
  print(ds,diff,total,'bw',bwmap.get(ds),'files',files)
PY"
