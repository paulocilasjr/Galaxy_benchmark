/bin/bash -lc "python3 - <<'PY'
import pyBigWig, numpy as np, time
bw=pyBigWig.open('ENCFF040SNS.bigWig')
track=[]
with open('track.tsv') as f:
 for line in f:
  p=line.split(); track.append((p[0],int(p[1]),int(p[2]),float(p[3])))
b=[]
for chrom,start,end,v in track:
 s=bw.stats(chrom,start,end,nBins=1,type='mean')[0]
 b.append(s if s is not None else 0.0)
t=np.array([v for *_,v in track]); b=np.array(b)
print('corr',np.corrcoef(t,b)[0,1])
nz=b>0
print('corr nz',np.corrcoef(t[nz],b[nz])[0,1], 'nz',nz.sum())
PY"
