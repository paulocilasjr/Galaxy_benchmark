/bin/bash -lc "python3 - <<'PY'
S=76275859
s=0; minc=10**9; maxc=0; odd=0; even=0
for line in open('track.tsv'):
 v=float(line.split()[3])
 k=round(v*S/1e6)
 s+=k
 if v>0:
  if k<minc:minc=k
  if k>maxc:maxc=k
  if k%2: odd+=1
  else: even+=1
print('sum',s,'min',minc,'max',maxc,'odd bins',odd,'even bins',even,'nonzero',odd+even)
# Factor S
n=S; fac=[]
d=2
while d*d<=n:
 while n%d==0:
  fac.append(d); n//=d
 d+=1
if n>1: fac.append(n)
print(fac)
PY"
