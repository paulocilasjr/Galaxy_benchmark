/bin/bash -lc "python3 - <<'PY'
import json
from collections import defaultdict
j=json.load(open('dnase_v3_bams_api.json'))
g=j['@graph']
exp=defaultdict(lambda:{'total':0,'files':[]})
for f in g:
 ds=f['dataset']; mapped=None
 for qm in f.get('quality_metrics',[]):
  if 'reads mapped' in qm: mapped=qm['reads mapped']
 e=exp[ds]
 if mapped is not None:e['total']+=mapped
 e['files'].append((f['accession'],f.get('mapped_run_type'),f.get('file_size'),mapped))
for T in [76275859,152551718]:
 lo=T; hi=int(T*1.25)
 rows=[(abs(e['total']-T),e['total'],ds,e['files']) for ds,e in exp.items() if lo<=e['total']<=hi]
 rows.sort()
 print('TARGET',T,'candidates in 0-25% above:',len(rows))
 for r in rows[:100]: print(r)
 print()
PY"
