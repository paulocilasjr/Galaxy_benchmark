/bin/bash -lc "python3 - <<'PY'
import numpy as np, time
vals=[]
with open('track.tsv') as f:
 for line in f:
  v=float(line.split()[3])
  if v>0: vals.append(v)
vals=np.array(vals)
# Subsample 10000 for search speed
idx=np.linspace(0,len(vals)-1,20000).astype(int)
sub=vals[idx]
best=[]
for S in range(20_000_000, 200_000_001, 100_000):
 a=1e6/S
 k=np.rint(sub/a)
 pred=np.rint(k*a*1000)/1000.0  # round to 3 decimals
 err=np.mean(np.abs(pred-sub))
 best.append((err,S))
best.sort()
print(best[:20])
PY"
