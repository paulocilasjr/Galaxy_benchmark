/bin/bash -lc "python3 - <<'PY'
import requests,pyBigWig,time
u='https://www.encodeproject.org/files/ENCFF389PZY/@@download/ENCFF389PZY.bigWig'
r=requests.get(u,allow_redirects=False,timeout=30)
print('redirect',r.status_code,len(r.content))
b=pyBigWig.open(r.headers['location'])
print('chrom',b.chroms()['chr1'])
t=time.time(); x=b.stats('chr1',0,100000,nBins=10,type='sum',exact=True); print(time.time()-t,x)
b.close()
PY"
