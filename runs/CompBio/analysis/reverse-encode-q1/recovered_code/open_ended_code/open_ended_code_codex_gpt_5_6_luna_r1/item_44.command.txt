import gzip
from collections import Counter
import numpy as np
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f: c=Counter(int(round(float(l.split('\\t')[3])*1000)) for l in f)

sol=[]
for N in range(76250000,76285001):

 if r==0: sol.append(N)
print('solutions',sol[:100],len(sol))
for N in range(76265000,76285001,100):
 r=int(np.sum(freq*((k*N+500_000_000)//1_000_000_000))-N)
 if abs(r)<1000: print(N,r)
PY"
