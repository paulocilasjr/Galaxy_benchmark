/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import Counter
import numpy as np
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f: c=Counter(int(round(float(l.split('\\t')[3])*1000)) for l in f)
k=np.array(list(c),dtype=np.int64); freq=np.array([c[x] for x in k],dtype=np.int64)
def res(N): return int(np.sum(freq*((k*N+500_000_000)//1_000_000_000))-N)
for N in range(76000000,76500001,10000): print(N,res(N))
PY"
