/bin/bash -lc "python3 - <<'PY'
import gzip, re
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
    for line in f:
        v=line.rstrip().split('\\t')[3]
        vals.append(int(round(float(v)*1000)))
from collections import Counter
c=Counter(vals)
print(len(vals),sum(vals),len(c),min(k for k in c if k),max(vals))
# rounded nearest integer raw count corresponding to each displayed y (thousandths)
def inferred(N):
    return sum(freq*((k*N+500_000_000)//1_000_000_000) for k,freq in c.items())
rows=[]
for line in open('candidate_totals.tsv'):
    e,fa,t,*rest=line.rstrip('\\n').split('\\t')
    if t=='0': continue
    N=int(t)
    rows.append((abs(inferred(N)-N),inferred(N)-N,N,e,fa))
for x in sorted(rows)[:100]: print(*x)
PY"
