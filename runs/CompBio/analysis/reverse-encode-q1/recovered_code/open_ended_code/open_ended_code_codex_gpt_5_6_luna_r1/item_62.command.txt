/bin/bash -lc "python3 - <<'PY'
import gzip, pyBigWig, numpy as np, math
# target
T={}
order=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  c,s,e,v=l.rstrip().split('\\t'); T.setdefault(c,[]).append(float(v));
  if c not in order: order.append(c)
# bigwig sums exact 10 kb chunks
b=pyBigWig.open('candidate_density.bigWig')
B={}; total=0
for c in order:
 L=b.chroms(c); n=L//10000; arr=[]
 vals=b.stats(c,0,n*10000,nBins=n,type='sum',exact=True)
 arr.extend(0.0 if x is None else x for x in vals)
 if n*10000<L:
  x=b.stats(c,n*10000,L,nBins=1,type='sum',exact=True)[0]
  arr.append(0.0 if x is None else x)
 B[c]=arr; total+=sum(arr)
 print(c,len(arr),len(T[c]),sum(arr),sum(T[c]))
b.close()
a=np.array([v for c in order for v in T[c]])
x=np.array([v for c in order for v in B[c]])
print('n',len(a),'target sum',a.sum(),'bw sum',x.sum())
print('corr',np.corrcoef(a,x)[0,1], 'corr log',np.corrcoef(np.log1p(a),np.log1p(x))[0,1])
# normalize BW sum and show ratio / diff
xn=x/x.sum()*1e6
print('norm corr',np.corrcoef(a,xn)[0,1], 'mae',np.mean(abs(a-xn)), 'max',max(abs(a-xn)))
for i in range(20): print(i,a[i],xn[i],x[i])
PY"
