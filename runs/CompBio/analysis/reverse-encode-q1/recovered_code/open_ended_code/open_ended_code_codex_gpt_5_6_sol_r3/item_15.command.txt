/bin/bash -lc "python - <<'PY'
import gzip,numpy as np
q=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for line in f:
  x=float(line.split()[3])
  if x>0:q.append(round(x*1000))
q=np.array(q,dtype=np.int64)
# candidates every integer N 50m..100m, filter using 200 diverse q constraints batches
cand=np.arange(50_000_000,100_000_001,dtype=np.int64)
vals=np.unique(q)
# choose quantiles broadly
checks=vals[np.linspace(0,len(vals)-1,300,dtype=int)]
for i,qq in enumerate(checks):
 c=np.rint(qq*cand/1e9).astype(np.int64)
 pred=np.floor(c*1e9/cand+0.5).astype(np.int64)
 cand=cand[pred==qq]
 if i%30==0:print(i,len(cand))
 if len(cand)==0:break
print('remaining',len(cand),cand[:20],cand[-20:])
PY"
