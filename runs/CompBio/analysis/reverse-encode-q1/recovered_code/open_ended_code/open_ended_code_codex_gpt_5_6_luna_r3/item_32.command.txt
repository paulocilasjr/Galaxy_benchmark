/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np
v=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  x=float(l.split()[3]);
  if x:v.append(x)
v=np.array(v)
# grid in chunks, use every unique value weighted counts
u,w=np.unique(v,return_counts=True)
print('u',len(u))
for lo,hi,step in [(.0128,.0134,1e-6),(.01295,.01325,1e-8)]:
 best=[]
 qs=np.arange(lo,hi+step/2,step)
 for qchunk in np.array_split(qs, max(1,len(qs)//10000)):
  n=np.rint(u[None,:]/qchunk[:,None])
  err=u[None,:]-n*qchunk[:,None]
  mse=(err*err*w).sum(1)
  mae=(np.abs(err)*w).sum(1)
  maxe=np.abs(err).max(1)
  for i in np.argsort(mse)[:5]:best.append((mse[i]/w.sum(),mae[i]/w.sum(),maxe[i],qchunk[i],sum(np.abs(err[i])<.0005)))
 print('range',lo,hi,'best')
 for z in sorted(best)[:20]:print(z,'N',1e6/z[3])
PY"
