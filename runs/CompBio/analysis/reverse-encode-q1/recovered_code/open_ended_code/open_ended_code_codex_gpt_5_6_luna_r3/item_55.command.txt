  if 'mapped' in q and isinstance(q['mapped'],int):vals.append(q['mapped'])
 return max(vals)
for up in [0,0.1,0.5,1,2,5,10,20,50,100,200]:
 lo=N;hi=N+int(up*1e6) if up else N
 print(f'{up}:',sum(lo<=m(x)<=hi for x in p))

# groups sum with all m; counts by candidate close
by=collections.defaultdict(list)
for x in p:by[x['dataset']].append(x)
for up in [0,0.1,0.5,1,2,5,10,20,50,100]:
