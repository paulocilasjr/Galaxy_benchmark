import gzip,numpy as np
v=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  x=float(l.split()[3]);
  if x:v.append(x)
v=np.array(v);u,w=np.unique(v,return_counts=True)
for N in [76326299,76217668,76275847,76608288,77054501,73827151]:
 q=1e6/N;n=np.rint(u/q);err=u-n*q
 print('\\nN',N,'q',q,'mse',np.average(err**2,weights=w),'mae',np.average(abs(err),weights=w),'max',abs(err).max(),'within',np.sum(w[abs(err)<.0005]),'of',w.sum())
 # most common residual range
 print('res range percentiles',np.percentile(np.repeat(err,w),[0,1,25,50,75,99,100]))
PY"
