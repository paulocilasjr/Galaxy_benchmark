/bin/bash -c "python3 - <<'PY'
import RNA, urllib.request
url='https://rnacentral.org/api/v1/rna/URS0000668495.fasta'
data=urllib.request.urlopen(url).read().decode()
seq=''.join(line.strip() for line in data.splitlines() if not line.startswith('>'))
print('len',len(seq))
print(seq)
(struct, mfe) = RNA.fold(seq)
print('structure',struct)
print('mfe raw',mfe)
print(f'{struct},{mfe:.2f}')
# also compare using fold_compound with mfe
fc=RNA.fold_compound(seq)
s,m=fc.mfe()
print(s,m, f'{m:.2f}')
PY"
