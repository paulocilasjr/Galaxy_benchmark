/bin/bash -c "python3 - <<'PY'
import RNA, urllib.request
data=urllib.request.urlopen('https://rnacentral.org/api/v1/rna/URS0000668495.fasta').read().decode()
seq=''.join(line.strip() for line in data.splitlines() if not line.startswith('>'))
s,m=RNA.fold(seq)
print(len(seq), len(s), s.count('('), s.count(')'), s.count('.'))
print(s)
print(f'{s},{m:.2f}')
# Validate balance and pseudoknot-free
st=[]; ok=True
for c in s:
 if c=='(': st.append(c)
 elif c==')':
  if not st: ok=False; break
  st.pop()
print('balanced',ok and not st)
PY"
