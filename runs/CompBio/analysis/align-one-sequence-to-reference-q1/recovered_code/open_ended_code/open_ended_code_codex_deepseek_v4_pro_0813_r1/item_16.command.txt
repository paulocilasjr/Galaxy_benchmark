/bin/bash -c "python3 - <<'PY'
from Bio import SeqIO
q='CACACACAGGAGAT'
for rec in SeqIO.parse('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna','fasta'):
    if rec.id=='NT_033779.5':
        s=str(rec.seq).upper()
        hits=[]
        start=0
        while True:
            i=s.find(q,start)
            if i<0:break
            hits.append(i)
            start=i+1
        print('hits', hits[:20], 'count',len(hits))
        i=hits[0]
        print('coords 0-based start',i,'end exclusive',i+len(q))
        print('context', s[i-20:i+len(q)+20])
        break
PY"
