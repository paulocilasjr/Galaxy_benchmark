/bin/bash -c "python3 - <<'PY'
from pathlib import Path
p=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
query='CACACACAGGAGAT'
comp=str.maketrans('ACGT','TGCA')
rc=query.translate(comp)[::-1]
print('rc',rc)
# streaming
name=None
seq=[]
pos=0
def scan(seqstr, name):
    hits=[]
    for q,label in [(query,'+'),(rc,'rc')]:
        start=0
        while True:
            i=seqstr.find(q,start)
            if i<0:break
            hits.append((name,i,i+len(q),label,seqstr[i:i+len(q)]))
            start=i+1
    return hits
with p.open() as f:
    for line in f:
        line=line.rstrip('\\n')
        if line.startswith('>'):
            if name is not None:
                s=''.join(seq)
                for h in scan(s,name):
                    print(h[0],h[1],h[2],h[3],h[4])
            name=line[1:].split()[0]
            seq=[]
            pos=0
        else:
            seq.append(line.strip())
    if name is not None:
        s=''.join(seq)
        for h in scan(s,name):
            print(h[0],h[1],h[2],h[3],h[4])
PY"
