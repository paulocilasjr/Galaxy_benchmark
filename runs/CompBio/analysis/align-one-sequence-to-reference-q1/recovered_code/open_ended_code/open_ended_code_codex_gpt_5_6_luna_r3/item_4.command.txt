        "'$p++;
      }
    }

    my $rc = reverse($q);
    $rc =~ tr/ACGT/TGCA/;
    my $p = 0;
    while (($p = index($seq, $rc, $p)) >= 0) {
      print "$id'"\\t"'$p-" . ($p + length($rc) - 1) . "'"\\t-\\n\";
      "'$p++;
    }
  }
'"' GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna"
