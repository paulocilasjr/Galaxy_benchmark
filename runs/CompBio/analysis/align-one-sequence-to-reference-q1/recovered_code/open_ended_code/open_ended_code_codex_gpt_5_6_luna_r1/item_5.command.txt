/bin/bash -lc "perl -e '
use strict; use warnings;
my "'$file = "GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna";
my $q = uc("CACACACAGGAGAT");
my $rc = reverse $q; $rc =~ tr/ACGT/TGCA/;
open my $fh, "<", $file or die $!;
local $/ = "'"\\n>\";
my "'$n = 0;
while (my $rec = <$fh>) {
  $rec =~ s/''^>//; $rec =~ s/''^'"\\n//;
  my ("'$head, @lines) = split /'"\\n/, "'$rec;
  my $id = (split /'"\\s+/, "'$head)[0];
  my $seq = uc(join("", @lines)); $seq =~ s/'"\\s+//g;
  for my "'$item ([$q,"+"] , [$rc,"-"]) {
    my ($needle,$strand) = @$item;
    my $p = 0;
    while (($p = index($seq, $needle, $p)) >= 0) {
      print "$id'"\\t"'$strand'"\\tstart="'$p'"\\tend_excl=\".("'$p+length($needle))."'"\\tend_incl=\".("'$p+length($needle)-1)."'"\\tcontext=\".substr("'$seq, $p-10, 34)."'"\\n\";
      "'$p++;
      $n++;
    }
  }
}
print STDERR "count=$n'"\\n\";
'"
