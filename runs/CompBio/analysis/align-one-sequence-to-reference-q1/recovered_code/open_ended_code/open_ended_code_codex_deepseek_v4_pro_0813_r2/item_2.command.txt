/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
p=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
seq='CACACACAGGAGAT'
comp=str.maketrans('ACGT','TGCA')
rev=seq.translate(comp)[::-1]
hits=[]
name=None; buf=[]
def scan(name,s):
    for label,q in [('fwd',seq),('rev',rev)]:
        start=0
        while True:
            i=s.find(q,start)
            if i<0: break
            hits.append((name,label,i,i+len(q)))
            start=i+1
for line in p.open():
    line=line.strip()
    if not line: continue
    if line.startswith('>'):
        if name is not None: scan(name,''.join(buf))
        name=line[1:].split()[0]; buf=[]
    else: buf.append(line.upper())
if name is not None: scan(name,''.join(buf))
print('hits',len(hits))
for h in hits[:100]: print(h)
PY"
