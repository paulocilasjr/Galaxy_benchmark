/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
f=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
query='CACACACAGGAGAT'.upper()
comp=str.maketrans('ACGTNacgtn','TGCANtgcan')
rc=query.translate(comp)[::-1]
print('query', query, len(query), 'rc', rc)
contig=None
seq=[]
matches=[]
with f.open() as fh:
    for line in fh:
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if contig is not None:
                s=''.join(seq).upper()
                for pat,strand in [(query,'+'),(rc,'-')]:
                    start=0
                    while True:
                        i=s.find(pat,start)
                        if i==-1: break
                        matches.append((contig,i,i+len(pat),strand,pat))
                        start=i+1
            contig=line[1:].split()[0]
            seq=[]
        else:
            seq.append(line)
    if contig is not None:
        s=''.join(seq).upper()
        for pat,strand in [(query,'+'),(rc,'-')]:
            start=0
            while True:
                i=s.find(pat,start)
                if i==-1: break
                matches.append((contig,i,i+len(pat),strand,pat))
                start=i+1
print('matches', len(matches))
for m in matches[:100]:
    print(*m)
PY"
