/bin/bash -lc "python - <<'PY'
import scanpy as sc
print('scanpy', sc.__version__, flush=True)
for fn in ['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']:
    ad = sc.read_h5ad(fn)
    print(fn, flush=True)
    print('shape', ad.shape, flush=True)
    print('var_names unique', ad.var_names.is_unique, flush=True)
    print('var columns', list(ad.var.columns), flush=True)
    if 'gene_symbols' in ad.var.columns:
        s=ad.var['gene_symbols']
        print('missing gene_symbols', int(s.isna().sum()), flush=True)
        print('blank gene_symbols', int((s.astype(str)=='').sum()), flush=True)
        print('first gene_symbols', list(s.head()), flush=True)
    print('first var_names', list(ad.var_names[:5]), flush=True)
PY"
