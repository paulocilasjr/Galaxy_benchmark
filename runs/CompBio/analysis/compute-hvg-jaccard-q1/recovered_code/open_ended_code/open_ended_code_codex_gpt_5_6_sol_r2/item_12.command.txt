/bin/bash -lc "PYTHONPATH=.scanpy_deps python - <<'PY'
import scanpy as sc
import pandas as pd
files=['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']
a=[sc.read_h5ad(f) for f in files]
common=a[0].var_names.intersection(a[1].var_names, sort=False)
print('common',len(common),common[:10].tolist())
sets=[]
for f,x in zip(files,a):
 x=x[:,common].copy()
 sc.pp.highly_variable_genes(x,flavor='seurat')
 accepted=['True','1','TRUE']
 mask=x.var['highly_variable'].astype(str).isin(accepted)
 s=set(x.var_names[mask].astype(str))
 sets.append(s)
 print(f,'HVG',len(s))
 print(sorted(s)[:30])
print('intersection',len(sets[0]&sets[1]),'union',len(sets[0]|sets[1]))
value=len(sets[0]&sets[1])/len(sets[0]|sets[1]) if sets[0]|sets[1] else float('nan')
print('jaccard',repr(value),'rounded',f'{value:.2f}')
PY"
