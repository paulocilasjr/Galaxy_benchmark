/bin/bash -lc "PYTHONPATH=.scanpy_deps python -u - <<'PY'
import scanpy as sc
import numpy as np
import pandas as pd

files=['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']
adatas=[sc.read_h5ad(f) for f in files]
print('loaded', [a.shape for a in adatas], flush=True)

# Neither dataset has gene_symbols; therefore there are no missing gene_symbols rows to exclude.
common=set(adatas[0].var_names.tolist()).intersection(adatas[1].var_names.tolist())
print('common',len(common),flush=True)
subs=[]
for a in adatas:
    names=[x for x in a.var_names if x in common]
    s=a[:,names].copy()
    sc.pp.highly_variable_genes(s, flavor='seurat')
    values=s.var['highly_variable'].astype(str)
    allowed=[]
    for x in ['True','True','1','TRUE']:
        if x not in allowed: allowed.append(x)
    hvgs=set(s.var_names[values.isin(allowed)].tolist())
    subs.append(hvgs)
    print('hvg',len(hvgs), sorted(hvgs)[:20], flush=True)
inter=subs[0] & subs[1]
union=subs[0] | subs[1]
print('intersection',len(inter),'union',len(union),flush=True)
print('JACCARD',len(inter)/len(union) if union else float('nan'),flush=True)
print('rounded',f'{len(inter)/len(union):.2f}' if union else 'NaN',flush=True)
PY"
