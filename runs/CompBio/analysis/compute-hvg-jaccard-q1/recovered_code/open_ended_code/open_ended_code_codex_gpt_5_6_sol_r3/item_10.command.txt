/bin/bash -lc "PYTHONPATH=.scanpy_deps NUMBA_DISABLE_JIT=1 python -u - <<'PY'
import scanpy as sc, numpy as np
from scipy import sparse
f='10x_pbmc68k_reduced.h5ad'
a=sc.read_h5ad(f)
print('FILE',f,'shape',a.shape,'X type',type(a.X),'dtype',a.X.dtype,flush=True)
print('var columns',list(a.var.columns),flush=True)
print(a.var.head(15).to_string(),flush=True)
print('var_names first',a.var_names[:15].tolist(),flush=True)
print('unique',a.var_names.is_unique,flush=True)
for c in a.var.columns:
 print('VARCOL',c,'dtype',a.var[c].dtype,'na',int(a.var[c].isna().sum()),'first',a.var[c].head().tolist(),flush=True)
x=a.X
vals=x.data if sparse.issparse(x) else np.asarray(x).ravel()
print('X min max mean',np.nanmin(vals),np.nanmax(vals),np.nanmean(vals),flush=True)
print('layers',list(a.layers.keys()),'raw',None if a.raw is None else a.raw.shape,flush=True)
PY"
