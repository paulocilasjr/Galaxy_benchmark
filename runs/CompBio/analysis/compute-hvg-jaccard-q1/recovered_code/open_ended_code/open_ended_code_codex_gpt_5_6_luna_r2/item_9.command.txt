/bin/bash -lc "python -u - <<'PY'
import warnings
warnings.filterwarnings('ignore')
import numpy as np
import scanpy as sc
print('scanpy', sc.__version__)
paths=['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']
ads=[sc.read_h5ad(p) for p in paths]
for p,a in zip(paths,ads):
 print(p, 'X', type(a.X), a.X.shape, 'min/max', float(a.X.min()), float(a.X.max()), 'layers', list(a.layers), 'raw', a.raw is not None)
 print('obs', list(a.obs.columns), 'uns keys', list(a.uns.keys()))
common = ads[0].var_names.intersection(ads[1].var_names)
print('common count', len(common), 'first', list(common[:10]))
sub=[a[:,common].copy() for a in ads]
for p,a in zip(paths,sub):
 print('sub', p, a.shape)
 try:
  sc.pp.highly_variable_genes(a, flavor='seurat')
  print('hvg columns', list(a.var.columns))
  print('hvg true count bool', int(a.var['highly_variable'].sum()))
  print('hvg type', a.var['highly_variable'].dtype, 'values', a.var['highly_variable'].value_counts(dropna=False).to_dict())
  print('hvg names', list(a.var_names[a.var['highly_variable']]))
 except Exception as e:
  print('HVG ERROR', type(e).__name__, str(e))
PY"
