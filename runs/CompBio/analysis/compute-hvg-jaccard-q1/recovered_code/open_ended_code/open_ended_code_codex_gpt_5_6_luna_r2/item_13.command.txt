/bin/bash -lc "PYTHONWARNINGS=ignore python -u - <<'PY'
import scanpy as sc
paths = ['pbmc3k.h5ad', '10x_pbmc68k_reduced.h5ad']
a, b = [sc.read_h5ad(p) for p in paths]
common = a.var_names.intersection(b.var_names)
sets = []
for adata in (a[:, common].copy(), b[:, common].copy()):
    sc.pp.highly_variable_genes(adata, flavor='seurat')
    valid = adata.var_names.notna()
    if 'gene_symbols' in adata.var:
        valid &= adata.var['gene_symbols'].notna().to_numpy()
    accepted = list(dict.fromkeys(['True', 'True', '1', 'TRUE']))
    hvg = set(adata.var_names[valid & adata.var['highly_variable'].astype(str).isin(accepted)])
    sets.append(hvg)
union = sets[0] | sets[1]
score = len(sets[0] & sets[1]) / len(union) if union else float('nan')
print(f'{score:.2f}')
PY"
