/bin/bash -c "python3 - <<'PY'
import pandas as pd, statsmodels.api as sm, numpy as np
from scipy.stats import chi2
f='work/features.tsv'
df=pd.read_csv(f,sep='\\t')
sub=df[df.label.isin(['fg','common'])].copy()
sub['y']=(sub.label=='fg').astype(int)
tf=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
mot=[c for c in df.columns if '_' in c and c.split('_')[0] in tf]
print('n fg', (sub.y==1).sum(), 'n common',(sub.y==0).sum())
print('mean gc fg',sub[sub.y==1].gc.mean(),'common',sub[sub.y==0].gc.mean())
rows=[]
for feat in tf+mot:
 X=sm.add_constant(sub[['gc',feat]])
 model=sm.Logit(sub.y,X).fit(disp=0)
 coef=model.params[feat]; se=model.bse[feat]; z=coef/se; p=model.pvalues[feat]
 or_=np.exp(coef)
 rows.append((feat,coef,se,z,p,or_))
rows=sorted(rows,key=lambda x:-x[3])
print('\\nRanked by z (positive=fg association):')
for feat,coef,se,z,p,or_ in rows:
 print(f'{feat:22s} coef {coef:+.4f} se {se:.4f} z {z:+.2f} p {p:.3e} OR {or_:.3f}')
# also GC coefficient
X=sm.add_constant(sub[['gc']]); m=sm.Logit(sub.y,X).fit(disp=0); print('\\nGC only',m.params)
PY"
