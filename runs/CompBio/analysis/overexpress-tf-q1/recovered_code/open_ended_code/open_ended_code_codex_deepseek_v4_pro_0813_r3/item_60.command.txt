/bin/bash -c "python3 - <<'PY'
from Bio import SeqIO
import time
seqs=list(SeqIO.parse('work/perturb_gained_200.fa','fasta'))
print(len(seqs), seqs[0].id, len(seqs[0].seq))
# test one motif scan first 100 seqs
from pyjaspar import jaspardb
import MOODS.scan as s, MOODS.tools as t
jdb=jaspardb()
mot=jdb.fetch_motifs(collection='CORE', tf_name='FOXA1')[0]
# construct 4xL matrix rows ACGT from counts
counts=mot.counts
m=[[counts[b][i] for i in range(mot.length)] for b in 'ACGT']
# add pseudocount maybe MOODS log_odds handles counts? It expects frequencies? pfm_to_log_odds uses counts.
lo=t.log_odds(m,t.flat_bg(4),0.01)
thr=t.threshold_from_p(lo,t.flat_bg(4),1e-4)
bg=t.flat_bg(4)
t0=time.time()
cnt=0
for r in seqs[:1000]:
 hits=s.scan_dna(str(r.seq),[lo],bg,[thr])[0]
 if len(hits): cnt+=1
print('cnt',cnt,'time',time.time()-t0)
print('first hits', list(s.scan_dna(str(seqs[0].seq),[lo],bg,[thr])[0])[:3])
PY"
