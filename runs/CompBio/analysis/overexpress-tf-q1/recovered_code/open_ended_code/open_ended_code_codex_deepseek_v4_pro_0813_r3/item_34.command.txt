/bin/bash -c "awk '{print "'$1"'"\\t\""'$2}'"' hg38.fa.fai > work/hg38.chrom.sizes
# gained perturb summits: perturb summit not within ref peaks (any overlap)
bedtools intersect -a work/perturb_peaks_summits.bed -b work/ref_peaks_peaks.narrowPeak -v > work/perturb_gained_summits.bed
# lost ref summits: ref summit not within perturb peaks
bedtools intersect -a work/ref_peaks_summits.bed -b work/perturb_peaks_peaks.narrowPeak -v > work/ref_lost_summits.bed
# common? all ref summits background maybe.
wc -l work/perturb_gained_summits.bed work/ref_lost_summits.bed
bedtools slop -i work/perturb_gained_summits.bed -g work/hg38.chrom.sizes -b 200 | bedtools getfasta -fi hg38.fa -bed - -fo work/perturb_gained_200.fa
bedtools slop -i work/ref_lost_summits.bed -g work/hg38.chrom.sizes -b 200 | bedtools getfasta -fi hg38.fa -bed - -fo work/ref_lost_200.fa
bedtools slop -i work/ref_peaks_summits.bed -g work/hg38.chrom.sizes -b 200 | bedtools getfasta -fi hg38.fa -bed - -fo work/ref_all_200.fa
ls -lh work/*200.fa"
