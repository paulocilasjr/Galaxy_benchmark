/bin/bash -c "python3 - <<'PY'
from Bio import SeqIO
from pyjaspar import jaspardb
import MOODS.scan as s, MOODS.tools as t, time
jdb=jaspardb(); mot=jdb.fetch_motifs(collection='CORE',tf_name='PAX7')[0]
m=[[mot.counts[b][i] for i in range(mot.length)] for b in 'ACGT']
lo=t.log_odds(m,t.flat_bg(4),0.01)
seqs=[str(r.seq) for r in list(SeqIO.parse('work/perturb_gained_200.fa','fasta'))[:1000]]
t0=time.time(); mx=[]
for seq in seqs:
 res=s.scan_best_hits_dna(seq,[lo],1)[0]
 mx.append(res[0].score if len(res) else None)
print('best',time.time()-t0,mx[:5])
t0=time.time(); mx=[]
thr=t.min_score(lo)
for seq in seqs:
 hits=s.scan_dna(seq,[lo],t.flat_bg(4),[thr])[0]
 mx.append(max(h.score for h in hits) if len(hits) else None)
print('scan all',time.time()-t0,mx[:5])
PY"
