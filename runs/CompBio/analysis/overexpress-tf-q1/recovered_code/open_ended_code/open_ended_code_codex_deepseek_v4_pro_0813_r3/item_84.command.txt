/bin/bash -c "python3 - <<'PY'
from Bio import SeqIO
from pyjaspar import jaspardb
import MOODS.scan as s, MOODS.tools as t, time
jdb=jaspardb(); tf_names=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
bg=t.flat_bg(4)
scans=[]
for tf in tf_names:
 for mot in jdb.fetch_motifs(collection='CORE',tf_name=tf):
  m=[[mot.counts[b][i] for i in range(mot.length)] for b in 'ACGT']
  lo=t.log_odds(m,bg,0.01); thr=t.min_score(lo)
  scans.append((tf,mot.matrix_id,lo,thr))
seqs=[str(r.seq) for r in list(SeqIO.parse('work/perturb_gained_200.fa','fasta'))[:1000]]
matrices=[x[2] for x in scans]; thresholds=[x[3] for x in scans]
t0=time.time()
for seq in seqs:
 res=s.scan_dna(seq,matrices,bg,thresholds)
 mx=[max((h.score for h in hits), default=None) for hits in res]
print('1000 seqs 13 motifs scan all time',time.time()-t0)
PY"
