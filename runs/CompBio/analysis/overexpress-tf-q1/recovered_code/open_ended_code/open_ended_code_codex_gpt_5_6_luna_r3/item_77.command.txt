/bin/bash -lc "python3 - <<'PY'
import json,numpy as np
p=json.load(open('candidate_motifs.json'));lens={k:len(v['pfm']['A']) for k,v in p.items()}
from collections import defaultdict
D={g:defaultdict(lambda:np.zeros(200)) for g in ['r','p']};N=defaultdict(int)
for g,path in [('r','all_motif_profile.ref.tsv'),('p','all_motif_profile.perturb.tsv')]:
 with open(path) as f:
  for line in f:
   a=line.split(); group=a[3].split('|')[0].split('_',1)[0];tf=a[3].split('|')[1];pos=int(a[5])-1
   if a[4]=='-':pos=199-pos
   D[g][(group,tf)][pos]+=float(a[6]);
   if int(a[5])==1:N[(group,tf)]+=1
scale=10141848/11257343
for tf,L in lens.items():
 for gr in ['gain','neutral']:
  if N[(gr,tf)]==0:continue
  lo=100-L//2;hi=lo+L
  vals=[]
  for g in ['r','p']:
   arr=D[g][(gr,tf)]/N[(gr,tf)]*(1 if g=='r' else scale); vals.append((arr[lo:hi].mean(),np.r_[arr[50:80],arr[120:150]].mean()))
  print(tf,gr,'ref',vals[0],'pert',vals[1])
PY"
