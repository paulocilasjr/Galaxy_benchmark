/bin/bash -lc "python3 - <<'PY'
from scipy.stats import fisher_exact
import csv
rows=list(csv.DictReader(open('motif_scan_peaks.tsv'),delimiter='\\t'))
for tf in sorted(set(r['TF'] for r in rows)):
 d={r['group']:r for r in rows if r['TF']==tf}
 a=int(d['gain']['hit_regions']); b=5000-a; c=int(d['loss']['hit_regions']); e=5000-c
 orr,p=fisher_exact([[a,b],[c,e]],alternative='two-sided')
 a2=int(d['gain']['hit_regions']); b2=5000-a2; c2=int(d['neutral']['hit_regions']); e2=4809-c2
 or2,p2=fisher_exact([[a2,b2],[c2,e2]],alternative='two-sided')
 print(tf,'gain/loss OR %.2f p %.1e gain/neutral OR %.2f p %.1e'%(orr,p,or2,p2))
PY"
