/bin/bash -lc "python3 - <<'PY'
from collections import Counter
from math import log2
from scipy.stats import fisher_exact
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s.translate(comp)[::-1]
def canon(s):
    r=rc(s); return s if s<r else r
def readseq(fn):
    seqs=[]; cur=[]
    for line in open(fn):
        if line[0]=='>':
            if cur: seqs.append(''.join(cur).upper()); cur=[]
        else: cur.append(line.strip())
    if cur: seqs.append(''.join(cur).upper())
    return seqs
g=readseq('gain_top3000.fa'); b=readseq('bg_top3000.fa')
for k in [6,7,8,9,10]:
    cg=Counter(); cb=Counter()
    for seqs,c in [(g,cg),(b,cb)]:
        for s in seqs:
            seen=set()
            for i in range(len(s)-k+1):
                mer=s[i:i+k]
                if 'N' in mer: continue
                seen.add(canon(mer))
            c.update(seen)
    rows=[]
    N=len(g); M=len(b)
    for mer,a in cg.items():
        c=cb.get(mer,0)
        if a>=50:
            odds,p=fisher_exact([[a,N-a],[c,M-c]], alternative='greater')
            rows.append((p,odds,a,c,mer))
    rows.sort(key=lambda x:(x[0],-x[1]))
    print('\\nk',k)
    for p,odds,a,c,mer in rows[:30]:
        print(f'{mer}\\tp={p:.1e}\\tOR={odds:.2f}\\tg={a}\\tb={c}')
PY"
