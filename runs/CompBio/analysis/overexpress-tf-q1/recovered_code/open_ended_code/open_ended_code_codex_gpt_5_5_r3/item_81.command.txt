/bin/bash -lc "python3 - <<'PY'
import re
canonical=re.compile(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y)$'"')
criteria=[(0.5,0.1),(1,0.2),(2,0.3),(3,0.5),(5,1.0)]
counts=[0]*len(criteria)
with open('diff_bins.tsv') as f:
    h=f.readline().strip().split('\\t'); idx={k:i for i,k in enumerate(h)}
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if not canonical.match(p[idx['chrom']]): continue
        z=float(p[idx['z']]); logfc=float(p[idx['logfc']]); r=int(p[idx['ref']]); q=int(p[idx['pert']])
        for i,(zc,lc) in enumerate(criteria):
            if abs(z)<zc and abs(logfc)<lc and r>=3 and q>=3:
                counts[i]+=1
print(list(zip(criteria,counts)))
PY"
