/bin/bash -lc "python3 - <<'PY'
import re
canonical=re.compile(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y)$'"')
seen=set(); i=0
with open('diff_bins.tsv') as f, open('bg_pool_z2_lfc03.bed','w') as w:
    h=f.readline().strip().split('\\t'); idx={k:i for i,k in enumerate(h)}
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        chrom=p[idx['chrom']]
        if not canonical.match(chrom): continue
        z=float(p[idx['z']]); logfc=float(p[idx['logfc']]); r=int(p[idx['ref']]); q=int(p[idx['pert']])
        if abs(z)<2 and abs(logfc)<0.3 and r>=3 and q>=3:
            start=int(p[idx['start']]); s=max(0,start+250-500); e=start+250+500
            key=(chrom,s,e)
            if key in seen: continue
            seen.add(key); w.write(f'{chrom}\\t{s}\\t{e}\\tpool_{i}\\n'); i+=1
print(i)
PY
tools/twoBitToFa hg38.2bit -bed=bg_pool_z2_lfc03.bed -bedPos -noMask bg_pool_z2_lfc03.fa && grep -c '"'^>'"' bg_pool_z2_lfc03.fa"
