/bin/bash -c "python3 - <<'EOF'
import json, numpy as np
m=json.load(open('candidate_motifs.json'))
for tf,d in m.items():
    p=d['pfm']; L=len(p['A'])
    a=np.array([p['A'],p['C'],p['G'],p['T']],float)
    consensus=''.join('ACGT'[i] for i in a.argmax(axis=0))
    gc=(a[1]+a[2]).sum()/a.sum()
    print(tf,d['id'],L,consensus,'GC',round(gc,3))
EOF"
