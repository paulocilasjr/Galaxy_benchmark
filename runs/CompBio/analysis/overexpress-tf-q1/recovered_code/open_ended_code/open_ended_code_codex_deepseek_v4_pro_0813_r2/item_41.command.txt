/bin/bash -c "Rscript - <<'EOF'
suppressMessages(library(edgeR))
ref <- read.table('ref_counts.txt', sep='\\t', header=FALSE, col.names=c('chr','start','end','count'))
pert <- read.table('perturb_counts.txt', sep='\\t', header=FALSE, col.names=c('chr','start','end','count'))
stopifnot(all(ref[,1:3]==pert[,1:3]))
counts <- cbind(ref=ref"'$count, perturb=pert$count)
rownames(counts) <- paste0(ref$chr, '"':', ref"'$start, '"'-', ref"'$end)
y <- DGEList(counts=counts, genes=ref[,1:3])
y <- calcNormFactors(y)
y$common.dispersion <- 0.1
y$trended.dispersion <- 0.1
y$tagwise.dispersion <- 0.1
et <- exactTest(y, pair=c('"'ref','perturb'))
res <- topTags(et, n=Inf)"'$table
res$FDR <- p.adjust(res$PValue, method='"'BH')
res"'$logCPM <- res$logCPM
write.table(res, file='"'edgeR_results.tsv', sep='\\t', quote=FALSE, row.names=FALSE)
cat('total', nrow(res), '\\n')
print(head(res, 10))
cat('up FDR<0.05', sum(res"'$FDR<0.05 & res$logFC>0), '"'down FDR<0.05', sum(res"'$FDR<0.05 & res$logFC<0), '"'\\n')
cat('up logFC>1', sum(res"'$logFC>1), '"'logFC>2', sum(res"'$logFC>2), '"'\\n')
EOF"
