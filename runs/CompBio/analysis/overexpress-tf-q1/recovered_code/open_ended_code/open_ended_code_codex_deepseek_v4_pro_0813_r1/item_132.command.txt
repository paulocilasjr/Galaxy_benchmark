/bin/bash -lc "python3 - <<'PY'
import re
path='data/knownTFs/all/known.motifs'
out=open('candidate_official_homer.motifs','w')
# Parse entries. Header starts with > at beginning of line.
cur=[]; header=None
pat=re.compile(r'(Klf4|KLF4|Tcf7|TCF7|TCFL2|TCF7L2|Lef1|LEF1|Pax7|PAX7|Ascl1|ASCL1|TFAP2A|AP-2alpha|AP-2gamma|RUNX1|Runx1|GATA3|Gata3|REST|NRSF|SPI1|PU\\.1|FOXA1)', re.I)
count=0
for line in open(path):
    if line.startswith('>'):
        if header is not None and cur and pat.search(header):
            out.write(header); out.writelines(cur); count+=1
        header=line; cur=[]
    else:
        if header is not None:
            cur.append(line)
if header is not None and cur and pat.search(header):
    out.write(header); out.writelines(cur); count+=1
out.close(); print('extracted',count)
PY
grep '"'^>'"' candidate_official_homer.motifs | wc -l; grep '"'^>'"' candidate_official_homer.motifs | cut -f2 | sort | head -100"
