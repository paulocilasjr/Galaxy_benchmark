/bin/bash -lc "python3 - <<'PY'
import pandas as pd, MOODS.parsers, MOODS.tools, MOODS.scan
# load a peak sequence from cond1.fa by name? names are NaN? Let's read fasta directly and find a sequence with anomaly
from Bio import SeqIO
# use simple parser
seqs=[]
for rec in SeqIO.parse('macs/bdgdiff/cond1.fa','fasta'):
    if len(rec.seq) in [538,1148,225]:
        seqs.append((rec.id,str(rec.seq).upper()))
        if len(seqs)>=1: break
seq=seqs[0][1]; print(seqs[0][0], len(seq))
pf='motif_pfms/ASCL1_MA1631.2.pfm'; bg=MOODS.tools.flat_bg(4)
lo=MOODS.parsers.pfm_to_log_odds(pf,bg,1); rc=MOODS.tools.reverse_complement(lo)
th_lo=MOODS.tools.threshold_from_p(lo,bg,1e-4); th_hi=MOODS.tools.threshold_from_p(lo,bg,1e-5)
print('th',th_lo,th_hi)
res=MOODS.scan.scan_dna(seq,[lo,rc],bg,[th_lo,th_lo],1)
for i,r in enumerate(res):
 print('res',i,len(r),[(m.pos,m.score,seq[m.pos:m.pos+4]) for m in r[:10]])
fw,rv=res
n4=len(fw)+len(rv)
n5=sum(m.score>=th_hi for m in fw)+sum(m.score>=th_hi for m in rv)
print('n4 n5',n4,n5)
# verify max and threshold
for m in list(fw)+list(rv):
 if m.score>=th_hi: print('hi',m.score,m.pos,seq[m.pos:m.pos+4])
PY"
