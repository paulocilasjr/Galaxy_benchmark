/bin/bash -lc "python3 - <<'PY'
import MOODS.scan, MOODS.tools
mat=[[0.8,0.066,0.066,0.066],[0.8,0.066,0.066,0.066],[0.066,0.8,0.066,0.066],[0.066,0.066,0.8,0.066]]
bg=MOODS.tools.flat_bg(4)
lo=MOODS.tools.log_odds(mat,bg,1)
th=MOODS.tools.threshold_from_p(lo,bg,0.01)
for seq in ['AACGAACG','TTTTTACGTTTT','AACG','CGTT','TTTTCGTTTTTT']:
 r=MOODS.scan.scan_dna(seq,[lo],bg,[th],7)
 print('seq',seq)
 for m in r[0]:
  print('pos',m.pos,'score',m.score,'strand?',getattr(m,'strand',None),'seqsub',seq[m.pos:m.pos+4])
 print()
PY"
