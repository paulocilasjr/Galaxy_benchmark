/bin/bash -lc "python3 - <<'PY'
import glob, os
from Bio import SeqIO
import MOODS.parsers, MOODS.tools, MOODS.scan
seq=str(list(SeqIO.parse('macs/bdgdiff/cond1.fa','fasta'))[1].seq).upper()
pfiles=sorted(glob.glob('motif_pfms/*.pfm'))
bg=MOODS.tools.flat_bg(4); pseudo=1.0; p_lo=1e-4; p_hi=1e-5
models=[]; mat_list=[]; th_lo_list=[]; th_hi_list=[]
for pf in pfiles:
 name=os.path.basename(pf)[:-4]
 lo=MOODS.parsers.pfm_to_log_odds(pf,bg,pseudo); rc=MOODS.tools.reverse_complement(lo)
 th_lo=MOODS.tools.threshold_from_p(lo,bg,p_lo); th_hi=MOODS.tools.threshold_from_p(lo,bg,p_hi)
 models.append((name,th_lo,th_hi)); mat_list.append(lo); mat_list.append(rc); th_lo_list.append(th_lo); th_lo_list.append(th_lo); th_hi_list.append(th_hi); th_hi_list.append(th_hi)
res=MOODS.scan.scan_dna(seq,mat_list,bg,th_lo_list,1)
for j,(name,thlo,thhi) in enumerate(models):
 fw=res[2*j]; rv=res[2*j+1]
 n4=len(fw)+len(rv)
 n5=sum(1 for m in fw if m.score>=th_hi_list[2*j])+sum(1 for m in rv if m.score>=th_hi_list[2*j+1])
 if name.startswith('ASCL1_MA1631.2'):
  print('name',name,'thlo',thlo,'thhi',thhi,'th_hi_list indices',th_hi_list[2*j],th_hi_list[2*j+1])
  print('fw',[(m.pos,m.score,seq[m.pos:m.pos+4]) for m in fw])
  print('rv',[(m.pos,m.score,seq[m.pos:m.pos+4]) for m in rv])
  print('n4 n5',n4,n5)
  print('fw score>=hi',[m.score for m in fw if m.score>=th_hi_list[2*j]])
  print('rv score>=hi',[m.score for m in rv if m.score>=th_hi_list[2*j+1]])
PY"
