/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
import pandas as pd
for s in ['cond1','common','cond2']:
 seqs=[]
 for rec in SeqIO.parse(f'macs/bdgdiff/{s}.fa','fasta'):
  seq=str(rec.seq).upper(); l=len(seq); gc=(seq.count('G')+seq.count('C'))/l if l else 0; seqs.append((l,gc))
 df=pd.DataFrame(seqs,columns=['length','gc'])
 print(s, 'peaks',len(df),'mean_len',df.length.mean(),'mean_gc',df.gc.mean(),'median_gc',df.gc.median(),'ATrich frac gc<0.35',(df.gc<0.35).mean())
PY"
