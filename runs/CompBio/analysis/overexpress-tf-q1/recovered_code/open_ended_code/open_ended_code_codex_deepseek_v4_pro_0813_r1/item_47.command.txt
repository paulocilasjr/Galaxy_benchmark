import MOODS.scan, MOODS.tools
mat=[[0.8,0.066,0.066,0.066],[0.8,0.066,0.066,0.066],[0.066,0.8,0.066,0.066],[0.066,0.066,0.8,0.066]]

for name,fn in [('scan_dna',MOODS.scan.scan_dna),('naive',MOODS.scan.naive_scan_dna),('best',MOODS.scan.scan_best_hits_dna),('scan',MOODS.scan.scan)]:
 try:
  if name in ['scan_dna','naive','best']:
   r=fn(seq,[lo],bg,[th],7)
  else:
   r=fn(seq,[lo],bg,[th],7)
  print(name,r)
  if r and r[0]:
   for m in r[0]: print(' ',m.pos,m.score,seq[m.pos:m.pos+4])
 except Exception as e: print(name,'err',e)
