/bin/bash -lc "mkdir -p motif_pfms && python3 - <<'PY'
import json, glob, os
for path in glob.glob('motifs/*.json'):
    d=json.load(open(path))
    pfm=d['pfm']  # dict A,C,G,T lists
    name=os.path.basename(path)[:-5]  # tf_mid
    out=f'motif_pfms/{name}.pfm'
    with open(out,'w') as f:
        for nuc in ['A','C','G','T']:
            vals=pfm[nuc]
            f.write(' '.join(str(int(round(x))) if float(x).is_integer() else str(x) for x in vals)+'\\n')
    print(out)
PY
head -4 motif_pfms/FOXA1_MA0148.5.pfm"
