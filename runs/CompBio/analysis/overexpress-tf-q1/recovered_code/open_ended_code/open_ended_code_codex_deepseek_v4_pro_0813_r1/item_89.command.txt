/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from scipy.stats import fisher_exact
sets=['cond1','common','cond2']; dfs={s:pd.read_csv(f'macs/bdgdiff/{s}.motif_scan.tsv',sep='\\t') for s in sets}
print('peaks', {s:len(d) for s,d in dfs.items()}, 'bp', {s:int(d.length.sum()) for s,d in dfs.items()})
for suffix in ['_p1e-4','_p1e-5']:
 cols=[c for c in dfs['cond2'].columns if c.endswith(suffix)]
 rows=[]
 for col in cols:
  motif=col[:-len(suffix)]
  a=(dfs['cond2'][col]>0).sum(); b=len(dfs['cond2'])-a
  c=(dfs['common'][col]>0).sum(); d=len(dfs['common'])-c
  odds,p=fisher_exact([[a,b],[c,d]],alternative='greater')
  a1=(dfs['cond2'][col]>0).sum(); b1=len(dfs['cond2'])-a1
  c1=(dfs['cond1'][col]>0).sum(); d1=len(dfs['cond1'])-c1
  odds1,p1=fisher_exact([[a1,b1],[c1,d1]],alternative='greater')
  rows.append({'motif':motif,'common_frac':c/len(dfs['common']),'cond2_frac':a/len(dfs['cond2']),'odds_common':odds,'p_common':p,'cond1_frac':c1/len(dfs['cond1']),'odds_cond1':odds1,'p_cond1':p1,'cond2_hits_per_kb':dfs['cond2'][col].sum()/dfs['cond2'].length.sum()*1000})
 res=pd.DataFrame(rows).sort_values('odds_common',ascending=False)
 pd.set_option('display.width',250); pd.set_option('display.max_rows',100)
 print('\\n===',suffix,'cond2 vs common')
 print(res.to_string(index=False, float_format=lambda x:f'{x:,.4g}'))
PY"
