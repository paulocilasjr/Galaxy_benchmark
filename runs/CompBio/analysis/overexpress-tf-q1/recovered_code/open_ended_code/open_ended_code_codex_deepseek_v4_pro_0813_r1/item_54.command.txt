import MOODS.scan, MOODS.tools
mat_rows = [[0.8,0.8,0.066,0.066],[0.066,0.066,0.8,0.066],[0.066,0.066,0.066,0.8],[0.066,0.066,0.066,0.066]]

for seq in ['AACG','AACGAACG','CGTT','TTTTCGTTTTTT']:
 r=MOODS.scan.scan_dna(seq,[lo,rcl],bg,[th,th],7)
 print(seq)
 for mi,res in enumerate(r):
  print(mi,[(m.pos,m.score,seq[m.pos:m.pos+4]) for m in res])
