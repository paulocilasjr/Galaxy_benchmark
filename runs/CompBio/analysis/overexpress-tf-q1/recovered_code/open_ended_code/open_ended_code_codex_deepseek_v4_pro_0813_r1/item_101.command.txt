/bin/bash -lc "cat > scan_shuffled_motifs.py <<'PY'
import os,glob,time,random
from Bio import SeqIO
import MOODS.parsers,MOODS.tools,MOODS.scan
random.seed(42)
pfiles=sorted(glob.glob('motif_pfms/*.pfm')); bg=MOODS.tools.flat_bg(4); pseudocount=1.0; p=1e-4
models=[]; mats=[]; ths=[]
for pf in pfiles:
 name=os.path.basename(pf)[:-4]; lo=MOODS.parsers.pfm_to_log_odds(pf,bg,pseudocount); rc=MOODS.tools.reverse_complement(lo); th=MOODS.tools.threshold_from_p(lo,bg,p)
 models.append(name); mats += [lo,rc]; ths += [th,th]
def shuffle_seq(s):
 arr=list(s); random.shuffle(arr); return ''.join(arr)
for s in ['cond1','common','cond2']:
 out=open(f'macs/bdgdiff/{s}.shuffled_motif_scan.tsv','w')
 out.write('set\\tlength\\t'+'\\t'.join([f'{m}_orig' for m in models]+[f'{m}_shuf' for m in models])+'\\n')
 t0=time.time()
 for i,rec in enumerate(SeqIO.parse(f'macs/bdgdiff/{s}.fa','fasta')):
  seq=str(rec.seq).upper(); sh=shuffle_seq(seq)
  res_o=MOODS.scan.scan_dna(seq,mats,bg,ths,1)
  res_s=MOODS.scan.scan_dna(sh,mats,bg,ths,1)
  vals=[]
  for j in range(len(models)):
   vals.append(len(res_o[2*j])+len(res_o[2*j+1]))
  for j in range(len(models)):
   vals.append(len(res_s[2*j])+len(res_s[2*j+1]))
  out.write(s+'\\t'+str(len(seq))+'\\t'+'\\t'.join(str(v) for v in vals)+'\\n')
  if (i+1)%10000==0: print(s,i+1,'elapsed',round(time.time()-t0,1),flush=True)
 out.close(); print(s,'done',round(time.time()-t0,1))
PY
python3 scan_shuffled_motifs.py"
