/bin/bash -lc "python3 - <<'PY'
import gzip
for fn in ['overexpress.tf.q1.ref.bed.gz','overexpress.tf.q1.perturb.bed.gz']:
 last={}; bad=[]; n=0
 with gzip.open(fn,'rt') as f:
  for line in f:
   c,s,e=line.split()[:3]; s=int(s)
   if c in last and s<last[c]:
    bad.append((n,c,last[c],s));
    if len(bad)>=3: break
   last[c]=s;n+=1
 print(fn,'n',n,'bad',bad)
PY"
