/bin/bash -lc "wc -l results/peaks/*narrowPeak results/peaks/*summits.bed
cut -f1-3 results/peaks/ref_peaks.narrowPeak results/peaks/perturb_peaks.narrowPeak | LC_ALL=C sort -k1,1 -k2,2n | bedtools merge -i - > results/union_peaks.bed
wc -l results/union_peaks.bed
bedtools coverage -a results/union_peaks.bed -b overexpress.tf.q1.ref.bed.gz -counts > results/union_ref_counts.bed
bedtools coverage -a results/union_peaks.bed -b overexpress.tf.q1.perturb.bed.gz -counts > results/union_perturb_counts.bed
paste results/union_ref_counts.bed results/union_perturb_counts.bed | awk 'BEGIN{OFS=\"\\t\"}{print "'$1,$2,$3,$4,$8}'"' > results/union_counts.tsv
head results/union_counts.tsv"
