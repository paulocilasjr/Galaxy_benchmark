def selective_scan_ref(u, delta, A, B, C, D, z, delta_bias):
    # copied from mamba_ssm selective_scan_ref
    dtype_in=u.dtype
    u=u.float(); delta=delta.float(); delta=delta+delta_bias[...,None].float(); delta=F.softplus(delta)
    batch,dim,dstate=u.shape[0],A.shape[0],A.shape[1]
    B=B.float(); C=C.float()
    x=A.new_zeros((batch,dim,dstate))
    ys=[]
    deltaA=torch.exp(torch.einsum('bdl,dn->bdln', delta, A))
    deltaB_u=torch.einsum('bdl,bnl,bdl->bdln', delta, B, u)
    for i in range(u.shape[2]):
        x=deltaA[:,:,i]*x + deltaB_u[:,:,i]
        y=torch.einsum('bdn,bn->bd', x, C[:,:,i])
        ys.append(y)
    y=torch.stack(ys,dim=2)
    out=y + u * D.view(1,-1,1)
    out=out*F.silu(z)
    return out.to(dtype=dtype_in)

def mamba_ref(hidden, m):
    B,L,_=hidden.shape
    xz=F.linear(hidden, m.in_proj.weight, m.in_proj.bias)
    x,z=xz.chunk(2,-1)
    x=x.transpose(1,2)
    x=F.silu(m.conv1d(x)[...,:L])

    return out+rev,residual
