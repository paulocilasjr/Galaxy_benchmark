/bin/bash -lc "cd /tmp
for host in hf-mirror.com huggingface.co cdn-lfs.huggingface.co cdn-lfs-us-1.hf.co raw.githubusercontent.com github.com api.genome.ucsc.edu; do
  printf '%s: ' \""'$host"
  host "$host" >/dev/null 2>&1 && echo -n resolve_ok || echo -n resolve_fail
  curl -4 -L --max-time 8 -s -o /dev/null -w '"' http=%{http_code} err=%{errormsg}' \"https://"'$host/" 2>/dev/null || true
  echo
done'
