/bin/bash -lc "W=[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-6e845b76af504ee4e726ec8c/workspace
cd /tmp
IP=128.114.119.132
for spec in '7792299_7793299' '7792298_7793299' '7792299_7793300'; do
  IFS='_' read start end <<< \""'$spec"
  curl -4 --resolve api.genome.ucsc.edu:443:$IP -L --max-time 60 -s "https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=$start;end=$end" -o "$W/seq_$spec.json"
  echo "$spec status=$? size=$(wc -c < "$W/seq_$spec.json")"
done
cd /tmp && python - <<'"'PY'
import json, os
W='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-6e845b76af504ee4e726ec8c/workspace'
for f in sorted(os.listdir(W)):
    if f.startswith('seq_'):
        d=json.load(open(os.path.join(W,f)))
        print(f, len(d['dna']), d['dna'][:80], d['dna'][-80:])
PY"
