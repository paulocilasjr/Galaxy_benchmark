/bin/bash -lc "python - <<'PY'
import sys, torch, pathlib
sys.path.insert(0,'.')
from safetensors.torch import load_file
from cadmodel.configuration_caduceus import CaduceusConfig
from cadmodel.modeling_caduceus import CaduceusForMaskedLM
import mamba_ssm.modules.mamba_simple as ms
from mamba_ssm.ops.selective_scan_interface import selective_scan_ref
saved={}; orig=ms.selective_scan_fn
def wrap(u,delta,A,B,C,D=None,z=None,delta_bias=None,delta_softplus=False,return_last_state=False):
    if not saved:
        saved.update(u=u.detach().clone(),delta=delta.detach().clone(),A=A.detach().clone(),B=B.detach().clone(),C=C.detach().clone(),D=D.detach().clone(),z=z.detach().clone(),delta_bias=delta_bias.detach().clone())
    return orig(u,delta,A,B,C,D,z,delta_bias,delta_softplus,return_last_state)
ms.selective_scan_fn=wrap
seq=pathlib.Path('sequence.fa').read_text().split('\\n')[1].strip(); mapping={'A':7,'C':8,'G':9,'T':10}
ids=torch.tensor([[mapping[c] for c in seq]],dtype=torch.long)
model=CaduceusForMaskedLM(CaduceusConfig.from_pretrained('./cadmodel')); model.load_state_dict(load_file('model.safetensors'),strict=False); model.tie_weights(); model.eval(); model.to('mps'); ids=ids.to('mps')
B=2; batch=[]
for i in range(B): x=ids[0].clone(); x[i]=3; batch.append(x)
with torch.no_grad(): out=model(input_ids=torch.stack(batch))
def call(dev):
 args={k:saved[k].to(dev) for k in ['u','delta','A','B','C','D','z','delta_bias']}
 o=selective_scan_ref(**args,delta_softplus=True)
 if isinstance(o,tuple): o=o[0]
 return o
print('CPU scan...'); oc=call('cpu'); print('cpu finite',torch.isfinite(oc).all().item(),'nan',torch.isnan(oc).sum().item(),'inf',torch.isinf(oc).sum().item())
print('MPS scan...'); om=call('mps'); print('mps finite',torch.isfinite(om).all().item(),'nan',torch.isnan(om).sum().item(),'inf',torch.isinf(om).sum().item())
print('cpu first',oc[0,0,:5]); print('mps first',om[0,0,:5].cpu())
PY"
