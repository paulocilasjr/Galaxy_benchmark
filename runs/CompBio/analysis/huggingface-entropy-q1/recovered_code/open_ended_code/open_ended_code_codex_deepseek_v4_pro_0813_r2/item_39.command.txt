/bin/bash -lc "python - <<'PY'
import sys, torch, time
sys.path.insert(0,'.')
from safetensors.torch import load_file
from cadmodel.configuration_caduceus import CaduceusConfig
from cadmodel.modeling_caduceus import CaduceusForMaskedLM
config=CaduceusConfig.from_pretrained('./cadmodel')
print('config d_model,n_layer,vocab,bidir,rms,fused',config.d_model,config.n_layer,config.vocab_size,config.bidirectional,config.rms_norm,config.fused_add_norm)
model=CaduceusForMaskedLM(config)
state=load_file('model.safetensors')
missing, unexpected = model.load_state_dict(state, strict=False)
print('missing', missing[:10], len(missing))
print('unexpected', unexpected[:10], len(unexpected))
print('params', sum(p.numel() for p in model.parameters()))
model.eval()
ids=torch.tensor([[7,8,9,10,3,7,8,9]], dtype=torch.long)
t=time.time()
with torch.no_grad():
 out=model(input_ids=ids)
print('time', time.time()-t, 'shape', out.logits.shape, out.logits[0,4,:])
PY"
