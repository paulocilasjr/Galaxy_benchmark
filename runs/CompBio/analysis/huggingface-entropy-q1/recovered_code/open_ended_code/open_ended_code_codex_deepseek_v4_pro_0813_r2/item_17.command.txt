/bin/bash -lc "python - <<'PY'
import requests, tarfile, io
for ver in ['1.2.2','2.2.2']:
 url=f'https://files.pythonhosted.org/packages/source/m/mamba-ssm/mamba_ssm-{ver}.tar.gz'
 r=requests.get(url)
 print(ver, r.status_code, len(r.content) if r.ok else '')
 if r.ok:
  tf=tarfile.open(fileobj=io.BytesIO(r.content))
  names=tf.getnames()
  setup=next(n for n in names if n.endswith('setup.py'))
  data=tf.extractfile(setup).read().decode()
  print('torch.cuda' in data, 'mac' in data, data[:150])
PY"
