/bin/bash -lc "python - <<'PY'
import requests, json, pathlib
url='https://api.genome.ucsc.edu/getData/sequence'
params={'genome':'hg38','chrom':'chr12','start':7792298,'end':7793298}
r=requests.get(url,params=params,timeout=20)
r.raise_for_status()
d=r.json()
seq=d['dna'].upper()
print('len',len(seq), 'unique', sorted(set(seq)), 'first', seq[:20], 'last', seq[-20:])
pathlib.Path('sequence.fa').write_text('>chr12:7792299-7793298\\n'+seq+'\\n')
print('saved')
print(seq)
PY"
