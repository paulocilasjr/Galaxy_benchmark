import time,json,torch
from pure_caduceus import load_model
print('torch',torch.__version__,'mps',torch.backends.mps.is_available(), flush=True)

for dev in ['cpu','mps']:

  t=time.time()
  with torch.no_grad(): y=m(x)
  if dev=='mps': torch.mps.synchronize()
  print(dev,'elapsed',time.time()-t,'shape',tuple(y.shape),'finite',bool(torch.isfinite(y).all()),'sample',y[0,0,:12].detach().cpu().tolist(),flush=True)
 except Exception as e:
  print(dev,'ERR',type(e).__name__,e,flush=True)
