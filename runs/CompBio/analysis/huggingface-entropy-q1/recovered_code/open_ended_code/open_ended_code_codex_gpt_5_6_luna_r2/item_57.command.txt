/bin/bash -lc "python - <<'PY'
import time,torch
from pure_caduceus import load_model
base=torch.tensor([[7,8,9,10]*250]*8)
for bs in [2,4,8,16,32]:
 try:
  m=load_model().eval().to('mps'); ids=base[:bs].to('mps'); t=time.time()
  with torch.no_grad(): y=m(ids)
  torch.mps.synchronize(); print('bs',bs,'sec',time.time()-t,'shape',tuple(y.shape),'finite',bool(torch.isfinite(y).all()),flush=True)
 except Exception as e: print('bs',bs,'ERR',type(e).__name__,e,flush=True)
PY"
