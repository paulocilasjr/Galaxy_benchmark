/bin/bash -lc "python -u - <<'PY'
import sys,time,torch
sys.path.insert(0,'scan_build')
from transformers import AutoTokenizer,AutoModelForMaskedLM
seq=open('data/sequence.txt').read().strip()
tok=AutoTokenizer.from_pretrained('model_files',trust_remote_code=True,local_files_only=True)
base=tok(seq,return_tensors='pt',add_special_tokens=False).input_ids
model=AutoModelForMaskedLM.from_pretrained('model_files',trust_remote_code=True,local_files_only=True).eval()
print('threads',torch.get_num_threads(),flush=True)
for b in [1,2,4,8]:
 x=base.repeat(b,1)
 with torch.no_grad():
  t=time.time(); y=model(x).logits; y.sum().item(); dt=time.time()-t
 print('b',b,'sec',dt,'per',dt/b,flush=True)
PY"
