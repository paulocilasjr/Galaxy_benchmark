/bin/bash -lc "mkdir -p data && curl -L --fail --silent --show-error 'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=7792299;end=7793299' -o data/hg38_chr12_7792299_7793299.json && python - <<'PY'
import json
p='data/hg38_chr12_7792299_7793299.json'
d=json.load(open(p))
seq=d['dna'].upper()
print(d['chrom'],d['start'],d['end'],len(seq),seq[:80],seq[-80:])
open('data/sequence.txt','w').write(seq+'\\n')
PY"
