/bin/bash -lc "PYTHONPATH=./packages:. python - <<'PY'
import os, sys, torch, json, urllib.request, time
sys.path[:0]=[os.path.abspath('.'), os.path.abspath('packages')]
from model.configuration_caduceus import CaduceusConfig
from model.modeling_caduceus import CaduceusForMaskedLM
from model.tokenization_caduceus import CaduceusTokenizer
seq=json.loads(urllib.request.urlopen('https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=7792299;end=7793299').read().decode())['dna'].upper()
tok=CaduceusTokenizer(model_max_length=131072)
config=CaduceusConfig.from_pretrained('model')
model=CaduceusForMaskedLM.from_pretrained('model', config=config).eval().to('mps')
bs=32
ids=torch.tensor(tok(seq)['input_ids'], device='mps').unsqueeze(0).repeat(bs,1)
for i in range(bs): ids[i,i]=tok.mask_token_id
st=time.time()
with torch.no_grad(): out=model(ids).logits
print(out.shape, time.time()-st)
print(out[0,0,7:11].cpu())
PY"
