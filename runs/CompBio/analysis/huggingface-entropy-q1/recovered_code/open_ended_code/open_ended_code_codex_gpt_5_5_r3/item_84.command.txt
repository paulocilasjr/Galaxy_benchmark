/bin/bash -lc ". .venv/bin/activate && python3 - <<'PY'
import json, statistics
r=json.load(open('outputs/caduceus_entropy_results.json'))

    print(key, 'min', min(vals), 'max', max(vals), 'mean', statistics.mean(vals))
    pairs=sorted(enumerate(vals), key=lambda x:x[1])[:20]
    print(pairs)
print(r['sequence'][:100])
PY"
