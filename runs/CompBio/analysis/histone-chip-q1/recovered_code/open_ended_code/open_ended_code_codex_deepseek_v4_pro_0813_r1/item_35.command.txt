/bin/bash -lc "cat > scripts/make_bins.py <<'PY'
import sys
genes='ref/genes.pc.longest.tsv'
out='bins/bins.bed'
idx=0
with open(genes) as f, open(out,'w') as o:
    next(f)
    for line in f:
        p=line.rstrip().split('\\t')
        gene,chrom,strand=p[0],p[1],p[2]; start=int(p[3]); end=int(p[4])
        if strand=='+':
            tss=start; tes=end
        else:
            tss=end; tes=start
        # TSS bins 100
        for b in range(100):
            s=tss-5000+b*100
            e=s+100
            o.write(f'g{idx}\\t{chrom}\\t{max(0,s)}\\t{e}\\t{strand}\\n')
        # TES bins 100
        for b in range(100):
            s=tes-5000+b*100
            e=s+100
            o.write(f'g{idx}\\t{chrom}\\t{max(0,s)}\\t{e}\\t{strand}\\n')
        # body bins 50 scaled over entire tx
        length=end-start
        for b in range(50):
            s=start + int(length*b/50)
            e=start + int(length*(b+1)/50)
            if e<=s: e=s+1
            o.write(f'g{idx}\\t{chrom}\\t{s}\\t{e}\\t{strand}\\n')
        idx+=1
print('genes',idx)
PY
mkdir -p bins && python3 scripts/make_bins.py && wc -l bins/bins.bed && head bins/bins.bed"
