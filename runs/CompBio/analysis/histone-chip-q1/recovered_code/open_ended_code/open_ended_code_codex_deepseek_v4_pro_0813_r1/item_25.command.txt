/bin/bash -lc "python3 - <<'PY'
import gzip
path='ref/refGene.txt.gz'
canon=set(['chr'+str(i) for i in range(1,23)]+['chrX','chrY','chrM'])
best={}
with gzip.open(path,'rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<16: continue
        chrom=p[2]; strand=p[3]; txStart=int(p[4]); txEnd=int(p[5]); cdsStart=int(p[6]); cdsEnd=int(p[7]); gene=p[12]
        if chrom not in canon: continue
        if cdsStart>=cdsEnd: continue
        length=txEnd-txStart
        if length<2000: continue
        if gene not in best or length>best[gene][4]:
            best[gene]=(chrom,strand,txStart,txEnd,length)
print('pc genes',len(best))
with open('ref/genes.pc.longest.tsv','w') as out:
    out.write('gene\\tchrom\\tstrand\\ttxStart\\ttxEnd\\n')
    for gene,(chrom,strand,txStart,txEnd,length) in sorted(best.items(), key=lambda x:(x[1][0],x[1][2])):
        out.write(f'{gene}\\t{chrom}\\t{strand}\\t{txStart}\\t{txEnd}\\n')
PY
wc -l ref/genes.pc.longest.tsv; head ref/genes.pc.longest.tsv"
