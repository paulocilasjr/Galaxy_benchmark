 print(f.get('accession'),f.get('file_size'),f.get('output_type'),f.get('href'),f.get('dataset'),f.get('biological_replicates'),f.get('lab'),f.get('target',{}).get('label') if isinstance(f.get('target'),dict) else f.get('target'))
PY"
