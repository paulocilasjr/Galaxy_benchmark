/bin/bash -lc 'sig_primary_total=$(awk '"'"'$1 ~ /''^chr([0-9]+|X|Y)$/ {s+=$2} END{print s}'"' signal.chrom_counts.tsv)
ctrl_primary_total="'$(awk '"'"'$1 ~ /''^chr([0-9]+|X|Y)$/ {s+=$2} END{print s}'"' control.chrom_counts.tsv)
primary_bp="'$(awk '"'{s+="'$2} END{print s}'"' hg38.primary.chrom.sizes)
for f in tss_pm1kb tss_pm2kb tss_upstream_5to10kb tss_downstream_5to10kb gene_body_excl_tss2kb; do
  bedtools coverage -sorted -g hg38.primary.chrom.sizes -a "'${f}.merged.primary.bed -b <(gzip -cd histone.chip.q1.signal.tagalign.gz | awk '"'"'$1 ~ /''^chr([0-9]+|X|Y)$/'"') -counts | awk -v f="'$f -v b=signal -v total=$sig_primary_total -v gbp=$primary_bp '"'BEGIN{OFS=\"\\t\"}{c+="'$4; bp+=$3-$2} END{print f,b,c,bp,total,(c/total)/(bp/gbp)}'"'
  bedtools coverage -sorted -g hg38.primary.chrom.sizes -a "'${f}.merged.primary.bed -b <(gzip -cd histone.chip.q1.control.tagalign.gz | awk '"'"'$1 ~ /''^chr([0-9]+|X|Y)$/'"') -counts | awk -v f="'$f -v b=control -v total=$ctrl_primary_total -v gbp=$primary_bp '"'BEGIN{OFS=\"\\t\"}{c+="'$4; bp+=$3-$2} END{print f,b,c,bp,total,(c/total)/(bp/gbp)}'"'
done > annotation_density.primary.tsv
cat annotation_density.primary.tsv"
