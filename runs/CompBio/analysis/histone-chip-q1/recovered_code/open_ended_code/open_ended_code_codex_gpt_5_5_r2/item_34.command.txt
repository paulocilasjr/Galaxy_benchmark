/bin/bash -lc "awk '"'$1 ~ /''^chr([0-9]+|X|Y)$/ {print}'"' hg38.chrom.sizes > hg38.main.chrom.sizes
bedtools complement -i genes_full.merged.bed -g hg38.main.chrom.sizes > intergenic.bed
for kind in signal control; do bedtools coverage -counts -a intergenic.bed -b <(gzip -cd histone.chip.q1."'${kind}.tagalign.gz) > intergenic.${kind}.counts.bed; done
python3 - <<'"'PY'
N={'signal':44898157,'control':27109157}
for base in ['intergenic.bed','genes_full.merged.bed','promoters_2kb.merged.bed','gene_bodies_no_prom.merged.bed']:
 print('\\n'+base)
 vals={}
 for kind in ['signal','control']:
  fn=f'{base}.{kind}.counts.bed' if base=='intergenic.bed' else f'{base}.{kind}.counts.bed'
  bp=cnt=0
  with open(fn) as f:
   for line in f:
    a=line.split('\\t'); bp+=int(a[2])-int(a[1]); cnt+=int(a[3])
  dens=cnt/N[kind]*1e6/bp
  vals[kind]=dens
  print(kind, 'Mb', round(bp/1e6,2), 'reads', cnt, 'rpm_per_bp', dens)
 print('norm fold S/C', vals['signal']/vals['control'])
PY"
