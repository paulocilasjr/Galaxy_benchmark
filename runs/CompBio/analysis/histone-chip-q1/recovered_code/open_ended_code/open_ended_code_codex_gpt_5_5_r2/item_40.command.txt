/bin/bash -lc "wget -q -O cpgIslandExt.txt.gz https://hgdownload.soe.ucsc.edu/goldenPath/hg38/database/cpgIslandExt.txt.gz && gzip -cd cpgIslandExt.txt.gz | awk 'BEGIN{OFS=\"\\t\"} "'$2 ~ /''^chr([0-9]+|X|Y)$/ {print $2,$3,$4}'"' | sort -k1,1 -k2,2n | bedtools merge -i - > cpgIslandExt.merged.bed && wc -l cpgIslandExt.merged.bed
for kind in signal control; do bedtools coverage -counts -a cpgIslandExt.merged.bed -b <(gzip -cd histone.chip.q1."'${kind}.tagalign.gz) > cpgIslandExt.${kind}.counts.bed; done
python3 - <<'"'PY'
N={'signal':44898157,'control':27109157}
vals={}
for kind in ['signal','control']:
 bp=cnt=0
 with open(f'cpgIslandExt.{kind}.counts.bed') as f:
  for line in f:
   a=line.split('\\t'); bp+=int(a[2])-int(a[1]); cnt+=int(a[3])
 dens=cnt/N[kind]*1e6/bp; vals[kind]=dens
 print(kind, 'Mb', bp/1e6, 'reads', cnt, 'rpm_per_bp', dens)
print('fold', vals['signal']/vals['control'])
PY"
