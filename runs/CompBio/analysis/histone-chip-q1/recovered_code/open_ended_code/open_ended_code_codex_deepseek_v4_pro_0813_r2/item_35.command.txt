/bin/bash -lc "python3 - <<'PY'
import gzip
order={}
with open('refs/hg38.chrom.sizes') as f:
    for i,line in enumerate(f): order[line.split()[0]]=i
# categorize cCREs; filter chrom in order; preserve all fields
cats={}
with open('refs/GRCh38-cCREs.bed') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<6 or p[0] not in order: continue
        group=p[5]
        # primary category: PLS if contains PLS, pELS, dELS, CTCF-only, DNase-H3K4me3
        if group.startswith('PLS'): cat='PLS'
        elif group.startswith('pELS'): cat='pELS'
        elif group.startswith('dELS'): cat='dELS'
        elif group.startswith('CTCF-only'): cat='CTCF_only'
        elif group.startswith('DNase-H3K4me3'): cat='DNase_H3K4me3'
        else: cat='other'
        cats.setdefault(cat,[]).append((order[p[0]], int(p[1]), int(p[2]), line))
for cat,rows in cats.items():
    rows.sort(key=lambda x:(x[0],x[1],x[2]))
    with open(f'refs/ccre_{cat}.bed','w') as f:
        for _,_,_,line in rows: f.write(line)
print({k:len(v) for k,v in cats.items()})
# cpg islands sorted by order
rows=[]
with gzip.open('refs/cpgIslandExt.txt.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<3 or p[0] not in order: continue
        rows.append((order[p[0]],int(p[1]),int(p[2]),f'{p[0]}\\t{p[1]}\\t{p[2]}\\t{p[4] if len(p)>4 else \"cpg\"}'))
rows.sort(key=lambda x:(x[0],x[1],x[2]))
with open('refs/cpg_islands.bed','w') as f:
    for _,_,_,line in rows: f.write(line+'\\n')
print('cpg islands',len(rows))
# blacklist sorted
rows=[]
with gzip.open('refs/hg38-blacklist.v2.bed.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<3 or p[0] not in order: continue
        rows.append((order[p[0]],int(p[1]),int(p[2]),line))
rows.sort(key=lambda x:(x[0],x[1],x[2]))
with open('refs/blacklist.bed','w') as f:
    for _,_,_,line in rows: f.write(line)
print('blacklist',len(rows))
PY
wc -l refs/ccre_*.bed refs/cpg_islands.bed refs/blacklist.bed"
