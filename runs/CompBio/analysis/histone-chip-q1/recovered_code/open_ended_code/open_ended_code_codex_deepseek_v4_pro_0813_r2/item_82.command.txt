/bin/bash -lc "curl -L -sS --get 'https://www.encodeproject.org/search/' --data-urlencode 'type=File' --data-urlencode 'file_format=tagAlign' --data-urlencode 'limit=all' --data-urlencode 'format=json' -o refs/encode_tagalign_allfmt.json; python3 - <<'PY'
import json,collections
j=json.load(open('refs/encode_tagalign_allfmt.json')); g=j.get('@graph',[]); print('total tagAlign files',len(g))
if g:
    c=collections.Counter(f.get('assembly') for f in g); print(c)
    # matching sizes
    target={321498264,199777948}
    for f in g:
        if f.get('file_size') in target:
            print('SIZE MATCH', f.get('accession'), f.get('file_size'), f.get('assembly'), f.get('assay_term_name'), f.get('dataset'))
PY"
