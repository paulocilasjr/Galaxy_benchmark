/bin/bash -lc "python3 - <<'PY'
import gzip
order={}
with open('refs/hg38.chrom.sizes') as f:
    for i,line in enumerate(f): order[line.split()[0]]=i
for fn in ['histone.chip.q1.signal.tagalign.gz','histone.chip.q1.control.tagalign.gz']:
    last_key=None; last_chrom=None; bad=0; total=0
    with gzip.open(fn,'rt') as f:
        for line in f:
            p=line.split('\\t')
            key=(order.get(p[0],999), int(p[1]))
            if last_key is not None and key<last_key:
                bad+=1
                if bad<=5: print('bad',fn,last_chrom,last_key,p[0],p[1])
                if bad>10: break
            last_key=key; last_chrom=p[0]; total+=1
    print(fn,'total',total,'bad',bad)
PY"
