/bin/bash -lc "python3 - <<'PY'
import gzip
order=[]; seen=set()
with gzip.open('histone.chip.q1.signal.tagalign.gz','rt') as f:
    for line in f:
        c=line.split('\\t')[0]
        if c not in seen:
            seen.add(c); order.append(c)
with open('refs/tagalign_order.txt','w') as f:
    for c in order: f.write(c+'\\n')
print('n chroms',len(order))
print('\\n'.join(order[:30]))
PY
cd refs && for f in canonical_genes.bed tss_2kb.bed gene_bodies.bed canonical_exons.bed ccre_CTCF_only.bed ccre_DNase_H3K4me3.bed ccre_PLS.bed ccre_pELS.bed ccre_dELS.bed cpg_islands.bed blacklist.bed; do bedtools sort -g tagalign_order.txt -i \""'$f" > tmp_sorted.bed && mv tmp_sorted.bed "$f"; done
head -5 refs/tss_2kb.bed'
