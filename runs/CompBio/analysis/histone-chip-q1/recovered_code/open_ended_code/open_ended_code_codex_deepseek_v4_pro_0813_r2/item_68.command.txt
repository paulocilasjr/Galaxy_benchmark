/bin/bash -lc "python3 - <<'PY'
order={}
with open('refs/lex.genome') as f:
    for i,line in enumerate(f): order[line.split()[0]]=i
fn='macs3_broad/histone_broad_peaks.broadPeak'
rows=[]
with open(fn) as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        rows.append((order.get(p[0],999),int(p[1]),int(p[2]),line))
rows.sort(key=lambda x:(x[0],x[1],x[2]))
with open('macs3_broad/peaks.sorted.bed','w') as f:
    for _,_,_,line in rows: f.write(line)
print(len(rows))
PY
for feat in refs/repeats/LINE.bed refs/repeats/LTR.bed refs/repeats/Satellite.bed refs/repeats/SINE.bed refs/repeats/Simple_repeat.bed refs/repeats/Retroposon.bed refs/cpg_islands.bed refs/tss_2kb.bed refs/canonical_genes.bed refs/blacklist.bed refs/ccre_PLS.bed refs/ccre_pELS.bed refs/ccre_dELS.bed refs/ccre_CTCF_only.bed; do
  n="'$(bedtools intersect -sorted -g refs/lex.genome -a macs3_broad/peaks.sorted.bed -b "$feat" -wa -u | wc -l | tr -d '"' ')
  echo \""'$(basename $feat) $n"
done'
