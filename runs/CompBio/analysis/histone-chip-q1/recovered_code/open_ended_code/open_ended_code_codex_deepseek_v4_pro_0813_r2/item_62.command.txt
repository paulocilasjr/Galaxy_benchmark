/bin/bash -lc 'set -e
for sample in signal control; do
  tag=histone.chip.q1.${sample}.lexsort.tagalign.gz
  for cls in SINE LINE LTR DNA Simple_repeat Satellite Low_complexity Retroposon snRNA rRNA tRNA; do
    echo "START $sample $cls $(date +%T)"
    bedtools map -sorted -g refs/lex.genome -a refs/repeats/${cls}.bed -b ${tag} -c 1 -o count > counts/${sample}_rmsk_${cls}.txt
    echo "DONE $sample $cls $(date +%T) lines $(wc -l < counts/${sample}_rmsk_${cls}.txt)"
  done
done'
