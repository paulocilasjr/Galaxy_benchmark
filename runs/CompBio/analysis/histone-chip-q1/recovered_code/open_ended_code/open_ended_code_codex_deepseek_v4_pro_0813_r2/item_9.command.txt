/bin/bash -lc "python3 - <<'PY'
import gzip, collections, sys, time
for fn in ['histone.chip.q1.signal.tagalign.gz','histone.chip.q1.control.tagalign.gz']:
    counts=collections.Counter(); tot=0; lens=0; strands=collections.Counter(); starts=collections.Counter(); ends=collections.Counter()
    t=time.time()
    with gzip.open(fn,'rt') as f:
        for line in f:
            p=line.rstrip('\\n').split('\\t')
            if len(p)<6: continue
            chrom=p[0]; s=int(p[1]); e=int(p[2]); strand=p[5]
            counts[chrom]+=1; tot+=1; lens += e-s; strands[strand]+=1; starts[chrom]+=s; ends[chrom]+=e
    print(fn, 'reads',tot, 'mean_len',lens/tot, 'strands',dict(strands), 'nchrom',len(counts), flush=True)
    for c,n in counts.most_common(30):
        print(' ', c,n, 'mean_start', starts[c]/n if n else 0, 'mean_end', ends[c]/n if n else 0)
PY"
