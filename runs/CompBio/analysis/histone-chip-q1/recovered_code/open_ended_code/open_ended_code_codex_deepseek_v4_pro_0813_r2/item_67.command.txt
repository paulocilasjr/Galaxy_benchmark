/bin/bash -lc "wc -l macs3_broad/histone_broad_peaks.broadPeak macs3_broad/histone_broad_peaks.xls; head -5 macs3_broad/histone_broad_peaks.broadPeak; python3 - <<'PY'
import statistics
fn='macs3_broad/histone_broad_peaks.broadPeak'
widths=[]; scores=[]; sig=[]
with open(fn) as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        w=int(p[2])-int(p[1]); widths.append(w); scores.append(float(p[8])); sig.append(float(p[6]))
print('peaks',len(widths))
print('width min/q1/median/q3/max',min(widths),statistics.quantiles(widths,n=4),max(widths))
print('mean width',statistics.mean(widths))
print('top by width')
for i in sorted(range(len(widths)), key=lambda i:-widths[i])[:5]:
    print(i,widths[i])
# fraction genome covered
cov=sum(widths)
print('total covered',cov, 'fraction',cov/(2.9e9))
PY"
