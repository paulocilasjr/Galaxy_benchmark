/bin/bash -lc 'set -e
for sample in signal control; do
  tag=histone.chip.q1.${sample}.lexsort.tagalign.gz
  for feat in ccre_PLS ccre_pELS ccre_dELS ccre_CTCF_only ccre_DNase_H3K4me3; do
    echo "START $sample $feat $(date +%T)"
    bedtools map -sorted -g refs/lex.genome -a refs/${feat}.bed -b ${tag} -c 1 -o count > counts/${sample}_${feat}.txt
    echo "DONE $sample $feat $(date +%T) lines $(wc -l < counts/${sample}_${feat}.txt)"
  done
done'
