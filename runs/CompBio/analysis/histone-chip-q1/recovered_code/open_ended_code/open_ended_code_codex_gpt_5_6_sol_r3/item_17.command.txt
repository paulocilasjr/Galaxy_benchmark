/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
D={}
with gzip.open('bins1k.tsv.gz','rt') as f:
 next(f)
 for line in f:
  ch,a,b,s,c=line.split('\\t'); D.setdefault(ch,[[],[]]); D[ch][0].append(int(s));D[ch][1].append(int(c))
scale=sum(sum(x[1]) for x in D.values())/sum(sum(x[0]) for x in D.values())
for ch in D:
 D[ch]=[np.array(D[ch][0]),np.array(D[ch][1])]
def region(ch,a,b):
 s,c=D[ch]; x=s[a//1000:(b+999)//1000].sum();y=c[a//1000:(b+999)//1000].sum();return int(x),int(y),round((x*scale+1)/(y+1),3)
regions=[('HOXA','chr7',27000000,27300000),('HOXB','chr17',46500000,46900000),('HOXC','chr12',54100000,54500000),('HOXD','chr2',175800000,176200000),('chr19 ZNF1','chr19',20000000,25000000),('chr19 ZNF2','chr19',35000000,39000000),('chr19 ZNF3','chr19',50000000,58600000),('chrY','chrY',0,57227415),('chrX','chrX',0,156040895)]
print('scale',scale)
for n,ch,a,b in regions: print(n,ch,a,b,region(ch,a,b))
print('top 100kb windows')
top=[]
for ch,(s,c) in D.items():
 n=len(s)//100
 ss=s[:n*100].reshape(n,100).sum(1); cc=c[:n*100].reshape(n,100).sum(1)
 rat=(ss*scale+10)/(cc+10)
 for i in np.argpartition(rat,-20)[-20:]:
  if cc[i]>=100: top.append((rat[i],ch,int(i*100000),int(ss[i]),int(cc[i])))
for row in sorted(top,reverse=True)[:60]: print(*(round(row[0],2),)+row[1:])
PY"
