/bin/bash -lc "curl -L --fail --silent --show-error 'https://www.encodeproject.org/search/?type=File&format=json&limit=all&assembly=GRCh38&file_format=bed' -o encode_bed_files.json && ls -lh encode_bed_files.json && python3 - <<'PY'
import json
p=json.load(open('encode_bed_files.json'))
print(len(p.get('@graph',[])))
print({k:p.get(k) for k in ['@id','total']})
for x in p.get('@graph',[])[:2]: print({k:x.get(k) for k in ['accession','filename','file_size','md5sum','output_type','file_format','assembly','dataset']})
PY"
