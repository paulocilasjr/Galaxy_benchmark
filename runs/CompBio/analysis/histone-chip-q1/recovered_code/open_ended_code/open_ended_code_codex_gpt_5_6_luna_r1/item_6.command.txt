/bin/bash -lc "python3 -u - <<'PY'
import gzip, collections, statistics, sys
fn='histone.chip.q1.signal.tagalign.gz'
chrom=collections.Counter(); lengths=collections.Counter(); n=0
with gzip.open(fn,'rt') as f:
 for line in f:
  p=line.rstrip().split('\\t'); n+=1; chrom[p[0]]+=1; lengths[int(p[2])-int(p[1])]+=1
print('N',n, 'chrom',chrom.most_common(30), 'lens', lengths.most_common(), flush=True)
PY"
