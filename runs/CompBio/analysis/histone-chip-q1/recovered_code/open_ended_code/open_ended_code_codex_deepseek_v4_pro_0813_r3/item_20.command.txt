/bin/bash -lc "python3 - <<'PY'
# Generate interval BEDs from coding_canonical_genes.tsv
# Gene bodies, promoters +/-1kb around TSS, TSS +/-500, exons need parse knownGene maybe later.
rows=[]
with open('coding_canonical_genes.tsv') as f:
    next(f)
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        chrom,strand,start,end,tss,gene_len,symbol=p[1],p[2],int(p[3]),int(p[4]),int(p[5]),int(p[6]),p[7]
        if gene_len<100: continue
        rows.append((chrom,start,end,symbol,strand,tss))
# gene bodies sorted
with open('gene_bodies.bed','w') as out:
    for r in sorted(rows,key=lambda x:(x[0],x[1])):
        out.write('\\t'.join(map(str,[r[0],r[1],r[2],r[3],'.',r[4]]))+'\\n')
# promoters +/-2000 around TSS
with open('promoters_2kb.bed','w') as out:
    intervals=[]
    for r in rows:
        s=max(0,r[5]-2000); e=r[5]+2000
        intervals.append((r[0],s,e,r[3],'.',r[4]))
    for r in sorted(intervals,key=lambda x:(x[0],x[1])):
        out.write('\\t'.join(map(str,r))+'\\n')
# TSS +/-500
with open('tss_500.bed','w') as out:
    intervals=[]
    for r in rows:
        s=max(0,r[5]-500); e=r[5]+500
        intervals.append((r[0],s,e,r[3],'.',r[4]))
    for r in sorted(intervals,key=lambda x:(x[0],x[1])):
        out.write('\\t'.join(map(str,r))+'\\n')
print('wrote', len(rows))
PY
wc -l gene_bodies.bed promoters_2kb.bed tss_500.bed
bedtools coverage -sorted -a gene_bodies.bed -b histone.chip.q1.signal.tagalign.gz | head -2"
