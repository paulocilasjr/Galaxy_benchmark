/bin/bash -lc "python3 - <<'PY'
ls=0; sc=0
with open('cpg_signal_counts.bed') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t'); ls+=int(p[2])-int(p[1]); sc+=int(p[-1])
lc=0; cc=0
with open('cpg_control_counts.bed') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t'); lc+=int(p[2])-int(p[1]); cc+=int(p[-1])
ts=44795356; tc=27124823
print('cpg length',ls/1e6,'signal_count',sc,'control_count',cc)
print('signal density',sc/(ts/1e6)/(ls/1e3),'control',cc/(tc/1e6)/(lc/1e3),'fold', (sc/(ls/1e3))/(cc/(lc/1e3))*(tc/ts))
PY"
