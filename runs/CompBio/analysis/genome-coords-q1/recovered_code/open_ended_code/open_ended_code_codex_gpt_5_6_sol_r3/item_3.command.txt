/bin/bash -lc "python - <<'PY'
import csv, math, collections
p='single_cell_dynamics_question.csv'
cells=collections.defaultdict(list)
with open(p,newline='') as f:
    for r in csv.DictReader(f):
        d=math.sqrt(sum((float(r[a])-float(r[b]))**2 for a,b in [('enh_x','prom_x'),('enh_y','prom_y'),('enh_z','prom_z')]))
        cells[int(r['cell_id'])].append((int(r['time']),d,int(r['transcription']),int(d<=260)))
print(len(cells), {len(v) for v in cells.values()})
# contemporaneous contingency
ct=collections.Counter()
for v in cells.values():
  for _,d,t,c in v: ct[c,t]+=1
print('ct',ct)
for c in [0,1]: print('P(T|C)',c,ct[c,1]/(ct[c,0]+ct[c,1]))
for t in [0,1]: print('P(C|T)',t,ct[1,t]/(ct[0,t]+ct[1,t]))
# lag conditional P T(t)|C(t-lag); positive contact earlier. also P C(t)|T(t-lag)
for lag in range(-20,21):
 nct=[[0,0],[0,0]]
 for v in cells.values():
  v=sorted(v)
  for i in range(max(0,lag),min(len(v),len(v)+lag)):
   # t index i, predictor at i-lag
   predc=v[i-lag][3]; outt=v[i][2]
   nct[predc][outt]+=1
 p0=nct[0][1]/sum(nct[0]); p1=nct[1][1]/sum(nct[1])
 if lag%2==0 or lag in [-1,1]: print('lagCtoT',lag,round(p1-p0,4),round(p0,4),round(p1,4))
print('---')
for lag in range(-20,21):
 nct=[[0,0],[0,0]]
 for v in cells.values():
  v=sorted(v)
  for i in range(max(0,lag),min(len(v),len(v)+lag)):
   predt=v[i-lag][2]; outc=v[i][3]
   nct[predt][outc]+=1
 p0=nct[0][1]/sum(nct[0]); p1=nct[1][1]/sum(nct[1])
 if lag%2==0 or lag in [-1,1]: print('lagTtoC',lag,round(p1-p0,4),round(p0,4),round(p1,4))
PY"
