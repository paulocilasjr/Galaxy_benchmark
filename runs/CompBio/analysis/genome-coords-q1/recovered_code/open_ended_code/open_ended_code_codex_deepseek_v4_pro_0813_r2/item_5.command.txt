import csv
ids=[]
with open('single_cell_dynamics_question.csv') as f:
 r=csv.DictReader(f)
 for row in r: ids.append(int(row['cell_id']))
from collections import Counter
print(len(ids), min(ids), max(ids), len(set(ids)))
c=Counter(ids)
print(c.most_common()[-10:])
print('minmax counts', min(c.values()), max(c.values()), sum(c.values()))
