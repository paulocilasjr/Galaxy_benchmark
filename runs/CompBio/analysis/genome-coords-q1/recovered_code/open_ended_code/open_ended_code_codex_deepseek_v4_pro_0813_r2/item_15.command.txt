import csv, math
import numpy as np
from scipy.stats import wilcoxon, ttest_1samp, binomtest
T=250; cells=[]
with open('single_cell_dynamics_question.csv') as f:
 r=csv.DictReader(f); cur=[]; last=None
 for row in r:
  cid=int(row['cell_id'])
  if last is not None and cid"'!=last: cells.append(cur); cur=[]

  ex=float(row['"'enh_x']);ey=float(row['enh_y']);ez=float(row['enh_z']);px=float(row['prom_x']);py=float(row['prom_y']);pz=float(row['prom_z'])

  if prevC: m1[curT]+=1
  else: m0[curT]+=1

  dT.append(m1[1]/sum(m1)-m0[1]/sum(m0))

PY'
