from collections import Counter
T=250
cells=[]
with open('single_cell_dynamics_question.csv') as f:
    r=csv.DictReader(f)
    current=[]; last=None
    for row in r:
        cid=int(row['cell_id'])
        if last is not None and cid"'!=last:
            cells.append(current); current=[]
        last=cid
        ex=float(row['"'enh_x']); ey=float(row['enh_y']); ez=float(row['enh_z'])
        px=float(row['prom_x']); py=float(row['prom_y']); pz=float(row['prom_z'])
        d=math.sqrt((ex-px)**2+(ey-py)**2+(ez-pz)**2)
        current.append((d, d<=260.0, int(row['transcription'])))
    if current: cells.append(current)
print('ncells',len(cells), 'lens', set(len(c) for c in cells))
# all lags conditional T_t given C_{t-l}, using all cells
base=sum(t for c in cells for _,_,t in c)/(len(cells)*T)
print('base T', base)
for l in range(-10,11):
    n=0; nc=0
    for c in cells:
        for t in range(T):
            j=t-l
            if 0<=j<T and c[j][1]:
                nc+=1; n+=c[t][2]
    rate=n/nc if nc else 0
    print(f'{\"pastC\" if l>0 else (\"futureC\" if l<0 else \"same\")} lag {l:+3d}: P(T_t|C_{t-l})={rate:.5f}, nC={nc}, diff={rate-base:+.5f}')

# event centered: contact onset defined C[t]=1 and C[t-1]=0 (for t>=1)

events=0

    for t in range(1,T):
        if c[t][1] and not c[t-1][1]:
            events+=1
            for k in range(-win,win+1):
                tt=t+k
                if 0<=tt<T:
                    sums[k+win]+=c[tt][2]; counts[k+win]+=1
print('contact onset events', events)

for c in cells:
    for t in range(1,T):
        if c[t][2] and not c[t-1][2]:
            events+=1
            for k in range(-win,win+1):
                tt=t+k

                    sums[k+win]+=c[tt][1]; counts[k+win]+=1
print('trans onset events', events)
for k in range(-win,win+1):
