/bin/bash -lc "python - <<'PY'
import csv,math
from collections import defaultdict,Counter
A=defaultdict(list)
with open('single_cell_dynamics_question.csv') as f:
 r=csv.DictReader(f)
 for x in r:
  ex=[float(x[k]) for k in ('enh_x','enh_y','enh_z')]; px=[float(x[k]) for k in ('prom_x','prom_y','prom_z')]
  d=math.sqrt(sum((a-b)**2 for a,b in zip(ex,px)))
  A[int(x['cell_id'])].append((int(x['time']),int(d<=260),int(x['transcription']),d))
for cell in A: A[cell].sort()
for lag in range(1,11):
 cc=[]; cy=[]; yy=[]
 for vals in A.values():
  c=[x[1] for x in vals]; y=[x[2] for x in vals]
  cc += [c[i-lag]==1 and c[i]==1 for i in range(lag,len(c))]
  cy += [y[i-lag]==1 and y[i]==1 for i in range(lag,len(y))]
  yy += [y[i-lag]==1 and c[i]==1 for i in range(lag,len(y))]
 print(lag,'C persistence',sum(cc)/len(cc),'Y persistence',sum(cy)/len(cy),'Y->C joint',sum(yy)/len(yy))
# transition tables lag1
for name, idx in [('C',1),('Y',2)]:
 n=Counter()
 for vals in A.values():
  v=[x[idx] for x in vals]
  for a,b in zip(v,v[1:]): n[a,b]+=1
 print(name, n, {a:{b:n[a,b]/sum(n[a,z] for z in [0,1]) for b in [0,1]} for a in [0,1]})
# contact intervals and transcription intervals distributions
for idx,name in [(1,'C'),(2,'Y')]:
 runs=[]
 for vals in A.values():
  v=[x[idx] for x in vals]; cur=0
  for x in v+[0]:
   if x: cur+=1
   elif cur: runs.append(cur);cur=0
 print(name,'runs',len(runs),'mean',sum(runs)/len(runs),'max',max(runs),'counts',Counter(runs).most_common(10))
PY"
