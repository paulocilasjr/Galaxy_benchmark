/bin/bash -lc "python - <<'PY'
import csv,math,statistics
from collections import defaultdict
A=defaultdict(list)
with open('single_cell_dynamics_question.csv') as f:
 for x in csv.DictReader(f):
  e=[float(x[k]) for k in ('enh_x','enh_y','enh_z')]; p=[float(x[k]) for k in ('prom_x','prom_y','prom_z')]
  d=math.sqrt(sum((a-b)**2 for a,b in zip(e,p)))
  A[int(x['cell_id'])].append((int(x['time']),int(d<=260),int(x['transcription'])))
for v in A.values(): v.sort()
# For each cell, compare lag1 effects and contemporaneous effects; aggregate sum centered covariance
for lag in [0,1,2,3,-1]:
 vals=[]; nums=[]
 for cell,v in A.items():
  if lag>=0:
   x=[z[1] for z in v[:len(v)-lag]]; y=[z[2] for z in v[lag:]]
  else:
   x=[z[1] for z in v[-lag:]]; y=[z[2] for z in v[:len(v)+lag]]
  mx=sum(x)/len(x); my=sum(y)/len(y)
  cov=sum((a-mx)*(b-my) for a,b in zip(x,y))/len(x)
  vals.append(cov)
  nums.append(sum(a*b for a,b in zip(x,y)))
 print('lag',lag,'mean per-cell covariance',sum(vals)/len(vals),'median',statistics.median(vals),'n positive',sum(z>0 for z in vals),'sum covariance',sum(vals))
# fixed cell overall lag table, each cell equal weighted differences
for lag in [0,1,2,3]:
 diffs=[]
 for cell,v in A.items():
  x=[z[1] for z in v[:len(v)-lag]]; y=[z[2] for z in v[lag:]]
  a=[b for aa,b in zip(x,y) if aa==1]; b=[bb for aa,bb in zip(x,y) if aa==0]
  diffs.append(sum(a)/len(a)-sum(b)/len(b))
 print('lag',lag,'mean within P(Y|C1)-P(Y|C0)',sum(diffs)/len(diffs),'median',statistics.median(diffs),'positive cells',sum(x>0 for x in diffs))
# cell means correlations and ranges
cms=[]; yms=[]
for cell,v in A.items():
 cms.append(sum(z[1] for z in v)/len(v)); yms.append(sum(z[2] for z in v)/len(v))
mx=sum(cms)/len(cms); my=sum(yms)/len(yms)
print('cell p means range',min(cms),max(cms),min(yms),max(yms),'cov',sum((a-mx)*(b-my) for a,b in zip(cms,yms))/len(cms))
PY"
