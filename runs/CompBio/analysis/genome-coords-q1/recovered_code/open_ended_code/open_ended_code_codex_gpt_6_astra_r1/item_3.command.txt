import numpy as np

 a,b=(c[:,:-lag],t[:,lag:]) if lag>0 else (c[:,-lag:],t[:,:lag]) if lag<0 else (c,t)
 print('lag',lag,'corr',np.corrcoef(a.ravel(),b.ravel())[0,1], 'pT_contact',b[a==1].mean(),'pT_no_contact',b[a==0].mean())

for a in [0,1]:
