/bin/bash -lc "python -S - <<'PY'
import csv,math
from collections import defaultdict,Counter
cells=defaultdict(list)
with open('single_cell_dynamics_question.csv') as f:
 for r in csv.DictReader(f):
  d=math.sqrt(sum((float(r['enh_'+a])-float(r['prom_'+a]))**2 for a in 'xyz'))
  cells[int(r['cell_id'])].append((int(r['time']),int(d<=260),int(r['transcription'])))
for start in range(0,600,100):
 print('cells',start,start+99)
 for lag in [-1,0,1]:
  tab=Counter()
  for k in range(start,start+100):
   v=sorted(cells[k])
   for i in range(max(0,-lag),min(len(v),len(v)-lag)):
    tab[v[i][1],v[i+lag][2]]+=1
  print(lag,round(tab[1,1]/(tab[1,0]+tab[1,1]),4),round(tab[0,1]/(tab[0,0]+tab[0,1]),4))
PY"
