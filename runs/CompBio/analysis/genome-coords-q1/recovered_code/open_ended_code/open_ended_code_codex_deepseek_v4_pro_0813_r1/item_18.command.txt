/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
import statsmodels.api as sm
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1)).reshape(600,250)
C=(D<=260).astype(int); T=df['transcription'].values.reshape(600,250)
# t=3..249, use lags 1,2,3 of both
L=3
Xcols=[]; names=[]
for lag in range(1,L+1):
    Xcols.append(T[:,L-lag:250-lag].ravel()); names.append('T_lag%d'%lag)
    Xcols.append(C[:,L-lag:250-lag].ravel()); names.append('C_lag%d'%lag)
X=np.c_[Xcols].T
yT=T[:,L:].ravel(); yC=C[:,L:].ravel()
for yname,y in [('T_t',yT),('C_t',yC)]:
    Xs=sm.add_constant(X)
    res=sm.Logit(y,Xs).fit(disp=0)
    print(yname)
    print(res.summary2().tables[1])
    print('odds',np.exp(res.params))
PY"
