/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1)).reshape(600,250)
C=(D<=260).astype(int); T=df['transcription'].values.reshape(600,250)
Tprev=T[:,:-2]; Tcur=T[:,1:-1]; Tnext=T[:,2:]
Cprev=C[:,:-2]; Ccur=C[:,1:-1]; Cnext=C[:,2:]
trans_onset=(Tcur==1)&(Tprev==0)
contact_onset=(Ccur==1)&(Cprev==0)
print('trans onset n',trans_onset.sum(),'contact onset n',contact_onset.sum())
for name, onset, Aprev,Acur,Anext in [('trans',trans_onset,Cprev,Ccur,Cnext),('contact',contact_onset,Tprev,Tcur,Tnext)]:
    print('\\n',name,'onset n',onset.sum())
    print('prev state prob',Aprev[onset].mean(),'cur',Acur[onset].mean(),'next',Anext[onset].mean())
print('\\nbaseline prev cur next')
print(Cprev.mean(),Ccur.mean(),Cnext.mean(),Tprev.mean(),Tcur.mean(),Tnext.mean())
# Around trans onset: contact probabilities at t+lag, with aligned interior subset
# onset defined for t in 1..248; for lag offset, t range [1,248] shifted to valid
print('\\nTrans onset contact lag')
for lag in range(-5,6):
    # t indices original = 1..248 => array onset index = original-1? onset array length 248 for orig t=1..248.
    # C at orig t+lag has array index orig+lag. valid 0..249.
    # Need intersection orig 1..248 and 0<=orig+lag<=249 -> orig low=max(1,-lag), high=min(248,249-lag)
    lo=max(1,-lag); hi=min(248,249-lag)
    onset_sub=trans_onset[lo-1:hi]  # orig t from lo..hi inclusive -> indices lo-1 .. hi-1
    C_sub=C[lo+lag:hi+lag+1]
    n=onset_sub.sum()
    print(lag, C_sub[onset_sub].mean() if n else np.nan, 'n',n)
print('\\nContact onset trans lag')
for lag in range(-5,6):
    lo=max(1,-lag); hi=min(248,249-lag)
    onset_sub=contact_onset[lo-1:hi]
    T_sub=T[lo+lag:hi+lag+1]
    n=onset_sub.sum()
    print(lag, T_sub[onset_sub].mean() if n else np.nan, 'n',n)
PY"
