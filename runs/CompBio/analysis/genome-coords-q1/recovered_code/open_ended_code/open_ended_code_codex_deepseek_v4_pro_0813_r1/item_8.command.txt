/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1)).reshape(600,250)
C=(D<=260).astype(int); T=df['transcription'].values.reshape(600,250)
print('Autocorr transition T_t -> T_t+lag for lag 1..6')
for lag in range(1,7):
    prev=T[:,:250-lag]; cur=T[:,lag:]
    for v in [0,1]:
        mask=prev==v
        print(lag,'prev',v,'P',cur[mask].mean(),'n',mask.sum())
    print()
print('C autocorr')
for lag in range(1,7):
    prev=C[:,:250-lag]; cur=C[:,lag:]
    for v in [0,1]:
        mask=prev==v
        print(lag,'prev',v,'P',cur[mask].mean(),'n',mask.sum())
    print()
# T_t -> C_t+lag
print('T_t -> C_t+lag')
for lag in range(1,7):
    prev=T[:,:250-lag]; cur=C[:,lag:]
    for v in [0,1]:
        mask=prev==v
        print(lag,'T prev',v,'C P',cur[mask].mean(),'n',mask.sum())
    print()
# C_t -> T_t+lag
print('C_t -> T_t+lag')
for lag in range(1,7):
    prev=C[:,:250-lag]; cur=T[:,lag:]
    for v in [0,1]:
        mask=prev==v
        print(lag,'C prev',v,'T P',cur[mask].mean(),'n',mask.sum())
    print()
PY"
