/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1)).reshape(600,250)
C=(D<=260).astype(int); T=df['transcription'].values.reshape(600,250)
# onsets among t=1..248 to have both prev/next
Tprev=T[:,1:-1] # t index? use arrays t from1 to248
Tcur=T[:,1:-1]; Tnext=T[:,2:]
Cprev=C[:,1:-1]; Ccur=C[:,1:-1]; Cnext=C[:,2:]
# Transcription onset at t: Tcur=1, Tprev=0
trans_onset=(Tcur==1)&(Tprev==0)
# Contact onset at t: Ccur=1, Cprev=0
contact_onset=(Ccur==1)&(Cprev==0)
# For transcription onset, contact before (t-1) and after (t+1), and same
for name, onset in [('trans_onset',trans_onset),('contact_onset',contact_onset)]:
    print('\\n',name,'n',onset.sum())
    print('Cprev P',Cprev[onset].mean(),'Ccur P',Ccur[onset].mean(),'Cnext P',Cnext[onset].mean())
    print('Tprev P',Tprev[onset].mean(),'Tcur P',Tcur[onset].mean(),'Tnext P',Tnext[onset].mean())
# baseline contact prob (all t 1..248)
print('\\nbaselines: Cprev, Ccur, Cnext, Tprev,Tcur,Tnext')
print(Cprev.mean(),Ccur.mean(),Cnext.mean(),Tprev.mean(),Tcur.mean(),Tnext.mean())
# For contact onset, look at trans around; for trans onset, look contact around
# also probabilities at lags -5..5 around onsets
print('\\nTrans onset contact around')
for lag in range(-5,6):
    if lag<0:
        l=-lag
        # t range start l to 248+l? onset t in 1..248; contact at t+lag (before if negative) within 0..249
        C_al = C[:, 1+l : 249+l]
        onset_al = trans_onset[:, l: trans_onset.shape[1]+l] if l>0 else trans_onset
    else:
        C_al=C[:, 1+lag : 249+lag]
        onset_al=trans_onset[:, lag: trans_onset.shape[1]+lag] if lag>0 else trans_onset
    print(lag, C_al[onset_al].mean())
print('\\nContact onset trans around')
for lag in range(-5,6):
    if lag<0:
        l=-lag
        T_al = T[:, 1+l : 249+l]
        onset_al = contact_onset[:, l: contact_onset.shape[1]+l] if l>0 else contact_onset
    else:
        T_al=T[:, 1+lag : 249+lag]
        onset_al=contact_onset[:, lag: contact_onset.shape[1]+lag] if lag>0 else contact_onset
    print(lag, T_al[onset_al].mean())
PY"
