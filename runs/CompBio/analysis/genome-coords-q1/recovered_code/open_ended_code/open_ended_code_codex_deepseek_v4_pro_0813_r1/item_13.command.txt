/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
E=df[['enh_x','enh_y','enh_z']].values.reshape(600,250,3)
P=df[['prom_x','prom_y','prom_z']].values.reshape(600,250,3)
D=np.sqrt(((E-P)**2).sum(2))
T=df['transcription'].values.reshape(600,250)
C=(D<=260).astype(int)
dD=D[:,1:]-D[:,:-1]
for c in [0,1]:
  for t in [0,1]:
    mask=(C[:,:-1]==c)&(T[:,:-1]==t)
    print('prev C,T=',c,t,'n',mask.sum(),'P(T next)',T[:,1:][mask].mean(),'P(C next)',C[:,1:][mask].mean(),'mean dD',dD[mask].mean())
# regression for movement direction maybe T_t vs C_t
print('\\nD_t+1 - D_t correlation')
print('baseline dD',dD.mean())
print('regress dD on T_t, C_t, D_t?')
from numpy.linalg import lstsq
X=np.c_[np.ones(dD.size), T[:,:-1].ravel(), C[:,:-1].ravel(), D[:,:-1].ravel()]
y=dD.ravel()
coef,res,rank,s=lstsq(X,y,rcond=None)
print(coef)
PY"
