/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D2=((df[['enh_x','enh_y','enh_z']].values - df[['prom_x','prom_y','prom_z']].values)**2).sum(axis=1)
df['dist']=np.sqrt(D2); df['contact']=(df.dist<=260).astype(int)
D=np.stack([s.values for _,s in df.groupby('cell_id')['dist']])
C=np.stack([s.values for _,s in df.groupby('cell_id')['contact']])
T=np.stack([s.values for _,s in df.groupby('cell_id')['transcription']])
for k in range(1,6):
 a=D[:,:-k].ravel(); b=D[:,k:].ravel()
 print('D autocorr lag',k,np.corrcoef(a,b)[0,1], 'mean diff', b[a<260].mean()-b[a>=260].mean())
print('P(C_t | D_{t-1}<260):', C[:,1:][D[:,:-1]<260].mean(), 'vs Dprev>=260', C[:,1:][D[:,:-1]>=260].mean())
# Actually threshold contact prev vs current
print('P(C_t|C_prev)',C[:,1:][C[:,:-1]==1].mean(), C[:,1:][C[:,:-1]==0].mean())
# distance at t conditional Cprev and Tprev
for Tprev in [0,1]:
 for Cprev in [0,1]:
  mask=(T[:,:-1]==Tprev)&(C[:,:-1]==Cprev)
  print('mean D_t',Tprev,Cprev,D[:,1:][mask].mean(),'n',mask.sum(),'P C_t',C[:,1:][mask].mean())
PY"
