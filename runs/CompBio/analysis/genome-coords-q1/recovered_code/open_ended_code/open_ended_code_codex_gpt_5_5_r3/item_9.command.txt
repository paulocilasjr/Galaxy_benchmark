/bin/bash -lc "python3 - <<'PY'
import csv, math
from collections import defaultdict, Counter
cells=defaultdict(list)
with open('single_cell_dynamics_question.csv', newline='') as f:
    for row in csv.DictReader(f):
        dx=float(row['enh_x'])-float(row['prom_x'])
        dy=float(row['enh_y'])-float(row['prom_y'])
        dz=float(row['enh_z'])-float(row['prom_z'])
        c=math.sqrt(dx*dx+dy*dy+dz*dz) <= 260.0
        cells[int(row['cell_id'])].append((int(row['time']), int(c), int(row['transcription'])))
for cid in cells: cells[cid].sort()
# P(tx next | current tx, current contact)
print('P(tx[t+1]=1 | tx[t], contact[t])')
ctr=Counter(); den=Counter()
for arr in cells.values():
    for i in range(len(arr)-1):
        key=(arr[i][2], arr[i][1]) # tx_t, contact_t
        den[key]+=1; ctr[key]+=arr[i+1][2]
for key in sorted(den): print(key, ctr[key], den[key], ctr[key]/den[key])
print('\\nP(contact[t+1]=1 | contact[t], tx[t])')
ctr=Counter(); den=Counter()
for arr in cells.values():
    for i in range(len(arr)-1):
        key=(arr[i][1], arr[i][2]) # c_t, tx_t
        den[key]+=1; ctr[key]+=arr[i+1][1]
for key in sorted(den): print(key, ctr[key], den[key], ctr[key]/den[key])

# More controls using previous state too: P(tx_{t+1}|tx_t,c_t,tx_{t-1},c_{t-1}); aggregate contact effect matched by tx states
print('\\nOnsets: P(tx[t+1]=1 | tx[t]=0, contact[t])')
# Already key tx=0.
print('\\nContact onset: P(contact[t+1]=1 | contact[t]=0, tx[t])')

# transitions (onsets only)
for label in ['tx_onset','contact_onset']:
    pass
# Contact effect on transcription onset conditioned on tx_t=0 and maybe contact_t
# Additional: P(tx_{t+1}=1 | tx_t=0, contact_t=1/0, contact_{t-1})
print('\\nP(tx[t+1]=1 | tx[t]=0, contact[t], contact[t-1])')
ctr=Counter(); den=Counter()
for arr in cells.values():
    for i in range(1,len(arr)-1):
        if arr[i][2]==0:
            key=(arr[i][1], arr[i-1][1])
            den[key]+=1; ctr[key]+=arr[i+1][2]
for key in sorted(den): print(key, ctr[key], den[key], ctr[key]/den[key])
print('\\nP(contact[t+1]=1 | contact[t]=0, tx[t], tx[t-1])')
ctr=Counter(); den=Counter()
for arr in cells.values():
    for i in range(1,len(arr)-1):
        if arr[i][1]==0:
            key=(arr[i][2], arr[i-1][2])
            den[key]+=1; ctr[key]+=arr[i+1][1]
for key in sorted(den): print(key, ctr[key], den[key], ctr[key]/den[key])

# enumerate event sequences pairs at lag +/-1 counts contact/tx transitions
# Cross-tab C_t vs T_t+1 controlling T_t? and T_t vs C_t+1 controlling C_t?
PY"
