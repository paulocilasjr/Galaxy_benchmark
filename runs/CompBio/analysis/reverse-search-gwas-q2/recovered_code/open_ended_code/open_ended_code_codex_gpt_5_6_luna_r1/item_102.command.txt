/bin/bash -lc "python3 - <<'PY'
import urllib.request
urls=['https://www.biorxiv.org/content/10.64898/2026.04.06.716850v1.full.pdf','https://www.biorxiv.org/content/biorxiv/early/2026/04/09/2026.04.06.716850.full.pdf','https://www.biorxiv.org/content/10.64898/2026.04.06.716850v1.pdf']
for u in urls:
 try:
  req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
  x=urllib.request.urlopen(req,timeout=60); b=x.read(100)
  print(x.status,len(b),x.geturl(),b[:10])
 except Exception as e: print('ERR',u,type(e).__name__,e)
PY"
