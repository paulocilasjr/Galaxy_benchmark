/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for u in ['https://www.ebi.ac.uk/gwas/rest/api/efoTraits/OBA_VT0001253/studies?size=1000','https://www.ebi.ac.uk/gwas/rest/api/efoTraits/OBA_VT0001253/associations?size=1']:
 print('\\n',u)
 try:
  d=json.load(urllib.request.urlopen(u,timeout=60));
  print('studies',len(d.get('_embedded',{}).get('studies',[])),d.get('page'))
  for s in d.get('_embedded',{}).get('studies',[]):
   p=s.get('publicationInfo',{})
   print(s.get('accessionId'),'|',s.get('initialSampleSize'),'|',s.get('replicationSampleSize'),'|',p.get('pubmedId'),'|',p.get('title'))
  if 'associations' in d.get('_embedded',{}): print('assoc',d['_embedded']['associations'][0])
 except Exception as e: print(type(e).__name__,e)
PY"
