/bin/bash -lc "python3 -u - <<'PY'
import urllib.request,json
for rs in ['rs1150779','rs6909732','rs2483211','rs7742369','rs4308051','rs8094261','rs7235010','rs4800455','rs7244739','rs2871960','rs6918981','rs224333']:
 print('begin',rs,flush=True)
 try:
  req=urllib.request.Request('https://rest.ensembl.org/variation/human/'+rs,headers={'Content-Type':'application/json','Accept':'application/json','User-Agent':'x'})
  d=json.load(urllib.request.urlopen(req,timeout=30))
  pops=[]
  for p in d.get('populations',[]):
   if p.get('population','').lower().find('1000genomes')>=0 or p.get('population','').lower().find('gnomad')>=0 or p.get('population','').lower().find('korean')>=0:
    pops.append((p.get('population'),p.get('allele'),p.get('frequency')))
  print(pops[:50],flush=True)
 except Exception as e: print('ERR',repr(e),flush=True)
PY"
