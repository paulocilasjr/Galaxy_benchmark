/bin/bash -lc "for rs in rs1150779 rs4308051 rs2871960 rs224333 rs2011603 rs3791679 rs3809128; do curl -s -H 'Content-Type: application/json' \"https://rest.ensembl.org/variation/human/"'$rs?pops=1" -o "$rs.json"; done
python - <<'"'PY'
import json
for rs in ['rs1150779','rs4308051','rs2871960','rs224333','rs2011603','rs3791679','rs3809128']:
 d=json.load(open(rs+'.json'))
 pops={p['population'].split(':')[-1]:p for p in d.get('populations',[]) if p['population'].split(':')[-1] in ['AFR','AMR','EAS','EUR','SAS']}
 print(rs,[(k,p.get('allele'),p.get('frequency')) for k,p in pops.items()])
PY"
