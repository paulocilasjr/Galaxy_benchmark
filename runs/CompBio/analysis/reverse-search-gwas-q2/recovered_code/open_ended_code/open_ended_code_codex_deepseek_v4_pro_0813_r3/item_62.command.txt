/bin/bash -lc "python3 - <<'PY'
import json, collections
j=json.load(open('gwasinfo.json')); dc=j['datasets_compressed']
counts=collections.Counter(); examples={}
for id,v in dc.items():
    if len(v)<2 or not isinstance(v[1],dict): continue
    main=v[0]
    if len(main)<=10: continue
    try: n=int(main[10])
    except: continue
    build=main[1]
    if build==0 and 40000<=n<=70000:
        trait=main[0]; cat=main[2]; sub=main[3]; year=main[7]; pmid=v[1].get('pmid'); nsnp=v[1].get('nsnp')
        key=(n,cat,sub,year)
        counts[key]+=1
        examples.setdefault(key,(id,trait,pmid,nsnp))
print('total',sum(counts.values()))
for k,c in counts.most_common(50):
    print(c,k,examples[k])
PY"
