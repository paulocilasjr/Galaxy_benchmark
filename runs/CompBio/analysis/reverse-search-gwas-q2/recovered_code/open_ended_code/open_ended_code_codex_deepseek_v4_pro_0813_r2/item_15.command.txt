/bin/bash -lc "curl -s 'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/rs1150779/associations?size=100' > rs1150779_assoc.json && python3 - <<'PY'
import json
j=json.load(open('rs1150779_assoc.json'))
for a in j['_embedded']['associations']:
    print(a.get('pvalue'), a.get('betaNum'), a.get('betaUnit'), a.get('riskFrequency'), a.get('standardError'))
    print(' snp type',a.get('snpType'), 'desc', a.get('pvalueDescription'))
    for trait in a.get('efoTraits',[]): print(' trait', trait.get('trait'))
    study=a['_links'].get('study',{}).get('href')
    print(' study',study)
PY"
