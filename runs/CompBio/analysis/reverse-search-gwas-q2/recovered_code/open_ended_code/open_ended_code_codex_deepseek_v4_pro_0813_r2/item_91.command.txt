/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, time
base='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByEfoTrait'
params={'efoTrait':'body height','size':500}
allstudies=[]
for page in range(20):
    p=params.copy(); p['page']=page
    url=base+'?'+urllib.parse.urlencode(p)
    try:
      data=json.load(urllib.request.urlopen(url))
    except Exception as e:
      print('err',page,e); break
    emb=data.get('_embedded',{})
    studies=emb.get('studies',[])
    if not studies: break
    allstudies += studies
    print('page',page,'got',len(studies),'total',len(allstudies))
    time.sleep(.2)
print('total',len(allstudies))
# unique
ids=set(); uniq=[]
for s in allstudies:
    if s['accessionId'] not in ids:
      ids.add(s['accessionId']); uniq.append(s)
print('unique',len(uniq))
# filter snpCount around 632553 and height/body height maybe all
for s in uniq:
    c=s.get('snpCount')
    if c and abs(int(c)-632553)<5000:
      print('MATCH',s['accessionId'],s.get('snpCount'),s.get('diseaseTrait'),s.get('publicationInfo'))
# print all with snpCount 500k-800k
for s in uniq:
    c=s.get('snpCount')
    if c and 500000 <= int(c) <= 800000:
      print('RANGE',s['accessionId'],c,s.get('diseaseTrait'),s.get('initialSampleSize'))
PY"
