/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
regions=['6:34000000-34400000','18:20500000-20950000','3:140500000-141700000','20:33700000-34300000','4:17700000-18300000','2:55800000-56400000','1:17000000-17600000','7:2500000-3100000']
for reg in regions:
 print('\\n===',reg)
 url=f'https://grch37.rest.ensembl.org/overlap/region/human/{reg}?feature=gene;content-type=application/json'
 try:
  data=json.load(urllib.request.urlopen(url))
  genes=[(x.get('external_name'),x.get('start'),x.get('end'),x.get('biotype')) for x in data if isinstance(x,dict) and x.get('biotype')=='protein_coding']
  print(sorted(genes,key=lambda x:x[1])[:50])
 except Exception as e: print(e)
PY"
