/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, time, re
base='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByEfoTrait'
allstudies=[]; seen=set()
for page in range(10):
    url=base+'?'+urllib.parse.urlencode({'efoTrait':'body height','size':200,'page':page})
    try: data=json.load(urllib.request.urlopen(url))
    except Exception as e: print(e); break
    studies=data.get('_embedded',{}).get('studies',[])
    if not studies: break
    for s in studies:
      if s['accessionId'] not in seen:
       seen.add(s['accessionId']); allstudies.append(s)
    print(page,len(studies),len(allstudies))
    time.sleep(.1)
print('unique',len(allstudies))
for s in allstudies:
    n=s.get('initialSampleSize','')
    # parse first number
    nums=re.findall(r'[\\d,]+', n)
    first=int(nums[0].replace(',','')) if nums else None
    if first and 30000 <= first <= 150000:
      print('candidate',s['accessionId'],s.get('snpCount'),n,s.get('fullPvalueSet'),s.get('publicationInfo',{}).get('pubmedId'),s.get('publicationInfo',{}).get('title'))
PY"
