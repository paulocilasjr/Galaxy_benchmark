/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, urllib.parse, sys, time
rsids=['rs1150779','rs4308051','rs2871960','rs224333','rs7742369','rs6909732','rs2483211','rs9846396','rs4800455']
for rs in rsids:
    print('\\n###',rs)
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?size=100'
    assocs=[]
    try:
        data=json.load(urllib.request.urlopen(url,timeout=20))
    except Exception as e:
        print('ERR',e); continue
    assocs=data.get('_embedded',{}).get('associations',[])
    for a in assocs[:50]:
        aid=a['_links']['self']['href'].rstrip('/').split('/')[-1]
        p=a.get('pvalue'); beta=a.get('betaNum'); unit=a.get('betaUnit'); desc=a.get('pvalueDescription')
        studyurl=a['_links']['study']['href']
        try:
            s=json.load(urllib.request.urlopen(studyurl,timeout=20))
            trait=s.get('diseaseTrait',{}).get('trait')
            info=s.get('publicationInfo') or {}
            pmid=info.get('pubmedId'); author=(info.get('author') or {}).get('fullname'); title=info.get('title')
            acc=s.get('accessionId'); n=s.get('initialSampleSize')
        except Exception as e:
            trait=pmid=author=title=acc=n=f'ERR {e}'
        print(aid, 'p',p,'beta',beta,unit,'desc',desc,'acc',acc,'pmid',pmid,'trait',trait,'n',n,'title',title[:60] if title else None)
PY"
