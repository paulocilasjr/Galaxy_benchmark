/bin/bash -lc "cat > studies.py <<'PY'
import requests,json,glob,concurrent.futures,time,os
alist=[]
for f in glob.glob('rs*.json'):
 for a in json.load(open(f)).get('_embedded',{}).get('associations',[]):alist.append((f,a))
urls=set(a['_links']['study']['href'] for f,a in alist)
def work(u):
 fn='study_'+u.split('/')[-1]+'.json'
 if os.path.exists(fn):return u,json.load(open(fn))
 for _ in range(4):
  try:
   r=requests.get(u,timeout=25);r.raise_for_status();j=r.json();json.dump(j,open(fn,'w'));return u,j
  except Exception:time.sleep(1)
 return u,{}
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as ex:d=dict(ex.map(work,urls))
for f,a in alist:
 s=d[a['_links']['study']['href']];p=s.get('publicationInfo',{})
 print(f,a.get('pvalue'),a.get('betaNum'),a.get('riskFrequency'),s.get('diseaseTrait'),s.get('accessionId'),p.get('pubmedId'),p.get('title'))
PY
python studies.py > matches.txt"
