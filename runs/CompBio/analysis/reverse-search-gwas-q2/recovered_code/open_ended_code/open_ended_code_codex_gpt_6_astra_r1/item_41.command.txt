/bin/bash -lc "python - <<'PY'
import requests,gzip,csv,json
j=requests.get('https://www.ebi.ac.uk/gwas/rest/api/v2/studies/GCST90245993',timeout=30).json();json.dump(j,open('verified_source.json','w'));print('PMID',j['pubmed_id'])
with gzip.open('reverse.search.gwas.q2.gz','rt') as f: local={r['RSID']:r for _,r in zip(range(1000),csv.DictReader(f,delimiter='\\t'))}
k=0
with open('yengo_excerpt.tsv') as f:
 for r in csv.DictReader(f,delimiter='\\t'):
  if r['variant_id'] in local:
   q=local[r['variant_id']]
   assert all(q[a]==r[b] for a,b in [('BETA','beta'),('SE','standard_error'),('P','p_value'),('EFFECT_ALLELE_FREQ','effect_allele_frequency')])
   k+=1
print('Exact matching variants',k)
assert k>=5
PY"
