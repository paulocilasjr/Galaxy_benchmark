import json

    if nsnp in [632552,632553,632551] or (sample and 40000 <= sample <= 70000):

        matches.append((id,trait,arr[6],arr[7],pop,sex,sample,nsnp,pmid,meta.get('note'),meta.get('doi')))
print('matches',len(matches))
for m in matches:
    print(m)
