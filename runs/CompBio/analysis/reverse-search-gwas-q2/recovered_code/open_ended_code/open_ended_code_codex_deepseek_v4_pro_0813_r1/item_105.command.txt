/bin/bash -lc "python3 - <<'PY'
import json, re
j=json.load(open('gwasinfo.json'))
ds=j['datasets_compressed']; cod=j['majority_fields_and_coding']
height=[]
for id,val in ds.items():
    arr=val[0]
    meta=val[1] if len(val)>1 else {}
    trait=arr[0]
    if 'height' in trait.lower() or 'height' in str(meta).lower():
        # reconstruct sample size from arr index 10 or meta
        sample=arr[10]
        nsnp=meta.get('nsnp')
        pmid=meta.get('pmid')
        pop=cod['population'][arr[4]] if arr[4] is not None else None
        sex=cod['sex'][arr[5]] if arr[5] is not None else None
        author=arr[6]; year=arr[7]
        height.append((id,trait,author,year,pop,sex,sample,nsnp,pmid,meta.get('doi'),meta.get('note')))
print('height datasets',len(height))
for h in height:
    print(h)
PY"
