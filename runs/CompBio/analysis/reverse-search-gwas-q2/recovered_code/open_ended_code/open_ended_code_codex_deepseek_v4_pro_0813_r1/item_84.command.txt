/bin/bash -lc "python3 - <<'PY'
import requests, json
base='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByDiseaseTrait?diseaseTrait=height&size=100'
items=[]; url=base
while url:
    r=requests.get(url,headers={'Accept':'application/json'},timeout=60)
    d=r.json(); items+=d.get('_embedded',{}).get('studies',[])
    url=d.get('_links',{}).get('next',{}).get('href')
print('total',len(items))
for s in items:
    snp=s.get('snpCount') or 0
    init=s.get('ancestries',[])
    initn=sum(a.get('numberOfIndividuals',0) for a in init if a.get('type')=='initial')
    if 500000 <= snp <= 800000:
        print(s.get('accessionId'), s.get('publicationInfo',{}).get('pubmedId'), initn, snp, s.get('diseaseTrait',{}).get('trait'), s.get('publicationInfo',{}).get('title'))
PY"
