/bin/bash -lc "python3 - <<'PY'
import requests, json
url='https://www.ebi.ac.uk/gwas/rest/api/studies/GCST90095033'
r=requests.get(url,headers={'Accept':'application/json'},timeout=30)
d=r.json()
print(json.dumps({k:d.get(k) for k in ['accessionId','diseaseTrait','initialSampleSize','snpCount','fullPvalueSet','publicationInfo']}, indent=2))
print('links',json.dumps(d.get('_links',{}),indent=2))
# associations summary count
for link in ['associations','associationsByStudySummary','snps']:
    if link in d['_links']:
        u=d['_links'][link]['href']
        r2=requests.get(u,headers={'Accept':'application/json'},timeout=30)
        print(link,u,r2.status_code,len(r2.text))
        if r2.ok:
            try:
                j=r2.json(); print(j.keys())
                emb=j.get('_embedded',{})
                print(list(emb.keys()), {k:len(v) for k,v in emb.items()})
                if 'associations' in emb: print(json.dumps(emb['associations'][0],indent=2)[:1000])
            except Exception as e: print('json err',e)
PY"
