/bin/bash -lc "python3 - <<'PY'
import requests
base='https://www.ncbi.nlm.nih.gov/pmc/articles/PMC9281111/bin/'
for f in ['12916_2022_2450_MOESM2_ESM.docx']:
    url=base+f
    r=requests.get(url,timeout=60)
    print(f,r.status_code,len(r.content),r.headers.get('content-type'))
    open(f,'wb').write(r.content)
PY
ls -l *.docx"
