/bin/bash -lc "python3 - <<'PY'
import subprocess, requests, json, collections, gzip
# get top 50 by p
cmd=\"gzip -cd reverse.search.gwas.q2.gz | awk -F \\\"\\\\t\\\" 'NR>1 && "'$10!='"\\\"\\\" && "'$10!='"\\\"NA\\\" {print "'$10'"\\\"\\\\t\\\""'$2}'"' | sort -g | head -50\"
out=subprocess.check_output(cmd,shell=True,text=True)
rslist=[line.split('\\t')[1] for line in out.strip().splitlines()]
print('top50:',rslist)
traits=collections.Counter(); studies=collections.Counter(); details=[]
for rs in rslist:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations'
    try:
        r=requests.get(url,headers={'Accept':'application/json'},timeout=30)
    except Exception as e:
        print('err',rs,e); continue
    if r.status_code"'!=200:
        continue
    assocs=r.json().get('"'_embedded',{}).get('associations',[])
    for a in assocs:
        p=a.get('pvalue'); beta=a.get('betaNum')
        study_url=a['_links']['study']['href']
        sr=requests.get(study_url,headers={'Accept':'application/json'},timeout=30)
        if sr.ok:
            st=sr.json()
            pub=st.get('publicationInfo',{}).get('pubmedId')
            trait=st.get('diseaseTrait',{}).get('trait')
            if trait:
                traits[(trait,pub)]+=1
            studies[(pub, st.get('publicationInfo',{}).get('title'))]+=1
    print('done',rs,'nassoc',len(assocs))
print('\\nTRAITS:')
for k,v in traits.most_common(30): print(v,k)
print('\\nSTUDIES:')
for k,v in studies.most_common(30): print(v,k)
