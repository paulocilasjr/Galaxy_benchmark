/bin/bash -lc "python3 - <<'PY'
import requests
for gcst in ['GCST90095033','GCST008053','GCST000644','GCST90225553','GCST000175','GCST002702']:
    # determine range based on number
    n=int(gcst[4:]) if gcst.startswith('GCST') else 0
    # EBI FTP uses ranges like GCST90095001-GCST90100000
    if n>=90095001:
        folder=f'GCST90095001-GCST90100000'
    elif n>=90000001:
        folder=f'GCST90000001-GCST90010000'
    elif n>=800001:
        folder=f'GCST{900001}-GCST{1000000}' # rough
    elif n>=8000:
        folder=f'GCST000001-GCST00100000' # maybe
    else:
        folder=None
    # Actually try common path from known URL patterns using ftp ebi
    for base in ['https://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/','http://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/']:
        # guess range by start
        start=(n//1000)*1000+1; end=(n//1000+1)*1000
        if gcst.startswith('GCST9'):
            # ranges of 10000? use 10000 chunks maybe
            start=((n-1)//10000)*10000+1; end=((n-1)//10000+1)*10000
        folder=f'GCST{start:09d}-GCST{end:09d}'
        url=f'{base}{folder}/{gcst}/'
        try:
            r=requests.get(url,timeout=20)
            print(gcst,url,r.status_code,r.url)
            if r.status_code==200:
                print(r.text[:1000]); break
        except Exception as e: print('err',e)
PY"
