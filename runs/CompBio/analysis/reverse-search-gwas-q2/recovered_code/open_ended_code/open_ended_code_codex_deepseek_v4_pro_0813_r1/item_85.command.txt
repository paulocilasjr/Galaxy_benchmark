/bin/bash -lc "python3 - <<'PY'
import requests, json
for term in ['body height','adult height','standing height','height adjusted','height (mean']:
    u=f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByDiseaseTrait?diseaseTrait={requests.utils.quote(term)}&size=100'
    r=requests.get(u,headers={'Accept':'application/json'},timeout=60)
    if r.ok:
        d=r.json(); emb=d.get('_embedded',{}).get('studies',[]); print(term,len(emb))
        for s in emb[:3]: print(' ',s.get('accessionId'),s.get('diseaseTrait',{}).get('trait'),s.get('snpCount'),s.get('initialSampleSize'))
    else: print(term,r.status_code)
PY"
