import requests,json
for gcst in ['GCST90225553','GCST008904','GCST000644']:

    items=[]; u=url
    while u:
        r=requests.get(u,headers={'Accept':'application/json'},timeout=60); d=r.json(); items+=d.get('_embedded',{}).get('associations',[]); u=d.get('_links',{}).get('next',{}).get('href')
    print('\\n',gcst,'n',len(items))
    for a in items[:20]:
        risk=a.get('loci',[{}])[0].get('strongestRiskAlleles',[{}])[0].get('riskAlleleName','')
        print(risk,a.get('pvalue'),a.get('betaNum'),a.get('standardError'),a.get('riskFrequency'))
PY"
