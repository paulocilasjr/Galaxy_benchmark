/bin/bash -lc "gzip -cd reverse.search.gwas.q2.gz | awk 'NR>1{print "'$10,$2}'"' | sort -g -k1,1 | head -25 | awk '{print "'$2}'"' > top25_snps.txt
xargs -P 4 -I{} sh -c 'curl -L --fail --silent \"https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{}/associations?projection=associationBySnp&size=1000\" -o \"topassoc_{}.json\" || true' < top25_snps.txt
ls topassoc_*.json 2>/dev/null | wc -l"
