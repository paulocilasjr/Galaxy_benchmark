/bin/bash -lc "awk '{print NR,"'$0}'"' assoc_study_urls.txt | head -100 | awk '{u="'$2; sub("/study$","",u); n=split(u,a,"/"); print $1,a[n]}'"' > index_associd.txt
for f in topstudy_[0-9]*.json; do i="'${f#topstudy_}; i=${i%.json}; jq -r --arg i "$i" '"'["'$i,.accessionId,.publicationInfo.pubmedId,.publicationInfo.title,.diseaseTrait.trait] | @tsv'"' \""'$f"; done > index_study_meta.tsv
join <(sort -k1,1n index_associd.txt) <(sort -k1,1n index_study_meta.tsv) > associd_meta_first100.tsv
for f in topassoc_*.json; do s=${f#topassoc_}; s=${s%.json}; jq -r --arg s "$s" '"'._embedded.associations[] | ["'$s,(._links.self.href|split("/")[-1]),.pvalue] | @tsv'"' \""'$f"; done > top_assoc_ids.tsv
awk '"'NR==FNR{m["'$1]=$2"'"\\t\""'$3"'"\\t\""'$4"'"\\t\""'$5;next} $2 in m{print $1,$2,$3,m[$2]}'"' associd_meta_first100.tsv top_assoc_ids.tsv | sort -k4,4 | head -100"
