/bin/bash -lc "awk 'NR>1{print "'$1,toupper($2),toupper($3),$5,$6}'"' randall_men.txt | sort -k1,1 > randall_men_full.txt
awk 'NR>1{print "'$1,toupper($2),toupper($3),$5,$6}'"' randall_women.txt | sort -k1,1 > randall_women_full.txt
gzip -cd reverse.search.gwas.q2.gz | awk 'NR>1{print "'$2,$5,$6,$8,$9}'"' | sort -k1,1 > q2_full.txt
for sex in men women; do join randall_"'${sex}_full.txt q2_full.txt | awk '"'"'$2==$7&&$3==$8{s=1} $2==$8&&$3==$7{s=-1} s!=0 && $4!="." {x=s*$4;y=$9;n++;sx+=x;sy+=y;sxx+=x*x;syy+=y*y;sxy+=x*y;s=0} END{print "'"'"'$sex'"'\",n,(sxy-sx*sy/n)/sqrt((sxx-sx*sx/n)*(syy-sy*sy/n))}'; done"
