/bin/bash -lc "python3 - <<'PY'
import requests,gzip,math
ids=['rs3131972','rs12562034','rs7537756','rs2340587','rs28576697','rs3791679','rs2871960','rs1150779','rs4308051','rs13273123','rs6918981','rs3750595','rs7086391']
q={}
for l in gzip.open('reverse.search.gwas.q2.gz','rt'):
 a=l.rstrip().split('\\t')
 if len(a)>6 and a[1] in ids:q[a[1]]=(a[4],float(a[6]))
d=requests.post('https://rest.ensembl.org/variation/homo_sapiens',headers={'Content-Type':'application/json','Accept':'application/json'},json={'ids':ids}).json()
pops=['afr','amr','eas','nfe','sas']
print('rs allele qfreq',' '.join(pops))
errs={p:[] for p in pops}
for rs in ids:
 ea,f=q.get(rs,('? ',float('nan')))
 vals={}
 for x in d[rs].get('populations',[]):
  if x['population'].startswith('gnomADg:') and x['population'].split(':')[-1] in pops and x['allele']==ea:
   vals[x['population'].split(':')[-1]]=x['frequency']
 print(rs,ea,round(f,3),*[round(vals.get(p,float('nan')),3) for p in pops])
 for p in pops:
  if p in vals:errs[p].append((f-vals[p])**2)
print('rmse', {p:math.sqrt(sum(v)/len(v)) for p,v in errs.items()})
PY"
