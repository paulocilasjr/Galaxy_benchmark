/bin/bash -lc "gzip -cd sample.swap.atac.q1.tsv.gz | wc -l && python - <<'PY'
import pandas as pd, numpy as np
p='sample.swap.atac.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t')
X=df.iloc[:,3:].to_numpy(dtype=np.float64)
names=list(df.columns[3:])
print('shape',X.shape)
print('libraries')
for n,v,z in zip(names,X.sum(0),np.count_nonzero(X,axis=0)): print(n,int(v),z)
# Pearson of log1p CPM across high-information bins
Y=np.log1p(X/X.sum(0)*1e6)
var=Y.var(1)
sel=var>=np.quantile(var,0.9)
C=np.corrcoef(Y[sel].T)
print('top correlations (10% variable bins)')
for i in range(len(names)):
    o=np.argsort(C[i])[::-1]
    print(names[i], ', '.join(f'{names[j]}:{C[i,j]:.3f}' for j in o[1:5]))
np.savez_compressed('atac_summary.npz', names=names, corr=C)
PY"
