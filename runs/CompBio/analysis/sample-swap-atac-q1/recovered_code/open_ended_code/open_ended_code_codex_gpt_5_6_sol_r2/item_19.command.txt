/bin/bash -lc "python - <<'PY'
import gzip,re,collections
f='AmexT_v47-AmexG_v6.0-DD.gtf.gz'
c=collections.Counter(); syms=collections.Counter(); ex=[]
with gzip.open(f,'rt') as h:
 for line in h:
  if line.startswith('#'):continue
  z=line.rstrip().split('\\t');
  if len(z)<9:continue
  if z[2]=='gene':
   c[z[0]]+=1
   m=re.search(r'gene_name \"(["'^"]+)'"',z[8]); gn=m.group(1) if m else ''
   for s in re.findall(r'(["'^|]+?) '"\\[(?:hs|mm|nr|dr)\\]',gn): syms[s.strip().upper()]+=1
   if len(ex)<8:ex.append((z[0],z[3],z[4],z[6],gn))
print('chrom gene counts',c.most_common(30))
print('unique symbols',len(syms),'total',sum(syms.values()))
print(*ex,sep='\\n')
PY"
