/bin/bash -lc "python3 - <<'PY'
import json,glob
import numpy as np
print('files',len(glob.glob('bw_coarse_*.npy')))
q=np.load('q_coarse_counts.npy'); s=json.load(open('q_profile_samples.json')); X=np.log1p(q/q.sum(0)*1e6)
tracks=[]; C=[]
for f in glob.glob('bw_coarse_*.npy'):
 tr=f[len('bw_coarse_'):-4]; y=np.log1p(np.load(f)); c=np.corrcoef(np.column_stack([X,y]).T)[:-1,-1]; tracks.append(tr); C.append(c)
C=np.array(C)
for i,tr in enumerate(tracks):
 o=np.argsort(C[i])[::-1]
 print(f'{tr:36s}', ' '.join(f'{s[j]}={C[i,j]:.3f}' for j in o[:5]))
print('mean ranks')
for j,x in enumerate(s):
 ranks=np.argsort(np.argsort(-C,axis=1),axis=1)[:,j]
 print(x, round(float(np.mean(ranks)),2), int(np.median(ranks)), 'top', int(sum(ranks==0)))
PY"
