/bin/bash -lc "python3 - <<'PY'
import gzip, collections, os
p='sample.swap.atac.q1.tsv.gz'
chrom=collections.Counter(); starts=[]; ends=[]; rows=0; zeros=0; sums=[0]*15; mins=[10**9]*15; maxs=[0]*15
with gzip.open(p,'rt') as f:
    header=f.readline().rstrip().split('\\t')
    for line in f:
        x=line.rstrip().split('\\t')
        rows+=1
        chrom[x[0]]+=1
        if rows<=3: starts.append((x[0],x[1],x[2]))
        vals=list(map(int,x[3:]))
        for j,v in enumerate(vals):
            sums[j]+=v; mins[j]=min(mins[j],v); maxs[j]=max(maxs[j],v)
        zeros+=sum(v==0 for v in vals)
print('header',header)
print('rows',rows,'chroms',len(chrom),'first',starts,'lastchrom',list(chrom.items())[-5:])
print('row count distribution',collections.Counter(chrom.values()).most_common(10))
print('sums',dict(zip(header[3:],sums)))
print('mins',mins,'maxs',maxs,'zero_fraction',zeros/(rows*15))
print('sizes lines',sum(1 for _ in open('sample.swap.atac.q1.chrom.sizes')))
PY"
