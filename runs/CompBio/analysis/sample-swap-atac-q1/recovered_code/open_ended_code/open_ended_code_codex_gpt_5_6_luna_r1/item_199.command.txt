/bin/bash -lc "python3 - <<'PY'
import pyBigWig,time
b=pyBigWig.open('https://bis.zju.edu.cn/uuatac/data/axolotl_bw/Cholangiocyte.bw')
for c in ['chr1p_1','chr1p_2','chr13p']:
 t=time.time()
 try:
  a=b.intervals(c,0,min(1000000,b.chroms(c)))
  print(c,len(a) if a else None,a[:3] if a else None,'time',time.time()-t)
 except Exception as e: print('err',e)
b.close()
PY"
