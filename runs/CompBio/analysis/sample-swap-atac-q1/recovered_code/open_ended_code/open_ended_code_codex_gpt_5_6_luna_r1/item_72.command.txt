/bin/bash -lc 'next=$(sed -n '"'s/"'^link: <'"\\(["'^>]*'"\\)>; rel=\"next\"/\\1/p' hf_headers.txt)
page=1
while [ -n \""'$next" ] && [ "$page" -le 10 ]
do
  curl -L --fail --silent --show-error "$next" -o "hf_tree_page_$page.json"
  next=$(curl -L --silent --show-error -D - -o /dev/null "$next" | sed -n '"'s/"'^link: <'"\\(["'^>]*'"\\)>; rel=\"next\"/\\1/p' | tr -d '\\r')
  page="'$((page+1))
done
python3 - <<'"'PY'
import glob,json
all=[]
for p in ['hf_tree_page.json']+sorted(glob.glob('hf_tree_page_*.json')): all.extend(json.load(open(p)))
print('n',len(all))
for z in all:
 p=z.get('path','')
 if 'sample.swap' in p.lower() or 'answer' in p.lower() or 'chrom.sizes' in p.lower(): print(p,z.get('size'))
PY"
