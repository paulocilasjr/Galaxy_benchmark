/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, time
from scipy.stats import spearmanr
p='sample.swap.atac.q1.tsv.gz'
t0=time.time()
with gzip.open(p,'rt') as f:
    names=f.readline().rstrip().split('\\t')[3:]
    a=np.loadtxt(f,delimiter='\\t',usecols=range(3,18),dtype=np.float32)
print('loaded',a.shape,'sec',round(time.time()-t0,1))
# transformations / correlations
for label,x in [('raw',a),('log1p',np.log1p(a)),('row-normalized',a/(a.sum(axis=1,keepdims=True)+1)),('zrow', (a-a.mean(axis=1,keepdims=True))/(a.std(axis=1,keepdims=True)+1e-5))]:
    # profile corr columns
    c=np.corrcoef(x,rowvar=False)
    print('\\n',label)
    for i in range(len(names)):
        order=np.argsort(c[i])[::-1]
        print(names[i], ' '.join(f'{names[j]}:{c[i,j]:.4f}' for j in order[1:4]))
# counts correlation only positive/variable rows maybe
for cutoff in [0,1,10,50,100]:
    x=np.log1p(a)
    keep=(a.max(axis=1)>=cutoff)
    c=np.corrcoef(x[keep],rowvar=False)
    iu=np.triu_indices(len(names),1)
    pairs=sorted(((c[i,j],names[i],names[j]) for i,j in zip(*iu)), reverse=True)
    print('top log corr cutoff',cutoff,'n',keep.sum(),pairs[:15])
PY"
