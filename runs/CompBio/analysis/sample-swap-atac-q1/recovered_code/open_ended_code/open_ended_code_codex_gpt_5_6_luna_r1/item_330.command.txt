/bin/bash -lc "python3 - <<'PY'
import csv,math
s=[]; rows=[]
for r in csv.reader(open('promoter_scores.tsv'),delimiter='\\t'):
 if not s: s=r[1:]; continue
 rows.append((r[0],[float(x) for x in r[1:]]))
import numpy as np
A=np.array([x[1] for x in rows]); A=np.log1p(A/ A.sum(0)*1e6)
for name in ['GallBladder','Cloaca','Stomach','Intestine','Liver','Gill','Bladder','Kidney','Limb']:
 j=s.index(name)
 # score vs mean other / t statistic maybe difference from mean others
 others=np.delete(A,j,axis=1); sc=A[:,j]-others.mean(1)
 o=np.argsort(sc)[::-1]
 print('\\n====',name)
 print(' '.join(f'{rows[i][0]}({sc[i]:.2f})' for i in o[:80]))
PY"
