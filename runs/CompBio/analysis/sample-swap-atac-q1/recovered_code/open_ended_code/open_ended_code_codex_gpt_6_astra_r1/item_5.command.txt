/bin/bash -lc "cat > inspect_counts.py <<'PY'
import pandas as pd, numpy as np
from scipy.stats import spearmanr
x=pd.read_csv('sample.swap.atac.q1.tsv.gz',sep='\\t')
x.to_pickle('counts.pkl')
a=x.iloc[:,3:].to_numpy(); names=x.columns[3:]
print('Shape',x.shape)
print(pd.DataFrame({'sum':a.sum(0),'median':np.median(a,axis=0),'zero':(a==0).mean(0)},index=names))
n=np.log1p(a/a.sum(0)*1e7)
print('Pearson log CPM')
print(pd.DataFrame(np.corrcoef(n.T),index=names,columns=names).round(2).to_string())
# select high variation rows
v=np.std(n,axis=1); sub=n[v>np.quantile(v,.95)]
print('Pearson variable')
print(pd.DataFrame(np.corrcoef(sub.T),index=names,columns=names).round(2).to_string())
np.save('lognorm.npy',n)
PY
OPENBLAS_NUM_THREADS=2 python inspect_counts.py > count_summary.txt"
