/bin/bash -lc "python3 - <<'PY'
import gzip
f='sample.swap.atac.q1.tsv.gz'
for chrom in ['chr1p_1','chr1p_2','chr1p_3','chr6q_4','chr14q']:
 first=[]; last=[]; n=0
 with gzip.open(f,'rt') as h:
  next(h)
  for line in h:
   a=line.rstrip().split('\\t')
   if a[0]==chrom:
    if not first:first=a
    last=a;n+=1
   elif first and a[0]"'!=chrom: break
 print(chrom,'"'n',n,'first',first[:3], 'last',last[:3])
PY"
