/bin/bash -lc "python3 - <<'PY'
import pandas as pd, pickle, json, numpy as np, gzip
from scipy.stats import spearmanr
cell=pd.read_csv('axolotl_cellColData.csv', index_col=0)
samples=['Bladder','Brain','Cloaca','GallBladder','Gill','Heart','Intestine','Kidney','Limb','Liver','Lung','Pancreas','Prostate','Spleen','Stomach']
lineages=sorted(cell['subanno'].unique())
counts=cell.groupby(['Sample','subanno']).size().unstack(fill_value=0).reindex(index=samples,columns=lineages,fill_value=0)
props=counts.div(counts.sum(axis=1),axis=0).values
d=pickle.load(open('cluster_exp_dict.pkl','rb')); gene_indices=json.load(open('selected_gene_indices.json')); names=list(gene_indices.keys())
acc=np.load('promoter_acc.npy')
tot=np.zeros(15)
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
 f.readline()
 for line in f:
  p=line.split('\\t'); tot += np.array([int(x) for x in p[3:]],float)
P=np.zeros((len(names),15))
for gi,name in enumerate(names): P[gi,:]=acc[gene_indices[name]].sum(axis=0)
P=np.log2(P/tot*1e6+1)
E=np.zeros_like(P)
for gi,name in enumerate(names):
 arr=np.array([d[name].get(L,0.0) for L in lineages])
 E[gi,:]=props.dot(arr)
# precompute current correlations? function
def diag_score(Pmat):
    s=0; pairs=[]
    for i in range(15):
        r=spearmanr(Pmat[:,i],E[:,i]).correlation
        s+=r; pairs.append(r)
    return s,pairs
base,pairs=diag_score(P)
print('base diag sum',base)
print(pairs)
results=[]
for i in range(15):
 for j in range(i+1,15):
    P2=P.copy(); P2[:,[i,j]]=P2[:,[j,i]]
    s,_=diag_score(P2)
    results.append((s,i,j))
results.sort(reverse=True)
print('top 20 swaps:')
for s,i,j in results[:20]:
 print(samples[i],samples[j],s, 'gain',s-base)
PY"
