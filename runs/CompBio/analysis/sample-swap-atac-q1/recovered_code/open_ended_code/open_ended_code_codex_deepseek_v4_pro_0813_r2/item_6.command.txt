/bin/bash -lc "python3 - <<'PY'
import gzip, csv, time, math
fn='sample.swap.atac.q1.tsv.gz'
names=None
n=0
sums=None
ss=None
pairs=None
with gzip.open(fn,'rt') as f:
    r=csv.reader(f,delimiter='\\t')
    names=next(r)[3:]
    m=len(names)
    sums=[0.0]*m
    ss=[0.0]*m
    pairs=[[0.0]*m for _ in range(m)]
    for row in r:
        vals=list(map(int,row[3:]))
        for i,v in enumerate(vals):
            sums[i]+=v
            ss[i]+=float(v)*v
            pi=pairs[i]
            for j in range(m):
                pi[j]+=float(v)*vals[j]
        n+=1
        if n%500000==0:
            print('rows',n, flush=True)
print('n',n)
# correlations
for i in range(m):
    for j in range(m):
        cov=pairs[i][j]-sums[i]*sums[j]/n
        den=math.sqrt((ss[i]-sums[i]*sums[i]/n)*(ss[j]-sums[j]*sums[j]/n))
        pairs[i][j]=cov/den if den else 0
print('\\t'.join(['']+names))
for i in range(m):
    print('\\t'.join([names[i]]+[f'{pairs[i][j]:.3f}' for j in range(m)]))
PY"
