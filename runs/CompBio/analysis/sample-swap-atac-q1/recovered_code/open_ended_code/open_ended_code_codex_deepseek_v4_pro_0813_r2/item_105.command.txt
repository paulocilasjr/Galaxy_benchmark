/bin/bash -lc "python3 - <<'PY'
import re, urllib.parse, json, numpy as np
html=open('cluster_exp_listing.html').read()
hrefs=re.findall(r'href=\"(["'^"]+'"\\.json)\"',html)
names=[urllib.parse.unquote(h) for h in hrefs]
print('listing',len(names), 'unique',len(set(names)))
gn=json.load(open('gene_names.json'))
# nonzero genes from acc
acc=np.load('promoter_acc.npy')
nz=np.where(acc.sum(axis=1)>0)[0]
gn_nz=[gn[i] for i in nz]
print('nonzero genes',len(gn_nz),'unique',len(set(gn_nz)))
setnames=set(names)
ov=[g for g in set(gn_nz) if g in setnames]
print('overlap unique',len(ov))
# How many nonzero gene indices have any name overlap? duplicate etc
print('indices with name in listing',sum(g in setnames for g in gn_nz))
PY"
