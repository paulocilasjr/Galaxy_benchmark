import pandas as pd,numpy as np
atac=pd.read_csv('atac_gene_promoter_counts.tsv',sep='\\t'); rna=pd.read_csv('bulk_rna_counts_clean.tsv',sep='\\t')
A=atac.groupby('symbol').sum(numeric_only=True); R=rna.set_index('symbol'); common=A.index.intersection(R.index)
Ac=A.loc[common].div(A.loc[common].sum(axis=0),axis=1)*1e6; Rc=R.loc[common].div(R.loc[common].sum(axis=0),axis=1)*1e6
mask=((Ac>0).sum(axis=1)>0)&((Rc>0).sum(axis=1)>0); Ac=Ac[mask]; Rc=Rc[mask]
Al=np.log1p(Ac); Rl=np.log1p(Rc)
def z(X): return ((X-X.mean(axis=1,keepdims=True))/(X.std(axis=1,keepdims=True).replace(0,1)))
Az=z(Al); Rz=z(Rl)
atac_cols=A.columns.tolist(); rna_cols=R.columns.tolist()
P=np.corrcoef(Az.T,Rz.T)[:len(atac_cols),len(atac_cols):]
df=pd.DataFrame(P,index=atac_cols,columns=rna_cols)
for swapped in [False,True]:
 print('\\nSwapped?',swapped)
 cols=atac_cols.copy()
 if swapped:
  i=cols.index('Cloaca'); j=cols.index('Stomach'); cols[i],cols[j]=cols[j],cols[i]

  if ref:
   k=rna_cols.index(ref)
   print(c, 'actual data col label ->', 'corr to ref', round(df.loc[c,ref],3))
