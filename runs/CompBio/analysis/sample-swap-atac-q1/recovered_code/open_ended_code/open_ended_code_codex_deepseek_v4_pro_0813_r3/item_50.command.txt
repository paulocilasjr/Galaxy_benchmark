/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
atac=pd.read_csv('atac_gene_promoter_counts.tsv',sep='\\t'); rna=pd.read_csv('bulk_rna_counts_clean.tsv',sep='\\t')
A=atac.groupby('symbol').sum(numeric_only=True); R=rna.set_index('symbol'); common=A.index.intersection(R.index)
Ac=A.loc[common].div(A.loc[common].sum(axis=0),axis=1)*1e6; Rc=R.loc[common].div(R.loc[common].sum(axis=0),axis=1)*1e6
clo=Rc['Cloacal_chamber_featurecounts']; sto=Rc['Stomach_featurecounts']; mask=((clo>1)|(sto>1)).values; clo=clo[mask]; sto=sto[mask]
fc=np.log2((sto+1)/(clo+1)); top_sto=fc.sort_values(ascending=False).head(100); top_clo=fc.sort_values(ascending=True).head(100)
for name,genes in [('stomach markers',top_sto),('cloaca markers',top_clo)]:
 sub=Ac.loc[Ac.index.isin(genes.index)]
 print('\\n',name, 'ATAC mean log1p CPM')
 print(np.log1p(sub).mean(axis=0).sort_values(ascending=False).round(3))
PY"
