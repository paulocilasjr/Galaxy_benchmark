/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
atac=pd.read_csv('atac_gene_promoter_counts.tsv',sep='\\t'); rna=pd.read_csv('bulk_rna_counts_clean.tsv',sep='\\t')
A=atac.groupby('symbol').sum(numeric_only=True); R=rna.set_index('symbol'); common=A.index.intersection(R.index)
Ac=A.loc[common].div(A.loc[common].sum(axis=0),axis=1)*1e6; Rc=R.loc[common].div(R.loc[common].sum(axis=0),axis=1)*1e6
mask=((Ac>0).sum(axis=1)>0)&((Rc>0).sum(axis=1)>0); Ac=Ac[mask]; Rc=Rc[mask]
Al=np.log1p(Ac); Rl=np.log1p(Rc)
def z(X):
 mu=X.mean(axis=1); sd=X.std(axis=1); sd=sd.replace(0,1)
 return X.sub(mu,axis=0).div(sd,axis=0)
Az=z(Al); Rz=z(Rl)
atac_cols=A.columns.tolist(); rna_cols=R.columns.tolist()
P=np.corrcoef(Az.T,Rz.T)[:len(atac_cols),len(atac_cols):]
df=pd.DataFrame(P,index=atac_cols,columns=rna_cols)
refmap={'Bladder':'Bladder_featurecounts','Brain':'Brain_featurecounts','Cloaca':'Cloacal_chamber_featurecounts','Gill':'Gill_featurecounts','Heart':'Heart_featurecounts','Intestine':'Intestine_featurecounts','Kidney':'Kidney_featurecounts','Limb':'Fore_limb_featurecounts','Liver':'Liver_featurecounts','Lung':'Lung_featurecounts','Spleen':'Spleen_featurecounts','Stomach':'Stomach_featurecounts'}
print('Original labels diagonal corr:')
for c in atac_cols:
 ref=refmap.get(c)
 if ref: print(c, round(df.loc[c,ref],3), 'best', df.loc[c].idxmax(), round(df.loc[c].max(),3))
print('\\nAfter swapping Cloaca/Stomach labels:')
for c in atac_cols:
 ref=refmap.get(c)
 if ref: print(c, round(df.loc[c,ref],3), 'best', df.loc[c].idxmax(), round(df.loc[c].max(),3))
PY"
