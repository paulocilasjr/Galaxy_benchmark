    for row in r:
        c=row[0]; rows[c]+=1
        arr=sums[c]
        vals=list(map(int,row[3:]))
        for i,v in enumerate(vals): arr[i]+=v
print('chrom nrows length total_sizes')
for c in sorted(sizes, key=lambda x:(int(x.split('p')[0].replace('chr','')), x)):
    arr=sums[c]
    # print only some
