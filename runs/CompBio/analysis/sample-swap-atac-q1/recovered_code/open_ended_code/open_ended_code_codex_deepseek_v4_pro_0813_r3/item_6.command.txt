/bin/bash -lc "python3 - <<'PY'
import gzip,csv,collections,re,math
sizes={}
for line in open('sample.swap.atac.q1.chrom.sizes'):
    c,s=line.split(); sizes[c]=int(s)
cols=None; sums=collections.defaultdict(lambda:[0]*15); rows=collections.defaultdict(int)
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t'); h=next(r); cols=h[3:]
    for row in r:
        c=row[0]; rows[c]+=1
        arr=sums[c]
        for i,v in enumerate(map(int,row[3:])): arr[i]+=v
def key(c):
    m=re.match(r'chr(\\d+)([pq])',c); return int(m.group(1)), m.group(2), c
# totals per col
totals=[sum(sums[c][i] for c in sizes) for i in range(len(cols))]
print('sample totals')
for c,t in zip(cols,totals): print(c,t)
# print fraction per chrom per sample maybe compact; find cols with zeros later
print('\\nlate chromosome fractions (x1000) and totals')
chroms=sorted(sizes,key=key)
for c in chroms:
    arr=sums[c]
    print(c, ' '.join(f'{a/totals[i]*1000:7.2f}' if totals[i] else '    nan' for i,a in enumerate(arr)))
PY"
