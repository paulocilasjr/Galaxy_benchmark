/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np, time
p='sample.swap.atac.q1.tsv.gz'
cols=list(pd.read_csv(p,sep='\\t',nrows=0).columns[3:]); m=len(cols)
tot=np.zeros(m,dtype=np.float64)
for ch in pd.read_csv(p,sep='\\t',usecols=range(3,18),chunksize=200000): tot += ch.to_numpy(dtype=np.float64).sum(0)
print('totals',dict(zip(cols,tot.astype(int))),flush=True)
# accumulate log CPM moments/cross products
s=np.zeros(m); cross=np.zeros((m,m)); n=0
for ch in pd.read_csv(p,sep='\\t',usecols=range(3,18),chunksize=200000):
    x=ch.to_numpy(dtype=np.float64)
    z=np.log1p(x/tot*1e8)
    s += z.sum(0); cross += z.T@z; n += len(z)
mean=s/n; cov=cross/n-np.outer(mean,mean); cor=cov/np.sqrt(np.outer(np.diag(cov),np.diag(cov)))
np.set_printoptions(precision=4,suppress=True,linewidth=240)
print('logCPM corr'); print('       '+' '.join(f'{c[:8]:>8}' for c in cols))
for c,row in zip(cols,cor): print(f'{c[:8]:>8} '+' '.join(f'{v:8.4f}' for v in row))
print('top pairs')
for i in range(m):
 for j in range(i): print(f'{cor[i,j]:.6f}\\t{cols[j]}\\t{cols[i]}')
PY"
