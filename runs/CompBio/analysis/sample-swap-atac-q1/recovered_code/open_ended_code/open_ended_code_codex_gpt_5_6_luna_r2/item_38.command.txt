/bin/bash -lc "python3 -u - <<'PY'
import gzip, urllib.request, pandas as pd, numpy as np
urls={
'Cloaca':'https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM8426nnn/GSM8426946/suppl/GSM8426946_Axolotl_Cloaca_Fragments.bed.gz',
'Stomach':'https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM8426nnn/GSM8426973/suppl/GSM8426973_Axolotl_Stomach_Fragments.bed.gz',
}
# Input bins and counts through 100 kb
x=pd.read_csv('sample.swap.atac.q1.tsv.gz',sep='\\t',nrows=20)
x=x[(x.chrom=='chr1p_1') & (x.start<100000)].copy()
print('input rows')
print(x[['start','end','Cloaca','Stomach']].to_string(index=False))
for name,url in urls.items():
    req=urllib.request.urlopen(url,timeout=60); gz=gzip.GzipFile(fileobj=req)
    bins={i:0 for i in range(0,100000,10000)}
    n=0
    for raw in gz:
        f=raw.split(b'\\t')
        if len(f)<3: continue
        if f[0]"'!=b'"'chr1p_1': break
        s,e=int(f[1]),int(f[2])
        if s>=100000: break
        if e>0:
            # fragment count assigned by overlap; all fragments here <1000
            for b in range(max(0,s//10000*10000), min(90000,(e-1)//10000*10000)+10000,10000):
                if s < b+10000 and e > b: bins[b]+=1
            n+=1
    print(name,'n<100k',n,'bins',[(b,bins[b]) for b in bins])
    req.close()
PY"
