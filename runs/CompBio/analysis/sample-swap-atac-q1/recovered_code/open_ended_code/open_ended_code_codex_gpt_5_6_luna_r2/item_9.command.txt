/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np, time, sys
p='sample.swap.atac.q1.tsv.gz'
cols=list(pd.read_csv(p,sep='\\t',nrows=0).columns[3:])
print('cols',cols,flush=True)
tot=np.zeros(len(cols),dtype=np.int64); nz=np.zeros(len(cols),dtype=np.int64); ss=np.zeros(len(cols),dtype=np.float64); sq=np.zeros(len(cols),dtype=np.float64); n=0
for i,ch in enumerate(pd.read_csv(p,sep='\\t',usecols=range(3,18),chunksize=100000)):
    x=ch.to_numpy(dtype=np.int64,copy=False)
    tot += x.sum(axis=0,dtype=np.int64); nz += (x>0).sum(axis=0); ss += x.sum(axis=0); sq += (x*x).sum(axis=0); n += len(x)
    if i%5==0: print(i,n,flush=True)
print('n',n,flush=True)
for c,a,b,d in zip(cols,tot,nz,sq): print(c, int(a), int(b), a/n, np.sqrt(max(0,d/n-(a/n)**2)),flush=True)
PY"
