def get_counts(acc, fname, limit=1000000):
    url='https://www.ncbi.nlm.nih.gov/geo/download/?acc=%s&format=file&file=%s' % (acc, urllib.parse.quote(fname))
    # shell handles broken pipe okay; use python streaming gzip? easier command gzip -dc pipe
    cmd=['bash','-lc', f\"set -o pipefail; curl -L --fail --silent {url"'!r} | gzip -dc"]
    proc=subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, bufsize=1)
    counts=defaultdict(int)
    try:
        for line in proc.stdout:
            p=line.rstrip('"'\\n').split('\\t')
            if len(p)<3: continue
            chrom=p[0]; st=int(p[1]); en=int(p[2])
            if chrom"'!='"'chr1p_1' or st>=limit:
                break
            sbin=(st//10000)*10000
            ebin=((en-1)//10000)*10000
            for b in range(sbin, min(ebin,limit-1)+1, 10000):
                counts[(b,b+10000)] += 1
    finally:
        try: proc.kill()
        except Exception: pass
        proc.stdout.close()
        # don't wait too long
        try: proc.wait(timeout=5)
        except Exception: pass
    return counts
