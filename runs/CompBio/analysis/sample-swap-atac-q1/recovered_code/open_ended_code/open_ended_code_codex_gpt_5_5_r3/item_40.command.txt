/bin/bash -lc "python3 - <<'PY'
import urllib.request,re
for path in ['assets/js/tissue_tree.js','call/mca_stats.php']:
    url='https://bis.zju.edu.cn/ACA/'+path
    try:
        txt=urllib.request.urlopen(url,timeout=20).read().decode('utf-8','replace')
        open('tmp/'+path.replace('/','_'),'w').write(txt)
        print(path, len(txt), txt[:500])
        for term in ['Cloaca','Gall','Limb','Liver']:
            print(term, [m.start() for m in re.finditer(term,txt)][:10])
    except Exception as e: print(e)
PY"
