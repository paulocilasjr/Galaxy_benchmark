/bin/bash -lc "cat > compute_corr.py <<'PY'
import gzip, csv, numpy as np
fn='sample.swap.atac.q1.tsv.gz'
names=None
sums=None; ss=None; dot=None; n=0
with gzip.open(fn,'rt') as f:
    r=csv.reader(f,delimiter='\\t')
    header=next(r)
    names=header[3:]
    p=len(names)
    sums=np.zeros(p); ss=np.zeros(p); dot=np.zeros((p,p))
    for row in r:
        vals=np.array(list(map(int,row[3:])),dtype=np.float64)
        sums+=vals; ss+=vals*vals; dot+=np.outer(vals,vals); n+=1
        if n%500000==0:
            print('processed',n,flush=True)
print('n',n)
means=sums/n
cov=dot/n - np.outer(means,means)
sd=np.sqrt(np.diag(cov))
corr=cov/np.outer(sd,sd)
print('names',names)
print('means')
for i,name in enumerate(names): print(i,name,means[i])
print('corr matrix')
print('\\t'.join(names))
for i,name in enumerate(names):
    print(name+'\\t'+'\\t'.join(f'{corr[i,j]:.4f}' for j in range(p)))
np.save('corr.npy',corr)
PY
python3 compute_corr.py"
