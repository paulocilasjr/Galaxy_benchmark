/bin/bash -lc "cat > gene_promoter.py <<'PY'
import gzip,csv,numpy as np,collections
# Load input keys and X
fn='sample.swap.atac.q1.tsv.gz'
keys=[]; X=[]
with gzip.open(fn,'rt') as f:
 r=csv.reader(f,delimiter='\\t'); header=next(r); names=header[3:]
 for row in r:
  keys.append((row[0],int(row[1])))
  X.append([int(v) for v in row[3:]])
X=np.array(X,dtype=np.float64)
key_to_idx={k:i for i,k in enumerate(keys)}
print('X',X.shape)
# Load genes
rows=[]
with open('gene_tss.tsv') as f:
 r=csv.DictReader(f,delimiter='\\t')
 for row in r:
  chrom=row['chrom']; pos=int(row['pos']); gene=row['gene']
  bs=(pos//10000)*10000
  idx=key_to_idx.get((chrom,bs))
  # neighbor bins idx-1 and +1 if same chrom
  vals=np.zeros(X.shape[1])
  for off in (-1,0,1):
   j=key_to_idx.get((chrom,bs+off*10000))
   if j is not None:
    vals += X[j]
  rows.append((gene,chrom,pos,bs,idx,vals))
G=np.array([r[5] for r in rows],dtype=np.float64)
genes=np.array([r[0] for r in rows],dtype=str)
chroms=np.array([r[1] for r in rows],dtype=str)
np.savez_compressed('promoter_matrix.npz', G=G, genes=genes, chroms=chroms, names=np.array(names,dtype=str))
print('saved',G.shape)
PY
python3 gene_promoter.py"
