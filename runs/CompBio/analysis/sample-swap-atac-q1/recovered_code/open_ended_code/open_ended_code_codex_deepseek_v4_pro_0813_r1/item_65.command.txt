/bin/bash -lc "gzip -cd sample.swap.atac.q1.tsv.gz | awk -f chromstats.awk | python3 -c '
import sys,re
names=[]
data=[]
for line in sys.stdin:
 line=line.rstrip(\"\\n\")
 # first token chrom n=...
 m=re.match(r\"(\\S+) n=(\\d+) (.*)\",line)
 if not m: continue
 chrom,n,rest=m.groups(); n=int(n)
 d={}
 for part in rest.split():
  mm=re.match(r\"(["'^_]+)_zero=('"\\d+)_avg=([\\d.]+)\",part)
  if mm:
   name=mm.group(1); z=int(mm.group(2)); avg=float(mm.group(3)); d[name]=(z/n,avg)
 data.append((chrom,n,d))
 if not names: names=list(d.keys())
# print max zero frac per sample and chrom
print(\"Sample\\\\Chrom maxzero fractions\")
for name in names:
 maxs=sorted([(d[name][0],chrom) for chrom,n,d in data],reverse=True)[:10]
 print(name, \", \".join(f\"{chrom}:{f:.3f}\" for f,c in maxs))
# print samples with zero frac >0.5 in each chrom
print(\"\\nChrom anomalies >0.5\")
for chrom,n,d in data:
 anom=[(name,round(d[name][0],3)) for name in names if d[name][0]>0.5]
 if anom: print(chrom,anom)
'"
