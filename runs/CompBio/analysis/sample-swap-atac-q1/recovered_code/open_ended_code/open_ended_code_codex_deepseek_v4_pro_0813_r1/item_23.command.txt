/bin/bash -lc "python3 - <<'PY'
import requests
url='https://ndownloader.figshare.com/files/43197084'
with requests.get(url,stream=True,timeout=60) as r:
    r.raise_for_status()
    total=int(r.headers.get('content-length',0))
    print('total',total)
    with open('proj_Axolotl.rds','wb') as f:
        done=0
        for chunk in r.iter_content(chunk_size=1<<20):
            f.write(chunk); done+=len(chunk)
            if done%(50<<20)==0 or done<0: print(done)
    print('done',done)
PY"
