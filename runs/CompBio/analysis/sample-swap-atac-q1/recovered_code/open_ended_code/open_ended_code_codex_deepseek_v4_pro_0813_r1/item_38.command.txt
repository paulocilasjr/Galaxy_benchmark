/bin/bash -lc "cat > compare_ref.py <<'PY'
import numpy as np
d=np.load('matrix_data.npz',allow_pickle=True)
X=d['X']; ref=d['ref']; names=list(d['names'])
print(X.shape, ref.shape, names)
# normalize each sample counts to library size? keep raw; pearson.
# compute correlation matrix sample x tissue
Xc=X-X.mean(axis=0)
rc=ref-ref.mean(axis=0)
denX=np.sqrt((Xc**2).sum(axis=0))
denR=np.sqrt((rc**2).sum(axis=0))
corr=(Xc.T@rc)/(np.outer(denX,denR)+1e-12)
print('Correlation rows=input labels, cols=reference tissues')
print('sample\\t'+'\\t'.join(names))
for i,n in enumerate(names):
 print(n+'\\t'+'\\t'.join(f'{corr[i,j]:.4f}' for j in range(len(names))))
# argmax per input column and score gap
for i,n in enumerate(names):
 order=np.argsort(corr[i])[::-1]
 print(f'{n:12s} best {names[order[0]]:12s} {corr[i,order[0]]:.4f} second {names[order[1]]:12s} {corr[i,order[1]]:.4f}')
# For each reference tissue, top matching sample
print('\\nRef tissue top samples')
for j,t in enumerate(names):
 order=np.argsort(corr[:,j])[::-1]
 print(f'{t:12s} best {names[order[0]]:12s} {corr[order[0],j]:.4f} second {names[order[1]]:12s} {corr[order[1],j]:.4f}')
np.save('corr_sample_ref.npy',corr)
PY
python3 compare_ref.py"
