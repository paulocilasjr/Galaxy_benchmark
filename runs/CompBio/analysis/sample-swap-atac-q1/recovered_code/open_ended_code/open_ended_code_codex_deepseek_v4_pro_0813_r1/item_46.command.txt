/bin/bash -lc "cat > compare_cellref.py <<'PY'
import numpy as np
d=np.load('matrix_data.npz',allow_pickle=True)
X=d['X']; names=list(d['names'])
ct=np.load('celltrack_data.npz',allow_pickle=True)
samples=list(ct['samples']); celltypes=list(ct['celltypes'])
# align sample order (same names)
R=ct['ref_frag']; Rc=ct['ref_count']
# Pearson correlations X col vs ref col (both dense)
for label,ref,title in [('ref_count',Rc,'count weighted'),('ref_frag',R,'frag weighted')]:
    Xc=X-X.mean(axis=0); rc=ref-ref.mean(axis=0)
    denX=np.sqrt((Xc**2).sum(axis=0)); denR=np.sqrt((rc**2).sum(axis=0))
    corr=(Xc.T@rc)/(np.outer(denX,denR)+1e-12)
    print('\\n===',title,'===')
    print('sample\\t'+'\\t'.join(samples))
    for i,n in enumerate(names):
        print(n+'\\t'+'\\t'.join(f'{corr[i,j]:.4f}' for j in range(len(samples))))
    print('Argmax per sample')
    for i,n in enumerate(names):
        order=np.argsort(corr[i])[::-1]
        print(f'{n:12s} best {samples[order[0]]:12s} {corr[i,order[0]]:.4f} second {samples[order[1]]:12s} {corr[i,order[1]]:.4f}')
    print('Ref top sample')
    for j,t in enumerate(samples):
        order=np.argsort(corr[:,j])[::-1]
        print(f'{t:12s} best {names[order[0]]:12s} {corr[order[0],j]:.4f} second {names[order[1]]:12s} {corr[order[1],j]:.4f}')
    np.save(title+'_corr.npy',corr)
PY
python3 compare_cellref.py"
