/bin/bash -lc "R --vanilla --quiet <<'RS'
x <- readRDS('proj_Axolotl.rds')
ns <- names(attributes(x))
cat('slot names:', paste(ns, collapse='\\n'), '\\n')
for(n in ns){
 obj <- attr(x,n)
 cat('\\n---', n, '---\\n')
 if(is.list(obj)) { cat('list length',length(obj),'names',paste(names(obj),collapse=','),'\\n') } else {cat('class',class(obj),'length',length(obj),'\\n')}
}
# sample metadata structure
sm <- attr(x,'sampleMetadata')
print(names(sm))
str(sm,max.level=1)
sc <- attr(x,'sampleColData')
print(sc)
cc <- attr(x,'cellColData')
print(dim(cc)); print(colnames(cc))
print(head(cc"'$Sample)); print(table(cc$Sample))
RS'
