import numpy as np
D=np.load('promoter_matrix.npz',allow_pickle=True); G=D['G']; genes=list(D['genes']); names=list(D['names'])
Md=np.load('matrix_data.npz',allow_pickle=True); totals=Md['X'].sum(axis=0)
for tissue,pats in {
'Heart':['TNNT2','NPPA.S[nr]','NKX2-5','MYH6'],
'Prostate':['NKX3-1','MSMB'],
'Kidney':['NPHS1[nr]','NPHS2','SLC12A1'],
'Spleen':['PTPRC','MPO','HBA1[hs]','HBZ[hs]'],
'Brain':['NEUROD1[nr]','SOX2','GFAP','RBFOX3'],
'Lung':['SFTPC','NKX2-1'],
'Liver':['ALB[nr]','APOA1','FGB','TTR[nr]'],
'Intestine':['VIL1','SI[nr]|MGAM[hs]','MUC2','FABP2'],
'Stomach':['GKN1','TFF2'],
'Bladder':['UPK1B','UPK2.S[nr]','UPK3A'],
'Gill':['FOXI1','ATP6V1B1','SLC4A4','STC1'],
}.items():
 print('\\n===',tissue)
