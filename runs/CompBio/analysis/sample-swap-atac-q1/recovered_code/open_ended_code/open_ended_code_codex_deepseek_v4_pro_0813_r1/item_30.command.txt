/bin/bash -lc "python3 - <<'PY'
import itertools, numpy as np
labels=['Bladder','Brain','Cloaca','GallBladder','Gill','Heart','Intestine','Kidney','Limb','Liver','Lung','Pancreas','Prostate','Spleen','Stomach']
input_tot=[701506327,1168148790,484000505,565994637,1094125058,868549497,699794434,745602917,1147926115,398317869,877686879,1398702406,973845979,497829833,455786546]
ref=[273265416,365993549,150065858,37722459,389787096,320246137,154154930,232490733,356774531,57871033,375099124,241875455,625904432,280236737,232112572]
d=dict(zip(labels,ref)); t=dict(zip(labels,input_tot))
base=np.array([t[l]/d[l] for l in labels])
print('current ratios')
for l,r in zip(labels,base): print(l,r)
print('base cv',base.std()/base.mean())
best=[]
for a,b in itertools.combinations(labels,2):
    ratios=[]
    for l in labels:
        if l==a: r=t[b]/d[l] # label a column actually b? If swap, data under a is actual b.
        elif l==b: r=t[a]/d[l]
        else: r=t[l]/d[l]
        ratios.append(r)
    ratios=np.array(ratios)
    cv=ratios.std()/ratios.mean()
    best.append((cv,a,b,ratios))
best.sort(key=lambda x:x[0])
for cv,a,b,ratios in best[:10]:
    print(a,b,'cv',cv,'range',ratios.min(),ratios.max())
    print(dict(zip(labels,ratios)))
PY"
