/bin/bash -lc "python - <<'PY'
import zipfile,re,statistics,math,os
num=r'[-+]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][-+]?\\d+)?'
pat_total=re.compile(r'Total tree length \\(sum of branch lengths\\):\\s*('+num+r')')
pat_int=re.compile(r'Sum of internal branch lengths:\\s*('+num+r')\\s*\\((["'^%]+)% of tree length'"\\)')
for zname in ['scogs_animals.zip','scogs_fungi.zip']:
    vals=[]; rows=[]; missing=[]
    with zipfile.ZipFile('/workspace/inputs/'+zname) as z:
        for n in sorted(x for x in z.namelist() if x.endswith('.iqtree')):
            txt=z.read(n).decode('utf-8', errors='replace')
            mt=pat_total.search(txt); mi=pat_int.search(txt)
            if not (mt and mi):
                missing.append(n); continue
            total=float(mt.group(1)); internal=float(mi.group(1)); pct=float(mi.group(2))
            val=internal/total if total else math.nan
            vals.append(val); rows.append((n,total,internal,pct/100,val))
    print(zname)
    print('parsed',len(vals),'missing summary',len(missing))
    print('min median max',min(vals),statistics.median(vals),max(vals))
    print('zeros',sum(v==0 for v in vals),'nonzero',sum(v>0 for v in vals))
    print('first5',rows[:5])
PY"
