/bin/bash -lc "python3 - <<'PY'
from zipfile import ZipFile
from statistics import median
import re

num = r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?'
branch_re = re.compile(r':(' + num + r')')
internal_re = re.compile(r'\\)["'^:(),;]*:('"' + num + r')')

def vals(path):
    out=[]
    with ZipFile(path) as z:
        names=sorted(n for n in z.namelist() if n.endswith('.treefile'))
        for n in names:
            nw=z.read(n).decode().strip()
            total=sum(float(x) for x in branch_re.findall(nw))
            internal=sum(float(x) for x in internal_re.findall(nw))
            if total <= 0:
                raise ValueError((path,n,nw,total))
            out.append((n, internal/total, internal, total, nw.count(':')))
    return out

animals=vals('inputs/scogs_animals.zip')
fungi=vals('inputs/scogs_fungi.zip')
for label, data in [('animals', animals), ('fungi', fungi)]:
    arr=[x[1] for x in data]
    print(label, 'n', len(arr), 'median', median(arr), 'min', min(arr), 'max', max(arr), 'zero_count', sum(v == 0 for v in arr))
    print('taxon branch count distribution', sorted({x[4] for x in data}), {k:sum(x[4]==k for x in data) for k in sorted({x[4] for x in data})})
print('difference fungi-animals', median([x[1] for x in fungi]) - median([x[1] for x in animals]))
PY"
