/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for group in ('animals','fungi'):
 with ZipFile(f'inputs/scogs_{group}.zip') as z:
  fasta=[n for n in z.namelist() if n.endswith('.faa')]
  aln=[n for n in z.namelist() if n.endswith('.mafft.clipkit')]
  trees=set(n.removesuffix('.faa.mafft.clipkit.treefile') for n in z.namelist() if n.endswith('.treefile'))
  c=Counter()
  success=Counter()
  for n in fasta:
   stem=n[:-4]
   k=sum(line.startswith(b'>') for line in z.read(n).splitlines())
   c[k]+=1
   success[(k,stem in trees)]+=1
  print(group,'input seq counts',c,'success',success)
  # examine one failed log by seq count
  for k in sorted(c):
   failed=[n[:-4] for n in fasta if sum(line.startswith(b'>') for line in z.read(n).splitlines())==k and n[:-4] not in trees]
   if failed:
    stem=failed[0]; log=f'{stem}.faa.mafft.clipkit.log'
    print('example failed',k,stem, z.read(log).decode(errors='replace')[-800:])
PY"
