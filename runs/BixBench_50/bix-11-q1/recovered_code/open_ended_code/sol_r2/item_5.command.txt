/bin/bash -lc 'for z in inputs/scogs_animals.zip inputs/scogs_fungi.zip; do
  echo "$z"
  printf '"'treefiles '; unzip -Z1 \""'$z" | grep -Ec '"'\\.treefile"'$'"'
  printf 'iqtree '; unzip -Z1 \""'$z" | grep -Ec '"'\\.iqtree"'$'"'
  printf 'trimmed_alignments '; unzip -Z1 \""'$z" | grep -Ec '"'\\.faa\\.mafft\\.clipkit"'$'"'
done
python - <<'PY'
import zipfile,re,statistics,collections
num=r'(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?'
for group in ('animals','fungi'):
 zpath=f'inputs/scogs_{group}.zip'
 vals=[]; details=[]
 with zipfile.ZipFile(zpath) as z:
  names=sorted(n for n in z.namelist() if n.endswith('.treefile'))
  for n in names:
   s=z.read(n).decode().strip()
   allbl=[float(x) for x in re.findall(r':('+num+r')',s)]
   internal=[float(x) for x in re.findall(r'\\)\\s*(?:["'^():;,]+)?'"\\s*:('+num+r')',s)]
   tips=len(re.findall(r'(?<=[(,])\\s*["'^():;,]+'"\\s*:',s))
   tr=sum(internal)/sum(allbl)
   vals.append(tr); details.append((n,tips,tr,sum(internal),sum(allbl),s))
 print(group,'n',len(vals),'tip counts',collections.Counter(d[1] for d in details))
 print('median',repr(statistics.median(vals)),'mean',sum(vals)/len(vals),'minmax',min(vals),max(vals),'zero',sum(x==0 for x in vals))
 sv=sorted(vals); k=len(vals)//2
 print('middle',[(k-1,repr(sv[k-1])),(k,repr(sv[k]))] if len(vals)%2==0 else [(k,repr(sv[k]))])
 globals()[group]=vals
print('fungi - animals',repr(statistics.median(fungi)-statistics.median(animals)))
print('animals - fungi',repr(statistics.median(animals)-statistics.median(fungi)))
print('absolute',repr(abs(statistics.median(fungi)-statistics.median(animals))))
PY"
