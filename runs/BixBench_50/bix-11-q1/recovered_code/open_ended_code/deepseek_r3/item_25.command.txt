/bin/bash -c "python3 - <<'PY'
import zipfile, collections, re
for zname in ['scogs_animals.zip','scogs_fungi.zip']:
    with zipfile.ZipFile('/workspace/inputs/'+zname) as z:
        names=z.namelist()
        faas=[n for n in names if n.endswith('.faa')]
        trees=[n for n in names if n.endswith('.treefile')]
        tree_ids={n.split('.faa.mafft')[0] for n in trees}
        # parse sequences in faa headers
        counts=collections.Counter(); total=0; no4=[]
        for n in faas:
            text=z.read(n).decode()
            seqs=re.findall(r'"'^>(.*)$'"', text, flags=re.M)
            counts[len(seqs)]+=1
            if len(seqs)"'!=4: no4.append((n,len(seqs),seqs[:5]))
        print('"'\\n',zname)
        print('faa files',len(faas),'counts by #seq',counts)
        print('tree ids',len(tree_ids))
        print('sample non4', no4[:10])
        # Count species coverage in faa
        allsp=collections.Counter()
        for n in faas:
            text=z.read(n).decode()
            seqs=re.findall(r'"'^>('"\\S+)', text, flags=re.M)
            for s in seqs:
                sp=s.split('_')[0] if '_' in s else s
                allsp[sp]+=1
        print('species faa coverage',allsp)
        # trees coverage similarly parse leaves manually? just count leaf species via regex '([A-Za-z_]+):'
        tree_sp=collections.Counter()
        leaf_counts=collections.Counter()
        for n in trees:
            text=z.read(n).decode()
            leaves=re.findall(r'([A-Za-z]+_[A-Za-z0-9]+):', text)
            # Better find taxon labels before colon
            labels=re.findall(r'([A-Za-z]+_[A-Za-z0-9]+):', text)
            leaf_counts[len(labels)]+=1
            for l in labels: tree_sp[l]+=1
        print('tree leaf count distribution',leaf_counts)
        print('tree species coverage',tree_sp)
PY"
