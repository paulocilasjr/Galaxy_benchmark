/bin/bash -c "python3 - <<'PY'
import zipfile, os, tempfile, statistics, numpy as np
from types import SimpleNamespace
from phykit.services.tree.treeness import Treeness

def phykit_treeness_full(path):
    t=Treeness(SimpleNamespace(tree=path))
    tree=t.read_tree_file()
    return t.calculate_treeness(tree)

for zname in ['scogs_animals.zip','scogs_fungi.zip']:
    vals=[]
    with zipfile.ZipFile('/workspace/inputs/'+zname) as z:
        treefiles=[n for n in z.namelist() if n.endswith('.treefile')]
        with tempfile.TemporaryDirectory() as td:
            for n in treefiles:
                p=os.path.join(td, os.path.basename(n))
                with open(p,'wb') as f: f.write(z.read(n))
                try:
                    v=phykit_treeness_full(p)
                except Exception as e:
                    print('ERR',zname,n,e); continue
                vals.append(v)
    print(zname, 'n',len(vals),'median full',repr(statistics.median(vals)),'numpy median',repr(float(np.median(vals))))
    print('round5', round(statistics.median(vals),5), 'round4', round(statistics.median(vals),4))
    print('mean',statistics.mean(vals))
PY"
