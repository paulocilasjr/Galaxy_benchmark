/bin/bash -lc "python - <<'PY'
import io, re, zipfile, statistics
from collections import Counter

paths = {'animals':'inputs/scogs_animals.zip','fungi':'inputs/scogs_fungi.zip'}
pat_total = re.compile(r'Total tree length \\(sum of branch lengths\\):\\s*([0-9.eE+-]+)')
pat_internal = re.compile(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)\\s*\\(([0-9.eE+-]+)% of tree length\\)')
for group,path in paths.items():
    vals=[]; ratios=[]; members=[]; missing=[]
    with zipfile.ZipFile(path) as z:
        iq = sorted(n for n in z.namelist() if n.endswith('.iqtree'))
        trees = {n[:-len('.iqtree')+'.treefile'] if False else n for n in []}
        for n in iq:
            s=z.read(n).decode('utf-8',errors='replace')
            mt=pat_total.search(s); mi=pat_internal.search(s)
            if not (mt and mi):
                missing.append(n); continue
            total=float(mt.group(1)); internal=float(mi.group(1)); pct=float(mi.group(2))
            vals.append(pct/100); members.append((n,total,internal,pct))
            ratios.append(internal/total if total else None)
    print(group, 'n',len(vals),'missing',len(missing))
    print(' pct/100 sorted first,last', sorted(vals)[:5], sorted(vals)[-5:])
    print('median displayed proportion', statistics.median(vals))
    print('median ratio rounded components', statistics.median(ratios))
    print('mean', statistics.mean(vals), statistics.mean(ratios))
    print('zero',sum(v==0 for v in vals), 'unique',len(set(vals)))
print('difference fungi - animals from displayed:', statistics.median([float(pat_internal.search(zipfile.ZipFile(paths['fungi']).read(n).decode()).group(2))/100 for n in []]) if False else '')
PY"
