/bin/bash -lc "python - <<'PY'
import re, zipfile
for group,path in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
  rows=[]
  with zipfile.ZipFile(path) as z:
    for n in z.namelist():
      if n.endswith('.iqtree'):
        s=z.read(n).decode('utf8','replace')
        m=re.search(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)\\s*\\(([0-9.eE+-]+)%',s)
        if m: rows.append((float(m.group(2)),n,float(m.group(1))))
  rows.sort()
  print('\\n',group,'n=',len(rows))
  for row in rows[max(0,len(rows)//2-4):len(rows)//2+5]: print(row)
PY"
