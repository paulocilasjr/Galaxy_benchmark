/bin/bash -lc "python3 - <<'PY'
import zipfile, re, statistics
# Cross-check total branch length by summing all Newick :length tokens; internal length by matching branch length immediately after a ')' group.
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    vals=[]
    with zipfile.ZipFile(zname) as z:
        for n in z.namelist():
            if n.endswith('.treefile'):
                s=z.read(n).decode().strip()
                all_lengths=[float(x) for x in re.findall(r':([0-9.eE+-]+)', s)]
                internal=[float(x) for x in re.findall(r'\\)(?:["'^:(),;]*)?:([0-9.eE+-]+)'"', s)]
                total=sum(all_lengths)
                vals.append(sum(internal)/total if total else float('nan'))
    print(zname, len(vals), statistics.median(vals), min(vals), max(vals))
PY"
