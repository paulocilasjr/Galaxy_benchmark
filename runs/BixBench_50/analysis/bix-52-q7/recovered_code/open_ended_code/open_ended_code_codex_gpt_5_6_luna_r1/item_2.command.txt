/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
p='inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
with open(p, newline='') as f:
    rows=list(csv.DictReader(f))
vals=[float(r['MethylationPercentage']) for r in rows]
kept=[v for v in vals if v > 90 or v < 10]
removed=[v for v in vals if not (v > 90 or v < 10)]
print('data_rows', len(rows))
print('kept_extreme_rows', len(kept))
print('removed_rows', len(removed))
print('missing_or_nonfinite', sum(not (v == v) for v in vals))
print('min_max', min(vals), max(vals))
print('exact_10', sum(v == 10 for v in vals), 'exact_90', sum(v == 90 for v in vals))
print('below10', sum(v < 10 for v in vals), 'above90', sum(v > 90 for v in vals))
print('between_inclusive', sum(10 <= v <= 90 for v in vals))
print('columns', list(rows[0]))
PY"
