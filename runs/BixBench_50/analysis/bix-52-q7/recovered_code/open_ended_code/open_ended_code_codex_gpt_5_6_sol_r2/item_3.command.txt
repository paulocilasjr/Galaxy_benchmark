/bin/bash -lc "python - <<'PY'
import csv, math
p='inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
with open(p,newline='') as f:
    rows=list(csv.DictReader(f))
vals=[]; missing=[]
for i,r in enumerate(rows, start=2):
    s=r['MethylationPercentage'].strip()
    try:
        v=float(s)
        if not math.isfinite(v): raise ValueError
        vals.append(v)
    except ValueError:
        missing.append((i,s))
kept=sum(v>90 or v<10 for v in vals)
removed=sum(not (v>90 or v<10) for v in vals)
print('data_rows',len(rows))
print('valid_values',len(vals))
print('missing_invalid',len(missing),missing[:10])
print('kept_extremes',kept)
print('removed',removed)
print('exact_10',sum(v==10 for v in vals),'exact_90',sum(v==90 for v in vals))
print('range',min(vals),max(vals))
PY
wc -l inputs/ZF_AgeRelated_CpG_noMT_Final.csv"
