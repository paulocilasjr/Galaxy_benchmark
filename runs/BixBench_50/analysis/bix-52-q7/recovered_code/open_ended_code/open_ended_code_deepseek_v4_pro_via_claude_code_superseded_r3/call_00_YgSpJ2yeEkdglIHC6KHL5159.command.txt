find /workspace/inputs -name "ZF_AgeRelatedCpG_noMT_Final.csv" -exec node -e '
const fs = require("fs");
const data = fs.readFileSync(process.argv[1], "utf8");
const lines = data.split("\n");
let total = 0, kept = 0;
for (let i = 1; i < lines.length; i++) {
  if (!lines[i].trim()) continue;
  total++;
  const fields = lines[i].split(",");
  const meth = parseFloat(fields[7]);
  if (meth > 90 || meth < 10) kept++;
}
console.log("Total:", total);
console.log("Kept:", kept);
console.log("Removed:", total - kept);
' {} \;
