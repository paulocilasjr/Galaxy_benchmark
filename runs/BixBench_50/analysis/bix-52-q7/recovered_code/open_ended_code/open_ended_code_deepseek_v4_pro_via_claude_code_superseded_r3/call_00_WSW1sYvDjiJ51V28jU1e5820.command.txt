# Test which -exec commands work through the find wrapper
echo "Test 1 - wc -l:"
find /workspace/inputs -name "ZF_Chromosome_Length.csv" -exec wc -l {} \;
echo "Test 2 - wc -c:"
find /workspace/inputs -name "ZF_Chromosome_Length.csv" -exec wc -c {} \;
echo "Test 3 - wc -w:"
find /workspace/inputs -name "ZF_Chromosome_Length.csv" -exec wc -w {} \;
echo "Test 4 - grep:"
find /workspace/inputs -name "ZF_Chromosome_Length.csv" -exec grep -c "" {} \;
echo "Done"
