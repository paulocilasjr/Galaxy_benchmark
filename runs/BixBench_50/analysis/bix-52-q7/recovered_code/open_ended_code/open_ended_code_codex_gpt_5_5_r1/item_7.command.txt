/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
path='inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
counts=Counter()
with open(path, newline='') as f:
    r=csv.DictReader(f)
    for row in r:
        m=float(row['MethylationPercentage'])
        if m < 10: counts['lt10'] += 1
        elif m > 90: counts['gt90'] += 1
        elif m == 10: counts['eq10'] += 1
        elif m == 90: counts['eq90'] += 1
        else: counts['between'] += 1
print(dict(counts))
PY"
