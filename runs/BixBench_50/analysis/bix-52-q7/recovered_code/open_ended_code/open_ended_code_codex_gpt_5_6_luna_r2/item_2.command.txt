/bin/bash -lc "python3 - <<'PY'
import csv
from collections import Counter
path='/workspace/inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
with open(path, newline='') as f:
    reader=csv.DictReader(f)
    rows=0
    retained=0
    removed=0
    missing=[]
    invalid=[]
    boundaries=Counter()
    minv=float('inf'); maxv=float('-inf')
    for line,row in enumerate(reader, start=2):
        rows += 1
        raw=row['MethylationPercentage']
        try:
            v=float(raw)
        except Exception:
            invalid.append((line,raw))
            continue
        minv=min(minv,v); maxv=max(maxv,v)
        if v > 90 or v < 10:
            retained += 1
        else:
            removed += 1
            if v in (10.0,90.0): boundaries[v] += 1
    print({'data_rows':rows,'retained':retained,'removed':removed,'min':minv,'max':maxv,'boundary_rows':dict(boundaries),'invalid':invalid[:5],'invalid_count':len(invalid)})
PY"
