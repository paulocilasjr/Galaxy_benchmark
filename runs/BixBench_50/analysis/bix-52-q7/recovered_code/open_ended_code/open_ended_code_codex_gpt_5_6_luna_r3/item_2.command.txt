/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
p='inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
rows=0
removed=0
retained=0
missing=[]
boundary=Counter()
mins=None
maxs=None
with open(p, newline='') as f:
    reader=csv.DictReader(f)
    print('fields:', reader.fieldnames)
    for line_no,row in enumerate(reader, start=2):
        rows += 1
        raw=row['MethylationPercentage']
        try:
            x=float(raw)
        except Exception:
            missing.append((line_no,raw))
            continue
        mins=x if mins is None else min(mins,x)
        maxs=x if maxs is None else max(maxs,x)
        if x > 90 or x < 10:
            retained += 1
        else:
            removed += 1
            if x == 10 or x == 90:
                boundary[x] += 1
print('data_rows', rows)
print('retained_extreme', retained)
print('removed_non_extreme', removed)
print('min_max', mins, maxs)
print('missing_or_invalid', len(missing), missing[:5])
print('exact_boundaries_removed', dict(boundary))
PY"
