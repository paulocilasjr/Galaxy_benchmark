/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
p='inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
with open(p, newline='') as f:
    reader=csv.DictReader(f)
    rows=list(reader)
vals=[]
missing=0
for i,row in enumerate(rows, start=2):
    s=row['MethylationPercentage'].strip()
    if not s:
        missing+=1
    else:
        vals.append(float(s))
counts=Counter()
for x in vals:
    if x < 10:
        counts['lt10']+=1
    elif x > 90:
        counts['gt90']+=1
    elif x == 10:
        counts['eq10']+=1
    elif x == 90:
        counts['eq90']+=1
    else:
        counts['between']+=1
print('rows',len(rows))
print('numeric',len(vals),'missing',missing)
print(dict(counts))
print('retained',counts['lt10']+counts['gt90'])
print('removed',len(rows)-(counts['lt10']+counts['gt90']))
print('min',min(vals),'max',max(vals))
PY"
