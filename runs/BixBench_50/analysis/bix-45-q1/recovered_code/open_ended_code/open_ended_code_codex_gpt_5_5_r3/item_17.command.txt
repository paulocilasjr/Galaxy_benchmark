/bin/bash -lc "rm -rf work_rcv && mkdir -p work_rcv/animals work_rcv/fungi && python - <<'PY'
import zipfile, pathlib
pairs=[('inputs/scogs_animals.zip','work_rcv/animals'),('inputs/scogs_fungi.zip','work_rcv/fungi')]
for zip_path,out_dir in pairs:
    out=pathlib.Path(out_dir)
    with zipfile.ZipFile(zip_path) as z:
        for name in z.namelist():
            if name.endswith('.faa.mafft'):
                target=out/pathlib.Path(name).name
                target.write_bytes(z.read(name))
    print(out_dir, len(list(out.glob('*.faa.mafft'))))
PY
printf 'sample file: '; ls work_rcv/animals/*.faa.mafft | head -1
.venv/bin/phykit rcv \""'$(ls work_rcv/animals/*.faa.mafft | head -1)"
