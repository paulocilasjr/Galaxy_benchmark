/bin/bash -c "python3 - <<'PY'
import urllib.request,json,time
url='https://api.github.com/repos/JLSteenwyk/PhyKIT/tags?per_page=100'
data=json.load(urllib.request.urlopen(url,timeout=20))
for t in data[:30]:
 sha=t['commit']['sha']
 try:
  c=json.load(urllib.request.urlopen(f'https://api.github.com/repos/JLSteenwyk/PhyKIT/commits/{sha}',timeout=20))
  d=c['commit']['committer']['date']
 except Exception as e: d='?'
 print(t['name'],d)
PY"
