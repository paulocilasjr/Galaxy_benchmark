/bin/bash -c "PYTHONPATH=/workspace/tmp/phykit_old python3 - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy import stats

groups = {
    'animals': Path('/workspace/tmp/extract_ani'),
    'fungi': Path('/workspace/tmp/extract_fun'),
}
values = {}
for label, d in groups.items():
    arr = []
    for p in sorted(d.glob('*.faa.mafft')):
        raw = Alignment(alignment_file_path=str(p)).calculate_rcv()
        arr.append(round(raw, 4))
    values[label] = arr
    print(label, len(arr), min(arr), max(arr), sum(arr)/len(arr))
u, p = stats.mannwhitneyu(values['animals'], values['fungi'], alternative='two-sided')
print('U', u)
print('p', repr(float(p)))
print('p sci', f'{float(p):.15e}')
PY"
