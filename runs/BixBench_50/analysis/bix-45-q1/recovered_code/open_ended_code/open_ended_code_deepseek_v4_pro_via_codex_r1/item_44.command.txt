/bin/bash -c "python3 - <<'PY'
import urllib.request,json
# search commits API for path base.py maybe paginated
url='https://api.github.com/repos/JLSteenwyk/PhyKIT/commits?path=phykit/services/alignment/base.py&per_page=100'
try:
 data=json.load(urllib.request.urlopen(url,timeout=20))
 print(len(data))
 for c in data:
  print(c['sha'][:10], c['commit']['author']['date'], c['commit']['message'].split('\\n')[0])
except Exception as e: print('ERR',e)
PY"
