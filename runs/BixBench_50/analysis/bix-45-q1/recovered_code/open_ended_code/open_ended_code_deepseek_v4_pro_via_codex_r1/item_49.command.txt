/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from Bio import AlignIO
from collections import Counter
from scipy import stats
import json, statistics
# old phykit formula

def old_rcv(p):
    aln=AlignIO.read(str(p),'fasta')
    aln_len=aln.get_alignment_length(); n=len(aln)
    concat=''.join(str(r.seq) for r in aln)
    total=Counter(concat)
    avg={c: total[c]/n for c in total}
    s=0
    for r in aln:
        rc=Counter(r.seq)
        s += sum(abs(rc.get(c,0)-avg[c]) for c in total)/(n*aln_len)
    return s

for grp in ['ani','fun']:
    vals={p.name.split('.')[0]: round(old_rcv(p),4) for p in Path('/workspace/tmp/extract_'+grp).glob('*.faa.mafft')}
    Path('/workspace/tmp/old_rcv_'+grp+'.json').write_text(json.dumps(vals,indent=2))
    a=list(vals.values())
    print(grp,len(a),'zeros',sum(v==0 for v in a),'median',statistics.median(a),'mean',statistics.mean(a),'minmax',min(a),max(a))
    globals()[grp]=a
u,p=stats.mannwhitneyu(ani,fun,alternative='two-sided')
print('old rounded p',repr(p),'u',u)
# raw
rawani=[]; rawfun=[]
for p in Path('/workspace/tmp/extract_ani').glob('*.faa.mafft'): rawani.append(old_rcv(p))
for p in Path('/workspace/tmp/extract_fun').glob('*.faa.mafft'): rawfun.append(old_rcv(p))
u,p=stats.mannwhitneyu(rawani,rawfun,alternative='two-sided')
print('old raw p',repr(p),'u',u)
PY"
