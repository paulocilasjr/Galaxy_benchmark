/bin/bash -lc "PYTHONPATH=/workspace/work/lib python - <<'PY'
from pathlib import Path
from collections import Counter
from Bio import AlignIO
for d in ['animals','fungi']:
 c=Counter(); lengths=Counter();
 for p in Path('work',d).glob('*.faa.mafft'):
  aln=AlignIO.read(p,'fasta'); c[len(aln)]+=1; lengths[aln.get_alignment_length()]+=1
 print(d,'records',c,'n lengths',len(lengths),'length range',min(lengths),max(lengths))
 print('less4',[(p.name,len(AlignIO.read(p,'fasta'))) for p in Path('work',d).glob('*.faa.mafft') if len(AlignIO.read(p,'fasta'))<4][:30])
PY"
