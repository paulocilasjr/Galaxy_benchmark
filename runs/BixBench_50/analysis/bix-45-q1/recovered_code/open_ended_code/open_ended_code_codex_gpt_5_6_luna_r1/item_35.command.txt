/bin/bash -lc "PYTHONPATH=/workspace/work/lib python - <<'PY'
from pathlib import Path
from phykit.services.alignment.rcv import RelativeCompositionVariability
from scipy.stats import mannwhitneyu

class Args:
    json=False

def score(p):
    Args.alignment=str(p)
    return RelativeCompositionVariability(Args()).calculate_rcv()

A=[score(p) for p in sorted(Path('work/animals').glob('*.faa.mafft'))]
F=[score(p) for p in sorted(Path('work/fungi').glob('*.faa.mafft'))]
for label, aa, ff in [('full',A,F),('round4',[round(x,4) for x in A],[round(x,4) for x in F]),('round3',[round(x,3) for x in A],[round(x,3) for x in F])]:
    for alternative in ['two-sided','less','greater']:
        r=mannwhitneyu(aa,ff,alternative=alternative)
        print(label, alternative, 'U',repr(float(r.statistic)), 'p',repr(float(r.pvalue)))
    print('ranges',min(aa),max(aa),min(ff),max(ff),'med',sorted(aa)[len(aa)//2],sorted(ff)[len(ff)//2])
print('n',len(A),len(F),'first',A[:5],F[:5])
print('ties full',len(A)+len(F)-len(set(A+F)),'rounded',len(A)+len(F)-len(set(round(x,4) for x in A+F)))
PY"
