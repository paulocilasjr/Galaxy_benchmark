rcv <- function(path) {
  z <- readLines(path)
  h <- which(substr(z,1,1)=='>')
  s <- character(length(h))
  for (i in seq_along(h)) { e <- if (i<length(h)) h[i+1]-1 else length(z); s[i] <- toupper(paste(z[(h[i]+1):e],collapse='')) }
  n <- length(s); if (n<2) return(0)
  chars <- sort(unique(strsplit(paste(s,collapse=''),'')[[1]]))
  chars <- chars["'!chars %in% c('"'-','?','*','X')]
  if ("'!length(chars)) return(0)
  cnt <- matrix(0,n,length(chars)); lens <- integer(n)
  for (i in seq_len(n)) { q <- strsplit(s[i],'"'')[[1]]; ok <- "'!q %in% c('"'-','?','*','X'); lens[i] <- sum(ok); cnt[i,] <- tabulate(match(q[ok],chars), nbins=length(chars)) }
  av <- colSums(cnt)/n
  sum(rowSums(abs(cnt-matrix(av,n,length(chars),byrow=TRUE)))/(n*lens))
}
get <- function(d, rounded=FALSE) { p <- list.files(d,pattern='\\\\.faa\\\\.mafft"'$'"',full.names=TRUE); x <- vapply(p,rcv,0.0); if(rounded) x <- round(x,4); x }
a <- get('work/animals'); f <- get('work/fungi'); ar <- get('work/animals',TRUE); fr <- get('work/fungi',TRUE)
for (q in list(list(a,f,'full'),list(ar,fr,'round4'))) {
  tt <- wilcox.test(q[[1]],q[[2]], exact=FALSE, correct=TRUE)
  cat(q[[3]],'W=',unname(tt"'$statistic),'"'p=',format(tt"'$p.value,digits=17), '"'\\n')
}
RS"
