/bin/bash -lc "python - <<'PY'
import json
from scipy import stats
a=json.load(open('/workspace/tmp/an_rcv.json')); b=json.load(open('/workspace/tmp/an_rcv_cli.json'))

a2=json.load(open('/workspace/tmp/fu_rcv.json')); b2=json.load(open('/workspace/tmp/fu_rcv_cli.json'))

av=[v for _,v in b]; fv=[v for _,v in b2]
for alt in ['two-sided','less','greater']:
 r=stats.mannwhitneyu(av,fv,alternative=alt,method='auto')
