/bin/bash -lc "python - <<'PY'
import json
from scipy import stats
for grp in ['an','fu']:
 a=json.load(open(f'/workspace/tmp/{grp}_rcv.json')); b=json.load(open(f'/workspace/tmp/{grp}_rcv_cli.json'))
 ma=dict(a); mb=dict(b)
 print(grp,'same ids',set(ma)==set(mb),'maxdiff',max(abs(ma[k]-round(mb[k],4)) for k in ma))
av=[v for _,v in json.load(open('/workspace/tmp/an_rcv_cli.json'))]
fv=[v for _,v in json.load(open('/workspace/tmp/fu_rcv_cli.json'))]
for alt in ['two-sided','less','greater']:
 r=stats.mannwhitneyu(av,fv,alternative=alt,method='auto')
 print(alt,r.statistic,repr(r.pvalue))
PY"
