/bin/bash -lc "python - <<'PY'
from Bio import AlignIO
from collections import Counter
import glob, os, statistics, json
for grp in ['an','fu']:
 d=f'/workspace/tmp/scogs_{grp}'
 vals=[]
 for f in glob.glob(d+'/*.mafft'):
  aln=AlignIO.read(f,'fasta'); n=len(aln); L=aln.get_alignment_length()
  all_chars=''.join(str(r.seq) for r in aln); total_count=Counter(all_chars); total_chars=sum(total_count.values())
  avg={c:total_count[c]/total_chars for c in total_count}; devsum=0
  for rec in aln:
   seq=str(rec.seq); sc=Counter(seq)
   for c,total_c in total_count.items(): devsum+=abs(sc.get(c,0)-avg[c]*len(seq))
  vals.append(round(devsum/(n*L),4))
 print(grp,'n',len(vals),'median',statistics.median(vals),'mean',statistics.mean(vals),'minmax',min(vals),max(vals),'pct>0.25',sum(v>0.25 for v in vals)/len(vals))
 print('quantiles',statistics.quantiles(vals))
PY"
