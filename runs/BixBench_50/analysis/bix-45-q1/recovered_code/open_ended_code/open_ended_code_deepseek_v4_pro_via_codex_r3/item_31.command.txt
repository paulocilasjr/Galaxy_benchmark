/bin/bash -lc "python - <<'PY'
from collections import Counter
from Bio import AlignIO
import argparse
from phykit.services.alignment.rcv import RelativeCompositionVariability
p='/workspace/tmp/scogs_fu/1260807at2759.faa.mafft'
a=argparse.Namespace(alignment=p,json=False)
print('phykit raw', RelativeCompositionVariability(a).calculate_rcv())
# tool script
aln=AlignIO.read(p,'fasta'); n=len(aln); L=aln.get_alignment_length()
all_chars=''.join(str(r.seq) for r in aln); total_count=Counter(all_chars); total_chars=sum(total_count.values())
avg={c:total_count[c]/total_chars for c in total_count}
devsum=0
for rec in aln:
 seq=str(rec.seq); sc=Counter(seq); dev=0
 for c,total_c in total_count.items(): dev+=abs(sc.get(c,0)-avg[c]*len(seq))
 devsum+=dev
print('tool raw', devsum/(n*L), 'rounded', round(devsum/(n*L),4))
PY"
