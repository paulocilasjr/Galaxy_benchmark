import subprocess
import sys

# Get the shared genes list
with open('animal_genes.txt') as f:
    animal_genes = set(line.strip() for line in f)
with open('fungi_genes.txt') as f:
    fungi_genes = set(line.strip() for line in f)

shared = sorted(animal_genes & fungi_genes)
print(f"Number of shared orthologs: {len(shared)}")

# Run phykit rcv on all alignments
animal_rcv = []
fungi_rcv = []

for gene in shared:
    animal_file = f"work_animals/{gene}.mafft"
    fungi_file = f"work_fungi/{gene}.mafft"
    
    # Run phykit rcv on animal alignment
    result = subprocess.run(['phykit', 'rcv', animal_file], capture_output=True, text=True)
    try:
        animal_rcv.append(float(result.stdout.strip()))
    except:
        print(f"Failed to parse animal RCV for {gene}: {result.stdout}, {result.stderr}")
    
    # Run phykit rcv on fungal alignment
    result = subprocess.run(['phykit', 'rcv', fungi_file], capture_output=True, text=True)
    try:
        fungi_rcv.append(float(result.stdout.strip()))
    except:
        print(f"Failed to parse fungi RCV for {gene}: {result.stdout}, {result.stderr}")

print(f"\nAnimal RCV values: {len(animal_rcv)}")
print(f"Fungi RCV values: {len(fungi_rcv)}")

# Mann-Whitney U test
from scipy.stats import mannwhitneyu
stat, pvalue = mannwhitneyu(animal_rcv, fungi_rcv, alternative='two-sided')
print(f"\nMann-Whitney U statistic: {stat}")
print(f"p-value: {pvalue}")

# Summary stats
import numpy as np
print(f"\nAnimal RCV: mean={np.mean(animal_rcv):.4f}, median={np.median(animal_rcv):.4f}")
print(f"Fungi RCV: mean={np.mean(fungi_rcv):.4f}, median={np.median(fungi_rcv):.4f}")
