from scipy import stats

# Reconstruct from the scores
import subprocess, os

phykit = "/workspace/venv/bin/phykit"

animal_vals = []
for fname in sorted(os.listdir("/workspace/tmp_animals")):
    if fname.endswith('.mafft'):
        r = subprocess.run([phykit, 'rcv', f"/workspace/tmp_animals/{fname}"], capture_output=True, text=True)
        animal_vals.append(float(r.stdout.strip()))

fungal_vals = []
for fname in sorted(os.listdir("/workspace/tmp_fungi")):
    if fname.endswith('.mafft'):
        r = subprocess.run([phykit, 'rcv', f"/workspace/tmp_fungi/{fname}"], capture_output=True, text=True)
        fungal_vals.append(float(r.stdout.strip()))

print(f"Animals: n={len(animal_vals)}, mean={sum(animal_vals)/len(animal_vals):.6f}")
print(f"Fungi:   n={len(fungal_vals)}, mean={sum(fungal_vals)/len(fungal_vals):.6f}")

# Test with different settings
# Default: use_continuity=True, method='auto' (asymptotic), alternative='two-sided'
s1, p1 = stats.mannwhitneyu(animal_vals, fungal_vals, alternative='two-sided')
print(f"\nDefault (continuity=True, two-sided):  U={s1}, p={p1:.15e}")

s2, p2 = stats.mannwhitneyu(animal_vals, fungal_vals, alternative='two-sided', use_continuity=False)
print(f"No continuity correction, two-sided: U={s2}, p={p2:.15e}")

s3, p3 = stats.mannwhitneyu(animal_vals, fungal_vals, alternative='greater')
print(f"Continuity=True, greater:            U={s3}, p={p3:.15e}")

s4, p4 = stats.mannwhitneyu(animal_vals, fungal_vals, alternative='less')
print(f"Continuity=True, less:               U={s4}, p={p4:.15e}")
