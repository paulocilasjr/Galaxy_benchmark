/bin/bash -lc "python - <<'PY'
import json, subprocess
from pathlib import Path
from scipy.stats import mannwhitneyu

scores = {}
failures = []
for group in ['animals','fungi']:
    vals=[]
    for path in sorted(Path('work_rcv', group).glob('*.mafft')):
        cp = subprocess.run(['python','-m','phykit','rcv',str(path),'--json'], text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        if cp.returncode "'!= 0:
            failures.append((group, str(path), cp.stderr.strip()))
            continue
        try:
            vals.append(float(json.loads(cp.stdout)['"'rcv']))
        except Exception as e:
            failures.append((group, str(path), f'parse: {e}; out={cp.stdout"'!r}; err={cp.stderr!r}'"'))
    scores[group]=vals
    print(group, len(vals), 'min', min(vals), 'median-ish', sorted(vals)[len(vals)//2], 'max', max(vals))

print('failures', len(failures))
if failures:
    print(failures[:5])

animals=scores['animals']; fungi=scores['fungi']
for alt in ['two-sided','less','greater']:
    res=mannwhitneyu(animals, fungi, alternative=alt, method='auto')
    print('auto', alt, 'stat', res.statistic, 'p', repr(float(res.pvalue)))
for method in ['asymptotic','exact']:
    try:
        res=mannwhitneyu(animals, fungi, alternative='two-sided', method=method)
        print(method, 'two-sided stat', res.statistic, 'p', repr(float(res.pvalue)))
    except Exception as e:
        print(method, 'err', e)

# Also no continuity for reference.
res=mannwhitneyu(animals, fungi, alternative='two-sided', method='asymptotic', use_continuity=False)
print('asym no continuity two-sided stat', res.statistic, 'p', repr(float(res.pvalue)))
PY"
