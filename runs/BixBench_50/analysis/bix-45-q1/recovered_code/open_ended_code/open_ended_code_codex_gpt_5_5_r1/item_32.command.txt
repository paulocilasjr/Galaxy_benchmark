/bin/bash -lc ".venv/bin/python - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy.stats import mannwhitneyu
import subprocess

def phykit_score(p):
    return round(Alignment(alignment_file_path=str(p)).calculate_rcv(), 4)

animals=sorted(Path('work/scogs_animals').glob('*.faa.mafft'))
fungi=sorted(Path('work/scogs_fungi').glob('*.faa.mafft'))
# Verify a few imported PhyKIT values match the CLI-reported values.
for p in animals[:3] + fungi[:3]:
    cli=float(subprocess.check_output(['.venv/bin/phykit','rcv',str(p)], text=True).strip())
    lib=phykit_score(p)
    if cli "'!= lib:
        raise SystemExit(f'"'mismatch {p}: cli={cli} lib={lib}')

a=[phykit_score(p) for p in animals]
f=[phykit_score(p) for p in fungi]
res=mannwhitneyu(a, f, alternative='two-sided', method='asymptotic', use_continuity=True)
print(len(a), len(f), res.statistic, format(res.pvalue, '.17g'))
Path('codex_output').mkdir(exist_ok=True)
Path('codex_output/answer.txt').write_text(format(res.pvalue, '.17g') + '\\n')
print(Path('codex_output/answer.txt').read_text(), end='')
PY"
