/bin/bash -lc "python - <<'PY'
import zipfile
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        names=f.namelist()
    print(z, len(names), 'mafft', sum(n.endswith('.mafft') for n in names), 'faa', sum(n.endswith('.faa') for n in names))
    ids_mafft={n.removesuffix('.faa.mafft') for n in names if n.endswith('.faa.mafft')}
    ids_faa={n.removesuffix('.faa') for n in names if n.endswith('.faa')}
    print(' unique ids', len(ids_mafft|ids_faa), 'both', len(ids_mafft&ids_faa), 'only_mafft', len(ids_mafft-ids_faa), 'only_faa', len(ids_faa-ids_mafft))
PY"
