/bin/bash -lc "python - <<'PY'
from pathlib import Path
from Bio import AlignIO
from collections import Counter
from scipy.stats import mannwhitneyu

def old(p):
 a=AlignIO.read(p,'fasta');L=a.get_alignment_length();n=len(a);t=Counter(''.join(map(lambda r:str(r.seq),a)));av={x:t[x]/n for x in t}
 return sum(sum(abs(Counter(r.seq)[x]-av[x]) for x in t)/(n*L) for r in a)
A=[old(p) for p in sorted(Path('work_rcv/animals').glob('*.mafft'))]
F=[old(p) for p in sorted(Path('work_rcv/fungi').glob('*.mafft'))]
print('full old',mannwhitneyu(A,F), 'ties', len(A)+len(F)-len(set(A+F)))
print('full old no cont',mannwhitneyu(A,F,use_continuity=False))
from phykit.services.alignment.base import Alignment
A2=[Alignment(alignment_file_path=str(p)).calculate_rcv() for p in sorted(Path('work_rcv/animals').glob('*.mafft'))]
F2=[Alignment(alignment_file_path=str(p)).calculate_rcv() for p in sorted(Path('work_rcv/fungi').glob('*.mafft'))]
print('full new',mannwhitneyu(A2,F2), 'ties', len(A2)+len(F2)-len(set(A2+F2)))
print('full new no cont',mannwhitneyu(A2,F2,use_continuity=False))
PY"
