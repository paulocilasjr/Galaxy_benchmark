/bin/bash -lc "PYTHONPATH=/workspace/work_rcv/phykit203/phykit-2.0.3-py2.py3-none-any.whl python - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy.stats import mannwhitneyu
scores={}
for group in ['animals','fungi']:
 vals=[]
 for p in sorted(Path('work_rcv',group).glob('*.mafft')):
  vals.append(round(Alignment(alignment_file_path=str(p)).calculate_rcv(),4))
 scores[group]=vals
 saved=[float(x) for x in Path('work_rcv',group+'.rcv203.txt').read_text().split()]
 print(group,'processed',len(vals),'identical_to_reconstruction',vals==saved,'zero_scores',sum(x==0 for x in vals))
r=mannwhitneyu(scores['animals'],scores['fungi'],alternative='two-sided',method='asymptotic',use_continuity=True)
print('U',format(float(r.statistic),'.17g'))
print('p',format(float(r.pvalue),'.17g'))
PY"
