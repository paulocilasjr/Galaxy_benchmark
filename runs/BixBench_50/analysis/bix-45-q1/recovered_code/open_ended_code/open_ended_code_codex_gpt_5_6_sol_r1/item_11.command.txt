/bin/bash -lc "python - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
for group in ['animals','fungi']:
 vals=[]
 for p in sorted(Path('work_rcv',group).glob('*.mafft')):
  v=round(Alignment(alignment_file_path=str(p)).calculate_rcv(),4)
  vals.append(v)
 Path('work_rcv',group+'.rcv.txt').write_text('\\n'.join(map(str,vals))+'\\n')
 print(group, len(vals), min(vals), max(vals), sum(vals)/len(vals), vals[:5])
PY
python - <<'PY'
from pathlib import Path
from scipy.stats import mannwhitneyu
A=[float(x) for x in Path('work_rcv/animals.rcv.txt').read_text().split()]
F=[float(x) for x in Path('work_rcv/fungi.rcv.txt').read_text().split()]
for method in ['auto','asymptotic','exact']:
 for use_cont in [True,False]:
  r=mannwhitneyu(A,F,alternative='two-sided',method=method,use_continuity=use_cont)
  print(method,use_cont,r)
print('greater',mannwhitneyu(A,F,alternative='greater'))
print('less',mannwhitneyu(A,F,alternative='less'))
print('ties A,F,combined',len(A)-len(set(A)),len(F)-len(set(F)),len(A)+len(F)-len(set(A+F)))
PY"
