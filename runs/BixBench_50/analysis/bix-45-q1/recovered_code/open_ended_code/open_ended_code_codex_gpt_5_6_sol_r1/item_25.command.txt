/bin/bash -lc "PYTHONPATH=/workspace/work_rcv/phykit203/phykit-2.0.3-py2.py3-none-any.whl python - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy.stats import mannwhitneyu
D={}
for g in ['animals','fungi']:
 D[g]={p.name.removesuffix('.faa.mafft'):round(Alignment(alignment_file_path=str(p)).calculate_rcv(),4) for p in Path('work_rcv',g).glob('*.mafft')}
common=set(D['animals'])&set(D['fungi'])
print(len(common),len(set(D['animals'])-common),len(set(D['fungi'])-common))
print(mannwhitneyu([D['animals'][k] for k in common],[D['fungi'][k] for k in common]))
PY"
