import zipfile
from collections import defaultdict
for group,species,zp in [('animals',['Cele','Ggal','Mmus','Xlae'],'inputs/scogs_animals.zip'),('fungi',['Ncra','Scer','Spom','Umay'],'inputs/scogs_fungi.zip')]:
 ds={}
 for s in species:
  with zipfile.ZipFile(f'inputs/{\"Animals\" if group==\"animals\" else \"Fungi\"}_{s}.busco.zip') as z:
   n=[x for x in z.namelist() if x.endswith('/full_table.tsv')][0]
   st=defaultdict(list)
   for ln in z.read(n).decode().splitlines():
    if ln and not ln.startswith('#'):
     x=ln.split('\\t'); st[x[0]].append(x[1])
   ds[s]=st
 with zipfile.ZipFile(zp) as z: ids={x.split('/')[-1].split('.')[0] for x in z.namelist() if x.endswith('.faa.mafft')}
 allids=set(next(iter(ds.values())))
 elig={i for i in allids if any('Complete' in ds[s][i] for s in species)}
 single={i for i in allids if any(ds[s][i]==['Complete'] for s in species)}
 print(group,len(ids),len(elig),len(single),'ids not elig',sorted(elig-ids), 'ids all dup',sorted(allids-ids)[:30])
PY"
