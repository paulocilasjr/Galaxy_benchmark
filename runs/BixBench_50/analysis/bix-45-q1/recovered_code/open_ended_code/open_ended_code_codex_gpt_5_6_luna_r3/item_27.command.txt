/bin/bash -lc "python - <<'PY'
import zipfile
from Bio import AlignIO
from io import StringIO
for label,zp in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
    dist={}; ex={}
    with zipfile.ZipFile(zp) as z:
      for n in z.namelist():
       if n.endswith('.faa.mafft'):
        aln=AlignIO.read(StringIO(z.read(n).decode()),'fasta'); k=len(aln); dist[k]=dist.get(k,0)+1; ex.setdefault(k,[]).append(n)
    print(label,dist)
    for k in sorted(ex): print(' ',k,ex[k][:10])
PY"
