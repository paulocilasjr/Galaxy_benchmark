/bin/bash -lc ".venv/bin/python - <<'PY'
import sys
sys.path.insert(0,'work_rcv/packages/phykit-2.0.3-py2.py3-none-any.whl')
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy.stats import mannwhitneyu
import numpy as np, phykit
print('loaded',phykit.__file__)
vals={}
for group in ('animals','fungi'):
 full=[]
 for p in sorted(Path('work_rcv',group).glob('*.mafft')):
  full.append(Alignment(alignment_file_path=str(p)).calculate_rcv())
 vals[group]=full
 print(group,len(full),'mean_full',np.mean(full),'median_full',np.median(full),'zero',sum(v==0 for v in full))
for precision in ('full','cli-rounded'):
 x=vals['animals']; y=vals['fungi']
 if precision=='cli-rounded': x=[round(v,4) for v in x]; y=[round(v,4) for v in y]
 print(precision,'unique',len(set(x)),len(set(y)),'U_and_default_p',mannwhitneyu(x,y,alternative='two-sided'))
 print('asymptotic cc',mannwhitneyu(x,y,alternative='two-sided',method='asymptotic',use_continuity=True))
 print('asymptotic no_cc',mannwhitneyu(x,y,alternative='two-sided',method='asymptotic',use_continuity=False))
PY"
