/bin/bash -lc ".venv/bin/python - <<'PY'
import sys, subprocess
sys.path.insert(0,'work_rcv/packages/phykit-2.0.3-py2.py3-none-any.whl')
from pathlib import Path
from phykit.services.alignment.base import Alignment
vals=[]
for group in ('animals','fungi'):
 vals.append([round(Alignment(alignment_file_path=str(p)).calculate_rcv(),4) for p in sorted(Path('work_rcv',group).glob('*.mafft'))])
data=' '.join(map(str,vals[0]+vals[1]))
for correction in ('TRUE','FALSE'):
 code=f\"z<-scan(file('stdin'),quiet=TRUE); x<-z[1:241]; y<-z[242:496]; w<-wilcox.test(x,y,alternative='two.sided',exact=FALSE,correct={correction}); cat(unname(w"'$statistic),format(w$p.value,digits=17,scientific=TRUE),'"'\\\\n')\"
 print('R correct='+correction, subprocess.check_output(['Rscript','-e',code],input=data,text=True).strip())
PY"
