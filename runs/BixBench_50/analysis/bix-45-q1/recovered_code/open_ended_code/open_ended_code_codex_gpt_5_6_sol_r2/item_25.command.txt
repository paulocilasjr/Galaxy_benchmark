/bin/bash -lc ".venv/bin/python - <<'PY'
import sys
sys.path.insert(0,'work_rcv/packages/phykit-2.0.3-py2.py3-none-any.whl')
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy.stats import mannwhitneyu
scores=[]
for group in ('animals','fungi'):
    scores.append([round(Alignment(alignment_file_path=str(p)).calculate_rcv(), 4)
                   for p in sorted(Path('work_rcv',group).glob('*.faa.mafft'))])
result=mannwhitneyu(scores[0],scores[1],alternative='two-sided',method='asymptotic',use_continuity=True)
print(f'n_animals={len(scores[0])} n_fungi={len(scores[1])} U={result.statistic} p={result.pvalue:.16e}')
PY
mkdir -p codex_output"
