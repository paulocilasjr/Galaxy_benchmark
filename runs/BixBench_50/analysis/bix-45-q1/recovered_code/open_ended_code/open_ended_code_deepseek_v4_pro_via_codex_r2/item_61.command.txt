/bin/bash -lc "cd /workspace && python3 - <<'PY'
# examine old A >.25 subsets
import sys
sys.path.insert(0,'/workspace/phykit203')
from phykit.services.alignment.rcv import RelativeCompositionVariability
from argparse import Namespace
from pathlib import Path
from Bio import AlignIO
vals=[]
for p in sorted(Path('work_animals').glob('*.faa.mafft')):
 v=round(RelativeCompositionVariability(Namespace(alignment=str(p))).calculate_rcv(),4)
 n=len(AlignIO.read(str(p),'fasta'))
 vals.append((p.name,n,v))
for cond,label in [(lambda x:True,'all'),(lambda x:x[1]>1,'nseq>1'),(lambda x:x[2]>0,'rcv>0')]:
 sub=[x for x in vals if cond(x)]
 print(label,len(sub),sum(x[2]>.25 for x in sub),sum(x[2]>.25 for x in sub)/len(sub), 'med', sorted(v for _,_,v in sub)[len(sub)//2])

PY"
