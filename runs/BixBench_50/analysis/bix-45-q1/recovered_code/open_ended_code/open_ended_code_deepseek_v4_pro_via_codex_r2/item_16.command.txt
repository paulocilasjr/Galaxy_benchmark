/bin/bash -lc "cd /workspace && python3 - <<'PY'
from pathlib import Path
for d in ['work_animals','work_fungi']:
    files=sorted(Path(d).glob('*'))
    mafft=sorted(Path(d).glob('*.faa.mafft'))
    fa=sorted(Path(d).glob('*.faa'))
    # exclude .mafft from .faa
    unaligned=[p for p in fa if not str(p).endswith('.faa.mafft')]
    print(d, 'total',len(files),'mafft',len(mafft),'fa',len(unaligned))
    ids_mafft={p.name[:-len('.faa.mafft')] for p in mafft}
    ids_fa={p.name[:-len('.faa')] for p in unaligned}
    print('both',len(ids_mafft&ids_fa),'mafft only',len(ids_mafft-ids_fa),'fa only',len(ids_fa-ids_mafft))
    print('mafft-only samples', sorted(ids_mafft-ids_fa)[:20])
    print('fa-only samples', sorted(ids_fa-ids_mafft)[:20])
PY"
