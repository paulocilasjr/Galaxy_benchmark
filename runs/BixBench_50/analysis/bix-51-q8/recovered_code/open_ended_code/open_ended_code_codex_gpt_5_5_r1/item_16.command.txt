/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import math
ws=load_workbook('/workspace/inputs/data.xlsx', read_only=True, data_only=True).active
headers=list(next(ws.iter_rows(values_only=True)))
idx={h:i for i,h in enumerate(headers) if h is not None}
data=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    eff=row[idx['Efficacy']]
    age=row[idx['Age']]
    if eff is None or age is None or age == ' ': continue
    y=1 if eff=='PR' else 0
    data.append((float(age), y, eff))
print('n', len(data), 'responses', sum(y for _,y,_ in data), 'nonresponses', len(data)-sum(y for _,y,_ in data))
# Newton-Raphson for intercept + age beta
b0=0.0; b1=0.0
for it in range(100):
    g0=g1=0.0
    h00=h01=h11=0.0
    ll=0.0
    for x,y,_ in data:
        eta=b0+b1*x
        if eta>=0:
            p=1/(1+math.exp(-eta))
            ll += y*eta - (eta + math.log1p(math.exp(-eta)))
        else:
            p=math.exp(eta)/(1+math.exp(eta))
            ll += y*eta - math.log1p(math.exp(eta))
        w=p*(1-p)
        g0 += y-p
        g1 += (y-p)*x
        h00 -= w
        h01 -= w*x
        h11 -= w*x*x
    det=h00*h11-h01*h01
    step0=( h11*g0 - h01*g1)/det
    step1=(-h01*g0 + h00*g1)/det
    b0 -= step0
    b1 -= step1
    if max(abs(step0),abs(step1))<1e-13:
        break
print('iter',it+1,'b0',repr(b0),'b_age_response_PR',repr(b1),'ll',ll)
# reverse coding nonresponse=1
print('b_age_nonresponse',repr(-b1))
# grouped summaries
for yval in [1,0]:
    vals=[x for x,y,_ in data if y==yval]
    print(yval, len(vals), sum(vals)/len(vals))
PY"
