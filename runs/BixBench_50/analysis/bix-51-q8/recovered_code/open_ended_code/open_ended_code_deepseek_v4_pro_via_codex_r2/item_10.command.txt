/bin/bash -c "python - <<'PY'
import openpyxl
from collections import Counter
p='/workspace/inputs/data.xlsx'
wb=openpyxl.load_workbook(p, data_only=True)
ws=wb.active
rows=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None or r[1] is None: # sequence & group actual
        continue
    rows.append(r)
print('actual',len(rows))
for g in sorted(set(r[1] for r in rows)):
    sub=[r for r in rows if r[1]==g]
    print('Group',g,'n',len(sub),'efficacy',Counter(r[2] for r in sub),'age mean',sum(r[3] for r in sub)/len(sub))
    print('PR age mean',sum(r[3] for r in sub if r[2]=='PR')/sum(1 for r in sub if r[2]=='PR'))
    print('nonPR age mean',sum(r[3] for r in sub if r[2]"'!='"'PR')/sum(1 for r in sub if r[2]"'!='"'PR'))
# cross group/efficacy ages
for g in sorted(set(r[1] for r in rows)):
 for e in ['PR','SD','PD']:
  vals=[r[3] for r in rows if r[1]==g and r[2]==e]
  print(g,e,len(vals),sum(vals)/len(vals) if vals else None, vals)
PY"
