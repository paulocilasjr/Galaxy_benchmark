/bin/bash -c "python - <<'PY'
import openpyxl
from collections import Counter
p='/workspace/inputs/data.xlsx'
wb=openpyxl.load_workbook(p, data_only=True)
ws=wb.active
rows=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None or r[1] is None:
        continue
    rows.append(r)
for g in [1,2]:
 sub=[r for r in rows if r[1]==g]
 print('Group',g,'n',len(sub),'efficacy',Counter(r[2] for r in sub))
 for e in ['PR','SD','PD']:
  vals=[r[3] for r in sub if r[2]==e]
  print(e,len(vals), vals)
PY"
