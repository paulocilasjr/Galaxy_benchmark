/bin/bash -c "python - <<'PY'
import openpyxl, math
p='/workspace/inputs/data.xlsx'
ws=openpyxl.load_workbook(p, data_only=True).active
X=[]; y=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None or not isinstance(r[0],(int,float)): continue
    if r[1] is None or r[3] is None: continue
    age=float(r[3]); grp=r[1]
    y.append(1.0 if grp==1 else 0.0)
    X.append((1.0, age))
n=len(y)
beta=[0.0,0.0]
for it in range(100):
    p=[1/(1+math.exp(-(beta[0]+beta[1]*x[1]))) for x in X]
    grad=[sum((p[i]-y[i])*X[i][j] for i in range(n)) for j in range(2)]
    H=[[0.0,0.0],[0.0,0.0]]
    for i in range(n):
        w=p[i]*(1-p[i])
        for j in range(2):
            for k in range(2):
                H[j][k]+=w*X[i][j]*X[i][k]
    # solve 2x2
    det=H[0][0]*H[1][1]-H[0][1]*H[1][0]
    inv=[[H[1][1]/det,-H[0][1]/det],[-H[1][0]/det,H[0][0]/det]]
    step=[-(inv[0][0]*grad[0]+inv[0][1]*grad[1]), -(inv[1][0]*grad[0]+inv[1][1]*grad[1])]
    beta=[beta[j]+step[j] for j in range(2)]
    if max(abs(s) for s in step)<1e-12: break
print('n',n,'iter',it,'beta',beta)
print('age coef',beta[1])
# SE and wald
p=[1/(1+math.exp(-(beta[0]+beta[1]*x[1]))) for x in X]
H=[[0.0,0.0],[0.0,0.0]]
for i in range(n):
 w=p[i]*(1-p[i])
 for j in range(2):
  for k in range(2):
   H[j][k]+=w*X[i][j]*X[i][k]
det=H[0][0]*H[1][1]-H[0][1]*H[1][0]
inv=[[H[1][1]/det,-H[0][1]/det],[-H[1][0]/det,H[0][0]/det]]
import math as m
print('se',m.sqrt(inv[1][1]),'z',beta[1]/m.sqrt(inv[1][1]))
# log likelihoods
def ll(b):
 return sum(y[i]*(b[0]+b[1]*X[i][1]) - m.log(1+m.exp(b[0]+b[1]*X[i][1])) for i in range(n))
print('ll',ll(beta),'ll0',ll([math.log(sum(y)/(n-sum(y))),0]))
PY"
