/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
import math
wb=load_workbook('inputs/data.xlsx',read_only=True,data_only=True)
ws=wb.active
records=[]
for row in ws.iter_rows(min_row=2,values_only=True):
    seq,grp,eff,age=row[:4]
    if seq is None: continue
    records.append((seq,grp,eff,float(age)))
print('n',len(records))
print('group x efficacy',Counter((g,e) for _,g,e,_ in records))
print('missing age',sum(not math.isfinite(a) for *_,a in records))
for yname, pred in [('PR event',lambda g,e:g==1),('nonresponse event',lambda g,e:g==2)]:
    vals=[a for _,g,e,a in records if pred(g,e)]
    print(yname,'n',len(vals),'mean',sum(vals)/len(vals),'range',min(vals),max(vals))
PY"
