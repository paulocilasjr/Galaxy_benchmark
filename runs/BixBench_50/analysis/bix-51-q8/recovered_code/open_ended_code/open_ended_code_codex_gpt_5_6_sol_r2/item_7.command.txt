/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
wb=load_workbook('inputs/data.xlsx',read_only=True,data_only=True)
ws=wb.active
rows=list(ws.iter_rows(values_only=True))
headers=rows[0][:12]
print('headers:',headers)
for j,h in enumerate(headers):
 vals=[r[j] for r in rows[1:] if r[j] is not None]
 print('\\n',j,h,'n=',len(vals),'unique=',len(set(vals)))
 if len(set(vals))<=20: print(Counter(vals))
 else: print('min/max',min(vals),max(vals),'first uniques',sorted(set(vals))[:20])
print('\\nRows with any populated values after col 12:',[(i+1,r[12:]) for i,r in enumerate(rows) if any(x is not None for x in r[12:])][:5])
PY"
