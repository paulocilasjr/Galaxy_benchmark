import openpyxl
import statsmodels.api as sm
import numpy as np

# Load data
wb = openpyxl.load_workbook('/workspace/inputs/data.xlsx')
ws = wb.active

# Extract data
data = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    group = row[1]
    efficacy = row[2]
    age = row[3]
    if group is not None and age is not None:
        data.append({
            'group': group,
            'efficacy': efficacy,
            'age': age
        })

print(f"Total patients with data: {len(data)}")

# Create binary outcome: Group 1 = responder, Group 2 = non-responder
# Or equivalently: efficacy PR = 1, PD/SD = 0
y = np.array([1 if d['group'] == 1 else 0 for d in data])
X = np.array([d['age'] for d in data])

print(f"Responders (Group=1): {sum(y)}")
print(f"Non-responders (Group=2): {len(y) - sum(y)}")
print(f"Age range: {X.min()} - {X.max()}")
print(f"Mean age responders: {X[y==1].mean():.2f}")
print(f"Mean age non-responders: {X[y==0].mean():.2f}")

# Add intercept
X_with_intercept = sm.add_constant(X)

# Fit logistic regression
model = sm.Logit(y, X_with_intercept)
result = model.fit(disp=False)

print("\nLogistic Regression Results:")
print(result.summary())

print(f"\nCoefficient for Age: {result.params[1]:.10f}")
print(f"Standard error: {result.bse[1]:.10f}")
print(f"p-value: {result.pvalues[1]:.10f}")
print(f"95% CI: [{result.conf_int().iloc[1, 0]:.10f}, {result.conf_int().iloc[1, 1]:.10f}]")
