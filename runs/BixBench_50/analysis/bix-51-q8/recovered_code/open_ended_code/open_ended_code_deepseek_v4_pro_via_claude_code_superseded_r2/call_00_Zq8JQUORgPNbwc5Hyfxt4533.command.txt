import openpyxl
wb = openpyxl.load_workbook('/workspace/inputs/data.xlsx')
ws = wb.active

# Get unique values for key columns
groups = set()
efficacy = set()
ages = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    groups.add(row[1])
    efficacy.add(row[2])
    ages.append(row[3])

print('Unique Groups:', sorted(groups, key=str))
print('Unique Efficacy:', sorted(efficacy, key=str))
print('Age range:', min(ages), '-', max(ages))
print('Total rows:', len(ages))
print()

# Count by group and efficacy
from collections import Counter
group_counts = Counter()
eff_counts = Counter()
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    group_counts[row[1]] += 1
    eff_counts[row[2]] += 1
print('Group counts:', dict(group_counts))
print('Efficacy counts:', dict(eff_counts))

# Cross-tab
cross = {}
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    key = (row[1], row[2])
    cross[key] = cross.get(key, 0) + 1
print('Cross-tab (Group x Efficacy):')
for k, v in sorted(cross.items()):
    print(f'  {k}: {v}')
