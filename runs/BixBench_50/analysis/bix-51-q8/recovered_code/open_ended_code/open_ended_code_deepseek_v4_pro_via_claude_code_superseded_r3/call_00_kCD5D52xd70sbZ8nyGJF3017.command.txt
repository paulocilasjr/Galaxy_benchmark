import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_excel('inputs/data.xlsx')
df['Age_num'] = pd.to_numeric(df['Age'], errors='coerce')
df['Response'] = df['Efficacy'].map({'PR': 1, 'SD': 0, 'PD': 0})
complete = df[df['Age_num'].notna() & df['Response'].notna()].copy()

# Also try with Group 1 only (camrelizumab group)
group1 = complete[complete['Groups'] == 1.0]
print(f'Group 1 patients: {len(group1)}')
print(f'Group 1 response distribution: {group1[\"Response\"].value_counts().to_dict()}')

# What about using Group as a covariate?
X2 = complete[['Age_num', 'Groups']]
X2 = sm.add_constant(X2)
y2 = complete['Response']
model2 = sm.Logit(y2, X2)
try:
    result2 = model2.fit(disp=0)
    print()
    print('Model with Age + Group:')
    print(result2.params)
except Exception as e:
    print(f'Model with Age + Group failed: {e}')

# Since Age alone works with the full dataset, let me also verify
# the univariate model more carefully
X1 = sm.add_constant(complete[['Age_num']])
y1 = complete['Response']
model1 = sm.Logit(y1, X1)
result1 = model1.fit(disp=0)
print()
print('Univariate model (Age only):')
print(f'Age coefficient: {result1.params[\"Age_num\"]:.10f}')
print(f'Age std err: {result1.bse[\"Age_num\"]:.10f}')
print(f'Age p-value: {result1.pvalues[\"Age_num\"]:.10f}')
print(f'Age 95% CI: [{result1.conf_int().loc[\"Age_num\", 0]:.10f}, {result1.conf_int().loc[\"Age_num\", 1]:.10f}]')
