/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from collections import Counter
p='inputs/data.xlsx'
ns={'m':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
with ZipFile(p) as z:
    ss=ET.fromstring(z.read('xl/sharedStrings.xml'))
    strings=[]
    for si in ss.findall('m:si',ns):
        strings.append(''.join(t.text or '' for t in si.iter('{%s}t'%ns['m'])))
    root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
    rows=[]
    for row in root.findall('.//m:sheetData/m:row',ns):
        vals={}
        for c in row.findall('m:c',ns):
            ref=c.attrib['r']
            col=''.join(x for x in ref if x.isalpha())
            v=c.find('m:v',ns)
            if v is None: val=''
            elif c.attrib.get('t')=='s': val=strings[int(v.text)]
            else: val=v.text
            vals[col]=val
        rows.append((int(row.attrib['r']),vals))
    cols=[]
    for c in rows[0][1]: cols.append(c)
    headers={c:rows[0][1][c] for c in cols}
    print('HEADERS',headers)
    print('ROW_COUNT_INCL_HEADER',len(rows))
    for rn,d in rows[1:]:
        print(rn, {headers.get(c,c):d.get(c,'') for c in cols})
    print('\\nCOLUMN SUMMARIES')
    for c in cols:
        name=headers[c]
        vv=[d.get(c,'') for _,d in rows[1:]]
        print(name, 'n=',len(vv),'missing=',sum(x=='' or x.strip()=='' for x in vv),'unique=',Counter(vv))
PY"
