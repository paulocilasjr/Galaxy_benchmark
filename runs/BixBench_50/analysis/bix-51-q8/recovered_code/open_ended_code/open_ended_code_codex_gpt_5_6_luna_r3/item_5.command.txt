/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from pathlib import Path
p=Path('/workspace/inputs/data.xlsx')
wb=load_workbook(p, data_only=False, read_only=False)
print('sheets:', wb.sheetnames)
for ws in wb.worksheets:
    print('\\nSHEET', ws.title, 'state=', ws.sheet_state, 'dims=', ws.max_row, ws.max_column)
    for row in ws.iter_rows(min_row=1, max_row=min(ws.max_row,12), values_only=True):
        print(row)
PY"
