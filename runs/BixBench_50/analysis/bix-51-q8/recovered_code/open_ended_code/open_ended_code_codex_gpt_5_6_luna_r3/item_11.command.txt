/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from math import exp,log
ws=load_workbook('/workspace/inputs/data.xlsx',data_only=True,read_only=True).active
rows=list(ws.iter_rows(min_row=2,max_row=81,max_col=4,values_only=True))
x=[float(r[3]) for r in rows]

def sig(z):
    if z>=0:
        e=exp(-z); return 1/(1+e)
    e=exp(z); return e/(1+e)
def coef(y):
    b0=b1=0.0
    for _ in range(100):
        s0=s1=a00=a01=a11=0.0
        for xi,yi in zip(x,y):
            p=sig(b0+b1*xi); q=p*(1-p); z=yi-p
            s0+=z;s1+=z*xi;a00+=q;a01+=q*xi;a11+=q*xi*xi
        det=a00*a11-a01*a01
        d0=(a11*s0-a01*s1)/det; d1=(-a01*s0+a00*s1)/det
        b0+=d0;b1+=d1
        if max(abs(d0),abs(d1))<1e-13: return b0,b1
    return b0,b1
for name,y in [
 ('PR vs SD/PD',[1 if r[2]=='PR' else 0 for r in rows]),
 ('non-PD vs PD',[1 if r[2] in ('PR','SD') else 0 for r in rows]),
 ('PD vs non-PD',[1 if r[2]=='PD' else 0 for r in rows]),
 ('group2 vs group1',[1 if r[1]==2 else 0 for r in rows]),
 ('group1 vs group2',[1 if r[1]==1 else 0 for r in rows]),
]:
    print(name, 'events',sum(y), 'beta0 beta_age',tuple(repr(z) for z in coef(y)))
PY"
