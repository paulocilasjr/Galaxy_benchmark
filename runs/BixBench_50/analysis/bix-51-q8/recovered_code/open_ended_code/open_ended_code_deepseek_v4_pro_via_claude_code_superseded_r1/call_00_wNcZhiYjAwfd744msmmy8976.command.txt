import pandas as pd
import statsmodels.api as sm

df = pd.read_excel('inputs/data.xlsx')

# Keep rows with actual data (80 rows)
data = df[df['Sequence number'].notna()].copy()

# Clean Age - it's stored as string, convert to float
data['Age'] = pd.to_numeric(data['Age'], errors='coerce')

# No missing values in the 80 rows - verify
print('Missing:', data[['Age', 'Efficacy']].isna().sum())
print('Rows:', len(data))

# Create binary response: PR = responder (1), SD/PD = non-responder (0)
data['Response'] = (data['Efficacy'] == 'PR').astype(int)

# Fit simple logistic regression: Response ~ Age
X = sm.add_constant(data['Age'])
y = data['Response']

model = sm.Logit(y, X)
result = model.fit(disp=False)

print(result.summary())
print()
print('Parameters:')
print(result.params)
print()
print('Coefficient for Age:', result.params['Age'])
