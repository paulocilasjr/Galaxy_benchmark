import pandas as pd
from sklearn.linear_model import LogisticRegression
import statsmodels.api as sm

df = pd.read_excel('inputs/data.xlsx')
data = df[df['Sequence number'].notna()].copy()
data['Age'] = pd.to_numeric(data['Age'], errors='coerce')
data['Response'] = (data['Efficacy'] == 'PR').astype(int)

X = data[['Age']].values
y = data['Response'].values

# sklearn with no penalty
model = LogisticRegression(penalty=None, solver='lbfgs', max_iter=1000)
model.fit(X, y)
print('sklearn:')
print(f'Intercept: {model.intercept_[0]:.16f}')
print(f'Age coef:  {model.coef_[0][0]:.16f}')

# statsmodels newton
X_sm = sm.add_constant(data['Age'])
result = sm.Logit(y, X_sm).fit(method='newton', disp=False)
print()
print('statsmodels:')
print(f'Intercept: {result.params[\"const\"]:.16f}')
print(f'Age coef:  {result.params[\"Age\"]:.16f}')
