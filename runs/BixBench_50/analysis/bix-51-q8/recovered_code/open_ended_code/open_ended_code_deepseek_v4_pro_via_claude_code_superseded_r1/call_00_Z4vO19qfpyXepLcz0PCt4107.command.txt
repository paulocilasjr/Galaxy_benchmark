import pandas as pd
import statsmodels.api as sm

df = pd.read_excel('inputs/data.xlsx')
data = df[df['Sequence number'].notna()].copy()
data['Age'] = pd.to_numeric(data['Age'], errors='coerce')
data['Response'] = (data['Efficacy'] == 'PR').astype(int)

X = sm.add_constant(data['Age'])
y = data['Response']

result = sm.Logit(y, X).fit(disp=False)

# Verify everything looks right
print(f'N = {len(data)}')
print(f'Age coefficient: {result.params[\"Age\"]:.16f}')
print(f'Std error: {result.bse[\"Age\"]:.6f}')
print(f'P-value: {result.pvalues[\"Age\"]:.6f}')
print(f'95% CI: [{result.conf_int().loc[\"Age\", 0]:.6f}, {result.conf_int().loc[\"Age\", 1]:.6f}]')
print(f'Converged: {result.mle_retvals[\"converged\"]}')
