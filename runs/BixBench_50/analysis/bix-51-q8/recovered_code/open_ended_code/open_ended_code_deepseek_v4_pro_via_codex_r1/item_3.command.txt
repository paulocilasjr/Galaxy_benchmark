/bin/bash -c "python3 - <<'PY'
import pandas as pd
p='/workspace/inputs/data.xlsx'
df=pd.read_excel(p)
print(df.shape)
print(df.columns.tolist())
for c in df.columns:
    print('\\nCOL', c, 'dtype', df[c].dtype)
    print('nunique', df[c].nunique(dropna=True), 'missing', df[c].isna().sum())
    if df[c].nunique(dropna=True)<=20:
        print(df[c].value_counts(dropna=False).to_string())
    else:
        print(df[c].describe().to_string())
print('\\nTail')
print(df.tail(20).to_string())
PY"
