/bin/bash -c "python3 - <<'PY'
import openpyxl, numpy as np, statsmodels.api as sm
p='/workspace/inputs/data.xlsx'
wb=openpyxl.load_workbook(p,data_only=True,read_only=True)
ws=wb.active
rows=list(ws.iter_rows(values_only=True))
h=rows[0]
data=[]
for r in rows[1:]:
    if any(v is not None and str(v).strip()"'!='"'' for v in r):
        d=dict(zip(h,r))
        if d['Groups'] is not None and d['Efficacy'] is not None and isinstance(d['Age'],(int,float)):
            data.append(d)
print('n',len(data), 'minmax age',min(d['Age'] for d in data),max(d['Age'] for d in data))
# y variants
age=np.array([d['Age'] for d in data],float)
ys={}
ys['PR_vs_other']=np.array([1 if d['Efficacy']=='PR' else 0 for d in data])
ys['group1']=np.array([1 if d['Groups']==1 else 0 for d in data])
ys['PR_or_SD?']=np.array([1 if d['Efficacy'] in ('PR','SD') else 0 for d in data]) # unlikely
for name,y in ys.items():
    X=sm.add_constant(age)
    try:
        model=sm.Logit(y,X).fit(disp=0)
        print('\\n',name,'n y1',sum(y))
        print(model.params)
        print(model.bse)
        print('coef age', model.params['x1'] if 'x1' in model.params else model.params.iloc[1])
        print('llf',model.llf)
    except Exception as e:
        print(name,e)
# maybe scipy exact
PY"
