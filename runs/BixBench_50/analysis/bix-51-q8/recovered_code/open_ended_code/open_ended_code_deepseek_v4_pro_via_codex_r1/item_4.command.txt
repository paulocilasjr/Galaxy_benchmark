/bin/bash -c "python3 - <<'PY'
import openpyxl, collections
p='/workspace/inputs/data.xlsx'
wb=openpyxl.load_workbook(p, data_only=True, read_only=True)
ws=wb.active
rows=list(ws.iter_rows(values_only=True))
header=rows[0]
data=[dict(zip(header,r)) for r in rows[1:] if any(v is not None for v in r)]
print('n data',len(data))
# remove Nones keys? header has None extras
cols=[h for h in header if h is not None]
print('cols', cols)
for c in cols:
    vals=[d[c] for d in data]
    print('\\n',c,'missing',sum(v is None for v in vals))
    cnt=collections.Counter(vals)
    if len(cnt)<=20:
        for k,v in sorted(cnt.items(), key=lambda kv: (kv[0] is None, str(kv[0]))):
            print('  ',repr(k),v)
    else:
        nums=[v for v in vals if isinstance(v,(int,float))]
        print(' min',min(nums),'max',max(nums),'mean',sum(nums)/len(nums), 'nonnum',sum(not isinstance(v,(int,float)) for v in vals))
# Print groups and efficacy by group
print('\\nGroups x Efficacy')
ct=collections.Counter((d['Groups'],d['Efficacy']) for d in data)
for k in sorted(ct, key=str): print(k,ct[k])
print('\\nAll rows compact')
for i,d in enumerate(data):
    print(i+1, [d[c] for c in cols])
PY"
