/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import math
ws=load_workbook('/workspace/inputs/data.xlsx',read_only=True,data_only=True).active
x=[]; y=[]
for row in ws.iter_rows(min_row=2,values_only=True):
    if row[0] is not None and isinstance(row[3],(int,float)):
        x.append(float(row[3])); y.append(1 if row[2]=='PR' else 0)
# Independently solve the score equations with Newton iterations.
b0=b1=0.0
for _ in range(100):
    g0=g1=h00=h01=h11=0.0
    for xi,yi in zip(x,y):
        z=b0+b1*xi
        p=1.0/(1.0+math.exp(-z))
        w=p*(1-p)
        g0 += yi-p; g1 += xi*(yi-p)
        h00 -= w; h01 -= xi*w; h11 -= xi*xi*w
    det=h00*h11-h01*h01
    d0=(-g0*h11+h01*g1)/det
    d1=(h01*g0-h00*g1)/det
    b0 += d0; b1 += d1
    if max(abs(d0),abs(d1)) < 1e-15:
        break
score0=score1=0.0
for xi,yi in zip(x,y):
    p=1.0/(1.0+math.exp(-(b0+b1*xi)))
    score0 += yi-p; score1 += xi*(yi-p)
print('n',len(x),'events',sum(y),'iterations',_+1)
print('intercept %.17g slope %.17g' % (b0,b1))
print('scores %.3g %.3g' % (score0,score1))
PY"
