import openpyxl

# Look at variant details for one carrier file
wb = openpyxl.load_workbook('/workspace/inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_185-PF.xlsx')
ws = wb[wb.sheetnames[0]]

print("Column headers (row 2):")
headers = []
for i, cell in enumerate(ws[2]):
    headers.append(cell.value)
    print(f"  Col {i}: {cell.value}")

# Now look at sample data - unique values for key columns
zygosity_vals = set()
so_vals = set()
vaf_vals = []
clinvar_vals = set()
in_chip_vals = set()

for row in ws.iter_rows(min_row=3, values_only=True):
    if row[4] is not None:
        zygosity_vals.add(row[4])
    if row[13] is not None:
        so_vals.add(row[13])
    if row[9] is not None:
        vaf_vals.append(row[9])
    if row[25] is not None:
        clinvar_vals.add(row[25])
    if row[29] is not None:
        in_chip_vals.add(str(row[29]))

print(f"\nZygosity values: {zygosity_vals}")
print(f"\nVAF sample (first 20 non-None): {[x for x in vaf_vals if x is not None][:20]}")
print(f"\nVAF type: {type(vaf_vals[0]) if vaf_vals else 'N/A'}")
print(f"\nSequence Ontology values: {sorted(so_vals)[:30]}")
print(f"\nIn_CHIP values: {in_chip_vals}")
print(f"\nClinVar Classification values (sample): {list(clinvar_vals)[:20]}")
