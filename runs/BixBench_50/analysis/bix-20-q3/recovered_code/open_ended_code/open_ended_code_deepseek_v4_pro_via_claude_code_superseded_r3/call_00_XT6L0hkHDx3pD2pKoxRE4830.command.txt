import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/230215_Trio_Status.xlsx')
ws = wb['Sheet1']

carriers = []
for row in ws.iter_rows(values_only=True):
    if row[3] == 'Carrier':
        carriers.append(row[0])

print("Carrier samples:")
for c in carriers:
    print(f"  {c}")
print(f"\nTotal carriers: {len(carriers)}")
