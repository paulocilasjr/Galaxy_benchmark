import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_185-PF.xlsx')
ws = wb[wb.sheetnames[0]]

# Look at examples of each ClinVar classification
for clinvar_val in ['Other', 'Conflicting']:
    print(f"\n=== ClinVar Classification: {clinvar_val} ===")
    count = 0
    for row in ws.iter_rows(min_row=3, values_only=True):
        if str(row[25]) == clinvar_val:
            print(f"  Gene: {row[12]}, SO: {row[13]}, Classification: {row[25]}, Aggregate: {row[26]}, VAF: {row[9]}")
            count += 1
            if count >= 3:
                break

# Also check if there are rows with None classification
null_count = 0
for row in ws.iter_rows(min_row=3, values_only=True):
    if row[25] is None:
        null_count += 1
print(f"\nRows with None ClinVar classification: {null_count}")
