/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter
from openpyxl import load_workbook
base=Path('inputs')
# gene list
wb=load_workbook(base/'230214_Schenz_et_al_2022_CHIP_Genes.xlsx',read_only=True,data_only=True); ws=wb.active; genes={str(row[0]).strip().upper() for row in ws.iter_rows(values_only=True) if row[0]}
# statuses
wb=load_workbook(base/'230215_Trio_Status.xlsx',read_only=True,data_only=True); ws=wb.active; carriers={str(row[0]).strip() for row in ws.iter_rows(min_row=2,values_only=True) if row[0] and str(row[3]).strip()=='Carrier'}
def fid(p):
 t=p.stem.split('_CHIP_',1)[1]; return t.split('-',1)[0] if t.split('-',1)[0].isdigit() else t

def rows(p):
 wb=load_workbook(p,read_only=True,data_only=True);ws=wb.active;ws.reset_dimensions();it=ws.iter_rows(values_only=True);next(it);hdr=next(it);idx={x:i for i,x in enumerate(hdr) if x}
 for row in it: yield {k:(row[i] if i<len(row) else None) for k,i in idx.items()}

def gene_in(g):
 if not g:return False
 # exact any gene symbol among comma-separated names, stripping transcript suffix aliases
 return any(x.strip().upper().split('-')[0] in genes for x in str(g).split(','))
# print counts under exact list, match CHIP flags, all
for scope in ['all','list','flag']:
 c=Counter(); bysample=Counter()
 for p in sorted((base/'CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
  if fid(p) not in carriers: continue
  for r in rows(p):
   if scope=='list' and not gene_in(r['Gene Names']): continue
   if scope=='flag' and str(r['In_CHIP']).strip().lower()"'!='"'true': continue
   c['rows']+=1
   try: v=float(r['Variant Allele Freq'])
   except: continue
   if v>=.3: continue
   c['vaf']+=1
   if r['Zygosity']=='Reference':continue
   c['nonref']+=1
   so=str(r['Sequence Ontology (Combined)'] or '').lower()
   if any(term in so for term in ('intron','intergenic','utr')): continue
   c['region']+=1
   cl=r['Classification']
   if cl is None or str(cl).strip()=='':continue
   c['classified']+=1
   if cl in ('Benign','Likely Benign'):c['blb']+=1
   else:c['other']+=1
   bysample[fid(p)]+=1
 print(scope,c,'classified genes by sample',dict(bysample))
PY"
