/bin/bash -c "python3 - <<'PY'
import os,glob
from openpyxl import load_workbook
from collections import Counter,defaultdict
# status carrier
wb=load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True); ws=wb.active
status={}
for row in ws.iter_rows(min_row=2,values_only=True):
    if row[0] is None: continue
    status[str(row[0]).rstrip('\\t').strip()]=row
wb.close()
carrier=set(k for k,r in status.items() if r[3]=='Carrier')
files=glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')
rows=[]
for f in files:
    sample=os.path.basename(f)[:-5].split('_CHIP_',1)[1]
    core=sample if sample.startswith('SRR') else sample.split('-')[0]
    if core not in carrier: continue
    wb=load_workbook(f,data_only=True); ws=wb.active
    for r in ws.iter_rows(min_row=3,values_only=True):
        chrpos=r[0]; refalt=r[1]; zyg=r[4]; vaf_raw=r[9]; gene=r[12]; seq=r[13]; cls=r[25]
        if zyg=='Reference': continue
        seq_s=str(seq or '').lower()
        if any(t in seq_s for t in ('intron','intergenic','utr')): continue
        if vaf_raw is None: continue
        try: vaf=float(str(vaf_raw).strip())
        except: continue
        if vaf<0.3 and str(cls or '').strip()"'!='"'':
            rows.append((sample,chrpos,refalt,zyg,vaf_raw,gene,seq,cls,str(r[26]),str(r[27])))
    wb.close()
print('count',len(rows))
print('benign count',sum(1 for r in rows if r[7] in ('Benign','Likely Benign')))
for i,r in enumerate(rows,1):
    print(i,r)
PY"
