/bin/bash -c "python3 - <<'PY'
# compute unique variant id among filtered rows
import os,glob
from openpyxl import load_workbook
wb=load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True);ws=wb.active
status={}
for r in ws.iter_rows(min_row=2,values_only=True):
 if r[0] is not None: status[str(r[0]).rstrip('\\t').strip()]=r
wb.close(); carrier=set(k for k,r in status.items() if r[3]=='Carrier')
ids=[]
class_by_id={}
for f in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'):
 sample=os.path.basename(f)[:-5].split('_CHIP_',1)[1]
 core=sample if sample.startswith('SRR') else sample.split('-')[0]
 if core not in carrier: continue
 wb=load_workbook(f,data_only=True);ws=wb.active
 for r in ws.iter_rows(min_row=3,values_only=True):
  zyg=r[4]; seq=str(r[13] or '').lower(); v=r[9]; cls=r[25]; vid=r[24]
  if zyg=='Reference': continue
  if any(t in seq for t in ('intron','intergenic','utr')): continue
  if v is None: continue
  try: val=float(str(v).strip())
  except: continue
  if val<0.3 and str(cls or '').strip()"'!='"'':
   ids.append(vid); class_by_id.setdefault(vid,set()).add(cls)
 wb.close()
print('rows',len(ids),'unique variant ids',len(set(ids)))
from collections import Counter
print('unique class sets',Counter(tuple(sorted(s)) for s in class_by_id.values()))
print('all unique B/LB', all(set(s)<= {'Benign','Likely Benign'} for s in class_by_id.values()))
PY"
