/bin/bash -c "python - <<'PY'
import openpyxl,glob,os,re,collections
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True); ws=wb.active
carriers=set(str(r[0]).strip() for r in ws.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier' and r[0] is not None)
cnt=collections.Counter(); cls_by_so=collections.defaultdict(collections.Counter)
for p in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'):
 b=os.path.basename(p); m=re.search(r'CHIP_(\\d+)-',b)
 if not m or m.group(1) not in carriers: continue
 wb=openpyxl.load_workbook(p,data_only=True); wsv=wb.active
 for r in wsv.iter_rows(min_row=3,values_only=True):
  if r[4]=='Reference': continue
  if r[9] is None: continue
  try: v=float(r[9])
  except: continue
  if v>=0.3: continue
  cls=r[25]
  if cls is None or str(cls).strip()=='': continue
  so=str(r[13])
  cnt[so]+=1; cls_by_so[so][str(cls)]+=1
print('total',sum(cnt.values()))
for so,c in cnt.most_common(): print(so,c,dict(cls_by_so[so]))
PY"
