/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter, defaultdict
import openpyxl

# status
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True,read_only=False)
ws=wb.active
carrier=[]
for r in ws.iter_rows(min_row=2,values_only=True):
    sid=str(r[0]).strip() if r[0] is not None else ''
    if r[3]=='Carrier': carrier.append(sid)
wb.close()
print('carrier IDs:',carrier,'n=',len(carrier))
files=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))
byid=defaultdict(list)
for p in files:
    sample=p.stem.split('_CHIP_',1)[1].split('-',1)[0]
    byid[sample].append(p)
print('n variant files',len(files))
print('carrier files:',[(x, len(byid.get(x,[])), [p.name for p in byid.get(x,[])]) for x in carrier])

all_counts=defaultdict(Counter); carrier_counts=defaultdict(Counter)
scalar_all=Counter(); scalar_car=Counter()
carrier_rows=[]
for p in files:
    sid=p.stem.split('_CHIP_',1)[1].split('-',1)[0]
    wb=openpyxl.load_workbook(p,data_only=True,read_only=False)
    ws=wb.active
    vals=list(ws.iter_rows(values_only=True))
    headers=list(vals[1])
    d={h:i for i,h in enumerate(headers) if h is not None}
    dat=vals[2:]
    c=carrier_counts if sid in carrier else all_counts
    sc=scalar_car if sid in carrier else scalar_all
    sc['files']+=1; sc['rows'] += len(dat)
    sc['nonempty_vaf'] += sum(r[d['Variant Allele Freq']] is not None for r in dat)
    sc['inchip_true'] += sum(str(r[d['In_CHIP']]).strip().lower()=='true' for r in dat)
    sc['zygosity_ref'] += sum(str(r[d['Zygosity']]).strip().lower()=='reference' for r in dat)
    sc['clinvar_nonnull'] += sum(r[d['Classification']] not in (None,'') for r in dat)
    for key in ['Zygosity','Sequence Ontology (Combined)','Effect (Combined)','Classification','In_CHIP','Filter']:
        c[key].update([r[d[key]] for r in dat])
    if sid in carrier:
        carrier_rows.extend((sid,p.name,r,d) for r in dat)
    wb.close()
print('\\nALL scalars',scalar_all)
print('CARRIER scalars',scalar_car)
for label,c in [('all',all_counts),('carrier',carrier_counts)]:
    print('\\n',label,'ontology',c['Sequence Ontology (Combined)'])
    print(label,'classification',c['Classification'])
    print(label,'zygosity',c['Zygosity'])
    print(label,'inchip',c['In_CHIP'])
    print(label,'filter',c['Filter'])
PY"
