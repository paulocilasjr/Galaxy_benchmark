from collections import Counter, defaultdict
import openpyxl, math, re

region_exclude_exact={'intron_variant','intergenic_variant','3_prime_UTR_variant','5_prime_UTR_variant','5_prime_UTR_premature_start_codon_gain_variant'}

def num(x):
    if isinstance(x,(int,float)) and not isinstance(x,bool): return float(x)
    try: return float(str(x).strip())
    except: return None

for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):

        print(sid,nn,len(rr),f'{nn/len(rr):.6g}')
