from pathlib import Path
from collections import Counter,defaultdict
import openpyxl,re
# Metadata
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',read_only=True,data_only=True)
rows=list(wb.active.iter_rows(values_only=True)); hdr=rows[0]

files={re.search(r'CHIP_(["'^.]*)'"\\.xlsx"'$'"',p.name).group(1).split('-')[0]:p for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')}

print('total rows',len(allcalls))
for col in ['Zygosity','Sequence Ontology (Combined)','Effect (Combined)','Classification','In_CHIP','Variant Allele Freq','Filter']:
