ws=wb.active
rows=list(ws.iter_rows(values_only=True))
for r in rows[1:]:
    sid=str(r[0]).strip()

carrier=set(k for k,v in meta.items() if v[1]=='Carrier')
print('carrier metadata', len(carrier), sorted(carrier))

files=sorted(glob.glob('/workspace/inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))
carr_files=[]
for f in files:
    base=os.path.basename(f)
    m=re.search(r'CHIP_(.+?)\\.xlsx"'$'"', base)
    sample=m.group(1) if m else base
    if sample in carrier:
        carr_files.append((sample,f))
print('carrier files',len(carr_files),[s for s,f in carr_files])

for sample,f in carr_files:

        d={hdr[i]:r[i] for i in range(min(len(hdr),len(r))) if hdr[i] is not None}

                    vals[z]+=1
                except Exception as e: pass
