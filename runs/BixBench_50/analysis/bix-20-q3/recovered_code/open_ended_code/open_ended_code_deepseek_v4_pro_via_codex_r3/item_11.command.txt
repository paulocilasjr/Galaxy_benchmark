/bin/bash -c "python - <<'PY'
import openpyxl, glob, os, re, collections
files=glob.glob('/workspace/inputs/CHIP_DP10_GQ20_PASS/*.xlsx')
allso=collections.Counter(); alleffect=collections.Counter(); allclass=collections.Counter(); allzyg=collections.Counter()
for f in files:
    wb=openpyxl.load_workbook(f, read_only=True, data_only=True)
    ws=wb.active
    rows=ws.iter_rows(values_only=True)
    header=None
    for i,r in enumerate(rows):
        if i==1:
            header=r
        elif i>1 and header:
            d={header[j]:r[j] for j in range(min(len(header),len(r))) if header[j] is not None}
            allso[d.get('Sequence Ontology (Combined)')]+=1
            alleffect[d.get('Effect (Combined)')]+=1
            allclass[d.get('Classification')]+=1
            allzyg[d.get('Zygosity')]+=1
print('SO values all files')
for k,v in allso.most_common(): print(repr(k),v)
print('Effect all')
for k,v in alleffect.most_common(): print(repr(k),v)
print('Class all')
for k,v in allclass.most_common(): print(repr(k),v)
print('Zyg all')
for k,v in allzyg.most_common(): print(repr(k),v)
PY"
