import pandas as pd
import os

trio = pd.read_excel("inputs/230215_Trio_Status.xlsx")
carriers = trio[trio['BLM Mutation Status'] == 'Carrier']

# Map sample IDs to files
variant_dir = "inputs/CHIP_DP10_GQ20_PASS"
files = os.listdir(variant_dir)
print("Available variant files:")
for f in sorted(files):
    print(f"  {f}")

# Check which carrier samples have files
print("\n--- Carrier sample to file mapping ---")
for _, row in carriers.iterrows():
    sid = row['Sample ID']
    status = row['Status']
    # The suffix is based on Status: Father→PF, Mother→PM, Patient→P
    # But we need to check actual file naming
    matching = [f for f in files if f'CHIP_{sid}-' in f]
    if matching:
        print(f"  {sid} ({status}): {matching[0]}")
    else:
        print(f"  {sid} ({status}): NO FILE FOUND")
