import pandas as pd
import os

# Check In_CHIP values and sequence ontology across a few Carrier files
carrier_files = [
    "230209_Exome_GRCh38_CHIP_185-PF.xlsx",
    "230209_Exome_GRCh38_CHIP_184-PM.xlsx",
    "230209_Exome_GRCh38_CHIP_381-PM.xlsx",
]

variant_dir = "inputs/CHIP_DP10_GQ20_PASS"

for fname in carrier_files:
    fpath = os.path.join(variant_dir, fname)
    df = pd.read_excel(fpath, header=None)
    data = df.iloc[2:].copy()
    
    print(f"\n=== {fname} ===")
    print(f"Total rows: {len(data)}")
    print(f"In_CHIP (Col 29) values: {data[29].value_counts().to_dict()}")
    
    # Sequence ontology
    seq_terms = data[13].value_counts()
    print(f"Sequence Ontology terms: {dict(seq_terms)}")
    
    # Check Zygosity
    zyg = data[4].value_counts()
    print(f"Zygosity: {dict(zyg)}")
    
    # ClinVar classification
    clinvar = data[25].value_counts(dropna=False)
    print(f"ClinVar Classification: {dict(clinvar)}")
