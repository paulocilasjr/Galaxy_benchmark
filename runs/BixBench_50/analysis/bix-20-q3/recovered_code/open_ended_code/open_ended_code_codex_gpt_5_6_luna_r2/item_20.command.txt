from pathlib import Path
from collections import Counter

# carriers
wb=load_workbook(base/'230215_Trio_Status.xlsx',read_only=True,data_only=True);ws=wb.active;ws.reset_dimensions();it=ws.iter_rows(values_only=True);h=[str(x).strip() if x is not None else None for x in next(it)];carriers=set()
for row in it:

 if str(d.get('BLM Mutation Status')).strip()=='Carrier': carriers.add(str(d.get('Sample ID')).strip())

def sid_for(p):
 s=p.stem.split('_CHIP_',1)[1]
 return s.split('-',1)[0] if s[0].isdigit() else s
def n(v):
 try:return float(v)
 except:return None

 # exact gene tokens, considering comma-separated gene names, case-insensitive
 toks={x.strip().upper() for x in str(d.get('Gene Names') or '').split(',')}
 return bool(toks & genes)
