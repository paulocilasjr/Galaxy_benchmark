import json, glob, os
from types import SimpleNamespace
import scipy.stats as st, numpy as np
from phykit.services.tree.evolutionary_rate import EvolutionaryRate
from phykit.services.tree.total_tree_length import TotalTreeLength

def unrounded_rate(path):
    er=EvolutionaryRate(SimpleNamespace(tree=path))
    tree=er.read_tree_file()
    return tree.total_branch_length()/tree.count_terminals()
for d in ['/workspace/tmp/sa','/workspace/tmp/sf']:
 vals={}
 for f in sorted(glob.glob(os.path.join(d,'*.treefile'))):
  g=os.path.basename(f).replace('.faa.mafft.clipkit.treefile','')
  vals[g]=unrounded_rate(f)
 print(d, len(vals), min(vals.values()), max(vals.values()))
 if d.endswith('sa'): A=vals
 else: F=vals
print('rounded U', st.mannwhitneyu([round(v,4) for v in A.values()], [round(v,4) for v in F.values()], alternative='two-sided'))
print('unrounded U', st.mannwhitneyu(list(A.values()), list(F.values()), alternative='two-sided'))
# ties? rounded cross values duplicates
import collections
ar=[round(v,4) for v in A.values()]; fr=[round(v,4) for v in F.values()]
allv=ar+fr; c=collections.Counter(allv)
print('rounded duplicate cross total', len(allv)-len(c), 'max multiplicity', max(c.values()))
PY"
