/bin/bash -lc "python - <<'PY'
import zipfile, collections
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    c=collections.Counter()
    with zipfile.ZipFile(z) as f:
        for n in f.namelist():
            base=n
            if base.endswith('.gz'):
                base=base[:-3]
            parts=base.split('.faa',1)
            suffix='.faa'+parts[1] if len(parts)>1 else base
            c[suffix]+=1
    print(z)
    for k,v in sorted(c.items()): print(v,k)
PY"
