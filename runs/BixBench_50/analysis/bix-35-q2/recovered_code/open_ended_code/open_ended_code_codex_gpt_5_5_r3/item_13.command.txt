/bin/bash -lc "python - <<'PY'
import zipfile
sets=[]
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        genes={n.split('.faa')[0] for n in f.namelist() if n.endswith('.faa.mafft.clipkit')}
    sets.append(genes)
print('animals', len(sets[0]))
print('fungi', len(sets[1]))
print('intersection', len(sets[0]&sets[1]))
print('animals only', len(sets[0]-sets[1]), sorted(sets[0]-sets[1])[:20])
print('fungi only', len(sets[1]-sets[0]), sorted(sets[1]-sets[0])[:20])
PY"
