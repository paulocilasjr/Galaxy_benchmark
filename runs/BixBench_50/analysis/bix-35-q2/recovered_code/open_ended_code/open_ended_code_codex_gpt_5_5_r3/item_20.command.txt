/bin/bash -lc "python - <<'PY'
import zipfile
sets=[]
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        genes={n.split('.faa')[0] for n in f.namelist() if n.endswith('.faa.mafft.clipkit.treefile')}
    sets.append(genes)
    print(z, len(genes), sorted(list(genes))[:10], sorted(list(genes))[-10:])
print('intersection', len(sets[0]&sets[1]))
print('animals only', len(sets[0]-sets[1]), sorted(sets[0]-sets[1])[:50])
print('fungi only', len(sets[1]-sets[0]), sorted(sets[1]-sets[0])[:50])
PY"
