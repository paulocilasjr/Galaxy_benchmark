/bin/bash -lc "python - <<'PY'
import zipfile
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('---',z)
    with zipfile.ZipFile(z) as f:
        missing=sorted({n.split('.faa')[0] for n in f.namelist() if n.endswith('.faa.mafft.clipkit')} - {n.split('.faa')[0] for n in f.namelist() if n.endswith('.faa.mafft.clipkit.treefile')})
        print('missing treefile count', len(missing), 'first', missing[:5])
        for g in missing[:2]:
            log=f.read(g+'.faa.mafft.clipkit.log').decode(errors='replace')
            print('log for',g)
            print('\\n'.join(log.splitlines()[:20]))
            print('...last...')
            print('\\n'.join(log.splitlines()[-10:]))
PY"
