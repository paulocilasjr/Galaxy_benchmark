# Find shared orthogroups between animals and fungi treefiles
unzip -l inputs/scogs_animals.zip | grep -oP '[0-9]+at[0-9]+(?=\.faa\.mafft\.clipkit\.treefile)' | sort > /tmp/animals_ogs.txt
unzip -l inputs/scogs_fungi.zip | grep -oP '[0-9]+at[0-9]+(?=\.faa\.mafft\.clipkit\.treefile)' | sort > /tmp/fungi_ogs.txt
echo "Animals unique: $(wc -l < /tmp/animals_ogs.txt)"
echo "Fungi unique: $(wc -l < /tmp/fungi_ogs.txt)"
echo "Shared: $(comm -12 /tmp/animals_ogs.txt /tmp/fungi_ogs.txt | wc -l)"
