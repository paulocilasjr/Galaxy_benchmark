mkdir -p /tmp/animals_trees /tmp/fungi_trees

# Get shared orthogroups
comm -12 /tmp/animals_ogs.txt /tmp/fungi_ogs.txt > /tmp/shared_ogs.txt

# Extract animal trees for shared OGs
for og in $(cat /tmp/shared_ogs.txt); do
  unzip -p inputs/scogs_animals.zip "*${og}.faa.mafft.clipkit.treefile" > /tmp/animals_trees/${og}.treefile 2>/dev/null
done

# Extract fungi trees for shared OGs  
for og in $(cat /tmp/shared_ogs.txt); do
  unzip -p inputs/scogs_fungi.zip "*${og}.faa.mafft.clipkit.treefile" > /tmp/fungi_trees/${og}.treefile 2>/dev/null
done

echo "Animal trees: $(ls /tmp/animals_trees/ | wc -l)"
echo "Fungi trees: $(ls /tmp/fungi_trees/ | wc -l)"
