from scipy.stats import mannwhitneyu
import json

# Re-run to collect rates
import subprocess
import os

animals_dir = "/tmp/animals_trees"
fungi_dir = "/tmp/fungi_trees"

shared_ogs = []
with open("/tmp/shared_ogs.txt") as f:
    for line in f:
        og = line.strip()
        if og:
            shared_ogs.append(og)

animal_rates = []
fungi_rates = []

for og in shared_ogs:
    animal_tree = os.path.join(animals_dir, f"{og}.treefile")
    fungi_tree = os.path.join(fungi_dir, f"{og}.treefile")
    
    if not os.path.exists(animal_tree) or not os.path.exists(fungi_tree):
        continue
    
    ar = subprocess.run(["phykit", "evo_rate", animal_tree], 
                      capture_output=True, text=True, check=True)
    animal_rate = float(ar.stdout.strip())
    
    fr = subprocess.run(["phykit", "evo_rate", fungi_tree],
                      capture_output=True, text=True, check=True)
    fungi_rate = float(fr.stdout.strip())
    
    animal_rates.append(animal_rate)
    fungi_rates.append(fungi_rate)

print(f"n_animals = {len(animal_rates)}")
print(f"n_fungi = {len(fungi_rates)}")

# Mann-Whitney U test
result = mannwhitneyu(animal_rates, fungi_rates, alternative='two-sided')
print(f"Mann-Whitney U statistic = {result.statistic}")
print(f"p-value = {result.pvalue}")

# Verify U manually
# U1 = R1 - n1*(n1+1)/2 where R1 is sum of ranks for group 1
# U = min(U1, U2) or we can use the scipy value

# Let's also compute U for animals manually
import numpy as np
all_rates = np.concatenate([animal_rates, fungi_rates])
ranks = np.argsort(np.argsort(all_rates)) + 1  # 1-based ranks
# Handle ties
from scipy.stats import rankdata
ranks = rankdata(all_rates)
animal_ranks = ranks[:len(animal_rates)]
fungi_ranks = ranks[len(animal_rates):]

R1 = np.sum(animal_ranks)
R2 = np.sum(fungi_ranks)
n1 = len(animal_rates)
n2 = len(fungi_rates)

U1 = R1 - n1*(n1+1)/2
U2 = R2 - n2*(n2+1)/2
U = min(U1, U2)

print(f"\nManual calculation:")
print(f"R1 (animal rank sum) = {R1}")
print(f"R2 (fungi rank sum) = {R2}")
print(f"U1 = {U1}")
print(f"U2 = {U2}")
print(f"U = min(U1, U2) = {U}")
