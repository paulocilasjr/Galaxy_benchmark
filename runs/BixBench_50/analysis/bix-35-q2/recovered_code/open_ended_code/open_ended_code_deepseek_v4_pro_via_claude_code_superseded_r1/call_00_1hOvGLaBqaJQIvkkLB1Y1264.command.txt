import subprocess
import os
import sys

animals_dir = "/tmp/animals_trees"
fungi_dir = "/tmp/fungi_trees"

# Get shared orthogroups
shared_ogs = []
with open("/tmp/shared_ogs.txt") as f:
    for line in f:
        og = line.strip()
        if og:
            shared_ogs.append(og)

print(f"Processing {len(shared_ogs)} shared orthogroups...")

animal_rates = []
fungi_rates = []

for og in shared_ogs:
    animal_tree = os.path.join(animals_dir, f"{og}.treefile")
    fungi_tree = os.path.join(fungi_dir, f"{og}.treefile")
    
    if not os.path.exists(animal_tree) or not os.path.exists(fungi_tree):
        print(f"Missing tree for {og}")
        continue
    
    # Run phykit evo_rate
    try:
        ar = subprocess.run(["phykit", "evo_rate", animal_tree], 
                          capture_output=True, text=True, check=True)
        animal_rate = float(ar.stdout.strip())
        
        fr = subprocess.run(["phykit", "evo_rate", fungi_tree],
                          capture_output=True, text=True, check=True)
        fungi_rate = float(fr.stdout.strip())
        
        animal_rates.append(animal_rate)
        fungi_rates.append(fungi_rate)
    except Exception as e:
        print(f"Error processing {og}: {e}")

print(f"\nSuccessfully processed {len(animal_rates)} orthogroups")
print(f"Animal rates: mean={sum(animal_rates)/len(animal_rates):.4f}, range=[{min(animal_rates):.4f}, {max(animal_rates):.4f}]")
print(f"Fungi rates: mean={sum(fungi_rates)/len(fungi_rates):.4f}, range=[{min(fungi_rates):.4f}, {max(fungi_rates):.4f}]")
PYEOF
