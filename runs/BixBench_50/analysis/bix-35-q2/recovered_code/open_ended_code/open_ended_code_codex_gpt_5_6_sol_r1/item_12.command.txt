/bin/bash -lc "PYTHONPATH=work/phykit_vendor python - <<'PY'
from Bio import Phylo
from collections import Counter
from pathlib import Path
for group in ('animals','fungi'):
 fs=sorted(Path(f'work/scogs_{group}').glob('*.treefile'))
 tips=Counter(); finite=0
 for p in fs:
  t=Phylo.read(p,'newick')
  tips[len(t.get_terminals())]+=1
  rate=round(sum(c.branch_length or 0 for c in t.find_clades())/len(t.get_terminals()),4)
  finite += rate == rate
 print(group,'files',len(fs),'terminal_counts',dict(sorted(tips.items())),'finite',finite)
a={p.name.split('.faa.',1)[0] for p in Path('work/scogs_animals').glob('*.treefile')}
f={p.name.split('.faa.',1)[0] for p in Path('work/scogs_fungi').glob('*.treefile')}
print('tree gene overlap',len(a&f),'animal_only',len(a-f),'fungi_only',len(f-a))
PY"
