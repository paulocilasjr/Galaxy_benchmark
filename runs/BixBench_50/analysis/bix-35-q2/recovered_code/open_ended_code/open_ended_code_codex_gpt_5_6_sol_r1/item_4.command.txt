from zipfile import ZipFile
from collections import Counter
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 with ZipFile(fn) as z:
  names=[x.filename for x in z.infolist() if not x.is_dir()]
  c=Counter()
  for n in names:
   if n.endswith('.treefile'): c['treefile']+=1
   elif n.endswith('.mafft.clipkit'): c['clipkit_alignment']+=1
   elif n.endswith('.mafft'): c['mafft_alignment']+=1
   elif n.endswith('.faa'): c['faa']+=1
   else: c[n.rsplit('.',1)[-1]]+=1
  trees={n.removesuffix('.faa.mafft.clipkit.treefile') for n in names if n.endswith('.treefile')}
  alns={n.removesuffix('.faa.mafft.clipkit') for n in names if n.endswith('.mafft.clipkit')}
  print(fn, len(names), c)
  print('trees no alns',len(trees-alns),'alns no trees',len(alns-trees),'intersection',len(trees&alns))
  for n in names:
   if n.endswith('.treefile'):
    print('example tree',n,z.read(n).decode().strip()); break
  for n in names:
   if n.endswith('.mafft.clipkit'):
