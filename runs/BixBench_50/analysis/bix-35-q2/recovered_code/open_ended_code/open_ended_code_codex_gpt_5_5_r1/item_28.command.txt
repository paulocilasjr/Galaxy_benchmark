/bin/bash -lc "python - <<'PY'
import zipfile
for zname in ['scogs_animals.zip','scogs_fungi.zip']:
    with zipfile.ZipFile('/workspace/inputs/'+zname) as z:
        raw={n[:-len('.faa.mafft.clipkit')] for n in z.namelist() if n.endswith('.faa.mafft.clipkit')}
        tree={n[:-len('.faa.mafft.clipkit.treefile')] for n in z.namelist() if n.endswith('.faa.mafft.clipkit.treefile')}
        logs=[g+'.faa.mafft.clipkit.log' for g in sorted(raw-tree)[:5]]
        print('\\n',zname,'missing logs')
        for lf in logs:
            if lf in z.namelist():
                txt=z.read(lf).decode(errors='replace').splitlines()
                print('##', lf)
                print('\\n'.join(txt[-15:]))
PY"
