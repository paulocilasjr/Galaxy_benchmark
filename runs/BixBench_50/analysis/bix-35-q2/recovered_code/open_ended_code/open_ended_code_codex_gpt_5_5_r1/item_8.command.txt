/bin/bash -lc "python - <<'PY'
import zipfile, collections, os
for name in ['scogs_animals.zip','scogs_fungi.zip']:
    with zipfile.ZipFile('/workspace/inputs/'+name) as z:
        ns=z.namelist()
    ext_counts=collections.Counter()
    gene_files=collections.defaultdict(list)
    for n in ns:
        base=n
        # common suffixes
        for suff in ['.faa.mafft.clipkit.treefile','.faa.mafft.clipkit','.faa.mafft','.faa']:
            if base.endswith(suff):
                gene_files[base[:-len(suff)]].append(suff)
                break
        if n.endswith('.treefile'): ext_counts['treefile']+=1
        elif n.endswith('.mafft.clipkit'): ext_counts['clipkit']+=1
        elif n.endswith('.mafft'): ext_counts['mafft']+=1
        elif n.endswith('.faa'): ext_counts['faa']+=1
        elif n.endswith('.iqtree'): ext_counts['iqtree']+=1
        elif n.endswith('.bionj'): ext_counts['bionj']+=1
        elif n.endswith('.mldist'): ext_counts['mldist']+=1
        elif n.endswith('.log'): ext_counts['log']+=1
        elif n.endswith('.model.gz'): ext_counts['model.gz']+=1
        elif n.endswith('.ckp.gz'): ext_counts['ckp.gz']+=1
        else: ext_counts['other']+=1
    print('\\n', name, 'members', len(ns))
    print(ext_counts)
    print('genes', len(gene_files))
    print('with tree+clipkit', sum({'.faa.mafft.clipkit.treefile','.faa.mafft.clipkit'} <= set(v) for v in gene_files.values()))
    print('sample gene', next(iter(gene_files.items())))
PY"
