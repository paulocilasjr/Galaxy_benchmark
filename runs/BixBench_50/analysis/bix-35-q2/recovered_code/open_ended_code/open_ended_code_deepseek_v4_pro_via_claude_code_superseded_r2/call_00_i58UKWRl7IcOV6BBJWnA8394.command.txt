# Get the sequence IDs for missing species for all 16 genes
python3 << 'EOF'
import csv

species_dirs = {
    'Animals_Cele': '/tmp/busco_tables/Animals_Cele',
    'Animals_Ggal': '/tmp/busco_tables/Animals_Ggal',
    'Animals_Mmus': '/tmp/busco_tables/Animals_Mmus',
    'Animals_Xlae': '/tmp/busco_tables/Animals_Xlae',
    'Fungi_Ncra': '/tmp/busco_tables/Fungi_Ncra',
    'Fungi_Scer': '/tmp/busco_tables/Fungi_Scer',
    'Fungi_Spom': '/tmp/busco_tables/Fungi_Spom',
    'Fungi_Umay': '/tmp/busco_tables/Fungi_Umay',
}

SHARED_GENES = [
    '1014314at2759', '1018517at2759', '1193442at2759', '1249159at2759',
    '1269649at2759', '1342242at2759', '1346432at2759', '1405073at2759',
    '1428854at2759', '1504863at2759', '1538526at2759', '1552706at2759',
    '1593937at2759', '160593at2759', '1645187at2759', '176625at2759'
]

# Build mapping: gene -> species -> sequence_id
gene_seq_ids = {g: {} for g in SHARED_GENES}

for sp, sp_dir in species_dirs.items():
    fpath = f"{sp_dir}/run_eukaryota_odb10/full_table.tsv"
    with open(fpath) as f:
        for line in f:
            if line.startswith('#') or not line.strip():
                continue
            parts = line.strip().split('\t')
            if len(parts) >= 3 and parts[1] == 'Complete':
                busco_id = parts[0]
                seq_id = parts[2]
                if busco_id in gene_seq_ids:
                    gene_seq_ids[busco_id][sp] = seq_id

# Check which are missing
for gene in SHARED_GENES:
    missing = [sp for sp in species_dirs if sp not in gene_seq_ids[gene]]
    if missing:
        print(f"{gene}: missing {missing}")
        for sp in missing:
            print(f"  Trying to find what happened for {sp}...")

print("\nComplete gene_seq_ids for all genes:")
for gene in SHARED_GENES:
    seqs = gene_seq_ids[gene]
    print(f"  {gene}: {len(seqs)} species - {sorted(seqs.keys())}")
EOF
