species_list = ['Animals_Cele', 'Animals_Ggal', 'Animals_Mmus', 'Animals_Xlae',
                'Fungi_Ncra', 'Fungi_Scer', 'Fungi_Spom', 'Fungi_Umay']

complete_genes = {}
for sp in species_list:
    fpath = f"/tmp/busco_tables/{sp}/run_eukaryota_odb10/full_table.tsv"
    genes = set()
    with open(fpath) as f:
        for line in f:
            if line.startswith('#') or not line.strip():
                continue
            parts = line.strip().split('\t')
            # parts[0] = Busco id, parts[1] = Status
            if len(parts) >= 2 and parts[1] == 'Complete':
                genes.add(parts[0])
    complete_genes[sp] = genes
    print(f"{sp}: {len(genes)} complete single-copy genes")

# Find intersection
shared = complete_genes[species_list[0]]
for sp in species_list[1:]:
    shared = shared & complete_genes[sp]

print(f"\nShared complete genes across all 8 species: {len(shared)}")
print("Genes:", sorted(shared))

# Check if they're in SCOS
import subprocess
a_genes = set()
with open('/tmp/scogs_a_genes.txt') as f:
    for line in f:
        a_genes.add(line.strip())
f_genes = set()
with open('/tmp/scogs_f_genes.txt') as f:
    for line in f:
        f_genes.add(line.strip())

in_both = shared & a_genes & f_genes
print(f"\nShared complete genes also in both SCOS sets: {len(in_both)}")
print("Genes:", sorted(in_both))
