/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for zpath in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 with ZipFile(zpath) as z:
  names=z.namelist()
  trees={n.split('.faa',1)[0] for n in names if n.endswith('.treefile')}
  fasta=[n for n in names if n.endswith('.faa')]
  counts=Counter(); missing=Counter()
  examples=[]
  for n in fasta:
   gid=n.split('.faa',1)[0]
   seqs=sum(1 for line in z.read(n).splitlines() if line.startswith(b'>'))
   counts[seqs]+=1
   if gid not in trees:
    missing[seqs]+=1
    if len(examples)<8: examples.append((gid,seqs))
 print(zpath,'all_faa',len(fasta),'sequence_counts',dict(sorted(counts.items())),
       'without_tree',sum(missing.values()),'without_tree_by_sequences',dict(sorted(missing.items())))
 print('examples',examples)
PY"
