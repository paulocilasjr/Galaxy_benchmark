python - \""'$analysis_tmp" <<'"'PY'
import pathlib, sys
from Bio import Phylo
from scipy.stats import mannwhitneyu, rankdata
root=pathlib.Path(sys.argv[1])
def rates(group):
 out=[]
 details=[]
 for p in sorted((root/group).glob('*.treefile')):
  t=Phylo.read(p,'newick')
  total=sum(c.branch_length or 0.0 for c in t.find_clades())
  n=len(t.get_terminals())
  out.append(total/n)
  details.append((p.name,n,total,total/n,round(total/n,4)))
 return out,details
a,ad=rates('animals'); f,fd=rates('fungi')
for label,x,y in [('full_precision',a,f),('rounded_4dp',[round(v,4) for v in a],[round(v,4) for v in f])]:
 res=mannwhitneyu(x,y,alternative='two-sided',method='asymptotic')
 print(label,res)
 print('U check from ranks',rankdata(x+y,method='average')[:len(x)].sum()-len(x)*(len(x)+1)/2)
from collections import Counter
for name,ds in [('animals',ad),('fungi',fd)]:
 print(name,'tip_counts',dict(sorted(Counter(d[1] for d in ds).items())))
# confirm PhyKIT output exactly matches Python round on all files via rates already gathered indirectly with prior run logic
PY"
