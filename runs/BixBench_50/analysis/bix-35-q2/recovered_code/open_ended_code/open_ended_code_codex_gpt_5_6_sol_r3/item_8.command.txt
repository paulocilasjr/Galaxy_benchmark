/bin/bash -lc 'analysis_tmp=$(mktemp -d -p /workspace evo_rate.XXXXXX)
mkdir "$analysis_tmp/animals" "$analysis_tmp/fungi"
unzip -q -j inputs/scogs_animals.zip '"'*.treefile' -d \""'$analysis_tmp/animals"
unzip -q -j inputs/scogs_fungi.zip '"'*.treefile' -d \""'$analysis_tmp/fungi"
printf '"'%s\\n' \""'$analysis_tmp" > /tmp/evo_rate_tmp_path
first_tree=$(find "$analysis_tmp/animals" -type f -name '"'*.treefile' -print -quit)
printf 'PhyKIT version: '
phykit --version
printf 'Example tree: %s\\n' \""'$(basename "$first_tree")"
printf '"'Example evo_rate output: '
phykit evo_rate \""'$first_tree"
printf '"'Extracted counts: animals='
find \""'$analysis_tmp/animals" -type f -name '"'*.treefile' | wc -l
printf 'fungi='
find \""'$analysis_tmp/fungi" -type f -name '"'*.treefile' | wc -l"
