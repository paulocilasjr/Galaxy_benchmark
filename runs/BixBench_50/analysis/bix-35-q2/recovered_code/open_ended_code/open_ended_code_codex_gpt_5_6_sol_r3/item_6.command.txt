/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import PurePosixPath
for zpath in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with ZipFile(zpath) as z:
        names=[n for n in z.namelist() if not n.endswith('/')]
    c=Counter()
    for n in names:
        b=PurePosixPath(n).name
        if b.endswith('.treefile'): ext='treefile'
        elif b.endswith('.clipkit'): ext='clipkit'
        elif b.endswith('.mafft'): ext='mafft'
        elif b.endswith('.faa'): ext='faa'
        else: ext='other:'+b.rsplit('.',1)[-1]
        c[ext]+=1
    trees=[PurePosixPath(n).name for n in names if n.endswith('.treefile')]
    ids={t.split('.faa',1)[0] for t in trees}
    print(zpath, 'members',len(names),'types',dict(sorted(c.items())))
    print('trees',len(trees),'gene_ids',len(ids),'dupes',len(trees)-len(ids),'first',trees[:5])
PY"
