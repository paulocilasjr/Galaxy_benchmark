/bin/bash -lc "python - <<'PY'
from pathlib import Path
from Bio import Phylo
from io import StringIO
from contextlib import redirect_stdout
from types import SimpleNamespace
from phykit.services.tree.evolutionary_rate import EvolutionaryRate
from scipy.stats import mannwhitneyu

# Independent check of PhyKIT's reported values against direct tree sums.
def rows(group):
    out=[]
    for p in sorted((Path('/workspace/.work_rates')/group).glob('*.treefile')):
        with p.open() as fh:
            tree=Phylo.read(fh,'newick')
        direct=sum(float(c.branch_length or 0.0) for c in tree.find_clades())/len(tree.get_terminals())
        b=StringIO()
        with redirect_stdout(b):
            EvolutionaryRate(SimpleNamespace(tree=str(p),json=False)).run()
        reported=float(b.getvalue().strip())
        out.append((reported,direct,p.name))
    return out
A=rows('animals'); F=rows('fungi')
print('animals mismatches after 4dp:',sum(round(x[1],4)"'!=x[0] for x in A))
print('"'fungi mismatches after 4dp:',sum(round(x[1],4)"'!=x[0] for x in F))
print('"'PhyKIT U:',mannwhitneyu([x[0] for x in A],[x[0] for x in F],alternative='two-sided').statistic)
print('full-precision U:',mannwhitneyu([x[1] for x in A],[x[1] for x in F],alternative='two-sided').statistic)
print('terminal-count range animals:',min(len(Phylo.read(str(p),'newick').get_terminals()) for p in (Path('/workspace/.work_rates/animals')).glob('*.treefile')),max(len(Phylo.read(str(p),'newick').get_terminals()) for p in (Path('/workspace/.work_rates/animals')).glob('*.treefile')))
print('terminal-count range fungi:',min(len(Phylo.read(str(p),'newick').get_terminals()) for p in (Path('/workspace/.work_rates/fungi')).glob('*.treefile')),max(len(Phylo.read(str(p),'newick').get_terminals()) for p in (Path('/workspace/.work_rates/fungi')).glob('*.treefile')))
PY"
