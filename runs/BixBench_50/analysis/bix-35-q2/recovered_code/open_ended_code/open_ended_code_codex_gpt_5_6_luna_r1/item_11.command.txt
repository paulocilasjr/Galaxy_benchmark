/bin/bash -lc 'for f in inputs/scogs_animals.zip inputs/scogs_fungi.zip; do
  echo "### $f"
  echo -n '"'treefiles '; unzip -Z1 \""'$f" | awk '"'/\\.treefile"'$/{n++} END{print n+0}'"'
  echo -n 'alignments '; unzip -Z1 \""'$f" | awk '"'/\\.faa\\.mafft\\.clipkit"'$/{n++} END{print n+0}'"'
  echo -n 'raw faa '; unzip -Z1 \""'$f" | awk '"'/\\.faa"'$/{n++} END{print n+0}'"'
  echo -n 'all entries '; unzip -Z1 \""'$f" | wc -l
  echo '"'treefile examples:'
  unzip -Z1 \""'$f" | awk '"'/\\.treefile"'$/{print}'"' | sed -n '1,12p'
done"
