/bin/bash -lc "python - <<'PY'
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO
from types import SimpleNamespace
from phykit.services.tree.evolutionary_rate import EvolutionaryRate
from scipy.stats import mannwhitneyu

def rates(group, suffix='treefile'):
    d={}
    for p in sorted((Path('/workspace/.work_rates')/group).glob('*.'+suffix)):
        b=StringIO()
        with redirect_stdout(b): EvolutionaryRate(SimpleNamespace(tree=str(p),json=False)).run()
        d[p.name.split('.')[0]]=float(b.getvalue().strip())
    return d
for suffix in ('treefile','bionj'):
    A=rates('animals',suffix); F=rates('fungi',suffix)
    print(suffix, len(A),len(F), 'all U',mannwhitneyu(list(A.values()),list(F.values())).statistic)
    shared=sorted(set(A)&set(F))
    print(' shared',len(shared),'U',mannwhitneyu([A[k] for k in shared],[F[k] for k in shared]).statistic)
PY"
