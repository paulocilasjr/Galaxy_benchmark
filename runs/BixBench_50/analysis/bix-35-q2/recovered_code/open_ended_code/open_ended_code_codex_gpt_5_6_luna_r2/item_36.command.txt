/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
from pathlib import Path
for group in ['animals','fungi']:
    fn=Path('inputs')/f'scogs_{group}.zip'
    with ZipFile(fn) as z:
        faa={n[:-4]: z.read(n).count(b'\\n>') + (1 if z.read(n).startswith(b'>') else 0)
             for n in z.namelist() if n.endswith('.faa')}
        trees={n[:-len('.faa.mafft.clipkit.treefile')] for n in z.namelist() if n.endswith('.treefile')}
        print(group, 'faa',len(faa),'tree',len(trees))
        print('faa header-count distribution',Counter(faa.values()))
        print('tree header-count distribution',Counter(faa[k] for k in trees))
        print('tree IDs with <4 headers',[(k,faa[k]) for k in sorted(trees) if faa[k]<4][:20], 'count',sum(faa[k]<4 for k in trees))
        print('non-tree ID header dist',Counter(v for k,v in faa.items() if k not in trees))
        print('IDs tree missing in other?', '')
PY"
