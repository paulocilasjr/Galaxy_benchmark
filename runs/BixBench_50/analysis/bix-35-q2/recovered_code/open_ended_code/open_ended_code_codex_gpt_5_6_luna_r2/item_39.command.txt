/bin/bash -lc "python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
from tempfile import TemporaryDirectory
from subprocess import check_output
from scipy.stats import mannwhitneyu

rates={}
with TemporaryDirectory() as td:
 td=Path(td)
 for g in ['animals','fungi']:
  vals={}
  with ZipFile(Path('inputs')/f'scogs_{g}.zip') as z:
   items={n[:-len('.faa.mafft.clipkit.treefile')]:n for n in z.namelist() if n.endswith('.treefile')}
   for i,(k,n) in enumerate(items.items()):
    p=td/f'{g}_{i}.tre'; p.write_bytes(z.read(n))
    vals[k]=float(check_output(['/opt/agent-venv/bin/pk_evo_rate',str(p)], text=True).strip())
  rates[g]=vals
for scope, keys in [('all',None),('common',set(rates['animals']) & set(rates['fungi']))]:
 if keys is None:
  a=list(rates['animals'].values()); f=list(rates['fungi'].values())
 else:
  a=[rates['animals'][k] for k in sorted(keys)]; f=[rates['fungi'][k] for k in sorted(keys)]
 r=mannwhitneyu(a,f,alternative='two-sided')
 print(scope,len(a),len(f),float(r.statistic),float(len(a)*len(f)-r.statistic),float(r.pvalue))
PY"
