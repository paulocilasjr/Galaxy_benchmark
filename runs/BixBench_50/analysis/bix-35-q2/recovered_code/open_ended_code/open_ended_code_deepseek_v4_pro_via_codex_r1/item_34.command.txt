/bin/bash -lc "cd /workspace/work
python - <<'PY'
import os,re,subprocess
from scipy.stats import mannwhitneyu, rankdata
pat=re.compile(r'"'^('"\\d+at\\d+)\\.faa\\.mafft\\.clipkit\\.treefile"'$'"')
rates={'A':[],'F':[]}
for d,k in [('scogs_animals','A'),('scogs_fungi','F')]:
 for fn in os.listdir(d):
  if pat.match(fn):
   out=subprocess.check_output(['python','-m','phykit','evo_rate',os.path.join(d,fn)],text=True).strip()
   rates[k].append(float(out))
a=rates['A']; f=rates['F']
print('len',len(a),len(f))
for meth in ['asymptotic','exact','auto']:
 for alt in ['two-sided','less','greater']:
  try:
   r=mannwhitneyu(a,f,alternative=alt,method=meth)
   print(meth,alt,'stat',r.statistic,'p',r.pvalue)
  except Exception as e: print(e)
# direct internal
from scipy.stats import _mannwhitneyu as mm
print('direct validation', mm.mannwhitneyu(a,f,method='asymptotic'))
# compute with rankdata
xy=a+f; ranks=rankdata(xy); R1=sum(ranks[:len(a)]); U1=R1-len(a)*(len(a)+1)/2
print('R1 U1 U2',R1,U1,len(a)*len(f)-U1)
PY"
