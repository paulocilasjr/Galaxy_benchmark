/bin/bash -lc "cd /workspace/work
python - <<'PY'
import os, re, subprocess
from scipy.stats import mannwhitneyu

root = '/workspace/work'
groups = {
    'animals': os.path.join(root, 'scogs_animals'),
    'fungi': os.path.join(root, 'scogs_fungi'),
}
pat = re.compile(r'"'^'"\\d+at\\d+\\.faa\\.mafft\\.clipkit\\.treefile"'$'"')
rates = {k: [] for k in groups}
files = {k: [] for k in groups}
for group, d in groups.items():
    for fn in sorted(os.listdir(d)):
        if pat.match(fn):
            path = os.path.join(d, fn)
            out = subprocess.check_output(['python', '-m', 'phykit', 'evo_rate', path], text=True).strip()
            rates[group].append(float(out))
            files[group].append(fn)

for group in groups:
    print(group, 'trees', len(rates[group]))

res = mannwhitneyu(rates['animals'], rates['fungi'], alternative='two-sided')
print('U statistic', repr(res.statistic))
print('p-value', res.pvalue)
print('animal median', sorted(rates['animals'])[len(rates['animals'])//2])
print('fungi median', sorted(rates['fungi'])[len(rates['fungi'])//2])
PY"
