/bin/bash -lc 'set -e
cd /workspace
mkdir -p work
cd work
for z in ../inputs/scogs_animals.zip ../inputs/scogs_fungi.zip; do d=$(basename "$z" .zip); mkdir -p "$d"; unzip -q -o "$z" -d "$d"; done
echo animals files:
find scogs_animals -type f | wc -l
echo fungi files:
find scogs_fungi -type f | wc -l
echo animals base genes:
find scogs_animals -type f -name '"'*.treefile' | sed 's#scogs_animals/##' | sort | head
echo fungi treefiles count:
find scogs_fungi -type f -name '*.treefile' | wc -l
echo animal faa count; find scogs_animals -type f -name '*.faa' | wc -l
echo fungi faa count; find scogs_fungi -type f -name '*.faa' | wc -l"
