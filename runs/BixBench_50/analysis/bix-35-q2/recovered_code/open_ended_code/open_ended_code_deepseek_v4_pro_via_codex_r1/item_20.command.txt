import os, subprocess, re, math
from scipy.stats import mannwhitneyu

    files=sorted(os.path.join(dp,fn) for dp,_,fns in os.walk(d) for fn in fns if fn.endswith('.treefile'))

        out=subprocess.check_output(['python','-m','phykit','evo_rate',f], text=True).strip()

    print(d, len(rates), min(rates), max(rates), sum(rates)/len(rates))

    globals()[d+'_files']=files
# save raw and rounded?
for use_rounded in [False, True]:
    a=globals()['scogs_animals']; f=globals()['scogs_fungi']
    if use_rounded:
        a=[round(x,4) for x in a]; f=[round(x,4) for x in f]
    for alt in ['two-sided','less','greater']:
        res=mannwhitneyu(a,f,alternative=alt,method='auto')
        print('rounded',use_rounded,'alt',alt,'U',res.statistic,'p',res.pvalue)
    # manual U1 U2
    import scipy.stats as st
    rank=st.rankdata(a+f)
    R1=sum(rank[:len(a)]); U1=len(a)*len(f)+len(a)*(len(a)+1)/2-R1
    U2=len(a)*len(f)-U1
    print('manual U1 (animals)',U1,'U2',U2,'n1 n2',len(a),len(f))
