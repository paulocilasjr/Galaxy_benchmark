import os,re,subprocess
from scipy.stats import mannwhitneyu
pat=re.compile(r'"'^('"\\d+at\\d+)\\.faa\\.mafft\\.clipkit\\.treefile"'$'"')
for d in ['scogs_animals','scogs_fungi']:
    files={}

    for g,f in files.items():
        out=subprocess.check_output(['python','-m','phykit','evo_rate',f],text=True).strip()
        rates[g]=float(out)
    print(d,len(rates))
    globals()[d+'_rates']=rates
A=globals()['scogs_animals_rates']; F=globals()['scogs_fungi_rates']
common=sorted(set(A)&set(F))
print('common',len(common))
for label,a,f in [('all',[A[g] for g in A],[F[g] for g in F]),('common',[A[g] for g in common],[F[g] for g in common])]:
    for order in ['AF','FA']:
        x,y=(a,f) if order=='AF' else (f,a)
        res=mannwhitneyu(x,y,alternative='two-sided')
        print(label,order,'U',res.statistic)
    # manual U1/U2
    import scipy.stats as st
    vals=a+f; ranks=st.rankdata(vals); R1=sum(ranks[:len(a)]); U1=len(a)*len(f)+len(a)*(len(a)+1)/2-R1; U2=len(a)*len(f)-U1
    print('manual',U1,U2,'n',len(a),len(f))
