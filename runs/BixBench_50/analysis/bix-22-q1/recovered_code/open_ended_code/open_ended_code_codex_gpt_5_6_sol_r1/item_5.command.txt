/bin/bash -lc "rg --files -g '"'!inputs/BatchCorrectedReadCounts_Zenodo.csv'"' /workspace | sort
python - <<'PY'
import csv, collections, math
expr='inputs/BatchCorrectedReadCounts_Zenodo.csv'; meta='inputs/GeneMetaInfo_Zenodo.csv'; ann='inputs/Sample_annotated_Zenodo.csv'
with open(expr,newline='') as f:
    r=csv.reader(f); eh=next(r); eids=[row[0] for row in r]
with open(meta,newline='') as f:
    r=csv.DictReader(f); mrows=list(r)
with open(ann,newline='') as f:
    r=csv.DictReader(f); arows=list(r)
print('expression dimensions',len(eids),len(eh)-1)
print('expression gene IDs unique',len(set(eids)),'duplicate groups',sum(n>1 for n in collections.Counter(eids).values()))
print('expression sample IDs unique',len(set(eh[1:])),'duplicate groups',sum(n>1 for n in collections.Counter(eh[1:]).values()))
print('metadata rows',len(mrows),'row index unique',len(set(x[''] for x in mrows)),'Geneid unique',len(set(x['Geneid'] for x in mrows)))
print('meta index==Geneid',sum(x['']==x['Geneid'] for x in mrows),'/',len(mrows))
print('gene order exact match',eids==[x[''] for x in mrows])
print('annotation rows',len(arows),'sample unique',len(set(x['sample'] for x in arows)),'matrix/annotation sets equal',set(eh[1:])==set(x['sample'] for x in arows))
print('annotation exact order',eh[1:]==[x['sample'] for x in arows])
for field in ['celltype','samtype','batch','sex']:
 print(field,dict(collections.Counter(x[field] for x in arows)))
print('samp unique',len(set(x['samp'] for x in arows)),'repeat multiplicities',dict(collections.Counter(collections.Counter(x['samp'] for x in arows).values())))
# Explicitly assess repeated subject IDs for per-celltype collisions and invariant conflicts
by=collections.defaultdict(list)
for x in arows: by[x['samp']].append(x)
print('repeated samp groups',sum(len(v)>1 for v in by.values()))
print('same samp + same celltype duplicates',[(k,ct,len(z)) for k,v in by.items() for ct in set(x['celltype'] for x in v) if len(z:=[x for x in v if x['celltype']==ct])>1])
for fld in ['samtype','batch','sex']:
 print('repeated samp conflicts',fld,sum(len(set(x[fld] for x in v))>1 for v in by.values()))
print('protein_coding',sum(x['gene_biotype']=='protein_coding' for x in mrows))
print('missing lengths',sum(not x['Length'] for x in mrows),'bad lengths',sum(1 for x in mrows if x['Length'] and not x['Length'].isdigit()))
PY"
