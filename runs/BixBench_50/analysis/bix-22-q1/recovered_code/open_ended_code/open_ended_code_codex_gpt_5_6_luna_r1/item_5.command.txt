import csv, os
from collections import Counter

base='inputs'
for fn in ['GeneMetaInfo_Zenodo.csv','Sample_annotated_Zenodo.csv','BatchCorrectedReadCounts_Zenodo.csv']:
    path=os.path.join(base,fn)
    with open(path, newline='') as f:
        rd=csv.reader(f)
        header=next(rd)
        n=0
        first=[]
        for row in rd:
            n+=1
            if len(first)<2: first.append(row[:8])
    print(fn, 'rows', n, 'header_cols', len(header), 'header_first', header[:8], 'first_rows', first)

with open(base+'/GeneMetaInfo_Zenodo.csv', newline='') as f:
    rd=csv.DictReader(f)
    bios=Counter(); gids=Counter(); missing_len=0; bad_len=0
    for row in rd:
        gid=row['Geneid']; gids[gid]+=1; bios[row['gene_biotype']]+=1
        try: float(row['Length'])
        except: bad_len+=1
print('meta duplicate Geneid rows', sum(v-1 for v in gids.values() if v>1), 'unique',len(gids),'bad length',bad_len)
print('top biotypes', bios.most_common(20))

with open(base+'/Sample_annotated_Zenodo.csv', newline='') as f:
    rd=csv.DictReader(f)
    rows=list(rd)
print('sample celltypes', Counter(r['celltype'] for r in rows))
print('sample columns', rd.fieldnames)
print('duplicate sample ids', [(x,n) for x,n in Counter(r['sample'] for r in rows).items() if n>1][:10])
with open(base+'/BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    hdr=next(csv.reader(f))[1:]
ann={r['sample']:r for r in rows}
print('count samples',len(hdr),'annotation overlap',sum(x in ann for x in hdr),'missing annotation', [x for x in hdr if x not in ann][:20], 'ann missing counts', [x for x in ann if x not in set(hdr)][:20])
print('count sample celltypes', Counter(ann[x]['celltype'] for x in hdr if x in ann))
PY"
