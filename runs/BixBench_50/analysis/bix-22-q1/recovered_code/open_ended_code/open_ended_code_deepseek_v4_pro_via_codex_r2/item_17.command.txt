/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, os
counts=pd.read_csv('/workspace/inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
meta=pd.read_csv('/workspace/inputs/GeneMetaInfo_Zenodo.csv', index_col=0)
samp=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv')
pc_idx=meta.index[meta['gene_biotype']=='protein_coding']
length=meta.loc[pc_idx,'Length'].astype(float)
res={}
for ct in ['CD4','CD8','CD14','CD19']:
    cols=samp.loc[samp['celltype']==ct,'sample'].tolist()
    avg=counts.loc[pc_idx, cols].mean(axis=1)
    r=np.corrcoef(length, avg)[0,1]
    res[ct]=r
    print(ct, 'n_samples', len(cols), 'n_genes', len(pc_idx), 'r', repr(r))
weakest=min(res, key=lambda ct: abs(res[ct]))
print('weakest', weakest)
os.makedirs('/workspace/codex_output', exist_ok=True)
open('/workspace/codex_output/answer.txt','w').write(weakest+'\\n')
print(open('/workspace/codex_output/answer.txt').read())
PY"
