/bin/bash -lc "python - <<'PY'
import pandas as pd
samp=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv')
for ct in ['CD4','CD8','CD14','CD19']:
    s=samp[samp['celltype']==ct]
    dup=s.groupby('samp').size()
    print(ct,'subjects',s['samp'].nunique(),'samples',len(s),'duplicated subjects', (dup>1).sum(), 'max per subject',dup.max())
    print(dup[dup>1].sort_values(ascending=False).head(20).to_dict())
PY"
