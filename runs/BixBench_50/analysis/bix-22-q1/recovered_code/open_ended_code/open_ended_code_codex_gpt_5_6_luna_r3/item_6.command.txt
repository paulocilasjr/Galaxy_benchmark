/bin/bash -lc "python - <<'PY'
import csv, collections, os
from pathlib import Path

base=Path('/workspace/inputs')
# metadata
with open(base/'GeneMetaInfo_Zenodo.csv', newline='') as f:
    rows=list(csv.DictReader(f))
print('metadata rows', len(rows), 'fields', rows[0].keys())
print('metadata first', rows[:2])
for col in ['Geneid','Length','gene_biotype']:
    vals=[r[col] for r in rows]
    print(col, 'missing', sum(v in ('',None) for v in vals), 'unique', len(set(vals)))
print('biotypes', collections.Counter(r['gene_biotype'] for r in rows).most_common(20))
print('duplicate Geneid examples', [(k,v) for k,v in collections.Counter(r['Geneid'] for r in rows).items() if v>1][:10])
print('duplicate first-col examples', [(k,v) for k,v in collections.Counter(r[''] for r in rows).items() if v>1][:10])
# annotation
with open(base/'Sample_annotated_Zenodo.csv', newline='') as f:
    srows=list(csv.DictReader(f))
print('\\nannotation rows',len(srows),'fields',srows[0].keys())
print('celltypes',collections.Counter(r['celltype'] for r in srows))
print('samtypes',collections.Counter(r['samtype'] for r in srows))
print('sample missing',sum(not r['sample'] for r in srows),'unique',len({r['sample'] for r in srows}))
# matrix header and first rows shape, labels
with open(base/'BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    reader=csv.reader(f)
    hdr=next(reader)
    first=[]
    for _,row in zip(range(3),reader): first.append(row)
print('\\nmatrix columns',len(hdr),'first header repr',repr(hdr[:5]),'last',hdr[-5:])
print('matrix first rows lens', [len(r) for r in first], 'labels', [r[0] for r in first])
ann={r['sample']:r for r in srows}
cols=hdr[1:]
print('matrix sample missing annotation',len(set(cols)-set(ann)), list(set(cols)-set(ann))[:20])
print('annotation sample missing matrix',len(set(ann)-set(cols)), list(set(ann)-set(cols))[:20])
print('matrix column celltype counts', collections.Counter(ann[c]['celltype'] for c in cols if c in ann))
for ct in ['CD4','CD8','CD14','CD19']:
    cs=[c for c in cols if c in ann and ann[c]['celltype']==ct]
    print(ct,len(cs),cs[:5])
# label relationship
print('metadata vs matrix labels first 10', [(rows[i][''], rows[i]['Geneid']) for i in range(10)])
PY"
