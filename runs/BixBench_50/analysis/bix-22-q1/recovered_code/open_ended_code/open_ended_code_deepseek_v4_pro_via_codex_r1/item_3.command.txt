/bin/bash -lc "python3 - <<'PY'
import csv
from collections import Counter
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
 r=csv.reader(f)
 header=next(r); print('header', header)
 bt=Counter()
 n=0
 for row in r:
  bt[row[4]]+=1; n+=1
print('rows', n, 'biotypes', len(bt))
for k,v in bt.most_common(20): print(k,v)
with open('inputs/Sample_annotated_Zenodo.csv', newline='') as f:
 r=csv.reader(f); h=next(r); print('ann header',h)
 ct=Counter(); st=Counter(); rows=0
 for row in r:
  ct[row[3]]+=1; st[row[2]]+=1; rows+=1
print('ann rows',rows,'ct',ct,'st',st)
PY"
