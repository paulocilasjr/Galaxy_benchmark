/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
base='inputs'
# Metadata checks
with open(base+'/GeneMetaInfo_Zenodo.csv', newline='') as f:
    meta=list(csv.DictReader(f))
for key in ['', 'Geneid']:
    c=Counter(r[key] for r in meta)
    print('meta key',repr(key),'unique',len(c),'duplicates',sum(v-1 for v in c.values() if v>1),'empty',c[''])
print('meta first/second differing=',sum(r['']"'!=r['"'Geneid'] for r in meta))
print('length nonnumeric=',sum(not r['Length'].strip().isdigit() for r in meta),'length minmax=',min(int(r['Length']) for r in meta),max(int(r['Length']) for r in meta))
# Sample/count header matching
with open(base+'/Sample_annotated_Zenodo.csv',newline='') as f:
    samples=list(csv.DictReader(f))
with open(base+'/BatchCorrectedReadCounts_Zenodo.csv',newline='') as f:
    rd=csv.reader(f); h=next(rd); first_rows=[]
    for i,row in zip(range(5),rd): first_rows.append(row)
count_samples=h[1:]
print('samples exact count/header=',len(set(r['sample'] for r in samples)&set(count_samples)),len(samples),len(count_samples))
print('in annotation not counts=',sorted(set(r['sample'] for r in samples)-set(count_samples))[:20])
print('in counts not annotation=',sorted(set(count_samples)-set(r['sample'] for r in samples))[:20])
print('celltype counts annotation=',Counter(r['celltype'] for r in samples))
print('first count row lengths=',[len(r) for r in first_rows])
# gene match preview
meta_keys=set(r[''] for r in meta)
print('meta/count first rows match=',[(r[0],r[0] in meta_keys) for r in first_rows])
PY"
