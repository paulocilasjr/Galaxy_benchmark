import csv, collections
with open('inputs/Sample_annotated_Zenodo.csv', newline='') as f:

    header=next(csv.reader(f))
cols=header[1:]
print('count cols', len(cols))
ann=set(r['sample'] for r in rows)
print('count cols in annotation', sum(c in ann for c in cols))
print('missing in annotation first20', [c for c in cols if c not in ann][:20])
print('annot not in counts first20', [s for s in ann if s not in cols][:20])
PY"
