/bin/bash -lc "python3 - <<'PY'
import csv, math, statistics, collections
# Read sample annotations and map sample -> celltype
ann={}
with open('inputs/Sample_annotated_Zenodo.csv',newline='') as f:
    r=csv.DictReader(f)
    for d in r:
        ann[d['sample']]=d['celltype']
# Read gene meta: gene id and length for exact protein_coding; preserve order
lengths={}
pc_order=[]
with open('inputs/GeneMetaInfo_Zenodo.csv',newline='') as f:
    r=csv.DictReader(f)
    for d in r:
        if d.get('gene_biotype')=='protein_coding':
            g=d['Geneid']
            L=float(d['Length'])
            lengths[g]=L
            pc_order.append(g)
print('protein coding exact',len(pc_order), 'unique',len(set(pc_order)))
# Read count header, build type positions and sample presence
with open('inputs/BatchCorrectedReadCounts_Zenodo.csv',newline='') as f:
    r=csv.reader(f)
    header=next(r)
    # first col blank/gene
    cols=header[1:]
    print('count cols',len(cols),'unique',len(set(cols)))
    missing=[c for c in cols if c not in ann]
    print('count cols missing annotation',len(missing),missing[:10])
    missingann=[s for s in ann if s not in set(cols)]
    print('ann samples missing counts',len(missingann),missingann[:10])
    typepos={t:[] for t in ['CD4','CD8','CD14','CD19']}
    for idx,c in enumerate(cols,start=1):
        t=ann.get(c)
        if t in typepos:
            typepos[t].append(idx)
    print({t:len(v) for t,v in typepos.items()})
    sums={t:[0.0]*len(pc_order) for t in typepos}
    counts_by_gene={g:i for i,g in enumerate(pc_order)}
    nrows=0; matched=0
    for row in r:
        nrows+=1
        g=row[0]
        i=counts_by_gene.get(g)
        if i is None:
            continue
        matched+=1
        for t,poslist in typepos.items():
            s=0.0; n=0
            for p in poslist:
                try:
                    s += float(row[p]); n+=1
                except Exception:
                    pass
            sums[t][i]=s
    print('rows',nrows,'matched pc',matched)
# Means
means={t:[sums[t][i]/len(typepos[t]) for i in range(len(pc_order))] for t in typepos}
length_vals=[lengths[g] for g in pc_order]
# utility pearson
def pearson(x,y):
    n=len(x)
    mx=statistics.fmean(x); my=statistics.fmean(y)
    sxy=0.0; sxx=0.0; syy=0.0
    for a,b in zip(x,y):
        da=a-mx; db=b-my
        sxy+=da*db; sxx+=da*da; syy+=db*db
    den=math.sqrt(sxx*syy)
    return sxy/den if den else float('nan')
for t in ['CD4','CD8','CD14','CD19']:
    r=pearson(length_vals,means[t])
    print(t, 'pearson raw mean', r, 'abs',abs(r))
# also rank / log transforms quickly for insight
import math
for t in ['CD4','CD8','CD14','CD19']:
    vals=means[t]
    for name, transform in [
        ('log1p(mean)', lambda v: math.log1p(v)),
        ('log10(mean+1)', lambda v: math.log10(v+1)),
        ('log2(mean+1)', lambda v: math.log2(v+1)),
    ]:
        x=[length_vals[i] for i in range(len(vals))]
        y=[transform(v) for v in vals]
        r=pearson(x,y)
        print(t,name,r,abs(r))
    print()
PY"
