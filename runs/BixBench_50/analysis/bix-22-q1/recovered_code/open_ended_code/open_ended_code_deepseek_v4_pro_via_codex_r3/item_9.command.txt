/bin/bash -lc "python3 - <<'PY'
import csv, collections
p='inputs/Sample_annotated_Zenodo.csv'
with open(p,newline='') as f:
    r=csv.reader(f)
    header=next(r)
    print('header',header)
    ct=collections.Counter(); st=collections.Counter(); ba=collections.Counter(); sx=collections.Counter()
    rows=[]
    for row in r:
        if not row: continue
        d=dict(zip(header,row))
        rows.append(d)
        ct[d.get('celltype','')]+=1
        st[d.get('samtype','')]+=1
        ba[d.get('batch','')]+=1
        sx[d.get('sex','')]+=1
print('nrows',len(rows),'unique sample',len({d['sample'] for d in rows}))
print('celltype',ct)
print('samtype',st)
print('batch',ba)
print('sex',sx)
for d in rows[:30]:
    print(d)
PY"
