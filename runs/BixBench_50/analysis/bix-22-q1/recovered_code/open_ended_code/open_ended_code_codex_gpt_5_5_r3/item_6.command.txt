/bin/bash -lc "python - <<'PY'
import csv, collections
path='inputs/Sample_annotated_Zenodo.csv'
with open(path, newline='') as f:
    r=csv.DictReader(f)
    print(r.fieldnames)
    counts={k:collections.Counter() for k in r.fieldnames}
    for row in r:
        for k,v in row.items():
            if len(counts[k])<20 or v in counts[k]: counts[k][v]+=1
            else: counts[k]['<OTHER>']+=1
    for k,c in counts.items():
        print('\\n', k, len(c))
        print(c.most_common(20))
PY"
