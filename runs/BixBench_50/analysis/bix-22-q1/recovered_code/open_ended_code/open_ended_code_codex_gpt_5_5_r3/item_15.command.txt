/bin/bash -lc "python - <<'PY'
# Recompute only final ordering as a lightweight check from saved logic-free assumptions.
import csv, math
cts=('CD4','CD8','CD14','CD19')
length={}
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
    for r in csv.DictReader(f):
        if r['gene_biotype']=='protein_coding': length[r['Geneid']]=float(r['Length'])
sample_ct={}
with open('inputs/Sample_annotated_Zenodo.csv', newline='') as f:
    for r in csv.DictReader(f):
        if r['celltype'] in cts: sample_ct[r['sample']]=r['celltype']
vals={ct:[] for ct in cts}
with open('inputs/BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    rd=csv.reader(f); h=next(rd)
    idx={ct:[i for i,s in enumerate(h) if sample_ct.get(s)==ct] for ct in cts}
    for row in rd:
        if row[0] not in length: continue
        x=length[row[0]]
        for ct,ii in idx.items(): vals[ct].append((x, sum(float(row[i]) for i in ii)/len(ii)))
def r(p):
    n=len(p); sx=sum(x for x,y in p); sy=sum(y for x,y in p); sxx=sum(x*x for x,y in p); syy=sum(y*y for x,y in p); sxy=sum(x*y for x,y in p)
    return (n*sxy-sx*sy)/math.sqrt((n*sxx-sx*sx)*(n*syy-sy*sy))
print({ct:r(vals[ct]) for ct in cts})
print(min(cts, key=lambda ct: abs(r(vals[ct]))))
PY"
