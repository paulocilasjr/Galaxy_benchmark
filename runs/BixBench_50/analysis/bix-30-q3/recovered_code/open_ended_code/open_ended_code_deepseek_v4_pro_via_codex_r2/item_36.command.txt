/bin/bash -c "python - <<'PY'
import xlrd,numpy as np
from scipy import stats
from statsmodels.stats.multitest import multipletests
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
names=[];A=[]
for r in range(2,sh.nrows):
    names.append(sh.cell_value(r,0));vals=[]
    for c in range(1,21):
        v=sh.cell_value(r,c); vals.append(np.nan if isinstance(v,str) and v.strip().upper()=='NAN' else float(v))
    A.append(vals)
A=np.array(A); P=list(range(10)); C=list(range(10,20))
means=np.nanmean(A,axis=0); X=means[None,:]-A
ps=[]
for i in range(A.shape[0]):
    xp=X[i,P];xc=X[i,C];xp=xp[~np.isnan(xp)];xc=xc[~np.isnan(xc)]
    ps.append(stats.ttest_ind(xp,xc,equal_var=True).pvalue)
ps=np.array(ps)
for meth in ['bonferroni','fdr_bh','fdr_by']:
    rej,adj,_,_=multipletests(ps,alpha=.05,method=meth)
    print(meth,'n',rej.sum(),[names[i] for i in np.where(rej)[0]], 'miR21 adj',adj[names.index('hsa-miR-21')])
PY"
