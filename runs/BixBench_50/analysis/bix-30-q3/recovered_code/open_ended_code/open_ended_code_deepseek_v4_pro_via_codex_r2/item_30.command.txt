import xlrd,numpy as np
from scipy import stats
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
names=[];rows=[]
for r in range(2,sh.nrows):
    names.append(sh.cell_value(r,0));vals=[]
    for c in range(1,21):
        v=sh.cell_value(r,c); vals.append(np.nan if isinstance(v,str) and v.strip().upper()=='NAN' else float(v))
    rows.append(vals)
A=np.array(rows); keep=[i for i in range(20) if i not in (2,17)]; A=A[:,keep]
P=[keep.index(i) for i in range(10) if i"'!=2]; C=[keep.index(i) for i in range(10,20) if i!=17]
for norm in ['"'all','detected']:
    means=[]
    for j in range(A.shape[1]):
        v=A[:,j]; fin=~np.isnan(v)
        if norm=='detected': fin &= v<34
        means.append(np.nanmean(v[fin]) if fin.any() else np.nan)
    X=np.array(means)[None,:]-A
    sig=[]
    for i in range(A.shape[0]):
        xp=X[i,P];xc=X[i,C];xp=xp[~np.isnan(xp)];xc=xc[~np.isnan(xc)]
        if len(xp)>=2 and len(xc)>=2:
            p=stats.ttest_ind(xp,xc,equal_var=True).pvalue
            if p<.05:
                direction='up' if np.nanmean(xp)>np.nanmean(xc) else 'down'
                sig.append((p,names[i],direction))
    print(norm,'n',len(sig),'up',sum(d=='up' for p,n,d in sig),'down',sum(d=='down' for p,n,d in sig))
    print([(n,d) for p,n,d in sorted(sig)])
