/bin/bash -c "python - <<'PY'
import urllib.request, re
url='https://pmc.ncbi.nlm.nih.gov/articles/PMC4769077/?report=classic'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
html=urllib.request.urlopen(req,timeout=30).read().decode('utf-8','ignore')
for pat in ['S3 Table','pone.0150501.s003','S1 Table','pone.0150501.s002','Supplementary']:
    print('PAT',pat)
    for m in re.finditer(pat,html,re.I):
        s=max(0,m.start()-300);e=min(len(html),m.end()+500)
        print(html[s:e].replace('\\n',' ')[:800])
        break
PY"
