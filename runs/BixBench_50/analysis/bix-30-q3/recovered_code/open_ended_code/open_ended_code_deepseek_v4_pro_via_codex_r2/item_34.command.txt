/bin/bash -c "python - <<'PY'
import xlrd,numpy as np
from scipy import stats
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
names=[];A=[]
for r in range(2,sh.nrows):
    names.append(sh.cell_value(r,0));vals=[]
    for c in range(1,21):
        v=sh.cell_value(r,c); vals.append(np.nan if isinstance(v,str) and v.strip().upper()=='NAN' else float(v))
    A.append(vals)
A=np.array(A);P=list(range(10));C=list(range(10,20));means=np.nanmean(A,axis=0);X=means[None,:]-A
for alternative in ['two-sided','less','greater']:
    ps=[]
    for i in range(A.shape[0]):
        xp=X[i,P];xc=X[i,C];xp=xp[~np.isnan(xp)];xc=xc[~np.isnan(xc)]
        ps.append(stats.ttest_ind(xp,xc,equal_var=True,alternative=alternative).pvalue)
    ps=np.array(ps)
    print(alternative,'raw',np.sum(ps<.05),'min',ps.min())
    for meth in ['bonf','bh','by']:
        m=len(ps);order=np.argsort(ps);q=np.empty(m);prev=1
        factor=m if meth=='bonf' else (m if meth=='bh' else m*sum(1/k for k in range(1,m+1)))
        if meth=='bonf':
            q=np.minimum(ps*m,1)
        else:
            for rank in range(m,0,-1):
                idx=order[rank-1]; val=min(factor/rank*ps[idx],1);prev=min(prev,val);q[idx]=prev
        sig=np.where(q<=.05)[0]
        print(' ',meth,len(sig),[names[i] for i in sig])
PY"
