from scipy import stats
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
names=[]; rows=[]
for r in range(2,sh.nrows):
    names.append(sh.cell_value(r,0))
    vals=[]
    for c in range(1,21):
        v=sh.cell_value(r,c)
        vals.append(np.nan if isinstance(v,str) and v.strip().upper()=='NAN' else (float(v) if isinstance(v,(int,float)) else np.nan))

A=np.array(rows)
keep=[i for i in range(20) if i not in (2,17)]
A=A[:,keep]
P=[keep.index(i) for i in range(10) if i"'!=2]
C=[keep.index(i) for i in range(10,20) if i!=17]
names=np.array(names)

def means_norm(norm):
    means=[]
    for j in range(A.shape[1]):
        vals=A[:,j]; finite=~np.isnan(vals)
        if norm=='"'detected': finite &= vals<34
        means.append(np.nanmean(vals[finite]) if finite.any() else np.nan)
    return np.array(means)

def count_raw(X, policy):
    pvals=[]; ns=[]
    for i in range(A.shape[0]):
        x=X[i]
        xp=x[P]; xc=x[C]

            if np.isnan(x[P]).any() or np.isnan(x[C]).any(): continue

        xp=xp[~np.isnan(xp)]; xc=xc[~np.isnan(xc)]
        if len(xp)>=2 and len(xc)>=2:
            pvals.append(stats.ttest_ind(xp,xc,equal_var=True).pvalue); ns.append(names[i])
    return np.array(pvals),ns

for norm in ['all','detected']:

        p,ns=count_raw(X,policy)

        sig=[ns[i] for i in np.where(p<.05)[0]]
