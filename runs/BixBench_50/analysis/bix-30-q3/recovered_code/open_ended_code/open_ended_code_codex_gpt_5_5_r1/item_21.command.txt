/bin/bash -lc "python - <<'PY'
import xlrd, subprocess, csv, io
s=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO(); w=csv.writer(out, delimiter='\\t', lineterminator='\\n')
w.writerow(['miRNA']+[s.cell_value(1,c) for c in range(1,s.ncols)])
for r in range(2,s.nrows): w.writerow([s.cell_value(r,0)]+[s.cell_value(r,c) for c in range(1,s.ncols)])
rscript=r'''
d <- read.delim('stdin', check.names=FALSE)
P <- grep('"'^P_'"', names(d)); C <- grep('"'^C_'"', names(d))
p <- vapply(seq_len(nrow(d)), function(i) t.test(as.numeric(d[i,P]), as.numeric(d[i,C]), paired=TRUE)"'$p.value, numeric(1))
cat('"'paired minp=', signif(min(p),4), 'raw=', sum(p<=0.05), 'bon=', sum(p.adjust(p,'bonferroni')<=0.05), 'BY=', sum(p.adjust(p,'BY')<=0.05), 'BH=', sum(p.adjust(p,'BH')<=0.05), '\\n')
ord <- order(p)[1:10]
print(data.frame(miRNA=d"'$miRNA[ord], p=signif(p[ord],4), bon=signif(p.adjust(p,'"'bonferroni')[ord],4), by=signif(p.adjust(p,'BY')[ord],4)))
'''
res=subprocess.run(['Rscript','-e',rscript], input=out.getvalue(), text=True, capture_output=True)
print(res.stdout)
if res.stderr: print('STDERR:',res.stderr)
PY"
