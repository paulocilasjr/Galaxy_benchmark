/bin/bash -lc "python - <<'PY'
import xlrd,numpy as np
from scipy.stats import ttest_ind
s=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
names=[s.cell_value(i,0) for i in range(2,s.nrows)]
X=np.array([[float(s.cell_value(i,j)) if isinstance(s.cell_value(i,j),(int,float)) else np.nan for j in range(1,s.ncols)] for i in range(2,s.nrows)])
samp=[s.cell_value(1,j) for j in range(1,s.ncols)]
for excl in [True,False]:
 keep=np.ones(20,bool)
 if excl:
  keep[samp.index('P_3')]=False; keep[samp.index('C_18')]=False
 for det in [False,True]:
  Y=np.where(X<34,X,np.nan) if det else X
  for equal in [True,False]:
   ps=[]
   for r in Y:
    a=r[np.array([q.startswith('P_') for q in samp])&keep];a=a[np.isfinite(a)]
    b=r[np.array([q.startswith('C_') for q in samp])&keep];b=b[np.isfinite(b)]
    ps.append(ttest_ind(a,b,equal_var=equal).pvalue)
   ps=np.array(ps); finite=np.isfinite(ps); m=finite.sum()
   bon=np.minimum(ps*m,1)
   c=np.sum(1/np.arange(1,m+1)); order=np.argsort(ps[finite]); ss=ps[finite][order]*m*c/np.arange(1,m+1); qq=np.minimum.accumulate(ss[::-1])[::-1]
   print(excl,det,equal,'m',m,'raw',np.sum(ps<=.05),'bon',np.sum(bon<=.05),'BY',np.sum(qq<=.05),'min',names[np.nanargmin(ps)],np.nanmin(ps))
PY"
