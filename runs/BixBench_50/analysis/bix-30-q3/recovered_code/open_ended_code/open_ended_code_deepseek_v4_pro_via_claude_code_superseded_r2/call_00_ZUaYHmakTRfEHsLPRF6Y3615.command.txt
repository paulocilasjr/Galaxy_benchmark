import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_excel('inputs/pone.0150501.s002.xls')

p_values = []
t_stats = []
mir_names = []

for i in range(1, len(df)):
    mir = df.iloc[i, 0]
    if pd.isna(mir):
        continue
    
    p_vals_raw = df.iloc[i, 1:11]
    c_vals_raw = df.iloc[i, 11:21]
    
    p_vals = pd.to_numeric(p_vals_raw, errors='coerce').dropna().values.astype(float)
    c_vals = pd.to_numeric(c_vals_raw, errors='coerce').dropna().values.astype(float)
    
    if len(p_vals) < 2 or len(c_vals) < 2:
        continue
    
    t_stat, p_val = stats.ttest_ind(p_vals, c_vals, equal_var=False)
    p_values.append(p_val)
    t_stats.append(t_stat)
    mir_names.append(mir)

p_values = np.array(p_values)

# Show p-value distribution
print('P-value distribution:')
for threshold in [0.001, 0.01, 0.05, 0.1, 0.2, 0.5]:
    count = np.sum(p_values <= threshold)
    print(f'  p <= {threshold}: {count}')

# Show top 10 most significant
sorted_idx = np.argsort(p_values)
print()
print('Top 10 most significant miRNAs:')
for i in range(10):
    idx = sorted_idx[i]
    print(f'  {mir_names[idx]}: p={p_values[idx]:.6f}, t={t_stats[idx]:.4f}')

# Print smallest Bonferroni-corrected p-values
bonf_p = np.minimum(p_values * len(p_values), 1.0)
sorted_bonf = np.sort(bonf_p)
print()
print('Smallest Bonferroni p-values:')
for i in range(10):
    print(f'  {sorted_bonf[i]:.4f}')
