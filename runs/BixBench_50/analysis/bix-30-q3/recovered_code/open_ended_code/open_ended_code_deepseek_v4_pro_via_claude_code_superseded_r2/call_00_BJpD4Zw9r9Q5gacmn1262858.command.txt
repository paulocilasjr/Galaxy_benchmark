import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_excel('inputs/pone.0150501.s002.xls')

# Raw data indices (in the original df):
# Col 0: miRNA name
# Cols 1-10: P_1 through P_10
# Cols 11-20: C_11 through C_20

# The data DataFrame (df.iloc[1:, 1:]) has:
# data col 0 = orig col 1 = P_1
# data col 1 = orig col 2 = P_2
# data col 2 = orig col 3 = P_3 (EXCLUDE)
# ...
# data col 9 = orig col 10 = P_10
# data col 10 = orig col 11 = C_11
# data col 16 = orig col 17 = C_17
# data col 17 = orig col 18 = C_18 (EXCLUDE)
# data col 18 = orig col 19 = C_19
# data col 19 = orig col 20 = C_20

patient_cols_data = [0, 1, 3, 4, 5, 6, 7, 8, 9]  # skip data col 2 = P_3
control_cols_data = [10, 11, 12, 13, 14, 15, 16, 18, 19]  # skip data col 17 = C_18

data = df.iloc[1:, 1:].apply(pd.to_numeric, errors='coerce')
mir_names = df.iloc[1:, 0].values

# Verify column mapping
print('Patient columns in data:')
for c in patient_cols_data:
    print(f'  data[{c}] = orig[{c+1}] = {df.columns[c+1]}')
print('Control columns in data:')
for c in control_cols_data:
    print(f'  data[{c}] = orig[{c+1}] = {df.columns[c+1]}')

# Step 1: Per-sample mean of detected miRNAs (Ct < 34)
all_data_cols = patient_cols_data + control_cols_data
sample_means = {}
for c in all_data_cols:
    vals = data.iloc[:, c].dropna().values.astype(float)
    detected = vals[vals < 34]
    sample_means[c] = np.mean(detected)
    print(f'  {df.columns[c+1]}: mean={sample_means[c]:.4f}, detected={len(detected)}/{len(vals)}')

# Step 2: ΔCt t-test
p_values = []
analysis_results = []

for i in range(data.shape[0]):
    if pd.isna(mir_names[i]):
        continue
    
    mir = mir_names[i]
    
    p_delta = []
    for c in patient_cols_data:
        val = data.iloc[i, c]
        if pd.notna(val):
            p_delta.append(float(val) - sample_means[c])
    
    c_delta = []
    for c in control_cols_data:
        val = data.iloc[i, c]
        if pd.notna(val):
            c_delta.append(float(val) - sample_means[c])
    
    p_delta = np.array(p_delta)
    c_delta = np.array(c_delta)
    
    if len(p_delta) < 2 or len(c_delta) < 2:
        continue
    
    t_stat, p_val = stats.ttest_ind(p_delta, c_delta, equal_var=True)
    p_values.append(p_val)
    
    mean_diff = np.mean(p_delta) - np.mean(c_delta)
    fc = 2**(-mean_diff)
    analysis_results.append((mir, np.mean(p_delta), np.mean(c_delta), mean_diff, fc, p_val, t_stat))

p_values = np.array(p_values)
n_tests = len(p_values)

print(f'\nTotal miRNAs: {n_tests}')
print(f'Raw p <= 0.05: {np.sum(p_values <= 0.05)}')

# Sort
sorted_results = sorted(analysis_results, key=lambda x: x[5])
print('\nTop 10:')
for mir, pm, cm, diff, fc, pv, ts in sorted_results[:10]:
    print(f'  {mir}: FC={fc:.3f}, p={pv:.6f}')

# Bonferroni
bonf_p = np.minimum(p_values * n_tests, 1.0)
bonf_sig = int(np.sum(bonf_p <= 0.05))
print(f'\nBonferroni significant: {bonf_sig}')

# Show which
for i in np.where(bonf_p <= 0.05)[0]:
    mir = analysis_results[i][0]
    print(f'  {mir}: p={analysis_results[i][5]:.6f}, adj_p={bonf_p[i]:.6f}')

# Benjamini-Yekutieli
sorted_idx = np.argsort(p_values)
sorted_p = p_values[sorted_idx]

n = n_tests
harmonic = np.sum(1.0 / np.arange(1, n + 1))

by_p = np.ones(n)
for i in range(n):
    rank = i + 1
    by_p[i] = min(sorted_p[i] * n * harmonic / rank, 1.0)

for i in range(n - 2, -1, -1):
    by_p[i] = min(by_p[i], by_p[i + 1])

by_sig = int(np.sum(by_p <= 0.05))
print(f'\nBY significant: {by_sig}')
for idx in np.where(by_p <= 0.05)[0]:
    mir = [analysis_results[j][0] for j in sorted_idx][idx]
    print(f'  {mir}: p={sorted_p[idx]:.6f}, BY_adj={by_p[idx]:.6f}')

print(f'\nANSWER: {bonf_sig}:{by_sig}')
