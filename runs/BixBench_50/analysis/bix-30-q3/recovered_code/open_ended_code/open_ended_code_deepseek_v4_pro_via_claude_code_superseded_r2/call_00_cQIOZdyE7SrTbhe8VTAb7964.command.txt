import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_excel('inputs/pone.0150501.s002.xls')

data = df.iloc[1:, 1:].apply(pd.to_numeric, errors='coerce')
mir_names = df.iloc[1:, 0].values

# Raw Ct t-test (no normalization)
p_values = []
meantable = []
for i in range(data.shape[0]):
    p_vals = data.iloc[i, 0:10].dropna().values.astype(float)
    c_vals = data.iloc[i, 10:20].dropna().values.astype(float)
    
    if len(p_vals) < 2 or len(c_vals) < 2:
        continue
    
    t_stat, p_val = stats.ttest_ind(p_vals, c_vals, equal_var=False)
    p_values.append(p_val)
    # mean difference (patient - control) for Ct: negative = higher expression in patients
    meantable.append((mir_names[i], np.mean(p_vals), np.mean(c_vals), np.mean(p_vals) - np.mean(c_vals), p_val))

p_values = np.array(p_values)
n_tests = len(p_values)

# Sort by p-value
meantable_sorted = sorted(meantable, key=lambda x: x[4])
print('Top 20 by p-value:')
for mir, pm, cm, diff, pv in meantable_sorted[:20]:
    direction = 'UP in patients' if diff < 0 else 'DOWN in patients'
    print(f'  {mir}: diff={diff:.3f} ({direction}), p={pv:.6f}')
