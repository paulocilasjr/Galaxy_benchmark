from scipy import stats
from statsmodels.stats.multitest import multipletests
p='/workspace/inputs/pone.0150501.s002.xls'
sh=xlrd.open_workbook(p).sheet_by_index(0)

        else:
            nums.append(np.nan)

A=np.array(mat, dtype=float)
P=[i for i,h in enumerate(headers) if h.startswith('P_')]
C=[i for i,h in enumerate(headers) if h.startswith('C_')]
exP={'P_3'}; exC={'C_18'}

def analysis(center_mode='allmean', exclude=False, threshold=None, sign='sample-gene', equal_var=True):
    B=A.copy()
    if threshold is not None:
        B[B>=threshold]=np.nan
    pidx=[i for i in P if not exclude or headers[i] not in exP]
    cidx=[i for i in C if not exclude or headers[i] not in exC]
    # center per sample based on available values among all 175
    if center_mode=='allmean':
        sample_means=np.nanmean(B, axis=0)

        # mean Ct of detected (<34) per sample; no threshold applied to B?

    elif center_mode=='none':
        sample_means=np.zeros(B.shape[1])
    if center_mode"'!='"'none':
