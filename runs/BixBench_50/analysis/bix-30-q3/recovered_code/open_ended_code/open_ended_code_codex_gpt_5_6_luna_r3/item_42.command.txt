/bin/bash -lc "python - <<'PY'
import xlrd,csv,io,subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO();w=csv.writer(out,delimiter='\\t',lineterminator='\\n');w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows):w.writerow([sh.cell_value(r,0)]+['NA' if isinstance(sh.cell_value(r,c),str) else sh.cell_value(r,c) for c in range(1,sh.ncols)])
rscript=r'''
d<-read.delim(file(\"stdin\"),check.names=FALSE,na.strings=\"NA\");allP<-grep(\""'^P_",names(d));allC<-grep("''^C_",names(d));
for(drop in c(FALSE,TRUE)){P<-allP;C<-allC;if(drop){P<-P[names(d)[P]!="P_3"];C<-C[names(d)[C]!="C_18"]};X<-as.matrix(d[c(P,C)]);for(cut in c(FALSE,TRUE)){means<-apply(X,2,function(z){z<-as.numeric(z);mean(z[is.finite(z)&(!cut|z<34)])});for(trans in c('"'delta','exp')){if(trans=='delta')E<-sweep(X,2,means,'-')*(-1) else E<-2"'^sweep(-X,2,means,'"'+');p<-apply(E,1,function(z){x<-z[P];y<-z[C];ok<-is.finite(x)&is.finite(y);if(sum(ok)<2)return(NA);t.test(x[ok],y[ok],paired=TRUE)"'$p.value});cat('"'drop',drop,'cut',cut,'trans',trans,'m',sum(is.finite(p)),'raw',sum(p<=.05,na.rm=T),'bonf',sum(p.adjust(p,'bonferroni')<=.05,na.rm=T),'BY',sum(p.adjust(p,'BY')<=.05,na.rm=T),'min',min(p,na.rm=T),'\\\\n')}}}
'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout);print(res.stderr)
PY"
