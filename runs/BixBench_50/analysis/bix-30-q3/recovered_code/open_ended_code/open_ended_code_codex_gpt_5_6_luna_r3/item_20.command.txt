/bin/bash -lc "python - <<'PY'
import xlrd,csv,io,subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO(); w=csv.writer(out,delimiter='\\t',lineterminator='\\n')
w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows):
 row=[sh.cell_value(r,0)]
 for c in range(1,sh.ncols):
  v=sh.cell_value(r,c); row.append('NA' if isinstance(v,str) else v)
 w.writerow(row)
rscript=r'''
d <- read.delim(file(\"stdin\"),check.names=FALSE,na.strings=\"NA\")
allP <- grep(\""'^P_",names(d)); allC <- grep("''^C_",names(d))
for (drop_bad in c(FALSE,TRUE)) for (drop_high in c(FALSE,TRUE)) for (mean_cut in c(FALSE,TRUE)) {
  P <- allP; C <- allC
  if (drop_bad) { P <- P[names(d)[P]!="P_3"]; C <- C[names(d)[C]!="C_18"] }
  X <- as.matrix(d[c(P,C)])
  # normalize by per-sample mean Ct for values <34 or all numeric
  means <- apply(X,2,function(z) { z <- as.numeric(z); z <- z[is.finite(z) & (!mean_cut | z<34)]; mean(z) })
  E <- sweep(X,2,means,"-") * -1 # log2 relative expression, sign immaterial
  if (drop_high) E[E <= -34+means[col(E)]] <- NA # wrong rowwise may not matter
  # Above drop high using original X instead
  if (drop_high) { E[is.finite(X) & X>=34] <- NA }
  p <- apply(E,1,function(z) {
    x <- as.numeric(z[P]); y <- as.numeric(z[C]); x<-x[is.finite(x)]; y<-y[is.finite(y)]
    t.test(x,y,var.equal=TRUE)$p.value
  })
  cat(sprintf("drop_bad=%s drop_high=%s mean_cut=%s raw=%d bonf=%d BY=%d min=%.8g'"\\n\",drop_bad,drop_high,mean_cut,sum(p<=.05),sum(p.adjust(p,\"bonferroni\")<=.05),sum(p.adjust(p,\"BY\")<=.05),min(p)))
}
'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout); print(res.stderr); print('status',res.returncode)
PY"
