/bin/bash -lc "python - <<'PY'
import xlrd,csv,io,subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO();w=csv.writer(out,delimiter='\\t',lineterminator='\\n')
w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows):
 row=[sh.cell_value(r,0)]+['NA' if isinstance(sh.cell_value(r,c),str) else sh.cell_value(r,c) for c in range(1,sh.ncols)];w.writerow(row)
rscript=r'''
d<-read.delim(file(\"stdin\"),check.names=FALSE,na.strings=\"NA\");P<-grep(\""'^P_",names(d));C<-grep("''^C_",names(d));i<-which(d$miRNA=="hsa-miR-21")
for(drop in c(FALSE,TRUE)) {pp<-P;cc<-C;if(drop){pp<-pp[names(d)[pp]!="P_3"];cc<-cc[names(d)[cc]!="C_18"]};x<-as.numeric(d[i,pp]);y<-as.numeric(d[i,cc]);print(c(drop,t.test(x,y,var.equal=TRUE)$p.value,t.test(x,y,var.equal=FALSE)$p.value,mean(x),mean(y),sd(x),sd(y)))}
'"'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout);print(res.stderr)
PY"
