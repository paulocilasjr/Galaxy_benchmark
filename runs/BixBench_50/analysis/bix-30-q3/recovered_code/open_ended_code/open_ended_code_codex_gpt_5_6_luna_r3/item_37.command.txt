/bin/bash -lc "python - <<'PY'
import xlrd,csv,io,subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO();w=csv.writer(out,delimiter='\\t',lineterminator='\\n');w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows):w.writerow([sh.cell_value(r,0)]+['NA' if isinstance(sh.cell_value(r,c),str) else sh.cell_value(r,c) for c in range(1,sh.ncols)])
rscript=r'''
d<-read.delim(file(\"stdin\"),check.names=FALSE,na.strings=\"NA\");allP<-grep(\""'^P_",names(d));allC<-grep("''^C_",names(d));
for(drop in c(FALSE,TRUE)){P<-allP;C<-allC;if(drop){P<-P[names(d)[P]!="P_3"];C<-C[names(d)[C]!="C_18"]};X<-as.matrix(d[c(P,C)]);for(cut in c(FALSE,TRUE))for(transform in c('"'delta','exp','logct','ratio')){
means<-apply(X,2,function(z){z<-as.numeric(z);mean(z[is.finite(z)&("'!cut|z<34)])});
if(transform=='"'delta') E<-sweep(X,2,means,'-')*(-1)
if(transform=='exp') E<-2"'^sweep(-X,2,means,'"'+')
if(transform=='logct') E<-log2(X)
if(transform=='ratio') E<-sweep(X,2,means,'/')
p<-apply(E,1,function(z){x<-z[P];y<-z[C];x<-x[is.finite(x)];y<-y[is.finite(y)];t.test(x,y,var.equal=TRUE)"'$p.value});i<-which(d$miRNA=='"'hsa-miR-21');cat(transform,'drop',drop,'cut',cut,'raw',sum(p<=.05),'bonf',sum(p.adjust(p,'bonferroni')<=.05),'BY',sum(p.adjust(p,'BY')<=.05),'min',min(p),'p21',p[i],'\\\\n')}}
'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout);print(res.stderr)
PY"
