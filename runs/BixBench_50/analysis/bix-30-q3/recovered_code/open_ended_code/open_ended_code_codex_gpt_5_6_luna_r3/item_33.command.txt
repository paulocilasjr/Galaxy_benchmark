/bin/bash -lc "python - <<'PY'
import xlrd,csv,io,subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO();w=csv.writer(out,delimiter='\\t',lineterminator='\\n')
w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows): w.writerow([sh.cell_value(r,0)]+['NA' if isinstance(sh.cell_value(r,c),str) else sh.cell_value(r,c) for c in range(1,sh.ncols)])
rscript=r'''
d<-read.delim(file(\"stdin\"),check.names=FALSE,na.strings=\"NA\");allP<-grep(\""'^P_",names(d));allC<-grep("''^C_",names(d));P<-allP[names(d)[allP]!="P_3"];C<-allC[names(d)[allC]!="C_18"];X<-as.matrix(d[c(P,C)]);means<-apply(X,2,function(z){z<-as.numeric(z);mean(z[is.finite(z)])});E<-sweep(X,2,means,"-")*(-1);p<-apply(E,1,function(z){x<-z[P];y<-z[C];x<-x[is.finite(x)];y<-y[is.finite(y)];t.test(x,y,var.equal=TRUE)$p.value});ix<-order(p)[1:10];print(data.frame(miRNA=d$miRNA[ix],p=p[ix],bonf=p.adjust(p,'"'bonferroni')[ix],BY=p.adjust(p,'BY')[ix]));print(data.frame(miRNA=d"'$miRNA[which(p.adjust(p,'"'bonferroni')<=.05)],p=p[p.adjust(p,'bonferroni')<=.05]))
'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout);print(res.stderr)
PY"
