/bin/bash -lc "python - <<'PY'
import xlrd, csv, io, subprocess, textwrap
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO(); w=csv.writer(out, delimiter='\\t', lineterminator='\\n')
w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows):
    row=[sh.cell_value(r,0)]
    for c in range(1,sh.ncols):
        v=sh.cell_value(r,c)
        row.append('NA' if isinstance(v,str) else v)
    w.writerow(row)
rscript=r'''
d <- read.delim(file(\"stdin\"), check.names=FALSE, na.strings=\"NA\")
P <- grep(\""'^P_", names(d)); C <- grep("''^C_", names(d))
methods <- list(
  welch=function(x,y) t.test(x,y, var.equal=FALSE)$p.value,
  student=function(x,y) t.test(x,y, var.equal=TRUE)$p.value,
  wilcox=function(x,y) wilcox.test(x,y, exact=FALSE)$p.value
)
for (nm in names(methods)) {
  p <- apply(d[c(P,C)],1,function(z) {
    x <- as.numeric(z[P]); y <- as.numeric(z[C]); x <- x[is.finite(x)]; y <- y[is.finite(y)]
    methods[[nm]](x,y)
  })
  b <- p.adjust(p, method="bonferroni")
  by <- p.adjust(p, method="BY")
  cat(nm, "raw<=.05",sum(p<=.05), "bonf<=.05",sum(b<=.05), "BY<=.05",sum(by<=.05), "min",format(min(p),digits=16),"'"\\n\")
  ix <- order(p)[1:min(15,length(p))]
  cat(paste(sprintf(\"%s=%.8g\",d"'$miRNA[ix],p[ix]), collapse="; "),"'"\\n\")
}
'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout)
print(res.stderr)
print('status',res.returncode)
PY"
