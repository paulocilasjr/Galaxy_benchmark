import xlrd,numpy as np
from scipy.stats import ttest_ind
s=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
X=np.array([[float(s.cell_value(i,j)) if isinstance(s.cell_value(i,j),(int,float)) else np.nan for j in range(1,21)] for i in range(2,s.nrows)])
X[X>=34]=np.nan
M=np.nanmean(X,axis=0); Z=M-X
P=[0,1,3,4,5,6,7,8,9];C=[10,11,12,13,14,15,16,18,19]
p=np.array([ttest_ind(r[P],r[C],equal_var=True,nan_policy='omit').pvalue for r in Z])
for family in [p,p[p<=.05]]:
 m=len(family); h=sum(1/i for i in range(1,m+1)); o=np.argsort(family); sp=family[o]; q=np.minimum.accumulate((sp*m*h/np.arange(1,m+1))[::-1])[::-1]
 print(m,'bonf',sum(family*m<=.05),'by',sum(q<=.05), 'minby',q.min())
PY"
