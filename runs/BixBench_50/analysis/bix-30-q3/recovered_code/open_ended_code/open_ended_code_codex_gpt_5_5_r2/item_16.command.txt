import xlrd, math

# Regularized incomplete beta implementation for Student-t CDF checks.
def betacf(a,b,x):
    MAXIT=200
    EPS=3e-14
    FPMIN=1e-300
    qab=a+b
    qap=a+1.0
    qam=a-1.0
    c=1.0
    d=1.0-qab*x/qap
    if abs(d)<FPMIN: d=FPMIN
    d=1.0/d
    h=d
    for m in range(1,MAXIT+1):
        m2=2*m
        aa=m*(b-m)*x/((qam+m2)*(a+m2))
        d=1.0+aa*d
        if abs(d)<FPMIN: d=FPMIN
        c=1.0+aa/c
        if abs(c)<FPMIN: c=FPMIN
        d=1.0/d
        h*=d*c
        aa=-(a+m)*(qab+m)*x/((a+m2)*(qap+m2))
        d=1.0+aa*d
        if abs(d)<FPMIN: d=FPMIN
        c=1.0+aa/c
        if abs(c)<FPMIN: c=FPMIN
        d=1.0/d
        delh=d*c
        h*=delh
        if abs(delh-1.0) < EPS:
            break
    else:
        raise RuntimeError('betacf did not converge')
    return h

def betai(a,b,x):
    if x<0 or x>1: raise ValueError(x)
    if x==0 or x==1: return x
    bt=math.exp(math.lgamma(a+b)-math.lgamma(a)-math.lgamma(b)+a*math.log(x)+b*math.log1p(-x))
    if x < (a+1)/(a+b+2):
        return bt*betacf(a,b,x)/a
    else:
        return 1-bt*betacf(b,a,1-x)/b

def t_two_sided_p(t, v):
    x=v/(v+t*t)
    # two-sided p = I_x(v/2, 1/2) for |t|
    return betai(v/2, 0.5, x)

def welch_p(a,b):
    n1,n2=len(a),len(b)
    m1=sum(a)/n1; m2=sum(b)/n2
    v1=sum((x-m1)**2 for x in a)/(n1-1)
    v2=sum((x-m2)**2 for x in b)/(n2-1)
    se=math.sqrt(v1/n1+v2/n2)
    t=(m1-m2)/se
    df=(v1/n1+v2/n2)**2/((v1/n1)**2/(n1-1)+(v2/n2)**2/(n2-1))
    return t,df,t_two_sided_p(abs(t),df)

def student_p(a,b):
    n1,n2=len(a),len(b)
    m1=sum(a)/n1; m2=sum(b)/n2
    v1=sum((x-m1)**2 for x in a)/(n1-1)
    v2=sum((x-m2)**2 for x in b)/(n2-1)
    sp=((n1-1)*v1+(n2-1)*v2)/(n1+n2-2)
    se=math.sqrt(sp*(1/n1+1/n2))
    t=(m1-m2)/se
    df=n1+n2-2
    return t,df,t_two_sided_p(abs(t),df)

def adjust_bonf(p):
    m=len(p)
    return [min(x*m,1.0) for x in p]

def adjust_by(p):
    m=len(p)
    cm=sum(1/i for i in range(1,m+1))
    order=sorted(range(m), key=lambda i:p[i])
    adj=[None]*m
    prev=1.0
    for rank,i in reversed(list(enumerate(order, start=1))):
        val=p[i]*m*cm/rank
        if val>prev: val=prev
        if val>1: val=1.0
        prev=val
        adj[i]=val
    return adj

s=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
headers=s.row_values(1)
P=[i for i,h in enumerate(headers) if isinstance(h,str) and h.startswith('P_')]
C=[i for i,h in enumerate(headers) if isinstance(h,str) and h.startswith('C_')]
for method, fn in [('welch',welch_p),('student',student_p)]:
    rows=[]; ps=[]
    for r in range(2,s.nrows):
        name=s.cell_value(r,0)
        a=[s.cell_value(r,c) for c in P if isinstance(s.cell_value(r,c),float) and not math.isnan(s.cell_value(r,c))]
        b=[s.cell_value(r,c) for c in C if isinstance(s.cell_value(r,c),float) and not math.isnan(s.cell_value(r,c))]
        t,df,p=fn(a,b)
        rows.append((name,p,t,df,len(a),len(b)))
        ps.append(p)
    bon=adjust_bonf(ps); by=adjust_by(ps)
    cb=sum(x<=0.05 for x in bon)
    cy=sum(x<=0.05 for x in by)
    print(method, 'm', len(ps), 'Bonf', cb, 'BY', cy, 'ratio', f'{cb}:{cy}')
    for name,p,t,df,n1,n2 in sorted(rows, key=lambda x:x[1])[:10]:
        print(' ',name,'p=',format(p,'.4g'),'t=',format(t,'.3f'),'df=',format(df,'.3f'),'n=',n1,n2)
PY"
