import xlrd, subprocess, json
s=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)

P=[i for i,h in enumerate(headers) if isinstance(h,str) and h.startswith('P_')]

    b=[s.cell_value(r,c) for c in C if isinstance(s.cell_value(r,c),float)]
    ap=[]; bp=[]
    for pc,cc in zip(P,C):
        av=s.cell_value(r,pc); bv=s.cell_value(r,cc)
        if isinstance(av,float) and isinstance(bv,float):
            ap.append(av); bp.append(bv)
    fmt=lambda xs:'c('+','.join(format(x,'.15g') for x in xs)+')'
    lines.append(f'names <- c(names, {json.dumps(name)})')
    lines.append(f'ps_w <- c(ps_w, t.test({fmt(a)}, {fmt(b)}, var.equal=FALSE)"'$p.value)'"')
    lines.append(f'ps_s <- c(ps_s, t.test({fmt(a)}, {fmt(b)}, var.equal=TRUE)"'$p.value)'"')
    lines.append(f'ps_paired <- c(ps_paired, t.test({fmt(ap)}, {fmt(bp)}, paired=TRUE)"'$p.value)'"')

PY"
