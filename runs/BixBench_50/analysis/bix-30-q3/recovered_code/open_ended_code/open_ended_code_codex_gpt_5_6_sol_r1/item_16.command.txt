/bin/bash -lc "python - <<'PY'
import pandas as pd,numpy as np
from scipy.stats import ttest_ind
from statsmodels.stats.multitest import multipletests
p='inputs/pone.0150501.s002.xls';d=pd.read_excel(p,header=1);d=d.rename(columns={d.columns[0]:'miRNA'}).set_index('miRNA').apply(pd.to_numeric,errors='coerce');d=d.drop(columns=['P_3','C_18']);P=[c for c in d if c[0]=='P'];C=[c for c in d if c[0]=='C']
for repl in [None,34,35,40]:
 for cap in [False,True]:
  x=d.copy()
  if cap:x=x.clip(upper=34)
  if repl is not None:x=x.fillna(repl)
  for mm in ['mean_x','mean_orig_lt34','mean_orig_all']:
   if mm=='mean_x':m=x.mean()
   elif mm=='mean_orig_lt34':m=d.where(d<34).mean()
   else:m=d.mean()
   z=m-x
   for eq in [True,False]:
    ps=[]
    for _,r in z.iterrows():
     a=r[P].dropna();b=r[C].dropna();ps.append(ttest_ind(a,b,equal_var=eq).pvalue if len(a)>1 and len(b)>1 else np.nan)
    ps=np.asarray(ps);ok=np.isfinite(ps); bon=multipletests(ps[ok],method='bonferroni')[1];by=multipletests(ps[ok],method='fdr_by')[1]
    if sum(ps<=.05) in range(33,38):print('repl',repl,'cap',cap,mm,'eq',eq,'raw',sum(ps<=.05),'bon/by',sum(bon<=.05),sum(by<=.05),'min',np.nanmin(ps))
PY"
