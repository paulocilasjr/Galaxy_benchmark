import pandas as pd,numpy as np
from scipy.stats import ttest_ind

print('m',M,'H',H)
for rank,(i,pv) in enumerate(ps[:15],1): print(rank,i,'p',pv,'bon',min(pv*M,1),'BY raw multiplier',min(pv*M*H/rank,1),'BY rejection threshold',rank*.05/(M*H))
