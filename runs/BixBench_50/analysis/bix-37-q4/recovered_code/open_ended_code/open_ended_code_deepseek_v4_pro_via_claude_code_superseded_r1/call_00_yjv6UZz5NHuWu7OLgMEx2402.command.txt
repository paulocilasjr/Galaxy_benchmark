import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/Proteomic_data .xlsx', data_only=True)
ws = wb['Tumor vs Normal']

# Find ENO1 and get the exact cell value for log2FC
for row in ws.iter_rows(min_row=2):
    gene = row[2].value  # gene column
    if gene and str(gene).strip().upper() == 'ENO1':
        log2fc_cell = row[8]  # log2FC column (0-indexed: col I)
        print(f"log2FC value: {log2fc_cell.value}")
        print(f"log2FC type: {type(log2fc_cell.value)}")
        print(f"log2FC number_format: {log2fc_cell.number_format}")
        
        # Also check Ratio for completeness
        ratio_cell = row[6]
        fc_cell = row[7]
        print(f"Ratio value: {ratio_cell.value} (type: {type(ratio_cell.value)})")
        print(f"FC value: {fc_cell.value} (type: {type(fc_cell.value)})")
        
        # Also print raw Normal and Tumor
        normal_cell = row[4]
        tumor_cell = row[5]
        print(f"Normal: {normal_cell.value}")
        print(f"Tumor: {tumor_cell.value}")
        
        # Compute log2FC from raw values
        ratio = tumor_cell.value / normal_cell.value
        import math
        computed_log2fc = math.log2(ratio)
        print(f"Computed ratio: {ratio}")
        print(f"Computed log2FC from raw values: {computed_log2fc:.10f}")
        print(f"Rounded to 2 decimal places: {round(computed_log2fc, 2)}")
        break
