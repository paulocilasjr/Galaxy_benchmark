wb = openpyxl.load_workbook('inputs/Proteomic_data .xlsx', data_only=True)

# Find header row and column indices

print('Headers:', headers)
gene_col = headers.index('gene')
log2fc_col = headers.index('log2FC')
compare_col = headers.index('compare')
print(f'gene col: {gene_col}, log2FC col: {log2fc_col}, compare col: {compare_col}')

# Find ENO1
for row in ws.iter_rows(min_row=2, values_only=True):

        print(f'  gene: {row[gene_col]}')
        print(f'  log2FC: {row[log2fc_col]}')

        print(f'  adj.Pval: {row[headers.index(\"adj.Pval\")]}')

        print(f'  Tumor: {row[headers.index(\"Tumor\")]}')
