/bin/bash -lc "python - <<'PY'
import math, openpyxl
p='inputs/Proteomic_data .xlsx'
wbv=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wbv['Tumor vs Normal']
headers=[c.value for c in next(ws.iter_rows(min_row=1,max_row=1))]
for row in ws.iter_rows(min_row=2, values_only=True):
    rec=dict(zip(headers,row))
    if rec['gene']=='ENO1':
        calc=math.log2(rec['Tumor']/rec['Normal'])
        print('Normal',repr(rec['Normal']))
        print('Tumor',repr(rec['Tumor']))
        print('Ratio',repr(rec['Ratio']))
        print('reported log2FC',repr(rec['log2FC']))
        print('calculated from displayed group abundances',repr(calc))
        print('rounded reported',format(rec['log2FC'],'.2f'))
        print('rounded calculated',format(calc,'.2f'))
PY"
