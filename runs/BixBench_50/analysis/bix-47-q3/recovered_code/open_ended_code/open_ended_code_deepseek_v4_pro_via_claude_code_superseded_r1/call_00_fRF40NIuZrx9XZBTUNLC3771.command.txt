import openpyxl
from collections import Counter

# Verify zygosity values
wb = openpyxl.load_workbook('inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx')
ws = wb.active
zyg_counter = Counter()
for row in range(2, ws.max_row + 1):
    zyg = ws.cell(row=row, column=5).value
    zyg_counter[zyg] += 1
print("Zygosity distribution for 556-PF:")
for z, c in zyg_counter.most_common():
    print(f"  {z}: {c}")

# Also verify: let's look at a few non-reference rows to understand the data
print("\n--- Sample non-reference rows ---")
count = 0
for row in range(2, ws.max_row + 1):
    zyg = ws.cell(row=row, column=5).value
    if zyg != 'Reference':
        gene = ws.cell(row=row, column=13).value
        in_chip = ws.cell(row=row, column=30).value
        ref_alt = ws.cell(row=row, column=2).value
        chr_pos = ws.cell(row=row, column=1).value
        print(f"  {chr_pos} | {ref_alt} | Zyg={zyg} | Gene={gene} | In_CHIP={in_chip}")
        count += 1
        if count >= 15:
            break
