import openpyxl
from collections import Counter

# Load CHIP genes (correctly)
wb_chip = openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx')
ws_chip = wb_chip.active
chip_genes = set()
for row in range(1, ws_chip.max_row + 1):
    gene = ws_chip.cell(row=row, column=1).value
    if gene:
        chip_genes.add(str(gene).strip())
print(f"CHIP genes count: {len(chip_genes)}")

# Load variant file - data starts at row 3 (rows 1-2 are headers)
wb = openpyxl.load_workbook('inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx')
ws = wb.active

gene_counter = Counter()
total = 0
non_ref = 0
chip_non_ref = 0

for row in range(3, ws.max_row + 1):
    zygosity = ws.cell(row=row, column=5).value
    gene_names = ws.cell(row=row, column=13).value
    in_chip = ws.cell(row=row, column=30).value
    
    total += 1
    
    if zygosity and str(zygosity).strip() != 'Reference':
        non_ref += 1
        if in_chip == 'True':
            chip_non_ref += 1
            if gene_names:
                genes = [g.strip() for g in str(gene_names).split(',')]
                for g in genes:
                    if g in chip_genes:
                        gene_counter[g] += 1

print(f"Total variants: {total}")
print(f"Non-reference: {non_ref}")
print(f"CHIP gene non-reference: {chip_non_ref}")

print(f"\nAll CHIP genes with non-ref variant counts:")
for gene, count in gene_counter.most_common():
    print(f"  {gene}: {count}")

print(f"\n=== Gene with most non-reference variants: {gene_counter.most_common(1)[0]} ===")
