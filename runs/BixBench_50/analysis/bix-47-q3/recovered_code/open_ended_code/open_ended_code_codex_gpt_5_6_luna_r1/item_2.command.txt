/bin/bash -lc "python - <<'PY'
from pathlib import Path
import openpyxl

for p in sorted(Path('inputs').glob('*.xlsx')):
    wb = openpyxl.load_workbook(p, read_only=True, data_only=True)
    print('\\nFILE', p.name)
    print('SHEETS', wb.sheetnames)
    for ws in wb.worksheets:
        print(' SHEET', ws.title, 'DIM', ws.max_row, ws.max_column)
        rows=[]
        for row in ws.iter_rows(min_row=1, max_row=min(ws.max_row,5), values_only=True):
            rows.append(list(row[:min(ws.max_column,12)]))
        for row in rows:
            print('  ', row)
    wb.close()
PY"
