/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from pathlib import Path
# Show all male carrier rows and whether a file exists
wb=load_workbook('inputs/230215_Trio_Status.xlsx', data_only=True)
ws=wb.active
headers=[c.value for c in ws[1]]
files={p.name for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')}
rows=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    d=dict(zip(headers,row))
    if d.get('Sex')=='M' and d.get('BLM Mutation Status')=='Carrier':
        sid=str(d['Sample ID'])
        matches=[f for f in files if f'_CHIP_{sid}-' in f or f'_CHIP_{sid}.' in f]
        rows.append((d['Age'], sid, d['Status'], matches))
for rec in sorted(rows, reverse=True):
    print(rec)
PY"
