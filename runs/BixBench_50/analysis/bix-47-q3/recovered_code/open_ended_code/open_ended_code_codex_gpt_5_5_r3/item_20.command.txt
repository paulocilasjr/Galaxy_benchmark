/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
wb=load_workbook('inputs/230215_Trio_Status.xlsx', data_only=True)
ws=wb.active
headers=[c.value for c in ws[1]]
print(headers)
counts=Counter()
for row in ws.iter_rows(min_row=2, values_only=True):
    d=dict(zip(headers,row))
    counts[(d.get('Sex'), d.get('BLM Mutation Status'))]+=1
print('counts')
for k,v in counts.items(): print(k,v)
print('all rows sorted by age desc:')
rows=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    d=dict(zip(headers,row))
    age=d.get('Age')
    try: agev=float(age)
    except (TypeError,ValueError): continue
    rows.append((agev,d.get('Sample ID'),d.get('Status'),d.get('Sex'),d.get('BLM Mutation Status')))
for rec in sorted(rows, reverse=True)[:25]: print(rec)
PY"
