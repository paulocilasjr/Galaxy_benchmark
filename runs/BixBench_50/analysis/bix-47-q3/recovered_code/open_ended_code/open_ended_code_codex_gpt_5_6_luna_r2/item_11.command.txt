/bin/bash -lc "python - <<'PY'
from collections import Counter,defaultdict
from openpyxl import load_workbook
p='inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx'
ws=load_workbook(p,data_only=True).active
headers=[ws.cell(2,c).value for c in range(1,ws.max_column+1)]
idx={h:i for i,h in enumerate(headers)}
print('rows',ws.max_row-2)
for h in ['Zygosity','Gene Names','In_CHIP','Match','Filter','Chr:Pos','Ref/Alt','Variant Allele Freq']:
 print(h,idx[h]+1)
rows=[]
for r in ws.iter_rows(min_row=3,values_only=True):
    d={h:r[i] if i<len(r) else None for h,i in idx.items()}
    rows.append(d)
print('zygosity',Counter(x['Zygosity'] for x in rows))
print('in_chip',Counter(x['In_CHIP'] for x in rows))
print('in_chip nonref',Counter(x['In_CHIP'] for x in rows if x['Zygosity']"'!='"'Reference'))
print('nonref total',sum(x['Zygosity']"'!='"'Reference' for x in rows))
print('\\nall nonref by Gene Names:')
for k,v in Counter(x['Gene Names'] for x in rows if x['Zygosity']"'!='"'Reference').most_common(): print(v,repr(k))
print('\\nnonref by Match (CHIP):')
for k,v in Counter(x['Match'] for x in rows if x['Zygosity']"'!='"'Reference' and x['In_CHIP']==True).most_common(): print(v,repr(k))
print('\\nnonref CHIP true Gene Names:')
for k,v in Counter(x['Gene Names'] for x in rows if x['Zygosity']"'!='"'Reference' and x['In_CHIP']==True).most_common(): print(v,repr(k))
print('\\nnonref no chip:')
for k,v in Counter(x['Gene Names'] for x in rows if x['Zygosity']"'!='"'Reference' and x['In_CHIP']"'!=True).most_common(): print(v,repr(k))
PY'
