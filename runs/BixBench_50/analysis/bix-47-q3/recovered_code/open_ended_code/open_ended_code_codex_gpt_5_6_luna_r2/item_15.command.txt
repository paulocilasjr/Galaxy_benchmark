/bin/bash -lc "python - <<'PY'
from collections import Counter
from openpyxl import load_workbook

p='inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx'
ws=load_workbook(p,data_only=True).active
headers=[ws.cell(2,c).value for c in range(1,ws.max_column+1)]
col={h:i+1 for i,h in enumerate(headers) if h is not None}
rows=[]
for r in range(3,ws.max_row+1):
    z=ws.cell(r,col['Zygosity']).value
    if z in ('Heterozygous','Homozygous Variant'):
        genes=str(ws.cell(r,col['Gene Names']).value or '')
        rows.append((r,ws.cell(r,col['Chr:Pos']).value,ws.cell(r,col['Variant ID']).value,genes,z))
counts=Counter(x[3] for x in rows)
print('top exact CHIP gene labels:')
for gene,count in counts.most_common(15): print(gene,count)
print('top gene label unique chrpos:',Counter(x[3] for x in rows if x[3]=='NOTCH1'),len({x[1] for x in rows if x[3]=='NOTCH1'}))
print('NOTCH1 records')
for x in rows:
    if x[3]=='NOTCH1': print(x)
# Check only supplied CHIP gene symbols and split annotations.
chip={str(load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx',data_only=True).active.cell(r,1).value).strip() for r in range(1,57)}
sym=Counter()
for _,_,_,genes,_ in rows:
    for g in genes.split(','):
        g=g.strip()
        if g in chip: sym[g]+=1
print('top split restricted CHIP:',sym.most_common(15))
PY"
