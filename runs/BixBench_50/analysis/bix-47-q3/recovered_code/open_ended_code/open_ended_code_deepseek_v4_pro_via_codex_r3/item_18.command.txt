/bin/bash -lc "python3 - <<'PY'
import openpyxl, collections, re
chip=[str(r[0]).strip() for r in openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx', data_only=True).active.iter_rows(values_only=True) if r[0]]
chipset=set(chip)
f='inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx'
ws=openpyxl.load_workbook(f, data_only=True).active
nr=[r for r in ws.iter_rows(min_row=3, values_only=True) if str(r[4]) not in ('Reference','None')]
# substring count per chip gene in gene names
subcnt=collections.Counter()
for r in nr:
 gs=str(r[12] or '')
 for g in chip:
  # word boundaries or comma separated token exact
  if re.search(r'(?<"'![A-Z0-9-])'"'+re.escape(g)+r'(?"'![A-Z0-9-])'"', gs):
   subcnt[g]+=1
print('substring chip counts top')
for g,c in subcnt.most_common(15): print(g,c)
# count rows containing exact token among chip genes, but one row per gene token
tokcnt=collections.Counter()
for r in nr:
 seen=set()
 for tok in re.split('[,;]', str(r[12] or '')):
  tok=tok.strip()
  if tok in chipset: seen.add(tok)
 for g in seen: tokcnt[g]+=1
print('token chip counts top')
for g,c in tokcnt.most_common(15): print(g,c)
# all genes token counts top
allcnt=collections.Counter()
for r in nr:
 seen=set()
 for tok in re.split('[,;]', str(r[12] or '')):
  tok=tok.strip()
  if tok: seen.add(tok)
 for g in seen: allcnt[g]+=1
print('all token unique per row counts top')
for g,c in allcnt.most_common(20): print(g,c)
PY"
