/bin/bash -lc "python - <<'PY'
import openpyxl, collections, re
p='inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx'
wb=openpyxl.load_workbook(p, data_only=True, read_only=True)
ws=wb.active
rows=ws.iter_rows(values_only=True)
_ = next(rows)  # group header
hdr = next(rows)
idx={h:i for i,h in enumerate(hdr) if h is not None}
print(idx)
zyg_i=idx['Zygosity']; gene_i=idx['Gene Names']
# duplicate headers, use fixed positions too
print('header len', len(hdr))
print('AD-AF', hdr[29:33])
counts=collections.Counter(); zcounts=collections.Counter(); examples=collections.defaultdict(list)
for r in rows:
    z=r[zyg_i]
    zcounts[z]+=1
    if z and str(z).strip().lower() "'!= '"'reference':
        gene=r[gene_i]
        counts[str(gene)] += 1
        if len(examples[str(gene)])<5:
            examples[str(gene)].append((r[0],r[1],z,r[9],r[13],r[17]))
print('zygosity counts', zcounts)
print('top gene names raw')
for g,c in counts.most_common(20):
    print(g,c,examples[g][:2])
# split by comma/semicolon, compare with In_CHIP Match col AE (0-based 30)
for colname, colidx in [('Gene Names', gene_i), ('In_CHIP Match AE',30)]:
    c=collections.Counter()
    for r in ws.iter_rows(min_row=3, values_only=True):
        z=r[zyg_i]
        if z and str(z).strip().lower() "'!= '"'reference':
            val=r[colidx]
            if val is None: continue
            parts=[x.strip() for x in re.split(r'[,;]', str(val)) if x.strip()]
            for part in parts:
                c[part]+=1
    print('\\nSplit count using',colname)
    print(c.most_common(20))
PY"
