/bin/bash -c "python3 - <<'PY'
import zipfile, xml.etree.ElementTree as ET
p='/workspace/inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx'
z=zipfile.ZipFile(p)
ns={'m':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
M='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
sroot=ET.fromstring(z.read('xl/sharedStrings.xml'))
items=[''.join(t.text or '' for t in si.iter(M+'t')) for si in sroot.findall('m:si',ns)]
def col_to_idx(ref):
    s=''
    for ch in ref:
        if ch.isalpha(): s+=ch
        else: break
    n=0
    for ch in s: n=n*26+ord(ch)-64
    return n-1
root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
rows=[]
for row in root.iter(M+'row'):
    r=int(row.get('r'))
    vals={}
    for c in row.findall(M+'c'):
        ref=c.get('r'); idx=col_to_idx(ref)
        typ=c.get('t')
        v=c.find(M+'v')
        if v is None:
            # inline string
            is_=c.find(M+'is')
            if is_ is not None:
                val=''.join(t.text or '' for t in is_.iter(M+'t'))
            else: val=''
        else:
            val=v.text or ''
            if typ=='s': val=items[int(val)]
            elif typ=='inlineStr':
                is_=c.find(M+'is'); val=''.join(t.text or '' for t in is_.iter(M+'t')) if is_ is not None else val
        vals[idx]=val
    rows.append((r,vals))
print('max row',max(r for r,v in rows), 'count',len(rows))
# print first few rows with keys
for r,vals in rows[:8]:
 print('ROW',r, vals)
PY"
