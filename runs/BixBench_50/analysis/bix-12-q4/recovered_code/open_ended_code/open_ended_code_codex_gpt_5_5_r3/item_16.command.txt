/bin/bash -lc "python - <<'PY'
import zipfile, collections
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    counts=collections.Counter()
    examples={}
    with zipfile.ZipFile(z) as f:
        for n in f.namelist():
            if not n.endswith('.mafft'): continue
            nt=sum(1 for line in f.read(n).decode().splitlines() if line.startswith('>'))
            counts[nt]+=1
            examples.setdefault(nt,n)
    print(z, dict(sorted(counts.items())), examples)
PY"
