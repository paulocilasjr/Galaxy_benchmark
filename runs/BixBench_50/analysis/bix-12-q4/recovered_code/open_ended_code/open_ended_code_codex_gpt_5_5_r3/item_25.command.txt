/bin/bash -lc "python - <<'PY'
# Compute U under exact percentages versus AMAS-like rounded proportions.
import zipfile
from collections import Counter

def read_fasta(text):
    seqs=[]; name=None; parts=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(parts))
            name=line[1:]; parts=[]
        else:
            parts.append(line)
    if name is not None: seqs.append(''.join(parts))
    return seqs

def p_prop(seqs):
    L=len(seqs[0]); pis=0
    miss=set('BJZX.*-?')
    for chars in zip(*seqs):
        site=[c for c in chars if c not in miss]
        cnt=Counter(site)
        if sum(v >= 2 for v in cnt.values()) >= 2:
            pis += 1
    return pis / L

def values(z, rounded=False):
    out=[]
    with zipfile.ZipFile(z) as f:
        for n in sorted(f.namelist()):
            if n.endswith('.mafft'):
                p=p_prop(read_fasta(f.read(n).decode()))
                out.append(round(p,3) if rounded else p*100)
    return out

def u(x,y):
    return sum(1 if a>b else 0.5 if a==b else 0 for a in x for b in y)
for rounded in [False, True]:
    a=values('inputs/scogs_animals.zip', rounded)
    f=values('inputs/scogs_fungi.zip', rounded)
    print('rounded', rounded, 'U', u(a,f), 'animal unique', len(set(a)), 'fungi unique', len(set(f)))
PY"
