/bin/bash -lc "python - <<'PY'
import zipfile
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        names=[n for n in f.namelist() if not n.endswith('/')]
    print(z)
    print('total',len(names),'mafft',sum(n.endswith('.mafft') for n in names),'faa',sum(n.endswith('.faa') for n in names))
    bases={n[:-6] if n.endswith('.mafft') else n for n in names}
    print('unique bases',len(bases),'with both',sum((b in names and b+'.mafft' in names) for b in bases))
    print('only raw',len([b for b in bases if b in names and b+'.mafft' not in names]))
    print('only mafft',len([b for b in bases if b not in names and b+'.mafft' in names]))
PY"
