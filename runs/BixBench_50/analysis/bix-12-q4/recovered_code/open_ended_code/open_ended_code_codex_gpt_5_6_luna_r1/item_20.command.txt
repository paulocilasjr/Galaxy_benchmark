import zipfile, collections, os, math
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 alph=collections.Counter(); nseq=collections.Counter(); gapcols=0; total=0; nonuniform=0; nali=0
 for name in zipfile.ZipFile(fn).namelist():
  if not name.endswith('.mafft'): continue
  nali+=1
  text=zipfile.ZipFile(fn).read(name).decode()
  seqs=[]; s=None
  for line in text.splitlines():
   if line.startswith('>'):
    if s is not None: seqs.append(''.join(s))
    s=[]
   elif s is not None: s.append(line.strip())
  if s is not None: seqs.append(''.join(s))
  nseq[len(seqs)]+=1
  L=len(seqs[0]); total+=L
  for seq in seqs: alph.update(seq)
  for col in zip(*seqs):
   if '-' in col: gapcols+=1
   if len(set(col))>1: nonuniform+=1
 print(fn,'alignments',nali,'nseq',nseq,'alphabet',alph,'gap-containing column fraction',gapcols/total,'variable col fraction',nonuniform/total)
