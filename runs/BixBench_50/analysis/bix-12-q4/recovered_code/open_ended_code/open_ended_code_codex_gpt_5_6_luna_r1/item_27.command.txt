/bin/bash -lc "python - <<'PY'
import zipfile,collections

def load(fn):
 out=[]
 with zipfile.ZipFile(fn) as z:
  for n in z.namelist():
   if n.endswith('.mafft'):
    seqs=[];cur=None
    for l in z.read(n).decode().splitlines():
     if l.startswith('>'):
      if cur is not None:seqs.append(''.join(cur))
      cur=[]
     else:cur.append(l.strip())
    seqs.append(''.join(cur));out.append(seqs)
 return out
def p(s, gap=False, denom='variable'):
 pi=var=0
 for c in zip(*s):
  v=list(c) if gap else [x for x in c if x"'!='"'-']
  if len(set(v))>1:var+=1
  co=collections.Counter(v)
  if sum(x>=2 for x in co.values())>=2:pi+=1
 return 100*pi/(var if denom=='variable' else len(s[0])) if (var if denom=='variable' else len(s[0])) else 0
def u(a,b):
 z=sorted(a+b); rr={};i=0
 while i<len(z):
  j=i
  while j<len(z) and z[j]==z[i]:j+=1
  rr[z[i]]=(i+j+2)/2;i=j
 return sum(rr[x] for x in a)-len(a)*(len(a)+1)/2
A=load('inputs/scogs_animals.zip');F=load('inputs/scogs_fungi.zip')
for gap in [False,True]:
 for den in ['variable','all']:
  a=[p(s,gap,den) for s in A];b=[p(s,gap,den) for s in F]
  print(gap,den,u(a,b),'ranges',min(a),max(a),min(b),max(b),'zeros',sum(x==0 for x in a),sum(x==0 for x in b))
PY"
