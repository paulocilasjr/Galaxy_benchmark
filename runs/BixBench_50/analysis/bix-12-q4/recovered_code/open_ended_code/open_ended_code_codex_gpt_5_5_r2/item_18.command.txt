/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse_count(data):
    return sum(1 for line in data.decode().splitlines() if line.startswith('>'))
for archive in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    c=Counter(); examples={}
    with ZipFile(archive) as z:
        for n in z.namelist():
            if n.endswith('.mafft'):
                k=parse_count(z.read(n)); c[k]+=1; examples.setdefault(k,n)
    print(archive, c, examples)
PY"
