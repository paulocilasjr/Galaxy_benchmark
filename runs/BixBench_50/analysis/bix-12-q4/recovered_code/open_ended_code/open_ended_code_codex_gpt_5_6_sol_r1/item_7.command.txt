/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import mean, median
AA=set('ACDEFGHIKLMNPQRSTVWY')

def fasta(txt):
 d={}; k=None
 for line in txt.splitlines():
  if line.startswith('>'): k=line[1:].split()[0]; d[k]=''
  elif k: d[k]+=line.strip().upper()
 return d

def vals(zp, mode):
 out=[]; desc=[]; chars=Counter()
 with ZipFile(zp) as z:
  for n in sorted(x for x in z.namelist() if x.endswith('.faa.mafft')):
   d=fasta(z.read(n).decode())
   lens={len(s) for s in d.values()}
   if len(lens)"'!=1: raise ValueError((n,lens))
   L=lens.pop(); pi=0
   for col in zip(*d.values()):
    chars.update(col)
    if mode=='"'strict': c=Counter(x for x in col if x in AA)
    elif mode=='gapstate': c=Counter(col)
    elif mode=='nongap_any': c=Counter(x for x in col if x not in '-?X')
    if sum(v>=2 for v in c.values())>=2: pi+=1
   out.append(100*pi/L)
   desc.append((n,len(d),L,pi,100*pi/L))
 return out,desc,chars

def ustat(x,y):
 tagged=[(v,0) for v in x]+[(v,1) for v in y]
 tagged.sort()
 rank_sum=0; i=0
 while i<len(tagged):
  j=i+1
  while j<len(tagged) and tagged[j][0]==tagged[i][0]: j+=1
  avg_rank=((i+1)+j)/2
  rank_sum += avg_rank*sum(g==0 for _,g in tagged[i:j])
  i=j
 return rank_sum-len(x)*(len(x)+1)/2
for mode in ['strict','nongap_any','gapstate']:
 a,ad,ac=vals('inputs/scogs_animals.zip',mode)
 f,fd,fc=vals('inputs/scogs_fungi.zip',mode)
 u=ustat(a,f)
 print('\\n',mode,'n',len(a),len(f),'U1',u,'U2',len(a)*len(f)-u)
 print('animal mean med range',mean(a),median(a),(min(a),max(a)))
 print('fungal mean med range',mean(f),median(f),(min(f),max(f)))
 print('seq count dist A',Counter(x[1] for x in ad),'F',Counter(x[1] for x in fd))
 print('chars A',ac,'F',fc)
PY"
