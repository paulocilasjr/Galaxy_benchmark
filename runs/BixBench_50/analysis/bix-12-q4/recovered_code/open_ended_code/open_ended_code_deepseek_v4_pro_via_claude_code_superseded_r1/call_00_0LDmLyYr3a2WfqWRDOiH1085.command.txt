import os

def parse_fasta(filepath):
    seqs = []
    cur_h, cur_s = None, []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if cur_h:
                    seqs.append((cur_h, ''.join(cur_s)))
                cur_h = line[1:]
                cur_s = []
            else:
                cur_s.append(line.upper())
        if cur_h:
            seqs.append((cur_h, ''.join(cur_s)))
    return seqs

def count_pis(sequences):
    '''Count parsimony informative sites. Gaps treated as missing data.'''
    if len(sequences) < 4:  # need >=4 seqs for >=2 states each in >=2 seqs
        # Actually even with 4 seqs, a site might need all 4
        # Let the main code handle it
        pass
    
    if len(sequences) < 2:
        return 0, 0
    
    seqs = [s for _, s in sequences if len(s) > 0]
    if not seqs:
        return 0, 0
    
    n = len(seqs[0])
    pis = 0
    for col in range(n):
        chars = [s[col] for s in seqs if col < len(s)]
        freq = {}
        for c in chars:
            if c == '-':
                continue
            freq[c] = freq.get(c, 0) + 1
        inf = sum(1 for v in freq.values() if v >= 2)
        if inf >= 2:
            pis += 1
    return pis, n

def mannwhitneyu(x, y):
    n1, n2 = len(x), len(y)
    combined = [(v, 0) for v in x] + [(v, 1) for v in y]
    combined.sort(key=lambda t: t[0])
    
    ranks = [0.0] * len(combined)
    i = 0
    while i < len(combined):
        j = i
        while j < len(combined) and combined[j][0] == combined[i][0]:
            j += 1
        midrank = (i + j + 1) / 2.0  # 1-indexed midrank
        for k in range(i, j):
            ranks[k] = midrank
        i = j
    
    r1 = sum(ranks[k] for k in range(len(combined)) if combined[k][1] == 0)
    u1 = r1 - n1 * (n1 + 1) / 2.0
    return u1
