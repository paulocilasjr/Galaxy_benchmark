import os

def parse_fasta(filepath):
    seqs = []
    cur_h, cur_s = None, []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if cur_h:
                    seqs.append((cur_h, ''.join(cur_s)))
                cur_h = line[1:]
                cur_s = []
            else:
                cur_s.append(line.upper())
        if cur_h:
            seqs.append((cur_h, ''.join(cur_s)))
    return seqs

def pis_details(filepath):
    seqs = parse_fasta(filepath)
    n_spp = len(seqs)
    if n_spp < 2 or not seqs:
        return n_spp, 0, 0, 0.0
    
    chars_seqs = [s for _, s in seqs if len(s) > 0]
    if not chars_seqs:
        return n_spp, 0, 0, 0.0
    
    n = len(chars_seqs[0])
    pis = 0
    total_non_gap_cols = 0
    for col in range(n):
        chars = [s[col] for s in chars_seqs if col < len(s)]
        non_gap = [c for c in chars if c != '-']
        if len(non_gap) >= 2:
            total_non_gap_cols += 1
        freq = {}
        for c in chars:
            if c == '-':
                continue
            freq[c] = freq.get(c, 0) + 1
        inf = sum(1 for v in freq.values() if v >= 2)
        if inf >= 2:
            pis += 1
    
    pct = (pis / n * 100.0) if n > 0 else 0.0
    return n_spp, n, pis, pct

# Check all shared OGs
animals_dir = '/workspace/tmp/animals'
fungi_dir = '/workspace/tmp/fungi'

animals_mafft = set(f for f in os.listdir(animals_dir) if f.endswith('.mafft'))

    nsp_f, alen_f, pis_f, pct_f = pis_details(ff)
