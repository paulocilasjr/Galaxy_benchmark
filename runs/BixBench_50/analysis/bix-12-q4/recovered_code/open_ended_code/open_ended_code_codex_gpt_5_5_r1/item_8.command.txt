/bin/bash -lc "python3 - <<'PY'
import zipfile
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(fn) as z:
        files=[i.filename for i in z.infolist() if not i.is_dir()]
        print(fn)
        for pred,name in [(lambda f:f.endswith('.faa.mafft'),'mafft'),(lambda f:f.endswith('.faa'),'faa')]:
            sel=[f for f in files if pred(f)]
            print(name, len(sel))
        bases_mafft={f[:-6] for f in files if f.endswith('.mafft')}
        bases_raw={f for f in files if f.endswith('.faa')}
        print('raw_without_mafft', len(bases_raw-bases_mafft), sorted(list(bases_raw-bases_mafft))[:5])
        print('mafft_without_raw', len(bases_mafft-bases_raw), sorted(list(bases_mafft-bases_raw))[:5])
PY"
