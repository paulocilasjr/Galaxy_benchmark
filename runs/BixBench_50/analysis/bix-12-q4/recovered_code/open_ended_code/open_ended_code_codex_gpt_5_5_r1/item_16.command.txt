import zipfile, collections, statistics

def read_fasta_bytes(b):
    seqs=[]; chunks=[]; name=None
    for raw in b.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(chunks))
            name=line[1:]; chunks=[]
        else: chunks.append(line)
    if name is not None: seqs.append(''.join(chunks))
    return seqs

def pi_pct(seqs, count_gaps=False, denom_all=True):
    pi=0
    cols=list(zip(*seqs))
    for col in cols:
        states=col if count_gaps else [c for c in col if c not in '-?.']
        counts=collections.Counter(states)
        if sum(v>=2 for v in counts.values()) >= 2:
            pi += 1
    return 100*pi/len(cols)

def values(fn, only4=False, common_ids=None):
    vals=[]; ids=[]

        for name in sorted(n for n in z.namelist() if n.endswith('.faa.mafft')):

    return ids, vals

def U(x,y): return sum(1 if a>b else .5 if a==b else 0 for a in x for b in y)
