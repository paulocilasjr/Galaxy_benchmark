import sys
sys.path.insert(0, '/tmp/venv/lib/python3.11/site-packages')
import os
from collections import Counter

def parse_mafft(filepath):
    sequences = []
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_seq:
                    sequences.append(''.join(current_seq))
                    current_seq = []
            else:
                current_seq.append(line)
        if current_seq:
            sequences.append(''.join(current_seq))
    return sequences

ANIMALS_DIR = '/tmp/scogs_animals'
FUNGI_DIR = '/tmp/scogs_fungi'

animal_mafft = set(f for f in os.listdir(ANIMALS_DIR) if f.endswith('.mafft'))
fungi_mafft = set(f for f in os.listdir(FUNGI_DIR) if f.endswith('.mafft'))
shared = sorted(animal_mafft & fungi_mafft)

animal_seq_counts = Counter()
fungi_seq_counts = Counter()

for ortholog in shared:
    animal_seqs = parse_mafft(os.path.join(ANIMALS_DIR, ortholog))
    fungi_seqs = parse_mafft(os.path.join(FUNGI_DIR, ortholog))
    animal_seq_counts[len(animal_seqs)] += 1
    fungi_seq_counts[len(fungi_seqs)] += 1

print('Animal sequence counts per alignment:')
for k in sorted(animal_seq_counts):
    print(f'  {k} seqs: {animal_seq_counts[k]} alignments')
print()
print('Fungi sequence counts per alignment:')
for k in sorted(fungi_seq_counts):
    print(f'  {k} seqs: {fungi_seq_counts[k]} alignments')
