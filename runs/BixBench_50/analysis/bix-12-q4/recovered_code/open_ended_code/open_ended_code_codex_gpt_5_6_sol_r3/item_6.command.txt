/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for p in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 c=Counter(); ns=Counter(); lens=[]; invalid=[]
 with ZipFile(p) as z:
  for n in z.namelist():
   if not n.endswith('.faa.mafft'): continue
   seqs=[]; cur=[]
   for line in z.read(n).decode().splitlines():
    if line.startswith('>'):
     if cur: seqs.append(''.join(cur)); cur=[]
    else: cur.append(line.strip())
   if cur: seqs.append(''.join(cur))
   ns[len(seqs)]+=1
   lengths={len(s) for s in seqs}
   if len(lengths)"'!=1: invalid.append((n,lengths))
   lens.extend(lengths)
   c.update('"''.join(seqs))
 print(p, 'sequence-count distribution',dict(sorted(ns.items())), 'alignlen min/max',min(lens),max(lens),'unequal',len(invalid))
 print('characters', ''.join(sorted(c)), c)
PY"
