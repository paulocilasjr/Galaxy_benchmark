/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from scipy.stats import mannwhitneyu
import numpy as np

def load(p,gap_state=False,only4=False):
 out=[]
 with ZipFile(p) as z:
  for n in z.namelist():
   if not n.endswith('.faa.mafft'): continue
   seqs=[]; cur=[]
   for line in z.read(n).decode().splitlines():
    if line.startswith('>'):
     if cur: seqs.append(''.join(cur)); cur=[]
    else: cur.append(line.strip())
   if cur: seqs.append(''.join(cur))
   if only4 and len(seqs)"'!=4: continue
   L=len(seqs[0])
   pis=0
   for col in zip(*seqs):
    cnt=Counter(col)
    if not gap_state: cnt.pop('"'-',None)
    if sum(v>=2 for v in cnt.values())>=2: pis+=1
   out.append((n, len(seqs), L, pis, 100*pis/L))
 return out
for gap in (False,True):
 for only4 in (False,True):
  a=load('inputs/scogs_animals.zip',gap,only4)
  f=load('inputs/scogs_fungi.zip',gap,only4)
  av=[x[-1] for x in a]; fv=[x[-1] for x in f]
  u=mannwhitneyu(av,fv,alternative='two-sided',method='asymptotic')
  print('gapstate',gap,'only4',only4,'n',len(a),len(f),'PI sums',sum(x[3] for x in a),sum(x[3] for x in f),'mean%',np.mean(av),np.mean(fv),'median',np.median(av),np.median(fv),'zeros',sum(x==0 for x in av),sum(x==0 for x in fv),'U1',u.statistic,'U2',len(a)*len(f)-u.statistic,'p',u.pvalue)
PY"
