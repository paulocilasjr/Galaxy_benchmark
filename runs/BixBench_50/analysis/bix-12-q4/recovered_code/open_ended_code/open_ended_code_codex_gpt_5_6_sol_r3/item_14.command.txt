/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from fractions import Fraction

def values(path):
    vals=[]
    with ZipFile(path) as z:
        for name in z.namelist():
            if not name.endswith('.faa.mafft'): continue
            seqs=[]; cur=[]
            for line in z.read(name).decode().splitlines():
                if line.startswith('>'):
                    if cur: seqs.append(''.join(cur)); cur=[]
                else: cur.append(line.strip())
            if cur: seqs.append(''.join(cur))
            length=len(seqs[0]); pi=0
            for col in zip(*seqs):
                counts=Counter(x for x in col if x not in set('BJZX.*-?'))
                if sum(n >= 2 for n in counts.values()) >= 2: pi += 1
            vals.append(Fraction(pi,length))
    return vals
A=values('inputs/scogs_animals.zip'); F=values('inputs/scogs_fungi.zip')
U_pair=sum(Fraction(a>b)+Fraction(a==b,2) for a in A for b in F)
# exact midrank sum independently
combined=sorted([(v,0) for v in A]+[(v,1) for v in F], key=lambda x:x[0])
ranksum=Fraction(); i=0
while i<len(combined):
    j=i+1
    while j<len(combined) and combined[j][0]==combined[i][0]: j+=1
    midrank=Fraction((i+1)+j,2)
    ranksum += midrank*sum(g==0 for _,g in combined[i:j])
    i=j
U_rank=ranksum-Fraction(len(A)*(len(A)+1),2)
print('n_animals',len(A),'n_fungi',len(F))
print('U_pairwise',U_pair,'U_rank_sum',U_rank,'complement',len(A)*len(F)-U_pair)
print('nonzero',sum(v>0 for v in A),sum(v>0 for v in F),'unique_percentages',len(set(A)),len(set(F)))
# Sensitivity to premature rounding: AMAS prop is rounded to 3 decimals
for digits in (3,4,5,6):
    ar=[round(float(x),digits) for x in A]; fr=[round(float(x),digits) for x in F]
    u=sum((a>b)+0.5*(a==b) for a in ar for b in fr)
    print('rounded proportion digits',digits,'U',u)
PY"
