STD=set('ACDEFGHIKLMNPQRSTVWY')

    cc=collections.Counter(x for x in c if x in STD)

    else:D+=1
   out.append(fractions.Fraction(pi,D) if D else fractions.Fraction())

  while j<len(a)and a[j][0]==a[i][0]:j+=1
  r=fractions.Fraction(i+j+1,2)

  i=j
 return rs[0]-len(x)*(len(x)+1)//2
