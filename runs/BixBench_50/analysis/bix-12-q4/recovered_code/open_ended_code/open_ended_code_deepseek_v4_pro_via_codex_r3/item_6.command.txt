/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
for d in ['tmp/scogs_a','tmp/scogs_f']:
    ids={p.name[:-len('.mafft')] for p in Path(d).glob('*.mafft')}
    faas={p.name for p in Path(d).glob('*.faa')}
    print(d, len(ids), 'mafft ids')
    print('ids no faa', len(ids-faas), sorted(ids-faas)[:5])
    print('faa no mafft', len(faas-ids), sorted(faas-ids)[:5])
an= {p.stem.replace('.faa','') for p in Path('tmp/scogs_a').glob('*.mafft')}
fu={p.stem.replace('.faa','') for p in Path('tmp/scogs_f').glob('*.mafft')}
print('intersection', len(an&fu),'animal-only',len(an-fu), sorted(an-fu)[:20],'fungi-only',len(fu-an), sorted(fu-an)[:20])
PY"
