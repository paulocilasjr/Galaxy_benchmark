/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
from collections import Counter
import scipy.stats as st

AMBIGUOUS=set('XBZJUO?*')

def read_fasta(path):
    seqs=[]; name=None; buf=[]
    with path.open() as f:
        for raw in f:
            line=raw.strip()
            if not line:
                continue
            if line.startswith('>'):
                if name is not None:
                    seqs.append((name,''.join(buf)))
                name=line[1:]; buf=[]
            else:
                buf.append(line)
    if name is not None:
        seqs.append((name,''.join(buf)))
    return seqs

def pis_percent(path):
    seqs=read_fasta(path)
    n=len(seqs)
    if n==0:
        raise ValueError(f'empty alignment: {path}')
    lengths={len(s) for _,s in seqs}
    if len(lengths)"'!=1:
        raise ValueError(f'"'ragged alignment: {path}: {lengths}')
    L=lengths.pop()
    informative=0
    for col in range(L):
        chars=[s[col] for _,s in seqs]
        counts=Counter(c for c in chars if c not in '-.' and c not in AMBIGUOUS)
        states=[c for c,v in counts.items() if v>=2]
        if len(set(states))>=2:
            informative += 1
    return 100.0*informative/L

for group in ['tmp/scogs_a','tmp/scogs_f']:
    vals=[pis_percent(p) for p in sorted(Path(group).glob('*.mafft'))]
    print(group, 'n=',len(vals), 'min=%.12g'%min(vals), 'median=%.12g'%sorted(vals)[len(vals)//2], 'max=%.12g'%max(vals), 'mean=%.12g'%(sum(vals)/len(vals)))

A=[pis_percent(p) for p in sorted(Path('tmp/scogs_a').glob('*.mafft'))]
F=[pis_percent(p) for p in sorted(Path('tmp/scogs_f').glob('*.mafft'))]
res=st.mannwhitneyu(A,F,alternative='two-sided',use_continuity=True,method='auto')
print('U animals-first =',repr(res.statistic))
print('U fungi-first =',len(A)*len(F)-res.statistic)
print('p-value =',res.pvalue)
PY"
