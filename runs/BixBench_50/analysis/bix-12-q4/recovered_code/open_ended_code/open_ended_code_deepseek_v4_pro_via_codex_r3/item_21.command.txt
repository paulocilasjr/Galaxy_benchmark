/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
from collections import Counter
for d in ['tmp/scogs_a','tmp/scogs_f']:
  allgap=0; total_cols=0; files_allgap=0
  for p in Path(d).glob('*.mafft'):
    seqs=[]; cur=[]; name=None
    for line in p.read_text().splitlines():
      if line.startswith('>'):
        if name is not None: seqs.append((name,''.join(cur)))
        name=line[1:].strip(); cur=[]
      else: cur.append(line.strip())
    if name is not None: seqs.append((name,''.join(cur)))
    n=len(seqs); L=len(seqs[0][1]); total_cols+=L
    fg=0
    for j in range(L):
      if all(s[j] in '-.' for n,s in seqs): fg+=1
    allgap+=fg; files_allgap += (fg>0)
  print(d,'total cols',total_cols,'allgap cols',allgap,'files with allgap',files_allgap)
PY"
