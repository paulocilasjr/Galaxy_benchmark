/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
import collections
base=Path('tmp/busco')
for clade, species_list in [('Animals',['Animals_Cele','Animals_Ggal','Animals_Mmus','Animals_Xlae']),('Fungi',['Fungi_Ncra','Fungi_Scer','Fungi_Spom','Fungi_Umay'])]:
  print('\\n',clade)
  counts=collections.Counter()
  for sp in species_list:
    ft=base/sp/'run_eukaryota_odb10'/'full_table.tsv'
    for line in ft.read_text().splitlines():
      if line.startswith('#'): continue
      parts=line.split('\\t')
      if len(parts)<2: continue
      busco,status=parts[0],parts[1]
      if status=='Complete': counts[busco]+=1
  dist=collections.Counter(counts.values())
  print('complete counts dist',dist)
  print('total distinct complete',len(counts))
  print('genes with 4 complete', sum(1 for g,c in counts.items() if c==4))
  print('genes with >=1',sum(1 for g,c in counts.items() if c>=1))
  scog={p.name.split('.faa')[0] for p in (Path('tmp/scogs_a') if clade=='Animals' else Path('tmp/scogs_f')).glob('*.mafft')}
  print('scogs count',len(scog))
  print('scogs not complete any', len(scog-set(counts)), 'complete not scogs', len(set(counts)-scog))
  scog_dist=collections.Counter(counts[g] for g in scog if g in counts)
  print('scog complete count dist',scog_dist)
PY"
