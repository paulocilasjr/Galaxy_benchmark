/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
from collections import Counter
chars=Counter()
for d in ['tmp/scogs_a','tmp/scogs_f']:
    for p in Path(d).glob('*.mafft'):
        for line in p.read_text().splitlines():
            if line.startswith('>'): continue
            for c in line.strip(): chars[c]+=1
print(chars)
PY"
