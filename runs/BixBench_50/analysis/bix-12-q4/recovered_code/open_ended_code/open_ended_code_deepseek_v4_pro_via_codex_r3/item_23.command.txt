from pathlib import Path
from collections import Counter
import scipy.stats as st
print('scipy',st.__version__)
def read(p):
  seqs=[]; name=None; cur=[]
  for line in p.read_text().splitlines():
    if line.startswith('>'):
      if name is not None: seqs.append((name,''.join(cur)))
      name=line[1:].strip(); cur=[]
    else: cur.append(line.strip())
  if name is not None: seqs.append((name,''.join(cur)))
  return seqs

def pis(p, exclude_gap=True, ambiguous='XBZJUO?*'):
  seqs=read(p); n=len(seqs); L=len(seqs[0][1]); cnt=0
  for j in range(L):
    col=[s[j] for n,s in seqs]
    syms=[c for c in col if c not in '-.' and (not ambiguous or c not in ambiguous)]
    counts=Counter(syms)
    states=[c for c,v in counts.items() if v>=2]
    if len(states)>=2: cnt+=1
  return cnt/L*100
A=sorted(pis(p) for p in Path('tmp/scogs_a').glob('*.mafft'))
F=sorted(pis(p) for p in Path('tmp/scogs_f').glob('*.mafft'))
for alt in ['two-sided','less','greater']:
  for cont in [True,False]:
    res=st.mannwhitneyu(A,F,alternative=alt,use_continuity=cont,method='auto')
    print('A,F',alt,cont,res.statistic,res.pvalue)
  res=st.mannwhitneyu(F,A,alternative=alt,use_continuity=True,method='auto')
  print('F,A',alt,True,res.statistic,res.pvalue)
PY"
