/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 c=Counter()
 for n in ZipFile(fn).namelist():
  if n.endswith('.mafft'):
   b=ZipFile(fn).read(n).decode()
   c.update(ch for ch in b if ch not in '>\\n\\r')
 print(fn, sorted(c.items()))
PY"
