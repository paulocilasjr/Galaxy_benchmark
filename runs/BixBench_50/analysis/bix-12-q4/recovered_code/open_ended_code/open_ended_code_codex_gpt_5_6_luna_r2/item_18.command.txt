from zipfile import ZipFile
from pathlib import PurePosixPath
from collections import Counter
from statistics import median

def fa(b):
 rec=[]; h=None;s=[]
 for l in b.decode().splitlines():
  if l.startswith('>'):
   if h is not None:rec.append((h,''.join(s)))
   h=l[1:].strip();s=[]
  else:s.append(l.strip())
 if h is not None:rec.append((h,''.join(s)))
 return rec

def m(rs):
 seq=[s.upper() for _,s in rs];L=len(seq[0]); p=0;pg=0
 for c in zip(*seq):
  ng=[x for x in c if x not in '-?']; co=Counter(ng); cg=Counter(c)
  p+=len(co)>=2 and min(co.values())>=2
  pg+=len(cg)>=2 and min(cg.values())>=2
 return len(seq),L,p,pg
D={}
for g,f in [('a','inputs/scogs_animals.zip'),('f','inputs/scogs_fungi.zip')]:
 with ZipFile(f) as z:
  D[g]={PurePosixPath(n).name[:-11]:m(fa(z.read(n))) for n in z.namelist() if n.endswith('.mafft')}
print('set sizes',len(D['a']),len(D['f']),'overlap',len(set(D['a'])&set(D['f'])))
# U function
def u(a,b):
 x=sorted([(v,0) for v in a]+[(v,1) for v in b]); rsum=0; i=0; rank=1
 while i<len(x):
  j=i
  while j<len(x) and x[j][0]==x[i][0]:j+=1
  ar=(rank+rank+j-i-1)/2
  rsum+=ar*sum(g==0 for v,g in x[i:j]); rank+=j-i;i=j
 return rsum-len(a)*(len(a)+1)/2
for scope in ['all','full','common','common_fullboth']:
 if scope=='all': ids_a=set(D['a']);ids_f=set(D['f'])
 elif scope=='full': ids_a={i for i,x in D['a'].items() if x[0]==4};ids_f={i for i,x in D['f'].items() if x[0]==4}
 elif scope=='common': ids_a=ids_f=set(D['a'])&set(D['f'])
 else: ids_a=ids_f={i for i in D['a'] if i in D['f'] and D['a'][i][0]==4 and D['f'][i][0]==4}

   if denom"'!='"'L':continue

   print(scope,'metric',metric_index,'n',len(av),len(fv),'U',u(av,fv))
