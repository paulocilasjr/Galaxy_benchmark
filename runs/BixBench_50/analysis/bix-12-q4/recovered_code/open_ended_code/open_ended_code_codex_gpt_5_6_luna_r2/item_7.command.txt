/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
from pathlib import PurePosixPath
import re, statistics

def fasta(txt):
    rec=[]; name=None; seq=[]
    for ln in txt.decode().splitlines():
        if ln.startswith('>'):
            if name is not None: rec.append((name,''.join(seq)))
            name=ln[1:].strip(); seq=[]
        else: seq.append(ln.strip())
    if name is not None: rec.append((name,''.join(seq)))
    return rec
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\n###',fn)
    with ZipFile(fn) as z:
      nms=[n for n in z.namelist() if n.endswith('.mafft')]
      stats=Counter(); labelsets=Counter(); lengths=Counter(); records=[]
      for n in nms:
        rs=fasta(z.read(n)); labels=tuple(x[0] for x in rs)
        lens=tuple(len(x[1]) for x in rs)
        stats[len(rs)]+=1; labelsets[labels]+=1; lengths[lens]+=1
        records.append((n,rs))
      print('n alignments',len(records),'seq counts',stats)
      print('label sets',len(labelsets))
      for labels,c in labelsets.most_common(10): print(c,labels)
      print('length consistency',sum(len(set(len(s) for _,s in rs))>1 for _,rs in [r for _,r in records]), 'of',len(records))
      print('alignment length summary',min(len(rs[0][1]) for _,rs in records),max(len(rs[0][1]) for _,rs in records),statistics.median(len(rs[0][1]) for _,rs in records))
      # print unusual nseq examples
      for n,rs in records:
        if len(rs)"'!=4:
          print('"'unusual',n,'n=',len(rs),'labels',[x[0] for x in rs],'L',[len(x[1]) for x in rs]);
          if sum(1 for _ in range(1))>10: break
      # exact group ID overlap with raw faa
      raw={PurePosixPath(n).name.removesuffix('.faa') for n in z.namelist() if n.endswith('.faa')}
      ma={PurePosixPath(n).name.removesuffix('.faa.mafft') for n in nms}
      print('raw',len(raw),'ma',len(ma),'overlap',len(raw&ma),'raw only',len(raw-ma),'ma only',len(ma-raw))
      print('raw-only sample',sorted(raw-ma)[:10])
PY"
