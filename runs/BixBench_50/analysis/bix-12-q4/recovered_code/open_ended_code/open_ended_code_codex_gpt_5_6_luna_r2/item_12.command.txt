/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import PurePosixPath
from statistics import median

def fasta(data):
    rec=[]; name=None; seq=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(seq)))
            name=line[1:].strip(); seq=[]
        elif line.strip(): seq.append(line.strip())
    if name is not None: rec.append((name,''.join(seq)))
    return rec

def metrics(rs):
    seqs=[s.upper() for _,s in rs]; n=len(seqs); L=len(seqs[0])
    out=dict(n=n,L=L,pis_ng=0,pis_gap=0,valid=0,complete=0,var_ng=0,var_gap=0)
    for col in zip(*seqs):
        c=list(col); cs=[x for x in c if x not in '-?']
        counts=Counter(cs); counts_gap=Counter(c)
        if len(cs)>=1: out['valid']+=1
        if len(cs)==n: out['complete']+=1
        if len(counts)>=2: out['var_ng']+=1
        if len(counts_gap)>=2: out['var_gap']+=1
        if len(counts)>=2 and min(counts.values())>=2: out['pis_ng']+=1
        if len(counts_gap)>=2 and min(counts_gap.values())>=2: out['pis_gap']+=1
    return out

def mann_u(a,b):
    # Exact midrank U for first sample. Equal values get average ranks.
    vals=sorted([(v,0) for v in a]+[(v,1) for v in b], key=lambda x:x[0])
    ranks=[0.0]*(len(vals)); i=0; rank=1
    while i<len(vals):
        j=i+1
        while j<len(vals) and vals[j][0]==vals[i][0]: j+=1
        av=(rank+(rank+(j-i)-1))/2
        for k in range(i,j): ranks[k]=av
        rank+=j-i; i=j
    r1=sum(r for (v,g),r in zip(vals,ranks) if g==0)
    return r1-len(a)*(len(a)+1)/2

data={}
for group,fn in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
  rows=[]
  with ZipFile(fn) as z:
    for nm in z.namelist():
      if nm.endswith('.mafft'):
        rows.append((PurePosixPath(nm).name,metrics(fasta(z.read(nm)))))
  data[group]=dict(rows)
  print('\\n',group,'n',len(rows),'n counts',Counter(x['n'] for _,x in rows))
  for k in ['pis_ng','pis_gap','L','valid','complete','var_ng','var_gap']:
    vals=[x[k] for _,x in rows]
    print(k,'sum',sum(vals),'min',min(vals),'max',max(vals),'median',median(vals))
  print('nonzero PIS ng',sum(x['pis_ng']>0 for _,x in rows),'gap',sum(x['pis_gap']>0 for _,x in rows))
  print('top PIS ng')
  for nm,x in sorted(rows,key=lambda t:t[1]['pis_ng'],reverse=True)[:5]: print(nm,x)

for taxscope in ['all','full4']:
  for num in ['pis_ng','pis_gap']:
    for den in ['L','valid','complete']:
      vals={}
      for g in ['animals','fungi']:
        rows=[x for x in data[g].values() if taxscope=='all' or x['n']==4]
        vals[g]=[100*x[num]/x[den] if x[den] else 0 for x in rows]
      u=mann_u(vals['animals'],vals['fungi'])
      print(taxscope,num,den,'n',len(vals['animals']),len(vals['fungi']),'U(a,f)',u,'med',median(vals['animals']),median(vals['fungi']))
PY"
