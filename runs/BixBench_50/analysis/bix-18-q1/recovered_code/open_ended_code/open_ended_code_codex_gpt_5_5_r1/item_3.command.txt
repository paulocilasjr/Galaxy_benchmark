/bin/bash -lc "mkdir -p /workspace/codex_output && python3 - <<'PY'
import csv
from collections import defaultdict
rows=[]
with open('/workspace/inputs/Swarm_1.csv', newline='') as f:
    for row in csv.DictReader(f):
        rows.append(row)
by=defaultdict(list)
for r in rows:
    by[r['Genotype']].append(r)
means={g: sum(float(r['Area']) for r in rs)/len(rs) for g, rs in by.items()}
best=max(means, key=means.get)
mean_circ=sum(float(r['Circularity']) for r in by[best])/len(by[best])
with open('/workspace/codex_output/answer.txt','w', newline='') as f:
    f.write(f'{mean_circ:g}\\n')
print(best, means[best], f'{mean_circ:g}')
PY
cat /workspace/codex_output/answer.txt"
