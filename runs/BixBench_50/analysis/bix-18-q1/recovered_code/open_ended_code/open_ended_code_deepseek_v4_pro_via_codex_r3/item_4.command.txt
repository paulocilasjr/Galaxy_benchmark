/bin/bash -lc "python3 - <<'PY'
import csv, collections
path='/workspace/inputs/Swarm_1.csv'
rows=list(csv.DictReader(open(path)))
areas=collections.defaultdict(list); circ=collections.defaultdict(list)
for r in rows:
    areas[r['Genotype']].append(float(r['Area']))
    circ[r['Genotype']].append(float(r['Circularity']))
means={g:sum(v)/len(v) for g,v in areas.items()}
g=max(means, key=means.get)
print(means)
print('max genotype:', g, means[g])
print('mean circularity:', sum(circ[g])/len(circ[g]))
PY"
