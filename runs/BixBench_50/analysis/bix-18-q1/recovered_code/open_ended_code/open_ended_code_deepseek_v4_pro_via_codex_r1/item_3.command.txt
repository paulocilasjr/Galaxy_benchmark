/bin/bash -lc "python3 - <<'PY'
import csv, statistics
from collections import defaultdict
p='/workspace/inputs/Swarm_1.csv'
areas=defaultdict(list); circ=defaultdict(list)
with open(p, newline='') as f:
    r=csv.DictReader(f)
    for row in r:
        areas[row['Genotype']].append(float(row['Area']))
        circ[row['Genotype']].append(float(row['Circularity']))
rows=[]
for g in areas:
    ma=statistics.mean(areas[g])
    mc=statistics.mean(circ[g])
    rows.append((ma,g,mc,len(areas[g])))
rows.sort(reverse=True)
for x in rows: print(x)
best=rows[0][1]
print('best', best, 'mean_circ', statistics.mean(circ[best]))
print('all circ', sorted(circ[best]))
print('mean',sum(circ[best])/len(circ[best]))
PY"
