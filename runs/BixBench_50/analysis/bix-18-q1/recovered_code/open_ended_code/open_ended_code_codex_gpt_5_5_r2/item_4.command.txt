/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal

rows=[]
with open('inputs/Swarm_1.csv', newline='') as f:
    reader=csv.DictReader(f)
    for row in reader:
        rows.append(row)

by=defaultdict(list)
for r in rows:
    by[r['Genotype']].append(r)

mean_area={g: sum(Decimal(r['Area']) for r in rs)/Decimal(len(rs)) for g,rs in by.items()}
top=max(mean_area, key=mean_area.get)
mean_circ=sum(Decimal(r['Circularity']) for r in by[top])/Decimal(len(by[top]))

import os
os.makedirs('codex_output', exist_ok=True)
with open('codex_output/answer.txt','w', newline='') as out:
    out.write(format(mean_circ, 'f') + '\\n')
print(top, mean_area[top], mean_circ)
PY
cat codex_output/answer.txt"
