import csv

areas = {'ΔlasR': [], 'Wildtype': []}
with open('inputs/Swarm_1.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        gt = row['Genotype']
        if gt in areas:
            areas[gt].append(float(row['Area']))

wt_mean = sum(areas['Wildtype']) / len(areas['Wildtype'])
mut_mean = sum(areas['ΔlasR']) / len(areas['ΔlasR'])
pct_reduction = (wt_mean - mut_mean) / wt_mean * 100

print(f'WT areas: {areas[\"Wildtype\"]}')
print(f'ΔlasR areas: {areas[\"ΔlasR\"]}')
print(f'WT mean: {wt_mean}')
print(f'ΔlasR mean: {mut_mean}')
print(f'Percent reduction: {pct_reduction:.4f}%')
print(f'Approximate: {round(pct_reduction)}%')
