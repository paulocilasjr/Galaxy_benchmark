import csv

with open('inputs/Swarm_1.csv') as f:
    reader = csv.DictReader(f)
    rows = list(reader)

# Extract areas by genotype
lasR_areas = [float(r['Area']) for r in rows if r['Genotype'] == 'ΔlasR']
wt_areas = [float(r['Area']) for r in rows if r['Genotype'] == 'Wildtype']

lasR_mean = sum(lasR_areas) / len(lasR_areas)
wt_mean = sum(wt_areas) / len(wt_areas)

pct_reduction = ((wt_mean - lasR_mean) / wt_mean) * 100

print(f'ΔlasR mean: {lasR_mean}')
print(f'Wildtype mean: {wt_mean}')
print(f'Percent reduction: {pct_reduction}')
