/bin/bash -lc "python3 - <<'PY'
import csv, statistics
with open('inputs/Swarm_1.csv', newline='') as f:
    rows=list(csv.DictReader(f))
for g in ['Wildtype','ΔlasR']:
    vals=[float(r['Area']) for r in rows if r['Genotype']==g]
    print(g, vals, 'mean', statistics.mean(vals))
lasr = statistics.mean([float(r['Area']) for r in rows if r['Genotype']=='ΔlasR'])
wt = statistics.mean([float(r['Area']) for r in rows if r['Genotype']=='Wildtype'])
red=(wt-lasr)/wt*100
print('reduction%', red)
print('rounded 1 decimal', round(red,1), 'rounded int', round(red))
PY"
