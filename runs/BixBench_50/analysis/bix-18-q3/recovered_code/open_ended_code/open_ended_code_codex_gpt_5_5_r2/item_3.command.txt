/bin/bash -lc "python - <<'PY'
import csv
from statistics import mean
from pathlib import Path
p=Path('/workspace/inputs/Swarm_1.csv')
vals={}
with p.open(newline='') as f:
    for r in csv.DictReader(f):
        vals.setdefault(r['Genotype'],[]).append(float(r['Area']))
wt=mean(vals['Wildtype'])
mut=mean(vals['ΔlasR'])
red=(wt-mut)/wt*100
print(wt, mut, red)
Path('/workspace/codex_output').mkdir(exist_ok=True)
Path('/workspace/codex_output/answer.txt').write_text(f'{red:.0f}%\\n')
PY
cat /workspace/codex_output/answer.txt"
