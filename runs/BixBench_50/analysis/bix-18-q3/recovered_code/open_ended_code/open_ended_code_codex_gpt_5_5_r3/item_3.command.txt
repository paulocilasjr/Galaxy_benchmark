/bin/bash -lc "python - <<'PY'
import csv
from statistics import mean
path='/workspace/inputs/Swarm_1.csv'
vals={}
with open(path, newline='') as f:
    for row in csv.DictReader(f):
        vals.setdefault(row['Genotype'], []).append(float(row['Area']))
wt=mean(vals['Wildtype'])
mut=mean(vals['ΔlasR'])
red=(wt-mut)/wt*100
print('WT mean', wt)
print('ΔlasR mean', mut)
print('percent reduction', red)
PY"
