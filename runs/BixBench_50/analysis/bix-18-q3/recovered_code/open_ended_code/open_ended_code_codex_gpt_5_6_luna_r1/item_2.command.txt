/bin/bash -lc "python - <<'PY'
import csv
from statistics import mean
with open('inputs/Swarm_1.csv', newline='') as f:
    rows = list(csv.DictReader(f))
for genotype in ('ΔlasR', 'Wildtype'):
    vals = [float(row['Area']) for row in rows if row['Genotype'] == genotype]
    print(genotype, vals, 'n=', len(vals), 'mean=', mean(vals))
mut = mean(float(r['Area']) for r in rows if r['Genotype'] == 'ΔlasR')
wt = mean(float(r['Area']) for r in rows if r['Genotype'] == 'Wildtype')
print('reduction_percent=', 100*(wt-mut)/wt)
PY"
