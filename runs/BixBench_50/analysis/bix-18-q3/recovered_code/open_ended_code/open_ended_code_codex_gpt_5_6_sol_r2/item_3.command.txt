/bin/bash -lc "python - <<'PY'
import csv
from pathlib import Path
p=Path('inputs/Swarm_1.csv')
vals={}
with p.open(newline='') as f:
    for row in csv.DictReader(f):
        vals.setdefault(row['Genotype'], []).append(float(row['Area']))
for g in ('ΔlasR','Wildtype'):
    print(g, vals[g], sum(vals[g])/len(vals[g]))
mut=sum(vals['ΔlasR'])/len(vals['ΔlasR'])
wt=sum(vals['Wildtype'])/len(vals['Wildtype'])
print('percent_reduction', (wt-mut)/wt*100)
PY"
