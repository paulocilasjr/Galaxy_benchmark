import csv
import math

rows = []
with open('inputs/Swarm_2.csv') as f:
    reader = csv.reader(f)
    for row in reader:
        rows.append(row)

# Skip header row (row 0)
data = []
for row in rows[1:]:
    data.append({
        'DishNumber': int(row[0]),
        'StrainNumber': row[1],
        'Ratio': row[2],
        'Replicate': row[3],
        'Area': float(row[4]),
        'Circularity': float(row[5]),
        'Round': float(row[6])
    })

# Strain 1 alone
strain1 = [d for d in data if d['StrainNumber'] == '1']
mean_area_1 = sum(d['Area'] for d in strain1) / len(strain1)
mean_circ_1 = sum(d['Circularity'] for d in strain1) / len(strain1)
print(f"Strain 1: Mean Area = {mean_area_1:.2f}, Mean Circularity = {mean_circ_1:.4f}")

# Mixed cultures
mixed = [d for d in data if d['StrainNumber'] == '287_98']
ratios = sorted(set(d['Ratio'] for d in mixed))

# Compute mean for each ratio
results = {}
for ratio in ratios:
    subset = [d for d in mixed if d['Ratio'] == ratio]
    mean_area = sum(d['Area'] for d in subset) / len(subset)
    mean_circ = sum(d['Circularity'] for d in subset) / len(subset)
    results[ratio] = (mean_area, mean_circ)
    print(f"Ratio {ratio}: Mean Area = {mean_area:.2f}, Mean Circularity = {mean_circ:.4f}")

print()

# Compute Euclidean distance in standardized space
# Use mean and std across all mixed samples for standardization
mixed_areas = [d['Area'] for d in mixed]
mixed_circs = [d['Circularity'] for d in mixed]

def mean(vals):
    return sum(vals) / len(vals)

def std(vals):
    m = mean(vals)
    return math.sqrt(sum((v - m)**2 for v in vals) / (len(vals) - 1))

mixed_area_mean = mean(mixed_areas)
mixed_area_std = std(mixed_areas)
mixed_circ_mean = mean(mixed_circs)
mixed_circ_std = std(mixed_circs)

# Standardize strain 1
z_area_1 = (mean_area_1 - mixed_area_mean) / mixed_area_std
z_circ_1 = (mean_circ_1 - mixed_circ_mean) / mixed_circ_std

print(f"Standardized Strain 1: z_area = {z_area_1:.4f}, z_circ = {z_circ_1:.4f}")

# Compute distances
best_ratio = None
best_dist = float('inf')
for ratio in ratios:
    mean_area, mean_circ = results[ratio]
    z_area = (mean_area - mixed_area_mean) / mixed_area_std
    z_circ = (mean_circ - mixed_circ_mean) / mixed_circ_std
    dist = math.sqrt((z_area - z_area_1)**2 + (z_circ - z_circ_1)**2)
    print(f"Ratio {ratio}: z_area = {z_area:.4f}, z_circ = {z_circ:.4f}, distance = {dist:.4f}")
    if dist < best_dist:
        best_dist = dist
        best_ratio = ratio

print()
print(f"Best matching ratio: {best_ratio} (distance = {best_dist:.4f})")
