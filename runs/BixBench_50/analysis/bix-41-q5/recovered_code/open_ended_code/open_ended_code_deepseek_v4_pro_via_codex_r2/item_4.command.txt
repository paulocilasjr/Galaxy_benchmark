/bin/bash -lc "python3 - <<'PY'
import csv, math, statistics
from collections import defaultdict
p='/workspace/inputs/Swarm_2.csv'
rows=[]
with open(p) as f:
    for r in csv.DictReader(f):
        r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity']); r['Round']=float(r['Round'])
        rows.append(r)
# means
groups=defaultdict(lambda: {'Area':[], 'Circularity':[], 'Round':[]})
for r in rows:
    key=(r['StrainNumber'],r['Ratio'])
    groups[key]['Area'].append(r['Area']); groups[key]['Circularity'].append(r['Circularity']); groups[key]['Round'].append(r['Round'])
for k,v in groups.items():
    print(k, {m:round(statistics.mean(v[m]),4) for m in ['Area','Circularity','Round']}, 'n',len(v['Area']))
refA=statistics.mean(groups[('1','1:0')]['Area']); refC=statistics.mean(groups[('1','1:0')]['Circularity'])
print('ref',refA,refC)
allA=[r['Area'] for r in rows]; allC=[r['Circularity'] for r in rows]
meanA=statistics.mean(allA); sdA=statistics.stdev(allA)
meanC=statistics.mean(allC); sdC=statistics.stdev(allC)
print('stats',meanA,sdA,meanC,sdC)
print('\\nraw, z-euclidean, range-normalized, minmax per')
for k in sorted(groups):
    if k[0]"'!='"'287_98': continue
    a=statistics.mean(groups[k]['Area']); c=statistics.mean(groups[k]['Circularity'])
    raw=math.hypot(a-refA,c-refC)
    z=math.hypot((a-meanA)/sdA-(refA-meanA)/sdA,(c-meanC)/sdC-(refC-meanC)/sdC)
    rng_a=max(allA)-min(allA); rng_c=max(allC)-min(allC)
    rng=math.hypot((a-refA)/rng_a,(c-refC)/rng_c)
    # log area
    zlog=math.hypot((math.log(a)-math.log(refA))/math.log(10), (c-refC)/(max(allC)-min(allC)))
    print(k[1], 'mean',round(a,2),round(c,4),'raw',round(raw,2),'z',round(z,4),'range',round(rng,4),'zlog-ish',round(zlog,4))
PY"
