import pandas as pd, numpy as np
p='/workspace/inputs/Swarm_2.csv'
df=pd.read_csv(p)
print(df.groupby(['StrainNumber','Ratio']).agg(area=('Area','mean'),circ=('Circularity','mean'),round=('Round','mean'),n=('Area','size')).round(4))

print('ref',ref.to_dict())
mix=df[df.StrainNumber.eq('287_98')].groupby('Ratio')[['Area','Circularity']].mean()
print('mix\\n',mix)
# distances raw
for norm in ['raw','z','range']:
    if norm=='raw':
        vals=df[['Area','Circularity']]
        refv=ref
        mixv=mix
        d=(mixv-refv).pow(2).sum(axis=1)**0.5
    elif norm=='z':
        vals=df[['Area','Circularity']]
        zdf=(df[['Area','Circularity']]-vals.mean())/vals.std(ddof=0)
        zref=(ref-vals.mean())/vals.std(ddof=0)
        zmix=(mix-vals.mean())/vals.std(ddof=0)
        d=(zmix-zref).pow(2).sum(axis=1)**0.5
    elif norm=='range':
        rng=df[['Area','Circularity']].max()-df[['Area','Circularity']].min()
        d=(((mix-ref)/rng)**2).sum(axis=1)**0.5
    print(norm, d.idxmin(), d.min(), '\\n', d.round(4))
# also distance based only mixed but scale std among all maybe same
