/bin/bash -lc "python - <<'PY'
import csv, math
from collections import defaultdict
rows=list(csv.DictReader(open('inputs/Swarm_2.csv')))
for r in rows:r['Area']=float(r['Area']);r['Circularity']=float(r['Circularity'])
mean=lambda rs,k:sum(r[k] for r in rs)/len(rs)
ref=[r for r in rows if r['StrainNumber']=='1']; ta,tc=mean(ref,'Area'),mean(ref,'Circularity')
g=defaultdict(list)
for r in rows:
 if r['StrainNumber']=='287_98':g[r['Ratio']].append(r)
vals=[]
for q,rs in g.items(): vals.append((q,mean(rs,'Area'),mean(rs,'Circularity')))
# distances with each metric normalized by target, pooled range, std across group means, reference replicate sd
import statistics
for name,sa,sc in [('target',ta,tc),('range',max(x[1] for x in vals)-min(x[1] for x in vals),max(x[2] for x in vals)-min(x[2] for x in vals)),('sdgroups',statistics.pstdev([x[1] for x in vals]),statistics.pstdev([x[2] for x in vals])),('refsd',statistics.pstdev([r['Area'] for r in ref]),statistics.pstdev([r['Circularity'] for r in ref]))]:
 d=[]
 for q,a,c in vals:
  d.append((math.hypot((a-ta)/sa,(c-tc)/sc),q))
 print(name,sorted(d))
PY"
