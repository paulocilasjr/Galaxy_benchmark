import csv, statistics, math
from collections import defaultdict
p='inputs/Swarm_2.csv'
rows=list(csv.DictReader(open(p)))

means={k:(statistics.mean(x[0] for x in v),statistics.mean(x[1] for x in v)) for k,v in by.items()}
t=means[('1','1:0')]
print('target',t)
print('ratio, mean_area, delta_area, mean_circ, delta_circ, raw_euclidean, relative_euclidean')
for (strain,ratio),(a,c) in means.items():
    if strain=='287_98':
        raw=math.hypot(a-t[0],c-t[1])
        rel=math.hypot((a-t[0])/t[0],(c-t[1])/t[1])
        print(ratio,a,a-t[0],c,c-t[1],raw,rel,sep=',')
PY"
