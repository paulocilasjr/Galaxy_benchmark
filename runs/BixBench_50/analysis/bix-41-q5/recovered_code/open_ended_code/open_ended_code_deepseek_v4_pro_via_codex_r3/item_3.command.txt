/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
from math import dist
p='/workspace/inputs/Swarm_2.csv'
df=pd.read_csv(p)
print(df.groupby(['StrainNumber','Ratio'])[['Area','Circularity']].agg(['mean','std','count']))
ref=df[(df.StrainNumber==1)].mean(numeric_only=True)
print('ref', ref)
mixed=df[df.StrainNumber=='287_98']
groups=mixed.groupby('Ratio')[['Area','Circularity']].mean()
print('groups')
print(groups)
# raw Euclidean distances
for r,g in groups.iterrows():
    print(r, 'area diff',g.Area-ref.Area,'circ diff',g.Circularity-ref.Circularity,'raw',np.sqrt((g.Area-ref.Area)**2+(g.Circularity-ref.Circularity)**2))
# normalize by range across mixed or all groups
allg=df.groupby(['StrainNumber','Ratio'])[['Area','Circularity']].mean()
rng=allg.max()-allg.min(); print('range all', rng)
norm=(allg-allg.min())/rng
refn=norm.loc[(1,'1:0')]
for idx,g in norm.loc['287_98'].iterrows():
    print('normall',idx,'d',np.sqrt(((g-refn)**2).sum()), 'coords',g.to_dict())
# z-score using group means std
zs=(allg-allg.mean())/allg.std()
refz=zs.loc[(1,'1:0')]
for idx,g in zs.loc['287_98'].iterrows():
    print('z',idx,'d',np.sqrt(((g-refz)**2).sum()), 'coords',g.to_dict())
PY"
