/bin/bash -c "python3 - <<'PY'
import csv, math, statistics
rows=[]
for r in csv.DictReader(open('/workspace/inputs/Swarm_2.csv')):
    r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity']); rows.append(r)
ref=[r for r in rows if r['StrainNumber']=='1']
ref_log=(statistics.mean(math.log(r['Area']) for r in ref), statistics.mean(r['Circularity'] for r in ref))
print('ref',ref_log)
for ratio in sorted(set(r['Ratio'] for r in rows if r['StrainNumber']=='287_98'), key=lambda s: eval(s.replace(':','/'))):
    rs=[r for r in rows if r['StrainNumber']=='287_98' and r['Ratio']==ratio]
    gm=(statistics.mean(math.log(r['Area']) for r in rs), statistics.mean(r['Circularity'] for r in rs))
    d=math.dist(gm,ref_log)
    print(ratio,gm,'d',d)
PY"
