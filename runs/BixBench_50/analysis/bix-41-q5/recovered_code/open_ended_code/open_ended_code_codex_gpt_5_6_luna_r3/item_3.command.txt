/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
p='inputs/Swarm_2.csv'
g=defaultdict(list)
with open(p,newline='') as f:
    for row in csv.DictReader(f):
        key=(row['StrainNumber'],row['Ratio'])
        g[key].append((float(row['Area']),float(row['Circularity'])))
for key, vals in g.items():
    ma=sum(x for x,y in vals)/len(vals)
    mc=sum(y for x,y in vals)/len(vals)
    print(key, len(vals), f'area={ma:.6f}', f'circ={mc:.6f}')
PY"
