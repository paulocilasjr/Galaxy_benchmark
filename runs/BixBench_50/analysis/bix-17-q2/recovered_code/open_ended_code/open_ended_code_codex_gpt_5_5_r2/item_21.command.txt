/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import glob, os, statistics
from collections import Counter

def iter_rows(path):
    wb=load_workbook(path, read_only=True, data_only=True)
    ws=wb[wb.sheetnames[0]]; ws.reset_dimensions()
    yield from ws.iter_rows(values_only=True)

def sid_from_path(f):
    return os.path.basename(f).replace('230209_Exome_GRCh38_CHIP_','').replace('.xlsx','').split('-')[0].strip()

def fnum(x):
    try: return float(str(x).strip()) if x is not None and str(x).strip()"'!='"'' else None
    except: return None

def excl(so):
    s=str(so or '').lower()
    return any(k in s for k in ['intron','intergenic','utr'])
files={sid_from_path(f):f for f in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')}
rows=list(iter_rows('inputs/230215_Trio_Status.xlsx'))
h=rows[0]; idx={v:i for i,v in enumerate(h) if v}
meta={str(r[idx['Sample ID']]).strip(): {'Status':str(r[idx['Status']]).strip(), 'BLM':str(r[idx['BLM Mutation Status']]).strip()} for r in rows[1:] if r and r[0] is not None}

def count(sid):
    rows=iter_rows(files[sid]); next(rows); next(rows)
    n=0
    for r in rows:
        if not r or r[0] is None: continue
        if str(r[4] or '').strip() == 'Reference': continue
        if excl(r[13] if len(r)>13 else None): continue
        v=fnum(r[9] if len(r)>9 else None)
        if v is None or v >= 0.3: continue
        n+=1
    return n
for label, pred in [
    ('BLM=Carrier', lambda m:m['BLM']=='Carrier'),
    ('BLM=Affected', lambda m:m['BLM']=='Affected'),
    ('BLM Carrier or Affected', lambda m:m['BLM'] in ['Carrier','Affected']),
    ('Status Patient', lambda m:m['Status']=='Patient'),
]:
    ids=[sid for sid,m in meta.items() if pred(m) and sid in files]
    vals=[count(sid) for sid in ids]
    print(label, 'n', len(vals), 'ids', ids, 'vals', vals, 'median', statistics.median(vals) if vals else None)
PY"
