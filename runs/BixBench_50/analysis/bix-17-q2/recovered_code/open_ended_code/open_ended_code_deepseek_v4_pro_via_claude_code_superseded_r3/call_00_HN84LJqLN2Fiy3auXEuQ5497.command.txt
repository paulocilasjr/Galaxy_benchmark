from openpyxl import load_workbook
import os

# Get all unique values in key columns across a few CHIP files
chip_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'
files = sorted(os.listdir(chip_dir))

# Check Sequence Ontology values and In_CHIP values across files
seq_onto = set()
in_chip_vals = set()
zyg_vals = set()

for f in files[:5]:
    wb = load_workbook(os.path.join(chip_dir, f))
    ws = wb.active
    header = [cell.value for cell in ws[1]]
    # Find column indices
    try:
        seq_idx = header.index('Sequence Ontology (Combined)')
        chip_idx = header.index('In_CHIP')
        zyg_idx = header.index('Zygosity')
        vaf_idx = header.index('Variant Allele Freq')
    except ValueError as e:
        print(f'Column not found in {f}: {e}')
        continue
    
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[seq_idx]:
            seq_onto.add(row[seq_idx])
        if row[chip_idx] is not None:
            in_chip_vals.add(str(row[chip_idx]))
        if row[zyg_idx]:
            zyg_vals.add(row[zyg_idx])

print("Sequence Ontology values (sample):")
for v in sorted(seq_onto)[:30]:
    print(f"  {v}")
print(f"  ... total unique: {len(seq_onto)}")
print()

print("In_CHIP values:", in_chip_vals)
print("Zygosity values:", zyg_vals)

# Print all Sequence Ontology values containing intron, intergenic, UTR
print("\n=== UTR/intronic/intergenic related values ===")
for v in sorted(seq_onto):
    vl = v.lower()
    if 'utr' in vl or 'intron' in vl or 'intergenic' in vl:
        print(f"  {v}")
