from openpyxl import load_workbook

# Trio Status - check all unique BLM Mutation Status values
wb = load_workbook('/workspace/inputs/230215_Trio_Status.xlsx')
ws = wb['Sheet1']
print("=== Trio Status ===")
header = [cell.value for cell in ws[1]]
print('Header:', header)

statuses = set()
all_rows = []
for row in ws.iter_rows(min_row=2, values_only=True):
    all_rows.append(row)
    statuses.add(row[3])  # BLM Mutation Status
print('Unique BLM Mutation Status:', statuses)
print('Total rows:', len(all_rows))
print()
