from pathlib import Path
from openpyxl import load_workbook
from collections import Counter, defaultdict
import re, statistics

base=Path('inputs/CHIP_DP10_GQ20_PASS')
files=sorted(base.glob('*.xlsx'))
print('variant files',len(files))
headers=Counter(); seq=Counter(); zyg=Counter(); inchip=Counter(); in_genes=defaultdict(set); rows_per=[]; samples=[]
for p in files:
    wb=load_workbook(p,read_only=False,data_only=True)
    ws=wb.active
    vals=list(ws.iter_rows(values_only=True))
    h1=vals[0]; h2=vals[1]
    headers[tuple(h2)]+=1
    samples.append((p.name, h1[4], len(vals)-2, ws.max_row, ws.max_column))
    for row in vals[2:]:
        if not row or all(x is None for x in row): continue
        seq[str(row[13]).strip() if row[13] is not None else None]+=1
        zyg[str(row[4]).strip() if row[4] is not None else None]+=1
        inchip[str(row[29]).strip() if row[29] is not None else None]+=1
        if str(row[29]).strip()=='True': in_genes[str(row[12]).strip()].add(p.name)
    wb.close()
print('header variants',len(headers), [v for v in headers.values()])
print('rows per file range',min(x[2] for x in samples),max(x[2] for x in samples), 'total',sum(x[2] for x in samples))
print('seq ontology unique/counts')
for k,v in seq.most_common(): print(repr(k),v)
print('zyg unique/counts',zyg)
print('inCHIP',inchip)
chip_wb=load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx',read_only=False,data_only=True)
chip_genes={str(r[0]).strip() for r in chip_wb.active.iter_rows(values_only=True) if r[0] is not None}
print('chip genes',len(chip_genes),sorted(chip_genes))
print('In_CHIP true gene set',len(in_genes), sorted(in_genes))
print('In_CHIP true not list',sorted(set(in_genes)-chip_genes))
print('list genes not In_CHIP',sorted(chip_genes-set(in_genes)))
print('first sample records')
for x in samples[:10]: print(x)
PY"
