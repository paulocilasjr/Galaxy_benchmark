import openpyxl, glob, collections
chip=set()
ws=openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx', data_only=True).active
for r in ws.iter_rows(values_only=True):
 if r[0]: chip.add(str(r[0]))
# For each gene string, get match values
matchby=collections.defaultdict(collections.Counter)
for f in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'):
 w=openpyxl.load_workbook(f,data_only=True).active
 for r in w.iter_rows(min_row=3,values_only=True):
  g=str(r[12]); parts=[p for p in g.split(',') if p]
  if not any(p in chip for p in parts):
   matchby[g][str(r[30])]+=1
for g,ms in matchby.items(): print(g, ms)
PY"
