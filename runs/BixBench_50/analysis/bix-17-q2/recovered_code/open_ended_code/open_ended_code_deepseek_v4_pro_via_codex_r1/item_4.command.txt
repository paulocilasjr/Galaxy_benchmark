import openpyxl, glob
files=sorted(glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))

        for r in rows[:12]: print(r)
        print('last rows:')
        for r in rows[-3:]: print(r)
    wb.close()
