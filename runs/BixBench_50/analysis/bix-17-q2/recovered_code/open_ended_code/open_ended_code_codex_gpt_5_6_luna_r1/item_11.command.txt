from pathlib import Path
from collections import Counter, defaultdict
import openpyxl, re
from statistics import median
root=Path('/workspace/inputs')
# load status and chip genes (the latter is used to validate In_CHIP/match semantics)
wb=openpyxl.load_workbook(root/'230215_Trio_Status.xlsx',data_only=True)
carrier={str(r[0]).strip() for r in wb.active.iter_rows(min_row=2,values_only=True) if r[0] is not None and str(r[3]).strip()=='Carrier'}
wb=openpyxl.load_workbook(root/'230214_Schenz_et_al_2022_CHIP_Genes.xlsx',data_only=True)
chip={str(r[0]).strip() for r in wb.active.iter_rows(values_only=True) if r[0] is not None}

def sample_id(p): return re.search(r'_CHIP_(["'^-.]+)'"',p.name).group(1)
def num(x):
    try:return float(x)
    except:return None

counts=[]
print('FILTERED_ROW_CHECKS')
for p in sorted((root/'CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    sid=sample_id(p)
    if sid not in carrier: continue
    wb=openpyxl.load_workbook(p,data_only=True); ws=wb.active
    it=ws.iter_rows(values_only=True); next(it); hdr=next(it); idx={str(v).strip():i for i,v in enumerate(hdr) if v is not None}
    rows=list(it)
    filt=[]
    for r in rows:
        v=num(r[idx['Variant Allele Freq']]); so=str(r[idx['Sequence Ontology (Combined)']] or '').lower(); z=str(r[idx['Zygosity']] or '').strip(); inchip=str(r[idx['In_CHIP']] or '').strip().lower()
        if inchip=='true' and v is not None and v<0.3 and z"'!='"'reference' and not any(t in so for t in ['intron','intergenic','utr']):
            filt.append(r)
    keys=[(r[idx['Chr:Pos']],r[idx['Ref/Alt']]) for r in filt]
    dup=[(k,n) for k,n in Counter(keys).items() if n>1]
    none_z=[r for r in filt if r[idx['Zygosity']] is None or str(r[idx['Zygosity']]).strip()=='']

    counts.append(len(filt))
