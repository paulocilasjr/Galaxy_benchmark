from pathlib import Path
from collections import Counter

 it=ws.iter_rows(values_only=True); next(it); h=next(it); idx={str(v).strip():i for i,v in enumerate(h) if v is not None}
 for row in it:
  g=str(row[idx['Gene Names']] or '').strip(); genes[g]+=1; examples.setdefault(g,(p.name,row[idx['Sequence Ontology (Combined)']],row[idx['Variant Allele Freq']],row[idx['Zygosity']]))

print('table unique',len(genes),sorted(genes))
print('not in chip',sorted(set(genes)-chip))
