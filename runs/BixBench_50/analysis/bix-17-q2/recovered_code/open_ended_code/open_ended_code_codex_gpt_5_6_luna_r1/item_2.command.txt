/bin/bash -lc "python - <<'PY'
from pathlib import Path
import openpyxl

root = Path('/workspace/inputs')
for path in sorted(root.rglob('*.xlsx')):
    wb = openpyxl.load_workbook(path, read_only=True, data_only=True)
    print(f'FILE\\t{path.relative_to(root)}')
    print(f'SHEETS\\t{wb.sheetnames}')
    for ws in wb.worksheets:
        print(f'SHEET\\t{ws.title}\\trows={ws.max_row}\\tcols={ws.max_column}')
        for row in ws.iter_rows(min_row=1, max_row=min(ws.max_row,5), values_only=True):
            print('ROW\\t' + repr(row))
    print()
PY"
