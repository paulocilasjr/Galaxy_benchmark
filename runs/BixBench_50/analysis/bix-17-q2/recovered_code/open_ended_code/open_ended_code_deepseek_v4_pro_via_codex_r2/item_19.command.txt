/bin/bash -lc "python3 - <<'PY'
import requests
url='https://pmc.ncbi.nlm.nih.gov/articles/instance/10871368/bin/NIHPP2024.02.02.24302163v1-supplement-4.pdf'
r=requests.get(url, timeout=20)
print(r.status_code, r.headers.get('content-type'), len(r.content))
open('/workspace/tmp/supp4.pdf','wb').write(r.content)
PY
file /workspace/tmp/supp4.pdf && ls -l /workspace/tmp/supp4.pdf"
