/bin/bash -lc "python3 - <<'PY'
import openpyxl, glob, os, re, statistics
base='/workspace/inputs'
# Trio status
trio={}
wb=openpyxl.load_workbook(f'{base}/230215_Trio_Status.xlsx', data_only=True)
for r in wb.active.iter_rows(min_row=2, values_only=True):
    if r[0] is None: continue
    sid=str(r[0]).strip()
    trio[sid]=(str(r[1]).strip() if r[1] is not None else '', str(r[3]).strip() if r[3] is not None else '')
# map file keys
files=sorted(glob.glob(f'{base}/CHIP_DP10_GQ20_PASS/*.xlsx'))
exclude_so={'intron_variant','intergenic_variant','3_prime_UTR_variant','5_prime_UTR_variant','5_prime_UTR_premature_start_codon_gain_variant'}
counts={}
for f in files:
    name=os.path.basename(f)
    # number IDs
    m=re.match(r'230209_Exome_GRCh38_CHIP_(\\d+)',name)
    key=m.group(1) if m else re.match(r'230209_Exome_GRCh38_CHIP_(SRR\\d+)',name).group(1)
    ws=openpyxl.load_workbook(f, data_only=False).active
    n=0
    for r in ws.iter_rows(min_row=3, values_only=True):
        zyg=r[4]
        so=r[13]
        vaf=r[9]
        if zyg == 'Reference':
            continue
        if so in exclude_so:
            continue
        if vaf is None:
            continue
        try:
            v=float(vaf)
        except (TypeError,ValueError):
            continue
        if v < 0.3:
            n += 1
    counts[key]=n
carrier_counts=sorted(n for sid,n in counts.items() if trio.get(sid,('',''))[1]=='Carrier')
print('carrier sample IDs/counts:', [(sid,counts[sid]) for sid in sorted(trio) if trio[sid][1]=='Carrier' and sid in counts])
print('carrier_counts_sorted',carrier_counts)
print('n carriers with files',len(carrier_counts))
print('median',statistics.median(carrier_counts))
print('>=2',sum(x>=2 for x in carrier_counts))
PY"
