from openpyxl import load_workbook
from collections import Counter
import glob, os, re, statistics

def read_rows(path):
    wb=load_workbook(path, read_only=True, data_only=True)
    ws=wb[wb.sheetnames[0]]
    try: ws.reset_dimensions()
    except Exception: pass
    return list(ws.iter_rows(values_only=True))

def file_sample(path):
    return os.path.basename(path).replace('.xlsx','').split('230209_Exome_GRCh38_CHIP_')[-1].split('-')[0].strip()

ci={x:i for i,x in enumerate(status_rows[0]) if x is not None}
carriers=[str(r[ci['Sample ID']]).strip() for r in status_rows[1:] if str(r[ci['BLM Mutation Status']]).strip()=='Carrier']
files={file_sample(p):p for p in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')}

def ffloat(x):
    try: return float(x)
    except Exception: return None

def excluded_so(x):
    s=str(x or '').lower()
    return 'intron' in s or 'intergenic' in s or 'utr' in s

def count(p, explicit_match=False):
    rows=read_rows(p); h=rows[1]
    # duplicate Match columns: col 30 is Match under In_CHIP based on workbook section layout.
    col={name:i for i,name in enumerate(h) if name is not None and name not in {}}
    n=0; counted=[]

        if not any(x is not None for x in r): continue
        if explicit_match:
            in_chip=str(r[29] if len(r)>29 else '').strip().lower()=='true'
