from openpyxl import load_workbook
from collections import Counter, defaultdict
import glob, os, re, statistics

def read_rows(path):
    wb=load_workbook(path, read_only=True, data_only=True)
    ws=wb[wb.sheetnames[0]]
    try: ws.reset_dimensions()
    except Exception: pass
    return list(ws.iter_rows(values_only=True))

def file_sample(path):
    return os.path.basename(path).replace('.xlsx','').split('230209_Exome_GRCh38_CHIP_')[-1].split('-')[0].strip()

print('chip genes n', len(chip_genes), sorted(list(chip_genes))[:10], sorted(list(chip_genes))[-10:])

in_chip_counts=Counter(); genes_not_in_list=Counter(); all_so=Counter(); all_effect=Counter(); all_zyg=Counter()
for p in sorted(glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')):
    rows=read_rows(p); header=rows[1]; col={h:i for i,h in enumerate(header) if h is not None}
    for r in rows[2:]:
        if not any(x is not None for x in r): continue
        in_chip_counts[r[col['In_CHIP']] if col['In_CHIP'] < len(r) else None]+=1
        all_so[r[col['Sequence Ontology (Combined)']] if col['Sequence Ontology (Combined)'] < len(r) else None]+=1
        all_effect[r[col['Effect (Combined)']] if col['Effect (Combined)'] < len(r) else None]+=1
        all_zyg[r[col['Zygosity']] if col['Zygosity'] < len(r) else None]+=1

print('In_CHIP counts', in_chip_counts)
print('Top SO', all_so.most_common(20))
print('Effect counts', all_effect)

def ffloat(x):

    s=str(x or '').lower()
    return 'intron' in s or 'intergenic' in s or 'utr' in s

            n += 1
    return n
