/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
from collections import Counter
import re, statistics
# cohort
ws=load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True).active
carriers={str(r[0]).strip() for r in ws.iter_rows(min_row=2,values_only=True) if str(r[3]).strip()=='Carrier'}
files={}
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
 raw=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name).group(1)
 sid=(raw.split('-')[0] if raw[0].isdigit() else raw).strip()
 files[sid]=p
print('carrier ids',sorted(carriers), 'N',len(carriers))
print('missing',sorted(carriers-files.keys()))
all_effect=Counter(); all_so=Counter(); counts={}; qualrows={}
exclude_effect={'Intronic','Intergenic',\"3' UTR\",\"5' UTR\",'3 Prime UTR','5 Prime UTR'}
for sid in sorted(carriers & files.keys()):
 s=load_workbook(files[sid],data_only=True,read_only=False).active
 heads=[c.value for c in s[2]]
 ix={x:i for i,x in enumerate(heads) if x is not None}
 q=[]
 for row in s.iter_rows(min_row=3,values_only=True):
  z=row[ix['Zygosity']]; eff=row[ix['Effect (Combined)']]; so=row[ix['Sequence Ontology (Combined)']]
  all_effect[eff]+=1; all_so[so]+=1
  v=row[ix['Variant Allele Freq']]
  try: v=float(v)
  except (ValueError,TypeError): continue
  if z=='Reference': continue
  # literal friendly Effect categories
  if eff in exclude_effect: continue
  if v < 0.3:
   q.append((row[ix['Chr:Pos']],row[ix['Gene Names']],so,eff,z,v))
 counts[sid]=len(q); qualrows[sid]=q
print('\\nEffects',all_effect)
print('\\nSOs')
for k,v in all_so.most_common(): print(repr(k),v)
print('\\nCounts')
for sid,n in sorted(counts.items(),key=lambda x:(x[1],x[0])):
 print(sid,n)
 for x in qualrows[sid]: print('  ',x)
print('N',len(counts),'sorted',sorted(counts.values()),'median',statistics.median(counts.values()))
PY"
