/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.stats as stats, time
T=time.time()
ch=pd.read_csv('inputs/CRISPRGeneEffect.csv', nrows=0); eh=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', nrows=0)
common=[c for c in ch.columns[1:] if c in eh.columns[1:]]; dt={c:np.float32 for c in common}
cd=pd.read_csv('inputs/CRISPRGeneEffect.csv', usecols=[ch.columns[0]]+common, index_col=0, dtype=dt)
ed=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', usecols=[eh.columns[0]]+common, index_col=0, dtype=dt)
ids=sorted(set(cd.index)&set(ed.index)); cd=cd.loc[ids]; ed=ed.loc[ids]
print('loaded',time.time()-T)
p=np.empty(len(common), dtype=float); rho=np.empty(len(common), dtype=float)
for i in range(len(common)):
    r, pp = stats.spearmanr(cd.iloc[:,i].to_numpy(), ed.iloc[:,i].to_numpy(), nan_policy='omit')
    rho[i]=r; p[i]=pp
m=p.size; order=np.argsort(p, kind='mergesort'); rank=np.empty(m); rank[order]=np.arange(1,m+1)
q=np.empty(m); q[order]=p[order]*m/rank[order]
for i in range(m-2,-1,-1):
    if q[order[i]] > q[order[i+1]]: q[order[i]]=q[order[i+1]]
q=np.minimum(q,1)
sig=q<0.05
print('finite',np.isfinite(p).sum(),'sig',sig.sum(),'percent',sig.sum()/m*100,'time',time.time()-T)
print('rho range',rho.min(),rho.max(),'p min',p.min())
PY"
