/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.stats as stats, time
ch=pd.read_csv('inputs/CRISPRGeneEffect.csv', nrows=0); eh=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', nrows=0)
common=[c for c in ch.columns[1:] if c in eh.columns[1:]]
cd=pd.read_csv('inputs/CRISPRGeneEffect.csv', usecols=[ch.columns[0]]+common, index_col=0)
ed=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', usecols=[eh.columns[0]]+common, index_col=0)
ids=sorted(set(cd.index)&set(ed.index)); cd=cd.loc[ids]; ed=ed.loc[ids]
max_r=0; max_p=0; argr=0; argp=0
for i in range(len(common)):
    x=cd.iloc[:,i].to_numpy(); y=ed.iloc[:,i].to_numpy()
    r1,p1=stats.spearmanr(x,y,nan_policy='omit')
    v=~(np.isnan(x)|np.isnan(y)); n=v.sum(); rx=stats.rankdata(x[v]); ry=stats.rankdata(y[v]); r2=np.corrcoef(rx,ry)[0,1]
    t=r2*np.sqrt((n-2)/(1-r2*r2)); p2=2*stats.t.sf(abs(t),n-2)
    dr=abs(r1-r2); dp=abs(p1-p2)
    if dr>max_r: max_r=dr; argr=i
    if dp>max_p: max_p=dp; argp=i
print('max r diff',max_r,'at',argr,common[argr])
print('max p diff',max_p,'at',argp,common[argp])
PY"
