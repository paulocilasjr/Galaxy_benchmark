/bin/bash -lc "python3 - <<'PY'
import csv
from pathlib import Path

def read_ids(fn):
    with open(fn,newline='') as f:
        rd=csv.reader(f)
        hdr=next(rd)
        ids=[]
        for row in rd:
            ids.append(row[0])
    return hdr, ids
cr_h, cr_ids=read_ids('inputs/CRISPRGeneEffect.csv')
ex_h, ex_ids=read_ids('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
mo_h, mo_ids=read_ids('inputs/Model.csv')
print('CRISPR rows',len(cr_ids),'unique',len(set(cr_ids)))
print('Expression rows',len(ex_ids),'unique',len(set(ex_ids)))
print('Model rows',len(mo_ids),'unique',len(set(mo_ids)))
print('common CRISPR/expression',len(set(cr_ids)&set(ex_ids)))
print('CRISPR not Model',len(set(cr_ids)-set(mo_ids)),'expression not Model',len(set(ex_ids)-set(mo_ids)))
print('Model common',len(set(mo_ids)&set(cr_ids)&set(ex_ids)))
print('gene columns CRISPR',len(cr_h)-1,'unique',len(set(cr_h[1:])))
print('gene columns Expression',len(ex_h)-1,'unique',len(set(ex_h[1:])))
print('gene intersection',len(set(cr_h[1:])&set(ex_h[1:])))
print('CRISPR-only genes',len(set(cr_h[1:])-set(ex_h[1:])))
print('Expression-only genes',len(set(ex_h[1:])-set(cr_h[1:])))
print('duplicate CRISPR', [x for x in set(cr_h[1:]) if cr_h[1:].count(x)>1][:5])
print('duplicate expression', [x for x in set(ex_h[1:]) if ex_h[1:].count(x)>1][:5])
print('common first ids', sorted(set(cr_ids)&set(ex_ids))[:10])
PY"
