/bin/bash -lc "python3 - <<'PY'
import csv,time
import numpy as np,pandas as pd

def h(f):
    with open(f,newline='') as x:return next(csv.reader(x))
crh=h('inputs/CRISPRGeneEffect.csv'); exh=h('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
es=set(exh[1:]); common=[g for g in crh[1:] if g in es]
cruse=[0]+[i for i,g in enumerate(crh) if i and g in es]; ep={g:i for i,g in enumerate(exh)}; exuse=[0]+[ep[g] for g in common]
dt={g:np.float32 for g in common}
cr=pd.read_csv('inputs/CRISPRGeneEffect.csv',usecols=cruse,index_col=0,dtype=dt)
ex=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=exuse,index_col=0,dtype=dt)
ids=cr.index.intersection(ex.index); cr=cr.loc[ids,common]; ex=ex.loc[ids,common]
n=cr.notna().sum(axis=0).to_numpy()
print('n complete range before expression missing',n.min(),n.max(),'mean',n.mean(),'quantiles',np.quantile(n,[0,.01,.05,.5,.95,.99,1]))
print('missing total',int(cr.isna().sum().sum()),'genes with missing',int((n<1103).sum()))
print('n<1000',int((n<1000).sum()),'n<500',int((n<500).sum()))
# Top missing gene labels
mi=np.argsort(n)[:20]
print('lowest',[(common[i],int(n[i])) for i in mi])
# Expression zero/ties fraction summary
x=ex.to_numpy(copy=False)
zero=(x==0).sum(axis=0)
print('expression exact zero range',zero.min(),zero.max(),'median',np.median(zero),'genes zero',int((zero>0).sum()))
print('expression unique count range',ex.nunique().min(),ex.nunique().max())
PY"
