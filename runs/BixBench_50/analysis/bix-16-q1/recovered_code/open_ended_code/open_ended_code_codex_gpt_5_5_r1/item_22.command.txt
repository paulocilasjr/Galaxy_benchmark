/bin/bash -lc "python - <<'PY'
import csv
import numpy as np
from scipy.stats import spearmanr

crispr_path='inputs/CRISPRGeneEffect.csv'
expr_path='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
label='CDKN1A (1026)'

def header(path):
    with open(path, newline='') as f: return next(csv.reader(f))
ch=header(crispr_path); eh=header(expr_path)
ci=ch.index(label); ei=eh.index(label)
expr={}
with open(expr_path, newline='') as f:
    r=csv.reader(f); next(r)
    for row in r:
        expr[row[0]]=float(row[ei])
xs=[]; ys=[]
with open(crispr_path, newline='') as f:
    r=csv.reader(f); next(r)
    for row in r:
        if row[0] in expr and row[ci]:
            xs.append(expr[row[0]])
            ys.append(-float(row[ci]))
print(label, len(xs), spearmanr(xs, ys).statistic)
PY
mkdir -p codex_output
printf '%s\\n' 'CDKN1A' > codex_output/answer.txt
wc -c codex_output/answer.txt && cat codex_output/answer.txt"
