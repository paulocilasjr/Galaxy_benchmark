/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from scipy.stats import spearmanr
c=pd.read_csv('inputs/CRISPRGeneEffect.csv',usecols=['Unnamed: 0','CCND1 (595)'],index_col=0)
e=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=['Unnamed: 0','CCND1 (595)'],index_col=0)
z=c.join(e,how='inner',lsuffix='_effect',rsuffix='_expr').dropna()
r=spearmanr(z.iloc[:,0],z.iloc[:,1])
print('paired_CCND1',len(z),'scipy_rho',r.statistic,'pvalue',r.pvalue)
print('effect_range',z.iloc[:,0].min(),z.iloc[:,0].max(),'expr_range',z.iloc[:,1].min(),z.iloc[:,1].max())
PY"
