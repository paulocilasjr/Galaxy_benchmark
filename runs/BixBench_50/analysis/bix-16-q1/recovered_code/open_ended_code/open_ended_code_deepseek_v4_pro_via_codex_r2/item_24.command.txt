/bin/bash -lc "python3 - <<'PY'
# Verify using scipy.stats.spearmanr direct for top genes, by loading just needed columns with genfromtxt/loadtxt
import csv,numpy as np,scipy.stats as st,time
# parse headers and ids
with open('inputs/CRISPRGeneEffect.csv',newline='') as f: eh=next(csv.reader(f))
with open('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',newline='') as f: xh=next(csv.reader(f))
genes=['CCND1 (595)','FERMT2 (10979)','KLF5 (688)','WWTR1 (25937)','ZEB2 (9839)','GRHL2 (79977)','ITGAV (3685)','MYC (4609)']
ec=[eh.index(g) for g in genes]; xc=[xh.index(g) for g in genes]
E=np.genfromtxt('inputs/CRISPRGeneEffect.csv',delimiter=',',skip_header=1,usecols=ec,dtype=np.float32,missing_values='',filling_values=np.nan)
X=np.loadtxt('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',delimiter=',',skiprows=1,usecols=xc,dtype=np.float32)
def ids(path):
 with open(path,newline='') as f:
  r=csv.reader(f);next(r);return [row[0] for row in r]
eids=ids('inputs/CRISPRGeneEffect.csv'); xids=ids('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'); xs={v:i for i,v in enumerate(xids)}
er=[i for i,v in enumerate(eids) if v in xs]; xr=[xs[v] for v in eids if v in xs]
for gi,g in enumerate(genes):
 x=E[er,gi]; y=X[xr,gi]
 res=st.spearmanr(x,y,nan_policy='omit')
 print(g, res.statistic, np.sum(np.isfinite(x)&np.isfinite(y)))
PY"
