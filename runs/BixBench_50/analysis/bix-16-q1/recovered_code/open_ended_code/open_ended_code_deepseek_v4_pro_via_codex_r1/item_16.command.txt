import pandas as pd, numpy as np, scipy.stats as st, time
# Load only top gene columns plus sample ids, to verify
# We'll parse headers and read necessary columns quickly.
eff_all=pd.read_csv('inputs/CRISPRGeneEffect.csv',nrows=0).columns
expr_all=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',nrows=0).columns
def sym(c): return str(c).split(' (')[0]
genes=['CCND1','FERMT2','KLF5','WWTR1','ZEB2']
eff_cols=[eff_all[0]]+[c for c in eff_all[1:] if sym(c) in genes]
expr_cols=[expr_all[0]]+[c for c in expr_all[1:] if sym(c) in genes]
eff=pd.read_csv('inputs/CRISPRGeneEffect.csv',usecols=eff_cols,index_col=0,dtype={c:np.float32 for c in eff_cols[1:]})
expr=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=expr_cols,index_col=0,dtype={c:np.float32 for c in expr_cols[1:]})
eff.columns=[sym(c) for c in eff.columns]
expr.columns=[sym(c) for c in expr.columns]
ids=eff.index.intersection(expr.index)
eff=eff.loc[ids,genes]; expr=expr.loc[ids,genes]
print(eff.shape)
for g in genes:
    a=eff[g].to_numpy(dtype=float); b=expr[g].to_numpy(dtype=float)
    m=np.isfinite(a)&np.isfinite(b)
    r,p=st.spearmanr(a[m],b[m])
    print(g, m.sum(), r)
