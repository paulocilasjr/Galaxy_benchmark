/bin/bash -lc "find inputs -maxdepth 1 -type f -printf '%f\\n' | sort
python - <<'PY'
import pandas as pd
from scipy.stats import spearmanr
pe='inputs/CRISPRGeneEffect.csv'; px='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
cands=['CDKN1A (1026)','KIRREL1 (55243)','EPHA2 (1969)','CCND1 (595)','FERMT2 (10979)']
e=pd.read_csv(pe,usecols=['Unnamed: 0']+cands,index_col=0)
x=pd.read_csv(px,usecols=['Unnamed: 0']+cands,index_col=0)
ids=e.index.intersection(x.index,sort=False)
print('models',len(ids))
for g in cands:
    z=pd.concat([x.loc[ids,g],e.loc[ids,g]],axis=1).dropna()
    raw=spearmanr(z.iloc[:,0],z.iloc[:,1]).statistic
    essentiality=spearmanr(z.iloc[:,0],-z.iloc[:,1]).statistic
    print(g,'n',len(z),'raw_gene_effect',format(raw,'.17g'),'essentiality',format(essentiality,'.17g'))
PY"
