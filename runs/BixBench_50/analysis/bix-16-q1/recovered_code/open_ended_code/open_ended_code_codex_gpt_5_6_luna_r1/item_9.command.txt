/bin/bash -lc "python - <<'PY'
import csv,re
from collections import Counter,defaultdict
files=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
headers=[]
for fn in files:
    with open(fn,newline='',encoding='utf-8') as f: headers.append(next(csv.reader(f)))
for fn,h in zip(files,headers):
    fs=h[1:]
    print('\\n',fn,'features',len(fs),'unique labels',len(set(fs)))
    print('duplicate labels',[(x,n) for x,n in Counter(fs).items() if n>1][:20])
    ids=[]; bad=[]
    for x in fs:
        m=re.match(r'"'^(.*) '"\\((\\d+)\\)"'$'"',x)
        if m: ids.append(m.group(2))
        else: bad.append(x)
    print('entrez parsed',len(ids),'unique ids',len(set(ids)),'bad',bad[:10])
    print('first5',fs[:5])
cr={x:i for i,x in enumerate(headers[0][1:])}; ex={x:i for i,x in enumerate(headers[1][1:])}
print('\\ncommon exact labels',len(set(cr)&set(ex)))
print('CR only',len(set(cr)-set(ex)),'Expr only',len(set(ex)-set(cr)))
# compare by entrez id, symbol
pat=re.compile(r'"'^(.*) '"\\((\\d+)\\)"'$'"')
def maps(h):
 d=defaultdict(list)
 for i,x in enumerate(h[1:]):
  m=pat.match(x)
  if m: d[m.group(2)].append((m.group(1),x,i))
 return d
cm,em=maps(headers[0]),maps(headers[1])
ci=set(cm)&set(em)
print('common entrez',len(ci),'id mismatches',sum(cm[i][0][0]"'!=em[i][0][0] for i in ci))
print('"'mismatch examples',[(i,cm[i],em[i]) for i in ci if cm[i][0][0]"'!=em[i][0][0]][:20])
print('"'CR dup entrez',[(i,v) for i,v in cm.items() if len(v)>1][:10])
print('Expr dup entrez',[(i,v) for i,v in em.items() if len(v)>1][:10])
PY"
