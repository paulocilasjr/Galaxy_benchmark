/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, re, time
start=time.time()
print('loading CRISPR...', flush=True)
eff=pd.read_csv('inputs/CRISPRGeneEffect.csv', index_col=0)
print('loading expression...', flush=True)
expr=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', index_col=0)
print('loaded', eff.shape, expr.shape, 'elapsed', time.time()-start, flush=True)
print('eff NaN', int(eff.isna().sum().sum()), 'expr NaN', int(expr.isna().sum().sum()))
print('eff inf', int(np.isinf(eff.to_numpy()).sum()), 'expr inf', int(np.isinf(expr.to_numpy()).sum()))
common=eff.index.intersection(expr.index)
print('common rows', len(common))
eff=eff.loc[common]
expr=expr.loc[common]
clean=lambda col: re.sub(r'\\s*\\(.*\\)"'$'"','',col)
eff.columns=[clean(c) for c in eff.columns]
expr.columns=[clean(c) for c in expr.columns]
common_genes=eff.columns.intersection(expr.columns)
print('common genes', len(common_genes))
print('dupe eff after clean', len(eff.columns)-len(set(eff.columns)), 'dupe expr', len(expr.columns)-len(set(expr.columns)))
PY"
