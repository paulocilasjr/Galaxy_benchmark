/bin/bash -lc "python - <<'PY'
import zipfile
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(zname) as z:
        names=z.namelist()
        suffixes={}
        for n in names:
            parts=n.split('.faa')
            suf='.faa'+parts[1] if len(parts)>1 else n
            suffixes[suf]=suffixes.get(suf,0)+1
        print(zname, 'members', len(names))
        for suf,c in sorted(suffixes.items(), key=lambda x:x[0]):
            if 'treefile' in suf or 'bionj' in suf or 'mldist' in suf or suf=='.faa': print(c, suf)
        t=[n for n in names if n.endswith('.treefile')]
        print('treefiles', len(t))
        for n in t[:5]:
            print(n, z.read(n).decode().strip())
        print()
PY"
