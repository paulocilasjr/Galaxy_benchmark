# Compute all-tree and shared-BUSCO-ID variants from treefiles to check sensitivity.
import zipfile, statistics, itertools
class Node:
    __slots__=('name','length','children','parent')
    def __init__(self,name=None,length=0.0):
        self.name=name; self.length=length; self.children=[]; self.parent=None

def parse(s):
    s=s.strip().rstrip(';'); n=len(s); i=0
    def ws(i):
        while i<n and s[i].isspace(): i+=1
        return i
    def name_len(i):
        i=ws(i); start=i
        while i<n and s[i] not in ':,()': i+=1
        name=s[start:i].strip() or None
        i=ws(i); L=0.0
        if i<n and s[i]==':':
            i+=1; start=i
            while i<n and s[i] not in ',()': i+=1
            L=float(s[start:i].strip())
        return name,L,i
    def sub(i):
        i=ws(i)
        if s[i]=='(':
            node=Node(); i+=1
            while True:
                c,i=sub(i); c.parent=node; node.children.append(c); i=ws(i)
                if i<n and s[i]==',': i+=1; continue
                if i<n and s[i]==')': i+=1; break

def tips(root):

    while stack:
        x=stack.pop()
        if x.children: stack.extend(x.children)
        else: out.append(x)
