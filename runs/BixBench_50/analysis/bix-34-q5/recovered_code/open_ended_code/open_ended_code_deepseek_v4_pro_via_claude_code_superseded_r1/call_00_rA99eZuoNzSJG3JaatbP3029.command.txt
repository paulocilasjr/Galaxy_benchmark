# Find shared ortholog IDs between animals and fungi
unzip -l inputs/scogs_animals.zip | grep '\.treefile$' | awk '{print $NF}' | sed 's/.faa.mafft.clipkit.treefile//' | sort > /tmp/animals_orthologs.txt
unzip -l inputs/scogs_fungi.zip | grep '\.treefile$' | awk '{print $NF}' | sed 's/.faa.mafft.clipkit.treefile//' | sort > /tmp/fungi_orthologs.txt
echo "Animals tree files: $(wc -l < /tmp/animals_orthologs.txt)"
echo "Fungi tree files: $(wc -l < /tmp/fungi_orthologs.txt)"
echo "Shared orthologs with trees: $(comm -12 /tmp/animals_orthologs.txt /tmp/fungi_orthologs.txt | wc -l)"
comm -12 /tmp/animals_orthologs.txt /tmp/fungi_orthologs.txt | head -20
