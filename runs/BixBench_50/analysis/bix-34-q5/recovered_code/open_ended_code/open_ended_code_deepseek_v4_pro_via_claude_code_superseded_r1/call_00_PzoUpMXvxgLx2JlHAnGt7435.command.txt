import json, statistics
from collections import defaultdict

with open('/workspace/codex_output/results.json') as f:
    results = json.load(f)

print(f"Number of gene trees: {len(results)}")

# ===== Interpretation 1: All-species mean (to all other species on tree) =====
animal_all = []
fungi_all = []
for r in results:
    for sp, val in r['all_means'].items():
        if sp.startswith('Animals'):
            animal_all.append(val)
        else:
            fungi_all.append(val)

am_all = statistics.median(animal_all)
fm_all = statistics.median(fungi_all)
ratio_all = fm_all / am_all
print(f"\n===== Interpretation 1: All-species mean =====")
print(f"Animal median: {am_all:.10f}")
print(f"Fungi median: {fm_all:.10f}")
print(f"Ratio fungi/animal: {ratio_all:.10f}")

# ===== Interpretation 2: Within-kingdom mean =====
animal_within = []
fungi_within = []
for r in results:
    for sp, val in r['within_means'].items():
        if sp.startswith('Animals'):
            animal_within.append(val)
        else:
            fungi_within.append(val)

am_within = statistics.median(animal_within)
fm_within = statistics.median(fungi_within)
ratio_within = fm_within / am_within
print(f"\n===== Interpretation 2: Within-kingdom mean =====")
print(f"Animal median: {am_within:.10f}")
print(f"Fungi median: {fm_within:.10f}")
print(f"Ratio fungi/animal: {ratio_within:.10f}")

# ===== Interpretation 3: Cross-kingdom mean =====
animal_cross = []
fungi_cross = []
for r in results:
    for sp, val in r['cross_means'].items():
        if sp.startswith('Animals'):
            animal_cross.append(val)
        else:
            fungi_cross.append(val)

am_cross = statistics.median(animal_cross)
fm_cross = statistics.median(fungi_cross)
ratio_cross = fm_cross / am_cross
print(f"\n===== Interpretation 3: Cross-kingdom mean =====")
print(f"Animal median: {am_cross:.10f}")
print(f"Fungi median: {fm_cross:.10f}")
print(f"Ratio fungi/animal: {ratio_cross:.10f}")

# ===== Interpretation 4: Per-gene median (all-species) =====
animal_gene_meds = []
fungi_gene_meds = []
for r in results:
    av = [v for k,v in r['all_means'].items() if k.startswith('Animals')]
    fv = [v for k,v in r['all_means'].items() if k.startswith('Fungi')]
    if av: animal_gene_meds.append(statistics.median(av))
    if fv: fungi_gene_meds.append(statistics.median(fv))
ratio_gene = statistics.median(fungi_gene_meds) / statistics.median(animal_gene_meds)
print(f"\n===== Interpretation 4: Per-gene median (all-species) =====")
print(f"Ratio fungi/animal: {ratio_gene:.10f}")

# ===== Interpretation 5: Per-species mean across genes, then median =====
animal_by_sp = defaultdict(list)
fungi_by_sp = defaultdict(list)
for r in results:
    for sp, val in r['all_means'].items():
        if sp.startswith('Animals'):
            animal_by_sp[sp].append(val)
        else:
            fungi_by_sp[sp].append(val)

animal_sp_means = [statistics.mean(v) for v in animal_by_sp.values()]
fungi_sp_means = [statistics.mean(v) for v in fungi_by_sp.values()]
ratio_sp = statistics.median(fungi_sp_means) / statistics.median(animal_sp_means)
print(f"\n===== Interpretation 5: Per-species mean across genes =====")
print(f"Animal per-species means: {[round(m,4) for m in animal_sp_means]}")
print(f"Fungi per-species means: {[round(m,4) for m in fungi_sp_means]}")
print(f"Animal median: {statistics.median(animal_sp_means):.10f}")
print(f"Fungi median: {statistics.median(fungi_sp_means):.10f}")
print(f"Ratio fungi/animal: {ratio_sp:.10f}")

# ===== Interpretation 6: Filter to trees where both groups are monophyletic =====
mono_results = []
for r in results:
    all_mono = True
    for sp in r['leaves']:
        within = r['within_means'].get(sp)
        cross = r['cross_means'].get(sp)
        if within is not None and cross is not None and within > cross:
            all_mono = False
            break
    if all_mono:
        mono_results.append(r)

print(f"\n===== Interpretation 6: Monophyletic-only trees ({len(mono_results)}/{len(results)}) =====")
animal_mono = []
fungi_mono = []
for r in mono_results:
    for sp, val in r['all_means'].items():
        if sp.startswith('Animals'):
            animal_mono.append(val)
        else:
            fungi_mono.append(val)
if animal_mono and fungi_mono:
    am_mono = statistics.median(animal_mono)
    fm_mono = statistics.median(fungi_mono)
    print(f"Ratio fungi/animal: {fm_mono/am_mono:.10f}")

# ===== Interpretation 7: Separate-tree approach (pre-computed trees) =====
print(f"\n===== Interpretation 7: Separate-tree approach =====")
print("Animal median mean patristic distance: 0.82419140555")
print("Fungi median mean patristic distance: 1.4714703755833332")
print("Ratio fungi/animal: 1.78535030294497")
