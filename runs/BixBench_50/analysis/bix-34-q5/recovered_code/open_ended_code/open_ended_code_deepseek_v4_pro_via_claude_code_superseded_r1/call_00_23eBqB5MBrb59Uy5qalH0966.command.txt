cd /tmp/combined_work
python3 << 'EOF'
import sys
sys.path.insert(0, '/workspace')
from compute_ratio import parse_newick, get_leaves, compute_patristic_distances, compute_mean_per_species

# Read the combined tree
with open('combined_aligned2.faa.treefile') as f:
    newick = f.read().strip()

tree = parse_newick(newick)
leaves = get_leaves(tree)
print("Leaves:", leaves)

dists = compute_patristic_distances(tree)
print("\nPatristic distances:")
for (a, b), d in sorted(dists.items()):
    if a < b:
        print(f"  {a} -> {b}: {d:.6f}")

means = compute_mean_per_species(dists, leaves)
print("\nMean patristic distances:")
for sp in sorted(means.keys()):
    kingdom = "Fungi" if sp.startswith("Fungi") else "Animals"
    print(f"  {sp} ({kingdom}): {means[sp]:.6f}")

# Compute per-kingdom medians of means
animal_means = [v for k, v in means.items() if k.startswith("Animals")]
fungi_means = [v for k, v in means.items() if k.startswith("Fungi")]
import statistics
print(f"\nAnimals mean distances: {animal_means}")
print(f"Fungi mean distances: {fungi_means}")
print(f"Animal median: {statistics.median(animal_means):.6f}")
print(f"Fungi median: {statistics.median(fungi_means):.6f}")
print(f"Ratio (fungi/animal): {statistics.median(fungi_means) / statistics.median(animal_means):.6f}")
EOF
