# Test MAFFT with the actual data
cd /tmp
mkdir -p combined_work
cd combined_work

# Extract a sample FAA from both archives
unzip -o /workspace/inputs/scogs_animals.zip "1003258at2759.faa" -d animal 2>&1 | grep -v Archive
unzip -o /workspace/inputs/scogs_fungi.zip "1003258at2759.faa" -d fungi 2>&1 | grep -v Archive

# Combine them
cat animal/1003258at2759.faa fungi/1003258at2759.faa > combined.faa
echo "Combined sequences:"
grep -c "^>" combined.faa
grep "^>" combined.faa

# Align with MAFFT
MAFFT_BINARIES=/tmp/mafft-linux64/mafftdir/libexec /tmp/mafft-linux64/mafftdir/bin/mafft --auto combined.faa > combined_aligned.faa 2>&1
echo "MAFFT exit: $?"
head -20 combined_aligned.faa
