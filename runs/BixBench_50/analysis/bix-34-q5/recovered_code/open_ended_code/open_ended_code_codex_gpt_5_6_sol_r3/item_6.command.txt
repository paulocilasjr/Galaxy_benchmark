class Parser:
    def __init__(self, s): self.s=s.strip(); self.i=0; self.nodes=[]
    def node(self):
        idx=len(self.nodes); self.nodes.append({'children':[], 'name':None, 'length':None})
        if self.s[self.i]=='(':
            self.i+=1

                c=self.node(); self.nodes[idx]['children'].append(c)

            # optional internal label/support
            start=self.i
            while self.i<len(self.s) and self.s[self.i] not in ':,();': self.i+=1
            if self.i>start: self.nodes[idx]['name']=self.s[start:self.i]
        else:
            start=self.i

            self.i+=1; start=self.i

        root=self.node()

def mean_patristic(newick):
    root,nodes=Parser(newick).parse()
    nleaf=sum(not x['children'] for x in nodes)
    if nleaf<2: raise ValueError('fewer than 2 leaves')
    missing=[]; negative=[]
    total=0.0
    def descend(u):
        nonlocal total
        if not nodes[u]['children']: return 1
        k=0
        for v in nodes[u]['children']:
            kv=descend(v); k+=kv
            w=nodes[v]['length']
            if w is None: missing.append(v)
            else:
                if w<0: negative.append(w)
                total += w*kv*(nleaf-kv)
        return k
    assert descend(root)==nleaf
    if missing: raise ValueError(('missing branch lengths',missing))
    return total/comb(nleaf,2),nleaf,negative
