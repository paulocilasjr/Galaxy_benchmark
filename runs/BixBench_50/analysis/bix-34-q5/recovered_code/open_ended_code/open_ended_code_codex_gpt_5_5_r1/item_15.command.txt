class Parser:

def mean_patristic(newick):
    root,nodes=Parser(newick).parse()
    adj=[[] for _ in nodes]
    for parent,node in enumerate(nodes):
        for child in node['children']:
            length=nodes[child]['length']
            if length is None:
                length=0.0
            adj[parent].append((child,length))
            adj[child].append((parent,length))
    leaves=[i for i,n in enumerate(nodes) if not n['children']]
    if len(leaves) < 2:
        raise ValueError('Tree has fewer than two leaves')
    # Check all non-root edges had a branch length; IQ-TREE treefiles should.
    missing=[i for i,n in enumerate(nodes) if i "'!= root and n['"'length'] is None]
    if missing:
        raise ValueError(f'Missing branch lengths for {len(missing)} nodes')

        seen=set()

                    stack.append((v,u,d+w))
        else:
            raise ValueError('"'Disconnected tree')
    return total/count, [nodes[i]['name'] for i in leaves]

def summarize(zpath):

            except Exception as e:

    return vals, med, leaves_count, failures
