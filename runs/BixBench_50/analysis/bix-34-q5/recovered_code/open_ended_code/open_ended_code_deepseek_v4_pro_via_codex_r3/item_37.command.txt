/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from statistics import median
import subprocess,re
for group in ['scogs_animals','scogs_fungi']:
    p=Path('/workspace/tmp')/group
    res={'all':[],'3':[],'4':[]}
    for f in sorted(p.glob('*.treefile')):
        # name pattern {id}.faa.mafft.clipkit.treefile
        base=f.name[:-len('.mafft.clipkit.treefile')]  # id.faa
        faa=p/base
        n=sum(1 for l in faa.open() if l.startswith('>'))
        cp=subprocess.run(['phykit','patristic_distances',str(f)],text=True,capture_output=True)
        m=float(re.search(r'"'^mean:'"\\s*([-+]?[0-9.eE]+)',cp.stdout,re.M).group(1))
        res['all'].append(m); res[str(n)].append(m)
    print(group)
    for k,v in res.items(): print(k,len(v), median(v) if v else None)
PY"
