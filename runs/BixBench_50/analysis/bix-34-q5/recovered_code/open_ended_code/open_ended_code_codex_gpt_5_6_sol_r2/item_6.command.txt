/bin/bash -lc "python - <<'PY'
import zipfile, io, statistics, itertools
from Bio import Phylo
for group in ('animals','fungi'):
    vals=[]; ntips=[]; bad=[]
    zpath=f'inputs/scogs_{group}.zip'
    with zipfile.ZipFile(zpath) as z:
        names=sorted(n for n in z.namelist() if n.endswith('.treefile'))
        for n in names:
            tree=Phylo.read(io.StringIO(z.read(n).decode()), 'newick')
            tips=tree.get_terminals(); ntips.append(len(tips))
            ds=[tree.distance(a,b) for a,b in itertools.combinations(tips,2)]
            if not ds: bad.append(n); continue
            vals.append(sum(ds)/len(ds))
    print(group, 'selected',len(names),'processed',len(vals),'bad',len(bad),'tip_counts', {k:ntips.count(k) for k in sorted(set(ntips))})
    print('median_mean_PD',repr(statistics.median(vals)),'min',min(vals),'max',max(vals),'mean',statistics.mean(vals))
    print('middle',sorted(vals)[len(vals)//2-2:len(vals)//2+2])
PY"
