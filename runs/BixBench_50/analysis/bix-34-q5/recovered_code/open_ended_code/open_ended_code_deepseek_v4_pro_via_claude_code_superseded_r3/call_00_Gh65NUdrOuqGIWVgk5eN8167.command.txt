import os, glob, statistics

def parse_and_mean(filepath):
    """Parse mldist and return (n_species, mean_pairwise_dist)."""
    with open(filepath) as f:
        content = f.read().strip()
    lines = content.split('\n')
    n = int(lines[0].strip())
    
    # Parse matrix
    matrix = {}
    species_order = []
    for line in lines[1:]:
        parts = line.strip().split()
        sp = parts[0]
        dists = [float(x) for x in parts[1:]]
        species_order.append(sp)
        matrix[sp] = dists
    
    # Extract all off-diagonal pairwise distances (only i<j to avoid double counting)
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append(matrix[species_order[i]][j])
    
    return n, statistics.mean(pairs)

# Process both kingdoms
results = {}
for label, directory in [('animals', '/tmp/work'), ('fungi', '/tmp/work_fungi')]:
    means = []
    species_counts = {}
    for fp in sorted(glob.glob(os.path.join(directory, '*.mafft.clipkit.mldist'))):
        n_sp, mean_dist = parse_and_mean(fp)
        means.append(mean_dist)
        species_counts[n_sp] = species_counts.get(n_sp, 0) + 1
    
    results[label] = {
        'n_groups': len(means),
        'median': statistics.median(means),
        'mean': statistics.mean(means),
        'min': min(means),
        'max': max(means),
        'species_counts': species_counts
    }

for kingdom, r in results.items():
    print(f"{kingdom}:")
    print(f"  Groups: {r['n_groups']}")
    print(f"  Median mean patristic distance: {r['median']:.10f}")
    print(f"  Mean of means: {r['mean']:.10f}")
    print(f"  Range: [{r['min']:.6f}, {r['max']:.6f}]")
    print(f"  Species per group: {r['species_counts']}")

ratio = results['fungi']['median'] / results['animals']['median']
print(f"\nRatio (fungi/animals): {ratio:.15f}")

# Also show the alternative: animals/fungi
alt = results['animals']['median'] / results['fungi']['median']
print(f"Alternative (animals/fungi): {alt:.15f}")
