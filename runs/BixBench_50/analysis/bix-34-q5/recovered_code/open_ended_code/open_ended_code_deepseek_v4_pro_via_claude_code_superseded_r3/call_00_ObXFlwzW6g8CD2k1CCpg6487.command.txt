import os
import glob
import statistics

def parse_mldist(filepath):
    with open(filepath) as f:
        lines = [l.strip() for l in f if l.strip()]
    if not lines:
        return None, 'empty'
    try:
        n = int(lines[0])
    except ValueError:
        return None, f'bad_n: {lines[0]}'
    if n < 2:
        return None, f'n<2: {n}'
    if len(lines) < n + 1:
        return None, f'too few lines: {len(lines)-1} vs {n}'
    
    distances = []
    for i in range(1, n + 1):
        parts = lines[i].split()
        if len(parts) < n + 1:
            return None, f'too few cols row {i}: {len(parts)-1} vs {n}'
        row_distances = [float(x) for x in parts[1:n+1]]
        distances.append(row_distances)
    
    pairwise = []
    for i in range(n):
        for j in range(i + 1, n):
            pairwise.append(distances[i][j])
    
    if not pairwise:
        return None, 'no pairs'
    return statistics.mean(pairwise), None

# Check all files
animals_means = []
fungi_means = []

for label, d in [('Animals', '/tmp/work'), ('Fungi', '/tmp/work_fungi')]:
    errors = []
    means = []
    for f in sorted(glob.glob(os.path.join(d, '*.mafft.clipkit.mldist'))):
        mean, err = parse_mldist(f)
        if err:
            errors.append((os.path.basename(f), err))
        else:
            means.append(mean)
    
    if label == 'Animals':
        animals_means = means
    else:
        fungi_means = means
    
    print(f'{label}: {len(means)} valid, {len(errors)} errors')
    if errors:
        for e in errors[:5]:
            print(f'  Error: {e}')
    med = statistics.median(means)
    print(f'  Median: {med:.10f}')
    print(f'  Min: {min(means):.6f}, Max: {max(means):.6f}')

median_animals = statistics.median(animals_means)
median_fungi = statistics.median(fungi_means)
ratio = median_fungi / median_animals
print(f'\nRatio (fungi/animals): {ratio:.15f}')
