import os
import glob
import statistics

def parse_mldist(filepath):
    with open(filepath) as f:
        lines = [l.strip() for l in f if l.strip()]
    if not lines:
        return None
    n = int(lines[0])
    if n < 2:
        return None
    distances = []
    for i in range(1, n + 1):
        parts = lines[i].split()
        row_distances = [float(x) for x in parts[1:]]
        distances.append(row_distances)
    pairwise = []
    for i in range(n):
        for j in range(i + 1, n):
            pairwise.append(distances[i][j])
    if not pairwise:
        return None
    return statistics.mean(pairwise)

# Animals
animals_means = []
for f in sorted(glob.glob('/tmp/work/*.mafft.clipkit.mldist')):
    mean_dist = parse_mldist(f)
    if mean_dist is not None:
        animals_means.append(mean_dist)

print(f'Animals: {len(animals_means)} groups')
median_animals = statistics.median(animals_means)
print(f'Median: {median_animals}')

# Fungi
fungi_means = []
for f in sorted(glob.glob('/tmp/work_fungi/*.mafft.clipkit.mldist')):
    mean_dist = parse_mldist(f)
    if mean_dist is not None:
        fungi_means.append(mean_dist)

print(f'Fungi: {len(fungi_means)} groups')
median_fungi = statistics.median(fungi_means)
print(f'Median: {median_fungi}')

ratio = median_fungi / median_animals
print(f'Ratio (fungi/animals): {ratio}')
