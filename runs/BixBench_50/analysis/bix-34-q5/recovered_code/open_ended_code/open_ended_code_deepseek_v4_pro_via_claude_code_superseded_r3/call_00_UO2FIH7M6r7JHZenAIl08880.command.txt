import os
import glob
import statistics
import numpy as np

def parse_mldist(filepath):
    """Parse a mldist file and return the mean of all pairwise patristic distances."""
    with open(filepath) as f:
        lines = [l.strip() for l in f if l.strip()]
    
    if not lines:
        return None
    
    n = int(lines[0])
    if n < 2:
        return None
    
    # Parse the distance matrix
    distances = []
    for i in range(1, n + 1):
        parts = lines[i].split()
        # parts[0] is species name, parts[1:] are distances
        species = parts[0]
        row_distances = [float(x) for x in parts[1:]]
        distances.append(row_distances)
    
    # Extract upper triangular (non-diagonal) pairwise distances
    pairwise = []
    for i in range(n):
        for j in range(i + 1, n):
            pairwise.append(distances[i][j])
    
    if not pairwise:
        return None
    
    return statistics.mean(pairwise)

# Process animals
animals_dir = '/tmp/work'
animals_means = []
for f in sorted(glob.glob(os.path.join(animals_dir, '*.mafft.clipkit.mldist'))):
    mean_dist = parse_mldist(f)
    if mean_dist is not None:
        animals_means.append(mean_dist)

print(f"Animals: {len(animals_means)} ortholog groups")
print(f"Animals means: {animals_means[:5]}...")
median_animals = statistics.median(animals_means)
print(f"Median mean patristic distance (animals): {median_animals}")

# Process fungi
fungi_dir = '/tmp/work_fungi'

    if mean_dist is not None:

median_fungi = statistics.median(fungi_means)
print(f"Median mean patristic distance (fungi): {median_fungi}")

ratio = median_fungi / median_animals
print(f"\nRatio (fungi/animal): {ratio}")
