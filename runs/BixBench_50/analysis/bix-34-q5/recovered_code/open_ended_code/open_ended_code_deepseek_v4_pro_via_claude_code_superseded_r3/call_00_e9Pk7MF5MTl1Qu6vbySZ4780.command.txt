# Manual verification: pick a few files and compute by hand
import os, glob

# Animals: pick the file with smallest and largest means
def parse_and_mean(filepath):
    with open(filepath) as f:
        content = f.read().strip()
    lines = content.split('\n')
    n = int(lines[0].strip())
    matrix = []
    for line in lines[1:]:
        parts = line.strip().split()
        dists = [float(x) for x in parts[1:]]
        matrix.append(dists)
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append(matrix[i][j])
    return n, pairs, sum(pairs)/len(pairs)

# Check a few files manually
for label, d in [('Animals', '/tmp/work'), ('Fungi', '/tmp/work_fungi')]:
    files = sorted(glob.glob(os.path.join(d, '*.mafft.clipkit.mldist')))
    # Check first, middle, and last
    for idx in [0, len(files)//2, -1]:
        fp = files[idx]
        n, pairs, mean = parse_and_mean(fp)
        print(f"{label} {os.path.basename(fp)[:20]}: n={n}, n_pairs={len(pairs)}, mean={mean:.10f}")
        print(f"  Pairs: {[f'{p:.6f}' for p in pairs[:5]]}{'...' if len(pairs)>5 else ''}")
        print(f"  Sum: {sum(pairs):.10f}, Mean: {sum(pairs)/len(pairs):.10f}")
        print()
