/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from pathlib import Path
from statistics import mean,median
from itertools import combinations
class N:
 def __init__(self,n=None,l=0): self.n=n; self.l=l; self.c=[]
def parse(s):
 s=s.strip().rstrip(';'); i=0
 def tok(stop):
  nonlocal i
  j=i
  while i<len(s) and s[i] not in stop:i+=1
  return s[j:i].strip()
 def sub():
  nonlocal i
  if s[i]=='(':
   i+=1; x=N()
   while 1:
    x.c.append(sub())
    if s[i]==',':i+=1;continue
    if s[i]==')':i+=1;break
   if i<len(s) and s[i] not in ',():':x.n=tok(',():')
   if i<len(s) and s[i]==':':i+=1;x.l=float(tok(',()'))
   return x
  x=N(tok(',():'))
  if i<len(s) and s[i]==':':i+=1;x.l=float(tok(',()'))
  return x
 return sub()
def avg(s):
 root=parse(s); ls=[]
 def walk(x,d,p):
  if not x.c:ls.append((x.n,p+[(x,d)]));return
  for y in x.c:walk(y,d+y.l,p+[(x,d)])
 walk(root,0,[]); ds=[]
 for a,b in combinations(ls,2):
  da={id(x):d for x,d in a[1]}; db={id(x):d for x,d in b[1]}; z=max(set(da)&set(db),key=lambda k:da[k]);ds.append(da[id(a[1][-1][0])]+db[id(b[1][-1][0])]-2*da[z])
 return mean(ds),len(ls)
D={}
for g in ['animals','fungi']:
 with ZipFile(Path('inputs')/f'scogs_{g}.zip') as z:
  D[g]={n.rsplit('/',1)[-1].split('.')[0]:avg(z.read(n).decode()) for n in z.namelist() if n.endswith('.treefile')}
common=set(D['animals'])&set(D['fungi'])
for kset,name in [(common,'common')]:
 a=[D['animals'][k][0] for k in kset];f=[D['fungi'][k][0] for k in kset]
 print(name,len(kset),'animal median',median(a),'fungi median',median(f),'ratio F/A',median(f)/median(a),'means',mean(a),mean(f),'complete counts',sum(D['animals'][k][1]==4 for k in kset),sum(D['fungi'][k][1]==4 for k in kset))
PY"
