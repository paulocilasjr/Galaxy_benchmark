import zipfile, io, re, statistics
from collections import defaultdict

def parse_newick(s):
    # return dict pair -> distance
    s=s.strip()
    if not s.endswith(';'): s+=';'
    # simple recursive parser for Newick
    pos=0
    def skip():
        nonlocal pos
        while pos<len(s) and s[pos] in ' \\t\\n': pos+=1
    def parse_node():
        nonlocal pos
        skip()
        if s[pos]=='(':
            pos+=1
            children=[]
            while True:
                children.append(parse_node())
                skip()
                if pos<len(s) and s[pos]==',':
                    pos+=1
                    continue
                elif pos<len(s) and s[pos]==')':
                    pos+=1
                    break
                else:
                    raise ValueError('bad newick')
            # branch length for internal node
            skip()
            bl=None
            if pos<len(s) and s[pos]==':':
                pos+=1; bl=parse_number()
            return ('internal',children,bl)
        else:
            # label
            m=re.match(r\"["'^(),;:]+", s[pos:])
            if not m: raise ValueError('"'bad label at '+s[pos:])
            label=m.group(0); pos+=len(label)
            skip()
            bl=None
            if pos<len(s) and s[pos]==':':
                pos+=1; bl=parse_number()
            return ('leaf',label,bl)
    def parse_number():
        nonlocal pos
        m=re.match(r\"[-+]?\\d+(?:\\.\\d+)?(?:[eE][-+]?\\d+)?\", s[pos:])
        if not m: raise ValueError('bad number')
        pos+=len(m.group(0)); return float(m.group(0))
    tree=parse_node()
    skip()
    return tree

def collect(tree):
    # return list leaf labels and paths with cumulative branch lengths to each leaf from root of subtree (undefined root)

def path_distances(s):
    # full recursive returns map leaf->distance from node
    def rec(node):
        if node[0]=='leaf':
            return {node[1]:0.0}

                d[lab]=v+child_len
