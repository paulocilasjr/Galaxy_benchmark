/bin/bash -c 'work=$(mktemp -d /tmp/scogs_work2.XXXXXX)
mkdir -p "$work/animals" "$work/fungi"
python3 - "$work" <<'"'PY'
import zipfile, pathlib, sys, statistics
from Bio import Phylo
from io import StringIO
work=pathlib.Path(sys.argv[1])
for group in ['animals','fungi']:
    z=zipfile.ZipFile(f'inputs/scogs_{group}.zip')
    out=work/group
    means=[]
    for n in z.namelist():
        if n.endswith('.treefile'):
            data=z.read(n)
            (out/n.split('/')[-1]).write_bytes(data)
            tree=Phylo.read(StringIO(data.decode()),'newick')
            terms=tree.get_terminals()
            vals=[]
            for i,a in enumerate(terms):
                for b in terms[i+1:]:
                    vals.append(tree.distance(a,b))
            means.append(sum(vals)/len(vals))
    print(group, 'n',len(means),'median',statistics.median(means),'mean',statistics.mean(means))
    print('first',means[:5])
print('ratio',1.7482006310000002/0.8796373416750001)
PY"
