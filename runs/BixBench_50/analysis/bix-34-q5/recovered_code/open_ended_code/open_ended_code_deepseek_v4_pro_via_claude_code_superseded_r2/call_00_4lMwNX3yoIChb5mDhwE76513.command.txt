import sys
sys.path.insert(0, '/workspace/venv/lib/python3.11/site-packages')
import dendropy

tree = dendropy.Tree.get(path='/workspace/extracted/1003258at2759.faa.mafft.clipkit.treefile', schema='newick', preserve_underscores=True)
print('Leaf nodes:')
for leaf in tree.leaf_node_iter():
    print(f'  {leaf.taxon.label!r}')
print()

pdm = tree.phylogenetic_distance_matrix()
taxa = [leaf.taxon for leaf in tree.leaf_node_iter()]
print('Testing distance matrix:')
for i in range(len(taxa)):
    for j in range(i+1, len(taxa)):
        try:
            d = pdm(taxa[i], taxa[j])
            print(f'  {taxa[i].label} - {taxa[j].label}: {d}')
        except Exception as e:
            print(f'  Error: {e}')
            # Try with labels
            try:
                d = pdm(taxa[i].label, taxa[j].label)
                print(f'  (with labels) {taxa[i].label} - {taxa[j].label}: {d}')
            except Exception as e2:
                print(f'  (with labels) also error: {e2}')
