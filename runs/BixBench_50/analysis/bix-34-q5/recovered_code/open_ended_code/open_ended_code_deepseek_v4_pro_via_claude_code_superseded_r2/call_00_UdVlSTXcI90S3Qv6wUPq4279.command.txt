import sys
sys.path.insert(0, '/workspace/venv/lib/python3.11/site-packages')
import dendropy
import os
import numpy as np

# Check species counts per orthogroup
animal_dir = '/workspace/extracted'
fungi_dir = '/workspace/extracted_fungi'

# Get shared OGs
animals = set()
for f in os.listdir(animal_dir):
    if f.endswith('.treefile'):
        animals.add(f.replace('.mafft.clipkit.treefile', ''))

fungi_set = set()
for f in os.listdir(fungi_dir):
    if f.endswith('.treefile'):
        fungi_set.add(f.replace('.mafft.clipkit.treefile', ''))

shared = sorted(animals & fungi_set)

full4_both = 0
full4_animal = 0
full4_fungi = 0

for og in shared:
    t_a = dendropy.Tree.get(path=os.path.join(animal_dir, og + '.mafft.clipkit.treefile'), schema='newick', preserve_underscores=True)
    t_f = dendropy.Tree.get(path=os.path.join(fungi_dir, og + '.mafft.clipkit.treefile'), schema='newick', preserve_underscores=True)
    
    n_a = len(list(t_a.leaf_node_iter()))
    n_f = len(list(t_f.leaf_node_iter()))
    
    if n_a == 4:
        full4_animal += 1
    if n_f == 4:
        full4_fungi += 1
    if n_a == 4 and n_f == 4:
        full4_both += 1

print(f'Animal orthogroups with 4 species: {full4_animal}')
print(f'Fungi orthogroups with 4 species: {full4_fungi}')
print(f'Both with 4 species: {full4_both}')
