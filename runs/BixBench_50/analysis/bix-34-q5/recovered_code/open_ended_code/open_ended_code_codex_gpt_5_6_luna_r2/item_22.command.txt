import zipfile
from decimal import Decimal, getcontext
getcontext().prec=80
class N:
 def __init__(self,name=None): self.name=name; self.children=[]; self.length=Decimal(0)
def parse(s):
 i=0
 def skip():
  nonlocal i
  while i<len(s) and s[i].isspace(): i+=1
 def lab():
  nonlocal i
  skip(); st=i
  while i<len(s) and s[i] not in '(),:;[]' and not s[i].isspace(): i+=1
  return s[st:i] or None
 def num():
  nonlocal i
  skip(); st=i
  while i<len(s) and s[i] not in ',();[]' and not s[i].isspace(): i+=1
  return Decimal(s[st:i])
 def sub():
  nonlocal i
  skip()
  if s[i]=='(':
   i+=1; n=N()
   while True:
    n.children.append(sub()); skip()
    if s[i]==',': i+=1; continue
    if s[i]==')': i+=1; break
   n.name=lab()
  else:n=N(lab())
  skip()
  if i<len(s) and s[i]==':': i+=1;n.length=num()
  skip()
  return n
 return sub()
def means(root):
 vals=[]
 def rec(n):
  if not n.children:return [(n.name,Decimal(0))]
  all=[]
  for c in n.children:
   g=[(x,d+c.length) for x,d in rec(c)]
   for x,d in g:
    vals.extend(d+e for _,e in all)
   all.extend(g)
  return all
 rec(root)
 return sum(vals,Decimal(0))/len(vals)
def get(zname):
 out=[]
 with zipfile.ZipFile(zname) as z:
  for p in z.namelist():
   if p.endswith('.treefile'):
    out.append((p.split('.')[0],means(parse(z.read(p).decode().strip()))))
 return sorted(out)
def median(a):
 a=sorted(x for _,x in a); n=len(a)
 return (a[(n-1)//2]+a[n//2])/Decimal(2)
A=get('inputs/scogs_animals.zip');F=get('inputs/scogs_fungi.zip')
ma=median(A);mf=median(F)
print('animal n',len(A),'median',ma)
print('fungi n',len(F),'median',mf)
print('ratio',mf/ma)
sa=sorted(A,key=lambda x:x[1]); sf=sorted(F,key=lambda x:x[1])

print('fungi median entry',sf[124])
print('ratio inverse',ma/mf)
PY"
