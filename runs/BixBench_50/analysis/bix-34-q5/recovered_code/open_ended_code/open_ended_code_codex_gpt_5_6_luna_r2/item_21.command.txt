/bin/bash -lc "python - <<'PY'
import zipfile

def ds(prefix,shorts):
 out=[]
 for x in shorts:
  p=f'inputs/{prefix}_{x}.busco.zip'
  with zipfile.ZipFile(p) as z:
   n=[x for x in z.namelist() if x.endswith('/full_table.tsv')][0]
   d={}
   for line in z.read(n).decode().splitlines():
    if line and not line.startswith('#'):
     q=line.split('\\t');d[q[0]]=q[1]
   out.append(d)
 return out
for kind,prefix,shorts in [('animals','Animals',['Cele','Ggal','Mmus','Xlae']),('fungi','Fungi',['Ncra','Scer','Spom','Umay'])]:
 d=ds(prefix,shorts); ids=set.intersection(*(set(x) for x in d))
 print(kind)
 for k in range(1,5): print('>=',k,'complete',sum(sum(x[i]=='Complete' for x in d)>=k for i in ids))
PY"
