/bin/bash -lc "python - <<'PY'
import zipfile, re, collections
for kind,paths in [('animals',[f'inputs/Animals_{x}.busco.zip' for x in ['Cele','Ggal','Mmus','Xlae']]),('fungi',[f'inputs/Fungi_{x}.busco.zip' for x in ['Ncra','Scer','Spom','Umay']])]:
    print('\\n',kind)
    for p in paths:
        with zipfile.ZipFile(p) as z:
            n=[x for x in z.namelist() if x.endswith('/full_table.tsv')][0]
            raw=z.read(n).decode('utf-8','replace').splitlines()
            rows=[]
            for line in raw:
                if not line or line.startswith('#'): continue
                parts=line.split('\\t')
                rows.append(parts)
            print(p, 'rows',len(rows),'cols',max(map(len,rows)),'statuses',collections.Counter(x[1] for x in rows))
            print('first',rows[0][:5])
PY"
