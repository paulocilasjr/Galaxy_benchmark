/bin/bash -c "python3 - <<'PY'
import zipfile, io, statistics, itertools
from Bio import Phylo

def per_tree_means(path):
    vals=[]
    with zipfile.ZipFile(path) as z:
        for n in z.namelist():
            if not n.endswith('.mafft.clipkit.treefile'):
                continue
            tree=Phylo.read(io.StringIO(z.read(n).decode()), 'newick')
            taxa=tree.get_terminals()
            ds=[tree.distance(a,b) for a,b in itertools.combinations(taxa,2)]
            if ds:
                vals.append(statistics.mean(ds))
    return vals
an=per_tree_means('/workspace/inputs/scogs_animals.zip')
fu=per_tree_means('/workspace/inputs/scogs_fungi.zip')
ma=statistics.median(an)
mf=statistics.median(fu)
print('animal trees',len(an),'median',ma)
print('fungi trees',len(fu),'median',mf)
print('fungi/animal',mf/ma)
print('rounded 2',round(mf/ma,2))
PY"
