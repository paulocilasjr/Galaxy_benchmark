/bin/bash -c "python - <<'PY'
from phykit.services.tree.base import Tree
import glob, statistics, time
for label,p in [('animals','tmp/scogs_animals/*.treefile'),('fungi','tmp/scogs_fungi/*.treefile')]:
    files=glob.glob(p); vals=[]; bad=[]
    t=time.time()
    for f in files:
        try:
            svc=Tree(tree_file_path=f)
            v=svc.calculate_treeness()
            vals.append(v)
        except Exception as e:
            bad.append((f,repr(e)))
    print(label,'n',len(vals),'median',statistics.median(vals),'mean',statistics.mean(vals),'min',min(vals),'max',max(vals),'time',time.time()-t,'bad',len(bad))
PY"
