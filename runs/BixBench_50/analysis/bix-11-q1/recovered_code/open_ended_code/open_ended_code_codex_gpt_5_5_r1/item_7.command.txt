/bin/bash -lc "python3 - <<'PY'
import zipfile
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(zname) as z:
        names=z.namelist()
        print(zname, 'treefile', sum(n.endswith('.treefile') for n in names), 'iqtree', sum(n.endswith('.iqtree') for n in names), 'bionj', sum(n.endswith('.bionj') for n in names))
        for n in names:
            if n.endswith('.treefile'):
                print('sample', n, z.read(n).decode()[:300])
                break
PY"
