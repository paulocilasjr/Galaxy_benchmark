import subprocess, os

# Count 4-species trees and check their treeness
four_species_zero = []
four_species_nonzero = []
three_species = []

for f in sorted(os.listdir('/tmp/scogs_animals/')):
    if f.endswith('.treefile'):
        path = os.path.join('/tmp/scogs_animals/', f)
        with open(path) as fh:
            tree = fh.read()
        n_species = tree.count('Animals_')
        result = subprocess.run(['/tmp/venv/bin/phykit', 'treeness', path], 
                              capture_output=True, text=True)
        val = float(result.stdout.strip())
        
        if n_species == 4:
            if val == 0.0:
                four_species_zero.append((f, val, tree.strip()[:120]))
            else:
                four_species_nonzero.append((f, val))
        elif n_species == 3:
            three_species.append((f, val))

print(f"4-species trees: {len(four_species_zero) + len(four_species_nonzero)}")
print(f"  4-species zero treeness: {len(four_species_zero)}")
print(f"  4-species non-zero treeness: {len(four_species_nonzero)}")
print(f"3-species trees: {len(three_species)}")

# Show a 4-species zero treeness example
if four_species_zero:
    print(f"\nExample 4-species zero treeness:")
    print(f"  {four_species_zero[0][0]}: {four_species_zero[0][2]}")
EOF
