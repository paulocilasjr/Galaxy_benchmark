# Let's also cross-check: manual computation on a random subset
python3 << 'EOF'
import subprocess, os, statistics, random

# Random sample of 20 animal trees and 20 fungi trees for manual verification
random.seed(42)

animal_trees = [f for f in os.listdir('/tmp/scogs_animals/') if f.endswith('.treefile')]
fungi_trees = [f for f in os.listdir('/tmp/scogs_fungi/') if f.endswith('.treefile')]

print(f"Animal: {len(animal_trees)} trees, Fungi: {len(fungi_trees)} trees")

# Verify the median manually with a different method
animal_subset = random.sample(animal_trees, min(20, len(animal_trees)))
fungi_subset = random.sample(fungi_trees, min(20, len(fungi_trees)))

print("\nManual verification on random subsets:")
print("Animal subset:")
for f in animal_subset:
    path = os.path.join('/tmp/scogs_animals/', f)
    result = subprocess.run(['/tmp/venv/bin/phykit', 'treeness', path], 
                          capture_output=True, text=True)
    val = result.stdout.strip()
    print(f"  {f}: {val}")

print("\nFungi subset:")
for f in fungi_subset:
    path = os.path.join('/tmp/scogs_fungi/', f)
    result = subprocess.run(['/tmp/venv/bin/phykit', 'treeness', path], 
                          capture_output=True, text=True)
    val = result.stdout.strip()
    print(f"  {f}: {val}")
EOF
