import subprocess, os

# Analyze fungi trees
four_species_zero = []
four_species_nonzero = []
three_species = []
other = []

for f in sorted(os.listdir('/tmp/scogs_fungi/')):
    if f.endswith('.treefile'):
        path = os.path.join('/tmp/scogs_fungi/', f)
        with open(path) as fh:
            tree = fh.read()
        n_species = tree.count('Fungi_')
        result = subprocess.run(['/tmp/venv/bin/phykit', 'treeness', path], 
                              capture_output=True, text=True)
        val = float(result.stdout.strip())
        
        if n_species == 4:
            if val == 0.0:
                four_species_zero.append((f, val))
            else:
                four_species_nonzero.append((f, val))
        elif n_species == 3:
            three_species.append((f, val))
        else:
            other.append((f, n_species, val))

print(f"4-species trees: {len(four_species_zero) + len(four_species_nonzero)}")
print(f"  4-species zero treeness: {len(four_species_zero)}")
print(f"  4-species non-zero treeness: {len(four_species_nonzero)}")
print(f"3-species trees: {len(three_species)}")
print(f"Other: {other}")

# Show a 4-species zero treeness example if any
if four_species_zero:
    print(f"\nExample 4-species zero treeness:")
    path = os.path.join('/tmp/scogs_fungi/', four_species_zero[0][0])
    with open(path) as fh:
        print(f"  {four_species_zero[0][0]}: {fh.read().strip()[:120]}")

# Show a 3-species example
if three_species:
    print(f"\nExample 3-species tree:")
    path = os.path.join('/tmp/scogs_fungi/', three_species[0][0])
    with open(path) as fh:
        print(f"  {three_species[0][0]}: {fh.read().strip()[:120]}")
EOF
