# Spot-check more trees
echo "=== Animal trees with treeness > 0 ==="
python3 << 'EOF'
import subprocess, os

# Get treeness for all animal treefiles and verify manually
animal_trees = []
for f in sorted(os.listdir('/tmp/scogs_animals/')):
    if f.endswith('.treefile'):
        path = os.path.join('/tmp/scogs_animals/', f)
        result = subprocess.run(['/tmp/venv/bin/phykit', 'treeness', path], 
                              capture_output=True, text=True)
        val = float(result.stdout.strip())
        animal_trees.append((f, val))

# Show non-zero ones
nonzero = [(f, v) for f, v in animal_trees if v > 0]
print(f"Total trees: {len(animal_trees)}, Non-zero: {len(nonzero)}")
for name, val in nonzero[:5]:
    print(f"  {name}: {val}")

# Check: do all 4-species trees have non-zero treeness?
# Let's count species in each tree
for name, val in nonzero:
    path = os.path.join('/tmp/scogs_animals/', name)
    with open(path) as f:
        tree = f.read()
    n_species = tree.count('Animals_')
    print(f"  {name}: treeness={val}, species={n_species}")
EOF
