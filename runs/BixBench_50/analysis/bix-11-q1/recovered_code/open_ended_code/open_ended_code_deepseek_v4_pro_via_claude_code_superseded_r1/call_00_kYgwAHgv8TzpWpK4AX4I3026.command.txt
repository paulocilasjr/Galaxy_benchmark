# Compute treeness for all animal treefiles
for f in /tmp/scogs_animals/*.treefile; do
    val=$(/tmp/venv/bin/phykit treeness "$f" 2>/dev/null)
    echo "$val"
done > /tmp/animal_treeness.txt
echo "Animal count: $(wc -l < /tmp/animal_treeness.txt)"
echo "First few:"
head -10 /tmp/animal_treeness.txt
