/bin/bash -c "python3 - <<'PY'
import zipfile,re,collections
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 z=zipfile.ZipFile(zname)
 counts=collections.Counter()
 lengths=[]
 for n in z.namelist():
  if n.endswith('.treefile'):
   s=z.read(n).decode(errors='replace').strip()
   taxa=len(re.findall(r'[A-Za-z_]+(?=:)', s))
   counts[taxa]+=1
   lengths.append(len(s))
 print(zname, 'taxa count', sorted(counts.items()), 'N',sum(counts.values()))
 # count all .faa sequences maybe taxa per gene for all faa
 counts2=collections.Counter()
 for n in z.namelist():
  if n.endswith('.faa') and '.mafft' not in n:
   s=z.read(n).decode(errors='replace')
   taxa=s.count('>')
   counts2[taxa]+=1
 print('faa taxa',sorted(counts2.items()))
PY"
