/bin/bash -c "python3 - <<'PY'
import zipfile
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 z=zipfile.ZipFile(zname)
 print('\\n===',zname)
 # choose one treefile and bionj/mldist
 for n in z.namelist():
  if n.endswith('.treefile') or n.endswith('.bionj') or n.endswith('.mldist'):
   print('\\n---',n)
   data=z.read(n).decode(errors='replace')
   print(data[:1000])
   break
 # .faa sample
 for n in z.namelist():
  if n.endswith('.faa') and not '.mafft' in n:
   print('\\n--- faa',n)
   print(z.read(n).decode()[:1000])
   break
PY"
