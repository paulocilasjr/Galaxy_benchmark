def parse_newick(s):
    s = s.strip()
    i = 0
    n = len(s)
    def skip_ws():
        nonlocal i
        while i < n and s[i].isspace():

        start = i
        while i < n and s[i] not in '(),:;[]':

        nonlocal i
        if i < n and s[i] == '[':

            while i < n:
                c = s[i]
                i += 1

                elif c == ']':

def metrics(tree_text):
    root = parse_newick(tree_text)
    leaves = []
    branches = []
    def walk(node, isroot=False):
        if not isroot:
            branches.append((node.length, bool(node.children)))
        if node.children:
            for c in node.children: walk(c)
        else:
            leaves.append(node.label)
    walk(root, True)
    if any(x is None for x,_ in branches):
        raise ValueError('missing branch length')
    total = sum(x for x,_ in branches)
    internal = sum(x for x,is_internal in branches if is_internal)
    terminal = total-internal
    return len(leaves), total, internal, terminal, internal/total if total else float('nan'), leaves
