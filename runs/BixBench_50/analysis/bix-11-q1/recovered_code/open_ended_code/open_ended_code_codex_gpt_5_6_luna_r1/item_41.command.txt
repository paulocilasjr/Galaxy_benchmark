from decimal import Decimal, getcontext
import zipfile
getcontext().prec=50

# Minimal exact parser for the IQ-TREE Newick files.
class Node:
    def __init__(self, children=None, length=None): self.children=children or []; self.length=length

def parse(s):
    i=0
    def subtree():
        nonlocal i
        if s[i]=='(':
            i+=1; children=[subtree()]
            while s[i]==',': i+=1; children.append(subtree())
            if s[i]"'!='"')': raise ValueError('bad newick')
            i+=1
            while i<len(s) and s[i] not in ':,();': i+=1
            length=None
            if i<len(s) and s[i]==':':
                i+=1; j=i
                while i<len(s) and s[i] not in ',();': i+=1
                length=Decimal(s[j:i])
            return Node(children,length)
        j=i
        while i<len(s) and s[i] not in ':,();': i+=1
        length=None
        if i<len(s) and s[i]==':':
            i+=1; j=i
            while i<len(s) and s[i] not in ',();': i+=1
            length=Decimal(s[j:i])
        return Node([],length)
    return subtree()
def treeness(s):
    root=parse(s.strip()); edges=[]
    def walk(x,root=False):
        if not root: edges.append((x.length,bool(x.children)))
        for c in x.children: walk(c)
    walk(root,True)
    total=sum((x for x,_ in edges),Decimal(0)); internal=sum((x for x,b in edges if b),Decimal(0))
    return internal/total, internal, total
for p in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    rows=[]
    with zipfile.ZipFile(p) as z:
        for name in z.namelist():
            if name.endswith('.treefile'):
                t,i,tot=treeness(z.read(name).decode())
                rows.append((t,name,i,tot,z.read(name).decode().strip()))
    rows.sort()
    n=len(rows)
    if n%2: med=rows[n//2][0]; meds=rows[n//2:n//2+1]
    else: med=(rows[n//2-1][0]+rows[n//2][0])/2; meds=rows[n//2-1:n//2+1]
    print(p,'n',n,'median',med)
    for x in meds: print('median contributor',x)
