import zipfile, tempfile

# Extract and verify a few sample trees from each group
for zip_path, label in [('/workspace/inputs/scogs_animals.zip', 'Animals'), ('/workspace/inputs/scogs_fungi.zip', 'Fungi')]:

    with zipfile.ZipFile(zip_path, 'r') as zf:
        files = [f for f in zf.namelist() if f.endswith('.treefile')]
        with tempfile.TemporaryDirectory() as tmpdir:

                zf.extract(tf, tmpdir)
                path = f'{tmpdir}/{tf}'
                with open(path) as f:
                    s = f.read().strip()
                print(f'{tf}:')
                print(f'  {s}')
                # Manual count
                import re
                s_clean = re.sub(r'\[[^\]]*\]', '', s)
                # Count lengths after :
                lengths = re.findall(r':([0-9.eE+-]+)', s_clean)
                print(f'  Branch lengths ({len(lengths)}): {[float(x) for x in lengths]}')
                
                # Count leaf labels (non-empty labels before : or , or ))
                # Rough: count comma-separated non-length tokens
