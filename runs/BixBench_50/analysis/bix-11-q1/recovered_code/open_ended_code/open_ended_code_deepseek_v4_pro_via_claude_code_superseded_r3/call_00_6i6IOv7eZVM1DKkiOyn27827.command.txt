# Compute from IQ-TREE outputs (IQ-TREE's own treeness values)
import zipfile, tempfile, os, re, statistics

def extract_treeness(zip_path):
    results = []
    with zipfile.ZipFile(zip_path, 'r') as zf:
        iqtree_files = sorted([f for f in zf.namelist() if f.endswith('.iqtree')])
        with tempfile.TemporaryDirectory() as tmpdir:
            for tf in iqtree_files:
                try:
                    zf.extract(tf, tmpdir)
                    path = os.path.join(tmpdir, tf)
                    with open(path, 'r') as f:
                        content = f.read()
                    match = re.search(r'Sum of internal branch lengths: [\d.]+ \(([\d.]+)% of tree length\)', content)
                    if match:
                        results.append(float(match.group(1)) / 100.0)
                except:
                    pass
    return sorted(results), statistics.median(results)

animal_vals, animal_med = extract_treeness('/workspace/inputs/scogs_animals.zip')
fungi_vals, fungi_med = extract_treeness('/workspace/inputs/scogs_fungi.zip')
diff = fungi_med - animal_med
print(f'IQ-TREE output method: {diff}')
