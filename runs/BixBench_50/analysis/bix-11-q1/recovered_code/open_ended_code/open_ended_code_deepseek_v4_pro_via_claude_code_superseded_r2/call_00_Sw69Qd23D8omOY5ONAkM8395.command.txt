import re
import os
import statistics

def extract_treeness(filepath):
    """Extract treeness from an IQ-TREE output file."""
    try:
        with open(filepath, 'r') as f:
            content = f.read()
        
        total_match = re.search(r'Total tree length \(sum of branch lengths\):\s+([\d.]+)', content)
        internal_match = re.search(r'Sum of internal branch lengths:\s+([\d.]+)\s+\(', content)
        
        if total_match and internal_match:
            total = float(total_match.group(1))
            internal = float(internal_match.group(1))
            if total > 0:
                return internal / total
            else:
                return None
    except Exception as e:
        print(f"Error processing {filepath}: {e}")
    return None

# Process animals with detailed debug
animals_dir = '/workspace/scogs_animals'
fungi_dir = '/workspace/scogs_fungi'

# Check animal files that have non-zero treeness
animal_data = {}
for f in sorted(os.listdir(animals_dir)):
    if f.endswith('.iqtree'):
        path = os.path.join(animals_dir, f)
        t = extract_treeness(path)
        gene_id = f.replace('.faa.mafft.clipkit.iqtree', '')
        animal_data[gene_id] = t

non_zero_animals = {k: v for k, v in animal_data.items() if v is not None and v > 0}
zero_animals = {k: v for k, v in animal_data.items() if v is not None and v == 0}

print(f"Animal SCOGs with treeness > 0: {len(non_zero_animals)}")
print(f"Animal SCOGs with treeness = 0: {len(zero_animals)}")
print(f"Total animal SCOGs with valid data: {len(animal_data)}")
print()

# Show all non-zero animal treeness values sorted
sorted_nonzero = sorted(non_zero_animals.items(), key=lambda x: x[1])
print("Non-zero animal treeness values (sorted):")
for gene, t in sorted_nonzero:
    print(f"  {gene}: {t:.10f}")

# Check median
values = [v for v in animal_data.values() if v is not None]
print(f"\nAnimals: {len(values)} values, median = {statistics.median(values)}")
print(f"Sorted values: {sorted(values)[:20]}")
