import re
import os
import statistics

def extract_treeness(filepath):
    """Extract treeness from an IQ-TREE output file."""
    try:
        with open(filepath, 'r') as f:
            content = f.read()
        
        total_match = re.search(r'Total tree length \(sum of branch lengths\):\s+([\d.]+)', content)
        internal_match = re.search(r'Sum of internal branch lengths:\s+([\d.]+)\s+\(', content)
        
        if total_match and internal_match:
            total = float(total_match.group(1))
            internal = float(internal_match.group(1))
            if total > 0:
                return internal / total
            else:
                return 0.0
    except Exception as e:
        print(f"Error processing {filepath}: {e}")
    return None

# Process animals
animals_dir = '/workspace/scogs_animals'
fungi_dir = '/workspace/scogs_fungi'

animal_treeness = []
for f in sorted(os.listdir(animals_dir)):
    if f.endswith('.iqtree'):
        path = os.path.join(animals_dir, f)
        t = extract_treeness(path)
        if t is not None:
            animal_treeness.append(t)

fungi_treeness = []
for f in sorted(os.listdir(fungi_dir)):
    if f.endswith('.iqtree'):
        path = os.path.join(fungi_dir, f)
        t = extract_treeness(path)
        if t is not None:
            fungi_treeness.append(t)

print(f"Animals: {len(animal_treeness)} treeness values extracted")
print(f"Fungi: {len(fungi_treeness)} treeness values extracted")

median_animals = statistics.median(animal_treeness)
median_fungi = statistics.median(fungi_treeness)

print(f"Median animals treeness: {median_animals}")
print(f"Median fungi treeness: {median_fungi}")
print(f"Difference (fungi - animals): {median_fungi - median_animals}")
print(f"Difference (animals - fungi): {median_animals - median_fungi}")

# Show distribution
print(f"\nAnimals treeness values: min={min(animal_treeness):.6f}, max={max(animal_treeness):.6f}, mean={statistics.mean(animal_treeness):.6f}")
print(f"Fungi treeness values: min={min(fungi_treeness):.6f}, max={max(fungi_treeness):.6f}, mean={statistics.mean(fungi_treeness):.6f}")

# Show a few values
print(f"\nAnimals (first 10): {animal_treeness[:10]}")
print(f"Fungi (first 10): {fungi_treeness[:10]}")
