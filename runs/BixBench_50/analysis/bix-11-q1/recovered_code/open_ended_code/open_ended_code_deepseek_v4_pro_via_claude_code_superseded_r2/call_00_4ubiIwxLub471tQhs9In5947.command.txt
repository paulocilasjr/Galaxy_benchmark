import re, os, statistics

def extract_treeness(filepath):
    with open(filepath, 'r') as f:
        content = f.read()
    total_match = re.search(r'Total tree length \(sum of branch lengths\):\s+([\d.]+)', content)
    internal_match = re.search(r'Sum of internal branch lengths:\s+([\d.]+)\s+\(', content)
    if total_match and internal_match:
        total = float(total_match.group(1))
        internal = float(internal_match.group(1))
        return internal / total if total > 0 else None
    return None

animals = []
for f in os.listdir('/workspace/scogs_animals'):
    if f.endswith('.iqtree'):
        t = extract_treeness(os.path.join('/workspace/scogs_animals', f))
        if t is not None:
            animals.append(t)

fungi = []
for f in os.listdir('/workspace/scogs_fungi'):
    if f.endswith('.iqtree'):
        t = extract_treeness(os.path.join('/workspace/scogs_fungi', f))
        if t is not None:
            fungi.append(t)

med_animals = statistics.median(animals)
med_fungi = statistics.median(fungi)
diff = med_fungi - med_animals

print(f"Animals: {len(animals)} SCOGs, median treeness = {med_animals}")
print(f"Fungi: {len(fungi)} SCOGs, median treeness = {med_fungi}")
print(f"Difference (fungi - animals): {diff}")
print(f"Difference (fungi - animals): {diff:.15f}")

# Also show mean and percentiles for context
print(f"\nAnimals: mean={statistics.mean(animals):.6f}, stdev={statistics.stdev(animals):.6f}")
print(f"Fungi: mean={statistics.mean(fungi):.6f}, stdev={statistics.stdev(fungi):.6f}")

# Verify: count zeros per group
print(f"\nAnimals zero treeness: {sum(1 for v in animals if v == 0)}/{len(animals)}")
print(f"Fungi zero treeness: {sum(1 for v in fungi if v == 0)}/{len(fungi)}")
