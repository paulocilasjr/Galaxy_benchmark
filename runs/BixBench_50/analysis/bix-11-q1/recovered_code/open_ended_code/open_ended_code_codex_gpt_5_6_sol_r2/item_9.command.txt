/bin/bash -lc "python - <<'PY'
from decimal import Decimal, getcontext
import zipfile,re
getcontext().prec=50
num=r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?'
for group in ('animals','fungi'):
 rows=[]
 with zipfile.ZipFile(f'inputs/scogs_{group}.zip') as z:
  for name in z.namelist():
   if not name.endswith('.treefile'): continue
   s=z.read(name).decode().strip()
   allbl=[Decimal(x) for x in re.findall(r':\\s*('+num+r')',s)]
   intl=[Decimal(x) for x in re.findall(r'\\)\\s*(?:["'^():,;]+)?'"\\s*:\\s*('+num+r')',s)]
   val=sum(intl,Decimal())/sum(allbl,Decimal())
   rows.append((val,name,s,sum(intl,Decimal()),sum(allbl,Decimal())))
 rows.sort()
 if len(rows)%2:
  med=rows[len(rows)//2][0]; selected=[rows[len(rows)//2]]
 else:
  selected=rows[len(rows)//2-1:len(rows)//2+1]
  med=sum((x[0] for x in selected),Decimal())/2
 print(group,'median',med)
 for val,name,s,intl,total in selected:
  print(name,'treeness',val,'internal',intl,'total',total)
  print(s)
PY
unzip -t inputs/scogs_animals.zip | tail -n 2
unzip -t inputs/scogs_fungi.zip | tail -n 2"
