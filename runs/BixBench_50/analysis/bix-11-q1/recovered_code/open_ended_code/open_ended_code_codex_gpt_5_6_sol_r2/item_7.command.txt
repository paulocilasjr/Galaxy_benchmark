/bin/bash -lc "python - <<'PY'
import zipfile,re,statistics,collections
float_re=r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?'
for group in ('animals','fungi'):
 with zipfile.ZipFile(f'inputs/scogs_{group}.zip') as z:
  rows=[]
  for name in z.namelist():
   if not name.endswith('.treefile'): continue
   s=z.read(name).decode().strip()
   if s.count(';') "'!= 1 or not s.endswith('"';'):
    raise ValueError((name,'not one Newick tree'))
   lengths=[float(x) for x in re.findall(r':\\s*('+float_re+r')',s)]
   internals=[float(x) for x in re.findall(r'\\)\\s*(?:["'^():,;]+)?'"\\s*:\\s*('+float_re+r')',s)]
   tips=re.findall(r'(?<=[(,])\\s*(["'^():,;]+)'"\\s*:',s)
   if len(lengths) "'!= len(tips)+len(internals):
    raise ValueError((name,len(lengths),len(tips),len(internals),s))
   rows.append((sum(internals)/sum(lengths),len(tips),name))
  vals=[x[0] for x in rows]
  vals4=[x[0] for x in rows if x[1]==4]
  print(group)
  print('"'all:',len(vals),'median',format(statistics.median(vals),'.17g'))
  print('4-tip only:',len(vals4),'median',format(statistics.median(vals4),'.17g'))
  print('missing-tip patterns:')
  miss=collections.Counter()
  expected={'animals':{'Animals_Cele','Animals_Ggal','Animals_Mmus','Animals_Xlae'},'fungi':{'Fungi_Ncra','Fungi_Scer','Fungi_Spom','Fungi_Umay'}}[group]
  for _,_,name in rows:
   s=z.read(name).decode()
   taxa=set(re.findall(r'(?<=[(,])\\s*(["'^():,;]+)'"\\s*:',s))
   miss[tuple(sorted(expected-taxa))]+=1
  print(miss)
PY
command -v R || true
command -v phykit || true"
