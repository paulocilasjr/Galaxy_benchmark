/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv',index_col=0, nrows=None)
# use per column sums via numpy? pandas compute
nonzero=(counts.to_numpy(dtype=np.float32)>0).sum(axis=0)
print('genes with any nonzero', (nonzero>0).sum(), 'of', counts.shape[0])
print('genes with sum >0', (counts.sum(axis=0)>0).sum())
print('genes with sum >=10', (counts.sum(axis=0)>=10).sum())
print('genes with sum >= 20', (counts.sum(axis=0)>=20).sum())
print('genes with >=10 nonzero samples', (nonzero>=10).sum())
PY"
