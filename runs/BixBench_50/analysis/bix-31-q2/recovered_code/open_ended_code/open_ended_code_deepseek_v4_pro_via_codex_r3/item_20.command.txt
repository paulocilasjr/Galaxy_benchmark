/bin/bash -c "python -u - <<'PY'
import pandas as pd, numpy as np, time, sys
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

print('loading', flush=True)
t=time.time()
meta=pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample')
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv',index_col=0).T
meta=meta.loc[counts.index]
meta['batch']='b'+meta['batch'].astype(str)
print('counts', counts.shape, 'meta', meta.shape, 'load sec', time.time()-t, flush=True)

dds=DeseqDataSet(counts=counts, metadata=meta, design='~ batch + sex', n_cpus=8, quiet=False)
print('DDS initialized', flush=True)
dds.deseq2()
print('deseq2 done', time.time()-t, flush=True)
stat=DeseqStats(dds, contrast=['sex','M','F'], n_cpus=8, quiet=False)
stat.summary()
print('unshrunk rows', stat.results_df.shape, flush=True)
print(stat.results_df.loc[['FAM138A']].to_string())
coeff='sex[T.M]'
print('LFC cols', stat.LFC.columns.tolist(), 'SE cols', stat.SE.columns.tolist(), flush=True)
stat.lfc_shrink(coeff=coeff)
print('shrunk done', time.time()-t, flush=True)
print(stat.results_df.loc[['FAM138A']].to_string())
print('FAM138A lfc from results_df:', float(stat.results_df.loc['FAM138A','log2FoldChange']))
print('FAM138A baseMean:', float(stat.results_df.loc['FAM138A','baseMean']))
print('FAM138A padj:', stat.results_df.loc['FAM138A','padj'])
PY"
