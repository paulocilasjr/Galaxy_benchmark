/bin/bash -c "python -u - <<'PY'
import pandas as pd, numpy as np, time, resource
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

def rss_gb():
    return resource.getrusage(resource.RUSAGE_SELF).ru_maxrss/1024/1024
print('loading', flush=True)
t=time.time()
meta=pd.read_csv('inputs/Sample_annotated_Zenodo.csv', dtype={'batch':'str'}).set_index('sample')
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv',index_col=0,dtype='int32').T
meta=meta.loc[counts.index]
meta['batch']='b'+meta['batch'].astype(str)
print('counts', counts.shape, 'load rss_gb', rss_gb(), flush=True)
dds=DeseqDataSet(counts=counts, metadata=meta, design='~ batch + sex', n_cpus=1, quiet=True, low_memory=True)
print('DDS init rss_gb', rss_gb(), flush=True)
dds.deseq2()
print('deseq2 done', time.time()-t, 'rss_gb', rss_gb(), flush=True)
stat=DeseqStats(dds, contrast=['sex','M','F'], n_cpus=1, quiet=True)
stat.summary()
print('unshrunk done rss_gb', rss_gb(), flush=True)
coeff='sex[T.M]'
stat.lfc_shrink(coeff=coeff)
print('shrunk done', time.time()-t, 'rss_gb', rss_gb(), flush=True)
print(stat.results_df.loc[['FAM138A']].to_string())
print('FAM138A lfc', float(stat.results_df.loc['FAM138A','log2FoldChange']))
print('FAM138A baseMean', float(stat.results_df.loc['FAM138A','baseMean']))
PY"
