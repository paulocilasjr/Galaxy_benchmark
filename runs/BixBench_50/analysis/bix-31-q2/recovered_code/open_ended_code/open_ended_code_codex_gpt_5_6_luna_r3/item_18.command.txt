/bin/bash -lc "python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv',index_col=0)
meta=pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample').loc[counts.columns].copy()
meta['batch']=pd.Categorical(meta['batch'].astype(str),categories=['1','2','3','4'],ordered=False)
meta['sex']=pd.Categorical(meta['sex'],categories=['F','M'],ordered=False)
# only instantiate with a handful of genes for design inspection
small=counts.loc[['FAM138A','WASH7P','DDX11L1']].T
print(meta.dtypes)
dds=DeseqDataSet(counts=small,metadata=meta,design='~ batch + sex',quiet=True,n_cpus=1)
print(dds.obsm['design_matrix'].head())
print(dds.obsm['design_matrix'].columns.tolist())
print(dds.obsm['design_matrix'].shape, dds.design)
PY"
