import pandas as pd
import numpy as np

counts_df = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
row_sums = counts_df.sum(axis=1)

for thresh in [5, 10, 20]:
    n_keep = (row_sums >= thresh).sum()
    print(f"Total count >= {thresh}: {n_keep} genes ({n_keep/len(row_sums)*100:.1f}%)")

# FAM138A
print(f"\nFAM138A total count: {row_sums.get('FAM138A', 'NOT FOUND')}")
