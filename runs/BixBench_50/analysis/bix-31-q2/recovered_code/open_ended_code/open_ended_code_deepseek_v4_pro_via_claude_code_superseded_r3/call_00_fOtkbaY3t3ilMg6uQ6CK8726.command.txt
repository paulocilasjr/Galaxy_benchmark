import pandas as pd
import numpy as np
import warnings
warnings.filterwarnings('ignore')

# Load data
counts_df = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
annot = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')

# Simple merge-based alignment
annot = annot.set_index('sample')
annot = annot.loc[counts_df.columns]  # Align to counts columns
annot['batch'] = annot['batch'].astype(str)
annot['sex'] = annot['sex'].astype(str)

print(f"Metadata shape: {annot.shape}")
print(f"Counts (genes x samples): {counts_df.shape}")
print(f"Alignment check: {(annot.index == counts_df.columns).all()}")

from pydeseq2.dds import DeseqDataSet

# Create DeseqDataSet - pydeseq2 expects samples x genes
counts_int = counts_df.T.round().astype(int)

dds = DeseqDataSet(
    counts=counts_int,
    metadata=annot,
    design_factors="batch + sex",
)

print("DDS created successfully")

# Run DESeq2
dds.deseq2()

print("\nDESeq2 completed")
print(f"Results shape: {dds.results_df.shape}")
print(f"Results columns: {dds.results_df.columns.tolist()}")
print(f"\nFirst 5 rows:")
print(dds.results_df.head())
print(f"\nResults index names: {dds.results_df.index[:10].tolist()}")
