import pandas as pd
import numpy as np

# Check if counts are integers
counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
vals = counts.values.astype(float)
is_int = np.allclose(vals, np.round(vals))
print(f"All values are effectively integers: {is_int}")
print(f"Any negative values: {(vals < 0).any()}")
print(f"Min: {vals.min():.1f}, Max: {vals.max():.1f}")

# Check sample alignment
annot = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')
count_samples = set(counts.columns)
annot_samples = set(annot['sample'])
print(f"\nCount matrix samples: {len(count_samples)}")
print(f"Annotation samples: {len(annot_samples)}")
print(f"Intersection: {len(count_samples & annot_samples)}")
print(f"Only in counts: {len(count_samples - annot_samples)}")
print(f"Only in annot: {len(annot_samples - count_samples)}")

# Check FAM138A data
if 'FAM138A' in counts.index:
    fam = counts.loc['FAM138A']
    print(f"\nFAM138A counts: min={fam.min()}, max={fam.max()}, mean={fam.mean():.2f}, nonzero={(fam > 0).sum()}/{len(fam)}")

# Batch and sex details
print(f"\nBatch unique values: {sorted(annot['batch'].unique())}")
print(f"Sex counts:\n{annot['sex'].value_counts()}")
