import pandas as pd
import numpy as np
import warnings
warnings.filterwarnings('ignore')

# Load data
counts_df = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
annot = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')

# Align samples
annot_aligned = annot[annot['sample'].isin(counts_df.columns)].set_index('sample').loc[counts_df.columns].reset_index()
annot_aligned['batch'] = annot_aligned['batch'].astype(str)
annot_aligned['sex'] = annot_aligned['sex'].astype(str)
metadata = annot_aligned.set_index('sample')

from pydeseq2.dds import DeseqDataSet

# Create DeseqDataSet - samples x genes
counts_int = counts_df.T.round().astype(int)  # samples x genes

dds = DeseqDataSet(
    counts=counts_int,
    metadata=metadata,
    design_factors="batch + sex",
)

print("DDS created")

# Run DESeq2
dds.deseq2()

print("\nDESeq2 completed")
print(f"Results shape: {dds.results_df.shape}")
print(f"Results columns: {dds.results_df.columns.tolist()}")
print(f"\nFirst 5 rows:")
print(dds.results_df.head())
print(f"\nLast few columns: {dds.results_df.columns[-10:].tolist()}")
