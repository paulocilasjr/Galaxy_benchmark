import pandas as pd
import numpy as np
import warnings
warnings.filterwarnings('ignore')

# Load data
counts_df = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
annot = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')

print(f"Counts columns (first 5): {counts_df.columns[:5].tolist()}")
print(f"Annot sample (first 5): {annot['sample'].head().tolist()}")

# Align samples
annot_aligned = annot.set_index('sample').loc[counts_df.columns].copy()
annot_aligned = annot_aligned.reset_index().rename(columns={'index': 'sample'})
# Or just: annot_aligned = annot[annot['sample'].isin(counts_df.columns)].set_index('sample').loc[counts_df.columns].reset_index()

annot_aligned['batch'] = annot_aligned['batch'].astype(str)
annot_aligned['sex'] = annot_aligned['sex'].astype(str)

print(f"\nAligned annotation shape: {annot_aligned.shape}")
print(annot_aligned.head())

from pydeseq2.dds import DeseqDataSet

# Round counts to integers
counts_int = counts_df.round().astype(int)

# Create DeseqDataSet - samples x genes
metadata = annot_aligned.set_index('sample')
# Ensure metadata is in same order as counts columns
metadata = metadata.loc[counts_int.columns]

print(f"\nMetadata index matches counts columns: {all(metadata.index == counts_int.columns)}")

dds = DeseqDataSet(
    counts=counts_int.T,  # pydeseq2 expects samples x genes
    metadata=metadata,
    design_factors="batch + sex",
)

print(f"DDS shape: {dds.counts.shape}")

# Run DESeq2
dds.deseq2()

print("\nDESeq2 completed")
print(f"Results shape: {dds.results_df.shape}")
print(f"Results columns: {dds.results_df.columns.tolist()}")
print(f"\nFirst 5 rows:")
print(dds.results_df.head())
