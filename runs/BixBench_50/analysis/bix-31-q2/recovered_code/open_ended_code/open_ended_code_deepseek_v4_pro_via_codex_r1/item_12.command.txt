/bin/bash -c "nproc && free -h && python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
meta=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv', index_col=0)
meta['batch']=meta['batch'].astype('category')
meta['sex']=pd.Categorical(meta['sex'], categories=['F','M'], ordered=False)
print(meta.dtypes)
# fake tiny counts to inspect design only
import numpy as np
counts=pd.DataFrame(np.random.randint(0,10,size=(len(meta),10)), index=meta.index, columns=[f'g{i}' for i in range(10)])
dds=DeseqDataSet(counts=counts, metadata=meta, design='~batch+sex', quiet=True)
print('obs dtypes')
print(dds.obs.dtypes)
print('design columns', dds.obsm['design_matrix'].columns.tolist())
print('formulaic contrasts available?')
fc=dds.formulaic_contrasts
print(type(fc), [x for x in dir(fc) if not x.startswith('_')])
print('contrasts attr?', getattr(fc,'contrasts',None))
print(dds.obsm['design_matrix'].head())
PY"
