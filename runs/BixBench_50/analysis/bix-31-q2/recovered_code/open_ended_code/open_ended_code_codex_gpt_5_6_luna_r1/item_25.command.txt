/bin/bash -lc "/workspace/tmp/pydeseq2-venv/bin/python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
base='/workspace/inputs'
meta=pd.read_csv(base+'/Sample_annotated_Zenodo.csv').set_index('sample')
meta['batch']=pd.Categorical(meta['batch'].astype(str),categories=['1','2','3','4'],ordered=True)
meta['sex']=pd.Categorical(meta['sex'],categories=['F','M'],ordered=True)
counts=pd.DataFrame({'g1':[10]*len(meta),'g2':[20]*len(meta)},index=meta.index)
dds=DeseqDataSet(counts=counts,metadata=meta,design='~ batch + sex',quiet=True)
print(dds.obsm['design_matrix'].head())
print(dds.obsm['design_matrix'].columns.tolist())
print(dds.formulaic_contrasts.contrasts)
PY"
