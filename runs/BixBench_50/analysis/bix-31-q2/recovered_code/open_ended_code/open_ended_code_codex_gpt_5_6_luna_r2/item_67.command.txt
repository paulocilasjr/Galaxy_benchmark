/bin/bash -lc "python - <<'PY'
import time, gc
import numpy as np
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

start=time.time()
print('loading counts', flush=True)
hdr = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', nrows=0).columns
sample_dtypes = {c: np.int32 for c in hdr[1:]}
counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0, dtype=sample_dtypes).T
counts.index.name='sample'
metadata = pd.read_csv('inputs/Sample_annotated_Zenodo.csv', index_col='sample')
metadata = metadata.loc[counts.index, ['batch','sex']].copy()
metadata['batch'] = pd.Categorical(metadata['batch'].astype(str), categories=['1','2','3','4'])
metadata['sex'] = pd.Categorical(metadata['sex'].astype(str), categories=['F','M'])
print('constructing dds', flush=True)
dds = DeseqDataSet(counts=counts, metadata=metadata, design='~ batch + sex', n_cpus=1, low_memory=True, quiet=True)
del counts, metadata
gc.collect()
print('running deseq2', flush=True)
dds.deseq2()
print('deseq2 done minutes', (time.time()-start)/60, flush=True)
stat_res = DeseqStats(dds, contrast=['sex','M','F'], n_cpus=1, quiet=True)
stat_res.summary()
coeff_candidates=[c for c in stat_res.LFC.columns if 'sex' in c]
assert coeff_candidates == ['sex[T.M]'], coeff_candidates
coeff=coeff_candidates[0]
pre = stat_res.results_df.loc['FAM138A'].copy()
print('pre-shrink', pre.to_dict(), flush=True)
# Restrict only the optimizer's gene list, while retaining the all-gene LFC/SE
# table so PyDESeq2's default adaptive prior variance is estimated identically.
target_idx = np.flatnonzero(dds.var_names.to_numpy() == 'FAM138A')
assert len(target_idx)==1
assert target_idx[0] in dds.non_zero_idx
dds.non_zero_idx = target_idx
dds.non_zero_genes = dds.var_names[target_idx]
stat_res.lfc_shrink(coeff=coeff)
row=stat_res.results_df.loc['FAM138A']
print('post-shrink', row.to_dict(), flush=True)
print('target_idx', target_idx.tolist(), 'baseMean_gt10', float(row['baseMean'])>10, 'abs_lfc_gt05', abs(float(row['log2FoldChange']))>0.5, flush=True)
print('minutes', (time.time()-start)/60, flush=True)
# Canonical numeric token, for later explicit verification before writing.
print('CANONICAL', format(float(row['log2FoldChange']), '.17g'), flush=True)
PY"
