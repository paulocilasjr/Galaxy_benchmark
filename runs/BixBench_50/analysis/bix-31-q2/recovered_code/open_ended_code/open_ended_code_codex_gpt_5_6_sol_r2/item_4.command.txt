 df=pd.read_csv(p/fn,index_col=0)
 print('\\n',fn,'shape',df.shape,'dup genes',df.index.duplicated().sum(),'target count',sum(df.index=='FAM138A'))
 row=df.loc['FAM138A']
 print('FAM138A min max sum nonzero',row.min(),row.max(),row.sum(),(row>0).sum())
