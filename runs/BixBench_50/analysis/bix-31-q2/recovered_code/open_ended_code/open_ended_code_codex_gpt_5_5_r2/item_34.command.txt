/bin/bash -lc "python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0).iloc[:100,:].T
meta = pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample').loc[counts.index, ['batch','sex']]
meta['batch']=meta['batch'].astype(str)
meta['sex']=pd.Categorical(meta['sex'], categories=['F','M'])
dds=DeseqDataSet(counts=counts, metadata=meta, design='~ batch + sex', quiet=True)
dds.fit_size_factors()
print(dds.obs.head().to_string())
print(dds.obs.columns.tolist())
print(type(dds.obsm), dds.obsm.keys())
PY"
