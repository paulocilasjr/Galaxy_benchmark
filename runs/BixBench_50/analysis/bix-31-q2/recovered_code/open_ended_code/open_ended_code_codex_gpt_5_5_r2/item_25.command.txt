/bin/bash -lc "python - <<'PY'
import os
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

os.makedirs('codex_output', exist_ok=True)

counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0).T
meta = pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample')

# Match and order samples exactly between counts and metadata.
common = meta.index.intersection(counts.index)
if len(common) "'!= len(meta) or len(common) != len(counts):
    raise SystemExit(f'"'sample mismatch: counts={len(counts)} meta={len(meta)} common={len(common)}')
counts = counts.loc[meta.index]
meta = meta.loc[counts.index, ['batch', 'sex']].copy()

# Batch is a categorical covariate; sex is ordered to make M vs F the coefficient.
meta['batch'] = meta['batch'].astype(str)
meta['sex'] = pd.Categorical(meta['sex'], categories=['F', 'M'])

if 'FAM138A' not in counts.columns:
    raise SystemExit('FAM138A not present')

print('counts', counts.shape, 'metadata', meta.shape, flush=True)
print('FAM138A raw mean', counts['FAM138A'].mean(), 'sum', counts['FAM138A'].sum(), flush=True)

dds = DeseqDataSet(
    counts=counts,
    metadata=meta,
    design='~ batch + sex',
    n_cpus=8,
    quiet=False,
    low_memory=True,
)
dds.deseq2()
print('design columns', dds.obsm['design_matrix'].columns.tolist(), flush=True)

stat = DeseqStats(dds, contrast=['sex', 'M', 'F'], quiet=False, n_cpus=8)
stat.summary()
stat.lfc_shrink(coeff='sex[T.M]')
res = stat.results_df
print(res.loc['FAM138A'].to_string(), flush=True)
val = res.loc['FAM138A', 'log2FoldChange']
if pd.isna(val):
    raise SystemExit('FAM138A shrunken log2FoldChange is NA')
with open('codex_output/answer.txt', 'w') as fh:
    fh.write(format(float(val), '.12g') + '\\n')
print('ANSWER', format(float(val), '.12g'), flush=True)
PY"
