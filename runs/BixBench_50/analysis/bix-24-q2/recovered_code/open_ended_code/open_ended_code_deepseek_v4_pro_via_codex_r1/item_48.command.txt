/bin/bash -c "python - <<'PY'
import pandas as pd, json, warnings
warnings.filterwarnings('ignore')
import gseapy
res=pd.read_csv('/workspace/de_cbd_vs_dmso.csv',index_col=0)
mp=json.load(open('/workspace/ensembl_mapping.json'))
bg_ids=res[res['padj'].notna()].index.tolist()
sig=res[(res['padj']<0.05)&(res['log2FoldChange'].abs()>0.5)]
def syms(ids):
 return sorted(set(v['symbol'] for i in ids if (v:=mp.get(i)) and v.get('symbol')))
bg=syms(bg_ids)
up=syms(sig[sig['log2FoldChange']>0].index.tolist())
down=syms(sig[sig['log2FoldChange']<0].index.tolist())
print(len(bg), len(up), len(down))
gs=json.load(open('/workspace/GO_Biological_Process_2023.json'))
for lab,gl in [('UP',up),('DOWN',down)]:
    enr=gseapy.enrich(gene_list=gl, gene_sets=gs, background=bg, outdir=None, cutoff=0.05, no_plot=True, verbose=False)
    r=enr.results.copy()
    print('\\n===',lab,'n results',len(r))
    print(r[['Term','Overlap','P-value','Adjusted P-value','Odds Ratio','Genes']].head(20).to_string(max_colwidth=80))
PY"
