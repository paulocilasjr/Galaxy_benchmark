/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np, time, warnings
warnings.filterwarnings('ignore')
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
meta = pd.read_csv('inputs/sample_layout.csv')
meta['sample_col'] = meta['SampleID'].str.replace('-', '_', regex=False)
meta = meta.set_index('sample_col')
# all groups; ensure columns in same order as meta
meta['group'] = meta['Group'].astype('category')
counts_all = counts.loc[:, meta.index].T.astype(int)
print('counts_all', counts_all.shape)
dds = DeseqDataSet(counts=counts_all, metadata=meta[['group']], design='~group', quiet=False)
t=time.time(); dds.deseq2(); print('deseq2 done', round(time.time()-t,1))
st = DeseqStats(dds, contrast=['group','CBD_IC50','DMSO'], alpha=0.05, quiet=False)
t=time.time(); st.summary(); print('stats done', round(time.time()-t,1))
res=st.results_df.copy()
res.to_csv('/workspace/de_cbd_vs_dmso_fullmodel.csv')
print(res.shape)
print('sig', ((res['padj']<0.05)&(res['log2FoldChange'].abs()>0.5)).sum())
print('up', ((res['padj']<0.05)&(res['log2FoldChange']>0.5)).sum())
print('down', ((res['padj']<0.05)&(res['log2FoldChange']< -0.5)).sum())
print('padj non-na', res['padj'].notna().sum())
PY"
