/bin/bash -lc "Rscript - <<'RS'
counts <- read.csv('inputs/counts_raw_unfiltered.csv', check.names=FALSE, row.names=1)
meta <- read.csv('inputs/sample_layout.csv', check.names=FALSE, stringsAsFactors=FALSE)
cat('dimensions', nrow(counts), ncol(counts), '\\n')
cat('unique genes', length(unique(rownames(counts))), 'duplicated', sum(duplicated(rownames(counts))), '\\n')
cat('columns', paste(colnames(counts), collapse=' | '), '\\n')
cat('metadata groups:\\n'); print(table(meta"'$Group))
cat('"'metadata sample IDs', paste(meta"'$SampleID, collapse='"' | '), '\\n')
cat('lib sizes:\\n'); print(round(colSums(counts)))
cat('count class', class(counts[[1]]), 'range', range(as.matrix(counts)), '\\n')
cat('genes with any count', sum(rowSums(counts)>0), '>=10 total', sum(rowSums(counts)>=10), '\\n')
RS"
