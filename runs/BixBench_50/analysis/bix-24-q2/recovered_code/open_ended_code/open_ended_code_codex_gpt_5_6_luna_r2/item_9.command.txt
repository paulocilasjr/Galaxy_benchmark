/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from pathlib import Path
c=Path('inputs/counts_raw_unfiltered.csv')
m=Path('inputs/sample_layout.csv')
counts=pd.read_csv(c)
meta=pd.read_csv(m)
print('counts shape', counts.shape)
print('counts columns first/last', counts.columns[:5].tolist(), counts.columns[-5:].tolist())
print('counts id col', counts.columns[0], 'unique', counts.iloc[:,0].nunique(), 'dups', counts.iloc[:,0].duplicated().sum())
print('count dtypes', counts.dtypes.iloc[1:].value_counts().to_dict())
print('meta shape', meta.shape)
print('groups')
print(meta.groupby('Group', sort=False)['SampleID'].agg(list).to_string())
print('sample ids match after hyphen/underscore:', set(meta.SampleID.str.replace('-','_',regex=False)) == set(counts.columns[1:]))
print('missing metadata cols', sorted(set(counts.columns[1:])-set(meta.SampleID.str.replace('-','_',regex=False))))
print('missing count cols', sorted(set(meta.SampleID.str.replace('-','_',regex=False))-set(counts.columns[1:])))
print('zero total genes', (counts.iloc[:,1:].sum(axis=1)==0).sum(), 'total genes', len(counts))
print('library sizes')
print(pd.DataFrame({'sample':counts.columns[1:], 'libsize':counts.iloc[:,1:].sum(axis=0).values}).to_string(index=False))
PY"
