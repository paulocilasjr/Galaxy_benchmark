/bin/bash -lc "PYTHONPATH=/workspace/tmp/pylib python -u - <<'PY'
import pandas as pd, traceback, sys
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
from gprofiler import GProfiler
try:
 counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
 meta = pd.read_csv('inputs/sample_layout.csv')
 meta['count_id'] = meta['SampleID'].str.replace('-', '_', regex=False)
 meta2 = meta[meta['Group'].isin(['DMSO','CBD_IC50'])].set_index('count_id')
 meta2 = meta2.loc[counts.columns.intersection(meta2.index)]
 meta2['condition'] = pd.Categorical(meta2['Group'], categories=['DMSO','CBD_IC50'])
 cts = counts.loc[:, meta2.index].astype('int64')
 cts = cts.loc[cts.sum(axis=1)>0]
 print('fitting', cts.shape, flush=True)
 dds = DeseqDataSet(counts=cts.T, metadata=meta2[['condition']], design='~condition', refit_cooks=True, n_cpus=1, quiet=True)
 dds.deseq2()
 stat = DeseqStats(dds, contrast=['condition','CBD_IC50','DMSO'], alpha=0.05, cooks_filter=True, independent_filter=True, quiet=True, n_cpus=1)
 stat.summary()
 res = stat.results_df
 sig = res[(res['padj'] < 0.05) & (res['log2FoldChange'].abs() > 0.5)].copy()
 up = sig.index[sig.log2FoldChange > 0].astype(str).tolist()
 down = sig.index[sig.log2FoldChange < 0].astype(str).tolist()
 bg = res.index.astype(str).tolist()
 print('DE counts', len(sig), len(up), len(down), 'background',len(bg), flush=True)
 gp = GProfiler(return_dataframe=True)
 outs={}
 for direction, genes in [('UP',up),('DOWN',down)]:
  print('enrich', direction, flush=True)
  tab=gp.profile(organism='hsapiens', query=genes, background=bg, sources=['GO:BP'], user_threshold=0.05, no_evidences=False, all_results=True)
  outs[direction]=tab
  print(direction, 'rows',len(tab), 'sig',int((tab['p_value']<0.05).sum()) if len(tab) else 0, flush=True)
  if len(tab):
   siggo=tab[tab.p_value<0.05].sort_values('p_value')
   cols=['native','name','p_value','intersection_size','term_size','effective_domain_size','intersection']
   print(siggo[cols].head(80).to_string(index=False), flush=True)
   met=siggo[siggo['name'].str.contains('metabol|energy|oxidative phosphorylation|respirat|lipid|amino acid|carbohydrate|nucleotide|mitochond',case=False,regex=True)]
   print('\\nMETABOLIC-LIKE', direction, len(met), flush=True)
   print(met[cols].head(80).to_string(index=False), flush=True)
except Exception:
 traceback.print_exc(); sys.exit(1)
PY"
