/bin/bash -lc "PYTHONPATH=/tmp/cbd_py_lib python - <<'PY'
import pandas as pd, numpy as np, warnings, os
warnings.filterwarnings('ignore')
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
from gprofiler import GProfiler

counts=pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
meta0=pd.read_csv('inputs/sample_layout.csv').set_index('SampleID')
meta=meta0.loc[[x.replace('_','-') for x in counts.columns]]
treat,control='CBD_IC50','DMSO'
samples=meta.index[meta.Group.isin([treat,control])]
cols=[x for x in counts.columns if x.replace('_','-') in set(samples)]
c0=counts[cols]
keep=(c0>=10).sum(axis=1)>=3
c=c0.loc[keep]
m=meta.loc[[x.replace('_','-') for x in cols]].copy(); m.index=cols
m['condition']=pd.Categorical(m['Group'], categories=[control,treat])
print('fitting',c.shape,flush=True)
dds=DeseqDataSet(counts=c.T, metadata=m[['condition']], design='~condition', refit_cooks=True, n_cpus=2, quiet=True)
dds.deseq2()
stat=DeseqStats(dds, contrast=['condition',treat,control], alpha=0.05, cooks_filter=True, independent_filter=True, quiet=True, n_cpus=2)
stat.summary(); r=stat.results_df.copy()
r.to_csv('/tmp/cbd_primary_deseq.csv')
sig=r[(r.padj<0.05)&(abs(r.log2FoldChange)>0.5)]
print('DE',len(sig),'up',(sig.log2FoldChange>0).sum(),'down',(sig.log2FoldChange<0).sum(),flush=True)
universe=r.index[r.pvalue.notna()].tolist()
up=sig.index[sig.log2FoldChange>0].tolist(); down=sig.index[sig.log2FoldChange<0].tolist()
gp=GProfiler(return_dataframe=True)
for label,genes in [('UP',up),('DOWN',down)]:
    print('\\nENRICH',label,'query',len(genes),'background',len(universe),flush=True)
    out=gp.profile(query=genes, organism='hsapiens', sources=['GO:BP'], user_threshold=0.05, all_results=True, ordered=False, no_evidences=True, background=universe)
    if out is None or len(out)==0:
        print('NO RESULTS'); continue
    out=out.sort_values('p_value')
    out.to_csv('/tmp/cbd_primary_'+label.lower()+'_go.csv',index=False)
    print('terms',len(out),'sig',int(out.significant.sum()) if 'significant' in out else 'NA')
    print(out[['native','name','p_value','significant','intersection_size','term_size','effective_domain_size']].head(30).to_string(index=False))
    keys=['metabol','catabol','biosynth','lipid','fatty','glucose','carbohyd','amino acid','nucleotide','respirat','oxidative','mitochond','energy','organic acid','small molecule']
    mask=out.name.str.lower().apply(lambda s:any(k in s for k in keys))
    print('\\nMETABOLIC-MATCHING')
    print(out.loc[mask,['native','name','p_value','significant','intersection_size','term_size']].head(50).to_string(index=False))
PY"
