import pandas as pd, numpy as np
counts=pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
meta=pd.read_csv('inputs/sample_layout.csv')
print('shape',counts.shape)
print('index unique',counts.index.is_unique,'duplicates',counts.index.duplicated().sum())
print('columns',list(counts.columns))
print('metadata',meta.shape)
print(meta.groupby('Group',sort=False).size().to_string())
print('all count integer', np.all(np.isfinite(counts.to_numpy())) and np.all(counts.to_numpy()>=0) and np.all(counts.to_numpy()==np.floor(counts.to_numpy())))
libs=counts.sum(axis=0)
print('library sizes')
print(libs.to_string())
print('library sizes by group')
print(pd.DataFrame({'lib':libs.values,'group':meta.Group.values}).groupby('group',sort=False).lib.agg(['min','median','max']))
print('genes with >0 counts:',(counts>0).any(axis=1).sum(),'genes >=10 total:',(counts.sum(axis=1)>=10).sum(),'genes >=10 in >=3 samples:',(counts>=10).sum(axis=1).ge(3).sum())
PY"
