/bin/bash -c "Rscript - <<'RS'
suppressPackageStartupMessages({library(DESeq2)})
counts <- read.csv('/workspace/inputs/counts_raw_unfiltered.csv', check.names=FALSE, row.names=1)
layout <- read.csv('/workspace/inputs/sample_layout.csv', stringsAsFactors=FALSE)
layout"'$SampleID <- gsub('"'-','_',layout"'$SampleID)
layout <- layout[layout$Group %in% c('"'DMSO','CBD_IC50'),]
layout <- layout[match(colnames(counts), layout"'$SampleID),]; layout <- layout[!is.na(layout$SampleID),]
layout$Group <- factor(layout$Group, levels=c('"'DMSO','CBD_IC50'))
cts <- counts[, layout"'$SampleID]
for (min_total in c(0,5,10,15)) {
  keep <- if(min_total==0) rep(TRUE,nrow(cts)) else rowSums(cts)>=min_total
  c2 <- cts[keep,]
  dds <- DESeqDataSetFromMatrix(as.matrix(c2), data.frame(row.names=layout$SampleID, Group=layout$Group), ~Group)
  dds <- DESeq(dds, quiet=TRUE)
  res <- results(dds, alpha=0.05)
  cat('"'min_total',min_total,'rows',nrow(c2),'nonzero',sum(rowSums(c2)>0),'nonNApadj',sum("'!is.na(res$padj)),'"'up',sum("'!is.na(res$padj)&res$padj<0.05&res$log2FoldChange>0.5),'"'down',sum("'!is.na(res$padj)&res$padj<0.05&res$log2FoldChange<(-0.5)),'"'\\n')
}
RS"
