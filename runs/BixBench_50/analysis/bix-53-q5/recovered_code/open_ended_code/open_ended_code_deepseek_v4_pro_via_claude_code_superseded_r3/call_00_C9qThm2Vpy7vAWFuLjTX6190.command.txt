from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
import pandas as pd

counts = pd.read_csv('inputs/count.csv', index_col=0)
counts_t = counts.T
metadata = pd.DataFrame({'condition': ['KL','KL','KL','WL','WL','WL']}, index=counts_t.index)

dds = DeseqDataSet(counts=counts_t, metadata=metadata, design='~condition')
dds.deseq2()

stat_res = DeseqStats(dds, contrast=['condition', 'KL', 'WL'])
stat_res.summary()

# Check available attributes/methods
print('Results columns:', stat_res.results_df.columns.tolist())
print('LFC column data type and sample:')
print(stat_res.results_df['log2FoldChange'].head())

# Try to figure out the coefficient
import inspect
sig = inspect.signature(stat_res.lfc_shrink)
print('lfc_shrink params:', list(sig.parameters.keys()))
