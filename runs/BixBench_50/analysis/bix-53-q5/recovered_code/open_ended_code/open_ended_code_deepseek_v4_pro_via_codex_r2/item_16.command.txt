/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time, warnings
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw = pd.read_csv('inputs/count.csv', index_col=0)
counts = raw.T.astype(int)
metadata = pd.DataFrame({'condition':['KL','KL','KL','WL','WL','WL']}, index=counts.index)
dds = DeseqDataSet(counts=counts, metadata=metadata, design='~condition', quiet=True)
dds.deseq2()
print('dds done')
print(dds.obs[['size_factors']] if 'size_factors' in dds.obs else 'no obs sf')
st = DeseqStats(dds, contrast=['condition','WL','KL'], alpha=0.05, quiet=True)
st.summary()
print('summary columns', st.results_df.columns.tolist())
st.lfc_shrink(coeff='condition_WL_vs_KL')
res = st.results_df.copy()
res.to_csv('/workspace/tmp/deseq_results.csv')
print('saved', res.shape)
print(res[['baseMean','log2FoldChange','pvalue','padj']].describe())
for pcol in ['pvalue','padj']:
 mask = (res[pcol] < 0.05) & (res['log2FoldChange'].abs()>1) & (res['baseMean']>10)
 print(pcol, mask.sum())
PY"
