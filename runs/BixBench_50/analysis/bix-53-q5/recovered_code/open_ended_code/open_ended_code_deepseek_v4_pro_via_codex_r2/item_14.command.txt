/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts = pd.read_csv('inputs/count.csv', index_col=0)
print(counts.shape, counts.index[:3].tolist(), counts.columns.tolist())
counts.columns = [c.strip() for c in counts.columns]
# ensure int
counts = counts.astype(int)
metadata = pd.DataFrame({
 'condition': ['KL','KL','KL','WL','WL','WL']
}, index=counts.columns)
print(metadata)
dds = DeseqDataSet(counts=counts, metadata=metadata, design='~condition', quiet=False)
dds.deseq2()
print('dds done', dds.obsm['size_factors'])
st = DeseqStats(dds, contrast=['condition','WL','KL'], alpha=0.05, quiet=False)
st.summary()
print(st.results_df.head())
print(st.results_df.columns)
st.lfc_shrink(coeff='condition_WL_vs_KL')
print('shrink done')
res = st.results_df.copy()
print(res.columns)
res.to_csv('/workspace/tmp/deseq_results.csv')
print(res.shape)
for col in ['baseMean','log2FoldChange','pvalue','padj']:
 print(col, res[col].describe())
for pcol in ['pvalue','padj']:
 mask = (res[pcol] < 0.05) & (res['log2FoldChange'].abs()>1) & (res['baseMean']>10)
 print(pcol, mask.sum())
PY"
