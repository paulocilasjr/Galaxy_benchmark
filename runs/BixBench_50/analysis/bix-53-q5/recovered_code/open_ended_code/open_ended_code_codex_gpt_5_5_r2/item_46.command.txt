/bin/bash -lc "python - <<'PY'
import gseapy as gp
for genes in [['Ccnd2','Slfn4','Itgb2l'], ['CCND2','SLFN4','ITGB2L']]:
 try:
  enr=gp.enrichr(gene_list=genes, gene_sets='WikiPathways_2019_Mouse', organism='Mouse', outdir=None, no_plot=True, verbose=False)
  print('genes', genes, 'rows', 0 if enr.res2d is None else len(enr.res2d))
  if enr.res2d is not None and len(enr.res2d): print(enr.res2d.head(3).to_string())
 except Exception as e:
  print('ERR', genes, type(e).__name__, str(e)[:400])
PY"
