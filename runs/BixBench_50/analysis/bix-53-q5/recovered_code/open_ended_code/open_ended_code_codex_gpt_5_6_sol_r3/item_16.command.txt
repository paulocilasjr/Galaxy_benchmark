/bin/bash -lc "OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
import mygene, gseapy as gp
x=pd.read_csv('inputs/count.csv',index_col=0).T
meta=pd.DataFrame({'condition':['KL']*3+['WL']*3},index=x.index)
dds=DeseqDataSet(counts=x,metadata=meta,design='~condition',n_cpus=1,quiet=True)
dds.deseq2()
st=DeseqStats(dds,contrast=['condition','WL','KL'],alpha=.05,n_cpus=1,quiet=True)
st.summary(); st.lfc_shrink(coeff='condition[T.WL]')
res=st.results_df
# Query union of IDs needed by either interpretation
needed=res[(res.pvalue<.05)&(res.log2FoldChange.abs()>1)&(res.baseMean>10)].index.tolist()
print('need map',len(needed),flush=True)
mg=mygene.MyGeneInfo()
hits=mg.querymany(needed,scopes='ensembl.gene',fields='symbol',species='mouse',as_dataframe=False,verbose=False)
print('mapping responses',len(hits), 'notfound',sum(bool(h.get('notfound')) for h in hits),flush=True)
mapd={h['query']:h['symbol'] for h in hits if 'symbol' in h and isinstance(h['symbol'],str)}
print('mapped unique ids',len(mapd),'unique symbols',len(set(mapd.values())),flush=True)
for pcol in ['pvalue','padj']:
 sig=res[(res[pcol]<.05)&(res.log2FoldChange.abs()>1)&(res.baseMean>10)]
 syms=list(dict.fromkeys(mapd[g] for g in sig.index if g in mapd))
 print('\\nFILTER',pcol,'genes',len(sig),'symbols',len(syms),flush=True)
 enr=gp.enrichr(gene_list=syms,gene_sets='WikiPathways_2019_Mouse',organism='mouse',outdir=None,no_plot=True,verbose=False)
 tab=enr.results.sort_values(['Adjusted P-value','P-value'],kind='stable').head(20)
 print(tab[['Term','P-value','Adjusted P-value','Overlap','Genes']].to_string(index=False),flush=True)
 ox=tab.Term.str.contains('oxidative',case=False,regex=False)
 print('oxidative terms:',tab.loc[ox,'Term'].tolist(), 'count',int(ox.sum()),'fraction',round(float(ox.sum())/20,1),flush=True)
PY"
