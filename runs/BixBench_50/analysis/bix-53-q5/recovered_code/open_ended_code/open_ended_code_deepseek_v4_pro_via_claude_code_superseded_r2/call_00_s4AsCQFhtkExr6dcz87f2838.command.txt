import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts = pd.read_csv('inputs/count.csv', index_col=0)
counts = counts[counts.sum(axis=1) > 0]
counts_t = counts.T.astype(int)
metadata = pd.DataFrame({'condition': ['KL','KL','KL','WL','WL','WL']}, index=counts.columns)

dds = DeseqDataSet(counts=counts_t, metadata=metadata, design='~condition')
dds.deseq2()
stat_res = DeseqStats(dds, contrast=['condition', 'KL', 'WL'])
stat_res.summary()
# Check LFC columns
print('LFC columns:', stat_res.LFCs.columns.tolist())
print('contrast_idx:', stat_res.contrast_idx)
print('LFCs shape:', stat_res.LFCs.shape)
print('LFCs:', stat_res.LFCs.head())
