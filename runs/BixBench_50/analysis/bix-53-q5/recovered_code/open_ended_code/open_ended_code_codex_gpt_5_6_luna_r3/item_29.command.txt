/bin/bash -lc "python -u - <<'PY'
import pandas as pd, numpy as np, time, sys
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
raw=pd.read_csv('inputs/count.csv')
counts=raw.set_index('geneID').T
metadata=pd.DataFrame({'condition':pd.Categorical(['KL']*3+['WL']*3,categories=['WL','KL'])},index=counts.index)
print('init',counts.shape,flush=True); t=time.time()
dds=DeseqDataSet(counts=counts,metadata=metadata,design='~condition',quiet=True,n_cpus=1,low_memory=True)
print('deseq2 init',time.time()-t,flush=True)
dds.deseq2()
print('done dds',time.time()-t,flush=True)
stat=DeseqStats(dds,contrast=['condition','KL','WL'],quiet=True,n_cpus=1)
print('summary',time.time()-t,flush=True)
stat.summary()
print('done stats',stat.results_df.shape,time.time()-t,flush=True)
stat.lfc_shrink(coeff='condition[T.KL]',adapt=True)
print('done shrink',time.time()-t,flush=True)
res=stat.results_df
sel=res['pvalue'].lt(.05)&res['log2FoldChange'].abs().gt(1)&res['baseMean'].gt(10)
print('sel',int(sel.sum()),'p',int(res.pvalue.lt(.05).sum()),'lfc',int(res.log2FoldChange.abs().gt(1).sum()),'base',int(res.baseMean.gt(10).sum()),flush=True)
res.to_csv('/tmp/pydeseq2_res.csv')
PY"
