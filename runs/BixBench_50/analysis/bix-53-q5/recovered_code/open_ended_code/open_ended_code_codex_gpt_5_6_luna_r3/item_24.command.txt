/bin/bash -lc "python -u - <<'PY'
import pandas as pd, numpy as np, time, sys
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
raw=pd.read_csv('inputs/count.csv').iloc[:1000]
counts=raw.set_index('geneID').T
metadata=pd.DataFrame({'condition':pd.Categorical(['KL']*3+['WL']*3,categories=['WL','KL'])},index=counts.index)
print('init',flush=True)
dds=DeseqDataSet(counts=counts,metadata=metadata,design='~condition',quiet=True,n_cpus=1)
print('deseq2',flush=True)
dds.deseq2()
print('done dds',flush=True)
stat=DeseqStats(dds,contrast=['condition','KL','WL'],quiet=True,n_cpus=1)
print('summary',flush=True)
stat.summary()
print('done stats',stat.results_df.shape,flush=True)
print(stat.LFC.columns,flush=True)
PY"
