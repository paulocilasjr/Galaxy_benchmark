/bin/bash -c "Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2))
cat('apeglm loaded', requireNamespace('apeglm', quietly=TRUE), '\\n')
counts <- read.csv('/workspace/inputs/count.csv', row.names=1, check.names=FALSE)
counts <- as.matrix(counts); mode(counts) <- 'integer'
samples <- data.frame(row.names=colnames(counts), condition=factor(c('KL','KL','KL','WL','WL','WL'), levels=c('KL','WL')))
dds <- DESeqDataSetFromMatrix(counts, samples, ~condition)
dds <- DESeq(dds, quiet=TRUE)
res <- lfcShrink(dds, contrast=c('condition','WL','KL'), type='apeglm', quiet=TRUE)
res <- as.data.frame(res); res"'$geneID <- rownames(res)
write.csv(res, '"'/workspace/tmp/deseq2_results_apeglm.csv', row.names=FALSE)
sig <- res["'!is.na(res$padj) & res$padj<0.05 & !is.na(res$log2FoldChange) & abs(res$log2FoldChange)>1 & res$baseMean>10,]
write.table(sig[,c('"'geneID','baseMean','log2FoldChange','padj')], '/workspace/tmp/sig_genes_apeglm.tsv', sep='\\t', quote=FALSE, row.names=FALSE)
cat('apeglm sig count', nrow(sig), '\\n')
RS
/opt/agent-venv/bin/python - <<'PY'
import pandas as pd, mygene, gseapy as gp
sig=pd.read_csv('/workspace/tmp/sig_genes_apeglm.tsv', sep='\\t')
ids=sig['geneID'].tolist(); mg=mygene.MyGeneInfo(); out=mg.querymany(ids, scopes='ensembl.gene', fields='symbol,name', species='mouse', verbose=False)
rows=[]
for item in out:
 q=item.get('query'); sym=item.get('symbol')
 if isinstance(sym,list): sym=sym[0] if sym else None
 if q: rows.append((q,sym))
ann=pd.DataFrame(rows,columns=['geneID','symbol']).drop_duplicates('geneID')
m=sig.merge(ann,on='geneID',how='left').dropna(subset=['symbol']).sort_values('padj').drop_duplicates('symbol')
print('mapped unique', len(m))
genes=m['symbol'].tolist()
enr=gp.enrichr(gene_list=genes, gene_sets='WikiPathways_2019_Mouse', organism='mouse', outdir=None, cutoff=1)
res=enr.results
res.to_csv('/workspace/tmp/enrichr_apeglm.tsv', sep='\\t', index=False)
top20=res.head(20)
print('top20 oxidative', top20['Term'].str.lower().str.contains('oxidative').sum(), 'fraction', top20['Term'].str.lower().str.contains('oxidative').sum()/20)
print(top20[['Term','P-value']].to_string(index=False))
PY"
