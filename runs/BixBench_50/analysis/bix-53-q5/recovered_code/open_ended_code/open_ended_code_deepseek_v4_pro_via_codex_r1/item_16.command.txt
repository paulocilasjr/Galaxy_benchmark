/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, json, os
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts = pd.read_csv('/workspace/inputs/count.csv', index_col=0)
print(counts.shape)
meta = pd.DataFrame({'condition':['KL']*3 + ['WL']*3}, index=counts.columns)
print(meta)
# set reference level KL
# In pydeseq2, design factors; reference_level param?
try:
    dds = DeseqDataSet(counts=counts, metadata=meta, design='~condition', reference_level=['condition','KL'])
except Exception as e:
    print('err with reference',e)
    dds = DeseqDataSet(counts=counts, metadata=meta, design='~condition')
print('obsm', dds.obsm)
print('obs', dds.obs)
print('design factors', dds.obsm.get('design_matrix').columns.tolist())
dds.deseq2()
stat = DeseqStats(dds)
stat.summary()
print(stat.results_df.head())
print(stat.results_df.columns)
stat.lfc_shrink(coeff='condition_WL_vs_KL')
res = stat.results_df
print(res.head())
print(res.columns)
print('sig raw p, abs lfc >1, baseMean >10:', ((res.pvalue<0.05)&(res.baseMean>10)&(res.log2FoldChange.abs()>1)).sum())
res.to_csv('/workspace/tmp/pydeseq2_results.csv')
PY"
