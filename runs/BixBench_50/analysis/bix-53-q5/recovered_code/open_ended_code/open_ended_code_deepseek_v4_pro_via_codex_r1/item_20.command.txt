/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, os, warnings, time
warnings.filterwarnings('ignore')
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw = pd.read_csv('/workspace/inputs/count.csv', index_col=0)
counts = raw.T.astype(int)
meta = pd.DataFrame({'condition':['KL']*3 + ['WL']*3}, index=counts.index)
dds = DeseqDataSet(counts=counts, metadata=meta, design='~condition', ref_level=['condition','KL'], quiet=True)
dds.deseq2()
print('design cols', dds.obsm['design_matrix'].columns.tolist())
stat = DeseqStats(dds, contrast=['condition','WL','KL'], alpha=0.05, quiet=True)
stat.summary()
print('pre columns', stat.results_df.columns)
stat.lfc_shrink(coeff='condition_WL_vs_KL')
res = stat.results_df
print('post columns', res.columns)
print(res[['baseMean','log2FoldChange','pvalue','padj']].head())
mask = (res.pvalue<0.05)&(res.baseMean>10)&(res.log2FoldChange.abs()>1)
print('sig genes', mask.sum())
print('padj sig 0.05 same mask?', ((res.padj<0.05)&(res.baseMean>10)&(res.log2FoldChange.abs()>1)).sum())
res.to_csv('/workspace/tmp/pydeseq2_results.csv')
PY"
