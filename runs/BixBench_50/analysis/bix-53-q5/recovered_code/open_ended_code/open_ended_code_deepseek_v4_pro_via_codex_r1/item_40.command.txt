/bin/bash -lc "python - <<'PY'
import pandas as pd, mygene, gseapy, warnings
warnings.filterwarnings('ignore')
res=pd.read_csv('/workspace/tmp/pydeseq2_results.csv',index_col=0)
for pcol in ['pvalue','padj']:
 mask=(res[pcol]<0.05)&(res.baseMean>10)&(res.log2FoldChange.abs()>1)
 genes=res.index[mask].tolist()
 mg=mygene.MyGeneInfo(); out=mg.querymany(genes, scopes='ensembl.gene', fields='symbol', species='mouse', verbose=False)
 sym=[]
 for g in genes:
  rs=[r for r in out if r.get('query')==g and 'symbol' in r]
  if rs: sym.append(rs[0]['symbol'].upper())
 uniq=[]; seen=set()
 for s in sym:
  if s not in seen: seen.add(s); uniq.append(s)
 print('\\n',pcol,len(genes),'mapped',len(uniq))
 enr=gseapy.enrichr(gene_list=uniq, gene_sets='WikiPathways_2019_Mouse', organism='mouse', no_plot=True, verbose=False)
 r=enr.results
 top=r.head(20)
 print('top20 oxidative', top.Term.str.contains('oxidative',case=False).sum())
 print(top[['Term','P-value']].to_string(index=False))
PY"
