import dendropy

tree_str = "(Fungi_Ncra:0.4916927411,(Fungi_Scer:0.4216513655,Fungi_Spom:0.4407010522):0.1387171356,Fungi_Umay:1.0828368619);"
tree = dendropy.Tree.get(data=tree_str, schema="newick")

# Use Taxon objects directly
taxon_objects = list(tree.taxon_namespace)
taxa = [t.label for t in taxon_objects]
print("Taxa:", taxa)

# Compute avg_dist (over all pairs, excluding self)
pdm = tree.phylogenetic_distance_matrix()
all_pair_dists = []
for i, t1 in enumerate(taxon_objects):
    for j, t2 in enumerate(taxon_objects):
        if i < j:
            d = pdm(t1, t2)
            all_pair_dists.append(d)
            print(f"  {t1.label} <-> {t2.label}: {d:.10f}")

tip_count = len(taxon_objects)
avg_dist = sum(all_pair_dists) / len(all_pair_dists)
print(f"\navg_dist (grand mean): {avg_dist:.10f}")
print(f"num pairs: {len(all_pair_dists)}")

# Now compute avg_PDis with the PhyKIT "bug" - denominator = tip_count (not tip_count-1)
# The bug means set(tip) creates set of chars, so tip_set - set(chars) = tip_set
# Therefore denominator = tip_count
for taxon_obj in taxon_objects:
    tip = taxon_obj.label
    distances_to_all = []
    for other_obj in taxon_objects:
        d = pdm(taxon_obj, other_obj)
        distances_to_all.append(d)
    print(f"\nDistances from {tip} to all: {[round(d, 10) for d in distances_to_all]}")
    sum_dist = sum(distances_to_all)  # includes 0 for self
    # PhyKIT denominator with bug: tip_count (not tip_count-1)
    avg_PDi = sum_dist / tip_count  # bug: divides by tip_count instead of (tip_count - 1)
    print(f"  Sum: {sum_dist:.10f}, Denominator: {tip_count}, avg_PDi: {avg_PDi:.10f}")
    
    # Compare to correct avg_PDi (excluding self)
    correct_avg_PDi = sum_dist / (tip_count - 1)
    print(f"  Correct avg_PDi (exclude self): {correct_avg_PDi:.10f}")

    # Now apply phykit formula
    scale = 100.0 / avg_dist
    lb_phykit = (avg_PDi * scale) - 100.0
    lb_correct_phykit = (correct_avg_PDi * scale) - 100.0
    print(f"  PhyKIT LB (with bug): {lb_phykit:.4f}")
    print(f"  Correct LB: {lb_correct_phykit:.4f}")

# PhyKIT reported:
# Fungi_Ncra: -29.4826, Fungi_Scer: -32.1541, Fungi_Spom: -31.4275, Fungi_Umay: -6.9357
print("\n=== PhyKIT reported values ===")
