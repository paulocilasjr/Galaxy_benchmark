# Compute patristic distances and LB scores manually
import dendropy

tree_str = "(Fungi_Ncra:0.4916927411,(Fungi_Scer:0.4216513655,Fungi_Spom:0.4407010522):0.1387171356,Fungi_Umay:1.0828368619);"
tree = dendropy.Tree.get(data=tree_str, schema="newick")

taxa = [t.taxon.label for t in tree.taxon_namespace]
print("Taxa:", taxa)

# Compute all pairwise patristic distances
pdm = tree.phylogenetic_distance_matrix()
print("\nPairwise patristic distance matrix:")
for t1 in taxa:
    for t2 in taxa:
        if t1 < t2:
            d = pdm(taxon1=t1, taxon2=t2)
            print(f"  {t1} <-> {t2}: {d}")

# Mean pairwise distance for each taxon
print("\nMean pairwise patristic distance for each taxon:")
mean_per_taxon = {}
all_pairs = []
for t1 in taxa:
    dists = []
    for t2 in taxa:
        if t1 != t2:
            d = pdm(taxon1=t1, taxon2=t2)
            dists.append(d)
            all_pairs.append(d)
    mean_per_taxon[t1] = sum(dists) / len(dists)
    print(f"  {t1}: {mean_per_taxon[t1]:.6f}")

# Grand mean of all pairwise distances
grand_mean = sum(all_pairs) / len(all_pairs)
print(f"\nGrand mean of all pairwise distances: {grand_mean:.6f}")

# LB scores (Struck 2014)
print("\nLB scores (Struck 2014 = mean_i / grand_mean):")
lb_scores = {}
for t in taxa:
    lb = mean_per_taxon[t] / grand_mean
    lb_scores[t] = lb
    print(f"  {t}: {lb:.6f}")

# PhyKIT reports negative numbers - maybe it's log-transformed?
import math
print("\nPhyKIT-style (log2? ln?):")
for t in taxa:
    print(f"  {t}: log2 ratio = {math.log2(lb_scores[t]):.4f}, ln ratio = {math.log(lb_scores[t]):.4f}")

print("\nPhyKIT reported values:")
print("  Fungi_Ncra: -29.4826")
print("  Fungi_Scer: -32.1541")
print("  Fungi_Spom: -31.4275")
print("  Fungi_Umay: -6.9357")

# Let me compute median of the raw Struck LB scores
sorted_scores = sorted(lb_scores.values())
print(f"\nStruck 2014 LB scores (sorted): {sorted_scores}")
n = len(sorted_scores)
median = (sorted_scores[n//2 - 1] + sorted_scores[n//2]) / 2
print(f"Median Struck LB score: {median:.6f}")

# Median of phykit values
phykit_scores = [-32.1541, -31.4275, -29.4826, -6.9357]
phykit_sorted = sorted(phykit_scores)
phykit_median = (phykit_sorted[1] + phykit_sorted[2]) / 2
print(f"Median PhyKIT score: {phykit_median:.4f}")
