/bin/bash -lc "python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
import os
import subprocess
import sys
import tempfile

member = '996662at2759.faa.mafft.clipkit.treefile'
with ZipFile('inputs/scogs_fungi.zip') as zf:
    names = set(zf.namelist())
    tree = zf.read(member)
    alignment = zf.read('996662at2759.faa.mafft.clipkit')
print('tree member present:', member in names)
print('tree:', tree.decode().strip())
print('alignment headers:', [line for line in alignment.decode().splitlines() if line.startswith('>')])
fd, tree_path = tempfile.mkstemp(suffix='.tree')
os.close(fd)
try:
    Path(tree_path).write_bytes(tree)
    result = subprocess.run([sys.executable, '-m', 'phykit', 'long_branch_score', tree_path, '--json'], text=True, capture_output=True, check=True)
    print('json output:')
    print(result.stdout, end='')
finally:
    os.unlink(tree_path)
PY"
