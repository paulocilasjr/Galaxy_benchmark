/bin/bash -lc "python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
import os
import subprocess
import tempfile

archive = Path('inputs/scogs_fungi.zip')
member = '996662at2759.faa.mafft.clipkit.treefile'
with ZipFile(archive) as zf:
    tree_text = zf.read(member)
fd, tree_path = tempfile.mkstemp(suffix='.tree')
os.close(fd)
try:
    Path(tree_path).write_bytes(tree_text)
    for args in (['phykit', 'long_branch_score', tree_path],
                 ['phykit', 'long_branch_score', tree_path, '--verbose']):
        result = subprocess.run(args, text=True, capture_output=True, check=True)
        print(' '.join(args[:-1]) if args[-1] == '--verbose' else 'summary')
        print(result.stdout, end='')
        if result.stderr:
            print(result.stderr, end='')
finally:
    os.unlink(tree_path)
PY"
