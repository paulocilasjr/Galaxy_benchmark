/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_1vs97.rds')
showattrs <- function(obj,label) {
  cat('\\n',label,'typeof=',typeof(obj),' isS4=',isS4(obj),'\\n',sep='')
  a <- attributes(obj)
  cat('attributes=',paste(names(a),collapse=', '),'\\n',sep='')
  for(n in names(a)) cat(' ',n,': typeof=',typeof(a[[n]]),' isS4=',isS4(a[[n]]),'\\n',sep='')
}
showattrs(x,'x')
for(n in names(attributes(x))) showattrs(attr(x,n), paste0('x@',n))
ld <- attr(x,'listData')
cat('\\nlistData inner names attr raw:\\n')
print(attr(ld,'names'))
cat('listData .Data attr? ', typeof(attr(ld,'.Data')), '\\n')
cat('listData listData attr? ', typeof(attr(ld,'listData')), '\\n')
cat('listData unclass attempt:\\n')
y <- unclass(ld)
showattrs(y,'unclass(ld)')
RS"
