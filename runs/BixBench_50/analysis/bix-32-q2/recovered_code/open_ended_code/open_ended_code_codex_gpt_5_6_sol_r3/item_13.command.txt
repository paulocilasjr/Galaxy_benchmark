/bin/bash -lc "Rscript - <<'RS'
read_res <- function(f) {x<-readRDS(f); d<-as.data.frame(attr(x,'listData')); rownames(d)<-attr(x,'rownames'); d}
res<-lapply(c("'`97`='"'inputs/res_1vs97.rds',"'`98`='"'inputs/res_1vs98.rds',"'`99`='"'inputs/res_1vs99.rds'),read_res)
ln<-read.delim('https://rest.kegg.jp/link/pathway/pau',header=F,stringsAsFactors=F); names(ln)<-c('gene','path'); ln"'$gene<-sub('"'pau:','',ln"'$gene);ln$path<-sub('"'path:','',ln"'$path)
ora<-function(fg,u){u<-intersect(unique(u),unique(ln$gene));g<-intersect(fg,u);gs<-lapply(split(ln$gene,ln$path),function(z)intersect(unique(z),u));gs<-gs[lengths(gs)>=10&lengths(gs)<=500];k<-vapply(gs,function(z)sum(z%in%g),integer(1));gs<-gs[k>0];k<-k[k>0];M<-lengths(gs);p<-phyper(k-1,M,length(u)-M,length(g),lower.tail=F);data.frame(path=names(gs),N=length(u),n=length(g),M=M,k=k,p=p,padj=p.adjust(p,'"'BH'))}
uin<-Reduce(intersect,lapply(res,rownames)); ukegg<-unique(ln"'$gene)
for(bg in c('"'input','kegg')) {u<-if(bg=='input')uin else ukegg; cat('\\nBACKGROUND',bg,'\\n');for(s in names(res)){fg<-rownames(res[[s]])[res[[s]]"'$log2FoldChange < -1.5];z<-ora(fg,u); print(z[z$path%in%c('"'pau01110','pau01310','pau04146'),],row.names=F)}}
RS"
