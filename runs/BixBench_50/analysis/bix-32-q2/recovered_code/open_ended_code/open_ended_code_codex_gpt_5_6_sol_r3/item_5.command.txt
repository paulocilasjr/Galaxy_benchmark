/bin/bash -lc "Rscript - <<'RS'
for (f in Sys.glob('inputs/*.rds')) {
  x <- readRDS(f)
  cat('\\nFILE',f,'typeof',typeof(x),'isS4',isS4(x),'\\n')
  a <- attributes(x)
  cat('attr names:',paste(names(a),collapse=', '),'\\n')
  for(n in names(a)) {
    cat('ATTR',n,' typeof=',typeof(a[[n]]),' length=',length(a[[n]]),'\\n',sep='')
    if (n %in% c('class','names','row.names','elementType','packageName')) print(a[[n]])
  }
}
RS"
