/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
files<-c('97'='inputs/res_1vs97.rds','98'='inputs/res_1vs98.rds','99'='inputs/res_1vs99.rds');res<-lapply(files,function(f){x<-readRDS(f);a<-attributes(x);d<-as.data.frame(a"'$listData);rownames(d)<-a$rownames;d})
ln<-readLines('"'https://rest.kegg.jp/link/pathway/pau',warn=F);z<-strsplit(ln,'\\\\t');links<-data.frame(gene=sub('pau:','',sapply(z,"'`[`,1)),path=sub('"'path:','',sapply(z,"'`[`,2)))
ora<-function(d,u){fg<-rownames(d)[d$log2FoldChange< -1.5];lk<-links[links$gene%in%u,];u<-unique(lk$gene);fg<-intersect(fg,u);ss<-lapply(split(lk$gene,lk$path),unique);ss<-ss[lengths(ss)>=10&lengths(ss)<=500];k<-sapply(ss,function(s)sum(fg%in%s));M<-lengths(ss);q<-k>0;p<-phyper(k[q]-1,M[q],length(u)-M[q],length(fg),lower.tail=F);o<-data.frame(path=names(k)[q],k=k[q],M=M[q],n=length(fg),N=length(u),p=p,padj=p.adjust(p,'"'BH'));o[order(o"'$p),]}
shared<-intersect(rownames(res[[1]]),unique(links$gene));kegg<-unique(links$gene);targets<-c('"'pau01110','pau01310','pau04146')
for(ut in c('shared','kegg')){cat('\\n',ut,'\\n');u<-if(ut=='shared')shared else kegg;for(nm in names(res)){o<-ora(res[[nm]],u);print(o[o"'$path%in%targets,]);cat('"'tested hits',nrow(o),'min padj',head(o,10)"'$padj,'"'\\n')}}
cat('\\nMissing mapped genes from results:\\n');print(setdiff(kegg,shared)); print(links[links"'$gene%in%setdiff(kegg,shared),])
RS'
