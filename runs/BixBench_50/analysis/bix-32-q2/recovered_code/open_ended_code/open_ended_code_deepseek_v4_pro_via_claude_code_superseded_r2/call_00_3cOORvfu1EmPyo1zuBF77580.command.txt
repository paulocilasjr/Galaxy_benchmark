import requests
import rdata
import numpy as np
import warnings
warnings.filterwarnings('ignore')

# Load all three datasets
res97 = rdata.read_rda('/workspace/inputs/res_1vs97.rds')
res98 = rdata.read_rda('/workspace/inputs/res_1vs98.rds')
res99 = rdata.read_rda('/workspace/inputs/res_1vs99.rds')

# Check if all three have the same genes
print('Same genes across all three:', 
      np.array_equal(res97.rownames, res98.rownames) and 
      np.array_equal(res97.rownames, res99.rownames))
print('Number of genes:', len(res97.rownames))

# Get KEGG pathway-gene links for pau
r = requests.get('https://rest.kegg.jp/link/pau/pathway', timeout=15)
links = r.text.strip().split('\n')

# Parse: path:pauXXXXX\tpau:PA14_XXXXX
kegg_genes = set()
for line in links:
    parts = line.strip().split('\t')
    if len(parts) == 2:
        gene = parts[1].replace('pau:', '')
        kegg_genes.add(gene)

# Our genes
our_genes = set(str(g) for g in res97.rownames)

print(f'KEGG genes in pau: {len(kegg_genes)}')
print(f'Our genes: {len(our_genes)}')
print(f'Overlap: {len(our_genes & kegg_genes)}')
print(f'Our genes not in KEGG: {len(our_genes - kegg_genes)}')
print(f'KEGG genes not in our data: {len(kegg_genes - our_genes)}')
