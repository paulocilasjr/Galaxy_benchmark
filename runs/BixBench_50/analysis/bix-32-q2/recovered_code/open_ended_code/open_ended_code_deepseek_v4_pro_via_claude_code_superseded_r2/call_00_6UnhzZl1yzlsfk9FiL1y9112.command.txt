import requests

# Check KEGG organism codes for Pseudomonas
# Try pau (PA14)
for org in ['pau', 'pae', 'paf']:
    url = f'https://rest.kegg.jp/list/{org}'
    try:
        r = requests.get(url, timeout=15)
        if r.status_code == 200 and len(r.text) > 0:
            lines = r.text.strip().split('\n')
            print(f'{org}: {len(lines)} genes, first: {lines[0][:100]}')
        else:
            print(f'{org}: status={r.status_code}, empty or not found')
    except Exception as e:
        print(f'{org}: {e}')
