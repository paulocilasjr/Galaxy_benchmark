import rdata, numpy as np, requests, warnings
from scipy.stats import hypergeom
from statsmodels.stats.multitest import multipletests
warnings.filterwarnings('ignore')

res97 = rdata.read_rda('/workspace/inputs/res_1vs97.rds')
res98 = rdata.read_rda('/workspace/inputs/res_1vs98.rds')
res99 = rdata.read_rda('/workspace/inputs/res_1vs99.rds')

def get_results(res):
    return {'genes': [str(g) for g in res.rownames],
            'lfc': res.listData['log2FoldChange']}

data97 = get_results(res97)
data98 = get_results(res98)
data99 = get_results(res99)

r_link = requests.get('https://rest.kegg.jp/link/pau/pathway', timeout=30)
links = r_link.text.strip().split('\n')
pathway_genes = {}
for line in links:
    parts = line.strip().split('\t')
    if len(parts) == 2:
        pid = parts[0].replace('path:', '')
        gene = parts[1].replace('pau:', '')
        pathway_genes.setdefault(pid, set()).add(gene)

r_list = requests.get('https://rest.kegg.jp/list/pathway/pau', timeout=30)
pathway_names = {}
for line in r_list.text.strip().split('\n'):
    parts = line.split('\t')
    if len(parts) >= 2:
        pathway_names[parts[0].replace('path:', '')] = parts[1]

all_genes_set = set(str(g) for g in res97.rownames)
kegg_mapped = set()
for gs in pathway_genes.values():
    kegg_mapped |= gs
universe = kegg_mapped & all_genes_set

for pid in pathway_genes:
    pathway_genes[pid] = pathway_genes[pid] & universe
pathway_genes = {pid: gs for pid, gs in pathway_genes.items() if len(gs) >= 5}

N = len(universe)
all_pids = sorted(pathway_genes.keys())

# Store all pvalues
all_pvals = {}
for label, data in [('97', data97), ('98', data98), ('99', data99)]:
    for direction in ['up', 'down']:
        genes_arr = np.array(data['genes'])
        lfc = data['lfc']
        if direction == 'up':
            mask = lfc > 1.5
        else:
            mask = lfc < -1.5
        fg = set(genes_arr[mask]) & universe
        K = len(fg)
        
        pvals_vec = []
        for pid in all_pids:
            pw_set = pathway_genes[pid]
            n = len(pw_set)
            k = len(fg & pw_set)
            if k == 0:
                pv = 1.0
            else:
                pv = hypergeom.sf(k - 1, N, n, K)
            pvals_vec.append(pv)
            all_pvals[(label, direction, pid)] = {'pval': pv, 'k': k, 'n': n, 'K': K}
        
        _, padjs, _, _ = multipletests(pvals_vec, method='fdr_bh')
        for i, pid in enumerate(all_pids):
            all_pvals[(label, direction, pid)]['padj'] = padjs[i]

# Show pathways with padj < 0.1 in all three for same direction
for direction in ['up', 'down']:
    print(f"\n=== {direction.upper()} - pathways with padj < 0.1 in all 3 ===")
    found = False
    for pid in all_pids:
        p97 = all_pvals[('97', direction, pid)]['padj']
        p98 = all_pvals[('98', direction, pid)]['padj']
        p99 = all_pvals[('99', direction, pid)]['padj']
        if max(p97, p98, p99) < 0.1:
            found = True
            name = pathway_names.get(pid, '')
            k97 = all_pvals[('97', direction, pid)]['k']
            k98 = all_pvals[('98', direction, pid)]['k']
            k99 = all_pvals[('99', direction, pid)]['k']
            n = all_pvals[('97', direction, pid)]['n']
            print(f"  {pid}: padj 97={p97:.4f}, 98={p98:.4f}, 99={p99:.4f}")
            print(f"    k/n: 97={k97}/{n}, 98={k98}/{n}, 99={k99}/{n}  ({name})")
    if not found:
        print("  (none)")

# Also show any pathway with padj < 0.05 in at least 2 of 3
print(f"\n=== Pathways significant (padj<0.05) in >=2 of 3 strains ===")
for direction in ['up', 'down']:
    for pid in all_pids:
        p97 = all_pvals[('97', direction, pid)]['padj']
        p98 = all_pvals[('98', direction, pid)]['padj']
        p99 = all_pvals[('99', direction, pid)]['padj']
        sig_count = sum(1 for p in [p97, p98, p99] if p < 0.05)
        if sig_count >= 2:
            name = pathway_names.get(pid, '')
            print(f"  {direction} {pid}: sig_in={sig_count}/3, 97={p97:.4f}, 98={p98:.4f}, 99={p99:.4f} ({name})")
