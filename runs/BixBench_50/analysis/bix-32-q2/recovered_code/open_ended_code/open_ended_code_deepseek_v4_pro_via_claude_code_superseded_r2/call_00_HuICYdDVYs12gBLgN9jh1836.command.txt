import requests

# Check KEGG pathway list for pau
r = requests.get('https://rest.kegg.jp/list/pathway/pau', timeout=15)
paths = r.text.strip().split('\n')
print(f'Total pathways: {len(paths)}')
for p in paths[:5]:
    print(p)
print('...')

# Check a specific pathway's genes
r2 = requests.get('https://rest.kegg.jp/link/pau/pathway', timeout=15)
links = r2.text.strip().split('\n')
print(f'\nTotal pathway-gene links: {len(links)}')
for l in links[:5]:
    print(l)
