/bin/bash -lc "Rscript - <<'RS'
parse <- function(f){a<-attributes(readRDS(f));list(g=a"'$rownames,l=a$listData$log2FoldChange,p=a$listData$padj)}
rr<-lapply(list.files("inputs",pattern="''^res_.*'"\\\\.rds"'$",full.names=TRUE),parse)
sp<-strsplit(readLines("https://rest.kegg.jp/link/pathway/pau",warn=FALSE),"'"\\t\",fixed=TRUE);mp<-data.frame(g=sub(\""'^pau:","",vapply(sp,`[`,"",1)),p=sub("''^path:","",vapply(sp,`[`,"",2)));ss<-lapply(split(mp$g,mp$p),unique); bg<-unique(mp$g)
calc<-function(r,d){b<-intersect(r$g,bg);s<-lapply(ss,intersect,y=b);s<-s[lengths(s)>=10&lengths(s)<=500];crit<-if(d=="up")r$l>1.5 else r$l< -1.5;sel<-intersect(r$g[crit],b);N<-length(b);n<-length(sel);K<-lengths(s);k<-vapply(s,function(x)sum(sel%in%x),integer(1));pv<-phyper(k-1,K,N-K,n,lower.tail=FALSE);data.frame(path=names(s),k=k,K=K,n=n,p=pv,adj=p.adjust(pv,"BH"))}
for(d in c("up","down")){cat("===",d,"==='"\\n\");zz<-lapply(rr,calc,d); common<-Reduce(intersect,lapply(zz,function(x)x"'$path[x$p<.05])); cat("raw common",paste(common,collapse=","),"'"\\n\"); common2<-Reduce(intersect,lapply(zz,function(x)x"'$path[x$adj<.05]));cat("adj common",paste(common2,collapse=","),"'"\\n\"); for(q in unique(c(common,common2))) { cat(q, paste(vapply(zz,function(x){y<-x[x"'$path==q,];sprintf("k%d/%d p=%.4g adj=%.4g",y$k,y$n,y$p,y$adj)},""),collapse=" | "),"'"\\n\")}}
RS"
