/bin/bash -lc "Rscript -e '
mp<-strsplit(readLines(\"https://rest.kegg.jp/link/pathway/pau\",warn=FALSE),\"\\\\t\"); map<-data.frame(gene=sub(\""'^pau:","",vapply(mp,`[[`,"",1)),path=sub("''^path:","",vapply(mp,`[[`,"",2))); sets<-split(map$gene,map$path); sets<-lapply(sets,unique); ids_by<-list(); lfc_by<-list(); for(f in list.files("inputs",pattern="'"\\\\.rds"'$",full.names=TRUE)){x<-readRDS(f);a<-attributes(x);ids_by[[basename(f)]]<-a$rownames;lfc_by[[basename(f)]]<-a$listData$log2FoldChange}
ora<-function(g,u,minGS=1,maxGS=Inf){g<-intersect(unique(g),unique(u));u<-intersect(unique(u),map$gene);ts<-names(sets)[vapply(sets,function(s){K<-sum(s%in%u);K>=minGS&&K<=maxGS},logical(1))];p<-vapply(ts,function(t){s<-sets[[t]];K<-sum(s%in%u);k<-sum(s%in%g);phyper(k-1,K,length(u)-K,length(g),lower.tail=FALSE)},numeric(1));a<-p.adjust(p,"BH");data.frame(path=ts,p=p,padj=a)}
for(bg in c("tested","kegg")){for(test in c("padj05","raw05","padj10","raw10")){up<-list();dn<-list();for(nm in names(ids_by)){ids<-ids_by[[nm]];l<-lfc_by[[nm]];u<-if(bg=="tested")ids else map$gene;eu<-ora(ids[l>1.5],u);ed<-ora(ids[l< -1.5],u);sel<-function(e) switch(test,padj05=e$padj<.05,raw05=e$p<.05,padj10=e$padj<.1,raw10=e$p<.1);up[[nm]]<-eu$path[sel(eu)];dn[[nm]]<-ed$path[sel(ed)]};cu<-Reduce(intersect,up);cd<-Reduce(intersect,dn);cat(bg,test,"common_up",length(cu),"common_down",length(cd),"total",length(union(cu,cd)),"'"\\n\")}}
'"
