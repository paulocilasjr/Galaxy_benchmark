/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
 con <- gzfile(f,'rb'); x <- unserialize(con); close(con)
 a <- attributes(x)
 cat('\\n',f,'\\n')
 cat('names attributes:',paste(names(a),collapse=','),'\\n')
 cat('priorInfo class',paste(class(a"'$priorInfo),collapse='"','),'len',length(a"'$priorInfo),'"'\\n')
 cat('metadata class',paste(class(a"'$metadata),collapse='"','),'len',length(a"'$metadata),'"'\\n')
 print(a"'$metadata)
 cat('"'elementMetadata class',paste(class(a"'$elementMetadata),collapse='"','),'len',length(a"'$elementMetadata),'"'\\n')
 em <- a"'$elementMetadata
 if (length(em)) {
  attrem <- attributes(em)
  cat('"'EM attr names',paste(names(attrem),collapse=','),'\\n')
  if ("'!is.null(attrem$listData)) { cat('"'EM listData',paste(names(attrem"'$listData),collapse='"','),'\\n') }
 }
}
RS"
