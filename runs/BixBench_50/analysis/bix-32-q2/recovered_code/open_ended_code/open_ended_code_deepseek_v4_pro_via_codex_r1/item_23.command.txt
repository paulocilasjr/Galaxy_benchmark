/bin/bash -lc "cat > /workspace/tmp/ora.py <<'PY'
import urllib.request, csv, math, os
from collections import defaultdict
from scipy.stats import hypergeom
from statsmodels.stats.multitest import multipletests

def fetch(url):
    req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    with urllib.request.urlopen(req, timeout=60) as r:
        return r.read().decode()

link=fetch('https://rest.kegg.jp/link/pathway/pau')
# parse gene -> set paths
g2p=defaultdict(set)
for line in link.splitlines():
    if '\\t' not in line: continue
    g,p=line.split('\\t')
    g=g.split(':')[1]; p=p.split(':')[1]
    g2p[g].add(p)
p2g=defaultdict(set)
for g,ps in g2p.items():
    for p in ps: p2g[p].add(g)
db_genes=set(g2p)
all_paths=sorted(p2g)

# load csv
data={}
for f in ['tmp/res_1vs97.csv','tmp/res_1vs98.csv','tmp/res_1vs99.csv']:
    with open(f) as fh: rows=list(csv.DictReader(fh))
    data[f]={r['gene']:float(r['log2FoldChange']) for r in rows}
all_measured=set(data[list(data)[0]])
print('db genes',len(db_genes),'pathways',len(p2g),'measured overlap',len(all_measured&db_genes))

def ora(genes, universe, min_g=10, max_g=500, direction_label=''):
    genes=set(genes)&universe
    res=[]
    N=len(universe); n=len(genes)
    paths=[]
    for p in all_paths:
        K=len(p2g[p]&universe)
        if K<min_g or K>max_g: continue
        k=len(genes & p2g[p])
        if k==0: continue
        pval=hypergeom.sf(k-1,N,K,n)
        paths.append((p,k,K,pval))
    if not paths: return res
    pvals=[x[3] for x in paths]
    rej,pcorr,_,_=multipletests(pvals,method='fdr_bh')
    for (p,k,K,pval),adj in zip(paths,pcorr):
        res.append({'path':p,'k':k,'K':K,'n':n,'p':pval,'padj':adj})
    return res

# interpretation A per strain up/down
for uni_name, universe in [('db',db_genes),('measured',all_measured),('measured_and_db',all_measured&db_genes)]:
    print('\\nUNIVERSE',uni_name,'N',len(universe))
    sig={}
    for f,vals in data.items():
        up={g for g,l in vals.items() if l>1.5}
        down={g for g,l in vals.items() if l< -1.5}
        for d,gs in [('up',up),('down',down)]:
            res=ora(gs,universe)
            sig[(f,d)]={r['path'] for r in res if r['padj']<0.05}
            print(f,d,'n query',len(gs&universe),'sig',len(sig[(f,d)]), 'top', sorted(sig[(f,d)])[:5])
    up3=set.intersection(*[sig[(f,'up')] for f in data])
    down3=set.intersection(*[sig[(f,'down')] for f in data])
    print('UP3',len(up3),sorted(up3))
    print('DOWN3',len(down3),sorted(down3))
    print('UNION',len(up3|down3))

# interpretation B common genes, ORA combined
for uni_name, universe in [('db',db_genes),('measured',all_measured),('measured_and_db',all_measured&db_genes)]:
    vals_list=list(data.values())
    up_common={g for g in all_measured if all(vals[g]>1.5 for vals in vals_list)}
    down_common={g for g in all_measured if all(vals[g]<-1.5 for vals in vals_list)}
    print('\\nB universe',uni_name,'up common',len(up_common),'down common',len(down_common))
    for d,gs in [('up',up_common),('down',down_common)]:
        res=ora(gs,universe)
        sig=[r for r in res if r['padj']<0.05]
        print(d,'sig',len(sig),sorted(r['path'] for r in sig)[:20])
PY
python3 /workspace/tmp/ora.py"
