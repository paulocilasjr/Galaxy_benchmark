/bin/bash -lc "Rscript - <<'RS'
files <- c('inputs/res_1vs97.rds','inputs/res_1vs98.rds','inputs/res_1vs99.rds')
for (f in files) {
 con <- gzfile(f,'rb'); x <- unserialize(con); close(con)
 a <- attributes(x); ld <- a"'$listData; rn <- a$rownames
 cat('"'\\n',f,'n',length(rn),'\\n')
 cat('NA lfc',sum(is.na(ld"'$log2FoldChange)),'"'NA padj',sum(is.na(ld"'$padj)),'"'\\n')
 cat('all IDs? ', all(grepl('"'^PA14_'"', rn)), ' unique? ', "'!anyDuplicated(rn), '"'\\n')
 cat('up lfc>1.5',sum(ld"'$log2FoldChange>1.5, na.rm=TRUE),'"'down lfc< -1.5',sum(ld"'$log2FoldChange< -1.5, na.rm=TRUE),'"'\\n')
 cat('up lfc>1.5 & padj<0.05',sum(ld"'$log2FoldChange>1.5 & ld$padj<0.05, na.rm=TRUE),'"'down lfc<-1.5 & padj<0.05',sum(ld"'$log2FoldChange< -1.5 & ld$padj<0.05, na.rm=TRUE),'"'\\n')
 cat('padj<0.05',sum(ld"'$padj<0.05, na.rm=TRUE),'"'lfc>1.5 OR <-1.5',sum(abs(ld"'$log2FoldChange)>1.5, na.rm=TRUE),'"'\\n')
}
RS"
