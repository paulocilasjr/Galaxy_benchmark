import requests
headers = {'User-Agent': 'Python/3.11'}

# Try a different KEGG endpoint
urls = [
    'https://rest.kegg.jp/list/pathway/pau',
    'https://rest.kegg.jp/get/pau:PA14_39780',
    'https://rest.kegg.jp/link/pathway/pau',
    'https://rest.kegg.jp/list/pau',
]
for url in urls:
    r = requests.get(url, headers=headers, timeout=10)
    print(f'{url}: {r.status_code} len={len(r.text)}')
    if r.status_code == 200:
        print(r.text[:300])
    print()
