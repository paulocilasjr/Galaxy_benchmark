R -e '
x <- readRDS("/workspace/inputs/res_1vs97.rds")
cat("Slot names:", slotNames(x), "\n\n")

# Try accessing slots directly
for (s in slotNames(x)) {
  cat("Slot:", s, "\n")
  val <- slot(x, s)
  cat("  Class:", class(val), "\n")
  cat("  Length:", length(val), "\n")
  if (length(val) <= 20) {
    cat("  Value:", toString(val), "\n")
  } else {
    cat("  Head:", toString(head(val)), "\n")
  }
  cat("\n")
}
' 2>&1
