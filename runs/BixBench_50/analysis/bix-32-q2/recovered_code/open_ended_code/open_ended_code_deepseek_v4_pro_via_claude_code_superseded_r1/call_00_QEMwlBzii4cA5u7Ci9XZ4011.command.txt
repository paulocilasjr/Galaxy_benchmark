R -e '
# Extract data from all three RDS files
extract_df <- function(path) {
  x <- readRDS(path)
  ld <- attr(x, "listData")
  rn <- attr(x, "rownames")
  df <- data.frame(
    gene = rn,
    baseMean = ld$baseMean,
    log2FoldChange = ld$log2FoldChange,
    lfcSE = ld$lfcSE,
    stat = ld$stat,
    pvalue = ld$pvalue,
    padj = ld$padj,
    stringsAsFactors = FALSE
  )
  return(df)
}

res97 <- extract_df("/workspace/inputs/res_1vs97.rds")
res98 <- extract_df("/workspace/inputs/res_1vs98.rds")
res99 <- extract_df("/workspace/inputs/res_1vs99.rds")

cat("res97 dimensions:", dim(res97), "\n")
cat("res98 dimensions:", dim(res98), "\n")
cat("res99 dimensions:", dim(res99), "\n")

# Find genes with |lfc| > 1.5 in each
up97 <- res97$gene[res97$log2FoldChange > 1.5]
down97 <- res97$gene[res97$log2FoldChange < -1.5]
up98 <- res98$gene[res98$log2FoldChange > 1.5]
down98 <- res98$gene[res98$log2FoldChange < -1.5]
up99 <- res99$gene[res99$log2FoldChange > 1.5]
down99 <- res99$gene[res99$log2FoldChange < -1.5]

cat("\n|LFC|>1.5 genes in each strain:\n")
cat("97: up=", length(up97), " down=", length(down97), "\n")
cat("98: up=", length(up98), " down=", length(down98), "\n")
cat("99: up=", length(up99), " down=", length(down99), "\n")

# Find genes consistently up (lfc > 1.5) in all three
consistently_up <- intersect(intersect(up97, up98), up99)
cat("\nConsistently upregulated (lfc > 1.5) in all 3:", length(consistently_up), "\n")

# Find genes consistently down (lfc < -1.5) in all three
consistently_down <- intersect(intersect(down97, down98), down99)
cat("Consistently downregulated (lfc < -1.5) in all 3:", length(consistently_down), "\n")

# Combined set for ORA
all_consistent <- union(consistently_up, consistently_down)
cat("Total consistently DE in same direction:", length(all_consistent), "\n")

# Save for later use
writeLines(consistently_up, "/workspace/consistent_up_genes.txt")
writeLines(consistently_down, "/workspace/consistent_down_genes.txt")
writeLines(all_consistent, "/workspace/consistent_genes.txt")

# Save full data
saveRDS(res97, "/workspace/res97_df.rds")
saveRDS(res98, "/workspace/res98_df.rds")
saveRDS(res99, "/workspace/res99_df.rds")

# Show some examples
cat("\nFirst 20 consistently upregulated:\n")
print(head(consistently_up, 20))
cat("\nFirst 20 consistently downregulated:\n")
print(head(consistently_down, 20))
' 2>&1
