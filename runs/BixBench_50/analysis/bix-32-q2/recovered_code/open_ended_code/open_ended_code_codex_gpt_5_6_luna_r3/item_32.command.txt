/bin/bash -lc "Rscript - <<'RS'
paths <- c(\"/workspace/inputs/res_1vs97.rds\", \"/workspace/inputs/res_1vs98.rds\", \"/workspace/inputs/res_1vs99.rds\")
res <- lapply(paths, function(p) { x <- readRDS(p); ld <- attr(x,\"listData\"); data.frame(gene=attr(x,\"rownames\"),lfc=ld"'$log2FoldChange) }); names(res)<-c("97","98","99")
links<-read.delim("https://rest.kegg.jp/link/pathway/pau",header=FALSE,sep="'"\\t\",quote=\"\",stringsAsFactors=FALSE); links"'$V1<-sub("''^pau:","",links$V1); links$V2<-sub("''^path:","",links$V2); gs0<-lapply(split(links$V1,links$V2),unique); uni<-intersect(unique(unlist(lapply(res,`[[`,"gene"))),unique(links$V1));
for(nm in names(res)){d<-res[[nm]];fg<-d$gene[d$lfc < -1.5];gs<-lapply(gs0,intersect,y=uni);sz<-lengths(gs);k<-vapply(gs,function(z)length(intersect(z,fg)),integer(1));test<-sz>=10&sz<=500&k>0;p<-rep(NA_real_,length(gs));p[test]<-phyper(k[test]-1,sz[test],length(uni)-sz[test],length(intersect(fg,uni)),lower.tail=FALSE);pa<-rep(NA_real_,length(gs));pa[test]<-p.adjust(p[test],"BH");z<-match(c("pau01110","pau01310","pau04146"),names(gs));cat(nm,"n",length(intersect(fg,uni)),"'"\\n\");for(i in z)cat(names(gs)[i],\"size\",sz[i],\"hit\",k[i],\"p\",format(p[i],digits=8),\"padj\",format(pa[i],digits=8),\"\\n\")}
RS"
