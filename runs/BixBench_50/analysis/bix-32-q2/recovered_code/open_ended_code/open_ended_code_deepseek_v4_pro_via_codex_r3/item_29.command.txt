/bin/bash -c "Rscript - <<'RS'
files <- c('res_1vs97.rds','res_1vs98.rds','res_1vs99.rds')
for (f in files) {
  x <- readRDS(file.path('/workspace/inputs', f))
  ld <- attr(x,'listData')
  rn <- attr(x,'rownames')
  d <- data.frame(gene=rn, baseMean=ld"'$baseMean, log2FoldChange=ld$log2FoldChange, lfcSE=ld$lfcSE, stat=ld$stat, pvalue=ld$pvalue, padj=ld$padj)
  out <- sub('"'.rds"'$'"','.csv', file.path('/workspace/tmp', f))
  write.csv(d, out, row.names=FALSE)
  cat(f, 'nrow', nrow(d), '\\n')
  cat('absLFC>1.5:', sum("'!is.na(d$log2FoldChange) & abs(d$log2FoldChange)>1.5), '"'\\n')
  cat('padj<0.05:', sum("'!is.na(d$padj) & d$padj<0.05), '"'\\n')
  cat('absLFC>1.5 & padj<0.05:', sum("'!is.na(d$log2FoldChange) & !is.na(d$padj) & abs(d$log2FoldChange)>1.5 & d$padj<0.05), '"'\\n')
  cat('up:', sum("'!is.na(d$log2FoldChange) & !is.na(d$padj) & d$log2FoldChange>1.5 & d$padj<0.05), '"'down:', sum("'!is.na(d$log2FoldChange) & !is.na(d$padj) & d$log2FoldChange<(-1.5) & d$padj<0.05), '"'\\n')
  cat('up no padj:', sum("'!is.na(d$log2FoldChange) & d$log2FoldChange>1.5), '"'down no padj:', sum("'!is.na(d$log2FoldChange) & d$log2FoldChange<(-1.5)), '"'\\n')
}
RS"
