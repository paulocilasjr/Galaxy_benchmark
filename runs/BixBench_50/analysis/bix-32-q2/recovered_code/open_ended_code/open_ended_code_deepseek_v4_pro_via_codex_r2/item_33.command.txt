/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
link=defaultdict(set); genes=set(); paths=set()
for line in open('/workspace/tmp/pau_link.tsv'):
    g,p=line.strip().split('\\t')
    g=g.split(':',1)[1]
    p=p.split(':',1)[1]
    link[p].add(g); genes.add(g); paths.add(p)
names={}
for line in open('/workspace/tmp/pau_pathways.tsv'):
    pid,name=line.strip().split('\\t',1)
    names[pid]=name
print('unique genes',len(genes),'unique paths',len(paths),'names',len(names))
print('path sizes min max',min(map(len,link.values())),max(map(len,link.values())))
print('between 10-500',sum(10<=len(v)<=500 for v in link.values()))
print('paths in names',len(paths & set(names)))
PY"
