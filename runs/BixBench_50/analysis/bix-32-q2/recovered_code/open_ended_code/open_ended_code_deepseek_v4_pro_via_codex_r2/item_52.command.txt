/bin/bash -lc "Rscript - <<'RS'
link <- read.table('/workspace/tmp/pau_link.tsv', sep='\\t', header=FALSE, stringsAsFactors=FALSE, quote='')
link[,1] <- sub('["'^:]+:'"', '', link[,1]); link[,2] <- sub('["'^:]+:'"', '', link[,2])
paths <- read.table('/workspace/tmp/pau_pathways.tsv', sep='\\t', header=FALSE, stringsAsFactors=FALSE, quote='')
link <- link[link[,2] %in% paths[,1],]
p2g <- split(link[,1], link[,2]); g2p <- split(link[,2], link[,1]); N <- length(unique(link[,1]))
d <- read.csv('/workspace/tmp/res_97.csv', stringsAsFactors=FALSE)
gs <- d"'$gene[d$lfc > 1.5]
gs <- intersect(unique(gs), names(g2p)); mapped <- g2p[gs]
qts <- unique(unlist(mapped)); qts <- qts[sapply(p2g[qts], length) >=10 & sapply(p2g[qts], length) <=500]
n <- length(mapped)
for(t in sort(qts)){k<-sum(p2g[[t]] %in% gs);M<-length(p2g[[t]]);p<-phyper(k-1,M,N-M,n,lower.tail=FALSE);cat(t,'"'k',k,'M',M,'n',n,'p',format(p,digits=15),'\\n')}
cat('p.adjust\\n'); ps<-sapply(sort(qts),function(t){k<-sum(p2g[[t]] %in% gs);M<-length(p2g[[t]]);phyper(k-1,M,N-M,n,lower.tail=FALSE)}); names(ps)<-sort(qts); print(p.adjust(ps,method='BH'))
RS"
