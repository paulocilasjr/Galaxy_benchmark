import requests

# Get total gene count in KEGG for pau
r = requests.get("https://rest.kegg.jp/list/pau", timeout=60)
total_kegg_genes = len(r.text.strip().split("\n"))
print(f"Total KEGG genes for pau: {total_kegg_genes}")
print(f"First 3 lines: {r.text.strip().split(chr(10))[:3]}")

# Also check the pathway-gene associations
r2 = requests.get("https://rest.kegg.jp/link/pau/pathway", timeout=60)
lines = r2.text.strip().split("\n")
total_links = len(lines)
unique_genes = set()
unique_pathways = set()
for line in lines:
    parts = line.split("\t")
    if len(parts) >= 2:
        unique_pathways.add(parts[0].replace("path:", ""))
        unique_genes.add(parts[1].replace("pau:", ""))
print(f"\nPathway-gene links: {total_links}")
print(f"Unique genes in pathways: {len(unique_genes)}")
print(f"Unique pathways: {len(unique_pathways)}")
