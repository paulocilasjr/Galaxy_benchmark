Rscript -e '
x <- readRDS("/workspace/inputs/res_1vs97.rds")
a <- attributes(x)

# Try accessing listData directly
cat("--- listData ---\n")
ld <- a$listData
cat("class:", class(ld), "\n")
cat("names:", names(ld), "\n")
cat("lengths:", sapply(ld, length), "\n")

# Try to print first few entries
for (nm in names(ld)) {
  cat("\nColumn:", nm, "\n")
  cat("Class:", class(ld[[nm]]), "\n")
  cat("Head:", head(ld[[nm]], 3), "\n")
}

cat("\n--- rownames ---\n")
cat("First few rownames:", head(a$rownames, 3), "\n")
cat("Number of rows:", a$nrows, "\n")
'
