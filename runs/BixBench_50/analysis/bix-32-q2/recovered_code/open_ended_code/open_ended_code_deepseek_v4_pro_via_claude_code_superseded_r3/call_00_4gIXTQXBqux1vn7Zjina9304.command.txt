import requests, json, csv, time
from collections import defaultdict

# Step 1: Get KEGG pathways for pau
print("=== Getting pathways for pau ===")
r = requests.get("https://rest.kegg.jp/list/pathway/pau", timeout=30)
pathways = {}
for line in r.text.strip().split("\n"):
    parts = line.split("\t")
    if len(parts) >= 2:
        pid, desc = parts[0], parts[1]
        pid_short = pid.replace("path:", "")
        pathways[pid_short] = desc

print(f"Total pathways: {len(pathways)}")
for pid, desc in list(pathways.items())[:10]:
    print(f"  {pid}: {desc[:80]}")
print("  ...")

# Step 2: Get pathway-gene links
print("\n=== Getting pathway-gene links ===")
time.sleep(0.5)
r = requests.get("https://rest.kegg.jp/link/pau/pathway", timeout=60)
pathway_genes = defaultdict(set)
for line in r.text.strip().split("\n"):
    parts = line.split("\t")
    if len(parts) >= 2:
        pid = parts[0].replace("path:", "")
        gene = parts[1].replace("pau:", "")
        pathway_genes[pid].add(gene)

print(f"Pathways with genes: {len(pathway_genes)}")
for pid, genes in list(pathway_genes.items())[:5]:
    print(f"  {pid}: {len(genes)} genes")

# Save to file for later use
import pickle
with open("/tmp/analysis/kegg_data.pkl", "wb") as f:
    pickle.dump({"pathways": pathways, "pathway_genes": dict(pathway_genes)}, f)

# Step 3: Show some gene IDs from pathway-gene links to understand format
all_genes_in_pathways = set()
for genes in pathway_genes.values():
    all_genes_in_pathways.update(genes)
print(f"\nTotal unique genes in pathways: {len(all_genes_in_pathways)}")
print(f"Example gene IDs: {list(all_genes_in_pathways)[:10]}")
