ann_path='inputs/Sample_annotated_Zenodo.csv'

    print('CD14 samples',len(cd14),'selected count columns',len(idx),'indices unique',len(set(idx)))

    print('missing CD14 columns', [s for s in cd14 if s not in h])
    row=next(reader)
    vals=[int(row[i]) for i in idx]
    print('first row',row[0],'CD14 total',sum(vals),'mean',sum(vals)/len(vals),'minmax',min(vals),max(vals))

with open(meta_path, newline='') as f:
    mr=csv.reader(f)
