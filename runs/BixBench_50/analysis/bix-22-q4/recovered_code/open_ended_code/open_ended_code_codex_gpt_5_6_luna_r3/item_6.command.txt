/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
from pathlib import Path

ann_path = Path('inputs/Sample_annotated_Zenodo.csv')
meta_path = Path('inputs/GeneMetaInfo_Zenodo.csv')
counts_path = Path('inputs/BatchCorrectedReadCounts_Zenodo.csv')

with ann_path.open(newline='') as f:
    ann = list(csv.DictReader(f))
print('annotation rows', len(ann))
print('annotation columns', list(ann[0]))
print('celltype counts', Counter(r['celltype'] for r in ann))
print('CD14 rows', sum(r['celltype']=='CD14' for r in ann))
print('CD14 sample first/last', [r['sample'] for r in ann if r['celltype']=='CD14'][:5], [r['sample'] for r in ann if r['celltype']=='CD14'][-5:])
print('duplicate sample annotation rows', sum(v>1 for v in Counter(r['sample'] for r in ann).values()))

with meta_path.open(newline='') as f:
    meta_reader = csv.DictReader(f)
    meta_rows = list(meta_reader)
print('metadata rows', len(meta_rows))
print('metadata columns', meta_reader.fieldnames)
print('gene biotype counts top', Counter(r['gene_biotype'] for r in meta_rows).most_common(20))
print('protein_coding rows', sum(r['gene_biotype']=='protein_coding' for r in meta_rows))
print('metadata duplicate gene names', sum(v>1 for v in Counter(r['Geneid'] for r in meta_rows).values()))

with counts_path.open(newline='') as f:
    reader = csv.reader(f)
    header = next(reader)
    first = next(reader)
print('counts columns', len(header), 'first blank?', repr(header[0]))
print('counts samples', len(header)-1)
print('counts first gene', first[0], 'first values', first[1:6])
print('counts duplicate sample columns', sum(v>1 for v in Counter(header[1:]).items() if v>1))
ann_names = {r['sample'] for r in ann}
count_names = set(header[1:])
print('annotation samples absent from counts', len(ann_names-count_names), sorted(ann_names-count_names)[:10])
print('count samples absent from annotation', len(count_names-ann_names), sorted(count_names-ann_names)[:10])
print('CD14 samples in counts', sum(x in count_names for x in [r['sample'] for r in ann if r['celltype']=='CD14']))
PY"
