/bin/bash -lc "Rscript - <<'RS'
s <- read.csv('inputs/Sample_annotated_Zenodo.csv', stringsAsFactors=FALSE, check.names=FALSE)
cd14 <- s"'$sample[s$celltype == '"'CD14']
h <- names(read.csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', nrows=0, check.names=FALSE))
classes <- ifelse(h %in% cd14, 'numeric', 'NULL')
classes[1] <- 'character'
x <- read.csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', check.names=FALSE,
              stringsAsFactors=FALSE, colClasses=classes)
names(x)[1] <- 'gene'
m <- read.csv('inputs/GeneMetaInfo_Zenodo.csv', stringsAsFactors=FALSE, check.names=FALSE)
names(m)[1] <- 'gene'
stopifnot(nrow(x)==59453L, ncol(x)==133L, length(cd14)==132L,
          setequal(names(x)[-1], cd14), identical(x"'$gene,m$gene),
          !anyNA(x), !anyNA(m$Length))
tot <- rowSums(x[-1])
mu <- rowMeans(x[-1])
keep <- m$gene_biotype == '"'protein_coding' & tot >= 10
cat('selected=',sum(keep),'\\n',sep='')
cat('pearson=',sprintf('%.17g',cor(m"'$Length[keep],mu[keep],method='"'pearson')),'\\n',sep='')
# Also verify using the covariance identity behind Pearson's r.
xv <- m"'$Length[keep]; yv <- mu[keep]
manual <- sum((xv-mean(xv))*(yv-mean(yv))) / sqrt(sum((xv-mean(xv))''^2)*sum((yv-mean(yv))''^2))
cat('"'manual=',sprintf('%.17g',manual),'\\n',sep='')
RS"
