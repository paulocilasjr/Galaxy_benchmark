/bin/bash -lc "python3 - <<'PY'
import csv, collections
biotypes=collections.Counter(); lengths=[]
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
    r=csv.reader(f); next(r)
    for row in r:
        biotypes[row[4]]+=1
        if row[4]=='protein_coding': lengths.append(int(row[3]))
print('biotype count', len(biotypes))
for k,v in biotypes.most_common(20): print(k,v)
print('protein coding', biotypes['protein_coding'], 'length range', min(lengths),max(lengths))
PY"
