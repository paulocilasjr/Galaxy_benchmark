/bin/bash -lc "python - <<'PY'
import csv, math, statistics
from pathlib import Path
p=Path('inputs')
with open(p/'Sample_annotated_Zenodo.csv',newline='') as f:
    cd14={r['sample'] for r in csv.DictReader(f) if r['celltype']=='CD14'}
with open(p/'GeneMetaInfo_Zenodo.csv',newline='') as f:
    dr=csv.DictReader(f); k=dr.fieldnames[0]; rows=list(dr)
print('metadata index "'!= Geneid'"',sum(r[k]"'!=r['"'Geneid'] for r in rows))
meta={r[k]:(int(r['Length']),r['gene_biotype']) for r in rows}
points=[]; all_threshold_points=[]
with open(p/'BatchCorrectedReadCounts_Zenodo.csv',newline='') as f:
    rd=csv.reader(f); h=next(rd)
    idx=[i for i,s in enumerate(h) if s in cd14]
    for row in rd:
        length,bio=meta[row[0]]
        if bio"'!='"'protein_coding': continue
        s_cd14=sum(int(row[i]) for i in idx)
        s_all=sum(int(v) for v in row[1:])
        if s_cd14>=10: points.append((length,s_cd14/len(idx)))
        if s_all>=10: all_threshold_points.append((length,s_cd14/len(idx)))
xs=[p[0] for p in points]; ys=[p[1] for p in points]
r=statistics.correlation(xs,ys)
print('statistics.correlation',len(points),format(r,'.17g'),repr(r))
print('all-sample threshold candidate',len(all_threshold_points),format(statistics.correlation([x for x,y in all_threshold_points],[y for x,y in all_threshold_points]),'.17g'))
# Algebraically independent raw-sum formula
n=len(points); sx=sum(xs); sy=sum(ys); sxx=sum(x*x for x in xs); syy=sum(y*y for y in ys); sxy=sum(x*y for x,y in points)
r2=(n*sxy-sx*sy)/math.sqrt((n*sxx-sx*sx)*(n*syy-sy*sy))
print('raw-moment check',format(r2,'.17g'),'absolute difference',abs(r-r2))
PY"
