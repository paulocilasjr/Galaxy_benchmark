/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, os
base='/workspace/inputs'
ann=pd.read_csv(base+'/Sample_annotated_Zenodo.csv')
cd14=ann.loc[ann['celltype']=='CD14','sample'].tolist()
print('cd14 n',len(cd14))
meta=pd.read_csv(base+'/GeneMetaInfo_Zenodo.csv')
print(meta.shape, meta.columns.tolist())
print(meta['gene_biotype'].value_counts().head(20))
# read counts only gene id + cd14 columns for memory
counts=pd.read_csv(base+'/BatchCorrectedReadCounts_Zenodo.csv', usecols=[''] + cd14)
print('counts shape', counts.shape)
print(counts.head())
print('missing in meta:', counts[''].isin(meta.iloc[:,1]).sum(), 'unique', counts[''].nunique(), 'dup', counts[''].duplicated().sum())
# identify geneid column in meta
meta_col=meta.columns[0] if meta.columns[0]=='' else 'Geneid'
print('meta first col repr',repr(meta.columns[0]), meta[[meta.columns[1], 'gene_biotype']].head())
pc=meta[meta['gene_biotype']=='protein_coding']
print('pc',len(pc), 'unique', pc['Geneid'].nunique())
# count matrix index is blank col first; second col gene id likely
df=counts.merge(pc[['Geneid','Length']], left_on=counts.columns[0], right_on='Geneid', how='inner')
print('merged',df.shape)
# total counts across cd14
vals=df[cd14].values
df['total']=vals.sum(axis=1)
df['mean']=df['total']/len(cd14)
exp=df[df['total']>=10].copy()
print('expressed protein coding >=10', len(exp), 'total range', exp['total'].min(), exp['total'].max())
print(exp[['Geneid','Length','total','mean']].head())
print('Length dtype', exp['Length'].dtype, 'mean dtype', exp['mean'].dtype)
print('pearson length vs mean', np.corrcoef(exp['Length'], exp['mean'])[0,1])
print('pearson length vs log mean', np.corrcoef(exp['Length'], np.log(exp['mean']))[0,1])
print('pearson log length vs log mean', np.corrcoef(np.log(exp['Length']), np.log(exp['mean']))[0,1])
print('spearman', exp['Length'].corr(exp['mean'], method='spearman'))
PY"
