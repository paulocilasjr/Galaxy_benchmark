import csv
from collections import Counter
samples=[]
with open('/workspace/inputs/Sample_annotated_Zenodo.csv') as fh:

print('samples',len(samples))
print(Counter(row['celltype'] for row in samples))
print(Counter(row['samtype'] for row in samples))
print('CD14 sample ids', [r['sample'] for r in samples if r['celltype']=='CD14'][:20], 'n', sum(r['celltype']=='CD14' for r in samples))
