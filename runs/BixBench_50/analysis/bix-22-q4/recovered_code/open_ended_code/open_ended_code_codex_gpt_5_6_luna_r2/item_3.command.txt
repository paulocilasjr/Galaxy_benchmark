/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
from pathlib import Path
p=Path('/workspace/inputs/GeneMetaInfo_Zenodo.csv')
with p.open(newline='') as f:
    rd=csv.DictReader(f)
    bi=Counter(); lengths=[]; bad=[]
    for r in rd:
        bi[r['gene_biotype']]+=1
        try: lengths.append(float(r['Length']))
        except: bad.append((r['Geneid'],r['Length']))
print('biotypes',len(bi))
print(bi.most_common())
print('length bad',len(bad),bad[:10], 'length range',min(lengths),max(lengths))
PY"
