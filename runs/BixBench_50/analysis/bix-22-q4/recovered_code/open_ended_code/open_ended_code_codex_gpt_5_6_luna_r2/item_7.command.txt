/bin/bash -lc "python - <<'PY'
import csv, math
from pathlib import Path
base=Path('/workspace/inputs')
meta={}
with (base/'GeneMetaInfo_Zenodo.csv').open(newline='') as f:
    for row in csv.DictReader(f):
        meta[row['']]=(float(row['Length']), row['gene_biotype'])
with (base/'Sample_annotated_Zenodo.csv').open(newline='') as f:
    ann=list(csv.DictReader(f))
cd14=[r['sample'] for r in ann if r['celltype']=='CD14']

def corr(xs, ys):
    n=len(xs)
    mx=math.fsum(xs)/n; my=math.fsum(ys)/n
    num=math.fsum((x-mx)*(y-my) for x,y in zip(xs,ys))
    dx=math.fsum((x-mx)*(x-mx) for x in xs)
    dy=math.fsum((y-my)*(y-my) for y in ys)
    return num/math.sqrt(dx*dy)

with (base/'BatchCorrectedReadCounts_Zenodo.csv').open(newline='') as f:
    rd=csv.reader(f)
    header=next(rd)
    idx={s:i for i,s in enumerate(header)}
    missing=[s for s in cd14 if s not in idx]
    if missing: raise RuntimeError(f'missing CD14 count columns: {missing}')
    cd14_idx=[idx[s] for s in cd14]
    all_idx=range(1,len(header))
    xs=[]; ys=[]; xs_allfilter=[]; ys_allfilter=[]
    n_protein=0; n_a=0; n_b=0
    for row in rd:
        length,biotype=meta[row[0]]
        if biotype"'!='"'protein_coding':
            continue
        n_protein+=1
        cd14_total=sum(int(row[i]) for i in cd14_idx)
        all_total=sum(int(row[i]) for i in all_idx)
        cd14_mean=cd14_total/len(cd14_idx)
        if cd14_total>=10:
            xs.append(length); ys.append(cd14_mean); n_a+=1
        if all_total>=10:
            xs_allfilter.append(length); ys_allfilter.append(cd14_mean); n_b+=1
print('CD14 samples:',len(cd14))
print('protein-coding genes:',n_protein)
print('CD14 total >= 10:',n_a)
print('Pearson:',repr(corr(xs,ys)))
print('all-sample total >= 10:',n_b)
print('all-sample-filter Pearson:',repr(corr(xs_allfilter,ys_allfilter)))
print('CD14 mean range:',min(ys),max(ys))
print('gene length range:',min(xs),max(xs))
PY"
