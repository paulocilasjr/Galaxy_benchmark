/bin/bash -lc "python - <<'PY'
import csv, math
from collections import Counter

sample_file='inputs/Sample_annotated_Zenodo.csv'
meta_file='inputs/GeneMetaInfo_Zenodo.csv'
counts_file='inputs/BatchCorrectedReadCounts_Zenodo.csv'

with open(sample_file, newline='') as fh:
    samples=list(csv.DictReader(fh))
print('sample rows', len(samples))
print('celltype counts', Counter(r['celltype'] for r in samples))
cd14_samples=[r['sample'] for r in samples if r['celltype']=='CD14']
print('cd14 n', len(cd14_samples), cd14_samples[:5], cd14_samples[-5:])

meta={}
meta_dupes=[]
with open(meta_file, newline='') as fh:
    rdr=csv.DictReader(fh)
    for r in rdr:
        gene=r['']
        if gene in meta:
            meta_dupes.append(gene)
        meta[gene]=(int(r['Length']), r['gene_biotype'], r['Geneid'])
print('meta genes', len(meta), 'dupes', len(meta_dupes))

n=0
sumx=sumy=sumxx=sumyy=sumxy=0.0
missing_meta=0
count_rows=0
protein_rows=0
expressed_rows=0
with open(counts_file, newline='') as fh:
    rdr=csv.reader(fh)
    header=next(rdr)
    sample_to_idx={s:i for i,s in enumerate(header)}
    idxs=[sample_to_idx[s] for s in cd14_samples]
    print('all cd14 headers found', len(idxs)==len(cd14_samples))
    for row in rdr:
        count_rows+=1
        gene=row[0]
        m=meta.get(gene)
        if m is None:
            missing_meta+=1
            continue
        length, biotype, geneid = m
        if biotype "'!= '"'protein_coding':
            continue
        protein_rows+=1
        vals=[float(row[i]) for i in idxs]
        total=sum(vals)
        if total < 10:
            continue
        expressed_rows+=1
        x=float(length)
        y=total/len(idxs)
        n+=1
        sumx+=x; sumy+=y; sumxx+=x*x; sumyy+=y*y; sumxy+=x*y

num=n*sumxy - sumx*sumy
den=math.sqrt((n*sumxx-sumx*sumx)*(n*sumyy-sumy*sumy))
r=num/den
print('count rows', count_rows, 'missing_meta', missing_meta, 'protein', protein_rows, 'expressed', expressed_rows, 'n', n)
print('r', repr(r))
print('r12', format(r,'.12g'))
print('r15', format(r,'.15g'))
PY"
