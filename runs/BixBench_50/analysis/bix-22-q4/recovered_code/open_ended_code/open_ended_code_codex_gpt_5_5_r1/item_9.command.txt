/bin/bash -lc "python - <<'PY'
import csv, math

sample_file='inputs/Sample_annotated_Zenodo.csv'
meta_file='inputs/GeneMetaInfo_Zenodo.csv'
counts_file='inputs/BatchCorrectedReadCounts_Zenodo.csv'

def corr(pairs):
    n=len(pairs)
    sx=sum(x for x,y in pairs); sy=sum(y for x,y in pairs)
    sxx=sum(x*x for x,y in pairs); syy=sum(y*y for x,y in pairs); sxy=sum(x*y for x,y in pairs)
    return (n*sxy-sx*sy)/(((n*sxx-sx*sx)*(n*syy-sy*sy))**0.5)

with open(sample_file, newline='') as fh:
    ann=list(csv.DictReader(fh))
cd14=[r['sample'] for r in ann if r['celltype']=='CD14']
all_samples=[r['sample'] for r in ann]
meta={}
with open(meta_file, newline='') as fh:
    for r in csv.DictReader(fh):
        meta[r['']]=(int(r['Length']), r['gene_biotype'])

pairs_cd14_thresh=[]
pairs_all_thresh=[]
with open(counts_file, newline='') as fh:
    rdr=csv.reader(fh); header=next(rdr)
    h={s:i for i,s in enumerate(header)}
    cd14_i=[h[s] for s in cd14]
    all_i=[h[s] for s in all_samples]
    for row in rdr:
        gene=row[0]
        length, bt=meta[gene]
        if bt"'!='"'protein_coding':
            continue
        cd14_vals=[float(row[i]) for i in cd14_i]
        cd14_total=sum(cd14_vals)
        all_total=sum(float(row[i]) for i in all_i)
        mean_cd14=cd14_total/len(cd14_i)
        if cd14_total>=10:
            pairs_cd14_thresh.append((float(length), mean_cd14))
        if all_total>=10:
            pairs_all_thresh.append((float(length), mean_cd14))

print('cd14 threshold', len(pairs_cd14_thresh), corr(pairs_cd14_thresh))
print('all-sample threshold, mean cd14', len(pairs_all_thresh), corr(pairs_all_thresh))
PY"
