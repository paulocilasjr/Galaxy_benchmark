    cd14_samples=[row['sample'] for row in csv.DictReader(f) if row['celltype']=='CD14']

    for row in csv.DictReader(f):
        gene=row['Geneid']
        length_by_gene[gene]=float(row['Length'])
        biotype_by_gene[gene]=row['gene_biotype']

n=0
sx=sy=sxx=syy=sxy=0.0
count_rows=0
missing_meta=0
protein_rows=0
protein_expressed=0

        gene=row[0]

        if biotype_by_gene[gene] "'!= '"'protein_coding':
