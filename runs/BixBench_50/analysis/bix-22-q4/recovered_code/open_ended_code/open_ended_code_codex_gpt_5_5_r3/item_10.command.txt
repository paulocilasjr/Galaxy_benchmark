import csv, collections
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:

print('gene meta rows', len(rows))
print('biotypes top20', collections.Counter(row['gene_biotype'] for row in rows).most_common(20))
print('protein_coding', sum(row['gene_biotype']=='protein_coding' for row in rows))
print('blank length', sum(not row['Length'] for row in rows))
