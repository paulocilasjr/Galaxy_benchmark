/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def readfa(txt):
    seqs=[]
    for line in txt.splitlines():
        if line.startswith('>'): seqs.append('')
        else: seqs[-1]+=line.strip()
    return seqs

def percentile(v,p):
    v=sorted(v); x=(len(v)-1)*p; lo=int(x); hi=min(lo+1,len(v)-1)
    return v[lo]+(v[hi]-v[lo])*(x-lo)

def metrics(zpath):
    out=[]; chars=Counter(); bad=[]
    with ZipFile(zpath) as z:
        for name in z.namelist():
            if not name.endswith('.mafft'): continue
            ss=readfa(z.read(name).decode())
            if len(ss)"'!=4 or len({len(s) for s in ss})!=1:
                bad.append((name,len(ss),[len(s) for s in ss])); continue
            chars.update('"''.join(ss))
            vals={}
            for mode, missing in [('gap_missing',set('-?X*')),('gap_state',set('?X*')),('all_state',set())]:
                n=0
                for col in zip(*ss):
                    c=Counter(x.upper() for x in col if x.upper() not in missing)
                    if sum(v>=2 for v in c.values())>=2: n+=1
                vals[mode]=n
            out.append((name,vals,len(ss[0])))
    return out,chars,bad

def midranks(v):
    order=sorted(range(len(v)),key=v.__getitem__); r=[None]*len(v); i=0
    while i<len(v):
        j=i+1
        while j<len(v) and v[order[j]]==v[order[i]]: j+=1
        rank=((i+1)+j)/2
        for k in range(i,j): r[order[k]]=rank
        i=j
    return r

def ustat(a,b):
    ranks=midranks(a+b); n=len(a)
    ua=sum(ranks[:n])-n*(n+1)/2
    return ua,len(a)*len(b)-ua,sum(ranks[:n])

allm={}
for group in ('animals','fungi'):
    m,ch,bad=metrics('inputs/scogs_'+group+'.zip'); allm[group]=m
    print(group,'n',len(m),'bad',bad,'chars',dict(ch))
    for mode in ('gap_missing','gap_state','all_state'):
        a=[x[1][mode] for x in m]
        print(mode,'min q25 med q75 max mean',min(a),percentile(a,.25),percentile(a,.5),percentile(a,.75),max(a),sum(a)/len(a),'zeros',a.count(0))
for mode in ('gap_missing','gap_state','all_state'):
    a=[x[1][mode] for x in allm['animals']]; f=[x[1][mode] for x in allm['fungi']]
    ua,uf,rs=ustat(a,f)
    print('ALL',mode,'U_animals',ua,'U_fungi',uf,'minU',min(ua,uf),'rank_sum_animal',rs,'ties',len(a+f)-len(set(a+f)))
sa={x[0].removesuffix('.faa.mafft') for x in allm['animals']}; sf={x[0].removesuffix('.faa.mafft') for x in allm['fungi']}
print('sets',len(sa),len(sf),'intersection',len(sa&sf),'animal-only',sorted(sa-sf),'fungi-only',sorted(sf-sa))
for mode in ('gap_missing','gap_state'):
    da={x[0].removesuffix('.faa.mafft'):x[1][mode] for x in allm['animals']}; df={x[0].removesuffix('.faa.mafft'):x[1][mode] for x in allm['fungi']}
    common=sorted(sa&sf); a=[da[x] for x in common]; f=[df[x] for x in common]
    ua,uf,rs=ustat(a,f)
    print('COMMON',mode,len(common),'U animal',ua,'other',uf,'min',min(ua,uf))
PY
command -v Rscript || true"
