/bin/bash -lc "PYTHONPATH=.local_pk python - <<'PY' | R --vanilla --slave
import io, zipfile
from Bio import AlignIO
from phykit.services.alignment.parsimony_informative_sites import ParsimonyInformative
for group,path in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
    with zipfile.ZipFile(path) as z:
        for name in z.namelist():
            if name.endswith('.faa.mafft'):
                aln=AlignIO.read(io.StringIO(z.read(name).decode()),'fasta')
                val,_,_=ParsimonyInformative.calculate_parsimony_informative_sites(None,aln,True)
                print(group,val)
PY
x <- read.table(file('stdin'), header=FALSE, col.names=c('group','pis'))
a <- x"'$pis[x$group == '"'animals']
f <- x"'$pis[x$group == '"'fungi']
res <- wilcox.test(a, f, paired=FALSE, exact=FALSE, correct=TRUE)
cat('n_animals=', length(a), ' n_fungi=', length(f), '\\n', sep='')
cat('sum_animals=', sum(a), ' sum_fungi=', sum(f), '\\n', sep='')
cat('median_animals=', median(a), ' median_fungi=', median(f), '\\n', sep='')
cat('R_statistic=', unname(res"'$statistic), '"' p=', res"'$p.value, '"'\\n', sep='')
RS"
