/bin/bash -lc "python - <<'PY'
import zipfile, collections
for zpath in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(zpath) as z:
        names=[i.filename for i in z.infolist() if not i.is_dir()]
        ext=collections.Counter('.faa.mafft' if n.endswith('.faa.mafft') else '.faa' if n.endswith('.faa') else 'other' for n in names)
        bases=collections.Counter(n.removesuffix('.mafft') for n in names)
        print(zpath, len(names), ext, 'paired', sum(v==2 for v in bases.values()), 'unique base ids',len(bases), 'dups',[(k,v) for k,v in bases.items() if v"'!=2][:10])
        for n in [x for x in names if x.endswith('"'.mafft')][:3]:
            s=z.read(n).decode()
            heads=[line for line in s.splitlines() if line.startswith('>')]
            seqs=[]
            cur=''
            for line in s.splitlines():
                if line.startswith('>'):
                    if cur: seqs.append(cur)
                    cur=''
                else: cur+=line.strip()
            if cur: seqs.append(cur)
            print(n, 'nseq',len(seqs),'lens',sorted(set(map(len,seqs))),'heads',heads[:5], 'chars',sorted(set(''.join(seqs))))
PY"
