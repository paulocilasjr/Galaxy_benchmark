/bin/bash -lc "python3 - <<'PY'
import zipfile
from collections import Counter
from statistics import mean, median
AA_MISSING_AMBIG=set('BJZX.*-?')

def parse_fasta(text):
    records=[]; name=None; chunks=[]
    for raw in text.splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                records.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else:
            chunks.append(line)
    if name is not None:
        records.append((name,''.join(chunks)))
    return records

def pis_count(records):
    seqs=[s.upper() for _,s in records]
    lens={len(s) for s in seqs}
    if len(lens)"'!=1:
        raise ValueError(lens)
    n=0
    for site in zip(*seqs):
        vals=[c for c in site if c not in AA_MISSING_AMBIG]
        counts=Counter(vals)
        if sum(1 for c in counts if counts[c] >= 2) >= 2:
            n += 1
    return n

def group_counts(zp):
    out=[]
    with zipfile.ZipFile(zp) as z:
        for name in sorted(n for n in z.namelist() if n.endswith('"'.faa.mafft')):
            records=parse_fasta(z.read(name).decode('utf-8'))
            out.append((name,len(records),pis_count(records)))
    return out

def mann_u(x,y):
    vals=x+y
    groups=[0]*len(x)+[1]*len(y)
    order=sorted(range(len(vals)), key=lambda i: vals[i])
    ranks=[0.0]*len(vals)
    i=0
    while i<len(order):
        j=i+1
        while j<len(order) and vals[order[j]]==vals[order[i]]:
            j+=1
        avg=(i+1+j)/2.0
        for k in order[i:j]:
            ranks[k]=avg
        i=j
    n1=len(x); n2=len(y)
    r1=sum(ranks[:n1])
    u1=r1-n1*(n1+1)/2.0
    return u1, n1*n2-u1

a=group_counts('inputs/scogs_animals.zip')
f=group_counts('inputs/scogs_fungi.zip')
av=[v for _,_,v in a]; fv=[v for _,_,v in f]
print('n',len(av),len(fv),'sum',sum(av),sum(fv),'mean',mean(av),mean(fv),'median',median(av),median(fv),'minmax',min(av),max(av),min(fv),max(fv))
print('record_counts animals',Counter(k for _,k,_ in a),'fungi',Counter(k for _,k,_ in f))
print('U animals, fungi',mann_u(av,fv),'product',len(av)*len(fv))
print('zeros',sum(v==0 for v in av),sum(v==0 for v in fv))
print('first10 animal counts',a[:10])
print('first10 fungi counts',f[:10])
PY"
