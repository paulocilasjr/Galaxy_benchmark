/bin/bash -lc "python3 - <<'PY'
import zipfile
from collections import Counter

def count_records(txt):
    return sum(1 for l in txt.splitlines() if l.startswith('>'))
for zp in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    c=Counter()
    examples={}
    with zipfile.ZipFile(zp) as z:
        for n in z.namelist():
            if n.endswith('.faa.mafft'):
                k=count_records(z.read(n).decode())
                c[k]+=1; examples.setdefault(k,n)
    print(zp,c,examples)
PY"
