/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    counts=Counter()
    bad=[]
    with zipfile.ZipFile(z) as f:
        for name in f.namelist():
            if not name.endswith('.mafft'): continue
            text=f.read(name).decode()
            seqs=[]; cur=None
            for line in text.splitlines():
                if not line: continue
                if line.startswith('>'):
                    seqs.append('')
                else:
                    seqs[-1]+=line.strip()
            counts[len(seqs)] += 1
            lens={len(s) for s in seqs}
            if len(lens)"'!=1: bad.append((name,lens))
    print(z, counts, '"'bad alignments', len(bad), bad[:3])
PY"
