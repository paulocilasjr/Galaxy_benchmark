from pathlib import Path
from collections import Counter
import scipy.stats as st

def parse(path):
    seqs=[]; name=None; chars=[]
    for line in path.read_text().splitlines():
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chars)))
            name=line[1:].strip(); chars=[]
        else:
            chars.append(line.strip())
    if name is not None: seqs.append((name,''.join(chars)))
    return seqs

def pis(seqs, gap='-'):
    n=len(seqs)
    if n<2: return 0
    L=len(seqs[0])
    for _,s in seqs:
        if len(s)"'!=L:
            print('"'unequal',len(s),L)
    c=0
    for j in range(L):
        chars=[s[j] for _,s in seqs if s[j]"'!=gap]
        if len(chars)<4: continue
        counts=Counter(chars)
        if len(counts)>=2 and sum(v>=2 for v in counts.values())>=2:
            c+=1
    return c

for scope in ['"'all','common','mafft']:
    pass
for group in ['animals','fungi']:
    files=sorted(Path('/workspace/tmp',group).glob('*.mafft'))
    vals=[]
    for f in files:
        vals.append((f.name,pis(parse(f))))
    Path('/workspace/tmp',f'{group}_pis.tsv').write_text('\\n'.join(f'{a}\\t{b}' for a,b in vals)+'\\n')
    print(group, len(vals), 'mean', sum(b for a,b in vals)/len(vals), 'minmax', min(b for a,b in vals), max(b for a,b in vals), 'zeros', sum(b==0 for a,b in vals))

an={a:b for a,b in (l.split('\\t') for l in Path('/workspace/tmp/animals_pis.tsv').read_text().splitlines())}
fu={a:b for a,b in (l.split('\\t') for l in Path('/workspace/tmp/fungi_pis.tsv').read_text().splitlines())}
common=sorted(set(an)&set(fu))
print('common n',len(common))
for label, x,y in [
    ('all animal vs all fungi', [int(an[k]) for k in an], [int(fu[k]) for k in fu]),
    ('common animal vs common fungi', [int(an[k]) for k in common], [int(fu[k]) for k in common]),
]:
    r=st.mannwhitneyu(x,y,alternative='two-sided')
    print(label, r)
    print('sum ranks x',sum(st.rankdata(x+y)[:len(x)]), 'U', len(x)*len(y)+len(x)*(len(x)+1)/2-sum(st.rankdata(x+y)[:len(x)]))
    # also alternative greater? 
    for alt in ['two-sided','less','greater']:
        print(alt, st.mannwhitneyu(x,y,alternative=alt).statistic)
PY"
