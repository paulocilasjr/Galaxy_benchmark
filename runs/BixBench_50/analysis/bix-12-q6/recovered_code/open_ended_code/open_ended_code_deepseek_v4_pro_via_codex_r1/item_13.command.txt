/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main?recursive=true&expand=false'
try:
    with urllib.request.urlopen(url, timeout=30) as r:
        data=json.load(r)
    print(len(data))
    for item in data[:50]:
        print(item.get('path'), item.get('type'), item.get('size'))
except Exception as e:
    print('ERR', type(e), e)
PY"
