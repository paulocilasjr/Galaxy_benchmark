/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter
x=list(map(int,Path('/workspace/tmp/animals_counts.txt').read_text().split()))
y=list(map(int,Path('/workspace/tmp/fungi_counts.txt').read_text().split()))
# rankdata average manually
combined=x+y
order=sorted(combined)
ranks={}
i=0
while i<len(order):
 j=i+1
 while j<len(order) and order[j]==order[i]: j+=1
 avg=(i+1+j)/2  # average of positions i+1..j
 for k in range(i,j): ranks[(order[k],k-i)] = avg # wrong key
 i=j
# simpler use sorted combined and average ranks by value, scipy-like
from collections import defaultdict
groups=defaultdict(list)
for idx,v in enumerate(sorted(combined),1): groups[v].append(idx)
avg={v:sum(pos)/len(pos) for v,pos in groups.items()}
rankx=sum(avg[v] for v in x)
ranky=sum(avg[v] for v in y)
n=len(x);m=len(y)
U1=rankx-n*(n+1)/2
U2=ranky-m*(m+1)/2
print('rankx',rankx,'ranky',ranky)
print('U1 animals',U1,'U2 fungi',U2,'min',min(U1,U2),'sum',U1+U2,'n*m',n*m)
PY"
