from pathlib import Path
from scipy.stats import mannwhitneyu
x=list(map(int,Path('/workspace/tmp/animals_counts.txt').read_text().split()))

 for method in ['auto','exact','asymptotic']:
  try:
   res=mannwhitneyu(x,y,alternative=alt,method=method)
   print(alt,method,res.statistic,res.pvalue)
  except Exception as e: print(e)
print('reverse')
for alt in ['two-sided','less','greater']:
 res=mannwhitneyu(y,x,alternative=alt,method='auto')
 print(alt,res.statistic,res.pvalue)
