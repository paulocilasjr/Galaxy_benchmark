/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse(d):
 r=[];b=[];h=None
 for l in d.decode().splitlines():
  if l.startswith('>'):
   if h is not None:r.append(''.join(b))
   h=l;b=[]
  else:b.append(l.strip())
 if h is not None:r.append(''.join(b))
 return r

def pi(s, gap=False):
 k=0
 for col in zip(*s):
  if gap and '-' in col: continue
  c=Counter(col if gap else [x for x in col if x"'!='"'-'])
  if sum(v>=2 for v in c.values())>=2:k+=1
 return k

def u(a,b):
 z=sorted([(x,0) for x in a]+[(x,1) for x in b]); r=[0,0];i=0
 while i<len(z):
  j=i+1
  while j<len(z) and z[j][0]==z[i][0]:j+=1
  for k in range(i,j):r[z[k][1]]+=(i+j+1)/2
  i=j
 return r[0]-len(a)*(len(a)+1)/2, r[1]-len(b)*(len(b)+1)/2
for nfilter in [None,4]:
 for gap in [False,True]:
  out=[]
  for f in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
   a=[]
   with ZipFile(f) as z:
    for n in z.namelist():
     if n.endswith('.faa.mafft'):
      s=parse(z.read(n))
      if nfilter is None or len(s)==nfilter:a.append(pi(s,gap))
   out.append(a)
  print('nfilter',nfilter,'gapflag',gap,'sizes',list(map(len,out)),'u',u(*out))
PY"
