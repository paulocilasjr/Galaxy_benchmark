from zipfile import ZipFile
from collections import Counter
from math import isnan


def parse(d):
    rec=[]; name=None; seq=[]
    for line in d.decode().splitlines():
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(seq)))
            name=line[1:].split()[0]; seq=[]
        else: seq.append(line.strip())
    if name is not None: rec.append((name,''.join(seq)))
    return rec

def count_pi(seqs, mode):
    if not seqs: return 0
    L=len(seqs[0])
    c=0
    for col in zip(*seqs):
        if mode=='allchars':
            vals=list(col)
        elif mode=='ignore_gaps':
            vals=[x for x in col if x not in '-.?']
        elif mode=='nogap_columns':
            if any(x in '-.?' for x in col): continue
            vals=list(col)
        elif mode=='nongap_only_charcount':
            vals=[x for x in col if x"'!='"'-']
        counts=Counter(vals)
        if sum(v>=2 for v in counts.values())>=2: c+=1
    return c

def u_stats(x,y):
    # exact average midranks
    a=[(v,0) for v in x]+[(v,1) for v in y]
    a.sort()
    ranks=[None]*len(a); i=0; rank=1
    while i<len(a):
        j=i+1
        while j<len(a) and a[j][0]==a[i][0]: j+=1
        mid=(i+1+j)/2
        for k in range(i,j): ranks[k]=mid
        i=j
    rx=sum(r for (_,g),r in zip(a,ranks) if g==0)
    u1=rx-len(x)*(len(x)+1)/2
    u2=len(x)*len(y)-u1
    return u1,u2,min(u1,u2),rx
for mode in ['allchars','ignore_gaps','nogap_columns','nongap_only_charcount']:
