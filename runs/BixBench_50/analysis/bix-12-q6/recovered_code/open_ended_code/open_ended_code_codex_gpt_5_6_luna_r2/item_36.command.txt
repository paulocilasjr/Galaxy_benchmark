from zipfile import ZipFile
from collections import Counter

def parse(d):
 r=[]; b=[]; n=None
 for l in d.decode().splitlines():
  if l.startswith('>'):
   if n is not None:r.append((n,''.join(b)))
   n=l[1:].split()[0];b=[]
  else:b.append(l.strip())
 if n is not None:r.append((n,''.join(b)))
 return r
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 print('\\n',fn)
 with ZipFile(fn) as z:
  d=[]
  for n in z.namelist():
   if n.endswith('.faa.mafft'):
    s=parse(z.read(n));
    if len(s)==4:
     raw=ign=0; types=Counter()
     for col in zip(*(x[1] for x in s)):
      c=Counter(col)
      if sum(v>=2 for v in c.values())>=2:
       raw+=1
       types['gap' if '-' in c else 'nogap']+=1
      cg=Counter(x for x in col if x"'!='"'-')
      if sum(v>=2 for v in cg.values())>=2:ign+=1
     d.append((raw,ign,n,types))
  print('n4',len(d),'raw sum',sum(x[0] for x in d),'ign sum',sum(x[1] for x in d),'extra',sum(x[0]-x[1] for x in d),'alignments extra',sum(x[0]"'!=x[1] for x in d))
  print('"'extra example',[(a,b,n,t) for a,b,n,t in d if a"'!=b][:10])
PY'
