/bin/bash -lc "python - <<'PY'
import zipfile, collections, statistics
from scipy.stats import mannwhitneyu, rankdata

zs=[('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]
results={}
for label,path in zs:
    vals=[]; vals_gapstate=[]; lens=[]; nseqs=[]; bad=[]; alph=collections.Counter()
    with zipfile.ZipFile(path) as z:
        members=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
        for n in members:
            text=z.read(n).decode()
            seqs=[]; cur=[]
            for line in text.splitlines():
                if line.startswith('>'):
                    if cur: seqs.append(''.join(cur)); cur=[]
                else: cur.append(line.strip().upper())
            if cur: seqs.append(''.join(cur))
            nseqs.append(len(seqs))
            if not seqs or len(set(map(len,seqs))) "'!= 1:
                bad.append(n); continue
            L=len(seqs[0]); lens.append(L)
            pi=0; pig=0
            for col in zip(*seqs):
                alph.update(col)
                c=collections.Counter(x for x in col if x not in '"'-.?X*')
                if sum(v>=2 for v in c.values()) >= 2: pi+=1
                cg=collections.Counter(col)
                if sum(v>=2 for v in cg.values()) >= 2: pig+=1
            vals.append(pi); vals_gapstate.append(pig)
    results[label]=vals
    print(label,'members',len(members),'processed',len(vals),'bad',len(bad),'nseq distribution',collections.Counter(nseqs))
    print('length min median max',min(lens),statistics.median(lens),max(lens))
    print('PI ignore missing/gap min median mean max',min(vals),statistics.median(vals),statistics.mean(vals),max(vals),'sum',sum(vals))
    print('PI gap-state min median mean max',min(vals_gapstate),statistics.median(vals_gapstate),statistics.mean(vals_gapstate),max(vals_gapstate),'sum',sum(vals_gapstate))
    print('alphabet',dict(alph))

x,y=results['animals'],results['fungi']
for method in ['auto','asymptotic','exact']:
    z=mannwhitneyu(x,y,alternative='two-sided',method=method)
    print('scipy',method,'U animals-first',z.statistic,'p',z.pvalue)
# manual rank U and tie counts
allv=x+y
r=rankdata(allv,method='average')
u1=sum(r[:len(x)])-len(x)*(len(x)+1)/2
u2=len(x)*len(y)-u1
print('manual U1 U2 min',u1,u2,min(u1,u2),'n1*n2',len(x)*len(y),'distinct',len(set(allv)),'tie groups',sum(v>1 for v in collections.Counter(allv).values()))
PY"
