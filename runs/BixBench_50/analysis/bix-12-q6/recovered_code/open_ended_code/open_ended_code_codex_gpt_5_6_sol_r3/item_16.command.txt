/bin/bash -lc "Rscript - <<'RS'
pi_counts <- function(zipfile) {
  listing <- unzip(zipfile, list=TRUE)
  keep <- grepl('\\\\.faa\\\\.mafft"'$'"', listing"'$Name)
  members <- listing[keep, ]
  members <- members[order(members$Name), ]
  ans <- numeric(nrow(members))
  for (j in seq_len(nrow(members))) {
    con <- unz(zipfile, members$Name[j], open='"'rb')
    txt <- rawToChar(readBin(con, what='raw', n=members"'$Length[j]))
    close(con)
    lines <- strsplit(txt, '"'\\n', fixed=TRUE)[[1]]
    lines <- sub('\\r"'$'"', '', lines)
    heads <- which(startsWith(lines, '>'))
    ends <- c(heads[-1] - 1L, length(lines))
    seqs <- vapply(seq_along(heads), function(i) paste0(lines[(heads[i]+1L):ends[i]], collapse=''), character(1))
    mat <- do.call(rbind, strsplit(seqs, '', fixed=TRUE))
    ans[j] <- sum(apply(mat, 2, function(site) {
      tab <- table(site[site "'!= '"'-'])
      sum(tab >= 2L) >= 2L
    }))
  }
  ans
}
a <- pi_counts('inputs/scogs_animals.zip')
f <- pi_counts('inputs/scogs_fungi.zip')
r <- rank(c(a,f), ties.method='average')
cat('n:', length(a), length(f), '\\n')
cat('rank sum animals:', sum(r[seq_along(a)]), '\\n')
cat('manual U:', sum(r[seq_along(a)]) - length(a)*(length(a)+1)/2, '\\n')
print(wilcox.test(a, f, alternative='two.sided', exact=FALSE, correct=FALSE))
RS"
