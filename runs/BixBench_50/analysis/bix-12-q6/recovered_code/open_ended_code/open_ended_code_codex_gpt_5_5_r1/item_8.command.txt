/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with ZipFile(fn) as z:
        names=z.namelist()
        ma=[n for n in names if n.endswith('.mafft')]
        raw=[n for n in names if n.endswith('.faa') and not n.endswith('.mafft')]
        print(fn, 'members', len(names), 'mafft', len(ma), 'raw', len(raw), 'unique base mafft', len(set(n.replace('.faa.mafft','') for n in ma)))
        print('first mafft', ma[:5])
PY"
