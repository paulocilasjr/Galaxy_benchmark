/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
VALID=set('ACDEFGHIKLMNPQRSTVWY')

def read_fasta(data):
    seqs=[]; name=None; parts=[]
    for line in data.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(parts)))
            name=line[1:].strip(); parts=[]
        else: parts.append(line)
    if name is not None: seqs.append((name,''.join(parts)))
    return seqs

def raw_pi_count(seqs):
    if not seqs: return 0
    lens={len(s) for _,s in seqs}
    if len(lens)"'!=1: raise ValueError(lens)
    count=0
    for col in zip(*(s for _,s in seqs)):
        states=Counter(c.upper() for c in col if c.upper() in VALID)
        if sum(1 for n in states.values() if n >= 2) >= 2:
            count += 1
    return count

def archive_records(path):
    rec=[]
    with ZipFile(path) as z:
        for name in sorted(n for n in z.namelist() if n.endswith('"'.faa.mafft')):
            seqs=read_fasta(z.read(name))
            rec.append((name,len(seqs),raw_pi_count(seqs)))
    return rec

def U(x,y):
    u=0.0
    for xv in x:
        for yv in y:
            u += (xv>yv) + 0.5*(xv==yv)
    return u
ar=archive_records('inputs/scogs_animals.zip')
fr=archive_records('inputs/scogs_fungi.zip')
for label, afilt, ffilt in [
    ('all', ar, fr),
    ('n>=4', [r for r in ar if r[1]>=4], [r for r in fr if r[1]>=4]),
    ('common all', [r for r in ar if r[0] in {x[0] for x in fr}], [r for r in fr if r[0] in {x[0] for x in ar}]),
    ('common n>=4 both', [], []),
]:
    if label=='common n>=4 both':
        da={r[0]:r for r in ar}; df={r[0]:r for r in fr}
        ids=sorted(k for k in da.keys() & df.keys() if da[k][1]>=4 and df[k][1]>=4)
        afilt=[da[k] for k in ids]; ffilt=[df[k] for k in ids]
    x=[r[2] for r in afilt]; y=[r[2] for r in ffilt]
    print(label, len(x), len(y), U(x,y), U(y,x), 'means', (sum(x)/len(x) if x else None), (sum(y)/len(y) if y else None))
PY"
