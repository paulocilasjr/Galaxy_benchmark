from zipfile import ZipFile
from collections import Counter

def names(fn):
    with ZipFile(fn) as z:
        return set(n for n in z.namelist() if n.endswith('.mafft'))
a=names('inputs/scogs_animals.zip'); f=names('inputs/scogs_fungi.zip')
print(len(a), len(f), 'overlap', len(a&f), 'animal_not_fungi', len(a-f), 'fungi_not_animal', len(f-a))
print('animal_not_fungi', sorted(a-f)[:20])
print('fungi_not_animal', sorted(f-a)[:30])
PY"
