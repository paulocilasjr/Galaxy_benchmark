from zipfile import ZipFile
from collections import Counter
from math import isnan

VALID_AA=set('ACDEFGHIKLMNPQRSTVWY')
MISSING=set('-?Xx*')

def read_fasta_bytes(b):
    seqs=[]; name=None; parts=[]
    for line in b.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                seqs.append((name,''.join(parts)))
            name=line[1:].strip().split()[0]
            parts=[]
        else:
            parts.append(line)
    if name is not None:
        seqs.append((name,''.join(parts)))
    return seqs

def pi_count(seqs, mode='ignore_gaps'):
    lengths={len(s) for _,s in seqs}
    if len(lengths)"'!=1:
        raise ValueError(lengths)
    n=0
    for col in zip(*(s for _,s in seqs)):
        if mode=='"'ignore_gaps':
            chars=[c.upper() for c in col if c.upper() in VALID_AA]
        elif mode=='gap_state':
            chars=[c.upper() for c in col if c not in '?Xx*']
        elif mode=='all_state':
            chars=[c.upper() for c in col]
        counts=Counter(chars)
        if sum(1 for v in counts.values() if v>=2) >= 2:
            n+=1
    return n

def counts(zipfn, mode):
    vals=[]; meta=[]
    with ZipFile(zipfn) as z:
        for name in sorted(n for n in z.namelist() if n.endswith('.mafft')):
            seqs=read_fasta_bytes(z.read(name))
            c=pi_count(seqs, mode)
            vals.append(c); meta.append((name,c,len(seqs),len(seqs[0][1]) if seqs else 0))
    return vals, meta

def mannwhitney_u(x,y):
    # U for first sample x, average ranks for ties
    vals=[(v,0) for v in x]+[(v,1) for v in y]
    vals.sort(key=lambda t:t[0])
    ranks=[0.0]*len(vals)
    i=0
    while i<len(vals):
        j=i+1
        while j<len(vals) and vals[j][0]==vals[i][0]: j+=1
        avg=(i+1+j)/2.0
        for k in range(i,j): ranks[k]=avg
        i=j
    R1=sum(r for r,(v,g) in zip(ranks, vals) if g==0)
    n1=len(x)
    return R1 - n1*(n1+1)/2.0
