# Manually verify PIS counting on a few example alignments
import os

def parse_fasta_alignment(filepath):
    sequences = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_header is not None:
                    sequences.append((current_header, ''.join(current_seq)))
                current_header = line
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            sequences.append((current_header, ''.join(current_seq)))
    return sequences

def count_pis(sequences):
    STANDARD_AA = set('ACDEFGHIKLMNPQRSTVWY')
    n_seqs = len(sequences)
    seqs = [s for _, s in sequences]
    aln_len = min(len(s) for s in seqs)
    pis_count = 0
    for pos in range(aln_len):
        chars = [seqs[i][pos] for i in range(n_seqs)]
        freq = {}
        for c in chars:
            if c in STANDARD_AA:
                freq[c] = freq.get(c, 0) + 1
        states_ge2 = sum(1 for cnt in freq.values() if cnt >= 2)
        if states_ge2 >= 2:
            pis_count += 1
    return pis_count

# Check a small alignment with known PIS
animal_dir = '/tmp/scogs_animals'
# Find a short alignment
files = [f for f in os.listdir(animal_dir) if f.endswith('.mafft')]
files.sort(key=lambda f: os.path.getsize(os.path.join(animal_dir, f)))

# Check the 3 smallest alignments manually
for fname in files[:3]:
    filepath = os.path.join(animal_dir, fname)
    seqs = parse_fasta_alignment(filepath)
    aln_len = min(len(s) for _, s in seqs)
    pis = count_pis(seqs)
    print(f"{fname}: alignment length={aln_len}, PIS={pis}")

# Also look at one with moderate PIS
for fname in files:
    filepath = os.path.join(animal_dir, fname)
    seqs = parse_fasta_alignment(filepath)
    pis = count_pis(seqs)
    if 5 <= pis <= 15:
        print(f"Moderate PIS: {fname}: PIS={pis}")
        break

# Check first few fungi alignments too
fungi_dir = '/tmp/scogs_fungi'
ffiles = [f for f in os.listdir(fungi_dir) if f.endswith('.mafft')]
ffiles.sort(key=lambda f: os.path.getsize(os.path.join(fungi_dir, f)))
for fname in ffiles[:3]:
    filepath = os.path.join(fungi_dir, fname)
    seqs = parse_fasta_alignment(filepath)
    aln_len = min(len(s) for _, s in seqs)
    pis = count_pis(seqs)
    print(f"{fname}: alignment length={aln_len}, PIS={pis}")
