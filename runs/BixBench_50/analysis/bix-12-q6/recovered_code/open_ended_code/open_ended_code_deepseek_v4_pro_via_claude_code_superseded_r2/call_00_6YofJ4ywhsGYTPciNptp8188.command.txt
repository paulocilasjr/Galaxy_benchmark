import zipfile

# Check which species are missing from orthologs with fewer than 4 sequences
with zipfile.ZipFile('/workspace/inputs/scogs_animals.zip', 'r') as zf:
    for name in sorted(zf.namelist()):
        if name.endswith('.faa.mafft'):
            content = zf.read(name).decode('utf-8')
            n_seqs = content.count('>')
            if n_seqs < 4:
                species = []
                for line in content.strip().split('\n'):
                    if line.startswith('>'):
                        species.append(line[1:].strip())
                ortho = name.replace('.faa.mafft','').split('/')[-1]
                print(f'{ortho}: {n_seqs} seqs, present={species}')
                if n_seqs == 1:
                    break  # just show first example of each count
