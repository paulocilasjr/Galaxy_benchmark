from pathlib import Path
from collections import Counter

def read_align(path):
    seqs=[]; cur=[]; name=None
    for line in open(path):
        line=line.rstrip('\\n')
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(cur))
            name=line[1:]; cur=[]
        else: cur.append(line.strip())
    if name is not None: seqs.append(''.join(cur))
    return seqs

def pic(seqs):
    L=min(map(len,seqs)) if seqs else 0
    count=0
    for col in zip(*seqs):
        counts=Counter(c for c in col if c not in '-?.')
        if sum(v>=2 for v in counts.values()) >=2:
            count+=1
    return count
arrs={}
for group in ['animals','fungi']:
    arr=[]
    for f in sorted(Path(group).glob('*.mafft')):
        seqs=read_align(f); arr.append(pic(seqs))
    arrs[group]=arr
    print(group, len(arr), sum(arr), min(arr), max(arr))
    print('top', sorted(arr, reverse=True)[:20])
    print('zeros', arr.count(0))
print('counts animals/fungi first 30')
