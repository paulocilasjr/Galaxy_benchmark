# Verify Mann-Whitney U implementation
def mannwhitneyu_manual(x, y):
    n1, n2 = len(x), len(y)
    combined = [(v, 0) for v in x] + [(v, 1) for v in y]
    combined.sort(key=lambda t: t[0])
    
    ranks = []
    i = 0
    while i < len(combined):
        j = i
        while j < len(combined) and combined[j][0] == combined[i][0]:
            j += 1
        avg_rank = (i + j + 1) / 2.0
        for k in range(i, j):
            ranks.append((avg_rank, combined[k][1]))
        i = j
    
    r1 = sum(r for r, group in ranks if group == 0)
    u1 = r1 - n1 * (n1 + 1) / 2.0
    u2 = n1 * n2 - u1
    return min(u1, u2)

# Test: x all smaller than y
x = [1, 2, 3]
y = [4, 5, 6]
print(f"x < y: U = {mannwhitneyu_manual(x, y)} (expect close to 0)")

# Test: x all larger than y  
x = [4, 5, 6]
y = [1, 2, 3]

# Test: perfect overlap
x = [1, 2, 3]
y = [1, 2, 3]
print(f"identical: U = {mannwhitneyu_manual(x, y)} (expect n1*n2/2 = 4.5)")
