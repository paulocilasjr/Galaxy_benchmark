import csv, collections, math
expr='inputs/ROSMAP_genexp_ad.csv'; meta='inputs/ROSMAP_meta_ad.csv'
with open(expr,newline='') as f:
    rd=csv.reader(f); hdr=next(rd); genes=[]; missing=0; nonnum=[]; nrows=0
    for row in rd:
        nrows+=1; genes.append(row[0])
        if len(row)"'!=len(hdr): print('"'bad row',nrows,len(row))
        for x in row[1:]:
            if x=='': missing+=1
            else:
                try:
                    v=float(x)
                    if not math.isfinite(v): nonnum.append((nrows,x))
                except: nonnum.append((nrows,x))
samples=hdr[1:]
with open(meta,newline='') as f:
    md=list(csv.DictReader(f)); mcols=list(md[0])
ids=[r['projid'] for r in md]
def dups(x): return {k:v for k,v in collections.Counter(x).items() if v>1}
print('expression matrix genes x samples',nrows,len(samples))
print('sample total unique duplicates',len(samples),len(set(samples)),dups(samples))
print('gene total unique duplicate groups',len(genes),len(set(genes)),dups(genes))
print('metadata rows unique IDs duplicates',len(ids),len(set(ids)),dups(ids))
print('expr-only samples',sorted(set(samples)-set(ids)))
print('meta-only IDs',sorted(set(ids)-set(samples)))
print('missing expr',missing,'nonfinite/bad',nonnum[:10], 'count',len(nonnum))
for k in dups(ids):
    print('META DUP GROUP',k)
    for r in md:
        if r['projid']==k: print(r)
PY"
