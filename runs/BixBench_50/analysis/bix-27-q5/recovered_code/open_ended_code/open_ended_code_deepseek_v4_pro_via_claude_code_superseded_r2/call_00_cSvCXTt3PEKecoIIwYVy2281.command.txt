import csv
import numpy as np
from sklearn.decomposition import PCA
from collections import Counter

# Read and filter data
with open('inputs/ROSMAP_genexp_ad.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    all_sample_ids = header[1:]
    raw_data = []
    for row in reader:
        raw_data.append([float(x) for x in row[1:]])

raw_matrix = np.array(raw_data, dtype=np.float64)
id_counts = Counter(all_sample_ids)
repeated_ids = {k for k, v in id_counts.items() if v > 1}
keep_mask = np.array([sid not in repeated_ids for sid in all_sample_ids])
filtered = raw_matrix[:, keep_mask]
expr = filtered.T
log_expr = np.log10(expr + 1.0)

# PCA with 100 components
pca_100 = PCA(n_components=100)
pca_100.fit(log_expr)

# PCA with all components
pca_full = PCA()
pca_full.fit(log_expr)

# Detailed comparison
ev_100 = pca_100.explained_variance_        # eigenvalues for 100 PCs
ev_full = pca_full.explained_variance_      # eigenvalues for all PCs

evr_100 = pca_100.explained_variance_ratio_  # ratios for 100 PCs
evr_full = pca_full.explained_variance_ratio_  # ratios for all PCs

print(f"Sum of eigenvalues (100 PCs): {ev_100.sum():.10f}")
print(f"Sum of eigenvalues (all PCs): {ev_full.sum():.10f}")
print(f"Total variance (ddof=1): {np.var(log_expr, axis=0, ddof=1).sum():.10f}")

print(f"\nev_100[0] = {ev_100[0]:.10f}")
print(f"ev_full[0] = {ev_full[0]:.10f}")
print(f"ev_100[0] == ev_full[0]: {np.isclose(ev_100[0], ev_full[0])}")

print(f"\nevr_100[0] = {evr_100[0]:.12f}")
print(f"evr_full[0] = {evr_full[0]:.12f}")
print(f"evr_100[0] == evr_full[0]: {np.isclose(evr_100[0], evr_full[0])}")

# How is evr computed?
print(f"\nevr_100[0] computation:")
print(f"  ev_100[0] / ev_100.sum() = {ev_100[0] / ev_100.sum():.12f}")
print(f"  ev_100[0] / ev_full.sum() = {ev_100[0] / ev_full.sum():.12f}")
print(f"  Actual evr_100[0] = {evr_100[0]:.12f}")

print(f"\nevr_full[0] computation:")
print(f"  ev_full[0] / ev_full.sum() = {ev_full[0] / ev_full.sum():.12f}")
print(f"  Actual evr_full[0] = {evr_full[0]:.12f}")

# So sklearn evr[i] = ev[i] / sum(ev) where sum(ev) is over stored components
# For 100 components, sum(ev) = 0.9928 * total_variance
# For all components, sum(ev) = total_variance
print(f"\nRatio of denominators: {ev_100.sum() / ev_full.sum():.10f}")
print(f"Expected difference in PC1%: {evr_100[0]*100:.8f}% vs {evr_full[0]*100:.8f}%")
