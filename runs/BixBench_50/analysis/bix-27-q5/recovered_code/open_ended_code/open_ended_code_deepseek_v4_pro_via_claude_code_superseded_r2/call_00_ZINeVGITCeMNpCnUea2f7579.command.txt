import csv
with open('inputs/ROSMAP_meta_ad.csv') as f:
    reader = csv.DictReader(f)
    rows = list(reader)

# For each repeated ID, show all records
from collections import defaultdict
by_id = defaultdict(list)
for r in rows:
    by_id[r['projid']].append(r)

for pid in sorted(by_id.keys()):
    if len(by_id[pid]) > 1:
        print(f'=== {pid} ({len(by_id[pid])} records) ===')
        for i, r in enumerate(by_id[pid]):
            print(f'  record {i}: msex={r[\"msex\"]}, educ={r[\"educ\"]}, apoe={r[\"apoe_genotype\"]}, mmse={r[\"cts_mmse30_lv\"]}, braak={r[\"braaksc\"]}, cerad={r[\"ceradsc\"]}, cogdx={r[\"cogdx\"]}, dcfdx={r[\"dcfdx_lv\"]}')
        # Check invariants
        sexes = set(r['msex'] for r in by_id[pid])
        educs = set(r['educ'] for r in by_id[pid])
        apoes = set(r['apoe_genotype'] for r in by_id[pid])
        if len(sexes) > 1:
            print(f'  *** SEX CONFLICT: {sexes}')
        if len(educs) > 1:
            print(f'  *** EDUC CONFLICT: {educs}')
        if len(apoes) > 1:
            print(f'  *** APOE CONFLICT: {apoes}')
        print()
