/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    r=csv.reader(f)
    header=next(r)
    n_genes=sum(1 for _ in r)
ids=header[1:]
counts=Counter(ids)
dups={k:v for k,v in counts.items() if v>1}
print('expression genes', n_genes)
print('expression sample columns', len(ids))
print('expression unique sample ids', len(counts))
print('duplicate ids count', len(dups))
print('duplicate ids', sorted(dups.items()))
with open('inputs/ROSMAP_meta_ad.csv', newline='') as f:
    mr=csv.DictReader(f)
    rows=list(mr)
meta_ids=[row['projid'] for row in rows]
mc=Counter(meta_ids)
print('metadata rows', len(rows))
print('metadata unique ids', len(mc))
print('metadata duplicate ids', sorted((k,v) for k,v in mc.items() if v>1))
print('expr ids missing meta', sorted(set(counts)-set(mc))[:20], len(set(counts)-set(mc)))
print('meta ids missing expr', sorted(set(mc)-set(counts))[:20], len(set(mc)-set(counts)))
PY"
