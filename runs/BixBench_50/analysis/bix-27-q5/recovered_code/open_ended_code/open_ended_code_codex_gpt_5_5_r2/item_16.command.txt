/bin/bash -lc "Rscript - <<'RS'
expr <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE)
meta <- read.csv('inputs/ROSMAP_meta_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
ids <- names(expr)[-1]
dup_ids <- names(which(table(ids) > 1))
# Repeated IDs disagree on stable attributes and have no secondary key; exclude entire collided groups.
keep <- "'!(ids %in% dup_ids)
mat_genes_by_samples <- as.matrix(expr[, c(FALSE, keep), drop=FALSE])
storage.mode(mat_genes_by_samples) <- '"'double'
rownames(mat_genes_by_samples) <- expr[[1]]
X <- t(log10(mat_genes_by_samples + 1))
cat('original columns:', length(ids), '\\n')
cat('duplicate id groups:', length(dup_ids), '\\n')
cat('fitted samples:', nrow(X), '\\n')
cat('genes:', ncol(X), '\\n')
cat('NA count:', sum(is.na(X)), '\\n')
vars <- apply(X, 2, var)
cat('zero variance genes:', sum(vars == 0), '\\n')
pca <- prcomp(X, center=TRUE, scale.=FALSE, rank.=100)
pc1_percent <- (pca"'$sdev[1]''^2 / sum(vars)) * 100
cat(sprintf('"'pc1_percent=%.15f\\n', pc1_percent))
# Compare with full prcomp denominator from full sdev as a sanity check.
pca_full <- prcomp(X, center=TRUE, scale.=FALSE)
pc1_percent_full <- (pca_full"'$sdev[1]''^2 / sum(pca_full$sdev''^2)) * 100
cat(sprintf('"'pc1_percent_full=%.15f\\n', pc1_percent_full))
RS"
