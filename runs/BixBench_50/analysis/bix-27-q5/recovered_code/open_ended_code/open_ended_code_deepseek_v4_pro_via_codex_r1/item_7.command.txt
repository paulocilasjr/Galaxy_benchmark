/bin/bash -c "python3 - <<'PY'
import csv, collections
meta_path='/workspace/inputs/ROSMAP_meta_ad.csv'
expr_path='/workspace/inputs/ROSMAP_genexp_ad.csv'
with open(meta_path, newline='') as f:
    r=csv.reader(f)
    header=next(r)
    meta_rows=list(r)
print('meta fields', header)
print('meta rows', len(meta_rows))
meta_ids=[row[0] for row in meta_rows]
print('meta ids', len(meta_ids), 'unique', len(set(meta_ids)))
print('meta id first', meta_ids[:20])
print('meta id last', meta_ids[-10:])
# duplicates metadata?
dup=[x for x,c in collections.Counter(meta_ids).items() if c>1]
print('meta duplicates', dup)
with open(expr_path, newline='') as f:
    r=csv.reader(f)
    h=next(r)
print('expr header cells', len(h), 'gene', h[0])
ids=h[1:]
print('expr sample cols', len(ids), 'unique', len(set(ids)))
cnt=collections.Counter(ids)
dups={x:c for x,c in cnt.items() if c>1}
print('num duplicate IDs', len(dups), 'dup occurrences', sum(c-1 for c in dups.values()))
print('max mult', max(dups.values()))
print('dups', dups)
print('top dups', sorted(dups.items(), key=lambda kv:-kv[1])[:50])
# compare IDs
expr_set=set(ids)
meta_set=set(meta_ids)
print('meta not expr', len(meta_set-expr_set), sorted(meta_set-expr_set))
print('expr not meta', len(expr_set-meta_set), sorted(expr_set-meta_set))
# all expr unique IDs count
print('expr unique total', len(expr_set))
PY"
