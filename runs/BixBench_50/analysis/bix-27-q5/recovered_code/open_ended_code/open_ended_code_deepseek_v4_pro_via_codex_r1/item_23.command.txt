/bin/bash -c "python3 - <<'PY'
import urllib.request, re
url='https://datadryad.org/dataset/doi:10.5061/dryad.ttdz08kxq'
req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
html=urllib.request.urlopen(req, timeout=30).read().decode('utf-8')
print(len(html))
for pat in ['ROSMAP', 'genexp', '.csv', 'download']:
 print(pat, html.find(pat))
# print around ROSMAP
for m in re.finditer('ROSMAP', html):
    s=max(0,m.start()-200); e=min(len(html),m.end()+500)
    print(html[s:e])
    print('---')
PY"
