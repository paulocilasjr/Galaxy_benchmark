/bin/bash -c "python3 - <<'PY'
import urllib.request
url='https://datadryad.org/downloads/file_stream/672449'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0','Referer':'https://datadryad.org/dataset/doi:10.5061/dryad.ttdz08kxq'})
try:
    with urllib.request.urlopen(req,timeout=30) as r:
        data=r.read()
    print('status', getattr(r,'status',None), 'len',len(data))
    print(data[:5000].decode('utf-8','replace'))
except Exception as e:
    print('ERR',e)
    if hasattr(e,'read'):
        print(e.read().decode('utf-8','replace')[:1000])
PY"
