/bin/bash -c "python3 - <<'PY'
import urllib.request,re
url='https://datadryad.org/dataset/doi:10.5061/dryad.ttdz08kxq'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
html=urllib.request.urlopen(req,timeout=30).read().decode()
# find all links with file names in download area
for m in re.finditer(r'<a["'^>]*href="([''^"]*)"[''^>]*>(.*?)</a>'"', html, flags=re.S):
    href,text=m.groups()
    text=re.sub('<["'^>]+>'"','',text).strip()
    if '.csv' in text.lower() or 'readme' in text.lower() or 'README' in text:
        print(repr(text), href)
PY"
