/bin/bash -c "python3 - <<'PY'
import csv, collections
meta_path='/workspace/inputs/ROSMAP_meta_ad.csv'
with open(meta_path, newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
by=collections.defaultdict(list)
for row in rows: by[row[0]].append(row)
dups=sorted([(k,v) for k,v in by.items() if len(v)>1], key=lambda kv:kv[0])
print('header', h)
for k,v in dups:
    print('\\nID',k,'count',len(v))
    for row in v:
        print(row)
PY"
