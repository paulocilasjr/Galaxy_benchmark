/bin/bash -c "python3 - <<'PY'
import csv, collections, numpy as np
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
X=np.array([[float(x) for x in row[1:]] for row in rows],float).T
Y=np.log10(X+1)
ids=h[1:]
groups=collections.defaultdict(list)
for i,pid in enumerate(ids): groups[pid].append(i)
# standardize rows (genes columns) for correlation-ish dot
Yc=Y-Y.mean(axis=1, keepdims=True)
norms=np.linalg.norm(Yc, axis=1)
Z=Yc/norms[:,None]
same=[]; out=[]
# sample random unrelated pairs
rng=np.random.default_rng(0)
# all same id pairs
for pid, idx in groups.items():
    if len(idx)>1:
        for a in range(len(idx)):
            for b in range(a+1,len(idx)):
                same.append(Z[idx[a]]@Z[idx[b]])
# random out from unique groups selecting singles only maybe all random
uids=list(groups.keys())
for _ in range(1000):
    p1,p2=rng.choice(uids, size=2, replace=False)
    i=rng.choice(groups[p1]); j=rng.choice(groups[p2])
    out.append(Z[i]@Z[j])
print('same id n',len(same),'mean',np.mean(same),'min',np.min(same),'max',np.max(same))
print('unrelated n',len(out),'mean',np.mean(out),'min',np.min(out),'max',np.max(out))
PY"
