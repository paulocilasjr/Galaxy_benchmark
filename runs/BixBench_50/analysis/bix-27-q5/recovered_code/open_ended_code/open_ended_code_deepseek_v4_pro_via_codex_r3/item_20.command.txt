/bin/bash -c "python3 - <<'PY'
import csv, numpy as np
from sklearn.decomposition import PCA
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
X=np.array([[float(x) for x in row[1:]] for row in rows], float).T
Y=np.log10(X+1)
for solver in ['auto','full','randomized']:
    pca=PCA(n_components=100, svd_solver=solver, random_state=0)
    pca.fit(Y)
    print(solver, 'ratio0', pca.explained_variance_ratio_[0]*100, 'ev0', pca.explained_variance_[0], 'total', pca.explained_variance_.sum() if hasattr(pca,'explained_variance_') else None)
PY"
