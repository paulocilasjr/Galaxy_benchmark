/bin/bash -c "python3 - <<'PY'
import csv, collections, math
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
# transpose matrix numeric; rows list genes, each row [gene, values]
gene_names=[row[0] for row in rows]
vals=[[float(x) for x in row[1:]] for row in rows]
# n_samples x n_genes
import numpy as np
X=np.array(vals, dtype=float).T
ids=h[1:]
groups=collections.defaultdict(list)
for i,pid in enumerate(ids): groups[pid].append(i)
for pid in sorted(groups):
    if len(groups[pid])>1:
        idx=groups[pid]
        # correlation all pairs after log10+1 perhaps
        Y=np.log10(X[idx]+1)
        pairs=[]
        for a in range(len(idx)):
            for b in range(a+1,len(idx)):
                x=Y[a]; y=Y[b]
                # centering
                xc=x-x.mean(); yc=y-y.mean()
                den=math.sqrt((xc@xc)*(yc@yc))
                corr=(xc@yc)/den if den else float('nan')
                pairs.append((idx[a], idx[b], corr))
        print(pid, [(i,j,round(c,4)) for i,j,c in pairs])
PY"
