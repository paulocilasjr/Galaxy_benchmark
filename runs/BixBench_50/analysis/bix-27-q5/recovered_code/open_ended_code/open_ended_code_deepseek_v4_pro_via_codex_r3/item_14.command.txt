/bin/bash -c "python3 - <<'PY'
import csv, collections, numpy as np
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
vals=[[float(x) for x in row[1:]] for row in rows]
X=np.array(vals, dtype=float).T
ids=h[1:]
groups=collections.defaultdict(list)
for i,pid in enumerate(ids): groups[pid].append(i)
Y=np.log10(X+1)
for pid in sorted(groups):
    if len(groups[pid])>1:
        idx=groups[pid]
        for a in range(len(idx)):
            for b in range(a+1,len(idx)):
                x=Y[idx[a]]; y=Y[idx[b]]
                xc=x-x.mean(); yc=y-y.mean()
                den=np.sqrt((xc@xc)*(yc@yc))
                corr=(xc@yc)/den if den else np.nan
                print(pid, idx[a], idx[b], f'{corr:.6f}')
PY"
