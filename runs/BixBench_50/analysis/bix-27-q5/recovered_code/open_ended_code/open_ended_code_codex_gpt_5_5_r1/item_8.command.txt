/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter,defaultdict
from pathlib import Path
expr=Path('inputs/ROSMAP_genexp_ad.csv')
meta=Path('inputs/ROSMAP_meta_ad.csv')
with expr.open(newline='') as f:
    header=next(csv.reader(f))
ids=header[1:]
c=Counter(ids)
dups={k:v for k,v in c.items() if v>1}
print('expr samples total',len(ids),'unique',len(c),'duplicates',len(dups))
print('duplicate ids', sorted(dups.items()))
with meta.open(newline='') as f:
    r=csv.DictReader(f)
    rows=list(r)
mc=Counter(row['projid'] for row in rows)
md={k:v for k,v in mc.items() if v>1}
print('meta rows',len(rows),'unique projid',len(mc),'duplicates',len(md))
print('meta duplicate ids', sorted(md.items()))
print('expr IDs missing in meta', sorted(set(c)-set(mc))[:20], len(set(c)-set(mc)))
print('meta IDs missing in expr', sorted(set(mc)-set(c))[:20], len(set(mc)-set(c)))
for sid in sorted(dups):
    print('\\nID',sid,'expr_n',dups[sid],'meta_n',mc.get(sid,0))
    for row in rows:
        if row['projid']==sid:
            print(row)
PY"
