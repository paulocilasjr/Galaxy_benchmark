/bin/bash -lc "Rscript - <<'RS'
expr_path <- 'inputs/ROSMAP_genexp_ad.csv'
header <- strsplit(readLines(expr_path, n = 1), ',', fixed = TRUE)[[1]]
sample_ids <- header[-1]
expr <- read.csv(expr_path, row.names = 1, check.names = FALSE)
calc <- function(mat) {
  X <- t(log10(as.matrix(mat) + 1))
  Xc <- scale(X, center=TRUE, scale=FALSE)
  sv <- svd(Xc, nu=0, nv=0)"'$d
  (sv[1]''^2 / sum(Xc''^2)) * 100
}
# The actual answer file uses the resolved unique-sample subset. These alternate values are only a sanity check.
all_pct <- calc(expr)
counts <- table(sample_ids)
keep_positions <- which(!(sample_ids %in% names(counts[counts > 1])))
resolved_pct <- calc(expr[, keep_positions, drop=FALSE])
cat('"'all_records_pc1_percent', format(all_pct, digits=17), '\\n')
cat('resolved_records_pc1_percent', format(resolved_pct, digits=17), '\\n')
cat('answer_file', readLines('codex_output/answer.txt'), '\\n')
RS"
