/bin/bash -lc "python - <<'PY'
import csv, collections
with open('inputs/ROSMAP_meta_ad.csv',newline='') as h:
    rows=list(csv.DictReader(h))
g=collections.defaultdict(list)
for i,r in enumerate(rows, start=2): g[r['projid']].append((i,r))
for k in sorted(g):
    if len(g[k])>1:
        print('\\n'+k)
        for i,r in g[k]: print(i, [r[x] for x in r])
PY"
