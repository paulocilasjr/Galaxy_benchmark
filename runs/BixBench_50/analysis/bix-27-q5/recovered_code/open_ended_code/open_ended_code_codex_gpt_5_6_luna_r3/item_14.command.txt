/bin/bash -lc "Rscript - <<'RS'
exp <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
X <- as.matrix(exp[-1]); storage.mode(X) <- 'double'; X <- t(log10(X + 1))
Xc <- scale(X, center=TRUE, scale=FALSE)
sv <- svd(Xc, nu=0, nv=0)"'$d
fit100 <- prcomp(X, center=TRUE, scale.=FALSE, rank.=100)
fitfull <- prcomp(X, center=TRUE, scale.=FALSE)
cat('"'manual sv top 5:', paste(sprintf('%.8f', sv[1:5]), collapse=' '), '\\n')
cat('prcomp rank100 sdev top 5:', paste(sprintf('%.8f', fit100"'$sdev[1:5]), collapse='"' '), 'len', length(fit100"'$sdev), '"'\\n')
cat('prcomp full sdev top 5:', paste(sprintf('%.8f', fitfull"'$sdev[1:5]), collapse='"' '), 'len', length(fitfull"'$sdev), '"'\\n')
cat('all.equal manual/(sqrt(n-1)), fitfull:', isTRUE(all.equal(sv/sqrt(nrow(X)-1), fitfull"'$sdev)), '"'\\n')
cat('fit100 sum squares:', sum(fit100"'$sdev''^2), '"'manual sum:', sum(sv"'^2)/(nrow(X)-1), '"'\\n')
RS"
