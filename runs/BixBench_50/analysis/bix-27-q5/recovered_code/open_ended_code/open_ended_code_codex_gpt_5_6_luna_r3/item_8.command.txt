/bin/bash -lc "Rscript - <<'RS'
exp <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
meta <- read.csv('inputs/ROSMAP_meta_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
exp_ids <- names(exp)[-1]
meta_ids <- as.character(meta[[1]])
genes <- as.character(exp[[1]])
Xraw <- as.matrix(exp[-1])
storage.mode(Xraw) <- 'double'
cat('expression shape genes x samples:', nrow(Xraw), 'x', ncol(Xraw), '\\n')
cat('metadata shape samples x fields:', nrow(meta), 'x', ncol(meta), '\\n')
cat('expression sample IDs unique:', length(unique(exp_ids)), 'duplicates:', length(exp_ids)-length(unique(exp_ids)), '\\n')
cat('metadata sample IDs unique:', length(unique(meta_ids)), 'duplicates:', length(meta_ids)-length(unique(meta_ids)), '\\n')
cat('same sample ID sets:', setequal(exp_ids, meta_ids), '\\n')
cat('metadata order equals expression order:', identical(meta_ids, exp_ids), '\\n')
cat('expression-only:', length(setdiff(exp_ids, meta_ids)), 'metadata-only:', length(setdiff(meta_ids, exp_ids)), '\\n')
cat('gene IDs unique:', length(unique(genes)), 'duplicates:', length(genes)-length(unique(genes)), 'blank:', sum("'!nzchar(genes)), '"'\\n')
cat('raw finite:', all(is.finite(Xraw)), 'min:', min(Xraw), 'max:', max(Xraw), 'zeros:', sum(Xraw == 0), 'negative:', sum(Xraw < 0), '\\n')
RS"
