/bin/bash -lc "python - <<'PY'
import csv
from pathlib import Path
import numpy as np

exp_path = Path('inputs/ROSMAP_genexp_ad.csv')
meta_path = Path('inputs/ROSMAP_meta_ad.csv')
with exp_path.open(newline='') as f:
    rows = list(csv.reader(f))
header, data = rows[0], rows[1:]
with meta_path.open(newline='') as f:
    meta_rows = list(csv.reader(f))
meta_header, meta_data = meta_rows[0], meta_rows[1:]
print('expression shape genes x samples:', (len(data), len(header)-1))
print('metadata shape samples x fields:', (len(meta_data), len(meta_header)))
print('expression sample IDs unique:', len(set(header[1:])), 'duplicates:', len(header[1:])-len(set(header[1:])))
print('metadata sample IDs unique:', len(set(r[0] for r in meta_data)), 'duplicates:', len(meta_data)-len(set(r[0] for r in meta_data)))
exp_ids = header[1:]
meta_ids = [r[0] for r in meta_data]
print('same sample ID sets:', set(exp_ids) == set(meta_ids))
print('expression-only:', sorted(set(exp_ids)-set(meta_ids))[:10], 'count', len(set(exp_ids)-set(meta_ids)))
print('metadata-only:', sorted(set(meta_ids)-set(exp_ids))[:10], 'count', len(set(meta_ids)-set(exp_ids)))
genes = [r[0] for r in data]
print('gene IDs unique:', len(set(genes)), 'duplicates:', len(genes)-len(set(genes)))
print('blank gene IDs:', sum(not x for x in genes))
arr = np.asarray([[float(x) for x in r[1:]] for r in data], dtype=np.float64)
print('raw finite:', np.isfinite(arr).all(), 'nan_count:', np.isnan(arr).sum(), 'inf_count:', np.isinf(arr).sum())
print('raw min max:', arr.min(), arr.max(), 'nonnegative:', (arr >= 0).all(), 'zeros:', int((arr == 0).sum()))
print('metadata order equals expression order:', meta_ids == exp_ids)
PY"
