/bin/bash -c "python3 - <<'PY'
import requests, re
url='https://journals.plos.org/plosone/article/file?id=10.1371/journal.pone.0250278&type=manuscript'
r=requests.get(url,timeout=30)
text=r.text
for pat in ['supplementary-material','xlink:href','S1','readme','dryad']:
 print(pat, text.lower().find(pat.lower()))
# print relevant snippets
for m in re.finditer(r'(supplementary-material|xlink:href=\"["'^"]+")'"', text):
 print(m.group(0)[:300])
PY"
