import csv, collections
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:

print('genexp dup count groups',len(dups),'total duplicate columns',sum(v-1 for v in dups.values()))
for k in sorted(dups, key=lambda x:int(x[2:])):
 print(k,dups[k])

 h=next(csv.reader(f)); mids=[r[0] for r in csv.reader(f)]
mcnt=collections.Counter(mids)
mdups={k:v for k,v in mcnt.items() if v>1}
print('meta dup count groups',len(mdups),'total duplicate rows',sum(v-1 for v in mdups.values()))
for k in sorted(mdups, key=lambda x:int(x[2:])):
 print(k,mdups[k])
# compare multiset
print('same multiset?', cnt==mcnt)
