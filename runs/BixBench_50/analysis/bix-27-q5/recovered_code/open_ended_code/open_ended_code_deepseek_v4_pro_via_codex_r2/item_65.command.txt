/bin/bash -c "python3 - <<'PY'
import requests, json, re, hashlib, time, html
s=requests.Session()
s.headers.update({'User-Agent':'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120 Safari/537.36','Referer':'https://datadryad.org/stash/dataset/doi:10.5061/dryad.ttdz08kxq'})
# get page first for cookies? 
s.get('https://datadryad.org/stash/dataset/doi:10.5061/dryad.ttdz08kxq',timeout=20)
url='https://datadryad.org/downloads/file_stream/672449'
r=s.get(url,timeout=20)
print('challenge status',r.status_code,'len',len(r.text),'title',re.search(r'<title>(.*?)</title>',r.text).group(1) if re.search(r'<title>(.*?)</title>',r.text) else 'no')
# extract challenge json script
m=re.search(r'<script id=\"anubis_challenge\" type=\"application/json\">(.*?)</script>',r.text,re.S)
print('challenge found',bool(m))
if m:
 ch=json.loads(html.unescape(m.group(1)))
 print(json.dumps(ch,indent=2)[:1000])
 base=re.search(r'<script id=\"anubis_base_prefix\" type=\"application/json\">\"(.*?)\"</script>',r.text).group(1)
 ver=re.search(r'<script id=\"anubis_version\" type=\"application/json\">\"(.*?)\"</script>',r.text).group(1)
 randomData=ch['challenge']['randomData']; difficulty=ch['rules']['difficulty']; cid=ch['challenge']['id']
 start=time.time(); nonce=0
 while True:
  digest=hashlib.sha256((randomData+str(nonce)).encode()).digest()
  if digest.hexdigest().startswith('0'*difficulty):
   break
  nonce+=1
 elapsed=int((time.time()-start)*1000)
 print('nonce',nonce,'hash',digest.hexdigest(),'elapsed',elapsed)
 passurl=f\"https://datadryad.org{base}/.within.website/x/cmd/anubis/api/pass-challenge\"
 params={'id':cid,'response':digest.hexdigest(),'nonce':nonce,'redir':url,'elapsedTime':elapsed}
 rr=s.get(passurl,params=params,timeout=30,allow_redirects=True)
 print('pass status',rr.status_code,'url',rr.url,'len',len(rr.content),'headers',dict(list(rr.headers.items())[:10]))
 print('content start',rr.text[:200])
PY"
