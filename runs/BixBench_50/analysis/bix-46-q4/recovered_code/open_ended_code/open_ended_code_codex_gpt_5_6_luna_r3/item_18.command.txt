/bin/bash -lc "Rscript - <<'RS'
ann <- read.csv(\"inputs/Pseudomonas_aeruginosa_UCBPP-PA14.csv\", check.names=FALSE, stringsAsFactors=FALSE)
ids <- ann[[\"Locus Tag\"]]
for (f in c(\"inputs/res_1vs97.rds\", \"inputs/res_1vs98.rds\")) {
  x <- readRDS(f)
  d <- x@listData
  tab <- data.frame(id=x@rownames, lfc=as.numeric(d"'$log2FoldChange), padj=as.numeric(d$padj), stringsAsFactors=FALSE)
  tab <- merge(tab, ann[, c("Locus Tag","Gene Name","Gene synonyns","Product Name")], by.x="id", by.y="Locus Tag", all.x=TRUE, sort=FALSE)
  cat("'"\\nFILE\", basename(f), \"\\n\")
  key <- grepl(\"rhlI|rhlA|rhlB|rhlC|rhlR|lasI|lasR|lasB|phz|phenazine|pqs|hcn|pyocyanin\", paste(tab"'$`Gene Name`,tab$`Gene synonyns`,tab$`Product Name`), ignore.case=TRUE)
  print(tab[key, c("id","Gene Name","Gene synonyns","Product Name","lfc","padj")], row.names=FALSE, digits=5)
  cat("TOP NEGATIVE'"\\n\")
  print(head(tab[order(tab"'$lfc), c("id","Gene Name","Product Name","lfc","padj")], 15), row.names=FALSE, digits=5)
  cat("TOP POSITIVE'"\\n\")
  print(head(tab[order(-tab"'$lfc), c("id","Gene Name","Product Name","lfc","padj")], 15), row.names=FALSE, digits=5)
}
RS'
