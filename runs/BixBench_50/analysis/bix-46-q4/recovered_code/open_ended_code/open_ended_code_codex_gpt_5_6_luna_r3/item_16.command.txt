/bin/bash -lc "Rscript - <<'RS'
for (f in c(\"inputs/res_1vs97.rds\", \"inputs/res_1vs98.rds\")) {
  x <- readRDS(f)
  idx <- match(\"PA14_35160\", x@rownames)
  d <- x@listData
  cat(\"FILE\", basename(f), \"INDEX\", idx,
      \"LOG2FC\", format(d[[\"log2FoldChange\"]][idx], digits=17),
      \"BASEMEAN\", format(d[[\"baseMean\"]][idx], digits=17),
      \"STAT\", format(d[[\"stat\"]][idx], digits=17),
      \"PADJ\", format(d[[\"padj\"]][idx], digits=17), \"\\n\")
  m <- x@metadata
  cat(\"METADATA_TYPE\", typeof(m), \"LENGTH\", length(m), \"NAMES\", paste(names(m), collapse=\" | \"), \"\\n\")
  if (length(m) > 0) {
    for (nm in names(m)) {
      v <- m[[nm]]
      cat(\"META\", nm, \"TYPE\", typeof(v), \"LEN\", length(v))
      if (is.atomic(v) && length(v) <= 20) cat(\" VALUE\", paste(as.character(v), collapse=\" || \"))
      if (is.list(v)) cat(\" LISTNAMES\", paste(names(v), collapse=\" | \"))
      cat(\"\\n\")
    }
  }
  p <- x@priorInfo
  cat(\"PRIOR_TYPE\", typeof(p), \"LENGTH\", length(p), \"NAMES\", paste(names(p), collapse=\" | \"), \"\\n\\n\")
}
RS"
