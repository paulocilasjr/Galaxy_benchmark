/bin/bash -c "python3 - <<'PY'
import urllib.request, urllib.parse, json
ids=['200122515','200122511','200122505']
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urllib.parse.urlencode({'db':'gds','id':','.join(ids),'retmode':'json'})
data=json.loads(urllib.request.urlopen(url,timeout=30).read())
for uid,res in data['result'].items():
    if uid=='uids': continue
    print('\\nUID',uid)
    for k in ['title','summary','gse','taxon','n_samples','GPL','PDAT']:
        print(k,res.get(k))
PY"
