/bin/bash -c "python3 - <<'PY'
import urllib.request, zipfile, io, re, xml.etree.ElementTree as ET
base='https://www.ncbi.nlm.nih.gov/pmc/articles/PMC5531660/bin/'
for fn in ['ppat.1006504.s012.docx','ppat.1006504.s013.docx']:
    url=base+fn
    print('FETCH',fn)
    data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read()
    print('size',len(data),data[:4])
    z=zipfile.ZipFile(io.BytesIO(data))
    xml=z.read('word/document.xml').decode('utf-8')
    text=''.join(re.findall(r'<w:t["'^>]*>([''^<]*)</w:t>'"',xml))
    print('text len',len(text))
    idx=text.find('PA14_35160')
    print('idx',idx)
    if idx>=0: print(text[max(0,idx-300):idx+800])
    print('---')
PY"
