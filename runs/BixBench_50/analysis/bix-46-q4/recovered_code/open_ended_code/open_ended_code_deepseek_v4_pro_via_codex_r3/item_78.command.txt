/bin/bash -c "python3 - <<'PY'
import urllib.request, urllib.parse, json
ids=['21284483','21284482','21284481','21284480','21284479','21284478','21284477','21284476','21284475','21284474','21284473','21284472','18596782','18596768','12118568','12118567','7752423','7752422','7752421','7752420','7752419','7752418','7619081','7619080','7619079','7619078','7619077','7619076','7619075','7619074']
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?'+urllib.parse.urlencode({'db':'sra','id':','.join(ids),'retmode':'json'})
data=json.loads(urllib.request.urlopen(url,timeout=60).read())
for uid in data['result'].get('uids',[]):
    res=data['result'][uid]
    exp=res.get('expxml','')
    # crude extract title and experiment acc
    import re
    title=re.search(r'<Title>(.*?)</Title>',exp).group(1) if '<Title>' in exp else ''
    expacc=re.search(r'<Experiment acc=\"(.*?)\"',exp).group(1) if '<Experiment acc=' in exp else ''
    print(uid, expacc, title[:180])
PY"
