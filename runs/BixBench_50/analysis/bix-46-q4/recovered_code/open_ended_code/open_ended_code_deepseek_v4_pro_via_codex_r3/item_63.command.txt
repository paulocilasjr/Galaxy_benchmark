/bin/bash -c "python3 - <<'PY'
import urllib.request, urllib.parse, json, re, time
# Search SRP525537
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':'sra','term':'SRP525537[Accession]','retmax':200,'retmode':'json'})
data=json.loads(urllib.request.urlopen(url,timeout=30).read()); ids=data['esearchresult']['idlist']; print('count',len(ids),ids)
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?'+urllib.parse.urlencode({'db':'sra','id':','.join(ids),'rettype':'full'})
xml=urllib.request.urlopen(url,timeout=120).read().decode('utf-8','ignore')
for package in re.findall(r'<EXPERIMENT_PACKAGE>(.*?)</EXPERIMENT_PACKAGE>',xml,re.S):
    exp=re.search(r'<EXPERIMENT["'^>]*alias="([''^"]+)"[''^>]*>(.*?)</EXPERIMENT>'"',package,re.S)
    sample=re.search(r'<SAMPLE["'^>]*alias="([''^"]+)"[''^>]*>(.*?)</SAMPLE>'"',package,re.S)
    attrs={}
    if sample:
        s=sample.group(2)
        attrs=dict(re.findall(r'<SAMPLE_ATTRIBUTE><TAG>(.*?)</TAG><VALUE>(.*?)</VALUE></SAMPLE_ATTRIBUTE>',s,re.S))
    title=''
    if exp:
        m=re.search(r'<TITLE>(.*?)</TITLE>',exp.group(2),re.S); title=m.group(1) if m else ''
    print('alias',exp.group(1) if exp else '?','sample',sample.group(1) if sample else '?','strain',attrs.get('strain'),'sample_type',attrs.get('sample_type'),'replicate',attrs.get('Replicate'))
PY"
