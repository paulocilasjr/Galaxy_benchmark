/bin/bash -c "python3 - <<'PY'
import urllib.request, urllib.parse, re, time
ids=['34678423','34678402','18596782','18596768','41093186','41093182']
for uid in ids:
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?'+urllib.parse.urlencode({'db':'sra','id':uid,'rettype':'full'})
    print('\\n===== UID',uid,'=====')
    try:
        xml=urllib.request.urlopen(url,timeout=60).read().decode('utf-8','ignore')
        # print relevant tags around SAMPLE and TITLE and attributes
        for tag in ['<SAMPLE','<TITLE>','<STUDY','<EXPERIMENT','<RUN ']:
            for m in list(re.finditer(tag,xml))[:5]:
                start=max(0,m.start()-100); end=min(len(xml),m.end()+800)
                print(xml[start:end].replace('\\n',' ')[:1200])
    except Exception as e: print('ERR',e)
    time.sleep(1)
PY"
