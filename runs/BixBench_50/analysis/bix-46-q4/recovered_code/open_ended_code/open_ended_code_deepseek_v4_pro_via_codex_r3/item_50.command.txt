/bin/bash -c "python3 - <<'PY'
import zipfile, io, re
z=zipfile.ZipFile('/tmp/codex_supp.zip')
for fn in ['ppat.1006504.s012.docx','ppat.1006504.s013.docx']:
    print('\\n===== ',fn,' =====')
    subz=zipfile.ZipFile(io.BytesIO(z.read(fn)))
    xml=subz.read('word/document.xml').decode('utf-8','ignore')
    # helper to get text in a segment
    def seg_text(seg):
        return ''.join(re.findall(r'<w:t["'^>]*>([''^<]*)</w:t>'"',seg))
    tables=re.findall(r'<w:tbl>.*?</w:tbl>',xml,re.S)
    print('num tables',len(tables))
    for ti,tbl in enumerate(tables):
        rows=re.findall(r'<w:tr[ >].*?</w:tr>',tbl,re.S)
        if not rows: rows=re.findall(r'<w:tr>.*?</w:tr>',tbl,re.S)
        print('table',ti,'rows',len(rows))
        for ri,row in enumerate(rows[:8]):
            cells=re.findall(r'<w:tc[ >].*?</w:tc>',row,re.S)
            if not cells: cells=re.findall(r'<w:tc>.*?</w:tc>',row,re.S)
            vals=[seg_text(c) for c in cells]
            print(ri,vals)
        # find row containing PA14_35160
        for row in rows:
            if 'PA14_35160' in seg_text(row):
                cells=re.findall(r'<w:tc[ >].*?</w:tc>',row,re.S)
                if not cells: cells=re.findall(r'<w:tc>.*?</w:tc>',row,re.S)
                print('TARGET', [seg_text(c) for c in cells])
PY"
