Rscript -e '
x <- readRDS("/workspace/inputs/res_1vs97.rds")
a <- attributes(x)
cat("Attribute names:", names(a), "\n")

# Access listData directly
ld <- a$listData
cat("listData class:", class(ld), "\n")
cat("listData names:", names(ld), "\n")
for(nn in names(ld)) {
  cat("  ", nn, ": length=", length(ld[[nn]]), " class=", class(ld[[nn]]), "\n")
}

# Check rownames
cat("\nNumber of rows:", length(a$rownames), "\n")
cat("First few rownames:", head(a$rownames, 10), "\n")

# Look for PA14_35160
rn <- a$rownames
idx <- which(rn == "PA14_35160")
cat("PA14_35160 index:", idx, "\n")
if(length(idx) > 0) {
  for(nn in names(ld)) {
    cat(nn, "=", ld[[nn]][idx], "\n")
  }
}
' 2>&1
