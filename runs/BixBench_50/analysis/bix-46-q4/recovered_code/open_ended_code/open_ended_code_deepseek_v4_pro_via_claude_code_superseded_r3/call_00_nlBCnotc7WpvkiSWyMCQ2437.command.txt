Rscript -e '
res1 <- readRDS("/workspace/inputs/res_1vs97.rds")
cat("=== res_1vs97.rds ===\n")
cat("Class:", class(res1), "\n")
if(is.data.frame(res1)) {
  cat("Dimensions:", nrow(res1), "x", ncol(res1), "\n")
  cat("Column names:", colnames(res1), "\n")
  cat("First few rows:\n")
  print(head(res1))
} else if(is.list(res1)) {
  cat("List names:", names(res1), "\n")
  for(n in names(res1)) {
    cat("\n---", n, "---\n")
    cat("Class:", class(res1[[n]]), "\n")
    if(is.data.frame(res1[[n]])) {
      cat("Dim:", dim(res1[[n]]), "\n")
      print(head(res1[[n]]))
    } else {
      print(res1[[n]])
    }
  }
}
'
