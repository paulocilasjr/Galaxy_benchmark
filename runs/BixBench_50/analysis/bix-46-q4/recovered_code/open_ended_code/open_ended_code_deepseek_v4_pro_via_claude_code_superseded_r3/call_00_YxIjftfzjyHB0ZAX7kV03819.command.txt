Rscript -e '
x97 <- readRDS("/workspace/inputs/res_1vs97.rds")
x98 <- readRDS("/workspace/inputs/res_1vs98.rds")

a97 <- attributes(x97)
a98 <- attributes(x98)
rn97 <- a97$rownames
rn98 <- a98$rownames
ld97 <- a97$listData
ld98 <- a98$listData

# Check some quorum sensing related genes
qs_genes <- c("PA14_35160",  # pumA - phenazine
              "rhlI", "rhlR", "lasI", "lasR", "rhlA", "rhlB", "rhlC")
cat("Checking for QS genes...\n")

# Search for rhlI, rhlR etc in CSV
con <- file("/workspace/inputs/Pseudomonas_aeruginosa_UCBPP-PA14.csv", "r")
lines <- readLines(con)
close(con)

cat("QS-related genes:\n")
qs_lines <- grep("rhl|las", lines, ignore.case=TRUE, value=TRUE)
for(l in qs_lines) {
  parts <- strsplit(l, ",")[[1]]
  if(grepl("rhl[IRABC]|las[IR]", parts[11], ignore.case=TRUE) || 
     grepl("autoinducer|quorum|homoserine|acyl", parts[11], ignore.case=TRUE) ||
     grepl("rhl[IRABC]|las[IR]", parts[10], ignore.case=TRUE)) {
    cat(sprintf("%s: %s / %s\n", parts[6], parts[10], parts[11]))
    tag <- parts[6]
    idx97 <- which(rn97 == tag)
    idx98 <- which(rn98 == tag)
    if(length(idx97) > 0) cat(sprintf("  97: log2FC=%.3f padj=%.3e\n", ld97$log2FoldChange[idx97], ld97$padj[idx97]))
    if(length(idx98) > 0) cat(sprintf("  98: log2FC=%.3f padj=%.3e\n", ld98$log2FoldChange[idx98], ld98$padj[idx98]))
  }
}
' 2>&1
