/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
 x <- readRDS(f)
 rn <- attr(x,'rownames',exact=TRUE)
 ld <- attr(x,'listData',exact=TRUE)
 em <- attr(x,'elementMetadata',exact=TRUE)
 emld <- attr(em,'listData',exact=TRUE)
 md <- attr(x,'metadata',exact=TRUE)
 cat('\\nFILE:',f,'\\n')
 cat('columns:',paste(names(ld),collapse=' | '),'\\n')
 cat('target index:',which(rn=='PA14_35160'),'\\n')
 i <- which(rn=='PA14_35160')
 for (nm in names(ld)) cat(nm,':',format(ld[[nm]][i], digits=15),'\\n')
 cat('column metadata names:',paste(names(emld),collapse=' | '),'\\n')
 for (nm in names(emld)) {cat('EM',nm,':\\n'); print(emld[[nm]])}
 cat('metadata names:',paste(names(md),collapse=' | '),'\\n')
 for (nm in names(md)) {cat('META',nm,':\\n'); print(md[[nm]])}
}
RS"
