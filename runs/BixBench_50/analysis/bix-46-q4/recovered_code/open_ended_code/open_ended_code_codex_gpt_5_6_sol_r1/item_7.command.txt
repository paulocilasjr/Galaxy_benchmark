/bin/bash -lc "Rscript - <<'RS'
showobj <- function(obj, name, depth=0) {
  cat(strrep(' ',depth), name, ': typeof=', typeof(obj), ' classattr=', paste(attr(obj,'class'),collapse=','), ' isS4=',isS4(obj),' length=',length(obj),'\\n',sep='')
  aa <- base::attributes(obj)
  cat(strrep(' ',depth), 'attrs=', paste(base::names(aa), collapse='|'),'\\n',sep='')
}
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
 x<-readRDS(f); cat('\\n',f,'\\n')
 showobj(x,'x')
 for (nm in c('priorInfo','rownames','nrows','elementType','elementMetadata','metadata','listData')) {
   y <- attr(x,nm,exact=TRUE); showobj(y,nm,2)
   for (nm2 in base::names(base::attributes(y))) {
     z <- attr(y,nm2,exact=TRUE); showobj(z,nm2,4)
   }
 }
}
RS"
