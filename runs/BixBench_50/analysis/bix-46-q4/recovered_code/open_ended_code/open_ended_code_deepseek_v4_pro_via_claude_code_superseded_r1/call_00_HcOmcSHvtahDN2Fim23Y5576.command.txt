Rscript -e '
res97 <- readRDS("inputs/res_1vs97.rds", refhook = NULL)
em <- attr(res97, "elementMetadata")
# Try slotNames

print("=== listData names ===")
print(names(ld_em))
for(nm in names(ld_em)) {
  cat("\n---", nm, "---\n")
  print(ld_em[[nm]])
}
' 2>&1
