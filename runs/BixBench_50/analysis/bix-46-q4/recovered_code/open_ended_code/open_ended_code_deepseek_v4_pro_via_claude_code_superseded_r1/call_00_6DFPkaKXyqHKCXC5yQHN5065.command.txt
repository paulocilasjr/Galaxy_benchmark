Rscript -e '
# Access listData directly without triggering S4 dispatch
res97 <- readRDS("inputs/res_1vs97.rds", refhook = NULL)
ld <- attr(res97, "listData")
print("=== listData names ===")
print(names(ld))
print("=== First few rows of each column ===")
for(nm in names(ld)) {
  cat("\n---", nm, "---\n")
  print(head(ld[[nm]]))
}
print("=== rownames ===")
print(head(attr(res97, "rownames")))
print("=== Number of rows ===")
print(length(attr(res97, "rownames")))
' 2>&1
