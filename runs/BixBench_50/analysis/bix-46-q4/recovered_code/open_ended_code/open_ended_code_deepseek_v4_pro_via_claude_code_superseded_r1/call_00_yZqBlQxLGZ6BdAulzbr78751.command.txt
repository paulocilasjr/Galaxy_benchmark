Rscript -e '
res97 <- readRDS("inputs/res_1vs97.rds", refhook = NULL)
res98 <- readRDS("inputs/res_1vs98.rds", refhook = NULL)

ld98 <- attr(res98, "listData")
rn97 <- attr(res97, "rownames")
rn98 <- attr(res98, "rownames")

# Check baseMean for rhlI in both comparisons
g <- "PA14_19130"  # rhlI

cat("res97 log2FC:", ld97$log2FoldChange[i97], "\n")
cat("res98 baseMean:", ld98$baseMean[i98], "\n")
cat("res98 log2FC:", ld98$log2FoldChange[i98], "\n")

' 2>&1
