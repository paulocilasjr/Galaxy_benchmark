/bin/bash -c "cd /workspace && python3 - <<'PY'
import subprocess,csv,math
# get RDS lfc
rscript=r'''
a<-readRDS('/workspace/inputs/res_1vs97.rds'); b<-readRDS('/workspace/inputs/res_1vs98.rds')
rna<-attr(a,'rownames',exact=T); rnb<-attr(b,'rownames',exact=T)
lda<-unclass(attr(a,'listData',exact=T)); ldb<-unclass(attr(b,'listData',exact=T))
cat('gene\\t97\\t98\\n')
for(g in sort(union(rna,rnb))) { ia<-match(g,rna); ib<-match(g,rnb); cat(g, format(lda"'$log2FoldChange[ia],digits=15), format(ldb$log2FoldChange[ib],digits=15), sep='"'\\t'); cat('\\n') }
'''
out=subprocess.check_output(['Rscript','-e',rscript]).decode(); rds={}
for line in out.strip().splitlines()[1:]:
 g,a,b=line.split('\\t'); rds[g]=(float(a),float(b))
rows=list(csv.DictReader(open('/workspace/tmp/QS_filtered.csv')))
maxd1=maxd2=0; n=0
for r in rows:
 g=r['GeneID']
 if g not in rds: continue
 d1=abs(float(r['log2FoldChange_ΔrhlI'])-rds[g][0])
 d2=abs(float(r['log2FoldChange_ΔlasI'])-rds[g][1])
 maxd1=max(maxd1,d1); maxd2=max(maxd2,d2); n+=1
print('n',n,'max diff rhlI vs rds97',maxd1,'max diff lasI vs rds98',maxd2)
print('target in rds',rds.get('PA14_35160'))
print('target in QS?', any(r['GeneID']=='PA14_35160' for r in rows))
PY"
