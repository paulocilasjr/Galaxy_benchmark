/bin/bash -c "Rscript - <<'RS'
x <- readRDS('/workspace/inputs/res_1vs97.rds')
attrs <- attributes(x)
for (nm in names(attrs)) {
  a <- attrs[[nm]]
  cat('\\nATTR', nm, 'typeof=', typeof(a), 'length=', length(a), 'names=', paste(if(is.list(a)) names(a) else '', collapse=','), '\\n')
}
ld <- attr(x,'listData')
cat('\\nlistData typeof', typeof(ld), 'length', length(ld), 'names', paste(names(ld), collapse=','), '\\n')
for(nm in names(ld)) {
  v <- ld[[nm]]
  cat('col', nm, 'typeof', typeof(v), 'length', length(v), '\\n')
  print(utils::head(v))
}
rn <- attr(x,'rownames')
cat('\\nrownames head:', paste(utils::head(rn), collapse=', '), '\\n')
RS"
