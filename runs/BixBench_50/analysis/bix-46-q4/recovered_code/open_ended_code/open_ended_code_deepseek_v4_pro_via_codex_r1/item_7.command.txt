/bin/bash -c "Rscript - <<'RS'
x <- readRDS('/workspace/inputs/res_1vs97.rds')
attrs <- attributes(x)
cat('typeof', typeof(x), '\\n')
cat('attrs:', paste(names(attrs), collapse=', '), '\\n')
for (nm in names(attrs)) {
  cat('\\nATTR', nm, 'class', class(attrs[[nm]]), 'typeof', typeof(attrs[[nm]]), 'length', length(attrs[[nm]]), '\\n')
  if (is.list(attrs[[nm]])) {
    print(names(attrs[[nm]]))
  } else if (length(attrs[[nm]]) < 20) {
    print(attrs[[nm]])
  }
}
ld <- attr(x,'listData')
if("'!is.null(ld)) {cat('"'\\nlistData names:', paste(names(ld), collapse=', '),'\\n'); for(nm in names(ld)) {cat('col',nm,'length',length(ld[[nm]]),'head'); print(utils::head(ld[[nm]]))}}
rn <- attr(x,'rownames')
cat('\\nrownames len', length(rn), 'head', paste(utils::head(rn), collapse=', '), '\\n')
RS"
