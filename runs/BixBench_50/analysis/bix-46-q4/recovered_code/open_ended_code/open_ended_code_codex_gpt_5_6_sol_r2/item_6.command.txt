/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  cat('FILE',f,'\\n')
  x <- readRDS(f)
  at <- attributes(x)
  cat('attrs:', paste(names(at),collapse=', '),'\\n')
  for (nm in names(at)) cat(nm, 'typeof=',typeof(at[[nm]]), 'length=',length(at[[nm]]),'class=',paste(attr(at[[nm]],'class'),collapse='/'),'\\n')
  rn <- attr(x,'rownames')
  i <- match('PA14_35160',rn)
  cat('target index',i,'\\n')
  ld <- attr(x,'listData')
  cat('listData names:',paste(names(ld),collapse=', '),' length',length(ld),'\\n')
  for (nm in names(ld)) cat(nm, format(ld[[nm]][i],digits=17), '\\n')
  md <- attr(x,'metadata')
  cat('metadata typeof',typeof(md),'names',paste(names(md),collapse=', '),'\\n')
  if(length(md)) dput(md)
  em <- attr(x,'elementMetadata')
  cat('elementMetadata attrs',paste(names(attributes(em)),collapse=', '),'\\n')
  eld <- attr(em,'listData')
  cat('element metadata listData names:',paste(names(eld),collapse=', '),'\\n')
  for(nm in names(eld)) { cat('emd',nm,':'); dput(eld[[nm]]) }
}
RS"
