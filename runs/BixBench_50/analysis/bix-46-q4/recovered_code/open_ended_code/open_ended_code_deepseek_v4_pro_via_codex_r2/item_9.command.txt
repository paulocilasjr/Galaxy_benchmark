/bin/bash -c "Rscript - <<'EOF'
f<-'/workspace/inputs/res_1vs97.rds'
con<-gzfile(f,'rb'); x<-unserialize(con); close(con)
cat('typeof', typeof(x), '\\n')
cat('class raw class', class(x), '\\n')
cat('length', length(x), '\\n')
cat('attributes names:', paste(names(attributes(x)), collapse=', '), '\\n')
for (nm in names(attributes(x))) {
  a <- attr(x,nm)
  cat('\\nATTR', nm, 'typeof', typeof(a), 'length', length(a), '\\n')
  if (nm "'!= '"'elementMetadata' && nm "'!= '"'rownames') {
    print(a)
  } else if (nm == 'rownames') {
    print(head(a)); print(tail(a))
  } else {
    cat('elementMetadata attributes names:', paste(names(attributes(a)), collapse=', '), '\\n')
    cat('elementMetadata length:', length(a), '\\n')
    for (anm in names(attributes(a))) {
      aa <- attr(a, anm)
      cat('  metadata attr', anm, 'typeof', typeof(aa), 'length', length(aa), '\\n')
      if (anm "'!= '"'rownames' && anm "'!= '"'elementMetadata') print(head(aa))
    }
  }
}
cat('\\nMaybe listData via attr:\\n')
ld <- attr(x,'listData')
cat('ld typeof:', typeof(ld), 'length:', length(ld), '\\n')
cat('ld names:', paste(names(ld), collapse=', '), '\\n')
for (nm in names(ld)) {
  cat('column', nm, 'typeof', typeof(ld[[nm]]), 'length', length(ld[[nm]]), '\\n')
  print(head(ld[[nm]]))
}
EOF"
