/bin/bash -c "Rscript - <<'EOF'
f<-'/workspace/inputs/res_1vs97.rds'
con<-gzfile(f,'rb'); x<-unserialize(con); close(con)
cat('typeof', typeof(x), '\\n')
cat('class', paste(class(x),collapse=','), '\\n')
cat('length', length(x), '\\n')
cat('isS4', isS4(x), '\\n')
cat('attributes names:', paste(names(attributes(x)), collapse=', '), '\\n')
for (nm in names(attributes(x))) {
  cat('\\nATTR', nm, 'typeof', typeof(attr(x,nm)), 'class', paste(class(attr(x,nm)), collapse=','), 'length', length(attr(x,nm)), '\\n')
  if (nm "'!= '"'elementMetadata' && nm "'!= '"'rownames') {
    print(attr(x,nm))
  }
}
cat('\\nMaybe listData via attr:\\n')
ld <- attr(x,'listData')
cat('ld typeof:', typeof(ld), 'class:', paste(class(ld),collapse=','), 'length:', length(ld), '\\n')
cat('ld names:', paste(names(ld), collapse=', '), '\\n')
for (nm in names(ld)) {
  cat('column', nm, 'typeof', typeof(ld[[nm]]), 'class', paste(class(ld[[nm]]), collapse=','), 'length', length(ld[[nm]]), '\\n')
  print(head(ld[[nm]]))
}
EOF"
