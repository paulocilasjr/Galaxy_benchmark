/bin/bash -c "Rscript - <<'EOF'
files<-c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')
xs<-list()
for(f in files){con<-gzfile(f,'rb');x<-unserialize(con);close(con);y<-unclass(x);ld<-attr(y,'listData');rn<-attr(y,'rownames');xs[[f]]<-list(rn=rn,bm=ld"'$baseMean,lfc=ld$log2FoldChange,padj=ld$padj)}
g<-intersect(xs[[1]]$rn,xs[[2]]$rn); cat('"'intersect genes',length(g),'\\n')
i1<-match(g,xs[[1]]"'$rn); i2<-match(g,xs[[2]]$rn)
cat('"'baseMean identical after align:', identical(as.numeric(xs[[1]]"'$bm[i1]),as.numeric(xs[[2]]$bm[i2])),'"'\\n')
cat('max bm diff',max(abs(xs[[1]]"'$bm[i1]-xs[[2]]$bm[i2])),'"'\\n')
cat('cor bm',cor(xs[[1]]"'$bm[i1],xs[[2]]$bm[i2]),'"'\\n')
cat('cor lfc',cor(xs[[1]]"'$lfc[i1],xs[[2]]$lfc[i2]),'"'\\n')
for(gid in c('PA14_35160','PA14_09540','PA14_39960','PA14_51380','PA14_19100')) {
  j1<-match(gid,xs[[1]]"'$rn);j2<-match(gid,xs[[2]]$rn)
  cat(gid,'"'file1 bm/lfc/padj',xs[[1]]"'$bm[j1],xs[[1]]$lfc[j1],xs[[1]]$padj[j1],'"' file2',xs[[2]]"'$bm[j2],xs[[2]]$lfc[j2],xs[[2]]$padj[j2],'"'\\n')
}
EOF"
