/bin/bash -c "which unzip || true; python3 - <<'PY'
import zipfile, xml.etree.ElementTree as ET
for fn in ['/workspace/ppat.1006504.s012.docx','/workspace/ppat.1006504.s013.docx']:
 print('\\n====',fn)
 with zipfile.ZipFile(fn) as z:
  data=z.read('word/document.xml')
 ns={'w':'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
 root=ET.fromstring(data)
 tables=root.findall('.//w:tbl',ns)
 print('tables',len(tables))
 for ti,t in enumerate(tables):
  rows=t.findall('w:tr',ns)
  print('table',ti,'rows',len(rows))
  for ri,row in enumerate(rows[:15]):
   cells=row.findall('w:tc',ns)
   texts=[]
   for c in cells:
    text=''.join(x.text or '' for x in c.findall('.//w:t',ns))
    texts.append(text)
   print(ri, texts)
PY"
