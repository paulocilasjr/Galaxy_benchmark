/bin/bash -c "python3 - <<'PY'
import requests
url='https://www.ebi.ac.uk/europepmc/webservices/rest/search'
r=requests.get(url,params={'query':'TITLE:\"RhlR quorum-sensing receptor ligand sensitivity regulates the differential expression of phenazine genes\"','format':'json','resultType':'core'},timeout=20)
print(r.status_code); j=r.json(); print(j.get('hitCount'))
for x in j.get('resultList',{}).get('result',[]): print(x.get('pmcid'),x.get('doi'),x.get('title'),x.get('supplementaryFiles'))
PY"
