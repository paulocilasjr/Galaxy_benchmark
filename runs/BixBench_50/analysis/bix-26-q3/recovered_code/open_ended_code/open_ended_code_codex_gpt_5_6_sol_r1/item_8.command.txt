/bin/bash -lc "Rscript - <<'RS'
for(f in list.files('inputs',full.names=TRUE)) {
 x<-readRDS(f); ld<-attr(x,'listData'); rn<-attr(x,'rownames'); em<-attr(x,'elementMetadata'); md<-attr(x,'metadata')
 cat('\\nFILE',f,'\\n')
 cat('listData type',typeof(ld),'names',paste(names(ld),collapse=', '),'lens',paste(lengths(ld),collapse=', '),'\\n')
 cat('rownames n',length(rn),'head',paste(head(rn),collapse=', '),'\\n')
 cat('elementMetadata attrs',paste(names(attributes(em)),collapse=', '),'\\n')
 cat('metadata type',typeof(md),'names',paste(names(md),collapse=', '),'\\n')
 for(n in names(ld)) cat(n,':',paste(head(ld[[n]],3),collapse=', '),'\\n')
}
RS"
