/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs',full.names=TRUE)) {
  cat('\\n',f,'\\n')
  x <- readRDS(f)
  cat('typeof',typeof(x),' isS4',isS4(x),' length',length(x),'\\n')
  cat('attributes names:',paste(names(attributes(x)),collapse=', '),'\\n')
  a <- attributes(x)
  for(n in names(a)) cat(n,': typeof=',typeof(a[[n]]),' len=',length(a[[n]]),' attrs=',paste(names(attributes(a[[n]])),collapse=','),'\\n',sep='')
}
RS"
