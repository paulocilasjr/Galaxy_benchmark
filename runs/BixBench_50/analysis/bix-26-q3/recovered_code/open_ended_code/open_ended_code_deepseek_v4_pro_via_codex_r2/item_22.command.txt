/bin/bash -lc "python3 - <<'EOF'
import rdata, numpy as np, requests, warnings
warnings.filterwarnings('ignore')
obj=rdata.read_rds('inputs/res_GluFevsGluFePlus.rds')
rows=obj.rownames
lfc=np.asarray(obj.listData['log2FoldChange']); padj=np.asarray(obj.listData['padj'])
r=requests.get('https://rest.kegg.jp/link/pau/path:pau02010',timeout=60)
genes=set(x.split(':')[1] for a,x in [line.split('\\t') for line in r.text.strip().split('\\n')])
for label, mask in [('up', (lfc>1.5)&(padj<.05)), ('down',(lfc< -1.5)&(padj<.05)), ('abs',(np.abs(lfc)>1.5)&(padj<.05))]:
 s=set(rows[mask]); ov=s&genes; print(label,len(s),len(ov),sorted(ov))
EOF"
