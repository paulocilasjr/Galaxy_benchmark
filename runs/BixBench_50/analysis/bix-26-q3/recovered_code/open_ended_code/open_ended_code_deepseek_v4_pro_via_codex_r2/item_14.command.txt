/bin/bash -lc "python3 - <<'EOF'
import rdata, warnings, numpy as np
warnings.filterwarnings('ignore')
obj=rdata.read_rds('inputs/res_GluFevsGluFePlus.rds')
res=obj
rows=res.rownames
lfc=np.asarray(res.listData['log2FoldChange'])
padj=np.asarray(res.listData['padj'])
up=rows[(lfc>1.5)&(padj<0.05)]
down=rows[(lfc< -1.5)&(padj<0.05)]
abs_up=rows[(np.abs(lfc)>1.5)&(padj<0.05)]
print('total',len(rows))
print('up >1.5 padj<.05',len(up))
print('down <-1.5 padj<.05',len(down))
print('abs up?',len(abs_up))
print('any NA padj',np.sum(np.isnan(padj)))
print('max/min', lfc.min(), lfc.max())
print('first up', up[:20])
EOF"
