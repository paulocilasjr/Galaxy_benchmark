/bin/bash -lc "Rscript - <<'RS'
for(p in c('inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')) {
 x <- readRDS(p)
 ld <- attr(x,'listData')
 rn <- attr(x,'rownames')
 cat('\\n',p,'\\n','nrows',attr(x,'nrows'),' rownames',length(rn),' listData typeof',typeof(ld),' len',length(ld),'\\n')
 cat('listData attrs:',paste(names(attributes(ld)),collapse=', '),'\\n')
 cat('listData names:',paste(names(ld),collapse=', '),'\\n')
 for(i in seq_along(ld)) cat(i,names(ld)[i],typeof(ld[[i]]),length(ld[[i]]),'head',paste(head(ld[[i]],3),collapse=', '),'\\n')
 cat('row names:',paste(head(rn,10),collapse=', '),'\\n')
 md <- attr(x,'metadata')
 cat('metadata raw type/len',typeof(md),length(md),' names',paste(names(md),collapse=', '),'\\n')
}
RS"
