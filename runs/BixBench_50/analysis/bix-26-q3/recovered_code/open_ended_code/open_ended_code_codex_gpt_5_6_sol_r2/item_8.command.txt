/bin/bash -lc "Rscript - <<'RS'
extract <- function(f) {
 x <- readRDS(f)
 d <- as.data.frame(attr(x, 'listData'), stringsAsFactors=FALSE)
 rownames(d) <- attr(x, 'rownames')
 cat('\\n', basename(f), '\\n', sep='')
 cat('dim', nrow(d), ncol(d), '\\n')
 cat('columns', paste(names(d), collapse=','), '\\n')
 cat('contrast', attr(attr(x,'elementMetadata'),'listData')"'$description[2], '"'\\n')
 cat('padj NA', sum(is.na(d"'$padj)), '"'\\n')
 cat('up LFC > 1.5 and padj < .05:', sum(d"'$log2FoldChange > 1.5 & d$padj < .05, na.rm=TRUE), '"'\\n')
 cat('down LFC < -1.5 and padj < .05:', sum(d"'$log2FoldChange < -1.5 & d$padj < .05, na.rm=TRUE), '"'\\n')
 cat('abs LFC > 1.5 and padj < .05:', sum(abs(d"'$log2FoldChange) > 1.5 & d$padj < .05, na.rm=TRUE), '"'\\n')
 cat('up IDs head:', paste(head(rownames(d)[which(d"'$log2FoldChange > 1.5 & d$padj < .05)]), collapse='"','), '\\n')
 print(head(d, 3))
}
lapply(list.files('inputs', pattern='rds"'$'"', full.names=TRUE), extract)
RS"
