/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n###', f, '\\n')
  x <- readRDS(f); a <- attributes(x)
  for (nm in c('metadata','elementMetadata')) {
    z <- a[[nm]]
    cat('ATTR', nm, 'typeof=',typeof(z),'\\n')
    az <- attributes(z)
    cat('attr names:', paste(base::names(az), collapse=' | '), '\\n')
    if ("'!is.null(az[['"'listData']])) {
      ld <- az[['listData']]
      cat('listData names:', paste(base::names(ld), collapse=' | '), '\\n')
      for (cn in base::names(ld)) {
        q <- ld[[cn]]
        cat(' ',cn, 'type',typeof(q),'len',base::length(q),'\\n')
        if (base::length(q) <= 20) print(q) else print(utils::head(q,3))
      }
    }
  }
}
RS"
