/bin/bash -lc "Rscript - <<'RS'
options(error=function() { traceback(2); quit(status=1) })
for (f in list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n###', f, '\\n')
  x <- readRDS(f)
  cat('typeof:', typeof(x), ' length:', length(x), ' object:', is.object(x), '\\n')
  cat('attributes:\\n'); print(attributes(x))
  cat('names:', paste(names(x), collapse=' | '), '\\n')
  cat('first element type/attrs:\\n'); print(typeof(x[[1]])); print(attributes(x[[1]])); print(utils::head(x[[1]], 3))
}
RS"
