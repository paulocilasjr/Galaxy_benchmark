/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n###', f, '\\n')
  x <- readRDS(f)
  a <- attributes(x)
  cat('attribute names:', paste(base::names(a), collapse=' | '), '\\n')
  for (nm in base::names(a)) {
    z <- a[[nm]]
    cat('attr', nm, 'typeof=', typeof(z), 'is.object=', is.object(z), 'base_len=', tryCatch(base::length(z), error=function(e) NA), '\\n')
    if (nm %in% c('elementMetadata','metadata')) {
      cat('  subattr names:', paste(base::names(attributes(z)), collapse=' | '), '\\n')
      cat('  names:', paste(base::names(z), collapse=' | '), '\\n')
    }
  }
}
RS"
