/bin/bash -lc "python3 - <<'PY'
import urllib.request,re
# fetch link/pathway/pau
data=urllib.request.urlopen('https://rest.kegg.jp/link/pathway/pau',timeout=30).read().decode()
lines=data.splitlines()
print('lines',len(lines))
pairs=[]
for l in lines:
    parts=l.split('\\t')
    if len(parts)==2:
        gene=parts[0].split(':')[-1]
        path=parts[1].split(':')[-1]
        pairs.append((gene,path))
print('pairs',len(pairs))
genes=set(g for g,p in pairs); paths=set(p for g,p in pairs)
print('unique genes',len(genes),'paths',len(paths))
# compare all genes in result annotated
allg=set(open('tmp/allgenes.txt').read().split())
print('result all',len(allg),'result annotated',len(allg & genes),'missing',len(allg-genes))
up=set(open('tmp/upgenes.txt').read().split())
print('up annotated',len(up & genes),'missing up',len(up-genes))
PY"
