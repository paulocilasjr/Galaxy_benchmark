import urllib.request

def kegg_get(url_suffix):
    url = f"https://rest.kegg.jp/{url_suffix}"
    try:
        with urllib.request.urlopen(url, timeout=30) as resp:
            return resp.read().decode('utf-8')
    except Exception as e:
        return f"ERROR: {e}"

# Try the correct API
result = kegg_get("list/organism")
print(f"Length: {len(result)}")
# Just show first 20 lines
for line in result.split("\n")[:20]:
    print(line)
print("...")
# Search for pseudomonas
for line in result.split("\n"):
    if "pseud" in line.lower():
        print("FOUND:", line)
