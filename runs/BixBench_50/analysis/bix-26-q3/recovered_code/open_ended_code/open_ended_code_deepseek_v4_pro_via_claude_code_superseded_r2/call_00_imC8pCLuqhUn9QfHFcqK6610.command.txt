import urllib.request

def kegg_get(url_suffix):
    url = f"https://rest.kegg.jp/{url_suffix}"
    try:
        with urllib.request.urlopen(url, timeout=30) as resp:
            return resp.read().decode('utf-8')
    except Exception as e:
        return f"ERROR: {e}"

# List organisms
orgs = kegg_get("list/organism")
# Find pseudomonas
for line in orgs.split("\n"):
    if "pseudo" in line.lower() or "aerug" in line.lower() or "PA14" in line:
        print(line)
