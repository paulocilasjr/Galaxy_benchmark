Rscript -e '
x <- readRDS("inputs/res_GluFevsGluFePlus.rds")
ld <- attr(x, "listData")

# Get rownames
rn <- attr(x, "rownames")
cat("Rownames length:", length(rn), "\n")
cat("First 20 rownames:\n")
cat(rn[1:20], sep="\n")

# Show the elementMetadata without triggering S4 dispatch
em <- attr(x, "elementMetadata")
cat("\nElementMetadata class:", class(em), "\n")
cat("ElementMetadata type:", typeof(em), "\n")

# Try to get data from listData directly
cat("\nFirst 10 rows of data:\n")
for (i in 1:min(10, length(ld$baseMean))) {
  cat(sprintf("Gene: %s, baseMean=%.2f, log2FC=%.4f, pvalue=%.4e, padj=%.4e\n",
    rn[i], ld$baseMean[i], ld$log2FoldChange[i], ld$pvalue[i], ld$padj[i]))
}
' 2>&1
