# Let me check the KEGG API
def kegg_api(operation, params=None):
    base = "https://rest.kegg.jp"
    url = f"{base}/{operation}"
    if params:
        url += "/" + "/".join(params)
    with urllib.request.urlopen(url) as resp:
        return resp.read().decode('utf-8')

# Try to list organisms
orgs = kegg_api("list", ["organism"])
# Find pseudomonas entries
for line in orgs.split("\n"):
    if "pseudo" in line.lower():
        print(line)
