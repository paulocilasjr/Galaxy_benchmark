/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_GluFevsGluFePlus.rds')
a <- attributes(x)
df <- data.frame(gene=a"'$rownames, a$listData, check.names=FALSE)
fg <- df$gene[!is.na(df$padj) & df$padj < 0.05 & !is.na(df$log2FoldChange) & df$log2FoldChange > 1.5]
links <- read.delim('"'https://rest.kegg.jp/link/pathway/pau', header=FALSE, stringsAsFactors=FALSE)
names(links) <- c('gene','pathway')
links"'$gene <- sub('"'"'^pau:'"', '', links"'$gene); links$pathway <- sub('"'"'^path:'"', '', links"'$pathway)
paths <- read.delim('"'https://rest.kegg.jp/list/pathway/pau', header=FALSE, stringsAsFactors=FALSE); names(paths)=c('pathway','description')
for (unimode in c('all_kegg','result_mapped')) {
 universe <- if (unimode=='all_kegg') unique(links"'$gene) else intersect(df$gene, unique(links$gene))
 fgm <- intersect(fg, universe)
 rows <- lapply(split(links$gene, links$pathway), function(g) {
  genes <- intersect(unique(g), universe); M <- length(genes); k <- sum(fgm %in% genes)
  c(M=M, Count=k, pvalue=phyper(k-1, M, length(universe)-M, length(fgm), lower.tail=FALSE))
 })
 out <- data.frame(pathway=names(rows), do.call(rbind, rows), row.names=NULL)
 for (minGS in c(1,10)) {
  tmp <- out[out$M >= minGS & out$M <= 500,]
  tmp$padj <- p.adjust(tmp$pvalue,'"'BH')
  tmp <- merge(tmp, paths, by='pathway')
  abc <- tmp[grep('"'^ABC transporters'"', tmp"'$description),]
  cat(unimode, '"'minGS', minGS, 'tested', nrow(tmp), 'ABC Count', abc"'$Count, '"'p', abc"'$pvalue, '"'padj', abc"'$padj, '"'\\n')
 }
}
RS"
