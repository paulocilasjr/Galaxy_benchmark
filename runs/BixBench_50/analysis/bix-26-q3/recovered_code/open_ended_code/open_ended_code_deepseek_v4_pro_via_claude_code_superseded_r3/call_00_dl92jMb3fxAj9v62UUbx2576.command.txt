source $HOME/miniforge3/bin/activate rtools && Rscript -e '
library(DESeq2)
library(clusterProfiler)

# Read the DESeqResults
res <- readRDS("inputs/res_GluFevsGluFePlus.rds")

cat("Contrast info:\n")
cat("ElementMetadata (mcols):\n")
print(mcols(res))

# Filter for upregulated genes: log2FC > 1.5 AND padj < 0.05
up_genes <- rownames(res)[which(res$log2FoldChange > 1.5 & res$padj < 0.05)]
cat("\nNumber of upregulated genes:", length(up_genes), "\n")
cat("Sample of upregulated genes:\n")
print(head(up_genes, 20))

# Check what the gene IDs look like and whether they are KEGG-compatible
cat("\nGene ID format check:\n")
cat("First gene:", up_genes[1], "\n")

# Get all genes (universe) - genes with non-NA padj
universe <- rownames(res)[which(!is.na(res$padj))]
cat("\nUniverse size:", length(universe), "\n")
'
