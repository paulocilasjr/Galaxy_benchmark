source $HOME/miniforge3/bin/activate rtools && Rscript -e '
library(DESeq2)
library(clusterProfiler)

res <- readRDS("inputs/res_GluFevsGluFePlus.rds")

# Upregulated genes
up_genes <- rownames(res)[which(res$log2FoldChange > 1.5 & res$padj < 0.05)]
cat("Number of upregulated genes:", length(up_genes), "\n")

# Universe
universe <- rownames(res)[which(!is.na(res$padj))]
cat("Universe size:", length(universe), "\n")

# Run KEGG enrichment
kegg_res <- enrichKEGG(
  gene = up_genes,
  universe = universe,
  organism = "pau",
  keyType = "kegg",
  pAdjustMethod = "BH",
  pvalueCutoff = 0.05,
  qvalueCutoff = 0.05
)

# Verify: how many foreground genes mapped to KEGG?
cat("\nForeground genes mapped to KEGG:", length(kegg_res@gene), "\n")
cat("Universe genes mapped to KEGG:", length(kegg_res@universe), "\n")

# Detailed ABC transporters info
abc_df <- as.data.frame(kegg_res)
abc_row <- abc_df[grep("ABC transporter", abc_df$Description, ignore.case=TRUE), ]
cat("\n--- ABC transporters detail ---\n")
cat("GeneRatio:", abc_row$GeneRatio, "\n")
cat("BgRatio:", abc_row$BgRatio, "\n")
cat("Count:", abc_row$Count, "\n")
cat("p.adjust:", abc_row$p.adjust, "\n")
cat("geneID:", abc_row$geneID, "\n")

# Verify the genes are actually in the upregulated set
abc_genes <- strsplit(abc_row$geneID, "/")[[1]]
cat("\nABC transporter genes:", length(abc_genes), "\n")
cat("All in upregulated set:", all(abc_genes %in% up_genes), "\n")

# Show the individual genes with their fold changes
cat("\nGene details:\n")
for (g in abc_genes) {
  idx <- which(rownames(res) == g)
  cat(sprintf("  %s: log2FC=%.4f, padj=%.4e\n", g, res$log2FoldChange[idx], res$padj[idx]))
}
' 2>&1
