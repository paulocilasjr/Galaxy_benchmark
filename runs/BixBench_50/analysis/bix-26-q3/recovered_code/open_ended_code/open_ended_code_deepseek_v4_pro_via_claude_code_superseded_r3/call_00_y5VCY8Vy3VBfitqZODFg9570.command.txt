source $HOME/miniforge3/bin/activate rtools && Rscript -e '
library(DESeq2)
library(clusterProfiler)

res <- readRDS("inputs/res_GluFevsGluFePlus.rds")

# Upregulated genes
up_genes <- rownames(res)[which(res$log2FoldChange > 1.5 & res$padj < 0.05)]
cat("Number of upregulated genes:", length(up_genes), "\n")

# Universe: all genes with non-NA padj (all genes tested)
universe <- rownames(res)[which(!is.na(res$padj))]
cat("Universe size:", length(universe), "\n")

# Try KEGG enrichment with organism pau (P. aeruginosa PA14)
cat("\nRunning enrichKEGG...\n")
kegg_res <- enrichKEGG(
  gene = up_genes,
  universe = universe,
  organism = "pau",
  keyType = "kegg",
  pAdjustMethod = "BH",
  pvalueCutoff = 0.05,
  qvalueCutoff = 0.05
)

cat("KEGG results:\n")
print(head(as.data.frame(kegg_res), 20))

# Check for ABC transporters
abc_rows <- grep("ABC transporter", as.data.frame(kegg_res)$Description, ignore.case=TRUE)
cat("\nABC transporters rows:\n")
if(length(abc_rows) > 0) {
  abc_data <- as.data.frame(kegg_res)[abc_rows, ]
  print(abc_data)
  cat("\nGene IDs in ABC transporters:\n")
  cat(abc_data$geneID, "\n")
  # Count the genes
  genes_in_abc <- strsplit(abc_data$geneID, "/")[[1]]
  cat("\nNumber of genes:", length(genes_in_abc), "\n")
  cat("Gene list:", paste(genes_in_abc, collapse=", "), "\n")
} else {
  cat("No ABC transporters found\n")
}
' 2>&1
