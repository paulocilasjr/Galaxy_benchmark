R --no-save -e '
raw <- readRDS("inputs/res_GluFevsGluFePlus.rds")

# Try to get attributes
a <- attributes(raw)
cat("Attributes names:", names(a), "\n")
for (nm in names(a)) {
  cat("\n--- attr:", nm, "---\n")
  val <- attr(raw, nm)
  cat("Class:", class(val), "\n")
  cat("Type:", typeof(val), "\n")
  if (is.atomic(val) && !is.list(val)) {
    cat("Length:", length(val), "\n")
    if (length(val) <= 10) cat("Values:", val, "\n")
    else cat("First 5:", val[1:5], "\n")
  }
}

# Also try: maybe we can see all contents with .Internal(inspect())
.Internal(inspect(raw))
' 2>&1
