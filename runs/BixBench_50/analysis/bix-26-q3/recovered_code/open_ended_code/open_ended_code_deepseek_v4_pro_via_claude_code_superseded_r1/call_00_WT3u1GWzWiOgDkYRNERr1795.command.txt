R --no-save -e '
raw <- readRDS("inputs/res_GluFevsGluFePlus.rds")

# Get listData names
listData <- attr(raw, "listData")
cat("List data names:", names(listData), "\n")
cat("List data lengths:", sapply(listData, length), "\n")

# Print first few of each
for (nm in names(listData)) {
  cat("\n--- ", nm, " ---\n")
  cat("First 5:", listData[[nm]][1:5], "\n")
}

# Get rownames
rnames <- attr(raw, "rownames")
cat("\nNumber of genes:", length(rnames), "\n")
cat("First 5 genes:", rnames[1:5], "\n")

# Get elementMetadata to understand column descriptions
em <- attr(raw, "elementMetadata")
cat("\nelementMetadata class:", class(em), "\n")
'
