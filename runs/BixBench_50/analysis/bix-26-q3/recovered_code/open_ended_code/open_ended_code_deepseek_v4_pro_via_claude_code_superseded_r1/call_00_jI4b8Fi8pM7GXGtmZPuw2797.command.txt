import requests

url = "http://rest.kegg.jp/link/pau/pathway"
response = requests.get(url, timeout=60)
lines = response.text.strip().split('\n')

# Correct parsing: line format is "path:pauXXXXX\tpau:PA14_XXXXX"
pathway_to_genes = {}
gene_to_pathways = {}
all_kegg_pathways = set()
all_kegg_genes = set()

for line in lines:
    parts = line.split('\t')
    if len(parts) == 2:
        path_raw = parts[0].strip()  # path:pau00010
        gene_raw = parts[1].strip()  # pau:PA14_07190
        
        path_id = path_raw.replace('path:', '')  # pau00010
        gene_id = gene_raw.replace('pau:', '')   # PA14_07190
        
        if path_id not in pathway_to_genes:
            pathway_to_genes[path_id] = set()
        pathway_to_genes[path_id].add(gene_id)
        
        if gene_id not in gene_to_pathways:
            gene_to_pathways[gene_id] = set()
        gene_to_pathways[gene_id].add(path_id)
        
        all_kegg_pathways.add(path_id)
        all_kegg_genes.add(gene_id)

print(f"Unique KEGG genes: {len(all_kegg_genes)}")
print(f"Unique KEGG pathways: {len(all_kegg_pathways)}")

# Check ABC transporters
for pid in pathway_to_genes:
    if '02010' in pid:
        print(f"\nABC transporters ({pid}): {len(pathway_to_genes[pid])} genes")
        print(f"  Genes: {sorted(pathway_to_genes[pid])[:20]}...")

# Intersection with our data - read our gene list
import csv
our_genes = set()
with open('/workspace/res_GluFevsGluFePlus.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        our_genes.add(row['gene'])

print(f"\nOur genes: {len(our_genes)}")
print(f"KEGG genes: {len(all_kegg_genes)}")
print(f"Intersection: {len(our_genes & all_kegg_genes)}")

# What format are our genes vs KEGG genes?
our_sample = list(our_genes)[:5]
kegg_sample = list(all_kegg_genes)[:5]
print(f"Our sample: {our_sample}")
print(f"KEGG sample: {kegg_sample}")
