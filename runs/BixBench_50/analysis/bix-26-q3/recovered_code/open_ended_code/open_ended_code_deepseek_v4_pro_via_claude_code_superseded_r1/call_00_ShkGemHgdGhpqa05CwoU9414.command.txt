R --no-save -e '
# Try to read the raw data without triggering S4 class loading
raw <- readRDS("inputs/res_GluFevsGluFePlus.rds")
# Try to access slots directly
cat("Slots:", slotNames(raw), "\n")

# Try to access the data as a list
slots <- slotNames(raw)
for (s in slots) {
  cat("\n--- Slot:", s, "---\n")
  obj <- slot(raw, s)
  cat("Class:", class(obj), "\n")
  cat("Type:", typeof(obj), "\n")
  if (is.vector(obj) && !is.list(obj)) {
    cat("Length:", length(obj), "\n")
    if (length(obj) > 0 && length(obj) <= 10) cat("Values:", obj, "\n")
    else if (length(obj) > 10) cat("First 5:", obj[1:5], "\n")
  } else if (is.data.frame(obj)) {
    cat("Dim:", dim(obj), "\n")
    cat("Names:", names(obj), "\n")
  } else if (is.list(obj)) {
    cat("Num elements:", length(obj), "\n")
    cat("Names:", names(obj), "\n")
  } else {
    cat("str:\n")
    str(obj, max.level=2)
  }
}
'
