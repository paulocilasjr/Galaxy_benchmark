R --no-save -e '
raw <- readRDS("inputs/res_GluFevsGluFePlus.rds")
listData <- attr(raw, "listData")
rnames <- attr(raw, "rownames")

# Create data frame
df <- data.frame(
  gene = rnames,
  baseMean = listData[["baseMean"]],
  log2FoldChange = listData[["log2FoldChange"]],
  lfcSE = listData[["lfcSE"]],
  stat = listData[["stat"]],
  pvalue = listData[["pvalue"]],
  padj = listData[["padj"]],
  stringsAsFactors = FALSE
)

write.csv(df, "/workspace/res_GluFevsGluFePlus.csv", row.names = FALSE)
cat("Total genes:", nrow(df), "\n")
cat("Upregulated (log2FC > 1.5 & padj < 0.05):", sum(df$log2FoldChange > 1.5 & df$padj < 0.05, na.rm = TRUE), "\n")
cat("Downregulated (log2FC < -1.5 & padj < 0.05):", sum(df$log2FoldChange < -1.5 & df$padj < 0.05, na.rm = TRUE), "\n")
cat("NA padj:", sum(is.na(df$padj)), "\n")

# Print some upregulated genes
up <- df[df$log2FoldChange > 1.5 & df$padj < 0.05 & !is.na(df$padj), ]
cat("\nFirst 10 upregulated genes:\n")
print(head(up[, c("gene", "log2FoldChange", "padj")], 10))
cat("\nNumber of upregulated:", nrow(up), "\n")
'
