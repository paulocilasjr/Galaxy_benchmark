R --no-save -e '
.libPaths(c(Sys.getenv("R_LIBS_USER"), .libPaths()))
# Install dependencies in order, starting with the most basic ones
pkgs <- c("cli", "glue", "lifecycle", "magrittr", "rlang", "vctrs", "Rcpp", "BH",
          "pillar", "tibble", "tidyselect", "purrr", "dplyr", "tidyr", "stringr", "stringi",
          "ggplot2", "gtable", "scales", "farver", "labeling", "munsell", "RColorBrewer", "viridisLite",
          "jsonlite", "igraph", "MASS", "plyr", "RcppArmadillo", "RcppEigen",
          "locfit", "matrixStats", "ape", "lazyeval", "fastmatch", "data.table",
          "reshape2", "ggrepel", "ggforce", "tweenr", "polyclip")
for (p in pkgs) {
  if (!requireNamespace(p, quietly = TRUE)) {
    cat("Installing", p, "...\n")
    install.packages(p, repos = "https://cran.r-project.org", lib = Sys.getenv("R_LIBS_USER"))
  }
}
' 2>&1 | tail -40
