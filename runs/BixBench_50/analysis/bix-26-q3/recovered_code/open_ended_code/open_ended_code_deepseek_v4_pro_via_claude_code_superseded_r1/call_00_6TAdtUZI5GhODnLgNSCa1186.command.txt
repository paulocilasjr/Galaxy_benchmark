R --no-save -e '
raw <- readRDS("inputs/res_GluFevsGluFePlus.rds")
cat("Class:", class(raw), "\n")
# Try unclass
uc <- unclass(raw)
cat("Unclassed class:", class(uc), "\n")
cat("Unclassed type:", typeof(uc), "\n")
cat("Unclassed length:", length(uc), "\n")
cat("Unclassed names:", names(uc), "\n")

# Check each element
for (nm in names(uc)) {
  cat("\n---", nm, "---\n")
  el <- uc[[nm]]
  cat("Class:", class(el), "\n")
  cat("Type:", typeof(el), "\n")
  cat("Length:", length(el), "\n")
  if (length(el) > 0 && length(el) <= 10) {
    cat("Values:", el, "\n")
  } else if (length(el) > 10) {
    cat("First 5:", el[1:5], "\n")
    cat("Last 5:", el[length(el) - 4:0], "\n")
  }
  if (!is.null(names(el))) cat("Names:", names(el)[1:min(5, length(el))], "\n")
}
'
