/bin/bash -lc "python - <<'PY'
import zipfile
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:

        tree=[n for n in names if n.endswith('.treefile')]
        bionj=[n for n in names if n.endswith('.bionj')]
        print(z, 'members', len(names), 'treefile', len(tree), 'bionj', len(bionj))
        print('first treefiles', tree[:5])
PY"
