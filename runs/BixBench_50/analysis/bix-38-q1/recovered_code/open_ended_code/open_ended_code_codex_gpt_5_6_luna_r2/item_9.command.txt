/bin/bash -lc 'for f in inputs/scogs_animals.zip inputs/scogs_fungi.zip; do
  echo "--- $f"
  unzip -Z1 "$f" | awk '"'
    /\\.treefile"'$/ {n_tree=n_tree+1}
    /'"\\.iqtree"'$/ {n_iq=n_iq+1}
    /'"\\.mafft\\.clipkit"'$/ {n_aln=n_aln+1}
    /'"\\.faa"'$/ {n_faa=n_faa+1}
    /'"\\.bionj"'$/ {n_bionj=n_bionj+1}
    /'"\\.mldist"'$/ {n_mldist=n_mldist+1}
    /'"\\.model\\.gz"'$/ {n_model=n_model+1}
    /'"\\.log"'$/ {n_log=n_log+1}
    END {printf "treefile=%d iqtree=%d clipkit=%d faa=%d bionj=%d mldist=%d model=%d log=%d'"\\n\",n_tree,n_iq,n_aln,n_faa,n_bionj,n_mldist,n_model,n_log}'
done
"
