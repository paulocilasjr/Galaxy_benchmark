/bin/bash -lc "python - <<'PY'
from pathlib import Path
import subprocess, statistics

def lengths(group):
    vals=[]
    failures=[]
    for p in sorted(Path('tmp/tree_lengths', group).glob('*.treefile')):
        cp = subprocess.run(['python','-m','phykit','total_tree_length',str(p)], text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        if cp.returncode "'!= 0:
            failures.append((str(p), cp.stderr.strip()))
            continue
        s = cp.stdout.strip().splitlines()[-1].strip()
        try:
            val = float(s)
        except ValueError:
            failures.append((str(p), '"'non-numeric output: '+repr(cp.stdout)))
            continue
        vals.append((p.name,val))
    if failures:
        raise SystemExit(f'failures: {failures[:3]} total={len(failures)}')
    return vals

animals = lengths('animals')
fungi = lengths('fungi')
for label, vals in [('animals',animals),('fungi',fungi)]:
    nums=[v for _,v in vals]
    print(label, 'n', len(nums), 'median', statistics.median(nums), 'min', min(nums), 'max', max(nums))
print('fungi/animals', statistics.median([v for _,v in fungi]) / statistics.median([v for _,v in animals]))
print('animals/fungi', statistics.median([v for _,v in animals]) / statistics.median([v for _,v in fungi]))
PY"
