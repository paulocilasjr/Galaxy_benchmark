#!/bin/bash
# Compute tree lengths for all animal trees
ANIMALS_FILE="/workspace/animals_tree_lengths.txt"
FUNGI_FILE="/workspace/fungi_tree_lengths.txt"

> "$ANIMALS_FILE"
> "$FUNGI_FILE"

echo "Computing animals tree lengths..."
for f in /workspace/trees_animals/*.treefile; do
    val=$(/workspace/venv/bin/phykit total_tree_length "$f" 2>/dev/null)
    echo "$val" >> "$ANIMALS_FILE"
done
wc -l "$ANIMALS_FILE"
echo "---"

echo "Computing fungi tree lengths..."
for f in /workspace/trees_fungi/*.treefile; do
    val=$(/workspace/venv/bin/phykit total_tree_length "$f" 2>/dev/null)
    echo "$val" >> "$FUNGI_FILE"
done
wc -l "$FUNGI_FILE"
