# Check PhyKIT's output precision - look at values in the files
echo "=== Animals (first 20) ==="
head -20 /workspace/phykit_animals.txt
echo ""
echo "=== Fungi (first 20) ==="
head -20 /workspace/phykit_fungi.txt
echo ""
echo "=== Check for scientific notation or high precision ==="
grep -E 'e|E|\.[0-9]{5,}' /workspace/phykit_animals.txt | head -5
grep -E 'e|E|\.[0-9]{5,}' /workspace/phykit_fungi.txt | head -5
echo ""
echo "=== Max precision seen ==="
awk '{print length($0)}' /workspace/phykit_animals.txt | sort -n | tail -5
awk '{print length($0)}' /workspace/phykit_fungi.txt | sort -n | tail -5
