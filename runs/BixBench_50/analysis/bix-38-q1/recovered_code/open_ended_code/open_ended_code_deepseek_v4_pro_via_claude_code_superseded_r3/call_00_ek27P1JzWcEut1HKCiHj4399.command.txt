python3 << 'EOF'
import statistics

with open('/workspace/animals_tree_lengths.txt') as f:
    animals = [float(line.strip()) for line in f if line.strip()]

with open('/workspace/fungi_tree_lengths.txt') as f:
    fungi = [float(line.strip()) for line in f if line.strip()]

animals_median = statistics.median(animals)
fungi_median = statistics.median(fungi)

print(f"Animals: n={len(animals)}, median={animals_median:.6f}, min={min(animals):.6f}, max={max(animals):.6f}")
print(f"Fungi: n={len(fungi)}, median={fungi_median:.6f}, min={min(fungi):.6f}, max={max(fungi):.6f}")
print(f"Fold-change (fungi/animals) = {fungi_median / animals_median:.6f}")
print(f"Fold-change (animals/fungi) = {animals_median / fungi_median:.6f}")
EOF
