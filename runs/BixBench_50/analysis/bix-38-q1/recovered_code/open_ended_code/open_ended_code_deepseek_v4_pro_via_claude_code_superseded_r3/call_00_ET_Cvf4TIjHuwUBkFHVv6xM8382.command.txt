python3 << 'EOF'
import statistics, math

with open('/workspace/phykit_animals.txt') as f:
    animals = [float(line.strip()) for line in f if line.strip()]

with open('/workspace/phykit_fungi.txt') as f:
    fungi = [float(line.strip()) for line in f if line.strip()]

am = statistics.median(animals)
fm = statistics.median(fungi)
fc = fm / am

print(f"Animals median tree length: {am}")
print(f"Fungi median tree length: {fm}")
print(f"Fold-change (fungi / animals): {fc}")
print(f"Fold-change (animals / fungi): {am / fm}")

# Print with high precision
print(f"\nPrecise fold-change: {fc:.10f}")
EOF
