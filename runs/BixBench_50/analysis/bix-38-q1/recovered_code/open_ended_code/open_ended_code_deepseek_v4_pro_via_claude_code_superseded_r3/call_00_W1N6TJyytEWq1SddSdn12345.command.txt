mkdir -p /workspace/codex_output && python3 << 'EOF'
import statistics

with open('/workspace/phykit_animals.txt') as f:
    animals = [float(line.strip()) for line in f if line.strip()]

with open('/workspace/phykit_fungi.txt') as f:
    fungi = [float(line.strip()) for line in f if line.strip()]

animals_median = statistics.median(animals)
fungi_median = statistics.median(fungi)
fold_change = fungi_median / animals_median

with open('/workspace/codex_output/answer.txt', 'w') as f:
    f.write(f"{fold_change}\n")

print(f"Written: {fold_change}")
EOF
