#!/bin/bash
# Use PhyKIT for the precise values as required
> /workspace/phykit_animals.txt
> /workspace/phykit_fungi.txt
for f in /workspace/trees_animals/*.treefile; do
    /workspace/venv/bin/phykit total_tree_length "$f" 2>/dev/null >> /workspace/phykit_animals.txt
done
for f in /workspace/trees_fungi/*.treefile; do
    /workspace/venv/bin/phykit total_tree_length "$f" 2>/dev/null >> /workspace/phykit_fungi.txt
done
echo "Animals: $(wc -l < /workspace/phykit_animals.txt) lines"
echo "Fungi: $(wc -l < /workspace/phykit_fungi.txt) lines"
