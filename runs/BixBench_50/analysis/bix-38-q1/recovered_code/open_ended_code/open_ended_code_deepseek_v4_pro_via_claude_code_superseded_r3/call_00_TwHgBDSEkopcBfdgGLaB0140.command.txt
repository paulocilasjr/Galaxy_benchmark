import re, os, statistics

def verify_trees(directory, label):
    """Verify all tree files and compute lengths"""
    lengths = []
    failures = []
    for fname in sorted(os.listdir(directory)):
        fpath = os.path.join(directory, fname)
        with open(fpath) as f:
            tree = f.read().strip()
        # Verify it's a valid Newick tree with branch lengths
        if not tree.endswith(';'):
            failures.append((fname, "doesn't end with semicolon"))
            continue
        branch_lengths = re.findall(r':([0-9.]+(?:[eE][+-]?\d+)?)', tree)
        if not branch_lengths:
            failures.append((fname, "no branch lengths found"))
            continue
        tl = sum(float(b) for b in branch_lengths)
        lengths.append(tl)
    
    print(f"{label}: {len(lengths)} valid trees, {len(failures)} failures, median={statistics.median(lengths):.6f}")
    if failures:
        for f, reason in failures[:5]:
            print(f"  FAIL: {f} - {reason}")
    return lengths

animals = verify_trees('/workspace/trees_animals', 'Animals')
fungi = verify_trees('/workspace/trees_fungi', 'Fungi')

am = statistics.median(animals)
fm = statistics.median(fungi)
print(f"\nFungi median / Animals median = {fm / am:.6f}")
