import re

def tree_length_from_newick(newick_str):
    """Sum all branch lengths in a Newick tree string."""
    # Match all :number patterns
    branch_lengths = re.findall(r':([0-9.]+(?:[eE][+-]?\d+)?)', newick_str)
    return sum(float(b) for b in branch_lengths)

    tree = f.read().strip()
