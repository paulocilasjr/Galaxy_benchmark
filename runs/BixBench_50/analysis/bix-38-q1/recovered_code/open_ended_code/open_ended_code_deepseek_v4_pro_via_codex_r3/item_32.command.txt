/bin/bash -lc "cd /workspace && python - <<'PY'
from pathlib import Path
import subprocess, statistics, json
PHYKIT='/opt/agent-venv/bin/phykit'
res={}
for g in ['scogs_animals','scogs_fungi']:
    files=sorted(Path('work',g).glob('*.treefile'))
    vals=[]
    for f in files:
        r=subprocess.run([PHYKIT,'total_tree_length',str(f)],capture_output=True,text=True,check=True)
        val=float(r.stdout.strip())
        vals.append(val)
    res[g]=vals
    print(g, 'n',len(vals),'median',statistics.median(vals),'min',min(vals),'max',max(vals),'mean',statistics.mean(vals))
    print('sorted first last', vals[:3], vals[-3:])
a=res['scogs_animals']; f=res['scogs_fungi']
ma=statistics.median(a); mf=statistics.median(f)
print('ma',repr(ma),'mf',repr(mf))
print('fungi/animal ratio', mf/ma, 'repr', repr(mf/ma))
print('animal/fungi ratio', ma/mf, repr(ma/mf))
PY"
