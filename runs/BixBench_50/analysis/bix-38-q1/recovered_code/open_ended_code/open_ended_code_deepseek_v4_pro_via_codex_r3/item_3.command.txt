/bin/bash -lc "cd /workspace && python - <<'PY'
import zipfile, collections
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    z=zipfile.ZipFile(fn)
    names=z.namelist()
    print(fn, len(names))
    ext=collections.Counter()
    for n in names:
        # suffix
        parts=n.split('.')
        # last meaningful extensions
        if n.endswith('.treefile'): ext['treefile']+=1
        elif n.endswith('.iqtree'): ext['iqtree']+=1
        elif n.endswith('.mafft'): ext['mafft']+=1
        elif n.endswith('.clipkit'): ext['clipkit']+=1
        elif n.endswith('.model.gz'): ext['model']+=1
        elif n.endswith('.bionj'): ext['bionj']+=1
        elif n.endswith('.mldist'): ext['mldist']+=1
        elif n.endswith('.ckp.gz'): ext['ckp']+=1
        elif n.endswith('.log'): ext['log']+=1
        elif n.endswith('.faa'): ext['faa']+=1
        else: ext['other']+=1
    print(ext)
PY"
