/bin/bash -lc "python - <<'PY'
from pathlib import Path
import subprocess, statistics
exe='/opt/agent-venv/bin/pk_total_tree_length'
for group in ('animals','fungi'):
    paths=sorted(Path(f'work/scogs_{group}').glob('*.treefile'))
    vals=[]
    failed=[]
    for p in paths:
        cp=subprocess.run([exe,str(p)],text=True,capture_output=True)
        try:
            vals.append(float(cp.stdout.strip()))
        except Exception:
            failed.append((p.name,cp.returncode,cp.stdout,cp.stderr))
    print(group, 'selected',len(paths),'processed',len(vals),'failed',len(failed))
    print('min',min(vals),'median',statistics.median(vals),'max',max(vals))
    print('middle', sorted(vals)[(len(vals)-1)//2:(len(vals)//2)+1])
    if failed: print(failed[:3])
    globals()[group]=vals
ma=statistics.median(animals); mf=statistics.median(fungi)
print('fungi/animals',mf/ma)
print('animals/fungi',ma/mf)
PY"
