/bin/bash -lc "python - <<'PY'
from pathlib import Path
from Bio import Phylo
from statistics import median
import subprocess
exe='/opt/agent-venv/bin/pk_total_tree_length'
for group in ('animals','fungi'):
    rows=[]
    for p in sorted(Path(f'work/scogs_{group}').glob('*.treefile')):
        t=Phylo.read(p,'newick')
        raw=sum((c.branch_length or 0.0) for c in t.find_clades())
        shown=float(subprocess.check_output([exe,str(p)],text=True).strip())
        rows.append((raw,shown,p.name,len(t.get_terminals())))
    s=sorted(rows)
    print(group,'taxon_counts',sorted(set(x[3] for x in rows)))
    print('raw_median',median(x[0] for x in rows),'display_median',median(x[1] for x in rows))
    for x in s[(len(s)-1)//2:(len(s)//2)+1]: print('central',x)
    print('max_abs_rounding_delta',max(abs(x[0]-x[1]) for x in rows))
    globals()[group]=rows
raw_ratio=median(x[0] for x in fungi)/median(x[0] for x in animals)
display_ratio=median(x[1] for x in fungi)/median(x[1] for x in animals)
print('raw_fungi_over_animals',format(raw_ratio,'.15g'))
print('display_fungi_over_animals',format(display_ratio,'.15g'))
print('rounded_2dp',format(display_ratio,'.2f'))
PY
sed -n '1p' work/scogs_animals/*.treefile | head -n 3
sed -n '1p' work/scogs_fungi/*.treefile | head -n 3"
