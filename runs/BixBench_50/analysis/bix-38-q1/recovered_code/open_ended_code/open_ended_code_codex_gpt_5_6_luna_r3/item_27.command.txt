/bin/bash -lc "PYTHONPATH=/workspace/.tools/phykit python - <<'PY'
import json
import os
import subprocess
from pathlib import Path
from statistics import median

phykit = '/workspace/.tools/phykit/bin/phykit'
env = dict(os.environ, PYTHONPATH='/workspace/.tools/phykit')
for group, root in [('animals', Path('/workspace/.work_animals')), ('fungi', Path('/workspace/.work_fungi'))]:
    paths = sorted(root.rglob('*.treefile'))
    vals = []
    for path in paths:
        result = subprocess.run([phykit, 'total_tree_length', '--json', str(path)], env=env, text=True, capture_output=True, check=True)
        vals.append(json.loads(result.stdout)['total_tree_length'])
    print(group, 'n=', len(vals), 'median=', repr(median(vals)), 'min=', repr(min(vals)), 'max=', repr(max(vals)))
    print(group, 'first5=', [repr(x) for x in vals[:5]])
PY"
