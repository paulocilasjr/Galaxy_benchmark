/bin/bash -lc "PYTHONPATH=/workspace/.tools/phykit python - <<'PY'
import json, os, subprocess
from pathlib import Path
from statistics import median
from Bio import Phylo

phykit='/workspace/.tools/phykit/bin/phykit'
env=dict(os.environ, PYTHONPATH='/workspace/.tools/phykit')
vals_by_group={}
for group, root in [('animals',Path('/workspace/.work_animals')),('fungi',Path('/workspace/.work_fungi'))]:
    vals=[]; raw=[]; mismatches=0
    for path in sorted(root.rglob('*.treefile')):
        out=subprocess.run([phykit,'total_tree_length','--json',str(path)],env=env,text=True,capture_output=True,check=True)
        v=json.loads(out.stdout)['total_tree_length']
        t=Phylo.read(str(path),'newick')
        r=sum(clade.branch_length or 0.0 for clade in t.find_clades())
        vals.append(v); raw.append(r)
        if round(r,4) "'!= v: mismatches += 1
    s=sorted(vals)
    print(group, '"'n',len(vals), 'central', s[(len(s)-1)//2:(len(s)+2)//2], 'median',median(vals), 'direct_mismatches',mismatches)
    vals_by_group[group]=median(vals)
print('fungi/animals', vals_by_group['fungi']/vals_by_group['animals'])
PY"
