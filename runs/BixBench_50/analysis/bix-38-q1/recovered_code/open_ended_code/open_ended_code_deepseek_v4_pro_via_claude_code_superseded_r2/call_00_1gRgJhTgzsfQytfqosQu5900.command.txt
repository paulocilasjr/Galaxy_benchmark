# Check overlap in orthologs
ls /workspace/animals_trees/*.treefile | sed 's|.*/||' | sed 's/.treefile$//' | sort > /tmp/animals_orthologs.txt
ls /workspace/fungi_trees/*.treefile | sed 's|.*/||' | sed 's/.treefile$//' | sort > /tmp/fungi_orthologs.txt
echo "Animal orthologs: $(wc -l < /tmp/animals_orthologs.txt)"
echo "Fungi orthologs: $(wc -l < /tmp/fungi_orthologs.txt)"
echo "Shared orthologs: $(comm -12 /tmp/animals_orthologs.txt /tmp/fungi_orthologs.txt | wc -l)"
