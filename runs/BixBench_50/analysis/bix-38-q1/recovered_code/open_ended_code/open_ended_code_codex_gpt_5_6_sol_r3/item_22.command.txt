/bin/bash -lc "python - <<'PY'
from pathlib import Path
from subprocess import run
from statistics import median
sets={g:{p.name.split('.faa')[0]:p for p in Path('.scratch_phylo',g).glob('*.treefile')} for g in ('animals','fungi')}
shared=sorted(sets['animals'].keys() & sets['fungi'].keys())
print('shared_tree_ids',len(shared))
for g in ('animals','fungi'):
    vals=[]
    for k in shared:
        p=run(['python','-m','phykit','total_tree_length',str(sets[g][k])],capture_output=True,text=True,check=True)
        vals.append(float(p.stdout.strip()))
    print(g,'shared_ID_median',median(vals))
PY"
