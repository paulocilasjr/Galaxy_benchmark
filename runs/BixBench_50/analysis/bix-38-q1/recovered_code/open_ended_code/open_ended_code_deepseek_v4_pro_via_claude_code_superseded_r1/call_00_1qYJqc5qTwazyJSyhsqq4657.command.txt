import json, subprocess, os

# Verify a few specific trees manually by parsing Newick
def parse_newick_lengths(newick):
    """Extract all branch lengths from a Newick string."""
    lengths = []
    i = 0
    while i < len(newick):
        if newick[i] == ':':
            i += 1
            start = i
            while i < len(newick) and newick[i] in '0123456789.-eE':
                i += 1
            val = float(newick[start:i])
            lengths.append(val)
        else:
            i += 1
    return lengths

# Recompute all medians with manual verification for a few
with open('/workspace/animal_lengths.jsonl') as f:
    animal_lengths = [json.loads(line)['total_tree_length'] for line in f if line.strip()]
with open('/workspace/fungi_lengths.jsonl') as f:
    fungi_lengths = [json.loads(line)['total_tree_length'] for line in f if line.strip()]

animal_lengths.sort()
fungi_lengths.sort()

def median(lst):
    n = len(lst)
    if n % 2 == 1:
        return lst[n // 2]
    else:
        return (lst[n // 2 - 1] + lst[n // 2]) / 2

animal_med = median(animal_lengths)
fungi_med = median(fungi_lengths)

print(f"\nAnimal median: {animal_med:.10f}")
print(f"Fungi median: {fungi_med:.10f}")
print(f"Fold-change (fungi/animal): {fungi_med/animal_med:.10f}")
