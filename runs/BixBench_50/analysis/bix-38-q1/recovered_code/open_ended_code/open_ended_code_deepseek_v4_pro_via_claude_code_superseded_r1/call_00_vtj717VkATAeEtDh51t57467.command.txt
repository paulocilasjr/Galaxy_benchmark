import json

# Animal tree lengths
with open('/workspace/animal_lengths.jsonl') as f:
    animal_lengths = [json.loads(line)['total_tree_length'] for line in f if line.strip()]

# Fungi tree lengths
with open('/workspace/fungi_lengths.jsonl') as f:
    fungi_lengths = [json.loads(line)['total_tree_length'] for line in f if line.strip()]

# Sort and compute medians
animal_lengths.sort()
fungi_lengths.sort()

def median(lst):
    n = len(lst)
    if n % 2 == 1:
        return lst[n // 2]
    else:
        return (lst[n // 2 - 1] + lst[n // 2]) / 2

animal_median = median(animal_lengths)
fungi_median = median(fungi_lengths)

print(f"Animals: {len(animal_lengths)} trees, median tree length = {animal_median:.6f}")
print(f"Fungi:   {len(fungi_lengths)} trees, median tree length = {fungi_median:.6f}")
print(f"Fold-change (fungi/animal): {fungi_median/animal_median:.6f}")
print(f"Fold-change (animal/fungi): {animal_median/fungi_median:.6f}")

# Also show range
print(f"\nAnimals min={min(animal_lengths):.6f}, max={max(animal_lengths):.6f}")
print(f"Fungi min={min(fungi_lengths):.6f}, max={max(fungi_lengths):.6f}")
