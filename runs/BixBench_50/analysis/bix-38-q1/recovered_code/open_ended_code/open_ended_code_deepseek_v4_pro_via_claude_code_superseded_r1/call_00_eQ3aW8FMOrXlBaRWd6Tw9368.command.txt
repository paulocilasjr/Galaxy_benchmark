import json

with open('/workspace/animal_lengths.jsonl') as f:
    animal_lengths = [json.loads(line)['total_tree_length'] for line in f if line.strip()]
with open('/workspace/fungi_lengths.jsonl') as f:
    fungi_lengths = [json.loads(line)['total_tree_length'] for line in f if line.strip()]

animal_lengths.sort()
fungi_lengths.sort()

def median(lst):
    n = len(lst)
    if n % 2 == 1:
        return lst[n // 2]
    else:
        return (lst[n // 2 - 1] + lst[n // 2]) / 2

animal_med = median(animal_lengths)
fungi_med = median(fungi_lengths)

fc = fungi_med / animal_med

print(f"N animals: {len(animal_lengths)}")
print(f"N fungi: {len(fungi_lengths)}")
print(f"Animal median: {animal_med}")
print(f"Fungi median: {fungi_med}")
print(f"Fold-change: {fc}")
print(f"Formatted: {fc:.6f}")
print(f"Formatted: {fc:.10f}")
