/bin/bash -lc 'calc_tmp=$(mktemp -d /workspace/.treecalc.XXXXXX)
unzip -q inputs/scogs_animals.zip -d "$calc_tmp/animals"
unzip -q inputs/scogs_fungi.zip -d "$calc_tmp/fungi"
first_animal=$(find "$calc_tmp/animals" -type f -name '"'*.treefile' | sort | head -n 1)
first_fungus="'$(find "$calc_tmp/fungi" -type f -name '"'*.treefile' | sort | head -n 1)
echo \"animal="'$first_animal"
/opt/agent-venv/bin/phykit tree_len "$first_animal"
echo "fungus=$first_fungus"
/opt/agent-venv/bin/phykit tree_len "$first_fungus"
echo "$calc_tmp"'
