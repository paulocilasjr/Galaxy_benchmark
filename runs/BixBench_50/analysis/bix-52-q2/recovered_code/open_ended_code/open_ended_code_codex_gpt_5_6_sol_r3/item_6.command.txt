/bin/bash -lc "command -v Rscript || true
python - <<'PY'
import csv
from fractions import Fraction
length={}
with open('inputs/JD_Chromosome_Length.csv',encoding='utf-8-sig',newline='') as f:
    length={r['Chromosome']:int(r['Length']) for r in csv.DictReader(f)}
kept=set()
qualifying_rows=0
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',encoding='utf-8-sig',newline='') as f:
    for r in csv.DictReader(f):
        m=Fraction(r['MethylationPercentage'])
        if m>90 or m<10:
            qualifying_rows += 1
            kept.add((r['Chromosome'],r['StartPosition']))
counts={}
for c,_ in kept: counts[c]=counts.get(c,0)+1
mean=sum((Fraction(n,length[c]) for c,n in counts.items()), Fraction())/len(counts)
print('qualifying rows',qualifying_rows)
print('distinct loci',len(kept))
print('counts',sorted(counts.items(),key=lambda x:(x[0] not in ('W','Z'), int(x[0]) if x[0].isdigit() else x[0])))
print('exact fraction',mean)
print('decimal',format(float(mean),'.17g'))
PY"
