/bin/bash -c "python3 - <<'PY'
import csv, collections, math
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8')))
lengths={}
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig')):
    lengths[r['Chromosome']]=int(r['Length'])
qual=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
seen=set(); counts=collections.defaultdict(int)
for r in qual:
    p=r['Pos']
    if p not in seen:
        seen.add(p)
        counts[r['Chromosome']]+=1
dens=[counts[ch]/lengths[ch] for ch in counts]
mean=math.fsum(dens)/len(dens)
print('filtered_rows',len(qual))
print('filtered_unique',len(seen))
print('chromosomes',len(counts))
print('mean',repr(mean))
print('counts',dict(sorted(counts.items(), key=lambda kv:(not kv[0].isdigit(), kv[0]))))
PY"
