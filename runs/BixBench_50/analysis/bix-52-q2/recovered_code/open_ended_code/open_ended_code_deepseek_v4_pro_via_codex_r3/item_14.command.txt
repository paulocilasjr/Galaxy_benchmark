/bin/bash -c "python3 - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='')))
for inclusive in [False,True]:
    def q(r):
        m=float(r['MethylationPercentage'])
        if inclusive: return m>=90 or m<=10
        return m>90 or m<10
    qual=[r for r in rows if q(r)]
    uniq=set(r['Pos'] for r in qual)
    print('inclusive',inclusive,'rows',len(qual),'uniq',len(uniq))
    lengths=dict((r['Chromosome'].strip(),int(r['Length'])) for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',encoding='utf-8-sig')))
    chrpos=collections.defaultdict(set)
    for r in qual: chrpos[r['Chromosome']].add(r['Pos'])
    dens=[len(v)/lengths[k] for k,v in chrpos.items()]
    print('chrs',len(dens),'mean',sum(dens)/len(dens))
PY"
