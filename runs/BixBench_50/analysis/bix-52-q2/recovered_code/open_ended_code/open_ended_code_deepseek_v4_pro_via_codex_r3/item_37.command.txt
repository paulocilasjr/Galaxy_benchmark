/bin/bash -c "python3 - <<'PY'
import csv, collections, statistics
for spec in ['JD','ZF']:
    rows=list(csv.DictReader(open(f'inputs/{spec}_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8')))
    lengths={}
    for r in csv.DictReader(open(f'inputs/{spec}_Chromosome_Length.csv',newline='',encoding='utf-8-sig')):
        lengths[r['Chromosome']]=int(r['Length'])
    # extremes retain (any row)
    q=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
    # intermediate rows (10<=m<=90)
    qmid=[r for r in rows if 10 <= float(r['MethylationPercentage']) <= 90]
    for name,qual in [('extremes',q),('middle',qmid)]:
        chrpos=collections.defaultdict(set)
        for r in qual: chrpos[r['Chromosome']].add(r['Pos'])
        counts={k:len(v) for k,v in chrpos.items()}
        z=len(chrpos.get('Z',set())); total=sum(counts.values()); total_len=sum(lengths[k] for k in counts)
        mean=sum(counts[k]/lengths[k] for k in counts)/len(counts)
        print(spec,name,'uniq',total,'Z',z,'Zdens',z/lengths.get('Z',1),'nchr',len(counts),'mean',mean)
PY"
