/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=50
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lp='inputs/JD_Chromosome_Length.csv'
# A locus is identified by its stable Pos label; chromosome/start are checked for consistency.
loc_by_pos={}
with open(cp, newline='') as f:
    for r in csv.DictReader(f):
        pos=r['Pos']
        locus=(r['Chromosome'], r['StartPosition'], r['EndPosition'])
        if pos in loc_by_pos and loc_by_pos[pos] "'!= locus:
            raise ValueError(f'"'inconsistent locus for {pos}')
        loc_by_pos[pos]=locus
# Filter rows first, then deduplicate loci.
filtered=set()
with open(cp, newline='') as f:
    for r in csv.DictReader(f):
        m=Decimal(r['MethylationPercentage'])
        if m > 90 or m < 10:
            filtered.add(r['Pos'])
counts=defaultdict(int)
for pos in filtered:
    chrom, start, end=loc_by_pos[pos]
    counts[chrom] += 1
lengths={}
with open(lp, newline='') as f:
    for r in csv.DictReader(f):
        chrom=r['Chromosome'].lstrip('\\ufeff')
        lengths[chrom]=int(r['Length'])
rows=[]
for chrom in sorted(counts, key=lambda x: (x not in ('W','Z'), int(x) if x.isdigit() else 10**9)):
    density=Decimal(counts[chrom])/Decimal(lengths[chrom])
    rows.append((chrom,counts[chrom],lengths[chrom],density))
mean=sum((x[3] for x in rows),Decimal(0))/Decimal(len(rows))
for x in rows:
    print(x[0],x[1],x[2],format(x[3],'.20E'))
print('n_chromosomes',len(rows))
print('mean',format(mean,'.30E'))
print('mean_fixed',mean)
PY"
