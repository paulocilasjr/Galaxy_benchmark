import pandas as pd
import numpy as np

df = pd.read_csv('inputs/JD_AgeRelated_CpG_noMT_Final.csv')
print("Shape:", df.shape)
print("\nColumns:", df.columns.tolist())
print("\nMethylationPercentage stats:")
print(df['MethylationPercentage'].describe())
print("\nMin:", df['MethylationPercentage'].min(), "Max:", df['MethylationPercentage'].max())

print("\nUnique Pos count:", df['Pos'].nunique())
print("Unique Chromosome values:", sorted(df['Chromosome'].unique()))

# Check how many rows have >90% or <10% methylation
high = df[df['MethylationPercentage'] > 90]
low = df[df['MethylationPercentage'] < 10]
print(f"\nRows with methylation > 90%: {len(high)}")
print(f"Rows with methylation < 10%: {len(low)}")

# Unique CpGs with >90% or <10%
filtered_rows = df[(df['MethylationPercentage'] > 90) | (df['MethylationPercentage'] < 10)]
unique_filtered_cpgs = filtered_rows['Pos'].nunique()
print(f"Unique CpGs in filtered rows: {unique_filtered_cpgs}")

# Per chromosome
chr_counts = filtered_rows.groupby('Chromosome')['Pos'].apply(lambda x: x.nunique())
print("\nFiltered unique CpGs per chromosome:")
print(chr_counts)
