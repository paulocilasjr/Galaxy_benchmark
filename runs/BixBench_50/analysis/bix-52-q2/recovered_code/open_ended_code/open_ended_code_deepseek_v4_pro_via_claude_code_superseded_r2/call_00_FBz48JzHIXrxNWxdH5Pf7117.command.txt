import csv
from collections import defaultdict

# Load chromosome lengths
chr_len = {}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chr_len[row['Chromosome']] = int(row['Length'])

print("Chromosome lengths loaded:", len(chr_len))
print("Sample chromosomes:", list(chr_len.keys())[:5])

# Load JD CpG data  
cpgs_by_chr = defaultdict(set)
total_rows = 0
filtered_rows = 0

with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    print("Fieldnames:", reader.fieldnames)
    for row in reader:
        total_rows += 1
        meth = float(row['MethylationPercentage'])
        if meth > 90 or meth < 10:
            filtered_rows += 1
            chrom = row['Chromosome']
            pos = row['Pos']
            cpgs_by_chr[chrom].add(pos)

print(f"Total rows: {total_rows}")
print(f"Filtered rows (>90% or <10%): {filtered_rows}")

total_unique = sum(len(v) for v in cpgs_by_chr.values())
print(f"Total unique filtered CpGs: {total_unique}")

# Per-chromosome density
densities = []
for chrom in sorted(cpgs_by_chr.keys(), key=lambda x: int(x) if x.isdigit() else 999):
    count = len(cpgs_by_chr[chrom])
    length = chr_len.get(chrom)
    if length:
        density = count / length
        densities.append(density)
        print(f"  Chr {chrom}: {count} CpGs, length={length}, density={density:.12e}")
    else:
        print(f"  Chr {chrom}: {count} CpGs, NO LENGTH DATA")

mean_density = sum(densities) / len(densities) if densities else 0
print(f"\nNumber of chromosomes with filtered CpGs: {len(densities)}")
print(f"Mean density: {mean_density:.15e}")
