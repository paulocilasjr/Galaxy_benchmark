import csv
from collections import defaultdict

# Load chromosome lengths
chr_len = {}
with open('inputs/JD_Chromosome_Length.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chr_len[row['Chromosome']] = int(row['Length'])

print("Chromosome lengths loaded:", len(chr_len))
print("Chromosomes:", sorted(chr_len.keys(), key=lambda x: int(x) if x.isdigit() else 999))

# Load JD CpG data
cpgs_by_chr = defaultdict(set)  # chr -> set of unique CpG positions
total_rows = 0
filtered_rows = 0

with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        total_rows += 1
        meth = float(row['MethylationPercentage'])
        if meth > 90 or meth < 10:
            filtered_rows += 1
            chrom = row['Chromosome']
            pos = row['Pos']  # unique CpG identifier
            cpgs_by_chr[chrom].add(pos)

print(f"Total rows: {total_rows}")
print(f"Filtered rows (>90% or <10%): {filtered_rows}")

# Check unique values
total_unique = sum(len(v) for v in cpgs_by_chr.values())
print(f"Total unique filtered CpGs: {total_unique}")

# Per-chromosome density
print("\nPer-chromosome filtered unique CpGs and densities:")
densities = []
for chrom in sorted(cpgs_by_chr.keys(), key=lambda x: int(x) if x.isdigit() else 999):
    count = len(cpgs_by_chr[chrom])
    length = chr_len.get(chrom)
    if length:
        density = count / length
        densities.append(density)
        print(f"  Chr {chrom}: {count} CpGs, length={length}, density={density:.10f}")
    else:
        print(f"  Chr {chrom}: {count} CpGs, NO LENGTH DATA")

# Mean density (only chromosomes with at least one filtered CpG)
mean_density = sum(densities) / len(densities) if densities else 0
print(f"\nNumber of chromosomes with filtered CpGs: {len(densities)}")
print(f"Mean density: {mean_density:.15f}")
print(f"Mean density (scientific): {mean_density:e}")
