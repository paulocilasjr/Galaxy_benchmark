/bin/bash -lc "python3 - <<'PY'
import csv, collections
fn='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
g=collections.defaultdict(list)
with open(fn,newline='') as f:
 for row in csv.DictReader(f):
  g[(row['Chromosome'],row['Pos'])].append(row)
# print first few
for i,(k,v) in enumerate(g.items()):
 print(k, len(v), [float(r['MethylationPercentage']) for r in v][:25], 'samples', sorted(set(r['Sample'] for r in v)))
 if i>=5: break
# distribution rows per unique
cnt=collections.Counter(len(v) for v in g.values())
print(cnt)
PY"
