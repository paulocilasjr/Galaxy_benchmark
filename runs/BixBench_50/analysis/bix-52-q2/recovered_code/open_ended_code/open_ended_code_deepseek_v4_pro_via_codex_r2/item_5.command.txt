/bin/bash -lc "python3 - <<'PY'
import csv
from collections import Counter
fn='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
# quick info
rows=[]
with open(fn,newline='') as f:
    r=csv.DictReader(f)
    for i,row in enumerate(r):
        rows.append(row)
print(len(rows), list(rows[0].keys()))
print('unique pos,chr', len(set((x['Chromosome'],x['Pos']) for x in rows)))
c=Counter(x['Chromosome'] for x in rows)
print('chr counts', c)
print('row high/low', sum(1 for x in rows if (p:=float(x['MethylationPercentage']))>90 or p<10))
uniq_pos={}
for x in rows:
    uni=(x['Chromosome'],x['Pos'])
    uniq_pos.setdefault(uniq,[]).append(float(x['MethylationPercentage']))
print('unique with any row high/low', sum(1 for v in uniq_pos.values() if any(p>90 or p<10 for p in v)))
print('unique with all high/low', sum(1 for v in uniq_pos.values() if all(p>90 or p<10 for p in v)))
print('unique mean high/low', sum(1 for v in uniq_pos.values() if (sum(v)/len(v))>90 or (sum(v)/len(v))<10))
PY"
