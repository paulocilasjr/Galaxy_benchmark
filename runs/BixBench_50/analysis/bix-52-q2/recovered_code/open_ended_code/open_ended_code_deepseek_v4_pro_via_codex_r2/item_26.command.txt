/bin/bash -lc "python3 - <<'PY'
import csv, collections
fn='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lens={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
 for row in csv.DictReader(f): lens[row['Chromosome']]=int(row['Length'])
# keep middle row then unique (if filtering out extremes)
s=set()
with open(fn,newline='') as f:
 for row in csv.DictReader(f):
  p=float(row['MethylationPercentage'])
  if p>=10 and p<=90:
   s.add((row['Chromosome'],row['Pos']))
print('middle filtered unique',len(s))
cnt=collections.Counter(c for c,_ in s)
dens={}
for c,n in cnt.items():
 L=lens[c]; dens[c]=n/L
print('chroms',len(dens))
print('mean middle', sum(dens.values())/len(dens), repr(sum(dens.values())/len(dens)))
# middle unique if all rows in site are middle
groups=collections.defaultdict(list)
with open(fn,newline='') as f:
 for row in csv.DictReader(f):
  groups[(row['Chromosome'],row['Pos'])].append(float(row['MethylationPercentage']))
s2={k for k,v in groups.items() if all(p>=10 and p<=90 for p in v)}
print('all middle unique', len(s2))
cnt=collections.Counter(c for c,_ in s2); dens={c:n/lens[c] for c,n in cnt.items()}
print('chroms2',len(dens),'mean2',sum(dens.values())/len(dens))
# sites with mean middle (actual pipeline step)
s3={k for k,v in groups.items() if 10<=sum(v)/len(v)<=90}
print('mean middle sites',len(s3))
cnt=collections.Counter(c for c,_ in s3); dens={c:n/lens[c] for c,n in cnt.items()}
print('chroms3',len(dens),'mean3',sum(dens.values())/len(dens))
PY"
