/bin/bash -lc "python3 - <<'PY'
import csv, collections, statistics
lens={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
 for row in csv.DictReader(f): lens[row['Chromosome']]=int(row['Length'])
S=set()
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='') as f:
 for row in csv.DictReader(f):
  p=float(row['MethylationPercentage'])
  if p>90 or p<10: S.add((row['Chromosome'],row['Pos']))
cnt=collections.Counter(c for c,_ in S)
print('sum counts',sum(cnt.values()),'sum lengths',sum(lens[c] for c in cnt),'weighted',sum(cnt.values())/sum(lens[c] for c in cnt))
# if include only counted chromosomes vs all lengths including zero? 
print('mean dens',statistics.mean([cnt[c]/lens[c] for c in cnt]))
# mean across all 43 chromosomes including zero? 
allchroms=list(lens) # includes MT
print('mean all incl MT zeros',sum(cnt.get(c,0)/lens[c] for c in allchroms)/len(allchroms))
# excluding MT? all nuclear 1-41,W,Z
nucl=[c for c in lens if c"'!='"'MT']
print('nuclear chroms',len(nucl),'mean incl zero nuclear',sum(cnt.get(c,0)/lens[c] for c in nucl)/len(nucl))
PY"
