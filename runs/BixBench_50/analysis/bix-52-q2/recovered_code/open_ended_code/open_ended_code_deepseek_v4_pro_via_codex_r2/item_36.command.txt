dens=[cnt[c]/lens[c] for c in cnt]
print('unsorted', repr(sum(dens)/len(dens)), format(sum(dens)/len(dens), '.17g'))
dens_sorted=[cnt[c]/lens[c] for c in sorted(cnt)]
s=sum(dens_sorted)/len(dens_sorted)
print('sorted', repr(s), format(s,'.17g'))

from fractions import Fraction
fs=[Fraction(cnt[c],lens[c]) for c in sorted(cnt)]
print(float(sum(fs)/len(fs)), repr(float(sum(fs)/len(fs))), format(float(sum(fs)/len(fs)), '.17g'))
