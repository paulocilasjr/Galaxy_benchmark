/bin/bash -lc "awk -F, 'NR>1 && ("'$7+0<10 || $7+0>90) {print $10 "," $1}'"' inputs/JD_AgeRelated_CpG_noMT_Final.csv | LC_ALL=C sort -u | awk -F, '{n["'$1]++} END {for (c in n) print c, n[c]}'"' | LC_ALL=C sort -k1,1V
python - <<'PY'
import pandas as pd
x=pd.read_csv('inputs/JD_AgeRelated_CpG_noMT_Final.csv')
l=pd.read_csv('inputs/JD_Chromosome_Length.csv', encoding='utf-8-sig')
y=x.loc[(x.MethylationPercentage < 10) | (x.MethylationPercentage > 90)].drop_duplicates(['Chromosome','StartPosition','EndPosition'])
c=y.groupby('Chromosome').size().rename('n').reset_index()
z=c.merge(l,on='Chromosome',validate='one_to_one')
print('pandas rows/chromosomes:',int(z.n.sum()),len(z))
print('pandas mean:',format((z.n/z.Length).mean(),'.17g'))
PY"
