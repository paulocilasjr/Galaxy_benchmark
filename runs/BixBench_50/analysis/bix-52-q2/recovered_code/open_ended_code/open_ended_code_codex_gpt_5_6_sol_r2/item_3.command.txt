/bin/bash -lc "python - <<'PY'
import csv, collections, statistics
from pathlib import Path
p=Path('inputs/JD_AgeRelated_CpG_noMT_Final.csv')
rows=0
sites=set(); filtered_rows=0; filtered_sites=set(); chr_rows=collections.Counter(); chr_sites=collections.defaultdict(set); chr_fsites=collections.defaultdict(set); methyl=collections.Counter(); per_site=collections.Counter()
with p.open(newline='', encoding='utf-8-sig') as f:
    rd=csv.DictReader(f)
    print('columns', rd.fieldnames)
    for x in rd:
        rows+=1
        pos=x['Pos']; ch=x['Chromosome']; m=float(x['MethylationPercentage'])
        sites.add(pos); chr_rows[ch]+=1; chr_sites[ch].add(pos); per_site[pos]+=1
        if m < 10 or m > 90:
            filtered_rows+=1; filtered_sites.add(pos); chr_fsites[ch].add(pos)
        if m in (0,10,90,100): methyl[m]+=1
print('rows',rows,'unique sites',len(sites),'filtered rows',filtered_rows,'filtered unique sites',len(filtered_sites))
print('chrom labels',sorted(chr_rows, key=lambda s:(not s.isdigit(), int(s) if s.isdigit() else s)))
print('rows/site min,max,counts',min(per_site.values()),max(per_site.values()),collections.Counter(per_site.values()).most_common())
print('boundary counts',methyl)
print('site chr counts',[(c,len(chr_sites[c]),len(chr_fsites[c])) for c in sorted(chr_rows,key=lambda s:(not s.isdigit(),int(s) if s.isdigit() else s))])
print('length file')
with open('inputs/JD_Chromosome_Length.csv',encoding='utf-8-sig') as f: print(f.read())
PY"
