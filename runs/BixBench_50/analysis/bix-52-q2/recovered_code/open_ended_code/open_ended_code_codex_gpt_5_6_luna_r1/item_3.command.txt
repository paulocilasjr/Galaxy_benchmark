/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
from decimal import Decimal
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
with open(p, newline='') as f:
    rows=list(csv.DictReader(f))
print('rows',len(rows))
print('fields',list(rows[0]))
print('chromosomes',sorted(Counter(r['Chromosome'] for r in rows), key=lambda x:(len(x),x)))
print('row counts by chromosome')
for c,n in sorted(Counter(r['Chromosome'] for r in rows).items(), key=lambda kv:(int(kv[0]) if kv[0].isdigit() else kv[0])):
    print(c,n)
print('unique Pos',len(set(r['Pos'] for r in rows)))
print('unique chrom,pos',len(set((r['Chromosome'],r['Pos']) for r in rows)))
print('unique chrom,start',len(set((r['Chromosome'],r['StartPosition']) for r in rows)))
print('methylation min max',min(float(r['MethylationPercentage']) for r in rows),max(float(r['MethylationPercentage']) for r in rows))
flt=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
print('filtered rows',len(flt),'filtered unique Pos',len(set(r['Pos'] for r in flt)),'filtered unique chrom,pos',len(set((r['Chromosome'],r['Pos']) for r in flt)))
print('filtered unique chrom,start',len(set((r['Chromosome'],r['StartPosition']) for r in flt)))
print('filtered counts by chromosome (rows, unique chrom,pos)')
for c in sorted(set(r['Chromosome'] for r in rows),key=lambda x:int(x) if x.isdigit() else x):
    rr=[r for r in flt if r['Chromosome']==c]
    print(c,len(rr),len(set(r['Pos'] for r in rr)),len(set(r['StartPosition'] for r in rr)))
print('methylation boundary counts',sum(float(r['MethylationPercentage'])==90 for r in rows),sum(float(r['MethylationPercentage'])==10 for r in rows))
PY
printf '%s\\n' '--- all chromosome lengths ---'
sed -n '1,100p' inputs/JD_Chromosome_Length.csv"
