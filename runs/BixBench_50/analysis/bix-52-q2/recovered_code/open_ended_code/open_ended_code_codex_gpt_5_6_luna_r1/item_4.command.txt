/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
with open(p, newline='') as f:
    rows=list(csv.DictReader(f))
key=lambda x: (0,int(x)) if x.isdigit() else (1,x)
chroms=sorted(Counter(r['Chromosome'] for r in rows), key=key)
print('rows',len(rows))
print('chromosomes',chroms)
print('row counts by chromosome')
for c in chroms: print(c,Counter(r['Chromosome'] for r in rows)[c])
for field in ['Pos','StartPosition','EndPosition']:
    print('unique',field,len(set(r[field] for r in rows)))
flt=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
print('filtered rows',len(flt))
for field in ['Pos','StartPosition','EndPosition']:
    print('filtered unique',field,len(set(r[field] for r in flt)))
print('filtered unique chrom,pos',len(set((r['Chromosome'],r['Pos']) for r in flt)))
print('filtered unique chrom,start',len(set((r['Chromosome'],r['StartPosition']) for r in flt)))
print('filtered counts by chromosome (rows, unique Pos, unique StartPosition)')
for c in chroms:
    rr=[r for r in flt if r['Chromosome']==c]
    print(c,len(rr),len(set(r['Pos'] for r in rr)),len(set(r['StartPosition'] for r in rr)))
print('methylation min max',min(float(r['MethylationPercentage']) for r in rows),max(float(r['MethylationPercentage']) for r in rows))
print('methylation boundary counts (==10, ==90)',sum(float(r['MethylationPercentage'])==90 for r in rows),sum(float(r['MethylationPercentage'])==10 for r in rows))
# per Pos record counts and whether mixed pass/fail
bypos=defaultdict(list)
for r in rows: bypos[r['Pos']].append(float(r['MethylationPercentage']))
mixed=[(p,vs) for p,vs in bypos.items() if any(v>90 or v<10 for v in vs) and any(not(v>90 or v<10) for v in vs)]
allpass=[p for p,vs in bypos.items() if all(v>90 or v<10 for v in vs)]
print('positions with records',len(bypos),'mixed pass/fail',len(mixed),'all records pass',len(allpass))
print('example mixed',mixed[:5])
PY"
