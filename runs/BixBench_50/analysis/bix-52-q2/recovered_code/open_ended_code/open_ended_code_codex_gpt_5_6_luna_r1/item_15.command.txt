/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from fractions import Fraction
from decimal import Decimal, getcontext
getcontext().prec=100
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='') as f:
    rows=list(csv.DictReader(f))
unique={r['Pos']:r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10}
counts=defaultdict(int)
for r in unique.values(): counts[r['Chromosome']]+=1
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    lengths={r['Chromosome']:int(r['Length']) for r in csv.DictReader(f)}
exact=sum((Fraction(n,lengths[c]) for c,n in counts.items()), Fraction(0))/len(counts)
answer=Decimal(open('codex_output/answer.txt').read().strip())
true_value=Decimal(exact.numerator)/Decimal(exact.denominator)
print('answer repr:',repr(open('codex_output/answer.txt').read()))
print('line count:',len(open('codex_output/answer.txt').read().splitlines()))
print('qualifying chromosomes:',len(counts))
print('unique filtered CpGs:',len(unique))
print('absolute rounding error:',abs(answer-true_value))
print('match at 60 significant digits:',format(answer,'.60E') == format(true_value,'.60E'))
assert len(open('codex_output/answer.txt').read().splitlines()) == 1
assert format(answer,'.60E') == format(true_value,'.60E')
PY"
