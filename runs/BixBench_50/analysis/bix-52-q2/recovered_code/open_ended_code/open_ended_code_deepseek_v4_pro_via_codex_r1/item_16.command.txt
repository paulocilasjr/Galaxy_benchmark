/bin/bash -lc "python3 - <<'PY'
import csv, collections
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lenp='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(lenp, newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f): lengths[r['Chromosome']]=int(r['Length'])
rows=[]
with open(cp,newline='') as f:
    for r in csv.DictReader(f): rows.append(r)
kept_unique={}
for r in rows:
    m=float(r['MethylationPercentage'])
    if m>90 or m<10:
        kept_unique[r['Pos']]=r['Chromosome']
chr_counts=collections.Counter(kept_unique.values())
# mean only filtered chrs
sel={c:chr_counts[c]/lengths[c] for c in chr_counts if c in lengths}
print(len(sel), sum(sel.values())/len(sel))
# mean all chromosomes in length file (including zeros), excluding MT? all length file has 44 incl MT; maybe no filtered MT, zero.
allc={c:chr_counts.get(c,0)/lengths[c] for c in lengths}
print('all incl MT mean',sum(allc.values())/len(allc), 'n',len(allc))
# excluding MT and perhaps chromosomes not observed? all noMT 43
nomt={c:chr_counts.get(c,0)/lengths[c] for c in lengths if c"'!='"'MT'}
print('all noMT mean',sum(nomt.values())/len(nomt),len(nomt))
# total genome length noMT and total filtered count/total length
print('total noMT length',sum(v for k,v in lengths.items() if k"'!='"'MT'), 'count',sum(chr_counts.values()), 'ratio',sum(chr_counts.values())/sum(v for k,v in lengths.items() if k"'!='"'MT'))
# total all length incl MT
print('total all len',sum(lengths.values()),'ratio',sum(chr_counts.values())/sum(lengths.values()))
# rows vs unique etc
print('unique filtered',len(kept_unique),'sum counts',sum(chr_counts.values()))
# per chrom strings maybe sex
PY"
