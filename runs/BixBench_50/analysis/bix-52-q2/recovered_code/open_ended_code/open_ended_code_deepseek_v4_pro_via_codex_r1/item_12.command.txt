/bin/bash -lc "python3 - <<'PY'
import csv, collections, math
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lenp='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(lenp, newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        lengths[r['Chromosome']]=int(r['Length'])
rows=[]
with open(cp,newline='') as f:
    for r in csv.DictReader(f): rows.append(r)
print('lengths',len(lengths))
# unique all positions count by chr
allunique={}
for r in rows:
    allunique.setdefault(r['Pos'], set()).add(r['Chromosome'])
print('all unique pos',len(allunique), 'chr mismatch',sum(len(v)>1 for v in allunique.values()))
allposchr=collections.Counter(list(v)[0] for v in allunique.values())
# filters
for name, pred in [
    ('strict',lambda m:m>90 or m<10),
    ('inclusive',lambda m:m>=90 or m<=10),
    ('strict_lt',lambda m:m>90 or m<10),
]:
    unique={}
    for r in rows:
        m=float(r['MethylationPercentage'])
        if pred(m):
            unique[r['Pos']]=r['Chromosome']
    chr_counts=collections.Counter(unique.values())
    densities={c:n/lengths[c] for c,n in chr_counts.items() if c in lengths}
    mean=sum(densities.values())/len(densities)
    print('\\n',name,'filtered rows',sum(1 for r in rows if pred(float(r['MethylationPercentage']))),'unique filtered',len(unique))
    print('chrs with filtered',sorted(densities, key=lambda x:int(x) if x.isdigit() else 9999))
    print('counts', {c:chr_counts[c] for c in sorted(densities, key=lambda x:int(x) if x.isdigit() else 9999)})
    print('densities', {c:densities[c] for c in sorted(densities, key=lambda x:int(x) if x.isdigit() else 9999)})
    print('mean',repr(mean),'sum count',sum(chr_counts.values()))
PY"
