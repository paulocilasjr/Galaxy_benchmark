/bin/bash -lc "python3 - <<'PY'
import csv, collections
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
rows=[]
with open(p, newline='') as f:
    r=csv.DictReader(f)
    for i,row in enumerate(r):
        if i<20:
            print(row)
        rows.append(row)
print('n',len(rows))
cols=set(rows[0].keys())
print('cols',cols)
chrom=collections.Counter(r['Chromosome'] for r in rows)
print('chrom n',len(chrom), chrom)
print('chrom list', sorted(chrom.keys(), key=lambda x: int(x) if x.isdigit() else 9999999))
posu=collections.Counter(r['Pos'] for r in rows)
print('pos unique',len(posu),'max count',max(posu.values()), [x for x in posu if posu[x]==max(posu.values())][:3])
# duplicate positions stats
print('dup pos count', sum(1 for v in posu.values() if v>1))
print('counts summary')
cnts=collections.Counter(posu.values())
print(cnts)
meths=[float(r['MethylationPercentage']) for r in rows]
print('meth min max',min(meths),max(meths))
print('>90',sum(m>90 for m in meths),'<10',sum(m<10 for m in meths), '>=90',sum(m>=90 for m in meths), '<=10',sum(m<=10 for m in meths))
print('coverage range',min(int(r['Coverage']) for r in rows),max(int(r['Coverage']) for r in rows))
print('Dage range',min(float(r['Dage']) for r in rows),max(float(r['Dage']) for r in rows))
PY"
