import csv

# Read chromosome lengths
chr_len = {}
with open('inputs/JD_Chromosome_Length.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row['Chromosome'].strip()
        length = int(row['Length'])
        chr_len[chrom] = length

print("Chromosomes and lengths:")
for c in sorted(chr_len.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else 999, x)):
    print(f"  {c}: {chr_len[c]}")

# Read age-related CpGs
# Filter rows with MethylationPercentage > 90 or < 10
# Then get unique CpG positions
filtered_positions = set()
all_positions = set()
total_rows = 0

with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        total_rows += 1
        pos = row['Pos'].strip()
        meth = float(row['MethylationPercentage'])
        all_positions.add(pos)
        if meth > 90 or meth < 10:
            filtered_positions.add(pos)

print(f"\nTotal rows: {total_rows}")
print(f"Total unique CpG positions: {len(all_positions)}")
print(f"Filtered unique CpG positions (meth >90% or <10%): {len(filtered_positions)}")

# Assign filtered CpGs to chromosomes based on Pos (chrom_position format)
chrom_counts = {}
for pos in sorted(filtered_positions):
    chrom = pos.split('_')[0]
    chrom_counts[chrom] = chrom_counts.get(chrom, 0) + 1

print(f"\nChromosomes with filtered CpGs:")
for c in sorted(chrom_counts.keys(), key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else 999, x)):
    print(f"  {c}: {chrom_counts[c]} CpGs")

# Compute densities (CpGs per base pair)
densities = {}
for chrom, count in chrom_counts.items():
    if chrom in chr_len:
        density = count / chr_len[chrom]
        densities[chrom] = density
    else:
        print(f"WARNING: Chromosome {chrom} not found in length file!")

print(f"\nPer-chromosome densities (CpGs per base pair):")
for c in sorted(densities.keys(), key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else 999, x)):
    print(f"  Chr{c}: {count} / {chr_len[c]} = {densities[c]:.6e}")

# Mean density
mean_density = sum(densities.values()) / len(densities)
print(f"\nNumber of chromosomes with filtered CpGs: {len(densities)}")
print(f"Mean density: {mean_density:.10e}")
print(f"Mean density (full precision): {mean_density}")
