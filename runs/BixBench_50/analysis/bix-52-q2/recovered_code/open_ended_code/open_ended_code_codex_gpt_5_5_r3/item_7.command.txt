/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict, Counter

cpg_file='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
len_file='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(len_file, newline='', encoding='utf-8-sig') as fh:
    for row in csv.DictReader(fh):
        lengths[row['Chromosome']]=int(row['Length'])

filtered=defaultdict(set)
all_unique=defaultdict(set)
rows=0
filtered_rows=0
examples=[]
with open(cpg_file, newline='', encoding='utf-8-sig') as fh:
    reader=csv.DictReader(fh)
    print('headers:', reader.fieldnames)
    for row in reader:
        rows += 1
        chrom=row['Chromosome']
        key=(chrom, row['StartPosition'], row['EndPosition'])
        all_unique[chrom].add(key)
        m=float(row['MethylationPercentage'])
        if m > 90 or m < 10:
            filtered_rows += 1
            filtered[chrom].add(key)
            if len(examples)<5:
                examples.append(row)

print('rows', rows)
print('filtered_rows', filtered_rows)
print('chromosomes in lengths', len(lengths), sorted(lengths, key=lambda x:int(x) if x.isdigit() else x)[:5], sorted(lengths, key=lambda x:int(x) if x.isdigit() else x)[-5:])
print('chromosomes with filtered CpGs', len(filtered), sorted(filtered, key=lambda x:int(x) if x.isdigit() else x))
print('missing length chroms', sorted(set(filtered)-set(lengths)))
counts={c:len(s) for c,s in filtered.items()}
for c in sorted(counts, key=lambda x:int(x) if x.isdigit() else x):
    print(c, counts[c], lengths.get(c), counts[c]/lengths[c])
mean=sum(counts[c]/lengths[c] for c in counts)/len(counts)
print('mean', mean)
print('repr', repr(mean))
PY"
