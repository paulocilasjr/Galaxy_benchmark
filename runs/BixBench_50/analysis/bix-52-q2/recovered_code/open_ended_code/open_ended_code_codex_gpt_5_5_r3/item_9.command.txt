/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict

def chrom_key(c):
    return (0, int(c)) if str(c).isdigit() else (1, str(c))

cpg_file='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
len_file='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(len_file, newline='', encoding='utf-8-sig') as fh:
    for row in csv.DictReader(fh):
        lengths[row['Chromosome']]=int(row['Length'])

filtered=defaultdict(set)
rows=filtered_rows=0
with open(cpg_file, newline='', encoding='utf-8-sig') as fh:
    for row in csv.DictReader(fh):
        rows += 1
        m=float(row['MethylationPercentage'])
        if m > 90 or m < 10:
            filtered_rows += 1
            chrom=row['Chromosome']
            # Pos is chrom_start, but use chromosome/start/end explicitly to avoid relying on formatting.
            filtered[chrom].add((row['StartPosition'], row['EndPosition']))

counts={c:len(s) for c,s in filtered.items()}
print('rows', rows)
print('filtered_rows', filtered_rows)
print('length chroms', len(lengths), sorted(lengths, key=chrom_key))
print('filtered chroms', len(counts), sorted(counts, key=chrom_key))
print('missing length chroms', sorted(set(counts)-set(lengths), key=chrom_key))
print('per_chromosome: chrom count length density')
for c in sorted(counts, key=chrom_key):
    d=counts[c]/lengths[c]
    print(c, counts[c], lengths[c], format(d,'.17g'))
mean=sum(counts[c]/lengths[c] for c in counts)/len(counts)
print('mean', format(mean,'.17g'))
PY"
