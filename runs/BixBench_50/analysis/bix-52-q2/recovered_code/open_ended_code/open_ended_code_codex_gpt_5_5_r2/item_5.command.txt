/bin/bash -lc "python3 - <<'PY'
import csv
from collections import Counter, defaultdict
path='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
with open(path, newline='') as f:
    r=csv.DictReader(f)
    rows=0
    filtered=0
    unique_all=set()
    unique_f=set()
    chroms=Counter()
    chroms_f=Counter()
    examples=[]
    for row in r:
        rows += 1
        key=(row['Chromosome'], row['StartPosition'], row['EndPosition'], row['Pos'])
        unique_all.add(key)
        m=float(row['MethylationPercentage'])
        if m > 90 or m < 10:
            filtered += 1
            unique_f.add(key)
            chroms_f[row['Chromosome']] += 1
            if len(examples)<3: examples.append(row)
    print('rows', rows)
    print('unique_all', len(unique_all))
    print('filtered_rows', filtered)
    print('unique_filtered', len(unique_f))
    print('chrom_count_filtered_rows', len(chroms_f), sorted(chroms_f.items(), key=lambda kv: (str(kv[0])) )[:10])
    print('examples', examples)
PY"
