import csv

# Read chromosome lengths
chrom_lengths = {}
with open('inputs/JD_Chromosome_Length.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom_lengths[row['Chromosome']] = int(row['Length'])

print("Chromosome lengths:")
for k, v in sorted(chrom_lengths.items(), key=lambda x: (not x[0].isdigit(), int(x[0]) if x[0].isdigit() else float('inf'), x[0])):
    print(f"  {k}: {v}")

# Read JD CpG data and filter
unique_pos_by_chrom = {}
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        meth = float(row['MethylationPercentage'])
        if meth > 90 or meth < 10:
            chrom = row['Chromosome']
            pos = row['Pos']
            if chrom not in unique_pos_by_chrom:
                unique_pos_by_chrom[chrom] = set()
            unique_pos_by_chrom[chrom].add(pos)

print("\nPer-chromosome densities (unique CpGs / bp):")
densities = []
for chrom in sorted(unique_pos_by_chrom.keys(), key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else float('inf'), x)):
    count = len(unique_pos_by_chrom[chrom])
    length = chrom_lengths[chrom]
    density = count / length
    densities.append(density)
    print(f"  Chr {chrom}: {count} CpGs / {length} bp = {density:.12f}")

print(f"\nNumber of chromosomes with filtered CpGs: {len(densities)}")
mean_density = sum(densities) / len(densities)
print(f"Mean density: {mean_density:.15f}")
print(f"Mean density (scientific): {mean_density:.15e}")
