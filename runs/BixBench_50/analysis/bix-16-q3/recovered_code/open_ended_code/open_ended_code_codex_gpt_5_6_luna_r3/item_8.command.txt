from pathlib import Path

def ids(name):
    with (Path('inputs')/name).open(newline='') as f:
        rd=csv.reader(f); next(rd); return [r[0] for r in rd]
cr=set(ids('CRISPRGeneEffect.csv')); ex=set(ids('OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'))
common=cr&ex
m={}
with open('inputs/Model.csv',newline='') as f:
    for r in csv.DictReader(f): m[r['ModelID']]=r
print('CRISPR',len(cr),'expression',len(ex),'common',len(common))
print('common not Model',len(common-set(m)))
print('types common',collections.Counter(m[x].get('ModelType','') for x in common))
print('types CRISPR',collections.Counter(m[x].get('ModelType','') for x in cr))
print('types expression',collections.Counter(m[x].get('ModelType','') for x in ex))
print('organoid shared',sorted((x,m[x]['CellLineName']) for x in common if m[x].get('ModelType')"'!='"'Cell Line')[:50])
PY"
