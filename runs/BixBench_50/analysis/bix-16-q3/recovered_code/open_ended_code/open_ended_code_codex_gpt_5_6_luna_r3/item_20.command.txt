/bin/bash -lc "python - <<'PY'
import csv
from pathlib import Path
import pandas as pd
from scipy.stats import spearmanr
base=Path('inputs')
def h(name):
    with (base/name).open(newline='') as f:return next(csv.reader(f))
ch=h('CRISPRGeneEffect.csv'); eh=h('OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
check=['ILK (3611)','CDIN1 (84529)','UFM1 (51569)','TRIO (7204)','VCL (7414)']
ci=[0]+[ch.index(g) for g in check]; ei=[0]+[eh.index(g) for g in check]
cr=pd.read_csv(base/'CRISPRGeneEffect.csv',usecols=ci,index_col=0)
ex=pd.read_csv(base/'OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=ei,index_col=0)
for g in check:
    z=pd.concat([cr[g],ex[g]],axis=1,join='inner').dropna()
    raw=spearmanr(z.iloc[:,0],z.iloc[:,1]).statistic
    print(g,'n',len(z),'raw',raw,'essentiality',-raw)
PY"
