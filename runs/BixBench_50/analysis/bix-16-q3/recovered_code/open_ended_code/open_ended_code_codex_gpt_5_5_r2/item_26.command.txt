/bin/bash -lc "python - <<'PY'
import csv
import pandas as pd
from scipy.stats import spearmanr

genes = ['CCND1 (595)', 'FERMT2 (10979)', 'KLF5 (688)', 'WWTR1 (25937)']

def read_cols(path, genes):
    with open(path, newline='') as f:
        header = next(csv.reader(f))
    idx = [0] + [header.index(g) for g in genes]
    return pd.read_csv(path, usecols=idx, index_col=0)

dep = read_cols('inputs/CRISPRGeneEffect.csv', genes)
expr = read_cols('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', genes)
common = dep.index.intersection(expr.index)
for g in genes:
    r = spearmanr(expr.loc[common, g], -dep.loc[common, g], nan_policy='omit').statistic
    print(g, f'{r:.9f}')
PY"
