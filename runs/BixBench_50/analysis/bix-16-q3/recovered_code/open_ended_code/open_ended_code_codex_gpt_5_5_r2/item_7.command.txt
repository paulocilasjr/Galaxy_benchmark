import csv
paths=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
headers=[]
ids=[]
for p in paths:
    with open(p,newline='') as f:
        r=csv.reader(f); h=next(r); headers.append(h[1:])
        first=[]; n=0
        for row in r:
            n+=1
            if len(first)<5: first.append(row[0])
        ids.append((n, first))
for p,(n,first),h in zip(paths,ids,headers):
    print(p, 'rows', n, 'genes', len(h), 'first_ids', first)
print('common_genes_exact', len(set(headers[0]).intersection(headers[1])))
print('dep_dup_cols', len(headers[0])-len(set(headers[0])))
print('expr_dup_cols', len(headers[1])-len(set(headers[1])))
PY"
