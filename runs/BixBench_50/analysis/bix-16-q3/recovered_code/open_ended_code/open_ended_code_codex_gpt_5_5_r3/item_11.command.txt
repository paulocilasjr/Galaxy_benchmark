/bin/bash -lc "python - <<'PY'
from pathlib import Path
import numpy as np
import pandas as pd
from scipy.stats import rankdata

crispr_path = 'inputs/CRISPRGeneEffect.csv'
expr_path = 'inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'

# Read matrices with model IDs as rows. Blanks in CRISPR become NaN.
ge = pd.read_csv(crispr_path, index_col=0, dtype=np.float32)
expr = pd.read_csv(expr_path, index_col=0, dtype=np.float32)

common_rows = ge.index.intersection(expr.index)
common_cols = ge.columns.intersection(expr.columns)
ge = ge.loc[common_rows, common_cols]
expr = expr.loc[common_rows, common_cols]

count = 0
valid_corrs = 0
near = []
for gene in common_cols:
    x = expr[gene].to_numpy(dtype=np.float64, copy=False)
    y = -ge[gene].to_numpy(dtype=np.float64, copy=False)
    mask = np.isfinite(x) & np.isfinite(y)
    n = int(mask.sum())
    if n < 2:
        continue
    xv = x[mask]
    yv = y[mask]
    # Spearman is Pearson correlation of ranks computed after pairwise filtering.
    rx = rankdata(xv, method='average')
    ry = rankdata(yv, method='average')
    rx -= rx.mean()
    ry -= ry.mean()
    denom = np.sqrt(np.dot(rx, rx) * np.dot(ry, ry))
    if denom == 0:
        continue
    rho = float(np.dot(rx, ry) / denom)
    valid_corrs += 1
    if rho >= 0.6:
        count += 1
    if 0.58 <= rho <= 0.62:
        near.append((gene, rho, n))

print('common_rows', len(common_rows))
print('common_cols', len(common_cols))
print('valid_corrs', valid_corrs)
print('count_ge_0.6', count)
print('near_threshold', len(near))
print('top_near', sorted(near, key=lambda t: abs(t[1]-0.6))[:10])
Path('codex_output').mkdir(exist_ok=True)
Path('codex_output/answer.txt').write_text(str(count) + '\\n')
PY"
