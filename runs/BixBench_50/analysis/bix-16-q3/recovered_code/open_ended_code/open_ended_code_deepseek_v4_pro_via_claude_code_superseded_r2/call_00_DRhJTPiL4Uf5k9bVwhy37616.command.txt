# Parse gene names (strip the Entrez ID part in parentheses)
def parse_gene_name(col_name):
    # Format: 'GENE (ID)' or just 'GENE'
    col_name = col_name.strip()
    if '(' in col_name:
        return col_name.split('(')[0].strip()
    return col_name

crispr_gene_names = set(parse_gene_name(g) for g in crispr_genes)
omics_gene_names = set(parse_gene_name(g) for g in omics_genes)

print(f'CRISPR unique gene names: {len(crispr_gene_names)}')
print(f'Omics unique gene names: {len(omics_gene_names)}')
print(f'Gene name intersection: {len(crispr_gene_names & omics_gene_names)}')
