static vector<string> split_csv(const string& line) {
    vector<string> out;
    size_t start = 0;
    while (true) {
        size_t pos = line.find(',', start);
        if (pos == string::npos) { out.emplace_back(line.substr(start)); break; }
        out.emplace_back(line.substr(start, pos - start));
        start = pos + 1;
    }
    return out;
}

static string first_field(const string& line) {
    size_t p = line.find(',');
    return p == string::npos ? line : line.substr(0, p);
}

static string feature_id(const string& x) {
    size_t p = x.rfind('(');
    if (p == string::npos || x.empty() || x.back() "'!= '"')') return x;
    return x.substr(p + 1, x.size() - p - 2);
}

static double parse_num(const char* b, const char* e) {
    if (b == e) return numeric_limits<double>::quiet_NaN();
    string s(b, e);
    if (s == \"NA\" || s == \"NaN\" || s == \"nan\") return numeric_limits<double>::quiet_NaN();
    char* endp = nullptr;
    double v = strtod(s.c_str(), &endp);
    if (endp == s.c_str()) return numeric_limits<double>::quiet_NaN();
    return v;
}

static vector<string> read_header(const string& path) {
    ifstream f(path);
    string line;
    if ("'!getline(f, line)) throw runtime_error("cannot read header: " + path);
    if (!line.empty() && line.back() == '"'\\r') line.pop_back();
    return split_csv(line);
}

static vector<string> read_ids(const string& path) {

    string line;

    vector<string> ids;

}

static void load_selected(const string& path,
                          const unordered_map<string,int>& row_idx,
                          const vector<int>& col_to_pair,
                          int nrows, int npairs,
                          vector<vector<double>>& mat,
                          long long& missing_count,
                          long long& parsed_selected) {
    ifstream f(path);
    string line;
    if (!getline(f, line)) throw runtime_error("cannot read data: " + path);
    int row_seen = 0;
    while (getline(f, line)) {
        if (!line.empty() && line.back() == '"'\\r') line.pop_back();
        if (line.empty()) continue;
        string id = first_field(line);
        auto rit = row_idx.find(id);
        int target = rit == row_idx.end() ? -1 : rit->second;
        size_t start = 0;
        int col = 0;
        while (start <= line.size()) {
            size_t pos = line.find(',', start);
            size_t end = (pos == string::npos) ? line.size() : pos;
            if (col > 0 && col < (int)col_to_pair.size()) {
                int p = col_to_pair[col];
                if (p >= 0 && target >= 0) {
                    double v = parse_num(line.data() + start, line.data() + end);
                    mat[target][p] = v;
                    ++parsed_selected;
                    if ("'!isfinite(v)) ++missing_count;
                }
            }
            ++col;
            if (pos == string::npos) break;
            start = pos + 1;
        }
        ++row_seen;
    }
    if (row_seen != (int)read_ids(path).size()) {
        throw runtime_error("row count changed unexpectedly: " + path);
    }
}

static double spearman_col(const vector<vector<double>>& a,
                           const vector<vector<double>>& b, int col,
                           int& nvalid) {
    vector<pair<double,int>> va, vb;
    int n = (int)a.size();
    va.reserve(n); vb.reserve(n);
    for (int i=0; i<n; ++i) {
        double x=a[i][col], y=b[i][col];
        if (isfinite(x) && isfinite(y)) {
            va.emplace_back(x,i);
            vb.emplace_back(y,i);
        }
    }
    nvalid = (int)va.size();
    if (nvalid < 2) return numeric_limits<double>::quiet_NaN();
    sort(va.begin(), va.end(), [](const auto& x, const auto& y) {
        if (x.first != y.first) return x.first < y.first;
        return x.second < y.second;
    });
    sort(vb.begin(), vb.end(), [](const auto& x, const auto& y) {
        if (x.first != y.first) return x.first < y.first;
        return x.second < y.second;
    });
    vector<double> ra(nvalid), rb(nvalid);
    for (int i=0; i<nvalid; ) {
        int j=i+1;
        while (j<nvalid && va[j].first == va[i].first) ++j;
        double rank = (i + 1 + j) / 2.0;
        for (int k=i; k<j; ++k) ra[va[k].second == va[k].second ? 0 : 0] = rank;
        i=j;
    }
    // The temporary rank arrays above are indexed by observation, not sorted position.
    // Rebuild them explicitly to keep the code clear and handle arbitrary row order.
    vector<double> rao(nvalid), rbo(nvalid);
    for (int i=0; i<nvalid; ) {
        int j=i+1;
        while (j<nvalid && va[j].first == va[i].first) ++j;
        double rank = (i + 1 + j) / 2.0;
        for (int k=i; k<j; ++k) rao[va[k].second] = rank;
        i=j;
    }
    for (int i=0; i<nvalid; ) {
        int j=i+1;
        while (j<nvalid && vb[j].first == vb[i].first) ++j;
        double rank = (i + 1 + j) / 2.0;
        for (int k=i; k<j; ++k) rbo[vb[k].second] = rank;
        i=j;
    }
    double mean=(nvalid+1)/2.0;
    long double num=0, dx=0, dy=0;
    for (int i=0; i<nvalid; ++i) {
        long double x=rao[i]-mean, y=rbo[i]-mean;
        num += x*y; dx += x*x; dy += y*y;
    }
    if (dx == 0 || dy == 0) return numeric_limits<double>::quiet_NaN();
    return (double)(num / sqrt(dx*dy));
}
