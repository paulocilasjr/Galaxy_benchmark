/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time, re
from collections import defaultdict
from scipy.stats import rankdata

t=time.time()
# headers
he=pd.read_csv('inputs/CRISPRGeneEffect.csv', nrows=0).columns.tolist()[1:]
hx=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', nrows=0).columns.tolist()[1:]
def sym(x): return re.sub(r'\\s*\\(\\d+\\)\\s*"'$'"','',x)
def ent(x): return int(re.search(r'\\((\\d+)\\)\\s*"'$'"',x).group(1))
ex_by=defaultdict(list)
for x in hx: ex_by[ent(x)].append(x)
pairs=[]
for e in he:
    ei=ent(e)
    opts=ex_by.get(ei)
    if not opts: continue
    if len(opts)==1:
        xcol=opts[0]
    else:
        es=sym(e)
        matches=[x for x in opts if sym(x)==es]
        xcol=matches[0] if matches else opts[0]
    pairs.append((e,xcol,ei))
print('entrez-matched pairs',len(pairs),flush=True)
eff=pd.read_csv('inputs/CRISPRGeneEffect.csv', index_col=0)
expr=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', index_col=0)
common=eff.index.intersection(expr.index)
eff=eff.loc[common]; expr=expr.loc[common]
E=eff[[p[0] for p in pairs]].to_numpy(dtype=np.float64)
X=expr[[p[1] for p in pairs]].to_numpy(dtype=np.float64)
mask=np.isfinite(E)&np.isfinite(X)
print(E.shape,'nan',np.isnan(E).sum(),np.isnan(X).sum(),'valid min max',mask.sum(0).min(),mask.sum(0).max(),flush=True)
count=0; rhos=np.full(len(pairs),np.nan); nvals=np.zeros(len(pairs),dtype=int)
for j in range(len(pairs)):
    m=mask[:,j]; n=int(m.sum()); nvals[j]=n
    if n<2: continue
    s=-E[m,j]; x=X[m,j]
    rs=rankdata(s); rx=rankdata(x)
    sds=rs.std(ddof=1); sdx=rx.std(ddof=1)
    if sds==0 or sdx==0: continue
    num=((rs-rs.mean())*(rx-rx.mean())).sum()
    den=np.sqrt(((rs-rs.mean())**2).sum()*((rx-rx.mean())**2).sum())
    rho=num/den; rhos[j]=rho
    if rho>=.6: count+=1
print('rho finite',np.isfinite(rhos).sum(),'min max',np.nanmin(rhos),np.nanmax(rhos),'count>=.6',count,'count<=-.6',int((rhos<=-.6).sum()),flush=True)
idx=np.argsort(rhos)
print('top 30:')
for i in idx[-30:][::-1]: print(pairs[i][0], '->', pairs[i][1], rhos[i], nvals[i])
print('bottom 10:')
for i in idx[:10]: print(pairs[i][0],'->',pairs[i][1],rhos[i],nvals[i])
print('sec',time.time()-t,flush=True)
PY"
