/bin/bash -lc "python - <<'PY'
import csv,re
from pathlib import Path
files=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
heads=[]
for fn in files:
    with open(fn,newline='') as f:
        h=next(csv.reader(f))
    vals=[]
    for x in h[1:]:
        m=re.fullmatch(r'(.*) \\((\\d+)\\)',x)
        vals.append((x,m.group(1) if m else None,int(m.group(2)) if m else None))
    ids=[x[2] for x in vals]
    labels=[x[0] for x in vals]
    print(fn,'features',len(vals),'unique labels',len(set(labels)),'unique entrez',len(set(ids)),'malformed',sum(x[2] is None for x in vals))
    heads.append(vals)
ids0={x[2] for x in heads[0]}; ids1={x[2] for x in heads[1]}
labels0={x[0] for x in heads[0]}; labels1={x[0] for x in heads[1]}
print('shared entrez',len(ids0&ids1),'shared exact labels',len(labels0&labels1))
# symbol discrepancies on same ID
m0={x[2]:x[1] for x in heads[0]}; m1={x[2]:x[1] for x in heads[1]}
diff=[(i,m0[i],m1[i]) for i in ids0&ids1 if m0[i]"'!=m1[i]]
print('"'same ID symbol differences',len(diff),diff[:30])
# row IDs, dupes, intersection, in model
rows=[]
for fn in files:
    with open(fn,newline='') as f:
        rr=csv.reader(f); next(rr); x=[r[0] for r in rr]
    print(fn,'rows',len(x),'unique',len(set(x)),'dupes',len(x)-len(set(x)))
    rows.append(x)
with open('inputs/Model.csv',newline='') as f:
    rr=csv.DictReader(f); mids=[r['ModelID'] for r in rr]
print('shared matrix row IDs',len(set(rows[0])&set(rows[1])))
print('matrix IDs absent Model',len((set(rows[0])|set(rows[1]))-set(mids)))
PY"
