/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time, warnings
from scipy.stats import rankdata
warnings.filterwarnings('ignore')
t=time.time()
ce=pd.read_csv('inputs/CRISPRGeneEffect.csv', index_col=0)
ex=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', index_col=0)
idx=ce.index.intersection(ex.index); cols=ce.columns.intersection(ex.columns)
X=ex.loc[idx,cols].to_numpy(dtype=np.float64)
Y=ce.loc[idx,cols].to_numpy(dtype=np.float64)
for label, yy in [('raw', Y), ('essentiality', -Y)]:
    count=0; top=[]
    for j in range(X.shape[1]):
        x=X[:,j]; y=yy[:,j]; mask=np.isfinite(y)
        if mask.sum()<2: continue
        xv=x[mask]; yv=y[mask]
        rx=rankdata(xv); ry=rankdata(yv)
        if np.std(rx)==0 or np.std(ry)==0: continue
        r=np.corrcoef(rx,ry)[0,1]
        if r>=0.6:
            count+=1
            if len(top)<5: top.append((cols[j], float(r)))
    print(label, 'count', count, 'top', top, 'time', time.time()-t)
PY"
