/bin/bash -lc "python - <<'PY'
import csv, time, os
import pandas as pd

crispr_path='inputs/CRISPRGeneEffect.csv'
expr_path='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
model_path='inputs/Model.csv'
start=time.time()
with open(crispr_path, newline='') as f: crispr_header=next(csv.reader(f))
with open(expr_path, newline='') as f: expr_header=next(csv.reader(f))
shared=[g for g in crispr_header[1:] if g in set(expr_header[1:])]
shared_set=set(shared)
crispr_usecols=[0]+[i for i,g in enumerate(crispr_header[1:], start=1) if g in shared_set]
expr_usecols=[0]+[i for i,g in enumerate(expr_header[1:], start=1) if g in shared_set]
ge=pd.read_csv(crispr_path, usecols=crispr_usecols, index_col=0)
expr=pd.read_csv(expr_path, usecols=expr_usecols, index_col=0)
model=pd.read_csv(model_path, usecols=['ModelID','ModelType']).set_index('ModelID')
models=ge.index.intersection(expr.index)
print('shared all', len(models), model.loc[models,'ModelType'].value_counts(dropna=False).to_dict(), flush=True)
cell_models=models[model.loc[models,'ModelType'].eq('Cell Line').to_numpy()]
print('cell models', len(cell_models), flush=True)

def count_for(idx):
    g=ge.loc[idx, shared]
    x=expr.loc[idx, shared]
    y=-g
    corr=x.corrwith(y, axis=0, method='spearman')
    return int((corr>=0.6).sum()), corr.sort_values(ascending=False).head(10)
for name, idx in [('all', models), ('cell', cell_models)]:
    c, top=count_for(idx)
    print(name, 'count', c, 'top')
    print(top.to_string())
print('elapsed', round(time.time()-start,2))
PY"
