/bin/bash -lc "PYTHONPATH=/workspace/.local_deps python3 - <<'PY'
import csv
from pathlib import Path
import numpy as np
import pandas as pd
from scipy.stats import rankdata

crispr_path=Path('inputs/CRISPRGeneEffect.csv')
expr_path=Path('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
model_path=Path('inputs/Model.csv')

def header(path):
    with path.open(newline='') as f:
        return next(csv.reader(f))

def first_col(path):
    with path.open(newline='') as f:
        rd=csv.reader(f); next(rd)
        return [row[0] for row in rd]

ch=header(crispr_path)
eh=header(expr_path)
epos={x:i for i,x in enumerate(eh)}
common=[x for x in ch[1:] if x in epos]
common_set=set(common)
cp=[0]+[i for i,x in enumerate(ch) if x in common_set]
ep=[0]+[epos[x] for x in common]
cs=set(first_col(crispr_path)); es=set(first_col(expr_path)); model_ids=first_col(model_path)
shared=[x for x in model_ids if x in cs and x in es]

dtypes={x:np.float32 for x in common}
print(f'Loading aligned matrices: {len(shared)} models x {len(common)} same-gene features',flush=True)
c=pd.read_csv(crispr_path,usecols=cp,index_col=0,dtype=dtypes)
e=pd.read_csv(expr_path,usecols=ep,index_col=0,dtype=dtypes)
c=c.loc[shared,common]
e=e.loc[shared,common]
ca=c.to_numpy(copy=False)
ea=e.to_numpy(copy=False)
print('Missing effect entries:',int(np.isnan(ca).sum()),'missing expression entries:',int(np.isnan(ea).sum()),flush=True)

n_features=len(common)
rhos_raw=np.full(n_features,np.nan,dtype=np.float64)
n_valid=np.zeros(n_features,dtype=np.int32)
for j in range(n_features):
    y=ca[:,j]
    valid=np.isfinite(y) & np.isfinite(ea[:,j])
    n=int(valid.sum()); n_valid[j]=n
    if n<2:
        continue
    # Spearman = Pearson correlation of average ranks after pairwise filtering.
    rx=rankdata(ea[valid,j],method='average')
    ry=rankdata(y[valid],method='average')
    rx-=rx.mean(); ry-=ry.mean()
    denom=np.sqrt(np.dot(rx,rx)*np.dot(ry,ry))
    if denom:
        rhos_raw[j]=np.dot(rx,ry)/denom

# DepMap gene-effect is a negative-going coordinate; biological essentiality is -gene_effect.
rhos_essentiality=-rhos_raw
count_essentiality=int(np.count_nonzero(rhos_essentiality >= 0.6))
count_raw=int(np.count_nonzero(rhos_raw >= 0.6))
print('valid n range/median:',int(n_valid.min()),int(np.median(n_valid)),int(n_valid.max()),flush=True)
print('Spearman count, biological essentiality (rho >= 0.6):',count_essentiality,flush=True)
print('Spearman count, raw gene-effect coordinate (rho >= 0.6):',count_raw,flush=True)
print('essentiality rho range:',float(np.nanmin(rhos_essentiality)),float(np.nanmax(rhos_essentiality)),flush=True)
# Report a compact threshold check for reproducibility without writing an audit file.
near=np.flatnonzero((rhos_essentiality>=0.58)&(rhos_essentiality<=0.62))
print('near-threshold feature count:',len(near),flush=True)
for j in near[np.argsort(rhos_essentiality[near])]:
    print(common[j], 'n=',int(n_valid[j]), 'rho=',format(float(rhos_essentiality[j]),'.12f'),flush=True)
PY"
