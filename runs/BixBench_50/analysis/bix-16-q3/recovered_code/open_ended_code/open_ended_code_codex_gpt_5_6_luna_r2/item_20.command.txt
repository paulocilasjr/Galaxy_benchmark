import csv
from collections import Counter, defaultdict
p='inputs/Model.csv'
with open(p,newline='') as f:

# shared ids from files, just get first cols (not parse numeric)

    with open(p,newline='') as f:
        rd=csv.reader(f);next(rd);return {r[0] for r in rd}
shared=get_ids('inputs/CRISPRGeneEffect.csv') & get_ids('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
sub=[r for r in rows if r['ModelID'] in shared]
print('shared rows',len(sub))
for col in ['DepmapModelType','CellLineName','StrippedCellLineName','OncotreeLineage','ModelDerivation','ModelType']:
    if col in rd.fieldnames:
        vals=Counter(r[col] for r in sub)
        print('\\n',col,'nunique',len(vals),'top',vals.most_common(20))
print('blank counts', {c:sum(r[c]=='' for r in sub) for c in rd.fieldnames})
