/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
for path in ['inputs/Proteomic_data .xlsx','inputs/MeRIP_RNA_result.xlsx']:
    print('\\nFILE', path)
    wb = load_workbook(path, read_only=True, data_only=True)
    print('sheets:', wb.sheetnames)
    for ws in wb.worksheets:
        print('SHEET', repr(ws.title), 'max_row', ws.max_row, 'max_col', ws.max_column)
        rows = list(ws.iter_rows(min_row=1, max_row=min(8, ws.max_row), values_only=True))
        for r in rows:
            print(r)
        print()
PY"
