from openpyxl import load_workbook
p='inputs/Proteomic_data .xlsx'

    desc=str(vals.get('Description') or '')
    if gene.upper()=='ENO1' or 'ENO1' in desc.upper() or 'ENOLASE' in desc.upper():
        print(i, vals)
