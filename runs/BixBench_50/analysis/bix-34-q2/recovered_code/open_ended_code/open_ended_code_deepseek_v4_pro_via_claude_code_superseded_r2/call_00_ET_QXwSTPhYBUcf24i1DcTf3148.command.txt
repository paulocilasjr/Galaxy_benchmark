import dendropy

# Read the tree
tree = dendropy.Tree.get(path='/tmp/gene_981902/981902at2759.faa.mafft.clipkit.treefile', schema='newick')

# Print tree info
print("Tree structure:")
print(tree.as_string(schema='newick'))
print()

# Get taxon names
taxa = [node.taxon.label for node in tree.leaf_node_iter()]
print("Taxa:", taxa)
print()

# Compute all pairwise patristic distances
pdm = tree.phylogenetic_distance_matrix()
print("Patristic distance matrix:")
for t1 in taxa:
    for t2 in taxa:
        if t1 < t2:
            d = pdm.distance(t1, t2)
            print(f"  {t1} <-> {t2}: {d}")

# Get all pairwise distances and compute median
distances = []
for i, t1 in enumerate(taxa):
    for t2 in taxa[i+1:]:
        distances.append(pdm.distance(t1, t2))

distances.sort()
print()
print("Sorted pairwise distances:")
for d in distances:
    print(f"  {d}")

n = len(distances)
if n % 2 == 0:
    median = (distances[n//2 - 1] + distances[n//2]) / 2
else:
    median = distances[n//2]

print()
print(f"Number of pairwise distances: {n}")
print(f"Median patristic distance: {median}")
print(f"Median patristic distance: {median:.10f}")
