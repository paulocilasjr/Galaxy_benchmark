import statistics

def parse(nwk):
    s=nwk.strip()
    if s.endswith(';'): s=s[:-1]
    pos=0
    def node():
        nonlocal pos
        if s[pos]=='(':
            pos+=1
            ch=[]
            while True:
                ch.append(node())
                if pos<len(s) and s[pos]==',':
                    pos+=1; continue
                if pos<len(s) and s[pos]==')':
                    pos+=1; break
                raise ValueError((s,pos))
            st=pos
            while pos<len(s) and s[pos] not in ',);': pos+=1
            lab=s[st:pos]
            length=float(lab.split(':')[1]) if ':' in lab else 0.0
            return ('int',ch,length)
        st=pos
        while pos<len(s) and s[pos] not in ',();': pos+=1
        tok=s[st:pos]
        if ':' in tok:
            name, length=tok.split(':')
            return ('leaf',name,float(length))
        return ('leaf',tok,0.0)
    return node()

def all_pair_dists(nwk):
    t=parse(nwk)
    edges=[]; idx=0
    def walk(n, parent=None, plen=0.0):
        nonlocal idx
        myid=idx; idx+=1
        if parent is not None:
            edges.append((parent,myid,plen))
        if n[0]=='int':
            for c in n[1]: walk(c,myid,c[2])

        if n[0]=='leaf': leaves.append(n[1])
