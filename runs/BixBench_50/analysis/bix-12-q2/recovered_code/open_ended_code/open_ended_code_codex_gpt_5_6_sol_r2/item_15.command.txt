/bin/bash -lc "PYTHONPATH=/workspace/phykit-check.IrBZeg python - <<'PY'
from io import StringIO
from statistics import median
from zipfile import ZipFile
from Bio import AlignIO
from phykit.services.alignment.parsimony_informative_sites import ParsimonyInformative
values=[]
raw_values=[]
with ZipFile('inputs/scogs_fungi.zip') as z:
    names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    for n in names:
        aln=AlignIO.read(StringIO(z.read(n).decode()), 'fasta')
        count,length,pct=ParsimonyInformative.calculate_parsimony_informative_sites(None,aln,True)
        values.append(round(pct,4))
        raw_values.append(pct)
print('selected',len(names))
print('processed',len(values))
print('failed',0)
print('median_raw',median(raw_values))
print('median_reported',median(values))
print('middle_file',names[sorted(range(len(values)),key=values.__getitem__)[len(values)//2]])
print('middle_value',sorted(values)[len(values)//2])
PY"
