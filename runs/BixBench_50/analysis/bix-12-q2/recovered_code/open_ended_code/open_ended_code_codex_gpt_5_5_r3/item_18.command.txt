from zipfile import ZipFile
from collections import Counter
from statistics import median


def parse_fasta(text):
    rec=[]; name=None; chunks=[]
    for raw in text.splitlines():
        line=raw.strip()
        if not line:
            continue
        if line.startswith('>'):
            if name is not None:
                rec.append((name, ''.join(chunks).upper()))
            name=line[1:].split()[0]
            chunks=[]
        else:
            chunks.append(line)
    if name is not None:
        rec.append((name, ''.join(chunks).upper()))
    return rec

values=[]
with ZipFile('inputs/scogs_fungi.zip') as z:
    for member in sorted(n for n in z.namelist() if n.endswith('.faa.mafft')):
        records=parse_fasta(z.read(member).decode('utf-8'))
        seqs=[seq for _,seq in records]
        lengths={len(seq) for seq in seqs}
        if len(lengths) "'!= 1:
            raise ValueError(f'"'unequal alignment lengths in {member}: {sorted(lengths)}')
        length=lengths.pop()
        pi=0
        for column in zip(*seqs):
            counts=Counter(c for c in column if c not in '-?')
            if sum(count >= 2 for count in counts.values()) >= 2:
                pi += 1
        values.append(pi / length * 100)

answer=f'{median(values)}%'
with open('codex_output/answer.txt','w',encoding='ascii') as out:
    out.write(answer + '\\n')
print(answer)
PY
printf 'answer.txt: '; cat codex_output/answer.txt"
