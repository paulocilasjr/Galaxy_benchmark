/bin/bash -lc "python - <<'PY'
import zipfile
from itertools import islice

def parse_fasta(txt):
    seqs=[]; name=None; chunks=[]
    for line in txt.splitlines():
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:].strip(); chunks=[]
        elif line.strip(): chunks.append(line.strip())
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    for n in list(islice(z.namelist(), 20)):
        if n.endswith('/') : continue
        seqs=parse_fasta(z.read(n).decode())
        lens=sorted(set(map(lambda x: len(x[1]), seqs)))
        print(n, 'nseq', len(seqs), 'lens', lens[:10], 'equal', len(lens)==1)
PY"
