/bin/bash -lc "python - <<'PY'
import zipfile, collections
chars=collections.Counter()
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    for n in z.namelist():
        if n.endswith('.faa.mafft'):
            for line in z.read(n).decode().splitlines():
                if line and not line.startswith('>'):
                    chars.update(line.strip())
print('chars:', ''.join(sorted(chars)))
for k in sorted(chars):
    print(repr(k), chars[k])
PY"
