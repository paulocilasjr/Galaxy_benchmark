/bin/bash -lc "python - <<'PY'
import zipfile, statistics, collections

def parse_fasta(data):
    seqs=[]; name=None; chunks=[]
    for line in data.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:]; chunks=[]
        else: chunks.append(line)
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs

def pi(seqs):
    count=0
    for col in zip(*(s for _,s in seqs)):
        c=collections.Counter(ch.upper() for ch in col if ch not in '-?' and ch.upper() not in {'X','N'})
        if sum(v>=2 for v in c.values()) >= 2: count += 1
    return count, len(seqs[0][1])
rows=[]
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    for n in sorted(x for x in z.namelist() if x.endswith('.faa.mafft')):
        seqs=parse_fasta(z.read(n)); p,L=pi(seqs)
        rows.append((n,len(seqs),p/L*100,p,L))
for label, sub in [('all',rows),('nseq4',[r for r in rows if r[1]==4]),('nseq>=4',[r for r in rows if r[1]>=4])]:
    vals=[r[2] for r in sub]
    print(label, len(vals), statistics.median(vals), 'zeros', sum(v==0 for v in vals))
print('all rows zero counts by nseq', collections.Counter(r[1] for r in rows if r[2]==0))
PY"
