/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import PurePosixPath
p='inputs/scogs_fungi.zip'
with ZipFile(p) as z:
    names=[x.filename for x in z.infolist() if not x.is_dir()]
    print('members',len(names))
    print('suffixes',Counter('.faa.mafft' if n.endswith('.faa.mafft') else '.faa' if n.endswith('.faa') else PurePosixPath(n).suffix for n in names))
    ma=[n for n in names if n.endswith('.faa.mafft')]
    raw=[n for n in names if n.endswith('.faa')]
    print('mafft',len(ma),'raw',len(raw),'matched',len({n[:-6] for n in ma}&set(raw)))
    for n in sorted(ma)[:3]:
        s=z.read(n).decode()
        print('\\nMEMBER',n)
        print('\\n'.join(s.splitlines()[:12]))
PY"
