/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median
P='inputs/scogs_fungi.zip'

def fasta(txt):
    rec=[]; name=None; seq=[]
    for line in txt.splitlines():
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(seq)))
            name=line[1:].strip(); seq=[]
        else: seq.append(line.strip())
    if name is not None: rec.append((name,''.join(seq)))
    return rec

def ispi(chars, missing):
    c=Counter(x for x in chars if x not in missing)
    return sum(v>=2 for v in c.values())>=2
rows=[]; alpha=Counter(); taxa=Counter(); bad=[]
with ZipFile(P) as z:
  for n in sorted(x.filename for x in z.infolist() if x.filename.endswith('.faa.mafft')):
    rec=fasta(z.read(n).decode())
    lens={len(s) for _,s in rec}; taxa[len(rec)]+=1
    if len(lens)"'!=1: bad.append((n,lens))
    L=next(iter(lens)); seqs=[s.upper() for _,s in rec]
    alpha.update('"''.join(seqs))
    vals={}
    for label,missing in [('gap_missing',set('-?')),('gap_x_missing',set('-?X*')),('nothing_missing',set())]:
      k=sum(ispi(cs,missing) for cs in zip(*seqs))
      vals[label]=(k,100*k/L)
    nogap=sum('-' not in cs and '?' not in cs for cs in zip(*seqs))
    kcg=sum(ispi(cs,set()) for cs in zip(*seqs) if '-' not in cs and '?' not in cs)
    vals['complete_cols']=(kcg,100*kcg/nogap if nogap else float('nan'))
    rows.append((n,L,nogap,vals))
print('taxon counts',taxa,'unequal',len(bad))
print('alphabet',alpha)
for key in ['gap_missing','gap_x_missing','nothing_missing','complete_cols']:
    ps=[r[3][key][1] for r in rows]
    print(key,'median',median(ps),'range',min(ps),max(ps))
    mid=sorted(rows,key=lambda r:r[3][key][1])[len(rows)//2]
    print(' median row',mid[0],'L',mid[1],'nogap',mid[2],'count,pct',mid[3][key])
PY"
