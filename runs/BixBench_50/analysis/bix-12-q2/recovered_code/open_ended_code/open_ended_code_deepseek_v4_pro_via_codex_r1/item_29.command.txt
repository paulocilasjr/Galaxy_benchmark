/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
from collections import Counter
import statistics
def parse(path):
    seqs=[]; cur=None; seq=[]
    for line in path.read_text().splitlines():
        if line.startswith('>'):
            if cur is not None:
                seqs.append((cur,''.join(seq)))
            cur=line[1:].strip(); seq=[]
        elif cur is not None:
            seq.append(line.strip())
    if cur is not None: seqs.append((cur,''.join(seq)))
    return seqs
vals=[]
for f in sorted(Path('/workspace/work/scogs_fungi').glob('*.faa.mafft')):
    seqs=parse(f)
    arr=[s for n,s in seqs]
    L=len(arr[0])
    pi=0
    # AMAS missing/ambiguous: determine alphabet/missing chars in source
    for j in range(L):
        chars=[a[j] for a in arr]
        if any(c in '-?X*BZJ.' for c in chars):
            pass
        else:
            c=Counter(chars)
            pattern=[ch for ch in c if c[ch]>=2]
            if len(pattern)>=2:
                pi+=1
    vals.append((f.name, pi/L*100, pi, L, len(arr)))
vals.sort(key=lambda x:x[1])
print('n',len(vals),'median',statistics.median([v[1] for v in vals]))
for i,x in enumerate(vals[120:135],start=120): print(i,x)
PY"
