/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
from collections import Counter
import statistics
amb=set('BJZX.*-?')
pcts=[]
for path in sorted(Path('/workspace/work/scogs_fungi').glob('*.faa.mafft')):
    seqs=[]; name=None; buf=[]
    for line in path.read_text().splitlines():
        if line.startswith('>'):
            if name is not None:
                seqs.append((name,''.join(buf)))
            name=line[1:].strip(); buf=[]
        else:
            buf.append(line.strip())
    if name is not None:
        seqs.append((name,''.join(buf)))
    cols=list(zip(*[s for _,s in seqs]))
    L=len(cols)
    pi=0
    for col in cols:
        if any(c in amb for c in col):
            continue
        counts=Counter(col)
        informative=sum(1 for c in counts if counts[c]>=2)
        if informative>=2:
            pi+=1
    pcts.append(pi/L*100)
print('alignments:',len(pcts))
print('median percentage:',statistics.median(pcts))
print('median as fraction:',sum(sorted(pcts)[125:128]) if False else statistics.median(pcts))
print('min/max',min(pcts),max(pcts),'mean',statistics.mean(pcts))
PY"
