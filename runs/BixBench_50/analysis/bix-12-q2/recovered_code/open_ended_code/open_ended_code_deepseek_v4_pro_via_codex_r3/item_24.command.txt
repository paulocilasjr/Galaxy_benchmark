/bin/bash -c 'work=$(find /workspace/tmp -maxdepth 1 -type d -name '"'scogs_fungi.*' | head -1) && python3 - <<'PY'
import glob, subprocess, os, statistics, math
work=glob.glob('/workspace/tmp/scogs_fungi.*')[0]
files=sorted(glob.glob(work+'/*.mafft'))
vals=[]
exact=[]
for f in files:
    out=subprocess.check_output(['phykit','parsimony_informative_sites',f], text=True).strip()
    n,total,p=out.split('\\t')
    vals.append(float(p))
    exact.append(int(n)/int(total)*100)
print(len(vals))
print('median rounded4',statistics.median(vals))
print('median exact',statistics.median(exact))
print('mean rounded4',statistics.mean(vals), 'mean exact',statistics.mean(exact))
print('min max',min(vals),max(vals), min(exact),max(exact))
print('quantiles rounded',[statistics.quantiles(vals,n=4), statistics.quantiles(exact,n=4)])
PY"
