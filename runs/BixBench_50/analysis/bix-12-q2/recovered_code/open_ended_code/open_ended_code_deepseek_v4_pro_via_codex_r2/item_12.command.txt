/bin/bash -c "python - <<'PY'
import zipfile, io, collections, statistics
path='inputs/scogs_fungi.zip'
with zipfile.ZipFile(path) as z:
 names=[n for n in z.namelist() if n.endswith('.faa.mafft')]
 print('alignments',len(names))
 lens=[]; nseqs=[]; coltypes=collections.Counter(); gaponly=[]
 for n in names:
  data=z.read(n).decode()
  seqs=[]; cur=None; name=''
  for line in data.splitlines():
   if line.startswith('>'):
    if cur is not None: seqs.append(cur)
    name=line[1:]; cur=[]
   else:
    cur.append(line.strip())
  if cur is not None: seqs.append(cur)
  seqs=[''.join(s) for s in seqs]
  L=len(seqs[0]) if seqs else 0
  lens.append(L); nseqs.append(len(seqs))
  ngap=0; nconst=0; nvar=0
  for col in zip(*seqs):
   chars=[c for c in col if c not in '-X?.']
   # gap only
   if not chars:
    gaponly.append(1); continue
   uniq=set(chars)
   if len(uniq)==1: nconst+=1
   else: nvar+=1
  coltypes[(len(seqs), 'gap', sum(1 for col in zip(*seqs) if all(c=='-' for c in col)))] += 1
 print('lens min max mean median', min(lens),max(lens),sum(lens)/len(lens),statistics.median(lens))
 print('seq counts',collections.Counter(nseqs))
 print('len=0',sum(1 for l in lens if l==0))
 print('sample first 5 lens',lens[:5])
 # print head of shortest
 minname=names[lens.index(min(lens))]
 print('shortest',minname)
PY"
