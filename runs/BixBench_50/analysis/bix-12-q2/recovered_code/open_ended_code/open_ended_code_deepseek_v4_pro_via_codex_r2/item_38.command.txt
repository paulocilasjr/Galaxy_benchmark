/bin/bash -c "python - <<'PY'
import zipfile, collections, statistics
AA='ACDEFGHIKLMNPQRSTVWYBZJUO'  # all non -?Xx
path='inputs/scogs_fungi.zip'
with zipfile.ZipFile(path) as z:
 names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
 vals=[]; raw=[]
 for n in names:
  data=z.read(n).decode(); seqs=[]; cur=None
  for line in data.splitlines():
   if line.startswith('>'):
    if cur is not None: seqs.append(''.join(cur))
    cur=[]
   else: cur.append(line.strip())
  if cur is not None: seqs.append(''.join(cur))
  L=len(seqs[0]); npi=0
  for col in zip(*seqs):
   c=collections.Counter(ch for ch in col if ch not in '-?Xx')
   if sum(1 for v in c.values() if v>=2)>=2:
    npi+=1
  pct=round(100.0*npi/L,4) if L else None
  vals.append(pct); raw.append(100.0*npi/L)
 print('n',len(vals))
 print('median of rounded per-file',statistics.median(vals))
 print('mean rounded per-file',statistics.mean(vals))
 print('median raw',statistics.median(raw))
 print('mean raw',statistics.mean(raw))
 print('sorted around median:')
 s=sorted(vals); n=len(s)
 for i in range(n//2-3,n//2+4): print(i,s[i])
PY"
