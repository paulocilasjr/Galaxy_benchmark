/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

with ZipFile('inputs/scogs_fungi.zip') as z:
    names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    result=[]; chars=Counter(); nseqs=Counter(); bad=[]
    for n in names:
        seq=[]
        for line in z.read(n).decode().splitlines():
            if line.startswith('>'): seq.append('')
            else: seq[-1]+=line.strip().upper()
        nseqs[len(seq)]+=1
        lens={len(s) for s in seq}
        if len(lens)"'!=1: bad.append((n,lens)); continue
        L=len(seq[0]); chars.update('"''.join(seq))
        pi_missing=0; pi_gapstate=0
        for col in zip(*seq):
            c=Counter(x for x in col if x not in '-?X')
            if sum(v>=2 for v in c.values())>=2: pi_missing+=1
            c2=Counter(col)
            if sum(v>=2 for v in c2.values())>=2: pi_gapstate+=1
        result.append((n,L,pi_missing,100*pi_missing/L,pi_gapstate,100*pi_gapstate/L))
    print('nseq distribution',nseqs,'bad',bad)
    print('character inventory',chars)
    print('median length',median(x[1] for x in result))
    print('median PI count',median(x[2] for x in result))
    print('median PI pct missing',median(x[3] for x in result))
    print('median PI pct gaps-as-state',median(x[5] for x in result))
    s=sorted(result,key=lambda x:x[3])
    for x in s[124:131]: print(x)
    print('pooled missing',100*sum(x[2] for x in result)/sum(x[1] for x in result))
PY"
