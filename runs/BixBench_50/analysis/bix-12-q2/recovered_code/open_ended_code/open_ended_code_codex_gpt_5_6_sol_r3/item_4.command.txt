/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import PurePosixPath
zpath='inputs/scogs_fungi.zip'
with ZipFile(zpath) as z:
    names=[n for n in z.namelist() if not n.endswith('/')]
    print('all members',len(names))
    print('suffix classes',Counter('.faa.mafft' if n.endswith('.faa.mafft') else '.faa' if n.endswith('.faa') else PurePosixPath(n).suffix for n in names))
    alns=[n for n in names if n.endswith('.faa.mafft')]
    print('alignments',len(alns),'unique names',len(set(alns)))
    for n in alns[:3]:
        text=z.read(n).decode()
        seqs=[]; name=None
        for line in text.splitlines():
            if line.startswith('>'):
                name=line[1:]; seqs.append([name,''])
            else: seqs[-1][1]+=line.strip()
        print('\\n',n,'nseq',len(seqs),'lengths',[len(s) for _,s in seqs])
        print('\\n'.join(text.splitlines()[:8]))
PY"
