/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

with ZipFile('inputs/scogs_fungi.zip') as z:
    rows=[]
    for name in sorted(n for n in z.namelist() if n.endswith('.faa.mafft')):
        seqs=[]
        for line in z.read(name).decode('ascii').splitlines():
            if line.startswith('>'): seqs.append('')
            else: seqs[-1] += line.strip().upper()
        L=len(seqs[0])
        assert L and all(len(s)==L for s in seqs)
        pis=sum(1 for col in zip(*seqs)
                if sum(n >= 2 for n in Counter(c for c in col if c "'!= '"'-').values()) >= 2)
        rows.append((name,len(seqs),L,pis,pis/L*100))

    values=[r[4] for r in rows]
    ordered=sorted(values)
    mid=ordered[len(ordered)//2]
    central=[r for r in rows if r[4]==mid]
    print('processed',len(rows))
    print('taxon counts',dict(sorted(Counter(r[1] for r in rows).items())))
    print('zero-PIS alignments',sum(r[3]==0 for r in rows))
    print('median rank',len(ordered)//2+1,'of',len(ordered))
    print('median exact',mid)
    print('median rounded 2dp',f'{median(values):.2f}%')
    print('central row(s)',central)
    print('complete-four-only median',median(r[4] for r in rows if r[1]==4))
PY"
