/bin/bash -lc "python3 - <<'PY'
import zipfile, collections, statistics
zf=zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip')
chars=collections.Counter()
for n in zf.namelist():
    if not n.endswith('.faa.mafft'): continue
    for line in zf.read(n).decode().splitlines():
        if line and not line.startswith('>'):
            chars.update(line.strip())
print(chars)
print('non standard', {c:v for c,v in chars.items() if c.upper() not in set('ACDEFGHIKLMNPQRSTVWY-')})
PY"
