/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median
path='inputs/scogs_fungi.zip'
def read_fasta(data):
    seqs=[]; parts=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if parts or (seqs and seqs[-1][1] is not None):
                pass
            seqs.append([line[1:].strip(), ''])
        else:
            seqs[-1][1] += line.strip()
    return [tuple(x) for x in seqs]
def stats(seqs):
    cols=list(zip(*(s for _,s in seqs)))
    pis_all=0; pis_nogap=0; var_all=0; valid=0; pis_valid=0
    for col in cols:
        c=Counter(col)
        if len(c)>1: var_all += 1
        if sum(v>=2 for v in c.values())>=2: pis_all += 1
        ng=[x for x in col if x not in '-?XxNn']
        if ng:
            valid += 1
        cn=Counter(ng)
        if len(cn)>1: pass
        if sum(v>=2 for v in cn.values())>=2: pis_nogap += 1
        if len(ng)==len(col): pis_valid += sum(v>=2 for v in cn.values())>=2
    complete=sum(all(x not in '-?XxNn' for x in col) for col in cols)
    pis_complete=0
    for col in cols:
        if all(x not in '-?XxNn' for x in col):
            c=Counter(col)
            pis_complete += sum(v>=2 for v in c.values())>=2
    return len(cols),len(seqs),pis_all,pis_nogap,complete,pis_complete,var_all
rows=[]
with ZipFile(path) as z:
    for n in z.namelist():
        if n.endswith('.faa.mafft'):
            rows.append((n,stats(read_fasta(z.read(n)))))
print('rows',len(rows))
vals=sorted((100*s[3]/s[0],n,s[0],s[3],s[4]) for n,s in rows)
for x in vals[123:132]: print(x)
print('median',median([100*s[3]/s[0] for _,s in rows]))
PY"
