/bin/bash -lc "python - <<'PY'
from collections import Counter
from fractions import Fraction
from zipfile import ZipFile

values=[]
with ZipFile('inputs/scogs_fungi.zip') as z:
    for name in z.namelist():
        if not name.endswith('.faa.mafft'):
            continue
        records=[]
        for block in z.read(name).decode().strip().split('>')[1:]:
            lines=block.splitlines()
            records.append(''.join(lines[1:]))
        length=len(records[0])
        informative=0
        for column in zip(*records):
            counts=Counter(state for state in column if state not in {'-','?','X'})
            informative += sum(n >= 2 for n in counts.values()) >= 2
        values.append(Fraction(100 * informative, length))
assert len(values)==255
expected=sorted(values)[127]
assert expected == Fraction(46,13)
assert open('codex_output/answer.txt').read() == '3.538461538461538%\\n'
print('verified', len(values), expected, 'answer_file_one_line=true')
PY
wc -l codex_output/answer.txt
sed -n 'l' codex_output/answer.txt"
