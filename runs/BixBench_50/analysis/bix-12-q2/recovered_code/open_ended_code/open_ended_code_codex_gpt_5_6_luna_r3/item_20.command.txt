/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

def parse(b):
 rec=[]; cur=None
 for line in b.decode().splitlines():
  if line.startswith('>'): cur=[line[1:].strip(),[]]; rec.append(cur)
  else: cur[1].append(line.strip())
 return [(n,''.join(p)) for n,p in rec]
vals=[]
with ZipFile('inputs/scogs_fungi.zip') as z:
 for n in z.namelist():
  if n.endswith('.faa.mafft'):
   ss=parse(z.read(n));
   if len(ss)<4: continue
   L=len(ss[0][1]); p=0
   for col in zip(*(s for _,s in ss)):
    c=Counter(x for x in col if x not in '-?X')
    p += sum(v>=2 for v in c.values())>=2
   vals.append(100*p/L)
print(len(vals),median(vals),sorted(vals)[105], sorted(vals)[104:107])
PY"
