/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

def fasta(b):
 out=[]
 for block in b.decode().strip().split('>')[1:]:
  ls=block.splitlines(); out.append(''.join(ls[1:]))
 return out
vals={key:[] for key in ('all/all','nogap/all','nogap/non-gap')}
with ZipFile('inputs/scogs_fungi.zip') as z:
 for n in z.namelist():
  if n.endswith('.faa.mafft'):
   ss=fasta(z.read(n));
   if len(ss)<4: continue
   L=len(ss[0]); p=0; pn=0; complete=0
   for col in zip(*ss):
    c=Counter(col)
    if sum(v>=2 for v in c.values())>=2: p+=1
    ng=[x for x in col if x not in '-?XxNn']; cn=Counter(ng)
    if sum(v>=2 for v in cn.values())>=2: pn+=1
    if len(ng)==len(col): complete+=1
   vals['all/all'].append(100*p/L)
   vals['nogap/all'].append(100*pn/L)
   vals['nogap/non-gap'].append(100*pn/complete if complete else 0)
for k,a in vals.items(): print(k,len(a),median(a),sorted(a)[104:107])
PY"
