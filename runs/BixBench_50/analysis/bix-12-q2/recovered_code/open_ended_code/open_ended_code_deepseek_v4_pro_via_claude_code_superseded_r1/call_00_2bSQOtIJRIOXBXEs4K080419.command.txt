import zipfile
from collections import Counter

def parse_fasta(text):
    sequences = []
    current_header = None
    current_seq = []
    for line in text.split('\n'):
        line = line.strip()
        if not line:
            continue
        if line.startswith('>'):
            if current_header is not None:
                sequences.append((current_header, ''.join(current_seq)))
            current_header = line[1:].strip()
            current_seq = []
        else:
            current_seq.append(line)
    if current_header is not None:
        sequences.append((current_header, ''.join(current_seq)))
    return sequences

# Check a specific alignment
with zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip', 'r') as zf:
    content = zf.read('1001705at2759.faa.mafft').decode('utf-8')
    seqs = parse_fasta(content)
    print(f'File: 1001705at2759.faa.mafft')
    print(f'Sequences: {len(seqs)}')
    for h, s in seqs:
        print(f'  {h}: len={len(s)}')
    
    # Show first 20 columns
    seqqs = [s for _, s in seqs]
    for col in range(min(20, len(seqqs[0]))):
        chars = [s[col] for s in seqqs]
        counts = Counter(chars)
        states_ge2 = sum(1 for c in counts.values() if c >= 2)
        is_pi = states_ge2 >= 2
