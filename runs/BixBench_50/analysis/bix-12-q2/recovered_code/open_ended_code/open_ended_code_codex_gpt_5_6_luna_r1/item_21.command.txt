/bin/bash -lc "python - <<'PY'
import zipfile, re
p='inputs/scogs_fungi.zip'
with zipfile.ZipFile(p) as z:
    names=[n for n in z.namelist() if n.endswith('.faa.mafft')]
    for n in names[:12]:
        txt=z.read(n).decode()
        rec=[]
        for h,s in re.findall(r'>(["'^'"\\n]+)\\n(["'^>]*)'"',txt):
            seq=''.join(s.split())
            rec.append((h, len(seq), seq[:8]))
        print(n, len(rec), sorted(set(x[1] for x in rec)), rec[:4])
PY"
