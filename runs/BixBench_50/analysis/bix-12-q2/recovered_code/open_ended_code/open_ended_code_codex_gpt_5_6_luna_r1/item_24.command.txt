/bin/bash -lc "python - <<'PY'
import zipfile, statistics, collections, math
p='inputs/scogs_fungi.zip'
def recs(txt):
  out=[]; h=None; a=[]
  for l in txt.splitlines():
    if l.startswith('>'):
      if h is not None: out.append((h,''.join(a)))
      h=l[1:].split()[0];a=[]
    else:a.append(l.strip())
  if h is not None:out.append((h,''.join(a)))
  return out
def pis(seqs):
  n=0
  for col in zip(*seqs):
    c=collections.Counter(x for x in col if x not in '-?.')
    if sum(v>=2 for v in c.values())>=2:n+=1
  return n,len(seqs[0])
with zipfile.ZipFile(p) as z:
 rows=[]
 for n in z.namelist():
  if not n.endswith('.faa.mafft'):continue
  rr=recs(z.read(n).decode()); p0,L=pis([s for _,s in rr]); rows.append((len(rr),p0,L,n))
 for k in sorted(set(x[0] for x in rows)):
  a=[100*p/L for n,p,L,_ in rows if n==k]
  print(k,len(a),statistics.median(a),min(a),max(a),statistics.mean(a))
 print('all',statistics.median([100*p/L for _,p,L,_ in rows]))
 print('4 only by maybe nonzero',statistics.median([100*p/L for n,p,L,_ in rows if n==4]))
 print('4 only count',sum(n==4 for n,_,_,_ in rows))
 print('all sorted around',sorted((100*p/L,nm) for n,p,L,nm in rows)[125-2:125+3])
PY"
