/bin/bash -lc "python - <<'PY'
import zipfile, collections, fractions, statistics
p='inputs/scogs_fungi.zip'
expected={'Fungi_Ncra','Fungi_Scer','Fungi_Spom','Fungi_Umay'}
rows=[]
with zipfile.ZipFile(p) as z:
  names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
  assert len(names)==255
  for name in names:
    rec=[]; header=None; seq=[]
    for line in z.read(name).decode().splitlines():
      if line.startswith('>'):
        if header is not None: rec.append((header,''.join(seq)))
        header=line[1:].split()[0]; seq=[]
      elif line.strip(): seq.append(line.strip())
    if header is not None: rec.append((header,''.join(seq)))
    ids=[h for h,s in rec]; seqs=[s for h,s in rec]
    assert set(ids) <= expected and len(ids)==len(set(ids))
    assert len(set(map(len,seqs)))==1
    pis=0
    for col in zip(*seqs):
      cnt=collections.Counter(c for c in col if c "'!= '"'-')
      if sum(v>=2 for v in cnt.values())>=2: pis += 1
    rows.append((100*pis/len(seqs[0]), name, pis, len(seqs[0]), len(seqs)))
rows.sort()
med=rows[len(rows)//2]
print('alignments',len(rows))
print('taxa counts',collections.Counter(r[4] for r in rows))
print('median index 1-based',len(rows)//2+1)
print('median row',med)
print('median rational percent',fractions.Fraction(med[2]*100,med[3]))
print('median decimal to 20 places',format(med[2]*100/med[3],'.20f'))
print('neighbors')
for x in rows[len(rows)//2-3:len(rows)//2+4]: print(x)
PY"
