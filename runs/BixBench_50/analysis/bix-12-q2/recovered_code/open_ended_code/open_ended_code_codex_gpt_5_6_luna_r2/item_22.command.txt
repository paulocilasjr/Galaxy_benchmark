/bin/bash -lc "python - <<'PY'
import zipfile

def parse(b):
    rec=[]; h=None; a=[]
    for l in b.decode().splitlines():
        if l.startswith('>'):
            if h is not None: rec.append((h,''.join(a)))
            h=l[1:].split()[0]; a=[]
        elif l.strip(): a.append(l.strip())
    if h is not None: rec.append((h,''.join(a)))
    return rec
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    for n in z.namelist():
        if n.endswith('.mafft'):
            rec=parse(z.read(n))
            if len(rec)<4:
                print(n, [x[0] for x in rec], [len(x[1]) for x in rec], [x[1].count('-') for x in rec])
                if sum(1 for _ in [n])>=10: pass
PY"
