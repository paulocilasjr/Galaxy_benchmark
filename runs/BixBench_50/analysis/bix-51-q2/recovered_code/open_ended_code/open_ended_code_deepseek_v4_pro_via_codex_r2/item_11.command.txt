/bin/bash -c "python - <<'PY'
import openpyxl
wb=openpyxl.load_workbook('/workspace/inputs/data.xlsx', read_only=True, data_only=True)
ws=wb['Sheet1']
rows=list(ws.iter_rows(values_only=True))
y=[];x=[];skipped=[]
for j,r in enumerate(rows[1:],2):
    grp=r[1]; eff=r[2]; bmi=r[10]
    if eff in ('PR','SD','PD'):
        if bmi is not None:
            y.append(1 if eff=='PR' else 0); x.append(float(bmi))
        else:
            skipped.append((j,'bmiNone',grp,eff,bmi))
    else:
        if grp is not None or bmi is not None:
            skipped.append((j,'noeff',grp,eff,bmi))
print(len(y),sum(y),len(x))
print('skipped',skipped)
print('last rows values', [(j,r[1],r[2],r[10]) for j,r in enumerate(rows[1:],2) if j>=79])
PY"
