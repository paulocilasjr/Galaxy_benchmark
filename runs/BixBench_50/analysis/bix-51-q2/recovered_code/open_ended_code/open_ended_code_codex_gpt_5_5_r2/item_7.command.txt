/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
wb=load_workbook('/workspace/inputs/data.xlsx', data_only=True, read_only=True)
ws=wb['Sheet1']
headers=[c for c in next(ws.iter_rows(values_only=True))]
cols={h:i for i,h in enumerate(headers) if h is not None}
rows=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if all(v is None for v in r):
        continue
    rows.append(r)
print('n rows', len(rows))
print('cols', cols)
print('Efficacy counts', Counter(r[cols['Efficacy']] for r in rows))
print('BMI missing', sum(r[cols['BMI']] is None for r in rows))
print('Efficacy missing', sum(r[cols['Efficacy']] is None for r in rows))
print('BMI min max', min(r[cols['BMI']] for r in rows if r[cols['BMI']] is not None), max(r[cols['BMI']] for r in rows if r[cols['BMI']] is not None))
# list by efficacy BMI summary
for eff in sorted(Counter(r[cols['Efficacy']] for r in rows), key=str):
    vals=[float(r[cols['BMI']]) for r in rows if r[cols['Efficacy']]==eff and r[cols['BMI']] is not None]
    print(eff, len(vals), sum(vals)/len(vals), min(vals), max(vals))
PY"
