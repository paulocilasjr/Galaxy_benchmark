wb = openpyxl.load_workbook('inputs/data.xlsx')
ws = wb['Sheet1']

# Get groups and BMI
from collections import Counter
groups_bmi = Counter()
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    g = row[1]
    bmi = row[10]
    eff = row[2]
    has_bmi = bmi is not None
    if g == 1:
        groups_bmi['group1_bmi'] += 1 if has_bmi else 0
        groups_bmi['group1_total'] += 1
    elif g == 2:
        groups_bmi['group2_bmi'] += 1 if has_bmi else 0
        groups_bmi['group2_total'] += 1

print('Groups BMI availability:', dict(groups_bmi))

# Also check which group has which efficacy
for g in [1, 2]:
    eff_count = Counter()
    for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
        if row[1] == g:
            eff_count[row[2]] += 1
    print(f'Group {g} efficacy:', dict(eff_count))
