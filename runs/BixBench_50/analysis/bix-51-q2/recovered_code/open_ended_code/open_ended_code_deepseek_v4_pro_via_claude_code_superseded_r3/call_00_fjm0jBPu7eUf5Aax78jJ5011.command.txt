import numpy as np
from scipy.special import expit  # sigmoid
from scipy.optimize import minimize

# Read data manually
import openpyxl
wb = openpyxl.load_workbook('inputs/data.xlsx')
ws = wb['Sheet1']

bmi_list = []
pr_list = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    bmi = row[10]
    eff = row[2]
    if bmi is not None:
        bmi_list.append(float(bmi))
        pr_list.append(1.0 if eff == 'PR' else 0.0)

X = np.array(bmi_list)
y = np.array(pr_list)
n = len(y)

# Add intercept
X_design = np.column_stack([np.ones(n), X])

# Negative log-likelihood for logistic regression
def neg_ll(beta):
    eta = X_design @ beta
    p = expit(eta)
    # Avoid log(0)
    p = np.clip(p, 1e-15, 1 - 1e-15)
    ll = np.sum(y * np.log(p) + (1 - y) * np.log(1 - p))
    return -ll

# Initial guess
beta0 = np.zeros(2)
result = minimize(neg_ll, beta0, method='BFGS')
beta_hat = result.x
ll_max = -result.fun  # log-likelihood at MLE
k = 2  # number of parameters
aic = 2 * k - 2 * ll_max

print(f'Beta: {beta_hat}')
print(f'Log-likelihood: {ll_max}')
print(f'AIC: {aic}')
print(f'k: {k}')
print(f'n: {n}')
