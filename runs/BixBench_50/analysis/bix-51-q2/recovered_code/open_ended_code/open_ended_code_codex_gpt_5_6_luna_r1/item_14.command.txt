/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter,defaultdict
wb=load_workbook('/workspace/inputs/data.xlsx',data_only=True,read_only=True)
ws=wb.active
rows=list(ws.iter_rows(min_row=2,values_only=True))
print('groups',Counter(r[1] for r in rows))
print('group x efficacy')
ct=defaultdict(Counter)
for r in rows: ct[r[1]][r[2]]+=1
for g,c in ct.items(): print(g,c)
print('sequence ranges by group/eff')
for g in sorted(ct):
    print('group',g)
    for e in ['PR','SD','PD',None]:
        seq=[r[0] for r in rows if r[1]==g and r[2]==e]
        if seq: print(repr(e),len(seq),min(seq),max(seq))
print('first and last rows:')
for r in rows[:10]+rows[-10:]: print(r[:12])
PY"
