/bin/bash -lc "python - <<'PY'
import pandas as pd
import statsmodels.api as sm
from scipy.optimize import minimize
from scipy.special import expit
path='/workspace/inputs/data.xlsx'
df=pd.read_excel(path)
print(df.shape)
df=df[['Efficacy','BMI']].dropna().copy()
df['y']=(df['Efficacy']=='PR').astype(int)
X=sm.add_constant(df['BMI'].astype(float),has_constant='add')
fit=sm.Logit(df['y'],X).fit(disp=False)
print('n',len(df),'counts',df.y.value_counts().to_dict())
print('params',fit.params.to_dict())
print('llf',fit.llf,'aic',fit.aic,'df_model',fit.df_model,'df_resid',fit.df_resid)
print(fit.summary())
# GLM binomial
fitg=sm.GLM(df.y,X,family=sm.families.Binomial()).fit()
print('glm llf',fitg.llf,'aic',fitg.aic,'deviance',fitg.deviance,'params',fitg.params.to_dict())
# calculate loglik directly at MLE
p=expit(X.values@fit.params.values)
ll=(df.y.values*__import__('numpy').log(p)+(1-df.y.values)*__import__('numpy').log1p(-p)).sum()
print('direct ll',ll,'direct AIC',-2*ll+2*len(fit.params))
# null
null=sm.Logit(df.y,sm.ones(len(df))).fit(disp=False)
print('null ll',null.llf,'null aic',null.aic)
PY"
