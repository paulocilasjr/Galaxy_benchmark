/bin/bash -c "python - <<'PY'
import math
# Parse and verify AIC manually with a Newton/optimization fallback
from openpyxl import load_workbook
import statsmodels.api as sm, numpy as np, pandas as pd
wb=load_workbook('/workspace/inputs/data.xlsx', data_only=True)
ws=wb['Sheet1']
data=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None: continue
    data.append(r)
df=pd.DataFrame(data, columns=['seq','Groups','Efficacy','Age','Family history','ECOG-PS','Child-Pugh','Chemotherapy experience','HBV infection','Gender','BMI','Toxic'])
y=(df.Efficacy=='PR').astype(float).to_numpy()
x=df.BMI.astype(float).to_numpy()
# simple coordinate optim
b0,b1=-3.631934,0.167746
for _ in range(100):
    eta=b0+b1*x
    p=1/(1+np.exp(-eta))
    res=y-p
    g0=res.sum(); g1=(res*x).sum()
    w=p*(1-p)
    H00=w.sum(); H01=(w*x).sum(); H11=(w*x*x).sum()
    det=H00*H11-H01**2
    db0=(H11*g0-H01*g1)/det
    db1=(-H01*g0+H00*g1)/det
    b0+=db0; b1+=db1
    if max(abs(db0),abs(db1))<1e-12: break
eta=b0+b1*x
p=1/(1+np.exp(-eta))
ll=np.sum(y*np.log(p)+(1-y)*np.log(1-p))
print(b0,b1,ll,-2*ll+2*2)
PY"
