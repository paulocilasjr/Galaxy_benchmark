/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
wb=load_workbook('inputs/data.xlsx', data_only=True, read_only=True)
ws=wb['Sheet1']
headers=[c.value for c in next(ws.iter_rows(min_row=1,max_row=1))]
headers=[h for h in headers if h is not None]
print(headers)
rows=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    vals=list(row[:len(headers)])
    if all(v is None or (isinstance(v,str) and v.strip()=='') for v in vals):
        continue
    if vals[0] is None:
        print('non-empty without sequence', vals)
        continue
    rows.append(dict(zip(headers, vals)))
print('n rows', len(rows))
print('Efficacy counts', Counter(r['Efficacy'] for r in rows))
print('BMI missing', sum(r['BMI'] is None or (isinstance(r['BMI'],str) and r['BMI'].strip()=='') for r in rows))
print('BMI type counts', Counter(type(r['BMI']).__name__ for r in rows))
print('Groups counts', Counter(r['Groups'] for r in rows))
print('First/last sequences', rows[0]['Sequence number'], rows[-1]['Sequence number'])
for r in rows[:5]+rows[-5:]:
    print(r['Sequence number'], r['Groups'], r['Efficacy'], r['BMI'])
PY"
