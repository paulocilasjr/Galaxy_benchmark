/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
wb=load_workbook('inputs/data.xlsx', data_only=True, read_only=True)
ws=wb.active
headers=[c.value for c in next(ws.iter_rows(min_row=1,max_row=1)) if c.value is not None]
y=[]; x=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    vals=row[:len(headers)]
    if all(v is None or (isinstance(v,str) and v.strip()=='') for v in vals):
        continue
    d=dict(zip(headers, vals))
    y.append('1' if d['Efficacy']=='PR' else '0')
    x.append(str(float(d['BMI'])))
print('Rscript -e \"y<-c(%s);x<-c(%s);m<-glm(y~x,family=binomial());cat(AIC(m),\\'\\\\n\\')\"' % (','.join(y), ','.join(x)))
PY"
