/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter, defaultdict
ws=load_workbook('inputs/data.xlsx', read_only=True, data_only=True).active
rows=list(ws.iter_rows(min_row=2, values_only=True))
headers=[c.value for c in ws[1]]
idx={h:i for i,h in enumerate(headers) if h is not None}
valid=[r for r in rows if r[idx['Sequence number']] is not None]
print('nonblank rows',len(valid))
c=defaultdict(Counter)
for r in valid: c[r[idx['Groups']]][r[idx['Efficacy']]] += 1
print('cross efficacy/group')
for k in sorted(c): print(k,dict(c[k]))
print('missing response/BMI among nonblank',sum(r[idx['Efficacy']] is None for r in valid),sum(r[idx['BMI']] is None for r in valid))
print('sequence range',valid[0][idx['Sequence number']],valid[-1][idx['Sequence number']])
for y in ['PR','SD','PD']:
 x=[float(r[idx['BMI']]) for r in valid if r[idx['Efficacy']]==y]
 print(y,len(x),min(x),max(x),sum(x)/len(x))
PY"
