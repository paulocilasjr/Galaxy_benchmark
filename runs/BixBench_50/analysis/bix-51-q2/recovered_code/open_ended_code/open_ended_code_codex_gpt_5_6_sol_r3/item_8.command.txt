/bin/bash -lc "command -v R || true
python - <<'PY'
import math
from openpyxl import load_workbook
ws=load_workbook('inputs/data.xlsx', read_only=True, data_only=True).active
headers=[c.value for c in ws[1]]
i_seq=headers.index('Sequence number'); i_y=headers.index('Efficacy'); i_x=headers.index('BMI')
data=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[i_seq] is None: continue
    data.append((float(r[i_x]), 1.0 if r[i_y]=='PR' else 0.0))

def sigmoid(z):
    return 1/(1+math.exp(-z)) if z>=0 else math.exp(z)/(1+math.exp(z))
def ll(b0,b1):
    # y*z - log(1+exp(z)), stable
    s=0.0
    for x,y in data:
        z=b0+b1*x
        s += y*z - (z+math.log1p(math.exp(-z)) if z>0 else math.log1p(math.exp(z)))
    return s
b0=b1=0.0
for it in range(100):
    g0=g1=0.0; h00=h01=h11=0.0
    for x,y in data:
        p=sigmoid(b0+b1*x); w=p*(1-p)
        g0 += y-p; g1 += (y-p)*x
        h00 += w; h01 += w*x; h11 += w*x*x
    det=h00*h11-h01*h01
    d0=(h11*g0-h01*g1)/det
    d1=(-h01*g0+h00*g1)/det
    b0 += d0; b1 += d1
    if max(abs(d0),abs(d1))<1e-14: break
loglik=ll(b0,b1); aic=-2*loglik+2*2
print('n',len(data),'iterations',it+1)
print('beta0',repr(b0),'beta1',repr(b1))
print('logLik',repr(loglik),'deviance',repr(-2*loglik),'AIC',repr(aic))
print('gradient',sum(y-sigmoid(b0+b1*x) for x,y in data),sum((y-sigmoid(b0+b1*x))*x for x,y in data))
# Null fit sanity check
p=sum(y for _,y in data)/len(data)
ll0=sum(y*math.log(p)+(1-y)*math.log1p(-p) for _,y in data)
print('null ll,aic',ll0,-2*ll0+2)
PY"
