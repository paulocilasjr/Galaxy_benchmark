/bin/bash -lc "python - <<'PY'
import pandas as pd
p='inputs/data.xlsx'
df=pd.read_excel(p)
print(df.shape)
print(df.columns.tolist())
for c in ['Sequence number','Groups','Efficacy','BMI']:
    print('\\n',c)
    print(df[c].value_counts(dropna=False).sort_index())
print('\\nCross-tab group x efficacy')
print(pd.crosstab(df['Groups'],df['Efficacy'],dropna=False))
print('\\nBMI summaries by efficacy')
print(df.groupby('Efficacy')['BMI'].agg(['count','mean','std','min','max']))
PY"
