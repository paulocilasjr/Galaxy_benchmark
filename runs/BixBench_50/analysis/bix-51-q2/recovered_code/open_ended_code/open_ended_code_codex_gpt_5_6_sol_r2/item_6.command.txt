from collections import Counter, defaultdict
from statistics import mean, stdev
import openpyxl
ws=openpyxl.load_workbook('inputs/data.xlsx',read_only=True,data_only=True)['Sheet1']
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0])
idx={h:i for i,h in enumerate(headers) if h is not None}
data=rows[1:]
print('records',len(data))
for c in ['Sequence number','Groups','Efficacy','BMI']:
    vals=[r[idx[c]] for r in data]
    print(c, Counter(vals))
print('group x efficacy', Counter((r[idx['Groups']],r[idx['Efficacy']]) for r in data))
by=defaultdict(list)
for r in data:
    if r[idx['Efficacy']] is not None and r[idx['BMI']] is not None:
        by[r[idx['Efficacy']]].append(float(r[idx['BMI']]))
for k,v in by.items():
    print(k,len(v),mean(v),stdev(v),min(v),max(v))
