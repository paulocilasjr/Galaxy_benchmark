from openpyxl import load_workbook
from collections import Counter, defaultdict
p='/workspace/inputs/data.xlsx'
ws=load_workbook(p, data_only=True, read_only=True).active
rows=list(ws.iter_rows(values_only=True))
h=list(rows[0][:12]); d=[dict(zip(h,r[:12])) for r in rows[1:]]
print('n',len(d))
for col in h:
    vals=[x[col] for x in d]
    c=Counter(vals)
    print('\\n',col,'missing',c.get(None,0),'unique',len(c))
    if len(c)<=20: print(c)
    elif col in ('BMI','Age'): print('range',min(v for v in vals if v is not None),max(v for v in vals if v is not None))
print('\\nGroups x Efficacy')
tab=defaultdict(Counter)
for x in d: tab[x['Groups']][x['Efficacy']]+=1
for k,v in tab.items(): print(k,v)
print('\\nSequence duplicate?',len({x['Sequence number'] for x in d}), sorted(x['Sequence number'] for x in d)[:5], sorted(x['Sequence number'] for x in d)[-5:])
PY"
