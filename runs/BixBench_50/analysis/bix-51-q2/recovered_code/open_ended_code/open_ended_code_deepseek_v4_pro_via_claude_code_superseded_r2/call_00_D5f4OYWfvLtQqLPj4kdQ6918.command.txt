import numpy as np
from scipy import stats
import statsmodels.api as sm

# Data
eff_map = {'PR': 1, 'SD': 0, 'PD': 0}

import openpyxl
wb = openpyxl.load_workbook('inputs/data.xlsx')
ws = wb.active

y = []
bmi = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    if row[2] is not None and row[10] is not None:
        y.append(eff_map[row[2]])
        bmi.append(row[10])

y = np.array(y)
X = sm.add_constant(np.array(bmi))

# Fit logistic regression
model = sm.Logit(y, X)
result = model.fit(disp=0)
print(result.summary())
print()
print(f'AIC: {result.aic}')
