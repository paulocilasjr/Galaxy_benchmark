/bin/bash -c "python - <<'PY'
import pandas as pd
res=pd.read_csv('/tmp/pydeseq2_res.csv', index_col=0)
m=pd.read_csv('inputs/ensg_to_gene_name.tsv', sep='\\t').set_index('ENSG')['gene_name']
mask=(res['padj']<0.05)&(res['log2FoldChange'].abs()>=0.5)
degs=res.loc[mask].copy()
print('DEG count', len(degs))
degs['symbol']=degs.index.map(m)
print('unmapped', degs['symbol'].isna().sum())
print(degs[['log2FoldChange','padj','symbol']].head(20))
mapped=degs.dropna(subset=['symbol'])
sym=mapped['symbol'].drop_duplicates().sort_values()
print('unique symbols', len(sym), 'duplicate symbols', mapped['symbol'].duplicated().sum())
print(sym.head(20), sym.tail(20))
print('contains TP53?', 'TP53' in set(sym), 'contains CDK?', [s for s in sym if 'CDK' in s][:20])
sym.to_csv('/tmp/deg_symbols.txt', index=False, header=False)
PY"
