import os
import pandas as pd
import gseapy as gp
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
counts.index.name = 'ENSG'
layout = pd.read_csv('inputs/sample_layout.csv')
layout['count_id'] = layout['SampleID'].str.replace('-', '_', regex=False)
control = 'DMSO'
treat = 'Cisplatin_IC50_CBD_IC50'
sel = layout[layout['Group'].isin([control, treat])].copy().set_index('count_id')

metadata['condition'] = pd.Categorical(metadata['condition'], categories=[control, treat])

    metadata=metadata,
    design='~condition',
    quiet=True,
    n_cpus=1,
)

deg = deg.merge(map_df, on='ENSG', how='left')
deg_symbols = sorted({x.strip() for x in deg['gene_name'].dropna().astype(str) if x.strip()})
print('DEG_ENSG', len(deg), 'DEG_SYMBOLS', len(deg_symbols))

    gene_list=deg_symbols,
    gene_sets='Reactome_2022',
    organism='human',

    cutoff=1.0,

    verbose=False,
)
