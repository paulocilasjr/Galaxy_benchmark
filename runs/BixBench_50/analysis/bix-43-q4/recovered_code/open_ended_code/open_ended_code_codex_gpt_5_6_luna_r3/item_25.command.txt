/bin/bash -lc "python3 - <<'PY'
# run pydeseq2 result and target intersection, no output files
import pandas as pd, gseapy
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
raw=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0)
cols=['3_1','3_2','3_3','9_1','9_2','9_3']
counts=raw[cols].astype(int).T
md=pd.DataFrame({'condition':['DMSO']*3+['CBD_cisplatin']*3},index=cols)
dds=DeseqDataSet(counts=counts,metadata=md,design_factors='condition',refit_cooks=True,n_cpus=1)
dds.deseq2()
st=DeseqStats(dds,contrast=['condition','CBD_cisplatin','DMSO'],n_cpus=1)
st.run_wald_test()
res=st.results_df
deg_ids=set(res.index[(res.padj<0.05)&(res.log2FoldChange.abs()>=0.5)])
mapdf=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t')
# one ensembl each, symbols can duplicate; keep mapped symbols
id_to_sym=mapdf.set_index('ENSG').gene_name
syms=set(id_to_sym.reindex(list(deg_ids)).dropna())
gs=gseapy.get_library(name='Reactome_2022',organism='Human')
name=[k for k in gs if k.startswith('TP53 Regulates Transcription Of Cell Cycle Genes')][0]
term=set(gs[name])
print('DEG IDs',len(deg_ids),'mapped symbols incl dup',len(id_to_sym.reindex(list(deg_ids)).dropna()),'unique syms',len(syms))
print(name,'term total',len(term),'overlap',len(syms&term), sorted(syms&term))
print('overlap fraction',f'{len(syms&term)}/{len(term)}')
PY"
