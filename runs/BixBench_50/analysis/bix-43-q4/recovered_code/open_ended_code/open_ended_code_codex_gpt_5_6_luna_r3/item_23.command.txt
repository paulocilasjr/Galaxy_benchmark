/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
cols=['3_1','3_2','3_3','9_1','9_2','9_3']
counts=raw[cols].astype(int)
metadata=pd.DataFrame({'condition':['DMSO']*3+['CBD_cisplatin']*3},index=cols)
print('counts',counts.shape, 'allzero', (counts.sum(axis=1)==0).sum())
print('metadata',metadata)
dds=DeseqDataSet(counts=counts, metadata=metadata, design_factors='condition', refit_cooks=True, n_cpus=1)
dds.deseq2()
stat=DeseqStats(dds, contrast=['condition','CBD_cisplatin','DMSO'], n_cpus=1)
stat.summary()
res=stat.results_df
print(res.head())
print(res.columns)
print('padj',res.padj.notna().sum())
deg=res[(res.padj<0.05)&(res.log2FoldChange.abs()>=0.5)]
print('DEG',len(deg))
print(deg.sort_values('padj').head(20).to_string())
PY"
