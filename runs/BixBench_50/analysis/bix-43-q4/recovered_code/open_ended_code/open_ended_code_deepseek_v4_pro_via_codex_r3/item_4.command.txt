/bin/bash -lc "python - <<'PY'
import csv
p='/workspace/inputs/counts_raw_unfiltered.csv'
with open(p) as f:
    r=csv.reader(f)
    header=next(r)
    print('header len',len(header),header[:5],header[-5:])
    n=0; rows=[]
    for row in r:
        n+=1
        rows.append(row)
    print('rows',n)
    # genes zero counts
    import collections
    nz=0; allzero=0; nonnumeric=0
    lengths=collections.Counter()
    for row in rows:
        lengths[len(row)]+=1
        vals=[]
        for x in row[1:]:
            try: vals.append(int(x))
            except: nonnumeric+=1
        if any(v"'!=0 for v in vals): nz+=1
        else: allzero+=1
    print('"'nonzero genes',nz,'allzero',allzero,'nonnum fields',nonnumeric,'lengths',lengths)
PY"
