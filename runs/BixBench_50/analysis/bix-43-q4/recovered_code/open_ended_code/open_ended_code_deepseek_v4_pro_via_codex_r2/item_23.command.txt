/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time, json
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
layout=pd.read_csv('inputs/sample_layout.csv')
counts=pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
layout['SampleID_us']=layout['SampleID'].str.replace('-','_', regex=False)
sub=layout[layout['Group'].isin(['DMSO','Cisplatin_IC50_CBD_IC50'])].copy().set_index('SampleID_us')
sub['condition']=pd.Categorical(sub['Group'], categories=['DMSO','Cisplatin_IC50_CBD_IC50'])
counts_sub=counts[sub.index].astype(int).T
dds=DeseqDataSet(counts=counts_sub, metadata=sub[['condition']], design='~condition', quiet=True)
t=time.time(); dds.deseq2(); print('fit',time.time()-t)
print('obsm keys', list(dds.obsm.keys()))
print('varm keys', list(dds.varm.keys()))
print(dds.obsm)
# try contrast
stat=DeseqStats(dds, contrast=['condition','Cisplatin_IC50_CBD_IC50','DMSO'], alpha=0.05, quiet=False)
t=time.time(); stat.summary(); print('stat', time.time()-t)
res=stat.results_df.copy()
print(res.shape)
print(res.head())
res.to_csv('/tmp/pydeseq2_results.csv')
print('saved /tmp/pydeseq2_results.csv')
PY"
