/bin/bash -lc "python - <<'PY'
import pandas as pd, gseapy
res=pd.read_csv('/tmp/pydeseq2_results.csv', index_col=0)
sig=res[(res.padj<.05)&(res.log2FoldChange.abs()>=.5)]
mapdf=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t',header=0)
syms=mapdf[mapdf.ENSG.isin(sig.index)].gene_name.dropna().astype(str).unique().tolist()
# pass ensembl ids plus symbols? test all sig ids directly
ens=sig.index.tolist()
for gl,name in [(syms,'syms'),(ens,'ens')]:
 try:
  e=gseapy.enrichr(gene_list=gl,gene_sets='Reactome_2022',organism='human',outdir=None).results
  row=e[e.Term.str.contains('TP53 Regulates Transcription Of Cell Cycle Genes',case=False,na=False)]
  print(name, len(gl), row[['Term','Overlap','Adjusted P-value']].to_string(index=False))
 except Exception as ex: print(name,'error',ex)
PY"
