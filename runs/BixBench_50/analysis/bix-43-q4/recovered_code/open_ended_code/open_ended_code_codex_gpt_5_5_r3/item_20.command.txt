/bin/bash -lc "python - <<'PY'
from pathlib import Path
import pandas as pd
import numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
layout = pd.read_csv('inputs/sample_layout.csv')
layout['count_id'] = layout['SampleID'].str.replace('-', '_', regex=False)
contrast_groups = ['DMSO', 'Cisplatin_IC50_CBD_IC50']
meta = layout[layout['Group'].isin(contrast_groups)].copy()
print(meta[['SampleID','count_id','Group']].to_string(index=False))
print('counts shape raw', counts.shape)
print('missing cols', set(meta['count_id']) - set(counts.columns))
# samples x genes
cts = counts.loc[:, meta['count_id']].T
cts.index = meta['count_id'].values
metadata = meta.set_index('count_id')[['Group']]
metadata['Group'] = pd.Categorical(metadata['Group'], categories=['DMSO','Cisplatin_IC50_CBD_IC50'])
# PyDESeq2 cannot handle all-zero genes; DESeq2 likewise has undefined size-independent info for them.
nonzero = cts.sum(axis=0) > 0
cts = cts.loc[:, nonzero]
print('counts shape contrast nonzero', cts.shape)

dds = DeseqDataSet(counts=cts.astype(int), metadata=metadata, design='~ Group', ref_level=['Group','DMSO'], quiet=False, n_cpus=1)
dds.deseq2()
stat_res = DeseqStats(dds, contrast=['Group','Cisplatin_IC50_CBD_IC50','DMSO'], alpha=0.05, quiet=False, n_cpus=1)
stat_res.summary()
res = stat_res.results_df.copy()
print(res.head().to_string())
print(res.columns)
print(res[['log2FoldChange','padj']].describe().to_string())
# Save temporary? allowed? no evidence/audit? Could use /tmp. avoid workspace extra except answer. Print counts.
degs = res[(res['padj'].notna()) & (res['padj'] < 0.05) & (res['log2FoldChange'].abs() >= 0.5)].copy()
print('DEGs ensg', len(degs))
mapdf = pd.read_csv('inputs/ensg_to_gene_name.tsv', sep='\\t')
merged = degs.reset_index(names='ENSG').merge(mapdf, on='ENSG', how='left')
print('mapped rows', merged['gene_name'].notna().sum(), 'unmapped', merged['gene_name'].isna().sum(), 'unique symbols', merged['gene_name'].dropna().nunique())
print('first symbols', sorted(merged['gene_name'].dropna().unique())[:20])
PY"
