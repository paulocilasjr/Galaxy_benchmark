# Count Ts/Tv matching bcftools convention: first ALT allele only, SNVs only
vcf_path = "/workspace/inputs/SRR35233585_raw_variants.vcf"
f = open(vcf_path, 'r')

transitions = {('A','G'), ('G','A'), ('C','T'), ('T','C')}
ts = 0
tv = 0

for line in f:
    if line.startswith('#'):
        continue
    fields = line.strip().split('\t')
    if len(fields) < 5:
        continue
    ref = fields[3]
    alt_field = fields[4]
    if len(ref) != 1:
        continue
    first_alt = alt_field.split(',')[0]
    if len(first_alt) != 1:
        continue
    if (ref, first_alt) in transitions:
        ts += 1
    else:
        tv += 1

f.close()
print(f"First-ALT only: Ts={ts}, Tv={tv}, Ts/Tv={ts/tv:.4f} -> {round(ts/tv, 2)}")
