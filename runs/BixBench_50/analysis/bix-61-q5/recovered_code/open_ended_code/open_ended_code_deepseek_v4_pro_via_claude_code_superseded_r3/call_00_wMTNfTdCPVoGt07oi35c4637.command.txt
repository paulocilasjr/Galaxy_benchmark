# Check for edge cases that bcftools might handle differently
vcf_path = "/workspace/inputs/SRR35233585_raw_variants.vcf"
f = open(vcf_path, 'r')

non_standard = []
for line in f:
    if line.startswith('#'):
        continue
    fields = line.strip().split('\t')
    if len(fields) < 5:
        continue
    ref = fields[3]
    alt_field = fields[4]
    alts = alt_field.split(',')
    
    # Check for non-standard alleles
    for a in alts:
        if a == '*' or a.startswith('<') or a == '.':
            non_standard.append(f"{fields[0]}:{fields[1]} REF={ref} ALT={a}")
            break
    # Check for empty or very long REF
    if len(ref) == 0 or len(ref) > 100:
        non_standard.append(f"{fields[0]}:{fields[1]} REF len={len(ref)}")

f.close()
print(f"Non-standard alleles found: {len(non_standard)}")
for ns in non_standard[:20]:
    print(f"  {ns}")

# Also check how many variants have '.' as ALT
f2 = open(vcf_path, 'r')
dot_alt = 0
for line in f2:
    if line.startswith('#'):
        continue
    if '\t.\t' in line.split('\t')[4] or line.split('\t')[4] == '.':
        dot_alt += 1
f2.close()
print(f"\nVariants with '.' as ALT: {dot_alt}")
