/bin/bash -lc "python3 - <<'PY'
from collections import Counter
path='inputs/SRR35233585_raw_variants.vcf'
classes=Counter(); examples={}
with open(path) as f:
    for line in f:
        if line.startswith('#'): continue
        c=line.rstrip('\\n').split('\\t')
        ref=c[3]; alts=c[4].split(',')
        key=(len(ref), tuple(sorted(Counter(len(a) for a in alts).items())), len(alts))
        classes[key]+=1
        examples.setdefault(key, line.rstrip('\\n'))
for key,n in classes.most_common(20):
    print(n, key, examples[key][:300])
PY"
