/bin/bash -lc "python3 - <<'PY'
from collections import Counter
path='inputs/SRR35233585_raw_variants.vcf'
trans={('A','G'),('G','A'),('C','T'),('T','C')}

def gt_alt_indices(sample, fmt):
    keys=fmt.split(':')
    vals=sample.split(':')
    d=dict(zip(keys, vals))
    gt=d.get('GT','')
    if gt in ('', '.', './.', '.|.'):
        return []
    parts=gt.replace('|','/').split('/')
    out=[]
    for p in parts:
        if p"'!='"'.' and p"'!='"'0':
            try: out.append(int(p))
            except ValueError: pass
    return out

counts={scope:Counter() for scope in ['all_site_alt','all_genotype_alt','nonlow_site_alt','nonlow_genotype_alt']}
filters=Counter(); gt_counts=Counter(); recs=0; snv_records=0; multi_snv_records=0
with open(path) as f:
    for line in f:
        if line.startswith('#'): continue
        recs+=1
        cols=line.rstrip('\\n').split('\\t')
        chrom,pos,vid,ref,alts,qual,flt,info,fmt,sample=cols[:10]
        filters[flt]+=1
        gt=sample.split(':',1)[0]
        gt_counts[gt]+=1
        alt_list=alts.split(',')
        snv_alt_indices=[i for i,a in enumerate(alt_list, start=1) if len(ref)==1 and len(a)==1 and ref in 'ACGT' and a in 'ACGT']
        if snv_alt_indices:
            snv_records+=1
            if len(snv_alt_indices)>1: multi_snv_records+=1
        for i,a in enumerate(alt_list, start=1):
            if len(ref)==1 and len(a)==1 and ref in 'ACGT' and a in 'ACGT':
                kind='ts' if (ref,a) in trans else 'tv'
                counts['all_site_alt'][kind]+=1
                if flt in ('.','PASS'):
                    counts['nonlow_site_alt'][kind]+=1
        present=set(gt_alt_indices(sample, fmt))
        for i in present:
            if 1 <= i <= len(alt_list):
                a=alt_list[i-1]
                if len(ref)==1 and len(a)==1 and ref in 'ACGT' and a in 'ACGT':
                    kind='ts' if (ref,a) in trans else 'tv'
                    counts['all_genotype_alt'][kind]+=1
                    if flt in ('.','PASS'):
                        counts['nonlow_genotype_alt'][kind]+=1
print('records', recs)
print('filters', dict(filters))
print('genotypes_top', gt_counts.most_common(10))
print('snv_records_with_snv_alt', snv_records, 'multi_snv_records', multi_snv_records)
for scope,c in counts.items():
    ts,tv=c['ts'],c['tv']
    print(scope, 'ts',ts,'tv',tv,'ratio', (ts/tv if tv else None), 'rounded', (format(ts/tv,'.2f') if tv else 'NA'))
PY"
