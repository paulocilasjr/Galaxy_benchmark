/bin/bash -lc "python3 - <<'PY'
from collections import Counter
p='inputs/SRR35233585_raw_variants.vcf'
base=set('ACGT')
trans={frozenset(('A','G')), frozenset(('C','T'))}
for mode in ('all_alt_once','sample_alt_once','sample_site_once'):
    c=Counter(); records=0; included_records=0; included_alts=0; gt_counts=Counter(); skipped=Counter(); multiallelic=0
    with open(p) as fh:
        for line in fh:
            if line.startswith('#'): continue
            f=line.rstrip('\\n').split('\\t'); records+=1
            ref=f[3].upper(); alts=f[4].upper().split(',')
            gt=f[9].split(':',1)[0] if len(f)>9 else './.'
            gt_counts[gt]+=1
            if mode=='all_alt_once':
                idxs=range(len(alts))
            else:
                toks=gt.replace('|','/').split('/')
                idxs=sorted({int(x) for x in toks if x not in ('.','0','')})
                if not idxs: continue
            if len(alts)>1: multiallelic+=1
            rec_has=False
            for i in idxs:
                if i<1 or i>len(alts):
                    skipped['bad_gt_index']+=1; continue
                alt=alts[i-1]
                if len(ref)==1 and len(alt)==1 and ref in base and alt in base and ref"'!=alt:
                    rec_has=True; included_alts+=1
                    c['"'Ts' if frozenset((ref,alt)) in trans else 'Tv']+=1
                else: skipped['non_snv']+=1
            if rec_has: included_records+=1
    ratio=c['Ts']/c['Tv']
    print(mode, 'records',records, 'included_records',included_records, 'included_alt_alleles',included_alts, 'Ts',c['Ts'],'Tv',c['Tv'], 'ratio',f'{ratio:.12f}', 'multi',multiallelic, 'skipped',dict(skipped), 'GTs',gt_counts)
PY"
