python3 << 'PYEOF'
# Final Ts/Tv computation for SRR35233585 raw variants
# Standard approach: SNP sites only, first ALT allele per site (site-level)
# Transitions: A<->G, C<->T
# Transversions: all other single-base changes (A<->C, A<->T, G<->C, G<->T)

vcf_path = '/workspace/inputs/SRR35233585_raw_variants.vcf'

transitions_set = {('A','G'), ('G','A'), ('C','T'), ('T','C')}

ts = 0  # transitions
tv = 0  # transversions

with open(vcf_path) as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        ref = parts[3]
        alt = parts[4]
        
        # Use first ALT allele only (standard site-level counting)
        first_alt = alt.split(',')[0]
        
        # Count only single-nucleotide substitutions
        if len(ref) == 1 and len(first_alt) == 1:
            if ref in 'ACGT' and first_alt in 'ACGT':
                if (ref, first_alt) in transitions_set:
                    ts += 1
                else:
                    tv += 1

ratio = ts / tv
print(f"Transitions: {ts}")
print(f"Transversions: {tv}")
print(f"Ts/Tv ratio: {ratio}")
print(f"Ts/Tv rounded to 2 decimal places: {ratio:.2f}")

PYEOF
