/bin/bash -lc "bcftools stats -s SRR35233585 inputs/SRR35233585_raw_variants.vcf | rg '"'^(SN|TSTV|PSC)'"'
bcftools query -f '%FILTER\\t%TYPE\\t%REF\\t%ALT[\\t%GT\\t%DP\\t%AD]\\n' inputs/SRR35233585_raw_variants.vcf | awk 'BEGIN{FS=\"\\t\"} {f["'$1]++; t[$2]++; gt[$5]++} END {print "FILTER"; for(k in f) print k,f[k]; print "TYPE"; for(k in t) print k,t[k]; print "GT"; for(k in gt) print k,gt[k]}'"' | sort -k1,1 -k2,2
bcftools norm -m -any -f inputs/GCF_000005845.2_ASM584v2_genomic.fna inputs/SRR35233585_raw_variants.vcf 2>/dev/null | bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT[\\t%GT]\\n' | awk 'BEGIN{FS=\"\\t\"; ts=tv=0} length("'$3)==1&&length($4)==1&&$5!~/''^(0['"\\/|]0|\\.\\/\\.|\\.)"'$/ {x=$3$4; if(x~/''^(AG|GA|CT|TC)$/) ts++; else tv++} END{printf "sample_nonref_biallelic_snvs ts=%d tv=%d ratio=%.12f'"\\n\",ts,tv,ts/tv}'"
