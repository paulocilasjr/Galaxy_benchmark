/bin/bash -c "which bcftools || true; which samtools || true; which vcftools || true; python3 - <<'PY'
import gzip
from pathlib import Path
p=Path('inputs/SRR35233585_raw_variants.vcf')
# count categories ignoring headers, classify records with single-char ref/alt
n=0; ts=0; tv=0; indel=0; multi=0; other=0
for line in open(p):
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    ref=f[3]; alts=f[4].split(',')
    # include only biallelic single base? count site-wise each ALT separately?
    if len(ref)==1 and all(len(a)==1 for a in alts):
        n+=len(alts)
        for a in alts:
            pair={ref,a}
            if pair in ({'A','G'},{'C','T'}): ts+=1
            else: tv+=1
    elif len(alts)==1 and (len(ref)"'!=1 or len(alts[0])!=1):
        indel+=1
    else:
        multi+=1
print('"'biallelic/site all alts n',n,'ts',ts,'tv',tv,'ratio',ts/tv if tv else None)
# Count records biallelic only and also alt-specific
nrec=0; ts1=0; tv1=0; nonbiallelic=0; multiallelic_records=0; indels=0; complex=0
for line in open(p):
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    ref=f[3]; alts=f[4].split(',')
    if len(alts)==1 and len(ref)==1 and len(alts[0])==1:
        nrec+=1
        pair={ref,alts[0]}
        if pair in ({'A','G'},{'C','T'}): ts1+=1
        else: tv1+=1
    else:
        nonbiallelic+=1
        if len(alts)>1: multiallelic_records+=1
        if len(alts)==1 and (len(ref)"'!=1 or len(alts[0])!=1): indels+=1
        if len(alts)==1 and len(ref)!=1 and len(alts[0])!=1: complex+=1
print('"'biallelic SNV records',nrec,'ts',ts1,'tv',tv1,'ratio',ts1/tv1 if tv1 else None,'nonbiallelic',nonbiallelic,'multiallelic_records',multiallelic_records,'indels',indels)
PY"
