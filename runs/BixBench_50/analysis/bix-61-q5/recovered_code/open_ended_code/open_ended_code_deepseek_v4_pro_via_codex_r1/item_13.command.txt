/bin/bash -c "python3 - <<'PY'
from pathlib import Path
rows_snp=0; rows_mixed=0; total_alts=0; total_alts_snp=0; total_records=0
for line in Path('inputs/SRR35233585_raw_variants.vcf').open():
    if line.startswith('#'): continue
    total_records+=1
    f=line.rstrip('\\n').split('\\t'); ref=f[3]; alts=f[4].split(',')
    snp=0; other=0
    for a in alts:
        if len(ref)==1 and len(a)==1 and a in 'ACGT': snp+=1
        else: other+=1
    total_alts += len(alts)
    total_alts_snp += snp
    if snp and not other: rows_snp+=1
    if snp and other: rows_mixed+=1
print('records', total_records, 'rows_snp_only', rows_snp, 'rows_mixed', rows_mixed, 'rows with snp', rows_snp+rows_mixed, 'total_alts', total_alts, 'total_snp_alts', total_alts_snp)
PY"
