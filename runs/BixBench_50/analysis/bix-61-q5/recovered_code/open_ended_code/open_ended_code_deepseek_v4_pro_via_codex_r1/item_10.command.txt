/bin/bash -c "python3 - <<'PY'
from pathlib import Path
p=Path('inputs/SRR35233585_raw_variants.vcf')
trans=set([('A','G'),('G','A'),('C','T'),('T','C')])
counts_all={'ts':0,'tv':0}
counts_first={'ts':0,'tv':0}
counts_geno={'ts':0,'tv':0}
n_snp_all=0;n_snp_first=0;n_snp_geno=0
multiallelic=[]
for line in p.open():
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    ref=f[3]; alts=f[4].split(','); gt=f[9].split(':')[0]
    snp_alts=[]; valid=[]
    for a in alts:
        if len(ref)==1 and len(a)==1:
            snp_alts.append(a); valid.append(True)
        else:
            valid.append(False)
    # all SNP alt alleles
    for a in snp_alts:
        n_snp_all+=1
        if (ref,a) in trans: counts_all['ts']+=1
        else: counts_all['tv']+=1
    # first ALT only
    if valid[0]:
        a=alts[0]; n_snp_first+=1
        if (ref,a) in trans: counts_first['ts']+=1
        else: counts_first['tv']+=1
    # genotype allele-specific among called non-ref alt alleles
    # unphased diploid tokens
    for tok in gt.replace('|','/').split('/'):
        if tok=='.' or tok=='0': continue
        idx=int(tok)-1
        if 0 <= idx < len(alts) and valid[idx]:
            a=alts[idx]; n_snp_geno+=1
            if (ref,a) in trans: counts_geno['ts']+=1
            else: counts_geno['tv']+=1
    if len(alts)>1:
        multiallelic.append((f[1], ref, alts, gt))
print('ALL SNP alts', n_snp_all, counts_all, 'ratio', counts_all['ts']/counts_all['tv'])
print('FIRST ALT', n_snp_first, counts_first, 'ratio', counts_first['ts']/counts_first['tv'])
print('GENO ALTS', n_snp_geno, counts_geno, 'ratio', counts_geno['ts']/counts_geno['tv'])
print('multiallelic count', len(multiallelic))
print('multiallelic with >1 SNV alts:')
for m in multiallelic:
    snp=[a for a in m[2] if len(a)==1]
    if len(snp)>1:
        print(m)
PY"
