/bin/bash -c "python3 - <<'PY'
from pathlib import Path
trans=set([('A','G'),('G','A'),('C','T'),('T','C')])
allc=firstc=genoc=0; allt=firstt=genot=0
for line in Path('inputs/SRR35233585_raw_variants.vcf').open():
    if line.startswith('#'): continue
    f=line.rstrip().split('\\t'); ref=f[3]; alts=f[4].split(','); gt=f[9].split(':')[0]
    valid=[]
    for a in alts:
        valid.append(len(ref)==1 and len(a)==1 and a in 'ACGT')
    for i,a in enumerate(alts):
        if valid[i]:
            allc+=1
            allt += 1 if (ref,a) in trans else 0
    if valid[0]:
        firstc+=1
        firstt += 1 if (ref,alts[0]) in trans else 0
    for tok in gt.replace('|','/').split('/'):
        if tok in ('.','0'): continue
        idx=int(tok)-1
        if 0<=idx<len(alts) and valid[idx]:
            genoc+=1
            genot += 1 if (ref,alts[idx]) in trans else 0
print('all', allc, allt, allc-allt, allt/(allc-allt))
print('first', firstc, firstt, firstc-firstt, firstt/(firstc-firstt))
print('geno', genoc, genot, genoc-genot, genot/(genoc-genot))
PY"
