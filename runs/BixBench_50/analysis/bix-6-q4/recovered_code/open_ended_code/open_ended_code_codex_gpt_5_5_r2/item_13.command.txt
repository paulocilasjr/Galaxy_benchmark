/bin/bash -lc "python - <<'PY'
# Alternative formula: Spearman as covariance of average ranks, computed from grouped value ranks.
import openpyxl, math
from collections import defaultdict
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
h=next(ws.iter_rows(values_only=True))
i1=h.index('Chronic Round1 S1'); i2=h.index('Chronic Round1 S2')
pairs=[(float(r[i1]), float(r[i2])) for r in ws.iter_rows(min_row=2, values_only=True)]
def ranks(vals):
    groups=defaultdict(list)
    for i,v in enumerate(vals): groups[v].append(i)
    out=[None]*len(vals); pos=1
    for v in sorted(groups):
        inds=groups[v]
        avg=(pos+pos+len(inds)-1)/2
        for i in inds: out[i]=avg
        pos += len(inds)
    return out
rx=ranks([p[0] for p in pairs]); ry=ranks([p[1] for p in pairs])
n=len(rx); mx=(n+1)/2; my=(n+1)/2
num=sum((a-mx)*(b-my) for a,b in zip(rx,ry))
den=(sum((a-mx)**2 for a in rx)*sum((b-my)**2 for b in ry))**0.5
print(format(num/den,'.17g'))
PY"
