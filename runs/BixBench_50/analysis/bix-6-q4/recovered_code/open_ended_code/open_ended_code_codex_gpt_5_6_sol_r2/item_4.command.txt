/bin/bash -lc "python - <<'PY'
import pandas as pd
from scipy.stats import spearmanr, rankdata
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
df=pd.read_excel(p, sheet_name='MAGeCK P-values', engine='openpyxl')
a='Chronic Round1 S1'; b='Chronic Round1 S2'
x=pd.to_numeric(df[a], errors='coerce'); y=pd.to_numeric(df[b], errors='coerce')
mask=x.notna() & y.notna()
print('shape',df.shape)
print('valid pairs',int(mask.sum()),'missing S1',int(x.isna().sum()),'missing S2',int(y.isna().sum()))
print('ranges',x[mask].min(),x[mask].max(),y[mask].min(),y[mask].max())
print('unique',x[mask].nunique(),y[mask].nunique())
print('duplicate RefSeq IDs',int(df['RefSeq ID'].duplicated().sum()))
r=spearmanr(x[mask],y[mask])
print('scipy',repr(float(r.statistic)),'p',repr(float(r.pvalue)))
print('pandas',repr(float(pd.concat([x[mask],y[mask]],axis=1).corr(method='spearman').iloc[0,1])))
print('rank_pearson',repr(float(pd.Series(rankdata(x[mask],method='average')).corr(pd.Series(rankdata(y[mask],method='average')),method='pearson'))))
print('zeros',int((x[mask]==0).sum()),int((y[mask]==0).sum()),'ones',int((x[mask]==1).sum()),int((y[mask]==1).sum()))
PY"
