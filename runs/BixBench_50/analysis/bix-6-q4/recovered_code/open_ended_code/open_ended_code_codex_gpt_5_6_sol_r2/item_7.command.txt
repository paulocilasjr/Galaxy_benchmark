/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
from math import fsum, sqrt
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
ws=load_workbook(p,read_only=True,data_only=True)['MAGeCK P-values']
it=ws.iter_rows(values_only=True)
head=next(it)
i0=head.index('RefSeq ID'); i1=head.index('Chronic Round1 S1'); i2=head.index('Chronic Round1 S2')
ids=[]; xs=[]; ys=[]; miss1=miss2=incomplete=0
for row in it:
    rid,x,y=row[i0],row[i1],row[i2]
    if not isinstance(x,(int,float)): miss1+=1
    if not isinstance(y,(int,float)): miss2+=1
    if isinstance(x,(int,float)) and isinstance(y,(int,float)):
        ids.append(rid); xs.append(float(x)); ys.append(float(y))
    else: incomplete+=1

def avg_ranks(vals):
    order=sorted(range(len(vals)), key=vals.__getitem__)
    ranks=[0.0]*len(vals)
    k=0
    while k<len(order):
        j=k+1
        v=vals[order[k]]
        while j<len(order) and vals[order[j]]==v:
            j+=1
        rank=(k+1+j)/2.0
        for q in range(k,j): ranks[order[q]]=rank
        k=j
    return ranks
rx=avg_ranks(xs); ry=avg_ranks(ys)
mx=fsum(rx)/len(rx); my=fsum(ry)/len(ry)
cov=fsum((a-mx)*(b-my) for a,b in zip(rx,ry))
vx=fsum((a-mx)**2 for a in rx); vy=fsum((b-my)**2 for b in ry)
rho=cov/sqrt(vx*vy)
print('headers',head)
print('rows',len(xs),'missing_s1',miss1,'missing_s2',miss2,'incomplete',incomplete)
print('ranges',min(xs),max(xs),min(ys),max(ys))
print('unique',len(set(xs)),len(set(ys)))
print('duplicate_ids',len(ids)-len(set(ids)))
print('zeros',xs.count(0.0),ys.count(0.0),'ones',xs.count(1.0),ys.count(1.0))
print('rho repr',repr(rho))
print('rho 15g',format(rho,'.15g'))
# alternative formula using centered rank and ordinary sum as cross-check
sx=sum((a-mx)*(b-my) for a,b in zip(rx,ry))
sxx=sum((a-mx)**2 for a in rx); syy=sum((b-my)**2 for b in ry)
print('rho ordinary',repr(sx/sqrt(sxx*syy)))
# tie group summary
cx=Counter(xs); cy=Counter(ys)
print('tied_value_groups',sum(n>1 for n in cx.values()),sum(n>1 for n in cy.values()),'max_tie',max(cx.values()),max(cy.values()))
PY"
