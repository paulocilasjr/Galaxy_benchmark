import openpyxl, math
from collections import Counter

def average_ranks(vals):
    order=sorted(range(len(vals)), key=vals.__getitem__)
    ranks=[0.0]*len(vals)
    i=0
    while i<len(vals):
        j=i+1
        v=vals[order[i]]
        while j<len(vals) and vals[order[j]]==v:
            j+=1
        avg=(i+1+j)/2.0
        for k in range(i,j): ranks[order[k]]=avg
        i=j
    return ranks

def pearson(a,b):
    ma=math.fsum(a)/len(a); mb=math.fsum(b)/len(b)
    da=[v-ma for v in a]; db=[v-mb for v in b]
    return math.fsum(u*v for u,v in zip(da,db))/math.sqrt(math.fsum(u*u for u in da)*math.fsum(v*v for v in db))

p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
ids=[]; x=[]; y=[]; invalid=[]
for rn,row in enumerate(ws.iter_rows(min_row=2, values_only=True),2):
    ident,a,b=row[0],row[5],row[6]
    ids.append(ident)
    if isinstance(a,(int,float)) and isinstance(b,(int,float)) and math.isfinite(a) and math.isfinite(b):
        x.append(float(a)); y.append(float(b))
    else: invalid.append((rn,ident,a,b))
counts=Counter(ids)
print('data_rows',len(ids),'valid_pairs',len(x),'invalid_pairs',len(invalid))
print('invalid',invalid[:20])
print('duplicate_rows_beyond_first',sum(v-1 for v in counts.values() if v>1),'duplicate_groups',sum(v>1 for v in counts.values()))
print('S1 min max zeros ones unique',min(x),max(x),sum(v==0 for v in x),sum(v==1 for v in x),len(set(x)))
print('S2 min max zeros ones unique',min(y),max(y),sum(v==0 for v in y),sum(v==1 for v in y),len(set(y)))
rho=pearson(average_ranks(x),average_ranks(y))
