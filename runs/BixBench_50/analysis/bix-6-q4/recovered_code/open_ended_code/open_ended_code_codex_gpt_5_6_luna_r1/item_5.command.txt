/bin/bash -lc "python3 - <<'PY'
from openpyxl import load_workbook
from math import sqrt
p='/workspace/inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
ws=load_workbook(p, read_only=True, data_only=True)['MAGeCK P-values']
x=[]; y=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    a,b=row[5],row[6]  # columns F/G in worksheet: Chronic Round1 S1/S2
    if a is not None and b is not None:
        x.append(float(a)); y.append(float(b))

def average_ranks(v):
    order=sorted(range(len(v)), key=v.__getitem__)
    r=[0.0]*len(v)
    i=0
    while i<len(v):
        j=i+1
        while j<len(v) and v[order[j]]==v[order[i]]:
            j+=1
        rank=(i+1+j)/2.0 # positions i+1 through j inclusive
        for k in order[i:j]: r[k]=rank
        i=j
    return r
rx=average_ranks(x); ry=average_ranks(y)
mx=sum(rx)/len(rx); my=sum(ry)/len(ry)
num=sum((a-mx)*(b-my) for a,b in zip(rx,ry))
den=sqrt(sum((a-mx)**2 for a in rx)*sum((b-my)**2 for b in ry))
rho=num/den
print('n',len(x))
print('mean ranks',repr(mx),repr(my))
print('rho',format(rho,'.17g'))
# equivalent squared-rank difference formula only valid no ties, report tie-aware check
print('sum rank products',repr(num),repr(den))
print('tie counts',len(x)-len(set(x)),len(y)-len(set(y)))
# Pearson raw values as sanity (not requested)
mx0=sum(x)/len(x); my0=sum(y)/len(y)
print('pearson raw',sum((a-mx0)*(b-my0) for a,b in zip(x,y))/sqrt(sum((a-mx0)**2 for a in x)*sum((b-my0)**2 for b in y)))
PY"
