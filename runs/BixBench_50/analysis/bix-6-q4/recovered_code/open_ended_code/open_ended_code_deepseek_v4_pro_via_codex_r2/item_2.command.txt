/bin/bash -lc "cd /workspace && python - <<'PY'
import pandas as pd
import scipy.stats as st
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
df=pd.read_excel(p, sheet_name='MAGeCK P-values')
print(df.shape)
print(df.columns)
sub=df[['Chronic Round1 S1','Chronic Round1 S2']]
print(sub.describe())
print('nulls', sub.isna().sum())
print('dups refseq', df['RefSeq ID'].duplicated().sum())
print('infs', (~sub.isin([float('inf'), float('-inf')])).all())
# spearman scipy
r,p=st.spearmanr(sub['Chronic Round1 S1'], sub['Chronic Round1 S2'], nan_policy='omit')
print('scipy spearman', r, p, 'n', len(sub), 'valid', sub.dropna().shape[0])
# pandas
print('pandas spearman', sub.corr(method='spearman').iloc[0,1])
# manual rank corr
ranked=sub.rank(method='average')
print('rank corr', ranked.corr().iloc[0,1])
PY"
