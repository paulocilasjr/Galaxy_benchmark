/bin/bash -lc "python3 - <<'PY'
import openpyxl, math
# manual rank with average ties
p='/workspace/inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
hdr=next(ws.iter_rows(values_only=True))
ci=[i for i,h in enumerate(hdr) if str(h).startswith('Chronic Round1')]
x=[];y=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    x.append(float(r[ci[0]])); y.append(float(r[ci[1]]))
def rankdata(vals):
    order=sorted(range(len(vals)), key=lambda i: vals[i])
    ranks=[0.0]*len(vals)
    i=0;n=len(order)
    while i<n:
        j=i+1
        while j<n and vals[order[j]]==vals[order[i]]:
            j+=1
        avg=(i+1+j)/2.0  # ranks i+1..j inclusive average
        for k in range(i,j): ranks[order[k]]=avg
        i=j
    return ranks
rx=rankdata(x); ry=rankdata(y)
mx=sum(rx)/len(rx); my=sum(ry)/len(ry)
num=sum((a-mx)*(b-my) for a,b in zip(rx,ry))
den=(sum((a-mx)**2 for a in rx)*sum((b-my)**2 for b in ry))**0.5
print(num/den, len(x))
from scipy.stats import spearmanr
print(spearmanr(x,y).statistic)
# print formatted variants
rho=num/den
print(repr(rho))
print(f'{rho:.15f}')
print(f'{rho:.12f}')
print(f'{rho:.6f}')
PY"
