/bin/bash -lc "python3 - <<'PY'
import openpyxl, math
from scipy import stats
p='/workspace/inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
rows=list(ws.iter_rows(values_only=True))
header=rows[0]
print(header)
ci=[i for i,h in enumerate(header) if 'Chronic Round1' in str(h)]
print('ci',ci, [header[i] for i in ci])
vals=[]
bad=[]
for idx,r in enumerate(rows[1:], start=2):
    try:
        a=float(r[ci[0]]); b=float(r[ci[1]])
    except Exception as e:
        bad.append((idx,repr(r),e)); continue
    if not (math.isfinite(a) and math.isfinite(b)):
        bad.append((idx,(a,b))); continue
    vals.append((a,b))
print('total rows',len(rows)-1,'good',len(vals),'bad',len(bad))
print('bad examples', bad[:10])
x=[v[0] for v in vals]; y=[v[1] for v in vals]
res=stats.spearmanr(x,y)
print(res)
print('corr', repr(res.statistic))
print('p', repr(res.pvalue))
# basic pearson correlation too, maybe spearman rank correlation
PY"
