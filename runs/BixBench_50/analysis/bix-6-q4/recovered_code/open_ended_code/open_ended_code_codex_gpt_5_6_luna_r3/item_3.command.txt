/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from scipy.stats import spearmanr
import math
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
rows=ws.iter_rows(min_row=2, values_only=True)
ids=[]; x=[]; y=[]; bad=[]
for i,row in enumerate(rows, start=2):
    ident,a,b,*rest=row
    ids.append(ident)
    if isinstance(a,(int,float)) and isinstance(b,(int,float)) and math.isfinite(a) and math.isfinite(b):
        x.append(a); y.append(b)
    else:
        bad.append((i,ident,a,b))
res=spearmanr(x,y)
print('n rows', len(ids), 'n valid', len(x), 'bad', len(bad))
print('bad sample', bad[:10])
print('unique IDs', len(set(ids)), 'duplicate count', len(ids)-len(set(ids)))
print('x unique', len(set(x)), 'y unique', len(set(y)))
print('x min max', min(x), max(x), 'y min max', min(y), max(y))
print('spearman statistic', repr(float(res.statistic)), 'pvalue', repr(float(res.pvalue)))
# rank Pearson explicit
from scipy.stats import rankdata, pearsonr
r2=pearsonr(rankdata(x, method='average'), rankdata(y, method='average'))[0]
print('rank Pearson', repr(float(r2)))
PY"
