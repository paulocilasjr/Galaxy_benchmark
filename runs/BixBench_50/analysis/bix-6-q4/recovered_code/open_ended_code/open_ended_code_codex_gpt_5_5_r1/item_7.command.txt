/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from math import isnan
path='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(path, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
header=next(ws.iter_rows(values_only=True))
print(header)
i1=header.index('Chronic Round1 S1')
i2=header.index('Chronic Round1 S2')
x=[]; y=[]; bad=0
for row in ws.iter_rows(values_only=True):
    # skip header by type of first? header already consumed? iter_rows starts new each call
    break
for row in ws.iter_rows(min_row=2, values_only=True):
    a=row[i1]; b=row[i2]
    if a is None or b is None:
        bad += 1
        continue
    try:
        af=float(a); bf=float(b)
    except Exception:
        bad += 1; continue
    if isnan(af) or isnan(bf):
        bad += 1; continue
    x.append(af); y.append(bf)
print('n', len(x), 'bad', bad, 'minmax', (min(x),max(x)), (min(y),max(y)))
try:
    import scipy.stats as st
    r,p=st.spearmanr(x,y)
    print('scipy', repr(r), repr(p))
except Exception as e:
    print('scipy err', e)
try:
    import pandas as pd
    s1=pd.Series(x)
    s2=pd.Series(y)
    print('pandas', repr(s1.corr(s2, method='spearman')))
except Exception as e:
    print('pandas err', e)
PY"
