/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from math import sqrt
path='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(path, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
header=next(ws.iter_rows(values_only=True))
cols=[header.index('Chronic Round1 S1'), header.index('Chronic Round1 S2')]
pairs=[(float(r[cols[0]]), float(r[cols[1]])) for r in ws.iter_rows(min_row=2, values_only=True)]

def ranks(vals):
    sorted_vals=sorted((v,i) for i,v in enumerate(vals))
    out=[None]*len(vals)
    pos=0
    while pos < len(sorted_vals):
        end=pos+1
        while end < len(sorted_vals) and sorted_vals[end][0] == sorted_vals[pos][0]:
            end += 1
        rank=(pos+1 + end)/2
        for _,idx in sorted_vals[pos:end]:
            out[idx]=rank
        pos=end
    return out
rx, ry = ranks([p[0] for p in pairs]), ranks([p[1] for p in pairs])
# Pearson via centered sums; sample/pop denominator cancels.
mx=sum(rx)/len(rx); my=sum(ry)/len(ry)
ssx=sum((v-mx)*(v-mx) for v in rx)
ssy=sum((v-my)*(v-my) for v in ry)
sp=sum((a-mx)*(b-my) for a,b in zip(rx,ry))
print(sp/(ssx*ssy)**0.5)
PY"
