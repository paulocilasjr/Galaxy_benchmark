/bin/bash -lc "python - <<'PY'
# compact duplicate/site check for carrier-only coding VAF<0.3
from zipfile import ZipFile
from pathlib import Path
import xml.etree.ElementTree as ET
from collections import Counter, defaultdict
NS={'a':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
def colnum(cellref):
    s=''.join(ch for ch in cellref if ch.isalpha()); n=0
    for ch in s: n=n*26+ord(ch.upper())-64
    return n-1
def read(path):
    with ZipFile(path) as z:
        ss=[]
        if 'xl/sharedStrings.xml' in z.namelist():
            root=ET.fromstring(z.read('xl/sharedStrings.xml'))
            ss=[''.join(t.text or '' for t in si.findall('.//a:t', NS)) for si in root.findall('a:si', NS)]
        root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
        out=[]
        for rr in root.findall('a:sheetData/a:row', NS):
            vals=[]
            for c in rr.findall('a:c', NS):
                idx=colnum(c.attrib.get('r','A1'))
                while len(vals)<idx: vals.append('')
                typ=c.attrib.get('t')
                if typ=='inlineStr': val=''.join(t.text or '' for t in c.findall('.//a:t', NS))
                else:
                    v=c.find('a:v', NS); val='' if v is None else v.text or ''
                    if typ=='s' and val"'!='"'': val=ss[int(val)]
                vals.append(val)
            out.append(vals)
        return out
meta=read('inputs/230215_Trio_Status.xlsx'); mi={h:i for i,h in enumerate(meta[0])}
carriers={r[mi['Sample ID']].strip() for r in meta[1:] if r[mi['BLM Mutation Status']].strip()=='Carrier'}
coding={'synonymous_variant','missense_variant','stop_gained','stop_lost','start_lost','frameshift_variant','inframe_deletion','inframe_insertion','protein_altering_variant','coding_sequence_variant'}
records=[]
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    sid=p.stem.replace('230209_Exome_GRCh38_CHIP_','').split('-')[0]
    if sid not in carriers: continue
    rows=read(p); hdr=rows[1]; idx={h:i for i,h in enumerate(hdr)}
    for r in rows[2:]:
        v=r[idx['Variant Allele Freq']] if idx['Variant Allele Freq']<len(r) else ''
        if not v: continue
        try: vf=float(v)
        except: continue
        so=r[idx['Sequence Ontology (Combined)']] if idx['Sequence Ontology (Combined)']<len(r) else ''
        terms={x.strip() for x in so.split(',')}
        if vf<0.3 and terms & coding:
            key=(r[idx['Chr:Pos']], r[idx['Ref/Alt']], so)
            records.append((sid,key,so))
print('sample-variant records',len(records), 'syn', sum(1 for _,_,so in records if 'synonymous_variant' in so), 'frac', sum(1 for _,_,so in records if 'synonymous_variant' in so)/len(records))
uniq={key:so for _,key,so in records}
print('unique sites',len(uniq), 'syn', sum(1 for so in uniq.values() if 'synonymous_variant' in so), 'frac', sum(1 for so in uniq.values() if 'synonymous_variant' in so)/len(uniq))
print('duplicates')
for key,c in Counter(key for _,key,_ in records).most_common():
    if c>1: print(key,c)
PY"
