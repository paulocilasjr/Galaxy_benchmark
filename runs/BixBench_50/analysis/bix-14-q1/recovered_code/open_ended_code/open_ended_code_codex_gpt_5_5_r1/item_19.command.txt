/bin/bash -lc "python - <<'PY'
# final deterministic recomputation and write canonical answer
from zipfile import ZipFile
from pathlib import Path
import xml.etree.ElementTree as ET
NS={'a':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}

def colnum(cellref):
    s=''.join(ch for ch in cellref if ch.isalpha())
    n=0
    for ch in s:
        n=n*26+ord(ch.upper())-64
    return n-1

def read_xlsx(path):
    with ZipFile(path) as z:
        ss=[]
        if 'xl/sharedStrings.xml' in z.namelist():
            root=ET.fromstring(z.read('xl/sharedStrings.xml'))
            for si in root.findall('a:si', NS):
                ss.append(''.join(t.text or '' for t in si.findall('.//a:t', NS)))
        root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
        rows=[]
        for rr in root.findall('a:sheetData/a:row', NS):
            vals=[]
            for c in rr.findall('a:c', NS):
                idx=colnum(c.attrib.get('r','A1'))
                while len(vals)<idx:
                    vals.append('')
                typ=c.attrib.get('t')
                if typ=='inlineStr':
                    val=''.join(t.text or '' for t in c.findall('.//a:t', NS))
                else:
                    v=c.find('a:v', NS)
                    val='' if v is None else v.text or ''
                    if typ=='s' and val"'!='"'':
                        val=ss[int(val)]
                vals.append(val)
            rows.append(vals)
        return rows

meta=read_xlsx(Path('inputs/230215_Trio_Status.xlsx'))
mi={h:i for i,h in enumerate(meta[0])}
carriers={r[mi['Sample ID']].strip() for r in meta[1:] if r[mi['BLM Mutation Status']].strip()=='Carrier'}

coding_terms={
    'synonymous_variant','missense_variant','stop_gained','stop_lost','start_lost',
    'frameshift_variant','inframe_deletion','inframe_insertion','protein_altering_variant',
    'coding_sequence_variant'
}

total=syn=0
used_samples=[]
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    sid=p.stem.replace('230209_Exome_GRCh38_CHIP_','').split('-')[0].strip()
    if sid not in carriers:
        continue
    used_samples.append(sid)
    rows=read_xlsx(p)
    hdr=rows[1]
    idx={h:i for i,h in enumerate(hdr)}
    for r in rows[2:]:
        vaf_str=r[idx['Variant Allele Freq']] if idx['Variant Allele Freq'] < len(r) else ''
        if not vaf_str:
            continue
        try:
            vaf=float(vaf_str)
        except ValueError:
            continue
        if vaf >= 0.3:
            continue
        so=r[idx['Sequence Ontology (Combined)']] if idx['Sequence Ontology (Combined)'] < len(r) else ''
        terms={term.strip() for term in so.split(',') if term.strip()}
        if not (terms & coding_terms):
            continue
        total += 1
        if 'synonymous_variant' in terms:
            syn += 1

if total == 0:
    raise SystemExit('No coding variants after filters')
answer=syn/total
Path('codex_output').mkdir(exist_ok=True)
Path('codex_output/answer.txt').write_text(f'{answer}\\n')
print('carrier metadata ids', len(carriers), sorted(carriers))
print('carrier workbooks used', len(used_samples), sorted(used_samples))
print('synonymous', syn, 'coding_total', total, 'fraction', answer)
PY"
