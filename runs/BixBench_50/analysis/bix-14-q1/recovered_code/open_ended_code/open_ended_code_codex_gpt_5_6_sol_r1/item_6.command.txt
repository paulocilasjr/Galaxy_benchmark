/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
p=Path('inputs/230215_Trio_Status.xlsx')
ws=load_workbook(p,read_only=True,data_only=True).active
rows=list(ws.iter_rows(min_row=2,values_only=True))
from collections import Counter
print('Status counts',Counter(r[1] for r in rows))
print('BLM counts',Counter(r[3] for r in rows))
print('Carrier rows:')
for r in rows:
    if r[3]=='Carrier': print(r[:6])
print('n carrier',sum(r[3]=='Carrier' for r in rows))
print('Affected rows', [r[0] for r in rows if r[3]=='Affected'])
print('Control rows', [(r[0],r[1],r[3]) for r in rows if r[3] not in ('Carrier','Affected')][:20])
PY"
