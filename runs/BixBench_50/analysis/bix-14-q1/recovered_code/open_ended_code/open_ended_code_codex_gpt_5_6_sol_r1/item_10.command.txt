/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
from collections import Counter
import re
base=Path('inputs')
meta=load_workbook(base/'230215_Trio_Status.xlsx',read_only=True,data_only=True).active
ids={str(r[0]).strip() for r in meta.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier'}
low=[]; boundary=[]
for p in (base/'CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
 m=re.search(r'CHIP_(["'^-.]+)'"',p.name)
 if not m or m.group(1) not in ids: continue
 sid=m.group(1); sh=load_workbook(p,read_only=True,data_only=True).active; sh.reset_dimensions(); it=sh.iter_rows(values_only=True); next(it); h=next(it); ix={v:i for i,v in enumerate(h) if v}
 for row in it:
  v=row[ix['Variant Allele Freq']]
  if v is None: continue
  v=float(v); so=row[ix['Sequence Ontology (Combined)']]
  r=(sid,row[ix['Chr:Pos']],v,row[ix['Zygosity']],row[ix['Gene Names']],so,row[ix['Effect (Combined)']],row[ix['HGVS c. (Clinically Relevant)']],row[ix['HGVS p. (Clinically Relevant)']])
  if v<.3: low.append(r)
  if .299<=v<=.301: boundary.append(r)
print('coding under common SO definition:',Counter(x[5] for x in low if x[5] in {'synonymous_variant','missense_variant','frameshift_variant','inframe_insertion','inframe_deletion','stop_gained','stop_lost','start_lost','protein_altering_variant'}))
print('Effect under .3',Counter(x[6] for x in low))
print('splice rows:')
for r in low:
 if r[5]=='splice_region_variant': print(r)
print('boundary rows:')
for r in boundary: print(r)
PY"
