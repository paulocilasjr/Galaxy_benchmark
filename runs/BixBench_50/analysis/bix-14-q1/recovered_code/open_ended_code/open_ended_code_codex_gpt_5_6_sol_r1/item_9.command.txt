/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
from collections import Counter
import re
base=Path('inputs')
meta=load_workbook(base/'230215_Trio_Status.xlsx',read_only=True,data_only=True).active
ids={str(r[0]).strip() for r in meta.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier'}
rec=[]
for p in (base/'CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
    m=re.search(r'CHIP_(["'^-.]+)'"',p.name)
    if not m or m.group(1) not in ids: continue
    sid=m.group(1); sh=load_workbook(p,read_only=True,data_only=True).active; sh.reset_dimensions(); it=sh.iter_rows(values_only=True); next(it); h=next(it); ix={v:i for i,v in enumerate(h) if v}
    for row in it:
        v=row[ix['Variant Allele Freq']]
        if v is None: continue
        v=float(v)
        so=row[ix['Sequence Ontology (Combined)']]
        if v<.3 and so in ('synonymous_variant','missense_variant'):
            rec.append((sid,row[ix['Chr:Pos']],row[ix['Ref/Alt']],row[ix['Zygosity']],v,row[ix['Gene Names']],so,row[ix['Effect (Combined)']],row[ix['HGVS c. (Clinically Relevant)']],row[ix['HGVS p. (Clinically Relevant)']]))
print('n',len(rec),Counter((x[6],x[3]) for x in rec))
print('sample',Counter((x[0],x[6]) for x in rec))
for x in sorted(rec): print(x)
PY"
