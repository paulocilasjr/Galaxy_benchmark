source /tmp/venv/bin/activate && python3 << 'EOF'
import pandas as pd
import os

df = pd.read_excel('inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_185-PF.xlsx', header=1)

# Check Zygosity values
print("Zygosity values:")
print(df['Zygosity'].value_counts())
print()

# Check VAF vs Zygosity
print("VAF stats by Zygosity:")
for z in df['Zygosity'].unique():
    subset = df[df['Zygosity'] == z]
    vaf = pd.to_numeric(subset['Variant Allele Freq'], errors='coerce')
    print(f"  {z}: count={len(subset)}, VAF NaN={vaf.isna().sum()}, VAF non-NaN={vaf.notna().sum()}")
    if vaf.notna().sum() > 0:
        print(f"       VAF min={vaf.min():.4f}, max={vaf.max():.4f}")

print()
# Check if VAF column is numeric
print("VAF dtype:", df['Variant Allele Freq'].dtype)
print("VAF sample values:", df['Variant Allele Freq'].dropna().head(20).tolist())
