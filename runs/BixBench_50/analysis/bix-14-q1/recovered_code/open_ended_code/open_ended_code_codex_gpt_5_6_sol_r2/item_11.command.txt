/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
from collections import Counter,defaultdict
# metadata
p=Path('inputs/230215_Trio_Status.xlsx'); w=load_workbook(p,read_only=True,data_only=True); ws=w.active; ws.reset_dimensions(); rr=list(ws.iter_rows(values_only=True));
carriers=[str(r[0]).strip() for r in rr[1:] if r[3]=='Carrier']
print('carriers',carriers)
files={p.stem.split('CHIP_',1)[1].split('-',1)[0]:p for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')}
print('missing',[x for x in carriers if x not in files])
cs=Counter(); ce=Counter(); cz=Counter(); cpair=Counter(); per=Counter()
allv=0
for sid in carriers:
 if sid not in files: continue
 p=files[sid]; wb=load_workbook(p,read_only=True,data_only=True); sh=wb.active; sh.reset_dimensions(); it=sh.iter_rows(values_only=True); next(it); hdr=next(it); idx={x:i for i,x in enumerate(hdr)}
 for row in it:
  v=row[idx['Variant Allele Freq']]
  try:v=float(v)
  except (TypeError,ValueError):continue
  if v<.3:
   allv+=1
   so=row[idx['Sequence Ontology (Combined)']]; ef=row[idx['Effect (Combined)']]; zy=row[idx['Zygosity']]
   cs[so]+=1; ce[ef]+=1; cz[zy]+=1; cpair[(so,ef)]+=1;per[sid]+=1
print('all VAF<.3',allv,'per',per)
print('\\nSO');
for k,v in cs.most_common(): print(v,repr(k))
print('\\nEffect');
for k,v in ce.most_common(): print(v,repr(k))
print('\\nZygosity',cz)
print('\\nPairs');
for k,v in cpair.most_common(): print(v,k)
PY"
