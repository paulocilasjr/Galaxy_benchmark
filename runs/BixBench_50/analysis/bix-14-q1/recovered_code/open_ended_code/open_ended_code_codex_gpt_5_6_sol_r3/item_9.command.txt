/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter,defaultdict
import openpyxl,re
meta=list(openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',read_only=True,data_only=True).active.iter_rows(values_only=True))[1:]
carrier_ids={str(r[0]).strip() for r in meta if r[3]=='Carrier'}
files=[]
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
 m=re.search(r'_CHIP_(["'^-'"\\.]+)',p.name)
 if m and m.group(1) in carrier_ids: files.append(p)
print('carrier files',len(files),[p.stem.split('_CHIP_')[1] for p in sorted(files)])
all_pairs=Counter(); low_pairs=Counter(); zyg=Counter(); per=defaultdict(Counter); low=[]
for p in sorted(files):
 ws=openpyxl.load_workbook(p,read_only=True,data_only=True).active
 ws.reset_dimensions()
 it=ws.iter_rows(min_row=2,values_only=True)
 header=next(it)
 idx={h:i for i,h in enumerate(header)}
 for row in it:
  so=row[idx['Sequence Ontology (Combined)']]
  eff=row[idx['Effect (Combined)']]
  all_pairs[(so,eff)] += 1
  v=row[idx['Variant Allele Freq']]
  try: vf=float(v)
  except (TypeError,ValueError): continue
  if vf<0.3:
   pair=(so,eff)
   low_pairs[pair]+=1
   zyg[row[idx['Zygosity']]]+=1
   per[p.name][pair]+=1
   low.append((p.name,row[idx['Chr:Pos']],row[idx['Gene Names']],so,eff,v,row[idx['Zygosity']],row[idx['HGVS c. (Clinically Relevant)']],row[idx['HGVS p. (Clinically Relevant)']]))
print('\\nLOW VAF PAIRS')
for k,v in low_pairs.most_common(): print(v,k)
print('\\nLOW ZYG',zyg,'TOTAL',len(low))
print('\\nALL PAIRS')
for k,v in all_pairs.most_common(): print(v,k)
print('\\nLOW ROWS')
for r in low: print(r)
PY"
