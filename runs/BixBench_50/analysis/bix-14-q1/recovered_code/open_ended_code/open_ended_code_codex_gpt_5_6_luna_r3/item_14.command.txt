/bin/bash -lc "python - <<'PY'
# Parse carrier low-VAF coding rows and report them for semantic validation.
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
from collections import Counter
import re, openpyxl
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def colnum(ref):
 s=''.join(c for c in ref if c.isalpha()); n=0
 for c in s:n=n*26+ord(c)-64
 return n
def load(p):
 with ZipFile(p) as z:
  ss=[''.join(t.text or '' for t in si.iter(NS+'t')) for si in ET.fromstring(z.read('xl/sharedStrings.xml')).findall(NS+'si')]
  root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
  rows=[]
  for row in root.find(NS+'sheetData').findall(NS+'row')[2:]:
   d={}
   for c in row.findall(NS+'c'):
    v=c.find(NS+'v'); x=None if v is None else v.text
    if c.attrib.get('t')=='s' and x is not None:x=ss[int(x)]
    d[colnum(c.attrib['r'])]=x
   rows.append(d)
 return rows
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True)
carriers=[str(r[0]).strip() for r in wb.active.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier']
by={}
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
 tok=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name).group(1); m=re.match(r'\\d+',tok); sid=m.group(0) if m else tok; by[sid]=p
for sid in carriers:
 if sid not in by:continue
 for r in load(by[sid]):
  try:vaf=float(r.get(10))
  except:continue
  if vaf < .3 and r.get(14) in ('synonymous_variant','missense_variant','inframe_insertion','inframe_deletion','frameshift_variant','stop_gained','stop_lost','start_lost','coding_sequence_variant'):
   print(sid, 'VAF',r.get(10),'variant',r.get(1),r.get(2),'gene',r.get(13),'SO',r.get(14),'effect',r.get(15),'zygo',r.get(5),'DP',r.get(8))
PY"
