/bin/bash -lc "python - <<'PY'
import openpyxl
from collections import Counter
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
import re
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def col(ref):return ''.join(c for c in ref if c.isalpha())
def rows(p):
 with ZipFile(p) as z:
  ss=[''.join(t.text or '' for t in si.iter(NS+'t')) for si in ET.fromstring(z.read('xl/sharedStrings.xml')).findall(NS+'si')]
  root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
  for row in root.find(NS+'sheetData').findall(NS+'row')[2:]:
   d={}
   for c in row.findall(NS+'c'):
    v=c.find(NS+'v')
    if v is None: continue
    x=v.text
    if c.attrib.get('t')=='s':x=ss[int(x)]
    d[col(c.attrib['r'])]=x
   yield d
chip=[r[0] for r in openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx',data_only=True).active.iter_rows(values_only=True) if r[0]]
print('chip count',len(chip),'contains extras',[g for g in ['MMAB','SF3B2','MMACHC'] if g in chip])
r=rows(sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))[0])
c=Counter((x.get('M'),x.get('AD'),x.get('AE')) for x in r)
for k,v in c.most_common():
 if k[0] in ('MMAB','SF3B2','MMACHC'): print(k,v)
print('all gene match pairs count',len(c))
print('genes',Counter(x.get('M') for x in r))
PY"
