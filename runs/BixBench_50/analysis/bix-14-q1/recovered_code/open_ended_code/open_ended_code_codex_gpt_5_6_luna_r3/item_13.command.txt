/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
from collections import Counter
import re, openpyxl
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def colnum(ref):
 s=''.join(c for c in ref if c.isalpha()); n=0
 for c in s:n=n*26+ord(c)-64
 return n
def load(p):
 with ZipFile(p) as z:
  root=ET.fromstring(z.read('xl/sharedStrings.xml'))
  ss=[''.join(t.text or '' for t in si.iter(NS+'t')) for si in root.findall(NS+'si')]
  root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
  out=[]
  for row in root.find(NS+'sheetData').findall(NS+'row')[2:]:
   d={}
   for c in row.findall(NS+'c'):
    v=c.find(NS+'v'); val=None if v is None else v.text
    if c.attrib.get('t')=='s' and val is not None: val=ss[int(val)]
    d[colnum(c.attrib['r'])]=val
   out.append(d)
 return out
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True)
meta={str(row[0]).strip(): row[3] for row in wb.active.iter_rows(min_row=2,values_only=True) if row[0] is not None}
files=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))
by={}
for p in files:
 tok=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name).group(1)
 sid=re.match(r'\\d+',tok)
 sid=sid.group(0) if sid else tok
 by[sid]=p
carriers=[sid for sid,status in meta.items() if status=='Carrier']
print('available files',len(files),'metadata IDs',len(meta),'carriers metadata',len(carriers))
print('carrier IDs',carriers)
print('carrier available', [x for x in carriers if x in by])
print('carrier missing', [x for x in carriers if x not in by])
print('available no metadata',[x for x in by if x not in meta])
agg=Counter(); agg_effect=Counter()
for sid in carriers:
 if sid not in by: continue
 rows=load(by[sid]); v=[]
 for r in rows:
  try: vaf=float(r.get(10))
  except (TypeError,ValueError): continue
  if vaf<.3: v.append(r)
 print(sid,by[sid].name,len(rows),'vaf<.3',len(v),'SO',Counter(r.get(14) for r in v),'effect',Counter(r.get(15) for r in v))
 for r in v:
  agg[r.get(14)]+=1; agg_effect[r.get(15)]+=1
print('aggregate SO',agg)
print('aggregate effect',agg_effect)
PY"
