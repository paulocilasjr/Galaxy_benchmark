/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
import re,openpyxl
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def col(ref):return ''.join(c for c in ref if c.isalpha())
def rows(p):
 with ZipFile(p) as z:
  ss=[''.join(t.text or '' for t in si.iter(NS+'t')) for si in ET.fromstring(z.read('xl/sharedStrings.xml')).findall(NS+'si')]
  root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
  for row in root.find(NS+'sheetData').findall(NS+'row')[2:]:
   d={}
   for c in row.findall(NS+'c'):
    v=c.find(NS+'v')
    if v is None:continue
    x=v.text
    if c.attrib.get('t')=='s':x=ss[int(x)]
    d[col(c.attrib['r'])]=x
   yield d
chip={str(r[0]).strip() for r in openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx',data_only=True).active.iter_rows(values_only=True) if r[0]}
meta=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True).active
carrier={str(r[0]).strip() for r in meta.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier'}
out={}
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
 token=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name).group(1);m=re.match(r'\\d+',token);sid=m.group(0) if m else token
 if sid not in carrier:continue
 for r in rows(p):
  genes=(r.get('M') or '').split(',')
  if not any(g in chip for g in genes):out[r.get('M')]=out.get(r.get('M'),0)+1
print('outside unique',len(out))
for k,v in sorted(out.items(),key=lambda x:-x[1])[:50]:print(v,repr(k))
PY"
