/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
import re
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def colnum(ref):
    s=''.join(c for c in ref if c.isalpha())
    n=0
    for c in s:n=n*26+ord(c.upper())-64
    return n
p=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))[0]
with ZipFile(p) as z:
    ss=[]
    root=ET.fromstring(z.read('xl/sharedStrings.xml'))
    for si in root.findall(NS+'si'):
        ss.append(''.join(t.text or '' for t in si.iter(NS+'t')))
    root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
    rows=[]
    for row in root.find(NS+'sheetData').findall(NS+'row')[:12]:
        vals={}
        for c in row.findall(NS+'c'):
            ref=c.attrib['r']; v=c.find(NS+'v'); val=None if v is None else v.text
            if c.attrib.get('t')=='s' and val is not None: val=ss[int(val)]
            vals[colnum(ref)]=val
        rows.append(vals)
    for i,d in enumerate(rows,1):
        print('ROW',i)
        print(' | '.join(f'{k}:{v}' for k,v in sorted(d.items())))
PY"
