from pathlib import Path
from openpyxl import load_workbook
import re
from collections import Counter, defaultdict
carrier={'184','185','285','287','381','353','354','396','397','489','490','499','500','503','504','556','557','613','614'}
def num(x):
 try:return float(x) if x not in (None,'') else None

for key in ['pos','posref','gene_pos']:

 print(key,'unique',len(d),'syn',sum(any(x['ont']=='synonymous_variant' for x in z) for z in d.values()),'miss',sum(any(x['ont']=='missense_variant' for x in z) for z in d.values()))
