/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
import re
carrier={'184','185','285','287','353','354','381','396','397','489','490','499','500','503','504','556','557','613','614'}
coding={'synonymous_variant','missense_variant','inframe_deletion','inframe_insertion','frameshift_variant','stop_gained','stop_lost','initiator_codon_variant','splice_acceptor_variant','splice_donor_variant'}
syn=total=0
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
    m=re.match(r'(\\d+)',p.stem.split('CHIP_',1)[1])
    if not m or m.group(1) not in carrier: continue
    ws=load_workbook(p,data_only=True).active
    for r in range(3,ws.max_row+1):
        try: v=float(ws.cell(r,10).value)
        except (TypeError,ValueError): continue
        ont=ws.cell(r,14).value
        if v < 0.3 and ont in coding:
            total += 1
            syn += (ont == 'synonymous_variant')
print(f'{syn}/{total} = {syn/total:.10f}')
print(Path('codex_output/answer.txt').read_text(), end='')
assert (syn,total)==(30,41)
PY"
