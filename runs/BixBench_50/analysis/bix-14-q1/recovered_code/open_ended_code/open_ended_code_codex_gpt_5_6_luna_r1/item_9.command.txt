from pathlib import Path
from openpyxl import load_workbook
from collections import Counter,defaultdict
import re, math
carrier_ids={'184','185','285','287','353','354','381','396','397','489','490','499','500','503','504','556','557','613','614','533'}
files=[]
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    sid=p.stem.split('CHIP_',1)[1]
    base=re.match(r'(\\d+)',sid)
    if base and base.group(1) in carrier_ids:
        files.append((base.group(1),sid,p))
print('selected',len(files),[sid for _,sid,_ in files])
all_ont=Counter(); all_eff=Counter(); all_vaf=[]; rows_by_file={}
for base,sid,p in files:
    wb=load_workbook(p,data_only=True,read_only=False)
    ws=wb.active
    headers=[ws.cell(2,c).value for c in range(1,ws.max_column+1)]
    idx={h:i+1 for i,h in enumerate(headers) if h}
    cnt=Counter(); eff=Counter(); vafs=[]
    for r in range(3,ws.max_row+1):
        vaf=ws.cell(r,10).value
        ont=ws.cell(r,14).value
        e=ws.cell(r,15).value
        if ont is not None: all_ont[str(ont)]+=1; cnt[str(ont)]+=1
        if e is not None: all_eff[str(e)]+=1; eff[str(e)]+=1
        try:
            x=float(vaf)
        except (ValueError,TypeError):
            x=None
        if x is not None: vafs.append((x,str(ont),str(e),ws.cell(r,13).value,ws.cell(r,1).value,ws.cell(r,2).value,ws.cell(r,5).value))
    rows_by_file[sid]=(cnt,eff,vafs)
    all_vaf += [(x,ont,e,sid) for x,ont,e,*_ in vafs]
    print(sid,'rows',ws.max_row-2,'numeric_vaf',len(vafs),'ont',dict(cnt),'eff',dict(eff),'vafminmax', (min([x for x,*_ in vafs],default=None),max([x for x,*_ in vafs],default=None)))
print('\\nALL ONT',all_ont)
print('ALL EFFECT',all_eff)
print('numeric vaf <.3',len([r for r in all_vaf if r[0]<.3]))
print('ont among vaf<.3',Counter(r[1] for r in all_vaf if r[0]<.3))
print('effect among vaf<.3',Counter(r[2] for r in all_vaf if r[0]<.3))
PY"
