import openpyxl
from collections import Counter

# Check VAF and zygosity distribution in one carrier file
wb = openpyxl.load_workbook('/workspace/inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_184-PM.xlsx')
ws = wb.active

zyg_counter = Counter()
vaf_present = 0
vaf_missing = 0
vaf_values = []

for row in ws.iter_rows(min_row=3, values_only=True):
    zyg = row[4]  # Zygosity column
    vaf = row[9]  # VAF column
    zyg_counter[str(zyg)] += 1
    if vaf is not None:
        vaf_present += 1
        vaf_values.append(float(vaf))
    else:
        vaf_missing += 1

print('Zygosity distribution:')
for k, v in zyg_counter.most_common():
    print(f'  {k}: {v}')

print(f'\nVAF present: {vaf_present}, VAF missing: {vaf_missing}')

if vaf_values:
    vaf_values.sort()
    print(f'\nVAF range: {min(vaf_values):.4f} - {max(vaf_values):.4f}')
    print(f'VAF < 0.3: {sum(1 for v in vaf_values if v < 0.3)}')
    print(f'VAF >= 0.3: {sum(1 for v in vaf_values if v >= 0.3)}')
    # sample some
    print(f'First 10 VAF values: {vaf_values[:10]}')
