import openpyxl
import os

coding_so_terms = {
    'synonymous_variant', 'missense_variant', 'stop_gained', 'stop_lost',
    'start_lost', 'frameshift_variant', 'inframe_insertion', 'inframe_deletion',
    'coding_sequence_variant', 'protein_altering_variant',
}

# Check VAF=0.0 variants across all carriers
chip_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'
carrier_ids = ['184', '185', '285', '287', '353', '354', '381', '396', '397',
               '489', '490', '499', '500', '503', '504', '556', '557', '613', '614']

vaf_zero = []
for sid in carrier_ids:
    for f in os.listdir(chip_dir):
        if f.startswith(f'230209_Exome_GRCh38_CHIP_{sid}-') and f.endswith('.xlsx'):
            fpath = os.path.join(chip_dir, f)
            break
    else:
        continue
    wb = openpyxl.load_workbook(fpath)
    ws = wb.active
    for row in ws.iter_rows(min_row=3, values_only=True):
        vaf = row[9]
        if vaf is not None and float(vaf) == 0.0:
            so = str(row[13]).strip() if row[13] else None
            zyg = row[4]
            vaf_zero.append((sid, row[0], row[12], so, zyg, row[14]))

print(f"Total VAF=0.0 variants across all carriers: {len(vaf_zero)}")
for v in vaf_zero[:20]:
    print(f"  Sample={v[0]}, Pos={v[1]}, Gene={v[2]}, SO={v[3]}, Zyg={v[4]}, Effect={v[5]}")

# Check if any VAF=0.0 variants are coding
coding_vaf_zero = [v for v in vaf_zero if v[3] in coding_so_terms]
print(f"\nOf these, coding variants: {len(coding_vaf_zero)}")
for v in coding_vaf_zero:
    print(f"  Sample={v[0]}, Pos={v[1]}, Gene={v[2]}, SO={v[3]}, Zyg={v[4]}, Effect={v[5]}")
