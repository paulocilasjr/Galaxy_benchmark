from openpyxl import load_workbook
from collections import Counter
import re

def norm(x):
    if x is None: return None
    if isinstance(x,(int,float)) and float(x).is_integer(): return str(int(x))
    return str(x).strip()

    return ids,used,missing,den,num,so
