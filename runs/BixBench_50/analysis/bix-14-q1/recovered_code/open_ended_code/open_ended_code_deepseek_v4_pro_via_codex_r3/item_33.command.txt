/bin/bash -lc "cd /workspace && python - <<'PY'
import urllib.request
url='https://pmc.ncbi.nlm.nih.gov/articles/PMC10871368/'
req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
data=urllib.request.urlopen(req, timeout=30).read().decode('utf-8','ignore')
# search for supplement hrefs
import re
for pat in ['supplement','xlsx','pdf','media']:
 print('\\nPAT',pat)
 for m in re.finditer(r'href=\"(["'^"]*(?:'"'+pat+r')["'^"]*)"'"', data, re.I):
  print(m.group(1))
# maybe print links around Table S5
for m in re.finditer(r'Table S5', data):
 print('table text pos',m.start())
 print(data[m.start()-300:m.start()+500])
PY"
