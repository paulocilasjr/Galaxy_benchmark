import pandas as pd
import os
import re

var_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'

# Define coding Sequence Ontology terms
# These are variants that affect the coding sequence (CDS)
coding_so_terms = {
    'synonymous_variant',
    'missense_variant',
    'inframe_insertion',
    'inframe_deletion',
    'frameshift_variant',
    'stop_gained',
    'stop_lost',
    'start_lost',
    'coding_sequence_variant',
    '5_prime_UTR_premature_start_codon_gain_variant',
}

# First, let's see all SO terms across carrier files
trio = pd.read_excel('/workspace/inputs/230215_Trio_Status.xlsx')
blm_carriers = trio[trio['BLM Mutation Status'] == 'Carrier']
carrier_ids = set(blm_carriers['Sample ID'].astype(str).tolist())

all_so_terms = set()
all_results = []

for fname in sorted(os.listdir(var_dir)):
    m = re.search(r'CHIP_(\d+)-([^.]+)\.xlsx$', fname)
    if not m:
        continue
    cid = m.group(1)
    if cid not in carrier_ids:
        continue
    
    fpath = os.path.join(var_dir, fname)
    df = pd.read_excel(fpath, header=None)
    
    # Data starts at row 2 (0-indexed)
    data = df.iloc[2:].copy()
    data.columns = range(data.shape[1])
    
    # col 4: Zygosity, col 9: VAF, col 13: Sequence Ontology
    # Filter to non-Reference (has a VAF value)
    data['VAF_num'] = pd.to_numeric(data[9], errors='coerce')
    
    # Count non-ref variants with VAF < 0.3
    low_vaf = data[(data['VAF_num'].notna()) & (data['VAF_num'] < 0.3)]
    
    for _, row in low_vaf.iterrows():
        all_so_terms.add(row[13])

print("All Sequence Ontology terms in carrier variants with VAF < 0.3:")
for t in sorted(all_so_terms, key=str):
    print(f"  '{t}'")
