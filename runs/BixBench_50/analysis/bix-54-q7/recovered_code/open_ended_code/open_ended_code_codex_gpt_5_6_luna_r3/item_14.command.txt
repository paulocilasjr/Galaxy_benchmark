/bin/bash -lc "Rscript -e '
d <- read.csv(\"inputs/Swarm_2.csv\", stringsAsFactors=FALSE)
m <- d[d"'$StrainNumber == "287_98", ]
parts <- strsplit(m$Ratio, ":", fixed=TRUE)
m$freq287 <- vapply(parts, function(p) {v <- as.numeric(p); v[1]/sum(v)}, numeric(1))
fits <- list(
 quadratic = lm(Area ~ freq287 + I(freq287''^2), data=m),
 cubic = lm(Area ~ freq287 + I(freq287''^2) + I(freq287''^3), data=m),
 natural_spline_df4 = lm(Area ~ splines::ns(freq287, df=4), data=m)
)
score <- sapply(fits, AIC)
best <- fits[[names(which.min(score))]]
lo <- min(m$freq287); hi <- max(m$freq287)
g <- seq(lo, hi, length.out=200001)
p <- as.numeric(predict(best, newdata=data.frame(freq287=g)))
ix <- which.max(p)
# Refine around the best grid point, and also check the full interval with optimize.
a <- max(lo, g[max(1, ix-1)]); b <- min(hi, g[min(length(g), ix+1)])
o <- optimize(function(x) as.numeric(predict(best, newdata=data.frame(freq287=x))), c(a,b), maximum=TRUE, tol=1e-14)
cat("best=", names(which.min(score)), "'"\\n\", sep=\"\")
cat(\"domain=\", format(lo, digits=17), \",\", format(hi, digits=17), \"\\n\", sep=\"\")
cat(\"grid_max=\", format(g[ix], digits=17), \",\", format(p[ix], digits=17), \"\\n\", sep=\"\")
cat(\"optimum_frequency=\", format(o"'$maximum, digits=17), "'"\\n\", sep=\"\")
cat(\"predicted_area=\", format(o"'$objective, digits=17), "'"\\n\", sep=\"\")
print(cbind(AIC=score, R2=sapply(fits, function(f) summary(f)"'$r.squared)))
'"'"
