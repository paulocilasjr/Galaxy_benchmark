/bin/bash -lc "Rscript -e '
d <- read.csv(\"inputs/Swarm_2.csv\", stringsAsFactors=FALSE)
m <- subset(d, StrainNumber == \"287_98\")
sp <- strsplit(m"'$Ratio, ":", fixed=TRUE)
m$freq <- vapply(sp, function(z) as.numeric(z[1]) / sum(as.numeric(z)), numeric(1))
cat("n", nrow(m), "range", min(m$freq), max(m$freq), "'"\\n\")
print(aggregate(Area ~ freq, m, function(x) c(n=length(x), mean=mean(x), sd=sd(x))))
fitq <- lm(Area ~ freq + I(freq"'^2), data=m)
fitc <- lm(Area ~ freq + I(freq''^2) + I(freq''^3), data=m)
fits <- lm(Area ~ splines::ns(freq, df=4), data=m)
mods <- list(quadratic=fitq, cubic=fitc, spline_df4=fits)
metrics <- do.call(rbind, lapply(mods, function(f) c(df=length(coef(f)), sigma=summary(f)$sigma, R2=summary(f)$r.squared, adjR2=summary(f)$adj.r.squared, AIC=AIC(f), BIC=BIC(f), SSE=deviance(f))))
print(metrics)
for (nm in names(mods)) {
  f <- mods[[nm]]
  opt <- optimize(function(x) -as.numeric(predict(f, newdata=data.frame(freq=x))), interval=c(min(m$freq),max(m$freq)), maximum=TRUE)
  cat("OPT", nm, "freq", opt$minimum, "pred", -opt$objective, "'"\\n\")
  print(coef(f))
}
'"
