/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv'); dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio,'"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
fit <- lm(Area~ns(p,df=4), data=dd)
x <- seq(min(dd"'$p),max(dd$p),length.out=100000)
y <- predict(fit,newdata=data.frame(p=x))
i <- which.max(y)
cat('"'max grid',x[i],y[i],'\\n')
# local maxima via diff
dy <- diff(y)
# print peaks if sign changes + to -
print(head(data.frame(x=x[which(dy[-length(dy)]>0 & dy[-1]<0)], y=y[which(dy[-length(dy)]>0 & dy[-1]<0)])))
# values at boundaries
cat('boundaries', y[1], y[length(y)], '\\n')
RS"
