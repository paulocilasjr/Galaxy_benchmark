/bin/bash -c "Rscript - <<'RS'
d <- read.csv('inputs/Swarm_2.csv'); dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio,'"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
for(df in c(4,5,6,7,8)) {
 s <- smooth.spline(dd"'$p, dd$Area, df=df)
 pr <- function(x) predict(s,x)$y
 o <- optimize(pr,range(dd$p),maximum=TRUE,tol=1e-12)
 cat('"'df',df,'maxp',o"'$maximum,'"'maxy',o"'$objective,'"'\\n')
}
# fit loess maybe
for(sp in c(0.5,0.6,0.7,0.75,0.8)) {
 fit <- loess(Area~p,data=dd,span=sp)
 pr <- function(x) predict(fit,newdata=data.frame(p=x))
 o <- optimize(pr,range(dd"'$p),maximum=TRUE,tol=1e-12)
 cat('"'loess',sp,'maxp',o"'$maximum,'"'maxy',o"'$objective,'"'\\n')
}
RS"
