/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
from patsy import dmatrix, cr
import statsmodels.api as sm
try:
    df=pd.read_csv('inputs/Swarm_2.csv')
    dd=df[df.StrainNumber=='287_98'].copy()
    dd['p']=dd.Ratio.str.split(':').apply(lambda z: float(z[0])/sum(map(float,z)))
    y=dd.Area.values; x=dd.p.values
    # fit cr df=4
    X=dmatrix('cr(p, df=4)', dd, return_type='dataframe')
    model=sm.OLS(y,X).fit()
    print(X.shape)
    print('AIC',model.aic,'params',model.params)
    grid=np.linspace(x.min(),x.max(),10001)
    Xg=dmatrix('cr(grid, df=4)', {'grid':grid}, return_type='dataframe')
    pred=model.predict(Xg)
    i=pred.argmax()
    print('max grid p',grid[i],'area',pred[i])
    # optimize exact
    from scipy.optimize import minimize_scalar
    def pr(p):
        Xp=dmatrix('cr(x, df=4)', {'x':[p]}, return_type='dataframe')
        return float(model.predict(Xp)[0])
    res=minimize_scalar(lambda p:-pr(p),bounds=(x.min(),x.max()),method='bounded',options={'xatol':1e-14})
    print('opt',res.x,pr(res.x))
except Exception as e:
    import traceback; traceback.print_exc()
PY"
