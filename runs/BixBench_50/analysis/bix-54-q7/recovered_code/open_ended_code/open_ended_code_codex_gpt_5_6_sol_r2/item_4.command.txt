/bin/bash -lc "Rscript -e '
d0 <- read.csv(\"inputs/Swarm_2.csv\", check.names=FALSE)
d <- subset(d0, StrainNumber == \"287_98\")
d"'$frequency <- vapply(strsplit(d$Ratio, ":", fixed=TRUE), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
library(splines)
mods <- list(quadratic=lm(Area ~ frequency + I(frequency''^2), d), cubic=lm(Area ~ frequency + I(frequency''^2) + I(frequency''^3), d), natural_spline_df4=lm(Area ~ ns(frequency, df=4), d))
print(data.frame(AIC=vapply(mods,AIC,numeric(1)), adjR2=vapply(mods,function(m) summary(m)$adj.r.squared,numeric(1))))
for(nm in names(mods)) { m <- mods[[nm]]; lo <- min(d$frequency); hi <- max(d$frequency); grid <- seq(lo,hi,length.out=10001); im <- which.max(predict(m,data.frame(frequency=grid))); br <- c(grid[max(1,im-2)],grid[min(length(grid),im+2)]); oo <- optimize(function(x)-predict(m,data.frame(frequency=x)),br,tol=1e-12); cat(nm,oo$minimum,-oo$objective,"'"\\n\") }
'"
