/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv')
m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':')
num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2))
m$p <- num/(num+den)
fits <- list(
  quadratic = lm(Area ~ p + I(p''^2), data=m),
  cubic = lm(Area ~ p + I(p''^2) + I(p''^3), data=m),
  spline = lm(Area ~ ns(p, df=4), data=m)
)
aic <- sapply(fits, AIC)
bic <- sapply(fits, BIC)
adjr <- sapply(fits, function(f) summary(f)$adj.r.squared)
cat('"'AIC\\n'); print(aic)
cat('BIC\\n'); print(bic)
cat('adjR2\\n'); print(adjr)
cat('selected by min AIC:', names(which.min(aic)), '\\n')
fit <- fits[[names(which.min(aic))]]
pred <- function(p) predict(fit, newdata=data.frame(p=p))
lo <- min(m"'$p); hi <- max(m$p)
# coarse grid
g <- seq(lo, hi, length.out=200001)
y <- pred(g)
i <- which.max(y)
p0 <- g[i]
# refine by optimize in small interval
o <- optimize(function(p) -pred(p), lower=max(lo, p0-1e-3), upper=min(hi, p0+1e-3), tol=1e-14)
# verify a few values around
xx <- seq(o$minimum-1e-8, o$minimum+1e-8, length.out=11)
yy <- pred(xx)
cat('"'p0',sprintf('%.17g',p0),'y0',sprintf('%.17g',y[i]),'\\n')
cat('optimized p',sprintf('%.17g',o"'$minimum),'"'area',sprintf('%.17g',-o"'$objective),'"'\\n')
print(data.frame(p=xx, area=yy), digits=17)
RS"
