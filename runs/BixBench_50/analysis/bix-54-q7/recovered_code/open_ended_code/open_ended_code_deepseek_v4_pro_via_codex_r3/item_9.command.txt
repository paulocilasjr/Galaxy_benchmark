/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv')
m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':')
num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2)); m$p <- num/(num+den)
fit <- lm(Area ~ ns(p, df=4), data=m)
o <- optimize(function(p) -predict(fit,newdata=data.frame(p=p)), lower=min(m$p), upper=max(m$p), tol=.Machine$double.eps''^0.25)
cat(sprintf('"'p %.17g area %.17g\\n', o"'$minimum, -o$objective))
# print with max digits
print(o$minimum, digits=22)
print(-o$objective, digits=22)
RS'
