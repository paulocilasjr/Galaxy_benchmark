/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv')
m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':')
num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2)); m$p <- num/(num+den)
fit <- lm(Area ~ ns(p, df=4), data=m)
pred <- function(p) predict(fit, newdata=data.frame(p=p))
lo <- min(m$p); hi <- max(m$p)
for (n in c(1e5,1e6,1e7,5e7)) {
  x <- seq(lo,hi,length.out=n)
  y <- pred(x)
  i <- which.max(y)
  cat('"'n',n,'maxp',sprintf('%.17g',x[i]),'maxy',sprintf('%.17g',y[i]),'\\n')
}
# print around max from coarse
x <- seq(.899,.905,length.out=10001); y<-pred(x); i<-which.max(y)
print(data.frame(p=x[pmax(1,i-5):pmin(length(x),i+5)], area=y[pmax(1,i-5):pmin(length(x),i+5)]), digits=17)
# optimize from many narrow intervals
best <- list(value=-Inf)
for (l in seq(lo, hi-0.001, length.out=500)) {
  o <- optimize(function(p) -pred(p), lower=l, upper=min(hi,l+0.002), tol=1e-14)
  if (-o"'$objective > best$value) best <- list(p=o$minimum, value=-o$objective)
}
cat('"'many optimize', sprintf('p %.17g area %.17g\\n', best"'$p,best$value))
# optimize with larger interval around coarse max
for (l in seq(.899,.905,length.out=100)) {
  o <- optimize(function(p) -pred(p), lower=l, upper=.905, tol=1e-14)
  if (-o$objective > best$value) { best <- list(p=o$minimum, value=-o$objective); cat('"'new',sprintf('%.17g %.17g\\n',best"'$p,best$value)) }
}
cat('"'best overall', sprintf('p %.17g area %.17g\\n',best"'$p,best$value))
RS'
