/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv'); m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':'); num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2)); m$p <- num/(num+den)
# leave-one-out RMSE and maybe leave one replicate group out
forms <- list(quadratic=Area~p+I(p''^2), cubic=Area~p+I(p''^2)+I(p''^3), spline=Area~ns(p,df=4))
loo <- sapply(forms, function(f) {
  pred <- numeric(nrow(m))
  for (i in 1:nrow(m)) {
    fi <- lm(f, data=m[-i,])
    pred[i] <- predict(fi, newdata=m[i,])
  }
  sqrt(mean((m$Area-pred)''^2))
})
print(loo)
# group-level leave-one-ratio out
grp <- unique(m$p)
gloo <- sapply(forms, function(f) {
  pred <- numeric(nrow(m))
  for (g in grp) {
    tr <- m[m$p != g,]; te <- m[m$p == g,]
    fi <- lm(f, data=tr)
    pred[m$p == g] <- predict(fi, newdata=te)
  }
  sqrt(mean((m$Area-pred)''^2))
})
print(gloo)
RS'
