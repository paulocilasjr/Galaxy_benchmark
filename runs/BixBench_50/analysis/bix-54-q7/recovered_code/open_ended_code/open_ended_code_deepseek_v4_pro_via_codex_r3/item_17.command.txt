/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv'); m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':'); num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2)); m$p <- num/(num+den)
agg <- aggregate(Area~p, data=m, FUN=mean)
fits <- list(q=lm(Area~p+I(p''^2),agg), c=lm(Area~p+I(p''^2)+I(p''^3),agg), s=lm(Area~ns(p,df=4),agg))
print(sapply(fits,AIC)); print(sapply(fits,function(f) summary(f)$adj.r.squared))
fit <- fits$s
pred <- function(p) predict(fit,newdata=data.frame(p=p))
g <- seq(min(agg$p),max(agg$p),length.out=1e6); y<-pred(g); i<-which.max(y)
cat('"'aggregated max',format(y[i],digits=16),'at',g[i],'\\n')
RS"
