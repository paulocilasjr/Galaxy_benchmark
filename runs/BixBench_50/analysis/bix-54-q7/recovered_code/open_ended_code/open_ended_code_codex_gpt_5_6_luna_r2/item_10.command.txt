/bin/bash -lc "Rscript --vanilla - <<'RS'
suppressPackageStartupMessages(library(splines))
d <- read.csv('inputs/Swarm_2.csv', stringsAsFactors = FALSE)
d <- d[d"'$StrainNumber == '"'287_98', , drop = FALSE]
parts <- do.call(rbind, strsplit(d"'$Ratio, '"':', fixed = TRUE))
d"'$freq287 <- as.numeric(parts[, 1]) / (as.numeric(parts[, 1]) + as.numeric(parts[, 2]))
fits <- list(
  quadratic = lm(Area ~ poly(freq287, 2, raw = TRUE), data = d),
  cubic = lm(Area ~ poly(freq287, 3, raw = TRUE), data = d),
  natural_spline_df4 = lm(Area ~ ns(freq287, df = 4), data = d)
)
aics <- vapply(fits, AIC, numeric(1))
best_name <- names(which.min(aics))
best <- fits[[best_name]]
predict_area <- function(x) as.numeric(predict(best, newdata = data.frame(freq287 = x)))
grid <- seq(min(d$freq287), max(d$freq287), length.out = 1000001)
grid_pred <- predict(best, newdata = data.frame(freq287 = grid))
i <- which.max(grid_pred)
lo <- if (i == 1) grid[i] else grid[i - 1]
hi <- if (i == length(grid)) grid[i] else grid[i + 1]
opt <- optimize(predict_area, c(lo, hi), maximum = TRUE)
cat('"'best=', best_name, '\\n', sep = '')
cat('AIC=', format(aics[best_name], digits = 15), '\\n', sep = '')
cat('frequency=', format(opt"'$maximum, digits = 15), '"'\\n', sep = '')
cat('maximum_area=', format(opt"'$objective, digits = 15), '"'\\n', sep = '')
cat('answer=', sprintf('%.2f', opt"'$objective), '"'\\n', sep = '')
RS"
