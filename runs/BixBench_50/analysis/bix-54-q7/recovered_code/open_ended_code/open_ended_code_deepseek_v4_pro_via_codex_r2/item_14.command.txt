/bin/bash -lc "cat > /tmp/optimize_swarm.R <<'EOF'
suppressPackageStartupMessages(library(splines))
dat <- read.csv('/workspace/inputs/Swarm_2.csv', stringsAsFactors=FALSE)
dat <- dat["'!(dat$StrainNumber %in% c('"'1','98')),]
parts <- strsplit(as.character(dat"'$Ratio), '"':', fixed=TRUE)
dat"'$x <- as.numeric(sapply(parts, `[`, 1))/(as.numeric(sapply(parts, `[`, 1))+as.numeric(sapply(parts, `[`, 2)))
fit <- lm(Area ~ ns(x, df=4), data=dat)
lo <- min(dat$x); hi <- max(dat$x)
f <- function(x) predict(fit, newdata=data.frame(x=x))
o <- optimize(f, lower=lo, upper=hi, maximum=TRUE)
cat(sprintf('"'optimize peak x=%.15f y=%.15f\\n', o"'$maximum, o$objective))
g <- seq(lo,hi,length.out=1000); pred <- f(g); i<-which.max(pred)
cat(sprintf('"'grid1000 peak x=%.15f y=%.15f\\n', g[i], pred[i]))
g2 <- seq(lo,hi,length.out=1000000); pred2 <- f(g2); i2<-which.max(pred2)
cat(sprintf('grid1e6 peak x=%.15f y=%.15f\\n', g2[i2], pred2[i2]))
EOF
Rscript /tmp/optimize_swarm.R"
