/bin/bash -lc "Rscript - <<'RS'
d <- read.csv('inputs/Swarm_2.csv', stringsAsFactors=FALSE)
d <- d[d"'$StrainNumber == '"'287_98', , drop=FALSE]
d"'$freq287 <- vapply(strsplit(d$Ratio, '"':', fixed=TRUE), function(z) {
  q <- as.numeric(z)
  q[1] / sum(q)
}, numeric(1))
models <- list(
  quadratic = lm(Area ~ poly(freq287, 2, raw=TRUE), data=d),
  cubic = lm(Area ~ poly(freq287, 3, raw=TRUE), data=d),
  natural_spline_df4 = lm(Area ~ splines::ns(freq287, df=4), data=d)
)
fit_table <- data.frame(
  model=names(models),
  AIC=vapply(models, AIC, numeric(1)),
  BIC=vapply(models, BIC, numeric(1)),
  adj_R2=vapply(models, function(m) summary(m)"'$adj.r.squared, numeric(1))
)
print(fit_table, row.names=FALSE)
best <- models[[which.min(fit_table$AIC)]]
objective <- function(x) -as.numeric(predict(best, newdata=data.frame(freq287=x)))
opt <- optimize(objective, interval=range(d$freq287), tol=.Machine$double.eps''^0.25)
endpoints <- range(d$freq287)
vals <- c(-opt$objective, -objective(endpoints))
ans <- max(vals)
cat(sprintf('"'best=%s\\noptimal_freq=%.15f\\nmax_predicted_area=%.15f\\n', fit_table"'$model[which.min(fit_table$AIC)], opt$minimum, ans))
RS'
