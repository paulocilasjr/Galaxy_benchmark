/bin/bash -lc "python - <<'PY'
import urllib.request, json
for channel in ['bioconda','conda-forge']:
    url=f'https://api.anaconda.org/package/{channel}/busco'
    try:
        with urllib.request.urlopen(url, timeout=30) as r:
            data=json.load(r)
        print(channel)
        for f in data.get('files',[])[:20]:
            print(f.get('version'), f.get('attrs',{}).get('subdir'), f.get('download_url'), f.get('dependencies'))
    except Exception as e: print(channel,e)
PY"
