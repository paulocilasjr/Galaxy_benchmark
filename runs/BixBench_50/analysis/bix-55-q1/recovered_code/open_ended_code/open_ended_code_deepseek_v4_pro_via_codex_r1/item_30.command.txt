/bin/bash -lc "python - <<'PY'
import urllib.request, json
for pkg in ['hmmer','metaeuk','miniprot','bbmap','blast','biopython','pandas']:
 url=f'https://api.anaconda.org/package/bioconda/{pkg}'
 try:
  with urllib.request.urlopen(url,timeout=20) as r:d=json.load(r)
  print('\\n',pkg,'latest',d.get('latest_version'))
  for f in d.get('files',[]):
   sub=f.get('attrs',{}).get('subdir','')
   if sub=='linux-aarch64' or pkg in ['biopython','pandas'] and sub=='noarch':
    print(f.get('version'),f.get('basename'),sub,f.get('download_url'))
 except Exception as e:print(pkg,e)
PY"
