from pathlib import Path
import csv

    dup={busco for busco,status in rows if status=='Duplicated'}
    inter_complete_single = complete - dup
    counts={}
    from collections import Counter
    c=Counter(status for busco,status in rows)
    print(sp, 'rows',len(rows), 'status counts', dict(c), 'Complete IDs',len(complete), 'single via Complete-Dup',len(inter_complete_single))
    sets[sp]=complete
inter=set.intersection(*sets.values())
print('intersection size all complete single', len(inter))
print('sorted first/last', sorted(inter)[:10], sorted(inter)[-10:])
# summary counts
for sp in species:
    txt=(base/sp/'run_eukaryota_odb10'/'short_summary.txt').read_text()
