/bin/bash -lc "PYTHONPATH=/workspace/tmp/pyhmmer python3 - <<'PY'
from pyhmmer import easel, plan7, hmmer
alpha=easel.Alphabet.amino()
with plan7.HMMFile('/workspace/tmp/busco_downloads/lineages/eukaryota_odb10/hmms/261328at2759.hmm') as hf: hmm=next(hf)
with easel.SequenceFile('/workspace/inputs/Fungi_Scer.faa', digital=True, alphabet=alpha) as sf: seqs=sf.read_block()
hits=next(hmmer.hmmsearch(hmm, seqs, cpus=1))
print(type(hits), len(hits)); print([a for a in dir(hits) if not a.startswith('_')][:100])
print('query?', getattr(hits,'query',None), getattr(hits,'query_name',None))
for h in hits[:3]:
 print('hit attrs', [a for a in dir(h) if not a.startswith('_')][:80])
 print('hit', h.name, h.accession, h.score, h.evalue, h.bias, h.included, h.reported, len(h.domains))
 d=h.domains[0]
 print('domain attrs', [a for a in dir(d) if not a.startswith('_')][:120])
 print('domain', d.score, d.bias, d.c_evalue, d.i_evalue, d.included, d.reported)
 ali=d.alignment
 print('ali attrs', [a for a in dir(ali) if not a.startswith('_')][:120])
 print('coords', ali.hmm_from, ali.hmm_to, ali.target_from, ali.target_to, ali.hmm_name, ali.target_name)
 break
PY"
