print('domain attrs',d.score,d.bias,d.c_evalue,d.i_evalue,d.env_from,d.env_to,d.included,d.reported)

for attr in ['hmm_from','hmm_to','target_from','target_to','sequence','hmm_sequence','target_sequence','identity','accuracy','posterior_probabilities']:

print(inspect.signature(res.write)); print(inspect.getdoc(res.write)[:3000])
