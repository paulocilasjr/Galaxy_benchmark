/bin/bash -lc "PYTHONPATH=/workspace/tmp/pydeps python - <<'PY'
from pyhmmer import easel,plan7,hmmer
h='/workspace/tmp/busco_downloads/lineages/eukaryota_odb10/hmms/160593at2759.hmm'; f='/workspace/inputs/Fungi_Scer.faa'
with plan7.HMMFile(h) as hf: q=hf.read()
with easel.SequenceFile(f,digital=True,alphabet=q.alphabet) as sf: hits=next(hmmer.hmmsearch(q,sf,cpus=1))
with open('/workspace/tmp/test.domtblout','wb') as out: hits.write(out,'domains',header=True)
PY
head -10 /workspace/tmp/test.domtblout; printf '%s\\n' '--- lines ---'; wc -l /workspace/tmp/test.domtblout"
