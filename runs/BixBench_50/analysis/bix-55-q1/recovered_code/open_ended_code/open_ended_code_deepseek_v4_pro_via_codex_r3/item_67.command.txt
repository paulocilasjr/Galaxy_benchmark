/bin/bash -c "python - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
p=Path('/workspace/busco_data/lineages/eukaryota_odb10/hmms/1001705at2759.hmm')
with plan7.HMMFile(str(p)) as hf: q=hf.read(); q=q.to_profile()
# emit a sequence from HMM as query target
text=q.consensus
seq=easel.TextSequence(name=b'test', sequence=text)
seq=seq.digitize(q.alphabet)
for E in [100000, 10000, 1000, 100, 10, 1, .1]:
    hits=next(hmmer.hmmsearch(q, [seq], E=E))
    n=sum(1 for _ in hits)
    print(E,n)
    for h in hits:
        print(h.name,h.score,h.evalue,len(h.domains))
        break
PY"
