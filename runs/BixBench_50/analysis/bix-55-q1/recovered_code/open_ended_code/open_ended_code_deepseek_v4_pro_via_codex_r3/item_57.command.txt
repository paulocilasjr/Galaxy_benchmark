import zipfile
zip_path='/workspace/inputs/busco_downloads.zip'
out='/workspace/busco_data'

            z.extract(n,out)
print('done')
PY
find /workspace/busco_data/lineages/eukaryota_odb10 -maxdepth 1 -type f -printf '%f\\n' | sort; ls /workspace/busco_data/lineages/eukaryota_odb10/hmms | wc -l"
