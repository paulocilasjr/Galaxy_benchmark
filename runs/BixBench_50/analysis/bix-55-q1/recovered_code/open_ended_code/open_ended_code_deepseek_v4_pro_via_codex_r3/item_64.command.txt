/bin/bash -c "python - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
import time, os
lineage=Path('/workspace/busco_data/lineages/eukaryota_odb10')
seq_path=Path('/workspace/inputs/Fungi_Scer.faa')
with easel.SequenceFile(str(seq_path), digital=True) as sf:
    seqs=list(sf)
print('seqs',len(seqs))
hmm_files=sorted((lineage/'hmms').glob('*.hmm'))
for hfpath in hmm_files[:20]:
    with plan7.HMMFile(str(hfpath)) as hf:
        q=hf.read(); q=q.to_profile()
    total=0
    best=None
    for hits in hmmer.hmmsearch([q], seqs, cpus=1):
        n=sum(1 for _ in hits)
        total=n
        for hit in hits:
            best=hit; break
        break
    print(hfpath.stem, 'hits', total, 'best', best.name if best else None, best.score if best else None)
PY"
