/bin/bash -c "python - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
p=Path('/workspace/busco_data/lineages/eukaryota_odb10/hmms/1001705at2759.hmm')
with plan7.HMMFile(str(p)) as hf: q=hf.read()
print(type(q),q.M,q.alphabet)
seq=easel.TextSequence(name=b'test', sequence='NLELLDPFQEYPEELEETLA')
seq=seq.digitize(q.alphabet)
print(seq)
hits=next(hmmer.hmmsearch(q, [seq], E=100000))
print('n',sum(1 for _ in hits))
for h in hits:
 print(h.name,h.score,h.evalue,len(h.domains),h.domains[0].score if h.domains else None)
 break
# try HMMFile as queries and SequenceFile
from tempfile import NamedTemporaryFile
with NamedTemporaryFile(mode='w',suffix='.faa') as f:
 f.write('>test\\nNLELLDPFQEYPEELEETLA\\n'); f.flush()
 with easel.SequenceFile(f.name, digital=True) as sf:
  seqs=list(sf)
  print('seqs',seqs)
  hits=next(hmmer.hmmsearch(q, seqs, E=100000))
  print('n2',sum(1 for _ in hits))
  for h in hits:
   print(h.name,h.score,h.evalue,len(h.domains))
   break
PY"
