# Check all BUSCO outputs
for sp in Cele Ggal Ncra Scer; do
  echo "=== $sp ==="
  if [ "$sp" = "Scer" ]; then
    tsv="/tmp/busco_run_scer/busco_scer/run_eukaryota_odb10/full_table.tsv"
  else
    tsv="/tmp/busco_run_${sp}/busco_${sp}/run_eukaryota_odb10/full_table.tsv"
  fi
  if [ -f "$tsv" ]; then
    cat "$tsv" | grep -v "^#" | awk -F'\t' '{print $2}' | sort | uniq -c
  else
    echo "File not found. Check log:"
    tail -5 /tmp/busco_${sp}.log 2>/dev/null
  fi
  echo ""
done
