/bin/bash -lc "find /workspace/busco_runs -name full_table.tsv -printf '%p\\n' | sort && python - <<'PY'
from pathlib import Path
root=Path('/workspace/busco_runs')
sets={}
for run in sorted(root.iterdir()):
    table=next(run.rglob('full_table.tsv'))
    statuses={}
    for line in table.read_text().splitlines():
        if line.startswith('#') or not line.strip():
            continue
        fields=line.split('\\t')
        statuses[fields[0]]=fields[1]
    single={bid for bid,status in statuses.items() if status=='Complete'}
    sets[run.name]=single
    print(run.name, 'rows',len(statuses),'complete_single',len(single),'status_counts', {s:sum(v==s for v in statuses.values()) for s in sorted(set(statuses.values()))})
common=set.intersection(*sets.values())
print('intersection',len(common))
print('common_ids', ' '.join(sorted(common)))
PY"
