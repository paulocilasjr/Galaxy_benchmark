import pyhmmer

# Test HMM search with DigitalSequenceBlock
with open('busco_downloads/lineages/eukaryota_odb10/hmms/1001705at2759.hmm', 'rb') as f:
    hmm_file = pyhmmer.plan7.HMMFile(f)
    hmm = next(hmm_file)

# Load sequences into DigitalSequenceBlock
alphabet = pyhmmer.easel.Alphabet.amino()
seq_block = pyhmmer.easel.DigitalSequenceBlock(alphabet)
with open('inputs/Animals_Cele.faa', 'rb') as f:
    with pyhmmer.easel.SequenceFile(f, format='fasta') as sf:
        for seq in sf:
            seq_block.append(seq.digitize(alphabet))

print(f'Loaded {len(seq_block)} sequences')

# Run search
pipeline = pyhmmer.plan7.Pipeline(
    alphabet, 
    Z=len(seq_block), 
    domZ=len(seq_block),
    bit_cutoffs=None,
    incE=100.0,
    incdomE=100.0,
)

count = 0
for top_hits in pipeline.search_hmm(hmm, seq_block):
    for hit in top_hits:
        if hit.included:
            best_score = max(d.score for d in hit.domains)
            print(f'  Hit: {hit.name.decode()}, best_score={best_score:.1f}')
            count += 1
print(f'Total included hits: {count}')
