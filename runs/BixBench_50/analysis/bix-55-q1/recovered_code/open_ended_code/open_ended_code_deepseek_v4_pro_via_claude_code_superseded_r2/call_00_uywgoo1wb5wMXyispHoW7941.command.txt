# Load scores
with zipfile.ZipFile('inputs/busco_downloads.zip') as z:

        parts = line.strip().split('\t')
        scores[parts[0]] = float(parts[1])

# Ribosomal proteins, tubulins, actin, etc.
test_buscos = [
    '261328at2759',  # Check one
    '293315at2759',  # Check another
    '396755at2759',
]
for bid in test_buscos:
    print(f'{bid}: threshold={scores.get(bid, \"N/A\")}')
