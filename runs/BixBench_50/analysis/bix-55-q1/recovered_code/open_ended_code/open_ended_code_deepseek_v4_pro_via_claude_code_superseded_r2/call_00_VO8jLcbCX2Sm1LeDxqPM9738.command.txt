import pyhmmer

# Test HMM search with OptimizedProfile
with open('busco_downloads/lineages/eukaryota_odb10/hmms/1001705at2759.hmm', 'rb') as f:
    hmm_file = pyhmmer.plan7.HMMFile(f)
    hmm = next(hmm_file)
    print(f'HMM: {hmm.name}')
    profile = hmm.to_profile()
    print(f'Profile: {profile}')
    opt_profile = profile.optimized()
    print(f'Optimized: {opt_profile}')

# Load sequences
sequences = []
with open('inputs/Animals_Cele.faa', 'rb') as f:
    with pyhmmer.easel.SequenceFile(f, format='fasta') as sf:
        for seq in sf:
            sequences.append(seq.digitize(pyhmmer.easel.Alphabet.amino()))

print(f'Loaded {len(sequences)} sequences')

# Run search
alphabet = pyhmmer.easel.Alphabet.amino()
pipeline = pyhmmer.plan7.Pipeline(
    alphabet, 
    Z=len(sequences), 
    domZ=len(sequences),
    bit_cutoffs=None,
    incE=100.0,
    incdomE=100.0,
)

count = 0
for top_hits in pipeline.search_hmm(hmm, sequences):
    for hit in top_hits:
        if hit.included:
            best_score = max(d.score for d in hit.domains)
            print(f'  Hit: {hit.name.decode()}, best_score={best_score:.1f}')
            count += 1
print(f'Total included hits: {count}')
