import time
import pyhmmer

# Test if we can use HMMPressedFile for faster searches
# Load all sequences once
alphabet = pyhmmer.easel.Alphabet.amino()
seq_block = pyhmmer.easel.DigitalSequenceBlock(alphabet)
with open('inputs/Animals_Cele.faa', 'rb') as f:
    with pyhmmer.easel.SequenceFile(f, format='fasta') as sf:
        for seq in sf:
            seq_block.append(seq.digitize(alphabet))

print(f'Cele: {len(seq_block)} sequences loaded')

# Run all HMMs sequentially - benchmark 5 HMMs
pipeline = pyhmmer.plan7.Pipeline(
    alphabet, Z=len(seq_block), domZ=len(seq_block),
    bit_cutoffs=None, incE=100.0, incdomE=100.0,
)

t0 = time.time()
count = 0
with open('eukaryota_odb10_all.hmm', 'rb') as f:
    hmm_file = pyhmmer.plan7.HMMFile(f)
    for hmm in hmm_file:
        top_hits = pipeline.search_hmm(hmm, seq_block)
        count += 1
        if count >= 5:
            break

elapsed = time.time() - t0
print(f'5 searches took {elapsed:.2f}s, ~{elapsed/5:.2f}s each')
print(f'Estimated for 255*4=1020: {elapsed/5 * 1020 / 60:.1f} min')
