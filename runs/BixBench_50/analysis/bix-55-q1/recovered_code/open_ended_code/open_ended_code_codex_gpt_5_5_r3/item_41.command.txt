/bin/bash -lc "python3 - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
lineage=Path('tmp/busco_downloads/lineages/eukaryota_odb10')
hmm_path=next((lineage/'hmms').glob('*.hmm'))
print('hmm',hmm_path)
alpha=easel.Alphabet.amino()
with plan7.HMMFile(str(hmm_path)) as hf:
    h=next(hf)
print('hmm name', h.name, 'M', h.M)
with easel.SequenceFile('inputs/Fungi_Scer.faa', digital=True, alphabet=alpha) as sf:
    seqs=sf.read_block()
print('seqs', len(seqs), seqs[0].name, seqs[0].accession, seqs[0].description)
hits=next(hmmer.hmmsearch(h, seqs, cpus=1))
print('hits len', len(hits), 'query', hits.query_name)
for hit in hits[:3]:
    print('hit', hit.name, 'score', hit.score, 'e', hit.evalue, 'domains', len(hit.domains), 'inc', hit.included, 'reported', hit.reported)
    for d in hit.domains[:5]:
        print(' dom', d.score, d.i_evalue, d.c_evalue, d.env_from, d.env_to, d.alignment.hmm_from, d.alignment.hmm_to, d.alignment.target_from, d.alignment.target_to)
PY"
