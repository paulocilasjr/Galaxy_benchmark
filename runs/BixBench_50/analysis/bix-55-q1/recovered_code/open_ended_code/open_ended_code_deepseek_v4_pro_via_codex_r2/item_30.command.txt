/bin/bash -c "python - <<'PY'
import urllib.request, tarfile, os
url='https://api.anaconda.org/download/bioconda/hmmer/3.4/linux-aarch64/hmmer-3.4-h7021222_2.tar.bz2'
out='/workspace/hmmer.tar.bz2'
print('downloading', url)
urllib.request.urlretrieve(url,out)
print('size',os.path.getsize(out))
os.makedirs('/workspace/tools',exist_ok=True)
with tarfile.open(out,'r:bz2') as t:
    members=[m for m in t.getmembers() if '/bin/' in m.name or m.name.startswith('bin/')]
    for m in members:
        print(m.name)
    t.extractall('/workspace/tools', members=members)
print('done')
PY
find /workspace/tools -maxdepth 3 -type f | head -30"
