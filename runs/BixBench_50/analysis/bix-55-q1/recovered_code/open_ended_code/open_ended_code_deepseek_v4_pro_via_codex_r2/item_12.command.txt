/bin/bash -c "python - <<'PY'
import json, urllib.request
for name in ['metaeuk','hmmer','augustus','blast','prodigal','bbmap','miniprot']:
 url=f'https://api.anaconda.org/package/bioconda/{name}'
 try:
  data=json.load(urllib.request.urlopen(url))
  print(name, 'platforms', data.get('platforms'), 'latest', data.get('latest_version'))
 except Exception as e: print(name,e)
PY"
