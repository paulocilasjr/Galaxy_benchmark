/bin/bash -c "python - <<'PY'
from pathlib import Path
runs=['Scer','Ncra','Cele','Ggal']
sets={}
counts={}
for name in runs:
    p=Path('/workspace/busco_runs')/name/'run_eukaryota_odb10'/'full_table.tsv'
    s=set()
    with p.open() as f:
        for line in f:
            if line.startswith('#'): continue
            parts=line.rstrip('\\n').split('\\t')
            if len(parts)<2: continue
            if parts[1]=='Complete':
                s.add(parts[0])
    sets[name]=s
    counts[name]=len(s)
print(counts)
inter=set.intersection(*sets.values())
print('intersection',len(inter))
print(sorted(inter)[:20])
# Also status complete or duplicated? complete any? duplicates union? 
for label, func in [('Complete only', lambda x: x=='Complete'), ('Complete or Duplicated', lambda x: x in ('Complete','Duplicated'))]:
    ss={}
    for name in runs:
        p=Path('/workspace/busco_runs')/name/'run_eukaryota_odb10'/'full_table.tsv'
        s=set()
        with p.open() as f:
            for line in f:
                if line.startswith('#'): continue
                parts=line.rstrip('\\n').split('\\t')
                if len(parts)<2: continue
                if func(parts[1]): s.add(parts[0])
        ss[name]=s
    print(label, len(set.intersection(*ss.values())))
PY"
