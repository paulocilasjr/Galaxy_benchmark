/bin/bash -c "python - <<'PY'
import json, urllib.request
for ver in ['5.4.7','5.8.3','6.0.0']:
 url=f'https://api.anaconda.org/release/bioconda/busco/{ver}'
 try:
  rel=json.load(urllib.request.urlopen(url))
  print('---',ver)
  for f in rel.get('distributions',[]):
   a=f.get('attrs',{})
   if a.get('subdir') in ('noarch','linux-aarch64'):
    print(a.get('subdir'), f.get('basename'))
    print(a.get('depends'))
 except Exception as e: print(e)
PY"
