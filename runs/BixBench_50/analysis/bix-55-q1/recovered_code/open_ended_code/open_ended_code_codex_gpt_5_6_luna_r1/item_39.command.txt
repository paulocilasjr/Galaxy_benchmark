/bin/bash -lc "python - <<'PY'
from pathlib import Path
base=Path('/workspace/tmp/busco_runs')
sets={}
for name in ['cele','ggal','ncra','scer']:
    t=base/name/'run_eukaryota_odb10'/'full_table.tsv'
    ids=set()
    rows=0
    for line in t.read_text().splitlines():
        if not line or line.startswith('#') or line.startswith('Busco id'):
            continue
        p=line.split('\\t')
        rows+=1
        if p[1]=='Complete': ids.add(p[0])
    d=base/name/'run_eukaryota_odb10'/'busco_sequences'/'single_copy_busco_sequences'
    dirids={p.stem for p in d.glob('*.faa')}
    sets[name]=ids
    print(name,'rows',rows,'Complete',len(ids),'single-copy-dir',len(dirids),'match',ids==dirids,'diff_table_dir',sorted(ids"'^dirids))
common=set.intersection(*sets.values())
print('"'common_complete_single_copy',len(common))
print('common ids', ' '.join(sorted(common)))
# Per-group membership count, useful for consistency but only stdout
from collections import Counter
c=Counter(x for s in sets.values() for x in s)
print('membership histogram', {k:sum(v==k for v in c.values()) for k in range(1,5)})
print('sets sizes', {k:len(v) for k,v in sets.items()})
PY"
