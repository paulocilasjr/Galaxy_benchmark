/bin/bash -lc "python - <<'PY'
from pathlib import Path
names = ['Cele','Ggal','Ncra','Scer']
sets = {}
for name in names:
    p = Path('work/results')/name/'run_eukaryota_odb10'/'full_table.tsv'
    statuses = {}
    singles = set()
    with p.open() as fh:
        for line in fh:
            if line.startswith('#'):
                continue
            cols = line.rstrip('\\n').split('\\t')
            statuses[cols[1]] = statuses.get(cols[1], 0) + 1
            if cols[1] == 'Complete':
                singles.add(cols[0])
    sets[name] = singles
    print(name, statuses, 'unique_single_copy_ids', len(singles))
shared = set.intersection(*(sets[n] for n in names))
print('intersection', len(shared))
print('pairwise_containment_check', {n: len(shared & sets[n]) for n in names})
PY
for n in Cele Ggal Ncra Scer; do rg 'Complete and single-copy BUSCOs|Complete and duplicated BUSCOs' work/results/"'$n/short_summary*.txt; done'
