/bin/bash -lc "python - <<'PY'
from pathlib import Path
samples=['Animals_Cele','Animals_Ggal','Fungi_Ncra','Fungi_Scer']
sets={}
for s in samples:
 p=Path('busco_results')/s/'run_eukaryota_odb10'/'full_table.tsv'
 complete=set()
 statuses={}
 with p.open() as fh:
  for line in fh:
   if line.startswith('#') or not line.strip(): continue
   cols=line.rstrip('\\n').split('\\t')
   bid,status=cols[:2]
   statuses.setdefault(bid,set()).add(status)
   if status=='Complete': complete.add(bid)
 sets[s]=complete
 print(s, 'single_complete_ids',len(complete), 'unique_BUSCO_ids',len(statuses))
common=set.intersection(*sets.values())
print('intersection',len(common))
print('IDs',','.join(sorted(common)))
# memberships by number of sets
from collections import Counter
allids=set.union(*sets.values())
print('membership_counts',dict(sorted(Counter(sum(x in sets[s] for s in samples) for x in allids).items())))
PY"
