mkdir -p /workspace/codex_output && python3 << 'EOF'
import re
import glob

iqtree_files = glob.glob('/tmp/work/fungi_scogs/*.iqtree')
total = len(iqtree_files)
above = 0

for f in iqtree_files:
    with open(f) as fh:
        content = fh.read()
    match = re.search(r'Sum of internal branch lengths: [\d.]+ \(([\d.]+)% of tree length\)', content)
    if match:
        treeness = float(match.group(1)) / 100.0
        if treeness > 0.06:
            above += 1

pct = above / total * 100
print(f"{pct:.2f}%")
EOF
