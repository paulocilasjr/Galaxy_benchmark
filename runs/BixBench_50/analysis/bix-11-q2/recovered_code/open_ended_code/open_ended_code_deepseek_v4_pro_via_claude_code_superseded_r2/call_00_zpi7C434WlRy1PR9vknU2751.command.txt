# Validate by computing treeness directly from Newick trees
# A 4-taxon unrooted tree has exactly one internal branch

def parse_newick(newick_str):
    """Extract all branch lengths from a Newick string."""
    # Find all branch lengths (numbers after colons)
    pattern = r':(\d+\.?\d*(?:[eE][+-]?\d+)?)'
    return [float(x) for x in re.findall(pattern, newick_str)]
