/bin/bash -lc "sed -n '1,240p' inputs_manifest.json
sed -n '1,120p' task.json
python - <<'PY'
import zipfile,re,collections
z=zipfile.ZipFile('inputs/scogs_fungi.zip')
files=z.namelist()
def strip(n,suf): return n[:-len(suf)]
fa={strip(n,'.faa') for n in files if n.endswith('.faa')}
trees={strip(n,'.faa.mafft.clipkit.treefile') for n in files if n.endswith('.faa.mafft.clipkit.treefile')}
iq={strip(n,'.faa.mafft.clipkit.iqtree') for n in files if n.endswith('.faa.mafft.clipkit.iqtree')}
print('base set counts',len(fa),len(trees),len(iq),'tree==iq',trees==iq)
print('without trees',sorted(fa-trees))
print('original FAA sequence count distribution')
d=collections.Counter(); missing=[]
for base in sorted(fa):
    txt=z.read(base+'.faa').decode(errors='replace')
    n=sum(line.startswith('>') for line in txt.splitlines())
    d[n]+=1
    if base not in trees: missing.append((base,n,[x for x in txt.splitlines() if x.startswith('>')]))
print(d)
for x in missing: print(x)
PY"
