/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
import re
with ZipFile('/workspace/inputs/scogs_fungi.zip') as z:
  trees=[(n,z.read(n).decode().strip()) for n in z.namelist() if n.endswith('.treefile')]
patterns={}
for n,s in trees:
  if not re.fullmatch(r'[A-Za-z0-9_.,:;()\\-+eE]+',s):
    patterns[n]=s
print('non-simple trees',len(patterns))
if patterns:
  for n,s in list(patterns.items())[:5]: print(n,repr(s))
print('all end semicolon',all(s.endswith(';') for _,s in trees))
print('internal branches',sum(s.count('):') for _,s in trees), 'treefiles',len(trees))
PY"
