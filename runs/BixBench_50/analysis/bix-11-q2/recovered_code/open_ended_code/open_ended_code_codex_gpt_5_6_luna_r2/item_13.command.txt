/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
import re, math
from collections import Counter

def count_leaves(s):
    return len(re.findall(r'(?:(?<=\\()|(?<=,))(["'^():,;]+)(?=[:),;])'"',s))

def branch_lengths(s):
    # all labels and lengths after colon
    return [float(x) for x in re.findall(r':([0-9.eE+-]+)',s)]

with ZipFile('/workspace/inputs/scogs_fungi.zip') as z:
    names=[n for n in z.namelist() if n.endswith('.treefile')]
    print('trees',len(names))
    leafcounts=Counter(); nbranches=Counter(); lens=[]; vals=[]
    for n in names:
      s=z.read(n).decode().strip()
      leaves=count_leaves(s)
      bl=branch_lengths(s)
      leafcounts[leaves]+=1; nbranches[len(bl)]+=1
      # Print some unusual
      vals.append((n, leaves, len(bl), sum(bl), s))
    print('leaf counts',leafcounts)
    print('branches',nbranches)
    print('total length range',min(x[3] for x in vals),max(x[3] for x in vals))
    print('examples min/max')
    for x in sorted(vals,key=lambda q:q[3])[:3]+sorted(vals,key=lambda q:q[3])[-3:]: print(x)
PY"
