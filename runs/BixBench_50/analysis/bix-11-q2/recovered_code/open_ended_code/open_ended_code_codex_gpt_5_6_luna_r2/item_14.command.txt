/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
import re
from collections import Counter

def parse_treeness(s):
    # iq-tree files are unrooted newicks with terminal lengths at leaves and
    # internal lengths on closing parentheses. Explicit root has no length.
    # Every colon length whose preceding non-space char is ')' is internal;
    # those whose preceding token is a taxon are terminal.
    pairs=[]
    for m in re.finditer(r':([0-9.eE+-]+)',s):
        p=s[:m.start()].rstrip()[-1]
        pairs.append((float(m.group(1)), p==')'))
    total=sum(x for x,_ in pairs)
    internal=sum(x for x,inside in pairs if inside)
    return internal/total if total else None, internal, total, len(pairs), sum(1 for _,inside in pairs if inside)

vals=[]
with ZipFile('/workspace/inputs/scogs_fungi.zip') as z:
    for n in z.namelist():
        if n.endswith('.treefile'):
            s=z.read(n).decode().strip()
            vals.append((n,)+parse_treeness(s))
print('n',len(vals))
print('internal count distribution',Counter(x[5] for x in vals))
for thresh in [0.06]:
  above=[x for x in vals if x[1]>thresh]
  print('above',thresh,len(above),'of',len(vals),'percent',len(above)/len(vals)*100)
  print('below/equal',len(vals)-len(above))
print('range',min(x[1] for x in vals),max(x[1] for x in vals))
print('sorted near threshold')
for x in sorted(vals,key=lambda q:q[1]):
    if 0.045<x[1]<0.075: print(x[0], 'treeness=',x[1],'int=',x[2],'total=',x[3],'branches=',x[4],'internal_n=',x[5])
PY"
