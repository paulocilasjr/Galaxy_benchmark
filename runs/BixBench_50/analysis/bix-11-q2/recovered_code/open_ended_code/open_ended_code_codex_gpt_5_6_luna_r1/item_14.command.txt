from zipfile import ZipFile

import re, math

class Node:
    __slots__=('children','length','label')
    def __init__(self, children=None, length=None, label=''):

        self.length=length
        self.label=label

def parse_newick(s):
    s=s.strip()
    if s.endswith(';'): s=s[:-1]
    i=0
    def skip():
        nonlocal i
        while i<len(s) and s[i].isspace(): i+=1
    def label():
        nonlocal i
        skip(); start=i
        while i<len(s) and s[i] not in ',():;[]': i+=1
        return s[start:i].strip()
    def length():
        nonlocal i
        skip()
        if i<len(s) and s[i]==':':
            i+=1; skip(); start=i
            while i<len(s) and s[i] not in ',();[]': i+=1
            return float(s[start:i].strip())
        return None
    def subtree():
        nonlocal i
        skip()
        if i<len(s) and s[i]=='(':
            i+=1; kids=[subtree()]
            while True:
                skip()
                if i<len(s) and s[i]==',': i+=1; kids.append(subtree())
                else: break
            skip()
            assert s[i]==')', (i,s[i:i+20],s)
            i+=1
            lab=label(); bl=length()
            return Node(kids,bl,lab)
        else:
            lab=label(); bl=length()
            return Node([],bl,lab)
    root=subtree(); skip()
    if i "'!= len(s): print('"'trailing',repr(s[i:]))
    return root

def metrics(root):
    vals=[]
    def walk(n, depth=0):
        if n.length is not None: vals.append((n.length, bool(n.children)))
        for c in n.children: walk(c,depth+1)
    walk(root)
    total=sum(x for x,_ in vals)
    internal=sum(x for x,intn in vals if intn)
    ext=sum(x for x,intn in vals if not intn)
