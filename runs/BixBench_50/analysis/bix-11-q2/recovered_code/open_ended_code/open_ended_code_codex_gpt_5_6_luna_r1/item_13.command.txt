/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
for fn in ['inputs/scogs_fungi.zip','inputs/scogs_animals.zip']:
    print('===',fn)
    with ZipFile(fn) as z:
        hits=[]
        for n in z.namelist():
            if n.endswith(('.iqtree','.treefile','.log')):
                try: b=z.read(n)
                except: continue
                for needle in [b'treeness', b'Treeness', b'tree length', b'Tree length']:
                    if needle in b:
                        hits.append((n,needle.decode(), b.lower().find(needle.lower())))
        print(hits[:50], 'total',len(hits))
PY"
