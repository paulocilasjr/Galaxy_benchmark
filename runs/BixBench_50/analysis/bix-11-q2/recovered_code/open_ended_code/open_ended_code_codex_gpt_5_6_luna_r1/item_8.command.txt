/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
from pathlib import PurePosixPath
for fn in ['inputs/scogs_fungi.zip','inputs/scogs_animals.zip','inputs/busco_downloads.zip']:
    print('\\n===',fn,'===')
    with ZipFile(fn) as z:
        infos=z.infolist()
        print('entries',len(infos),'compressed',sum(i.compress_size for i in infos),'uncompressed',sum(i.file_size for i in infos))
        c=Counter()
        for i in infos:
            name=i.filename
            if '/' in name: c[name.rsplit('/',1)[-1].split('.',1)[-1] if '.' in name.rsplit('/',1)[-1] else '<none>'] += 1
            else: c[name.split('.',1)[-1] if '.' in name else '<none>'] += 1
        print('extensions',c)
        if fn.endswith('scogs_fungi.zip'):
            for ext in ['treefile','faa','iqtree','mafft','clipkit']:
                names=[i.filename for i in infos if i.filename.endswith('.'+ext)]
                print(ext,len(names),'examples',names[:5])
            print('tree first',next(i.filename for i in infos if i.filename.endswith('.treefile')))
PY"
