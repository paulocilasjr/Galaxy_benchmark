/bin/bash -lc "python3 - <<'PY'
import zipfile, re, statistics
z='inputs/scogs_fungi.zip'
rows=[]
pattern=re.compile(r'Sum of internal branch lengths:\\s+([0-9.Ee+-]+)\\s+\\(([0-9.Ee+-]+)% of tree length\\)')
with zipfile.ZipFile(z) as fh:
    for n in fh.namelist():
        if n.endswith('.iqtree'):
            text=fh.read(n).decode('utf-8', errors='replace')
            m=pattern.search(text)
            if not m:
                raise SystemExit(f'missing treeness line in {n}')
            internal=float(m.group(1))
            pct=float(m.group(2))
            treeness=pct/100.0
            gid=n.split('.faa')[0]
            rows.append((gid, treeness, pct, internal, n))
rows.sort()
above=[r for r in rows if r[1] > 0.06]
print('n_values', len(rows))
print('n_above', len(above))
print('percent_over_values', len(above)/len(rows)*100)
print('percent_over_all_255', len(above)/255*100)
print('min max mean', min(r[1] for r in rows), max(r[1] for r in rows), statistics.mean(r[1] for r in rows))
print('near_threshold')
for r in sorted(rows, key=lambda r: abs(r[1]-0.06))[:20]:
    print(r[0], r[1], r[2])
PY"
