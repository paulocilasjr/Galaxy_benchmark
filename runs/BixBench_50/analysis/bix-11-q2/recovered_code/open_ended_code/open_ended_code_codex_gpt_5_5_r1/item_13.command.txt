/bin/bash -lc "python3 - <<'PY'
import zipfile, re, collections
z='inputs/scogs_fungi.zip'
with zipfile.ZipFile(z) as fh:
    names=[n for n in fh.namelist() if not n.endswith('/')]
print('members',len(names))
# suffix counts after .faa or .faa.mafft...
suffix=collections.Counter()
ids=collections.defaultdict(set)
for n in names:
    m=re.match(r'(["'^/]+?at2759)'"\\.faa(.*)"'$'"', n)
    if m:
        ids[m.group(1)].add(m.group(2) or '.faa')
        suffix[m.group(2) or '.faa'] += 1
    else:
        suffix['OTHER'] += 1
print('unique ids',len(ids))
for k,v in suffix.most_common(): print(v, repr(k))
print('ids with iqtree', sum(any(s.endswith('.iqtree') for s in ss) for ss in ids.values()))
print('ids with treefile', sum(any(s.endswith('.treefile') for s in ss) for ss in ids.values()))
print('ids with log', sum(any(s.endswith('.log') for s in ss) for ss in ids.values()))
print('ids only no iqtree example', [i for i,ss in ids.items() if not any(s.endswith('.iqtree') for s in ss)][:20])
PY"
