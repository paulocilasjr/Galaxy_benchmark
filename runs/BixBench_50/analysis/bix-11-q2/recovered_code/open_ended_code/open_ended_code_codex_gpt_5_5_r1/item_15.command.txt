/bin/bash -lc "python3 - <<'PY'
import zipfile, re
z='inputs/scogs_fungi.zip'
with zipfile.ZipFile(z) as fh:
    names=fh.namelist()
    ids={re.match(r'(["'^/]+?at2759)'"\\.faa', n).group(1) for n in names if re.match(r'(["'^/]+?at2759)'"\\.faa', n)}
    iq={re.match(r'(["'^/]+?at2759)'"\\.faa', n).group(1) for n in names if n.endswith('.iqtree')}
    missing=sorted(ids-iq)
    print(missing)
    for gid in missing:
        print('\\n===', gid, 'members ===')
        for n in names:
            if n.startswith(gid+'.faa'):
                print(n)
        log=gid+'.faa.mafft.clipkit.log'
        if log in names:
            print('--- log tail ---')
            data=fh.read(log).decode('utf-8', errors='replace').splitlines()
            for line in data[-60:]: print(line)
PY"
