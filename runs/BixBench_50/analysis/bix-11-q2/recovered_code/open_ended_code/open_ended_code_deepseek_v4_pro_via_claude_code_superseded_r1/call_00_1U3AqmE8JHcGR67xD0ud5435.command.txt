import os, re

os.chdir('/workspace/tmp')

# Count all fungal genes
all_faa = set()
for f in os.listdir('.'):
    if f.endswith('.faa') and '.mafft' not in f:
        all_faa.add(f.replace('.faa', ''))

# Extract treeness from IQ-TREE reports  
treeness = {}
for f in os.listdir('.'):
    if f.endswith('.iqtree'):
        gene_id = f.replace('.faa.mafft.clipkit.iqtree', '')
        content = open(f).read()
        match = re.search(r'Sum of internal branch lengths:\s+[\d.]+\s+\(([\d.]+)% of tree length\)', content)
        if match:
            treeness[gene_id] = float(match.group(1)) / 100.0

# Genes with treeness > 0.06
above = sum(1 for v in treeness.values() if v > 0.06)
total_with_trees = len(treeness)
total_all = len(all_faa)

print(f"Fungal genes total: {total_all}")
print(f"Fungal genes with trees: {total_with_trees}")
print(f"Fungal genes without trees: {total_all - total_with_trees}")
print(f"Fungal genes with treeness > 0.06: {above}")
print(f"Percentage (of genes with trees): {100*above/total_with_trees:.2f}%")
print(f"Percentage (of all fungal genes): {100*above/total_all:.2f}%")

# Double-check: what are the treeness values of the 88 genes just above 0.06?
above_vals = sorted([v for v in treeness.values() if v > 0.06])
print(f"\nLowest treeness > 0.06: {above_vals[:5]}")
# What about just below 0.06?
below_vals = sorted([v for v in treeness.values() if v <= 0.06], reverse=True)
print(f"Highest treeness <= 0.06: {below_vals[:5]}")
