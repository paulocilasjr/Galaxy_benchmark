def parse_fasta(filepath):
    seqs = {}
    current_header = None
    current_seq = []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if current_header is not None:
                    seqs[current_header] = ''.join(current_seq)
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            seqs[current_header] = ''.join(current_seq)
    return seqs

def p_distance(seq1, seq2):
    n_diff = 0
    n_compared = 0
    for a, b in zip(seq1, seq2):
        if a == '-' and b == '-':
            continue
        n_compared += 1
        if a != b:
            n_diff += 1
    if n_compared == 0:
        return 0.0
    return n_diff / n_compared

def compute_patristic(tree_str):
    s = tree_str.strip().rstrip(';')
    nodes = []
    i = 0
    
    def parse_node():
        nonlocal i
        node = {'children': [], 'length': 0.0, 'name': None}
        node_id = len(nodes)
        nodes.append(node)
        
        if s[i] == '(':
            i += 1
            while True:
                child_id = parse_node()
                nodes[node_id]['children'].append(child_id)
                if s[i] == ',':
                    i += 1
                elif s[i] == ')':
                    i += 1
                    break
            if i < len(s) and s[i] == ':':
                i += 1
                j = i
                while j < len(s) and s[j] not in ',):;':
                    j += 1
                if j > i:
                    nodes[node_id]['length'] = float(s[i:j])
                    i = j
        else:
            j = i
            while j < len(s) and s[j] not in ':),;':
                j += 1
            nodes[node_id]['name'] = s[i:j]
            i = j
            if i < len(s) and s[i] == ':':
                i += 1
                j = i
                while j < len(s) and s[j] not in ',):;':
                    j += 1
                if j > i:
                    nodes[node_id]['length'] = float(s[i:j])
                    i = j
        
        return node_id
    
    root = parse_node()
    
    parent = {}
    dist_to_root = {}
    
    def dfs(node_id, d):
        dist_to_root[node_id] = d
        for child_id in nodes[node_id]['children']:
            parent[child_id] = node_id
            dfs(child_id, d + nodes[child_id]['length'])
    
    dfs(root, 0)
    parent[root] = None
    
    # Find leaf nodes
    leaf_to_nid = {}
    for nid, node in enumerate(nodes):
        if node['name'] is not None:
            leaf_to_nid[node['name']] = nid
    
    leaf_names = sorted(leaf_to_nid.keys())
    
    # Build ancestor sets for each leaf
    leaf_ancestors = {}
    for name, nid in leaf_to_nid.items():
        ancestors = set()
        p = nid
        while p is not None:
            ancestors.add(p)
            p = parent.get(p)
        leaf_ancestors[name] = ancestors
    
    patristic = {}
    for i in range(len(leaf_names)):
        for j in range(i+1, len(leaf_names)):
            n1, n2 = leaf_names[i], leaf_names[j]
            nid1, nid2 = leaf_to_nid[n1], leaf_to_nid[n2]
            anc1, anc2 = leaf_ancestors[n1], leaf_ancestors[n2]
            common = anc1 & anc2
            lca = max(common, key=lambda x: dist_to_root.get(x, 0))
            d = dist_to_root[nid1] + dist_to_root[nid2] - 2 * dist_to_root[lca]
            patristic[(n1, n2)] = d
    
    return patristic, leaf_names

def parse_mldist(filepath):
    with open(filepath) as f:
        n = int(f.readline().strip())
        order = []
        matrix = []
        for i in range(n):
            parts = f.readline().strip().split()
            order.append(parts[0])
            matrix.append([float(x) for x in parts[1:]])
    return order, matrix

def compute_r2(y, x):
    """R² from linear regression y ~ x with intercept."""
    n = len(y)
    if n < 3:
        return 0.0
    mean_y = sum(y) / n
    mean_x = sum(x) / n
    
    # slope = sum((x_i - mean_x)*(y_i - mean_y)) / sum((x_i - mean_x)^2)
    ss_xx = sum((xi - mean_x)**2 for xi in x)
    ss_xy = sum((xi - mean_x)*(yi - mean_y) for xi, yi in zip(x, y))
    ss_yy = sum((yi - mean_y)**2 for yi in y)
    
    if ss_xx == 0 or ss_yy == 0:
        return 0.0
    
    slope = ss_xy / ss_xx
    intercept = mean_y - slope * mean_x
    ss_res = sum((yi - (slope * xi + intercept))**2 for xi, yi in zip(x, y))
    r2 = 1.0 - ss_res / ss_yy
    return r2
