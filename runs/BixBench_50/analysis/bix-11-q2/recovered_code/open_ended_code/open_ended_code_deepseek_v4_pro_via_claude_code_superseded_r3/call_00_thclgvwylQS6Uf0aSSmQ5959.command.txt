cd /tmp/fungi_all

# Get total genes across ALL orthogroups
total_all=0
for f in *.faa; do
    if [[ "$f" != *"mafft"* ]]; then
        count=$(grep -c "^>" "$f")
        total_all=$((total_all + count))
    fi
done

# Get genes in orthogroups WITH trees
total_with_trees=0
for f in *.treefile; do
    base=$(basename "$f" .faa.mafft.clipkit.treefile)
    faa_file="${base}.faa"
    if [ -f "$faa_file" ]; then
        count=$(grep -c "^>" "$faa_file")
        total_with_trees=$((total_with_trees + count))
    fi
done

# Get genes in orthogroups WITHOUT trees
total_without_trees=0
for f in *.faa; do
    if [[ "$f" != *"mafft"* ]]; then
        base=$(basename "$f" .faa)
        tree_file="${base}.faa.mafft.clipkit.treefile"
        if [ ! -f "$tree_file" ]; then
            count=$(grep -c "^>" "$f")
            total_without_trees=$((total_without_trees + count))
            echo "No tree: $base ($count genes)"
        fi
    done
done

echo ""
echo "Total genes across all 255 orthogroups: $total_all"
echo "Genes in 249 orthogroups WITH trees: $total_with_trees"
echo "Genes in 6 orthogroups WITHOUT trees: $total_without_trees"
echo ""
echo "All 249 orthogroups with trees have treeness > 0.06"
echo "Percentage (of all genes): $total_with_trees / $total_all"
python3 -c "print(f'{100.0 * $total_with_trees / $total_all:.4f}%')"
