for f in /tmp/busco_fungi2/run_eukaryota_odb10/full_table.tsv; do
    sp=$(echo "$f" | sed 's|.*/||')
    # Actually the species info is in the path

# Let me just check each BUSCO table
for sp in Ncra Scer Spom Umay; do
    d=$(find /tmp/busco_fungi2 -name "full_table.tsv" -path "*${sp}*" 2>/dev/null | head -1)
    if [ -z "$d" ]; then
        d="/tmp/busco_fungi2/run_eukaryota_odb10/full_table.tsv"
    fi
    echo "=== $sp ==="
    tail -n +4 "$d" | awk -F'\t' '{print $1, $2}' | sort | head -3
    echo "..."
    tail -n +4 "$d" | awk -F'\t' '{print $2}' | sort | uniq -c
    echo ""
done
