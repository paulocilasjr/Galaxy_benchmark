/bin/bash -lc "python - <<'PY'
import zipfile
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    names=set(z.namelist())
    bases=sorted({n[:-4] for n in names if n.endswith('.faa')})
    missing=[]
    for b in bases:
        tree=b+'.mafft.clipkit.treefile'
        iq=b+'.mafft.clipkit.iqtree'
        if tree not in names or iq not in names:
            missing.append((b, tree in names, iq in names, b+'.mafft.clipkit.log' in names))
    print('faa bases', len(bases))
    print('missing tree/iq', len(missing))
    for row in missing:
        print(*row)
PY"
