/bin/bash -lc "python - <<'PY'
import re, zipfile
from decimal import Decimal, getcontext
getcontext().prec = 40
n = 0
above = 0
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    for name in z.namelist():
        if not name.endswith('.iqtree'):
            continue
        s = z.read(name).decode('utf-8', 'replace')
        a = re.search(r'Sum of internal branch lengths:\\s*([0-9.]+)', s)
        b = re.search(r'Total tree length \\(sum of branch lengths\\):\\s*([0-9.]+)', s)
        if not (a and b):
            raise SystemExit(f'Could not parse {name}')
        n += 1
        above += Decimal(a.group(1)) / Decimal(b.group(1)) > Decimal('0.06')
print(f'{above}/{n} = {Decimal(above)*100/Decimal(n)}%')
PY"
