/bin/bash -lc "python - <<'PY'
import re
import zipfile
from decimal import Decimal
p = 'inputs/scogs_fungi.zip'
vals = []
with zipfile.ZipFile(p) as z:
    names = [n for n in z.namelist() if n.endswith('.iqtree')]
    for n in names:
        s = z.read(n).decode('utf-8', errors='replace')
        m = re.search(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+).*?\\(([0-9.eE+-]+)% of tree length\\)', s)
        t = re.search(r'Total tree length \\(sum of branch lengths\\):\\s*([0-9.eE+-]+)', s)
        if not m:
            print('NO TREEness', n)
            continue
        internal = Decimal(m.group(1))
        pct = Decimal(m.group(2))
        total = Decimal(t.group(1)) if t else None
        vals.append((n, internal, pct, total))
print('reports', len(names), 'parsed', len(vals))
print('above_0.06_using_percent_field', sum(v[2] > Decimal('6') for v in vals))
print('above_0.06_using_ratio_field', sum((v[2]/Decimal(100)) > Decimal('0.06') for v in vals))
print('equal_6_percent', sum(v[2] == Decimal('6') for v in vals))
print('min_pct', min(v[2] for v in vals), 'max_pct', max(v[2] for v in vals))
print('near threshold')
for n, internal, pct, total in sorted(vals, key=lambda x: x[2]):
    if Decimal('4.5') <= pct <= Decimal('7.5'):
        print(n.split('.')[0], 'internal', internal, 'total', total, 'pct', pct)
print('distribution bins')
for lo, hi in [(0,2),(2,4),(4,6),(6,8),(8,10),(10,100)]:
    print(lo, hi, sum(Decimal(str(lo)) < v[2] <= Decimal(str(hi)) for v in vals))
PY"
