/bin/bash -lc 'tmpdir=$(cat /workspace/fungi_trees_dir.txt); TREEDIR="$tmpdir" python - <<'"'PY'
import os, glob, statistics
from phykit.services.tree.treeness import Treeness
files = sorted(glob.glob(os.path.join(os.environ['TREEDIR'], '*.treefile')))
vals=[]
for f in files:
    s=Treeness._scan_simple_newick_summary(f)
    if s is None:
        raise RuntimeError('non-simple '+f)
    _, total, internal = s
    v=float(internal/total) if total else 0.0
    vals.append((os.path.basename(f).split('.faa')[0], v))
print('n', len(vals))
print('above', sum(1 for _,v in vals if v>0.06))
print('pct', sum(1 for _,v in vals if v>0.06)/len(vals)*100)
print('rounded pct', round(sum(1 for _,v in vals if v>0.06)/len(vals)*100, 2))
print('mean median min max', statistics.mean([v for _,v in vals]), statistics.median([v for _,v in vals]), min(v for _,v in vals), max(v for _,v in vals))
print('count >=?', sum(1 for _,v in vals if v>=0.06))
for name,v in vals:
    if abs(v-0.06)<0.01:
        print(name, v)
PY"
