/bin/bash -lc "python3 - <<'PY'
import zipfile,re
z=zipfile.ZipFile('inputs/scogs_fungi.zip')
rows=[]
for n in z.namelist():
    if n.endswith('.iqtree'):
        txt=z.read(n).decode('utf-8','replace')
        m=re.search(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)\\s*\\(([0-9.eE+-]+)% of tree length\\)', txt)
        if m:
            gene=n.split('.')[0]
            rows.append((gene,float(m.group(1)),float(m.group(2))/100))
        else:
            print('missing metric', n)
print('parsed',len(rows))
print('above',sum(v>0.06 for _,_,v in rows))
print('percent',sum(v>0.06 for _,_,v in rows)/len(rows)*100 if rows else None)
print('first5', rows[:5])
PY"
