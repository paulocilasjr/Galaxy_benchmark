/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
ln='inputs/JD_Chromosome_Length.csv'
rows=Counter(); sites=defaultdict(set); pos_labels=defaultdict(set); shared=Counter()
with open(cp, newline='', encoding='utf-8-sig') as f:
    rd=csv.DictReader(f)
    print('fields', rd.fieldnames)
    for x in rd:
        c=x['Chromosome'].strip()
        rows[c]+=1
        sites[c].add((x['StartPosition'],x['EndPosition']))
        pos_labels[c].add(x['Pos'])
        shared[x['SampleShared']]+=1
lengths={}
with open(ln,newline='',encoding='utf-8-sig') as f:
    for x in csv.DictReader(f): lengths[x['Chromosome'].strip()]=int(x['Length'])
print('length_rows', len(lengths), 'labels', list(lengths))
print('data_rows',sum(rows.values()),'unique_coords',sum(map(len,sites.values())),'unique_pos',sum(map(len,pos_labels.values())))
print('sample_shared', shared.most_common())
print('chr\\trows\\tsites\\tlength\\tdensity_per_bp\\tdensity_per_Mb')
result=[]
for c,L in lengths.items():
    n=len(sites[c]); d=n/L
    result.append((d,c,n,L,rows[c]))
for d,c,n,L,r in sorted(result, reverse=True):
    print(f'{c}\\t{r}\\t{n}\\t{L}\\t{d:.12g}\\t{d*1e6:.6f}')
print('data chromosomes absent lengths', sorted(set(rows)-set(lengths)))
PY"
