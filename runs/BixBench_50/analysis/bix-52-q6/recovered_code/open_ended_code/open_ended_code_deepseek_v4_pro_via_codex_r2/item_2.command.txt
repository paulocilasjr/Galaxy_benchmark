import csv, collections
for fn in ['inputs/JD_AgeRelated_CpG_noMT_Final.csv','inputs/ZF_AgeRelated_CpG_noMT_Final.csv']:
    with open(fn, newline='') as f:
        r=csv.reader(f)

        rows=list(r)
    print(fn)
    print('header',header)
    print('n rows',len(rows))
    chromidx=header.index('Chromosome'); posidx=header.index('Pos')
    chroms=[x[chromidx] for x in rows]
    poss=[x[posidx] for x in rows]
    print('unique pos',len(set(poss)),'unique chrom',len(set(chroms)))
    print('chrom sorted', sorted(set(chroms), key=lambda x:(len(x),x))[:50])
    print('chrom counts')
    for c,n in collections.Counter(chroms).most_common(30): print(c,n)
    print('first rows')
    for row in rows[:3]: print(row)
