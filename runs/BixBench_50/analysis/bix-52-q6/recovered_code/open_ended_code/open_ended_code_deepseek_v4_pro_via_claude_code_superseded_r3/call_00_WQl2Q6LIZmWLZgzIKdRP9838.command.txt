import csv
from collections import defaultdict

# Read CpG data - count unique sites per chromosome
cpg_sites = defaultdict(set)
with open("inputs/JD_AgeRelated_CpG_noMT_Final.csv", newline="") as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row["Chromosome"]
        pos = row["Pos"]
        cpg_sites[chrom].add(pos)

# Read chromosome lengths
lengths = {}
with open("inputs/JD_Chromosome_Length.csv", newline="") as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row["Chromosome"].strip()
        if chrom and row["Length"].strip():
            lengths[chrom] = int(row["Length"])

print("Lengths loaded:", len(lengths))

# Compute densities (only for chromosomes present in both datasets)
print("\nChromosome | Sites | Length (bp) | Density (sites/bp)")
print("-" * 65)
max_density = -1
max_chrom = None

for chrom in sorted(cpg_sites.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else 0, x)):
    n_sites = len(cpg_sites[chrom])
    length = lengths.get(chrom, None)
    if length is None:
        print(f"  {chrom:>2}       | {n_sites:>5} | MISSING     | N/A")
        continue
    density = n_sites / length
    print(f"  {chrom:>2}       | {n_sites:>5} | {length:>11} | {density:.10f}")
    if density > max_density:
        max_density = density
        max_chrom = chrom

print(f"\nHighest density: Chromosome {max_chrom} at {max_density:.10f} sites/bp")
print(f"  = {max_density * 1_000_000:.4f} sites per Mb")
