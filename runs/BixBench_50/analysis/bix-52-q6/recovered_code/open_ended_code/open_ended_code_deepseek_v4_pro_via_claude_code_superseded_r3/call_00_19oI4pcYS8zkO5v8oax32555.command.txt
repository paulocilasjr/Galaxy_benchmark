import csv
from collections import defaultdict

# Read CpG data
cpg_sites = defaultdict(set)
with open("inputs/JD_AgeRelated_CpG_noMT_Final.csv", newline="") as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row["Chromosome"]
        pos = row["Pos"]
        cpg_sites[chrom].add(pos)

print("Chromosomes in CpG data:", sorted(cpg_sites.keys()))
for chrom in sorted(cpg_sites.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else 0, x)):
    print(f"  Chromosome {chrom}: {len(cpg_sites[chrom])} unique CpG sites")

# Read chromosome lengths
lengths = {}
with open("inputs/JD_Chromosome_Length.csv", newline="") as f:
    reader = csv.DictReader(f)
    for row in reader:
        lengths[row["Chromosome"]] = int(row["Length"])

print("\nChromosome lengths:", sorted(lengths.keys()))
for chrom in sorted(lengths.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else 0, x)):
    print(f"  Chromosome {chrom}: {lengths[chrom]} bp")
