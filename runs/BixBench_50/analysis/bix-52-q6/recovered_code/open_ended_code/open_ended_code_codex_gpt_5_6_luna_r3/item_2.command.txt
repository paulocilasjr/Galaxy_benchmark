/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
with open(p, newline='') as f:
    rows=list(csv.DictReader(f))
print('rows', len(rows))
print('chromosomes', sorted(set(r['Chromosome'] for r in rows), key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else x)))
for key in ['Pos','StartPosition','EndPosition']:
    vals=[r[key] for r in rows]
    print(key, 'unique', len(set(vals)), 'duplicate_rows', len(vals)-len(set(vals)))
# number observations per Pos, and whether Pos maps 1:1 to chromosome/start
pos_to_pairs=defaultdict(set)
for r in rows: pos_to_pairs[r['Pos']].add((r['Chromosome'],r['StartPosition'],r['EndPosition']))
print('Pos with multiple genomic coordinates', sum(len(v)>1 for v in pos_to_pairs.values()))
print('top rows per Pos', Counter(r['Pos'] for r in rows).most_common(10))
print('Sample unique', len(set(r['Sample'] for r in rows)), sorted(set(r['Sample']))[:10])
print('Dage min max', min(float(r['Dage']) for r in rows), max(float(r['Dage']) for r in rows))
PY"
