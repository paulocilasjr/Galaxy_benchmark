/bin/bash -lc "python3 - <<'PY'
import csv, collections
for name in ['JD','ZF']:
    with open(f'/workspace/inputs/{name}_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
        r=csv.DictReader(f)
        pos_counts=collections.Counter()
        rows=0
        chrom_pos=collections.defaultdict(set)
        chrom_rows=collections.Counter()
        for row in r:
            rows+=1
            chrom=row['Chromosome'].strip()
            pos=row['Pos'].strip()
            chrom_rows[chrom]+=1
            chrom_pos[chrom].add(pos)
    with open(f'/workspace/inputs/{name}_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
        r=csv.DictReader(f)
        lengths={row['Chromosome'].strip(): int(row['Length']) for row in r}
    print(name, 'rows',rows, 'unique Pos', len({p for s in chrom_pos.values() for p in s}), 'chroms', len(chrom_pos))
    print('top row counts by chrom', chrom_rows.most_common(5))
    print('top site counts by chrom', sorted(((len(v),k) for k,v in chrom_pos.items()), reverse=True)[:10])
    print('chrom not in length', set(chrom_pos)-set(lengths))
    print('length chrom not in sites', set(lengths)-set(chrom_pos))
PY"
