# Debug: check the exact keys in both files
import csv

# Check CpG file chromosome values
chroms_cpg = set()
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', 'r') as f:
    reader = csv.DictReader(f)
    for i, row in enumerate(row for row in reader):
        if i < 3:
            print(f"CpG row {i}: keys={list(row.keys())}, Chromosome={repr(row['Chromosome'])}")
        chroms_cpg.add(row['Chromosome'])

print(f"\nCpG chromosomes: {sorted(chroms_cpg, key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else x))}")

# Check length file
chroms_len = set()
with open('inputs/JD_Chromosome_Length.csv', 'r') as f:
    reader = csv.DictReader(f)
    for i, row in enumerate(row for row in reader):
        chrom = row.get('Chromosome', 'MISSING')
        length = row.get('Length', 'MISSING')
        if i < 3:
            print(f"Len row {i}: keys={list(row.keys())}, Chromosome={repr(chrom)}, Length={repr(length)}")
        if chrom.strip():
            chroms_len.add(chrom.strip())

print(f"\nLength chromosomes: {sorted(chroms_len, key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else x))}")

# Check intersection
print(f"\nIntersection: {chroms_cpg & chroms_len}")
print(f"CpG only: {chroms_cpg - chroms_len}")
print(f"Length only: {chroms_len - chroms_cpg}")

# Check byte-by-byte
print("\nByte check of CpG chromosome '1':", [hex(b) for b in list(chroms_cpg)[0].encode('utf-8')])
print("Byte check of Len chromosome '1':", [hex(b) for b in list(chroms_len)[0].encode('utf-8')])
