/bin/bash -c "python3 - <<'PY'
import csv, collections
for fn in ['JD_AgeRelated_CpG_noMT_Final.csv','ZF_AgeRelated_CpG_noMT_Final.csv']:
    with open('/workspace/inputs/'+fn, newline='') as f:
        r=csv.DictReader(f)
        rows=list(r)
    print(fn, 'rows', len(rows))
    print('cols', list(rows[0].keys()))
    poss=set(r['Pos'] for r in rows)
    chrs=set(r['Chromosome'] for r in rows)
    print('unique Pos',len(poss),'unique chr',len(chrs))
    cp=set((r['Chromosome'], r['StartPosition']) for r in rows)
    print('unique (chr,start)',len(cp))
    c=collections.Counter(r['Chromosome'] for r in rows)
    print('chr row counts top', c.most_common(10))
    cp_count=collections.Counter(ch for ch,st in cp)
    print('chr unique-site counts top', cp_count.most_common(10))
PY"
