/bin/bash -lc "echo '--- full JD chromosome lengths ---' && sed -n '1,60p' inputs/JD_Chromosome_Length.csv && echo '--- unique chromosome/row/site counts ---' && python - <<'PY'
import csv
from collections import Counter, defaultdict
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
with open(p,newline='') as f:
    rows=list(csv.DictReader(f))
print('rows',len(rows))
print('fields',list(rows[0]))
print('chromosome values',sorted({r['Chromosome'] for r in rows}, key=lambda x:(len(x),x)))
rowc=Counter(r['Chromosome'] for r in rows)
sitec=defaultdict(set)
for r in rows:
    sitec[r['Chromosome']].add((r['Pos'],r['StartPosition'],r['EndPosition']))
for c in sorted(rowc,key=lambda x:(len(x),x)):
    print(c, 'rows',rowc[c], 'unique_sites',len(sitec[c]), 'unique_Pos',len({r['Pos'] for r in rows if r['Chromosome']==c}), 'samples',len({r['Sample'] for r in rows if r['Chromosome']==c}))
PY"
