/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lp='inputs/JD_Chromosome_Length.csv'
by=defaultdict(set)
with open(cp,newline='',encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        by[r['Chromosome']].add((r['StartPosition'], r['EndPosition']))
with open(lp,newline='',encoding='utf-8-sig') as f:
    length={r['Chromosome']:int(r['Length']) for r in csv.DictReader(f)}
rank=[]
for chrom,sites in by.items():
    d=len(sites)/length[chrom]
    rank.append((d,chrom,len(sites),length[chrom],d*1_000_000))
for d,c,n,L,permb in sorted(rank,reverse=True):
    print(f'{c}\\t{n}\\t{L}\\t{d:.12g}\\t{permb:.6f}')
print('top',max(rank))
print('missing lengths', sorted(set(by)-set(length)))
print('lengths without sites', sorted(set(length)-set(by)))
PY"
