/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
path='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
rows=0
chrom_counts=Counter()
chrom_pos_counts=Counter()
pos_to_chrom={}
dup_conflicts=[]
with open(path, newline='', encoding='utf-8-sig') as f:
    r=csv.DictReader(f)
    print('fields', r.fieldnames)
    for row in r:
        rows+=1
        chrom=row['Chromosome']
        pos=row['Pos']
        chrom_counts[chrom]+=1
        key=(chrom,pos)
        chrom_pos_counts[key]+=1
        if pos in pos_to_chrom and pos_to_chrom[pos]"'!=chrom:
            dup_conflicts.append((pos,pos_to_chrom[pos],chrom))
        pos_to_chrom[pos]=chrom
print('"'rows', rows)
print('unique Pos', len(pos_to_chrom))
print('unique chrom,pos', len(chrom_pos_counts))
print('conflicts', dup_conflicts[:5], len(dup_conflicts))
print('top raw rows', chrom_counts.most_common(10))
print('top unique sites', Counter({c:sum(1 for (cc,p) in chrom_pos_counts if cc==c) for c in chrom_counts}).most_common(10))
PY"
