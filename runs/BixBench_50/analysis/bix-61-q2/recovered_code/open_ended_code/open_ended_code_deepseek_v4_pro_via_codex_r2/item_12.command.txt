/bin/bash -lc "cd /workspace/tmp && curl -sSL --fail --retry 3 -o bwa-conda.tar.bz2 https://api.anaconda.org/download/bioconda/bwa/0.7.18/linux-aarch64/bwa-0.7.18-h0cbc5ad_2.tar.bz2 && python3 - <<'PY'
import tarfile
with tarfile.open('bwa-conda.tar.bz2','r:bz2') as t:
    t.extractall('bwa-conda')
PY
find bwa-conda -maxdepth 4 -type f -o -type l | sort && file bwa-conda/bin/bwa && ldd bwa-conda/bin/bwa || true && chmod +x bwa-conda/bin/bwa && ./bwa-conda/bin/bwa 2>&1 | head -8"
