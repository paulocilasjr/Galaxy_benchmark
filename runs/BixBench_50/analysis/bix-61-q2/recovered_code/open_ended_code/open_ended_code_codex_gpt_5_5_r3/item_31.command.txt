/bin/bash -lc "mkdir -p /workspace/tools/bwa-conda && curl -L https://anaconda.org/bioconda/bwa/0.7.19/download/linux-aarch64/bwa-0.7.19-h0cbc5ad_1.tar.bz2 -o /workspace/tools/bwa-conda/bwa.tar.bz2 && python3 - <<'PY'
import tarfile
archive='/workspace/tools/bwa-conda/bwa.tar.bz2'
out='/workspace/tools/bwa-conda'
with tarfile.open(archive, 'r:bz2') as tf:
    names=[n for n in tf.getnames() if n == 'bin/bwa' or n.startswith('lib/')]
    tf.extractall(out, members=[tf.getmember(n) for n in names])
PY
/workspace/tools/bwa-conda/bin/bwa 2>&1 | sed -n '1,8p'"
