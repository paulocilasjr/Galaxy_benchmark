/bin/bash -lc "python3 - <<'PY'
import urllib.request,re,time
ids=[f'GSM723{i}' for i in range(2572,2591)]
for gsm in ids:
    url=f'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc={gsm}&targ=self&form=text&view=quick'
    try:
        data=urllib.request.urlopen(url,timeout=30).read().decode('utf-8','ignore')
        title=''; chars=[]
        for line in data.splitlines():
            if line.startswith('"'!Sample_title'"'): title=line.split('=',1)[1].strip()
            if line.startswith('"'!Sample_characteristics'"'): chars.append(line.split('=',1)[1].strip())
        print(gsm,title,' | '.join(chars))
    except Exception as e:
        print(gsm,'ERR',e)
PY"
