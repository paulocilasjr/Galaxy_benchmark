# Try a more targeted approach: set R_LIBS_USER and install with full dependency resolution
export R_LIBS_USER=~/R/library
R -e '
.libPaths(c(Sys.getenv("R_LIBS_USER"), .libPaths()))
install.packages("Rcpp", repos="https://cran.r-project.org", lib=Sys.getenv("R_LIBS_USER"), dependencies=TRUE)
' 2>&1 | tail -15
