/bin/bash -lc "curl -Ls https://micro.mamba.pm/api/micromamba/linux-arm64/latest -o /workspace/micromamba-arm64.tar.bz2; python - <<'PY'
import tarfile
src='/workspace/micromamba-arm64.tar.bz2'
dst='/workspace/.micromamba_bin'
with tarfile.open(src,'r:bz2') as tf:
    member=next(m for m in tf.getmembers() if m.name=='bin/micromamba')
    tf.extract(member,dst)
PY
/workspace/.micromamba_bin/bin/micromamba --version"
