/bin/bash -lc "python3 - <<'PY'
import io, os, tarfile, urllib.request
url='https://micro.mamba.pm/api/micromamba/linux-aarch64/latest'
with urllib.request.urlopen(url) as r:
    data=r.read()
with tarfile.open(fileobj=io.BytesIO(data), mode='r:bz2') as tf:
    member=tf.getmember('bin/micromamba')
    src=tf.extractfile(member)
    path='.micromamba-download/bin/micromamba-aarch64'
    with open(path,'wb') as out: out.write(src.read())
    os.chmod(path, 0o755)
PY
.micromamba-download/bin/micromamba-aarch64 --version"
