/bin/bash -lc "Rscript - <<'RS'
.libPaths(c('/workspace/tmp/Rlib', .libPaths()))
suppressPackageStartupMessages(library(DESeq2))
o <- readRDS('/workspace/tmp/deseq_result.rds')
d <- o"'$dds
cn <- '"'condition_ASXL1_vs_Control'
for (a in c(0.05,0.1)) {
  x <- results(d,name=cn,alpha=a)
  cat('alpha',a,'name',sum(x"'$padj<.05,na.rm=TRUE),'"'tested',sum("'!is.na(x$pvalue)),'"'padj_non_na',sum("'!is.na(x$padj)),'"'\\n')
  y <- results(d,contrast=c('condition','ASXL1','Control'),alpha=a)
  cat('alpha',a,'contrast',sum(y"'$padj<.05,na.rm=TRUE),'"'same_p',isTRUE(all.equal(x"'$pvalue,y$pvalue,check.attributes=FALSE)),'"'same_padj',isTRUE(all.equal(x"'$padj,y$padj,check.attributes=FALSE)),'"'\\n')
}
x <- results(d,name=cn,alpha=.05,independentFiltering=FALSE)
cat('no_filter',sum(x"'$padj<.05,na.rm=TRUE),'"'padj_non_na',sum("'!is.na(x$padj)),'"'\\n')
cat('saved_shrunk',sum(o"'$res_shrunk$padj<.05,na.rm=TRUE),'"'\\n')
RS"
