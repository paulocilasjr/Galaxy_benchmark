/bin/bash -c "Rscript - <<'EOF'
for (f in c('GluFevsGluFePlus','SuccvsGluFePlus')) {
  x <- readRDS(file.path('inputs', paste0('res_', f, '.rds')))
  y <- attr(x, 'listData', exact=TRUE)
  ids <- attr(x, 'rownames', exact=TRUE)
  d <- data.frame(gene=ids, baseMean=y"'$baseMean, log2FoldChange=y$log2FoldChange,
                  lfcSE=y$lfcSE, stat=y$stat, pvalue=y$pvalue, padj=y$padj,
                  stringsAsFactors=FALSE)
  write.csv(d, file.path('"'tmp', paste0(f, '.csv')), row.names=FALSE, quote=TRUE)
  cat(f, nrow(d), 'rows written; non-NA padj', sum("'!is.na(d$padj)), '"'; DEG thresholds', sum("'!is.na(d$padj) & abs(d$log2FoldChange)>1.5 & d$padj<0.05), '"'\\n')
}
EOF"
