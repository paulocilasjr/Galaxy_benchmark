/bin/bash -c "cat > tmp/answer_calc.R <<'EOF'
links <- read.delim('tmp/pau_pathway_links.txt', header=FALSE, sep='\\t', stringsAsFactors=FALSE, quote='')
links[,1] <- sub('"'^[''^:]+:'"', '', links[,1]); links[,2] <- sub('"'^[''^:]+:'"', '', links[,2])
pl <- read.delim('tmp/pau_pathway_list.txt', header=FALSE, sep='\\t', stringsAsFactors=FALSE, quote='')
links <- unique(links[links[,2] %in% pl[,1],])
extid <- unique(links[,1]); N <- length(extid)
run <- function(csv) {
  d <- read.csv(csv, stringsAsFactors=FALSE)
  gene <- unique(d"'$gene[!is.na(d$padj) & abs(d$log2FoldChange) > 1.5 & d$padj < 0.05])
  q <- links[links[,1] %in% gene,,drop=FALSE]
  qmap <- split(q[,2], q[,1]); n <- length(qmap)
  terms <- unique(q[,2])
  fullsizes <- sapply(terms, function(t) length(unique(links[links[,2]==t,1])))
  terms <- terms[fullsizes >= 10 & fullsizes <= 500]
  out <- lapply(terms, function(t) {
    k <- length(unique(q[q[,2]==t,1])); M <- fullsizes[[t]]
    data.frame(ID=t, k=k, M=M, p=phyper(k-1, M, N-M, n, lower.tail=FALSE))
  })
  out <- do.call(rbind, out); out$padj <- p.adjust(out$p, method='"'BH')
  out <- out[out"'$padj < 0.05,]
  out[order(out$p),]
}
a <- run('"'tmp/GluFevsGluFePlus.csv')
b <- run('tmp/SuccvsGluFePlus.csv')
cat('iron-depleted significant:\\n'); print(a)
cat('innate-media significant:\\n'); print(b)
cat('iron-only count:', length(setdiff(a"'$ID,b$ID)), '"'\\n'); print(setdiff(a"'$ID,b$ID))
EOF
Rscript tmp/answer_calc.R'
