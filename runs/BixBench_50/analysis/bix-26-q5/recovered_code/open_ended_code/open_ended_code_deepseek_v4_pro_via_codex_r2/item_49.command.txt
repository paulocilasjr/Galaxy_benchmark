/bin/bash -c "Rscript - <<'EOF'
links <- read.delim('tmp/pau_pathway_links.txt', header=FALSE, sep='\\t', stringsAsFactors=FALSE)
links[,1] <- sub('"'^[''^:]+:'"', '', links[,1])
links[,2] <- sub('"'^[''^:]+:'"', '', links[,2])
pl <- read.delim('tmp/pau_pathway_list.txt', header=FALSE, sep='\\t', stringsAsFactors=FALSE)
links <- unique(links[links[,2] %in% pl[,1],])
extid <- unique(links[,1]); N <- length(extid)
d <- read.csv('tmp/GluFevsGluFePlus.csv', stringsAsFactors=FALSE)
gene <- unique(d"'$gene[abs(d$log2FoldChange)>1.5 & d$padj<0.05])
q <- links[links[,1] %in% gene,,drop=FALSE]
qmap <- split(q[,2], q[,1]); n <- length(qmap)
terms <- unique(q[,2])
sizes <- sapply(terms, function(t) length(unique(links[links[,2]==t,1])))
terms <- terms[sizes>=10 & sizes<=500]
res <- do.call(rbind, lapply(terms, function(t){
 k <- length(unique(q[q[,2]==t,1])); M <- sizes[[t]]
 p <- phyper(k-1, M, N-M, n, lower.tail=FALSE)
 data.frame(ID=t,k=k,M=M,p=p)
}))
res$padj <- p.adjust(res$p, method='"'BH')
print(res[order(res"'$p),], row.names=FALSE)
print(res[res$padj<0.05, c('"'ID','p','padj')], row.names=FALSE)
EOF"
