Rscript -e '
con <- gzfile("inputs/res_GluFevsGluFePlus.rds", "rb")
data1 <- unserialize(con)
close(con)

attrs1 <- attributes(data1)

con <- gzfile("inputs/res_SuccvsGluFePlus.rds", "rb")
data2 <- unserialize(con)
close(con)

attrs2 <- attributes(data2)

cat("=== res_GluFevsGluFePlus (Glucose + Fe-depleted vs Glucose + Fe-Plus) ===\n")
cat("rownames length:", length(attrs1$rownames), "\n")
cat("First 20 rownames:", paste(head(attrs1$rownames, 20), collapse=", "), "\n")
cat("nrows:", attrs1$nrows, "\n")
cat("elementMetadata:", class(attrs1$elementMetadata), "\n")

cat("\n=== res_SuccvsGluFePlus (Succinate vs Glucose + Fe-Plus) ===\n")
cat("rownames length:", length(attrs2$rownames), "\n")
cat("First 20 rownames:", paste(head(attrs2$rownames, 20), collapse=", "), "\n")
cat("nrows:", attrs2$nrows, "\n")

# Create data frames
df1 <- data.frame(
  gene = attrs1$rownames,
  baseMean = attrs1$listData$baseMean,
  log2FoldChange = attrs1$listData$log2FoldChange,
  lfcSE = attrs1$listData$lfcSE,
  stat = attrs1$listData$stat,
  pvalue = attrs1$listData$pvalue,
  padj = attrs1$listData$padj,
  stringsAsFactors = FALSE
)

df2 <- data.frame(
  gene = attrs2$rownames,
  baseMean = attrs2$listData$baseMean,
  log2FoldChange = attrs2$listData$log2FoldChange,
  lfcSE = attrs2$listData$lfcSE,
  stat = attrs2$listData$stat,
  pvalue = attrs2$listData$pvalue,
  padj = attrs2$listData$padj,
  stringsAsFactors = FALSE
)

# Filter genes with |log2FC| > 1.5 and padj < 0.05
sig1 <- df1[!is.na(df1$padj) & df1$padj < 0.05 & abs(df1$log2FoldChange) > 1.5, ]
sig2 <- df2[!is.na(df2$padj) & df2$padj < 0.05 & abs(df2$log2FoldChange) > 1.5, ]

cat("\n=== Significant genes (|log2FC|>1.5, padj<0.05) ===\n")
cat("GluFe vs GluFePlus:", nrow(sig1), "genes\n")
cat("Succ vs GluFePlus:", nrow(sig2), "genes\n")

# Save to CSV for Python
write.csv(df1, "/workspace/df1_full.csv", row.names = FALSE)
write.csv(df2, "/workspace/df2_full.csv", row.names = FALSE)
' 2>&1
