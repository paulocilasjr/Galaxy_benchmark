import pandas as pd
import numpy as np
from Bio.KEGG import REST
import time
import urllib.error

# Read significant gene lists
sig_iron = pd.read_csv("genes_iron_depleted.txt", header=None)[0].tolist()
sig_innate = pd.read_csv("genes_innate_media.txt", header=None)[0].tolist()
all_genes = pd.read_csv("all_genes.txt", header=None)[0].tolist()

print(f"Iron-depleted DEGs: {len(sig_iron)}")
print(f"Innate media DEGs: {len(sig_innate)}")
print(f"Total genes in data: {len(all_genes)}")

# Map PA14 IDs to KEGG IDs: our format is PA14_33530, KEGG format is pau:PA14_33530
# So we just prepend "pau:"
sig_iron_kegg = [f"pau:{g}" for g in sig_iron]
sig_innate_kegg = [f"pau:{g}" for g in sig_innate]
all_kegg = [f"pau:{g}" for g in all_genes]

# Get all KEGG pathways for pau
print("\nFetching KEGG pathways for P. aeruginosa PA14...")
try:
    pathways_raw = REST.kegg_list("pathway", "pau").read()
    pw_lines = [l for l in pathways_raw.strip().split('\n') if l]
    print(f"Total pathway entries: {len(pw_lines)}")
    for line in pw_lines[:5]:
        print(line)
except Exception as e:
    print(f"Error: {e}")

# Parse pathway IDs
pathway_ids = []
pathway_names = {}
for line in pw_lines:
    parts = line.split('\t')
    if len(parts) >= 2:
        pid = parts[0].replace('path:', '')
        pathway_ids.append(pid)
        pathway_names[pid] = parts[1]

print(f"\nParsed {len(pathway_ids)} pathways")
