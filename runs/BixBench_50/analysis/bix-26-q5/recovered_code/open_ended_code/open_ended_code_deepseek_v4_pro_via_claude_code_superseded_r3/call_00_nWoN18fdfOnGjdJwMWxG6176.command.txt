Rscript -e '
con <- gzfile("inputs/res_GluFevsGluFePlus.rds", "rb")
data <- unserialize(con)
close(con)

# Access attributes without triggering S4
attrs <- attributes(data)
cat("Attribute names:", paste(names(attrs), collapse=", "), "\n")

# Try to get the data slot directly using .Data
# For S4 objects, the data is stored in the .Data slot
cat("\nTrying to access .Data slot...\n")
dd <- attrs[[".Data"]]
cat("Class of .Data:", class(dd), "\n")

# .Data is an environment for reference classes
cat("\nTrying to access names in .Data:\n")
print(ls(dd))
' 2>&1
