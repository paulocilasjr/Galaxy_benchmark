R --no-save -e '
res1 <- readRDS("inputs/res_GluFevsGluFePlus.rds")
# Access listData 
ld <- attr(res1, "listData")
cat("listData class:", class(ld), "\n")
cat("listData names:", names(ld), "\n")
for(nm in names(ld)) {
  cat("\n---", nm, "---\n")
  cat("class:", class(ld[[nm]]), "\n")
  cat("length:", length(ld[[nm]]), "\n")
  print(head(ld[[nm]]))
}

# Also get rownames
rn <- attr(res1, "rownames")
cat("\nRownames (first 10):", head(rn, 10), "\n")
cat("Total rows:", length(rn), "\n")
' 2>&1
