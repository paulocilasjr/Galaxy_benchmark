R --no-save -e '
res1 <- readRDS("inputs/res_GluFevsGluFePlus.rds")
res2 <- readRDS("inputs/res_SuccvsGluFePlus.rds")

save_csv <- function(res, path) {
  ld <- attr(res, "listData")
  rn <- attr(res, "rownames")
  df <- data.frame(
    gene = rn,
    baseMean = ld$baseMean,
    log2FoldChange = ld$log2FoldChange,
    lfcSE = ld$lfcSE,
    stat = ld$stat,
    pvalue = ld$pvalue,
    padj = ld$padj,
    stringsAsFactors = FALSE
  )
  write.csv(df, path, row.names = FALSE)
  cat("Saved", path, ":", nrow(df), "rows\n")
}

save_csv(res1, "/workspace/res_GluFevsGluFePlus.csv")
save_csv(res2, "/workspace/res_SuccvsGluFePlus.csv")
cat("Done\n")
' 2>&1
