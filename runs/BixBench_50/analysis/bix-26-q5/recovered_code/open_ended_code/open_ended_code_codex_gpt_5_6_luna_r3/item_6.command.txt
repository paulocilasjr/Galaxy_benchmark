/bin/bash -lc "Rscript - <<'RS'
options(error=function(){traceback(2); q(status=1)})
for (f in list.files('/workspace/inputs', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  cat('typeof=', typeof(x), 'isS4=', isS4(x), '\\n')
  cat('attributes:\\n'); print(attributes(x))
  cat('slot names raw:\\n'); print(try(slotNames(x)))
  cat('internal inspect:\\n'); .Internal(inspect(x))
}
RS"
