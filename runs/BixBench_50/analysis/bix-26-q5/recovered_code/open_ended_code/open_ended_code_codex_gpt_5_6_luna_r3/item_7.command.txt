/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  cat('isS4=', isS4(x), '\\n')
  cat('attribute names=', paste(names(attributes(x)), collapse=' | '), '\\n')
  ca <- tryCatch(attr(x, 'class', exact=TRUE), error=function(e) paste('ERR', conditionMessage(e)))
  cat('class attr=', paste(ca, collapse=' | '), '\\n')
  rn <- attr(x, 'rownames', exact=TRUE)
  cat('rownames length=', length(rn), 'first=', paste(head(rn,3), collapse=' | '), '\\n')
  cat('nrows attr=', attr(x,'nrows', exact=TRUE), '\\n')
  for (sn in c('listData','elementType','elementMetadata','nrows','rownames','metadata')) {
    cat('slot',sn,': ')
    y <- tryCatch(x[[sn]], error=function(e) structure(list(msg=conditionMessage(e)), class='ERR'))
    cat('[[ type=', typeof(y), ' class=', paste(class(y),collapse=','), ' len=', length(y), '\\n', sep='')
    if ("'!inherits(y,'"'ERR')) {
      if (is.list(y)) cat(' names=', paste(head(names(y),20), collapse=' | '), '\\n', sep='')
      if (is.atomic(y)) cat(' head=', paste(head(y,5), collapse=' | '), '\\n', sep='')
    } else cat(' err=', y"'$msg,'"'\\n',sep='')
  }
  cat('direct slots:\\n')
  for (sn in c('listData','elementType','elementMetadata','nrows','rownames','metadata')) {
    y <- tryCatch(eval(parse(text=paste0('x@',sn))), error=function(e) structure(list(msg=conditionMessage(e)), class='ERR'))
    cat(sn, ': type=', typeof(y), ' class=', paste(class(y),collapse=','), ' len=', length(y), sep='')
    if (is.list(y)) cat(' names=', paste(head(names(y),20), collapse=' | '), sep='')
    if (is.atomic(y)) cat(' head=', paste(head(y,5), collapse=' | '), sep='')
    if (inherits(y,'ERR')) cat(' err=',y"'$msg,sep='"'')
    cat('\\n')
  }
}
RS"
