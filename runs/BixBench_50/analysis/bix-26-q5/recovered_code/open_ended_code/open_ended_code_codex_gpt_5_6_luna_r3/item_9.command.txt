/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', full.names=TRUE)) {
  x <- readRDS(f)
  d <- x@listData
  rn <- attr(x,'rownames', exact=TRUE)
  cat('\\n===', basename(f), '===\\n')
  cat('n=', length(rn), 'cols=', paste(names(d),collapse=','), '\\n')
  cat('first rows:\\n')
  print(data.frame(gene=head(rn,10), lfc=head(d"'$log2FoldChange,10), p=head(d$pvalue,10), padj=head(d$padj,10), row.names=NULL))
  for (nm in names(d)) {
    z <- d[[nm]]
    if (is.numeric(z)) cat(nm, '"'NA=',sum(is.na(z)), 'finite=',sum(is.finite(z)), 'range=',paste(range(z,na.rm=TRUE),collapse=':'), '\\n')
  }
  sig <- "'!is.na(d$log2FoldChange) & !is.na(d$padj) & abs(d$log2FoldChange)>1.5 & d$padj<0.05
  cat('"'filter count=',sum(sig),'up=',sum(sig & d"'$log2FoldChange>0),'"'down=',sum(sig & d"'$log2FoldChange<0),'"'padj<0.05=',sum("'!is.na(d$padj)&d$padj<0.05),'"'absLFC>1.5=',sum("'!is.na(d$log2FoldChange)&abs(d$log2FoldChange)>1.5),'"'\\n')
  cat('genes sig first=', paste(head(rn[sig],20),collapse=','), '\\n')
  cat('metadata attr names=',paste(names(attr(x,'metadata',exact=TRUE)),collapse=','),'\\n')
}
RS"
