/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  cat('class:', paste(class(x), collapse=', '), '\\n')
  cat('typeof:', typeof(x), '\\n')
  cat('length:', length(x), '\\n')
  cat('names:', paste(names(x), collapse=' | '), '\\n')
  str(x, max.level=2)
}
RS"
