import re
import requests
import time
from scipy.stats import fisher_exact
from statsmodels.stats.multitest import multipletests

# Read gene lists
with open("/tmp/de_iron.txt") as f:
    iron_de = [l.strip() for l in f if l.strip()]
with open("/tmp/de_innate.txt") as f:
    innate_de = [l.strip() for l in f if l.strip()]
with open("/tmp/all_genes.txt") as f:
    all_genes = [l.strip() for l in f if l.strip()]

print(f"Total genes: {len(all_genes)}")
print(f"Iron-depleted DE genes: {len(iron_de)}")
print(f"Innate media DE genes: {len(innate_de)}")

# Get KEGG pathway-gene mapping for PA14 (pau)
def fetch_kegg(url, desc="", retries=3):
    for i in range(retries):
        try:
            r = requests.get(url, timeout=60)
            if r.status_code == 200:
                return r.text
            time.sleep(2)
        except:
            time.sleep(2)
    return None

print("Fetching KEGG data...")
link_text = fetch_kegg("https://rest.kegg.jp/link/pau/pathway", "pathway-gene links")
lines = [l for l in link_text.strip().split('\n') if l]

pw2genes = {}
for line in lines:
    parts = line.strip().split('\t')
    if len(parts) >= 2:
        pw = parts[0].strip()  # path:pau01100
        gene = parts[1].strip()  # pau:PA14_00010
        if pw not in pw2genes:
            pw2genes[pw] = set()
        pw2genes[pw].add(gene)

print(f"KEGG pathway-gene links: {len(lines)} links, {len(pw2genes)} pathways")

# Get pathway names
pw_text = fetch_kegg("https://rest.kegg.jp/list/pathway/pau", "pathway names")
pw_name = {}
for line in pw_text.strip().split('\n'):
    if not line: continue
    parts = line.split('\t')
    if len(parts) >= 2:
        pid = parts[0].strip()
        name = re.sub(r'\s+-\s+Pseudomonas aeruginosa.*', '', parts[1].strip())
        pw_name['path:' + pid] = name

def to_kegg(gene):
    return f"pau:{gene}"

universe = set(to_kegg(g) for g in all_genes)
iron_fg = set(to_kegg(g) for g in iron_de)
innate_fg = set(to_kegg(g) for g in innate_de)

print(f"Universe: {len(universe)}, Iron FG: {len(iron_fg)}, Innate FG: {len(innate_fg)}")

# Filter pathways: at least 3 genes in universe
pathways_in_universe = {}
for pw, genes in pw2genes.items():
    genes_in_universe = genes & universe
    if len(genes_in_universe) >= 3:
        pathways_in_universe[pw] = genes_in_universe

print(f"Pathways with >= 3 genes in universe: {len(pathways_in_universe)}")

# ORA with Fisher's exact test
def ora(fg, universe, pw2genes):
    results = {}
    universe_size = len(universe)
    fg_size = len(fg)
    
    for pw, pw_genes in pw2genes.items():
        a = len(pw_genes & fg)
        if a == 0:
            continue
        
        b = len(pw_genes) - a
        c = fg_size - a
        d = universe_size - a - b - c
        
        table = [[a, b], [c, d]]
        odds_ratio, p_value = fisher_exact(table, alternative='two-sided')
        
        results[pw] = {
            'pathway': pw,
            'name': pw_name.get(pw, pw),
            'count': a,
            'pw_size': len(pw_genes),
            'pvalue': p_value,
            'or': odds_ratio
        }
    
    return results

iron_results = ora(iron_fg, universe, pathways_in_universe)
innate_results = ora(innate_fg, universe, pathways_in_universe)

print(f"Iron pathways tested: {len(iron_results)}, Innate pathways tested: {len(innate_results)}")

# Benjamini-Hochberg adjustment
for label, results in [("iron", iron_results), ("innate", innate_results)]:
    pvals = [r['pvalue'] for r in results.values()]
    pw_list = list(results.keys())
    if len(pvals) > 0:
        _, padj, _, _ = multipletests(pvals, method='fdr_bh')
        for i, pw in enumerate(pw_list):
            results[pw]['padj'] = padj[i]

# Significant pathways
iron_sig = {pw: r for pw, r in iron_results.items() if r['padj'] < 0.05}
innate_sig = {pw: r for pw, r in innate_results.items() if r['padj'] < 0.05}

print(f"\n=== RESULTS ===")
print(f"Iron-depleted significant pathways: {len(iron_sig)}")
print(f"Innate media significant pathways: {len(innate_sig)}")

iron_only = set(iron_sig.keys()) - set(innate_sig.keys())
innate_only = set(innate_sig.keys()) - set(iron_sig.keys())
both = set(iron_sig.keys()) & set(innate_sig.keys())

print(f"\nSignificant in iron-depleted ONLY: {len(iron_only)}")
print(f"Significant in innate media ONLY: {len(innate_only)}")
print(f"Significant in BOTH: {len(both)}")

if len(iron_only) > 0:
    print("\n--- Iron-only pathways ---")
    for pw in sorted(iron_only):
        r = iron_results[pw]
        print(f"  {pw} {r['name']}: padj={r['padj']:.4e}, genes={r['count']}/{r['pw_size']}")

print(f"\n=== ANSWER: {len(iron_only)} ===")
