import re
import requests
import time
from scipy.stats import fisher_exact
from statsmodels.stats.multitest import multipletests

# Read gene lists
with open("/tmp/de_iron.txt") as f:
    iron_de = [l.strip() for l in f if l.strip()]
with open("/tmp/de_innate.txt") as f:
    innate_de = [l.strip() for l in f if l.strip()]
with open("/tmp/all_genes.txt") as f:
    all_genes = [l.strip() for l in f if l.strip()]

# Get KEGG data
def fetch_kegg(url):
    for i in range(5):
        try:
            r = requests.get(url, timeout=60)
            if r.status_code == 200:
                return r.text
            time.sleep(3)
        except:
            time.sleep(3)
    return None

link_text = fetch_kegg("https://rest.kegg.jp/link/pau/pathway")
lines = [l for l in link_text.strip().split('\n') if l]

pw2genes = {}
for line in lines:
    parts = line.strip().split('\t')
    if len(parts) >= 2:
        pw = parts[0].strip()
        gene = parts[1].strip()
        if pw not in pw2genes:
            pw2genes[pw] = set()
        pw2genes[pw].add(gene)

pw_text = fetch_kegg("https://rest.kegg.jp/list/pathway/pau")
pw_name = {}
for line in pw_text.strip().split('\n'):
    if not line: continue
    parts = line.split('\t')
    if len(parts) >= 2:
        pid = parts[0].strip()
        name = re.sub(r'\s+-\s+Pseudomonas aeruginosa.*', '', parts[1].strip())
        pw_name['path:' + pid] = name

def to_kegg(gene):
    return f"pau:{gene}"

universe = set(to_kegg(g) for g in all_genes)
iron_fg = set(to_kegg(g) for g in iron_de)
innate_fg = set(to_kegg(g) for g in innate_de)

# Check mapping coverage
print("=== MAPPING COVERAGE ===")
all_kegg = set()
for genes in pw2genes.values():
    all_kegg.update(genes)
print(f"All KEGG pau genes: {len(all_kegg)}")
print(f"Universe genes in KEGG: {len(universe & all_kegg)}")
print(f"Universe genes NOT in KEGG: {len(universe - all_kegg)}")

# Show first few unmapped
unmapped = universe - all_kegg
print(f"Unmapped examples: {list(unmapped)[:10]}")

# Filter pathways
pathways_in_universe = {}
for pw, genes in pw2genes.items():
    genes_in_universe = genes & universe
    if len(genes_in_universe) >= 3:
        pathways_in_universe[pw] = genes_in_universe

# ORA
def ora(fg, universe, pw2genes):
    results = {}
    universe_size = len(universe)
    fg_size = len(fg)
    
    for pw, pw_genes in pw2genes.items():
        a = len(pw_genes & fg)
        if a == 0:
            continue
        b = len(pw_genes) - a
        c = fg_size - a
        d = universe_size - a - b - c
        table = [[a, b], [c, d]]
        odds_ratio, p_value = fisher_exact(table, alternative='two-sided')
        results[pw] = {
            'pathway': pw,
            'name': pw_name.get(pw, pw),
            'count': a,
            'pw_size': len(pw_genes),
            'pvalue': p_value,
            'or': odds_ratio
        }
    return results

iron_results = ora(iron_fg, universe, pathways_in_universe)
innate_results = ora(innate_fg, universe, pathways_in_universe)

# BH adjustment
for label, results in [("iron", iron_results), ("innate", innate_results)]:
    pvals = [r['pvalue'] for r in results.values()]
    pw_list = list(results.keys())
    if len(pvals) > 0:
        _, padj, _, _ = multipletests(pvals, method='fdr_bh')
        for i, pw in enumerate(pw_list):
            results[pw]['padj'] = padj[i]

iron_sig = {pw: r for pw, r in iron_results.items() if r['padj'] < 0.05}
innate_sig = {pw: r for pw, r in innate_results.items() if r['padj'] < 0.05}

iron_only = set(iron_sig.keys()) - set(innate_sig.keys())

print(f"\n=== DETAILED IRON-DEPLETED SIGNIFICANT PATHWAYS ===")
for pw in sorted(iron_sig.keys(), key=lambda p: iron_sig[p]['padj']):
    r = iron_sig[pw]
    in_innate = "YES" if pw in innate_sig else "NO"
    innate_padj = f"{innate_sig[pw]['padj']:.4e}" if pw in innate_sig else "N/A"
    print(f"  {r['name']}: padj={r['padj']:.4e}, genes={r['count']}/{r['pw_size']}, OR={r['or']:.2f}, in_innate={in_innate}({innate_padj})")

# Also check the innate results for the iron-only pathways
print(f"\n=== INNATE RESULTS FOR IRON-ONLY PATHWAYS ===")
for pw in sorted(iron_only):
    if pw in innate_results:
        r = innate_results[pw]
        print(f"  {r['name']}: padj={r['padj']:.4e}, genes={r['count']}/{r['pw_size']}")
    else:
        print(f"  {pw_name.get(pw, pw)}: NOT TESTED (no genes in FG)")

    print(f"  In innate FG: {kegg_genes & innate_fg}")
