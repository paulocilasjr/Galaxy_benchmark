Rscript -e '
obj1 <- suppressWarnings(readRDS("inputs/res_GluFevsGluFePlus.rds"))
obj2 <- suppressWarnings(readRDS("inputs/res_SuccvsGluFePlus.rds"))

a1 <- attributes(obj1)
a2 <- attributes(obj2)

# Show rownames
cat("=== res_GluFevsGluFePlus (Iron-depleted) ===\n")
cat("Number of genes:", length(a1$rownames), "\n")
cat("First 20 genes:", head(a1$rownames, 20), "\n\n")

cat("=== res_SuccvsGluFePlus (Innate media) ===\n")
cat("Number of genes:", length(a2$rownames), "\n")
cat("First 20 genes:", head(a2$rownames, 20), "\n")

# Filter criteria: |log2FoldChange| > 1.5, padj < 0.05
lfc1 <- a1$listData$log2FoldChange
padj1 <- a1$listData$padj
names1 <- a1$rownames

lfc2 <- a2$listData$log2FoldChange
padj2 <- a2$listData$padj
names2 <- a2$rownames

# Filter (handle NA padj)
keep1 <- !is.na(padj1) & padj1 < 0.05 & abs(lfc1) > 1.5
keep2 <- !is.na(padj2) & padj2 < 0.05 & abs(lfc2) > 1.5

cat("\n=== Filtered genes ===\n")
cat("Iron-depleted DE genes:", sum(keep1), "\n")
cat("Innate media DE genes:", sum(keep2), "\n")

de1 <- names1[keep1]
de2 <- names2[keep2]

cat("First 20 Iron-depleted DE genes:", head(de1, 20), "\n")
cat("First 20 Innate media DE genes:", head(de2, 20), "\n")

# Save for later use
writeLines(de1, "/tmp/de_iron.txt")
writeLines(de2, "/tmp/de_innate.txt")
writeLines(names1, "/tmp/all_genes.txt")
' 2>&1
