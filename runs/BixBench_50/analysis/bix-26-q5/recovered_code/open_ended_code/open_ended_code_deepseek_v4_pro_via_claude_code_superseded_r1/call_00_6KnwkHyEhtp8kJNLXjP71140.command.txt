Rscript -e '
# Try to read RDS without DESeq2 package loading
# The key is to suppress the class definition loading

# Method 1: try using tryCatch with suppressPackageStartupMessages
obj <- tryCatch(
  suppressWarnings(readRDS("inputs/res_GluFevsGluFePlus.rds")),
  error = function(e) NULL
)

# Try to see if we can access data without triggering class loading
cat("Class:", class(obj), "\n")
cat("Is S4:", isS4(obj), "\n")

# Try to get the data directly from the serialized object
# Use .jcopy or similar
# Actually, try to coerce
df <- tryCatch(
  as.data.frame(obj),
  error = function(e) { cat("as.data.frame error:", e$message, "\n"); NULL }
)

cat("Data frame success:", !is.null(df), "\n")

# Try accessing attributes directly
attrs <- attributes(obj)
cat("Attribute names:", names(attrs), "\n")
for (a in names(attrs)) {
  cat("  ", a, ":", class(attrs[[a]]), "\n")
}
' 2>&1
