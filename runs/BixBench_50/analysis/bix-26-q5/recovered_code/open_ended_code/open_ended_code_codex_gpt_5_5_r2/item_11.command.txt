}
for (f in c('"'inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')) {
  d <- extract(f)
  fg <- subset(d, "'!is.na(padj) & !is.na(log2FoldChange) & abs(log2FoldChange) > 1.5 & padj < 0.05)
  cat(basename(f), '"'universe', nrow(d), 'foreground', nrow(fg), 'up', sum(fg"'$log2FoldChange>0), '"'down', sum(fg"'$log2FoldChange<0), '"'\\n')
}
RS"
