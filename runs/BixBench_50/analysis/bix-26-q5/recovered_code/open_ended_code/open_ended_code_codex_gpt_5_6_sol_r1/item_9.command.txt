/bin/bash -lc "Rscript - <<'RS'
read_de <- function(f) {x<-readRDS(f);data.frame(gene=attr(x,'rownames'),as.data.frame(attr(x,'listData'),check.names=FALSE))}
fe<-read_de('inputs/res_GluFevsGluFePlus.rds'); su<-read_de('inputs/res_SuccvsGluFePlus.rds')
fg1<-fe"'$gene[abs(fe$log2FoldChange)>1.5 & fe$padj<.05]; fg2<-su$gene[abs(su$log2FoldChange)>1.5 & su$padj<.05]
l<-read.delim('"'/tmp/pau_pathway_links.tsv',header=FALSE,col.names=c('gene','path'));l"'$gene<-sub('"'"'^pau:'"','',l"'$gene);l$path<-sub('"'"'^path:'"','',l"'$path);l<-unique(l)
nm<-read.delim('"'/tmp/pau_pathways.tsv',header=FALSE,col.names=c('path','name'))
ora<-function(fg,universe){
 universe<-intersect(unique(universe),unique(l"'$gene)); fg<-intersect(unique(fg),universe)
 ll<-l[l$gene %in% universe,]; gs<-split(ll$gene,ll$path);gs<-gs[lengths(gs)>=10 & lengths(gs)<=500]
 z<-do.call(rbind,lapply(names(gs),function(id){M<-length(gs[[id]]);k<-sum(fg%in%gs[[id]]);if(!k)return(NULL);data.frame(path=id,k,M,n=length(fg),N=length(universe),p=phyper(k-1,M,length(universe)-M,length(fg),lower.tail=FALSE))}))
 z$padj<-p.adjust(z$p,'"'BH');merge(z,nm,by='path')[order(z"'$padj),]
}
for(uname in c('"'KEGG annotated','DE-table eligible mapped')){
 u<-if(uname=='KEGG annotated')unique(l"'$gene) else intersect(fe$gene,su$gene)
 a<-ora(fg1,u);b<-ora(fg2,u);sa<-a$path[a$padj<.05];sb<-b$path[b$padj<.05]
 cat('"'\\n',uname,'universe N=',unique(a"'$N),'"'iron sig',length(sa),'innate sig',length(sb),'iron-only',length(setdiff(sa,sb)),'\\n')
 print(a[a"'$path%in%setdiff(sa,sb),c('"'path','name','k','M','n','N','p','padj')],row.names=FALSE)
 cat('iron sig:',paste(sa,collapse=', '),'\\ninnate sig:',paste(sb,collapse=', '),'\\n')
}
RS"
