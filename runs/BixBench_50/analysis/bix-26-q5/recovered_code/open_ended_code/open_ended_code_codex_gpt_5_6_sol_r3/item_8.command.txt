/bin/bash -lc "Rscript - <<'RS'
for (p in list.files('inputs', pattern='rds"'$'"', full.names=TRUE)) {
 x <- readRDS(p)
 rn <- attr(x, 'rownames')
 ld <- attr(x, 'listData')
 cat('\\nFILE', p, '\\n')
 cat('rows', length(rn), 'first', paste(head(rn), collapse=', '), '\\n')
 cat('listData typeof', typeof(ld), 'len', length(ld), 'names', paste(names(ld), collapse=', '), '\\n')
 for (nm in names(ld)) {
   z <- ld[[nm]]
   cat(nm, typeof(z), length(z), 'head=', paste(head(z), collapse=', '), '\\n')
 }
 cat('metadata typeof', typeof(attr(x,'metadata')), 'len', length(attr(x,'metadata')), '\\n')
 cat('priorInfo typeof', typeof(attr(x,'priorInfo')), '\\n')
}
RS"
