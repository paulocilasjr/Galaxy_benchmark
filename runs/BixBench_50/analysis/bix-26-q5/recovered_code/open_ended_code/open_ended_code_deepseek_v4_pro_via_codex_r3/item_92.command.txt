/bin/bash -lc "cd /workspace && Rscript - <<'RS'
library(jsonlite)
D <- fromJSON('tmp/kegg_pau.json')
g2p <- D"'$g2p
p2g <- split(rep(names(g2p), lengths(g2p)), unlist(g2p))
all_annotated <- unique(names(g2p))
ora <- function(degs, universe=NULL, minsize=1, maxsize=Inf) {
  genes <- unique(degs)
  mapped <- genes[genes %in% names(g2p)]
  qpaths <- unique(unlist(g2p[mapped], use.names=FALSE))
  extID <- if (is.null(universe)) all_annotated else intersect(universe, all_annotated)
  res <- list()
  for (p in sort(qpaths)) {
    pathway_genes <- intersect(p2g[[p]], extID)
    M <- length(pathway_genes)
    if (M < minsize || M > maxsize) next
    k <- length(intersect(mapped, pathway_genes))
    if (k == 0) next
    N <- length(extID); n <- length(mapped)
    pv <- phyper(k - 1, M, N - M, n, lower.tail = FALSE)
    res[[p]] <- c(k=k, M=M, pvalue=pv)
  }
  if (length(res)==0) return(NULL)
  df <- do.call(rbind, lapply(names(res), function(p) data.frame(ID=p,k=res[[p]]['"'k'],M=res[[p]]['M'],pvalue=res[[p]]['pvalue'],stringsAsFactors=FALSE)))
  df"'$padj <- p.adjust(df$pvalue, '"'BH')
  df[order(df"'$pvalue),]
}
gl <- readLines('"'tmp/GluFe_degs.txt'); sl <- readLines('tmp/Succ_degs.txt')
for (mode in c('no_size_filter','min1max500','min0max500')) {
  if(mode=='no_size_filter'){mn<-1; mx<-Inf}
  else if(mode=='min1max500'){mn<-1;mx<-500}
  else {mn<-0;mx<-500}
  g<-ora(gl,NULL,mn,mx); s<-ora(sl,NULL,mn,mx)
  cat('\\n',mode,'\\n')
  cat('GluFe tested',nrow(g),'sig',sum(g"'$padj<.05),paste(g$ID[g$padj<.05],collapse='"','),'\\n')
  cat('Succ tested',nrow(s),'sig',sum(s"'$padj<.05),paste(s$ID[s$padj<.05],collapse='"','),'\\n')
  cat('only',length(setdiff(g"'$ID[g$padj<.05],s$ID[s$padj<.05])),paste(setdiff(g$ID[g$padj<.05],s$ID[s$padj<.05]),collapse='"','),'\\n')
}
RS"
