/bin/bash -lc "cd /workspace && python3 - <<'PY'
import json
p='run_trace/codex_events.jsonl'
items=[]
for line in open(p):
    try:
        obj=json.loads(line)
    except: continue
    if obj.get('type')=='item.completed' and obj.get('item',{}).get('type')=='command_execution':
        cmd=obj['item']['command']
        out=obj.get('item',{}).get('aggregated_output','')
        # print last 15 commands and summaries
        items.append((cmd,out,obj.get('item',{}).get('exit_code')))
for i,(cmd,out,code) in enumerate(items[-25:],1):
    print(f'\\n===== ITEM {i} exit={code} =====')
    print('CMD:', cmd[:2000])
    print('OUT (last 2000):')
    print(out[-2000:])
PY"
