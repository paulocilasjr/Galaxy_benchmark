/bin/bash -lc "Rscript - <<'RS'
m <- read.delim(url('https://rest.kegg.jp/link/pathway/pau'), header=FALSE, stringsAsFactors=FALSE, col.names=c('gene','path'))
cat('pathway ids:\\n'); print(sort(unique(sub('"'^path:'"','',m"'$path))))
cat('"'counts sorted:\\n'); print(sort(table(sub('"'^path:'"','',m"'$path)), decreasing=TRUE))
# all pathway names
nm <- read.delim(url('"'https://rest.kegg.jp/list/pathway/pau'), header=FALSE, stringsAsFactors=FALSE, col.names=c('path','name'))
nm"'$path <- sub('"'"'^path:'"','',nm"'$path)
cat('"'names n=',nrow(nm),'\\n'); print(head(nm,20), row.names=FALSE)
RS"
