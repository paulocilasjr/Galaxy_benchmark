/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')) {
  cat('\\n===', f, '===\\n')
  x <- readRDS(f)
  cat('class:', paste(class(x), collapse=', '), '\\n')
  cat('typeof:', typeof(x), '\\n')
  cat('length:', length(x), '\\n')
  if (is.data.frame(x)) {
    cat('dim:', paste(dim(x), collapse=' x '), '\\n')
    cat('names:', paste(names(x), collapse=' | '), '\\n')
    print(utils::head(x, 5))
    cat('rownames head:', paste(utils::head(rownames(x)), collapse=' | '), '\\n')
    cat('metadata attrs:\\n')
    print(attributes(x)[setdiff(names(attributes(x)), c('names','class','row.names'))])
    cat('column classes:\\n')
    print(vapply(x, function(z) paste(class(z), collapse='/'), character(1)))
  } else {
    str(x, max.level=2)
  }
}
RS"
