/bin/bash -lc "Rscript - <<'RS'
read_res <- function(f) {
  a <- attributes(readRDS(f))
  z <- as.data.frame(a"'$listData, stringsAsFactors=FALSE)
  rownames(z) <- a$rownames
  z
}
for (f in c('"'inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')) {
  z <- read_res(f)
  keep <- is.finite(z"'$log2FoldChange) & is.finite(z$padj) & abs(z$log2FoldChange) > 1.5 & z$padj < 0.05
  cat(basename(f), '"'rows=', nrow(z), 'NA_lfc=', sum("'!is.finite(z$log2FoldChange)), '"'NA_padj=', sum("'!is.finite(z$padj)), '"'foreground=', sum(keep), 'up=', sum(keep & z"'$log2FoldChange>1.5), '"'down=', sum(keep & z"'$log2FoldChange < -1.5), '"'\\n')
  cat('first foreground:', paste(head(rownames(z)[keep]), collapse=','), '\\n')
}
# inspect KEGG mappings and overlap using live REST responses in memory
m <- read.delim(url('https://rest.kegg.jp/link/pathway/pau'), header=FALSE, stringsAsFactors=FALSE, col.names=c('gene','path'))
cat('KEGG mapping rows=', nrow(m), 'genes=', length(unique(m"'$gene)), '"'pathways=', length(unique(m"'$path)), '"'\\n')
for (f in c('inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')) {
  z <- read_res(f); keep <- is.finite(z"'$log2FoldChange) & is.finite(z$padj) & abs(z$log2FoldChange)>1.5 & z$padj<0.05
  ids <- paste0('"'pau:', rownames(z)[keep])
  cat(basename(f), 'fg=',sum(keep), 'fg mapped=',sum(ids %in% m"'$gene), '"'unique mapped=',length(unique(ids[ids %in% m"'$gene])), '"'\\n')
  cat('universe mapped=',sum(paste0('pau:',rownames(z)) %in% m"'$gene), '"'\\n')
}
RS"
