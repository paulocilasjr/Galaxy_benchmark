import pandas as pd, numpy as np, os
from scipy.stats import hypergeom

def read_csv(p):

# parse mapping

rows=[]

 if len(parts)>=2: rows.append((parts[1].split(':',1)[-1],parts[0].split(':',1)[-1]))

p2n={l.split('\\t')[0]:l.split('\\t',1)[1] for l in names if '\\t' in l}
p2g=p2g[p2g.path.isin(p2n)]
def rbh(p):
 p=np.asarray(p,float); n=len(p); i=np.arange(n,0,-1); o=np.argsort(p,kind='stable')[::-1]; ro=np.argsort(o,kind='stable'); return np.minimum(1,np.minimum.accumulate((n/i)*p[o]))[ro]
