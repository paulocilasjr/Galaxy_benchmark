/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
  cat('\\n', f, '\\n')
  x <- readRDS(f)
  cat('typeof', typeof(x), 'isS4', isS4(x), '\\n')
  cat('attr class raw:', paste(attr(x,'class'), collapse=','), '\\n')
  cat('slotNames may fail...\\n')
  print(try(slotNames(x), silent=TRUE))
  print(try(attributes(x), silent=TRUE))
}
RS"
