/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
read_tab <- function(u) { con<-url(u,'rb'); on.exit(close(con)); z<-readLines(con,warn=FALSE); z<-z[nzchar(z)]; do.call(rbind,strsplit(z,'\\\\t')) }
pw <- read_tab('https://rest.kegg.jp/list/pathway/pau'); rownames(pw)<-sub('"'^pau'"','',pw[,1]);
for (id in c('00405','01053','00190','02024','01110','00860','00030','02010','01200','04146','01503')) print(pw[paste0('pau',id),,drop=FALSE])
RS"
