/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  a <- attributes(x)
  cat('attr names:', paste(names(a), collapse=' | '), '\\n')
  for (nm in names(a)) {
    z <- a[[nm]]
    ty <- typeof(z); s4 <- isS4(z)
    if ("'!s4) {
      le <- length(z); di <- tryCatch(paste(dim(z),collapse='"' x '),error=function(e)'<err>'); na <- tryCatch(paste(head(names(z),20),collapse=','),error=function(e)'<err>')
      cl <- paste(class(z),collapse=',')
      cat('attr',nm,'typeof',ty,'class',cl,'len',le,'dim',di,'names',na,'\\n')
    } else cat('attr',nm,'typeof',ty,'S4 TRUE\\n')
  }
  z <- a[['listData']]
  cat('listData typeof',typeof(z),'S4',isS4(z),'len',length(z),'names',paste(names(z),collapse=' | '),'\\n')
  if (is.list(z)) {
    for (nn in names(z)) {
      zz <- z[[nn]]
      cat('  ',nn,'typeof',typeof(zz),'S4',isS4(zz),'len',length(zz),'dim',paste(dim(zz),collapse=' x '),'head',paste(head(zz,5),collapse=' | '),'\\n')
    }
  }
  em <- a[['elementMetadata']]
  cat('elementMetadata typeof',typeof(em),'S4',isS4(em),'attr names',paste(names(attributes(em)),collapse=' | '),'\\n')
}
RS"
