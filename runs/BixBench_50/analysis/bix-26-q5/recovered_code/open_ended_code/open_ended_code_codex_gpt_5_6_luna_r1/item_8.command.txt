/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  a <- attributes(x)
  cat('attr names:', paste(names(a), collapse=' | '), '\\n')
  for (nm in names(a)) {
    z <- a[[nm]]
    ty <- typeof(z); le <- length(z); di <- tryCatch(paste(dim(z),collapse=' x '),error=function(e)'<err>'); na <- tryCatch(paste(head(names(z),20),collapse=','),error=function(e)'<err>')
    cl <- if (ty %in% c('list','character','integer','double','logical','raw','NULL')) paste(class(z),collapse=',') else '<S4/other>'
    cat('attr',nm,'typeof',ty,'class',cl,'len',le,'dim',di,'names',na,'\\n')
  }
  z <- a[['listData']]
  cat('listData typeof',typeof(z),'len',length(z),'names',paste(names(z),collapse=' | '),'\\n')
  if (is.list(z)) {
    for (nn in names(z)) {
      zz <- z[[nn]]
      cat('  ',nn,'typeof',typeof(zz),'len',length(zz),'dim',paste(dim(zz),collapse=' x '),'head',paste(head(zz,5),collapse=' | '),'\\n')
    }
  }
  em <- a[['elementMetadata']]
  cat('elementMetadata typeof',typeof(em),'len',length(em),'attr names',paste(names(attributes(em)),collapse=' | '),'\\n')
}
RS"
