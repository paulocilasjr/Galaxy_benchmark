/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
 cat('\\n===', basename(f), '===\\n')
 x <- readRDS(f)
 tests <- list(
   isS4=function() isS4(x),
   attr=function() attributes(x),
   slotNames=function() slotNames(x),
   names=function() names(x),
   length=function() length(x),
   dim=function() dim(x),
   unclass=function() unclass(x),
   str=function() str(x, max.level=1)
 )
 for (nm in names(tests)) {
   cat('--',nm,'--\\n')
   z <- tryCatch(tests[[nm]](), error=function(e) paste('ERROR:', conditionMessage(e)))
   if (is.character(z) && length(z)==1 && startsWith(z,'ERROR:')) print(z) else { print(z); }
 }
}
RS"
