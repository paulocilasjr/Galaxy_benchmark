/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  cat('isS4', isS4(x), '\\n')
  a <- attributes(x)
  cat('attr names:', paste(names(a), collapse=' | '), '\\n')
  for (nm in names(a)) {
    z <- a[[nm]]
    cat('attr',nm,'typeof',typeof(z),'class',paste(class(z),collapse=','),'len',length(z),'dim',paste(dim(z),collapse=' x '),'names',paste(head(names(z),20),collapse=','),'\\n')
  }
  for (nm in c('listData','rownames','nrows','elementType','elementMetadata','priorInfo')) {
    cat('slot/attr',nm,': ')
    z <- tryCatch(slot(x,nm), error=function(e) tryCatch(attr(x,nm), error=function(e2) paste0('ERR ',conditionMessage(e2))))
    cat('typeof',typeof(z),'class',paste(class(z),collapse=','),'len',length(z),'dim',paste(dim(z),collapse=' x '),'names',paste(head(names(z),20),collapse=','),'\\n')
    if (nm == 'listData' && is.list(z)) {
      for (nn in names(z)) { zz <- z[[nn]]; cat('  ',nn,'type',typeof(zz),'class',paste(class(zz),collapse=','),'len',length(zz),'head',paste(head(zz,3),collapse=' | '),'\\n') }
    }
  }
}
RS"
