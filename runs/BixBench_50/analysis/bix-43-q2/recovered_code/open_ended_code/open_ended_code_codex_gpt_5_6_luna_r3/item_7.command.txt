/bin/bash -lc "python - <<'PY'
import importlib.util
mods=['pandas','numpy','scipy','statsmodels','pydeseq2','gseapy','requests','mygene']
for m in mods:
    s=importlib.util.find_spec(m)
    print(m, s.origin if s else None)
    if s:
        try:
            mod=__import__(m)
            print('  version',getattr(mod,'__version__','?'))
        except Exception as e: print('  import error',type(e).__name__,str(e)[:160])
PY"
