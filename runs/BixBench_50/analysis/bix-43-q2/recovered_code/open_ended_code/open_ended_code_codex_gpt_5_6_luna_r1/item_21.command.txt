/bin/bash -lc "python -u - <<'PY'
import pandas as pd
import numpy as np
import gseapy as gp
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
layout = pd.read_csv('inputs/sample_layout.csv')
layout['count_col'] = layout['SampleID'].str.replace('-', '_', regex=False)
counts = counts.loc[(counts > 10).any(axis=1)]
wanted = ['DMSO','DMSO_Serum_starvation','Cisplatin_IC50_CBD_IC50','Cisplatin_IC50_CBD_IC50_Serum_starvation_16h']
sel = layout[layout['Group'].isin(wanted)].copy()
sel['Group'] = pd.Categorical(sel['Group'], categories=wanted, ordered=True)
sel = sel.sort_values(['Group','SampleID'])
metadata = sel.set_index('count_col')[['Group']]
counts = counts[sel['count_col'].tolist()].T
print('DE start',counts.shape,flush=True)
dds = DeseqDataSet(counts=counts, metadata=metadata, design_factors='Group', refit_cooks=True, n_cpus=4, quiet=True)
dds.deseq2()
stat_res = DeseqStats(dds, contrast=['Group','Cisplatin_IC50_CBD_IC50','DMSO'], alpha=0.05, quiet=True, n_cpus=4)
stat_res.summary()
res = stat_res.results_df
sig = res.loc[res['padj'].notna() & (res['padj']<=0.05) & (res['log2FoldChange'].abs()>=0.5) & (res['baseMean']>=10)].copy()
print('DE selected',len(sig),flush=True)
mapdf = pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t',dtype=str).set_index('ENSG')['gene_name']
genes = mapdf.reindex(sig.index)
print('mapped missing',int(genes.isna().sum()),'symbols total',len(genes),'unique',genes.nunique(),'duplicates',len(genes)-genes.nunique(),flush=True)
gene_list = genes.dropna().tolist()
lib = gp.get_library('Reactome_2022', organism='Human')
term='TP53 Regulates Transcription Of Cell Cycle Genes R-HSA-6791312'
print('target genes in list',len(set(gene_list)&set(lib[term])),sorted(set(gene_list)&set(lib[term])),flush=True)
print('enrichr call',flush=True)
enr = gp.enrichr(gene_list=gene_list, gene_sets='Reactome_2022', organism='human', outdir=None, cutoff=1.0, no_plot=True, verbose=True)
print('columns',enr.results.columns.tolist(),flush=True)
rows = enr.results[enr.results['Term'].str.contains('TP53|p53|Cell Cycle',case=False,regex=True,na=False)]
print(rows.to_string(index=False),flush=True)
PY"
