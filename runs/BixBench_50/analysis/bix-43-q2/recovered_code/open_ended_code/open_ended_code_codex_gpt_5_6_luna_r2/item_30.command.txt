/bin/bash -lc "PYTHONPATH=/workspace/.deps python - <<'PY'
import pandas as pd, numpy as np, warnings, gseapy
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
warnings.filterwarnings('ignore')
counts=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0)
layout=pd.read_csv('inputs/sample_layout.csv')
layout['count_col']=layout.SampleID.str.replace('-','_',regex=False)
keep=(counts>10).any(axis=1)
cc=counts.loc[keep,layout.count_col.tolist()].T
cc.index=layout.SampleID.tolist()
md=layout.set_index('SampleID')[['Group']].rename(columns={'Group':'condition'}).loc[cc.index]
print('all-group fit',cc.shape,md.condition.nunique(),flush=True)
dds=DeseqDataSet(counts=cc,metadata=md,design='~condition',ref_level=['condition','DMSO'],quiet=True,n_cpus=1)
dds.deseq2()
ds=DeseqStats(dds,contrast=['condition','Cisplatin_IC50_CBD_IC50','DMSO'],quiet=True,n_cpus=1); ds.summary()
res=ds.results_df
sel=res[(res.padj<=0.05)&(res.log2FoldChange.abs()>=0.5)&(res.baseMean>=10)]
ann=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t').set_index('ENSG')
names=sel.join(ann).gene_name.dropna().drop_duplicates().tolist()
lib=gseapy.get_library('Reactome_2022',organism='human')
term=[k for k in lib if k.startswith('TP53 Regulates Transcription Of Cell Cycle Genes ')][0]
path=set(lib[term]); hits=set(names)&path
N=20000;m=len(path);k=len(names);x=len(hits)
orr=x*(N-m-k+x)/((m-x)*(k-x)) if (m-x)*(k-x) else float('inf')
print('all-group selected',len(sel),'mapped',k,'up',sum(sel.log2FoldChange>0),'down',sum(sel.log2FoldChange<0),'overlap',x,'OR_N20000',repr(orr),'hits',sorted(hits))
print('target size',m)
PY"
