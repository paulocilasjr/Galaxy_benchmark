/bin/bash -lc "python - <<'PY'
import pandas as pd
import gseapy as gp
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts_all=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0)
layout=pd.read_csv('inputs/sample_layout.csv')
layout['sample']=layout['SampleID'].str.replace('-','_',regex=False)
layout=layout.set_index('sample')
counts_all=counts_all.loc[counts_all.gt(10).any(axis=1)]
fit_groups=['DMSO','DMSO_Serum_starvation','Cisplatin_IC50_CBD_IC50','Cisplatin_IC50_CBD_IC50_Serum_starvation_16h']
meta=layout.loc[layout.Group.isin(fit_groups),['Group']].copy()
meta['Group']=pd.Categorical(meta.Group,categories=fit_groups)
counts=counts_all.loc[:,meta.index].T
dds=DeseqDataSet(counts=counts,metadata=meta,design='~Group',n_cpus=4,quiet=True)
dds.deseq2()
stats=DeseqStats(dds,contrast=['Group','Cisplatin_IC50_CBD_IC50','DMSO'],alpha=0.05,n_cpus=4,quiet=True)
stats.summary()
res=stats.results_df
sig=res[(res.padj<=0.05)&(res.log2FoldChange.abs()>=0.5)&(res.baseMean>=10)]
ann=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t').set_index('ENSG')
fg=sig.join(ann,how='left').gene_name.dropna().drop_duplicates().tolist()
print('FOREGROUND',len(sig),len(fg),flush=True)
enr=gp.enrichr(gene_list=fg,gene_sets='Reactome_2022',organism='human',outdir=None,no_plot=True,verbose=True)
out=enr.results
print('ROWS_COLS',out.shape,list(out.columns),flush=True)
target='TP53 Regulates Transcription Of Cell Cycle Genes R-HSA-6791312'
sel=out.loc[out.Term.eq(target)]
print('TARGET_MATCHES',len(sel),flush=True)
print(sel.to_string(index=False),flush=True)
PY"
