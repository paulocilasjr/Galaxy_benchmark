/bin/bash -c "python - <<'PY'
import pandas as pd
from pathlib import Path
counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
print('counts shape', counts.shape)
print('columns', list(counts.columns))
print('index head', list(counts.index[:5]))
print('duplicated index', counts.index.duplicated().sum())
print('non-integer?', counts.dtypes.value_counts().to_dict())
layout = pd.read_csv('inputs/sample_layout.csv')
print(layout)
# mapping header names
for s in layout['SampleID']:
    print(s, s.replace('-','_') in counts.columns)
print('all match', all(s.replace('-','_') in counts.columns for s in layout['SampleID']))
# prefilter all 30
keep = (counts > 10).any(axis=1)
print('prefilter keep', keep.sum(), 'out of', len(counts))
# subset groups
sel_groups = ['DMSO','DMSO_Serum_starvation','Cisplatin_IC50_CBD_IC50','Cisplatin_IC50_CBD_IC50_Serum_starvation_16h']
sel = layout[layout.Group.isin(sel_groups)].copy()
cols = [s.replace('-','_') for s in sel.SampleID]
print('sel samples', len(sel), sel)
sub = counts.loc[keep, cols]
print('subset shape', sub.shape)
print('any NA', sub.isna().sum().sum())
print('min', sub.values.min(), 'max', sub.values.max())
print('sums by group')
# show mapping coverage
mp = pd.read_csv('inputs/ensg_to_gene_name.tsv', sep='\\t')
print('mapping shape', mp.shape, 'unique ensg', mp.ENSG.nunique(), 'unique gene', mp.gene_name.nunique())
print('dup ensg', mp.ENSG.duplicated().sum(), 'dup gene', mp.gene_name.duplicated().sum())
print('missing genes', sub.index[~sub.index.isin(mp.ENSG)].shape)
PY"
