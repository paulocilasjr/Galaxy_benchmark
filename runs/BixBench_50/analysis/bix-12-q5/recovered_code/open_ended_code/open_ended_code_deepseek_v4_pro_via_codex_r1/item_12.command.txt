/bin/bash -c "cd /workspace && python3 - <<'PY'
import glob, os
from collections import Counter

def parse(f):
    seqs=[]; cur=[]
    for line in open(f):
        if line.startswith('>'):
            if cur: seqs.append(''.join(cur)); cur=[]
        else:
            s=line.strip()
            if s: cur.append(s)
    if cur: seqs.append(''.join(cur))
    return seqs

for kind in ['tmp/sca','tmp/scf']:
    print('\\n',kind)
    res=[]
    for f in glob.glob(kind+'/*.mafft'):
        seqs=parse(f); n=len(seqs)
        if n"'!=4: continue
        L=len(seqs[0])
        assert all(len(s)==L for s in seqs)
        # gap as state
        counts=[]
        # gaps ignored for each column (treat gap missing)
        counts_ignore=[]
        # only columns no gaps, counts aa
        counts_nogap=[]
        for j in range(L):
            chars=[s[j] for s in seqs]
            cnt=Counter(chars)
            # informative standard each state count>=2
            states=[c for c,v in cnt.items() if v>=2]
            counts.append(len(states)>=2)
            non_gap=[c for c in chars if c!='"'-']
            if len(non_gap)>=4:
                cnt2=Counter(non_gap)
                states2=[c for c,v in cnt2.items() if v>=2]
                counts_ignore.append(len(states2)>=2)
            elif len(non_gap)>=2:
                # What definition ignoring gaps? can't have each state count 2 unless all 4 non-gap? with 3 non-gap one singleton
                cnt2=Counter(non_gap); states2=[c for c,v in cnt2.items() if v>=2]
                counts_ignore.append(len(states2)>=2)
            else:
                counts_ignore.append(False)
            if all(c"'!='"'-' for c in chars):
                cnt3=Counter(chars); states3=[c for c,v in cnt3.items() if v>=2]
                counts_nogap.append(len(states3)>=2)
            else:
                counts_nogap.append(False)
        res.append((sum(counts),sum(counts_ignore),sum(counts_nogap),L,n,os.path.basename(f)))
    for i,(a,b,c,L,n,name) in enumerate(sorted(res, reverse=True)[:30],1):
        print(i,name,L,a,b,c)
    print('max gapstate',max(r[0] for r in res),'max ignore',max(r[1] for r in res),'max nogap',max(r[2] for r in res),'num',len(res))
PY"
