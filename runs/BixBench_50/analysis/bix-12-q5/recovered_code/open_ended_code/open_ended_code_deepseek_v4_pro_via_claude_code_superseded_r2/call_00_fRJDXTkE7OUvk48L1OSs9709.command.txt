import zipfile
from collections import Counter

def parse_fasta(text):
    entries = []
    current_header = None
    current_seq = []
    for line in text.split('\n'):
        line = line.strip()
        if not line:
            continue
        if line.startswith('>'):
            if current_header is not None:
                entries.append((current_header, ''.join(current_seq)))
            current_header = line[1:].strip()
            current_seq = []
        else:
            current_seq.append(line)
    if current_header is not None:
        entries.append((current_header, ''.join(current_seq)))
    return entries

def count_parsimony_informative_sites(sequences):
    if not sequences:
        return 0
    seqlen = len(sequences[0])
    for s in sequences:
        if len(s) != seqlen:
            raise ValueError('Sequences have different lengths')
    
    count = 0
    for i in range(seqlen):
        col = [s[i] for s in sequences]
        counter = Counter(col)
        for gap_char in ['-', '.', '?', 'X']:
            counter.pop(gap_char, None)
        if len(counter) >= 2 and all(v >= 2 for v in counter.values()):
            count += 1
    return count

with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    mafft_files = sorted([n for n in z.namelist() if n.endswith('.faa.mafft')])
    
    results = []
    for mf in mafft_files:
        with z.open(mf) as f:
            content = f.read().decode('utf-8')
        entries = parse_fasta(content)
        seqs = [s for _, s in entries]
        pis = count_parsimony_informative_sites(seqs)
        results.append((pis, mf.replace('.faa.mafft', ''), len(seqs), len(seqs[0]) if seqs else 0))
    
    # Sort by PIS descending
    results.sort(key=lambda x: x[0], reverse=True)
    
    print("Top 20 genes by parsimony informative sites:")
    print(f"{'Rank':<6} {'PIS':<8} {'Gene':<25} {'Seqs':<6} {'Cols':<8}")
    for rank, (pis, gene, nseqs, cols) in enumerate(results[:20], 1):
        print(f"{rank:<6} {pis:<8} {gene:<25} {nseqs:<6} {cols:<8}")
    
    # Show distribution of PIS
    from collections import Counter as Ctr
    pis_dist = Ctr(r[0] for r in results)
    print(f"\nPIS distribution:")
    for pis in sorted(pis_dist):
        print(f"  {pis}: {pis_dist[pis]} genes")
