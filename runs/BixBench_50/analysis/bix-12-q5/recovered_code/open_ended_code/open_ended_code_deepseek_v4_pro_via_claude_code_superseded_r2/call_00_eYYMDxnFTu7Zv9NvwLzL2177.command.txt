import zipfile
from collections import Counter

def parse_fasta(text):
    entries = []
    current_header = None
    current_seq = []
    for line in text.split('\\n'):
        line = line.strip()
        if not line:
            continue
        if line.startswith('>'):
            if current_header is not None:
                entries.append((current_header, ''.join(current_seq)))
            current_header = line[1:].strip()
            current_seq = []
        else:
            current_seq.append(line)
    if current_header is not None:
        entries.append((current_header, ''.join(current_seq)))
    return entries

with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    mafft_files = [n for n in z.namelist() if n.endswith('.faa.mafft')]
    
    # Check seq counts distribution
    seq_counts = Counter()
    for mf in mafft_files:
        with z.open(mf) as f:
            content = f.read().decode('utf-8')
        entries = parse_fasta(content)
        seq_counts[len(entries)] += 1
    
    print('Distribution of sequence counts per alignment:')
    for k in sorted(seq_counts):
        print(f'  {k} seqs: {seq_counts[k]} alignments')

    # Show the max PIS gene details
    with z.open('1428854at2759.faa.mafft') as f:
        content = f.read().decode('utf-8')
    entries = parse_fasta(content)
    print(f'\\nMax PIS gene: 1428854at2759, {len(entries)} seqs, {len(entries[0][1])} cols')
    for h, s in entries:
        print(f'  {h}: len={len(s)}')
