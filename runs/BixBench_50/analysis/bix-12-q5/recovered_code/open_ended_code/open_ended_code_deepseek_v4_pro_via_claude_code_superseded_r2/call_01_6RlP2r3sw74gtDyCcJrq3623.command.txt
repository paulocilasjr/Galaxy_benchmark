import zipfile

# Check what's in scogs_animals.zip - are there nucleotide alignments too?
with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    names = z.namelist()
    # Show all unique extensions
    exts = set()
    for n in names:
        parts = n.rsplit('.', 1)
        if len(parts) > 1:
            exts.add(parts[1])
        else:
            exts.add('(none)')
    print('Unique extensions:')
    for e in sorted(exts):
        count = sum(1 for n in names if n.endswith('.' + e) if e != '(none)')
        print(f'  .{e}: {count}')
    
    # Check for nucleotide files
    nuc = [n for n in names if 'fna' in n or 'nuc' in n.lower()]
    print(f'\\nNucleotide files: {nuc}')
