/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse(s):
    out=[]; name=None; parts=[]
    for line in s.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: out.append((name,''.join(parts)))
            name=line[1:].split()[0]; parts=[]
        else: parts.append(line)
    if name is not None: out.append((name,''.join(parts)))
    return out

rows=[]; alphabet=Counter(); bad=[]
with ZipFile('inputs/scogs_animals.zip') as z:
  for n in z.namelist():
    if not n.endswith('.faa.mafft'): continue
    recs=parse(z.read(n).decode())
    lens={len(s) for _,s in recs}
    if len(recs)"'!=4 or len(lens)!=1: bad.append((n,len(recs),lens)); continue
    seqs=[s.upper() for _,s in recs]
    alphabet.update('"''.join(seqs))
    vals={}
    for mode in ['gap_state','nongap_complete','missing_gap_x','all_symbol']:
      k=0
      for col in zip(*seqs):
        if mode=='gap_state':
          obs=[x for x in col if x not in 'X?*']
        elif mode=='nongap_complete':
          if any(x in '-X?*' for x in col): continue
          obs=list(col)
        elif mode=='missing_gap_x':
          obs=[x for x in col if x not in '-X?*']
        else:
          obs=list(col)
        c=Counter(obs)
        if sum(v>=2 for v in c.values())>=2: k+=1
      vals[mode]=k
    rows.append((n,len(seqs[0]),vals,recs))
print('files',len(rows),'bad',bad)
print('alphabet',alphabet)
for mode in ['gap_state','nongap_complete','missing_gap_x','all_symbol']:
  top=sorted(rows,key=lambda x:x[2][mode],reverse=True)[:15]
  print('\\nMODE',mode,'MAX',top[0][2][mode])
  for n,L,v,recs in top[:10]: print(n,'L',L,'PI',v[mode], 'other',v)
PY"
