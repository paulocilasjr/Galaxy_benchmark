/bin/bash -c "python - <<'PY'
from pathlib import Path
import collections
for kind in ['scogs_animals','scogs_fungi']:
    d=Path('/workspace/work')/kind
    chars=collections.Counter()
    for f in d.glob('*.mafft'):
        lines=f.read_text().splitlines()
        for l in lines:
            if l.startswith('>'): continue
            chars.update(l.rstrip())
    print(kind, ''.join(sorted(chars)), {c:n for c,n in chars.most_common()})
PY"
