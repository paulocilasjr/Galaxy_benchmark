/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import Path
zpath=Path('inputs/scogs_animals.zip')

def parse_fasta(text):
    records=[]
    name=None; parts=[]
    for line in text.splitlines():
        if line.startswith('>'):
            if name is not None: records.append((name,''.join(parts)))
            name=line[1:].strip(); parts=[]
        elif line.strip(): parts.append(line.strip())
    if name is not None: records.append((name,''.join(parts)))
    return records

rows=[]; invalid=[]; symbols=Counter()
with ZipFile(zpath) as z:
    names=[n for n in z.namelist() if n.endswith('.faa.mafft')]
    for n in names:
        rec=parse_fasta(z.read(n).decode())
        lens={len(s) for _,s in rec}
        if len(rec)"'!=4 or len(lens)!=1:
            invalid.append((n,len(rec),lens)); continue
        seqs=[s.upper() for _,s in rec]
        symbols.update('"''.join(seqs))
        pi_all=pi_nogap=pi_canonical=0
        for col in zip(*seqs):
            vals=Counter(col)
            if sum(v>=2 for v in vals.values())>=2: pi_all+=1
            vals=Counter(c for c in col if c not in '-?')
            if sum(v>=2 for v in vals.values())>=2: pi_nogap+=1
            vals=Counter(c for c in col if c in 'ACDEFGHIKLMNPQRSTVWY')
            if sum(v>=2 for v in vals.values())>=2: pi_canonical+=1
        rows.append((n,len(seqs[0]),pi_all,pi_nogap,pi_canonical,[x for x,_ in rec]))
print('files',len(names),'valid',len(rows),'invalid',invalid)
print('symbols',dict(symbols))
for idx,label in [(2,'all symbols'),(3,'exclude gap/?'),(4,'canonical amino acids')]:
    m=max(r[idx] for r in rows)
    best=[r for r in rows if r[idx]==m]
    print(label,'max',m,'best_count',len(best))
    for r in best[:10]: print(' ',r)
print('top 15 by canonical')
for r in sorted(rows,key=lambda x:(-x[4],x[0]))[:15]: print(r[:5])
PY"
