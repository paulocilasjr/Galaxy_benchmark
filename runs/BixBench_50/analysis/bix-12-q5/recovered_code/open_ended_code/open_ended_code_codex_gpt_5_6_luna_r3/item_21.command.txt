/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse(b):
    out=[]; name=None; s=[]
    for line in b.decode().splitlines():
        if line.startswith('>'):
            if name is not None: out.append((name,''.join(s)))
            name=line[1:].split()[0]; s=[]
        elif line.strip(): s.append(line.strip())
    if name is not None: out.append((name,''.join(s)))
    return out

def std(a):
    n=0; idx=[]; patterns=[]
    for i,col in enumerate(zip(*(s for _,s in a)),1):
        c=Counter(x for x in col if x"'!='"'-')
        if sum(v>=2 for v in c.values())>=2:
            n+=1; idx.append(i); patterns.append(''.join(col))
    return n,idx,patterns
with ZipFile('inputs/scogs_animals.zip') as z:
  a=parse(z.read('1428854at2759.faa.mafft'))
  n,idx,pats=std(a)
  print('taxa',[x[0] for x in a], 'length',len(a[0][1]), 'std PI',n)
  print('PI indices',idx)
  print('PI patterns',pats)
  print('pattern counts',Counter(pats))
PY"
