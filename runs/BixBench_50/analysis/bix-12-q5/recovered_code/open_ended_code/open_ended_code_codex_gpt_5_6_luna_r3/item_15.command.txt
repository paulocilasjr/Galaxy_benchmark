/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter


def parse_fasta(data):
    seqs=[]; name=None; chunks=[]
    for line in data.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else: chunks.append(line)
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs

def pi_count(seqs, ignore=frozenset('-?Xx')):
    if not seqs: return 0
    L={len(s) for _,s in seqs}
    if len(L)"'!=1: raise ValueError(L)
    n=0
    for col in zip(*(s for _,s in seqs)):
        c=Counter(x for x in col if x not in ignore)
        if sum(v>=2 for v in c.values())>=2: n+=1
    return n

def pi_count_allchars(seqs):
    if not seqs: return 0
    n=0
    for col in zip(*(s for _,s in seqs)):
        c=Counter(col)
        if sum(v>=2 for v in c.values())>=2: n+=1
    return n

rows=[]
with ZipFile('"'inputs/scogs_animals.zip') as z:
    for fn in z.namelist():
        if not fn.endswith('.faa.mafft'): continue
        seqs=parse_fasta(z.read(fn))
        chars=Counter(''.join(s for _,s in seqs))
        rows.append((fn,len(seqs),len(seqs[0][1]) if seqs else 0,pi_count(seqs),pi_count_allchars(seqs),''.join(sorted(chars))))

print('alignments',len(rows))
print('taxa counts',Counter(r[1] for r in rows))
print('length range',min(r[2] for r in rows),max(r[2] for r in rows))
print('alphabet union',''.join(sorted(set(''.join(r[5] for r in rows)))))
print('top standard:')
for r in sorted(rows,key=lambda x:(x[3],x[2],x[0]),reverse=True)[:25]: print(r)
print('top allchars:')
for r in sorted(rows,key=lambda x:(x[4],x[2],x[0]),reverse=True)[:25]: print(r)
print('max standard',max(r[3] for r in rows))
print('max allchars',max(r[4] for r in rows))
PY"
