/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for fn in ['inputs/scogs_animals.zip']+[f'inputs/Animals_{x}.busco.zip' for x in ['Cele','Ggal','Mmus','Xlae']]:
    with ZipFile(fn) as z:
        names=z.namelist()
    print('\\n',fn,'members',len(names))
    print(Counter(n.rsplit('/',1)[-1].split('.',1)[-1] if '.' in n.rsplit('/',1)[-1] else '<noext>' for n in names).most_common(20))
    print('alignment-like',[n for n in names if any(k in n.lower() for k in ['mafft','align','aln'])][:10], 'count',sum(any(k in n.lower() for k in ['mafft','align','aln']) for n in names))
PY"
