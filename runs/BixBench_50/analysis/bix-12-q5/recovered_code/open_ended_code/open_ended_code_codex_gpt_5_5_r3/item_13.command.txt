/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter

zip_path='inputs/scogs_animals.zip'

def parse_fasta(data):
    seqs=[]; name=None; parts=[]
    for raw in data.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                seqs.append((name, ''.join(parts)))
            name=line[1:].split()[0]
            parts=[]
        else:
            parts.append(line)
    if name is not None:
        seqs.append((name, ''.join(parts)))
    return seqs

def pi_count(seqs, gaps_missing=True):
    lengths={len(s) for _,s in seqs}
    if len(lengths)"'!=1:
        raise ValueError(f'"'unequal lengths {lengths}')
    total=0
    for col in zip(*(s for _,s in seqs)):
        if gaps_missing:
            states=[c.upper() for c in col if c not in '-?Xx']
        else:
            states=[c.upper() for c in col if c "'!= '"'?']
        cnt=Counter(states)
        if sum(1 for v in cnt.values() if v>=2) >= 2:
            total += 1
    return total

rows=[]
bad=[]
with zipfile.ZipFile(zip_path) as z:
    for n in z.namelist():
        if not n.endswith('.faa.mafft'):
            continue
        seqs=parse_fasta(z.read(n))
        taxa=[t for t,_ in seqs]
        lengths={len(s) for _,s in seqs}
        if len(seqs)"'!=4 or len(lengths)!=1:
            bad.append((n,len(seqs),lengths,taxa))
        rows.append((n, len(seqs), next(iter(lengths)) if len(lengths)==1 else None, pi_count(seqs, True), pi_count(seqs, False)))

print('"'alignments',len(rows),'bad',len(bad))
print('top gaps_missing')
for r in sorted(rows, key=lambda x:x[3], reverse=True)[:15]: print(r)
print('top gaps_as_state')
for r in sorted(rows, key=lambda x:x[4], reverse=True)[:15]: print(r)
print('max_missing', max(r[3] for r in rows), 'max_gapstate', max(r[4] for r in rows))
print('bad_examples', bad[:5])
PY"
