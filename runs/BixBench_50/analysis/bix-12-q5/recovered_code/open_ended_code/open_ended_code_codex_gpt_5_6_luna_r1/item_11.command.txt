from zipfile import ZipFile
from collections import Counter

path='inputs/scogs_animals.zip'
def fasta(data):
    recs=[]; h=None; seq=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if h is not None: recs.append((h,''.join(seq)))
            h=line[1:].strip(); seq=[]
        elif line.strip(): seq.append(line.strip())
    if h is not None: recs.append((h,''.join(seq)))
    return recs

def pis(seq_list, gap_missing=True):
    n=len(seq_list); L=len(seq_list[0]) if n else 0
    counts=[]
    for j in range(L):
        col=[s[j] for s in seq_list]
        if gap_missing: col=[c for c in col if c not in '-?X']

        if sum(v>=2 for v in c.values()) >= 2:
            counts.append(j+1)
    return counts
