# Let me do a quick sanity check: create a small test alignment and verify the count
from collections import Counter

# Test 1: Simple case
test_seqs = [
    "AABB",  # A, B each >=2 -> informative
    "AABC",  # A=2, B=1, C=1 -> NOT informative (only A >=2)
    "ABCC",  # C=2, A=1, B=1 -> NOT informative
    "ABCD",  # all 1 -> NOT informative
    "AAAA",  # only 1 state -> NOT informative
    "A-BB",  # A=2, B=2, -=1 -> A and B >=2 -> informative (gaps are a state, but does '-' count?)
    "A--B",  # A=1, B=1, -=2 -> only '-' >=2 -> NOT informative
]

for test in test_seqs:
    counts = Counter(test)
    valid = [c for c, cnt in counts.items() if cnt >= 2]
    is_informative = len(valid) >= 2
    print(f"  {test}: counts={dict(counts)}, valid_states={valid}, informative={is_informative}")
    
print("\nExpected: 1st=informative, 6th depends on gap treatment")
