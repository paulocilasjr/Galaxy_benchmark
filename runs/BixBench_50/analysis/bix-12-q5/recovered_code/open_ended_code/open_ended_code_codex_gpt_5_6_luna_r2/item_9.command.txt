/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse_fasta(data):
    name=None; seq=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if name is not None: yield name, ''.join(seq)
            name=line[1:].strip(); seq=[]
        else: seq.append(line.strip())
    if name is not None: yield name, ''.join(seq)

def counts(rec):
    cols=zip(*(s for _,s in rec))
    incl=excl=0; patterns=[]
    for i,col in enumerate(cols):
        ci=Counter(col)
        inf_incl=sum(v>=2 for v in ci.values())>=2
        obs=[x for x in col if x"'!='"'-']
        ce=Counter(obs)
        inf_excl=sum(v>=2 for v in ce.values())>=2
        incl += inf_incl; excl += inf_excl
        if inf_incl or inf_excl: patterns.append((i+1,''.join(col),inf_incl,inf_excl,dict(ci)))
    return incl,excl,patterns
with ZipFile('inputs/scogs_animals.zip') as z:
    rows=[]
    for n in z.namelist():
        if n.endswith('.faa.mafft'):
            rec=list(parse_fasta(z.read(n)))
            a,b,p=counts(rec)
            rows.append((a,b,n,rec,p))
rows.sort(reverse=True)
print('top by gaps-as-state')
for a,b,n,rec,p in rows[:30]:
    if a or b:
        print(f'{n}\\ttaxa={len(rec)}\\tlen={len(rec[0][1])}\\tincl={a}\\texcl={b}\\tfirst={p[:4]}')
print('nonzero total',sum(a>0 for a,_,*x in rows),'top excl',max(b for a,b,*x in rows),'max incl',max(a for a,b,*x in rows))
print('top by gaps-missing')
for a,b,n,rec,p in sorted(rows,key=lambda x:(x[1],x[0],x[2]),reverse=True)[:30]:
    if b or a:
        print(f'{n}\\ttaxa={len(rec)}\\tlen={len(rec[0][1])}\\tincl={a}\\texcl={b}')
PY"
