/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import PurePosixPath

p='inputs/scogs_animals.zip'

def parse_fasta(data):
    name=None; seq=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if name is not None:
                yield name, ''.join(seq)
            name=line[1:].strip().split()[0]
            seq=[]
        else:
            seq.append(line.strip())
    if name is not None:
        yield name, ''.join(seq)

with ZipFile(p) as z:
    aligned=[n for n in z.namelist() if n.endswith('.faa.mafft')]
    print('aligned files',len(aligned))
    taxon_counts=Counter(); lengths=Counter(); alph=Counter(); bad=[]; amb=Counter(); gaps=0
    rows=[]
    for n in aligned:
        rec=list(parse_fasta(z.read(n)))
        taxon_counts[len(rec)]+=1
        ls={len(s) for _,s in rec}; lengths.update(ls)
        chars=Counter(c for _,s in rec for c in s); alph.update(chars)
        if len(ls)"'!=1: bad.append((n, [len(s) for _,s in rec]))
        amb.update(c for c in chars if c not in set('"'ACDEFGHIKLMNPQRSTVWY-'))
        gaps += chars.get('-',0)
        rows.append((n,rec,len(rec),next(iter(ls))))
    print('taxon counts',taxon_counts)
    print('alignment lengths',lengths.most_common(20), 'distinct',len(lengths))
    print('alphabet', ''.join(sorted(alph)), 'counts', dict(sorted(alph.items())))
    print('nonstandard',dict(amb), 'gap chars',gaps, 'bad lengths',len(bad))
    print('first/last', aligned[:3], aligned[-3:])
PY"
