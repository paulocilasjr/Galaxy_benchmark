/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter


def fasta_records(raw):
    records=[]
    name=None
    seq=[]
    for line in raw.decode('ascii').splitlines():
        line=line.strip()
        if not line:
            continue
        if line[0]=='>':
            if name is not None:
                records.append((name,''.join(seq)))
            name=line[1:].split()[0]
            seq=[]
        else:
            seq.append(line)
    if name is not None:
        records.append((name,''.join(seq)))
    return records

results=[]
with ZipFile('inputs/scogs_animals.zip') as z:
    for name in z.namelist():
        if not name.endswith('.faa.mafft'):
            continue
        recs=fasta_records(z.read(name))
        assert recs and len({len(s) for _,s in recs})==1
        pis=0
        for column in zip(*(s for _,s in recs)):
            states=Counter(c for c in column if c "'!= '"'-')
            if sum(n >= 2 for n in states.values()) >= 2:
                pis += 1
        results.append((pis,name,len(recs),len(recs[0][1])))

m=max(x[0] for x in results)
print('aligned files:', len(results))
print('maximum:', m)
print('maximal files:', [(n,t,l) for p,n,t,l in results if p==m])
print('second maximum:', sorted((p,n) for p,n,_,_ in results)[-5:])
assert m == 29
PY"
