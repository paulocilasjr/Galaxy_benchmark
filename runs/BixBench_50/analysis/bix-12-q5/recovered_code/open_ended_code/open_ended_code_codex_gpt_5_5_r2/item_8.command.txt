/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter
zp='inputs/scogs_animals.zip'

def read_fasta(data):
    seqs=[]; name=None; chunks=[]
    for line in data.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else:
            chunks.append(line)
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs

def pis(seqs, include_gap=False):
    if not seqs: return None
    lens={len(s) for _,s in seqs}
    if len(lens)"'!=1: raise ValueError(('"'unaligned', lens))
    n=0
    for col in zip(*(s for _,s in seqs)):
        if include_gap:
            states=[c.upper() for c in col if c not in {'?'}]
        else:
            states=[c.upper() for c in col if c not in {'-','?'}]
        cnt=Counter(states)
        if sum(1 for v in cnt.values() if v>=2) >= 2:
            n+=1
    return n

with zipfile.ZipFile(zp) as z:
    names=z.namelist()
    mafft=[n for n in names if n.endswith('.faa.mafft')]
    raw=[n for n in names if n.endswith('.faa')]
    print('members', len(names), 'mafft', len(mafft), 'raw', len(raw))
    rows=[]
    bad=[]
    for n in mafft:
        seqs=read_fasta(z.read(n))
        lens={len(s) for _,s in seqs}
        if len(lens)"'!=1: bad.append((n, len(seqs), sorted(lens)[:5]))
        rows.append((pis(seqs, False), pis(seqs, True), n, len(seqs), next(iter(lens)) if len(lens)==1 else None))
    print('"'bad alignments', bad[:5], 'count', len(bad))
    rows.sort(reverse=True)
    print('top ignore gaps:')
    for r in rows[:15]: print(r)
    rows_gap=sorted(rows, key=lambda x:x[1], reverse=True)
    print('top include gaps:')
    for r in rows_gap[:15]: print(r)
    print('seq counts', Counter(r[3] for r in rows))
PY"
