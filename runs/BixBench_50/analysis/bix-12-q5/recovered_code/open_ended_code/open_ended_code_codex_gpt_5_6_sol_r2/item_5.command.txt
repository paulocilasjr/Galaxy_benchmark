/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for zp in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with ZipFile(zp) as z:
        names=[n for n in z.namelist() if not n.endswith('/')]
        kinds=Counter('mafft' if n.endswith('.faa.mafft') else 'faa' if n.endswith('.faa') else 'other' for n in names)
        print(zp, len(names), dict(kinds), 'comment=',repr(z.comment))
        for n in [x for x in names if x.endswith('.faa.mafft')][:3]:
            print('\\n###',n)
            print(z.read(n).decode()[:800])
PY"
