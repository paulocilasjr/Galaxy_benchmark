/bin/bash -lc "python3 - <<'PY'
import zipfile, collections

def parse_fasta(b):
    seqs=[]; name=None; chunks=[]
    for raw in b.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line[0]=='>':
            if name is not None: seqs.append((name, ''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else: chunks.append(line)
    if name is not None: seqs.append((name, ''.join(chunks)))
    return seqs

with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    mafft=[n for n in z.namelist() if n.endswith('.faa.mafft')]
    raw=[n for n in z.namelist() if n.endswith('.faa')]
    raw_equal=sum(1 for n in raw if len({len(s) for _,s in parse_fasta(z.read(n))})==1)
    print('raw_equal_length_files', raw_equal, 'of', len(raw))
    best=(-1,None)
    distribution=collections.Counter()
    for n in mafft:
        seqs=parse_fasta(z.read(n))
        assert len(seqs)==4, (n,len(seqs))
        Ls={len(s) for _,s in seqs}
        assert len(Ls)==1, (n,Ls)
        count=0
        for i in range(next(iter(Ls))):
            c=collections.Counter(s[i].upper() for _,s in seqs if s[i] not in '-?*.Xx')
            inf=sum(v>=2 for v in c.values())>=2
            count += inf
        distribution[count]+=1
        if count>best[0]: best=(count,n)
    print('best',best)
    print('num_best', distribution[best[0]])
    print('top_counts', sorted(distribution.items(), reverse=True)[:8])
PY"
