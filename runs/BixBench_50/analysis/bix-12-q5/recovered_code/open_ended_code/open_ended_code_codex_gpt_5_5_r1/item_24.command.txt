/bin/bash -lc "python3 - <<'PY'
import zipfile, collections

def parse(b):
    seqs=[]; name=None; chunks=[]
    for line in b.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else: chunks.append(line)
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs

def pi(seqs):
    lens={len(s) for _,s in seqs}
    if len(lens)"'!=1: return None
    total=0
    for col in zip(*(s for _,s in seqs)):
        c=collections.Counter(ch.upper() for ch in col if ch not in '"'-?*.Xx')
        total += (sum(v>=2 for v in c.values()) >= 2)
    return total
with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    best_mafft=(-1,None); best_raw=(-1,None); raw_equal=[]
    for n in z.namelist():
        if not (n.endswith('.faa.mafft') or n.endswith('.faa')): continue
        seqs=parse(z.read(n)); val=pi(seqs)
        if n.endswith('.faa.mafft') and val is not None and val>best_mafft[0]: best_mafft=(val,n)
        if n.endswith('.faa') and val is not None:
            raw_equal.append((val,n,len(seqs),len(seqs[0][1])))
            if val>best_raw[0]: best_raw=(val,n)
    print('best_mafft', best_mafft)
    print('raw_equal_count', len(raw_equal), 'best_raw_equal', best_raw)
    print('top_raw_equal', sorted(raw_equal, reverse=True)[:5])
PY"
