from collections import Counter

def parse_fasta(filepath):
    records = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if current_header is not None:
                    records.append((current_header, ''.join(current_seq)))
                current_header = line
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            records.append((current_header, ''.join(current_seq)))
    return records

records = parse_fasta('/workspace/extracted/scogs_animals/1428854at2759.faa.mafft')
for h, s in records:
    print(f'{h}: len={len(s)}')
    print(s[:100])

# Verify manually for a few key columns
seqs = [s for _, s in records]
aln_len = len(seqs[0])

# Count informative sites again with more detail
informative = 0
for i in range(aln_len):
    col = [s[i] for s in seqs]
    counts = Counter(col)
    n_informative = sum(1 for v in counts.values() if v >= 2)
    if n_informative >= 2:
        informative += 1
        if informative <= 5:
            print(f'Col {i}: {col} -> {counts}, informative_states={n_informative}')

print(f'Total informative: {informative}')
