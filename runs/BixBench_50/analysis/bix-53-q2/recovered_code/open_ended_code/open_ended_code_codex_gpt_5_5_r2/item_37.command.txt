/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts_raw = pd.read_csv('inputs/count.csv').set_index('geneID')

def run(samples, label):
    counts = counts_raw[samples].T.astype(int)
    cond = ['KL' if s.startswith('KL') else 'WL' for s in samples]
    metadata = pd.DataFrame({'condition': pd.Categorical(cond, categories=['WL','KL'])}, index=samples)
    dds = DeseqDataSet(counts=counts, metadata=metadata, design='~condition', quiet=True, n_cpus=1)
    dds.deseq2()
    stat = DeseqStats(dds, contrast=['condition','KL','WL'], alpha=0.05, quiet=True, n_cpus=1)
    stat.summary()
    print(label, 'LFC columns', list(dds.varm['LFC'].columns))
    print(label, stat.results_df.head(1).to_string())
    coeffs = [c for c in dds.varm['LFC'].columns if c "'!= '"'Intercept']
    assert coeffs == ['condition[T.KL]'], coeffs
    stat.lfc_shrink(coeff='condition[T.KL]')
    res = stat.results_df.copy()
    sig = res[(res['baseMean'] > 10) & (res['pvalue'] < 0.05) & (res['log2FoldChange'].abs() > 1)]
    print(label, 'n_sig', int(sig.shape[0]), 'p_nonNA', int(res['pvalue'].notna().sum()), 'padj_nonNA', int(res['padj'].notna().sum()))
    print(label, 'up', int((sig['log2FoldChange'] > 1).sum()), 'down', int((sig['log2FoldChange'] < -1).sum()))
    return int(sig.shape[0])
full = run(['KL1','KL2','KL3','WL1','WL2','WL3'], 'full')
reduced = run(['KL1','KL2','WL1','WL2'], 'reduced')
direction = 'increase' if reduced > full else 'decrease' if reduced < full else 'no change'
print('RESULT', full, reduced, direction)
PY"
