/bin/bash -lc "python3 - <<'PY'
import pandas as pd
import numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw = pd.read_csv('inputs/count.csv')
raw = raw.set_index('geneID')


def run(samples):
    counts = raw[samples].T
    counts.index.name = None
    metadata = pd.DataFrame({
        'condition': ['KL' if s.startswith('KL') else 'WL' for s in samples]
    }, index=samples)
    dds = DeseqDataSet(
        counts=counts,
        metadata=metadata,
        design='~condition',
        ref_level=['condition', 'WL'],
        quiet=True,
        n_cpus=1,
    )
    dds.deseq2()
    stat = DeseqStats(dds, contrast=['condition', 'KL', 'WL'], alpha=0.05, quiet=True, n_cpus=1)
    stat.summary()
    print('variables', stat.variables)
    # Find the KL-vs-WL coefficient generated by the design.
    coeffs = [x for x in stat.variables if 'condition' in x and 'KL' in x]
    if len(coeffs) "'!= 1:
        raise RuntimeError(f'"'Could not identify coefficient from {stat.variables}')
    stat.lfc_shrink(coeff=coeffs[0])
    res = stat.results_df.copy()
    sig = res[(res['pvalue'] < 0.05) & (res['baseMean'] > 10) & (res['log2FoldChange'].abs() > 1)]
    return len(sig), res

full_n, full_res = run(['KL1','KL2','KL3','WL1','WL2','WL3'])
sub_n, sub_res = run(['KL1','KL2','WL1','WL2'])
print('COUNTS', full_n, sub_n)
if sub_n > full_n:
    direction = 'increase'
elif sub_n < full_n:
    direction = 'decrease'
else:
    direction = 'no change'
print('DIRECTION', direction)
PY"
