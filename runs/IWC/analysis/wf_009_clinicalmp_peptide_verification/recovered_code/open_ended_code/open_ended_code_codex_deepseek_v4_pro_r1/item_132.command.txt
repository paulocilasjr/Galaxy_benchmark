/usr/bin/bash -lc "micromamba run -n pepquery python3 - <<'PY'
from pathlib import Path
files=list(Path('work/pepquery_out').glob('*_nodb_5g/psm_rank.txt'))
print(files)
peps=set()
for f in files:
    with open(f) as fh:
        header=fh.readline().rstrip('\\n').split('\\t')
        ci=header.index('confident')
        pi=header.index('peptide')
        for line in fh:
            parts=line.rstrip('\\n').split('\\t')
            if parts[ci]=='Yes': peps.add(parts[pi].upper())
print('unique confident peptides',len(peps))
# compare reference peprep unique
ref=set()
with open('/tmp/ref/Peptide_and_Protein_from_Peptide_Reports.tabular') as fh:
    next(fh)
    for line in fh:
        ref.add(line.split('\\t',1)[0].upper())
print('ref unique',len(ref))
print('missing vs ref', sorted(ref-peps))
print('extra vs ref', sorted(peps-ref))
PY"
