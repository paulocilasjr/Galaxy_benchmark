/usr/bin/bash -lc "micromamba run -n pepquery python3 - <<'PY'
import zipfile,re
p='/home/benchmark/.local/share/mamba/envs/pepquery/share/pepquery-2.0.2-0/pepquery-2.0.2.jar'
with zipfile.ZipFile(p) as z:
  for cls in ['main/java/pg/CParameter.class','main/java/pg/IMain.class','main/java/pg/PeptideSearchMT.class']:
    data=z.read(cls)
    s=re.findall(rb'[\\x20-\\x7e]{2,}', data)
    text=b' '.join(s).decode('latin1','ignore')
    print('\\n###',cls)
    for term in ['unmodif','unMod','fixMod','varMod','aa','hc','-um','maxVar','um']:
      idx=text.lower().find(term.lower())
      if idx>=0:
        print('TERM',term, ':', text[max(0,idx-300):idx+500])
PY"
