/usr/bin/bash -lc "micromamba run -n pepquery python3 - <<'PY'
import re
ids=set()
for line in open('/tmp/ref/Peptide_and_Protein_from_Peptide_Reports.tabular'):
    if line.startswith('mpep'): continue
    parts=line.rstrip('\\n').split('\\t')
    if len(parts)<2: continue
    prot=parts[1]
    # step20 normalize ; then replace tr|sp| then spaces then |.*"'$ then comma
    toks=[]
    for x in prot.split('"';'):
        x=re.sub(r'(tr|sp)[|]','',x)
        x=re.sub(r'[ ]+','',x)
        x=re.sub(r'[|].*"'$'"','',x)
        toks.append(x)
    for x in toks:
        for y in x.split(','):
            y=y.strip()
            if y: ids.add(y)
print(len(ids))
print('has A0A8H8ZU94', 'A0A8H8ZU94' in ids)
for i in sorted(ids): print(i)
PY"
