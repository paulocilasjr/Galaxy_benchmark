/usr/bin/bash -lc "micromamba run -n pepquery python3 - <<'PY'
import zipfile,re
p='/home/benchmark/.local/share/mamba/envs/pepquery/share/pepquery-2.0.2-0/pepquery-2.0.2.jar'
with zipfile.ZipFile(p) as z:
    names=z.namelist()
    # main class candidates
    for n in names:
        if n.endswith('.class') and ('main' in n.lower() or 'PepQuery' in n or 'Cmd' in n):
            data=z.read(n)
            # extract printable strings len>=2
            s=re.findall(rb'[\\x20-\\x7e]{2,}', data)
            text=b' '.join(s).decode('latin1','ignore')
            if 'unmodified' in text.lower() or '-um' in text or 'unMod' in text:
                print('FOUND',n)
                for term in ['unmodified','-um','unMod','um ']:
                    i=text.lower().find(term.lower())
                    if i>=0: print(term, text[i-200:i+300])
                break
    else:
        print('not found in class files')
PY"
