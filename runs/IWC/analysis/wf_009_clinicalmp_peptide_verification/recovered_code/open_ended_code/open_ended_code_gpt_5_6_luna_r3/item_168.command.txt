/usr/bin/bash -lc "awk -F '\\t' '
NR==1 {next}
{
  mode="'$5; d=$6-$14;
  n[mode]++;
  if ($6>$14) {pos[mode]++; if (!(mode in max) || d>max[mode]) max[mode]=d; if (d>=5) ge5[mode]++; if (d>=10) ge10[mode]++; if ($7>=5) m5[mode]++;}
}
END {for (m in n) print m, "rows="n[m], "positive="(pos[m]+0), ">=5="(ge5[m]+0), ">=10="(ge10[m]+0), "positive_matched>=5="(m5[m]+0), "maxdiff="(max[m]+0)}'"' /tmp/selected_variants.tsv | sort

echo '--- by2 positives ---'
awk -F '\\t' 'NR>1 && "'$5=="by2" && $6>$14 {print $4 "'"\\t\" "'$6 "'"\\t\" "'$7 "'"\\t\" "'$13 "'"\\t\" "'$14 "'"\\t\" ("'$6-$14)}'"' /tmp/selected_variants.tsv | sort -k6,6nr

echo '--- bya positives ---'
awk -F '\\t' 'NR>1 && "'$5=="bya" && $6>$14 {print $4 "'"\\t\" "'$6 "'"\\t\" "'$7 "'"\\t\" "'$13 "'"\\t\" "'$14 "'"\\t\" ("'$6-$14)}'"' /tmp/selected_variants.tsv | sort -k6,6nr"
