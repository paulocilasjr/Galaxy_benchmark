/usr/bin/bash -lc "set -euo pipefail
out=/workspace/final_answer/verified_peptides.tsv
printf 'deliverable_files\\n'; find /workspace/final_answer -mindepth 1 -maxdepth 1 -type f -printf '%f\\n' | sort
[ \""'$(find /workspace/final_answer -mindepth 1 -maxdepth 1 -type f | wc -l)" -eq 1 ]
[ "$(head -1 "$out")" = $'"'peptide\\tprotein_id' ]
awk -F'\\t' 'NR==1{if(NF"'!=2 || $1!="peptide" || $2!="protein_id")exit 1;next} NF!=2 || $1!~/''^[A-Z]+$/ || $2==""{print "invalid row " NR > "/dev/stderr";exit 1} END{if(NR<2)exit 1}'"' \""'$out"
[ "$(tail -n +2 "$out" | sort -u | wc -l)" -eq "$(($(wc -l < "$out")-1))" ]
cmp <(tail -n +2 "$out") /workspace/work/verified_pairs.sorted.tsv
cmp <(awk -F'"'\\t' 'NR>1{print "'$1}'"' \""'$out" | sort -u) /workspace/work/verified_peptides.txt
for f in /workspace/work/pepquery_by_mgf/*/parameter.txt; do
  rg -q '"'Precursor mass tolerance: 10.0' \""'$f"
  rg -q '"'Fragment ion mass tolerance: 0.6' \""'$f"
  rg -q '"'Max Missed cleavages: 2' \""'$f"
  rg -q '"'Fixed modification: 1,13,14 = Carbamidomethylation of C,TMT 11-plex of K,TMT 11-plex of peptide N-term' \""'$f"
  rg -q '"'Variable modification: 2 = Oxidation of M' \""'$f"
done
printf '"'rows_with_header\\t'; wc -l < \""'$out"
printf '"'data_rows\\t'; awk 'END{print NR-1}' \""'$out"
printf '"'unique_peptides\\t'; awk -F'\\t' 'NR>1{p["'$1]=1} END{for(x in p)n++;print n}'"' \""'$out"
printf '"'sha256\\t'; sha256sum \""'$out" | cut -d'"' ' -f1
printf 'bytes\\t'; stat -c '%s' \""'$out"
printf '"'trace_files\\n'; find /workspace/run_trace -maxdepth 1 -type f -printf '%f\\n' | sort
printf 'status\\tPASS\\n'"
