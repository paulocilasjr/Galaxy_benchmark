/usr/bin/bash -lc "set -euo pipefail
mkdir -p /workspace/work/ms_inputs
gzip -dc /workspace/data/inputs/human_uniprot_isoforms_fasta/stream > /workspace/work/human_crap_reference.fasta
sed -n '1,"'$p'"' '/workspace/data/inputs/crap/Contaminants%28cRAP%29-Protein-Database.fasta' >> /workspace/work/human_crap_reference.fasta
for f in /workspace/data/inputs/tandem_mass_spectrometry_ms_ms_datasets/*/*.mgf; do ln -sf \""'$f" "/workspace/work/ms_inputs/$(basename "$f")"; done
cp /workspace/data/inputs/distinct_peptides_for_pepquery/Distinct_Peptides_for_PepQuery.tabular /workspace/work/candidate_peptides.txt
printf '"'reference_headers\\t'; rg -c '"'^>'"' /workspace/work/human_crap_reference.fasta
printf 'reference_bytes\\t'; stat -c '%s' /workspace/work/human_crap_reference.fasta
printf 'human_sha256\\t'; sha256sum /workspace/data/inputs/human_uniprot_isoforms_fasta/stream | cut -d' ' -f1
printf 'crap_sha256\\t'; sha256sum '/workspace/data/inputs/crap/Contaminants%28cRAP%29-Protein-Database.fasta' | cut -d' ' -f1
printf 'candidate_sha1\\t'; sha1sum /workspace/work/candidate_peptides.txt | cut -d' ' -f1
printf 'invalid_candidates\\n'; awk 'BEGIN{bad=0} "'!/''^[A-Z]+$/{print NR ":" $0; bad++} END{print "count=" bad}'"' /workspace/work/candidate_peptides.txt
printf 'candidate_length_range\\t'; awk 'NR==1{a=length;b=length} {if(length<a)a=length;if(length>b)b=length} END{print a \"-\" b}' /workspace/work/candidate_peptides.txt
ls -l /workspace/work/ms_inputs"
