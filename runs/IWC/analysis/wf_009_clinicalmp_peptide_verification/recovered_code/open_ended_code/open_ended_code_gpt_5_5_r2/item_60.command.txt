/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,re,os
rows=list(csv.DictReader(open('final_answer/verified_peptides.tsv'),delimiter='\\t'))
print('columns', rows and list(rows[0].keys()))
print('rows',len(rows),'peptides',len({r['peptide'] for r in rows}),'pairs',len({(r['peptide'],r['protein_id']) for r in rows}))
print('bad_peptides',sum(1 for r in rows if not re.fullmatch(r'[A-Z]+',r['peptide'])))
print('pipe_protein_ids',sum(1 for r in rows if '|' in r['protein_id']))
print('blank_fields',sum(1 for r in rows if not r['peptide'] or not r['protein_id']))
print('final_answer_files',sorted(os.listdir('final_answer')))
PY"
