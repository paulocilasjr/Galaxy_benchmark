echo '--- counts ---'; /opt/conda/bin/python - <<'PY'
import csv, collections
for name in ['psm.txt','psm_rank.txt','detail.txt','ptm.txt']:
 p='/workspace/analysis_work/pq_all_n100_fast/'+name
 with open(p) as f:
  rows=list(csv.DictReader(f,delimiter='\\t'))
 print(name,len(rows), 'fields', list(rows[0]) if rows else [])
 if rows and 'confident' in rows[0]: print('confident',collections.Counter(r['confident'] for r in rows),'peptides',len(set(r['peptide'] for r in rows if r['confident']=='Yes')))
 if rows and 'pvalue' in rows:
  print('pvalue range',min(float(r['pvalue']) for r in rows),max(float(r['pvalue']) for r in rows), 'p<=.05',sum(float(r['pvalue'])<=.05 for r in rows))
PY"
