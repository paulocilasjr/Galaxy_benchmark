/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
from pathlib import Path
base=Path('data/inputs/tandem_mass_spectrometry_ms_ms_datasets')
out=Path('/workspace/analysis_work/all_spectra.mgf')
with out.open('wb') as dst:
 for p in sorted(base.glob('*/*.mgf')):
  print('adding',p.name,p.stat().st_size,flush=True)
  with p.open('rb') as src:
   while True:
    b=src.read(8*1024*1024)
    if not b: break
    dst.write(b)
print('combined',out.stat().st_size)
PY
ls -lh /workspace/analysis_work/all_spectra.mgf"
