/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
from pathlib import Path
import csv, gzip, glob, collections, re, time
D=[x[0].strip().upper() for x in csv.reader(open(glob.glob('data/inputs/distinct_peptides_for_pepquery/*.tabular')[0]),delimiter='\\t') if x]
found_raw=collections.defaultdict(list); found_try=collections.defaultdict(list); nseq=0; npep=0; unique=set(); t=time.time()
def parse(path, gz=False):
  op=gzip.open if gz else open
  with op(path,'rt',errors='replace') as f:
   h=None; seq=[]
   for line in f:
    if line.startswith('>'):
     if h is not None: yield h,''.join(seq)
     h=line[1:].strip();seq=[]
    else: seq.append(line.strip())
   if h is not None: yield h,''.join(seq)

def addprot(h,s):
 global nseq,npep
 nseq+=1
 for p in D:
  if p in s: found_raw[p].append(h.split()[0])
 # trypsin, cleavage after K/R unless next P, fully specific; generate <=2 missed
 cuts=[0]
 for i,a in enumerate(s[:-1]):
  if a in 'KR' and s[i+1]"'!='"'P': cuts.append(i+1)
 cuts.append(len(s))
 # unique peptide sequence? include protein per peptide
 for i in range(len(cuts)-1):
  for j in range(i+1,min(len(cuts),i+4)):
   pep=s[cuts[i]:cuts[j]]
   npep+=1
   if pep in D: found_try[pep].append(h.split()[0])
for path,gz in [('data/inputs/human_uniprot_isoforms_fasta/stream',True), (glob.glob('data/inputs/crap/*')[0],False)]:
 for h,s in parse(path,gz): addprot(h,s)
 print(path,'done nseq',nseq,'npep',npep,'sec',round(time.time()-t,1))
print('raw matches',sum(bool(found_raw[p]) for p in D),'tryptic',sum(bool(found_try[p]) for p in D))
for p in D:
 if found_raw[p] or found_try[p]: print(p,'raw',len(set(found_raw[p])),'try',len(set(found_try[p])),found_try[p][:4], 'rawfirst',found_raw[p][:2])
PY"
