/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,re,collections
verified={r['peptide'] for r in csv.DictReader(open('/workspace/analysis_work/pq_all_n100_fast/psm_rank.txt'),delimiter='\\t') if r['confident']=='Yes'}

def toks(v): return [x.strip() for x in re.split(r'[;,]',v or '') if x.strip()]
def canon(x):
 x=x.strip()
 if '|' in x:
  q=x.split('|')
  if len(q)>=3 and q[1]: return q[1]
 return x
m=collections.defaultdict(set)
for p,field in [('/workspace/data/inputs/sgps_peptide_report/SGPS_peptide-report.tabular','Protein(s)'),('/workspace/data/inputs/maxquant_peptide_report/MaxQuant-peptide-report.tabular','Proteins')]:
 with open(p,newline='') as f:
  for r in csv.DictReader(f,delimiter='\\t'):
   seq=(r.get('Sequence') or '').strip()
   if seq in verified:
    for x in toks(r.get(field,'')): m[seq].add(canon(x))
print('verified',len(verified),'mapped',len([x for x in verified if m[x]]),'pairs',sum(map(len,m.values())))
print('missing',sorted(verified-set(m)))
for s in sorted(m):
 bad=[x for x in sorted(m[s]) if any(y in x.upper() for y in ['HUMAN','RABBIT','CON_'])]
 if bad: print('suspicious',s,bad)
print('all canonical ids',len(set().union(*m.values())))
print('multiples')
for s in sorted(m):
 if len(m[s])>1: print(s, sorted(m[s]))
PY"
