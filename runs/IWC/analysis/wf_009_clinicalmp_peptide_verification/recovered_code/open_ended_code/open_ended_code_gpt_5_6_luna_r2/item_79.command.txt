 if r['Sequence'] in ver:

 if r['Sequence'] in ver:
  for p in r['Protein(s)'].split(','):add(r['Sequence'],p)
print('raw pairs',sum(len(raw[p]) for p in ver),'canon pairs',sum(len(canon[p]) for p in ver),'unique proteins canon',len(set().union(*canon.values())))
print('multiple canon')
for p in sorted(ver):
 if len(canon[p])>1: print(p, sorted(canon[p]),'raw',sorted(raw[p]))
