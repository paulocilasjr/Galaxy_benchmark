/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
from pathlib import Path
import re, time
files=sorted(Path('data/inputs/tandem_mass_spectrometry_ms_ms_datasets').glob('*/*.mgf'))
for path in files:
 t=time.time(); n=0; peaks=0; charges={}; mzmin=1e9; mzmax=0; pmin=1e9;pmax=0
 inions=False; np=0; pm=None; ch=None
 with path.open('r', errors='replace') as f:
  for line in f:
   if line.startswith('BEGIN IONS'):
    n+=1; np=0; pm=None;ch=None; inions=True
   elif inions and line.startswith('PEPMASS='):
    pm=float(line.split('=',1)[1].split()[0]); mzmin=min(mzmin,pm);mzmax=max(mzmax,pm)
   elif inions and line.startswith('CHARGE='):
    s=line.split('=',1)[1].strip().rstrip('+').rstrip('-');ch=int(s); charges[ch]=charges.get(ch,0)+1
   elif inions and line.startswith('END IONS'):
    inions=False
   elif inions and line and line[0].isdigit():
    np+=1; peaks+=1
 print(path.name, 'spectra',n,'peaks',peaks,'charge',charges,'mz',mzmin,mzmax,'sec',round(time.time()-t,1))
PY"
