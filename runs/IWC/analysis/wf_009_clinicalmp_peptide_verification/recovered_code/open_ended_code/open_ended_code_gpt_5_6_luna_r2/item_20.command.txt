import csv,gzip,glob,collections,time,sys
D={x[0].strip().upper() for x in csv.reader(open(glob.glob('data/inputs/distinct_peptides_for_pepquery/*.tabular')[0]),delimiter='\\t') if x}
found={p:set() for p in D}; nseq=0;npep=0

def process(path,gz):
 global nseq,npep
 f=gzip.open(path,'rt',errors='replace') if gz else open(path,errors='replace')
 h=None;seq=[];t=time.time()
 def one(h,s):
  global nseq,npep
  nseq+=1
  cuts=[0]
  for i,a in enumerate(s[:-1]):
   if a in 'KR' and s[i+1]"'!='"'P':cuts.append(i+1)
  cuts.append(len(s))
  for i in range(len(cuts)-1):
   for j in range(i+1,min(len(cuts),i+4)):
    p=s[cuts[i]:cuts[j]];npep+=1
    if p in found: found[p].add(h.split()[0])
 for line in f:
  if line.startswith('>'):
   if h is not None: one(h,''.join(seq))
   h=line[1:].strip();seq=[]
  else:seq.append(line.strip())
 if h is not None: one(h,''.join(seq))
 f.close()
 print('done',path,'nseq',nseq,'npep',npep,'seconds',time.time()-t,flush=True)
process('data/inputs/human_uniprot_isoforms_fasta/stream',True)
process(glob.glob('data/inputs/crap/*')[0],False)
print('found',sum(bool(v) for v in found.values()),flush=True)
for p in sorted(found):
 if found[p]:print(p,len(found[p]),sorted(found[p])[:5],flush=True)
