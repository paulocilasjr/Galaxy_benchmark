/usr/bin/bash -lc "micromamba run -n pepq python - <<'PY'
from pathlib import Path
vp=set(Path('work/verified_peptides_from_pepquery.txt').read_text().splitlines())
report=Path('work/peptide_protein_reports.tsv').read_text().splitlines()
out=[]
for line in report:
    p,pr=line.split('\\t',1)
    if p in vp:
        out.append((p,pr))
print('join rows',len(out),'unique pairs',len(set(out)),'unique pep',len(set(p for p,_ in out)),'unique raw prot',len(set(pr for _,pr in out)))
Path('work/query_result_ours.tsv').write_text(''.join(f'{p}\\t{pr}\\n' for p,pr in out))
# compare to expected file without header
exp=Path('work/expected_report_pairs.tabular').read_text().splitlines()
expdata=exp[1:]
print('expected rows',len(expdata),'our rows',len(out))
# set compare
E=set(tuple(l.split('\\t',1)) for l in expdata)
O=set(out)
print('missing ours from expected',len(E-O))
print('extra ours',len(O-E))
print('--- missing ---')
for x in sorted(E-O): print(x)
print('--- extra ---')
for x in sorted(O-E): print(x)
PY"
