/usr/bin/bash -lc "micromamba run -n pepq python - <<'PY'
from pathlib import Path
import re
vp=set(Path('work/verified_peptides_from_pepquery.txt').read_text().splitlines())
report=Path('work/peptide_protein_reports.tsv').read_text().splitlines()
pairs=[]
for line in report:
    p,prot=line.split('\\t',1)
    if p not in vp: continue
    for item in prot.split(';'):
        item=re.sub(r'(tr|sp)[|]','',item)
        item=re.sub(r'[ ]+','',item)
        item=re.sub(r'[|].*"'$'"','',item)
        for sub in item.split(','):
            if sub: pairs.append((p,sub))
seen=set(); uniq=[]
for p,pid in pairs:
    if (p,pid) not in seen:
        seen.add((p,pid)); uniq.append((p,pid))
print(len(uniq),len(set(p for p,_ in uniq)),len(set(pid for _,pid in uniq)))
ids=sorted(set(pid for _,pid in uniq))
Path('work/actual_normalized_pairs.tsv').write_text(''.join(f'{p}\\t{pid}\\n' for p,pid in uniq))
# diff set with expected normalized
exp=Path('work/expected_normalized_pairs.tsv').read_text().splitlines()
E=set(tuple(l.split('\\t',1)) for l in exp)
O=set(uniq)
print('missing',len(E-O)); [print('M',x) for x in sorted(E-O)]
print('extra',len(O-E)); [print('X',x) for x in sorted(O-E)]
PY"
