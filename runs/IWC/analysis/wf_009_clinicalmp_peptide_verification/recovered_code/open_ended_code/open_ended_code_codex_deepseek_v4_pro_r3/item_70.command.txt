/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv, glob, collections
conf=[]
for f in glob.glob('working/pepquery_ptrc_*/psm_rank.txt'):
    with open(f) as fh:
        rows=list(csv.DictReader(fh, delimiter='\\t'))
    c=[r['peptide'].upper() for r in rows if r.get('confident')=='Yes']
    print(f, 'rows', len(rows), 'confident rows', len(c), 'unique', len(set(c)))
    conf+=c
print('all confident rows', len(conf), 'unique peptides', len(set(conf)))
for p in sorted(set(conf)): print(p)
PY"
