/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv
from collections import defaultdict
rows=list(csv.reader(open('final_answer/verified_peptides.tsv'),delimiter='\\t'))
print('header',rows[0], 'data',len(rows)-1)
d=defaultdict(list)
for p,pid in rows[1:]: d[p].append(pid)
print('multi')
for p,ids in d.items():
    if len(ids)>1: print(p, ids)
# validate no non-uppercase, no mods
import re
bad=[(p,pid) for p,pid in rows[1:] if p"'!=p.upper() or not re.fullmatch(r'"'[A-Z]+',p)]
print('bad peptides',bad[:10],len(bad))
print('unique peptides',len(d),'unique pairs',len(set(rows[1:])))
PY"
