/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,collections,itertools
from statistics import median
rows=list(csv.DictReader(open('work/fragment_scores.tsv'),delimiter='\\t'))
by=collections.defaultdict(list)
for r in rows: by[r['peptide']].append(r)
print('rows',len(rows),'peptides',len(by))
print('pep\\tspecs\\tbest_match\\tbest_b\\tbest_y\\tbest_cov\\tbest_int\\tmedian_match\\tmax_ppm')
for p in sorted(by):
 rs=by[p]
 best=max(rs,key=lambda r:(int(r['matched_ions']),float(r['coverage']),float(r['intensity_fraction'])))
 print('\\t'.join([p,str(len(rs)),best['matched_ions'],best['b_ions'],best['y_ions'],best['coverage'],best['intensity_fraction'],f'{median(int(r[\"matched_ions\"]) for r in rs):.1f}',str(min(abs(float(r['ppm'])) for r in rs))]))
print('\\nBest matched ion distribution')
print(collections.Counter(max(int(r['matched_ions']) for r in rs) for rs in by.values()))
print('\\nTop 50 by best matched ions/coverage')
for p,rs in sorted(by.items(),key=lambda pr:max((int(r['matched_ions']),float(r['coverage']),float(r['intensity_fraction'])) for r in pr[1]),reverse=True)[:50]:
 b=max(rs,key=lambda r:(int(r['matched_ions']),float(r['coverage']),float(r['intensity_fraction'])))
 print(p,b['matched_ions'],b['b_ions'],b['y_ions'],b['coverage'],b['intensity_fraction'],b['charge'],b['oxidations'],b['oxidation_positions'],b['file'][-27:])
PY"
