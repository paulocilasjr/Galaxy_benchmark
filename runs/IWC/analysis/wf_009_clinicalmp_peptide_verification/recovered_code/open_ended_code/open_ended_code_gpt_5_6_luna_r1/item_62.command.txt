/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv,collections
rows=list(csv.DictReader(open('work/reference_near_masses.tsv'),delimiter='\\t'))
by=collections.defaultdict(set); byprot=collections.defaultdict(set)
for r in rows:
 by[r['candidate']].add((r['reference_peptide'],r['reference_oxidations']))
 byprot[r['candidate']].add(r['protein_id'])
print('rows',len(rows),'unique ref variants total',len(set((r['reference_peptide'],r['reference_oxidations']) for r in rows)))
print('per candidate min/max/med',min(map(len,by.values())),max(map(len,by.values())),sorted(map(len,by.values()))[len(by)//2])
for p in sorted(by,key=lambda p:len(by[p]),reverse=True)[:20]:print(p,len(by[p]),len(byprot[p]))
print('ref exact candidates',sum(r['candidate']==r['reference_peptide'] for r in rows))
PY"
