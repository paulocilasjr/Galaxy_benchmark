import csv,collections
allfr=list(csv.DictReader(open('work/fragment_scores.tsv'),delimiter='\\t'))

for p in sorted(q):
 if p in fr:
  b=max(fr[p],key=lambda r:(int(r['matched_ions']),float(r['coverage']),float(r['intensity_fraction'])))
  print(p,q[p]['Score'],q[p]['PEP'],b['matched_ions'],b['coverage'])
