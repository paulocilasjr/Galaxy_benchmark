/usr/bin/bash -lc "/opt/conda/bin/python3.11 - <<'PY'
import xml.etree.ElementTree as ET
for path in ['work/ncbi_sra_PRJNA1339024.xml','work/ncbi_sra_PRJNA1354216.xml']:
 root=ET.parse(path).getroot()
 for p in root.findall('EXPERIMENT_PACKAGE'):
  run=p.find('RUN_SET/RUN'); layout_el=p.find('EXPERIMENT/DESIGN/LIBRARY_DESCRIPTOR/LIBRARY_LAYOUT')
  layout='PAIRED' if layout_el.find('PAIRED') is not None else 'SINGLE' if layout_el.find('SINGLE') is not None else 'UNKNOWN'
  stats=run.find('Statistics')
  nreads=int(stats.get('nreads')) if stats is not None else -1
  expected=2 if layout=='PAIRED' else 1 if layout=='SINGLE' else -1
  assert nreads == expected, (run.get('accession'),layout,nreads)
 print(path, 'runs',len(root.findall('EXPERIMENT_PACKAGE')), 'all nreads match layout')
PY"
