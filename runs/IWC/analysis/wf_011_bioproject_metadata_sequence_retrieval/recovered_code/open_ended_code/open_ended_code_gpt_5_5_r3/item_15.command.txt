/usr/bin/bash -lc "printf '%s\\n' '{\"stage\":\"runinfo_trace_sra_endpoint\",\"status\":\"failed\",\"tool_or_command\":\"curl https://trace.ncbi.nlm.nih.gov/Traces/sra/sra.cgi\",\"effective_parameters\":{\"rettype\":\"runinfo\",\"term_source\":\"BioProject accessions\"},\"inputs\":[\"data/inputs/bioproject_ids/bioproject_ids.txt\"],\"outputs\":[],\"validation\":{\"error\":\"HTTP 400 for both BioProject accessions\"}}' >> run_trace/analysis_steps.jsonl
for acc in "'$(sed '"'/"'^$/d'"' data/inputs/bioproject_ids/bioproject_ids.txt); do
  echo \"== "'$acc =="
  curl -fsSL "https://trace.ncbi.nlm.nih.gov/Traces/sra-db-be/runinfo?acc=${acc}" | tee "work/${acc}.runinfo.csv" | head -n 3
  echo lines $(wc -l < "work/${acc}.runinfo.csv")
done'
