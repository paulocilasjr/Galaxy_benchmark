/usr/bin/bash -lc 'set -o pipefail
for bp in PRJNA1354216 PRJNA1339024; do
  ids=$(python3 -c '"'import json,sys; print(\",\".join(json.load(open(sys.argv[1]))[\"esearchresult\"][\"idlist\"]))' \"/tmp/"'${bp}.esearch.json")
  echo "=== $bp runinfo ==="
  curl -L --fail --silent --show-error --retry 3 --retry-delay 2 "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=sra&id=${ids}&rettype=runinfo&retmode=text" | tee "/tmp/${bp}.runinfo.csv" | sed -n '"'1,30p'
  echo
 done"
