/usr/bin/bash -lc 'for acc in SRR35895060 SRR35763874; do
  echo "=== $acc XML head ==="
  curl -L --fail --silent --show-error --retry 3 --retry-delay 2 "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=sra&id=${acc}&rettype=xml&retmode=xml" | tee "/tmp/${acc}.xml" | sed -n '"'1,80p'
  echo
 done"
