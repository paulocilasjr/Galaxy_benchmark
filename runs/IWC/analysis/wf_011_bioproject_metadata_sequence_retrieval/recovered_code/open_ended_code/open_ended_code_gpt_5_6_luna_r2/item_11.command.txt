/usr/bin/bash -lc 'set -o pipefail
for bp in PRJNA1354216 PRJNA1339024; do
  ids=$(jq -r '"'.esearchresult.idlist | join(\",\")' \"/tmp/"'${bp}.esearch.json")
  echo "=== $bp runinfo ==="
  curl -L --fail --silent --show-error --retry 3 --retry-delay 2 "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=sra&id=${ids}&rettype=runinfo&retmode=text" | tee "/tmp/${bp}.runinfo.csv" | sed -n '"'1,35p'
  echo
 done"
