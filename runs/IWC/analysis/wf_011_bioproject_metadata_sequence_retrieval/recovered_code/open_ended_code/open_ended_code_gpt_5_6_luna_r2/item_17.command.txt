/usr/bin/bash -lc 'for bp in PRJNA1354216 PRJNA1339024; do
  url="https://www.ebi.ac.uk/ena/portal/api/filereport?accession=${bp}&result=read_run&fields=run_accession%2Cfastq_ftp%2Cfastq_bytes%2Cfastq_md5&format=tsv&download=false"
  echo "=== $bp ==="
  curl --http1.1 -L --fail --silent --show-error --retry 5 --retry-delay 2 "$url" | tee "/tmp/${bp}.ena.tsv" | sed -n '"'1,20p'
  echo \"rows="'$(($(wc -l < /tmp/${bp}.ena.tsv 2>/dev/null || echo 0)-1))"
 done'
