/usr/bin/bash -lc 'for acc in SRR35895060 SRR35763874; do
  echo "=== $acc ENA ==="
  curl -L --fail --silent --show-error --retry 3 --retry-delay 2 "https://www.ebi.ac.uk/ena/portal/api/filereport?accession=${acc}&result=read_run&fields=run_accession,study_accession,experiment_accession,sample_accession,scientific_name,tax_id,library_strategy,library_source,library_selection,library_layout,instrument_model,read_count,base_count,fastq_ftp,fastq_bytes,fastq_md5&format=tsv&download=false" | tee "/tmp/${acc}.ena.tsv"
  echo
 done'
