/usr/bin/bash -lc "mkdir -p /workspace/work /workspace/run_trace /workspace/final_answer/metadata /workspace/final_answer/reads
curl -sS 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=sra&term=PRJNA1354216%5BBioProject%5D&retmax=1000&retmode=json' -o /workspace/work/esearch_PRJNA1354216.json
curl -sS 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=sra&term=PRJNA1339024%5BBioProject%5D&retmax=1000&retmode=json' -o /workspace/work/esearch_PRJNA1339024.json
ids1="'$(jq -r '"'.esearchresult.idlist[]' /workspace/work/esearch_PRJNA1354216.json | paste -sd, -)
ids2="'$(jq -r '"'.esearchresult.idlist[]' /workspace/work/esearch_PRJNA1339024.json | paste -sd, -)
curl -sS \"https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=sra&id="'${ids1}&rettype=runinfo&retmode=text" -o /workspace/work/runinfo_PRJNA1354216.csv
curl -sS "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=sra&id=${ids2}&rettype=runinfo&retmode=text" -o /workspace/work/runinfo_PRJNA1339024.csv
echo IDs1=$ids1; echo IDs2=$ids2; wc -l /workspace/work/runinfo_*.csv; head -2 /workspace/work/runinfo_PRJNA1354216.csv; head -2 /workspace/work/runinfo_PRJNA1339024.csv'
