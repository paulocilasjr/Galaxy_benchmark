/usr/bin/bash -lc "tail -n +2 work/downloads.tsv | xargs -P 8 -n 5 sh -c '
  run=\""'$1"; mate="$2"; url="$3"; dest="$4"; expected="$5"
  part="$dest.part"
  if [ ! -s "$part" ]; then
    curl -fsSL --retry 4 --retry-delay 2 "$url" -o "$part" || exit 1
  fi
  actual=$(md5sum "$part")
  actual=${actual%% *}
  if [ "$actual" != "$expected" ]; then
    curl -fsSL --retry 4 --retry-delay 2 "$url" -o "$part" || exit 1
    actual=$(md5sum "$part")
    actual=${actual%% *}
  fi
  if [ "$actual" != "$expected" ]; then
    echo "MD5 mismatch for $run mate $mate" >&2
    exit 1
  fi
  mv "$part" "$dest"
  echo "$run mate $mate verified"
'"' _"
