/usr/bin/bash -lc "/opt/conda/bin/python3.11 - <<'PY'
import h5py, numpy as np, pandas as pd, math
from scipy.sparse import csr_matrix
p='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
def decarr(a):
    return np.array([x.decode() if isinstance(x, bytes) else str(x) for x in a], dtype=object)
with h5py.File(p,'r') as f:
    cell_type=decarr(f['obs/cell_type/categories'][()])[f['obs/cell_type/codes'][()]]
    disease=decarr(f['obs/disease/categories'][()])[f['obs/disease/codes'][()]]
    individual=decarr(f['obs/individual/categories'][()])[f['obs/individual/codes'][()]]
    gene=decarr(f['var/gene_symbol'][()])
    print('gene unique',len(np.unique(gene)),'of',len(gene),'duplicates',len(gene)-len(np.unique(gene)))
    dup=pd.Series(gene).value_counts()
    print('dup symbols',dup[dup>1].head(30).to_string())
    meta=pd.DataFrame({'individual':individual,'cell_type':cell_type,'disease':disease})
    combo=meta.groupby(['individual','cell_type'],sort=False,observed=True).agg(n_cells=('disease','size'),n_disease=('disease','nunique'),disease=('disease','first'))
    combo['col']=[str(i).replace(' ','_').replace('-','_')+'_'+str(c).replace(' ','_').replace('-','_') for i,c in combo.index]
    print('\\ncombos all',len(combo))
    print(combo.to_string())
    retained=combo[combo.n_cells>=10]
    print('\\nretained',len(retained),'disease counts',retained.disease.value_counts().to_dict())
    print('retained samples per individual')
    print(retained.reset_index().groupby('individual').size().to_string())
PY"
