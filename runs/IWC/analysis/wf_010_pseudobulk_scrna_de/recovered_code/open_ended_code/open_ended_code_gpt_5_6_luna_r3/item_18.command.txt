/usr/bin/bash -lc "micromamba run -n sc_pseudobulk python - <<'PY'
import anndata as ad
import numpy as np
from pathlib import Path
p=Path('data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad')
a=ad.read_h5ad(p, backed='r')
print('counts type', type(a.layers['counts']))
print('counts shape',a.layers['counts'].shape, 'dtype',getattr(a.layers['counts'],'dtype',None))
print('counts repr',repr(a.layers['counts'])[:500])
print('obs categorical details')
for c in ['individual','disease','cell_type']:
    print(c, list(a.obs[c].cat.categories))
print('var index name',a.var.index.name,'unique',a.var.index.is_unique,'n',len(a.var.index),'head',a.var.index[:10].tolist(),'tail',a.var.index[-5:].tolist())
# group sizes, disease per individual and combo
obs=a.obs
print('\\nindividual disease cross-tab')
print(__import__('pandas').crosstab(obs['individual'],obs['disease']).to_string())
print('\\ncombo sizes')
print(obs.groupby(['individual','cell_type'],observed=True).size().to_string())
print('\\ncombo disease unique')
print(obs.groupby(['individual','cell_type'],observed=True)['disease'].agg(lambda s: ','.join(map(str,s.unique()))).to_string())
PY"
