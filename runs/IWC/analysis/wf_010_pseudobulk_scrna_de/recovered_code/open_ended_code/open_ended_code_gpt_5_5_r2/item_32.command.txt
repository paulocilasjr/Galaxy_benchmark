/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import h5py
path='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
with h5py.File(path,'r') as f:
    def visit(name, obj):
        if len(name.split('/'))<=3:
            if isinstance(obj,h5py.Dataset):
                print('D', name, obj.shape, obj.dtype)
            else:
                print('G', name, dict(obj.attrs))
    f.visititems(visit)
PY"
