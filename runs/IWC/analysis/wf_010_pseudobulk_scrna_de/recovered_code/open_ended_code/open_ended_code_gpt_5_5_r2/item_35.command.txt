/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import h5py, numpy as np, pandas as pd
path='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
def cat(g, key):
    grp=g['obs'][key]
    cats=[x.decode() if isinstance(x,bytes) else str(x) for x in grp['categories'][()]]
    codes=grp['codes'][()]
    return np.array([cats[c] if c>=0 else None for c in codes], dtype=object), cats
with h5py.File(path,'r') as f:
    ind, inds=cat(f,'individual'); dis, diss=cat(f,'disease'); ct, cts=cat(f,'cell_type')
    print('disease categories:', diss)
    print('individual categories:', inds)
    print('cell_type categories:', cts)
    df=pd.DataFrame({'individual':ind,'disease':dis,'cell_type':ct})
    print('\\nCells by disease')
    print(df['disease'].value_counts().to_string())
    print('\\nindividual disease')
    print(df.drop_duplicates(['individual','disease']).sort_values(['disease','individual']).to_string(index=False))
    print('\\ncombos >=10')
    tab=df.groupby(['individual','disease','cell_type']).size().reset_index(name='n_cells')
    print(tab[tab.n_cells>=10].to_string(index=False))
    print('N retained combos', (tab.n_cells>=10).sum())
PY"
