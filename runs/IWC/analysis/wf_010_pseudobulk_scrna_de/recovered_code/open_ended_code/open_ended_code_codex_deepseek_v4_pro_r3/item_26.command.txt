/usr/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
meta=pd.read_csv('/workspace/pseudobulk_metadata.tsv',sep='\\t')
df=pd.read_csv('/workspace/pseudobulk_counts_raw_all.tsv',sep='\\t',index_col=0)
samples=meta['sample'].values
N=len(samples); M=np.median(meta['total_counts']); S=int(np.ceil(10+0.7*(N-10)))
thr=10/M*1e6
print('N',N,'M',M,'S',S,'thr_cpm',thr)
cpm=df.values / meta['total_counts'].values * 1e6
npass=(cpm>=thr).sum(axis=1)
sums=df.values.sum(axis=1)
keep=(npass>=S)&(sums>=1000)
print('pass counts', keep.sum(), 'only npass', (npass>=S).sum(), 'only sum', (sums>=1000).sum())
print('npass distribution top', pd.Series(npass).value_counts().sort_index().tail(10).to_dict())
print('sums min among pass', sums[keep].min(), 'max', sums[keep].max())
print('examples keep', df.index[keep][:5].tolist())
PY"
