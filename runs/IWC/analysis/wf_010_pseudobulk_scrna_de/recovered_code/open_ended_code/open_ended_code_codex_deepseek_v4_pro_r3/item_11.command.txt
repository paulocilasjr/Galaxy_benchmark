/usr/bin/bash -c "python - <<'PY'
import h5py
p='/workspace/data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
with h5py.File(p,'r') as f:
    for group in ['obs','var']:
      g=f[group]
      for k in g.keys():
        item=g[k]
        if isinstance(item,h5py.Group) and 'categories' in item:
          cats=item['categories'][:]
          codes=item['codes'][:]
          try: cats=[x.decode() if isinstance(x,bytes) else x for x in cats]
          except: pass
          print(group,k,'cats=',cats,'code counts:')
          import collections
          cnt=collections.Counter(codes.tolist())
          for c,n in sorted(cnt.items()): print(' ',c,cats[c] if c>=0 and c<len(cats) else 'NA',n)
        else:
          pass
PY"
