/usr/bin/bash -lc "/workspace/.analysis_env/bin/python - <<'PY'
import anndata as ad, numpy as np, pandas as pd
from pathlib import Path
p=Path('/workspace/data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad')
a=ad.read_h5ad(p)
print('var index name:', a.var_names.name)
print('var index unique:', a.var_names.is_unique, 'null:', a.var_names.isna().sum())
print('first var names:', list(a.var_names[:20]))
print('gene_symbol attr present?', 'gene_symbol' in a.var.columns)
print('\\nindividual -> diseases')
print(a.obs.groupby('individual', observed=True)['disease'].agg(lambda x: ','.join(sorted(set(map(str,x))))).to_string())
print('\\nall individual/cell_type cell counts')
g=a.obs.groupby(['individual','cell_type'], observed=True).size().rename('cells').reset_index()
print(g.to_string(index=False))
print('retained >=10:', (g.cells>=10).sum(), 'of',len(g))
print('sanitized names unique:', (g.loc[g.cells>=10,'individual'].astype(str)+'_'+g.loc[g.cells>=10,'cell_type'].astype(str)).str.replace(r'[ -]','_',regex=True).is_unique)
X=a.layers['counts']
print('nnz',X.nnz,'min stored',X.data.min(),'max',X.data.max(),'all finite',np.isfinite(X.data).all(),'all nonnegative',(X.data>=0).all(),'all integer',np.equal(X.data,np.rint(X.data)).all())
print('total raw count',X.sum())
PY
