/usr/bin/bash -lc "/workspace/.analysis_env/bin/python - <<'PY'
from pathlib import Path
import numpy as np, pandas as pd, json
root=Path('/workspace')
de=pd.read_csv(root/'final_answer/differential_expression.tsv',sep='\\t')
p=de.p_value.to_numpy(); n=len(p); order=np.argsort(p,kind='mergesort')
q0=np.minimum.accumulate((p[order]*n/np.arange(1,n+1))[::-1])[::-1]
q=np.empty(n); q[order]=np.clip(q0,0,1)
d=np.abs(q-de.fdr.to_numpy())
i=np.argmax(d)
print('max_abs_diff',d[i], 'at', de.gene.iloc[i])
print('recomputed',repr(q[i]),'serialized',repr(de.fdr.iloc[i]),'p',repr(p[i]))
print('max_rel_diff',np.max(d/np.maximum(np.abs(q),np.finfo(float).tiny)))
print('method',json.loads((root/'final_answer/method.json').read_text()))
print(de.iloc[i].to_dict())
PY
head -n 3 final_answer/pseudobulk_counts.tsv
