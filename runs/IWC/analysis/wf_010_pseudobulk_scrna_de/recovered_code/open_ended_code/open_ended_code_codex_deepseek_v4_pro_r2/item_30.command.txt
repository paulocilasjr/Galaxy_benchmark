/usr/bin/bash -c "python - <<'PY'
import math, numpy as np, pandas as pd
counts=pd.read_csv('final_answer/pseudobulk_counts.tsv', sep='\\t')
cm=counts.set_index('gene')
totals=cm.sum(axis=0)
N=cm.shape[1]; M=float(np.median(totals)); S=math.ceil(10+0.7*(N-10))
cpm=cm.values/totals.values[None,:]*1_000_000
samples_above=(cpm>=10/M*1_000_000).sum(axis=1)
gene_sums=cm.values.sum(axis=1)
keep=(samples_above>=S)&(gene_sums>=1000)
print(N,M,S,10/M*1e6, keep.sum(), len(cm))
bad=cm.index[~keep]
print('bad count',len(bad))
for g in bad[:20]:
    i=cm.index.get_loc(g)
    print(g, 'above', samples_above[i], 'sum', gene_sums[i], 'cpm threshold', 10/M*1e6, 'maxcpm', cpm[i].max())
# compare for first good?
print(cm.index[keep][:5])
PY"
