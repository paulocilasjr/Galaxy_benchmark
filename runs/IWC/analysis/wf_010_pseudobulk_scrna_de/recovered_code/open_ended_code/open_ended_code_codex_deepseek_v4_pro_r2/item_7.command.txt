/usr/bin/bash -c "python - <<'PY'
import h5py
p='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
f=h5py.File(p,'r')
def visit(name,obj):
    if isinstance(obj,h5py.Dataset):
        print(name, obj.shape, obj.dtype, dict(obj.attrs))
    elif isinstance(obj,h5py.Group):
        print('GRP', name, dict(obj.attrs))
f.visititems(visit)
print('root attrs', dict(f.attrs))
f.close()
PY"
