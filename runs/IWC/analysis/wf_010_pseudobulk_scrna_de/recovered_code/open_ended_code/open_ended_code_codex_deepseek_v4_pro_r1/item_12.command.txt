/usr/bin/bash -c "python - <<'PY'
import anndata as ad, pandas as pd, numpy as np, scipy.sparse as sp
p='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
print('reading')
a=ad.read_h5ad(p)
print('loaded', a.shape, 'X dtype', a.X.dtype, 'X type', type(a.X))
counts=a.layers['counts'].tocsr()
print('counts nnz', counts.nnz, 'min', counts.data.min(), 'max', counts.data.max(), 'dtype', counts.dtype)
obs=a.obs.copy()
obs['group']=obs['individual'].astype(str)+'|'+obs['cell_type'].astype(str)
vc=obs.groupby(['individual','cell_type']).size().rename('ncells').reset_index()
print('combos all:', len(vc)); print(vc.to_string(index=False))
ret=vc[vc.ncells>=10].copy()
print('retained combos', len(ret), 'cells retained', ret.ncells.sum())
print(ret.to_string(index=False))
for i,r in ret.iterrows():
    idx=np.where((obs['individual']==r['individual'])&(obs['cell_type']==r['cell_type']))[0]
    s=counts[idx].sum(axis=0)
    arr=np.asarray(s).ravel()
    print(r['individual'], r['cell_type'], 'cells', len(idx), 'sum', arr.sum(), 'genes_nonzero', (arr>0).sum(), 'max', arr.max())
PY"
