/usr/bin/bash -c "python - <<'PY'
import anndata as ad, numpy as np, pandas as pd
p='data/inputs/source_anndata_file/Source%20AnnData%20file.h5ad'
a=ad.read_h5ad(p, backed='r')
print('shape', a.shape)
print('layers', list(a.layers.keys()))
print('obs columns', list(a.obs.columns))
print('var columns', list(a.var.columns))
for c in a.obs.columns:
    col=a.obs[c]
    if str(col.dtype) == 'object' or str(col.dtype) == 'category':
        print('obs', c, 'nunique', col.nunique(dropna=False), 'dtype', col.dtype, 'vals', list(col.drop_duplicates().astype(str).head(20)))
    else:
        print('obs', c, 'dtype', col.dtype, 'minmax', (float(col.min()), float(col.max())) if len(col) else None, 'nunique', col.nunique(dropna=False))
for c in a.var.columns:
    col=a.var[c]
    print('var', c, 'dtype', col.dtype, 'nunique', col.nunique(dropna=False), 'head', list(col.drop_duplicates().astype(str).head(5)))
print('obs index name', a.obs.index.name, a.obs.index[:5].tolist())
print('var index name', a.var.index.name, a.var.index[:5].tolist())
print('X type', type(a.X), getattr(a.X,'dtype',None), a.X.shape)
print('layer counts type', type(a.layers['counts']), getattr(a.layers['counts'],'dtype',None), a.layers['counts'].shape)
print(a.obs[['individual','cell_type']].head())
PY"
