/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
from pathlib import Path
import json, math
import numpy as np
import pandas as pd
pb_path=Path('final_answer/pseudobulk_counts.tsv')
de_path=Path('final_answer/differential_expression.tsv')
method_path=Path('final_answer/method.json')
pb=pd.read_csv(pb_path,sep='\\t')
de=pd.read_csv(de_path,sep='\\t')
print('pb shape',pb.shape,'columns',len(pb.columns),'unique genes',pb.gene.is_unique)
print('de shape',de.shape,'columns',de.columns.tolist(),'unique genes',de.gene.is_unique)
print('method',json.loads(method_path.read_text()))
print('pb schema',pb.columns[:5].tolist(),'...',pb.columns[-3:].tolist())
print('pb numeric all',pb.iloc[:,1:].apply(lambda s: pd.api.types.is_numeric_dtype(s)).all(),'integer all',np.all(np.isfinite(pb.iloc[:,1:].to_numpy())) and np.all(pb.iloc[:,1:].to_numpy()==np.floor(pb.iloc[:,1:].to_numpy())),'nonnegative',pb.iloc[:,1:].to_numpy().min()>=0)
print('gene sets same',set(pb.gene)==set(de.gene),'same order',pb.gene.tolist()==de.gene.tolist())
for col in ['log2_fold_change','p_value','fdr']:
 x=de[col].to_numpy(float)
 print(col,'finite',np.isfinite(x).all(),'range',x.min(),x.max())
print('BH monotone check sorted p/fdr',np.all(np.diff(de.sort_values('p_value').fdr.to_numpy())>=-1e-14))
# Independently calculate BH and compare.
p=de.p_value.to_numpy(float); n=len(p); order=np.argsort(p,kind='mergesort'); q=np.minimum.accumulate((p[order]*n/np.arange(1,n+1))[::-1])[::-1]; q=np.clip(q,0,1); out=np.empty(n); out[order]=q
print('BH exact max abs diff',np.max(np.abs(out-de.fdr.to_numpy(float))))
