/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv, re, yaml, pathlib
out=pathlib.Path('final_answer/asv_abundance.tsv')
expected=[e['identifier'] for e in yaml.safe_load(open('codex_visible/task_input.yaml'))['datasets']['Paired input data']['elements']]
with out.open() as fh:
    rows=list(csv.reader(fh,delimiter='\\t'))
assert rows and rows[0][0]=='sequence'
assert rows[0][1:]==expected, (rows[0][1:],expected)
seqs=[]; total=0; nonzero=0
for i,row in enumerate(rows[1:],2):
    assert len(row)==18, (i,len(row))
    s=row[0]; assert re.fullmatch(r'[ACGT]+',s), (i,s)
    seqs.append(s)
    for x in row[1:]:
        assert re.fullmatch(r'\\d+',x), (i,x)
        v=int(x); assert v>=0
        total+=v; nonzero+=v>0
assert len(seqs)==len(set(seqs))==(len(rows)-1)
assert total==223938
print('VALID', 'rows',len(seqs),'samples',len(expected),'total',total,'nonzero_cells',nonzero,'lengths',sorted(set(map(len,seqs))))
print('columns:', '\\t'.join(rows[0]))
PY"
