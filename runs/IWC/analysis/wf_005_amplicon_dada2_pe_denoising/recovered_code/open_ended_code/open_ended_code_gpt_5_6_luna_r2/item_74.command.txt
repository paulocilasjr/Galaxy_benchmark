/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import pandas as pd, numpy as np
p='work/dada2_runs/t240_r200_ee2/asv_abundance.tsv'; q='work/dada2_runs/t240_r160_ee2/asv_abundance.tsv'
a=pd.read_csv(p,sep='\\t').set_index('sequence'); b=pd.read_csv(q,sep='\\t').set_index('sequence')
print('shapes',a.shape,b.shape,'intersection',len(a.index.intersection(b.index)))
print('totals',a.to_numpy().sum(),b.to_numpy().sum())
print('sample totals r200',a.sum().to_dict()); print('sample totals r160',b.sum().to_dict())
# Correlation on shared ASV sample counts
ix=a.index.intersection(b.index); aa=a.loc[ix].to_numpy().ravel();bb=b.loc[ix].to_numpy().ravel();print('shared corr',np.corrcoef(aa,bb)[0,1],'shared sums',aa.sum(),bb.sum())
print('only r200 reads',a.drop(index=ix).to_numpy().sum(),'only r160 reads',b.drop(index=ix).to_numpy().sum())
print('sequence length counts r200',a.reset_index().sequence.str.len().value_counts().sort_index().to_dict(),'r160',b.reset_index().sequence.str.len().value_counts().sort_index().to_dict())
PY"
