from pathlib import Path
import gzip
from Bio.Seq import Seq
base=Path('data/inputs/paired_input_data')
def first(p):
    with gzip.open(p,'rt') as h:
        h.readline(); s=h.readline().strip(); h.readline(); q=h.readline().strip(); return s,q
for name in ['mfmt_cecal_1191_2','hfmt_cecal_1297_2','nofmt_cecal_1294_3']:
    f,qf=first(base/name/'forward/forward.fastqsanger.gz'); r,qr=first(base/name/'reverse/reverse.fastqsanger.gz'); rc=str(Seq(r).reverse_complement())
    print(name,len(f),len(r), 'F',f, 'RRC',rc)
    best=[]
    for off in range(-len(rc)+1,len(f)):
        # rc starts at position off in f coordinate; overlap
        lo=max(0,off); hi=min(len(f),off+len(rc));
        if hi<=lo: continue
        a=f[lo:hi]; b=rc[lo-off:hi-off]; matches=sum(x==y for x,y in zip(a,b)); mm=len(a)-matches
        best.append((matches/len(a),matches,mm,len(a),off))
    print('best',sorted(best,reverse=True)[:10])
