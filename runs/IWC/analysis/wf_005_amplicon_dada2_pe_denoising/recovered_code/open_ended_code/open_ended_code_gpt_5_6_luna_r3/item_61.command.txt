/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import csv, re
from pathlib import Path
p=Path('final_answer/asv_abundance.tsv')
expected=[]
for line in Path('codex_visible/task_input.yaml').read_text().splitlines():
    m=re.match(r'    - identifier: (.+)',line)
    if m: expected.append(m.group(1).strip())
with p.open() as h:
    rd=csv.reader(h,delimiter='\\t')
    header=next(rd); rows=list(rd)
assert header == ['sequence']+expected, (header, expected)
assert len(rows)==len({r[0] for r in rows})
assert all(re.fullmatch(r'[ACGT]+', r[0]) for r in rows)
assert all(len(r)==len(header) for r in rows)
vals=[[int(x) for x in r[1:]] for r in rows]
assert all(x>=0 for row in vals for x in row)
assert all(any(x>0 for x in row) for row in vals)
assert all(len(r[0])>=250 for r in rows)
print('VALID', 'samples=',len(header)-1,'ASVs=',len(rows),'columns=',len(header),'lengths=',sorted(set(map(lambda r:len(r[0]),rows))))
print('totals=', [sum(row[i] for row in vals) for i in range(len(expected))])
PY"
