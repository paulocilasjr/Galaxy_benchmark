/usr/bin/bash -c "python3 - <<'PY'
import gzip, glob, os
samples=sorted(set(p.split('/data/inputs/paired_input_data/')[1].split('/')[0] for p in glob.glob('/workspace/data/inputs/paired_input_data/*/forward/*.gz')))
print('samples',len(samples))
for s in samples[:3]:
    f=f'/workspace/data/inputs/paired_input_data/{s}/forward/forward.fastqsanger.gz'
    r=f'/workspace/data/inputs/paired_input_data/{s}/reverse/reverse.fastqsanger.gz'
    for file in [f,r]:
        n=0; lens=[]; qsum=None
        with gzip.open(file,'rt') as fh:
            for i,line in enumerate(fh):
                if i==0: print(file.split('/')[-4], 'first header', line.strip())
                if i%4==1:
                    seq=line.strip(); n+=1; lens.append(len(seq))
                if i%4==3:
                    q=line.strip();
                    if qsum is None: qsum=[0]*len(q)
                    for j,ch in enumerate(q): qsum[j]+=ord(ch)-33
                if n>=5000: break
        print(os.path.basename(file), 'n',n,'lens',sorted(set(lens))[:20], 'max',max(lens),'min',min(lens), 'mean first100 Q', [round(x/n,1) for x in qsum[:10]], 'mean last20 Q', [round(x/n,1) for x in qsum[-20:]])
    print()
PY"
