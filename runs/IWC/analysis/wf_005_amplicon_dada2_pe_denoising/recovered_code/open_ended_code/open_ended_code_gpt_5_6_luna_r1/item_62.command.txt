/usr/bin/bash -lc "/workspace/.denoise/bin/Rscript - <<'RS'
suppressPackageStartupMessages(library(dada2))
root <- 'data/inputs/paired_input_data'
samples <- c('mFMT_cecal_1191_2','mFMT_cecal_1191_3','mFMT_cecal_1275_1','mFMT_cecal_1275_2','mFMT_cecal_1275_4','hFMT_cecal_1296_1','hFMT_cecal_1296_2','hFMT_cecal_1296_3','hFMT_cecal_1297_1','hFMT_cecal_1297_2','hFMT_cecal_1297_4','noFMT_cecal_1192_1','noFMT_cecal_1192_3','noFMT_cecal_1294_1','noFMT_cecal_1294_2','noFMT_cecal_1294_3','noFMT_cecal_1295_4')
dirs <- list.dirs(root, recursive=FALSE, full.names=TRUE); lookup <- setNames(dirs,tolower(basename(dirs)))
indir <- vapply(samples,function(s) lookup[[tolower(s)]], character(1)); f <- file.path(indir,'forward','forward.fastqsanger.gz'); r <- file.path(indir,'reverse','reverse.fastqsanger.gz'); out <- 'work/filter_no_phix'; ff <- file.path(out,paste0(samples,'_F.fastq.gz')); rr <- file.path(out,paste0(samples,'_R.fastq.gz'))
y <- filterAndTrim(f,ff,r,rr,truncLen=c(240,200),maxN=0,maxEE=c(2,2),truncQ=2,rm.phix=FALSE,compress=TRUE,multithread=TRUE,matchIDs=FALSE,verbose=FALSE)
z <- read.table('work/dada2_240_200/filter_stats.tsv',header=TRUE,check.names=FALSE)
cat('same_filter_counts=',identical(unname(y[,1:2]),unname(as.matrix(z[,2:3]))),'\\n')
print(y)
RS"
