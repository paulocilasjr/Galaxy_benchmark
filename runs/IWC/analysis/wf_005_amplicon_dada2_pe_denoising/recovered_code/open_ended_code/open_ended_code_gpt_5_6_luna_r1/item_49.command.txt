/usr/bin/bash -lc "/workspace/.denoise/bin/Rscript - <<'RS'
suppressPackageStartupMessages(library(dada2))
f <- 'data/inputs/paired_input_data/hfmt_cecal_1296_1/forward/forward.fastqsanger.gz'
for (ee in c(2,5,10,20,Inf)) {
  out <- paste0('work/test_ee_',ee,'.fastq.gz')
  x <- tryCatch(fastqFilter(f, out, truncLen=240, maxN=0, maxEE=ee, truncQ=2, rm.phix=FALSE, compress=TRUE, verbose=TRUE, qualityType='Auto'), error=function(e)e)
  print(c(ee=ee)); print(x)
}
RS
ls -lh work/test_ee_* 2>/dev/null || true"
