sed -n '1,60p' run_trace/filter_candidates.tsv
/workspace/work/dada2_env/bin/python - <<'PY'
import json
q=json.load(open('run_trace/input_qc.json'))
for mate in ('forward','reverse'):
    total=sum(x[mate]['reads'] for x in q.values())
    weighted={}
    for p in ('200','220','230','240'):
        vals=[(x[mate]['quality_profile'][p]['mean_q'],x[mate]['reads']) for x in q.values() if p in x[mate]['quality_profile']]
        weighted[p]=round(sum(v*n for v,n in vals)/sum(n for _,n in vals),2)
    print(mate,total,weighted)
PY"
