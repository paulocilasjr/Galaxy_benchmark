/usr/bin/bash -lc "set -euo pipefail
PATH=/workspace/tools/bin:/opt/conda/bin:/usr/bin:/bin
samtools quickcheck -v /workspace/work/filtered.bam
samtools index -@ 4 /workspace/work/filtered.bam
samtools flagstat -@ 4 /workspace/work/filtered.bam > /workspace/work/qc/final_flagstat.txt
samtools stats -@ 4 /workspace/work/filtered.bam > /workspace/work/qc/final_samtools_stats.txt
samtools idxstats /workspace/work/filtered.bam > /workspace/work/qc/final_idxstats.txt
printf '%s\\n' '--- filtered BAM ---'
ls -lh /workspace/work/filtered.bam /workspace/work/filtered.bam.bai
printf '%s\\n' '--- flagstat ---'
sed -n '1,30p' /workspace/work/qc/final_flagstat.txt
printf '%s\\n' '--- markdup ---'
sed -n '1,80p' /workspace/work/qc/markdup_stats.txt
printf '%s\\n' '--- mitochondrial idxstats ---'
rg '"'^chrM'"\\t' /workspace/work/qc/final_idxstats.txt
printf '%s\\n' '--- core stats ---'
rg '"'^SN'"\\t(raw total sequences|reads mapped:|reads properly paired:|reads duplicated:|insert size average:|insert size standard deviation:)' /workspace/work/qc/final_samtools_stats.txt || true"
