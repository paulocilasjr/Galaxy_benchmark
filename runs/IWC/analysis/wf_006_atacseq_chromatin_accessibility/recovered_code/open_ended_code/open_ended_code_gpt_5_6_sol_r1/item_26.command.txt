/usr/bin/bash -lc "set -euo pipefail
PATH=/workspace/tools/bin:/opt/conda/bin:/usr/bin:/bin
fastp \\
  -i /workspace/data/inputs/pe_fastq_input/srr891269/forward/forward.fastqsanger.gz \\
  -I /workspace/data/inputs/pe_fastq_input/srr891269/reverse/reverse.fastqsanger.gz \\
  --stdout \\
  --adapter_sequence CTGTCTCTTATACACATCT \\
  --adapter_sequence_r2 CTGTCTCTTATACACATCT \\
  --overlap_len_require 10 \\
  --qualified_quality_phred 20 \\
  --unqualified_percent_limit 70 \\
  --n_base_limit 50 \\
  --length_required 15 \\
  --disable_trim_poly_g \\
  --dont_eval_duplication \\
  --thread 3 \\
  --json /workspace/work/qc/fastp.json \\
  --html /workspace/work/qc/fastp.html \\
  2>/workspace/work/qc/fastp.log \\
| bowtie2 \\
  -x /workspace/work/reference/hg19 \\
  --interleaved - \\
  --very-sensitive \\
  -X 2000 \\
  --no-mixed \\
  --no-discordant \\
  --threads 6 \\
  -S - \\
  2>/workspace/work/qc/bowtie2.log \\
| samtools view -h -f 2 -F 2304 - \\
| awk 'BEGIN { OFS=\"\\t\"; pending=0 }
       /"'^@/ { print; next }
       {
         if (!pending) {
           first=$0; qname=$1; chr=$3; mapq=$5+0; pending=1
         } else {
           if ($1 != qname) {
             print "ERROR: non-adjacent mates after primary proper-pair filter: " qname " vs " $1 > "/dev/stderr"
             exit 2
           }
           if (mapq >= 30 && ($5+0) >= 30 && chr != "chrM" && $3 != "chrM") {
             print first
             print $0
           }
           pending=0
         }
       }
       END {
         if (pending) {
           print "ERROR: unpaired terminal SAM record: " qname > "/dev/stderr"
           exit 3
         }
       }'"' \\
| samtools sort -n -@ 2 -m 768M -T /workspace/work/tmp/namesort - \\
| samtools fixmate -m -u - - \\
| samtools sort -@ 2 -m 768M -T /workspace/work/tmp/coordsort - \\
| samtools markdup -@ 2 -r -s -f /workspace/work/qc/markdup_stats.txt - /workspace/work/filtered.bam 
"
