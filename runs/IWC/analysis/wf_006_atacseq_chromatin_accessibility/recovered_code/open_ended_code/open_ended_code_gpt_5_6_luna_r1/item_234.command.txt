/usr/bin/bash -lc "./analysis/project_peaks.sh && wc -l final_answer/peaks.bed && awk -F '\\t' 'NF<3 || "'$1=="" || $2 !~ /''^[0-9]+$/ || $3 !~ /''^[0-9]+$/ || $2 >= $3 {bad++} END {print "invalid_records=" (bad+0)}'"' final_answer/peaks.bed && /opt/conda/bin/python - <<'PY'
from collections import Counter
n=0
c=Counter()
with open('final_answer/peaks.bed') as fh:
    for line in fh:
        f=line.rstrip('\\n').split('\\t')
        n += 1
        c[tuple(f[:3])] += 1
print('records=', n)
print('unique_coordinates=', len(c))
print('duplicate_coordinate_records=', sum(v-1 for v in c.values() if v > 1))
PY"
