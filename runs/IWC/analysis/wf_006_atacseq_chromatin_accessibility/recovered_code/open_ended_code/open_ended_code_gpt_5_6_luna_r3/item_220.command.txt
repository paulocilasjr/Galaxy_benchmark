/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import json
import pysam
p='/workspace/work/align/final.bam'
counts={'records':0,'bad_proper_pair':0,'bad_mapq':0,'bad_mito':0,'bad_duplicate':0,'bad_secondary_or_supplementary':0,'bad_unmapped':0,'read1':0,'read2':0}
with pysam.AlignmentFile(p,'rb') as bam:
    for rec in bam.fetch(until_eof=True):
        counts['records'] += 1
        counts['bad_proper_pair'] += not rec.is_proper_pair
        counts['bad_mapq'] += rec.mapping_quality < 30
        counts['bad_mito'] += rec.reference_name == 'chrM' or rec.next_reference_name == 'chrM'
        counts['bad_duplicate'] += rec.is_duplicate
        counts['bad_secondary_or_supplementary'] += rec.is_secondary or rec.is_supplementary
        counts['bad_unmapped'] += rec.is_unmapped or rec.mate_is_unmapped
        counts['read1'] += rec.is_read1
        counts['read2'] += rec.is_read2
print(json.dumps(counts,sort_keys=True))
assert counts['records'] == 24435892
assert counts['read1'] == counts['read2'] == 12217946
assert all(counts[k] == 0 for k in counts if k.startswith('bad_'))
PY"
