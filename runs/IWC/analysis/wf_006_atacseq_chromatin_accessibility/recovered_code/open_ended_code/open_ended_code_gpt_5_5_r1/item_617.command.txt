/usr/bin/bash -lc "/opt/conda/bin/python - <<'PY'
import json
required={'stage','status','tool_or_command','effective_parameters','inputs','outputs','validation'}
for i,line in enumerate(open('run_trace/analysis_steps.jsonl'),1):
    rec=json.loads(line)
    missing=required-set(rec)
    if missing:
        raise SystemExit(f'line {i} missing {missing}')
print('jsonl_ok')
PY
wc -l run_trace/analysis_steps.jsonl"
