/usr/bin/bash -lc 'mkdir -p /tmp/atac_fifo_bwa_test2
mkfifo /tmp/atac_fifo_bwa_test2/R1.fq /tmp/atac_fifo_bwa_test2/R2.fq
export PATH="/opt/conda/bin:/workspace/atac_env/bin:$PATH"
set -o pipefail
(
  fastp     --in1 /workspace/data/inputs/pe_fastq_input/srr891269/forward/forward.fastqsanger.gz     --in2 /workspace/data/inputs/pe_fastq_input/srr891269/reverse/reverse.fastqsanger.gz     --out1 /tmp/atac_fifo_bwa_test2/R1.fq     --out2 /tmp/atac_fifo_bwa_test2/R2.fq     --reads_to_process 100000     --detect_adapter_for_pe     --adapter_sequence CTGTCTCTTATACACATCT     --adapter_sequence_r2 CTGTCTCTTATACACATCT     --qualified_quality_phred 20     --unqualified_percent_limit 70     --length_required 15     --n_base_limit 50     --disable_trim_poly_g     --thread 4     --json /tmp/atac_fifo_bwa_test2/fastp.json     --html /tmp/atac_fifo_bwa_test2/fastp.html     2> /tmp/atac_fifo_bwa_test2/fastp.stderr
  echo $? > /tmp/atac_fifo_bwa_test2/fastp.status
) &
fp=$!
bwa mem -t 8 -M /workspace/atac_work/ref/hg19.fa   /tmp/atac_fifo_bwa_test2/R1.fq /tmp/atac_fifo_bwa_test2/R2.fq   > /tmp/atac_fifo_bwa_test2/out.sam   2> /tmp/atac_fifo_bwa_test2/bwa.stderr
bs=$?
wait $fp
fs=$(sed -n '"'1p' /tmp/atac_fifo_bwa_test2/fastp.status)
printf 'bwa=%s fastp=%s\\n' \""'$bs" "$fs"
sed -n '"'1,16p' /tmp/atac_fifo_bwa_test2/out.sam
tail -20 /tmp/atac_fifo_bwa_test2/bwa.stderr
wc -l /tmp/atac_fifo_bwa_test2/out.sam"
