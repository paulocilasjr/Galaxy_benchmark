/usr/bin/bash -lc 'mkdir -p /workspace/atac_work/fifo_test
mkfifo /workspace/atac_work/fifo_test/R1.fastq.gz /workspace/atac_work/fifo_test/R2.fastq.gz
gzip -cd /workspace/atac_work/fifo_test/R1.fastq.gz > /workspace/atac_work/fifo_test/R1.capture.fastq &
p1=$!
gzip -cd /workspace/atac_work/fifo_test/R2.fastq.gz > /workspace/atac_work/fifo_test/R2.capture.fastq &
p2=$!
/workspace/atac_env/bin/fastp   --in1 /workspace/data/inputs/pe_fastq_input/srr891269/forward/forward.fastqsanger.gz   --in2 /workspace/data/inputs/pe_fastq_input/srr891269/reverse/reverse.fastqsanger.gz   --out1 /workspace/atac_work/fifo_test/R1.fastq.gz   --out2 /workspace/atac_work/fifo_test/R2.fastq.gz   --reads_to_process 100000   --detect_adapter_for_pe   --adapter_sequence CTGTCTCTTATACACATCT   --adapter_sequence_r2 CTGTCTCTTATACACATCT   --qualified_quality_phred 20   --unqualified_percent_limit 70   --length_required 15   --n_base_limit 50   --disable_trim_poly_g   --thread 4   --json /workspace/atac_work/fifo_test/fastp.json   --html /workspace/atac_work/fifo_test/fastp.html   2> /workspace/atac_work/fifo_test/fastp.stderr
fp=$?
wait $p1
s1=$?
wait $p2
s2=$?
printf '"'statuses fastp=%s r1=%s r2=%s\\n' \""'$fp" "$s1" "$s2"
gzip -t /workspace/atac_work/fifo_test/R1.fastq.gz 2>/dev/null || true
wc -l /workspace/atac_work/fifo_test/R1.capture.fastq /workspace/atac_work/fifo_test/R2.capture.fastq
