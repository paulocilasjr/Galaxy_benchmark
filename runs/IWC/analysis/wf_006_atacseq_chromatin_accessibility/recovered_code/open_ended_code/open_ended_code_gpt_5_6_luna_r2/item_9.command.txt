/usr/bin/bash -lc "printf '%s\\n' '--- running related processes ---'
ps -eo pid,etime,cmd | rg 'gzip -t|md5sum|forward.fastqsanger|reverse.fastqsanger' || true
printf '%s\\n' '--- tool paths ---'
for x in fastp cutadapt bbduk.sh bowtie2 bowtie2-build bwa bwa-mem2 minimap2 samtools sambamba bedtools macs2 macs3 Genrich picard java seqkit pigz; do
  p="'$(command -v "$x" 2>/dev/null || true)
  if [ -n "$p" ]; then printf '"'%-14s %s\\n' \""'$x" "$p"; else printf '"'%-14s MISSING\\n' \""'$x"; fi
done
printf '"'%s\\n' '--- possible reference/index files ---'
find /workspace /data /opt /usr/local /reference /references /mnt -maxdepth 5 -type f \\( -iname '*hg19*' -o -iname '*b37*' -o -iname '*human*' -o -iname '*.fa' -o -iname '*.fasta' -o -iname '*.fai' -o -iname '*.bt2' -o -iname '*.bwt' -o -iname '*.amb' -o -iname '*.ann' \\) -printf '%s %p\\n' 2>/dev/null | sed -n '1,240p'"
