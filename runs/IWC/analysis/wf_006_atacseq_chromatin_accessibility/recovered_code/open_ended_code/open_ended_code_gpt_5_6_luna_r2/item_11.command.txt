/usr/bin/bash -lc 'export PATH="/opt/conda/bin:$PATH"
printf '"'%s\\n' '--- tool versions ---'
for x in bwa samtools bedtools bamToBed bamPEFragmentSize; do
  printf '%s: ' \""'$x"
  command -v "$x" || true
done
bwa 2>&1 | sed -n '"'1,3p'
samtools --version | sed -n '1,4p'
bedtools --version 2>&1
printf '%s\\n' '--- conda executables/packages ---'
command -v conda || true
find /opt/conda/bin -maxdepth 1 -type f -printf '%f\\n' | rg 'fast|macs|bowtie|cutadapt|trim|adapter|bam|sam' | sort
find /opt/conda/conda-meta -maxdepth 1 -type f -printf '%f\\n' | rg 'macs|fastp|cutadapt|bowtie|samtools|bwa|bedtools|sra' | sort"
