from pathlib import Path
L=16279
for mq in [0,1,10,20,40,60]:
    cov=[0]*L; ids=set(); kept=0
    for line in Path('work/reads/raw_vs_polished2.paf').open():
        f=line.split('\\t')
        if len(f)<12: continue
        q,ql,qs,qe,strand,t,tl,ts,te,match,alen,mapq=f[:12]
        match=int(match); alen=int(alen); ts=int(ts); te=int(te); mapq=int(mapq)
        ident=match/alen if alen else 0
        if alen>=1000 and ident>=0.97 and mapq>=mq:
            ids.add(q); kept+=1
            for i in range(max(0,ts), min(L,te)): cov[i]+=1
    print('mq',mq,'ids',len(ids),'alns',kept,'covmin',min(cov),'mean',round(sum(cov)/L,1),'max',max(cov),'lt10',sum(1 for x in cov if x<10))
PY"
