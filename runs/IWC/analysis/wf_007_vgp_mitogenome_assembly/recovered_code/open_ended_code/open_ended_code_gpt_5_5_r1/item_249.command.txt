/usr/bin/bash -lc 'PATH=/workspace/tools/mitoenv/bin:$PATH /workspace/tools/mitoenv/bin/minimap2 -x map-hifi -t4 work/assemblies/raven_anchor_unitig1_p0/candidate_16279.polished2.fasta data/inputs/collection_of_pacbio_data/err13917223/err13917223.fastqsanger.gz > work/reads/raw_vs_polished2.paf 2> work/logs/raw_vs_polished2.paf.log
/workspace/tools/mitoenv/bin/python - <<'"'PY'
from pathlib import Path
ids=set(); rows=0; kept=0; cov=[0]*16279
for line in Path('work/reads/raw_vs_polished2.paf').open():
    f=line.split('\\t')
    if len(f)<12: continue
    rows+=1
    q,ql,qs,qe,strand,t,tl,ts,te,match,alen,mapq=f[:12]
    match=int(match); alen=int(alen); ts=int(ts); te=int(te); mapq=int(mapq)
    ident=match/alen if alen else 0
    if alen>=1000 and ident>=0.97 and mapq>=20:
        ids.add(q); kept+=1
        for i in range(max(0,ts), min(len(cov),te)): cov[i]+=1
Path('work/reads/raw_candidate_read_ids.txt').write_text('\\n'.join(sorted(ids))+'\\n')
print('paf_rows',rows)
print('kept_alignments',kept)
print('selected_ids',len(ids))
print('paf_target_cov_min',min(cov),'mean',sum(cov)/len(cov),'max',max(cov),'lt10',sum(1 for x in cov if x<10))
PY
/workspace/tools/mitoenv/bin/seqkit grep -f work/reads/raw_candidate_read_ids.txt data/inputs/collection_of_pacbio_data/err13917223/err13917223.fastqsanger.gz | /workspace/tools/mitoenv/bin/seqkit fq2fa > work/reads/raw_candidate_reads.fasta
/workspace/tools/mitoenv/bin/seqkit stats -a work/reads/raw_candidate_reads.fasta > work/reads/raw_candidate_reads.stats.tsv
cat work/reads/raw_candidate_reads.stats.tsv"
