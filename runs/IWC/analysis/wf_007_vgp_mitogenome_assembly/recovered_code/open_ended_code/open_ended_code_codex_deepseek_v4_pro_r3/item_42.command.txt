from pathlib import Path
import re
ref=''.join(l.strip() for l in Path('reference/OZ203683.1.fa').read_text().splitlines() if not l.startswith('>'))
q=''.join(l.strip() for l in Path('assembly/mito_r_utg_500.fa').read_text().splitlines() if not l.startswith('>'))
# Parse primary SAM alignment from existing file (minimap output)
def parse_cigar(c):
    return [(int(n),op) for n,op in re.findall(r'(\\d+)([MIDNSHP=X])', c)]
for line in Path('assembly/r_to_ref.sam').read_text().splitlines():
    if line.startswith('@'): continue
    a=line.split('\\t')
    flag=int(a[1])
    if flag & 2048: continue
    # primary: target positions 1-based, q string same
    refstart=int(a[3])-1
    cig=parse_cigar(a[5])
    ri=refstart; qi=0; pairs=[]
    for n,op in cig:
        if op in 'M=X':
            for k in range(n):
                pairs.append((qi,ri,q[qi],ref[ri])); qi+=1; ri+=1
        elif op=='I':
            for k in range(n):
                pairs.append((qi,None,q[qi],'-')); qi+=1
        elif op=='D':
            for k in range(n):
                pairs.append((None,ri,'-',ref[ri])); ri+=1
        elif op=='S':
            qi+=n
        elif op=='H':
            pass
    for idx,(qi,ri,qc,rc) in enumerate(pairs):
        if qc"'!=rc:
            lo=max(0,idx-15); hi=min(len(pairs),idx+16)
            print('"'DIFF',idx,'qi',qi,'ri',ri,'q',qc,'r',rc)
            print('qctx',''.join((p[2] if p[2] else '') for p in pairs[lo:hi]))
            print('rctx',''.join((p[3] if p[3] else '') for p in pairs[lo:hi]))
    print('total pairs',len(pairs),'diffs',sum(1 for p in pairs if p[2]"'!=p[3]))
    break
