/usr/bin/bash -lc 'mkdir -p work/resfinder_official
zcat data/inputs/input_sequence_fasta/GCA_053668395.1_ASM5366839v1_genomic.fna.gz > work/input_sequence.fna
PATH=/workspace/.env/bin:$PATH run_resfinder.py '"\\
  -ifa work/input_sequence.fna \\
  -o work/resfinder_official \\
  -db_res refs/resfinder_db \\
  -db_point refs/pointfinder_db \\
  -acq -c \\
  -s \"Escherichia coli\" \\
  -l 0.60 -t 0.90 \\
  -l_p 0.95 -t_p 0.90 \\
  -b /workspace/.env/bin/blastn \\
  > work/resfinder_official/run.stdout 2> work/resfinder_official/run.stderr
find work/resfinder_official -maxdepth 2 -type f -printf '%p\\n' | sort
sed -n '1,160p' work/resfinder_official/run.stderr"
