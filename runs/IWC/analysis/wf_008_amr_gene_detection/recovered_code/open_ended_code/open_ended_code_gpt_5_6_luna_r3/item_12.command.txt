/usr/bin/bash -lc 'workdir=$(cat /workspace/.amr_work_path)
printf '"'%s\\n' '--- resolve target revisions ---'
for repo_commit in 'resfinder_db d1e607b' 'pointfinder_db 694919f'; do
  set -- "'$repo_commit
  printf '"'%s: ' \""'$1"
  git -C "$workdir/$1" rev-parse "$2''^{commit}"
done
printf '"'%s\\n' '--- target database summaries ---'
for repo_commit in 'resfinder_db d1e607b' 'pointfinder_db 694919f'; do
  set -- "'$repo_commit
  rev=$(git -C "$workdir/$1" rev-parse "$2''^{commit}")
  printf '"'\\n[%s %s]\\n' \""'$1" "$rev"
  git -C "$workdir/$1" show -s --format='"'%H%n%ad%n%s' --date=iso-strict \""'$rev"
  git -C "$workdir/$1" show "$rev:VERSION" 2>&1 || true
  git -C "$workdir/$1" show "$rev:README.md" 2>&1 | sed -n '"'1,80p'
done
printf '%s\\n' '--- ResFinder README usage ---'
sed -n '1,260p' \""'$workdir/resfinder/README.md"
printf '"'%s\\n' '--- pyproject ---'
sed -n '1,260p' \""'$workdir/resfinder/pyproject.toml"
printf '"'%s\\n' '--- scripts ---'
find \""'$workdir/resfinder" -maxdepth 3 -type f '"\\( -name '*.py' -o -name '*.sh' \\) -print | sort | sed -n '1,240p'"
