/usr/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
from pathlib import Path
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
from statsmodels.stats.multitest import multipletests

samples={'SRR5085167':'reference','SRR5085168':'reference','SRR5085169':'changed','SRR5085170':'changed'}
rows=[]
for s, cond in samples.items():
    group='counts_from_reference_condition' if cond=='reference' else 'counts_from_changed_condition'
    p=next(Path('/workspace/data/inputs').joinpath(group).glob(f'{s.lower()}_counts_table/*.dat'))
    d=pd.read_csv(p,sep='\\t'); d.columns=['Geneid',s]; rows.append(d)
counts=rows[0]
for d in rows[1:]: counts=counts.merge(d,on='Geneid',how='outer')
counts=counts.set_index('Geneid')
keep=(counts>=10).sum(axis=1)>=2
counts_filt=counts.loc[keep].astype(int).T
meta=pd.DataFrame({'condition': pd.Categorical([samples[c] for c in counts_filt.index], categories=['reference','changed'], ordered=True)}, index=counts_filt.index)
dds=DeseqDataSet(counts=counts_filt, metadata=meta, design='~condition', quiet=True)
dds.deseq2()
stat=DeseqStats(dds, contrast=['condition','changed','reference'], independent_filter=False, cooks_filter=False, quiet=True)
stat.summary()
res=stat.results_df.copy()
bh=multipletests(res['pvalue'], method='fdr_bh')[1]
print('max diff padj vs statsmodels', np.max(np.abs(res['padj'].values-bh)))
print('same?', np.allclose(res['padj'].values,bh,equal_nan=False))
print('finite check', np.isfinite(res[['log2FoldChange','pvalue','padj']].to_numpy()).all())
print('within range', (res['pvalue']>=0).all() & (res['pvalue']<=1).all() & (res['padj']>=0).all() & (res['padj']<=1).all())
print('gene id types', res.index[:5].tolist())
PY"
