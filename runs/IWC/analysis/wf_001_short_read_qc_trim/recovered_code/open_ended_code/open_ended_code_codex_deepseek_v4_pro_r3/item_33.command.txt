/usr/bin/bash -c "python3 - <<'PY'
import gzip, sys
p1='final_answer/trimmed_forward.fastqsanger.gz'
p2='final_answer/trimmed_reverse.fastqsanger.gz'
n=0
minlen=10**9
maxlen=0
low_pass=True
n_pass=True
sync=True
with gzip.open(p1,'rt') as f1, gzip.open(p2,'rt') as f2:
    while True:
        try:
            h1=next(f1); s1=next(f1); x1=next(f1); q1=next(f1)
            h2=next(f2); s2=next(f2); x2=next(f2); q2=next(f2)
        except StopIteration:
            break
        n+=1
        id1=h1.split()[0]; id2=h2.split()[0]
        if id1"'!=id2: sync=False
        for s,q in [(s1.strip(),q1.strip()),(s2.strip(),q2.strip())]:
            l=len(s)
            minlen=min(minlen,l); maxlen=max(maxlen,l)
            if l<15: low_pass=False
            if s.count('"'N')>5: n_pass=False
            low=sum(1 for c in q if ord(c)-33<20)
            if low > 0.4*l: low_pass=False
print('pairs', n, 'sync', sync, 'all_filters_pass', low_pass and n_pass, 'minlen', minlen, 'maxlen', maxlen)
PY"
