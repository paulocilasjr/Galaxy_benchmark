/usr/bin/bash -c "python3 - <<'PY'
import gzip, collections
for fn in ['data/inputs/raw_reads/pair/forward/forward.fastqsanger.gz','data/inputs/raw_reads/pair/reverse/reverse.fastqsanger.gz']:
    n=0; lens=collections.Counter(); minl=10**9; maxl=0
    with gzip.open(fn,'rt') as f:
      while True:
        h=f.readline()
        if not h: break
        s=f.readline().strip(); q=f.readline().strip(); q2=f.readline()
        n+=1; l=len(s); lens[l]+=1; minl=min(minl,l); maxl=max(maxl,l)
    print(fn,'n',n,'min',minl,'max',maxl,'lens',lens.most_common(15))
PY"
