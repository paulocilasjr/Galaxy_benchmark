/usr/bin/bash -c "python3 - <<'PY'
import gzip, pathlib, json
names = set()
with open('/workspace/work/valid_pair_names.txt') as f:
    names = {line.strip() for line in f}

def records(path):
    with gzip.open(path,'rt') as f:
        while True:
            h=f.readline().strip()
            if not h: break
            s=f.readline().strip(); p=f.readline().strip(); q=f.readline().strip()
            yield h,s,p,q

for inpath,outpath in [
('/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz','/workspace/final_answer/filtered_forward.fastqsanger.gz'),
('/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz','/workspace/final_answer/filtered_reverse.fastqsanger.gz')]:
    in_recs=list(records(inpath)); out_recs=list(records(outpath))
    # check exact record preservation and only expected removals
    exp=[r for r in in_recs if r[0].split()[0].lstrip('@') not in names]
    assert exp==out_recs, (inpath, len(exp), len(out_recs))
    print(pathlib.Path(inpath).name, 'preserved exactly', len(out_recs))
print('method:', open('/workspace/final_answer/method.json').read())
PY"
