/usr/bin/bash -c "gzip -t final_answer/filtered_forward.fastqsanger.gz && gzip -t final_answer/filtered_reverse.fastqsanger.gz && echo gzip_ok; python3 - <<'PY'
import gzip,json,os
for p in ['final_answer/filtered_forward.fastqsanger.gz','final_answer/filtered_reverse.fastqsanger.gz']:
    n=0
    with gzip.open(p,'rt') as f:
        while True:
            h=f.readline(); s=f.readline(); plus=f.readline(); q=f.readline()
            if not h: break
            assert h.startswith('@') and plus.startswith('+') and len(s)==len(q)
            n+=1
    print(p,n)
with open('final_answer/method.json') as f:
    m=json.load(f)
print(m)
print('final_answer files:', sorted(os.listdir('final_answer')))
PY"
