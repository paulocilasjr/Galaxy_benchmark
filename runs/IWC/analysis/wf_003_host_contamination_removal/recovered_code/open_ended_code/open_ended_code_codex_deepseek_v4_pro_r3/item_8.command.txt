/usr/bin/bash -c "python3 - <<'PY'
import gzip
for p in ['data/inputs/short_reads/pair/forward/forward.fastqsanger.gz','data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz']:
    n=0
    with gzip.open(p,'rt') as f:
        for line in f:
            if line.startswith('@'):
                n+=1
    print(p,n)
PY"
