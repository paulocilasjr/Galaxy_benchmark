import gzip, sys

def first_token(h):
    return h.rstrip('\\n').split()[0]

def read_records(path):
    recs={}
    order=[]
    with gzip.open(path,'rt') as f:
        while True:
            h=f.readline(); s=f.readline(); plus=f.readline(); q=f.readline()
            if not h: break
            if not (h.startswith('@') and plus.startswith('+') and len(s)==len(q)):
                raise RuntimeError(f'bad record in {path}: {h"'!r}'"')
            name=first_token(h)
            recs[name]=(h,s,plus,q)
            order.append(name)
    return recs, order

for fwd,rev in [('/workspace/analysis/unconc_1.gz','/workspace/analysis/unconc_2.gz')]:
    rf,of=read_records(fwd)
    rr,orv=read_records(rev)
    print(fwd, len(rf), len(rr), 'order_same', of==orv)
    if set(rf)"'!=set(rr):
        print('"'name mismatch', len(set(rf)-set(rr)), len(set(rr)-set(rf)))
    # verify all records match originals exactly for first few and spot random
    inf,ino=read_records('/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz')
    inr,inro=read_records('/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz')
    mismatch_fwd=[n for n in rf if n in inf and inf[n]"'!=rf[n]]
    mismatch_rev=[n for n in rr if n in inr and inr[n]!=rr[n]]
    print('"'record_mismatches_vs_input', len(mismatch_fwd), len(mismatch_rev))
