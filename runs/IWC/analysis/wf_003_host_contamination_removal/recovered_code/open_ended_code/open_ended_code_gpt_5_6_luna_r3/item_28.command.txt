/usr/bin/bash -lc "set -o pipefail
F=/workspace/final_answer/filtered_forward.fastqsanger.gz
R=/workspace/final_answer/filtered_reverse.fastqsanger.gz
IF=/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz
IR=/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz
printf '%s\\n' '--- corrected FASTQ structure checks ---'
for x in \""'$F" "$R"; do
  gzip -cd "$x" | awk -v name="$x" '"'NR%4==2{s="'$0} NR%4==3{if(substr($0,1,1)!="+") bad_plus++} NR%4==0{if(length(s)!=length($0)) bad_len++} END{if(NR%4!=0 || bad_plus || bad_len){printf "%s invalid: lines=%d records=%d bad_plus=%d bad_lengths=%d'"\\n\",name,NR,NR/4,bad_plus+0,bad_len+0; exit 1} printf \"%s valid: records=%d\\n\",name,NR/4}'
done
printf '%s\\n' '--- exact retained-record preservation and membership ---'
for pair in forward reverse; do
  if [ \""'$pair" = forward ]; then src="$IF"; out="$F"; else src="$IR"; out="$R"; fi
  awk '"'
  NR==FNR { r=(FNR-1)%4; if(r==0){h="'$0; input[h]=1; order[h]=++n} else if(r==1) seq[h]=$0; else if(r==2) plus[h]=$0; else qual[h]=$0; next }
  { r=(FNR-1)%4; if(r==0){h=$0; seen[h]++; if(!(h in input)) missing++; if((h in order) && order[h]<last) reordered++; last=order[h]} else if(r==1 && $0!=seq[h]) diff++; else if(r==2 && $0!=plus[h]) diff++; else if(r==3 && $0!=qual[h]) diff++ }
  END{for(h in seen) if(seen[h]>1) dup++; printf "%s: missing=%d changed_lines=%d duplicate_headers=%d order_decreases=%d'"\\n\", \"'\""'$pair"'"'\", missing+0,diff+0,dup+0,reordered+0; exit (missing+diff+dup)"'!=0}'"' <(gzip -cd \""'$src") <(gzip -cd "$out")
done
printf '"'%s\\n' '--- method JSON ---'
micromamba run -p /workspace/work/bowtie2_env python -c 'import json,sys; p=\"/workspace/final_answer/method.json\"; x=json.load(open(p)); assert list(x)==[\"aligner\"] and isinstance(x[\"aligner\"],str); print(x)'"
