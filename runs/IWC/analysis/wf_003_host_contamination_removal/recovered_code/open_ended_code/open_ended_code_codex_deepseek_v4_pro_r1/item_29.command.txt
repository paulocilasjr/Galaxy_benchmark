# Read all records and compare header key sets/order.
def read_records(path):
    recs=[]
    with gzip.open(path,'rb') as h:
        while True:
            header=h.readline()
            if not header: break
            seq=h.readline(); plus=h.readline(); qual=h.readline()
            assert seq and plus and qual, 'truncated'
            key=header.split()[0]
            if key.startswith(b'@'): key=key[1:]
            key=re.sub(rb'/[12]"'$'"',b'',key).decode()
            recs.append((key,header,seq,plus,qual))
    return recs
fwd=read_records(os.path.join(root,files[0]))

print('unique_forward_keys',len({k for k,_,_,_,_ in fwd}))

# Verify retained records exist byte-for-byte in the original files.
def original_by_key(path):
    d={}
    with gzip.open(path,'rb') as h:
        while True:
            header=h.readline()
            if not header: break
            seq=h.readline(); plus=h.readline(); qual=h.readline()
            key=header.split()[0]
            if key.startswith(b'@'): key=key[1:]
            key=re.sub(rb'/[12]"'$'"',b'',key).decode()
            d[key]=(header,seq,plus,qual)
    return d
