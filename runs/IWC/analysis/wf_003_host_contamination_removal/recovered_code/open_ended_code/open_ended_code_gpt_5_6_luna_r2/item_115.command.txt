/usr/bin/bash -lc "set -euo pipefail
gzip -t /workspace/final_answer/filtered_forward.fastqsanger.gz
gzip -t /workspace/final_answer/filtered_reverse.fastqsanger.gz
cmp <(gzip -cd /workspace/final_answer/filtered_forward.fastqsanger.gz) /workspace/work_hg38/expected_forward_original_v1.fastq
cmp <(gzip -cd /workspace/final_answer/filtered_reverse.fastqsanger.gz) /workspace/work_hg38/expected_reverse_original_v1.fastq
gzip -cd /workspace/final_answer/filtered_forward.fastqsanger.gz | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/final_forward_qnames_v1.txt
gzip -cd /workspace/final_answer/filtered_reverse.fastqsanger.gz | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/final_reverse_qnames_v1.txt
cmp /workspace/work_hg38/final_forward_qnames_v1.txt /workspace/work_hg38/final_reverse_qnames_v1.txt
printf 'final_forward_records='; awk 'END {print NR/4}' /workspace/work_hg38/expected_forward_original_v1.fastq
printf 'final_reverse_records='; awk 'END {print NR/4}' /workspace/work_hg38/expected_reverse_original_v1.fastq
awk 'NR%4==1 {if (substr("'$0,1,1)!="@") bad++} NR%4==2 {s=$0} NR%4==3 {if ($0!="+") bad++} NR%4==0 {if (length(s)!=length($0)) bad++} END {if (bad+0) exit 1; print "final_forward_fastq_errors=0"}'"' <(gzip -cd /workspace/final_answer/filtered_forward.fastqsanger.gz)
awk 'NR%4==1 {if (substr("'$0,1,1)!="@") bad++} NR%4==2 {s=$0} NR%4==3 {if ($0!="+") bad++} NR%4==0 {if (length(s)!=length($0)) bad++} END {if (bad+0) exit 1; print "final_reverse_fastq_errors=0"}'"' <(gzip -cd /workspace/final_answer/filtered_reverse.fastqsanger.gz)
micromamba run -n base python -c 'import json; x=json.load(open(\"/workspace/final_answer/method.json\")); assert list(x)==[\"aligner\"]; assert x[\"aligner\"]==\"Bowtie2\"; print(\"method_json_valid=yes\")'
find /workspace/final_answer -maxdepth 1 -type f -printf '%f\\n' | sort
printf '%s\\n' '---FINAL SHA256---'
sha256sum /workspace/final_answer/filtered_forward.fastqsanger.gz /workspace/final_answer/filtered_reverse.fastqsanger.gz /workspace/final_answer/method.json
printf '%s\\n' 'all_final_validations_passed=yes'"
