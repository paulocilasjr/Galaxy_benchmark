/usr/bin/bash -lc "gzip -cd /workspace/work_hg38/nonconcordant_v1.1 | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/retained_qnames_v1.txt
gzip -cd /workspace/work_hg38/nonconcordant_v1.2 | awk 'NR%4==1 {id="'$1; sub(/''^@/,"",id); print id}'"' > /workspace/work_hg38/retained_qnames_v1_r.txt
printf 'retained_qnames_forward='; wc -l < /workspace/work_hg38/retained_qnames_v1.txt
printf 'retained_qnames_reverse='; wc -l < /workspace/work_hg38/retained_qnames_v1_r.txt
cmp /workspace/work_hg38/retained_qnames_v1.txt /workspace/work_hg38/retained_qnames_v1_r.txt
printf '%s\\n' 'qname_lists_synchronized=yes'
awk 'NR==FNR {keep["'$1]=1; next} FNR%4==1 {id=$1; sub(/''^@/,"",id); take=(id in keep)} take'"' \\
  /workspace/work_hg38/retained_qnames_v1.txt \\
  <(gzip -cd /workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz) \\
  > /workspace/work_hg38/expected_forward_v1.fastq
awk 'NR==FNR {keep["'$1]=1; next} FNR%4==1 {id=$1; sub(/''^@/,"",id); take=(id in keep)} take'"' \\
  /workspace/work_hg38/retained_qnames_v1.txt \\
  <(gzip -cd /workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz) \\
  > /workspace/work_hg38/expected_reverse_v1.fastq
gzip -cd /workspace/work_hg38/nonconcordant_v1.1 > /workspace/work_hg38/nonconcordant_v1.1.fastq
gzip -cd /workspace/work_hg38/nonconcordant_v1.2 > /workspace/work_hg38/nonconcordant_v1.2.fastq
cmp /workspace/work_hg38/expected_forward_v1.fastq /workspace/work_hg38/nonconcordant_v1.1.fastq
cmp /workspace/work_hg38/expected_reverse_v1.fastq /workspace/work_hg38/nonconcordant_v1.2.fastq
printf '%s\\n' 'bowtie2_unconcordant_records_match_original_exactly=yes'
printf 'expected_forward_records='; awk 'END {print NR/4}' /workspace/work_hg38/expected_forward_v1.fastq
printf 'expected_reverse_records='; awk 'END {print NR/4}' /workspace/work_hg38/expected_reverse_v1.fastq
printf '%s\\n' '---UNCONCORDANT MD5---'
md5sum /workspace/work_hg38/nonconcordant_v1.1 /workspace/work_hg38/nonconcordant_v1.2"
