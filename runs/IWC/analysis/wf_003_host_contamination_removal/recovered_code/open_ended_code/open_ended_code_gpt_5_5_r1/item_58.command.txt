/usr/bin/bash -lc "set -euo pipefail
R1=/workspace/data/inputs/short_reads/pair/forward/forward.fastqsanger.gz
R2=/workspace/data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz
RM=/workspace/work/filter/remove_proper_pair_qnames.txt
TMP1=/workspace/work/filter/filtered_forward.fastqsanger.gz.tmp
TMP2=/workspace/work/filter/filtered_reverse.fastqsanger.gz.tmp
FILTER_PERL='use strict; use warnings; my "'$rmfile = shift @ARGV; open my $rm, "<", $rmfile or die "open remove list: $!'"\\n\"; my %remove; while (my "'$id = <$rm>) { chomp $id; $remove{$id} = 1; } my $records = 0; my $kept = 0; while (defined(my $h = <STDIN>)) { my $s = <STDIN>; my $p = <STDIN>; my $q = <STDIN>; die "truncated FASTQ after $h'"\\n\" unless defined "'$s && defined $p && defined $q; my ($id) = ($h =~ /''^@('"\\S+)/); die \"bad FASTQ header: "'$h'"\\n\" unless defined "'$id; $records++; if (!$remove{$id}) { print $h, $s, $p, $q; $kept++; } } warn "records=$records kept=$kept'"\\n\";'
printf 'filter_start_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /workspace/run_trace/execution.log
gzip -dc "$R1" | perl -e "$FILTER_PERL" "$RM" 2> /workspace/run_trace/filter_forward.stderr.log | gzip -c > "$TMP1"
gzip -dc "$R2" | perl -e "$FILTER_PERL" "$RM" 2> /workspace/run_trace/filter_reverse.stderr.log | gzip -c > "$TMP2"
gzip -t "$TMP1" "$TMP2"
mv "$TMP1" /workspace/final_answer/filtered_forward.fastqsanger.gz
mv "$TMP2" /workspace/final_answer/filtered_reverse.fastqsanger.gz
printf '"'filter_done_utc=%s\\n' \""'$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /workspace/run_trace/execution.log'
