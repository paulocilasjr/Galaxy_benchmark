/usr/bin/bash -lc 'for mate_file in data/inputs/short_reads/pair/forward/forward.fastqsanger.gz data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz; do gzip -cd "$mate_file" | awk -v label="$mate_file" '"'NR%4==1{headers++} NR%4==2{n=length("'$0); if(min==0||n<min)min=n; if(n>max)max=n} END{printf "%s: records=%d lines_mod4=%d read_length=%d-%d'"\\n\",label,headers,NR%4,min,max}'; done
paste <(gzip -cd data/inputs/short_reads/pair/forward/forward.fastqsanger.gz | awk 'NR%4==1{print "'$1}'"') <(gzip -cd data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz | awk 'NR%4==1{print "'$1}'"') | awk 'NF"'!=2{bad++; if(bad<=3) print "missing_mate",NR,$0; next} {a=$1;b=$2;sub(/''^@/,"",a);sub(/''^@/,"",b);sub(/'"\\/1"'$/, "", a);sub(/'"\\/2"'$/, "", b); if(a!=b){bad++; if(bad<=3)print "mismatch",NR,$1,$2}} END{print "paired_header_mismatches_or_missing=" bad+0; exit(bad?1:0)}'"'
gzip -cd data/inputs/short_reads/pair/forward/forward.fastqsanger.gz | sed -n '1,8p'
gzip -cd data/inputs/short_reads/pair/reverse/reverse.fastqsanger.gz | sed -n '1,8p'"
