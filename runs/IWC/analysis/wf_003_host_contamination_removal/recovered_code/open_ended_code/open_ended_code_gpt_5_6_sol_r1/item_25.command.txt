/usr/bin/bash -lc "paste <(gzip -cd /tmp/retained_1.fastqsanger.gz | awk 'NR%4==1{print "'$0}'"') <(gzip -cd /tmp/retained_2.fastqsanger.gz | awk 'NR%4==1{print "'$0}'"') | awk '
NF<4 {bad++; if(bad<=3) print \"missing mate at pair\",NR; next}
{
  f="'$1; r=$3
  sub(/''^@/,"",f); sub(/''^@/,"",r); sub(/'"\\/1"'$/,"",f); sub(/'"\\/2"'$/,"",r)
  if(f!=r){bad++; if(bad<=3) print "ID mismatch at pair",NR,$1,$3}
  if($2 !~ /'"\\/1"'$/ || $4 !~ /'"\\/2"'$/){labels++; if(labels<=3) print "mate label mismatch at pair",NR,$2,$4}
}
END{printf "retained_pair_sync: pairs=%d id_or_missing_errors=%d mate_label_errors=%d'"\\n\",NR,bad+0,labels+0; exit(bad||labels)}'
for retained_file in /tmp/retained_1.fastqsanger.gz /tmp/retained_2.fastqsanger.gz; do gzip -cd \""'$retained_file" | awk -v label="$retained_file" '"'
NR%4==1 && substr("'$0,1,1)!="@" {bad_header++}
NR%4==2 {sequence=$0}
NR%4==3 && substr($0,1,1)!="+" {bad_plus++}
NR%4==0 && length($0)!=length(sequence) {bad_length++}
END {printf "%s structure: records=%d lines_mod4=%d bad_headers=%d bad_plus=%d seq_qual_length_errors=%d'"\\n\",label,NR/4,NR%4,bad_header+0,bad_plus+0,bad_length+0; exit((NR%4)||bad_header||bad_plus||bad_length)}'; done"
