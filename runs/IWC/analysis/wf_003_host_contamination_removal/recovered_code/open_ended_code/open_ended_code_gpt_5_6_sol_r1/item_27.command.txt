/usr/bin/bash -lc "awk -v label='forward' '
FILENAME==ARGV[1] {
  h="'$0; if ((getline s)<=0 || (getline p)<=0 || (getline q)<=0) {truncated_orig++; next}
  split(h,a," "); key=a[1]; rec=h ORS s ORS p ORS q
  if (key in original) dup_orig++
  original[key]=rec; position[key]=++orig_n
  next
}
{
  h=$0; if ((getline s)<=0 || (getline p)<=0 || (getline q)<=0) {truncated_ret++; next}
  split(h,a," "); key=a[1]; rec=h ORS s ORS p ORS q
  if (key in seen_ret) dup_ret++
  seen_ret[key]=1; ret_n++
  if (!(key in original)) missing++
  else {
    if (original[key] != rec) modified++
    if (position[key] <= last_position) out_of_order++
    last_position=position[key]
  }
}
END {
  printf "%s: original=%d retained=%d missing=%d modified=%d out_of_order=%d duplicate_original_ids=%d duplicate_retained_ids=%d truncated=%d'"\\n\", label,orig_n,ret_n,missing,modified,out_of_order,dup_orig,dup_ret,truncated_orig+truncated_ret
  exit(missing||modified||out_of_order||dup_orig||dup_ret||truncated_orig||truncated_ret)
}' <(gzip -cd 'data/inputs/short_reads/pair/forward/forward.fastqsanger.gz') <(gzip -cd '/tmp/retained_1.fastqsanger.gz')"
