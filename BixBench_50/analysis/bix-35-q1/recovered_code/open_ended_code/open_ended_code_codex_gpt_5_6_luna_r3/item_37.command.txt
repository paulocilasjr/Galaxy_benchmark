/bin/bash -lc "PYTHONPATH=/workspace/.phykit_pkgs python - <<'PY'
import io, zipfile, hashlib
from Bio import SeqIO
base='/workspace/inputs'
with zipfile.ZipFile(base+'/scogs_animals.zip') as z:
    target=z.read('156083at2759.faa').decode()

def norm(txt):
    return [(r.id, str(r.seq)) for r in SeqIO.parse(io.StringIO(txt), 'fasta')]
print('scog_records', len(norm(target)))
for stem in ['Animals_Cele','Animals_Ggal','Animals_Mmus','Animals_Xlae']:
    with zipfile.ZipFile(f'{base}/{stem}.busco.zip') as z:
        txt=z.read('run_eukaryota_odb10/busco_sequences/single_copy_busco_sequences/156083at2759.faa').decode()
    # BUSCO entry can have a record identifier; compare sequence only.
    print(stem, len(norm(txt)), hashlib.sha256(norm(target)[['Animals_Cele','Animals_Ggal','Animals_Mmus','Animals_Xlae'].index(stem)][1].encode()).hexdigest()[:12], hashlib.sha256(norm(txt)[0][1].encode()).hexdigest()[:12])
PY"
