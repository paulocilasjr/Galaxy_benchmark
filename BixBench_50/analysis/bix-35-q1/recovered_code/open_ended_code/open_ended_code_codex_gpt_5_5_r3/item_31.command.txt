/bin/bash -lc "python - <<'PY'
from phykit.service_factories import EvolutionaryRate
import argparse, tempfile, zipfile, os
with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    tree=z.read('156083at2759.faa.mafft.clipkit.treefile').decode()
with tempfile.NamedTemporaryFile('w', dir='tmp', suffix='.treefile', delete=False) as fh:
    fh.write(tree)
    path=fh.name
try:
    EvolutionaryRate(argparse.Namespace(tree=path, json=False)).run()
finally:
    os.remove(path)
PY"
