/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
raw = pd.read_csv('inputs/count.csv', nrows=10)
counts=raw[['geneID','KL1','KL2','WL1','WL2']].set_index('geneID').T.astype(int)
for cats in [None, ['WL','KL'], ['KL','WL']]:
    meta=pd.DataFrame({'condition':['KL','KL','WL','WL']}, index=counts.index)
    if cats:
        meta['condition']=pd.Categorical(meta['condition'], categories=cats)
    dds=DeseqDataSet(counts=counts, metadata=meta, design='~ condition', quiet=True, n_cpus=1)
    print('cats', cats, 'obs dtype', dds.obs['condition'].dtype)
    print('obsm keys', dds.obsm.keys())
    print('design_matrix attr', getattr(dds,'design_matrix', 'NA'))
    print('formulaic design columns?', getattr(getattr(dds,'design_matrix',None),'columns',None))
PY"
