import pandas as pd
import numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

# Quick test to see the LFC column names
count_df = pd.read_csv("inputs/count.csv", index_col=0)
samples_all = list(count_df.columns)
condition_all = ["KL" if "KL" in s else "WL" for s in samples_all]
metadata_all = pd.DataFrame({"condition": condition_all}, index=samples_all)

dds_all = DeseqDataSet(counts=count_df, metadata=metadata_all, design="~condition")
dds_all.deseq2()

stat_res_all = DeseqStats(dds_all, contrast=["condition", "KL", "WL"])
stat_res_all.summary()

print("LFC columns:", stat_res_all.LFC.columns.tolist())
print("LFC shape:", stat_res_all.LFC.shape)
print("First few LFC rows:")
print(stat_res_all.LFC.head())
