import pandas as pd
import numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

count_df = pd.read_csv("inputs/count.csv", index_col=0)
counts_t = count_df.T
samples_all = list(counts_t.index)
condition_all = ["KL" if "KL" in s else "WL" for s in samples_all]
metadata_all = pd.DataFrame({"condition": condition_all}, index=samples_all)

dds = DeseqDataSet(counts=counts_t, metadata=metadata_all, design="~condition")
dds.deseq2()
stat = DeseqStats(dds, contrast=["condition", "KL", "WL"])
stat.summary()

print("LFC columns:", stat.LFC.columns.tolist())
print("Design matrix:")
print(dds.obsm["design_matrix"])
print()
print("First 5 LFC values:")
print(stat.LFC.head())
print()
# Check: which LFC column matches the results_df log2FoldChange?
# The contrast "KL" vs "WL" with WL as reference
# If KL is the reference level in the model, then condition[T.WL] = log2(WL/KL)
# And log2FoldChange in results should be log2(KL/WL) = -condition[T.WL]
print("Checking if results log2FoldChange = -condition[T.WL]:")
diff = stat.results_df["log2FoldChange"].dropna() + stat.LFC["condition[T.WL]"].dropna()
print("Max abs diff:", diff.abs().max())
print("Mean abs diff:", diff.abs().mean())

# Check the contrast info
print("\nContrast info:")
print(stat.contrast)
