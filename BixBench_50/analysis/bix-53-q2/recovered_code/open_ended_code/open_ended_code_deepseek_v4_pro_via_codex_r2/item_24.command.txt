/bin/bash -lc "python3 - <<'PY'
import pandas as pd
import numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw = pd.read_csv('/workspace/inputs/count.csv', index_col=0)

def analyze(samples):
    cnt = raw[samples].T.copy()
    meta = pd.DataFrame({'condition': [s[:2] for s in samples]}, index=samples)
    dds = DeseqDataSet(counts=cnt, metadata=meta, design=\"~C(condition, levels=['WL','KL'])\", quiet=True)
    dds.deseq2()
    ds = DeseqStats(dds, contrast=['condition','KL','WL'], alpha=0.05, quiet=True)
    ds.summary()
    coeff = [c for c in ds.LFC.columns if c "'!= '"'Intercept'][0]
    ds.lfc_shrink(coeff=coeff)
    r = ds.results_df.copy()
    print('\\nsamples', samples, 'shape', r.shape)
    print(r[['baseMean','log2FoldChange','pvalue','padj']].head(8).to_string())
    for pcol in ['padj','pvalue']:
        sig = (r[pcol] < 0.05) & (r['baseMean'] > 10) & (r['log2FoldChange'].abs() > 1)
        print(pcol, 'count', int(sig.sum()), 'up', int(((r.log2FoldChange>1)&sig).sum()), 'down', int(((r.log2FoldChange<-1)&sig).sum()), 'NA', r[pcol].isna().sum())
    return r

r_all = analyze(['KL1','KL2','KL3','WL1','WL2','WL3'])
r_sub = analyze(['KL1','KL2','WL1','WL2'])
PY"
