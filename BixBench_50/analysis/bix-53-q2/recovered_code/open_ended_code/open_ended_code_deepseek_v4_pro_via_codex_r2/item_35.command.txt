/bin/bash -lc "Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2))
counts <- read.csv('/workspace/inputs/count.csv', row.names=1, check.names=FALSE)
run_analysis <- function(samples, label) {
  ct <- counts[, samples, drop=FALSE]
  condition <- factor(sub(\"\\\\d+"'$","",samples), levels=c("WL","KL"))
  coldata <- data.frame(row.names=samples, condition=condition)
  dds <- DESeqDataSetFromMatrix(countData=ct, colData=coldata, design=~condition)
  dds <- DESeq(dds, quiet=TRUE)
  res <- results(dds, contrast=c("condition","KL","WL"), alpha=0.05)
  shr <- lfcShrink(dds, res=res, coef="condition_KL_vs_WL", type="apeglm", quiet=TRUE)
  dat <- as.data.frame(shr)
  cat('"'\\n', label, '\\n', sep='')
  cat('dim', nrow(dat), 'baseMean>10', sum("'!is.na(dat$baseMean) & dat$baseMean>10), '"'\\n')
  cat('padj NA', sum(is.na(dat"'$padj)), '"'pvalue NA', sum(is.na(dat"'$pvalue)), '"'\\n')
  for (pc in c('padj','pvalue')) {
    p <- dat[[pc]]
    keep <- "'!is.na(p) & p < 0.05 & !is.na(dat$baseMean) & dat$baseMean > 10 & !is.na(dat$log2FoldChange) & abs(dat$log2FoldChange) > 1
    cat(pc, '"'significant', sum(keep), 'up', sum(keep & dat"'$log2FoldChange>1), '"'down', sum(keep & dat"'$log2FoldChange< -1), '"'\\n')
  }
  invisible(dat)
}
run_analysis(c('KL1','KL2','KL3','WL1','WL2','WL3'), 'full KL1-3 vs WL1-3')
run_analysis(c('KL1','KL2','WL1','WL2'), 'no third replicates KL1-2 vs WL1-2')
RS"
