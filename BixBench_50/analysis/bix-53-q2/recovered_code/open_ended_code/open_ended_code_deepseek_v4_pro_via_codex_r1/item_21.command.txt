/bin/bash -c "Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2))
counts <- read.csv('inputs/count.csv', row.names=1, check.names=FALSE)
cat('dims', dim(counts), '\\n')
cat('colSums\\n'); print(colSums(counts))

run_analysis <- function(samples, alpha=0.05) {
  cmat <- counts[, samples, drop=FALSE]
  coldata <- data.frame(condition=factor(ifelse(grepl('"'^KL'"', samples), 'KL', 'WL'), levels=c('WL','KL')), row.names=samples)
  dds <- DESeqDataSetFromMatrix(countData=round(as.matrix(cmat)), colData=coldata, design=~condition)
  dds <- DESeq(dds, quiet=TRUE)
  res <- results(dds, alpha=alpha)
  # apeglm needs coef
  resLFC <- lfcShrink(dds, coef=resultsNames(dds)[2], type='apeglm', res=res, quiet=TRUE)
  print(resultsNames(dds))
  print(head(resLFC))
  sig <- resLFC[which(resLFC"'$pvalue < 0.05 & abs(resLFC$log2FoldChange) > 1 & resLFC$baseMean > 10),]
  cat('"'samples', paste(samples, collapse=','), 'n_nonNA', sum("'!is.na(resLFC$pvalue)), '"'sig_p_abs_bm', nrow(sig), '\\n')
  sig_padj <- resLFC[which(resLFC"'$padj < 0.05 & abs(resLFC$log2FoldChange) > 1 & resLFC$baseMean > 10),]
  cat('"'sig_padj_abs_bm', nrow(sig_padj), '\\n')
  return(list(res=resLFC, sig=sig))
}
full <- run_analysis(c('KL1','KL2','KL3','WL1','WL2','WL3'), alpha=0.05)
red <- run_analysis(c('KL1','KL2','WL1','WL2'), alpha=0.05)
RS"
