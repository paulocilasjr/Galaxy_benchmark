/bin/bash -c "Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2))
counts <- read.csv('inputs/count.csv', row.names=1, check.names=FALSE)
run <- function(samples, alpha=0.1, shrink_type='apeglm') {
  cmat <- round(as.matrix(counts[,samples]))
  coldata <- data.frame(condition=factor(ifelse(grepl('"'^KL'"', samples), 'KL', 'WL'), levels=c('WL','KL')), row.names=samples)
  dds <- DESeqDataSetFromMatrix(cmat, coldata, ~condition)
  dds <- DESeq(dds, quiet=TRUE)
  res <- results(dds, alpha=alpha)
  resLFC <- lfcShrink(dds, coef=resultsNames(dds)[2], type=shrink_type, res=res, quiet=TRUE)
  c(p=sum(resLFC"'$pvalue < 0.05 & abs(resLFC$log2FoldChange)>1 & resLFC$baseMean>10, na.rm=TRUE),
    padj=sum(resLFC$padj < 0.05 & abs(resLFC$log2FoldChange)>1 & resLFC$baseMean>10, na.rm=TRUE))
}
for (a in c(0.05,0.1)) {
 cat('"'alpha', a, 'full', run(c('KL1','KL2','KL3','WL1','WL2','WL3'), a), '\\n')
 cat('alpha', a, 'red', run(c('KL1','KL2','WL1','WL2'), a), '\\n')
}
# normal shrinkage alpha .05
for (s in c('normal')) {
 cat('type',s,'full',run(c('KL1','KL2','KL3','WL1','WL2','WL3'),0.05,s),'\\n')
 cat('type',s,'red',run(c('KL1','KL2','WL1','WL2'),0.05,s),'\\n')
}
RS"
