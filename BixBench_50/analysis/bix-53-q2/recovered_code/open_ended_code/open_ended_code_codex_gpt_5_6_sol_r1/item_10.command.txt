/bin/bash -lc "python - <<'PY'
import csv
p='inputs/count.csv'
with open(p,newline='') as f:
 r=csv.reader(f); h=next(r); seen=set(); dup=0; n=0; nonint=neg=0; allzero=0
 for row in r:
  n+=1
  dup += row[0] in seen; seen.add(row[0])
  try: vals=[int(x) for x in row[1:]]
  except ValueError: nonint+=1; continue
  neg += any(x<0 for x in vals)
  allzero += all(x==0 for x in vals)
print('header',h)
print('genes',n,'duplicates',dup,'noninteger_rows',nonint,'negative_rows',neg,'all_zero_genes',allzero)
PY"
