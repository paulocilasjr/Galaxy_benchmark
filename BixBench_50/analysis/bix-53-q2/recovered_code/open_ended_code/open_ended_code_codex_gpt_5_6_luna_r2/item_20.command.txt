/bin/bash -lc "PYTHONPATH=/workspace/pylib python3 - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw = pd.read_csv('inputs/count.csv')
counts = raw.set_index('geneID').T
counts.index = counts.index.astype(str)
metadata = pd.DataFrame({'condition': ['KL','KL','KL','WL','WL','WL']}, index=counts.index)
print(counts.shape, metadata)
dds = DeseqDataSet(counts=counts, metadata=metadata, design='~condition', ref_level=['condition','WL'], quiet=True, n_cpus=1)
dds.deseq2()
print('LFC columns', dds.varm['LFC'].shape, dds.varm['LFC'].columns if hasattr(dds.varm['LFC'],'columns') else 'array')
print('nonzero', len(dds.non_zero_genes), 'size factors', dds.obs['size_factors'].to_dict())
stats = DeseqStats(dds, contrast=['condition','KL','WL'], quiet=True, n_cpus=1)
stats.summary()
print(stats.results_df.head())
print('attrs', [x for x in dir(stats) if 'shrink' in x.lower()])
stats.lfc_shrink(coeff='condition_KL_vs_WL')
print('shrunk head', stats.results_df.head())
print('shrunk stats', stats.results_df[['log2FoldChange','pvalue','padj']].describe())
PY"
