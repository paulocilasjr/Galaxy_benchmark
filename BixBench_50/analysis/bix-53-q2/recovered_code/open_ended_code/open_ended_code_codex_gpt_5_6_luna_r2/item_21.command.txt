/bin/bash -lc "PYTHONPATH=/workspace/pylib python3 -u - <<'PY'
import pandas as pd, traceback
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
try:
 raw = pd.read_csv('inputs/count.csv')
 counts = raw.set_index('geneID').T
 metadata = pd.DataFrame({'condition': ['KL']*3+['WL']*3}, index=counts.index)
 print('start',counts.shape,flush=True)
 dds = DeseqDataSet(counts=counts, metadata=metadata, design='~condition', ref_level=['condition','WL'], quiet=True, n_cpus=1)
 print('dds',flush=True)
 dds.deseq2()
 print('fit done',flush=True)
 stats = DeseqStats(dds, contrast=['condition','KL','WL'], quiet=True, n_cpus=1)
 stats.summary()
 print('summary done',stats.results_df.head().to_string(),flush=True)
 print('cols',list(dds.varm['LFC'].columns),flush=True)
 stats.lfc_shrink(coeff='condition_KL_vs_WL')
 print('shrink done',stats.results_df.head().to_string(),flush=True)
except Exception as e:
 traceback.print_exc()
 raise
PY"
