/bin/bash -lc "python3 - <<'PY'
import json, urllib.request
# Debian package search API
for suite in ['bookworm','bullseye','sid','trixie']:
    url=f'https://sources.debian.org/api/src/bwa/'
    try:
        data=json.load(urllib.request.urlopen(url))
        print(data.get('package'), [(v['version']) for v in data.get('versions',[])[:10]])
        break
    except Exception as e:
        print('err',suite,e)
PY
curl -sSL --max-time 30 https://deb.debian.org/debian/pool/main/b/bwa/ | head -c 30000"
