/bin/bash -lc 'set -o pipefail
mkdir -p /workspace/.work
BWA=/workspace/.tools/bwa/root/usr/bin/bwa
REF=/workspace/inputs/GCF_000005845.2_ASM584v2_genomic.fna
R1=/workspace/inputs/SRR35233585_1_paired.fastq.gz
R2=/workspace/inputs/SRR35233585_2_paired.fastq.gz
BAM=/workspace/.work/SRR35233585.trimmed.sorted.bam
"$BWA" mem -t 8 -R '"'@RG\\tID:SRR35233585\\tSM:SRR35233585\\tPL:ILLUMINA\\tLB:lib1\\tPU:unit1' \""'$REF" "$R1" "$R2" 2>/workspace/.work/bwa_mem.log | samtools sort -@ 8 -m 64M -o "$BAM" - 2>/workspace/.work/samtools_sort.log
samtools index -@ 8 "$BAM"
echo '"'ALIGNMENT_DONE'
ls -lh \""'$BAM" "$BAM.bai"
echo '"'READ_GROUP:'
samtools view -H \""'$BAM" | grep '"'"'^@RG'"'
echo 'FLAGSTAT:'
samtools flagstat \""'$BAM" | head -n 12
echo '"'BWA_LOG:'
cat /workspace/.work/bwa_mem.log"
