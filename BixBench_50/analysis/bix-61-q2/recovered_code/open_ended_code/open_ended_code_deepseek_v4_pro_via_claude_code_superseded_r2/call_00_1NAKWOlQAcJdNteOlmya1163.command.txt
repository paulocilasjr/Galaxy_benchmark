/tmp/mm_extract/bin/micromamba run -n bwa_env bash -c '
set -euo pipefail

REF=/workspace/inputs/GCF_000005845.2_ASM584v2_genomic.fna
R1=/workspace/inputs/SRR35233585_1_paired.fastq.gz
R2=/workspace/inputs/SRR35233585_2_paired.fastq.gz
RG="@RG\\tID:SRR35233585\\tSM:SRR35233585\\tPL:ILLUMINA\\tLB:lib1\\tPU:unit1"
OUTDIR=/workspace/tmp_bwa

echo "=== Running BWA-MEM ==="
bwa mem -M -t 4 -R "$RG" "$REF" "$R1" "$R2" | \
  samtools sort -@ 4 -o "$OUTDIR/SRR35233585.sorted.bam" -

echo "=== Indexing BAM ==="
samtools index "$OUTDIR/SRR35233585.sorted.bam"

echo "=== Calculating depth ==="
samtools depth -a "$OUTDIR/SRR35233585.sorted.bam" > "$OUTDIR/SRR35233585.depth.txt"
wc -l "$OUTDIR/SRR35233585.depth.txt"

echo "=== Computing average depth ==="
awk "{sum+=\$3; count++} END {if(count>0) printf \"%.4f\\n\", sum/count}" "$OUTDIR/SRR35233585.depth.txt"
'
