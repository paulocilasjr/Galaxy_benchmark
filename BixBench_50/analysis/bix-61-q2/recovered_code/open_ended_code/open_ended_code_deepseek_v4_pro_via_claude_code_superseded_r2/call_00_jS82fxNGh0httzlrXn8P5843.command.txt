import bz2, tarfile, io, os, stat

with open('/tmp/micromamba', 'rb') as f:
    raw = f.read()

data = bz2.decompress(raw)
print(f"Decompressed: {len(data)} bytes")

with open('/tmp/micromamba.tar', 'wb') as f:
    f.write(data)

os.makedirs('/tmp/mm_extract', exist_ok=True)
with tarfile.open('/tmp/micromamba.tar', 'r') as tar:
    tar.extractall('/tmp/mm_extract')

for root, dirs, files in os.walk('/tmp/mm_extract'):
    for f in files:
        fpath = os.path.join(root, f)
        if 'micromamba' in f or 'mamba' in f or 'bin/' in root:
            sz = os.path.getsize(fpath)
            print(f"Found: {fpath} ({sz} bytes)")
            st = os.stat(fpath)
            if st.st_mode & stat.S_IXUSR:
