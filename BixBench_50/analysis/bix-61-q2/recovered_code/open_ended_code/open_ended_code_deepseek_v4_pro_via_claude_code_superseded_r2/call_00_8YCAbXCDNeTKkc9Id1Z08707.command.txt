import bz2
with open('/tmp/micromamba', 'rb') as f:
    raw = f.read()
    # BZh9 starts bzip2
    if raw[:3] == b'BZh':
        print('Is bzip2, decompressing...')
        data = bz2.decompress(raw)
        print(f'Decompressed size: {len(data)}')
        # Check if it's a tar
        if data[:2] == b'\x1f\x8b':
            print('Is gzip within')
        else:
            print('First 100 bytes:', data[:100])
" 2>&1
