python3 -c "
import urllib.request
import tarfile
import io

url = 'https://micro.mamba.pm/api/micromamba/linux-aarch64/latest'
print('Downloading...')
data = urllib.request.urlopen(url).read()
print(f'Downloaded {len(data)} bytes')

# Try to decompress bzip2 in python
import bz2
decompressed = bz2.decompress(data)
print(f'Decompressed to {len(decompressed)} bytes')

with tarfile.open(fileobj=io.BytesIO(decompressed), mode='r') as tf:
    import os
    os.makedirs('/tmp/micromamba', exist_ok=True)
    tf.extractall('/tmp/micromamba')
print('Extracted')
" 2>&1
