/bin/bash -lc "set -o pipefail
printf '%s\\n' 'Reference index stats:'
samtools idxstats /workspace/work/SRR35233585.sorted.bam
printf '%s\\n' 'Depth extrema and zero count (no depth cap):'
samtools depth -aa -d 0 /workspace/work/SRR35233585.sorted.bam \\
  | awk 'BEGIN{n=0;s=0;z=0;mx=0} {n++;s+="'$3;if($3==0)z++;if($3>mx)mx=$3} END{printf "positions=%d zeros=%d max=%d mean=%.10f'"\\n\",n,z,mx,s/n}'
printf '%s\\n' 'Default samtools depth comparison:'
samtools depth -aa /workspace/work/SRR35233585.sorted.bam \\
  | awk 'BEGIN{n=0;s=0} {n++;s+="'$3} END{printf "positions=%d mean=%.10f'"\\n\",n,s/n}'"
