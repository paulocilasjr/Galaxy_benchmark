printf '%s\\n' 'Depth command options:'

samtools depth -aa -d 0 /workspace/work/SRR35233585.sorted.bam \\
  | awk 'BEGIN{n=0; s=0} {n++; s += "'$3} END {printf "positions=%d total_depth=%.0f mean=%.10f'"\\n\", n, s, s/n}'"
