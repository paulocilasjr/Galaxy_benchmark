/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
x=defaultdict(list)
with open('inputs/Swarm_1.csv', newline='') as f:
    for row in csv.DictReader(f):
        x[row['Genotype']].append(float(row['Area']))
for g in ('Wildtype','ΔlasR'):
    print(g, x[g], sum(x[g])/len(x[g]))
wt=sum(x['Wildtype'])/len(x['Wildtype'])
mut=sum(x['ΔlasR'])/len(x['ΔlasR'])
print('percent reduction', (wt-mut)/wt*100)
PY"
