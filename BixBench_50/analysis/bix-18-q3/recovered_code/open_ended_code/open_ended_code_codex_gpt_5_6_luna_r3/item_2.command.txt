/bin/bash -lc "python - <<'PY'
from csv import DictReader
from statistics import mean
from pathlib import Path
p=Path('inputs/Swarm_1.csv')
with p.open(newline='', encoding='utf-8') as f:
    rows=list(DictReader(f))
areas={g: mean(float(row['Area']) for row in rows if row['Genotype']==g) for g in ('ΔlasR','Wildtype')}
reduction=(areas['Wildtype']-areas['ΔlasR'])/areas['Wildtype']*100
print(areas)
print(reduction)
PY"
