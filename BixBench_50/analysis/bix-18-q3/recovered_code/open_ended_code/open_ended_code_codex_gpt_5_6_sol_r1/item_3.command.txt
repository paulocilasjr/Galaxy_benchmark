/bin/bash -lc "python - <<'PY'
import csv
from pathlib import Path
p=Path('inputs/Swarm_1.csv')
rows=list(csv.DictReader(p.open()))
means={}
for genotype in ('Wildtype','ΔlasR'):
    vals=[float(r['Area']) for r in rows if r['Genotype']==genotype]
    means[genotype]=sum(vals)/len(vals)
reduction=(means['Wildtype']-means['ΔlasR'])/means['Wildtype']*100
print(means)
print(repr(reduction))
PY"
