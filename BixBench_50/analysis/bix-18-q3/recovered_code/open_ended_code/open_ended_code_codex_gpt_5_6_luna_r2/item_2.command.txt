/bin/bash -lc "python3 - <<'PY'
import csv
from pathlib import Path
p=Path('inputs/Swarm_1.csv')
with p.open(newline='') as f:
    rows=list(csv.DictReader(f))
for g in ('ΔlasR','Wildtype'):
    vals=[float(r['Area']) for r in rows if r['Genotype']==g]
    print(g, len(vals), sum(vals)/len(vals))
wt=sum(float(r['Area']) for r in rows if r['Genotype']=='Wildtype')/3
mut=sum(float(r['Area']) for r in rows if r['Genotype']=='ΔlasR')/3
reduction=(wt-mut)/wt*100
print('reduction', reduction)
PY"
