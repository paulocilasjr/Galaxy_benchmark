/bin/bash -lc "python - <<'PY'
from io import StringIO
from itertools import combinations
from statistics import median
from zipfile import ZipFile
from Bio import Phylo
zpath='inputs/scogs_fungi.zip'
member='981902at2759.faa.mafft.clipkit.treefile'
with ZipFile(zpath) as z:
    newick=z.read(member).decode().strip()
tree=Phylo.read(StringIO(newick), 'newick')
names=[t.name for t in tree.get_terminals()]
dists=[(a,b,tree.distance(a,b)) for a,b in combinations(names,2)]
print('tree:',newick)
print('taxa:',len(names),names)
for row in dists:
    print(*row,sep='\\t')
vals=[d for _,_,d in dists]
print('sorted:', sorted(vals))
print('median:', format(median(vals),'.10f'))
PY"
