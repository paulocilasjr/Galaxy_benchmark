import dendropy
import numpy as np

# Read the tree
tree = dendropy.Tree.get(
    path="/workspace/work/981902at2759.faa.mafft.clipkit.treefile",
    schema="newick",
    preserve_underscores=True
)

# Get leaf nodes
leaf_nodes = list(tree.leaf_node_iter())
print("Leaves:", [n.taxon.label for n in leaf_nodes])

# Compute all pairwise patristic distances
pdm = tree.phylogenetic_distance_matrix()

distances = []
for i in range(len(leaf_nodes)):
    for j in range(i+1, len(leaf_nodes)):
        taxon1 = leaf_nodes[i].taxon
        taxon2 = leaf_nodes[j].taxon
        d = pdm.patristic_distance(taxon1, taxon2)
        distances.append(d)
        print(f"{taxon1.label} <-> {taxon2.label}: {d}")

print()
sorted_distances = sorted(distances)
print("Sorted distances:", sorted_distances)
print(f"Median: {np.median(sorted_distances)}")
