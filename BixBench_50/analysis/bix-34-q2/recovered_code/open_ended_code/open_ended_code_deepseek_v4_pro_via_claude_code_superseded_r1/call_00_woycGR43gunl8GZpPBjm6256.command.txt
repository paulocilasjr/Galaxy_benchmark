import dendropy
import numpy as np

# Read the tree - dendropy replaces underscores with spaces by default
# Let's preserve underscores
tree = dendropy.Tree.get(
    path="/workspace/work/981902at2759.faa.mafft.clipkit.treefile",
    schema="newick",
    preserve_underscores=True
)

# Print tree structure
print("Tree:", tree.as_string(schema="newick"))
print()

# Get all leaf nodes
taxa = [leaf.taxon.label for leaf in tree.leaf_node_iter()]
print("Taxa:", taxa)
print()

# Compute all pairwise patristic distances
pdm = tree.phylogenetic_distance_matrix()
distances = []
for i, taxon1 in enumerate(taxa):
    for j, taxon2 in enumerate(taxa):
        if i < j:
            d = pdm(taxon1, taxon2)
            distances.append(d)
            print(f"{taxon1} <-> {taxon2}: {d}")

print()
sorted_distances = sorted(distances)
print("Sorted distances:", sorted_distances)
print(f"Median: {np.median(sorted_distances)}")

# Also compute per-taxon distances
print()
for taxon in taxa:
    dists = [pdm(taxon, other) for other in taxa if other != taxon]
    print(f"Distances from {taxon}: {dists}, median: {np.median(dists)}")
