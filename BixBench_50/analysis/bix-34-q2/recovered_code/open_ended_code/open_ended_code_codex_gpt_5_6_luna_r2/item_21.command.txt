/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from statistics import median
from itertools import combinations

member = '981902at2759.faa.mafft.clipkit.treefile'
with ZipFile('inputs/scogs_fungi.zip') as z:
    newick = z.read(member).decode().strip()

# Minimal Newick parser retaining branch lengths and parent/child topology.
class Node:
    def __init__(self, name=None, length=0.0):
        self.name = name
        self.length = length
        self.children = []
        self.parent = None

i = 0

def parse_subtree():
    global i
    if newick[i] == '(':
        i += 1
        node = Node()
        while True:
            child = parse_subtree()
            child.parent = node
            node.children.append(child)
            if newick[i] == ',':
                i += 1
                continue
            if newick[i] == ')':
                i += 1
                break
            raise ValueError(f'unexpected {newick[i]"'!r} at {i}'"')
        start = i
        while i < len(newick) and newick[i] not in ':,();':
            i += 1
        if i > start:
            node.name = newick[start:i]
    else:
        start = i
        while i < len(newick) and newick[i] not in ':,();':
            i += 1
        node = Node(newick[start:i])
    if i < len(newick) and newick[i] == ':':
        i += 1
        start = i
        while i < len(newick) and newick[i] not in ',();':
            i += 1
        node.length = float(newick[start:i])
    return node

root = parse_subtree()
leaves = []
def walk(n):
    if n.children:
        for c in n.children:
            walk(c)
    else:
        leaves.append(n)
walk(root)

def distance(a, b):
    anc = {}
    d = 0.0
    x = a
    while x is not None:
        anc[id(x)] = d
        d += x.length
        x = x.parent
    d = 0.0
    x = b
    while id(x) not in anc:
        d += x.length
        x = x.parent
    return d + anc[id(x)]

pairs = [(a.name, b.name, distance(a,b)) for a,b in combinations(leaves,2)]
vals = [x[2] for x in pairs]
print('tree:', newick)
for p in pairs:
    print('%s-%s %.10f' % p)
print('n_pairs:', len(vals))
print('median: %.10f' % median(vals))
PY"
