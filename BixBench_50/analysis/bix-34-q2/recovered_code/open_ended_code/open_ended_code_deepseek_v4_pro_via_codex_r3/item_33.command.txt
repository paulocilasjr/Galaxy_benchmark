/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://api.github.com/repos/Future-House/BixBench/git/trees/main?recursive=1'
try:
 data=json.load(urllib.request.urlopen(url, timeout=20))
 for item in data.get('tree',[]):
  p=item['path']
  if any(s in p.lower() for s in ['phyl','busco','scog','bixbench','capsule']):
   print(item['type'], p)
except Exception as e: print('ERR',e)
PY"
