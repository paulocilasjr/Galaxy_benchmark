import zipfile, collections
p='inputs/scogs_fungi.zip'
with zipfile.ZipFile(p) as z:
    c=collections.Counter()
    for n in z.namelist():
        if n.endswith('.treefile'): k='treefile'
        elif n.endswith('.mldist'): k='mldist'
        elif n.endswith('.bionj'): k='bionj'
        elif n.endswith('.iqtree'): k='iqtree'
        elif n.endswith('.log'): k='log'
        elif n.endswith('.model.gz'): k='model.gz'
        elif n.endswith('.ckp.gz'): k='ckp.gz'
        elif n.endswith('.mafft.clipkit'): k='clipkit'
        elif n.endswith('.mafft'): k='mafft'
        elif n.endswith('.faa'): k='faa'
        else: k=n.rsplit('.',1)[-1]
        c[k]+=1
    print('members',len(z.namelist()))
    print(c)
PY
python - <<'PY'
from io import StringIO
import zipfile, statistics
try:
    from Bio import Phylo
except Exception as e:
    print('Bio import failed:',e)
    raise
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    nwk=z.read('981902at2759.faa.mafft.clipkit.treefile').decode()
tree=Phylo.read(StringIO(nwk),'newick')
leaves=tree.get_terminals()
vals=[]
for i,a in enumerate(leaves):
  for b in leaves[i+1:]:
    d=tree.distance(a,b)
    vals.append(d)
    print(a.name,b.name,format(d,'.10f'))
print('sorted', [format(x,'.10f') for x in sorted(vals)])
print('median',format(statistics.median(vals),'.10f'))
PY"
