with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:

# Minimal Newick parser for labels and branch lengths; builds an undirected weighted graph.
s = newick.rstrip(';')
i = 0
node_id = 0
graph = defaultdict(list)
labels = {}

def new_node(label=None):
    global node_id
    n = f'__n{node_id}'
    node_id += 1
    if label:
        labels[label] = n
    return n

def parse_label_len():
    global i
    label_chars = []
    while i < len(s) and s[i] not in ':,()':
        label_chars.append(s[i]); i += 1
    label = ''.join(label_chars).strip() or None
    length = 0.0
    if i < len(s) and s[i] == ':':
        i += 1
        start = i
        while i < len(s) and s[i] not in ',()':
            i += 1
        length = float(s[start:i])
    return label, length

def parse_subtree():
    global i
    if s[i] == '(':
        i += 1
        parent = new_node()
        while True:
            child, child_len = parse_subtree()
            graph[parent].append((child, child_len))
            graph[child].append((parent, child_len))
            if s[i] == ',':
                i += 1
                continue
            if s[i] == ')':
                i += 1
                break
        label, length = parse_label_len()
        if label:
            labels[label] = parent
        return parent, length
    label, length = parse_label_len()
    return new_node(label), length

root, root_len = parse_subtree()
assert i == len(s), (i, len(s), s[i:])
assert abs(root_len) < 1e-15

def dist(a, b):
    start, goal = labels[a], labels[b]
    stack = [(start, None, 0.0)]
    while stack:
        node, parent, d = stack.pop()
        if node == goal:
            return d
        for nxt, w in graph[node]:
            if nxt "'!= parent:
                stack.append((nxt, node, d + w))
    raise RuntimeError('"'disconnected')

pairs = []
for a, b in itertools.combinations(sorted(labels), 2):
