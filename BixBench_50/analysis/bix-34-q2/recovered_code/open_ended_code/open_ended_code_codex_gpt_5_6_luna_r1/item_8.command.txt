# Minimal Newick parser for named tips and branch lengths.
class Node:
    def __init__(self, name=None, length=0.0, children=None):
        self.name = name
        self.length = length
        self.children = children or []
        self.parent = None
        for child in self.children:
            child.parent = self

def parse_newick(s):
    s = s.rstrip(';').strip()
    i = 0
    def subtree():
        nonlocal i
        children = []
        if i < len(s) and s[i] == '(':
            i += 1
            while True:
                children.append(subtree())
                if s[i] == ',':
                    i += 1
                    continue
                if s[i] == ')':
                    i += 1
                    break
                raise ValueError(f'bad Newick at {i}: {s[i:]}')
        start = i
        while i < len(s) and s[i] not in ',():':
            i += 1
        name = s[start:i].strip() or None
        length = 0.0
        if i < len(s) and s[i] == ':':
            i += 1
            start = i
            while i < len(s) and s[i] not in ',():' :
                i += 1
            length = float(s[start:i])
        return Node(name, length, children)
    root = subtree()
    if i "'!= len(s):
        raise ValueError(f'"'unparsed Newick at {i}: {s[i:]}')
    return root

root = parse_newick(newick)
tips = []
def collect(n):
    if n.children:
        for c in n.children:
            collect(c)
    else:
        tips.append(n)
collect(root)
assert len(tips) == 4

def patristic(a, b):
    da = {}
    x = a
    dist = 0.0
    while x is not None:
        da[x] = dist
        dist += x.length
        x = x.parent
    x = b
    dist = 0.0
    while x not in da:
        dist += x.length
        x = x.parent
    return da[x] + dist

    pairs.append((a.name, b.name, patristic(a, b)))
