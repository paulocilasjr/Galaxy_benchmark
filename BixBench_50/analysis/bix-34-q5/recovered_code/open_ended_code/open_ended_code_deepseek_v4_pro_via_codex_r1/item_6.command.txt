/bin/bash -c "python3 - <<'PY'
import zipfile
for z in ['/workspace/inputs/scogs_animals.zip','/workspace/inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        names=f.namelist()
    print(z, len(names))
    from collections import Counter
    ext=Counter()
    prefixes=set()
    for n in names:
        base=n.split('/')[-1]
        # extensions after .faa
        parts=base.split('.')
        if 'faa' in parts:
            i=parts.index('faa')
            suffix='.'.join(parts[i+1:]) or 'faa'
        else:
            suffix='?'
        ext[suffix]+=1
    print(ext.most_common(30))
    # list basenames prefixes by extension
    for suff in ['treefile','iqtree','mldist','bionj','model.gz','mafft.clipkit','mafft']:
        ex=[n for n in names if n.endswith('.'+suff)]
        print(suff, len(ex), ex[:5])
PY"
