/bin/bash -c "python3 - <<'PY'
import zipfile
for z in ['/workspace/inputs/scogs_animals.zip','/workspace/inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        for n in f.namelist():
            if n.endswith('.treefile'):
                print('===',z,n)
                data=f.read(n).decode()
                print(data[:2000])
                break
        # sample mldist
        for n in f.namelist():
            if n.endswith('.mldist'):
                print('--- mldist',n)
                print(f.read(n).decode()[:2000])
                break
PY"
