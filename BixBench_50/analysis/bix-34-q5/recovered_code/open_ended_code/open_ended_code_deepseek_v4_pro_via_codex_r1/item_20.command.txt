/bin/bash -c "python3 - <<'PY'
import zipfile, os
for group in ['animals','fungi']:
 z=f'/workspace/inputs/scogs_{group}.zip'
 with zipfile.ZipFile(z) as f: names=f.namelist()
 bases=set()
 types={}
 for n in names:
  b=n.split('/')[-1]
  if '.faa' in b:
   stem=b.split('.faa')[0]
  else: continue
  bases.add(stem)
  parts=b.split('.faa',1)[1] if '.faa' in b else ''
  # suffix
  suff=parts.lstrip('.') if parts else 'faa'
  types.setdefault(stem,[]).append(suff)
 print('\\n',group,'faa bases',len(bases))
 from collections import Counter
 c=Counter(tuple(sorted(v)) for v in types.values())
 print(c.most_common(20))
 # for bases with treefile count etc
 has_tree=sum(1 for v in types.values() if 'mafft.clipkit.treefile' in v)
 print('has tree',has_tree)
 # example types
 for k,v in list(types.items())[:3]: print(k,v)
 # missing tree bases
 missing=[k for k,v in types.items() if 'mafft.clipkit.treefile' not in v]
 print('missing tree',len(missing), missing[:10])
PY"
