/bin/bash -c "python3 - <<'PY'
import zipfile,io,statistics,itertools
from Bio import Phylo
from collections import defaultdict

def tree_dists(newick):
    tree=Phylo.read(io.StringIO(newick),'newick')
    terms=tree.get_terminals()
    d={}
    for a,b in itertools.combinations(terms,2): d[(a.name,b.name)]=tree.distance(a,b)
    return d
for group in ['animals','fungi']:
 with zipfile.ZipFile(f'/workspace/inputs/scogs_{group}.zip') as f:
  per_taxon=[]
  per_tree_means=[]
  per_tree_medians=[]
  for n in f.namelist():
   if n.endswith('.treefile'):
    d=tree_dists(f.read(n).decode())
    # mean per tree
    per_tree_means.append(statistics.mean(d.values()))
    # per taxon mean within tree
    taxa=set()
    for a,b in d:
     taxa.add(a); taxa.add(b)
    for t in taxa:
     vals=[v for (a,b),v in d.items() if a==t or b==t]
     per_taxon.append(statistics.mean(vals))
 print(group,'per taxon mean median',statistics.median(per_taxon), 'mean',statistics.mean(per_taxon), 'n',len(per_taxon))
print('ratio', statistics.median([]))
PY"
