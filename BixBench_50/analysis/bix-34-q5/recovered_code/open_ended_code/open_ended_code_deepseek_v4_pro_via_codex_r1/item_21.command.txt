/bin/bash -c "python3 - <<'PY'
import zipfile, glob, os
from collections import defaultdict

def read_statuses(path):
    with zipfile.ZipFile(path) as f:
        data=f.read('run_eukaryota_odb10/full_table.tsv').decode()
    lines=data.splitlines()[3:]
    out=defaultdict(list)
    for line in lines:
        parts=line.split('\\t')
        if len(parts)<3: continue
        out[parts[0]].append(parts[1])
    return out
for group, species in [('animals',['Cele','Ggal','Mmus','Xlae']),('fungi',['Ncra','Scer','Spom','Umay'])]:
    stats=[read_statuses(f'/workspace/inputs/{group.capitalize()}_{sp}.busco.zip') for sp in species]
    # single copy complete in all
    genes=set(stats[0])
    for st in stats[1:]: genes &= set(st)
    sc=set(g for g in genes if all(st[g]==['Complete'] for st in stats))
    print(group, 'all complete single',len(sc))
    # compare scogs tree stems
    with zipfile.ZipFile(f'/workspace/inputs/scogs_{group}.zip') as f:
        tree_stems=set(n.split('/')[-1].split('.faa')[0] for n in f.namelist() if n.endswith('.treefile'))
    print('tree count',len(tree_stems),'intersection with sc',len(tree_stems & sc),'only tree not sc',len(tree_stems-sc), list(tree_stems-sc)[:5])
    print('only sc not tree',len(sc-tree_stems),list(sc-tree_stems)[:5])
PY"
