/bin/bash -c "python3 - <<'PY'
import zipfile,io,statistics,itertools
from Bio import Phylo

def analyze(group):
    z=f'/workspace/inputs/scogs_{group}.zip'
    with zipfile.ZipFile(z) as f:
        means=[]; counts={}; missing=[]
        for n in f.namelist():
            if not n.endswith('.treefile'): continue
            tree=Phylo.read(io.StringIO(f.read(n).decode()),'newick')
            terms=tree.get_terminals()
            ds=[tree.distance(a,b) for a,b in itertools.combinations(terms,2)]
            means.append(statistics.mean(ds))
            counts[len(terms)]=counts.get(len(terms),0)+1
            if len(terms)<4: missing.append((n, len(terms), [t.name for t in terms], statistics.mean(ds)))
    print(group, 'n',len(means),'counts',counts,'median',statistics.median(means))
    print('all4 median',statistics.median([m for m,c in zip(means, [None]) if True]))
    # recreate with taxon count
    all4=[]; 
    with zipfile.ZipFile(z) as f:
        for n in f.namelist():
            if not n.endswith('.treefile'): continue
            tree=Phylo.read(io.StringIO(f.read(n).decode()),'newick')
            terms=tree.get_terminals()
            if len(terms)==4:
                ds=[tree.distance(a,b) for a,b in itertools.combinations(terms,2)]
                all4.append(statistics.mean(ds))
    print('all4 n',len(all4),'median',statistics.median(all4))
    print('missing sample',missing[:10], 'total missing', len(missing))
for g in ['animals','fungi']:
    analyze(g)
PY"
