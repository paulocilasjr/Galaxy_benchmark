from Bio import Phylo
import io, itertools, zipfile, statistics

def custom_dists(s):
    pos=0
    def parse_node():
        nonlocal pos
        while pos < len(s) and s[pos].isspace(): pos+=1
        children=[]; label=''; length=None
        if s[pos]=='(':
            pos+=1
            while True:
                children.append(parse_node())
                while pos < len(s) and s[pos].isspace(): pos+=1
                if s[pos]==',': pos+=1; continue
                elif s[pos]==')': pos+=1; break
        else:
            st=pos
            while pos<len(s) and s[pos] not in ':(),;': pos+=1
            label=s[st:pos].strip()
        while pos<len(s) and s[pos].isspace(): pos+=1
        if pos<len(s) and s[pos]==':':
            pos+=1
            st=pos
            while pos<len(s) and (s[pos].isdigit() or s[pos] in '.eE+-'): pos+=1
            length=float(s[st:pos])
        return {'label':label,'length':length,'children':children}
    root=parse_node(); leaves=[]
    def collect(node,d):
        if not node['children']:
            leaves.append((node['label'],d)); return
        for ch in node['children']:
            collect(ch,d+(ch['length'] if ch['length'] is not None else 0.0))
    collect(root,0.0)
    return [(a[0],b[0],a[1]+b[1]) for a,b in itertools.combinations(leaves,2)]

s='(Animals_Cele:1.7692000750,(Animals_Ggal:0.2602162586,Animals_Mmus:0.2641311680):0.0900936810,Animals_Xlae:0.2951345137);'
print('custom', custom_dists(s))
t=Phylo.read(io.StringIO(s),'newick')
terms=t.get_terminals()
print('bio',[(a.name,b.name,t.distance(a,b)) for a,b in itertools.combinations(terms,2)])
print('terms',[x.name for x in terms])
# custom mean
print('custom mean', statistics.mean([d for _,_,d in custom_dists(s)]))
print('bio mean', statistics.mean([t.distance(a,b) for a,b in itertools.combinations(terms,2)]))
PY"
