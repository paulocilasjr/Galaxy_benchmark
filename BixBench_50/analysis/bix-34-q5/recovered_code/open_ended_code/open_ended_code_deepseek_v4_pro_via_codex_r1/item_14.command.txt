/bin/bash -c "python3 - <<'PY'
import zipfile, io, statistics, itertools, math
from Bio import Phylo
from collections import defaultdict

def tree_dists(newick):
    tree=Phylo.read(io.StringIO(newick),'newick')
    terms=tree.get_terminals()
    out={}
    for a,b in itertools.combinations(terms,2):
        out[(a.name,b.name)]=tree.distance(a,b)
    return out

def get_tree_dist_dict(group):
    z=f'/workspace/inputs/scogs_{group}.zip'
    with zipfile.ZipFile(z) as f:
        arr=[]
        for n in f.namelist():
            if n.endswith('.treefile'):
                d=tree_dists(f.read(n).decode())
                arr.append(d)
    return arr

for group in ['animals','fungi']:
    arr=get_tree_dist_dict(group)
    print('\\n',group,'trees',len(arr))
    # per tree mean/median/gmean
    means=[statistics.mean(d.values()) for d in arr]
    medians=[statistics.median(d.values()) for d in arr]
    print('per-tree means median',statistics.median(means),'mean',statistics.mean(means))
    print('per-tree medians median',statistics.median(medians))
    # per pair means across trees
    pairs=sorted(set().union(*[set(d.keys()) for d in arr]))
    pairvals={p:[d[p] for d in arr if p in d] for p in pairs}
    pm=[statistics.mean(v) for v in pairvals.values()]
    print('pair mean median',statistics.median(pm),'mean',statistics.mean(pm))
    print('pair mean range',min(pm),max(pm))
    # maybe median per pair across trees then mean? 
    pm2=[statistics.median(v) for v in pairvals.values()]
    print('pair median median',statistics.median(pm2),'mean',statistics.mean(pm2))
    for p,v in sorted(pairvals.items()): print(p,len(v),round(statistics.mean(v),4),round(statistics.median(v),4))
PY"
