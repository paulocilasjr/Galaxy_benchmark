from decimal import Decimal, getcontext
from statistics import median
from collections import Counter
getcontext().prec=50
class N:
    __slots__=('ch','p','bl','name')
    def __init__(self): self.ch=[]; self.p=None; self.bl=Decimal(0); self.name=''
def parse(s):

                if s[i]==',': i+=1

            st=i
            while i<len(s) and s[i] not in ',():;': i+=1
            n.name=s[st:i].strip()

            st=i
            while i<len(s) and s[i] not in ',():;': i+=1
            n.name=s[st:i].strip()

            i+=1; st=i

def check(s):

    def walk(n):

        for c in n.ch:
