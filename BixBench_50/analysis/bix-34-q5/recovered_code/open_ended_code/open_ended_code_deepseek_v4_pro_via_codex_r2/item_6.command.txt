/bin/bash -c "python3 - <<'PY'
import zipfile, glob
for f in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    z=zipfile.ZipFile(f)
    print('\\n===',f,'===')
    trees=[n for n in z.namelist() if n.endswith('.treefile')]
    mlds=[n for n in z.namelist() if n.endswith('.mldist')]
    print('trees',len(trees),'mld',len(mlds))
    for arr in [trees,mlds]:
      if arr:
        n=arr[0]
        print('sample',n)
        print(z.read(n).decode('utf-8','replace')[:1000])
    # inspect bionj too
    b=[n for n in z.namelist() if n.endswith('.bionj')]
    print('bionj',len(b))
    if b: print(z.read(b[0]).decode()[:1000])
PY"
