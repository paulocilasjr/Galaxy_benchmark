/bin/bash -c "python3 - <<'PY'
import urllib.request
base='https://raw.githubusercontent.com/mims-harvard/ToolUniverse/main/skills/tooluniverse-phylogenetics/scripts/'
for f in ['scogs_paired_compare.py','scogs_phykit_pipeline.py','phykit_batch.py']:
    url=base+f
    try:
        data=urllib.request.urlopen(url, timeout=30).read().decode()
        print('---',f,len(data),'---')
        print(data[:1000])
        open('/tmp/'+f,'w').write(data)
    except Exception as e: print(f,e)
PY"
