import zipfile, statistics, re

def parse_mldist(text):
    lines=[l.strip() for l in text.splitlines() if l.strip()]
    n=int(lines[0])
    labels=[]; vals=[]
    for row in lines[1:]:
        parts=row.split()
        labels.append(parts[0]); vals.append([float(x) for x in parts[1:]])
    ds=[]
    for i in range(n):
        for j in range(i+1,n):
            ds.append(vals[i][j])
    return ds

for f in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    z=zipfile.ZipFile(f)
    means=[]; ns=[]
    for name in z.namelist():
        if name.endswith('.mldist'):
            s=z.read(name).decode()
            ds=parse_mldist(s)
            means.append(sum(ds)/len(ds)); ns.append(len(ds))
    med=statistics.median(means)
    print(f, len(means), 'median mean mldist', med, 'mean',statistics.mean(means))
