/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter,defaultdict
from pathlib import Path
import re
for p in [Path('inputs/scogs_animals.zip'),Path('inputs/scogs_fungi.zip')]:
    print('\\n',p)
    with ZipFile(p) as z:
        names=z.namelist()
        suf=Counter()
        trees=[]
        for n in names:
            base=n.rsplit('/',1)[-1]
            if base.endswith('.treefile'):
                suf['treefile']+=1; trees.append(n)
            elif base.endswith('.iqtree'): suf['iqtree']+=1
            elif base.endswith('.mafft.clipkit'): suf['clipkit']+=1
            elif base.endswith('.mafft'): suf['mafft']+=1
            elif base.endswith('.faa'): suf['faa']+=1
            elif base.endswith('.mldist'): suf['mldist']+=1
            elif base.endswith('.model.gz'): suf['model.gz']+=1
            elif base.endswith('.log'): suf['log']+=1
            elif base.endswith('.bionj'): suf['bionj']+=1
            elif base.endswith('.ckp.gz'): suf['ckp.gz']+=1
            else: suf['other']+=1
        print('entries',len(names),'suffix counts',dict(suf))
        taxa=Counter(); nleaves=Counter(); examples=[]; all_names=[]
        for n in trees:
            s=z.read(n).decode().strip()
            labels=re.findall(r'(?:(?<=\\()|(?<=,))(["'^():,]+):'"',s)
            taxa.update(labels); nleaves[len(labels)]+=1
            all_names.append(n.rsplit('/',1)[-1].split('.')[0])
            if len(examples)<5: examples.append((n,s,labels))
        print('tree leaves',dict(nleaves),'taxa counts',dict(taxa))
        print('distinct tree IDs',len(set(all_names)),'duplicate IDs',len(all_names)-len(set(all_names)))
        print('examples')
        for x in examples: print(x)
PY"
