/bin/bash -lc "python - <<'PY'
import zipfile, collections
for zpath in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(zpath) as z:
        c=collections.Counter()
        for n in z.namelist():
            if n.endswith('.treefile'): c['treefile']+=1
            if n.endswith('.bionj'): c['bionj']+=1
            if n.endswith('.mldist'): c['mldist']+=1
            if n.endswith('.iqtree'): c['iqtree']+=1
            if n.endswith('.faa.mafft.clipkit'): c['clipkit']+=1
            if n.endswith('.faa.mafft'): c['mafft']+=1
            if n.endswith('.faa'): c['faa']+=1
        print(zpath, len(z.namelist()), c)
PY"
