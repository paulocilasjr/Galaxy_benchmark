import zipfile, collections

def statuses(p):
 with zipfile.ZipFile(p) as z:
  n=[x for x in z.namelist() if x.endswith('/full_table.tsv')][0]
  d={}
  for line in z.read(n).decode().splitlines():
   if line and not line.startswith('#'):
    x=line.split('\\t'); d[x[0]]=x[1]
  return d
for kind,shorts in [('animals',['Cele','Ggal','Mmus','Xlae']),('fungi',['Ncra','Scer','Spom','Umay'])]:
 ds=[statuses(f'inputs/{kind.title() if False else (\"Animals\" if kind==\"animals\" else \"Fungi\")}_{x}.busco.zip') for x in shorts]
 ids=set.intersection(*(set(d) for d in ds))
 print(kind,'common',len(ids),'all complete',sum(all(d[i]=='Complete' for d in ds) for i in ids),'all nonmissing',sum(all(d[i] in ('Complete','Duplicated','Fragmented') for d in ds) for i in ids))
 print('statuses combined',collections.Counter(tuple(d[i] for d in ds) for i in ids).most_common(10))
PY"
