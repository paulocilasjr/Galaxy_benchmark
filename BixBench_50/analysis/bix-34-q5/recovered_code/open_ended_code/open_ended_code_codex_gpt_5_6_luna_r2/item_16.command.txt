/bin/bash -lc "python - <<'PY'
import zipfile
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\nARCHIVE', zname)
    with zipfile.ZipFile(zname) as z:
        names=[n for n in z.namelist() if n.endswith('.treefile')]
        for n in names[:5]:
            print(n, repr(z.read(n).decode('utf-8','replace').strip()))
        print('trees',len(names))
PY"
