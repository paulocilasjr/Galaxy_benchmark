/bin/bash -lc "python - <<'PY'
import zipfile,re,collections
for group in ['animals','fungi']:
 with zipfile.ZipFile(f'inputs/scogs_{group}.zip') as z:
  rows=[]; bad=[]
  for p in z.namelist():
   if not p.endswith('.treefile'):continue
   s=z.read(p).decode('utf-8','replace').strip()
   rows.append((p,s))
   if s.count(';')"'!=1 or not s.endswith('"';') or not s.startswith('('): bad.append((p,repr(s)))
  print(group,'treefiles',len(rows),'malformed',len(bad),'newline variants',sum('\\n' in s for _,s in rows),'lengths',collections.Counter(len(s) for _,s in rows).most_common(5))
  print('all labels expected',all(set(re.findall(r'(?:(?<=\\()|(?<=,))([A-Za-z_]+)(?=:)',s)) <= ({'Animals_Cele','Animals_Ggal','Animals_Mmus','Animals_Xlae'} if group=='animals' else {'Fungi_Ncra','Fungi_Scer','Fungi_Spom','Fungi_Umay'}) for _,s in rows))
PY"
