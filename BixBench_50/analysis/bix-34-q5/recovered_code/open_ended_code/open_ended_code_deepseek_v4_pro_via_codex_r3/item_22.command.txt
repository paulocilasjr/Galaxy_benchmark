/bin/bash -c "find /workspace/tmp/scogs_animals -name '*.treefile' | wc -l; find /workspace/tmp/scogs_fungi -name '*.treefile' | wc -l; echo 'max lines/tree weird'; python3 - <<'PY'
from pathlib import Path
for group in ['scogs_animals','scogs_fungi']:
    files=list(Path('/workspace/tmp',group).glob('*.treefile'))
    maxlen=0;minlen=999999
    weird=[]
    tips=[]
    for f in files:
        s=f.read_text().strip()
        maxlen=max(maxlen,len(s)); minlen=min(minlen,len(s))
        if any(c in s for c in [\"[&\",\"'\",\":\"]): pass
        if '(' not in s or ';' not in s or ':' not in s:
            weird.append(f.name)
        # tip count roughly count : after )
        tips.append((f.name, s.count(':')))
    print(group,len(files),'minmax',minlen,maxlen,'weird',weird[:3], 'tips range', min(t for _,t in tips), max(t for _,t in tips))
    print('sample files', [f.name for f in files[:5]])
PY"
