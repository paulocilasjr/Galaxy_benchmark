/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
for group in ['scogs_animals','scogs_fungi']:
    p=Path('/workspace/tmp')/group
    cnt=Counter()
    files=sorted(p.glob('*.faa'))
    with_ml=[]
    print(group, 'faa files', len(files))
    for f in files:
        n=sum(1 for line in f.open() if line.startswith('>'))
        cnt[n]+=1
        base=f.name
        ml=p/(base+'.mafft.clipkit.mldist')
        if ml.exists(): with_ml.append((f.name,n))
    print('seq counts:', cnt)
    print('with mldist:', len(with_ml), Counter(n for _,n in with_ml))
    print('no mldist sample', [f for f in files if not (p/(f.name+'.mafft.clipkit.mldist')).exists()][:10])
    # check which suffix exist
    bybase={f.name.split('.')[0] for f in files}
    print('unique bases',len(bybase))
PY"
