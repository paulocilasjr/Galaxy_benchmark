/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from statistics import median
import subprocess, re
for group in ['scogs_animals','scogs_fungi']:
    p=Path('/workspace/tmp')/group
    means=[]
    errors=[]
    for f in sorted(p.glob('*.treefile')):
        cp=subprocess.run(['phykit','patristic_distances',str(f)],text=True,capture_output=True)
        if cp.returncode"'!=0:
            errors.append((f.name,cp.stderr[:200])); continue
        m=re.search(r'"'"'^mean:'"\\s*([-+]?[0-9.eE]+)',cp.stdout,re.M)
        if not m:
            errors.append((f.name,cp.stdout[:200])); continue
        means.append(float(m.group(1)))
    print(group,'n',len(means),'median',median(means),'errors',len(errors))
    print('first 5',means[:5])
PY"
