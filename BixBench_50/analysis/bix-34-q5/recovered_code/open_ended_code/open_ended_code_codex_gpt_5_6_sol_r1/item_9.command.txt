/bin/bash -lc "python -m pip install --quiet biopython && python - <<'PY'
from Bio import Phylo
from io import StringIO
from itertools import combinations
from statistics import median
from zipfile import ZipFile

out={}
for group in ('animals','fungi'):
    values=[]
    with ZipFile(f'inputs/scogs_{group}.zip') as z:
        members=sorted(x for x in z.namelist() if x.endswith('.treefile'))
        for member in members:
            tree=Phylo.read(StringIO(z.read(member).decode()), 'newick')
            tips=tree.get_terminals()
            distances=[tree.distance(a,b) for a,b in combinations(tips,2)]
            values.append(sum(distances)/len(distances))
    out[group]=median(values)
    print(group, len(values), repr(out[group]))
print('fungi/animals',repr(out['fungi']/out['animals']))
PY"
