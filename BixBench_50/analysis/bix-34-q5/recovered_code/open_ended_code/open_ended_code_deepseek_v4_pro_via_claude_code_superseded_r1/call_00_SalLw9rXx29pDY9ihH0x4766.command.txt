# Final verification: trace through the computation step by step
import json, statistics

with open('/workspace/codex_output/results.json') as f:
    results = json.load(f)

# For each gene tree, we have per-species mean distances to all other species
# Pool them all by kingdom
animal_vals = []
fungi_vals = []
for r in results:
    for sp, val in r['all_means'].items():
        if sp.startswith('Animals'):
            animal_vals.append(val)
        else:
            fungi_vals.append(val)

animal_vals.sort()
fungi_vals.sort()

print(f"Number of gene trees: {len(results)}")
print(f"Animal observations: {len(animal_vals)}")
print(f"Fungi observations: {len(fungi_vals)}")

# Verify median computation
def my_median(vals):
    n = len(vals)
    if n % 2 == 0:
        return (vals[n//2 - 1] + vals[n//2]) / 2
    else:
        return vals[n//2]

am = my_median(animal_vals)
fm = my_median(fungi_vals)
ratio = fm / am

print(f"\nAnimal median (verified): {am}")
print(f"Fungi median (verified): {fm}")
print(f"Ratio fungi/animal: {ratio}")
print(f"  = {fm} / {am}")

# Check against statistics.median
print(f"\nstatistics.median animal: {statistics.median(animal_vals)}")
print(f"statistics.median fungi: {statistics.median(fungi_vals)}")
print(f"Match: {abs(am - statistics.median(animal_vals)) < 1e-10 and abs(fm - statistics.median(fungi_vals)) < 1e-10}")

# Print distribution summary
print(f"\nAnimal distribution:")
print(f"  Min: {min(animal_vals):.4f}")
print(f"  Q1: {animal_vals[len(animal_vals)//4]:.4f}")
print(f"  Median: {am:.4f}")
print(f"  Q3: {animal_vals[3*len(animal_vals)//4]:.4f}")
print(f"  Max: {max(animal_vals):.4f}")

print(f"\nFungi distribution:")
print(f"  Min: {min(fungi_vals):.4f}")
print(f"  Q1: {fungi_vals[len(fungi_vals)//4]:.4f}")
print(f"  Median: {fm:.4f}")
print(f"  Q3: {fungi_vals[3*len(fungi_vals)//4]:.4f}")
print(f"  Max: {max(fungi_vals):.4f}")
EOF
