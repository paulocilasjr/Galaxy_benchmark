import json, statistics

with open('/workspace/codex_output/results.json') as f:
    results = json.load(f)

# Per-gene within-group mean pairwise distance
# For each gene tree, compute the mean of all pairwise distances within each kingdom
animal_gene_means = []
fungi_gene_means = []

for r in results:
    # Get all animal leaves and fungi leaves
    animal_leaves = [l for l in r['leaves'] if l.startswith('Animals')]
    fungi_leaves = [l for l in r['leaves'] if l.startswith('Fungi')]
    
    # Compute all pairwise within-kingdom distances from the distance matrix
    # We need to reconstruct the distances
    from full_analysis import parse_newick, get_leaves, compute_patristic_distances
    # Actually, we can use the within_means to reconstruct
    
    # within_means gives per-species mean to same-kingdom members
    # Mean within-kingdom pairwise distance for the whole group:
    # For n species, there are n*(n-1)/2 pairs
    # Total distance = sum of (per-species mean * (n-1)) / 2
    # Mean pairwise = total / (n*(n-1)/2) = sum of per-species means / n
    
    animal_within_vals = [v for k, v in r['within_means'].items() if k.startswith('Animals')]
    fungi_within_vals = [v for k, v in r['within_means'].items() if k.startswith('Fungi')]
    
    if animal_within_vals:
        animal_gene_means.append(statistics.mean(animal_within_vals))
    if fungi_within_vals:
        fungi_gene_means.append(statistics.mean(fungi_within_vals))

print("Within-group mean pairwise distances per gene:")
am = statistics.median(animal_gene_means)
fm = statistics.median(fungi_gene_means)
print(f"Animal median: {am:.10f}")
print(f"Fungi median: {fm:.10f}")
print(f"Ratio (fungi/animal): {fm/am:.10f}")
print(f"Ratio (animal/fungi): {am/fm:.10f}")

# Also compute per-gene ratio then median
gene_ratios = []
for i in range(min(len(animal_gene_means), len(fungi_gene_means))):
    if animal_gene_means[i] > 0:
        gene_ratios.append(fungi_gene_means[i] / animal_gene_means[i])
print(f"\nPer-gene ratio (fungi_within / animal_within), median: {statistics.median(gene_ratios):.10f}")
print(f"Per-gene ratio mean: {statistics.mean(gene_ratios):.10f}")
