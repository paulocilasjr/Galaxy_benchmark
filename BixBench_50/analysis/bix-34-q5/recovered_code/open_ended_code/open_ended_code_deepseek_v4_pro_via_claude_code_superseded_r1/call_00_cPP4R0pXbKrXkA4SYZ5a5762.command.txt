cd /tmp && mkdir -p trees_animals trees_fungi

# Extract all shared animal trees
for og in $(cat /tmp/shared_with_trees.txt); do
  unzip -o /workspace/inputs/scogs_animals.zip "${og}.faa.mafft.clipkit.treefile" -d trees_animals/ 2>&1 | grep -v Archive | grep -v inflating | grep -v "caution: filename"
done 2>/dev/null

# Extract all shared fungi trees
for og in $(cat /tmp/shared_with_trees.txt); do
  unzip -o /workspace/inputs/scogs_fungi.zip "${og}.faa.mafft.clipkit.treefile" -d trees_fungi/ 2>&1 | grep -v Archive | grep -v inflating | grep -v "caution: filename"
done 2>/dev/null

echo "Animal trees extracted: $(ls trees_animals/ | wc -l)"
echo "Fungi trees extracted: $(ls trees_fungi/ | wc -l)"
