import os, glob, statistics

def parse_and_mean(filepath):
    with open(filepath) as f:
        content = f.read().strip()
    lines = content.split('\n')
    n = int(lines[0].strip())
    matrix = []
    for line in lines[1:]:
        parts = line.strip().split()
        dists = [float(x) for x in parts[1:]]
        matrix.append(dists)
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append(matrix[i][j])
    return sum(pairs)/len(pairs)

# Animals
animal_means = sorted([parse_and_mean(f) for f in glob.glob('/tmp/work/*.mafft.clipkit.mldist')])
print(f"Animals: {len(animal_means)} groups")
print(f"Sorted first 5: {animal_means[:5]}")
print(f"Sorted last 5: {animal_means[-5:]}")
med_a = statistics.median(animal_means)
print(f"Median: {med_a}")

# Fungi
fungi_means = sorted([parse_and_mean(f) for f in glob.glob('/tmp/work_fungi/*.mafft.clipkit.mldist')])
print(f"\nFungi: {len(fungi_means)} groups")
print(f"Sorted first 5: {fungi_means[:5]}")
print(f"Sorted last 5: {fungi_means[-5:]}")
med_f = statistics.median(fungi_means)
print(f"Median: {med_f}")

ratio = med_f / med_a
print(f"\nRatio: {ratio}")
print(f"Ratio with 15 digits: {ratio:.15f}")
