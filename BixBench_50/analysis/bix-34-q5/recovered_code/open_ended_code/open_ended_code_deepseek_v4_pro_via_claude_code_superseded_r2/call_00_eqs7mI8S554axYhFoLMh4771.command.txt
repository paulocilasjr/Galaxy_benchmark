import sys
sys.path.insert(0, '/workspace/venv/lib/python3.11/site-packages')
import dendropy
import numpy as np

# Verify with manual computation
# Tree: (Animals_Cele:1.5246608168,Animals_Ggal:0.4600442996,Animals_Mmus:0.3539942333);
tree_str = '(Animals_Cele:1.5246608168,Animals_Ggal:0.4600442996,Animals_Mmus:0.3539942333);'

# Dendropy
tree = dendropy.Tree.get(data=tree_str, schema='newick', preserve_underscores=True)
taxa = [leaf.taxon for leaf in tree.leaf_node_iter()]
pdm = tree.phylogenetic_distance_matrix()

print('Dendropy distances:')
for i in range(3):
    for j in range(i+1, 3):
        d = pdm(taxa[i], taxa[j])
        print(f'  {taxa[i].label} - {taxa[j].label}: {d}')

# Manual: for star tree, distance = branch_i + branch_j
# But let's verify by traversing the tree
print()
print('Tree structure:')
for node in tree.preorder_node_iter():
    if node.is_leaf():
        print(f'  Leaf: {node.taxon.label}, edge_length: {node.edge_length}')
    else:
        child_labels = [c.taxon.label if c.is_leaf() else \"internal\" for c in node.child_node_iter()]
        print(f'  Internal node, edge_length: {node.edge_length}, children: {child_labels}')

# Manual computation from Newick
# Star tree: dist(a,b) = edge_a + edge_b
bl = {'Animals_Cele': 1.5246608168, 'Animals_Ggal': 0.4600442996, 'Animals_Mmus': 0.3539942333}
print()
print('Manual distances (star tree):')
print(f'  Cele-Ggal: {bl[\"Animals_Cele\"] + bl[\"Animals_Ggal\"]}')
print(f'  Cele-Mmus: {bl[\"Animals_Cele\"] + bl[\"Animals_Mmus\"]}')
print(f'  Ggal-Mmus: {bl[\"Animals_Ggal\"] + bl[\"Animals_Mmus\"]}')
