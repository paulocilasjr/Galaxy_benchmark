import dendropy

# Test with a resolved tree
# (Animals_Cele:1.0919629990,(Animals_Ggal:0.0207717524,Animals_Mmus:0.0205050160):0.1098961236,Animals_Xlae:0.0000027032);
tree_str = '(Animals_Cele:1.0919629990,(Animals_Ggal:0.0207717524,Animals_Mmus:0.0205050160):0.1098961236,Animals_Xlae:0.0000027032);'

tree = dendropy.Tree.get(data=tree_str, schema='newick', preserve_underscores=True)
taxa = [leaf.taxon for leaf in tree.leaf_node_iter()]
labels = [t.label for t in taxa]
pdm = tree.phylogenetic_distance_matrix()

print('Dendropy distances:')
for i in range(len(taxa)):
    for j in range(i+1, len(taxa)):
        d = pdm(taxa[i], taxa[j])
        print(f'  {labels[i]} - {labels[j]}: {d}')
