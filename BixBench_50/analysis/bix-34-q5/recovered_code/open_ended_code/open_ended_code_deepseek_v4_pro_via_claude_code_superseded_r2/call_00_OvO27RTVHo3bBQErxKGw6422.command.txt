import sys
sys.path.insert(0, '/workspace/venv/lib/python3.11/site-packages')
import os
import numpy as np

# ML distances as sanity check
animal_dir = '/workspace/extracted'
fungi_dir = '/workspace/extracted_fungi'

def compute_mean_mldist(mldist_file):
    with open(mldist_file) as fh:
        lines = fh.read().strip().split('\n')
    n = int(lines[0])
    taxa = []
    matrix = {}
    for line in lines[1:]:
        parts = line.strip().split()
        name = parts[0]
        taxa.append(name)
        matrix[name] = [float(x) for x in parts[1:]]
    
    distances = []
    for i in range(n):
        for j in range(i+1, n):
            distances.append(matrix[taxa[i]][j])
    
    return np.mean(distances)

# Get shared OGs
animals_s = set()
for f in os.listdir(animal_dir):
    if f.endswith('.treefile'):
        animals_s.add(f.replace('.mafft.clipkit.treefile', ''))

fungi_s = set()
for f in os.listdir(fungi_dir):
    if f.endswith('.treefile'):
        fungi_s.add(f.replace('.mafft.clipkit.treefile', ''))

shared = sorted(animals_s & fungi_s)

animal_means = []
fungi_means = []

for og in shared:
    am = compute_mean_mldist(os.path.join(animal_dir, og + '.mafft.clipkit.mldist'))
    fm = compute_mean_mldist(os.path.join(fungi_dir, og + '.mafft.clipkit.mldist'))
    animal_means.append(am)
    fungi_means.append(fm)

animal_median = np.median(animal_means)
fungi_median = np.median(fungi_means)
ratio = fungi_median / animal_median

print(f'ML distances (sanity check):')
print(f'Animal median: {animal_median:.6f}')
print(f'Fungi median: {fungi_median:.6f}')
print(f'Ratio (fungi/animal): {ratio:.6f}')
print(f'Per-gene median ratio: {np.median([f/a for f,a in zip(fungi_means, animal_means)]):.6f}')
