/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  x <- readRDS(f)
  cat('\\nFILE', f, '\\n')
  cat('class:', paste(class(x), collapse=','), '\\n')
  cat('dim:', paste(dim(x), collapse='x'), '\\n')
  cat('colnames:', paste(colnames(x), collapse=' | '), '\\n')
  cat('metadata:\\n')
  print(metadata(x))
  cat('mcols:\\n')
  print(mcols(x))
  cat('attrs:', paste(names(attributes(x)), collapse=' | '), '\\n')
  ii <- grep('35160|PGD1656520|pumA', rownames(x), ignore.case=TRUE)
  cat('matches:', paste(ii, collapse=','), '\\n')
  if (length(ii)) print(as.data.frame(x[ii,,drop=FALSE]))
  cat('rownames first:', paste(head(rownames(x)), collapse=' | '), '\\n')
}
RS"
