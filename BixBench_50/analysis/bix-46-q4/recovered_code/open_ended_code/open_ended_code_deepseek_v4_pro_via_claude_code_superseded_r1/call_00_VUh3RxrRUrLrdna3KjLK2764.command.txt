Rscript -e '
res97 <- readRDS("inputs/res_1vs97.rds", refhook = NULL)
res98 <- readRDS("inputs/res_1vs98.rds", refhook = NULL)

ld97 <- attr(res97, "listData")
ld98 <- attr(res98, "listData")
rn97 <- attr(res97, "rownames")
rn98 <- attr(res98, "rownames")

# Check rhl genes
cat("=== Quorum sensing genes ===\n")
qs_genes <- c("PA14_19100", "PA14_19110", "PA14_19120", "PA14_19130",  # rhlA, rhlB, rhlR, rhlI
              "PA14_16250", "PA14_16260")  # lasI, lasR
for(g in qs_genes) {
  i97 <- which(rn97 == g)
  i98 <- which(rn98 == g)
  cat("\n", g, ":\n", sep="")
  if(length(i97) > 0) cat("  res97 log2FC:", round(ld97$log2FoldChange[i97], 4), "padj:", format(ld97$padj[i97], digits=3), "\n")
  else cat("  res97: NOT FOUND\n")
  if(length(i98) > 0) cat("  res98 log2FC:", round(ld98$log2FoldChange[i98], 4), "padj:", format(ld98$padj[i98], digits=3), "\n")
  else cat("  res98: NOT FOUND\n")
}

# Confirm PA14_35160 value precisely
idx97 <- which(rn97 == "PA14_35160")
cat("\n=== PA14_35160 precise value ===\n")
cat("log2FoldChange:", ld97$log2FoldChange[idx97], "\n")
cat("Rounded to 2 dp:", round(ld97$log2FoldChange[idx97], 2), "\n")
' 2>&1
