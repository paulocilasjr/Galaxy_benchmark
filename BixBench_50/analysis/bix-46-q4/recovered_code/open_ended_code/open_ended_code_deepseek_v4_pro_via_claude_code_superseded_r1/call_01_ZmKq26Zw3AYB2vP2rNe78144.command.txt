Rscript -e '
res97 <- readRDS("inputs/res_1vs97.rds", refhook = NULL)
res98 <- readRDS("inputs/res_1vs98.rds", refhook = NULL)

ld97 <- attr(res97, "listData")
ld98 <- attr(res98, "listData")
rn97 <- attr(res97, "rownames")
rn98 <- attr(res98, "rownames")

# Get PA14_35160 from both
idx97 <- which(rn97 == "PA14_35160")
idx98 <- which(rn98 == "PA14_35160")

cat("=== PA14_35160 ===\n")
cat("res_1vs97 (Strain 97 vs 1):\n")
cat("  log2FoldChange:", ld97$log2FoldChange[idx97], "\n")
cat("  padj:", ld97$padj[idx97], "\n")
cat("res_1vs98 (Strain 98 vs 1):\n")
if(length(idx98) > 0) {
  cat("  log2FoldChange:", ld98$log2FoldChange[idx98], "\n")
  cat("  padj:", ld98$padj[idx98], "\n")
} else {
  cat("  NOT FOUND\n")
}

# Also check known phenazine genes (phzA1, phzB1, etc.)
cat("\n=== Phenazine biosynthesis genes ===\n")
phz_genes <- c("PA14_09420", "PA14_09440", "PA14_09450", "PA14_09460", 
               "PA14_09470", "PA14_09480", "PA14_09490",
               "PA14_39890", "PA14_39910", "PA14_39925", "PA14_39945",
               "PA14_39960", "PA14_39970", "PA14_00640")
for(g in phz_genes) {
  i97 <- which(rn97 == g)
  i98 <- which(rn98 == g)
  cat("\n", g, ":\n", sep="")
  if(length(i97) > 0) cat("  res97 log2FC:", round(ld97$log2FoldChange[i97], 4), "padj:", format(ld97$padj[i97], digits=3), "\n")
  if(length(i98) > 0) cat("  res98 log2FC:", round(ld98$log2FoldChange[i98], 4), "padj:", format(ld98$padj[i98], digits=3), "\n")
}
' 2>&1
