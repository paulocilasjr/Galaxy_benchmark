Rscript -e '
x97 <- readRDS("/workspace/inputs/res_1vs97.rds")
x98 <- readRDS("/workspace/inputs/res_1vs98.rds")

# Check other phenazine genes from the CSV
# First, find all phenazine-related genes
cat("Looking for phenazine genes in both comparisons...\n")

# Known phenazine genes in PA14
phz_genes <- c("PA14_35160", "PA14_09400", "PA14_09410", "PA14_09420", 
               "PA14_09430", "PA14_09440", "PA14_09450", "PA14_09460",
               "PA14_09470", "PA14_09480", "PA14_09490", "PA14_09500",
               "PA14_09510", "PA14_09900", "PA14_39940", "PA14_39950")

# Get phenazine related genes from CSV
con <- file("/workspace/inputs/Pseudomonas_aeruginosa_UCBPP-PA14.csv", "r")
lines <- readLines(con)
close(con)
phz_lines <- grep("phenazine", lines, ignore.case=TRUE, value=TRUE)
cat("Phenazine-related genes in CSV:\n")
for(l in phz_lines) {
  parts <- strsplit(l, ",")[[1]]
  cat("  ", parts[6], ": ", parts[11], "\n", sep="")
}

# Now check what happens to these genes in both comparisons
cat("\n=== Phenazine gene log2FC ===\n")
cat(sprintf("%-15s %12s %12s\n", "Gene", "res_1vs97", "res_1vs98"))
cat(strrep("-", 42), "\n")
for(name in c("res_1vs97", "res_1vs98")) {
  # NA
}

a97 <- attributes(x97)
a98 <- attributes(x98)
rn97 <- a97$rownames
rn98 <- a98$rownames
ld97 <- a97$listData
ld98 <- a98$listData

# Get all phenazine locus tags
phz_tags <- character(0)
for(l in phz_lines) {
  parts <- strsplit(l, ",")[[1]]
  phz_tags <- c(phz_tags, parts[6])
}
phz_tags <- unique(phz_tags)

for(tag in phz_tags) {
  idx97 <- which(rn97 == tag)
  idx98 <- which(rn98 == tag)
  fc97 <- if(length(idx97) > 0) round(ld97$log2FoldChange[idx97], 3) else NA
  fc98 <- if(length(idx98) > 0) round(ld98$log2FoldChange[idx98], 3) else NA
  cat(sprintf("%-15s %12s %12s\n", tag, fc97, fc98))
}
' 2>&1
