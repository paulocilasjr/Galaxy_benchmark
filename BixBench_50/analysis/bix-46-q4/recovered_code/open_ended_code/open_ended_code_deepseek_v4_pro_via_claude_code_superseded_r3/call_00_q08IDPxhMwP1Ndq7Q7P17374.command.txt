Rscript -e '
x97 <- readRDS("/workspace/inputs/res_1vs97.rds")
x98 <- readRDS("/workspace/inputs/res_1vs98.rds")

# Get elementMetadata description for both
for(name in c("res_1vs97", "res_1vs98")) {
  x <- if(name == "res_1vs97") x97 else x98
  cat("\n=== ", name, " ===\n")
  em <- attributes(x)$elementMetadata
  ld <- attributes(em)$listData
  cat("descriptions:\n")
  for(i in seq_along(ld$description)) {
    cat("  [", i, "] ", ld$description[i], "\n", sep="")
  }
}

# Get PA14_35160 values from both
cat("\n=== PA14_35160 values ===\n")
for(name in c("res_1vs97", "res_1vs98")) {
  x <- if(name == "res_1vs97") x97 else x98
  a <- attributes(x)
  rn <- a$rownames
  ld <- a$listData
  idx <- which(rn == "PA14_35160")
  cat("\n", name, ":\n", sep="")
  cat("  baseMean:", ld$baseMean[idx], "\n")
  cat("  log2FoldChange:", ld$log2FoldChange[idx], "\n")
  cat("  lfcSE:", ld$lfcSE[idx], "\n")
  cat("  stat:", ld$stat[idx], "\n")
  cat("  pvalue:", ld$pvalue[idx], "\n")
  cat("  padj:", ld$padj[idx], "\n")
}
' 2>&1
