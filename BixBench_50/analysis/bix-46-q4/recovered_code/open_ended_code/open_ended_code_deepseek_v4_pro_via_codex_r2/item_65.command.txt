/bin/bash -c "python3 - <<'PY'
import requests
for q in ['Pseudomonas aeruginosa rhlI','PA14 rhlI rhlR','Pseudomonas aeruginosa UCBPP-PA14 RNA-seq']:
 r=requests.get('https://zenodo.org/api/records',params={'q':q,'size':10},timeout=20)
 print('\\nQUERY',q,r.status_code)
 try:
  data=r.json(); print('hits',data.get('hits',{}).get('total'))
  for h in data.get('hits',{}).get('hits',[])[:5]:
   print(h.get('id'), h.get('metadata',{}).get('title'), h.get('links',{}).get('self_html'))
 except Exception as e: print(r.text[:500])
PY"
