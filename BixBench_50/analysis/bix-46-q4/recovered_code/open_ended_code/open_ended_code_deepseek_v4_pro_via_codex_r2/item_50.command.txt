/bin/bash -c "Rscript - <<'EOF'
for (f in c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')) {
 cat('\\n====',f,'\\n')
 con<-gzfile(f,'rb'); x<-unserialize(con); close(con); y<-unclass(x)
 em<-unclass(attr(y,'elementMetadata'))
 cat('em attrs:', paste(names(attributes(em)), collapse=', '), '\\n')
 for(nm in names(attributes(em))) {
   a<-attr(em,nm)
   cat('  ',nm,'typeof',typeof(a),'length',length(a),'\\n')
   if(nm=='listData') { cat('    list names:', paste(names(a),collapse=', '),'\\n'); for(cn in names(a)) print(a[[cn]]) }
   else if(nm=='metadata') {
      if(typeof(a)=='list') { cat('    metadata names:',paste(names(a),collapse=', '),'\\n'); for(mn in names(a)) {cat('      ',mn,'typeof',typeof(a[[mn]]),'len',length(a[[mn]]),'\\n'); if(typeof(a[[mn]])%in%c('character','integer','double','logical')) print(a[[mn]])}}
   } else if(typeof(a)%in%c('character','integer','double','logical')) { print(head(a,20)) }
 }
}
EOF"
