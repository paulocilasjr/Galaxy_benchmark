/bin/bash -c "python3 - <<'PY'
import requests,json
for uid in ['1144160','1086972','816529','816523','816522','816521','815834']:
 # summary
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'bioproject','id':uid,'retmode':'json'},timeout=20)
 d=r.json()['result'][uid]
 print('\\n',uid,d.get('project_title'))
 for k in ['project_description','description','taxname','relevance']:
  if d.get(k): print(k, d[k][:1000])
 # sra links
 r2=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/elink.fcgi',params={'dbfrom':'bioproject','db':'sra','id':uid,'retmode':'json'},timeout=20)
 print('sra link',r2.json().get('linksets',[{}])[0].get('linksetdbs',[]))
PY"
