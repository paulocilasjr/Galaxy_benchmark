/bin/bash -c "Rscript - <<'EOF'
for (f in c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')) {
 cat('\\n===== FILE', f, '=====\\n')
 con<-gzfile(f,'rb'); x<-unserialize(con); close(con); y<-unclass(x)
 atts <- attributes(y)
 # metadata attr
 a <- atts[['metadata']]
 cat('metadata typeof', typeof(a), 'length', length(a), '\\n')
 if (typeof(a)=='list') {
   cat('metadata names:', paste(names(a), collapse=', '), '\\n')
   for (nm in names(a)) {
      cat('META', nm, 'typeof', typeof(a[[nm]]), 'length', length(a[[nm]]), '\\n')
      if (typeof(a[[nm]]) %in% c('character','integer','double','logical')) print(a[[nm]])
      else if (is.list(a[[nm]]) || typeof(a[[nm]])=='list') { cat('names:', paste(names(a[[nm]]), collapse=', '), '\\n') }
   }
 }
 # element metadata descriptions
 em <- unclass(atts[['elementMetadata']])
 eld <- attr(em,'listData')
 for (cn in names(eld)) { cat('EM', cn, ': '); print(eld[[cn]]) }
}
EOF"
