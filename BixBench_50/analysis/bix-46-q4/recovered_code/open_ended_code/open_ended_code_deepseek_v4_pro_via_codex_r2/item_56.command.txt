/bin/bash -c "python3 - <<'PY'
import requests,re
url='https://journals.plos.org/plospathogens/article?id=10.1371/journal.ppat.1006504'
r=requests.get(url,headers={'User-Agent':'Mozilla/5.0'},timeout=30)
text=r.text
for pat in ['s01','xlsx','csv','count','Count','RNA-seq','read']:
 print('\\nPAT',pat)
 for m in re.finditer(pat,text,re.I):
  s=max(0,m.start()-200); print(text[s:m.start()+300].replace('\\n',' ')[:500]); print('---')
  break
# list hrefs with supplementary doi
hrefs=re.findall(r'href=\"(["'^"]+)"'"',text)
for h in hrefs:
 if 'ppat.1006504.s0' in h or 'supplementary' in h.lower():
  print(h)
PY"
