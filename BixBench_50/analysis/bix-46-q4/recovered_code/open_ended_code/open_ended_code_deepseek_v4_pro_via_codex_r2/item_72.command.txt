/bin/bash -c "python3 - <<'PY'
import subprocess, zipfile, xml.etree.ElementTree as ET, math
# load res lfc and se
rscript=r'''
for (f in c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')) { con<-gzfile(f,'rb'); x<-unserialize(con); close(con); y<-unclass(x); ld<-attr(y,'listData'); rn<-attr(y,'rownames'); cat('###FILE',basename(f),'\\n'); for(i in seq_along(rn)) cat(rn[i],ld"'$log2FoldChange[i],ld$lfcSE[i],'"'\\n',sep='\\t') }
'''
out=subprocess.run(['Rscript','-e',rscript],capture_output=True,text=True,check=True).stdout
res={};cur=None
for line in out.splitlines():
 if line.startswith('###FILE'):cur=line.split()[1];res[cur]={};continue
 p=line.split('\\t');
 if len(p)>=3:res[cur][p[0]]=(float(p[1]),float(p[2]))
ns={'w':'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
def parse(fn):
 with zipfile.ZipFile(fn) as z:root=ET.fromstring(z.read('word/document.xml'))
 t=root.find('.//w:tbl',ns);d={}
 for tr in t.findall('w:tr',ns)[2:]:
  cells=[''.join(x.text or '' for x in c.findall('.//w:t',ns)) for c in tr.findall('w:tc',ns)]
  if len(cells)>=4 and cells[0]:
   def f(s):
    s=s.strip();return None if s in ('','-') else float(s)
   d[cells[0]]={'rhlR':f(cells[1]),'rhlI':f(cells[2])}
 return d
tabs={'HCD':parse('/workspace/ppat.1006504.s012.docx'),'biofilm':parse('/workspace/ppat.1006504.s013.docx')}

def fit(fname,col,cond):
 rfcs=res[fname]; tab=tabs[cond]
 xs=[]; ys=[]; ses=[]
 for g,d in tab.items():
  t=d[col]
  if t is None or g not in rfcs:continue
  x=-t; y=rfcs[g][0]; se=rfcs[g][1]
  xs.append(x); ys.append(y); ses.append(se)
 n=len(xs)
 if n==0:return None
 # slope through origin weighted by 1/se"'^2
 w=[1/(s*s) for s in ses]
 b=sum(wi*x*y for wi,x,y in zip(w,xs,ys))/sum(wi*x*x for wi,x in zip(w,xs))
 rss=sum(wi*(y-b*x)**2 for wi,x,y in zip(w,xs,ys))
 tss=sum(wi*y*y for wi,y in zip(w,ys))
 return {'"'n':n,'b':b,'rss':rss,'tss':tss,'r2':1-rss/tss if tss else 0,'rmse':math.sqrt(sum((y-b*x)**2 for x,y in zip(xs,ys))/n)}
for case,cond,c97,c98 in [
 ('HCD A 97=rhlR 98=rhlI','HCD','rhlR','rhlI'),
 ('HCD B 97=rhlI 98=rhlR','HCD','rhlI','rhlR'),
 ('bio C 97=rhlR 98=rhlI','biofilm','rhlR','rhlI'),
 ('bio D 97=rhlI 98=rhlR','biofilm','rhlI','rhlR')]:
 m1=fit('res_1vs97.rds',c97,cond);m2=fit('res_1vs98.rds',c98,cond)
 print('\\n',case)
 for nm,m in [('res97',m1),('res98',m2)]:
  print(nm,m)
PY"
