/bin/bash -c "Rscript - <<'EOF'
for (f in c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')) {
 cat('\\n===== FILE', f, '=====\\n')
 con<-gzfile(f,'rb'); x<-unserialize(con); close(con); y<-unclass(x)
 atts <- attributes(y)
 for (nm in names(atts)) {
  a <- atts[[nm]]
  cat('\\nATTR', nm, 'typeof', typeof(a), '\\n')
  if (nm == 'listData') {
    cat('listData names:', paste(names(a),collapse=', '), '\\n')
    next
  }
  if (nm == 'rownames') {
    cat('first:', paste(head(a,10),collapse=', '), '\\n')
    next
  }
  if (nm == 'elementMetadata') {
    em <- unclass(a)
    cat('elementMetadata after unclass attrs:', paste(names(attributes(em)), collapse=', '), '\\n')
    eld <- attr(em,'listData')
    cat('elementMetadata listData names:', paste(names(eld), collapse=', '), '\\n')
    for (cn in names(eld)) {
      cat('  emcol', cn, 'typeof', typeof(eld[[cn]]), 'length', length(eld[[cn]]), '\\n')
      print(utils::head(eld[[cn]], 20))
    }
    cat('elementMetadata rownames:', paste(utils::head(attr(em,'rownames'),20), collapse=', '), '\\n')
    next
  }
  if (typeof(a) %in% c('list','character','integer','double','logical')) {
    str(a, max.level=2)
  } else {
    cat('skip print for type', typeof(a), '\\n')
  }
 }
}
EOF"
