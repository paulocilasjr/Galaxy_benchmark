/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  cat('\\nFILE', f, '\\n')
  x <- readRDS(f)
  cat('typeof', typeof(x), 'isS4', isS4(x), '\\n')
  cat('attributes names:', paste(names(attributes(x)), collapse=', '), '\\n')
  print(attributes(x))
  cat('slotNames attempt:\\n')
  print(try(slotNames(x), silent=TRUE))
  cat('str attempt:\\n')
  print(try(str(x, max.level=2), silent=TRUE))
}
RS"
