/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  x <- readRDS(f)
  rn <- attr(x, 'rownames')
  ld <- attr(x, 'listData')
  md <- attr(x, 'metadata')
  em <- attr(x, 'elementMetadata')
  cat('\\nFILE', f, '\\n')
  cat('nrows_attr=', attr(x,'nrows'), ' rownames_len=', length(rn), ' listData_len=', length(ld), '\\n', sep='')
  cat('listData names:', paste(names(ld), collapse=', '), '\\n')
  idx <- match('PA14_35160', rn)
  cat('target index:', idx, '\\n')
  for (nm in names(ld)) {
    v <- ld[[nm]]
    cat(nm, ': type=', typeof(v), ' class=', paste(class(v), collapse='|'), ' length=', length(v), ' target=', if ("'!is.na(idx)) as.character(v[[idx]]) else '"'NA', '\\n', sep='')
  }
  cat('metadata names:', paste(names(md), collapse=', '), '\\n')
  cat('metadata compact:\\n')
  print(utils::str(md, max.level=3))
  cat('elementMetadata attrs/names:\\n')
  cat('attrs:', paste(names(attributes(em)), collapse=', '), '\\n')
  cat('listData names in elementMetadata:', paste(names(attr(em, 'listData')), collapse=', '), '\\n')
}
RS"
