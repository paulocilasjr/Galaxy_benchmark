/bin/bash -lc "Rscript - <<'RS'
files <- c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')
for (f in files) {
  cat('\\nFILE', f, '\\n')
  x <- readRDS(f)
  cat('class:', paste(class(x), collapse=', '), '\\n')
  cat('dim:', paste(dim(x), collapse=' x '), '\\n')
  cat('colnames:', paste(colnames(x), collapse=', '), '\\n')
  cat('names:', paste(names(x), collapse=', '), '\\n')
  cat('metadata names:', paste(names(metadata(x)), collapse=', '), '\\n')
  if (length(metadata(x))) print(metadata(x))
  cat('mcols names:', paste(names(mcols(x)), collapse=', '), '\\n')
  if ('PA14_35160' %in% rownames(x)) {
    print(as.data.frame(x['PA14_35160',,drop=FALSE]))
  } else {
    cat('target not in rownames. sample rownames:', paste(head(rownames(x)), collapse=', '), '\\n')
  }
}
RS"
