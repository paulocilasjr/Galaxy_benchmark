/bin/bash -lc "Rscript - <<'RS'
ann <- read.csv('inputs/Pseudomonas_aeruginosa_UCBPP-PA14.csv', check.names=FALSE)
qs <- ann[grepl('rhlI|rhlR|lasI|lasR|pqs|mvfR|pqsR|phz', paste(ann[['Gene Name']], ann[['Product Name']], ann[['Product Synonyms']], sep=' '), ignore.case=TRUE), c('Locus Tag','Gene Name','Product Name')]
print(qs, row.names=FALSE)
# Extract DE rows for rhl/las/pqs/phz genes from each result
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  x <- readRDS(f); rn <- attr(x,'rownames'); ld <- attr(x,'listData')
  df <- data.frame(Locus.Tag=rn, log2FoldChange=ld"'$log2FoldChange, padj=ld$padj, stringsAsFactors=FALSE)
  m <- merge(qs, df, by='"'Locus.Tag', all.x=TRUE, sort=FALSE)
  cat('\\n', f, '\\n')
  print(m[order(m"'$Locus.Tag),], row.names=FALSE)
}
RS'
