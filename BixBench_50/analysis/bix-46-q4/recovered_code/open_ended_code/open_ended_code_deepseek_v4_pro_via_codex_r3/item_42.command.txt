/bin/bash -c "python3 - <<'PY'
import urllib.request
urls=[
'https://www.ebi.ac.uk/europepmc/webservices/rest/PMC5531660/supplementary/ppat.1006504.s012.docx',
'https://www.ebi.ac.uk/europepmc/webservices/rest/PMC5531660/supplementaryFiles/ppat.1006504.s012.docx',
'https://europepmc.org/articles/PMC5531660/bin/ppat.1006504.s012.docx',
'https://europepmc.org/api/rest/PMC5531660/supplementary/ppat.1006504.s012.docx'
]
for url in urls:
 try:
  req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
  r=urllib.request.urlopen(req,timeout=20)
  data=r.read(); print(url, r.status, len(data), data[:8], r.headers.get('content-type'))
 except Exception as e: print(url, type(e).__name__, e)
PY"
