/bin/bash -c "python3 - <<'PY'
import urllib.request,re
url='https://www.ebi.ac.uk/europepmc/webservices/rest/PMC5531660/fullTextXML'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read().decode()
for term in ['Supporting','Data Availability','Repository','publicly','Sequence Read Archive','NCBI','RNA-seq', 'Accession']:
    positions=[m.start() for m in re.finditer(term,data,re.I)]
    print(term, positions[:20])
    if positions:
        idx=positions[0]
        print(data[max(0,idx-300):idx+1000].replace('\\n',' ')[:1300])
PY"
