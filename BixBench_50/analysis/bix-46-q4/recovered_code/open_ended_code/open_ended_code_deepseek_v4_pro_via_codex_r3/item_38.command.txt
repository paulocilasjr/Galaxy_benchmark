/bin/bash -c "python3 - <<'PY'
import urllib.request,re
url='https://www.ebi.ac.uk/europepmc/webservices/rest/PMC5531660/fullTextXML'

for m in re.finditer(r'<supplementary-material id=\"(ppat\\.1006504\\.s0[0-9]+)\"',data):
    id=m.group(1)
    start=m.start(); end=data.find('</supplementary-material>',start)
    print('\\nID',id, data[start:end+25][:1500])
PY"
