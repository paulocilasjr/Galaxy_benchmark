/bin/bash -c "python3 - <<'PY'
import urllib.request, zipfile, io, re
base='https://www.ebi.ac.uk/europepmc/webservices/rest/PMC5531660/supplementaryFiles/'
for fn in ['ppat.1006504.s012.docx','ppat.1006504.s013.docx']:
    url=base+fn
    print('\\nFETCH',fn)
    data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read()
    print('size',len(data),'sig',data[:4])
    try:
        z=zipfile.ZipFile(io.BytesIO(data))
        print('names', z.namelist()[:20])
        xml=z.read('word/document.xml').decode('utf-8','ignore')
        # Extract all w:t text
        text=''.join(re.findall(r'<w:t["'^>]*>([''^<]*)</w:t>'"',xml))
        print('text len',len(text))
        idx=text.find('PA14_35160')
        print('idx',idx)
        if idx>=0:
            print(text[max(0,idx-500):idx+1000])
        else:
            print('not found')
            print(text[:2000])
    except Exception as e:
        print('ERR',type(e).__name__,e)
PY"
