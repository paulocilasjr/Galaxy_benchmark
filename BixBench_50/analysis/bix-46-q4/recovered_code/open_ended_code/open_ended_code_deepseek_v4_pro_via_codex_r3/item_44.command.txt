/bin/bash -c "python3 - <<'PY'
import urllib.request, zipfile, io, re
url='https://www.ebi.ac.uk/europepmc/webservices/rest/PMC5531660/supplementaryFiles/ppat.1006504.s012.docx'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read()
z=zipfile.ZipFile(io.BytesIO(data))
for fn in ['ppat.1006504.s012.docx','ppat.1006504.s013.docx']:
    print('\\nFILE in archive',fn)
    sub=z.read(fn)
    subz=zipfile.ZipFile(io.BytesIO(sub))
    print('sub names',subz.namelist())
    xml=subz.read('word/document.xml').decode('utf-8','ignore')
    # extract paragraphs/cells more robustly, preserving breaks
    # simple all text
    text=''.join(re.findall(r'<w:t["'^>]*>([''^<]*)</w:t>'"',xml))
    print('text len',len(text))
    idx=text.find('PA14_35160')
    print('idx',idx)
    if idx>=0: print(text[max(0,idx-500):idx+1500])
    else: print(text[:3000])
PY"
