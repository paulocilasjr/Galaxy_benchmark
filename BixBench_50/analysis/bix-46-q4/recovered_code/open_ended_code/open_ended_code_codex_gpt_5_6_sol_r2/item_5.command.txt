for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  cat('FILE', f, '\\n')
  x <- readRDS(f)
  cat('typeof', typeof(x), 'isS4', isS4(x), '\\n')
  print(attributes(x))
  if(isS4(x)) {
    print(try(slotNames(x), silent=TRUE))
    for(n in c('listData','elementMetadata','metadata','partitioning','rownames','NAMES')) {
      y <- try(slot(x,n), silent=TRUE)
      cat('slot',n,'class',paste(class(y),collapse=','),'typeof',typeof(y),'length',length(y),'\\n')
      print(try(str(y, max.level=2),silent=TRUE))
    }
  }
}
RS
