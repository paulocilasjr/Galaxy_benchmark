/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_1vs97.rds','inputs/res_1vs98.rds')) {
  cat('FILE',f,'\\n')
  x <- readRDS(f)
  rn <- attr(x,'rownames',exact=TRUE)
  i <- match('PA14_35160',rn)
  ld <- attr(x,'listData',exact=TRUE)
  cat('target index',i,'\\n')
  cat('listData names:',paste(names(ld),collapse=', '),'\\n')
  for (j in seq_along(ld)) cat(names(ld)[j], format(ld[[j]][i],digits=17), '\\n')
  md <- attr(x,'metadata',exact=TRUE)
  cat('metadata names',paste(names(md),collapse=', '),'\\n')
  if(is.list(md)) dput(md)
  em <- attr(x,'elementMetadata',exact=TRUE)
  eld <- attr(em,'listData',exact=TRUE)
  cat('element metadata listData names:',paste(names(eld),collapse=', '),'\\n')
  for(j in seq_along(eld)) { cat('emd',names(eld)[j],':'); dput(eld[[j]]) }
}
RS"
