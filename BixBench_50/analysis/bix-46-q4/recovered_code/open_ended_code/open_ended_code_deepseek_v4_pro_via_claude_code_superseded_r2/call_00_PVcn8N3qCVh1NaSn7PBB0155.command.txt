Rscript -e '
r1 <- readRDS("/workspace/inputs/res_1vs97.rds")
r2 <- readRDS("/workspace/inputs/res_1vs98.rds")

ld1 <- attr(r1, "listData")
ld2 <- attr(r2, "listData")
rn1 <- attr(r1, "rownames")
rn2 <- attr(r2, "rownames")

# Phenazine genes from the CSV
phenazine_genes <- c(
  "PA14_09420", "PA14_09440", "PA14_09450", "PA14_09460", "PA14_09470", "PA14_09480",
  "PA14_09490", "PA14_00640", 
  "PA14_39890", "PA14_39910", "PA14_39925", "PA14_39945", "PA14_39960", "PA14_39970",
  "PA14_35160", "PA14_09520", "PA14_18020", "PA14_28280"
)

cat(sprintf("%-15s %12s %12s %12s %12s\n", "Gene", "97_lfc", "97_padj", "98_lfc", "98_padj"))
cat(strrep("-", 65), "\n")
for (gene in phenazine_genes) {
  idx1 <- which(rn1 == gene)
  idx2 <- which(rn2 == gene)
  lfc1 <- if(length(idx1)) sprintf("%.4f", ld1$log2FoldChange[idx1]) else "NA"
  lfc2 <- if(length(idx2)) sprintf("%.4f", ld2$log2FoldChange[idx2]) else "NA"
  padj1 <- if(length(idx1)) sprintf("%.2e", ld1$padj[idx1]) else "NA"
  padj2 <- if(length(idx2)) sprintf("%.2e", ld2$padj[idx2]) else "NA"
  cat(sprintf("%-15s %12s %12s %12s %12s\n", gene, lfc1, padj1, lfc2, padj2))
}
' 2>&1
