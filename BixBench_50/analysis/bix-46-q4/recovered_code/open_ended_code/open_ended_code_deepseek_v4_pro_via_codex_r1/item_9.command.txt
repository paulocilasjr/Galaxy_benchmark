/bin/bash -c "Rscript - <<'RS'
x <- readRDS('/workspace/inputs/res_1vs97.rds')
ld0 <- attr(x, 'listData', exact=TRUE)
cat('ld0 typeof', typeof(ld0), '\\n')
# unclass to plain list of vectors
ld <- unclass(ld0)
cat('ld typeof after unclass', typeof(ld), 'len', length(ld), '\\n')
cat('ld names:', paste(names(ld), collapse=', '), '\\n')
for(nm in names(ld)) {
  v <- ld[[nm]]
  cat('\\ncol', nm, 'typeof', typeof(v), 'len', length(v), '\\n')
  print(utils::head(v))
}
rn <- attr(x, 'rownames', exact=TRUE)
cat('\\nrn len', length(rn), 'head', paste(utils::head(rn), collapse=', '), '\\n')
# print specific gene row
idx <- which(rn == 'PA14_35160')
cat('\\nidx for PA14_35160:', idx, '\\n')
if(length(idx)) {
 for(nm in names(ld)) {cat(nm, '=', ld[[nm]][idx], '\\n')}
}
RS"
