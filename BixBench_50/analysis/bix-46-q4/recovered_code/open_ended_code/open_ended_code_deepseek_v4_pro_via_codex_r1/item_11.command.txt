/bin/bash -c "Rscript - <<'RS'
for(f in c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')) {
 cat('\\n=====', f, '=====\\n')
 x <- readRDS(f)
 attrs <- attributes(x)
 cat('all attrs names:', paste(names(attrs), collapse=', '), '\\n')
 m <- attr(x,'metadata', exact=TRUE)
 cat('metadata typeof', typeof(m), 'len', length(m), '\\n')
 if(typeof(m)=='list') {
  cat('metadata names:', paste(names(m), collapse=', '), '\\n')
  for(nm in names(m)) {
   cat('metadata item', nm, 'typeof', typeof(m[[nm]]), 'len', length(m[[nm]]), '\\n')
   y <- m[[nm]]
   if(length(y) <= 100) { print(unclass(y)) } else { str(unclass(y), max.level=1) }
  }
 }
 em <- attr(x,'elementMetadata', exact=TRUE)
 cat('elementMetadata typeof', typeof(em), 'len', length(em), '\\n')
 emu <- unclass(em)
 cat('emu names:', paste(names(emu), collapse=', '), '\\n')
 for(nm in names(emu)) {
  cat('elem col', nm, 'typeof', typeof(emu[[nm]]), 'len', length(emu[[nm]]), '\\n')
  if(length(emu[[nm]]) <= 30) print(emu[[nm]])
 }
 pr <- attr(x,'priorInfo', exact=TRUE)
 cat('priorInfo:\\n'); print(unclass(pr))
}
RS"
