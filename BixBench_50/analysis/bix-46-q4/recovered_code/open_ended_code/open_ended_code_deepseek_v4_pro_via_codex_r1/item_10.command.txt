/bin/bash -c "Rscript - <<'RS'
for(f in c('/workspace/inputs/res_1vs97.rds','/workspace/inputs/res_1vs98.rds')) {
 cat('\\n=====', f, '=====\\n')
 x <- readRDS(f)
 attrs <- attributes(x)
 cat('all attrs names:', paste(names(attrs), collapse=', '), '\\n')
 # metadata
 m <- attr(x,'metadata', exact=TRUE)
 cat('\\nmetadata typeof', typeof(m), 'len', length(m), '\\n')
 if(typeof(m)=='list') {
  cat('metadata names:', paste(names(m), collapse=', '), '\\n')
  for(nm in names(m)) { cat('metadata item', nm, 'typeof', typeof(m[[nm]]), 'len', length(m[[nm]]), ':\\n'); if(length(m[[nm]])<=100) print(unclass(m[[nm]]) else str(unclass(m[[nm]]), max.level=2)) }
 }
 em <- attr(x,'elementMetadata', exact=TRUE)
 cat('\\nelementMetadata typeof', typeof(em), 'len', length(em), 'names?')
 emu <- unclass(em)
 cat(' after unclass typeof', typeof(emu), 'len', length(emu), 'names', paste(names(emu), collapse=','), '\\n')
 for(nm in names(emu)) {cat('elem col', nm, 'typeof', typeof(emu[[nm]]), 'len', length(emu[[nm]])); if(length(emu[[nm]])<=30) {cat(':'); print(emu[[nm]])} else cat('\\n')}
 # prior
 pr <- attr(x,'priorInfo', exact=TRUE)
 cat('\\npriorInfo unclass:'); print(unclass(pr))
}
RS"
