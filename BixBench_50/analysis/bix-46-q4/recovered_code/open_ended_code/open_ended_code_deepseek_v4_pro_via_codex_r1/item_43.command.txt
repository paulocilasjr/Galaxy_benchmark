/bin/bash -c "python3 - <<'PY'
import zipfile,re,html,xml.etree.ElementTree as ET
ns={'w':'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
def table_text(path):
 z=zipfile.ZipFile(path)
 root=ET.fromstring(z.read('word/document.xml'))
 tables=[]
 for tbl in root.findall('.//w:tbl',ns):
  rows=[]
  for tr in tbl.findall('.//w:tr',ns):
   cells=[]
   for tc in tr.findall('.//w:tc',ns):
    parts=[]
    for p in tc.findall('.//w:t',ns):
     if p.text: parts.append(p.text)
    cells.append(''.join(parts))
   rows.append(cells)
  tables.append(rows)
 return tables
for fn in ['/workspace/tmp/s012.docx','/workspace/tmp/s013.docx']:
 tabs=table_text(fn)
 print('\\n',fn,'tables',len(tabs))
 for ti,t in enumerate(tabs):
  print('table',ti,'rows',len(t),'first5',t[:5])
  # find PA14_35160
  for row in t:
   if row and 'PA14_35160' in row[0]: print('target',row)
PY"
