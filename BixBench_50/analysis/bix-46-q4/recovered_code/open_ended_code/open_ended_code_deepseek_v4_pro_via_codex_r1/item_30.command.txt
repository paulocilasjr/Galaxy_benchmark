/bin/bash -c "Rscript - <<'RS'
a<-readRDS('/workspace/inputs/res_1vs97.rds'); b<-readRDS('/workspace/inputs/res_1vs98.rds')
rna<-attr(a,'rownames',exact=T); rnb<-attr(b,'rownames',exact=T)
lda<-unclass(attr(a,'listData',exact=T)); ldb<-unclass(attr(b,'listData',exact=T))
cat('same order?', identical(rna,rnb), '\\n')
cat('common genes', sum(rna %in% rnb), length(rna), length(rnb), '\\n')
i<-match(rna,rnb)
cat('baseMean same aligned?', identical(lda"'$baseMean, ldb$baseMean[i]), '"'\\n')
cat('max diff', max(abs(lda"'$baseMean - ldb$baseMean[i])), '"'\\n')
# correlations lfc between contrasts
cat('cor lfc', cor(lda"'$log2FoldChange, ldb$log2FoldChange[i]), '"'\\n')
# print known phz genes values
for(g in c('PA14_09400','PA14_09480','PA14_09490','PA14_09500','PA14_09520','PA14_09530','PA14_09540','PA14_35160','PA14_00640')) {
 ia<-match(g,rna); ib<-match(g,rnb)
 cat(g,'97',lda"'$log2FoldChange[ia],'"'98',ldb"'$log2FoldChange[ib],'"'base',lda"'$baseMean[ia],'"'\\n')
}
RS"
