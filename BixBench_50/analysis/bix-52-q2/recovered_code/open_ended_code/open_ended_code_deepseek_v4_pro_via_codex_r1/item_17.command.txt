/bin/bash -lc "python3 - <<'PY'
import csv, collections
from fractions import Fraction
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lenp='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(lenp,newline='',encoding='utf-8-sig') as f:
    for r in csv.DictReader(f): lengths[r['Chromosome']]=int(r['Length'])
unique={}
with open(cp,newline='') as f:
    for r in csv.DictReader(f):
        m=float(r['MethylationPercentage'])
        if m>90 or m<10:
            unique[r['Pos']]=r['Chromosome']
counts=collections.Counter(unique.values())
fracs=[Fraction(counts[c],lengths[c]) for c in counts if c in lengths]
mean=sum(fracs, Fraction(0))/len(fracs)
print(mean)
print(float(mean))
print(format(float(mean), '.17g'))
print(format(float(mean), '.20f'))
print(format(float(mean), '.20e'))
# also using Decimal high precision from floats
from decimal import Decimal, getcontext
getcontext().prec=80
d=Decimal(0)
for c in counts:
    if c in lengths:
        d += Decimal(counts[c])/Decimal(lengths[c])
d /= Decimal(len(fracs))
print(d)
PY"
