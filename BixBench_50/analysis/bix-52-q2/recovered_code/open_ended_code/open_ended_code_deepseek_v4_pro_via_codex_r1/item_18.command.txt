/bin/bash -lc "python3 - <<'PY'
import csv, collections
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'; lenp='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(lenp,newline='',encoding='utf-8-sig') as f:
    for r in csv.DictReader(f): lengths[r['Chromosome']]=int(r['Length'])
rows=collections.Counter(); unique={}; duplicates=[]
with open(cp,newline='') as f:
    for r in csv.DictReader(f):
        m=float(r['MethylationPercentage'])
        if m>90 or m<10:
            rows[r['Chromosome']]+=1
            if r['Pos'] in unique and unique[r['Pos']]"'!=r['"'Chromosome']:
                print('mismatch')
            unique[r['Pos']]=r['Chromosome']
            duplicates.append(r['Pos'])
dups=[p for p,c in collections.Counter(duplicates).items() if c>1]
print('duplicate filtered positions',dups)
print('rows counts',rows)
sel=[rows[c]/lengths[c] for c in rows if c in lengths]
print('rows mean',sum(sel)/len(sel))
# position with duplicate filter rows details
with open(cp,newline='') as f:
    for r in csv.DictReader(f):
        if r['Pos'] in dups:
            print(r)
PY"
