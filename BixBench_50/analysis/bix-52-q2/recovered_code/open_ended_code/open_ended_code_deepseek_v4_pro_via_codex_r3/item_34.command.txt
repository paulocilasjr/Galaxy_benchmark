/bin/bash -c "python3 - <<'PY'
import csv, collections
from fractions import Fraction
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8')))
lengths={}
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig')):
    lengths[r['Chromosome']]=int(r['Length'])
q=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
seen=set(); counts=collections.defaultdict(int)
for r in q:
    pos=r['Pos']; ch=r['Chromosome']
    if pos not in seen:
        seen.add(pos); counts[ch]+=1
# if any pos has different chrom? not.
# exact mean
total=Fraction(0)
for ch,n in counts.items():
    total += Fraction(n, lengths[ch])
mean=total/Fraction(len(counts))
print(float(mean))
print(mean.numerator, mean.denominator)
print(mean.limit_denominator())
# full decimal from numerator/denominator high precision
from decimal import Decimal, getcontext
getcontext().prec=80
print(Decimal(mean.numerator)/Decimal(mean.denominator))
PY"
