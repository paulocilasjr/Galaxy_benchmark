import csv, collections, statistics
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='')))
groups=collections.defaultdict(list)
for r in rows:
    groups[(r['Chromosome'],r['Pos'])].append(float(r['MethylationPercentage']))
means={k:statistics.mean(v) for k,v in groups.items()}
# filter unique cpg by mean >90 or <10
q={k for k,m in means.items() if m>90 or m<10}
print('unique',len(groups),'filtered by mean',len(q))
# count by chr
chrq=collections.Counter(chr for chr,pos in q)
lengths=dict((r['Chromosome'].strip(),int(r['Length'])) for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',encoding='utf-8-sig')))
dens=[chrq[c]/lengths[c] for c in chrq]
print(chrq)
print('mean',sum(dens)/len(dens))
# filter unique by median
med={k:statistics.median(v) for k,v in groups.items()}
q2={k for k,m in med.items() if m>90 or m<10}
chrq2=collections.Counter(chr for chr,pos in q2)
dens2=[chrq2[c]/lengths[c] for c in chrq2]
print('median filter',len(q2),chrq2,'mean',sum(dens2)/len(dens2))
PY"
