/bin/bash -c "python3 - <<'PY'
import csv, collections
spec='ZF'
rows=list(csv.DictReader(open(f'inputs/{spec}_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8')))
# first row per pos and last
first={}; last={}
for r in rows:
    if r['Pos'] not in first: first[r['Pos']]=r
    last[r['Pos']]=r
for name,d in [('first',first),('last',last)]:
    q=[r for r in d.values() if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
    chrpos=collections.Counter(r['Chromosome'] for r in q)
    print(name,len(q),chrpos.get('Z'))
# any row
q=set(r['Pos'] for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10)
# count Z by looking up
print('any uniq total',len(q),'Z',sum(1 for p in q if p.startswith('Z_')))
PY"
