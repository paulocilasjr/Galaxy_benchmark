/bin/bash -c "python3 - <<'PY'
import csv, collections, math
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8')))
lengths={}
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig')):
    lengths[r['Chromosome']]=int(r['Length'])
q=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
seen=set(); counts=collections.defaultdict(int)
for r in q:
    if r['Pos'] not in seen:
        seen.add(r['Pos']); counts[r['Chromosome']]+=1
auto=[ch for ch in counts if ch not in ('W','Z','MT')]
dens=[counts[ch]/lengths[ch] for ch in auto]
print(len(auto), math.fsum(dens)/len(dens))
# excluding W only / Z only
print('no W', math.fsum(counts[ch]/lengths[ch] for ch in counts if ch"'!='"'W')/sum(1 for ch in counts if ch"'!='"'W'))
print('no Z', math.fsum(counts[ch]/lengths[ch] for ch in counts if ch"'!='"'Z')/sum(1 for ch in counts if ch"'!='"'Z'))
PY"
