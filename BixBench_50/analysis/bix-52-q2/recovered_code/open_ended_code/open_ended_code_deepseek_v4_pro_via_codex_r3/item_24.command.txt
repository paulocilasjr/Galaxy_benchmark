/bin/bash -c "python3 - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='')))
lengths={}
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',encoding='utf-8-sig',newline='')):
    lengths[r['Chromosome'].strip()]=int(r['Length'])
q=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
chrpos=collections.defaultdict(set)
for r in q: chrpos[r['Chromosome']].add(r['Pos'])
counts={k:len(v) for k,v in chrpos.items()}
total_cpgs=sum(counts.values())
total_len=sum(lengths[k] for k in counts)
print(total_cpgs,total_len,total_cpgs/total_len)
print('mean individual', sum(counts[k]/lengths[k] for k in counts)/len(counts))
PY"
