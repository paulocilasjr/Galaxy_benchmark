import pandas as pd
for f in ['inputs/JD_AgeRelated_CpG_noMT_Final.csv','inputs/ZF_AgeRelated_CpG_noMT_Final.csv']:

    print('SampleShared count', df.groupby('Pos')['SampleShared'].nunique().describe())

    print('rows >90', (df['MethylationPercentage']>90).sum(), 'rows <10', (df['MethylationPercentage']<10).sum())
    print('unique positions with any qualifying row', df.groupby('Pos')['MethylationPercentage'].apply(lambda s: ((s>90)|(s<10)).any()).sum())
