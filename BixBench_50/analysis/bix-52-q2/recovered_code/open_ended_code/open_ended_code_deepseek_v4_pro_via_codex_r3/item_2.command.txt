/bin/bash -c "python3 - <<'PY'
import csv, collections
for f in ['inputs/JD_AgeRelated_CpG_noMT_Final.csv','inputs/ZF_AgeRelated_CpG_noMT_Final.csv']:
    with open(f,newline='') as fh:
        r=csv.DictReader(fh)
        cols=r.fieldnames
        rows=list(r)
    print('\\n',f, len(rows), cols)
    print(rows[0])
    poscounts=collections.Counter(x['Pos'] for x in rows)
    print('unique positions',len(poscounts),'duplicate rows',sum(1 for v in poscounts.values() if v>1),'max copies',max(poscounts.values()))
    chrs=sorted(set(x['Chromosome'] for x in rows), key=lambda z:int(z) if z.isdigit() else z)
    print('chromosomes',chrs)
    qual_rows=[x for x in rows if float(x['MethylationPercentage'])>90 or float(x['MethylationPercentage'])<10]
    print('qual rows',len(qual_rows))
    qual_pos=set(x['Pos'] for x in qual_rows)
    print('qual unique pos',len(qual_pos))
    bychr=collections.Counter(x['Chromosome'] for x in qual_rows)
    print('qual rows chr counts',bychr)
    # Count unique pos per chromosome among qual rows
    bychr_pos={}
    for x in qual_rows:
        bychr_pos.setdefault(x['Chromosome'],set()).add(x['Pos'])
    print('qual unique pos chr counts', {k:len(v) for k,v in sorted(bychr_pos.items(), key=lambda kv:int(kv[0]) if kv[0].isdigit() else kv[0])})
PY"
