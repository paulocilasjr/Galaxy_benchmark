/bin/bash -c "python3 - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='')))
lengths={}
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',encoding='utf-8-sig',newline='')):
    lengths[r['Chromosome'].strip()]=int(r['Length'])
groups=collections.defaultdict(list)
for r in rows:
    groups[r['Pos']].append((r['Chromosome'],float(r['MethylationPercentage'])))
for mode in ['any','all']:
    chrpos=collections.defaultdict(set)
    for pos,vals in groups.items():
        ext=[m>90 or m<10 for ch,m in vals]
        if (mode=='any' and any(ext)) or (mode=='all' and all(ext)):
            chrpos[vals[0][0]].add(pos)
    counts={k:len(v) for k,v in chrpos.items()}
    dens=[counts[k]/lengths[k] for k in counts]
    print(mode,len(counts),sum(dens)/len(dens),counts)
PY"
