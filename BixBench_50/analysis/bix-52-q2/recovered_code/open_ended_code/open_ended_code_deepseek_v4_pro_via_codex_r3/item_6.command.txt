/bin/bash -c "python3 - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='')))
lengths={}
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig')):
    lengths[r['Chromosome'].strip()]=int(r['Length'])
# filter rows
q=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
print('qual rows',len(q))
seen=set(); chr_pos=collections.defaultdict(set)
for r in q:
    pos=r['Pos']; chr=r['Chromosome']
    if pos not in seen:
        seen.add(pos); chr_pos[chr].add(pos)
    else:
        # check duplicate pos chromosome consistency? add anyway no effect if unique only
        chr_pos[chr].add(pos)
print('unique filtered positions total', sum(len(v) for v in chr_pos.values()))
print('chrs with counts:')
dens=[]
for c,poss in sorted(chr_pos.items(), key=lambda kv: (not kv[0].isdigit(), kv[0])):
    L=lengths.get(c)
    d=len(poss)/L
    print(c,len(poss),L,d)
    dens.append(d)
print('n chrs',len(dens),'mean',sum(dens)/len(dens))
# alternative unique first then filter any qual row; should same
# alternative aggregate mean and filter unique (check if results differ maybe)
PY"
