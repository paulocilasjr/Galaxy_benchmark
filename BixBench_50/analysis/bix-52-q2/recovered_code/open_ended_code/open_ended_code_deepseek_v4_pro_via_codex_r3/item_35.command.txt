/bin/bash -c "python3 - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8')))
lengths=[]
for r in csv.DictReader(open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig')):
    lengths.append((r['Chromosome'],int(r['Length'])))
q=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
seen=set(); counts=collections.defaultdict(int)
for r in q:
    p=r['Pos']
    if p not in seen:
        seen.add(p); counts[r['Chromosome']]+=1
# average in length file order including only chrs with counts
vals=[]
for ch,L in lengths:
    if ch in counts:
        vals.append(counts[ch]/L)
print(len(vals))
s=sum(vals)
print(repr(s/len(vals)))
# average in sorted counts order
vals2=[counts[ch]/dict(lengths)[ch] for ch in counts]
print(repr(sum(vals2)/len(vals2)))
# math.fsum
import math
print(repr(math.fsum(vals)/len(vals)))
print(repr(math.fsum(vals2)/len(vals2)))
PY"
