/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=50
lengths={}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f): lengths[r['Chromosome']]=int(r['Length'])
filtered=defaultdict(set)
filtered_by_coord=defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        m=Decimal(r['MethylationPercentage'])
        if m < Decimal(10) or m > Decimal(90):
            filtered[r['Chromosome']].add(r['Pos'])
            filtered_by_coord[r['Chromosome']].add((r['StartPosition'],r['EndPosition']))
assert set(filtered) <= set(lengths)
assert {c:len(v) for c,v in filtered.items()} == {c:len(v) for c,v in filtered_by_coord.items()}
order=lambda c:(c in ('W','Z','MT'), int(c) if c.isdigit() else c)
densities=[]
for c in sorted(filtered,key=order):
    n=len(filtered[c]); d=Decimal(n)/Decimal(lengths[c]); densities.append(d)
    print(f'{c}\\t{n}\\t{lengths[c]}\\t{d}')
mean=sum(densities,Decimal(0))/Decimal(len(densities))
pooled=Decimal(sum(map(len,filtered.values())))/Decimal(sum(lengths[c] for c in filtered))
print('chromosomes',len(densities))
print('unique_cpgs',sum(map(len,filtered.values())))
print('arithmetic_mean_density',mean)
print('pooled_density_for_contrast',pooled)
PY"
