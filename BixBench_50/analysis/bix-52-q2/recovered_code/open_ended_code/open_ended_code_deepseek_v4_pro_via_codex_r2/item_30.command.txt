/bin/bash -lc "python3 - <<'PY'
from decimal import Decimal, getcontext
import csv, collections
getcontext().prec=80
lens={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
 for row in csv.DictReader(f): lens[row['Chromosome']]=int(row['Length'])
S=set()
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='') as f:
 for row in csv.DictReader(f):
  p=float(row['MethylationPercentage'])
  if p>90 or p<10: S.add((row['Chromosome'],row['Pos']))
cnt=collections.Counter(c for c,_ in S)
dens=[Decimal(cnt[c])/Decimal(lens[c]) for c in cnt]
mean=sum(dens)/Decimal(len(dens))
print(mean)
# also print each decimal for inspection
for c,n in sorted(cnt.items()):
 print(c,n,lens[c], (Decimal(n)/Decimal(lens[c])))
PY"
