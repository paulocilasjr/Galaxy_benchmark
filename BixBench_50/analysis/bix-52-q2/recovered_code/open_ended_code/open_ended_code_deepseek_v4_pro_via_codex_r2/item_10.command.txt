/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict
fn='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lens={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        lens[row['Chromosome']]=int(row['Length'])
filtered=set()
with open(fn,newline='') as f:
    for row in csv.DictReader(f):
        pct=float(row['MethylationPercentage'])
        if pct>90 or pct<10:
            filtered.add((row['Chromosome'], row['Pos']))
print('n filtered unique', len(filtered))
counts=defaultdict(int)
for c,p in filtered:
    counts[c]+=1
dens={}
for c,n in counts.items():
    L=lens[c]
    dens[c]=n/L
    print(c,n,L,dens[c])
print('count chroms', len(dens))
print('mean', sum(dens.values())/len(dens))
# rounded variants
print('repr mean', repr(sum(dens.values())/len(dens)))
PY"
