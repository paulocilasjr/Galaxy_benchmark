/bin/bash -lc "python3 - <<'PY'
import csv, collections, statistics
lens={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
 for row in csv.DictReader(f): lens[row['Chromosome']]=int(row['Length'])
rows=list(csv.DictReader(open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='')))
for label,pred in [
 ('strict >90 or <10',lambda p:p>90 or p<10),
 ('>=90 or <=10',lambda p:p>=90 or p<=10),
 ('>90 or <=10',lambda p:p>90 or p<=10),
 ('>=90 or <10',lambda p:p>=90 or p<10),
]:
 S={(r['Chromosome'],r['Pos']) for r in rows if pred(float(r['MethylationPercentage']))}
 cnt=collections.Counter(c for c,_ in S)
 dens=[cnt[c]/lens[c] for c in cnt]
 print(label,len(S),len(dens),statistics.mean(dens))
 print(collections.Counter(c for c,_ in S))
PY"
