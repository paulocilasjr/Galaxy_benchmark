/bin/bash -lc "python3 - <<'PY'
import pandas as pd
cpgs = pd.read_csv('inputs/JD_AgeRelated_CpG_noMT_Final.csv')
lengths = pd.read_csv('inputs/JD_Chromosome_Length.csv', encoding='utf-8-sig')
f = cpgs[(cpgs['MethylationPercentage'] > 90) | (cpgs['MethylationPercentage'] < 10)].drop_duplicates(subset=['Chromosome','Pos'])
counts = f.groupby('Chromosome').size().rename('n').reset_index()
m = counts.merge(lengths, on='Chromosome')
m['density'] = m['n'] / m['Length']
print(m[['Chromosome','n','Length','density']].to_string(index=False))
print('mean', m['density'].mean(), repr(m['density'].mean()))
PY"
