/bin/bash -lc "python3 - <<'PY'
import csv
fn='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
rows=[]
with open(fn,newline='') as f:
    for row in csv.DictReader(f):
        rows.append(row)
uniq_pos={}
for x in rows:
    uni=(x['Chromosome'],x['Pos'])
    uniq_pos.setdefault(uni,[]).append(float(x['MethylationPercentage']))
print('unique total', len(uniq_pos))
print('unique with any row high/low', sum(1 for v in uniq_pos.values() if any(p>90 or p<10 for p in v)))
print('unique with all high/low', sum(1 for v in uniq_pos.values() if all(p>90 or p<10 for p in v)))
print('unique mean high/low', sum(1 for v in uniq_pos.values() if (sum(v)/len(v))>90 or (sum(v)/len(v))<10))
# rows high/low unique positions
hi=set((x['Chromosome'],x['Pos']) for x in rows if float(x['MethylationPercentage'])>90 or float(x['MethylationPercentage'])<10)
print('unique positions from filtered rows', len(hi))
# counts per chr
from collections import Counter
print(Counter(c for c,_ in hi))
PY"
