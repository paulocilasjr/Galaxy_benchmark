/bin/bash -lc "python3 - <<'PY'
import csv, collections
fn='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
g=collections.defaultdict(list)
with open(fn,newline='') as f:
 for row in csv.DictReader(f):
  pct=float(row['MethylationPercentage'])
  if pct>90 or pct<10:
   g[(row['Chromosome'],row['Pos'])].append((row['Sample'],pct))
for k,v in g.items():
 if len(v)>1: print(k,v)
PY"
