import csv, collections
lens={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
 for row in csv.DictReader(f): lens[row['Chromosome']]=int(row['Length'])
S=set()
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='') as f:
 for row in csv.DictReader(f):
  p=float(row['MethylationPercentage'])
  if p>90 or p<10:
   S.add((row['Chromosome'],row['Pos']))
cnt=collections.Counter(c for c,_ in S)
dens=[cnt[c]/lens[c] for c in cnt]
print('verification filtered unique:',len(S),'chromosomes:',len(dens),'mean:',sum(dens)/len(dens))
PY"
