/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=50
age='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lengths='inputs/JD_Chromosome_Length.csv'
with open(age,newline='') as f:
    rows=list(csv.DictReader(f))
filtered={r['Pos']: r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10}
counts=defaultdict(int)
for r in filtered.values(): counts[r['Chromosome']]+=1
with open(lengths,newline='',encoding='utf-8-sig') as f:
    lens={r['Chromosome']: Decimal(r['Length']) for r in csv.DictReader(f)}
used=sorted(counts,key=lambda c: (0,int(c)) if c.isdigit() else (1,c))
densities={c: Decimal(counts[c])/lens[c] for c in used}
mean=sum(densities.values())/Decimal(len(densities))
print('used chromosomes:',','.join(used))
print('number used:',len(used))
print('total unique filtered:',sum(counts.values()))
for c in used: print(f'{c}\\t{counts[c]}\\t{lens[c]}\\t{densities[c]}')
print('sum densities',sum(densities.values()))
print('mean exact decimal',mean)
print('mean scientific 20 sig',format(mean,'.20E'))
PY"
