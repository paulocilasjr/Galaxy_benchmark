/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=60
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='') as f:
    rows=list(csv.DictReader(f))
unique={row['Pos']: row for row in rows if (float(row['MethylationPercentage']) > 90.0 or float(row['MethylationPercentage']) < 10.0)}
counts=defaultdict(int)
for row in unique.values(): counts[row['Chromosome']] += 1
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    lengths={row['Chromosome']: Decimal(row['Length']) for row in csv.DictReader(f)}
mean=sum((Decimal(n)/lengths[c] for c,n in counts.items()), Decimal(0))/Decimal(len(counts))
with open('codex_output/answer.txt') as f:
    content=f.read()
print('answer repr:',repr(content))
print('line count:',len(content.splitlines()))
print('numeric:',Decimal(content.strip()))
print('recomputed:',mean)
print('match:',Decimal(content.strip()) == mean)
assert len(content.splitlines()) == 1
assert Decimal(content.strip()) == mean
PY"
