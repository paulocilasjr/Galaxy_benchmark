import csv
from collections import defaultdict

# Verify: are all filtered chromosomes in the length file?
chr_len = {}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chr_len[row['Chromosome']] = int(row['Length'])

# Recompute and verify
cpgs_by_chr = defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        meth = float(row['MethylationPercentage'])
        if meth > 90 or meth < 10:
            chrom = row['Chromosome']
            pos = row['Pos']
            cpgs_by_chr[chrom].add(pos)

# Verify all chromosomes have length data
missing = [c for c in cpgs_by_chr if c not in chr_len]
print(f"Chromosomes missing length data: {missing}")

# Recompute with verification
densities = []
for chrom in sorted(cpgs_by_chr.keys(), key=lambda x: int(x) if x.isdigit() else 999):
    count = len(cpgs_by_chr[chrom])
    length = chr_len[chrom]
    density = count / length
    densities.append(density)

mean_density = sum(densities) / len(densities)
print(f"Sum of densities: {sum(densities):.15e}")
print(f"Number of chromosomes: {len(densities)}")
print(f"Mean density: {mean_density:.15e}")

# Also compute a quick weighted alternative (global density) for sanity check
total_cpgs = sum(len(v) for v in cpgs_by_chr.values())
total_length = sum(chr_len[c] for c in cpgs_by_chr)
print(f"Global density (all CpGs / all lengths): {total_cpgs/total_length:.15e}")
print(f"Total filtered unique CpGs: {total_cpgs}")
print(f"Total length of filtered chromosomes: {total_length}")
