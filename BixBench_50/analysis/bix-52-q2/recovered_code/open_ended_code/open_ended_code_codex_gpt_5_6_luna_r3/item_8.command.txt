/bin/bash -lc "python - <<'PY'
import csv, collections, statistics

age_path = 'inputs/JD_AgeRelated_CpG_noMT_Final.csv'
length_path = 'inputs/JD_Chromosome_Length.csv'

filtered_rows = []
with open(age_path, newline='') as f:
    for r in csv.DictReader(f):
        m = float(r['MethylationPercentage'])
        if m > 90.0 or m < 10.0:
            filtered_rows.append(r)

# A CpG is uniquely identified by its genomic position; Pos is verified as chr_start.
unique_by_pos = {}
for r in filtered_rows:
    unique_by_pos.setdefault(r['Pos'], r)

lengths = {}
with open(length_path, newline='') as f:
    rd = csv.reader(f)
    header = next(rd)
    chromosome_col = next(i for i, name in enumerate(header) if name.lstrip('\\ufeff') == 'Chromosome')
    length_col = header.index('Length')
    for row in rd:
        lengths[row[chromosome_col]] = int(row[length_col])

counts = collections.Counter(r['Chromosome'] for r in unique_by_pos.values())
included = [c for c in lengths if counts[c] > 0]
densities = {c: counts[c] / lengths[c] for c in included}
mean_density = statistics.fmean(densities.values())

print('filtered rows =', len(filtered_rows))
print('filtered unique CpGs =', len(unique_by_pos))
print('included chromosomes =', len(included))
print('included details:')
for c in sorted(included, key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else x)):
    print(c, counts[c], lengths[c], repr(densities[c]))
print('sum densities =', repr(sum(densities.values())))
print('mean density =', repr(mean_density))
print('mean density .17g =', format(mean_density, '.17g'))
print('mean density .15e =', format(mean_density, '.15e'))
PY"
