/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec = 50

lengths={}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as fh:
    for row in csv.DictReader(fh):
        lengths[row['Chromosome']]=int(row['Length'])

filtered=defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as fh:
    for row in csv.DictReader(fh):
        m=Decimal(row['MethylationPercentage'])
        if m > Decimal('90') or m < Decimal('10'):
            filtered[row['Chromosome']].add((row['StartPosition'], row['EndPosition']))

mean=sum(Decimal(len(s))/Decimal(lengths[c]) for c,s in filtered.items())/Decimal(len(filtered))
answer=format(mean, 'f').rstrip('0').rstrip('.')
print(answer)
print(format(mean, '.30E'))
PY"
