/bin/bash -lc "python - <<'PY'
import csv, collections, statistics
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
rows=0; filt=0
pos_chrs=collections.defaultdict(set); all_pos=set(); filt_pos=set(); exact=collections.Counter(); chr_rows=collections.Counter(); chr_filt_rows=collections.Counter()
with open(p, newline='', encoding='utf-8-sig') as f:
    rd=csv.DictReader(f)
    print('columns', rd.fieldnames)
    for x in rd:
        rows+=1
        chrom=x['Chromosome'].strip(); pos=x['Pos'].strip(); m=float(x['MethylationPercentage'])
        all_pos.add(pos); pos_chrs[pos].add(chrom); chr_rows[chrom]+=1
        if m in (0,10,90,100): exact[m]+=1
        if m>90 or m<10:
            filt+=1; filt_pos.add((chrom,pos)); chr_filt_rows[chrom]+=1
print('rows',rows,'unique Pos',len(all_pos),'filtered rows',filt,'filtered unique chr+Pos',len(filt_pos))
print('Pos on multiple chromosomes',sum(len(v)>1 for v in pos_chrs.values()))
print('exact boundary/common extrema',dict(exact))
print('chromosomes all',sorted(chr_rows, key=lambda s:(not s.isdigit(),int(s) if s.isdigit() else s)))
print('filtered row chrom counts',dict(chr_filt_rows))
print('sample unique filtered loci',sorted(filt_pos)[:10])

with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    lens=list(csv.DictReader(f))
print('length rows',len(lens),'chromosomes',[x['Chromosome'] for x in lens])
PY"
