/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=40
length={}
with open('inputs/JD_Chromosome_Length.csv',newline='',encoding='utf-8-sig') as f:
    for r in csv.DictReader(f): length[r['Chromosome']]=int(r['Length'])
loci=defaultdict(set); filtered_rows=defaultdict(int)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv',newline='',encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        m=Decimal(r['MethylationPercentage'])
        if m>90 or m<10:
            loci[r['Chromosome']].add(r['Pos'])
            filtered_rows[(r['Chromosome'],r['Pos'])]+=1
order=lambda c:(c not in ('W','Z') and not c.isdigit(), int(c) if c.isdigit() else 10**9+(0 if c=='W' else 1))
vals=[]
for c in sorted(loci,key=order):
    d=Decimal(len(loci[c]))/Decimal(length[c])
    vals.append(d)
    print(f'{c:>2} count={len(loci[c]):2d} length={length[c]:9d} density={d}')
mean=sum(vals)/Decimal(len(vals))
print('represented_chromosomes',len(vals))
print('filtered_unique_loci',sum(map(len,loci.values())))
print('mean_density',mean)
print('mean_scientific',format(mean,'.15E'))
print('duplicate_filtered_loci',[(k,n) for k,n in filtered_rows.items() if n>1])
PY
