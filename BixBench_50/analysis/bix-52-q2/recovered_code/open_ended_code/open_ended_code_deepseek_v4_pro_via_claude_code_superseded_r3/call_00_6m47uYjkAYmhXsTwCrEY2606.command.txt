import csv

# Read chromosome lengths
chrom_lengths = {}
with open('inputs/JD_Chromosome_Length.csv', 'r', encoding='utf-8-sig', newline='') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom_lengths[row['Chromosome']] = int(row['Length'])

# Read JD CpG data and filter
unique_pos_by_chrom = {}
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', 'r', encoding='utf-8-sig', newline='') as f:
    reader = csv.DictReader(f)
    for row in reader:
        meth = float(row['MethylationPercentage'])
        if meth > 90 or meth < 10:
            chrom = row['Chromosome']
            pos = row['Pos']
            if chrom not in unique_pos_by_chrom:
                unique_pos_by_chrom[chrom] = set()
            unique_pos_by_chrom[chrom].add(pos)

densities = []
for chrom in unique_pos_by_chrom:
    count = len(unique_pos_by_chrom[chrom])
    length = chrom_lengths[chrom]
    density = count / length
    densities.append(density)

mean_density = sum(densities) / len(densities)

# Verify: sum all unique CpGs and total lengths of used chromosomes
total_cpgs = sum(len(v) for v in unique_pos_by_chrom.values())
total_length = sum(chrom_lengths[chrom] for chrom in unique_pos_by_chrom)
print(f"Total unique CpGs: {total_cpgs}")
print(f"Total length of used chromosomes: {total_length}")
print(f"Number of chromosomes: {len(densities)}")
print(f"Mean density: {mean_density:.18e}")
print(f"Mean density: {mean_density}")

# Write answer
import os
os.makedirs('codex_output', exist_ok=True)
with open('codex_output/answer.txt', 'w') as f:
    f.write(f"{mean_density:.15e}\n")
print("Answer written to codex_output/answer.txt")
