import csv

# Read JD CpG data
rows = []
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        rows.append(row)

print(f"Total rows: {len(rows)}")
print(f"Columns: {list(rows[0].keys())}")

# Check unique positions
unique_pos = set(row['Pos'] for row in rows)
print(f"Unique positions: {len(unique_pos)}")

# Check methylation range
meth_values = [float(row['MethylationPercentage']) for row in rows]
print(f"Methylation range: {min(meth_values):.2f} to {max(meth_values):.2f}")

# Filter >90 or <10
filtered = [row for row in rows if float(row['MethylationPercentage']) > 90 or float(row['MethylationPercentage']) < 10]
print(f"Filtered rows (>90% or <10%): {len(filtered)}")

# Unique filtered positions
filtered_unique = set(row['Pos'] for row in filtered)
print(f"Unique filtered positions: {len(filtered_unique)}")

# Check chromosome distribution
from collections import Counter
chrom_counts_filtered = Counter(row['Chromosome'] for row in filtered)
print("\nFiltered rows per chromosome:")
for chrom in sorted(chrom_counts_filtered.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else float('inf'), x)):
    print(f"  Chr {chrom}: {chrom_counts_filtered[chrom]}")

# Unique filtered positions per chromosome
unique_pos_by_chrom = {}
for row in filtered:
    chrom = row['Chromosome']
    pos = row['Pos']
    if chrom not in unique_pos_by_chrom:
        unique_pos_by_chrom[chrom] = set()
    unique_pos_by_chrom[chrom].add(pos)

print("\nUnique filtered positions per chromosome:")
for chrom in sorted(unique_pos_by_chrom.keys(), key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else float('inf'), x)):
    print(f"  Chr {chrom}: {len(unique_pos_by_chrom[chrom])}")
