import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=50
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lp='inputs/JD_Chromosome_Length.csv'
lengths={}
with open(lp, newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f): lengths[row['Chromosome']]=int(row['Length'])
rows=0; all_pos=set(); filtered_pos=defaultdict(set); posmap={}; inconsistent=[]; exact10=exact90=0; minm=Decimal('Infinity'); maxm=Decimal('-Infinity')
with open(cp, newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        rows+=1
        pos=row['Pos']; chrom=row['Chromosome']; start=row['StartPosition']; m=Decimal(row['MethylationPercentage'])
        all_pos.add(pos); minm=min(minm,m); maxm=max(maxm,m)
        if m==10: exact10+=1
        if m==90: exact90+=1
        old=posmap.setdefault(pos,(chrom,start))
        if old"'!=(chrom,start): inconsistent.append((pos,old,(chrom,start)))
        if m>90 or m<10: filtered_pos[chrom].add(pos)
print('"'rows',rows,'unique positions',len(all_pos),'inconsistent',len(inconsistent),'range',minm,maxm,'exact10',exact10,'exact90',exact90)
print('length chromosomes',len(lengths),'filtered chromosomes',len(filtered_pos),'unmatched',set(filtered_pos)-set(lengths))
densities=[]
for chrom in lengths:
    n=len(filtered_pos.get(chrom,set()))
    if n:
        density=Decimal(n)/Decimal(lengths[chrom])
        densities.append(density)
        print(chrom,n,lengths[chrom],density)
mean=sum(densities,Decimal(0))/Decimal(len(densities))
print('mean',mean)
print('scientific',f'{mean:.15E}')
print('filtered unique total',sum(map(len,filtered_pos.values())))
PY"
