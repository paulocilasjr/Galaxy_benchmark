/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict, Counter
age='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lengths='inputs/JD_Chromosome_Length.csv'
# Read lengths, handling BOM.
chrom_len={}
with open(lengths, newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        chrom_len[row['Chromosome']]=int(row['Length'])
# Unique filtered CpGs keyed by stable locus identity. Pos also encodes chrom_start, but include fields to guard consistency.
loci=set()
with open(age, newline='') as f:
    for row in csv.DictReader(f):
        meth=float(row['MethylationPercentage'])
        if meth > 90 or meth < 10:
            chrom=row['Chromosome']
            locus=(chrom, int(row['StartPosition']), int(row['EndPosition']), row['Pos'])
            loci.add(locus)
counts=Counter(chrom for chrom, start, end, pos in loci)
missing=sorted(set(counts)-set(chrom_len))
if missing:
    raise SystemExit(f'Missing lengths for {missing}')
densities={chrom: counts[chrom]/chrom_len[chrom] for chrom in counts}
mean=sum(densities.values())/len(densities)
print('unique_filtered_loci', len(loci))
print('included_chromosomes', len(densities))
for chrom in sorted(densities, key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else x)):
    print(chrom, counts[chrom], chrom_len[chrom], densities[chrom])
print('mean', repr(mean), format(mean, '.17g'))
PY"
