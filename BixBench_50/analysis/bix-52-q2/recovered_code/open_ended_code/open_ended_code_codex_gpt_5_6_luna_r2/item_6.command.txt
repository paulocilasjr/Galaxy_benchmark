/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
from decimal import Decimal, getcontext
getcontext().prec=50
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lp='inputs/JD_Chromosome_Length.csv'
loc_by_pos={}
with open(cp, newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        pos=r['Pos']
        locus=(r['Chromosome'], r['StartPosition'], r['EndPosition'])
        if pos in loc_by_pos and loc_by_pos[pos] "'!= locus:
            raise ValueError(f'"'inconsistent locus for {pos}')
        loc_by_pos[pos]=locus
filtered=set()
with open(cp, newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        m=Decimal(r['MethylationPercentage'])
        if m > 90 or m < 10:
            filtered.add(r['Pos'])
counts=defaultdict(int)
for pos in filtered:
    chrom, start, end=loc_by_pos[pos]
    counts[chrom] += 1
lengths={}
with open(lp, newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        lengths[r['Chromosome']]=int(r['Length'])
rows=[]
for chrom in sorted(counts, key=lambda x: (0, int(x)) if x.isdigit() else (1, x)):
    density=Decimal(counts[chrom])/Decimal(lengths[chrom])
    rows.append((chrom,counts[chrom],lengths[chrom],density))
mean=sum((x[3] for x in rows),Decimal(0))/Decimal(len(rows))
for chrom,count,length,density in rows:
    print(f'{chrom}\\t{count}\\t{length}\\t{density}')
print(f'n_chromosomes\\t{len(rows)}')
print(f'n_unique_filtered\\t{len(filtered)}')
print(f'mean\\t{mean}')
PY"
