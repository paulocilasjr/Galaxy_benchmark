/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
p='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
rows=[]
with open(p, newline='') as f:
    rd=csv.DictReader(f)
    print('fields:', rd.fieldnames)
    for row in rd:
        rows.append(row)
print('rows',len(rows))
print('chroms', sorted(Counter(r['Chromosome'] for r in rows), key=lambda x: (len(x), x)))
print('unique Pos',len({r['Pos'] for r in rows}))
print('unique chrom,start,end',len({(r['Chromosome'],r['StartPosition'],r['EndPosition']) for r in rows}))
print('unique pos/chrom/start',len({(r['Pos'],r['Chromosome'],r['StartPosition']) for r in rows}))
print('rows per Pos distribution',Counter(Counter(r['Pos'] for r in rows).values()).most_common(10))
vals=[float(r['MethylationPercentage']) for r in rows]
print('meth min/max',min(vals),max(vals))
flt=[r for r in rows if float(r['MethylationPercentage'])>90 or float(r['MethylationPercentage'])<10]
print('filtered rows',len(flt),'unique Pos',len({r['Pos'] for r in flt}))
print('filtered unique pos by chrom',Counter(r['Chromosome'] for r in {r['Pos']:r for r in flt}.values()))
print('filtered row chrom',Counter(r['Chromosome'] for r in flt))
print('positions with conflicting chrom',sum(len({r['Chromosome'] for r in rows if r['Pos']==pos})>1 for pos in {r['Pos'] for r in rows}))
PY"
