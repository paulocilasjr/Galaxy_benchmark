/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
import re

def tv(s):
    b=[]
    for m in re.finditer(r':([0-9.eE+-]+)',s):
        b.append((float(m.group(1)), s[:m.start()].rstrip()[-1]==')'))
    return sum(v for v,i in b if i)/sum(v for v,i in b) if b else None, len(b),sum(i for _,i in b)
for fn in ['scogs_fungi.zip','scogs_animals.zip']:
  with ZipFile('/workspace/inputs/'+fn) as z:
    vals=[]
    for n in z.namelist():
      if n.endswith('.treefile'):
        vals.append(tv(z.read(n).decode()))
  print(fn,len(vals),Counter((x[1],x[2]) for x in vals))
  for th in [.04,.05,.06,.07,.1]: print(th,sum(x[0]>th for x in vals),100*sum(x[0]>th for x in vals)/len(vals))
PY"
