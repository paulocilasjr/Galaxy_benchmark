    END {printf "treefile=%d iqtree=%d clipkit=%d faa=%d mldist=%d bionj=%d log=%d model=%d checkpoint=%d other=%d total=%d'"\\n\",t,i,a,f,m,b,l,g,c,o,t+i+a+f+m+b+l+g+c+o}'
  unzip -Z1 \""'$f" | awk -F/ '"'{n="'$NF; if (n ~ /'"\\.treefile"'$/) print n}'"' | sort -V | sed -n '1,20p'
done"
