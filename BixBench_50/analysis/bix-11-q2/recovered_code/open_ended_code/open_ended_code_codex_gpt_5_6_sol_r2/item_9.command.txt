/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
import re
with ZipFile('inputs/scogs_fungi.zip') as z:
    vals=[]
    for n in z.namelist():
        if n.endswith('.iqtree'):
            text=z.read(n).decode()
            m=re.search(r'Sum of internal branch lengths: ["'^'"\\n]*\\((\\d+(?:\\.\\d+)?)% of tree length\\)', text)
            if not m:
                raise RuntimeError(f'missing metric: {n}')
            vals.append(float(m.group(1))/100)
    above=sum(v > 0.06 for v in vals)
    assert len(vals)==249
    assert above==88
    print(f'valid_values={len(vals)} above={above} percentage={100*above/len(vals):.15f}%')
PY
unzip -t inputs/scogs_fungi.zip | tail -1"
