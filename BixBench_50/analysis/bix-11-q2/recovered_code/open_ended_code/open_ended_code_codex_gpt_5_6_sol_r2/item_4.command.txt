/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter,defaultdict
from pathlib import Path
for zpath in ['inputs/scogs_fungi.zip','inputs/scogs_animals.zip']:
  with ZipFile(zpath) as z:
    names=z.namelist()
    ext=Counter()
    bases=defaultdict(set)
    for n in names:
      parts=n.split('.')
      base=parts[0]
      suffix='.'.join(parts[1:]) if len(parts)>1 else ''
      ext[suffix]+=1
      bases[base].add(suffix)
    print(zpath, 'members',len(names),'unique genes',len(bases))
    for k,v in sorted(ext.items()): print(v,k)
    print('treefiles',sum(x.endswith('.treefile') for x in names),'unique treefile bases',len({x.split('.')[0] for x in names if x.endswith('.treefile')}))
    # exact duplicate names?
    c=Counter(names); print('duplicate member names',sum(v>1 for v in c.values()))
PY
