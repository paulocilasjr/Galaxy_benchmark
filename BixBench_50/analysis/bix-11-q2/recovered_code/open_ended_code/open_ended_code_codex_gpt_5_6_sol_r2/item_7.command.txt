/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
import re, statistics
zp='inputs/scogs_fungi.zip'
num_pat=r'[-+]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][-+]?\\d+)?'
rows=[]
with ZipFile(zp) as z:
 for n in z.namelist():
  if not n.endswith('.treefile'): continue
  s=z.read(n).decode().strip()
  allbl=[float(x) for x in re.findall(r':('+num_pat+r')',s)]
  intbl=[float(x) for x in re.findall(r'\\):('+num_pat+r')',s)]
  total=sum(allbl); internal=sum(intbl)
  iq=z.read(n[:-8]+'iqtree').decode()
  m=re.search(r'Sum of internal branch lengths: ('+num_pat+r') \\(('+num_pat+r')% of tree length\\)',iq)
  tips=len(re.findall(r'(?:\\(|,)(["'^():,;]+):'"'+num_pat,s))
  rows.append((n.split('.')[0],tips,total,internal,internal/total if total else float('nan'),float(m.group(2))/100 if m else None,s))
 treebases={n.split('.')[0] for n in z.namelist() if n.endswith('.treefile')}
 failed=[]
 for n in z.namelist():
  if n.endswith('.log') and n.split('.')[0] not in treebases:
   failed.append((n,z.read(n).decode(errors='replace')))
print('trees',len(rows),'tips counts',Counter(x[1] for x in rows))
num=sum(x[4]>.06 for x in rows)
print('above .06 precise',num,'den',len(rows),'pct',100*num/len(rows))
print('above .06 printed IQtree',sum(x[5]>.06 for x in rows))
print('ratio ranges by tips')
for k in sorted(set(x[1] for x in rows)):
 a=[x[4] for x in rows if x[1]==k]
 print(k,len(a),min(a),statistics.median(a),max(a),'above',sum(v>.06 for v in a))
print('nearest threshold:')
for x in sorted(rows,key=lambda x:abs(x[4]-.06))[:15]: print(x[:6])
print('internal mismatch printed tolerance',max(abs(x[4]-x[5]) for x in rows))
print('non-tree logs',len(failed))
for n,txt in failed:
 print('\\n',n, '\\n', '\\n'.join(txt.splitlines()[-18:]))
print('sample 4-tip',next(x[-1] for x in rows if x[1]==4))
PY"
