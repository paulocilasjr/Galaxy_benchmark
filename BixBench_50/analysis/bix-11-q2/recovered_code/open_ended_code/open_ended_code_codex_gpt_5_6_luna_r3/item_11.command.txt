def base(n):
    return n.rsplit('/',1)[-1].split('.faa',1)[0]
def ids(suffix):
    out=[]

        if n.endswith(suffix): out.append(base(n))
    return set(out)
for suf in ['.faa','.faa.mafft','.faa.mafft.clipkit','.faa.mafft.clipkit.iqtree','.faa.mafft.clipkit.treefile','.faa.mafft.clipkit.log']:
    print(suf, len(ids(suf)))
missing = sorted(ids('.faa.mafft.clipkit') - ids('.faa.mafft.clipkit.iqtree'))
print('alignments lacking iqtree:', len(missing), missing)
for ident in missing:
    log = next((n for n in z.namelist() if n.endswith(ident+'.faa.mafft.clipkit.log')), None)
    print('\\n---', ident, log)
    if log:
        s=z.read(log).decode('utf8','replace')
