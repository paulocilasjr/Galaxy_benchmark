/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter
from io import StringIO
from Bio import Phylo
z=zipfile.ZipFile('inputs/scogs_fungi.zip')
rows=[]
for name in sorted(n for n in z.namelist() if n.endswith('.treefile')):
    nw=z.read(name).decode().strip()
    t=Phylo.read(StringIO(nw),'newick')
    terms=t.get_terminals()
    total=sum((c.branch_length or 0.0) for c in t.find_clades() if c is not t.root)
    internal=sum((c.branch_length or 0.0) for c in t.get_nonterminals() if c is not t.root)
    rows.append((name,len(terms),internal,total,internal/total if total else None,nw))
print('valid trees',len(rows))
print('tip counts',Counter(r[1] for r in rows))
print('above',sum(r[4] > 0.06 for r in rows))
print('at/below',sum(r[4] <= 0.06 for r in rows))
print('percentage',100*sum(r[4] > 0.06 for r in rows)/len(rows))
print('treeness ranges by tips')
for tips in sorted(set(r[1] for r in rows)):
    vals=[r[4] for r in rows if r[1]==tips]
    print(tips,len(vals),min(vals),max(vals),sum(v>0.06 for v in vals))
print('nearest threshold')
for r in sorted(rows,key=lambda r:abs(r[4]-.06))[:10]: print(r[:5])
PY
unzip -p inputs/scogs_fungi.zip 160593at2759.faa.mafft.clipkit.iqtree | rg -n -i 'tree length|internal branch|BEST SCORE FOUND|NOTE' | head -30"
