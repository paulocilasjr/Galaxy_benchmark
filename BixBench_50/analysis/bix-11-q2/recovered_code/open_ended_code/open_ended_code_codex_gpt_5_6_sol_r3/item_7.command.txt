/bin/bash -lc "python - <<'PY'
import zipfile,re,math
from collections import Counter
z=zipfile.ZipFile('inputs/scogs_fungi.zip')
pat=re.compile(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)\\s*\\(([0-9.eE+-]+)% of tree length\\)')
rows=[]; bad=[]
for name in sorted(n for n in z.namelist() if n.endswith('.iqtree')):
    s=z.read(name).decode(errors='replace')
    m=pat.search(s)
    if not m: bad.append(name)
    else: rows.append((name,float(m.group(1)),float(m.group(2))))
print('iqtree reports',len(rows),'missing metric',len(bad))
print('reported percent >6',sum(p>6 for _,_,p in rows),'percentage of genes',100*sum(p>6 for _,_,p in rows)/len(rows))
print('reported percent ==6',sum(p==6 for _,_,p in rows))
print('nearest reported threshold')
for x in sorted(rows,key=lambda x:abs(x[2]-6))[:12]: print(x)
print('iqtree filenames without treefile', sorted(set(n[:-7] for n in z.namelist() if n.endswith('.iqtree'))-set(n[:-8] for n in z.namelist() if n.endswith('.treefile'))))
print('treefile filenames without iqtree', sorted(set(n[:-8] for n in z.namelist() if n.endswith('.treefile'))-set(n[:-7] for n in z.namelist() if n.endswith('.iqtree'))))
PY"
