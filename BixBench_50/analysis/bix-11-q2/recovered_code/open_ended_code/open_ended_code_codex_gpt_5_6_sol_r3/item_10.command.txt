/bin/bash -lc "python - <<'PY'
import zipfile,re,statistics
z=zipfile.ZipFile('inputs/scogs_fungi.zip')
num=r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?'
allpat=re.compile(r':('+num+r')')
intpat=re.compile(r'\\)(?:["'^,:;()]*)?:('"'+num+r')')
rep_pat=re.compile(r'Sum of internal branch lengths:\\s*('+num+r')\\s*\\(('+num+r')% of tree length\\)')
rows=[]
for n in sorted(x for x in z.namelist() if x.endswith('.treefile')):
    nw=z.read(n).decode().strip()
    lengths=[float(x) for x in allpat.findall(nw)]
    internals=[float(x) for x in intpat.findall(nw)]
    total=sum(lengths); inner=sum(internals); tr=inner/total
    iq=n[:-len('.treefile')]+'.iqtree'
    m=rep_pat.search(z.read(iq).decode(errors='replace'))
    report=float(m.group(2))/100
    rows.append((n,tr,report,len(lengths)-len(internals),len(internals)))
print('tree count',len(rows))
print('structure counts',dict(__import__('collections').Counter((r[3],r[4]) for r in rows)))
print('full-precision above 0.06',sum(r[1]>0.06 for r in rows))
print('percentage',sum(r[1]>0.06 for r in rows)/len(rows)*100)
print('max abs report rounding delta',max(abs(r[1]-r[2]) for r in rows))
print('nearest exact')
for r in sorted(rows,key=lambda r:abs(r[1]-.06))[:8]: print(r[:3])
PY"
