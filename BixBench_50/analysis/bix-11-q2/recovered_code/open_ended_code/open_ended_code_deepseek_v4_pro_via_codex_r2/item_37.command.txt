/bin/bash -lc 'tmpdir=$(cat /workspace/fungi_trees_dir.txt); TREEDIR="$tmpdir" python - <<'"'PY'
import os, glob
from phykit.services.tree.treeness import Treeness
files=sorted(glob.glob(os.path.join(os.environ['TREEDIR'],'*.treefile')))
vals=[]
for f in files:
    s=Treeness._scan_simple_newick_summary(f)
    _,total,internal=s
    v=float(internal/total) if total else 0.0
    vals.append(v)
for thr in [0.06]:
 print('raw >', sum(v>thr for v in vals))
 print('rounded >', sum(round(v,4)>thr for v in vals))
 print('rounded >=', sum(round(v,4)>=thr for v in vals))
print('raw max <=.06', max(v for v in vals if v<=0.06))
print('raw min >.06', min(v for v in vals if v>0.06))
print('count total', len(vals))
PY"
