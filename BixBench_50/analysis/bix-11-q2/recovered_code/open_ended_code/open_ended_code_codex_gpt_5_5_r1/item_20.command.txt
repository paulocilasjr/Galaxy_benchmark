/bin/bash -lc "python3 - <<'PY'
import zipfile, re, math
num_re=r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[Ee][+-]?\\d+)?'
all_pat=re.compile(':'+num_re)
int_pat=re.compile(r'\\)(?:["'^:(),;]*)?:('"'+num_re+')')
rows=[]
with zipfile.ZipFile('inputs/scogs_fungi.zip') as fh:
    for n in fh.namelist():
        if n.endswith('.treefile'):
            s=fh.read(n).decode().strip()
            vals=[float(x[1:]) for x in all_pat.findall(s)]
            ints=[float(x) for x in int_pat.findall(s)]
            if not vals: raise Exception(n)
            treeness=sum(ints)/sum(vals)
            rows.append((n.split('.faa')[0], treeness, sum(ints), sum(vals), len(vals), len(ints)))
rows.sort()
above=sum(t>0.06 for _,t,*rest in rows)
print('treefiles', len(rows), 'above', above, 'pct', above/len(rows)*100)
# Compare to iqtree displayed percentages max abs difference
pat=re.compile(r'Sum of internal branch lengths:\\s+([0-9.Ee+-]+)\\s+\\(([0-9.Ee+-]+)% of tree length\\)')
maxdiff=0; diffs=[]
with zipfile.ZipFile('inputs/scogs_fungi.zip') as fh:
    d={gid:t for gid,t,*_ in rows}
    for n in fh.namelist():
        if n.endswith('.iqtree'):
            text=fh.read(n).decode(errors='replace')
            m=pat.search(text)
            pct=float(m.group(2))/100
            gid=n.split('.faa')[0]
            diffs.append((abs(d[gid]-pct), gid, d[gid], pct))
print('maxdiff', max(diffs))
print('near exact')
for r in sorted(rows, key=lambda r: abs(r[1]-0.06))[:20]: print(r)
PY"
