/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
z='inputs/scogs_fungi.zip'
with ZipFile(z) as f:
 names=f.namelist()
 bases=sorted({n.split('.')[0] for n in names})
 trees={n.split('.')[0] for n in names if n.endswith('.treefile')}
 print('without_tree', *[b for b in bases if b not in trees])
 for b in bases:
  if b not in trees:
   print('\\n###',b,'alignment')
   print(f.read(b+'.faa.mafft.clipkit').decode()[:3000])
   print('### log tail')
   print('\\n'.join(f.read(b+'.faa.mafft.clipkit.log').decode(errors='replace').splitlines()[-35:]))
PY"
