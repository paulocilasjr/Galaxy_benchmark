from zipfile import ZipFile
import re, statistics
z='inputs/scogs_fungi.zip'
num=r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[Ee][+-]?\\d+)?'
rows=[]
with ZipFile(z) as f:
  for n in f.namelist():
    if not n.endswith('.treefile'): continue
    s=f.read(n).decode().strip()
    allbl=[float(x) for x in re.findall(r':('+num+r')',s)]
    intbl=[float(x) for x in re.findall(r'\\):('+num+r')',s)]
    tr=sum(intbl)/sum(allbl) if sum(allbl)>0 else float('nan')
    tips=len(re.findall(r'(?:\\(|,)(["'^():,;]+):'"'+num,s))
    rows.append((n,tips,tr,s,sum(intbl),sum(allbl)))
print('tree_count',len(rows))
print('tip_counts', {k:sum(r[1]==k for r in rows) for k in sorted(set(r[1] for r in rows))})
print('above_0.06',sum(r[2]>.06 for r in rows))
print('equal_0.06',sum(r[2]==.06 for r in rows))
print('below_or_equal',sum(r[2]<=.06 for r in rows))
print('percentage_successful_trees',100*sum(r[2]>.06 for r in rows)/len(rows))
print('percentage_all_255_groups',100*sum(r[2]>.06 for r in rows)/255)
print('range',min(r[2] for r in rows),max(r[2] for r in rows))
print('nearest_threshold')

print('three_tip_trees')
for r in rows:
  if r[1]==3: print(r[0].split('.')[0],r[2],r[3])
PY"
