/bin/bash -lc "command -v phykit || true
python - <<'PY'
try:
 import Bio
 print('biopython', Bio.__version__)
except Exception as e: print('no biopython',e)
try:
 import phykit
 print('phykit', getattr(phykit,'__version__','installed'))
except Exception as e: print('no phykit',e)
PY
python - <<'PY'
from zipfile import ZipFile
from io import StringIO
from Bio import Phylo
z='inputs/scogs_fungi.zip'; vals=[]
with ZipFile(z) as f:
 for n in f.namelist():
  if n.endswith('.treefile'):
   t=Phylo.read(StringIO(f.read(n).decode()),'newick')
   total=sum(c.branch_length or 0 for c in t.find_clades())
   internal=sum(c.branch_length or 0 for c in t.get_nonterminals())
   vals.append((n,internal/total if total else float('nan')))
print('biopython trees',len(vals),'above',sum(v>.06 for _,v in vals),'pct',sum(v>.06 for _,v in vals)/len(vals)*100)
PY"
