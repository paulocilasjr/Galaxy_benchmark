/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from io import StringIO
from Bio import Phylo
rows=[]
with ZipFile('inputs/scogs_fungi.zip') as f:
 for n in f.namelist():
  if n.endswith('.treefile'):
   t=Phylo.read(StringIO(f.read(n).decode()),'newick')
   total=sum((c.branch_length or 0.0) for c in t.find_clades())
   internal=sum((c.branch_length or 0.0) for c in t.get_nonterminals())
   rows.append((n,internal/total))
print('parsed',len(rows))
print('above',sum(v > 0.06 for _,v in rows))
print('percentage',format(100*sum(v > 0.06 for _,v in rows)/len(rows),'.15g'))
print('at_or_below',sum(v <= 0.06 for _,v in rows))
PY"
