cd /tmp/fungi_all

total_all=0
for f in *.faa; do
    case "$f" in
        *mafft*) continue ;;
    esac
    count=$(grep -c "^>" "$f")
    total_all=$((total_all + count))
done
echo "Total genes across all orthogroups: $total_all"

total_with_trees=0
for f in *.treefile; do
    base=$(basename "$f" .faa.mafft.clipkit.treefile)
    faa_file="${base}.faa"
    if [ -f "$faa_file" ]; then
        count=$(grep -c "^>" "$faa_file")
        total_with_trees=$((total_with_trees + count))
    fi
done
echo "Genes in orthogroups WITH trees: $total_with_trees"

total_without_trees=0
for f in *.faa; do
    case "$f" in
        *mafft*) continue ;;
    esac
    base=$(basename "$f" .faa)
    tree_file="${base}.faa.mafft.clipkit.treefile"
    if [ ! -f "$tree_file" ]; then
        count=$(grep -c "^>" "$f")
        total_without_trees=$((total_without_trees + count))
        echo "No tree: $base ($count genes)"
    fi
done

echo ""
echo "Genes in orthogroups WITHOUT trees: $total_without_trees"
