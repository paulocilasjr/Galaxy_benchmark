cd /tmp/fungi_all && total_genes=0
for f in *.faa; do
    if [[ "$f" != *"mafft"* ]]; then
        count=$(grep -c "^>" "$f")
        total_genes=$((total_genes + count))
    fi
done
echo "Total gene copies in all orthogroups: $total_genes"

# Count by number of sequences per orthogroup
for f in *.faa; do
    if [[ "$f" != *"mafft"* ]]; then
        grep -c "^>" "$f"
    fi
done | sort | uniq -c
