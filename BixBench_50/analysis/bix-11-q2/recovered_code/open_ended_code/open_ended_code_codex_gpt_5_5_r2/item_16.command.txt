/bin/bash -lc "python3 - <<'PY'
import zipfile,re,math
z=zipfile.ZipFile('inputs/scogs_fungi.zip')
# simple Newick branch parser: every :number branch length belongs to one branch; internal branches are lengths after a closing parenthesis.
rows=[]
for n in z.namelist():
    if n.endswith('.treefile'):
        s=z.read(n).decode().strip()
        nums=[float(x) for x in re.findall(r':([0-9]+(?:\\.[0-9]*)?(?:[eE][+-]?\\d+)?)', s)]
        intern=[float(x) for x in re.findall(r'\\)(?:["'^,():;]*)?:([0-9]+(?:'"\\.[0-9]*)?(?:[eE][+-]?\\d+)?)', s)]
        total=sum(nums)
        rows.append((n.split('.')[0], sum(intern), total, sum(intern)/total if total else math.nan, s))
print('treefiles',len(rows))
print('above',sum(r[3]>0.06 for r in rows))
print('percent',sum(r[3]>0.06 for r in rows)/len(rows)*100 if rows else None)
print('minmax', min(r[3] for r in rows), max(r[3] for r in rows))
print('first5', [(r[0], r[3]) for r in rows[:5]])
PY"
