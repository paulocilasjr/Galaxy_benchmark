import os

def parse_fasta(filepath):
    seqs = {}
    current_header = None
    current_seq = []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if current_header is not None:
                    seqs[current_header] = ''.join(current_seq)
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            seqs[current_header] = ''.join(current_seq)
    return seqs

def p_distance(seq1, seq2):
    n_diff = 0
    n_compared = 0
    for a, b in zip(seq1, seq2):
        if a == '-' and b == '-':
            continue
        n_compared += 1
        if a != b:
            n_diff += 1
    if n_compared == 0:
        return 0.0
    return n_diff / n_compared

def compute_patristic(tree_str):
    s = tree_str.strip().rstrip(';')
    nodes = []
    i = 0
    
    def parse_node():
        nonlocal i
        node = {'children': [], 'length': 0.0, 'name': None}
        node_id = len(nodes)
        nodes.append(node)
        
        if s[i] == '(':
            i += 1
            while True:
                child_id = parse_node()
                nodes[node_id]['children'].append(child_id)
                if s[i] == ',':
                    i += 1
                elif s[i] == ')':
                    i += 1
                    break
            if i < len(s) and s[i] == ':':
                i += 1
                j = i
                while j < len(s) and s[j] not in ',):;':
                    j += 1
                if j > i:
                    nodes[node_id]['length'] = float(s[i:j])
                    i = j
        else:
            j = i
            while j < len(s) and s[j] not in ':),;':
                j += 1
            nodes[node_id]['name'] = s[i:j]
            i = j
            if i < len(s) and s[i] == ':':
                i += 1
                j = i
                while j < len(s) and s[j] not in ',):;':
                    j += 1
                if j > i:
                    nodes[node_id]['length'] = float(s[i:j])
                    i = j
        
        return node_id
    
    root = parse_node()
    
    parent = {}
    dist_to_root = {}
    
    def dfs(node_id, d):
        dist_to_root[node_id] = d
        for child_id in nodes[node_id]['children']:
            parent[child_id] = node_id
            dfs(child_id, d + nodes[child_id]['length'])
    
    dfs(root, 0)
    parent[root] = None
    
    leaf_to_nid = {}
    for nid, node in enumerate(nodes):
        if node['name'] is not None:
            leaf_to_nid[node['name']] = nid
    
    leaf_names = sorted(leaf_to_nid.keys())
    
    leaf_ancestors = {}
    for name, nid in leaf_to_nid.items():
        ancestors = set()
        p = nid
        while p is not None:
            ancestors.add(p)
            p = parent.get(p)
        leaf_ancestors[name] = ancestors
    
    patristic = {}
    for i in range(len(leaf_names)):
        for j in range(i+1, len(leaf_names)):
            n1, n2 = leaf_names[i], leaf_names[j]
            nid1, nid2 = leaf_to_nid[n1], leaf_to_nid[n2]
            anc1, anc2 = leaf_ancestors[n1], leaf_ancestors[n2]
            common = anc1 & anc2
            lca = max(common, key=lambda x: dist_to_root.get(x, 0))
            d = dist_to_root[nid1] + dist_to_root[nid2] - 2 * dist_to_root[lca]
            patristic[(n1, n2)] = d
    
    return patristic, leaf_names

# Compute treeness using through-origin regression:
# treeness = 1 - min_β sum((tree_i - β * uncorrected_i)²) / sum(tree_i²)
# Optimal β = sum(tree_i * uncorrected_i) / sum(uncorrected_i²)
# treeness = 1 - sum((tree_i - β*uncorrected_i)²) / sum(tree_i²)

all_faa = set()
for f in os.listdir('.'):
    if f.endswith('.faa') and '.mafft' not in f:
        all_faa.add(f.replace('.faa', ''))

tree_genes = set()
for f in os.listdir('.'):
    if f.endswith('.treefile'):
        tree_genes.add(f.replace('.faa.mafft.clipkit.treefile', ''))

treeness_vals = {}

for gene_id in sorted(tree_genes):
    aln_file = f"{gene_id}.faa.mafft.clipkit"
    tree_file = f"{gene_id}.faa.mafft.clipkit.treefile"
    
    if not all(os.path.exists(f) for f in [aln_file, tree_file]):
        continue
    
    seqs = parse_fasta(aln_file)
    species_list = sorted(seqs.keys())
    
    # Uncorrected p-distances
    uncorr = []
    pairs = []
    for i in range(len(species_list)):
        for j in range(i+1, len(species_list)):
            d = p_distance(seqs[species_list[i]], seqs[species_list[j]])
            uncorr.append(d)
            pairs.append((species_list[i], species_list[j]))
    
    # Tree patristic distances
    tree_str = open(tree_file).read().strip()
    patristic, leaf_names = compute_patristic(tree_str)
    
    # Match
    tree_dist = []
    matched_uncorr = []
    for (sp1, sp2), ud in zip(pairs, uncorr):
        td = patristic.get((sp1, sp2))
        if td is None:
            td = patristic.get((sp2, sp1))
        if td is not None:
            tree_dist.append(td)
            matched_uncorr.append(ud)
    
    if len(tree_dist) < 3:
        continue
    
    n = len(tree_dist)
    sum_xy = sum(t*u for t, u in zip(tree_dist, matched_uncorr))
    sum_xx = sum(u*u for u in matched_uncorr)
    sum_yy = sum(t*t for t in tree_dist)
    
    if sum_xx > 0 and sum_yy > 0:
        beta = sum_xy / sum_xx
        ss_res = sum((t - beta*u)**2 for t, u in zip(tree_dist, matched_uncorr))
        treeness = 1.0 - ss_res / sum_yy
        treeness_vals[gene_id] = treeness

vals = list(treeness_vals.values())
mean_v = sum(vals) / len(vals)
sorted_v = sorted(vals)
median_v = sorted_v[len(sorted_v)//2]
min_v = min(vals)
max_v = max(vals)

print(f"Through-origin regression R² (tree ~ uncorrected):")
print(f"  n={len(vals)}, mean={mean_v:.4f}, median={median_v:.4f}, min={min_v:.4f}, max={max_v:.4f}")

above_006 = sum(1 for v in vals if v > 0.06)
