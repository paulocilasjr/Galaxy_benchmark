import os, re

os.chdir('/workspace/tmp')

# Extract treeness directly from IQ-TREE report files
# IQ-TREE reports: "Sum of internal branch lengths: X.XXXX (Y.YYYY% of tree length)"

treeness_from_iqtree = {}
for f in os.listdir('.'):
    if f.endswith('.iqtree'):
        gene_id = f.replace('.faa.mafft.clipkit.iqtree', '')
        content = open(f).read()
        # Search for "Sum of internal branch lengths:"
        match = re.search(r'Sum of internal branch lengths:\s+[\d.]+\s+\(([\d.]+)% of tree length\)', content)
        if match:
            pct = float(match.group(1))
            treeness_from_iqtree[gene_id] = pct / 100.0

# Now also check with my tree-parsing method
def compute_treeness_from_tree(tree_str):
    """Compute internal branch proportion from Newick tree."""
    s = tree_str.strip().rstrip(';')
    nodes = []
    i = 0
    
    def parse_node():
        nonlocal i
        node = {'children': [], 'length': 0.0, 'name': None}
        node_id = len(nodes)
        nodes.append(node)
        
        if s[i] == '(':
            i += 1
            while True:
                child_id = parse_node()
                nodes[node_id]['children'].append(child_id)
                if s[i] == ',':
                    i += 1
                elif s[i] == ')':
                    i += 1
                    break
            if i < len(s) and s[i] == ':':
                i += 1
                j = i
                while j < len(s) and s[j] not in ',):;':
                    j += 1
                if j > i:
                    nodes[node_id]['length'] = float(s[i:j])
                    i = j
        else:
            j = i
            while j < len(s) and s[j] not in ':),;':
                j += 1
            nodes[node_id]['name'] = s[i:j]
            i = j
            if i < len(s) and s[i] == ':':
                i += 1
                j = i
                while j < len(s) and s[j] not in ',):;':
                    j += 1
                if j > i:
                    nodes[node_id]['length'] = float(s[i:j])
                    i = j
        
        return node_id
    
    root = parse_node()
    
    total_length = 0.0
    internal_length = 0.0
    
    for nid, node in enumerate(nodes):
        if nid == root and node['name'] is None and node['length'] > 0:
            total_length += node['length']
            internal_length += node['length']
            continue
        if node['length'] > 0:
            total_length += node['length']
            if node['name'] is None:
                internal_length += node['length']
    
    if total_length == 0:
        return 0.0
    return internal_length / total_length

tree_genes = set()
for f in os.listdir('.'):
    if f.endswith('.treefile'):
        tree_genes.add(f.replace('.faa.mafft.clipkit.treefile', ''))

treeness_from_tree = {}
errors = []
for gene_id in sorted(tree_genes):
    tree_file = f"{gene_id}.faa.mafft.clipkit.treefile"
    tree_str = open(tree_file).read().strip()
    try:
        treeness_from_tree[gene_id] = compute_treeness_from_tree(tree_str)
    except Exception as e:
        errors.append((gene_id, str(e)))

# Compare IQ-TREE values with tree-parsed values
print(f"IQ-TREE treeness values: {len(treeness_from_iqtree)}")
print(f"Tree-parsed treeness values: {len(treeness_from_tree)}")
print(f"Errors in tree parsing: {len(errors)}")

if errors:
    for gene_id, err in errors[:5]:
        print(f"  {gene_id}: {err}")

# Compare values

print(f"\nCommon genes: {len(common)}")

for g in common:
    diff = abs(treeness_from_iqtree[g] - treeness_from_tree[g])
    diffs.append(diff)
if diffs:
    print(f"Max diff: {max(diffs):.6f}")
    print(f"Mean diff: {sum(diffs)/len(diffs):.6f}")
    big_diffs = [(g, treeness_from_iqtree[g], treeness_from_tree[g]) for g in common 
                 if abs(treeness_from_iqtree[g] - treeness_from_tree[g]) > 0.001]
    if big_diffs:
        print(f"Genes with diff > 0.001: {len(big_diffs)}")
        for g, iq, tp in big_diffs[:5]:
            print(f"  {g}: IQ-TREE={iq:.6f}, tree={tp:.6f}")
    else:
        print("All differences < 0.001")
