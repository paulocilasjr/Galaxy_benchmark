import os
import math

# Parse trimmed alignments and compute uncorrected p-distances
def parse_fasta(filepath):
    """Parse a FASTA file, return dict of {header: sequence}"""
    seqs = {}
    current_header = None
    current_seq = []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if current_header is not None:
                    seqs[current_header] = ''.join(current_seq)
                current_header = line[1:]  # Remove '>'
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            seqs[current_header] = ''.join(current_seq)
    return seqs

def p_distance(seq1, seq2):
    """Compute uncorrected p-distance between two protein sequences.
    Excludes positions where both sequences have gaps.
    Returns proportion of differing residues at aligned positions.
    """
    n_diff = 0
    n_compared = 0
    for a, b in zip(seq1, seq2):
        if a == '-' and b == '-':
            continue  # Skip positions where both are gapped
        n_compared += 1
        if a != b:
            n_diff += 1
    if n_compared == 0:
        return 0.0
    return n_diff / n_compared

def parse_mldist(filepath):
    """Parse IQ-TREE mldist file, return dict mapping species pairs to distance"""
    species_order = []
    dist_matrix = []
    with open(filepath) as f:
        n = int(f.readline().strip())
        for i in range(n):
            parts = f.readline().strip().split()
            species_order.append(parts[0])
            dist_matrix.append([float(x) for x in parts[1:]])
    # Extract upper triangle
    distances = {}
    for i in range(n):
        for j in range(i+1, n):
            distances[(species_order[i], species_order[j])] = dist_matrix[i][j]
    return distances, species_order

# Find all genes with alignments and trees
all_faa = set()
for f in os.listdir('.'):
    if f.endswith('.faa') and '.mafft' not in f:
        gene_id = f.replace('.faa', '')
        all_faa.add(gene_id)

tree_genes = set()
for f in os.listdir('.'):
    if f.endswith('.treefile'):
        gene_id = f.replace('.faa.mafft.clipkit.treefile', '')
        tree_genes.add(gene_id)

print(f"Total fungal genes: {len(all_faa)}")
print(f"Genes with trees: {len(tree_genes)}")
print(f"Genes without trees: {len(all_faa - tree_genes)}")

# Compute treeness for each gene with a tree
treeness_values = {}
for gene_id in sorted(tree_genes):
    aln_file = f"{gene_id}.faa.mafft.clipkit"
    mldist_file = f"{gene_id}.faa.mafft.clipkit.mldist"
    
    if not os.path.exists(aln_file) or not os.path.exists(mldist_file):
        continue
    
    # Parse alignment
    seqs = parse_fasta(aln_file)
    species_list = sorted(seqs.keys())
    
    # Compute uncorrected p-distances
    uncorrected_dists = []
    pairs = []
    for i in range(len(species_list)):
        for j in range(i+1, len(species_list)):
            d = p_distance(seqs[species_list[i]], seqs[species_list[j]])
            uncorrected_dists.append(d)
            pairs.append((species_list[i], species_list[j]))
    
    # Parse ML distances
    ml_dists_dict, ml_order = parse_mldist(mldist_file)
    
    # Match pairs - the mldist file uses full species names, alignment headers match
    patristic_dists = []
    for (sp1, sp2) in pairs:
        # Try both orderings
        d = ml_dists_dict.get((sp1, sp2))
        if d is None:
            d = ml_dists_dict.get((sp2, sp1))
        if d is not None:
            patristic_dists.append(d)
        else:
            print(f"WARNING: Could not match pair ({sp1}, {sp2}) for gene {gene_id}")
    
    if len(uncorrected_dists) != len(patristic_dists):
        print(f"WARNING: Mismatch in distances for {gene_id}: {len(uncorrected_dists)} vs {len(patristic_dists)}")
        continue
    
    # Treeness = 1 - sum((patristic - uncorrected)^2) / sum(uncorrected^2)
    ss_residual = sum((p - u) ** 2 for p, u in zip(patristic_dists, uncorrected_dists))
    ss_total = sum(u ** 2 for u in uncorrected_dists)
    
    if ss_total == 0:
        treeness = 0.0  # All sequences identical
    else:
        treeness = 1.0 - ss_residual / ss_total
    
    treeness_values[gene_id] = treeness

# Statistics
values = list(treeness_values.values())
mean_v = sum(values) / len(values)
