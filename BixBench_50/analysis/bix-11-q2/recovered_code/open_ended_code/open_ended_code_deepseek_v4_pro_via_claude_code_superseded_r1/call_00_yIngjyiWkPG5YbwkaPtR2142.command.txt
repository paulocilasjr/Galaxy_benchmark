import os

# Check a specific gene manually
gene_id = '1001705at2759'

# Parse alignment
seqs = {}
with open(f"{gene_id}.faa.mafft.clipkit") as f:
    current_header = None
    current_seq = []
    for line in f:
        line = line.strip()
        if line.startswith('>'):
            if current_header is not None:
                seqs[current_header] = ''.join(current_seq)
            current_header = line[1:]
            current_seq = []
        else:
            current_seq.append(line)
    if current_header is not None:
        seqs[current_header] = ''.join(current_seq)

species_list = sorted(seqs.keys())
print(f"Species: {species_list}")
print(f"Sequence lengths: {[(s, len(seqs[s])) for s in species_list]}")

# Compute uncorrected p-distances
print("\nUncorrected p-distances:")
for i in range(len(species_list)):
    for j in range(i+1, len(species_list)):
        s1, s2 = species_list[i], species_list[j]
        seq1, seq2 = seqs[s1], seqs[s2]
        n_diff = 0
        n_comp = 0
        for a, b in zip(seq1, seq2):
            if a == '-' and b == '-':
                continue
            n_comp += 1
            if a != b:
                n_diff += 1
        d = n_diff / n_comp if n_comp > 0 else 0
        print(f"  {s1} - {s2}: {d:.6f} (diffs={n_diff}, compared={n_comp})")

# Parse mldist
with open(f"{gene_id}.faa.mafft.clipkit.mldist") as f:
    n = int(f.readline().strip())
    mldist_order = []
    mldist_matrix = []
    for i in range(n):
        parts = f.readline().strip().split()
        mldist_order.append(parts[0])
        mldist_matrix.append([float(x) for x in parts[1:]])

print(f"\nML distances ({n}x{n}):")
print(f"  Order: {mldist_order}")
for i in range(n):
    for j in range(i+1, n):
        print(f"  {mldist_order[i]} - {mldist_order[j]}: {mldist_matrix[i][j]:.6f}")

# Compute treeness
uncorrected = []
patristic = []
for i in range(len(species_list)):
    for j in range(i+1, len(species_list)):
        s1, s2 = species_list[i], species_list[j]
        seq1, seq2 = seqs[s1], seqs[s2]
        n_diff = 0
        n_comp = 0
        for a, b in zip(seq1, seq2):
            if a == '-' and b == '-':
                continue
            n_comp += 1
            if a != b:
                n_diff += 1
        ud = n_diff / n_comp if n_comp > 0 else 0
        uncorrected.append(ud)
        
        # Find matching species in mldist
        for mi in range(n):
            for mj in range(mi+1, n):
                if (mldist_order[mi] == s1 and mldist_order[mj] == s2) or \
                   (mldist_order[mi] == s2 and mldist_order[mj] == s1):
                    patristic.append(mldist_matrix[mi][mj])

print(f"\nPaired distances:")
for i, (u, p) in enumerate(zip(uncorrected, patristic)):
    print(f"  Pair {i}: uncorrected={u:.6f}, patristic={p:.6f}")

ss_res = sum((p - u)**2 for p, u in zip(patristic, uncorrected))
ss_tot = sum(u**2 for u in uncorrected)
treeness = 1.0 - ss_res / ss_tot
print(f"\nSS_residual: {ss_res:.6f}")
print(f"SS_total: {ss_tot:.6f}")
print(f"Treeness: {treeness:.6f}")

# Also compute regular R^2 with intercept
n_pairs = len(uncorrected)
mean_p = sum(patristic) / n_pairs
mean_u = sum(uncorrected) / n_pairs
# Simple linear regression y = a*x + b
# a = sum((x_i - mean_x)*(y_i - mean_y)) / sum((x_i - mean_x)^2)
# b = mean_y - a*mean_x
ss_xx = sum((u - mean_u)**2 for u in uncorrected)
ss_xy = sum((u - mean_u)*(p - mean_p) for u, p in zip(uncorrected, patristic))
ss_yy = sum((p - mean_p)**2 for p in patristic)
if ss_xx > 0:
    a = ss_xy / ss_xx
    b = mean_p - a * mean_u
    ss_res_reg = sum((p - (a*u + b))**2 for u, p in zip(uncorrected, patristic))
    r_squared = 1.0 - ss_res_reg / ss_yy
    print(f"\nRegression with intercept:")
    print(f"  Slope a: {a:.6f}")
    print(f"  Intercept b: {b:.6f}")
    print(f"  R^2: {r_squared:.6f}")
else:
    print("  Cannot compute (zero variance in uncorrected distances)")
