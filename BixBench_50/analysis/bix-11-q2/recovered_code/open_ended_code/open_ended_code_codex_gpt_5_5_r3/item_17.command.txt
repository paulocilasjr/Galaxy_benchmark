/bin/bash -lc "python - <<'PY'
import zipfile, re
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    all_bases=sorted({n.split('.faa')[0] for n in z.namelist() if '.faa' in n})
    tree_bases={n.split('.faa')[0] for n in z.namelist() if n.endswith('.treefile')}
    for b in all_bases:
        if b not in tree_bases:
            log=b+'.faa.mafft.clipkit.log'
            print('---', b, '---')
            if log in z.namelist():
                txt=z.read(log).decode(errors='replace')
                lines=[ln for ln in txt.splitlines() if re.search('ERROR|failed|Failed|WARNING|Alignment|ERROR|NOTE|sequences|site|ERROR', ln)]
                print('\\n'.join(lines[-20:]))
            else:
                print('no log')
PY"
