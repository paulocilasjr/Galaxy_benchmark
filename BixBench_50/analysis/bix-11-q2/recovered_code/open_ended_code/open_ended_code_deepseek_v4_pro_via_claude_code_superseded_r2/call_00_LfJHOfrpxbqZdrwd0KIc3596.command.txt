python3 << 'EOF'
import re
import glob

iqtree_files = glob.glob('/tmp/work/fungi_scogs/*.iqtree')
total_genes = len(iqtree_files)

above = 0
equal = 0
below = 0

for f in iqtree_files:
    with open(f) as fh:
        content = fh.read()
    
    match = re.search(r'Sum of internal branch lengths: [\d.]+ \(([\d.]+)% of tree length\)', content)
    if match:
        pct = float(match.group(1))
        treeness = pct / 100.0
        
        if treeness > 0.06:
            above += 1
        elif treeness == 0.06:
            equal += 1
            print(f"EXACTLY 0.06: {f} -> {pct}%")
        else:
            below += 1

print(f"Total: {total_genes}")
print(f"Above 0.06: {above}")
print(f"Exactly 0.06: {equal}")
print(f"Below 0.06: {below}")
print(f"Percentage above 0.06: {above/total_genes*100:.10f}%")
print(f"Answer as percentage: {above/total_genes*100:.4f}%")

# More precise calculation
ratio = above / total_genes
percentage = ratio * 100
print(f"Precise: {above}/{total_genes} = {ratio:.10f} = {percentage:.10f}%")
EOF
