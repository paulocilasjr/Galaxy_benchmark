# Extract all treeness percentages and compute the answer
python3 << 'EOF'
import re
import glob

# Parse all fungal IQ-TREE files
iqtree_files = glob.glob('/tmp/work/fungi_scogs/*.iqtree')
total_genes = len(iqtree_files)
above_threshold = 0

treeness_values = []

for f in iqtree_files:
    with open(f) as fh:
        content = fh.read()
    
    # Extract "Sum of internal branch lengths: X.XXXX (Y.YYYY% of tree length)"
    match = re.search(r'Sum of internal branch lengths: [\d.]+ \(([\d.]+)% of tree length\)', content)
    if match:
        pct = float(match.group(1))
        treeness = pct / 100.0
        treeness_values.append(treeness)
        if treeness > 0.06:
            above_threshold += 1
    else:
        print(f"WARNING: No match in {f}")

print(f"Total fungal genes: {total_genes}")
print(f"Genes with treeness > 0.06: {above_threshold}")
print(f"Percentage: {above_threshold/total_genes*100:.2f}%")

# Show distribution
print(f"\nTreeness distribution:")
print(f"  Min: {min(treeness_values):.4f}")
print(f"  Max: {max(treeness_values):.4f}")
print(f"  Mean: {sum(treeness_values)/len(treeness_values):.4f}")
print(f"  Median: {sorted(treeness_values)[len(treeness_values)//2]:.4f}")

# Count below/above
below = sum(1 for t in treeness_values if t <= 0.06)
print(f"  <= 0.06: {below}")
print(f"  > 0.06: {above_threshold}")
EOF
