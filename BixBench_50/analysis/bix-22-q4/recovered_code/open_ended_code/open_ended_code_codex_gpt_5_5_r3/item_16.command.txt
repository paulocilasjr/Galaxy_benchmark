/bin/bash -lc "python - <<'PY'
# Recompute with a two-pass/centered formula to confirm the streamed result.
import csv, math
with open('inputs/Sample_annotated_Zenodo.csv', newline='') as f:
    cd14=[r['sample'] for r in csv.DictReader(f) if r['celltype']=='CD14']
meta={}
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
    for r in csv.DictReader(f):
        if r['gene_biotype']=='protein_coding':
            meta[r['Geneid']]=float(r['Length'])
vals=[]
with open('inputs/BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    reader=csv.reader(f)
    header=next(reader)
    pos={h:i for i,h in enumerate(header)}
    idx=[pos[s] for s in cd14]
    for row in reader:
        gene=row[0]
        if gene not in meta:
            continue
        total=math.fsum(float(row[i]) for i in idx)
        if total>=10:
            vals.append((meta[gene], total/len(idx)))
mx=math.fsum(x for x,y in vals)/len(vals)
my=math.fsum(y for x,y in vals)/len(vals)
num=math.fsum((x-mx)*(y-my) for x,y in vals)
dx=math.fsum((x-mx)*(x-mx) for x,y in vals)
dy=math.fsum((y-my)*(y-my) for x,y in vals)
r=num/math.sqrt(dx*dy)
print(len(vals), format(r,'.17g'))
PY"
