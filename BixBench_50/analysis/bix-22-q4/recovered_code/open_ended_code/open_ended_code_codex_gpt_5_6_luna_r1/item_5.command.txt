import importlib.util

python - <<'PY'
import csv
from collections import Counter
for fn in ['inputs/Sample_annotated_Zenodo.csv','inputs/GeneMetaInfo_Zenodo.csv']:
    with open(fn, newline='') as f:
        rd=csv.reader(f)
        header=next(rd)
        rows=list(rd)
    print(fn, 'header=',header, 'rows=',len(rows))
    if 'Sample' in fn:
        print('celltypes',Counter(row[3] for row in rows))
        print('duplicate sample',len(rows)-len(set(row[0] for row in rows)))
    else:
        print('duplicate first',len(rows)-len(set(row[0] for row in rows)), 'duplicate geneid',len(rows)-len(set(row[1] for row in rows)))
        print('biotypes',Counter(row[4] for row in rows).most_common(20))
PY"
