import csv, collections
p='inputs/Sample_annotated_Zenodo.csv'

    cnt=collections.Counter(row[c] for row in rows)
    if len(cnt)<=20:
        print(c, len(cnt), cnt)
    else:
        print(c, len(cnt), list(cnt.items())[:5])
