/bin/bash -lc "Rscript - <<'RS'
options(digits=17)
a <- read.csv('inputs/Sample_annotated_Zenodo.csv', check.names=FALSE, stringsAsFactors=FALSE)
g <- read.csv('inputs/GeneMetaInfo_Zenodo.csv', check.names=FALSE, stringsAsFactors=FALSE)
c <- read.csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', check.names=FALSE, stringsAsFactors=FALSE)
s <- a"'$sample[a$celltype == '"'CD14']
p <- g"'$gene_biotype == '"'protein_coding'
z <- as.matrix(c[match(g[[1]][p],c[[1]]),s]); storage.mode(z) <- 'double'
x <- as.double(g"'$Length[p]); y <- rowMeans(z); keep <- rowSums(z) >= 10
x <- x[keep]; y <- y[keep]
manual <- sum((x-mean(x))*(y-mean(y))) / sqrt(sum((x-mean(x))''^2)*sum((y-mean(y))''^2))
cat(format(manual,digits=17), '"'\\n')
cat(format(cor(x,y),digits=17), '\\n')
cat('boundary totals:',sum(rowSums(z)==9),sum(rowSums(z)==10),'retained',length(x),'\\n')
RS"
