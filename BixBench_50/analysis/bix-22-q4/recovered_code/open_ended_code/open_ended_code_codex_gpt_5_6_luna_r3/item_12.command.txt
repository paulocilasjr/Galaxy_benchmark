/bin/bash -lc "python - <<'PY'
import csv, math

with open('inputs/Sample_annotated_Zenodo.csv', newline='') as f:
    ann=list(csv.DictReader(f))
cd14=[r['sample'] for r in ann if r['celltype']=='CD14']
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
    meta={r['']: (int(r['Length']), r['gene_biotype']) for r in csv.DictReader(f)}

with open('inputs/BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    rd=csv.reader(f); h=next(rd)
    cd_idx=[h.index(s) for s in cd14]
    all_idx=list(range(1,len(h)))
    values=[]
    for row in rd:
        length, bio=meta[row[0]]
        if bio "'!= '"'protein_coding':
            continue
        cd_total=sum(int(row[i]) for i in cd_idx)
        all_total=sum(int(row[i]) for i in all_idx)
        values.append((length,cd_total,all_total,row[0]))

def corr_exact(pairs):
    n=len(pairs)
    sx=sum(x for x,y in pairs); sy=sum(y for x,y in pairs)
    sxx=sum(x*x for x,y in pairs); syy=sum(y*y for x,y in pairs); sxy=sum(x*y for x,y in pairs)
    num=n*sxy-sx*sy
    denx=n*sxx-sx*sx; deny=n*syy-sy*sy
    return num, denx, deny, num/math.sqrt(denx*deny)

cd_filtered=[(x,y,g) for x,y,t,g in values if y>=10]
cd_pairs=[(x,y) for x,y,g in cd_filtered]
num,dx,dy,r=corr_exact(cd_pairs)
print('CD14 filter n',len(cd_filtered),'exact-formula r %.18f'%r)
print('exact numerator',num,'denominator factors',dx,dy)
all_filtered=[(x,y,g) for x,y,t,g in values if t>=10]
all_pairs=[(x,y) for x,y,g in all_filtered]
print('all-sample filter n',len(all_filtered),'CD14 means among these r %.18f'%corr_exact(all_pairs)[3])
print('all-sample filter genes excluded by CD14 filter',len(all_filtered)-len(cd_filtered))
PY"
