/bin/bash -lc "python3 - <<'PY'
import pandas as pd
import numpy as np
from scipy.stats import pearsonr, spearmanr

samp = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')
cd14_samples = samp.loc[samp['celltype']=='CD14','sample'].tolist()
meta = pd.read_csv('inputs/GeneMetaInfo_Zenodo.csv', index_col=0)
counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
print('counts shape', counts.shape, counts.index.is_unique)
cd14 = counts[cd14_samples]
print('cd14 shape', cd14.shape, 'na', cd14.isna().sum().sum(), 'min', cd14.values.min(), 'max', cd14.values.max())
total = cd14.sum(axis=1)
mean = cd14.mean(axis=1)
length = meta['Length'].reindex(counts.index)
biotype = meta['gene_biotype'].reindex(counts.index)
mask = biotype.eq('protein_coding') & (total >= 10)
print('eligible', mask.sum())
print('total min max', total[mask].min(), total[mask].max())
print('length missing', length[mask].isna().sum())
r, p = pearsonr(length[mask].to_numpy(dtype='float64'), mean[mask].to_numpy(dtype='float64'))
print('Pearson r', repr(r))
print('p', p)
print('length stats', length[mask].describe().to_dict())
print('mean stats', mean[mask].describe().to_dict())
print('Pearson log2mean', pearsonr(length[mask].to_numpy(float), np.log2(mean[mask].to_numpy(float)))[0])
print('Spearman', spearmanr(length[mask], mean[mask]).statistic)
PY"
