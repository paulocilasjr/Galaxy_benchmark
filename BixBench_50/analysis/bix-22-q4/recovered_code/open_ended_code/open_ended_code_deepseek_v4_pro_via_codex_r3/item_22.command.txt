/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import pearsonr
samp=pd.read_csv('inputs/Sample_annotated_Zenodo.csv')
cd14_samples=samp.loc[samp['celltype']=='CD14','sample'].tolist()
meta=pd.read_csv('inputs/GeneMetaInfo_Zenodo.csv', index_col=0)
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
cd14=counts[cd14_samples]
total_all=counts.sum(axis=1)
total_cd14=cd14.sum(axis=1)
mean_cd14=cd14.mean(axis=1)
length=meta['Length'].reindex(counts.index)
bio=meta['gene_biotype'].reindex(counts.index)
for label, mask in [
 ('cd14_threshold', bio.eq('protein_coding') & (total_cd14>=10)),
 ('all_threshold', bio.eq('protein_coding') & (total_all>=10)),
 ('cd14_mean>=10', bio.eq('protein_coding') & (mean_cd14>=10)),
 ('all_threshold_mean_cd14', bio.eq('protein_coding') & (total_all>=10)),
]:
 r=pearsonr(length[mask].to_numpy(float), mean_cd14[mask].to_numpy(float))[0]
 print(label, mask.sum(), repr(r))
PY"
