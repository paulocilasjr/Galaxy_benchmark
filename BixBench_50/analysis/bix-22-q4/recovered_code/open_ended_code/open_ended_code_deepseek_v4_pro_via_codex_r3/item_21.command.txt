/bin/bash -lc "python3 - <<'PY'
import csv, math
# read sample annotations with csv
cd14=set()
with open('inputs/Sample_annotated_Zenodo.csv', newline='') as f:
    r=csv.DictReader(f)
    for row in r:
        if row['celltype']=='CD14':
            cd14.add(row['sample'])
print('n cd14', len(cd14))
# read meta
length={}; biotype={}
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
    r=csv.DictReader(f)
    for row in r:
        length[row['Geneid']]=float(row['Length'])
        biotype[row['Geneid']]=row['gene_biotype']
# stream count matrix, compute total and mean for protein_coding only
with open('inputs/BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    r=csv.DictReader(f)
    headers=r.fieldnames
    cols=[i for i,h in enumerate(headers) if h in cd14]
    print('cd14 cols found', len(cols))
    totals=[]; means=[]; lens=[]; genes=[]
    for row in r:
        gene=row[headers[0]]
        if biotype.get(gene)=='protein_coding':
            s=0.0
            for i in cols:
                val=row[headers[i]]
                s += float(val)
            if s>=10:
                n=len(cols)
                totals.append(s)
                means.append(s/n)
                lens.append(length[gene])
                genes.append(gene)
print('eligible', len(totals))
# Pearson pure
n=len(lens)
mx=sum(lens)/n; my=sum(means)/n
sxy=sum((lens[i]-mx)*(means[i]-my) for i in range(n))
sxx=sum((lens[i]-mx)**2 for i in range(n))
syy=sum((means[i]-my)**2 for i in range(n))
r=sxy/math.sqrt(sxx*syy)
print('r pure', repr(r))
print('meanx', mx, 'meany', my)
PY"
