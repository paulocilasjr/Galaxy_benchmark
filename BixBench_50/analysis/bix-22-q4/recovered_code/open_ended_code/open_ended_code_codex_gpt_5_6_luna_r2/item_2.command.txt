/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
from pathlib import Path
base=Path('/workspace/inputs')
for fn in ['BatchCorrectedReadCounts_Zenodo.csv','GeneMetaInfo_Zenodo.csv','Sample_annotated_Zenodo.csv']:
    p=base/fn
    with p.open(newline='') as f:
        rd=csv.reader(f)
        header=next(rd)
        n=0
        first=[]
        firstcol=[]
        for row in rd:
            n+=1
            if len(first)<3: first.append(row)
            firstcol.append(row[0] if row else '')
    print(fn, 'rows', n, 'columns', len(header), 'header_first', header[:8], 'header_last', header[-5:])
    c=Counter(firstcol)
    print('  first-column duplicates', sum(v>1 for v in c.values()), 'maxdup', max(c.values(), default=0), 'blank', c.get('',0))

p=base/'Sample_annotated_Zenodo.csv'
with p.open(newline='') as f:
    rows=list(csv.DictReader(f))
print('sample annotation rows',len(rows))
print('celltypes',Counter(r['celltype'] for r in rows))
print('sample duplicates',[(x,n) for x,n in Counter(r['sample'] for r in rows).items() if n>1][:20])
print('CD14 samples',sum(r['celltype']=='CD14' for r in rows))
print('CD14 examples',[r['sample'] for r in rows if r['celltype']=='CD14'][:20])
PY"
