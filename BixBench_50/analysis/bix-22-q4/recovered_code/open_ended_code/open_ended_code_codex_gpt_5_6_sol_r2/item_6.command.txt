/bin/bash -lc "python - <<'PY'
import csv, math
from pathlib import Path
p=Path('inputs')
with open(p/'Sample_annotated_Zenodo.csv', newline='') as f:
    cd14=[r['sample'] for r in csv.DictReader(f) if r['celltype']=='CD14']
with open(p/'GeneMetaInfo_Zenodo.csv', newline='') as f:
    rd=csv.DictReader(f); key=rd.fieldnames[0]
    meta={r[key]:(float(r['Length']),r['gene_biotype']) for r in rd}

def corr(xs,ys):
    n=len(xs); mx=math.fsum(xs)/n; my=math.fsum(ys)/n
    sxx=math.fsum((x-mx)**2 for x in xs)
    syy=math.fsum((y-my)**2 for y in ys)
    sxy=math.fsum((x-mx)*(y-my) for x,y in zip(xs,ys))
    return sxy/math.sqrt(sxx*syy)
with open(p/'BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    rd=csv.reader(f); hdr=next(rd); pos={s:i for i,s in enumerate(hdr)}; ix=[pos[s] for s in cd14]
    xs=[]; ys=[]; xcontains=[]; ycontains=[]; allsum=[]
    odd=[]
    for row in rd:
        g=row[0]; length,bio=meta[g]
        vals=[float(row[i]) for i in ix]
        total=math.fsum(vals)
        total_all=math.fsum(float(v) for v in row[1:])
        if bio=='protein_coding' and total>=10:
            xs.append(length); ys.append(total/len(ix)); allsum.append((length,total/len(ix),total_all))
        if 'protein_coding' in bio and total>=10:
            xcontains.append(length); ycontains.append(total/len(ix))
            if bio"'!='"'protein_coding': odd.append((g,length,bio.count('protein_coding'),total))
print('exact protein-coding eligible n',len(xs))
print('r CD14 total threshold, CD14 mean',repr(corr(xs,ys)),format(corr(xs,ys),'.17g'))
xa=[x for x,y,t in allsum if t>=10]; ya=[y for x,y,t in allsum if t>=10]
print('r all-sample total threshold, CD14 mean',len(xa),repr(corr(xa,ya)))
print('contains protein_coding n/r',len(xcontains),repr(corr(xcontains,ycontains)))
print('odd included',[(g,l,n,t) for g,l,n,t in odd])
print('mean expression min/max',min(ys),max(ys),'length min/max',min(xs),max(xs))
PY"
