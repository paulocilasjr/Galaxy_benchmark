/bin/bash -lc "python - <<'PY'
import csv, statistics
from collections import defaultdict
rows=[]
with open('inputs/Swarm_2.csv', newline='') as f:
    for r in csv.DictReader(f):
        r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity'])
        rows.append(r)
target_rows=[r for r in rows if r['StrainNumber']=='1']
target=(statistics.mean(r['Area'] for r in target_rows), statistics.mean(r['Circularity'] for r in target_rows))
by=defaultdict(list)
for r in rows:
    if r['StrainNumber']=='287_98': by[r['Ratio']].append(r)
best=None
for ratio,rs in by.items():
    mean=(statistics.mean(r['Area'] for r in rs), statistics.mean(r['Circularity'] for r in rs))
    diffs=(abs(mean[0]-target[0]), abs(mean[1]-target[1]))
    if best is None or diffs < best[2]:
        best=(ratio,mean,diffs)
print('target', target)
print('best', best)
PY"
