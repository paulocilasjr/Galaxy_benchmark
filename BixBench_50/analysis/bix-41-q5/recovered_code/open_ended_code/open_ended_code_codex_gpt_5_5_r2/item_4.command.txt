/bin/bash -lc "python - <<'PY'
import csv, math, statistics
from collections import defaultdict
path='inputs/Swarm_2.csv'
rows=[]
with open(path,newline='') as f:
    for r in csv.DictReader(f):
        r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity'])
        rows.append(r)

strain1=[r for r in rows if r['StrainNumber']=='1']
target={k:statistics.mean(r[k] for r in strain1) for k in ['Area','Circularity']}
print('target',target)
by=defaultdict(list)
for r in rows:
    if r['StrainNumber']=='287_98':
        by[r['Ratio']].append(r)
means={ratio:{k:statistics.mean(r[k] for r in rs) for k in ['Area','Circularity']} for ratio,rs in by.items()}
print('ratio, mean area, mean circ, area diff, circ diff, raw euclidean')
for ratio,m in means.items():
    ad=abs(m['Area']-target['Area']); cd=abs(m['Circularity']-target['Circularity'])
    print(ratio,m['Area'],m['Circularity'],ad,cd,math.hypot(ad,cd))
# normalized by target values, by range of mixed+target, z-score all group means?
print('\\nrelative euclidean')
for ratio,m in sorted(means.items(), key=lambda x: math.hypot((x[1]['Area']-target['Area'])/target['Area'], (x[1]['Circularity']-target['Circularity'])/target['Circularity'])):
    print(ratio, math.hypot((m['Area']-target['Area'])/target['Area'], (m['Circularity']-target['Circularity'])/target['Circularity']))
print('\\nrange normalized')
allm=list(means.values())+[target]
mins={k:min(m[k] for m in allm) for k in ['Area','Circularity']}
maxs={k:max(m[k] for m in allm) for k in ['Area','Circularity']}
for ratio,m in sorted(means.items(), key=lambda x: math.hypot((x[1]['Area']-target['Area'])/(maxs['Area']-mins['Area']), (x[1]['Circularity']-target['Circularity'])/(maxs['Circularity']-mins['Circularity']))):
    print(ratio, math.hypot((m['Area']-target['Area'])/(maxs['Area']-mins['Area']), (m['Circularity']-target['Circularity'])/(maxs['Circularity']-mins['Circularity'])))
PY"
