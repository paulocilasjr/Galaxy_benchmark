import csv
import math
from collections import defaultdict

# Read the CSV
rows = []
with open('inputs/Swarm_2.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        rows.append(row)

# Strain 1 data (pure strain 1)
strain1_areas = []
strain1_circs = []
for r in rows:
    if r['StrainNumber'] == '1':
        strain1_areas.append(float(r['Area']))
        strain1_circs.append(float(r['Circularity']))

strain1_mean_area = sum(strain1_areas) / len(strain1_areas)
strain1_mean_circ = sum(strain1_circs) / len(strain1_circs)
print(f"Strain 1 - Mean Area: {strain1_mean_area:.4f}, Mean Circularity: {strain1_mean_circ:.6f}")

# Group by Ratio for mixed cultures (StrainNumber = '287_98')
ratio_groups = defaultdict(lambda: {'areas': [], 'circs': []})
for r in rows:
    if r['StrainNumber'] == '287_98':
        ratio = r['Ratio']
        ratio_groups[ratio]['areas'].append(float(r['Area']))
        ratio_groups[ratio]['circs'].append(float(r['Circularity']))

# Compute group means
ratio_means = {}
for ratio, vals in ratio_groups.items():
    mean_area = sum(vals['areas']) / len(vals['areas'])
    mean_circ = sum(vals['circs']) / len(vals['circs'])
    ratio_means[ratio] = (mean_area, mean_circ)
    print(f"Ratio {ratio}: Mean Area = {mean_area:.4f}, Mean Circularity = {mean_circ:.6f}")

# Collect all group means to compute z-scores
all_mean_areas = [v[0] for v in ratio_means.values()]
all_mean_circs = [v[1] for v in ratio_means.values()]

# Mean and std of group means
mu_area = sum(all_mean_areas) / len(all_mean_areas)
mu_circ = sum(all_mean_circs) / len(all_mean_circs)

var_area = sum((a - mu_area)**2 for a in all_mean_areas) / len(all_mean_areas)
var_circ = sum((c - mu_circ)**2 for c in all_mean_circs) / len(all_mean_circs)

std_area = math.sqrt(var_area)
std_circ = math.sqrt(var_circ)

print(f"\nGroup means: mu_area={mu_area:.4f}, std_area={std_area:.4f}")
print(f"Group means: mu_circ={mu_circ:.6f}, std_circ={std_circ:.6f}")

# Z-score Strain 1
z1_area = (strain1_mean_area - mu_area) / std_area
z1_circ = (strain1_mean_circ - mu_circ) / std_circ
print(f"\nStrain 1 z-scores: area={z1_area:.6f}, circ={z1_circ:.6f}")

# Compute Euclidean distance in z-score space for each ratio
distances = {}
for ratio, (area, circ) in ratio_means.items():
    z_area = (area - mu_area) / std_area
    z_circ = (circ - mu_circ) / std_circ
    dist = math.sqrt((z_area - z1_area)**2 + (z_circ - z1_circ)**2)
    distances[ratio] = dist
    print(f"Ratio {ratio}: z_area={z_area:.6f}, z_circ={z_circ:.6f}, dist={dist:.6f}")

# Find closest
closest = min(distances.items(), key=lambda x: x[1])
print(f"\nClosest ratio: {closest[0]} with distance {closest[1]:.6f}")
