import pandas as pd
import numpy as np

df = pd.read_csv('inputs/Swarm_2.csv')

# Strain 1 alone
strain1 = df[df['StrainNumber'] == 1]
print("Strain 1 (alone):")
print(strain1[['Area', 'Circularity']])
mean_area_1 = strain1['Area'].mean()
mean_circ_1 = strain1['Circularity'].mean()
print(f"Mean Area: {mean_area_1}, Mean Circularity: {mean_circ_1}")
print()

# Mixed cultures (Strain 287 and 98 at various ratios)
mixed = df[df['StrainNumber'] == '287_98']
ratios = mixed['Ratio'].unique()
print("Ratios:", ratios)
print()

# Compute mean Area and Circularity for each ratio
results = []
for ratio in ratios:
    subset = mixed[mixed['Ratio'] == ratio]
    mean_area = subset['Area'].mean()
    mean_circ = subset['Circularity'].mean()
    results.append((ratio, mean_area, mean_circ))
    print(f"Ratio {ratio}: Mean Area = {mean_area:.2f}, Mean Circularity = {mean_circ:.4f}")

print()

# Compute Euclidean distance in standardized space
# Standardize using mean and std across all mixed samples
mixed_area_mean = mixed['Area'].mean()
mixed_area_std = mixed['Area'].std()
mixed_circ_mean = mixed['Circularity'].mean()
mixed_circ_std = mixed['Circularity'].std()

# Standardize strain 1
z_area_1 = (mean_area_1 - mixed_area_mean) / mixed_area_std
z_circ_1 = (mean_circ_1 - mixed_circ_mean) / mixed_circ_std

print(f"Standardized Strain 1: z_area = {z_area_1:.4f}, z_circ = {z_circ_1:.4f}")
print()

# Compute distances
best_ratio = None
best_dist = float('inf')
for ratio, mean_area, mean_circ in results:
    z_area = (mean_area - mixed_area_mean) / mixed_area_std
    z_circ = (mean_circ - mixed_circ_mean) / mixed_circ_std
    dist = np.sqrt((z_area - z_area_1)**2 + (z_circ - z_circ_1)**2)
    print(f"Ratio {ratio}: z_area = {z_area:.4f}, z_circ = {z_circ:.4f}, distance = {dist:.4f}")
    if dist < best_dist:
        best_dist = dist
        best_ratio = ratio

print()
print(f"Best matching ratio: {best_ratio} (distance = {best_dist:.4f})")
