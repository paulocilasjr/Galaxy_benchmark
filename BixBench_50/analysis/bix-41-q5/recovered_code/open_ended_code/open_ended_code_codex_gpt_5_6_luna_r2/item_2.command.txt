/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
p='inputs/Swarm_2.csv'
rows=list(csv.DictReader(open(p)))
for r in rows:
    r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity'])

def mean(rs,k): return sum(r[k] for r in rs)/len(rs)
ref=[r for r in rows if r['StrainNumber']=='1']
refa, refc=mean(ref,'Area'),mean(ref,'Circularity')
print('Strain1', refa, refc)
groups=defaultdict(list)
for r in rows:
    if r['StrainNumber']=='287_98': groups[r['Ratio']].append(r)
for ratio,rs in groups.items():
    a,c=mean(rs,'Area'),mean(rs,'Circularity')
    print(ratio, len(rs), a,c,'diff',abs(a-refa),abs(c-refc),'rel',abs(a-refa)/refa,abs(c-refc)/refc)
PY"
