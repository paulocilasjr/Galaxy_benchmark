/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='/workspace/inputs/Swarm_2.csv'
df=pd.read_csv(p)
print(df.dtypes)
print(df.groupby(['StrainNumber','Ratio'])[['Area','Circularity']].mean())
# distances raw, standardized across all groups, standardized across mixed groups, min-max scaling etc
strain1=df[df.StrainNumber==1][['Area','Circularity']].mean()
mixed=df[df.StrainNumber=='287_98']
print('strain1', strain1)
# scale using all data including strain1
for label, subset in [('all', df), ('mixed', mixed)]:
    means=subset.groupby('Ratio')[['Area','Circularity']].mean()
    X=means
    # zscore over ratio means
    zmu=X.mean(); zsd=X.std(ddof=0)
    z=(X-zmu)/zsd
    ztarget=(strain1-zmu)/zsd
    dist=(z-ztarget).pow(2).sum(1).pow(.5)
    print('\\n', label, 'zscore ratio means')
    print(pd.DataFrame({'dist':dist,'zarea':z.Area,'zcirc':z.Circularity}).sort_values('dist'))
    # zscore over individual observations
    obs=subset[['Area','Circularity']]
    omu=obs.mean(); osd=obs.std(ddof=0)
    zg=(X-omu)/osd; zt=(strain1-omu)/osd
    dist2=(zg-zt).pow(2).sum(1).pow(.5)
    print('zscore obs')
    print(pd.DataFrame({'dist':dist2,'zarea':zg.Area,'zcirc':zg.Circularity}).sort_values('dist'))
    # feature range minmax using individual obs
    omin=obs.min(); omax=obs.max()
    norm=(X-omin)/(omax-omin); nt=(strain1-omin)/(omax-omin)
    dist3=(norm-nt).pow(2).sum(1).pow(.5)
    print('minmax obs')
    print(pd.DataFrame({'dist':dist3,'a':norm.Area,'c':norm.Circularity}).sort_values('dist'))
PY"
