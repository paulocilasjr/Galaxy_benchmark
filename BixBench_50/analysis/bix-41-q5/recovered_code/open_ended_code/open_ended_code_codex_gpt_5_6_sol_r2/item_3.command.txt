/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
rows=list(csv.DictReader(open('inputs/Swarm_2.csv', newline='')))
g=defaultdict(list)
for r in rows:
    g[(r['StrainNumber'],r['Ratio'])].append((float(r['Area']),float(r['Circularity'])))
for k,v in g.items():
    print(k, sum(x for x,_ in v)/len(v), sum(y for _,y in v)/len(v))
PY"
