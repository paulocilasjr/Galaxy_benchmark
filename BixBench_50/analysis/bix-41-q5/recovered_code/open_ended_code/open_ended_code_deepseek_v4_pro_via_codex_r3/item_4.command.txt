/bin/bash -c "python3 - <<'PY'
import csv, math, statistics
p='/workspace/inputs/Swarm_2.csv'
rows=list(csv.DictReader(open(p)))
for r in rows:
    r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity'])
groups={}
for r in rows:
    key=(r['StrainNumber'],r['Ratio'])
    groups.setdefault(key,[]).append(r)
means={k:(statistics.mean(x['Area'] for x in v), statistics.mean(x['Circularity'] for x in v)) for k,v in groups.items()}
ref=means[('1','1:0')]
print('ref',ref)
for k,v in means.items():
    if k[0]=='287_98':
        print(k[1], v, 'diff',v[0]-ref[0], v[1]-ref[1], 'raw', math.dist(v,ref))
# all group means
allkeys=list(means)
allarr=[means[k] for k in allkeys]
# range norm over all group means
mins=[min(x[i] for x in allarr) for i in range(2)]
maxs=[max(x[i] for x in allarr) for i in range(2)]
rng=[maxs[i]-mins[i] for i in range(2)]
norm={k: tuple((means[k][i]-mins[i])/rng[i] for i in range(2)) for k in allkeys}
refn=norm[('1','1:0')]
print('\\nrange norm distances')
for k in allkeys:
    if k[0]=='287_98':
        print(k[1],norm[k], math.dist(norm[k],refn))
# z-score over group means
mu=[statistics.mean([x[i] for x in allarr]) for i in range(2)]
sd=[statistics.pstdev([x[i] for x in allarr]) for i in range(2)]
print('mu sd',mu,sd)
zs={k: tuple((means[k][i]-mu[i])/sd[i] for i in range(2)) for k in allkeys}
refz=zs[('1','1:0')]
print('\\nz distances')
for k in allkeys:
    if k[0]=='287_98':
        print(k[1],zs[k], math.dist(zs[k],refz))
PY"
