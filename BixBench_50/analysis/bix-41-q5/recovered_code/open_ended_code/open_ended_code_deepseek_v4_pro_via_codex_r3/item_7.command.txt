rows=[]

print('ref Round',refm)

    rs=[r for r in rows if r['StrainNumber']=='287_98' and r['Ratio']==ratio]
    gm=(statistics.mean(r['Area'] for r in rs), statistics.mean(r['Round'] for r in rs))
    print(ratio,gm,'raw',math.dist(gm,refm),'log',math.dist((math.log(gm[0]),gm[1]),(math.log(refm[0]),refm[1])))
PY"
