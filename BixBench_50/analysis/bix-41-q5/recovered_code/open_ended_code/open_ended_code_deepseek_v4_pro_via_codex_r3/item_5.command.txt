/bin/bash -c "python3 - <<'PY'
import csv, math, statistics
rows=[]
for r in csv.DictReader(open('/workspace/inputs/Swarm_2.csv')):
    r['Area']=float(r['Area']); r['Circularity']=float(r['Circularity']); rows.append(r)
ref=[r for r in rows if r['StrainNumber']=='1']
mu=[statistics.mean(r[x] for r in ref) for x in ['Area','Circularity']]
sd=[statistics.pstdev(r[x] for r in ref) for x in ['Area','Circularity']]
print('ref mean sd',mu,sd)
# each mixed group mean standardized by ref sd
groups={}
for r in rows:
    if r['StrainNumber']=='287_98':
        groups.setdefault(r['Ratio'],[]).append(r)
for ratio,rs in sorted(groups.items(), key=lambda kv: float(kv[0].split(':')[0])/float(kv[0].split(':')[1])):
    gm=[statistics.mean(r[x] for r in rs) for x in ['Area','Circularity']]
    d=math.dist([(gm[i]-mu[i])/sd[i] for i in range(2)],[0,0])
    print(ratio,gm,'z',[(gm[i]-mu[i])/sd[i] for i in range(2)],'d',d)
PY"
