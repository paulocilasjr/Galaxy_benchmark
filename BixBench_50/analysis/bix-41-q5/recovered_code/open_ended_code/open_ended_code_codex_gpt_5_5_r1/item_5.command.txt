import csv, statistics, math
from collections import defaultdict
rows=list(csv.DictReader(open('inputs/Swarm_2.csv', newline='')))
print('n',len(rows))
by=defaultdict(list)
for r in rows:
    by[(r['StrainNumber'],r['Ratio'])].append(r)
for k,rs in by.items():
    area=statistics.mean(float(r['Area']) for r in rs)
    circ=statistics.mean(float(r['Circularity']) for r in rs)
    print(k, len(rs), area, circ)

s1=by[('1','1:0')]
s1_area=statistics.mean(float(r['Area']) for r in s1)
s1_circ=statistics.mean(float(r['Circularity']) for r in s1)
print('Strain1',s1_area,s1_circ)
# Mixed ratios only
mixed=[]
for (strain,ratio),rs in by.items():
    if strain=='287_98':
        area=statistics.mean(float(r['Area']) for r in rs)
        circ=statistics.mean(float(r['Circularity']) for r in rs)
        mixed.append((ratio,area,circ))
# raw euclid, standardized by mixed+strain1 SD of condition means? by overall variable range? z by replicate sd? rank? print differences
areas=[float(r['Area']) for r in rows]
circs=[float(r['Circularity']) for r in rows]
mean_area=statistics.mean(areas); sd_area=statistics.stdev(areas)
mean_circ=statistics.mean(circs); sd_circ=statistics.stdev(circs)
print('overall sd',sd_area,sd_circ,'range',max(areas)-min(areas),max(circs)-min(circs))
for ratio,area,circ in mixed:
    raw=math.hypot(area-s1_area,circ-s1_circ)
    z=math.hypot((area-s1_area)/sd_area,(circ-s1_circ)/sd_circ)
    rng=math.hypot((area-s1_area)/(max(areas)-min(areas)),(circ-s1_circ)/(max(circs)-min(circs)))
    # absolute relative? maybe percent area plus abs circularity
    print(ratio, 'area',area,'circ',circ,'diffs', area-s1_area, circ-s1_circ, 'raw',raw,'z',z,'range',rng)
