/bin/bash -lc "python - <<'PY'
import csv, math
from collections import defaultdict
p='/workspace/inputs/Swarm_2.csv'
rows=list(csv.DictReader(open(p, newline='')))

def means(rs):
    return {k:sum(float(r[k]) for r in rs)/len(rs) for k in ('Area','Circularity')}
ref=means([r for r in rows if r['StrainNumber']=='1'])
groups=defaultdict(list)
for r in rows:
    if r['StrainNumber']=='287_98': groups[r['Ratio']].append(r)
print('reference', ref)
print('ratio,mean_area,mean_circularity,area_diff,circularity_diff,relative_euclidean')
for ratio, rs in groups.items():
    m=means(rs)
    da=m['Area']-ref['Area']; dc=m['Circularity']-ref['Circularity']
    rel=math.hypot(da/ref['Area'], dc/ref['Circularity'])
    print(ratio, m['Area'], m['Circularity'], da, dc, rel)
PY"
