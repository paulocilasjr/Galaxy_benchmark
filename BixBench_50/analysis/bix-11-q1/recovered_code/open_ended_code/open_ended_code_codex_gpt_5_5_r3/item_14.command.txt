/bin/bash -lc "python - <<'PY'
import zipfile, collections, re
for zname in ['scogs_animals.zip','scogs_fungi.zip']:
  with zipfile.ZipFile('/workspace/inputs/'+zname) as z:
    stems=collections.defaultdict(list)
    for n in z.namelist():
      stem=n.split('.faa')[0]
      stems[stem].append(n)
    counts=collections.Counter(tuple(sorted(x.split('.faa')[-1] or '.faa' for x in files)) for files in stems.values())
    print('\\n',zname,'orthogroups',len(stems))
    for pat,c in counts.most_common(10): print(c, pat)
PY"
