# Let me do a more careful analysis
import statistics

with open('/tmp/animal_treeness.txt') as f:
    animal_vals = [float(line.strip()) for line in f if line.strip()]

with open('/tmp/fungi_treeness.txt') as f:
    fungi_vals = [float(line.strip()) for line in f if line.strip()]

# Count zeros vs non-zeros
animal_zeros = sum(1 for v in animal_vals if v == 0.0)
animal_nonzero = sum(1 for v in animal_vals if v > 0.0)
print(f"Animal: {len(animal_vals)} trees, {animal_zeros} zero treeness, {animal_nonzero} non-zero")

fungi_zeros = sum(1 for v in fungi_vals if v == 0.0)
fungi_nonzero = sum(1 for v in fungi_vals if v > 0.0)
print(f"Fungi: {len(fungi_vals)} trees, {fungi_zeros} zero treeness, {fungi_nonzero} non-zero")

# Sort and show
animal_sorted = sorted(animal_vals)
fungi_sorted = sorted(fungi_vals)

print(f"\nAnimal median: {statistics.median(animal_vals)}")
print(f"Fungi median: {statistics.median(fungi_vals)}")
print(f"Animal sorted (all): {animal_sorted}")
print(f"\nFungi non-zero values count: {fungi_nonzero}")
print(f"Fungi zeros: {fungi_zeros}")
print(f"Fungi sorted first 10: {fungi_sorted[:10]}")

# What proportion of animal trees have treeness > 0?
print(f"\nAnimal non-zero treeness rate: {animal_nonzero}/{len(animal_vals)} = {animal_nonzero/len(animal_vals):.3f}")
print(f"Fungi non-zero treeness rate: {fungi_nonzero}/{len(fungi_vals)} = {fungi_nonzero/len(fungi_vals):.3f}")
EOF
