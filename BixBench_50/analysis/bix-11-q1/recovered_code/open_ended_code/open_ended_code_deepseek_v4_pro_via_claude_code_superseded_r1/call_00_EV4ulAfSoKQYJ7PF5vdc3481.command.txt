import statistics
import numpy as np

with open('/tmp/animal_treeness.txt') as f:
    animal_vals = [float(line.strip()) for line in f if line.strip()]

with open('/tmp/fungi_treeness.txt') as f:
    fungi_vals = [float(line.strip()) for line in f if line.strip()]

# Multiple ways to compute median
print("=== Animal ===")
print(f"statistics.median: {statistics.median(animal_vals)}")
print(f"numpy.median: {np.median(animal_vals)}")
print(f"statistics.median_low: {statistics.median_low(animal_vals)}")
print(f"statistics.median_high: {statistics.median_high(animal_vals)}")

print("\n=== Fungi ===")
print(f"statistics.median: {statistics.median(fungi_vals)}")
print(f"numpy.median: {np.median(fungi_vals)}")
print(f"statistics.median_low: {statistics.median_low(fungi_vals)}")
print(f"statistics.median_high: {statistics.median_high(fungi_vals)}")

# Difference
diff = statistics.median(fungi_vals) - statistics.median(animal_vals)
print(f"\n=== Difference (fungi - animals) ===")
print(f"{diff}")
print(f"{diff:.6f}")

# Let's also look at the values around the median for fungi
fungi_sorted = sorted(fungi_vals)
n = len(fungi_sorted)
print(f"\nFungi sorted, n={n}")
print(f"Index {n//2 - 1}: {fungi_sorted[n//2 - 1]}")
print(f"Index {n//2}: {fungi_sorted[n//2]}")
print(f"Index {n//2 + 1}: {fungi_sorted[n//2 + 1]}")

# How many zeros in fungi?
fungi_zero_indices = [i for i, v in enumerate(fungi_sorted) if v == 0.0]
print(f"Number of zeros in fungi: {len(fungi_zero_indices)}")
print(f"Last zero index: {fungi_zero_indices[-1] if fungi_zero_indices else 'N/A'}")
print(f"First non-zero index: {fungi_zero_indices[-1] + 1 if fungi_zero_indices else 0}")
print(f"First non-zero value: {fungi_sorted[fungi_zero_indices[-1] + 1] if fungi_zero_indices and fungi_zero_indices[-1] + 1 < n else 'N/A'}")
EOF
