python3 << 'EOF'
import statistics

# Read animal treeness values
with open('/tmp/animal_treeness.txt') as f:
    animal_vals = [float(line.strip()) for line in f if line.strip()]

# Read fungi treeness values
with open('/tmp/fungi_treeness.txt') as f:
    fungi_vals = [float(line.strip()) for line in f if line.strip()]

print(f"Animal count: {len(animal_vals)}")
print(f"Fungi count: {len(fungi_vals)}")

animal_median = statistics.median(animal_vals)
fungi_median = statistics.median(fungi_vals)

print(f"Animal median treeness: {animal_median}")
print(f"Fungi median treeness: {fungi_median}")
print(f"Difference (fungi - animals): {fungi_median - animal_median}")
print(f"Difference (animals - fungi): {animal_median - fungi_median}")

# Show distribution
print(f"\nAnimal stats: min={min(animal_vals)}, max={max(animal_vals)}, mean={statistics.mean(animal_vals):.4f}, median={animal_median}")
print(f"Fungi stats: min={min(fungi_vals)}, max={max(fungi_vals)}, mean={statistics.mean(fungi_vals):.4f}, median={fungi_median}")
EOF
