/bin/bash -lc "python3 - <<'PY'
from decimal import Decimal
import re, statistics, zipfile

def reported_values(path):
    vals=[]
    with zipfile.ZipFile(path) as z:
        for n in z.namelist():
            if n.endswith('.iqtree'):
                s=z.read(n).decode()
                m=re.search(r'Sum of internal branch lengths:\\s*[0-9.eE+-]+\\s*\\(([0-9.eE+-]+)% of tree length\\)',s)
                assert m, n
                vals.append(Decimal(m.group(1))/Decimal(100))
    return sorted(vals)

animals=reported_values('inputs/scogs_animals.zip')
fungi=reported_values('inputs/scogs_fungi.zip')
assert len(animals)==100 and len(fungi)==249
animal_median=animals[len(animals)//2-1:len(animals)//2+1]
fungal_median=fungi[len(fungi)//2]
assert animal_median==[Decimal(0),Decimal(0)]
assert fungal_median==Decimal('0.050140')
diff=fungal_median - Decimal(0)
print('animal_n',len(animals),'animal_median',Decimal(0))
print('fungi_n',len(fungi),'fungi_median',fungal_median)
print('difference',diff)
PY
