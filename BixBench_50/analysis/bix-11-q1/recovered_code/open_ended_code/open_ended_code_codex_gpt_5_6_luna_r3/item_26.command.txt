/bin/bash -lc "python3 - <<'PY'
import zipfile, re, collections

for archive in ('inputs/scogs_animals.zip','inputs/scogs_fungi.zip'):
    with zipfile.ZipFile(archive) as z:
        faas=sorted(n for n in z.namelist() if re.search(r'["'^.]+at2759'"\\.faa"'$'"',n))
        treeids={n.split('.')[0] for n in z.namelist() if n.endswith('.treefile')}
        dist=collections.Counter()
        missing=[]
        for n in faas:
            s=z.read(n).decode('utf8','replace')
            cnt=s.count('>')
            dist[(cnt, n.split('.')[0] in treeids)]+=1
            if n.split('.')[0] not in treeids:
                missing.append((n,cnt,len(s)))
        print('\\n',archive,'fasta',len(faas),'tree',len(treeids))
        print('records x tree status',sorted(dist.items()))
        print('missing first 20',missing[:20])
        print('missing record counts',collections.Counter(x[1] for x in missing))
PY"
