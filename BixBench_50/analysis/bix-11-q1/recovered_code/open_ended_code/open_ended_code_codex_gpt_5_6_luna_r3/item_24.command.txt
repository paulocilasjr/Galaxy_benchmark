/bin/bash -lc "python3 - <<'PY'
import re, zipfile, statistics, math

class Node:
    __slots__ = (\"children\",\"length\",\"name\")
    def __init__(self, children=None, length=None, name=None):
        self.children = children or []
        self.length = length
        self.name = name

def parse_newick(s):
    s=s.strip()
    i=0
    def skip():
        nonlocal i
        while i<len(s) and s[i].isspace(): i+=1
    def token(stop):
        nonlocal i
        start=i
        while i<len(s) and s[i] not in stop: i+=1
        return s[start:i].strip()
    def length():
        nonlocal i
        skip()
        if i<len(s) and s[i]==\":\":
            i+=1; skip()
            start=i
            while i<len(s) and s[i] not in \",();\": i+=1
            return float(s[start:i].strip())
        return None
    def subtree():
        nonlocal i
        skip()
        if s[i]==\"(\":
            i+=1
            kids=[subtree()]
            skip()
            while i<len(s) and s[i]==\",\":
                i+=1; kids.append(subtree()); skip()
            if i>=len(s) or s[i]"'!=")": raise ValueError((i,s[max(0,i-20):i+20]))
            i+=1; skip()
            # optional internal label, not expected
            if i<len(s) and s[i] not in ":,();":
                _=token(":,();")
            L=length()
            return Node(kids,L)
        else:
            name=token(":,();")
            L=length()
            return Node([],L,name)
    root=subtree()
    skip()
    if i<len(s) and s[i]==";": i+=1
    skip()
    if i != len(s): raise ValueError(("trailing",i,s[i:]))
    return root

def metrics(root):
    edges=[]
    internal=[]
    def walk(n, isroot=False):
        if not isroot:
            edges.append(n.length or 0.0)
            if n.children:
                internal.append(n.length or 0.0)
        for c in n.children: walk(c)
    walk(root,True)
    return sum(internal),sum(edges),len(edges),len(internal)

for archive in ("inputs/scogs_animals.zip", "inputs/scogs_fungi.zip"):
    with zipfile.ZipFile(archive) as z:
        names=sorted(n for n in z.namelist() if n.endswith(".treefile"))
        vals=[]; diffs=[]; rows=[]
        for n in names:
            root=parse_newick(z.read(n).decode())
            internal,total,ne,ni=metrics(root)
            vals.append(internal/total if total else math.nan)
            iq=n[:-len(".treefile")]+".iqtree"
            report=z.read(iq).decode()
            mt=float(re.search(r"Total tree length '"\\(sum of branch lengths\\):\\s*([0-9.eE+-]+)\", report).group(1))
            mi=float(re.search(r\"Sum of internal branch lengths:\\s*([0-9.eE+-]+)\", report).group(1))
            diffs.append((total-mt, internal-mi))
            rows.append((n, internal,total,internal/total,ne,ni))
        print(\"\\n\",archive,\"trees\",len(vals))
        print(\"median_exact_decimal\",format(statistics.median(vals),\".17g\"))
        print(\"median_exact_percent\",format(statistics.median(vals)*100,\".17g\"))
        print(\"median_iqtree_report_decimal\",format(statistics.median([float(re.search(r\"\\(([0-9.eE+-]+)% of tree length\\)\",z.read(n[:-len('.treefile')]+'.iqtree').decode()).group(1))/100 for n in names]),\".17g\"))
        print(\"range_exact\",min(vals),max(vals))
        print(\"taxa(edge count) distribution\", {k:sum(1 for r in rows if r[4]==k) for k in sorted(set(r[4] for r in rows))})
        print(\"internal edge distribution\", {k:sum(1 for r in rows if r[5]==k) for k in sorted(set(r[5] for r in rows))})
        print(\"max report-vs-newick rounded-total diffs\",max(abs(a) for d in diffs for a in d))
        print(\"median neighborhood\")
        for row in sorted(rows,key=lambda x:x[3])[max(0,len(rows)//2-3):len(rows)//2+3]:
            print(row)
PY"
