import zipfile, tempfile, os, re, statistics

def extract_treeness_from_iqtree(zip_path, label):
    results = []
    with zipfile.ZipFile(zip_path, 'r') as zf:
        iqtree_files = sorted([f for f in zf.namelist() if f.endswith('.iqtree')])
        with tempfile.TemporaryDirectory() as tmpdir:
            for tf in iqtree_files:
                try:
                    zf.extract(tf, tmpdir)
                    path = os.path.join(tmpdir, tf)
                    with open(path, 'r') as f:
                        content = f.read()
                    match = re.search(r'Sum of internal branch lengths: [\d.]+ \(([\d.]+)% of tree length\)', content)
                    if match:
                        pct = float(match.group(1))
                        treeness = pct / 100.0
                        results.append(treeness)
                except:
                    pass
    
    results.sort()
    med = statistics.median(results)
    print(f'{label}: N={len(results)}')
    print(f'  Median: {med:.10f}')
    print(f'  Q1: {statistics.median(results[:len(results)//2]):.10f}')
    print(f'  Q3: {statistics.median(results[(len(results)+1)//2:]):.10f}')
    # Show the values around the median
    if len(results) % 2 == 1:
        mid = len(results) // 2
        print(f'  Values around median: {results[mid-1]:.10f}, [{results[mid]:.10f}], {results[mid+1]:.10f}')
    return results, med

animal_vals, animal_med = extract_treeness_from_iqtree('/workspace/inputs/scogs_animals.zip', 'ANIMALS')
print()
fungi_vals, fungi_med = extract_treeness_from_iqtree('/workspace/inputs/scogs_fungi.zip', 'FUNGI')
print()
diff = fungi_med - animal_med
print(f'Difference (fungi - animals): {diff:.10f}')
