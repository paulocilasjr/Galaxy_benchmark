import dendropy, zipfile, tempfile, os, statistics

def compute_treeness(zip_path, label):
    results = []
    with zipfile.ZipFile(zip_path, 'r') as zf:
        tree_files = sorted([f for f in zf.namelist() if f.endswith('.treefile')])
        with tempfile.TemporaryDirectory() as tmpdir:
            for tf in tree_files:
                zf.extract(tf, tmpdir)
                tree_path = os.path.join(tmpdir, tf)
                tree = dendropy.Tree.get(path=tree_path, schema='newick', preserve_underscores=True)
                total_bl = 0.0; internal_bl = 0.0
                for edge in tree.postorder_edge_iter():
                    if edge.length is not None:
                        total_bl += edge.length
                        if not edge.head_node.is_leaf():
                            internal_bl += edge.length
                treeness = internal_bl / total_bl if total_bl > 0 else 0.0
                results.append(treeness)
    results.sort()
    med = statistics.median(results)
    print(f'{label}: N={len(results)}, median={med}')
    # Print every value for verification
    for i, v in enumerate(results):
        if i == len(results)//2 or i == len(results)//2 - 1:
            print(f'  [{i}] = {v}')
    return results, med

animal_vals, animal_med = compute_treeness('/workspace/inputs/scogs_animals.zip', 'ANIMALS')
print()
fungi_vals, fungi_med = compute_treeness('/workspace/inputs/scogs_fungi.zip', 'FUNGI')
print()
diff = fungi_med - animal_med
print(f'Difference: {diff}')
print(f'Difference (formatted): {diff:.10f}')
