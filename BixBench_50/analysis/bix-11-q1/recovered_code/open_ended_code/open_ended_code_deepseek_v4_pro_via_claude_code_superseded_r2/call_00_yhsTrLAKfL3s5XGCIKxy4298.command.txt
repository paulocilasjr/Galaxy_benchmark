import re, os

def extract_treeness_raw(filepath):
    with open(filepath, 'r') as f:
        content = f.read()
    total_match = re.search(r'Total tree length \(sum of branch lengths\):\s+([\d.]+)', content)
    internal_match = re.search(r'Sum of internal branch lengths:\s+([\d.]+)\s+\(', content)
    return (total_match.group(1) if total_match else None, 
            internal_match.group(1) if internal_match else None)

# Check for files without matches
for d, label in [('/workspace/scogs_animals', 'Animals'), ('/workspace/scogs_fungi', 'Fungi')]:
    missing = []
    for f in sorted(os.listdir(d)):
        if f.endswith('.iqtree'):
            total, internal = extract_treeness_raw(os.path.join(d, f))
            if total is None or internal is None:
                missing.append(f)
    print(f"{label}: {len(missing)} files with missing data")
    if missing:
        for m in missing[:5]:
            print(f"  {m}")

PYEOF
