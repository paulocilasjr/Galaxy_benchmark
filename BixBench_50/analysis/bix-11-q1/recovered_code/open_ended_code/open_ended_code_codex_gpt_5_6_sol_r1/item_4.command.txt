/bin/bash -lc 'for a in inputs/scogs_animals.zip inputs/scogs_fungi.zip; do
  echo "$a"
  unzip -Z1 "$a" | sed '"'s/.*\\.//' | sort | uniq -c | sort -nr
  echo treefiles
  unzip -Z1 \""'$a" | rg '"'\\.treefile"'$'"' | wc -l
  unzip -Z1 \""'$a" | rg '"'\\.treefile"'$'"' | head -3
  n="'$(unzip -Z1 "$a" | rg '"'\\.treefile"'$'"' | head -1); unzip -p \""'$a" "$n"
  echo iqtree treeness mentions
  unzip -Z1 "$a" | rg '"'\\.iqtree"'$'"' | head -1 | xargs -I{} unzip -p \""'$a" '"'{}' | rg -in 'tree length|treeness|sum of branch' | head -20
 done"
