/bin/bash -lc "python - <<'PY'
import zipfile,re,statistics
for group in ('animals','fungi'):
 vals=[]
 with zipfile.ZipFile(f'inputs/scogs_{group}.zip') as z:
  for n in z.namelist():
   if not n.endswith('.iqtree'): continue
   s=z.read(n).decode(errors='replace')
   m=re.search(r'Sum of internal branch lengths:\\s*[0-9.eE+-]+\\s*\\(([0-9.eE+-]+)% of tree length\\)',s)
   if not m: raise ValueError(n)
   vals.append(float(m.group(1))/100)
 print(group,len(vals),format(statistics.median(vals),'.17g'))
PY"
