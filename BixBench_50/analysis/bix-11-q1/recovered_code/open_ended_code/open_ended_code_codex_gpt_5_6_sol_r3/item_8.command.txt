/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from decimal import Decimal, getcontext
from statistics import median
import re
getcontext().prec=50
allrows={}
for group in ('animals','fungi'):
 rows=[]
 with ZipFile(f'inputs/scogs_{group}.zip') as z:
  for n in z.namelist():
   if not n.endswith('.treefile'): continue
   s=z.read(n).decode().strip()
   total=sum(Decimal(x) for x in re.findall(r':([+\\-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[Ee][+\\-]?\\d+)?)',s))
   intern=sum(Decimal(x) for x in re.findall(r'\\)(?:["'^:,();]*):([+'"\\-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[Ee][+\\-]?\\d+)?)',s))
   tips=s.count(':')-s.count('):')
   rows.append((n.split('.')[0],intern/total,tips,intern,total))
 allrows[group]=rows
 for scope, rr in [('all',rows),('four-tip',[r for r in rows if r[2]==4])]:
  vals=[r[1] for r in rr]
  print(group,scope,'n',len(vals),'zeros',sum(v==0 for v in vals),'median',median(vals))
 print('all tree lengths positive',all(r[4]>0 for r in rows),'all branches count structure',sorted(set((r[2],int(r[3]>0)) for r in rows)))
for scope in ('all','four'):
 a=[r[1] for r in allrows['animals'] if scope=='all' or r[2]==4]
 f=[r[1] for r in allrows['fungi'] if scope=='all' or r[2]==4]
 print(scope,'F-A',median(f)-median(a))
# common loci where both trees and four tips in both
ad={r[0]:r for r in allrows['animals']}; fd={r[0]:r for r in allrows['fungi']}
common4=sorted(k for k in ad.keys()&fd.keys() if ad[k][2]==fd[k][2]==4)
print('common4',len(common4), 'med A',median([ad[k][1] for k in common4]), 'med F',median([fd[k][1] for k in common4]), 'diff',median([fd[k][1] for k in common4])-median([ad[k][1] for k in common4]))
PY"
