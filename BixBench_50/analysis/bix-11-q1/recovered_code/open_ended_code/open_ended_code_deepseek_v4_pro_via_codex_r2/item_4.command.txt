/bin/bash -c "python3 - <<'PY'
import zipfile, collections, os
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    z=zipfile.ZipFile(zname)
    names=z.namelist()
    print('\\n===',zname,len(names))
    # Count by extension/suffix and patterns
    ext=collections.Counter()
    genes=collections.defaultdict(list)
    for n in names:
        base=n.split('/')[-1]
        # gene id prefix up to .faa
        if '.faa' in base:
            g=base.split('.faa')[0]
            genes[g].append(n)
        ext[base.split('.',1)[-1] if '.' in base else ''] +=1
    for e,c in ext.most_common(): print(e,c)
    print('genes',len(genes))
    # list sample gene files
    for g in list(genes)[:10]:
        print(g, genes[g])
    # list treefiles count
    tfs=[n for n in names if n.endswith('.treefile')]
    print('treefiles',len(tfs))
PY"
