/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
cp='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
lp='inputs/JD_Chromosome_Length.csv'
with open(cp, newline='') as f:
    rows=list(csv.DictReader(f))
site_counts=Counter((r['Chromosome'], r['StartPosition']) for r in rows)
unique_counts=Counter(ch for ch,pos in site_counts)
row_counts=Counter(r['Chromosome'] for r in rows)
with open(lp, newline='') as f:
    lengths={r['Chromosome'].lstrip('\\ufeff'): int(r['Length']) for r in csv.DictReader(f)}
print('chrom,unique_sites,rows,length,density_per_Mb')
for ch in sorted(unique_counts, key=lambda x: (not x.isdigit(), int(x) if x.isdigit() else x)):
    d=unique_counts[ch]/lengths[ch]*1e6
    print(f'{ch},{unique_counts[ch]},{row_counts[ch]},{lengths[ch]},{d:.9f}')
print('TOP unique', max(unique_counts, key=lambda ch: unique_counts[ch]/lengths[ch]), max(unique_counts[ch]/lengths[ch] for ch in unique_counts)*1e6)
print('TOP rows', max(row_counts, key=lambda ch: row_counts[ch]/lengths[ch]), max(row_counts[ch]/lengths[ch] for ch in row_counts)*1e6)
print('all rows site multiplicities', Counter(site_counts.values()))
PY"
