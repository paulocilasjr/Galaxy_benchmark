/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict

lengths = {}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        lengths[row['Chromosome']] = int(row['Length'])

sites = defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        chrom = row['Chromosome']
        # Pos is the CpG coordinate identifier; use its coordinate fields as a check.
        sites[chrom].add((row['Pos'], row['StartPosition'], row['EndPosition']))

ranked = []
for chrom, positions in sites.items():
    if chrom not in lengths:
        raise ValueError(f'missing length for chromosome {chrom}')
    count = len(positions)
    length = lengths[chrom]
    ranked.append((count / length, chrom, count, length))

for density, chrom, count, length in sorted(ranked, reverse=True):
    print(f'{chrom}\\t{count}\\t{length}\\t{density:.12g}\\t{density*1_000_000:.12g}')
PY"
