/bin/bash -lc "python3 - <<'PY'
import csv, collections
# load lengths JD
lengths={}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    r=csv.reader(f); next(r)
    for c,l in r:
        lengths[c.strip()]=int(l)
# parse CpG
sites=collections.defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='') as f:
    r=csv.reader(f); h=next(r)
    ci=h.index('Chromosome'); pi=h.index('Pos'); si=h.index('StartPosition')
    for row in r:
        sites[row[ci]].add((row[pi], row[si]))
print('chrom','unique_sites','length','density_per_bp','density_per_Mb','rows')
allrows=collections.Counter()
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='') as f:
    r=csv.reader(f); next(r); ci=1
    h=['Pos','SampleShared','Sample','SampleL','StartPosition','EndPosition','MethylationPercentage','CountMethylated','CountNonMethylated','Chromosome','Coverage','Dage']; ci=h.index('Chromosome')
    for row in r: allrows[row[ci]]+=1
res=[]
for c,st in sites.items():
    l=lengths.get(c)
    res.append((len(st)/l if l else None,c,len(st),l,len(st)/(l/1e6) if l else None,allrows[c]))
for x in sorted(res, reverse=True):
    print(x)
print('max by density per bp:', max(res, key=lambda x:x[0]))
PY"
