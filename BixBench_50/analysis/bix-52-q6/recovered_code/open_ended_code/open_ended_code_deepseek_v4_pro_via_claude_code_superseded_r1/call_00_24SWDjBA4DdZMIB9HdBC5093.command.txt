import csv
from collections import defaultdict

# Count unique CpG sites per chromosome from JD age-related CpG file
# Position format: Chromosome_Position (e.g., "1_10453")
# A unique CpG site is defined by its position identifier

cpgs_per_chrom = defaultdict(set)

with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        pos = row['Pos']
        # Pos format: Chromosome_Position
        chrom = row['Chromosome']
        cpgs_per_chrom[chrom].add(pos)

# Count unique sites
cpgs_count = {chrom: len(sites) for chrom, sites in cpgs_per_chrom.items()}
print("Unique CpG sites per chromosome:")
for chrom in sorted(cpgs_count.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else x)):
    print(f"  Chromosome {chrom}: {cpgs_count[chrom]}")

# Load chromosome lengths
chrom_lengths = {}
with open('inputs/JD_Chromosome_Length.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row['Chromosome'].strip()
        length = int(row['Length'].strip())
        if length > 0:
            chrom_lengths[chrom] = length

print("\nChromosome lengths:")
for chrom in sorted(chrom_lengths.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else x)):
    print(f"  Chromosome {chrom}: {chrom_lengths[chrom]}")

# Compute density: unique CpG sites per base pair
print("\nDensity (unique CpG sites per base pair):")
max_density = -1
max_chrom = None
for chrom in sorted(cpgs_count.keys(), key=lambda x: (x.isdigit(), int(x) if x.isdigit() else x)):
    if chrom in chrom_lengths:
        density = cpgs_count[chrom] / chrom_lengths[chrom]
        print(f"  Chromosome {chrom}: {density:.10f} ({cpgs_count[chrom]} sites / {chrom_lengths[chrom]} bp)")
        if density > max_density:
            max_density = density
            max_chrom = chrom

print(f"\nHighest density: Chromosome {max_chrom} with density {max_density:.10f}")
