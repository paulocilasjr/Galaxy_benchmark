import csv
from collections import defaultdict

# Read CpG data - count unique sites per chromosome
cpg_sites = defaultdict(set)
with open("inputs/JD_AgeRelated_CpG_noMT_Final.csv", newline="", encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row["Chromosome"]
        pos = row["Pos"]
        cpg_sites[chrom].add(pos)

# Read chromosome lengths
lengths = {}
with open("inputs/JD_Chromosome_Length.csv", newline="", encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row["Chromosome"].strip()
        if chrom and row["Length"].strip():
            lengths[chrom] = int(row["Length"])

print(f"Loaded {len(lengths)} chromosome lengths")
print(f"Found CpG sites on {len(cpg_sites)} chromosomes")

# Compute densities
max_density = -1
max_chrom = None

print(f"\n{'Chr':>4} | {'Sites':>6} | {'Length(bp)':>12} | {'Density':>14}")
print("-" * 44)

for chrom in sorted(cpg_sites.keys(), key=lambda x: (not x.isdigit(), x.isdigit() and int(x) or 0, x)):
    n_sites = len(cpg_sites[chrom])
    length = lengths.get(chrom)
    if length is None:
        continue
    density = n_sites / length
    print(f"{chrom:>4} | {n_sites:>6} | {length:>12} | {density:>14.10f}")
    if density > max_density:
        max_density = density
        max_chrom = chrom

print(f"\nHighest density: Chromosome {max_chrom}")
print(f"Density: {max_density:.10f} sites/bp ({max_density * 1e6:.4f} sites/Mb)")
