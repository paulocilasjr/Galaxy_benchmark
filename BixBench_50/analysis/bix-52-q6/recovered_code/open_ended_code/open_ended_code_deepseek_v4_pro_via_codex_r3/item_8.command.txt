/bin/bash -c "python3 - <<'PY'
import csv, collections
lengths={}
with open('/workspace/inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    for r in csv.DictReader(f):
        lengths[r['Chromosome']] = int(r['Length'])
print('length entries',len(lengths))
with open('/workspace/inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='') as f:
    rows=list(csv.DictReader(f))
# unique sites
sites=set((r['Chromosome'], int(r['StartPosition'])) for r in rows)
# count unique per chr
uniq=collections.Counter(ch for ch,pos in sites)
rowcnt=collections.Counter(r['Chromosome'] for r in rows)
# length denominator possibilities: provided all length. density sites / bp
print('\\nunique site density (sites/bp) top:')
for ch,n in sorted(uniq.items(), key=lambda kv: n/lengths[kv[0]], reverse=True):
    print(ch, n, lengths.get(ch), n/lengths[ch])
print('\\nrow density top:')
for ch,n in sorted(rowcnt.items(), key=lambda kv: n/lengths[kv[0]], reverse=True):
    print(ch, n, lengths.get(ch), n/lengths[ch])
PY"
