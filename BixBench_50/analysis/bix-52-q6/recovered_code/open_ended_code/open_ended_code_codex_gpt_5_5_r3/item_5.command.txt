/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict

age_path='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
len_path='inputs/JD_Chromosome_Length.csv'

lengths={}
with open(len_path, newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        lengths[row['Chromosome']] = int(row['Length'])

sites=defaultdict(set)
rows=defaultdict(int)
with open(age_path, newline='', encoding='utf-8-sig') as f:
    reader=csv.DictReader(f)
    for row in reader:
        chrom=row['Chromosome']
        # Pos encodes chrom_position; include chrom/start/end for robustness.
        site=(row['Pos'], row['StartPosition'], row['EndPosition'])
        sites[chrom].add(site)
        rows[chrom]+=1

result=[]
for chrom, ss in sites.items():
    if chrom not in lengths:
        raise SystemExit(f'missing length for {chrom}')
    density=len(ss)/lengths[chrom]
    result.append((density, chrom, len(ss), lengths[chrom], rows[chrom]))

for density, chrom, n, length, row_count in sorted(result, reverse=True)[:10]:
    print(chrom, n, length, density, 'rows', row_count)
print('winner', max(result)[1])
PY"
