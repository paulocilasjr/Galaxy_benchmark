import csv
from collections import defaultdict

# Load CpG sites - count unique positions per chromosome
cpg_by_chrom = defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', 'r', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        pos = row['Pos']
        chrom = row['Chromosome']
        cpg_by_chrom[chrom].add(pos)

# Load chromosome lengths
chrom_lengths = {}
with open('inputs/JD_Chromosome_Length.csv', 'r', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row['Chromosome']
        length = int(row['Length'])
        chrom_lengths[chrom] = length

# Calculate density and sort
results = []
for chrom, sites in cpg_by_chrom.items():
    count = len(sites)
    length = chrom_lengths.get(chrom)
    if length and length > 0:
        density = count / length
        results.append((chrom, count, length, density))

# Sort by density descending
results.sort(key=lambda x: x[3], reverse=True)

print(f"{'Chrom':>6} | {'Sites':>8} | {'Length':>12} | {'Density':>14}")
print("-" * 52)
for chrom, count, length, density in results:
    marker = " <-- HIGHEST" if density == results[0][3] else ""
    print(f"{chrom:>6} | {count:>8} | {length:>12} | {density:>14.10f}{marker}")

print(f"\nAnswer: Chromosome {results[0][0]}")
