import csv
from collections import defaultdict

# Load CpG sites - count unique positions per chromosome
cpg_by_chrom = defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        pos = row['Pos']
        chrom = row['Chromosome']
        cpg_by_chrom[chrom].add(pos)

# Load chromosome lengths
chrom_lengths = {}
with open('inputs/JD_Chromosome_Length.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        chrom = row['Chromosome']
        length = int(row['Length'])
        chrom_lengths[chrom] = length

# Calculate density and find max
print("Chromosome | CpG Sites | Length | Density")
print("-" * 55)
max_density = -1
max_chrom = None
results = []
for chrom in sorted(cpg_by_chrom.keys(), key=lambda x: int(x) if x.isdigit() else float('inf')):
    count = len(cpg_by_chrom[chrom])
    length = chrom_lengths.get(chrom, None)
    if length:
        density = count / length
        results.append((chrom, count, length, density))
        marker = ""
        if density > max_density:
            max_density = density
            max_chrom = chrom
            marker = " <--"

# Sort by density descending
results.sort(key=lambda x: x[3], reverse=True)
for chrom, count, length, density in results:
    print(f"{chrom:>4}     | {count:>9} | {length:>12} | {density:.10f}")

print(f"\nHighest density: Chromosome {max_chrom} with density {max_density:.10f}")
