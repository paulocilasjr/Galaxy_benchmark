/bin/bash -lc "mkdir -p /workspace/codex_output && python3 - <<'PY'
import csv, collections
name='JD'
with open('/workspace/inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
    sites_by_chrom=collections.defaultdict(set)
    for row in csv.DictReader(f):
        sites_by_chrom[row['Chromosome'].strip()].add(row['Pos'].strip())
with open('/workspace/inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    lengths={row['Chromosome'].strip(): int(row['Length']) for row in csv.DictReader(f)}
ranked=sorted(((len(s)/lengths[c], c, len(s), lengths[c]) for c,s in sites_by_chrom.items()), reverse=True)
for density,c,n,l in ranked[:10]:
    print(c, n, l, f'{density*1e6:.6f} per Mb')
print('ANSWER', ranked[0][1])
PY"
