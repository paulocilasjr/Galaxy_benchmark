/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict, Counter

def chrom_key(c):
    return (0, int(c)) if c.isdigit() else (1, c)

lengths={}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        lengths[row['Chromosome']] = int(row['Length'])

sites=defaultdict(set)
raw=Counter()
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='') as f:
    for row in csv.DictReader(f):
        chrom=row['Chromosome']
        # Pos is chromosome_start, but use explicit genomic coordinate tuple for clarity.
        sites[chrom].add((int(row['StartPosition']), int(row['EndPosition'])))
        raw[chrom]+=1

rank=[]
for chrom, site_set in sites.items():
    if chrom not in lengths:
        raise SystemExit(f'missing length for {chrom}')
    count=len(site_set)
    length=lengths[chrom]
    rank.append((count/length, chrom, count, length, raw[chrom]))
rank.sort(reverse=True)
print('top unique-site densities:')
for density, chrom, count, length, raw_rows in rank[:15]:
    print(chrom, count, length, density, density*1_000_000, 'perMb', 'raw_rows', raw_rows)
print('\\ntop raw-row densities (diagnostic only):')
rawrank=[]
for chrom, count in raw.items():
    rawrank.append((count/lengths[chrom], chrom, count, lengths[chrom]))
for density, chrom, count, length in sorted(rawrank, reverse=True)[:10]:
    print(chrom, count, length, density, density*1_000_000)
PY"
