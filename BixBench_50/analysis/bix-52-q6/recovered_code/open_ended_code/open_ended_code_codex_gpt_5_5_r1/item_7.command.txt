/bin/bash -lc "python3 - <<'PY'
import csv
from collections import Counter, defaultdict
path='inputs/JD_AgeRelated_CpG_noMT_Final.csv'
rows=0
pos=Counter()
chrom_pos=defaultdict(set)
chrom_rows=Counter()
with open(path,newline='') as f:
    r=csv.DictReader(f)
    print('headers', r.fieldnames)
    for row in r:
        rows+=1
        chrom=row['Chromosome']
        p=row['Pos']
        pos[p]+=1
        chrom_rows[chrom]+=1
        chrom_pos[chrom].add((row['StartPosition'], row['EndPosition']))
print('rows', rows)
print('unique_pos', len(pos))
print('n_chrom', len(chrom_pos))
print('top duplicate counts', Counter(pos.values()).most_common(10))
print('chrom_rows first', sorted(chrom_rows.items(), key=lambda x: (int(x[0]) if x[0].isdigit() else x[0]))[:10])
print('chrom_unique first', sorted(((c,len(s)) for c,s in chrom_pos.items()), key=lambda x: (int(x[0]) if x[0].isdigit() else x[0]))[:10])
PY"
