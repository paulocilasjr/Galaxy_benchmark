/bin/bash -lc "python - <<'PY'
import csv, os
from collections import defaultdict

sites_by_chrom=defaultdict(set)
with open('inputs/JD_AgeRelated_CpG_noMT_Final.csv', newline='', encoding='utf-8-sig') as f:
    reader=csv.DictReader(f)
    for row in reader:
        chrom=row['Chromosome']
        # Pos is already chrom_position; Start/End available too. Use full site ID to avoid collapsing across chromosomes.
        sites_by_chrom[chrom].add(row['Pos'])

lengths={}
with open('inputs/JD_Chromosome_Length.csv', newline='', encoding='utf-8-sig') as f:
    reader=csv.DictReader(f)
    for row in reader:
        lengths[row['Chromosome']]=int(row['Length'])

rows=[]
for chrom, sites in sites_by_chrom.items():
    if chrom not in lengths:
        raise SystemExit(f'Missing length for chromosome {chrom}')
    count=len(sites)
    length=lengths[chrom]
    density=count/length
    rows.append((density, count, length, chrom))
rows.sort(reverse=True)
for density,count,length,chrom in rows[:15]:
    print(chrom, count, length, density, density*1_000_000)

answer=rows[0][3]
os.makedirs('codex_output', exist_ok=True)
with open('codex_output/answer.txt','w',encoding='utf-8',newline='') as f:
    f.write(answer+'\\n')
print('ANSWER', answer)
PY
printf 'answer.txt: '; cat codex_output/answer.txt"
