from pyhmmer.plan7 import Pipeline, HMMFile
from pyhmmer.easel import Alphabet, SequenceFile, DigitalSequenceBlock

alphabet = Alphabet.amino()
lineage_dir = '/workspace/busco_downloads/lineages/eukaryota_odb10'
hmm_path = os.path.join(lineage_dir, 'hmms', '261328at2759.hmm')
proteome = '/workspace/inputs/Animals_Cele.faa'

pipeline = Pipeline(alphabet)

block = DigitalSequenceBlock(alphabet)
with SequenceFile(proteome, format='fasta', digital=True, alphabet=alphabet) as sf:
    for seq in sf:
        block.append(seq)
        if len(block) >= 5000:
            break

with HMMFile(hmm_path) as hf:
    hmm = hf.read()
    hits = list(pipeline.search_hmm(hmm, block))

for hit in hits[:3]:
    print(f'Hit: {hit.name}')
    for di in range(len(hit.domains)):
        d = hit.domains[di]
        print(f'  Domain {di} attributes:')
        for attr in dir(d):

                print(f'    {attr}')
        break  # just first domain
    break  # just first hit
