source venv/bin/activate && for proteome in Cele Ggal Ncra; do
    echo "Running BUSCO on $proteome..."
    case $proteome in
        Cele) input="/workspace/inputs/Animals_Cele.faa" ;;
        Ggal) input="/workspace/inputs/Animals_Ggal.faa" ;;
        Ncra) input="/workspace/inputs/Fungi_Ncra.faa" ;;
    esac
    busco -m proteins -i "$input" -l eukaryota_odb10 --download_path /workspace/busco_downloads/lineages --out "${proteome}_verify" --out_path /workspace/busco_real 2>&1 | grep -E "C:|Complete|Fragmented|Missing|Total|Results"
done
