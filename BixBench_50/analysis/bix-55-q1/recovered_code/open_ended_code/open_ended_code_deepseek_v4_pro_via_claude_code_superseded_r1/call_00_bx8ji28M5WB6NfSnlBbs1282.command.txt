# Extract single-copy complete BUSCO IDs from each proteome
for name in Cele Ggal Ncra Scer; do
    awk -F'\t' '!/^#/ && $2=="Complete" {print $1}' /workspace/busco_real/${name}_verify/run_eukaryota_odb10/full_table.tsv | sort > /tmp/${name}_sc.txt
    count=$(wc -l < /tmp/${name}_sc.txt)
    echo "$name: $count single-copy complete BUSCOs"
done

echo ""
echo "Intersection across all four:"
comm -12 /tmp/Cele_sc.txt /tmp/Ggal_sc.txt > /tmp/animal.txt
comm -12 /tmp/Ncra_sc.txt /tmp/Scer_sc.txt > /tmp/fungi.txt
comm -12 /tmp/animal.txt /tmp/fungi.txt > /tmp/all_four.txt
count=$(wc -l < /tmp/all_four.txt)
echo "Shared in ALL four: $count"

echo ""
echo "Also checking pairwise:"
for pair in "Cele Ggal" "Ncra Scer" "Cele Ncra" "Ggal Scer"; do
    a=$(echo $pair | cut -d' ' -f1)
    b=$(echo $pair | cut -d' ' -f2)
    overlap=$(comm -12 /tmp/${a}_sc.txt /tmp/${b}_sc.txt | wc -l)
    echo "  $a ∩ $b: $overlap"
done
