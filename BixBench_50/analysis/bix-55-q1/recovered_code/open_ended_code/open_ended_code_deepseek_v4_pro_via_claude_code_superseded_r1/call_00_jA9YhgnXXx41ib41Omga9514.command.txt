source venv/bin/activate && python3 -c "
from pyhmmer.plan7 import Pipeline, HMMFile
from pyhmmer.easel import Alphabet, SequenceFile
import os

alphabet = Alphabet.amino()
lineage_dir = '/workspace/busco_downloads/lineages/eukaryota_odb10'
hmm_path = os.path.join(lineage_dir, 'hmms', '261328at2759.hmm')
proteome = '/workspace/inputs/Animals_Cele.faa'

pipeline = Pipeline(alphabet)

# Load sequences
with SequenceFile(proteome, format='fasta') as sf:
    seqs = list(sf)

# Run hmmsearch
with HMMFile(hmm_path) as hf:
    hmm = hf.read()  # Read the HMM
    for hit in pipeline.search_hmm(hmm, seqs):
        print(f'Hit: {hit.name}, score={hit.score}, evalue={hit.evalue}')
        for d in hit.domains:
            print(f'  Domain: score={d.score}, hmm_from={d.hmm_from}, hmm_to={d.hmm_to}, env_from={d.env_from}, env_to={d.env_to}, ali_from={d.ali_from}, ali_to={d.ali_to}')
        break  # just first hit
    break  # just first HMM
" 2>&1
