import time
import pyhmmer

# Benchmark a single HMM search
with open('busco_downloads/lineages/eukaryota_odb10/hmms/1001705at2759.hmm', 'rb') as f:
    hmm_file = pyhmmer.plan7.HMMFile(f)
    hmm = next(hmm_file)

# Load sequences
alphabet = pyhmmer.easel.Alphabet.amino()
seq_block = pyhmmer.easel.DigitalSequenceBlock(alphabet)
with open('inputs/Animals_Ggal.faa', 'rb') as f:
    with pyhmmer.easel.SequenceFile(f, format='fasta') as sf:
        for seq in sf:
            seq_block.append(seq.digitize(alphabet))

print(f'Galgal: {len(seq_block)} sequences')

pipeline = pyhmmer.plan7.Pipeline(
    alphabet, Z=len(seq_block), domZ=len(seq_block),
    bit_cutoffs=None, incE=100.0, incdomE=100.0,
)

t0 = time.time()
top_hits = pipeline.search_hmm(hmm, seq_block)
count = sum(1 for hit in top_hits if hit.included)
elapsed = time.time() - t0
print(f'Search took {elapsed:.2f}s, found {count} included hits')
print(f'Estimated for 255*4=1020 searches: {elapsed * 1020 / 60:.0f} minutes')
