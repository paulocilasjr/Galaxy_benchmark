import pyhmmer

# Test a single HMM search with permissive settings
# Load HMM
with open('busco_downloads/lineages/eukaryota_odb10/hmms/1001705at2759.hmm', 'rb') as f:
    hmm_file = pyhmmer.plan7.HMMFile(f)
    hmm = next(hmm_file)
    print(f'HMM: {hmm.name}, length: {hmm.M}')
    print(f'TC cutoffs: trusted1={hmm.cutoffs.trusted1}, trusted2={hmm.cutoffs.trusted2}')
    print(f'GA cutoffs: gathering1={hmm.cutoffs.gathering1}, gathering2={hmm.cutoffs.gathering2}')
    print(f'NC cutoffs: noise1={hmm.cutoffs.noise1}, noise2={hmm.cutoffs.noise2}')

# Load sequences
sequences = []
with open('inputs/Animals_Cele.faa', 'rb') as f:
    with pyhmmer.easel.SequenceFile(f, format='fasta') as sf:
        for seq in sf:
            sequences.append(seq.digitize(pyhmmer.easel.Alphabet.amino()))

print(f'Loaded {len(sequences)} sequences')

# Run search with permissive thresholds
alphabet = pyhmmer.easel.Alphabet.amino()
pipeline = pyhmmer.plan7.Pipeline(
    alphabet, 
    Z=len(sequences), 
    domZ=len(sequences),
    bit_cutoffs=None,  # Don't use model bit thresholds
    incE=100.0,        # Report everything
    incdomE=100.0,
)

with pyhmmer.plan7.HMMPressed(hmm) as pressed:
    for top_hits in pipeline.search_hmm(pressed, sequences):
        for hit in top_hits:
            if hit.included:
                best_score = max(d.score for d in hit.domains)
                print(f'  Hit: {hit.name}, best_score={best_score:.1f}, included={hit.included}')
