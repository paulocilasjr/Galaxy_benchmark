import pyhmmer

# Test a single HMM search
# Load HMM
with open('busco_downloads/lineages/eukaryota_odb10/hmms/1001705at2759.hmm', 'rb') as f:
    hmm_file = pyhmmer.plan7.HMMFile(f)
    hmm = next(hmm_file)
    hmm.cutoffs.trusted1 = 0.0  # Ensure no trusted cutoff filter
    hmm.cutoffs.trusted2 = 0.0

# Load sequences
sequences = []
with open('inputs/Animals_Cele.faa', 'rb') as f:
    with pyhmmer.easel.SequenceFile(f, format='fasta') as sf:
        for seq in sf:
            sequences.append(seq.digitize(pyhmmer.easel.Alphabet.amino()))

print(f'Loaded {len(sequences)} sequences')

# Run search
alphabet = pyhmmer.easel.Alphabet.amino()
pipeline = pyhmmer.plan7.Pipeline(alphabet, Z=len(sequences), domZ=len(sequences))

with pyhmmer.plan7.HMMPressed(hmm) as pressed:
    hits = list(pipeline.search_hmm(pressed, sequences))
    
total_hits = sum(1 for top_hits in hits for _ in top_hits)
print(f'Found {total_hits} hits in first result')

# Check the TopHits structure
for i, top_hits in enumerate(hits[:2]):
    print(f'\\nTopHits {i}: {len(top_hits)} hits')
    for hit in top_hits:
        print(f'  Hit: {hit.name}, score={hit.score}, included={hit.included}')
        if hit.included:
            for dom in hit.domains:
                print(f'    Domain: score={dom.score}, bias={dom.bias}, i-evalue={dom.i_evalue}')
        if hasattr(hit, 'best_domain'):
            bd = hit.best_domain
            print(f'    Best domain: score={bd.score}')
