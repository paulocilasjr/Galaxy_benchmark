/bin/bash -lc "python - <<'PY'
import urllib.request, json
for ver in ['5.8.0','5.7.1','5.7.0','5.6.1','5.5.0','5.4.7']:
 url=f'https://api.anaconda.org/package/bioconda/busco/{ver}'
 try:
  with urllib.request.urlopen(url,timeout=20) as r: d=json.load(r)
  print('\\nVERSION',ver)
  for f in d.get('files',[]):
   print(f['basename'], f.get('attrs',{}).get('subdir'), f.get('download_url'))
   print('deps', f.get('dependencies'))
 except Exception as e: print(ver,e)
PY"
