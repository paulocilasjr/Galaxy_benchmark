/bin/bash -lc "set -e
mkdir -p /workspace/tools/pkgs
cd /workspace/tools/pkgs
curl -L --fail -o hmmer.tar.bz2 'https://conda.anaconda.org/bioconda/linux-aarch64/hmmer-3.4-hfe13ca0_4.tar.bz2'
python - <<'PY'
import bz2
src='/workspace/tools/pkgs/hmmer.tar.bz2'; out=src.replace('.tar.bz2','.tar')
open(out,'wb').write(bz2.decompress(open(src,'rb').read()))
PY
mkdir -p hmmer_pkg && tar --no-same-owner -xf hmmer.tar -C hmmer_pkg
find hmmer_pkg -maxdepth 3 -type f | head -50
ls -l hmmer_pkg/bin"
