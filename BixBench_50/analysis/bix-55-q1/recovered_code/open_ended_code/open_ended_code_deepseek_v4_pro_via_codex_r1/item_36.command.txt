/bin/bash -lc "python - <<'PY'
import urllib.request,json
for pkg in ['openmpi','gsl','libgcc-ng']:
 url=f'https://api.anaconda.org/package/conda-forge/{pkg}'
 try:
  with urllib.request.urlopen(url,timeout=20) as r:d=json.load(r)
  print('\\n',pkg,'latest',d.get('latest_version'))
  for f in d.get('files',[]):
   if f.get('attrs',{}).get('subdir')=='linux-aarch64':
    print(f.get('version'), f['basename'], f.get('download_url'))
 except Exception as e:print(pkg,e)
PY"
