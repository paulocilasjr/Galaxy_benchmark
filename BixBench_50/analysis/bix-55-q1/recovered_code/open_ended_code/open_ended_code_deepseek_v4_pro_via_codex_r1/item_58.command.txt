/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import defaultdict
base=Path('/workspace/busco_results')
sp=['Cele','Ggal','Ncra','Scer']
complete_sets={}; complete_single_sets={}; all_complete_sets={}
for s in sp:
    rows=[l.rstrip('\\n').split('\\t') for l in (base/s/'run_eukaryota_odb10'/'full_table.tsv').open() if not l.startswith('#') and l.strip()]
    statuses=defaultdict(list)
    for r in rows: statuses[r[1]].append(r[0])
    complete_single_sets[s]=set(statuses.get('Complete',[]))
    # Complete BUSCO groups (all complete copies) = Complete status or Duplicated status
    all_complete_sets[s]=set(statuses.get('Complete',[])) | set(statuses.get('Duplicated',[]))
print('intersection S only', len(set.intersection(*complete_single_sets.values())))
print('intersection C groups S+D', len(set.intersection(*all_complete_sets.values())))
# maybe \"present and complete\" requiring at least one complete copy per species = C groups intersection
for name,sets in [('single',complete_single_sets),('allC',all_complete_sets)]:
 print(name, {k:len(v) for k,v in sets.items()})
PY"
