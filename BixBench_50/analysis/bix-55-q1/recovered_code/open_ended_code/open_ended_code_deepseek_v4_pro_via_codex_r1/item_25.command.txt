/bin/bash -lc "set -e
mkdir -p /workspace/tools/pkgs
cd /workspace/tools/pkgs
curl -L --fail -o busco-5.8.0.tar.bz2 https://conda.anaconda.org/bioconda/noarch/busco-5.8.0-pyhdfd78af_0.tar.bz2
ls -lh busco-5.8.0.tar.bz2
python - <<'PY'
import bz2
src='/workspace/tools/pkgs/busco-5.8.0.tar.bz2'; out=src.replace('.tar.bz2','.tar')
b=bz2.decompress(open(src,'rb').read())
open(out,'wb').write(b)
print(len(b))
PY
mkdir -p /workspace/tools/pkgs/busco_pkg && tar --no-same-owner -xf /workspace/tools/pkgs/busco-5.8.0.tar -C /workspace/tools/pkgs/busco_pkg
find /workspace/tools/pkgs/busco_pkg -maxdepth 3 -type f | head -50"
