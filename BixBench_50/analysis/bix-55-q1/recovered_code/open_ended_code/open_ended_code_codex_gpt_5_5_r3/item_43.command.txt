/bin/bash -lc "python3 - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
alpha=easel.Alphabet.amino(); lineage=Path('tmp/busco_downloads/lineages/eukaryota_odb10')
hmm_path=next((lineage/'hmms').glob('*.hmm'))
with plan7.HMMFile(str(hmm_path)) as hf: h=next(hf)
with easel.SequenceFile('inputs/Fungi_Scer.faa', digital=True, alphabet=alpha) as sf: seqs=sf.read_block()
hits=next(hmmer.hmmsearch(h, seqs, cpus=1))
print(type(hits), [a for a in dir(hits) if not a.startswith('_')][:80])
print('len', len(hits))
if len(hits):
 hit=hits[0]
 print('hit attrs', [a for a in dir(hit) if not a.startswith('_')][:80])
 print('hit name', hit.name, type(hit.name), 'score', hit.score, 'e', hit.evalue, 'inc', hit.included, 'rep', hit.reported)
 print('domains attrs', [a for a in dir(hit.domains) if not a.startswith('_')][:80])
 d=hit.domains[0]
 print('domain attrs', [a for a in dir(d) if not a.startswith('_')][:100])
 print('dom fields', d.score, d.i_evalue, d.c_evalue, d.env_from, d.env_to, d.included, d.reported)
 print('ali attrs', [a for a in dir(d.alignment) if not a.startswith('_')][:100])
 print('ali', d.alignment.hmm_from, d.alignment.hmm_to, d.alignment.target_from, d.alignment.target_to)
PY"
