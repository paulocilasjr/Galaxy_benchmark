# Verify the intersection count more carefully
echo "Intersection count: $(wc -l < /tmp/intersection.txt)"

# Also verify by counting BUSCO IDs that are Complete in each species
echo ""
echo "Detailed verification:"
for sp in Cele Ggal Ncra Scer; do
  if [ "$sp" = "Scer" ]; then
    tsv="/tmp/busco_run_scer/busco_scer/run_eukaryota_odb10/full_table.tsv"
  else
    tsv="/tmp/busco_run_${sp}/busco_${sp}/run_eukaryota_odb10/full_table.tsv"
  fi
  # Count unique BUSCO IDs per status
  echo -n "$sp: "
  grep -v "^#" "$tsv" | awk -F'\t' '{print $1, $2}' | sort -u | awk '{print $2}' | sort | uniq -c | tr '\n' ' '
  echo ""
done
