# Extract the BUSCO IDs marked as "Complete" (single-copy) in each species
# Column 1 = BUSCO ID, Column 2 = Status
for sp in Cele Ggal Ncra Scer; do
  if [ "$sp" = "Scer" ]; then
    tsv="/tmp/busco_run_scer/busco_scer/run_eukaryota_odb10/full_table.tsv"
  else
    tsv="/tmp/busco_run_${sp}/busco_${sp}/run_eukaryota_odb10/full_table.tsv"
  fi
  # Extract BUSCO IDs with "Complete" status (single-copy)
  grep -v "^#" "$tsv" | awk -F'\t' '$2=="Complete" {print $1}' | sort > /tmp/${sp}_complete_buscos.txt
  echo "$sp: $(wc -l < /tmp/${sp}_complete_buscos.txt) Complete single-copy BUSCOs"
  # Also show unique count (should be same for Complete, different for Duplicated)
  grep -v "^#" "$tsv" | awk -F'\t' '$2=="Complete" {print $1}' | sort -u | wc -l | xargs echo "  Unique:"
done

# Find intersection
echo ""
echo "Intersection (single-copy in ALL four):"
comm -12 /tmp/Cele_complete_buscos.txt /tmp/Ggal_complete_buscos.txt > /tmp/tmp1.txt
comm -12 /tmp/Ncra_complete_buscos.txt /tmp/Scer_complete_buscos.txt > /tmp/tmp2.txt
comm -12 /tmp/tmp1.txt /tmp/tmp2.txt > /tmp/intersection.txt
wc -l < /tmp/intersection.txt
echo ""
echo "First 10 shared BUSCOs:"
head -10 /tmp/intersection.txt
