python3 -c "
import sys

# Read scores_cutoff
scores = {}
for line in open('/workspace/busco_downloads/lineages/eukaryota_odb10/scores_cutoff'):
    parts = line.strip().split('\t')
    scores[parts[0]] = float(parts[1])

# Read lengths_cutoff for odb10
lengths = {}
for line in open('/workspace/busco_downloads/lineages/eukaryota_odb10/lengths_cutoff'):
    parts = line.strip().split('\t')
    lengths[parts[0]] = (float(parts[2]), float(parts[3]))

# sample some lines from domtbl to understand format
print('Checking Cele domtbl...')
max_scores = {}
with open('/tmp/Cele_hmmscan.domtbl') as f:
    for i, line in enumerate(f):
        if line.startswith('#'):
            continue
        parts = line.strip().split()
        if len(parts) < 22:
            continue
        hmm = parts[0]  # target = HMM
        try:
            full_score = float(parts[7])
            dom_score = float(parts[13])
        except:
            continue
        if hmm not in max_scores or full_score > max_scores[hmm][0]:
            max_scores[hmm] = (full_score, dom_score, parts[3])  # query = protein

# Show top hits vs cutoff
passing = 0
total = 0
for hmm in sorted(max_scores.keys()):
    if hmm in scores:
        total += 1
        full_s, dom_s, prot = max_scores[hmm]
        if full_s >= scores[hmm]:
            passing += 1
            if passing <= 10:
                print(f'  PASS: {hmm}: score={full_s:.1f} (cutoff={scores[hmm]:.1f}), protein={prot}')

print(f'\n{passing}/{total} BUSCOs have hits passing score cutoff in Cele')
print(f'(using full sequence score at parts[7])')

# Also check using domain score
passing2 = 0
for hmm in sorted(max_scores.keys()):
    if hmm in scores:
        full_s, dom_s, prot = max_scores[hmm]
        if dom_s >= scores[hmm]:
            passing2 += 1
print(f'{passing2}/{total} BUSCOs passing score cutoff using domain score (parts[13])')
