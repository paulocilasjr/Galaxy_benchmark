export PATH="/tmp/pylib/hmmer/data/bin:$PATH"

# Run BUSCO on all 4 proteomes in parallel
for sp in Cele Ggal Ncra; do
  case $sp in
    Cele) fa="/workspace/inputs/Animals_Cele.faa" ;;
    Ggal) fa="/workspace/inputs/Animals_Ggal.faa" ;;
    Ncra) fa="/workspace/inputs/Fungi_Ncra.faa" ;;
  esac

  PYTHONPATH=/tmp/pylib python3 /tmp/pylib/busco/run_BUSCO.py \
    -i "$fa" \
    -l /workspace/busco_downloads/lineages/eukaryota_odb10 \
    -o "busco_${sp}" \
    -m prot \
    --cpu 4 \
    --out_path /tmp/busco_run_${sp} \
    --force \
    --download_path /tmp/busco_lineages \
    > /tmp/busco_${sp}.log 2>&1 &
  echo "Started BUSCO for $sp (PID: $!)"
done
echo "Waiting..."
wait
echo "All done"
