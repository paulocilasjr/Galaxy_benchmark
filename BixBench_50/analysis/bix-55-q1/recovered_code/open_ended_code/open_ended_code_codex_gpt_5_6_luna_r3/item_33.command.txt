/bin/bash -lc "mkdir -p /workspace/test_hmmer && unzip -q -o inputs/busco_downloads.zip 'lineages/eukaryota_odb10/hmms/1001705at2759.hmm' -d /workspace/test_hmmer && python - <<'PY'
from pyhmmer.easel import Alphabet, SequenceFile
from pyhmmer.plan7 import HMMFile
from pyhmmer.hmmer import hmmsearch
from pathlib import Path
alphabet=Alphabet.amino()
with HMMFile('/workspace/test_hmmer/lineages/eukaryota_odb10/hmms/1001705at2759.hmm') as hf:
    hmm=next(hf)
with SequenceFile('inputs/Fungi_Scer.faa', digital=True, alphabet=alphabet) as sf:
    seqs=list(sf)
print('hmm',hmm.name,hmm.M,'seqs',len(seqs))
hits=next(hmmsearch(hmm,seqs,cpus=1))
print('hits',len(hits), 'first', hits[0].name if len(hits) else None)
with open('/workspace/test_hmmer/test.domtblout','wb') as out:
    hits.write(out,format='domains')
PY
sed -n '1,20p' /workspace/test_hmmer/test.domtblout"
