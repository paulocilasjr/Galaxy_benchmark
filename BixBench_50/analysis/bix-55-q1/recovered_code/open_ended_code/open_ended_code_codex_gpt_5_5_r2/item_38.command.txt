/bin/bash -lc "python3 - <<'PY'
from pyhmmer import easel, plan7
from pathlib import Path
seq_path='inputs/Fungi_Scer.faa'
hmm_path=next(Path('tmp/busco/lineages/eukaryota_odb10/hmms').glob('*.hmm'))
abc=easel.Alphabet.amino()
with easel.SequenceFile(seq_path, digital=True, alphabet=abc) as sf:
    seqs=sf.read_block()
with plan7.HMMFile(str(hmm_path)) as hf:
    hmm=hf.read()
pipeline=plan7.Pipeline(abc)
hits=pipeline.search_hmm(hmm, seqs)
print('hmm', hmm.name.decode(), hmm.M, 'hits', len(hits))
for i,h in zip(range(3),hits):
    print('hit', i, h.name.decode(), 'score', h.score, 'pre_score', getattr(h,'pre_score',None), 'sum_score', getattr(h,'sum_score',None), 'e', h.evalue, 'inc', h.included, 'rep', h.reported, 'domains', len(h.domains))
    print([a for a in dir(h) if not a.startswith('_')][:80])
    for j,d in zip(range(2),h.domains):
        print(' domain', j, 'score', d.score, 'bias', d.bias, 'c_evalue', d.c_evalue, 'i_evalue', d.i_evalue, 'env', d.env_from, d.env_to, 'ali', d.alignment.hmm_from, d.alignment.hmm_to, d.alignment.target_from, d.alignment.target_to, 'inc', d.included)
        print('d attrs', [a for a in dir(d) if not a.startswith('_')][:80])
PY"
