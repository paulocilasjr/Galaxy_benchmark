from pyhmmer import easel, hmmer, plan7
from pathlib import Path
abc=easel.Alphabet.amino()

    seqs=sf.read_block()

    h=hf.read()
th=next(hmmer.hmmsearch([h], seqs, cpus=1))
print([a for a in dir(th) if not a.startswith('_')])
print('query?', getattr(th,'query',None))
print('attrs maybe', getattr(th,'query_name',None), getattr(th,'query_accession',None))
PY"
