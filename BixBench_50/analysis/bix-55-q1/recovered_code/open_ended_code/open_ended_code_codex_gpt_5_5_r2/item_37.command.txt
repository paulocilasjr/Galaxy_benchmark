/bin/bash -lc "python3 - <<'PY'
from pyhmmer import easel, plan7
from pathlib import Path
seq_path='inputs/Fungi_Scer.faa'
hmm_path=next(Path('tmp/busco/lineages/eukaryota_odb10/hmms').glob('*.hmm'))
abc=easel.Alphabet.amino()
with easel.SequenceFile(seq_path, digital=True, alphabet=abc) as sf:
    seqs=sf.read_block()
print('seqs', len(seqs), 'first', seqs[0].name)
with plan7.HMMFile(str(hmm_path)) as hf:
    hmm=hf.read()
print('hmm', hmm.name, hmm.M)
pipeline=plan7.Pipeline(abc)
hits=pipeline.search_hmm(hmm, seqs)
print('hits', len(hits))
for i,h in enumerate(hits[:3]):
    print('hit', i, h.name, h.score, h.evalue, h.included, h.reported, len(h.domains))
    for d in h.domains[:2]:
        print(' domain', d.score, d.i_evalue, d.env_from, d.env_to, d.alignment.hmm_from, d.alignment.hmm_to, d.alignment.target_from, d.alignment.target_to, d.included)
PY"
