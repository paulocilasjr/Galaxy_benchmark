tools/busco-env/bin/python - <<'PY'
from pathlib import Path
sets={}
for p in sorted(Path('busco_results').glob('*/run_eukaryota_odb10/full_table.tsv')):
    status={}
    with p.open() as fh:
        for line in fh:
            if line.startswith('#'): continue
            f=line.rstrip('\\n').split('\\t')
            status[f[0]]=f[1]
    sets[p.parents[1].name]={x for x,s in status.items() if s=='Complete'}
    print(p.parents[1].name, len(status), len(sets[p.parents[1].name]))
shared=set.intersection(*sets.values())
print('shared',len(shared))
print('IDs')
print('\\n'.join(sorted(shared)))
PY"
