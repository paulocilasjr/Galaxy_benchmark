/bin/bash -lc "PYTHONPATH=/workspace/tmp/pydeps python - <<'PY'
from pyhmmer import easel, plan7, hmmer
hmp='/workspace/tmp/busco_downloads/lineages/eukaryota_odb10/hmms/160593at2759.hmm'
fap='/workspace/inputs/Fungi_Scer.faa'
with plan7.HMMFile(hmp) as hf:
    hmm=hf.read()
print(hmm.name, hmm.accession, hmm.M, hmm.alphabet)
sf=easel.SequenceFile(fap, digital=True, alphabet=hmm.alphabet)
print(sf.name, sf.format, sf.alphabet)
res=next(hmmer.hmmsearch(hmm, sf, cpus=1))
print(type(res), [x for x in dir(res) if not x.startswith('_')])
print('len',len(res), 'included', getattr(res,'included',None))
if len(res):
  hit=res[0]
  print('hit type',type(hit), [x for x in dir(hit) if not x.startswith('_')])
  print('name',hit.name,'acc',hit.accession,'score',hit.score,'evalue',hit.evalue,'included',hit.included,'domains',len(hit.domains))
  d=hit.domains[0]
  print('dom type',type(d),[x for x in dir(d) if not x.startswith('_')])
  print('dom',d.score,d.i_evalue,d.c_evalue,d.hmm_from,d.hmm_to,d.ali_from,d.ali_to,d.env_from,d.env_to,d.bias)
PY"
