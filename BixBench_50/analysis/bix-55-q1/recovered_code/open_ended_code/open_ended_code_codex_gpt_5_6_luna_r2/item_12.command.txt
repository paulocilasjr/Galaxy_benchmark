/bin/bash -lc "command -v curl || true; command -v wget || true; command -v busybox || true; python - <<'PY'
import urllib.request
for u in ['https://raw.githubusercontent.com/ezlab/busco/master/README.md','https://api.github.com/repos/ezlab/busco/releases/latest']:
  try:
    with urllib.request.urlopen(u,timeout=15) as f:
      print(u, f.status, f.read(200).decode('utf-8','replace').replace('\\n',' '))
  except Exception as e:
    print(u, type(e).__name__, str(e))
PY"
