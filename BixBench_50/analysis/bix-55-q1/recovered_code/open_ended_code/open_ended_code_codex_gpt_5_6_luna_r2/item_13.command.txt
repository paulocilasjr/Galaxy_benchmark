/bin/bash -lc "python - <<'PY'
import urllib.request
for u in ['https://api.github.com/repos/ezlab/BUSCO','https://raw.githubusercontent.com/ezlab/BUSCO/master/README.md','https://raw.githubusercontent.com/ezlab/BUSCO/main/README.md']:
  try:
    with urllib.request.urlopen(u,timeout=15) as f:
      d=f.read(500).decode('utf-8','replace')
      print(u, f.status, d[:250].replace('\\n',' '))
  except Exception as e:
    print(u, type(e).__name__, str(e))
PY"
