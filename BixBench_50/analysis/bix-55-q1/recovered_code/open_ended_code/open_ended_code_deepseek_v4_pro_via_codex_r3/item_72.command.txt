/bin/bash -c "python - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
import io
p=Path('/workspace/busco_data/lineages/eukaryota_odb10/hmms/1001705at2759.hmm')
with plan7.HMMFile(str(p)) as hf: q=hf.read()
seq_path=Path('/workspace/inputs/Fungi_Scer.faa')
with easel.SequenceFile(str(seq_path), digital=True) as sf: seqs=list(sf)
hits=next(hmmer.hmmsearch(q, seqs, cpus=1))
buf=io.BytesIO()
hits.write(buf, format='domains', header=True)
print(buf.getvalue().decode()[:2000])
PY"
