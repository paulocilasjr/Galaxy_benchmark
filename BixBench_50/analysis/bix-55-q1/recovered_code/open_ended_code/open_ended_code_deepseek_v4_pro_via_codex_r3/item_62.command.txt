/bin/bash -c "python - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
import time
lineage=Path('/workspace/busco_data/lineages/eukaryota_odb10')
hmm_path=lineage/'hmms'/'261328at2759.hmm'
seq_path=Path('/workspace/inputs/Fungi_Scer.faa')
with plan7.HMMFile(str(hmm_path)) as hf:
    query=hf.read(); query=query.to_profile()
with easel.SequenceFile(str(seq_path), digital=True) as sf:
    seqs=list(sf)
for hits in hmmer.hmmsearch([query], seqs, cpus=8):
    q=hits.query
    print('query name acc len',q.name,q.accession,q.M, 'reported',hits.reported,'included',hits.included)
    for j,hit in enumerate(hits):
        if j>=5: break
        print(' HIT', hit.name, hit.accession, hit.length, hit.score, hit.bias, 'ndom', len(hit.domains), repr(hit.description))
        for d in hit.domains:
            al=d.alignment
            print('  DOM', d.score,d.bias,d.c_evalue,d.i_evalue,d.env_from,d.env_to,al.hmm_from,al.hmm_to,al.target_from,al.target_to,al.target_name,al.hmm_length)
    break
PY"
