/bin/bash -c "python - <<'PY'
from pyhmmer import easel, plan7, hmmer
from pathlib import Path
lineage=Path('/workspace/busco_data/lineages/eukaryota_odb10')
seq_path=Path('/workspace/inputs/Fungi_Scer.faa')
with easel.SequenceFile(str(seq_path), digital=True) as sf:
    seqs=list(sf)
print('seqs',len(seqs))
for hfpath in sorted((lineage/'hmms').glob('*.hmm'))[:20]:
    with plan7.HMMFile(str(hfpath)) as hf:
        q=hf.read()
    hits=next(hmmer.hmmsearch(q, seqs, cpus=1))
    arr=list(hits)
    best=arr[0] if arr else None
    print(hfpath.stem, 'hits', len(arr), 'best', best.name if best else None, best.score if best else None)
PY"
