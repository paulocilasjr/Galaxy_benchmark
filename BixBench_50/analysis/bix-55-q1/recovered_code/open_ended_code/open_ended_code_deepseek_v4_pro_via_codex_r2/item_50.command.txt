/bin/bash -c "python - <<'PY'
import urllib.request, tarfile, os, shutil
out='/workspace/busco-5.4.7.tar.gz'
if not os.path.exists(out):
    urllib.request.urlretrieve('https://gitlab.com/ezlab/busco/-/archive/5.4.7/busco-5.4.7.tar.gz',out)
os.makedirs('/workspace/busco_547_full',exist_ok=True)
with tarfile.open(out,'r:gz') as t:
    # Extract all files, but not preserve ownership
    for m in t.getmembers():
        t.extract(m, path='/workspace/busco_547_full')
print('extracted')
for root,dirs,files in os.walk('/workspace/busco_547_full'):
    if 'setup.py' in files:
        print(root)
        print(open(os.path.join(root,'setup.py')).read())
        break
PY"
