/bin/bash -c "python - <<'PY'
from pathlib import Path
runs=['Scer','Ncra','Cele','Ggal']
for prefix in ['/workspace/busco_runs','/workspace/busco_547_runs']:
    sets={}
    for name in runs:
        if prefix.endswith('busco_runs'):
            p=Path(prefix)/name/'run_eukaryota_odb10/full_table.tsv'
        else:
            p=Path(prefix)/f'{name}_547/run_eukaryota_odb10/full_table.tsv'
        s=set(); c=set()
        for line in p.read_text().splitlines():
            if line.startswith('#'): continue
            cols=line.split('\\t')
            if len(cols)>1:
                if cols[1]=='Complete': s.add(cols[0])
                if cols[1] in ('Complete','Duplicated'): c.add(cols[0])
        sets[name]=(s,c)
    inter_s=set.intersection(*[v[0] for v in sets.values()])
    inter_c=set.intersection(*[v[1] for v in sets.values()])
    print(prefix, 'Complete',len(inter_s),'CompleteOrDup',len(inter_c))
    print({n: (len(s),len(c)) for n,(s,c) in sets.items()})
PY"
