/bin/bash -c "mkdir -p /workspace/codex_output && python - <<'PY'
from pathlib import Path
# recompute exact and print a compact audit
runs=['Scer','Ncra','Cele','Ggal']
sets={}
for name in runs:
    p=Path('/workspace/busco_runs')/name/'run_eukaryota_odb10'/'full_table.tsv'
    s=set()
    with p.open() as f:
        for line in f:
            if line.startswith('#'):
                continue
            cols=line.rstrip('\\n').split('\\t')
            if len(cols)>1 and cols[1]=='Complete':
                s.add(cols[0])
    sets[name]=s
    print(name, len(s))
inter=set.intersection(*sets.values())
print('intersection', len(inter))
Path('/workspace/codex_output/answer.txt').write_text(str(len(inter))+'\\n')
PY
cat /workspace/codex_output/answer.txt"
