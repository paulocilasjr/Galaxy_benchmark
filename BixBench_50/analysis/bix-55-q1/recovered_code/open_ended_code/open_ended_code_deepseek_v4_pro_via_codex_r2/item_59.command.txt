/bin/bash -c "python - <<'PY'
import urllib.request, tarfile, os
for ver in ['5.7.1','5.8.3']:
 out=f'/workspace/busco-{ver}.tar.gz'
 url=f'https://gitlab.com/ezlab/busco/-/archive/{ver}/busco-{ver}.tar.gz'
 print('download',ver)
 urllib.request.urlretrieve(url,out)
 print(os.path.getsize(out))
 path=f'/workspace/busco_{ver.replace(\".\",\"_\")}_full'
 os.makedirs(path,exist_ok=True)
 with tarfile.open(out,'r:gz') as t:
  for m in t.getmembers(): t.extract(m,path=path)
PY
find /workspace/busco_5_8_3_full -maxdepth 2 -type d; find /workspace/busco_5_7_1_full -maxdepth 2 -type d | head"
