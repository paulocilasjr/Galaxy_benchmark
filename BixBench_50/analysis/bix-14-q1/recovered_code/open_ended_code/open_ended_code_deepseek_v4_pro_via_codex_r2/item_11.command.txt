/bin/bash -lc "python - <<'PY'
import openpyxl, glob, re, os, collections, math
carrier_ids={'185','184','285','287','353','354','556','557','381','396','397','489','490','499','500','503','504','613','614'}
files=glob.glob('/workspace/inputs/CHIP_DP10_GQ20_PASS/*.xlsx')
# load headers/rows for carrier files
all_rows=[]
headers=None
for f in files:
    base=os.path.basename(f)
    m=re.match(r'230209_Exome_GRCh38_CHIP_(\\d+)-',base)
    if not m or m.group(1) not in carrier_ids: continue
    wb=openpyxl.load_workbook(f, data_only=True)
    ws=wb.active
    rows=ws.iter_rows(values_only=True)
    h=next(rows)
    if headers is None: headers=h
    for r in rows:
        if r[0] is None: continue
        all_rows.append(r)
print('carrier rows',len(all_rows))
print('headers')
for i,h in enumerate(headers): print(i,h)
# distributions of relevant columns
for idx in [3,4,9,13,14,19,25,30,31]:
    print('\\ncol',idx, headers[idx])
    c=collections.Counter()
    for r in all_rows:
        v=r[idx]
        if v is None: v='<None>'
        else: v=str(v)
        c[v]+=1
    for k,n in c.most_common(50): print(repr(k),n)
PY"
