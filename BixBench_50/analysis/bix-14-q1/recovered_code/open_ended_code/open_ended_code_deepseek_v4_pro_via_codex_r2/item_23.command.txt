/bin/bash -lc "python - <<'PY'
import requests
urls=[
'https://www.medrxiv.org/content/medrxiv/early/2024/02/06/2024.02.02.24302163.full.pdf',
'https://www.medrxiv.org/content/10.1101/2024.02.02.24302163v1.full.pdf',
'https://www.medrxiv.org/content/10.1101/2024.02.02.24302163v1.full.pdf?download=1',
]
for u in urls:
 r=requests.get(u,timeout=30)
 print(u, r.status_code, r.headers.get('content-type'), len(r.content), r.url)
 if 'pdf' in r.headers.get('content-type',''):
  open('/workspace/tmp/paper.pdf','wb').write(r.content)
  break
PY
ls -l /workspace/tmp"
