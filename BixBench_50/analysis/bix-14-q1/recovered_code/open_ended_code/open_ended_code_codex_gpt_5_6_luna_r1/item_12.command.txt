/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
import re
ids={'184','185','285','287','353','354','381','396','397','489','490','499','500','503','504','556','557','613','614'}
rows=[]
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    sid=p.stem.split('CHIP_',1)[1]; m=re.match(r'(\\d+)',sid)
    if not m or m.group(1) not in ids: continue
    ws=load_workbook(p,data_only=True).active
    for r in range(3,ws.max_row+1):
        try: v=float(ws.cell(r,10).value)
        except: continue
        ont=ws.cell(r,14).value
        if v<.3 and ont in {'synonymous_variant','missense_variant','inframe_deletion','inframe_insertion','frameshift_variant','stop_gained','stop_lost','splice_acceptor_variant','splice_donor_variant','initiator_codon_variant','disruptive_inframe_deletion','disruptive_inframe_insertion'}:
            rows.append({'sid':sid,'vaf':v,'ont':ont,'gene':ws.cell(r,13).value,'pos':ws.cell(r,1).value,'alleles':ws.cell(r,2).value,'hgvs_c':ws.cell(r,18).value,'hgvs_p':ws.cell(r,19).value})
from collections import Counter
for key in [('pos','alleles'),('pos','alleles','ont'),('hgvs_c',)]:
    d={tuple(x[k] for k in key):x for x in rows}
    print(key,'unique',len(d),'syn',sum(x['ont']=='synonymous_variant' for x in d.values()),'total',len(d),'ratio',sum(x['ont']=='synonymous_variant' for x in d.values())/len(d))
    for k,x in d.items(): print(k, x['ont'], sum(tuple(y[z] for z in key)==k for y in rows))
PY"
