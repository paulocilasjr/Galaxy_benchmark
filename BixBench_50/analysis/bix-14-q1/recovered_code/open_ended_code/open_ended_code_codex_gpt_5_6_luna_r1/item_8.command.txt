/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
import re
trio=load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True).active
status={str(row[0]).strip(): row[3] for row in trio.iter_rows(min_row=2,values_only=True) if row[0] is not None}
files=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))
print('files',len(files))
for p in files:
    m=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name)
    sid=m.group(1) if m else p.name
    print(sid, '=>',status.get(sid,'MISSING'))
print('carrier IDs', [sid for sid,st in status.items() if st=='Carrier'])
print('carrier files', [p.stem.split('CHIP_',1)[1] for p in files if status.get(p.stem.split('CHIP_',1)[1])=='Carrier'])
PY"
