source /tmp/venv/bin/activate && python3 << 'EOF'
import pandas as pd
import os

# Check for multi-term SO annotations
files_dir = 'inputs/CHIP_DP10_GQ20_PASS'
carrier_files = [
    '230209_Exome_GRCh38_CHIP_185-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_184-PM.xlsx',
]

for fname in carrier_files:
    fpath = os.path.join(files_dir, fname)
    df = pd.read_excel(fpath, header=1)
    so = df['Sequence Ontology (Combined)'].dropna()
    multi = so[so.str.contains('&|,|;')]
    if len(multi) > 0:
        print(f"{fname}: Multi-term SO values:")
        print(multi.value_counts())
        print()
    else:
        print(f"{fname}: No multi-term SO values found")
EOF
