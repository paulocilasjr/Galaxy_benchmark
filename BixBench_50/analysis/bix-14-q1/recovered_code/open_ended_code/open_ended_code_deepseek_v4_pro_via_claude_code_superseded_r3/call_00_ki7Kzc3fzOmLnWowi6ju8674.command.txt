import pandas as pd
import os

files_dir = 'inputs/CHIP_DP10_GQ20_PASS'
carrier_files = [
    '230209_Exome_GRCh38_CHIP_185-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_184-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_285-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_287-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_353-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_354-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_381-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_396-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_397-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_489-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_490-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_499-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_500-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_503-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_504-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_556-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_557-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_613-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_614-PM.xlsx',
]

coding_so_terms = {
    'synonymous_variant',
    'missense_variant',
    'inframe_insertion',
    'inframe_deletion',
    'frameshift_variant',
    'stop_gained',
    'stop_lost',
    'start_lost',
    'coding_sequence_variant',
    'initiator_codon_variant',
}

total_coding_low_vaf = 0
total_synonymous_low_vaf = 0
all_rows = []

for fname in carrier_files:
    fpath = os.path.join(files_dir, fname)
    df = pd.read_excel(fpath, header=1)
    
    mask_coding = df['Sequence Ontology (Combined)'].isin(coding_so_terms)
    vaf = df['Variant Allele Freq']
    mask_vaf = vaf < 0.3
    
    coding_low_vaf = df[mask_coding & mask_vaf]
    total_coding_low_vaf += len(coding_low_vaf)
    
    for _, row in coding_low_vaf.iterrows():
        all_rows.append({
            'file': fname,
            'gene': row['Gene Names'],
            'so': row['Sequence Ontology (Combined)'],
            'vaf': row['Variant Allele Freq'],
            'zygosity': row['Zygosity'],
            'hgvs_c': row['HGVS c. (Clinically Relevant)'],
        })

print(f"Total coding variants with VAF < 0.3: {total_coding_low_vaf}")

# Breakdown by SO term
df_all = pd.DataFrame(all_rows)
print("\nBreakdown by SO term:")
print(df_all['so'].value_counts())
print(f"\nSynonymous count: {len(df_all[df_all['so'] == 'synonymous_variant'])}")
print(f"Fraction: {len(df_all[df_all['so'] == 'synonymous_variant'])} / {total_coding_low_vaf} = {len(df_all[df_all['so'] == 'synonymous_variant']) / total_coding_low_vaf:.6f}")
