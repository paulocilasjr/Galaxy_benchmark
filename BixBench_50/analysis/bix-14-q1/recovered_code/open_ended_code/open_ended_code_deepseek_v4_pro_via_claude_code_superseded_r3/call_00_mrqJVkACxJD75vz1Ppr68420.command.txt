source /tmp/venv/bin/activate && python3 << 'EOF'
import pandas as pd
import os

# Map carrier sample IDs to file names
carrier_ids = [185, 184, 285, 287, 353, 354, 556, 557, 533, 381, 396, 397, 489, 490, 499, 500, 503, 504, 613, 614]

files_dir = 'inputs/CHIP_DP10_GQ20_PASS'
files = os.listdir(files_dir)

# Map IDs to files
id_to_file = {}
for f in files:
    for cid in carrier_ids:
        if f'_{cid}-' in f or f'_{cid}.' in f:
            id_to_file[cid] = os.path.join(files_dir, f)
            break

print("Carrier ID -> File mapping:")
for cid in sorted(id_to_file.keys()):
    print(f"  {cid} -> {id_to_file[cid]}")

print(f"\nFound {len(id_to_file)} out of {len(carrier_ids)} carriers")
missing = set(carrier_ids) - set(id_to_file.keys())
if missing:
    print(f"Missing: {missing}")
