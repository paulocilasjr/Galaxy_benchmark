/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
from collections import Counter
import math
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def colnum(ref):
 s=''.join(c for c in ref if c.isalpha()); n=0
 for c in s:n=n*26+ord(c)-64
 return n
def load(p):
 with ZipFile(p) as z:
  root=ET.fromstring(z.read('xl/sharedStrings.xml'))
  ss=[''.join(t.text or '' for t in si.iter(NS+'t')) for si in root.findall(NS+'si')]
  root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
  out=[]
  for row in root.find(NS+'sheetData').findall(NS+'row'):
   d={}
   for c in row.findall(NS+'c'):
    v=c.find(NS+'v'); val=None if v is None else v.text
    if c.attrib.get('t')=='s' and val is not None: val=ss[int(val)]
    d[colnum(c.attrib['r'])]=val
   out.append(d)
 return out
p=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))[0]
rows=load(p)
print(p.name, 'rows',len(rows), 'maxcol',max(map(lambda d:max(d),rows)))
print('headers',rows[1])
for ci,name in [(1,'variant'),(2,'refalt'),(5,'zygosity'),(8,'DP'),(9,'GQ'),(10,'VAF'),(13,'gene'),(14,'SO'),(15,'effect'),(20,'filter'),(30,'In_CHIP')]:
 vals=[r.get(ci) for r in rows[2:]]
 print(name, 'n',len(vals), 'missing',sum(v is None for v in vals),'unique',Counter(vals).most_common(20))
print('last 5')
for r in rows[-5:]: print(r)
PY"
