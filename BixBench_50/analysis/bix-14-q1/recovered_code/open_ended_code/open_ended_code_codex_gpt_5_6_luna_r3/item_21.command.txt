/bin/bash -lc "python - <<'PY'
# Independent XML-path check of the final count, using lxml and explicit column letters.
from zipfile import ZipFile
from pathlib import Path
from lxml import etree
import re, openpyxl
NS='http://schemas.openxmlformats.org/spreadsheetml/2006/main'

def cell_col(ref):
    return re.match(r'[A-Z]+', ref).group(0)
def rows(p):
    with ZipFile(p) as z:
        ssroot=etree.fromstring(z.read('xl/sharedStrings.xml'))
        ss=[''.join(si.xpath('.//m:t/text()',namespaces={'m':NS})) for si in ssroot.xpath('./m:si',namespaces={'m':NS})]
        sheet=etree.fromstring(z.read('xl/worksheets/sheet1.xml'))
        for row in sheet.xpath('./m:sheetData/m:row[position()>2]',namespaces={'m':NS}):
            d={}
            for c in row.xpath('./m:c',namespaces={'m':NS}):
                v=c.find('{%s}v'%NS)
                if v is None: continue
                value=v.text
                if c.get('t')=='s': value=ss[int(value)]
                d[cell_col(c.get('r'))]=value
            yield d
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True)
carrier={str(r[0]).strip() for r in wb.active.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier'}
counts={'coding':0,'synonymous':0,'boundary_exact_03':0}
coding={'synonymous_variant','missense_variant','inframe_insertion','inframe_deletion','frameshift_variant','stop_gained','stop_lost','start_lost','coding_sequence_variant'}
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
    tok=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name).group(1); m=re.match(r'\\d+',tok); sid=m.group(0) if m else tok
    if sid not in carrier: continue
    for d in rows(p):
        try: vaf=float(d.get('J'))
        except (TypeError,ValueError): continue
        if abs(vaf-0.3)<1e-12: counts['boundary_exact_03']+=1
        if vaf<0.3 and d.get('N') in coding:
            counts['coding']+=1
            if d.get('N')=='synonymous_variant': counts['synonymous']+=1
print(counts)
assert counts == {'coding':41,'synonymous':30,'boundary_exact_03':0}
print('fraction_decimal', format(counts['synonymous']/counts['coding'], '.17g'))
PY"
