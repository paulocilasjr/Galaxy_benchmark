/bin/bash -lc "python - <<'PY'
# Check threshold boundary and annotation labels across supplied carrier workbooks.
from zipfile import ZipFile
from xml.etree import ElementTree as ET
from pathlib import Path
from collections import Counter
import re,openpyxl
NS='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
def colnum(ref):
 s=''.join(c for c in ref if c.isalpha());n=0
 for c in s:n=n*26+ord(c)-64
 return n
def rows(p):
 with ZipFile(p) as z:
  ss=[''.join(t.text or '' for t in si.iter(NS+'t')) for si in ET.fromstring(z.read('xl/sharedStrings.xml')).findall(NS+'si')]
  root=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
  for row in root.find(NS+'sheetData').findall(NS+'row')[2:]:
   d={}
   for c in row.findall(NS+'c'):
    v=c.find(NS+'v');x=None if v is None else v.text
    if c.attrib.get('t')=='s' and x is not None:x=ss[int(x)]
    d[colnum(c.attrib['r'])]=x
   yield d
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True)
carrier={str(r[0]).strip() for r in wb.active.iter_rows(min_row=2,values_only=True) if r[3]=='Carrier'}
for p in Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'):
 tok=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name).group(1);m=re.match(r'\\d+',tok);sid=m.group(0) if m else tok
 if sid not in carrier:continue
 vals=[]
 for r in rows(p):
  try: v=float(r.get(10))
  except: continue
  vals.append((v,r.get(14),r.get(1),r.get(13)))
 near=[x for x in vals if .29<=x[0]<=.31]
 if near:print(sid,near)
print('done')
PY"
