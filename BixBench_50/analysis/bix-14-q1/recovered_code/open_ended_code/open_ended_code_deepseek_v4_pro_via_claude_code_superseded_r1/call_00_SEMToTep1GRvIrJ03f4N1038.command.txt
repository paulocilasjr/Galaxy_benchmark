import openpyxl

# Check sample 396 specifically
wb = openpyxl.load_workbook('/workspace/inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_396-PF.xlsx')
ws = wb.active

coding_so_terms = {
    'synonymous_variant', 'missense_variant', 'stop_gained', 'stop_lost',
    'start_lost', 'frameshift_variant', 'inframe_insertion', 'inframe_deletion',
    'coding_sequence_variant', 'protein_altering_variant',
}

total_with_vaf = 0
coding_with_vaf_below_03 = 0
syn_with_vaf_below_03 = 0

for row in ws.iter_rows(min_row=3, values_only=True):
    so_term = str(row[13]).strip() if row[13] is not None else None
    vaf = row[9]
    if vaf is not None:
        total_with_vaf += 1
        vaf_val = float(vaf)
        if vaf_val < 0.3:
            if so_term and so_term in coding_so_terms:
                coding_with_vaf_below_03 += 1
                if so_term == 'synonymous_variant':
                    syn_with_vaf_below_03 += 1

print(f"Sample 396: total variants with VAF={total_with_vaf}")
print(f"  Coding with VAF<0.3: {coding_with_vaf_below_03}")
print(f"  Synonymous with VAF<0.3: {syn_with_vaf_below_03}")
