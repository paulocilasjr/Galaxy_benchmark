import pandas as pd
import os
import re

# Check: which carriers have files, which don't
trio = pd.read_excel('/workspace/inputs/230215_Trio_Status.xlsx')
blm_carriers = trio[trio['BLM Mutation Status'] == 'Carrier']
carrier_ids = set(blm_carriers['Sample ID'].astype(str).tolist())

var_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'
files = os.listdir(var_dir)

# Extract all IDs from filenames
file_ids = set()
for f in files:
    # Try numeric pattern: CHIP_<digits>-
    m = re.search(r'CHIP_(\d+)-', f)
    if m:
        file_ids.add(m.group(1))
    # Try SRR pattern: CHIP_SRR<digits>
    m = re.search(r'CHIP_(SRR\d+)', f)
    if m:
        file_ids.add(m.group(1))

print("Carriers without files:", carrier_ids - file_ids)
print("Carriers with files:", carrier_ids & file_ids)

# Check what status 533 has in trio
print("\nSample 533 details:")
print(trio[trio['Sample ID'].astype(str) == '533'].to_string())

# List all files that exist for carriers
print("\nFiles for carriers:")
carrier_with_files = carrier_ids & file_ids
for cid in sorted(carrier_with_files):
    for f in files:
        if cid in f:
            print(f"  {cid}: {f}")
