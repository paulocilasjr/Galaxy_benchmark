import pandas as pd
import os
import re

var_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'
files = sorted(os.listdir(var_dir))

# Proper mapping: only use files with pattern CHIP_<digits>-<suffix>.xlsx
# These are the carrier/patient files (not SRR controls)
carrier_file_map = {}
for f in files:
    m = re.search(r'CHIP_(\d+)-([^.]+)\.xlsx$', f)
    if m:
        cid = m.group(1)
        carrier_file_map[cid] = f

trio = pd.read_excel('/workspace/inputs/230215_Trio_Status.xlsx')
blm_carriers = trio[trio['BLM Mutation Status'] == 'Carrier']
carrier_ids = set(blm_carriers['Sample ID'].astype(str).tolist())

print("Carriers with data files:")
for cid in sorted(carrier_ids & set(carrier_file_map.keys())):
    print(f"  {cid}: {carrier_file_map[cid]}")
print(f"\nCarriers without data files: {carrier_ids - set(carrier_file_map.keys())}")
print(f"Total carriers with data: {len(carrier_ids & set(carrier_file_map.keys()))}")
