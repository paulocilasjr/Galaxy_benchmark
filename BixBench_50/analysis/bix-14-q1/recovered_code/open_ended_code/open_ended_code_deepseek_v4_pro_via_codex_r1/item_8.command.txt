/bin/bash -lc "python3 - <<'PY'
import openpyxl, glob, collections, os
files=sorted(glob.glob('/workspace/inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))
seq=collections.Counter(); eff=collections.Counter(); pairs=collections.Counter(); vaf_nonnum=collections.Counter()
for f in files:
    wb=openpyxl.load_workbook(f, read_only=False, data_only=True)
    ws=wb.active
    rows=list(ws.iter_rows(min_row=3, values_only=True))
    for r in rows:
        seq[str(r[13])]+=1
        eff[str(r[14])]+=1
        pairs[(str(r[13]),str(r[14]))]+=1
        v=r[9]
        if v is not None:
            try: float(v)
            except: vaf_nonnum[repr(v)]+=1
    wb.close()
print('files',len(files),'rows',sum(seq.values()))
print('\\nSEQ top all')
for k,v in seq.most_common(): print(v,k)
print('\\nEFF')
for k,v in eff.most_common(): print(v,k)
print('\\nPAIRS')
for k,v in pairs.most_common(): print(v,k)
print('vaf nonnum',vaf_nonnum)
PY"
