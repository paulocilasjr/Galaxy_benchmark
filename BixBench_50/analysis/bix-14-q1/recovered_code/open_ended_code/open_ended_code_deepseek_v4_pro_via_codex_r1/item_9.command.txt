import openpyxl, glob, os, re
statuses={}
wb=openpyxl.load_workbook('/workspace/inputs/230215_Trio_Status.xlsx', data_only=True)
ws=wb.active
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None: continue
    sid=str(r[0]).strip()
    statuses[sid]=(r[1],r[3])
wb.close()
files=sorted(glob.glob('/workspace/inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))
fmap={}
for f in files:
    b=os.path.basename(f).replace('230209_Exome_GRCh38_CHIP_','').replace('.xlsx','')
    fmap[b]=f
# handle SRR with tab in status
for sid,(st,blm) in sorted(statuses.items(), key=lambda x:(str(x[0]),)):
    present = sid in fmap
    if blm=='Carrier' or st=='Carrier' or present:
        pass
print('total statuses', len(statuses), 'files',len(files))
print('Carriers')
for sid,(st,blm) in sorted(statuses.items(), key=lambda x:str(x[0])):
    if blm=='Carrier':
        print(repr(sid), st, 'file', 'yes' if sid in fmap else 'NO')
print('Affected')
for sid,(st,blm) in sorted(statuses.items(), key=lambda x:str(x[0])):
    if blm=='Affected':
        print(repr(sid), st, 'file', 'yes' if sid in fmap else 'NO')
print('Unaffected count',sum(1 for st,blm in statuses.values() if blm=='Unaffected'))
print('unmapped files')
for sid in sorted(fmap):
    if sid not in statuses:
        print(repr(sid))
PY"
