def colnum(cellref):
    s=''.join(ch for ch in cellref if ch.isalpha())
    n=0
    for ch in s:
        n=n*26+ord(ch.upper())-64
    return n-1

                    if typ=='s' and val"'!='"'': val=ss[int(val)]
