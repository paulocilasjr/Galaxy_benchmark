/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from math import log2
p='/workspace/inputs/Proteomic_data .xlsx'
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['Tumor vs Normal']
headers=[c.value for c in next(ws.iter_rows(min_row=1,max_row=1))]
idx={h:i for i,h in enumerate(headers)}
print(headers)
matches=[]
for rno,row in enumerate(ws.iter_rows(min_row=2, values_only=True), start=2):
    gene=row[idx['gene']]
    desc=row[idx['Description']]
    if gene == 'ENO1' or (isinstance(desc,str) and 'ENO1' in desc):
        rec={h: row[i] for h,i in idx.items()}
        rec['_row']=rno
        matches.append(rec)
print('matches', len(matches))
for rec in matches:
    normal=rec['Normal']; tumor=rec['Tumor']
    print('row',rec['_row'])
    for k in ['protein','Description','gene','gene_id','Normal','Tumor','Ratio','FC','log2FC','p.value','adj.Pval','is_sig']:
        print(k, rec[k])
    print('calc_ratio', tumor/normal, 'calc_log2', log2(tumor/normal), 'rounded', round(log2(tumor/normal),2))
PY"
