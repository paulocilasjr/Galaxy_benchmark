/bin/bash -lc "pwd
rg --files -g '"'!*'"' -g '*' /workspace | sed -n '1,80p'
python - <<'PY'
from openpyxl import load_workbook
from pathlib import Path
for p in Path('/workspace/inputs').glob('*.xlsx'):
    wb=load_workbook(p, read_only=True, data_only=True)
    print('\\nFILE',p.name)
    for ws in wb.worksheets:
        print('SHEET',repr(ws.title),'rows',ws.max_row,'cols',ws.max_column)
        for row in ws.iter_rows(min_row=1,max_row=min(8,ws.max_row),values_only=True):
            print(row[:min(15,ws.max_column)])
PY"
