/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from pathlib import Path
from math import log2
p=Path('/workspace/inputs/Proteomic_data .xlsx')
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['Tumor vs Normal']
headers=next(ws.iter_rows(min_row=1, max_row=1, values_only=True))
idx={h:i for i,h in enumerate(headers)}
for row in ws.iter_rows(min_row=2, values_only=True):
    if row[idx['gene']] == 'ENO1':
        normal=row[idx['Normal']]
        tumor=row[idx['Tumor']]
        ratio=row[idx['Ratio']]
        log2fc=row[idx['log2FC']]
        calc=log2(tumor/normal)
        print('normal', normal)
        print('tumor', tumor)
        print('ratio', ratio)
        print('reported_log2FC', log2fc)
        print('calculated_log2FC', calc)
        print('rounded', f'{log2fc:.2f}')
        break
PY"
