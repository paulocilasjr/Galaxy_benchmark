import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/Proteomic_data .xlsx', data_only=True)
print("Sheet names:", wb.sheetnames)

for sn in wb.sheetnames:
    ws = wb[sn]
    print(f"\n=== Sheet: {sn} ===")
    print(f"Dimensions: {ws.dimensions}, rows={ws.max_row}, cols={ws.max_column}")
    # Print first 15 rows
    for i, row in enumerate(ws.iter_rows(min_row=1, max_row=min(15, ws.max_row), values_only=False)):
        vals = [str(c.value)[:80] if c.value is not None else 'None' for c in row]
        print(f"  Row {i+1}: {vals}")
