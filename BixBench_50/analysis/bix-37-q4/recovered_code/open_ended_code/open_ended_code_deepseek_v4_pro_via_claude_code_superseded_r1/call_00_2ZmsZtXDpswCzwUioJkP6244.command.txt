import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/Proteomic_data .xlsx', data_only=True)
ws = wb['Tumor vs Normal']

# Find ENO1 in the gene column (column C, index 3)
for row in ws.iter_rows(min_row=2, values_only=True):
    gene = row[2]  # gene column
    if gene and str(gene).strip().upper() == 'ENO1':
        print(f"Found ENO1:")
        print(f"  Protein: {row[0]}")
        print(f"  Description: {row[1]}")
        print(f"  Gene: {row[2]}")
        print(f"  Gene ID: {row[3]}")
        print(f"  Normal: {row[4]}")
        print(f"  Tumor: {row[5]}")
        print(f"  Ratio: {row[6]}")
        print(f"  FC: {row[7]}")
        print(f"  log2FC: {row[8]}")
        print(f"  p.value: {row[9]}")
        print(f"  adj.Pval: {row[10]}")
        print(f"  is_sig: {row[11]}")
        print(f"  compare: {row[12]}")
        break
else:
    print("ENO1 not found in gene column")
