import pandas as pd
import numpy as np
import warnings
warnings.filterwarnings('ignore')
import gc
import os

os.makedirs('codex_output', exist_ok=True)

# Load data
counts_df = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
annot = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')

annot = annot.set_index('sample')
annot = annot.loc[counts_df.columns]
annot['batch'] = annot['batch'].astype(str)
annot['sex'] = annot['sex'].astype(str)

# Prefilter: keep genes with total count >= 50 (keeps FAM138A with 292 total)
row_sums = counts_df.sum(axis=1)
keep = row_sums >= 50
keep = keep | (counts_df.index == 'FAM138A')
print(f"Keeping {keep.sum()} / {len(keep)} genes")
counts_filtered = counts_df.loc[keep]
del counts_df, row_sums
gc.collect()

from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts_int = counts_filtered.T.round().astype(int)
del counts_filtered
gc.collect()

dds = DeseqDataSet(
    counts=counts_int,
    metadata=annot,
    design_factors="batch + sex",
    refit_cooks=False,
    n_cpus=1,
)
del counts_int
gc.collect()

print("Starting deseq2 fitting...")
dds.deseq2()
print("DESeq2 fitting done!")

# Immediately run DeseqStats and LFC shrink
print("\nRunning DeseqStats (M vs F)...")
stat = DeseqStats(dds, contrast=["sex", "M", "F"], n_cpus=1, cooks_filter=False, independent_filter=False)
stat.summary()
print("Wald test done")

print("Running LFC shrinkage (default apeglm)...")
stat.lfc_shrink()
print("LFC shrinkage done")

res = stat.results_df
print(f"\nResults columns: {res.columns.tolist()}")

# Save results so we don't lose them
res.to_csv('codex_output/deseq2_results.csv')
print("Results saved to codex_output/deseq2_results.csv")

# FAM138A
if 'FAM138A' in res.index:
    fam = res.loc['FAM138A']
    print(f"\n=== FAM138A RESULTS ===")
    print(fam.to_string())
    
    # Write answer
    lfc = fam.get('log2FoldChange', fam.get('lfc', None))
    if lfc is None:

else:
