import pandas as pd
import numpy as np
import warnings
warnings.filterwarnings('ignore')

# Load data
counts_df = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
annot = pd.read_csv('inputs/Sample_annotated_Zenodo.csv')

# Ensure alignment
annot = annot.set_index('sample').loc[counts_df.columns].reset_index()

# Ensure batch and sex are proper types
annot['batch'] = annot['batch'].astype(str)  # treat as categorical
annot['sex'] = annot['sex'].astype(str)

print("Annotation head:")
print(annot.head())
print(f"\nSex value counts: {annot['sex'].value_counts().to_dict()}")
print(f"Batch value counts: {annot['batch'].value_counts().to_dict()}")

from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

# Round counts to integers (they should already be integers)
counts_df = counts_df.round().astype(int)

# Create DeseqDataSet
# Design: ~ batch + sex
dds = DeseqDataSet(
    counts=counts_df.T,  # pydeseq2 expects samples x genes
    metadata=annot.set_index('sample'),
    design_factors="batch + sex",  # batch as covariate, sex as main effect
)

print(f"\nDDS object shape: {dds.counts.shape}")
print(f"Design factors: {dds.design_factors}")

# Run DESeq2
dds.deseq2()

print("\nDESeq2 completed")
print(f"Available results columns: {dds.results_df.columns.tolist()}")

# Check coefficient names
print(f"\nResults index names sample: {dds.results_df.index[:5].tolist()}")
print(f"Results columns: {dds.results_df.columns.tolist()}")

# Show some results to understand coefficient names
print("\nFirst few rows of results:")
print(dds.results_df.head())
