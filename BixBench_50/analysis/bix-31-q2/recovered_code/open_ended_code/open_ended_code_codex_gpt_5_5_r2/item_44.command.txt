import os, gc
import pandas as pd
import numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

os.makedirs('codex_output', exist_ok=True)

meta_all = pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample')

meta['sex'] = pd.Categorical(meta['sex'], categories=['F', 'M'])

# First compute DESeq2/PyDESeq2 size factors from all provided batch-corrected counts.
counts_all = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0).T
if list(counts_all.index) "'!= list(meta.index):
    counts_all = counts_all.loc[meta.index]

full_dds.fit_size_factors()
size_factors = full_dds.obs['size_factors'].copy()
normed_means = pd.Series(np.asarray(full_dds.var['_normed_means']), index=full_dds.var_names)
eligible = normed_means[normed_means > 10].index.tolist()
if 'FAM138A' not in counts_all.columns:

# Copy subset before freeing full object.
counts = counts_all.loc[:, eligible].copy()
del full_dds, counts_all
_gc = gc.collect()

# Run the DESeq2 pipeline on the requested baseMean scope, preserving full-table size factors.
dds = DeseqDataSet(counts=counts, metadata=meta, design='~ batch + sex', quiet=False, low_memory=True, n_cpus=1)
