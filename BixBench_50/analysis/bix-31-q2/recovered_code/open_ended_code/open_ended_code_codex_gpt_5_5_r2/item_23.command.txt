/bin/bash -lc "python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0, nrows=300).T
meta = pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample')
counts = counts.loc[meta.index]
meta = meta.loc[counts.index, ['batch','sex']].copy()
meta['batch'] = meta['batch'].astype(str)
meta['sex'] = pd.Categorical(meta['sex'], categories=['F','M'])
# keep genes with at least one nonzero in subset
counts = counts.loc[:, counts.sum(axis=0) > 0]
print(counts.shape, meta.dtypes.to_dict())
dds = DeseqDataSet(counts=counts, metadata=meta, design='~ batch + sex', ref_level=['sex','F'], n_cpus=4, quiet=False)
dds.deseq2()
print('design columns', dds.obsm['design_matrix'].columns.tolist())
print('LFC cols', dds.varm['LFC'].columns.tolist())
stat = DeseqStats(dds, contrast=['sex','M','F'], quiet=False, n_cpus=4)
stat.summary()
print(stat.results_df.head().to_string())
print('coeff names maybe', dds.varm['LFC'].columns.tolist())
PY"
