/bin/bash -lc "python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts = pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0, nrows=200).T
meta = pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample')
meta = meta.loc[counts.index].copy()
meta['batch'] = pd.Categorical(meta['batch'].astype(str))
meta['sex'] = pd.Categorical(meta['sex'], categories=['F','M'])
print(counts.shape, meta.shape)
dds = DeseqDataSet(counts=counts, metadata=meta, design='~ batch + sex', quiet=True, n_cpus=4)
dds.deseq2()
stat = DeseqStats(dds, contrast=['sex','M','F'], quiet=True, n_cpus=4)
stat.summary()
print('LFC cols:', list(stat.LFC.columns))
print('results cols:', stat.results_df.columns.tolist())
print(stat.results_df.head().to_string())
PY"
