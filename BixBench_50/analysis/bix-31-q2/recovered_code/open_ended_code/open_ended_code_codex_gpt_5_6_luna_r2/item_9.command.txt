/bin/bash -lc "python - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('inputs/Sample_annotated_Zenodo.csv')))
by=collections.defaultdict(list)
for r in rows: by[r['samp']].append(r)
print('subjects',len(by))
print('subject sizes',collections.Counter(len(v) for v in by.values()))
print('samtype by subject',collections.Counter(tuple(sorted(set(r['samtype'] for r in v))) for v in by.values()))
print('sex by subject',collections.Counter(tuple(sorted(set(r['sex'] for r in v))) for v in by.values()))
print('batch by subject',collections.Counter(tuple(sorted(set(r['batch'] for r in v))) for v in by.values()))
for samp in list(by)[:20]:
    print(samp, [(r['sample'],r['samtype'],r['celltype'],r['batch'],r['sex']) for r in by[samp]])
PY"
