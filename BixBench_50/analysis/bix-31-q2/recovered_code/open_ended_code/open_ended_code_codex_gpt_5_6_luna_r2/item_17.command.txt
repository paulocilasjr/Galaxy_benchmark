/bin/bash -lc "python - <<'PY'
import csv, statistics
for fn in ['BatchCorrectedReadCounts_Zenodo.csv','RawReadCounts_Zenodo.csv']:
    sums=None; n=0
    with open('inputs/'+fn,newline='') as f:
        rd=csv.reader(f); h=next(rd); sums=[0]*(len(h)-1)
        for row in rd:
            n+=1
            for i,s in enumerate(row[1:]): sums[i]+=int(float(s))
    print(fn,'rows',n,'colsum min med mean max',min(sums),statistics.median(sums),statistics.mean(sums),max(sums),'first',sums[:5])
PY"
