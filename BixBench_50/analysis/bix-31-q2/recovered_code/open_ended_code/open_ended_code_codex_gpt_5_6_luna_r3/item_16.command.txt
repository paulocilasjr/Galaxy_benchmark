/bin/bash -lc "python - <<'PY'
import pandas as pd
m=pd.read_csv('inputs/Sample_annotated_Zenodo.csv')
print(m.groupby(['samtype','sex','batch']).size().to_string())
print('\\nindividuals',m.groupby(['samtype','sex','batch'])['samp'].nunique().to_string())
print('\\nby samtype sex',m.groupby(['samtype','sex']).agg(samples=('sample','size'), donors=('samp','nunique')).to_string())
print('\\nDonor multiplicity counts')
print(m.groupby('samp').size().value_counts().sort_index().to_string())
print('\\nsamples per celltype/samtype')
print(pd.crosstab(m['samtype'],m['celltype']).to_string())
print('\\nDonors sex consistency:',m.groupby('samp')['sex'].nunique().value_counts().to_dict())
print('Donors batch consistency:',m.groupby('samp')['batch'].nunique().value_counts().to_dict())
print('\\nfirst 20 donors')
print(m.groupby('samp').agg(sex=('sex','first'),batch=('batch','first'),samtype=('samtype','first'),n=('sample','size')).head(20).to_string())
PY"
