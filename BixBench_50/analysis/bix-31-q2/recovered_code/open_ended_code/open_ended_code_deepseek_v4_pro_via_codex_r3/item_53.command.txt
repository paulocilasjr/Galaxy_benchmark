/bin/bash -c "python -u - <<'PY'
import pandas as pd, numpy as np, gc
from pydeseq2.dds import DeseqDataSet
from pydeseq2.utils import robust_method_of_moments_disp, n_or_more_replicates
from scipy.stats import f

def fit_genewise_dispersions_lowmem(dds):
    if \"size_factors\" not in dds.obs:
        dds.fit_size_factors(fit_type=dds.size_factors_fit_type)
    dds.var[\"non_zero\"] = ~(dds.X == 0).all(axis=0)
    dds.non_zero_idx = np.arange(dds.n_vars)[dds.var[\"non_zero\"]]
    dds.non_zero_genes = dds.var_names[dds.var[\"non_zero\"]]
    dds._fit_MoM_dispersions()
    if \"normed_counts\" in dds.layers:
        del dds.layers[\"normed_counts\"]
    gc.collect()
    design_matrix=dds.obsm['design_matrix'].values
    size_factors=dds.obs['size_factors'].values
    if len(dds.obsm['design_matrix'].value_counts())==dds.obsm['design_matrix'].shape[-1]:
        mu_hat_=dds.inference.lin_reg_mu(counts=dds.X[:,dds.non_zero_idx],size_factors=size_factors,design_matrix=design_matrix,min_mu=dds.min_mu)
    else:
        _,mu_hat_,_,_=dds.inference.irls(counts=dds.X[:,dds.non_zero_idx],size_factors=size_factors,design_matrix=design_matrix,disp=dds.var.loc[dds.var['non_zero'],'_MoM_dispersions'].values,min_mu=dds.min_mu,beta_tol=dds.beta_tol)
    dds.layers['_mu_hat']=np.full((dds.n_obs,dds.n_vars),np.nan)
    dds.layers['_mu_hat'][:,dds.var['non_zero']]=mu_hat_
    del mu_hat_; gc.collect()
    dispersions_,conv=dds.inference.alpha_mle(counts=dds.X[:,dds.non_zero_idx],design_matrix=design_matrix,mu=dds.layers['_mu_hat'][:,dds.non_zero_idx],alpha_hat=dds.var.loc[dds.var['non_zero'],'_MoM_dispersions'].values,min_disp=dds.min_disp,max_disp=dds.max_disp)
    dds.var['genewise_dispersions']=np.full(dds.n_vars,np.nan)
    dds.var.loc[dds.var['non_zero'],'genewise_dispersions']=np.clip(dispersions_,dds.min_disp,dds.max_disp)
    dds.var['_genewise_converged']=pd.array([pd.NA]*dds.n_vars,dtype='boolean')
    dds.var.loc[dds.var['non_zero'],'_genewise_converged']=conv

meta=pd.read_csv('inputs/Sample_annotated_Zenodo.csv').set_index('sample')
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv',index_col=0).astype(np.int32).T
meta=meta.loc[counts.index]; meta['batch']='b'+meta['batch'].astype(str)
dds=DeseqDataSet(counts=counts,metadata=meta,design='~ batch + sex',n_cpus=4,quiet=True,low_memory=True,refit_cooks=False)
dds.fit_size_factors()
fit_genewise_dispersions_lowmem(dds)
dds.fit_dispersion_trend()
dds.fit_dispersion_prior()
dds.fit_MAP_dispersions()
gene='FAM138A'; gi=dds.var_names.get_loc(gene)
dm=dds.obsm['design_matrix'].values; sf=dds.obs['size_factors'].values
disp=np.array([dds.var.loc[gene,'dispersions']],dtype=float)
beta,mu,hat,conv=dds.inference.irls(counts=dds.X[:,gi:gi+1],size_factors=sf,design_matrix=dm,disp=disp,min_mu=dds.min_mu,beta_tol=dds.beta_tol)
mu=mu[:,0]; hat=hat[:,0]
normed_col=dds.X[:,gi]/sf
rdisp=robust_method_of_moments_disp(normed_col[:,None], dds.obsm['design_matrix'])
res=(dds.X[:,gi]-mu)**2
V=mu**2*rdisp[0]+mu
res=res/V/5
res=res*hat/(1-hat)**2
cutoff=f.ppf(0.99,5,dds.n_obs-5)
use_for_max=n_or_more_replicates(dds.obsm['design_matrix'],3)
print('gene',gene)
print('dispersion',float(dds.var.loc[gene,'dispersions']))
print('mu_LFC',mu)
print('hat_diagonals',hat)
print('rdisp',rdisp)
print('cooks',res)
print('cooks_cutoff',cutoff)
print('use_for_max count',use_for_max.sum(),'of',len(use_for_max))
print('max cooks sample index',np.argmax(res),'sample',dds.obs_names[np.argmax(res)],'maxcooks',res.max())
print('outlier among use_for_max',bool((res[use_for_max]>cutoff).any()))
pos=np.argmax(res)
print('pos sample',dds.obs_names[pos], 'count at pos',dds.X[pos,gi], 'counts greater than pos', int((dds.X[:,gi]>dds.X[pos,gi]).sum()))
print('would be final outlier:', bool((res[use_for_max]>cutoff).any()) and int((dds.X[:,gi]>dds.X[pos,gi]).sum())<3)
PY"
