/bin/bash -c "python - <<'PY'
from pydeseq2.ds import DeseqStats
import inspect
src=inspect.getsource(DeseqStats)
for needle in ['base_mean','def LFC','def SE']:
    print('---',needle)
    idx=0
    while True:
        idx=src.find(needle,idx)
        if idx==-1: break
        print(idx, src[idx-150:idx+500].replace('\\n',' '))
        idx+=1
PY"
