/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np, gc
n_genes=0; n_nonzero=0; n_total_ge10=0; n_total_ge1=0
chunks=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv',index_col=0,chunksize=2000)
for chunk in chunks:
    # chunk genes x samples
    a=chunk.to_numpy(dtype=np.int64)
    n_genes += a.shape[0]
    n_nonzero += (a>0).any(axis=1).sum()
    sums=a.sum(axis=1)
    n_total_ge1 += (sums>=1).sum()
    n_total_ge10 += (sums>=10).sum()
    del a, chunk; gc.collect()
print('genes',n_genes,'any nonzero',n_nonzero,'sum>=1',n_total_ge1,'sum>=10',n_total_ge10)
PY"
