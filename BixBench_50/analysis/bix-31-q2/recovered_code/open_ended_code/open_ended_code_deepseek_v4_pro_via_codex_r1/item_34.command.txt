/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
from pydeseq2.dds import DeseqDataSet
meta=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv', index_col=0)
meta['sex']=meta['sex'].astype('category')
counts=pd.DataFrame(np.random.randint(0,10,size=(len(meta),5)), index=meta.index, columns=[f'g{i}' for i in range(5)])
for desc, m in [('int batch', meta.copy()), ('cat batch', meta.assign(batch=meta['batch'].astype('category'))), ('sex obj', meta.assign(batch=meta['batch'].astype('category'), sex=meta['sex'].astype('object')))]:
    print('\\n', desc, m.dtypes.to_dict())
    try:
        d=DeseqDataSet(counts=counts, metadata=m, design='~batch+sex', quiet=True)
        print(d.obsm['design_matrix'].columns.tolist())
    except Exception as e:
        print('ERR', type(e), e)
PY"
