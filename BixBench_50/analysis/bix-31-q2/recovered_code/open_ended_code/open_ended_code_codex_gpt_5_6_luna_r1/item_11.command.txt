/bin/bash -lc "python - <<'PY'
import csv, collections, os, math
base='/workspace/inputs'
for fn in ['BatchCorrectedReadCounts_Zenodo.csv','RawReadCounts_Zenodo.csv','GeneMetaInfo_Zenodo.csv','Sample_annotated_Zenodo.csv','FlowSorterFraction_Zenodo.csv']:
    path=os.path.join(base,fn)
    with open(path,newline='') as f:
        rd=csv.reader(f)
        header=next(rd)
        nrow=0; first=[]; dups=collections.Counter()
        for row in rd:
            nrow+=1
            if len(first)<3: first.append(row[:8])
            if row: dups[row[0]]+=1
    print(fn, 'header_cols=',len(header), 'data_rows=',nrow, 'first_header=',header[:8], 'first_rows=',first[:2])
    if fn.endswith('ReadCounts_Zenodo.csv'):
        print('  duplicate first-column IDs:',sum(v>1 for v in dups.values()), 'maxdup',max(dups.values()))

# Annotation details
with open(os.path.join(base,'Sample_annotated_Zenodo.csv'),newline='') as f:
    rows=list(csv.DictReader(f))
print('annotation rows',len(rows),'columns',list(rows[0]))
for c in rows[0]:
    vals=collections.Counter(r[c] for r in rows)
    print(c, 'n=',len(vals), 'values=',vals.most_common(20))

# count header exact matching annotation samples, gene metadata ids
with open(os.path.join(base,'BatchCorrectedReadCounts_Zenodo.csv'),newline='') as f:
    h=next(csv.reader(f))[1:]
ann={r['sample'] for r in rows}
print('count samples',len(h),'annotation samples',len(ann),'missing ann',len(set(h)-ann),'extra ann',len(ann-set(h)))
print('duplicate sample names',len(h)-len(set(h)))
print('sample order first/last',h[:5],h[-5:])
with open(os.path.join(base,'GeneMetaInfo_Zenodo.csv'),newline='') as f:
    gm=list(csv.DictReader(f))
gids=[r['Geneid'] for r in gm]
print('gene metadata rows',len(gm),'unique Geneid',len(set(gids)),'FAM138A rows',[r for r in gm if r['Geneid']=='FAM138A'])
PY"
