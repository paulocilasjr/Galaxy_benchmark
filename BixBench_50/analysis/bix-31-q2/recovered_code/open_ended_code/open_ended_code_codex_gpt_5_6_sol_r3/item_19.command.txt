/bin/bash -lc "python - <<'PY'
import pandas as pd
import numpy as np
p='inputs/BatchCorrectedReadCounts_Zenodo.csv'
cols=pd.read_csv(p,nrows=0).columns
dtypes={c:'int32' for c in cols[1:]}
zero=0
sum_lt10=0
sum_lt100=0
n=0
fam=None
for ch in pd.read_csv(p,index_col=0,dtype=dtypes,chunksize=2000):
    s=ch.sum(axis=1)
    zero += int((s==0).sum())
    sum_lt10 += int((s<10).sum())
    sum_lt100 += int((s<100).sum())
    n += len(ch)
    if 'FAM138A' in ch.index: fam=int(s.loc['FAM138A'])
print({'genes':n,'all_zero':zero,'sum_lt10':sum_lt10,'sum_lt100':sum_lt100,'FAM138A_sum':fam})
PY"
