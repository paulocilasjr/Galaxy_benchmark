/bin/bash -lc "PYTHONPATH=analysis_work/phykit_source python - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
from scipy.stats import mannwhitneyu

def scores(folder):
    paths=sorted(Path(folder).glob('*.faa.mafft'))
    exact=[Alignment(alignment_file_path=str(p)).calculate_rcv() for p in paths]
    return paths, exact, [round(v,4) for v in exact]

ap, ae, ar = scores('analysis_work/animals')
fp, fe, fr = scores('analysis_work/fungi')
print('first_validation', ap[0].name, ae[0], ar[0])
for label, a, f in [('cli_rounded', ar, fr), ('internal_exact', ae, fe)]:
    print(label, 'n',len(a),len(f), 'unique',len(set(a)),len(set(f)), 'ties_combined',len(a)+len(f)-len(set(a+f)))
    print('two_sided_auto', mannwhitneyu(a,f,alternative='two-sided',method='auto'))
    print('two_sided_asymptotic_cc', mannwhitneyu(a,f,alternative='two-sided',method='asymptotic',use_continuity=True))
    print('two_sided_asymptotic_no_cc', mannwhitneyu(a,f,alternative='two-sided',method='asymptotic',use_continuity=False))
    print('less', mannwhitneyu(a,f,alternative='less',method='asymptotic',use_continuity=True))
    print('greater', mannwhitneyu(a,f,alternative='greater',method='asymptotic',use_continuity=True))
PY"
