/bin/bash -lc "rm -rf work_rcv && mkdir -p work_rcv/animals work_rcv/fungi && python - <<'PY'
import zipfile, pathlib
for group, zpath in [('animals','inputs/scogs_animals.zip'), ('fungi','inputs/scogs_fungi.zip')]:
    out = pathlib.Path('work_rcv')/group
    with zipfile.ZipFile(zpath) as z:
        for info in z.infolist():
            if info.filename.endswith('.faa.mafft'):
                target = out / pathlib.Path(info.filename).name
                target.write_bytes(z.read(info))
    print(group, len(list(out.glob('*.mafft'))))
PY"
