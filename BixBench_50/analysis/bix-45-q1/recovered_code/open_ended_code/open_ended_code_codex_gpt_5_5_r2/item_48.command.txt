/bin/bash -lc "python - <<'PY'
from pathlib import Path
from statistics import median
from scipy.stats import mannwhitneyu
from phykit.services.alignment.rcv import Alignment

scores={}
for group in ['animals','fungi']:
    vals=[]
    bad=[]
    for p in sorted(Path('work_rcv', group).glob('*.mafft')):
        try:
            vals.append(float(Alignment(alignment_file_path=str(p)).calculate_rcv()))
        except Exception as e:
            bad.append((p.name, repr(e)))
    scores[group]=vals
    print(group, 'n', len(vals), 'bad', len(bad), 'min', min(vals), 'median', median(vals), 'max', max(vals))
    if bad: print(bad[:3])

animals=scores['animals']; fungi=scores['fungi']
for alt in ['two-sided','less','greater']:
    res=mannwhitneyu(animals, fungi, alternative=alt, method='auto')
    print('auto', alt, 'U', repr(float(res.statistic)), 'p', repr(float(res.pvalue)))
for method in ['asymptotic','exact']:
    res=mannwhitneyu(animals, fungi, alternative='two-sided', method=method)
    print(method, 'two-sided U', repr(float(res.statistic)), 'p', repr(float(res.pvalue)))
res=mannwhitneyu(animals, fungi, alternative='two-sided', method='asymptotic', use_continuity=False)
print('asymptotic_no_continuity two-sided U', repr(float(res.statistic)), 'p', repr(float(res.pvalue)))
PY"
