import subprocess
import os

PHYKIT = '/workspace/venv/bin/phykit'
animals_dir = '/workspace/work'
fungi_dir = '/workspace/work/fungi_scogs'

# Get common orthologs
animal_set = set(f.replace('.faa.mafft', '') for f in os.listdir(animals_dir) if f.endswith('.mafft'))
fungal_set = set(f.replace('.faa.mafft', '') for f in os.listdir(fungi_dir) if f.endswith('.mafft'))
common = sorted(animal_set & fungal_set)
print(f'Common orthologs: {len(common)}')

# Verify a few alignments manually
for oid in common[:3]:
    a_path = os.path.join(animals_dir, oid + '.faa.mafft')
    f_path = os.path.join(fungi_dir, oid + '.faa.mafft')
    a_rcv = float(subprocess.run([PHYKIT, 'rcv', a_path], capture_output=True, text=True).stdout.strip())
    f_rcv = float(subprocess.run([PHYKIT, 'rcv', f_path], capture_output=True, text=True).stdout.strip())
    print(f'{oid}: animal={a_rcv}, fungal={f_rcv}')
