/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from Bio import AlignIO
from collections import Counter
import statistics
for grp in ['ani','fun']:
 vals=[]
 for p in Path('/workspace/tmp/extract_'+grp).glob('*.faa.mafft'):
  aln=AlignIO.read(str(p),'fasta'); n=len(aln); L=aln.get_alignment_length()
  all_chars=''.join(str(r.seq) for r in aln); total=Counter(all_chars); total_chars=sum(total.values()); avg={c:total[c]/total_chars for c in total}
  ds=0
  for rec in aln:
   seq=str(rec.seq); sc=Counter(seq)
   ds += sum(abs(sc.get(c,0)-avg[c]*len(seq)) for c in total)
  vals.append(ds/(n*L))
 print(grp,'median',statistics.median(vals),'pct>0.25',sum(v>0.25 for v in vals)/len(vals)*100,'mean',statistics.mean(vals))
PY"
