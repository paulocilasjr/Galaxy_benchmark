/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from Bio import AlignIO
from collections import Counter
from scipy import stats
import statistics

def rcv_bio(p):
    aln=AlignIO.read(str(p),'fasta')
    n=len(aln); L=aln.get_alignment_length()
    all_chars=''.join(str(r.seq) for r in aln)
    total=Counter(all_chars); total_chars=sum(total.values())
    avg={c: total[c]/total_chars for c in total}
    devsum=0
    for rec in aln:
        seq=str(rec.seq); sc=Counter(seq)
        dev=sum(abs(sc.get(c,0)-avg[c]*len(seq)) for c in total)
        devsum+=dev
    return devsum/(n*L)
for grp in ['ani','fun']:
    vals={p.name.split('.')[0]: round(rcv_bio(p),4) for p in Path('/workspace/tmp/extract_'+grp).glob('*.faa.mafft')}
    print(grp,len(vals),'zeros',sum(v==0 for v in vals.values()),'minmax',min(vals.values()),max(vals.values()),'median',statistics.median(vals.values()))
    globals()[grp]=list(vals.values())
u,p=stats.mannwhitneyu(ani,fun,alternative='two-sided')
print('p',repr(p),'u',u)
PY"
