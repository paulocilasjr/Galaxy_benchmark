/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from phykit.services.alignment.base import Alignment
import subprocess
for grp in ['ani','fun']:
    for p in list(Path('/workspace/tmp/extract_'+grp).glob('*.faa.mafft'))[:3]:
        v=Alignment(alignment_file_path=str(p)).calculate_rcv()
        out=subprocess.check_output(['phykit','rcv',str(p)],text=True).strip()
        print(p.name,repr(v),out, 'match',round(v,4)==float(out))
PY"
