/bin/bash -lc "Rscript - <<'EOF'
a=scan('/workspace/old_A.csv',quiet=TRUE); b=scan('/workspace/old_F.csv',quiet=TRUE)
for (exact in c(FALSE,TRUE)) for (corr in c(FALSE,TRUE)) {
 if(exact) {res=wilcox.test(a,b,alternative='two.sided',exact=TRUE)} else {res=wilcox.test(a,b,alternative='two.sided',exact=FALSE,correct=corr)}
 cat('exact',exact,'corr',corr,'p=',format(res"'$p.value,digits=22,scientific=TRUE),'"'\\n')
}
EOF"
