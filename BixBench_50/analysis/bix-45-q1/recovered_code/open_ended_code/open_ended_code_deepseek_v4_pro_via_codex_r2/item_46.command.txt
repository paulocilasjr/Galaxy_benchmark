/bin/bash -lc "cd /workspace && python3 - <<'PY'
import sys, json
sys.path.insert(0,'/workspace/phykit203')
from phykit.services.alignment.rcv import RelativeCompositionVariability
from argparse import Namespace
from pathlib import Path
from scipy import stats
from statistics import median,mean

def vals(d):
    vv={}
    for p in sorted(Path(d).glob('*.faa.mafft')):
        raw=RelativeCompositionVariability(Namespace(alignment=str(p))).calculate_rcv()
        vv[p.name[:-len('.faa.mafft')]]=round(raw,4)
    return vv
A=vals('/workspace/work_animals'); F=vals('/workspace/work_fungi')
print('n',len(A),len(F))
print('medA medF',median(A.values()),median(F.values()))
print('means',mean(A.values()),mean(F.values()))
u,p=stats.mannwhitneyu(list(A.values()),list(F.values()),alternative='two-sided')
print('p all old',repr(p),'U',u)
u2,p2=stats.mannwhitneyu(list(A.values()),[F[g] for g in sorted(set(A)&set(F))],alternative='two-sided')
print('p common old',repr(p2),'U',u2)
# outputs first few
for g in list(A)[:5]: print(g,A[g],F.get(g))
PY"
