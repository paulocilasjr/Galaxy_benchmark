/bin/bash -lc "cd /workspace && python3 - <<'PY'
# Recompute quickly by direct phykit class for all to ensure and variants
from pathlib import Path
import json, subprocess, math
from scipy import stats
from statistics import median,mean
from phykit.services.alignment.rcv import RelativeCompositionVariability
from argparse import Namespace

def phykit_rcv(path):
    inst=RelativeCompositionVariability(Namespace(alignment=str(path), json=False))
    return inst.calculate_rcv()
# But run method rounds to 4 decimals. Use run? We'll use calculate and round.
def vals(d):
    vv={}
    for p in sorted(Path(d).glob('*.faa.mafft')):
        x=phykit_rcv(p)
        vv[p.name[:-len('.faa.mafft')]]=round(x,4)
    return vv
A=vals('/workspace/work_animals'); F=vals('/workspace/work_fungi')
common=sorted(set(A)&set(F))
for name, fa in [('all',F),('common',{g:F[g] for g in common})]:
    u,p=stats.mannwhitneyu(list(A.values()), list(fa.values()), alternative='two-sided')
    print(name, len(A), len(fa), 'p', repr(p), 'U',u, 'medA',median(A.values()),'medF',median(fa.values()))
# MWU method='asymptotic' perhaps scipy default exact? let's check
PY"
