/bin/bash -lc "cd /workspace && python3 - <<'PY'
import sys
sys.path.insert(0,'/workspace/phykit203')
from phykit.services.alignment.rcv import RelativeCompositionVariability
from argparse import Namespace
from pathlib import Path
from scipy import stats

def vals_raw(d):
    return [RelativeCompositionVariability(Namespace(alignment=str(p))).calculate_rcv() for p in sorted(Path(d).glob('*.faa.mafft'))]
def vals_round(d):
    return [round(v,4) for v in vals_raw(d)]
A=vals_raw('/workspace/work_animals');F=vals_raw('/workspace/work_fungi')
Ar=vals_round('/workspace/work_animals');Fr=vals_round('/workspace/work_fungi')
print('raw p',stats.mannwhitneyu(A,F,alternative='two-sided'))
print('round p',stats.mannwhitneyu(Ar,Fr,alternative='two-sided'))
PY"
