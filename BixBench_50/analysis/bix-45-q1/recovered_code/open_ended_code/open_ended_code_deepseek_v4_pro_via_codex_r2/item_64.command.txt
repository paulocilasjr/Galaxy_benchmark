# produce final summary and save answer candidate? not yet
import sys
sys.path.insert(0,'/workspace/phykit203')
from phykit.services.alignment.rcv import RelativeCompositionVariability
from argparse import Namespace
from pathlib import Path
from scipy import stats
from collections import Counter

def compute(d):
    vals={}
    for p in sorted(Path(d).glob('*.faa.mafft')):
        raw=RelativeCompositionVariability(Namespace(alignment=str(p))).calculate_rcv()
        vals[p.name[:-len('.faa.mafft')]]=round(raw,4)
    return vals
A=compute('work_animals'); F=compute('work_fungi')
res=stats.mannwhitneyu(list(A.values()),list(F.values()),alternative='two-sided')
print('A count',len(A),'F count',len(F),'common',len(set(A)&set(F)))
print('p all scipy',format(res.pvalue,'.17g'))
print('U all',res.statistic)
print('p common scipy',format(stats.mannwhitneyu(list(A.values()),[F[g] for g in set(A)&set(F)],alternative='two-sided').pvalue,'.17g'))
print('A median/mean/min/max',__import__('statistics').median(A.values()),sum(A.values())/len(A),min(A.values()),max(A.values()))
print('F median/mean/min/max',__import__('statistics').median(F.values()),sum(F.values())/len(F),min(F.values()),max(F.values()))
# R-style continuity correct p
# calculate using scipy default (already)
PY
