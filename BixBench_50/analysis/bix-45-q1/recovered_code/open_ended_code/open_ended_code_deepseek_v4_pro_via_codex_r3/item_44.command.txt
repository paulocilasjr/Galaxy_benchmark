/bin/bash -lc "/workspace/tmp/venv203/bin/python - <<'PY'
import glob, os, json, argparse, statistics
from phykit.services.alignment.rcv import RelativeCompositionVariability
from scipy import stats
for grp in ['an','fu']:
 d=f'/workspace/tmp/scogs_{grp}'
 vals=[]
 for f in sorted(glob.glob(d+'/*.mafft')):
  a=argparse.Namespace(alignment=f)
  v=RelativeCompositionVariability(a).calculate_rcv()
  vals.append((os.path.basename(f),round(v,4),v))
 json.dump(vals, open(f'/workspace/tmp/{grp}_rcv_203.json','w'))
 print(grp,'n',len(vals),'median',statistics.median([x[1] for x in vals]),'mean',statistics.mean([x[1] for x in vals]),'minmax',min(x[1] for x in vals),max(x[1] for x in vals))
A=json.load(open('/workspace/tmp/an_rcv_203.json')); F=json.load(open('/workspace/tmp/fu_rcv_203.json'))
av=[x[1] for x in A]; fv=[x[1] for x in F]
for alt in ['two-sided','less','greater']:
 r=stats.mannwhitneyu(av,fv,alternative=alt,method='auto')
 print(alt,r.statistic,repr(r.pvalue))
print('raw p')
av=[x[2] for x in A]; fv=[x[2] for x in F]
r=stats.mannwhitneyu(av,fv,alternative='two-sided',method='auto'); print(r.statistic,repr(r.pvalue))
PY"
