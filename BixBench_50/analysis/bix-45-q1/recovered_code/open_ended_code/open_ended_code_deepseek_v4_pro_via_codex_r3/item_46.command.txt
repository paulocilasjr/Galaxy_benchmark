/bin/bash -lc "python - <<'PY'
import json, urllib.request
D=json.load(urllib.request.urlopen('https://pypi.org/pypi/phykit/json',timeout=20))
rel=D['releases']
def key(s):
 parts=s.split('.')
 return tuple(int(x) if x.isdigit() else 0 for x in parts)
for v in sorted(rel,key=key):
 fs=rel[v]
 if fs:
  t=fs[0].get('upload_time_iso_8601','')
  if t < '2025-09-01':
   print(v,t)
PY"
