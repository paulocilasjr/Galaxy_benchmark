/bin/bash -lc "python - <<'PY'
import subprocess, glob, os, json, sys
for grp in ['an','fu']:
 d=f'/workspace/tmp/scogs_{grp}'
 out=[]
 for f in sorted(glob.glob(d+'/*.mafft')):
  r=subprocess.run([sys.executable,'-m','phykit','rcv',f], capture_output=True, text=True, timeout=60)
  if r.returncode:
   print('ERR',grp,f,r.stderr); raise SystemExit
  out.append((os.path.basename(f), float(r.stdout.strip())))
 print(grp, len(out), out[:2])
 json.dump(out, open(f'/workspace/tmp/{grp}_rcv_cli.json','w'))
PY"
