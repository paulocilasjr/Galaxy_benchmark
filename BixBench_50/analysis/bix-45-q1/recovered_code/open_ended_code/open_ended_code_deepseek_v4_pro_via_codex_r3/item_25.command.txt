/bin/bash -lc "python - <<'PY'
import json
from scipy import stats
for typ in ['raw','rounded']:
 a=json.load(open('/workspace/tmp/an_rcv.json'))
 b=json.load(open('/workspace/tmp/fu_rcv.json'))
 idx=1 if typ=='raw' else 2
 av=[x[idx] for x in a]; fv=[x[idx] for x in b]
 for alt in ['two-sided','less','greater']:
  r=stats.mannwhitneyu(av,fv,alternative=alt, method='auto')
  print(typ, alt, r.statistic, r.pvalue)
 print('means',sum(av)/len(av),sum(fv)/len(fv))
PY"
