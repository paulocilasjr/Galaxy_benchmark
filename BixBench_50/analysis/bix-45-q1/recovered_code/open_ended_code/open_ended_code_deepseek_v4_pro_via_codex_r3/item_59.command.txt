/bin/bash -lc "/workspace/tmp/venv203/bin/python - <<'PY'
import argparse, glob, os
from phykit.services.alignment.rcv import RelativeCompositionVariability
from scipy import stats
vals={}
for grp in ['an','fu']:
    d=f'/workspace/tmp/scogs_{grp}'
    a=[]
    for f in sorted(glob.glob(d+'/*.mafft')):
        ns=argparse.Namespace(alignment=f)
        a.append(RelativeCompositionVariability(ns).calculate_rcv())
    vals[grp]=[round(v,4) for v in a]
    print(grp, len(a), 'median', __import__('statistics').median(vals[grp]))
r=stats.mannwhitneyu(vals['an'], vals['fu'], alternative='two-sided')
print('U=',r.statistic)
print('p=',repr(r.pvalue))
PY"
