/bin/bash -lc "python - <<'PY'
import json, urllib.request
for url in ['https://pypi.org/pypi/phykit/json']:
 d=json.load(urllib.request.urlopen(url,timeout=20))
 rel=d['releases']
 print('latest',d['info']['version'])
 # print release versions and upload times for relevant
 for v in sorted(rel.keys(), key=lambda s: [int(x) if x.isdigit() else 0 for x in s.split('.')])[-30:]:
  files=rel[v]
  if files:
   t=files[0].get('upload_time_iso_8601')
   print(v,t)
PY"
