/bin/bash -lc "python - <<'PY'
import json
from scipy import stats
A={x[0]:round(x[1],4) for x in json.load(open('/workspace/tmp/an_rcv.json'))}
F={x[0]:round(x[1],4) for x in json.load(open('/workspace/tmp/fu_rcv.json'))}
common=sorted(set(A)&set(F))
print('common',len(common))
av=[A[k] for k in common]; fv=[F[k] for k in common]
r=stats.mannwhitneyu(av,fv,alternative='two-sided')
print(r)
PY"
