import argparse, glob, os, json, time
from phykit.services.alignment.rcv import RelativeCompositionVariability
from scipy import stats
for grp in ['an','fu']:
 d=f'/workspace/tmp/scogs_{grp}'
 vals=[]

  vals.append((os.path.basename(f), v, round(v,4)))
 print(grp, len(vals), 'range raw', min(x[1] for x in vals), max(x[1] for x in vals), 'rounded range', min(x[2] for x in vals), max(x[2] for x in vals))
 print('first 10', vals[:3])
 # store maybe json for next use
 with open(f'/workspace/tmp/{grp}_rcv.json','w') as fh: json.dump(vals,fh)
PY"
