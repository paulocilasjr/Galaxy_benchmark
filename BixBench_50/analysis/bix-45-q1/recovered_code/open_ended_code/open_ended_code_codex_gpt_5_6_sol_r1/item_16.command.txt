/bin/bash -lc "python - <<'PY'
from pathlib import Path
from Bio import AlignIO
from collections import Counter

def old_rcv(p):
 aln=AlignIO.read(p,'fasta')
 L=aln.get_alignment_length(); n=len(aln)
 total=Counter(''.join(str(r.seq) for r in aln))
 avg={x:total[x]/n for x in total}
 vals=[]
 for r in aln:
  c=Counter(r.seq)
  vals.append(sum(abs(c[x]-avg[x]) for x in total)/(n*L))
 return sum(vals)

for group in ['animals','fungi']:
 vals=[]
 for p in sorted(Path('work_rcv',group).glob('*.mafft')):
  vals.append(round(old_rcv(p),4))
 Path('work_rcv',group+'.rcv203.txt').write_text('\\n'.join(map(str,vals))+'\\n')
 print(group,len(vals),min(vals),max(vals),sum(vals)/len(vals),vals[:5])
PY
python - <<'PY'
from pathlib import Path
from scipy.stats import mannwhitneyu
A=[float(x) for x in Path('work_rcv/animals.rcv203.txt').read_text().split()]
F=[float(x) for x in Path('work_rcv/fungi.rcv203.txt').read_text().split()]
for method in ['auto','asymptotic','exact']:
 for use_cont in [True,False]:
  r=mannwhitneyu(A,F,alternative='two-sided',method=method,use_continuity=use_cont)
  print(method,use_cont,'U=',r.statistic,'p=',format(r.pvalue,'.17g'))
for alt in ['less','greater']:
 r=mannwhitneyu(A,F,alternative=alt)
 print(alt,'U=',r.statistic,'p=',format(r.pvalue,'.17g'))
print('ties A,F,combined',len(A)-len(set(A)),len(F)-len(set(F)),len(A)+len(F)-len(set(A+F)))
PY"
