/bin/bash -lc 'work_tmp=$(mktemp -d /tmp/rcv_final.XXXXXX)
unzip -qq inputs/scogs_animals.zip -d "$work_tmp/animals"
unzip -qq inputs/scogs_fungi.zip -d "$work_tmp/fungi"
WORK_TMP="$work_tmp" python - <<'"'PY'
import contextlib, io, os, pathlib
from types import SimpleNamespace
from scipy.stats import mannwhitneyu
from phykit.services.alignment.rcv import RelativeCompositionVariability

root = pathlib.Path(os.environ['WORK_TMP'])
def cli_scores(domain):
    scores = {}
    for path in sorted((root / domain).glob('*.faa.mafft')):
        buf = io.StringIO()
        args = SimpleNamespace(alignment=str(path), json=False)
        with contextlib.redirect_stdout(buf):
            RelativeCompositionVariability(args).run()
        out = buf.getvalue().strip().splitlines()
        if len(out) "'!= 1:
            raise RuntimeError((path, out))
        scores[path.name.removesuffix('"'.faa.mafft')] = float(out[0])
    return scores
animals = cli_scores('animals')
fungi = cli_scores('fungi')
result = mannwhitneyu(list(animals.values()), list(fungi.values()), alternative='two-sided')
print('n_animals', len(animals), 'n_fungi', len(fungi))
print('u', float(result.statistic))
print('p', format(float(result.pvalue), '.17g'))
PY"
