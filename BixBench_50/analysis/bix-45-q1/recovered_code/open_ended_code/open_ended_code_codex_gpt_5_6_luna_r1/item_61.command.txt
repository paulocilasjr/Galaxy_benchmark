/bin/bash -lc "PYTHONPATH=/workspace/work/lib203:/workspace/work/lib python - <<'PY'
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO
from phykit.services.alignment.rcv import RelativeCompositionVariability
from scipy.stats import mannwhitneyu

class Args: pass

def cli_score(path):
    Args.alignment=str(path)
    b=StringIO()
    with redirect_stdout(b): RelativeCompositionVariability(Args()).run()
    out=b.getvalue().strip()
    assert out and '\\n' not in out, (path,out)
    return float(out)

A=[cli_score(p) for p in sorted(Path('work/animals').glob('*.faa.mafft'))]
F=[cli_score(p) for p in sorted(Path('work/fungi').glob('*.faa.mafft'))]
r=mannwhitneyu(A,F)
print('counts',len(A),len(F))
print('U',repr(float(r.statistic)))
print('p',repr(float(r.pvalue)))
print('first',A[:5],F[:5])
print('medians',sorted(A)[len(A)//2],sorted(F)[len(F)//2])
print('zero',sum(x==0 for x in A),sum(x==0 for x in F))
PY"
