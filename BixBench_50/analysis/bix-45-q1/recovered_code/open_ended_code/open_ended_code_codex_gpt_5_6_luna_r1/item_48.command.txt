/bin/bash -lc "PYTHONPATH=/workspace/work/lib python - <<'PY'
from pathlib import Path
from phykit.services.alignment.rcv import RelativeCompositionVariability
from scipy.stats import mannwhitneyu
class Args: json=False

def score(p): Args.alignment=str(p); return RelativeCompositionVariability(Args()).calculate_rcv()
A={p.stem.removesuffix('.faa'):score(p) for p in Path('work/animals').glob('*.faa.mafft')}
F={p.stem.removesuffix('.faa'):score(p) for p in Path('work/fungi').glob('*.faa.mafft')}
for name,keys in [('common',sorted(A.keys()&F.keys())),('all',sorted(A.keys()|F.keys()))]:
 aa=[A[k] for k in keys if k in A]; ff=[F[k] for k in keys if k in F]
 for lab,x,y in [('full',aa,ff),('round4',[round(z,4) for z in aa],[round(z,4) for z in ff])]:
  r=mannwhitneyu(x,y)
  print(name,lab,len(x),len(y),repr(float(r.statistic)),repr(float(r.pvalue)))
PY"
