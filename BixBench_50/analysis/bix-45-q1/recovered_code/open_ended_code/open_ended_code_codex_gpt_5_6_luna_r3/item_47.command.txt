/bin/bash -lc "wc -l -c codex_output/answer.txt; sed -n 'l' codex_output/answer.txt; PYTHONPATH=/workspace/.phykit203_tmp python - <<'PY'
import zipfile, tempfile, io, contextlib
from pathlib import Path
from types import SimpleNamespace
from scipy.stats import mannwhitneyu
from phykit.services.alignment.rcv import RelativeCompositionVariability
vals=[]
with tempfile.TemporaryDirectory(dir='/workspace') as d:
    groups=[]
    for label,zp in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
        scores=[]
        with zipfile.ZipFile(zp) as z:
            for name in sorted(z.namelist()):
                if not name.endswith('.faa.mafft'): continue
                p=Path(d)/(label+'_'+Path(name).name); p.write_bytes(z.read(name))
                obj=RelativeCompositionVariability(SimpleNamespace(alignment=str(p)))
                out=io.StringIO()
                with contextlib.redirect_stdout(out): obj.run()
                scores.append(float(out.getvalue()))
        groups.append(scores)
res=mannwhitneyu(*groups,alternative='two-sided',method='auto')
expected='7.696760829801303e-54'
print('n=',len(groups[0]),len(groups[1]),'U=',res.statistic,'p=',repr(float(res.pvalue)),'match=',format(float(res.pvalue),'.17g')==expected)
PY"
