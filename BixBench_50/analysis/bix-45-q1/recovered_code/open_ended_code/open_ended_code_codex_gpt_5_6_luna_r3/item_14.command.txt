import zipfile, os
for p in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(p) as z:
        names=z.namelist()
        ids=sorted({n[:-9] for n in names if n.endswith('.faa.mafft')})
        raws=sorted({n[:-4] for n in names if n.endswith('.faa')})
        print(p, 'aligned_ids',len(ids),'raw_ids',len(raws),'aligned_only',len(set(ids)-set(raws)),'raw_only',len(set(raws)-set(ids)))
        print('first',ids[:5],'last',ids[-5:])
PY"
