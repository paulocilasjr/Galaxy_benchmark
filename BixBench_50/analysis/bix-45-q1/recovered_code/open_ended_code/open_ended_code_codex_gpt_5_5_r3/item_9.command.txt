/bin/bash -lc "python - <<'PY'
import zipfile
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        mafft=sorted(n for n in f.namelist() if n.endswith('.faa.mafft'))
        faa=sorted(n for n in f.namelist() if n.endswith('.faa'))
    print(z)
    print('mafft count', len(mafft), 'faa count', len(faa))
    print('mafft sample', mafft[:10])
    print('faa sample', faa[:10])
PY"
