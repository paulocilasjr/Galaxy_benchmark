/bin/bash -lc ".venv/bin/python - <<'PY'
import subprocess, pathlib, json, math, statistics
from scipy.stats import mannwhitneyu

def rcv(path):
    p=subprocess.run(['.venv/bin/phykit','rcv',str(path),'--json'], text=True, capture_output=True, check=True)
    return float(json.loads(p.stdout)['rcv'])

data={}
for group in ['animals','fungi']:
    vals=[]
    files=sorted(pathlib.Path('work_rcv',group).glob('*.faa.mafft'))
    for i,path in enumerate(files,1):
        vals.append(rcv(path))
        if i % 50 == 0:
            print(group, i, flush=True)
    data[group]=vals
    print(group, 'count', len(vals), 'min', min(vals), 'median', statistics.median(vals), 'mean', statistics.mean(vals), 'max', max(vals), 'zeros', sum(v==0 for v in vals), flush=True)

x=data['animals']; y=data['fungi']
for kwargs in [
    {'alternative':'two-sided'},
    {'alternative':'two-sided','method':'asymptotic'},
    {'alternative':'two-sided','method':'asymptotic','use_continuity':True},
    {'alternative':'two-sided','method':'asymptotic','use_continuity':False},
    {'alternative':'less','method':'asymptotic'},
    {'alternative':'greater','method':'asymptotic'},
]:
    res=mannwhitneyu(x,y,**kwargs)
    print('mw', kwargs, 'U', res.statistic, 'p', format(res.pvalue,'.17g'))
PY"
