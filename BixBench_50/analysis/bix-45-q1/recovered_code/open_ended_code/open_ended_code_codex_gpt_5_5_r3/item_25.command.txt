/bin/bash -lc ".venv/bin/python - <<'PY'
import pathlib, subprocess, json, statistics
from scipy.stats import mannwhitneyu

def ids(group):
    return {p.name.replace('.faa.mafft',''):p for p in pathlib.Path('work_rcv',group).glob('*.faa.mafft')}
A=ids('animals'); F=ids('fungi')
Ak=set(A); Fk=set(F)
common=sorted(Ak & Fk)
print('animal only', len(Ak-Fk), sorted(Ak-Fk)[:20])
print('fungi only', len(Fk-Ak), sorted(Fk-Ak)[:20])
print('common', len(common))

def rcv(path):
    p=subprocess.run(['.venv/bin/phykit','rcv',str(path),'--json'], text=True, capture_output=True, check=True)
    return float(json.loads(p.stdout)['rcv'])
a=[]; f=[]
for i,k in enumerate(common,1):
    a.append(rcv(A[k])); f.append(rcv(F[k]))
    if i % 80 == 0:
        print('common', i, flush=True)
res=mannwhitneyu(a,f,alternative='two-sided',method='asymptotic')
print('common subset U', res.statistic, 'p', format(res.pvalue,'.17g'))
print('common medians', statistics.median(a), statistics.median(f))
PY"
