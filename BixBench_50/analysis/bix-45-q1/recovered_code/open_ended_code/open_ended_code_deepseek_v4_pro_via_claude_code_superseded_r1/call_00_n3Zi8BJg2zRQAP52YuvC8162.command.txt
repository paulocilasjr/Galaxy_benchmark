    animal_mafft = f"work_animals/{gene}.mafft"

    r1 = subprocess.run(['phykit', 'rcv', animal_mafft], capture_output=True, text=True)
    r2 = subprocess.run(['phykit', 'rcv', fungi_mafft], capture_output=True, text=True)
    
    try:
        animal_rcv2.append(float(r1.stdout.strip()))
    except:
        pass
    try:
        fungi_rcv2.append(float(r2.stdout.strip()))
    except:
        pass
    
    if (i+1) % 50 == 0:
