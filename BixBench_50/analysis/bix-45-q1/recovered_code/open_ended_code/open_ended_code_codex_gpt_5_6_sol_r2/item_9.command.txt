        if len({len(r.seq) for r in a}) "'!= 1: invalid.append(f.name)
    print(group, '"'files',len(files),'records',sorted(set(counts)),'length',min(lens),max(lens),'invalid',len(invalid))
    f=files[0]; a=AlignIO.read(f,'fasta'); print(f.name, [(r.id,len(str(r.seq).replace('-',''))) for r in a])

f={p.name.removesuffix('.faa.mafft') for p in Path('work_rcv/fungi').glob('*.mafft')}
