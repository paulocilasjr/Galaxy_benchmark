/bin/bash -lc ".venv/bin/python - <<'PY'
import json, urllib.request
with urllib.request.urlopen('https://pypi.org/pypi/phykit/json') as f: j=json.load(f)
rels=[]
for v,files in j['releases'].items():
 if files:
  t=min(x['upload_time_iso_8601'] for x in files)
  if t<'2025-08-29': rels.append((t,v))
for t,v in sorted(rels)[-15:]: print(t,v)
PY
.venv/bin/pip download --no-deps phykit==1.20.0 -d work_rcv/packages
