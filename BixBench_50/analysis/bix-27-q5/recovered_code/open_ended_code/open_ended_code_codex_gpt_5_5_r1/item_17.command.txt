/bin/bash -lc "Rscript - <<'RS'
expr_path <- 'inputs/ROSMAP_genexp_ad.csv'
meta_path <- 'inputs/ROSMAP_meta_ad.csv'
header <- strsplit(readLines(expr_path, n=1), ',', fixed=TRUE)[[1]]
expr_ids <- header[-1]
counts <- table(expr_ids)
collided <- names(counts[counts > 1])
keep <- "'!(expr_ids %in% collided)
cat('"'header sample ids total/unique:', length(expr_ids), length(unique(expr_ids)), '\\n')
cat('collided IDs/columns removed:', length(collided), sum("'!keep), '"'\\n')
meta <- read.csv(meta_path, check.names=FALSE, stringsAsFactors=FALSE)
cat('metadata rows/unique projid:', nrow(meta), length(unique(meta"'$projid)), '"'\\n')
# Summarize stable conflicts for the repeated groups for sanity.
conflict_cols <- c('msex','educ','apoe_genotype')
conflicted <- 0L
for (sid in collided) {
  rows <- meta[meta"'$projid == sid, , drop=FALSE]
  if (any(vapply(conflict_cols, function(col) length(unique(rows[[col]])) > 1, logical(1)))) conflicted <- conflicted + 1L
}
cat('"'collided groups with stable-attribute conflicts:', conflicted, '\\n')

expr <- read.csv(expr_path, check.names=FALSE, stringsAsFactors=FALSE, row.names=1)
X_gene_sample <- as.matrix(expr)
storage.mode(X_gene_sample) <- 'double'
cat('raw genes x columns:', paste(dim(X_gene_sample), collapse=' x '), '\\n')
X <- t(X_gene_sample[, keep, drop=FALSE])
cat('fitted samples x genes before filters:', paste(dim(X), collapse=' x '), '\\n')
cat('missing values:', sum(is.na(X)), '\\n')
if (anyNA(X)) stop('missing values present')
if (any(X < 0)) stop('negative expression values present')
X <- log10(X + 1)
vars <- apply(X, 2, var)
nonzero <- vars > 0
cat('zero variance genes removed:', sum("'!nonzero), '"'\\n')
X <- X[, nonzero, drop=FALSE]
Xc <- scale(X, center=TRUE, scale=FALSE)
s <- svd(Xc, nu=0, nv=0)"'$d
ratio <- s[1]''^2 / sum(s''^2)
cat('"'rank singular values:', length(s), 'requested components: 100\\n')
cat('pc1_fraction:', format(ratio, digits=17), '\\n')
cat('pc1_percent:', format(ratio*100, digits=17), '\\n')

# Diagnostic all-column value only, to make the duplicate-label decision visible.
Xall <- t(X_gene_sample)
Xall <- log10(Xall + 1)
Xall <- Xall[, apply(Xall, 2, var) > 0, drop=FALSE]
Xallc <- scale(Xall, center=TRUE, scale=FALSE)
sall <- svd(Xallc, nu=0, nv=0)"'$d
cat('"'diagnostic_all_columns_pc1_percent:', format((sall[1]"'^2/sum(sall''^2))*100, digits=17), '"'\\n')
RS"
