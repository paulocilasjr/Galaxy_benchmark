import csv
import numpy as np
from sklearn.decomposition import PCA
from collections import Counter

# ---- 1. Read expression matrix ----
with open('inputs/ROSMAP_genexp_ad.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    all_sample_ids = header[1:]
    raw_data = []
    for row in reader:
        raw_data.append([float(x) for x in row[1:]])

raw_matrix = np.array(raw_data, dtype=np.float64)  # 2456 genes x 222 samples

# ---- 2. Identity resolution: exclude all repeated-ID samples ----
id_counts = Counter(all_sample_ids)
repeated_ids = {k for k, v in id_counts.items() if v > 1}
keep_mask = np.array([sid not in repeated_ids for sid in all_sample_ids])
filtered = raw_matrix[:, keep_mask]  # 2456 genes x 178 samples

# ---- 3. Transpose: samples as rows, genes as columns ----
expr = filtered.T  # 178 samples x 2456 genes

# ---- 4. Log10(x + 1) transform ----
log_expr = np.log10(expr + 1.0)

# ---- 5. PCA with 100 components (center=True, no scaling) ----
pca = PCA(n_components=100)
pca.fit(log_expr)

# ---- 6. Extract PC1 % variance explained ----
pc1_pct = pca.explained_variance_ratio_[0] * 100

# Verification
total_var = pca.explained_variance_.sum() / pca.explained_variance_ratio_.sum()
print(f"PC1 explained variance ratio: {pca.explained_variance_ratio_[0]:.12f}")
print(f"PC1 % variance explained: {pc1_pct:.8f}%")

# Format for answer
print(f"\nAnswer: {pc1_pct:.4f}%")
