import csv
# Read header
with open('inputs/ROSMAP_genexp_ad.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    n_genes = sum(1 for _ in reader)
    
sample_ids = header[1:]  # skip 'gene'
print(f'Total genes: {n_genes}')
print(f'Total sample columns: {len(sample_ids)}')
print(f'Unique sample IDs: {len(set(sample_ids))}')

from collections import Counter
counts = Counter(sample_ids)
dups = {k:v for k,v in counts.items() if v > 1}
print(f'Number of repeated IDs: {len(dups)}')
print('Repeated IDs and their counts:')
for k,v in sorted(dups.items(), key=lambda x: -x[1]):
    print(f'  {k}: {v}')
