/bin/bash -lc "python -c 'import numpy as np, pandas as pd
from sklearn.decomposition import PCA
expr=pd.read_csv(\"/workspace/inputs/ROSMAP_genexp_ad.csv\")
meta=pd.read_csv(\"/workspace/inputs/ROSMAP_meta_ad.csv\")
ids=expr.columns[1:].to_numpy()
counts=pd.Series(ids).value_counts()
keep_ids=set(counts[counts==1].index)
keep_cols=[c for c in expr.columns[1:] if c in keep_ids]
A=expr[keep_cols].to_numpy(dtype=np.float64)
print(\"eligible_shape_features_by_samples\",A.shape,\"fitted_samples_by_features\",(A.T).shape)
print(\"raw_min_max\",A.min(),A.max(),\"missing\",np.isnan(A).sum(),\"nonfinite\",np.count_nonzero(~np.isfinite(A)))
X=np.log10(A.T+1.0)
print(\"log_min_max\",X.min(),X.max(),\"zero_var_features\",np.count_nonzero(X.var(axis=0)==0))
model=PCA(n_components=100,svd_solver=\"full\")
model.fit(X)
print(\"n_samples\",X.shape[0],\"n_features\",X.shape[1],\"components\",model.n_components_)
print(\"pc1_ratio\",repr(model.explained_variance_ratio_[0]))
print(\"pc1_percent\",repr(model.explained_variance_ratio_[0]*100.0))
print(\"sum_all_ratio_retained100\",repr(model.explained_variance_ratio_.sum()))
print(\"first5_ratios\",[repr(x) for x in model.explained_variance_ratio_[:5]])
# Direct centered SVD validation against all-rank total variance
Xc=X-X.mean(axis=0)
s=np.linalg.svd(Xc,full_matrices=False,compute_uv=False)
print(\"direct_pc1_percent\",repr((s[0]*s[0]/np.sum(s*s))*100.0))
print(\"total_var_sum\",repr(np.sum(Xc*Xc)),\"direct_rank\",np.count_nonzero(s>1e-12))
# Reference only: what naive all-column fit would produce
Xa=np.log10(expr.iloc[:,1:].to_numpy(dtype=np.float64).T+1.0)
ma=PCA(n_components=100,svd_solver=\"full\").fit(Xa)
print(\"all_columns_pc1_percent_reference\",repr(ma.explained_variance_ratio_[0]*100.0),\"all_shape\",Xa.shape)'"
