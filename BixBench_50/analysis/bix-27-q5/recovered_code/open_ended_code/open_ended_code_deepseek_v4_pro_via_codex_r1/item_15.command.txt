/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
# read raw header with duplicate names preserved; pandas mangles but numeric values okay.
expr=pd.read_csv('/workspace/inputs/ROSMAP_genexp_ad.csv', index_col=0)
print('expr shape', expr.shape)
X=expr.T.astype(float) # samples x genes 222 x 2456
Xlog=np.log10(X.values + 1.0)
# zero variance features after log
nonconst=np.var(Xlog, axis=0) > 0
print('nonconst genes', nonconst.sum(), 'zero variance', (~nonconst).sum())
Y=Xlog[:, nonconst]
print('Y shape', Y.shape)
# center features
Yc=Y - Y.mean(axis=0)
print('finite?', np.isfinite(Yc).all())
# use numpy svd
U,S,Vt=np.linalg.svd(Yc, full_matrices=False)
eig=S**2/(Y.shape[0]-1)
total=eig.sum()
print('rank/eigs count', len(eig), 'total variance', total)
print('svd first ratio', eig[0]/total*100)
print('sum first 100 ratio', eig[:100].sum()/total*100)
# sklearn verify
pca=PCA(n_components=100, svd_solver='full')
pca.fit(Y)
print('sklearn first', pca.explained_variance_ratio_[0]*100)
print('sklearn sum100', pca.explained_variance_ratio_.sum()*100)
print('sklearn total denom?', pca.explained_variance_.sum(), 'n_features', pca.n_features_in_)
# compare eigen first
print('sklearn ev first', pca.explained_variance_[0])
print('numpy eig first', eig[0])
PY"
