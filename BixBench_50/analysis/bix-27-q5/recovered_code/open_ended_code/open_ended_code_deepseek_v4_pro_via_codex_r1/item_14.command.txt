/bin/bash -c "python3 - <<'PY'
import csv, collections
meta_path='/workspace/inputs/ROSMAP_meta_ad.csv'
with open(meta_path, newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
by=collections.defaultdict(list)
for row in rows: by[row[0]].append(row)
dups={k:v for k,v in by.items() if len(v)>1}
for field in h[1:]:
    idx=h.index(field)
    conflict=[]
    same=[]
    for k,v in dups.items():
        vals=[row[idx] for row in v]
        if len(set(vals))>1: conflict.append((k,vals))
        else: same.append((k,vals))
    print(field, 'groups conflict', len(conflict), 'same', len(same))
    if conflict:
        print(' ', conflict)
PY"
