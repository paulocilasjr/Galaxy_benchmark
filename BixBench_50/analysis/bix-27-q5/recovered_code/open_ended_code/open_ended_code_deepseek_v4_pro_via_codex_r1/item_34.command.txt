/bin/bash -c "python3 - <<'PY'
import csv, collections, numpy as np
path='/workspace/inputs/ROSMAP_genexp_ad.csv'
with open(path, newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
ids=h[1:]
# transpose parse to columns floats
mat=np.array([[float(x) for x in row[1:]] for row in rows], dtype=float) # genes x samples
print('mat',mat.shape)
by=collections.defaultdict(list)
for j,name in enumerate(ids): by[name].append(j)
for name in sorted(by):
    if len(by[name])>1:
        cols=by[name]
        sub=mat[:,cols]
        exact=np.array_equal(sub[:,0], sub[:,1]) if len(cols)==2 else False
        # mean abs diff between first and others
        diffs=[]
        for k in range(1,len(cols)):
            d=np.abs(sub[:,0]-sub[:,k])
            diffs.append((d.mean(), d.max(), float(np.corrcoef(sub[:,0], sub[:,k])[0,1])))
        print(name, 'cols',cols, 'exact_pair0_1', exact, 'mean/max diff', [(round(a,3),round(b,3),round(c,5)) for a,b,c in diffs])
PY"
