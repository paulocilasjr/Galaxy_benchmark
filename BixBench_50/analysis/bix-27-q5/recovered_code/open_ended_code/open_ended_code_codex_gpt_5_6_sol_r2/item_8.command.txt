/bin/bash -lc "Rscript - <<'RS'
options(digits=17)
expr <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
meta <- read.csv('inputs/ROSMAP_meta_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
ids <- names(expr)[-1]
dup_ids <- unique(ids[duplicated(ids) | duplicated(ids, fromLast=TRUE)])
keep <- "'!(ids %in% dup_ids)
stopifnot(length(ids)==222L, length(unique(ids))==199L, sum(keep)==178L)
stopifnot(setequal(ids, meta$projid), all(table(ids)==table(meta$projid)))
raw <- as.matrix(expr[-1])
storage.mode(raw) <- '"'double'
stopifnot(all(is.finite(raw)), all(raw >= 0))
X <- t(log10(raw[, keep, drop=FALSE] + 1))
stopifnot(nrow(X)==178L, ncol(X)==2456L, all(is.finite(X)))
Xc <- scale(X, center=TRUE, scale=FALSE)
ss <- sum(Xc"'^2)
G <- tcrossprod(Xc)
ev <- eigen(G, symmetric=TRUE, only.values=TRUE)$values
ratio_gram <- ev[1] / ss
p <- prcomp(X, center=TRUE, scale.=FALSE, rank.=100)
ratio_prcomp <- p$sdev[1]''^2 / sum(apply(X,2,var))
cat('"'duplicate IDs:', paste(sort(dup_ids), collapse=','), '\\n')
cat('fitted dimensions:', paste(dim(X), collapse=' x '), '\\n')
cat('zero variance genes:', sum(apply(X,2,var)==0), '\\n')
cat('components requested: 100; returned scores:', ncol(p"'$x), '"'\\n')
cat('PC1 ratio Gram:', ratio_gram, '\\n')
cat('PC1 percent Gram:', 100*ratio_gram, '\\n')
cat('PC1 ratio prcomp:', ratio_prcomp, '\\n')
cat('absolute difference:', abs(ratio_gram-ratio_prcomp), '\\n')
# Diagnostic only: ordinary record-level result retaining ambiguous duplicates.
Xa <- t(log10(raw + 1)); Xac <- scale(Xa, center=TRUE, scale=FALSE)
eva <- eigen(tcrossprod(Xac), symmetric=TRUE, only.values=TRUE)"'$values
cat('"'all-records diagnostic PC1 percent:', 100*eva[1]/sum(Xac"'^2), '"'\\n')
RS"
