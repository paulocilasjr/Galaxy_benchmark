/bin/bash -lc "python - <<'PY'
import csv, collections, math
expr='inputs/ROSMAP_genexp_ad.csv'; meta='inputs/ROSMAP_meta_ad.csv'
with open(expr,newline='') as h:
    r=csv.reader(h); hd=next(r); samples=hd[1:]
    genes=[]; missing=0; nonfinite=0; invalid_log=0; nvals=0; minv=float('inf'); maxv=-float('inf')
    for row in r:
        genes.append(row[0])
        for s in row[1:]:
            nvals+=1
            if s=='': missing+=1; continue
            try: v=float(s)
            except: nonfinite+=1; continue
            if not math.isfinite(v): nonfinite+=1
            elif v <= -1: invalid_log+=1
            minv=min(minv,v); maxv=max(maxv,v)
with open(meta,newline='') as h:
    r=csv.DictReader(h); mr=list(r); mids=[x['projid'] for x in mr]
print('expression samples total unique duplicates:',len(samples),len(set(samples)),[(x,n) for x,n in collections.Counter(samples).items() if n>1])
print('genes total unique duplicate groups:',len(genes),len(set(genes)),sum(n>1 for n in collections.Counter(genes).values()))
print('metadata IDs total unique duplicates:',len(mids),len(set(mids)),[(x,n) for x,n in collections.Counter(mids).items() if n>1])
print('expression-not-meta:',sorted(set(samples)-set(mids)))
print('meta-not-expression:',sorted(set(mids)-set(samples)))
print('values:',nvals,'missing',missing,'nonfinite',nonfinite,'<=-1',invalid_log,'min',minv,'max',maxv)
PY
command -v R || true
python -m pip --version"
