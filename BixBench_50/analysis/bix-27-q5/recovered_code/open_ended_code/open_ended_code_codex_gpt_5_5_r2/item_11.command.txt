/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
with open('inputs/ROSMAP_meta_ad.csv', newline='') as f:
    rows=list(csv.DictReader(f))
g=defaultdict(list)
for i,row in enumerate(rows, start=2):
    g[row['projid']].append((i,row))
for pid, items in sorted(g.items()):
    if len(items)>1:
        print(pid)
        for line,row in items:
            print(' line', line, row)
PY"
