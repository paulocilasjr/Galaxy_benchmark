/bin/bash -lc "Rscript - <<'RS'
expr <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE)
ids <- names(expr)[-1]
M <- as.matrix(expr[, -1, drop=FALSE]); storage.mode(M) <- 'double'; rownames(M) <- expr[[1]]
calc <- function(M, label) {
  X <- t(log10(M + 1))
  vars <- apply(X, 2, var)
  p <- prcomp(X, center=TRUE, scale.=FALSE, rank.=100)
  pct <- p"'$sdev[1]''^2 / sum(vars) * 100
  cat(label, nrow(X), sprintf('"'%.15f', pct), '\\n')
}
calc(M, 'all_columns')
dup_ids <- names(which(table(ids) > 1))
calc(M[, "'!(ids %in% dup_ids), drop=FALSE], '"'exclude_duplicate_groups')
# Mean duplicate columns on raw scale by ID before log transform, if treating same IDs as one sample.
unique_ids <- unique(ids)
Mmean_raw <- sapply(unique_ids, function(id) rowMeans(M[, ids == id, drop=FALSE]))
calc(Mmean_raw, 'average_duplicates_raw_then_log')
# Mean duplicate columns after log transform, then PCA.
Xlog <- t(log10(M + 1))
Xmean_log <- t(sapply(unique_ids, function(id) colMeans(Xlog[ids == id, , drop=FALSE])))
vars <- apply(Xmean_log, 2, var)
p <- prcomp(Xmean_log, center=TRUE, scale.=FALSE, rank.=100)
cat('average_duplicates_after_log', nrow(Xmean_log), sprintf('%.15f', p"'$sdev[1]''^2 / sum(vars) * 100), '"'\\n')
RS"
