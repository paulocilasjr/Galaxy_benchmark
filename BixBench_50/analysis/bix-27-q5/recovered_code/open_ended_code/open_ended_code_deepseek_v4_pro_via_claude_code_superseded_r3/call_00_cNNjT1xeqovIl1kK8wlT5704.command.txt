import csv

# Read metadata
meta_rows = []
with open('/workspace/inputs/ROSMAP_meta_ad.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        meta_rows.append(row)

# Group by projid
from collections import defaultdict
meta_by_id = defaultdict(list)
for row in meta_rows:
    meta_by_id[row['projid']].append(row)

# Print repeated groups
for pid, rows in sorted(meta_by_id.items()):
    if len(rows) > 1:
        print(f"\n=== {pid} ({len(rows)} rows) ===")
        for i, r in enumerate(rows):
            print(f"  row {i}: msex={r['msex']}, educ={r['educ']}, apoe_genotype={r['apoe_genotype']}, "
                  f"cts_mmse30_lv={r['cts_mmse30_lv']}, braaksc={r['braaksc']}, "
                  f"ceradsc={r['ceradsc']}, cogdx={r['cogdx']}, dcfdx_lv={r['dcfdx_lv']}")
