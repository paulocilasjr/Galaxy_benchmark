import csv
import numpy as np
from sklearn.decomposition import PCA
from collections import defaultdict

# ---- 1. Read metadata and identify collisions vs aggregates ----
meta_rows = []
with open('/workspace/inputs/ROSMAP_meta_ad.csv') as f:
    reader = csv.DictReader(f)
    for row in reader:
        meta_rows.append(row)

meta_by_id = defaultdict(list)
for row in meta_rows:
    meta_by_id[row['projid']].append(row)

# Identify collision groups vs aggregate groups
collision_ids = set()
aggregate_ids = {}  # id -> list of row indices to average

for pid, rows in meta_by_id.items():
    if len(rows) <= 1:
        continue
    # Check invariant attributes: msex, educ, apoe_genotype
    invariant_keys = ['msex', 'educ', 'apoe_genotype']
    all_match = True
    for key in invariant_keys:
        vals = set(r[key] for r in rows)
        if len(vals) > 1:
            all_match = False
            break
    if all_match:
        aggregate_ids[pid] = len(rows)
    else:
        collision_ids.add(pid)

print("Collision IDs (exclude all occurrences):", sorted(collision_ids))
print("Aggregate IDs (average):", aggregate_ids)
print(f"Total collision IDs: {len(collision_ids)}")
print(f"Total aggregate IDs: {len(aggregate_ids)}")

# ---- 2. Read expression matrix ----
with open('/workspace/inputs/ROSMAP_genexp_ad.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    
sample_ids = header[1:]  # skip the 'gene' label
gene_names = []
data_rows = []
for row in reader:
    gene_names.append(row[0])
    data_rows.append([float(x) for x in row[1:]])

# Matrix: genes x samples
X_raw = np.array(data_rows)  # shape: (n_genes, n_samples)
n_genes, n_samples = X_raw.shape
print(f"\nExpression matrix: {n_genes} genes x {n_samples} samples")
print(f"Sample IDs (first 10): {sample_ids[:10]}")

# Build index map: sample_id -> list of column indices
sample_idx_map = defaultdict(list)
for i, sid in enumerate(sample_ids):
    sample_idx_map[sid].append(i)

print(f"Unique sample IDs in expression data: {len(sample_idx_map)}")

# ---- 3. Resolve duplicates in expression data ----
# Strategy:
# - Exclude all columns where sample_id is in collision_ids
# - Average columns where sample_id is in aggregate_ids
# - Keep single-occurrence samples as-is

cols_to_keep = []  # list of (sample_id, column_index_or_list)
for sid, indices in sorted(sample_idx_map.items()):
    if sid in collision_ids:
        # Exclude entirely
        continue
    elif sid in aggregate_ids:
        # Average the columns
        cols_to_keep.append((sid, indices))
    else:
        # Single occurrence
        assert len(indices) == 1, f"Expected 1 occurrence for {sid}, got {len(indices)}"
        cols_to_keep.append((sid, indices[0]))

print(f"\nSamples after resolution: {len(cols_to_keep)}")

# Build the resolved expression matrix (genes x resolved_samples)
resolved_samples = []
resolved_data = np.zeros((n_genes, len(cols_to_keep)))
for j, (sid, col_info) in enumerate(cols_to_keep):
    resolved_samples.append(sid)
    if isinstance(col_info, list):
        # Average the columns
        resolved_data[:, j] = X_raw[:, col_info].mean(axis=1)
    else:
        resolved_data[:, j] = X_raw[:, col_info]

print(f"Resolved matrix: {resolved_data.shape[1]} samples x {resolved_data.shape[0]} genes")

# ---- 4. Transpose: samples as rows, genes as columns ----
X = resolved_data.T  # shape: (n_samples, n_genes)
print(f"Transposed: {X.shape[0]} samples (rows) x {X.shape[1]} genes (columns)")

# ---- 5. log10(x + 1) transform ----
X_log = np.log10(X + 1)
print(f"After log10(x+1): min={X_log.min():.4f}, max={X_log.max():.4f}")

# ---- 6. Remove zero-variance genes ----
gene_vars = np.var(X_log, axis=0)
nzv_mask = gene_vars > 0
print(f"Zero-variance genes removed: {(~nzv_mask).sum()} out of {len(gene_vars)}")
X_filt = X_log[:, nzv_mask]
print(f"Filtered matrix: {X_filt.shape}")

# ---- 7. Center features (genes) ----
# Standard PCA: center the features (columns)
X_centered = X_filt - X_filt.mean(axis=0)
print(f"Centered: mean per gene approx 0, abs max mean: {np.abs(X_centered.mean(axis=0)).max():.2e}")

# ---- 8. PCA with 100 components ----
pca = PCA(n_components=100)
pca.fit(X_centered)

# ---- 9. Extract PC1 explained variance ratio ----
explained_var_ratios = pca.explained_variance_ratio_
pc1_var_pct = explained_var_ratios[0] * 100

print(f"\n===== RESULTS =====")
print(f"PC1 explained variance ratio: {explained_var_ratios[0]:.6f}")
print(f"PC1 percentage of total variance: {pc1_var_pct:.4f}%")
print(f"Sum of {len(explained_var_ratios)} PCs explained variance: {explained_var_ratios.sum():.4f}")
print(f"Total variance (sum of eigenvalues / n_features): {sum(pca.explained_variance_ / pca.explained_variance_ratio_[0] * pca.explained_variance_ratio_[0]) if False else 'N/A'}")

# Verify denominator: sum of all eigenvalues = total variance
# explained_variance_ratio_[i] = explained_variance_[i] / sum of all eigenvalues
# So sum of explained_variance_ is the total variance of retained components
print(f"Sum of explained_variance_: {pca.explained_variance_.sum():.4f}")
print(f"Total variance (trace of covariance matrix): {np.var(X_centered, axis=0).sum():.4f}")
