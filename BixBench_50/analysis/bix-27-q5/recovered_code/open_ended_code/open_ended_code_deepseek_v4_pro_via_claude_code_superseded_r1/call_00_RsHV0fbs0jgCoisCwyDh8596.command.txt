source /workspace/venv/bin/activate && python3 << 'EOF'
import pandas as pd
import numpy as np
from sklearn.decomposition import PCA

# Load the gene expression matrix
df = pd.read_csv("inputs/ROSMAP_genexp_ad.csv")

# Set gene names as index
df = df.set_index("gene")

# df is now genes x samples. Transpose to samples x genes.
df_t = df.T  # samples as rows, genes as columns

print(f"Shape after transpose (samples x genes): {df_t.shape}")

# Apply log10 transformation with pseudocount of 1
log_expr = np.log10(df_t.values + 1)

print(f"Log-transformed matrix shape: {log_expr.shape}")

# Check for any NaN or inf values
print(f"Any NaN: {np.any(np.isnan(log_expr))}")
print(f"Any Inf: {np.any(np.isinf(log_expr))}")

# Run PCA with 100 components
pca = PCA(n_components=100)
pca.fit(log_expr)

# Get the explained variance ratio for the first component
evr_pc1 = pca.explained_variance_ratio_[0]
pct = evr_pc1 * 100

print(f"\nPC1 explained variance ratio: {evr_pc1}")
print(f"PC1 percentage of total variance: {pct}%")

# Also print cumulative for first few
cumsum = np.cumsum(pca.explained_variance_ratio_)
print(f"\nFirst 10 components explained variance (%):")
for i in range(10):
    print(f"  PC{i+1}: {pca.explained_variance_ratio_[i]*100:.4f}% (cumulative: {cumsum[i]*100:.4f}%)")

print(f"\nTotal variance explained by 100 PCs: {cumsum[-1]*100:.4f}%")
EOF
