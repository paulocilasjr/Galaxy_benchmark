/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
from pathlib import Path
expr=Path('inputs/ROSMAP_genexp_ad.csv')
meta=Path('inputs/ROSMAP_meta_ad.csv')
with expr.open(newline='') as f:
    rd=csv.reader(f); hdr=next(rd)
    genes=[]; widths=Counter()
    for row in rd:
        genes.append(row[0]); widths[len(row)]+=1
samples=hdr[1:]
sc=Counter(samples); gc=Counter(genes)
print('expression_shape_genes_samples',len(genes),len(samples))
print('row_width_counts',widths)
print('sample_total_unique',len(samples),len(sc))
print('repeated_sample_ids',[(k,v) for k,v in sc.items() if v>1])
print('gene_total_unique',len(genes),len(gc))
print('repeated_gene_count',sum(v>1 for v in gc.values()),'examples',[(k,v) for k,v in gc.items() if v>1][:30])
with meta.open(newline='') as f:
    rows=list(csv.DictReader(f))
mc=Counter(r['projid'] for r in rows)
print('metadata_columns',list(rows[0]))
print('metadata_total_unique',len(rows),len(mc))
print('metadata_repeated_ids',[(k,v) for k,v in mc.items() if v>1])
print('expr_vs_meta_multiset_equal',sc==mc)
by=defaultdict(list)
for i,r in enumerate(rows, start=2): by[r['projid']].append((i,r))
for sid,n in sc.items():
    if n>1:
        print('\\nGROUP',sid,'expr_n',n,'meta_n',len(by[sid]))
        for lineno,row in by[sid]: print('line',lineno,row)
PY"
