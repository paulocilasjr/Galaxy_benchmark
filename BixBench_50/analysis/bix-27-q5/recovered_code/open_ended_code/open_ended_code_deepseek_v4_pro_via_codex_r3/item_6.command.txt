/bin/bash -c "python3 - <<'PY'
import csv, collections
with open('inputs/ROSMAP_meta_ad.csv', newline='') as f:
    rows=list(csv.reader(f))
h=rows[0]
data=rows[1:]
groups=collections.defaultdict(list)
for r in data: groups[r[0]].append(r)
for pid in sorted(groups):
    if len(groups[pid])>1:
        print('\\n', pid)
        for r in groups[pid]: print(r)
        for j in range(1,len(h)):
            vals=[r[j] for r in groups[pid]]
            if len(set(vals))>1:
                print(' DIFF', h[j], vals)
PY"
