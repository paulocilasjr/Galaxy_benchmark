/bin/bash -c "python3 - <<'PY'
import csv, collections, numpy as np
from sklearn.decomposition import PCA
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    r=csv.reader(f); h=next(r); rows=list(r)
X=np.array([[float(x) for x in row[1:]] for row in rows], float).T
ids=h[1:]
cnt=collections.Counter(ids)
keep=[i for i,pid in enumerate(ids) if cnt[pid]==1]
Y=np.log10(X[keep]+1)
print('n', Y.shape, 'kept ids unique', len(set(ids[i] for i in keep)), 'excluded total', len(ids)-len(keep))
pca=PCA(n_components=100, svd_solver='full')
pca.fit(Y)
print('ratio0', pca.explained_variance_ratio_[0]*100)
print('full precision', repr(pca.explained_variance_ratio_[0]*100))
# R-like check
Yc=Y-Y.mean(axis=0)
tot=(Yc**2).sum()/(Y.shape[0]-1)
u,s,vt=np.linalg.svd(Yc, full_matrices=False)
print('svd pct', s[0]**2/(Y.shape[0]-1)/tot*100)
print('denom total var', tot)
print('first5', (s[:5]**2/(Y.shape[0]-1))/tot*100)
PY"
