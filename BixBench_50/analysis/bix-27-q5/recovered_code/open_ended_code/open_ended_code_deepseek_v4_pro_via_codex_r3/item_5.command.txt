/bin/bash -c "python3 - <<'PY'
import csv, collections
p='inputs/ROSMAP_genexp_ad.csv'
with open(p, newline='') as f:
    r=csv.reader(f)
    header=next(r)
print('header total', len(header), 'first', header[:5], 'last', header[-5:])
ids=header[1:]
print('sample cols', len(ids), 'unique', len(set(ids)))
cnt=collections.Counter(ids)
print('duplicate ids count', len([x for x,n in cnt.items() if n>1]))
print('duplicates:', sorted([(x,n) for x,n in cnt.items() if n>1]))
# metadata
mp='inputs/ROSMAP_meta_ad.csv'
with open(mp, newline='') as f:
    rows=list(csv.reader(f))
print('meta header', rows[0])
mids=[r[0] for r in rows[1:]]
print('meta rows', len(mids), 'unique', len(set(mids)))
mcnt=collections.Counter(mids)
print('meta dups', sorted([(x,n) for x,n in mcnt.items() if n>1]))
print('matrix ids not in meta', len(set(ids)-set(mids)), sorted(set(ids)-set(mids))[:20])
print('meta ids not in matrix', len(set(mids)-set(ids)), sorted(set(mids)-set(ids))[:20])
PY"
