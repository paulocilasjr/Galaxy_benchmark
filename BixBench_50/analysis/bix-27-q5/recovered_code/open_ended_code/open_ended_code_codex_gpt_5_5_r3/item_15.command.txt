/bin/bash -lc "Rscript - <<'RS'
expr_path <- 'inputs/ROSMAP_genexp_ad.csv'
meta_path <- 'inputs/ROSMAP_meta_ad.csv'

header <- strsplit(readLines(expr_path, n = 1), ',', fixed = TRUE)[[1]]
sample_ids <- header[-1]
counts <- table(sample_ids)
collided <- names(counts[counts > 1])
keep_positions <- which("'!(sample_ids %in% collided))
keep_ids <- sample_ids[keep_positions]

meta <- read.csv(meta_path, stringsAsFactors = FALSE, check.names = FALSE)
stopifnot(identical(sort(as.character(table(sample_ids))), sort(as.character(table(meta$projid)))))
# Stronger multiset check by label.
stopifnot(identical(table(sample_ids), table(meta$projid)[names(table(sample_ids))]))
stopifnot(length(keep_ids) == length(unique(keep_ids)))
stopifnot(all(table(meta$projid)[keep_ids] == 1))

expr <- read.csv(expr_path, row.names = 1, check.names = FALSE)
mat <- as.matrix(expr[, keep_positions, drop = FALSE])
storage.mode(mat) <- '"'double'
cat('genes', nrow(mat), 'unique_genes', length(unique(rownames(mat))), '\\n')
cat('original_samples', length(sample_ids), 'unique_primary_ids', length(unique(sample_ids)), '\\n')
cat('collided_ids', length(collided), 'excluded_records', length(sample_ids) - length(keep_ids), 'fitted_samples', length(keep_ids), '\\n')
cat('na_count', sum(is.na(mat)), 'min_value', min(mat, na.rm = TRUE), '\\n')

X <- t(log10(mat + 1))
vars <- apply(X, 2, var)
nonzero <- vars > 0
cat('zero_variance_genes', sum("'!nonzero), '"'\\n')
X <- X[, nonzero, drop = FALSE]
Xc <- scale(X, center = TRUE, scale = FALSE)
sv <- svd(Xc, nu = 0, nv = 0)"'$d
percent <- (sv[1]''^2 / sum(Xc''^2)) * 100
cat('"'pc1_percent', format(percent, digits = 17), '\\n')

dir.create('codex_output', showWarnings = FALSE)
writeLines(paste0(format(percent, digits = 12, scientific = FALSE, trim = TRUE), '%'), 'codex_output/answer.txt')
RS
cat codex_output/answer.txt"
