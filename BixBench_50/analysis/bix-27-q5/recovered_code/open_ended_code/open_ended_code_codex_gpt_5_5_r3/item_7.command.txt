/bin/bash -lc "python - <<'PY'
import csv, collections, json, math
from pathlib import Path
expr=Path('inputs/ROSMAP_genexp_ad.csv')
meta=Path('inputs/ROSMAP_meta_ad.csv')
with expr.open(newline='') as f:
    reader=csv.reader(f)
    header=next(reader)
cols=header[1:]
print('expr sample cols', len(cols), 'unique', len(set(cols)))
dups={k:v for k,v in collections.Counter(cols).items() if v>1}
print('expr duplicate groups', len(dups), dups)
with meta.open(newline='') as f:
    rows=list(csv.DictReader(f))
meta_ids=[r['projid'] for r in rows]
print('meta rows', len(rows), 'unique', len(set(meta_ids)))
mdups={k:v for k,v in collections.Counter(meta_ids).items() if v>1}
print('meta duplicate groups', len(mdups), mdups)
print('expr-meta count equality', collections.Counter(cols)==collections.Counter(meta_ids))
# print repeated metadata rows grouped
fields=rows[0].keys()
for sid in sorted(set(dups)|set(mdups), key=lambda x:int(x[2:])):
    group=[r for r in rows if r['projid']==sid]
    print('\\n', sid, 'expr_count', dups.get(sid,1), 'meta_count', len(group))
    for r in group:
        print({k:r[k] for k in fields})
PY"
