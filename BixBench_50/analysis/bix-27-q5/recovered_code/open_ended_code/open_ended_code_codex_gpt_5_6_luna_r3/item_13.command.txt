/bin/bash -lc "Rscript - <<'RS'
exp <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
X <- as.matrix(exp[-1])
storage.mode(X) <- 'double'
X <- t(log10(X + 1))
fit <- prcomp(X, center=TRUE, scale.=FALSE, rank.=100)
Xc <- scale(X, center=TRUE, scale=FALSE)
sv_full <- svd(Xc, nu=0, nv=0)"'$d
ratio_from_full <- fit$sdev[1]''^2 / sum(sv_full''^2)
ratio_top100 <- fit$sdev[1]''^2 / sum(fit$sdev''^2)
cat(sprintf('"'prcomp_components=%d\\n', length(fit"'$sdev)))
cat(sprintf('"'ratio_using_total_variance=%.15f\\n', ratio_from_full))
cat(sprintf('percentage_using_total_variance=%.12f%%\\n', ratio_from_full*100))
cat(sprintf('ratio_if_100_component_sum_denominator=%.15f\\n', ratio_top100))
RS"
