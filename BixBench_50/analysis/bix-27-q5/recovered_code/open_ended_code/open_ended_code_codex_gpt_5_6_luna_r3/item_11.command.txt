/bin/bash -lc "Rscript - <<'RS'
exp <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
X <- as.matrix(exp[-1])
storage.mode(X) <- 'double'
X <- log10(X + 1)
X <- t(X)
Xc <- scale(X, center=TRUE, scale=FALSE)
sv <- svd(Xc, nu=0, nv=0)"'$d
ratio1 <- sv[1]''^2 / sum(sv''^2)
cat(sprintf('"'n_samples=%d n_genes=%d rank=%d\\n', nrow(X), ncol(X), sum(sv > 0)))
cat(sprintf('first_singular_value=%.17g\\n', sv[1]))
cat(sprintf('first_variance_ratio=%.17g\\n', ratio1))
cat(sprintf('first_percentage=%.17g\\n', ratio1 * 100))
cat(sprintf('top100_cumulative_percentage=%.17g\\n', sum(sv[1:100]"'^2)/sum(sv''^2)*100))
RS'
