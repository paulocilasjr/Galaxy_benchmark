/bin/bash -lc "Rscript - <<'RS'
exp <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
meta <- read.csv('inputs/ROSMAP_meta_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
ids <- names(exp)[-1]
counts <- table(ids)
rep_ids <- names(counts[counts > 1])
cat('repeated IDs and counts:\\n')
for (id in rep_ids) {
  idx <- which(ids == id)
  diffs <- sapply(idx[-1], function(j) max(abs(as.numeric(exp[[j]]) - as.numeric(exp[[idx[1]]]))))
  midx <- which(as.character(meta[[1]]) == id)
  cat(id, 'expr_cols=', paste(idx, collapse=','), 'meta_rows=', paste(midx, collapse=','), 'max_expr_diffs=', paste(diffs, collapse=','), '\\n')
}
RS"
