/bin/bash -lc "Rscript - <<'RS'
meta <- read.csv('inputs/ROSMAP_meta_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
ids <- as.character(meta[[1]])
for (id in names(table(ids)[table(ids)>1])) {
  rows <- meta[ids == id, , drop=FALSE]
  cat('\\n', id, 'rows', paste(which(ids == id), collapse=','), '\\n', sep='')
  print(rows, row.names=FALSE)
}
RS"
