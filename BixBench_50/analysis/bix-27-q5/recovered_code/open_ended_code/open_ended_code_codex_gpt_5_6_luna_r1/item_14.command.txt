/bin/bash -lc "Rscript - <<'RS'
expr <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
ids <- names(expr)[-1]
M <- as.matrix(expr[-1]); storage.mode(M) <- 'double'
collided <- names(table(ids))[table(ids) > 1]
keep <- "'!(ids %in% collided)
X <- t(log10(M[,keep,drop=FALSE] + 1))
# Explicit 100-component fit. The denominator is total centered variance over all fitted genes.
fit100 <- prcomp(X, center=TRUE, scale.=FALSE, rank.=100)
Xc <- sweep(X, 2, colMeans(X), '"'-')
total_ss <- sum(Xc"'^2)
pc1_ss <- sum(fit100$x[,1]''^2)
ratio <- pc1_ss/total_ss
cat('"'n_components returned:', ncol(fit100"'$x), '"'\\n')
cat('total centered SS:', sprintf('%.17g', total_ss), '\\n')
cat('PC1 SS:', sprintf('%.17g', pc1_ss), '\\n')
cat('PC1 percent:', sprintf('%.17g', 100*ratio), '\\n')
cat('prcomp PC1 percent using full total:', sprintf('%.17g', 100*fit100"'$sdev[1]''^2/(sum(Xc''^2)/(nrow(X)-1))), '"'\\n')
cat('finite output:', all(is.finite(fit100"'$x)), '"'\\n')
RS"
