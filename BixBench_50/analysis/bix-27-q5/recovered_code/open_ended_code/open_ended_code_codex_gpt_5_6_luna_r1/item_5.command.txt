/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter, defaultdict
from pathlib import Path

expr = Path('inputs/ROSMAP_genexp_ad.csv')
meta = Path('inputs/ROSMAP_meta_ad.csv')
with expr.open(newline='') as f:
    h = next(csv.reader(f))
expr_ids = h[1:]
with meta.open(newline='') as f:
    rows = list(csv.DictReader(f))
meta_ids = [r['projid'] for r in rows]

def dup(c):
    return {k:v for k,v in Counter(c).items() if v>1}
print('expression samples:', len(expr_ids), 'unique:', len(set(expr_ids)), 'duplicate labels:', dup(expr_ids))
print('metadata rows:', len(meta_ids), 'unique:', len(set(meta_ids)), 'duplicate labels:', dup(meta_ids))
print('expr IDs absent metadata:', sorted(set(expr_ids)-set(meta_ids)))
print('metadata IDs absent expression:', sorted(set(meta_ids)-set(expr_ids)))
print('repeated expression labels and metadata records:')
by = defaultdict(list)
for r in rows: by[r['projid']].append(r)
for k,n in sorted(dup(expr_ids).items()):
    print(k, 'expr_count=', n, 'meta_count=', len(by.get(k,[])), 'meta=', by.get(k,[]))
print('metadata columns:', list(rows[0]) if rows else [])
print('all metadata rows:')
for r in rows:
    print(','.join(r.values()))
PY"
