/bin/bash -lc "Rscript - <<'RS'
expr <- read.csv('inputs/ROSMAP_genexp_ad.csv', check.names=FALSE, stringsAsFactors=FALSE, row.names=NULL)
meta <- read.csv('inputs/ROSMAP_meta_ad.csv', check.names=FALSE, stringsAsFactors=FALSE)
ids <- names(expr)[-1]
genes <- expr[[1]]
M <- as.matrix(expr[-1])
storage.mode(M) <- 'double'
cat('raw genes x samples:', nrow(M), 'x', ncol(M), '\\n')
cat('sample labels unique:', length(unique(ids)), 'of', length(ids), '\\n')
cat('raw finite:', all(is.finite(M)), 'min:', min(M), 'max:', max(M), 'zeros:', sum(M==0), '\\n')
counts <- table(ids)
collided <- names(counts[counts > 1])
keep <- "'!(ids %in% collided)
X <- t(log10(M[, keep, drop=FALSE] + 1))
cat('"'collided labels:', length(collided), 'occurrences:', sum("'!keep), '"'\\n')
cat('fit samples x genes:', nrow(X), 'x', ncol(X), '\\n')
cat('fit finite:', all(is.finite(X)), 'zero-var genes:', sum(apply(X, 2, var) == 0), '\\n')
# Full PCA, then take PC1 over the total variance of the complete fitted matrix.
fit <- prcomp(X, center=TRUE, scale.=FALSE)
ratios <- fit"'$sdev''^2 / sum(fit$sdev''^2)
cat('"'PC1 fraction:', sprintf('%.17g', ratios[1]), '\\n')
cat('PC1 percent:', sprintf('%.17g', 100*ratios[1]), '\\n')
cat('first 5 percent:', paste(sprintf('%.12g', 100*ratios[1:5]), collapse=', '), '\\n')
cat('first 100 percent sum:', sprintf('%.17g', sum(100*ratios[1:100])), '\\n')
# Independent covariance calculation (same ratio; covariance denominator cancels).
Xc <- sweep(X, 2, colMeans(X), '-')
sv <- svd(Xc, nu=0, nv=0)"'$d
ind <- 100*sv[1]''^2/sum(sv''^2)
cat('"'independent SVD percent:', sprintf('%.17g', ind), 'abs diff:', sprintf('%.3g', abs(ind-100*ratios[1])), '\\n')
# Diagnostic variants.
allX <- t(log10(M+1))
allfit <- prcomp(allX, center=TRUE, scale.=FALSE)
allrat <- allfit"'$sdev''^2/sum(allfit$sdev''^2)
cat('"'all 222 rows diagnostic percent:', sprintf('%.17g', 100*allrat[1]), '\\n')
first <- "'!duplicated(ids)
onefit <- prcomp(t(log10(M[,first,drop=FALSE]+1)), center=TRUE, scale.=FALSE)
onerat <- onefit$sdev''^2/sum(onefit$sdev''^2)
cat('"'one occurrence per ID diagnostic percent:', sprintf('%.17g', 100*onerat[1]), '\\n')
RS"
