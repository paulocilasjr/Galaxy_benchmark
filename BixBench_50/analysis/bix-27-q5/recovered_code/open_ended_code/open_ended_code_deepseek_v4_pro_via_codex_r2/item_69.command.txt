/bin/bash -c "python3 - <<'PY'
import csv, numpy as np
from sklearn.decomposition import PCA
with open('inputs/ROSMAP_genexp_ad.csv', newline='') as f:
    header=next(csv.reader(f))
    n_samples=len(header)-1
    X=np.array([[float(v) for v in row[1:]] for row in csv.reader(f)], dtype=float)
print('raw shape',X.shape,'samples',n_samples,'genes',X.shape[0],'NaN',np.isnan(X).any())
Y=np.log10(X.T + 1.0)  # samples x genes
print('Y',Y.shape,'finite',np.isfinite(Y).all(),'zero genes after log',np.sum(Y.var(axis=0)==0))
pca=PCA(n_components=100, svd_solver='full')
pca.fit(Y)
pc1=pca.explained_variance_ratio_[0]
total_ratio=pca.explained_variance_ratio_.sum()
print('PC1 ratio',repr(pc1),'percent',pc1*100)
print('sum first100 ratios',total_ratio)
print('PC1 eigenvalue',pca.explained_variance_[0],'sum eigenvalues all?',pca.explained_variance_.sum())
# denominator should be trace of centered covariance / n using np.cov ddof=0
Z=Y-Y.mean(axis=0)
denom=np.sum(Z*Z)/Y.shape[0]
print('denom trace',denom,'eig/denom',pca.explained_variance_[0]/denom)
# compare singular value squared/n
U,s,Vt=np.linalg.svd(Z,full_matrices=False)
print('svd first ratio',(s[0]**2/Y.shape[0])/denom)
print('all svd sum',np.sum(s**2/Y.shape[0])/denom)
PY"
