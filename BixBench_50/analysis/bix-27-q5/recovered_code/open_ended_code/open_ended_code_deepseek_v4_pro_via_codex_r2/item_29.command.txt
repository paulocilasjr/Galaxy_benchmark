/bin/bash -c "python3 - <<'PY'
import csv, numpy as np, collections
with open('inputs/ROSMAP_genexp_ad.csv') as f:
 h=next(csv.reader(f)); ids=h[1:]; X=np.array([[float(x) for x in row[1:]] for row in csv.reader(f)])
print(X.shape)
# log transform for correlation maybe Pearson
Xl=np.log10(X.T+1)
cnt=collections.Counter(ids)
groups={}
for i,s in enumerate(ids):
 groups.setdefault(s,[]).append(i)
print('duplicate groups correlations (log space)')
for s in sorted(groups, key=lambda x:int(x[2:])):
 if len(groups[s])>1:
  inds=groups[s]
  cors=[]
  for a in range(len(inds)):
   for b in range(a+1,len(inds)):
    cors.append(np.corrcoef(Xl[inds[a]],Xl[inds[b]])[0,1])
  print(s, inds, ['%.4f'%c for c in cors])
PY"
