/bin/bash -c "python3 - <<'PY'
import requests, re
s=requests.Session()
s.headers.update({'User-Agent':'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 Chrome/120 Safari/537.36','Referer':'https://datadryad.org/stash/dataset/doi:10.5061/dryad.ttdz08kxq'})
# get page and cookies
r=s.get('https://datadryad.org/stash/dataset/doi:10.5061/dryad.ttdz08kxq',timeout=30)
print('page',r.status_code,'cookies',s.cookies.get_dict())
# download individual readme
for url in ['https://datadryad.org/downloads/file_stream/672449','https://datadryad.org/api/v2/files/672449/download']:
 rr=s.get(url,timeout=30)
 print(url,rr.status_code,rr.url,len(rr.content),rr.text[:100].replace('\\n',' '))
# zip form token
m=re.search(r'name=\"authenticity_token\" value=\"(["'^"]+)"'"',r.text)
print('token',m.group(1) if m else None)
if m:
 data={'authenticity_token':m.group(1)}
 rr=s.post('https://datadryad.org/dataset/downloadZip/doi_10_5061_dryad_ttdz08kxq__v20210422.zip',data=data,timeout=60)
 print('zip',rr.status_code,rr.url,len(rr.content),rr.headers.get('content-type'))
PY"
