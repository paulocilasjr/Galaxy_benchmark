/bin/bash -c "python3 - <<'PY'
import requests,re
url='https://datadryad.org/stash/dataset/doi:10.5061/dryad.ttdz08kxq'
r=requests.get(url,headers={'User-Agent':'Mozilla/5.0'})
text=r.text
for pat in ['file_stream','download_resource','downloadZip','file_downloads','js-individual-dl']:
 print('\\nPAT',pat)
 for m in re.finditer(pat, text):
  s=max(0,m.start()-300); e=min(len(text),m.end()+500)
  print(text[s:e].replace('\\n',' ')[:900])
PY"
