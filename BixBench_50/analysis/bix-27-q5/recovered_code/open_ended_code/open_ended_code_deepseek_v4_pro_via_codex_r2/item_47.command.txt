/bin/bash -c "python3 - <<'PY'
import csv, numpy as np
from sklearn.decomposition import PCA
with open('inputs/ROSMAP_genexp_ad.csv') as f:
 h=next(csv.reader(f)); X=np.array([[float(x) for x in r[1:]] for r in csv.reader(f)])
Y=np.log10(X.T+1)
for scale in [False,True]:
 if scale:
  Z=(Y-Y.mean(axis=0))/Y.std(axis=0,ddof=0)
 else: Z=Y
 p=PCA(n_components=100, svd_solver='full'); p.fit(Z)
 print('scale',scale,'pc1',p.explained_variance_ratio_[0]*100,'sum',p.explained_variance_ratio_.sum())
PY"
