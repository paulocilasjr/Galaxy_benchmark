/bin/bash -lc "python - <<'PY'
import gseapy as gp
lib=gp.get_library(name='Reactome_2022',organism='Human')
term=[k for k in lib if k.startswith('TP53 Regulates Transcription Of Cell Cycle Genes')][0]
path=lib[term]
deg=['BAX','BTG2','CCNE1','CCNE2','CDKN1A','PCNA','PLK2','RGCC']
res=gp.enrich(gene_list=deg,gene_sets={term:path},background=list(set().union(*[set(v) for v in lib.values()])),outdir=None,no_plot=True,verbose=False)
print('term',term,'pathway genes',len(path))
print(res.results[['Term','Overlap','Genes']].to_string(index=False))
PY"
