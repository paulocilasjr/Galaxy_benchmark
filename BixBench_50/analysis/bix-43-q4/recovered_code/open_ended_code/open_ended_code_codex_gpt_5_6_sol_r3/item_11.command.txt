/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
import gseapy as gp
counts=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0).T
layout=pd.read_csv('inputs/sample_layout.csv')
layout['sid']=layout['SampleID'].str.replace('-','_',regex=False)
meta=layout.set_index('sid').loc[counts.index,['Group']]
print('input',counts.shape,'zero genes',(counts.sum(axis=0)==0).sum(),flush=True)
dds=DeseqDataSet(counts=counts,metadata=meta,design='~Group',refit_cooks=True,n_cpus=4,quiet=False)
dds.deseq2()
st=DeseqStats(dds,contrast=['Group','Cisplatin_IC50_CBD_IC50','DMSO'],alpha=0.05,n_cpus=4,quiet=False)
st.summary()
r=st.results_df
sig=r[(r.padj<0.05)&(r.log2FoldChange.abs()>=0.5)].copy()
ann=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t').set_index('ENSG')
sig=sig.join(ann,how='left')
genes=set(sig.gene_name.dropna())
lib=gp.get_library('Reactome_2022','Human')
key=[k for k in lib if k.startswith('TP53 Regulates Transcription Of Cell Cycle Genes ')][0]
hits=sorted(genes & set(lib[key]))
print('RESULT_SHAPE',r.shape,'PADJ_NONNA',r.padj.notna().sum(),'DEGS',len(sig),'UNIQUE_NAMES',len(genes),flush=True)
print('TARGET',key,'OVERLAP',f'{len(hits)}/{len(lib[key])}',flush=True)
print('HITS',','.join(hits),flush=True)
print('TARGET_TABLE',flush=True)
rr=r.join(ann,how='left')
print(rr[rr.gene_name.isin(lib[key])][['gene_name','log2FoldChange','pvalue','padj']].sort_values('gene_name').to_string(index=False),flush=True)
PY"
