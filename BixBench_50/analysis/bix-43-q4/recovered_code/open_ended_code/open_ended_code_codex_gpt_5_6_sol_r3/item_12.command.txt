/bin/bash -lc "python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
counts=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0).T
counts=counts.loc[:,counts.sum()>0]
layout=pd.read_csv('inputs/sample_layout.csv'); layout['sid']=layout.SampleID.str.replace('-','_',regex=False)
meta=layout.set_index('sid').loc[counts.index,['Group']]
print(counts.shape, flush=True)
dds=DeseqDataSet(counts=counts,metadata=meta,design='~Group',n_cpus=1,quiet=False,low_memory=True)
dds.fit_size_factors()
print('size factors done',dds.obs['size_factors'].tolist(),flush=True)
PY"
