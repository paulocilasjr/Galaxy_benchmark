/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
from scipy.stats import ttest_ind
from statsmodels.stats.multitest import multipletests
import gseapy
raw=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0)
cols=['3_1','3_2','3_3','9_1','9_2','9_3']; x=raw[cols].to_numpy(float)
lib=x.sum(0); logc=np.log2(x/lib*1e6+1)
lfc=logc[:,3:].mean(1)-logc[:,:3].mean(1)
p=ttest_ind(logc[:,3:],logc[:,:3],axis=1,equal_var=False,nan_policy='omit').pvalue
padj=np.full(p.shape,np.nan); ok=np.isfinite(p); padj[ok]=multipletests(p[ok],method='fdr_bh')[1]
ids=set(raw.index[(padj<.05)&(abs(lfc)>=.5)])
mp=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t'); syms=set(mp.set_index('ENSG').gene_name.reindex(list(ids)).dropna())
gs=gseapy.get_library(name='Reactome_2022',organism='Human'); term=set(gs[[k for k in gs if k.startswith('TP53 Regulates Transcription Of Cell Cycle Genes')][0]])
print(len(ids),len(syms&term),sorted(syms&term),f'{len(syms&term)}/{len(term)}')
PY"
