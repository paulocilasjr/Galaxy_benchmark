/bin/bash -lc "python3 - <<'PY'
import pandas as pd, gseapy
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
raw=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0)
cols=list(raw.columns)
# 10 condition levels, three replicates each; use all samples, contrast combo vs DMSO.
levels=['Untreated','Untreated','Untreated','Serum_starvation','Serum_starvation','Serum_starvation','DMSO','DMSO','DMSO','DMSO_Serum_starvation','DMSO_Serum_starvation','DMSO_Serum_starvation','Cisplatin_IC50','Cisplatin_IC50','Cisplatin_IC50','Cisplatin_IC50_Serum_starvation_16h','Cisplatin_IC50_Serum_starvation_16h','Cisplatin_IC50_Serum_starvation_16h','CBD_IC50','CBD_IC50','CBD_IC50','CBD_IC50_Serum_starvation_16h','CBD_IC50_Serum_starvation_16h','CBD_IC50_Serum_starvation_16h','CBD_cisplatin','CBD_cisplatin','CBD_cisplatin','CBD_cisplatin_Serum_starvation_16h','CBD_cisplatin_Serum_starvation_16h','CBD_cisplatin_Serum_starvation_16h']
# Note count group 9 is named combo; use only group label to avoid slash in contrast.
md=pd.DataFrame({'condition':levels},index=cols)
counts=raw.astype(int).T
dds=DeseqDataSet(counts=counts,metadata=md,design_factors='condition',refit_cooks=True,n_cpus=1)
dds.deseq2()
st=DeseqStats(dds,contrast=['condition','CBD_cisplatin','DMSO'],n_cpus=1)
st.summary()
res=st.results_df
ids=set(res.index[(res.padj<0.05)&(res.log2FoldChange.abs()>=0.5)])
mp=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t')
syms=set(mp.set_index('ENSG').gene_name.reindex(list(ids)).dropna())
gs=gseapy.get_library(name='Reactome_2022',organism='Human')
name=[k for k in gs if k.startswith('TP53 Regulates Transcription Of Cell Cycle Genes')][0]
term=set(gs[name])
print('all30 DEGs',len(ids),'mapped syms',len(syms),'overlap',len(syms&term),sorted(syms&term),'ratio',f'{len(syms&term)}/{len(term)}')
# print target results
print(res.loc[[i for i in res.index if i in set(mp[mp.gene_name.isin(term)].ENSG)]].sort_values('padj').to_string())
PY"
