/bin/bash -lc "/tmp/mamba/bin/micromamba run -p /tmp/renv Rscript - <<'RS'
suppressMessages(library(DESeq2))
layout <- read.csv('inputs/sample_layout.csv', stringsAsFactors=FALSE)
counts <- read.csv('inputs/counts_raw_unfiltered.csv', row.names=1, check.names=FALSE)
layout"'$SampleID_us <- gsub('"'-', '_', layout"'$SampleID)
sub <- layout[layout$Group %in% c('"'DMSO','Cisplatin_IC50_CBD_IC50'),]
rownames(sub) <- sub"'$SampleID_us
sub$condition <- factor(sub$Group, levels=c('"'DMSO','Cisplatin_IC50_CBD_IC50'))
cts <- counts[, sub"'$SampleID_us, drop=FALSE]
cts <- as.matrix(cts)
storage.mode(cts) <- '"'integer'
print(dim(cts))
print(table(sub"'$condition))
dds <- DESeqDataSetFromMatrix(countData=cts, colData=sub, design=~condition)
dds <- DESeq(dds, quiet=TRUE)
res <- results(dds, alpha=0.05)
cat('"'results dim', nrow(res), '\\n')
cat('num nonNA padj', sum("'!is.na(res$padj)), '"'\\n')
cat('padj<.05', sum(res"'$padj < .05, na.rm=TRUE), '"'\\n')
cat('sig abs lfc >= .5', sum(res"'$padj < .05 & abs(res$log2FoldChange) >= .5, na.rm=TRUE), '"'\\n')
write.csv(as.data.frame(res), '/tmp/desq2_results.csv', row.names=TRUE)
cat('saved\\n')
RS"
