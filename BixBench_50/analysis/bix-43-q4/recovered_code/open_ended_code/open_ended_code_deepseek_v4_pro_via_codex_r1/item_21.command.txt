/bin/bash -c "python - <<'PY'
import gseapy, pandas as pd, time, sys
symbols=pd.read_csv('/tmp/deg_symbols.txt', header=None)[0].tolist()
print('n genes', len(symbols))
t=time.time()
try:
    enr = gseapy.enrichr(gene_list=symbols, gene_sets='Reactome_2022', organism='human', outdir=None, cutoff=1.0)
    print('done', time.time()-t)
    res=enr.results
    print(type(res), res.shape)
    print(res.columns.tolist())
    print(res.head().to_string())
    # search pathway exact/contain
    mask=res['Term'].str.contains('TP53 Regulates Transcription Of Cell Cycle Genes', case=False, na=False)
    print('matches', mask.sum())
    if mask.sum(): print(res.loc[mask].to_string(index=False))
    enr.results.to_csv('/tmp/gseapy_enrichr_results.csv', index=False)
except Exception as e:
    print('ERROR', type(e), e)
    raise
PY"
