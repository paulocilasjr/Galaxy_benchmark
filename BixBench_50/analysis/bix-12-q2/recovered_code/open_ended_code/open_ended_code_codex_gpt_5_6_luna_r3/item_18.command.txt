/bin/bash -lc "python - <<'PY'
from collections import Counter
from statistics import median
from zipfile import ZipFile

def parse(b):
    rec=[]; cur=None
    for line in b.decode().splitlines():
        if line.startswith('>'):
            cur=[line[1:].strip(), []]; rec.append(cur)
        elif line.strip():
            cur[1].append(line.strip())
    return [(n,''.join(p)) for n,p in rec]

def calc(ss):
    L=len(ss[0][1]); allp=nogapp=complete=comp_p=0; chars=set()
    for col in zip(*(s for _,s in ss)):
        chars.update(col)
        c=Counter(col)
        allp += sum(v>=2 for v in c.values())>=2
        obs=[x for x in col if x not in {'-','?','X'}]
        cn=Counter(obs)
        nogapp += sum(v>=2 for v in cn.values())>=2
        if len(obs)==len(col):
            complete += 1
            comp_p += sum(v>=2 for v in cn.values())>=2
    return L,len(ss),allp,nogapp,complete,comp_p,chars
rows=[]
with ZipFile('inputs/scogs_fungi.zip') as z:
    for n in z.namelist():
        if n.endswith('.faa.mafft'):
            rows.append((n,calc(parse(z.read(n)))))
print('n',len(rows),'seqcounts',Counter(s[1] for _,s in rows),'chars',sorted(set().union(*(s[-1] for _,s in rows))))
for label, fun in [
 ('gap-as-state/all-columns',lambda s:100*s[2]/s[0]),
 ('gaps-missing/all-columns',lambda s:100*s[3]/s[0]),
 ('gaps-missing/complete-columns',lambda s:100*s[5]/s[4]),
]:
 vals=sorted((fun(s),n,s[0],s[3],s[4]) for n,s in rows)
 print(label,'median',median(x[0] for x in vals),'middle',vals[127], 'zeros',sum(x[0]==0 for x in vals))
 print('around',*[x[:2] for x in vals[125:130]])
PY"
