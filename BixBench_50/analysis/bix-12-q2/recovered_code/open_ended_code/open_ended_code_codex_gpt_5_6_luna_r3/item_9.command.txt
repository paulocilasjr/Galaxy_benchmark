/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
import statistics

path='inputs/scogs_fungi.zip'

def read_fasta(data):
    seqs=[]; name=None; parts=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(parts)))
            name=line[1:].strip(); parts=[]
        else:
            parts.append(line.strip())
    if name is not None: seqs.append((name,''.join(parts)))
    return seqs

with ZipFile(path) as z:
    names=[n for n in z.namelist() if n.endswith('.faa.mafft')]
    stats=[]
    for n in names:
        seqs=read_fasta(z.read(n))
        lens={len(s) for _,s in seqs}
        chars=Counter(c for _,s in seqs for c in s)
        stats.append((n,len(seqs),min(lens),max(lens),''.join(sorted(chars))))
print('n alignments',len(stats))
print('sequence counts',Counter(x[1] for x in stats))
print('length range',min(x[2] for x in stats),max(x[3] for x in stats))
print('unequal lengths',sum(x[2]"'!=x[3] for x in stats))
print('"'characters overall',sorted(set(''.join(x[4] for x in stats))))
print('first 10 stats')
for x in stats[:10]: print(x)
PY"
