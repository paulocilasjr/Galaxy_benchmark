/bin/bash -lc "python - <<'PY'
from collections import Counter
from statistics import median
from zipfile import ZipFile

aligned = []
with ZipFile('inputs/scogs_fungi.zip') as archive:
    for name in archive.namelist():
        if not name.endswith('.faa.mafft'):
            continue
        records = []
        current = None
        for line in archive.read(name).decode('utf-8').splitlines():
            if line.startswith('>'):
                current = [line[1:].strip(), []]
                records.append(current)
            else:
                current[1].append(line.strip())
        sequences = [ ''.join(parts) for _, parts in records ]
        assert sequences and len({len(s) for s in sequences}) == 1, name
        length = len(sequences[0])
        informative = 0
        for column in zip(*sequences):
            observed = [state for state in column if state not in {'-', '?'}]
            counts = Counter(observed)
            if sum(count >= 2 for count in counts.values()) >= 2:
                informative += 1
        aligned.append(100.0 * informative / length)

assert len(aligned) == 255
answer = median(aligned)
ordered = sorted(aligned)
assert answer == ordered[len(ordered)//2]
print(f'alignment_count={len(aligned)}')
print(f'median={answer:.15f}%')
print(f'middle_index={len(ordered)//2}')
PY"
