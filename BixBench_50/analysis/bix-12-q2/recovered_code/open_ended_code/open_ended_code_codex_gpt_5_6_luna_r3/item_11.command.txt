/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

def fasta(b):
 out=[]
 for block in b.decode().strip().split('>')[1:]:
  ls=block.splitlines(); out.append((ls[0],''.join(ls[1:])))
 return out
with ZipFile('inputs/scogs_fungi.zip') as z:
 vals=[]
 for n in z.namelist():
  if n.endswith('.faa.mafft'):
   ss=fasta(z.read(n)); L=len(ss[0][1]); pis=0
   for col in zip(*(s for _,s in ss)):
    c=Counter(x for x in col if x not in '-?XxNn')
    if sum(v>=2 for v in c.values())>=2: pis+=1
   vals.append((len(ss),100*pis/L,n,L,pis))
for k in sorted(set(x[0] for x in vals)):
 a=[x[1] for x in vals if x[0]==k]
 print(k,len(a),median(a),min(a),max(a),sorted(a)[len(a)//2])
a=[x[1] for x in vals if x[0]>=4]
print('>=4',len(a),median(a),sorted(a)[len(a)//2-1:len(a)//2+1])
PY"
