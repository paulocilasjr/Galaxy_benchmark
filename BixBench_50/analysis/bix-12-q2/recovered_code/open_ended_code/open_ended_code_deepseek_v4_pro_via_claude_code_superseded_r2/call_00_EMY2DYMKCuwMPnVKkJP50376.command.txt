import os, glob

def read_fasta(filepath):
    sequences = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_header is not None:
                    sequences.append((current_header, ''.join(current_seq)))
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            sequences.append((current_header, ''.join(current_seq)))
    return sequences

# Check a few .faa files (non-mafft) to see if they're aligned
import random
faas = sorted(glob.glob('/workspace/work/scogs_fungi/*.faa'))

# Get non-mafft faa files
non_mafft = [f for f in faas if not f.endswith('.mafft')]

print(f'Total .faa files: {len(faas)}')
print(f'Total .mafft files: {len(maffts)}')
print(f'Non-mafft .faa files: {len(non_mafft)}')

# Check if non-mafft files are aligned
import random
sample = random.sample(non_mafft, min(10, len(non_mafft)))
for f in sample:
    seqs = read_fasta(f)
    lengths = [len(s[1]) for s in seqs]
    aligned = len(set(lengths)) == 1
    print(f'{os.path.basename(f)}: lengths={lengths}, aligned={aligned}, nseqs={len(seqs)}')
