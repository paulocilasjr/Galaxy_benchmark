/bin/bash -lc "python - <<'PY'
import zipfile, statistics, collections, re
zpath='inputs/scogs_fungi.zip'
with zipfile.ZipFile(zpath) as z:
    names=[n for n in z.namelist() if n.endswith('.mafft')]
    rec_counts=[]; lens=[]; chars=collections.Counter(); bad=[]
    examples=[]
    for n in names:
        text=z.read(n).decode()
        rec=[]; h=None; seq=[]
        for line in text.splitlines():
            if line.startswith('>'):
                if h is not None: rec.append((h,''.join(seq)))
                h=line[1:].strip(); seq=[]
            elif line.strip(): seq.append(line.strip())
        if h is not None: rec.append((h,''.join(seq)))
        rec_counts.append(len(rec)); lens.append(len(rec[0][1]) if rec else 0)
        if len({len(s) for _,s in rec}) "'!= 1: bad.append((n,[(h,len(s)) for h,s in rec]))
        for _,s in rec: chars.update(s)
        if len(examples)<4: examples.append((n,[(h,len(s),s.count('"'-')) for h,s in rec]))
    print('alignments',len(names))
    print('record counts',collections.Counter(rec_counts))
    print('lengths min/median/max',min(lens),statistics.median(lens),max(lens))
    print('bad length files',len(bad),bad[:3])
    print('chars',chars)
    print('examples')
    for x in examples: print(x)
PY"
