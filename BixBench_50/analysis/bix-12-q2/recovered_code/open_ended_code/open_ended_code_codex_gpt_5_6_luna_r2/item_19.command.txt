/bin/bash -lc "python - <<'PY'
import zipfile, statistics, collections, math

def parse(s):
    out=[]; h=None; buf=[]
    for line in s.decode().splitlines():
        if line.startswith('>'):
            if h is not None: out.append((h,''.join(buf)))
            h=line[1:].split()[0]; buf=[]
        elif line.strip(): buf.append(line.strip())
    if h is not None: out.append((h,''.join(buf)))
    return out

def metrics(seqs):
    L=len(seqs[0]); n=len(seqs)
    allstate=0; non_gap=0; strict=0; nongap_inform=0
    for col in zip(*seqs):
        c=collections.Counter(col)
        if len(c)>=2 and min(c.values())>=2: allstate += 1
        ng=[x for x in col if x not in '-.?']
        if len(ng)>0: non_gap+=1
        cn=collections.Counter(ng)
        if len(ng)==n and len(cn)>=2 and min(cn.values())>=2: strict+=1
        if len(cn)>=2 and min(cn.values())>=2: nongap_inform+=1
    return L, allstate, strict, nongap_inform, non_gap

with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    data=[]
    for name in z.namelist():
        if name.endswith('.mafft'):
            rec=parse(z.read(name)); m=metrics([s for _,s in rec])
            data.append((name,len(rec),*m))

print('n',len(data),'n4',sum(r[1]==4 for r in data),'n3',sum(r[1]==3 for r in data),'n2',sum(r[1]==2 for r in data))
for label,idx,den_idx,filterfn in [
    ('allstate / L',3,2,lambda r:True),
    ('strict / L',4,2,lambda r:True),
    ('nongap / L',5,2,lambda r:True),
    ('strict / complete',4,2,lambda r:True),
]:
    vals=[]
    for r in data:
        den=r[den_idx]
        if label=='strict / complete': den=r[6]
        vals.append(100*r[idx]/den if den else float('nan'))
    vals=sorted(vals)
    print(label,'median',statistics.median(vals),'p25,p75',statistics.quantiles(vals,n=4),'minmax',min(vals),max(vals))

# explicit useful variants
for desc, select, num, den in [
 ('all 255, residues/gaps states, L',lambda r:True,3,2),
 ('all 255, ignore gaps, L',lambda r:True,5,2),
 ('all 255, strict no gaps, L',lambda r:True,4,2),
 ('complete 211, strict no gaps, L',lambda r:r[1]==4,4,2),
 ('complete 211, strict no gaps, nongap columns',lambda r:r[1]==4,4,6),
 ('complete 211, ignore gaps, L',lambda r:r[1]==4,5,2),
 ('complete 211, ignore gaps, nongap columns',lambda r:r[1]==4,5,6),
]:
    vals=[100*r[num]/r[den] for r in data if select(r) and r[den]>0]
    print(desc, 'N',len(vals),'median',repr(statistics.median(vals)))

# show median-neighbor values sorted for likely variant
for desc, select, num, den in [
 ('all strict/L',lambda r:True,4,2),
 ('all ignore/L',lambda r:True,5,2),
 ('complete strict/L',lambda r:r[1]==4,4,2),
 ('complete strict/non-gap',lambda r:r[1]==4,4,6),
 ('all gapstate/L',lambda r:True,3,2),
]:
    vals=sorted((100*r[num]/r[den],r[0],r[1],r[2],r[3],r[4],r[5],r[6]) for r in data if select(r) and r[den]>0)
    mid=(len(vals)-1)//2
    print('\\n',desc,'N',len(vals))
    for x in vals[max(0,mid-2):min(len(vals),mid+4)]: print(x)
PY"
