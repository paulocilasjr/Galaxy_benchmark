import zipfile, collections
files=['Ncra','Scer','Spom','Umay']
sets={}
rows={}
for f in files:
  with zipfile.ZipFile(f'inputs/Fungi_{f}.busco.zip') as z:
    b=z.read('run_eukaryota_odb10/full_table.tsv').decode()
  d=collections.defaultdict(list)
  for line in b.splitlines():
    if not line or line.startswith('#') or line.startswith('Busco id'): continue
    p=line.split('\\t')
    d[p[0]].append(p[1])
  rows[f]=d
  sets[f]=set(d)
  print(f,'ids',len(d),'complete single',sum(v==['Complete'] for v in d.values()),'complete any',sum(all(s in ('Complete','Duplicated') for s in v) for v in d.values()),'missing/frag',sum(v==['Missing'] for v in d.values()),sum(v==['Fragmented'] for v in d.values()))
inter=set.intersection(*[set(k for k,v in rows[f].items() if v==['Complete']) for f in files])
print('intersection complete status',len(inter))
inter_sc=set.intersection(*[set(k for k,v in rows[f].items() if v==['Complete']) for f in files])
print('same',len(inter_sc))
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
 names=[n for n in z.namelist() if n.endswith('.mafft')]
 four=[]; less=[]
 for n in names:
  txt=z.read(n).decode(); ids=[]
  for line in txt.splitlines():
   if line.startswith('>'): ids.append(line[1:].split()[0])
  (four if len(ids)==4 else less).append(n.removesuffix('.faa.mafft'))
print('four',len(four),'less',len(less),'four_set_intersection',len(set(four)&inter),'extra_four',sorted(set(four)-inter)[:20],'missing_inter',sorted(inter-set(four))[:20])
print('less ids status')
for x in less:
 print(x,{f:rows[f].get(x) for f in files})
PY"
