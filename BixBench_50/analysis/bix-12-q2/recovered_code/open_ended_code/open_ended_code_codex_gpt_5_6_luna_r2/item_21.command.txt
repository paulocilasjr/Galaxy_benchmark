/bin/bash -lc "python - <<'PY'
import zipfile,collections,statistics

def parse(b):
    rec=[]; h=None; a=[]
    for l in b.decode().splitlines():
        if l.startswith('>'):
            if h is not None: rec.append((h,''.join(a)))
            h=l[1:].split()[0]; a=[]
        elif l.strip(): a.append(l.strip())
    if h is not None: rec.append((h,''.join(a)))
    return rec
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    stats=[]
    for n in z.namelist():
        if not n.endswith('.mafft'): continue
        r=parse(z.read(n)); cols=list(zip(*(s for _,s in r)))
        ng=[sum(c"'!='"'-' for c in col) for col in cols]
        stats.append((n,len(r),len(cols),ng.count(0),ng.count(1),ng.count(2),ng.count(3),ng.count(4)))
print('all-gap files',sum(x[3] for x in stats),'total all-gap cols',sum(x[3] for x in stats))
print('non-full columns across files',sum(sum(x[4:]) for x in stats))
print('n by count non-gap col distribution')
for k in range(2,9):
 print(k,collections.Counter(x[k] for x in stats).most_common(10))
print('example max all-gap',sorted(stats,key=lambda x:x[3],reverse=True)[:5])
PY"
