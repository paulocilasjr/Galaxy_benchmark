/bin/bash -lc "python - <<'PY'
import zipfile,statistics,collections

def parse(b):
 rec=[]; h=None; p=[]
 for l in b.decode().splitlines():
  if l.startswith('>'):
   if h is not None: rec.append(''.join(p))
   h=l[1:]; p=[]
  elif l.strip(): p.append(l.strip())
 if h is not None: rec.append(''.join(p))
 return rec
rows=[]
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
 for n in z.namelist():
  if not n.endswith('.mafft'): continue
  ss=parse(z.read(n)); L=len(ss[0]); pi=0; complete=0; observed=0; variable=0
  for col in zip(*ss):
   nongap=[c for c in col if c"'!='"'-']
   if len(nongap)==len(ss): complete+=1
   if nongap: observed+=1
   if len(set(nongap))>1: variable+=1
   if len(collections.Counter(nongap).values())>=2 and sum(v>=2 for v in collections.Counter(nongap).values())>=2: pi+=1
  rows.append((n,len(ss),L,pi,complete,observed,variable))
for selname,sel in [('all',lambda r:True),('4taxa',lambda r:r[1]==4)]:
 for denidx,name in [(2,'L'),(4,'complete columns'),(5,'observed columns'),(6,'variable columns')]:
  vals=[100*r[3]/r[denidx] if r[denidx] else 0 for r in rows if sel(r)]
  print(selname,name,len(vals),statistics.median(vals),min(vals),max(vals))
PY"
