/bin/bash -c "python - <<'PY'
import zipfile, statistics, collections, math
AA=set('ACDEFGHIKLMNPQRSTVWY')
path='inputs/scogs_fungi.zip'
with zipfile.ZipFile(path) as z:
 names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
 vals=[]
 rows=[]
 for n in names:
  data=z.read(n).decode(); seqs=[]; cur=None
  for line in data.splitlines():
   if line.startswith('>'):
    if cur is not None: seqs.append(''.join(cur))
    cur=[]
   else: cur.append(line.strip())
  if cur is not None: seqs.append(''.join(cur))
  L=len(seqs[0]); pis=0; variable=0; constant=0; singleton=0
  for col in zip(*seqs):
   counts=collections.Counter(c for c in col if c in AA)
   if counts:
    n_states=len(counts)
    if n_states>1: variable+=1
    else: constant+=1
    if sum(1 for c in counts if counts[c]>=2)>=2:
     pis+=1
    elif n_states>1: singleton+=1
   else: constant+=1
  pct=pis/L*100
  vals.append(pct)
  rows.append((n,len(seqs),L,pis,pct))
 print('median',statistics.median(vals),'mean',statistics.mean(vals),'min',min(vals),'max',max(vals))
 print('count',len(vals))
 print('first rows')
 for r in rows[:10]: print(r)
 print('files with 0',sum(1 for v in vals if v==0))
 print('non-zero median excluding zero',statistics.median([v for v in vals if v>0]))
 print('nonzero count',sum(v>0 for v in vals))
 # summary with all names
 for r in sorted(rows,key=lambda x:x[3])[:10]: print('low',r)
 for r in sorted(rows,key=lambda x:-x[3])[:10]: print('high',r)
PY"
