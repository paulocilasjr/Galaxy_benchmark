from zipfile import ZipFile
from collections import Counter
from statistics import median
P='inputs/scogs_fungi.zip'
def parse(t):
 d=[]; s=[]
 for line in t.splitlines():
  if line.startswith('>'):
   if s:d.append(''.join(s));s=[]
  else:s.append(line.strip().upper())
 if s:d.append(''.join(s))
 return d
rows=[]
with ZipFile(P) as z:
 for n in sorted(x.filename for x in z.infolist() if x.filename.endswith('.mafft')):
  ss=parse(z.read(n).decode()); L=len(ss[0]); pi=0

   c=Counter(x for x in col if x not in set('BJZX.*-?'))
   pi += sum(v>=2 for v in c.values())>=2
  rows.append((n,len(ss),L,pi,pi/L,round(pi/L,3)))
for nt in [None,4,3,2]:
 q=[x for x in rows if nt is None or x[1]==nt]
