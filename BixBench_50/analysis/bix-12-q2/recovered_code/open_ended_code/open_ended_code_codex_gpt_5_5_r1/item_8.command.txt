/bin/bash -lc "python - <<'PY'
import zipfile, collections
for zpath in ['inputs/scogs_fungi.zip','inputs/scogs_animals.zip']:
    with zipfile.ZipFile(zpath) as z:
        names=[n for n in z.namelist() if not n.endswith('/')]
    print(zpath, len(names), 'mafft', sum(n.endswith('.mafft') for n in names), 'faa_only', sum(n.endswith('.faa') for n in names))
    bases=collections.defaultdict(list)
    for n in names:
        b=n[:-6] if n.endswith('.mafft') else n
        bases[b].append(n)
    print('unique bases', len(bases), 'both', sum(len(v)>1 for v in bases.values()), 'mafft only', sum(len(v)==1 and v[0].endswith('.mafft') for v in bases.values()), 'faa only', sum(len(v)==1 and v[0].endswith('.faa') for v in bases.values()))
PY"
