/bin/bash -lc "python - <<'PY'
import zipfile, statistics, collections, math
from pathlib import Path

zpath=Path('inputs/scogs_fungi.zip')
missing=set('-?XxNn') # for protein, treat ambiguous/missing/gap as missing in this variant

def parse_fasta(data):
    seqs=[]; name=None; chunks=[]
    for raw in data.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                seqs.append((name,''.join(chunks)))
            name=line[1:]; chunks=[]
        else:
            chunks.append(line)
    if name is not None:
        seqs.append((name,''.join(chunks)))
    return seqs

def pi_count(seqs, missing_policy='ignore'):
    lens={len(s) for _,s in seqs}
    if len(lens)"'!=1: raise ValueError(lens)
    L=lens.pop(); pi=0
    for col in zip(*(s for _,s in seqs)):
        if missing_policy=='"'ignore':
            chars=[c.upper() for c in col if c not in '-?' and c.upper() not in {'X','N'}]
        elif missing_policy=='gap_missing_only':
            chars=[c.upper() for c in col if c not in '-?']
        elif missing_policy=='gap_state':
            chars=[c.upper() for c in col]
        counts=collections.Counter(chars)
        if sum(1 for v in counts.values() if v>=2)>=2:
            pi+=1
    return pi,L

with zipfile.ZipFile(zpath) as z:
    names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    rows=[]
    for n in names:
        seqs=parse_fasta(z.read(n))
        pi,L=pi_count(seqs,'ignore')
        rows.append((n,len(seqs),L,pi,pi/L*100))
print('n alignments', len(rows))
print('nseq distribution', collections.Counter(r[1] for r in rows))
print('length min max', min(r[2] for r in rows), max(r[2] for r in rows))
print('pi min max', min(r[3] for r in rows), max(r[3] for r in rows))
vals=[r[4] for r in rows]
print('median', statistics.median(vals))
print('median repr', repr(statistics.median(vals)))
print('middle two sorted', sorted(vals)[126:129])
# variants
for pol in ['gap_missing_only','gap_state']:
    vals2=[]
    with zipfile.ZipFile(zpath) as z:
        for n in names:
            pi,L=pi_count(parse_fasta(z.read(n)),pol)
            vals2.append(pi/L*100)
    print(pol, statistics.median(vals2), sorted(vals2)[126:129])
# print rows around median
for v in sorted(rows, key=lambda r:r[4])[123:132]:
    print(v)
PY"
