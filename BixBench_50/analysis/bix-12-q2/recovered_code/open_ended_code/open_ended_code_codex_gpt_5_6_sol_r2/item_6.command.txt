/bin/bash -lc "command -v phykit || true
command -v amas || true
command -v AMAS.py || true
python - <<'PY'
from zipfile import ZipFile
from collections import Counter
z=ZipFile('inputs/scogs_fungi.zip')
summary=Counter(); chars=Counter(); bad=[]
for n in z.namelist():
    if not n.endswith('.mafft'): continue
    rec=[]
    for line in z.read(n).decode().splitlines():
        if line.startswith('>'): rec.append([line[1:].strip(),''])
        else: rec[-1][1]+=line.strip()
    lens={len(s) for _,s in rec}
    summary[(len(rec),len(lens))]+=1
    chars.update(''.join(s for _,s in rec))
    if len(rec)"'!=4 or len(lens)!=1: bad.append((n,len(rec),lens))
print('"'alignment validation',summary,'bad',bad[:5])
print('character alphabet',chars)
PY"
