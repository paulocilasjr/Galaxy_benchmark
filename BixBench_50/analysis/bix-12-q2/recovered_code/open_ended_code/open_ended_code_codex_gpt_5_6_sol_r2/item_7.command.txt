/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median
z=ZipFile('inputs/scogs_fungi.zip')
rows=[]
for n in z.namelist():
    if not n.endswith('.mafft'): continue
    seq=[]
    for line in z.read(n).decode().splitlines():
        if line.startswith('>'): seq.append('')
        else: seq[-1]+=line.strip()
    L=len(seq[0])
    pis_missing=pis_gapstate=0
    variable_missing=0
    valid_cols=0
    for col in zip(*seq):
        counts=Counter(col)
        nongap=Counter(c for c in col if c"'!='"'-')
        pis_missing += sum(v>=2 for v in nongap.values()) >= 2
        pis_gapstate += sum(v>=2 for v in counts.values()) >= 2
        variable_missing += len(nongap)>=2
        valid_cols += '-' not in col
    rows.append((n,len(seq),L,pis_missing,pis_gapstate,100*pis_missing/L,100*pis_gapstate/L,valid_cols))
for ix,label in [(2,'length'),(3,'pis missing'),(4,'pis gapstate'),(5,'pct missing/total'),(6,'pct gapstate/total')]:
 print(label,'median',median(x[ix] for x in rows),'min',min(x[ix] for x in rows),'max',max(x[ix] for x in rows))
print('n',len(rows),'middle percentages missing')
for x in sorted(rows,key=lambda x:x[5])[124:131]: print(x)
print('middle gapstate')
for x in sorted(rows,key=lambda x:x[6])[124:131]: print(x)
print('median by taxa')
for nt in [2,3,4]:
 rr=[x for x in rows if x[1]==nt]
 print(nt,len(rr),median(x[5] for x in rr),median(x[6] for x in rr))
PY"
