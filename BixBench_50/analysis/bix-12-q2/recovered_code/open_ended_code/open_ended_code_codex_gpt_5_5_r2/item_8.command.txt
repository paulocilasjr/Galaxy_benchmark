/bin/bash -lc "python3 - <<'PY'
import zipfile, collections
zf=zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip')
names=[n for n in zf.namelist() if not n.endswith('/')]
print('members', len(names))
print('mafft', sum(n.endswith('.faa.mafft') for n in names))
print('faa_only', sum(n.endswith('.faa') for n in names))
base=collections.Counter(n.replace('.faa.mafft','').replace('.faa','') for n in names)
print('unique genes', len(base))
print('with both', sum(c==2 for c in base.values()), 'single', sum(c==1 for c in base.values()))
print('first names', names[:10])
for suffix in ['.faa.mafft','.faa']:
    lens=[]; nseqs=[]; bad=[]
    for n in names:
        if not n.endswith(suffix): continue
        seqs=[]; seq=''
        for line in zf.read(n).decode().splitlines():
            if line.startswith('>'):
                if seq: seqs.append(seq); seq=''
            elif line.strip():
                seq+=line.strip()
        if seq: seqs.append(seq)
        nseqs.append(len(seqs))
        ls=set(map(len,seqs))
        lens.append(next(iter(ls)) if len(ls)==1 else None)
        if len(ls)"'!=1: bad.append((n, sorted(ls), len(seqs)))
    print(suffix, '"'files', len(nseqs), 'nseq_counts', collections.Counter(nseqs).most_common(), 'bad unequal', len(bad), 'len min/median/max', (min(x for x in lens if x), sorted(x for x in lens if x)[len([x for x in lens if x])//2], max(x for x in lens if x)) if any(lens) else None)
    print('bad sample', bad[:3])
PY"
