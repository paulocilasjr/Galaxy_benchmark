import os
from collections import Counter

def parse_fasta_alignment(filepath):
    records = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_header is not None:
                    records.append((current_header, ''.join(current_seq)))
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            records.append((current_header, ''.join(current_seq)))
    return records

def count_pis_verbose(filepath):
    """Detailed count of PIS in an alignment with per-column info."""
    records = parse_fasta_alignment(filepath)
    seqs = [seq for _, seq in records]
    aln_length = len(seqs[0])
    gap_chars = set('-Xx.?*')
    
    pis_count = 0
    total_const = 0
    total_singleton = 0
    total_gap = 0
    
    for i in range(aln_length):
        column = [seq[i] for seq in seqs]
        valid = [c.upper() for c in column if c not in gap_chars]
        
        if len(valid) == 0:
            total_gap += 1
            continue
        
        counts = Counter(valid)
        
        if len(counts) == 1:
            total_const += 1
            continue
        
        # Check if parsimony informative
        is_pis = all(count >= 2 for count in counts.values())
        if is_pis:
            pis_count += 1
        else:
            total_singleton += 1
    
    return aln_length, pis_count, total_const, total_singleton, total_gap
