import os
from collections import Counter

def parse_fasta_alignment(filepath):
    records = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_header is not None:
                    records.append((current_header, ''.join(current_seq)))
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            records.append((current_header, ''.join(current_seq)))
    return records

mafft_dir = '/workspace/tmp_fungi'
files = sorted([f for f in os.listdir(mafft_dir) if f.endswith('.mafft')])
gap_chars = set('-Xx.?*')

percentages = []
for fname in files:
    fp = os.path.join(mafft_dir, fname)
    records = parse_fasta_alignment(fp)
    seqs = [seq for _, seq in records]
    aln_len = len(seqs[0])
    
    pis_count = 0
    for i in range(aln_len):
        column = [seq[i] for seq in seqs]
        valid = [c.upper() for c in column if c not in gap_chars]
        counts = Counter(valid)
        if len(counts) >= 2 and all(c >= 2 for c in counts.values()):
            pis_count += 1
    
    percentages.append((pis_count / aln_len) * 100.0)

percentages.sort()
n = len(percentages)

# Median of odd-length list: middle element
median = percentages[n // 2]

# Show values around median
mid = n // 2
print("Values around median:")
for i in range(max(0, mid-3), min(n, mid+4)):
    marker = " <-- MEDIAN" if i == mid else ""
    print(f"  [{i}] = {percentages[i]:.10f}%{marker}")

# Write final answer
os.makedirs('/workspace/codex_output', exist_ok=True)
with open('/workspace/codex_output/answer.txt', 'w') as f:
    f.write(f"{median}\n")

print(f"\nFinal median: {median}")
print(f"As percentage: {median:.4f}%")
print(f"Rational: PIS count is based on integer counts, so the percentage is rational.")
