/bin/bash -lc "python - <<'PY'
import zipfile
for zname in ['inputs/scogs_fungi.zip','inputs/scogs_animals.zip']:
    with zipfile.ZipFile(zname) as z:
        names=[n for n in z.namelist() if not n.endswith('/')]
        print(zname, 'total', len(names), 'mafft', sum(n.endswith('.mafft') for n in names), 'faa_only', sum(n.endswith('.faa') for n in names))
PY"
