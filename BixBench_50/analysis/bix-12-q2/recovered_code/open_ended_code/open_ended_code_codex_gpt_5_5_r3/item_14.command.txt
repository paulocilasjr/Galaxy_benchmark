/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse_fasta(text):
    rec=[]; name=None; chunks=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else: chunks.append(line)
    if name is not None: rec.append((name,''.join(chunks)))
    return rec
with ZipFile('inputs/scogs_fungi.zip') as z:
    dist=Counter(); examples=[]
    for name in sorted(n for n in z.namelist() if n.endswith('.faa.mafft')):
        rec=parse_fasta(z.read(name).decode())
        lens=tuple(sorted({len(s) for _,s in rec}))
        dist[(len(rec), lens)] += 1
        if len(rec)"'!=4 or len(lens)!=1:
            examples.append((name, [h for h,_ in rec], lens))
    print('"'dist')
    for k,v in sorted(dist.items()): print(k, v)
    print('examples')
    for e in examples[:20]: print(e)
PY"
