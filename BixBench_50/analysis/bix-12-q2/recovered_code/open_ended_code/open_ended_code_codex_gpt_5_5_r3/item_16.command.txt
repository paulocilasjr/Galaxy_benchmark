/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

zip_path='inputs/scogs_fungi.zip'

def parse_fasta(text):
    rec=[]; name=None; chunks=[]
    for raw in text.splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                rec.append((name,''.join(chunks).upper()))
            name=line[1:].split()[0]
            chunks=[]
        else:
            chunks.append(line)
    if name is not None:
        rec.append((name,''.join(chunks).upper()))
    return rec

def pi_count(seqs, gaps_missing=True):
    if not seqs: return 0,0
    lengths={len(s) for s in seqs}
    if len(lengths)"'!=1:
        raise ValueError(f'"'unequal lengths {lengths}')
    L=lengths.pop()
    pi=0
    for col in zip(*seqs):
        vals=[c for c in col if c not in '-?'] if gaps_missing else [c for c in col if c"'!='"'?']
        counts=Counter(vals)
        if sum(v>=2 for v in counts.values()) >= 2:
            pi += 1
    return pi,L

for gaps_missing in [True, False]:
    rows=[]
    with ZipFile(zip_path) as z:
        for name in sorted(n for n in z.namelist() if n.endswith('.faa.mafft')):
            rec=parse_fasta(z.read(name).decode('utf-8'))
            pi,L=pi_count([s for _,s in rec], gaps_missing)
            rows.append((name,len(rec),L,pi,pi/L*100 if L else 0.0))
    vals=[r[4] for r in rows]
    print('mode', 'gaps_missing' if gaps_missing else 'gap_state')
    print('n',len(rows),'median',median(vals),'zeros',sum(v==0 for v in vals),'min',min(vals),'max',max(vals))
    # print median bracketing for even/odd? n=255 odd
    sv=sorted(vals)
    print('middle_index_value', sv[len(sv)//2])
    print('taxon_counts', Counter(r[1] for r in rows))
PY"
