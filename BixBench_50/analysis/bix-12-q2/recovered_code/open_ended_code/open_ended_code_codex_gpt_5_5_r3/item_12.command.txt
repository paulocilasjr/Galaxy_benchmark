/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from statistics import median

zip_path='inputs/scogs_fungi.zip'
missing=set('-?X')  # just for testing below? don't include X maybe amino acid unknown, not missing generally? We'll compute variants.

def parse_fasta(text):
    rec=[]; name=None; chunks=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(chunks).upper()))
            name=line[1:].split()[0]; chunks=[]
        else:
            chunks.append(line)
    if name is not None: rec.append((name,''.join(chunks).upper()))
    return rec

def count_pi(seqs, mode):
    L=len(seqs[0])
    pi=0
    for col in zip(*seqs):
        if mode=='gaps_missing':
            vals=[c for c in col if c not in '-?']
        elif mode=='gaps_state':
            vals=[c for c in col if c"'!='"'?']
        elif mode=='gaps_x_missing':
            vals=[c for c in col if c not in '-?X']
        cnt=Counter(vals)
        if sum(1 for v in cnt.values() if v>=2)>=2:
            pi+=1
    return pi,L

for mode in ['gaps_missing','gaps_state','gaps_x_missing']:
    vals=[]; n=0; bad=[]
    with ZipFile(zip_path) as z:
        for name in sorted(n for n in z.namelist() if n.endswith('.faa.mafft')):
            rec=parse_fasta(z.read(name).decode())
            lens={len(s) for _,s in rec}
            if len(rec)"'!=4 or len(lens)!=1:
                bad.append((name,len(rec),lens)); continue
            pi,L=count_pi([s for _,s in rec], mode)
            vals.append(pi/L*100 if L else float('"'nan'))
            n+=1
    print(mode, 'n', n, 'bad', len(bad), 'median', median(vals), 'min', min(vals), 'max', max(vals))
    print(' first5', vals[:5])
PY"
