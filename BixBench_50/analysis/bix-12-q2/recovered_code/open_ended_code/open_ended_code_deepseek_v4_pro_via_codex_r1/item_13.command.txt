from collections import Counter
p=Path('/workspace/work/scogs_fungi')

    lens={len(''.join(s)) for n,s in seqs}
    cnt[(len(seqs), len(lens), tuple(sorted(lens)))]+=1
    if len(seqs)"'!=4 or len(lens)!=1:
        bad.append((f.name,len(seqs),lens))
print('"'files',sum(cnt.values()))
for k,v in cnt.items(): print(k,v)
