/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from math import exp
ws=load_workbook('inputs/data.xlsx',read_only=True,data_only=True).active
data=[(float(r[3]),1.0 if r[1]==1 else 0.0) for r in ws.iter_rows(min_row=2,values_only=True) if r[0] is not None]
b0=b1=0.0
for it in range(100):
    s0=s1=0.0; h00=h01=h11=0.0
    for x,y in data:
        eta=b0+b1*x
        p=1/(1+exp(-eta))
        w=p*(1-p)
        s0+=y-p; s1+=(y-p)*x
        h00+=w; h01+=w*x; h11+=w*x*x
    det=h00*h11-h01*h01
    d0=(h11*s0-h01*s1)/det
    d1=(-h01*s0+h00*s1)/det
    b0+=d0; b1+=d1
    if max(abs(d0),abs(d1))<1e-14: break
print('iterations',it+1)
print('intercept',format(b0,'.17g'))
print('age',format(b1,'.17g'))
print('score',format(s0,'.3g'),format(s1,'.3g'))
print('rows',len(data),'events',sum(y for _,y in data))
PY"
