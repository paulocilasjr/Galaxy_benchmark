/bin/bash -lc "unzip -l inputs/data.xlsx && python - <<'PY'
import zipfile, re
p='inputs/data.xlsx'
with zipfile.ZipFile(p) as z:
 for n in z.namelist():
  if n.endswith('.xml') or n.endswith('.rels'):
   b=z.read(n)
   txt=b.decode('utf-8','replace')
   hits=[]
   for pat in ['camrel','treatment','group','efficacy','response','药','卡瑞']:
    if re.search(pat,txt,re.I): hits.append(pat)
   if hits: print(n,hits)
PY"
