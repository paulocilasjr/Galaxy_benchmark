/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter, defaultdict
ws=load_workbook('/workspace/inputs/data.xlsx', read_only=True, data_only=True).active
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0])
# trim trailing none headers
cols=[i for i,h in enumerate(headers) if h is not None]
print('headers')
for i in cols: print(i, headers[i])
records=[]
for row in rows[1:]:
    if all(v is None for v in row): continue
    records.append({headers[i]:row[i] for i in cols})
print('n', len(records))
for col in ['Groups','Efficacy','Age','Family history','ECOG-PS','Child-Pugh','Chemotherapy experience','HBV infection','Gender','Toxic and side effect']:
    c=Counter(r[col] for r in records)
    print('\\n',col, c)
print('\\nEfficacy by Groups')
d=defaultdict(Counter)
for r in records: d[r['Groups']][r['Efficacy']]+=1
for k in sorted(d): print(k, d[k])
print('\\nAge by efficacy')
for eff in sorted(Counter(r['Efficacy'] for r in records), key=str):
    vals=[r['Age'] for r in records if r['Efficacy']==eff]
    print(eff, len(vals), min(vals), max(vals), sum(vals)/len(vals))
PY"
