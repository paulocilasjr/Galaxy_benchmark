/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import subprocess, textwrap
ws=load_workbook('/workspace/inputs/data.xlsx', read_only=True, data_only=True).active
headers=list(next(ws.iter_rows(values_only=True)))
idx={h:i for i,h in enumerate(headers) if h is not None}
ages=[]; ys=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    eff=row[idx['Efficacy']]
    age=row[idx['Age']]
    if eff is None or age is None or age == ' ': continue
    ages.append(float(age))
    ys.append(1 if eff=='PR' else 0)
rcode = f\"\"\"
age <- c({','.join(str(x) for x in ages)})
y <- c({','.join(str(y) for y in ys)})
fit <- glm(y ~ age, family=binomial())
print(coef(fit))
cat(unname(coef(fit)['age']), '\\\\n')
\"\"\"
res=subprocess.run(['Rscript','-'], input=rcode, text=True, capture_output=True)
print(res.stdout)
if res.returncode:
    print(res.stderr)
    raise SystemExit(res.returncode)
PY"
