/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter, defaultdict
ws=load_workbook('/workspace/inputs/data.xlsx', read_only=True, data_only=True).active
headers=list(next(ws.iter_rows(values_only=True)))
cols=[i for i,h in enumerate(headers) if h is not None]
records=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    rec={headers[i]:row[i] for i in cols}
    if rec['Sequence number'] is None and rec['Age'] is None and rec['Efficacy'] is None: continue
    records.append(rec)
print('n', len(records))
print('records with missing key')
for r in records:
    if r['Age'] in [None,' '] or r['Efficacy'] is None or r['Groups'] is None:
        print(r)
print('Efficacy by Groups')
d=defaultdict(Counter)
for r in records:
    d[r['Groups']][r['Efficacy']]+=1
for k in sorted([x for x in d if x is not None]): print(k, d[k])
PY"
