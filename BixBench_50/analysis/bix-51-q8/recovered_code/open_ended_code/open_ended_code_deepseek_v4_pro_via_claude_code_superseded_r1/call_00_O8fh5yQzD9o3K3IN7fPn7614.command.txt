import pandas as pd
from sklearn.linear_model import LogisticRegression
import numpy as np

df = pd.read_excel('inputs/data.xlsx')
data = df[df['Sequence number'].notna()].copy()
data['Age'] = pd.to_numeric(data['Age'], errors='coerce')
data['Response'] = (data['Efficacy'] == 'PR').astype(int)

X = data[['Age']].values
y = data['Response'].values

# sklearn with no regularization (C=1e10 effectively removes L2 penalty)
model = LogisticRegression(penalty=None, solver='lbfgs', max_iter=1000)
model.fit(X, y)

print('sklearn coefficients:')
print('Intercept:', model.intercept_[0])
print('Age coef:', model.coef_[0][0])

# Also try statsmodels with different method
import statsmodels.api as sm
X_sm = sm.add_constant(data['Age'])
model_sm = sm.Logit(y, X_sm)
result_sm = model_sm.fit(method='newton', disp=False)
print()
print('statsmodels (newton):')
print('Age coef:', result_sm.params['Age'])
print('Intercept:', result_sm.params['const'])

# Print all with high precision
print()
print('Full precision Age coefficient:')
print(f'{result_sm.params[\"Age\"]:.16f}')
