/bin/bash -c "python - <<'PY'
import openpyxl, statsmodels.api as sm
import numpy as np
ws=openpyxl.load_workbook('/workspace/inputs/data.xlsx', data_only=True).active
rows=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is not None and isinstance(r[0],(int,float)) and r[1] is not None and r[3] is not None:
        rows.append(r)
age=np.array([float(r[3]) for r in rows])
y=np.array([1.0 if r[1]==1 else 0.0 for r in rows])
X=sm.add_constant(age, has_constant='add')
model=sm.Logit(y,X).fit(disp=0)
print(model.summary())
print('params', model.params)
print('coef_age', model.params[1])
print('nobs',int(model.nobs))
print('llf',model.llf)
PY"
