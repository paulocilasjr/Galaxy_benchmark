/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
p='/workspace/inputs/data.xlsx'
wb=load_workbook(p,read_only=False,data_only=False)
print('props',wb.properties.title,wb.properties.subject,wb.properties.creator,wb.properties.description)
print('defined names',list(wb.defined_names))
for ws in wb.worksheets:
 print('sheet',ws.title,'state',ws.sheet_state,'freeze',ws.freeze_panes,'merged',list(ws.merged_cells.ranges))
 print('hidden rows',[i for i,d in ws.row_dimensions.items() if d.hidden][:20])
 print('hidden cols',[i for i,d in ws.column_dimensions.items() if d.hidden][:20])
 print('comments',[(c.coordinate,c.comment.text) for row in ws.iter_rows() for c in row if c.comment])
 print('tables',list(ws.tables))
PY"
