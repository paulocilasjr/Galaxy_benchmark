/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from decimal import Decimal, getcontext
getcontext().prec=80
ws=load_workbook('/workspace/inputs/data.xlsx',read_only=True,data_only=True).active
x=[]; y=[]
for row in ws.iter_rows(min_row=2,values_only=True):
    if row[0] is not None and isinstance(row[3],(int,float)):
        x.append(Decimal(str(row[3]))); y.append(Decimal(1 if row[2]=='PR' else 0))
b0=Decimal(0); b1=Decimal(0)
for it in range(100):
    g0=g1=h00=h01=h11=Decimal(0)
    for xi,yi in zip(x,y):
        z=b0+b1*xi
        ez=(-z).exp()
        p=Decimal(1)/(Decimal(1)+ez)
        w=p*(Decimal(1)-p)
        d=yi-p
        g0+=d; g1+=xi*d
        h00-=w; h01-=xi*w; h11-=xi*xi*w
    det=h00*h11-h01*h01
    d0=(-g0*h11+h01*g1)/det
    d1=(h01*g0-h00*g1)/det
    b0+=d0; b1+=d1
    if max(abs(d0),abs(d1)) < Decimal('1e-70'): break
print('iterations',it+1)
print('intercept',b0)
print('slope',b1)
# scores
s0=s1=Decimal(0)
for xi,yi in zip(x,y):
    p=Decimal(1)/(Decimal(1)+(-(b0+b1*xi)).exp())
    s0+=yi-p; s1+=xi*(yi-p)
print('scores',s0,s1)
PY"
