/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter,defaultdict
p='/workspace/inputs/data.xlsx'
ws=load_workbook(p,read_only=True,data_only=True).active
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0][:12])
data=[dict(zip(headers,row[:12])) for row in rows[1:] if row[0] is not None]
print('n',len(data))
for h in headers:
    c=Counter(d[h] for d in data)
    print('\\n',h,'nunique',len(c),c)
print('\\nGroup x efficacy')
ct=defaultdict(Counter)
for d in data: ct[d['Groups']][d['Efficacy']]+=1
for g,c in ct.items(): print(g,c)
print('\\nEfficacy x ages')
for e in sorted(set(d['Efficacy'] for d in data)):
    vals=[d['Age'] for d in data if d['Efficacy']==e]
    print(e,len(vals),'min',min(vals),'max',max(vals),'mean',sum(vals)/len(vals))
print('\\nRows all:')
for d in data:
    print(d)
PY"
