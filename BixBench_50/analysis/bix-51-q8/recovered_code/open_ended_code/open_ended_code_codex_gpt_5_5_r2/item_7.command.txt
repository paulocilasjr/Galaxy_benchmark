/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
import xml.etree.ElementTree as ET
from pathlib import Path
import re
p=Path('/workspace/inputs/data.xlsx')
ns={'a':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}

def col_index(cell_ref):
    m=re.match(r'([A-Z]+)', cell_ref)
    s=m.group(1)
    n=0
    for ch in s:
        n=n*26+ord(ch)-64
    return n-1
with ZipFile(p) as z:
    ss=[]
    root=ET.fromstring(z.read('xl/sharedStrings.xml'))
    for si in root.findall('a:si', ns):
        texts=[]
        for t in si.findall('.//a:t', ns):
            texts.append(t.text or '')
        ss.append(''.join(texts))
    sheet=ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
    rows=[]
    for row in sheet.findall('.//a:sheetData/a:row', ns):
        vals={}
        for c in row.findall('a:c', ns):
            ref=c.attrib['r']
            t=c.attrib.get('t')
            v=c.find('a:v', ns)
            val='' if v is None else v.text
            if t=='s' and val"'!='"'': val=ss[int(val)]
            elif t=='inlineStr':
                val=''.join(tn.text or '' for tn in c.findall('.//a:t', ns))
            vals[col_index(ref)]=val
        maxcol=max(vals) if vals else -1
        rows.append([vals.get(i,'') for i in range(maxcol+1)])
    width=max(len(r) for r in rows)
    rows=[r+['']*(width-len(r)) for r in rows]
    print('rows', len(rows), 'cols', width)
    for i,r in enumerate(rows[:15],1):
        print(i, r)
    print('last rows')
    for i,r in list(enumerate(rows,1))[-5:]:
        print(i,r)
PY"
