/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter, defaultdict
from math import sqrt
p='inputs/data.xlsx'
ws=load_workbook(p, data_only=True, read_only=True).active
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0][:12])
data=[dict(zip(headers, r[:12])) for r in rows[1:] if any(x is not None for x in r[:12])]
print('shape', len(data), len(headers))
print(headers)
for c in headers:
    vals=[d[c] for d in data]
    print('\\n',c,'nunique',len(set(vals)))
    for k,v in Counter(vals).most_common(): print(repr(k),v)
print('\\nCross Groups/Efficacy')
ct=defaultdict(Counter)
for d in data: ct[d['Groups']][d['Efficacy']]+=1
for g,c in sorted(ct.items()): print(g,dict(c))
print('\\nAge by Efficacy')
for e in sorted(set(d['Efficacy'] for d in data)):
    a=[d['Age'] for d in data if d['Efficacy']==e]
    mean=sum(a)/len(a); sd=sqrt(sum((x-mean)**2 for x in a)/(len(a)-1)) if len(a)>1 else 0
    print(e,len(a),mean,sd,min(a),max(a))
PY"
