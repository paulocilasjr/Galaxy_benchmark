    if (group==1) "'!= (efficacy=='"'PR'):

def sigmoid(z):
    if z >= 0:
        q=exp(-z)
        return 1.0/(1.0+q)
    q=exp(z)
    return q/(1.0+q)

def fit(xs, ys):
    b0=b1=0.0
    for _ in range(100):
        s0=s1=a00=a01=a11=0.0
        for x,y in zip(xs,ys):
            p=sigmoid(b0+b1*x)
            residual=y-p
            w=p*(1.0-p)
            s0 += residual
            s1 += residual*x
            a00 += w
            a01 += w*x
            a11 += w*x*x
        det=a00*a11-a01*a01
        if det <= 0.0:
            raise RuntimeError('Singular information matrix')
        d0=(a11*s0-a01*s1)/det
        d1=(-a01*s0+a00*s1)/det
        b0 += d0
        b1 += d1
        if max(abs(d0),abs(d1)) < 1e-13:
            break
    else:
        raise RuntimeError('MLE did not converge')
    score0=score1=0.0
    for x,y in zip(xs,ys):
        residual=y-sigmoid(b0+b1*x)
        score0 += residual
        score1 += residual*x
    if max(abs(score0),abs(score1)) > 1e-9:
        raise RuntimeError('Score check failed')
    return b0,b1
