/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
p='/workspace/inputs/data.xlsx'
ws=load_workbook(p, data_only=True, read_only=True).active
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0])
# retain cols with header and any nonempty data
idx=[i for i,h in enumerate(headers) if h is not None and any(r[i] is not None for r in rows[1:])]
print('data rows',len(rows)-1,'used cols',len(idx))
for i in idx:
    vals=[r[i] for r in rows[1:]]
    cnt=Counter(vals)
    print(f'{i+1}: {headers[i]"'!r} n={sum(v is not None for v in vals)} nunique={len([x for x in cnt if x is not None])}'"')
    print(' ',cnt.most_common(20))
print('\\nlast 15 relevant rows:')
for r in rows[-15:]: print(tuple(r[i] for i in idx))
PY"
