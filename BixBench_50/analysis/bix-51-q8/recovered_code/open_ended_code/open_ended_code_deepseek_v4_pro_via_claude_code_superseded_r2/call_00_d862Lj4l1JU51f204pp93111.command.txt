import openpyxl
wb = openpyxl.load_workbook('/workspace/inputs/data.xlsx')
ws = wb.active

# Get unique values for key columns
groups = set()
efficacy = set()
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    groups.add(row[1])
    efficacy.add(row[2])

print('Unique Groups:', sorted(groups, key=str))
print('Unique Efficacy:', sorted(efficacy, key=str))

# Count by group and efficacy
from collections import Counter
cross = Counter()
none_group_rows = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    cross[(row[1], row[2])] += 1
    if row[1] is None:
        none_group_rows.append(row)

print('Cross-tab:')
for k, v in sorted(cross.items(), key=lambda x: str(x[0])):
    print(f'  Group={k[0]}, Efficacy={k[1]}: {v}')

# None Group rows
print()
print('Rows with None group:', len(none_group_rows))
for r in none_group_rows[:5]:
    print(r[:4])
