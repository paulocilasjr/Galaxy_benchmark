import pandas as pd
import numpy as np
from scipy.optimize import minimize

df = pd.read_excel('inputs/data.xlsx')
df['Age_num'] = pd.to_numeric(df['Age'], errors='coerce')
df['Response'] = df['Efficacy'].map({'PR': 1, 'SD': 0, 'PD': 0})
complete = df[df['Age_num'].notna() & df['Response'].notna()].copy()

X_val = complete['Age_num'].values
y_val = complete['Response'].values

# Manual logistic regression via MLE (sanity check)
def neg_log_likelihood(beta):
    b0, b1 = beta
    logit = b0 + b1 * X_val
    p = 1 / (1 + np.exp(-logit))
    # avoid log(0)
    p = np.clip(p, 1e-15, 1 - 1e-15)
    ll = y_val * np.log(p) + (1 - y_val) * np.log(1 - p)
    return -np.sum(ll)

result = minimize(neg_log_likelihood, [0, 0], method='BFGS')
print('Manual MLE:')
print(f'Intercept: {result.x[0]:.10f}')
print(f'Age coefficient: {result.x[1]:.10f}')
print()

# Also verify with statsmodels
import statsmodels.api as sm
X_sm = sm.add_constant(complete[['Age_num']])
model = sm.Logit(y_val, X_sm)
fit = model.fit(disp=0)
print('Statsmodels:')
print(f'Age coefficient: {fit.params[\"Age_num\"]:.15f}')
print(f'OR: {np.exp(fit.params[\"Age_num\"]):.15f}')
