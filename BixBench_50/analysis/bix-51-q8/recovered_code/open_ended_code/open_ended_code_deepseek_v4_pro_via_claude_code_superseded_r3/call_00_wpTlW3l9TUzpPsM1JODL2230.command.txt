import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_excel('inputs/data.xlsx')

# Clean and prepare data
df['Age_num'] = pd.to_numeric(df['Age'], errors='coerce')
df['Response'] = df['Efficacy'].map({'PR': 1, 'SD': 0, 'PD': 0})

complete = df[df['Age_num'].notna() & df['Response'].notna()].copy()

X = complete[['Age_num']]
X = sm.add_constant(X)
y = complete['Response']

# Fit logistic regression
model = sm.Logit(y, X)
result = model.fit(disp=0)

print(result.summary())
print()
print('Coefficient for Age:', result.params['Age_num'])
print('Std error:', result.bse['Age_num'])
print('P-value:', result.pvalues['Age_num'])
print('95% CI:', result.conf_int().loc['Age_num'].tolist())
