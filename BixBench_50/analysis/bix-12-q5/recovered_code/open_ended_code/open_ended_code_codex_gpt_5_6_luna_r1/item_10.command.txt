/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
import re

path='inputs/scogs_animals.zip'

def parse_fasta(data):
    text=data.decode('ascii')
    recs=[]
    h=None; chunks=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if h is not None: recs.append((h,''.join(chunks)))
            h=line[1:].strip(); chunks=[]
        else: chunks.append(line)
    if h is not None: recs.append((h,''.join(chunks)))
    return recs

rows=[]
with ZipFile(path) as z:
    names=[n for n in z.namelist() if n.endswith('.faa.mafft')]
    for n in names:
        recs=parse_fasta(z.read(n))
        lens=Counter(len(s) for _,s in recs)
        taxa=[h for h,_ in recs]
        rows.append((n,len(recs),lens, taxa, Counter(c for _,s in recs for c in s)))

print('alignments',len(rows))
print('taxon-count distribution',Counter(r[1] for r in rows))
print('length distribution',Counter(next(iter(r[2])) for r in rows).most_common(10))
print('unequal-length alignments',sum(len(r[2])>1 for r in rows))
print('taxon-label distribution')
alltax=Counter(t for r in rows for t in r[3])
print(alltax)
print('character set across alignments',Counter(''.join(sorted(r[4])) for r in rows).most_common(10))
print('max gaps per alignment',max((r[4]['-'],r[0]) for r in rows))
for r in sorted(rows,key=lambda x: (-next(iter(x[2])),x[0]))[:5]:
    print('long',r[0],r[1],next(iter(r[2])),r[3])
PY"
