/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
name='1428854at2759.faa.mafft'
with ZipFile('inputs/scogs_animals.zip') as z:
    records=[]; header=None; seq=[]
    for line in z.read(name).decode().splitlines():
        if line.startswith('>'):
            if header is not None: records.append((header,''.join(seq)))
            header=line[1:].strip(); seq=[]
        elif line.strip(): seq.append(line.strip())
    records.append((header,''.join(seq)))
for i in range(len(records[0][1])):
    col=''.join(s[i] for _,s in records)
    c=Counter(col)
    if sum(v>=2 for v in c.values())>=2 and '-' in c:
        print(i+1,col,dict(c))
PY"
