/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

name='1428854at2759.faa.mafft'
with ZipFile('inputs/scogs_animals.zip') as z:
    lines=z.read(name).decode().splitlines()
seqs={}; h=None
for line in lines:
    if line.startswith('>'):
        h=line[1:].strip(); seqs[h]=''
    else: seqs[h]+=line.strip()
headers=list(seqs); L=len(next(iter(seqs.values())))
print(headers,L)
for j in range(L):
    raw=''.join(seqs[h][j] for h in headers)
    obs=[c for c in raw if c"'!='"'-']
    c=Counter(obs)
    if sum(n>=2 for n in c.values())>=2:
        print(j+1, raw, dict(c))
PY"
