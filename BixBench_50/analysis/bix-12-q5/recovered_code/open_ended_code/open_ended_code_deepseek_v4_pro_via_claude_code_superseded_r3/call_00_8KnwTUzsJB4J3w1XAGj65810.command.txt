import os
from collections import Counter

def parse_fasta(filepath):
    records = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if current_header is not None:
                    records.append((current_header, ''.join(current_seq)))
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            records.append((current_header, ''.join(current_seq)))
    return records

# Check non-mafft .faa files
aligned_non_mafft = []
for fname in sorted(os.listdir('/workspace/tmp_animals')):
    if fname.endswith('.mafft'):
        continue
    fpath = os.path.join('/workspace/tmp_animals', fname)
    records = parse_fasta(fpath)
    lengths = [len(r[1]) for r in records]
    if len(set(lengths)) == 1:  # all same length = aligned
        aligned_non_mafft.append((fname, lengths[0], len(records)))

print(f"Non-mafft files that are aligned: {len(aligned_non_mafft)}")
for fname, length, nseq in aligned_non_mafft[:10]:
    print(f"  {fname}: {nseq} seqs, {length} aa")
