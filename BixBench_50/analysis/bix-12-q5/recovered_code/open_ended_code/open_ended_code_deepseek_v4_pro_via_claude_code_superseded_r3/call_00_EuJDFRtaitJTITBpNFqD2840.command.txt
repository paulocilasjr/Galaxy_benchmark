from collections import Counter

def parse_fasta(filepath):
    records = []
    current_header = None
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if current_header is not None:
                    records.append((current_header, ''.join(current_seq)))
                current_header = line[1:]
                current_seq = []
            else:
                current_seq.append(line)
        if current_header is not None:
            records.append((current_header, ''.join(current_seq)))
    return records

def count_parsimony_informative_sites(records, verbose=False):
    if len(records) < 2:
        return 0
    
    seqs = [seq for _, seq in records]
    aln_len = len(seqs[0])
    informative_count = 0
    informative_cols = []
    
    for col in range(aln_len):
        chars = [s[col] for s in seqs]
        counts = Counter(chars)
        valid_states = [char for char, count in counts.items() if count >= 2]
        if len(valid_states) >= 2:
            informative_count += 1
            informative_cols.append((col, chars, dict(counts)))
    
    if verbose:
        for col, chars, counts in informative_cols:
            print(f"  Col {col}: {''.join(chars)} -> {counts}")
    
    return informative_count
