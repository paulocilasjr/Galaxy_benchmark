/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
zp='inputs/scogs_animals.zip'
std=set('ACDEFGHIKLMNPQRSTVWY')

def parse(txt):
    rec=[]; name=None; seq=[]
    for raw in txt.splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(seq).upper()))
            name=line[1:].split()[0]; seq=[]
        else: seq.append(line)
    if name is not None: rec.append((name,''.join(seq).upper()))
    return rec

rows=[]; alphabet=Counter(); failures=[]
with ZipFile(zp) as z:
    names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    for n in names:
        rec=parse(z.read(n).decode())
        lens={len(s) for _,s in rec}
        if len(rec)"'!=4 or len(lens)!=1:
            failures.append((n,len(rec),lens))
            continue
        seqs=[s for _,s in rec]
        alphabet.update('"''.join(seqs))
        pi_std=0; pi_gapstate=0; variable_std=0
        for col in zip(*seqs):
            c=Counter(x for x in col if x in std)
            if len(c)>=2: variable_std+=1
            if sum(v>=2 for v in c.values())>=2: pi_std+=1
            cg=Counter(col)
            if sum(v>=2 for v in cg.values())>=2: pi_gapstate+=1
        rows.append((pi_std,n,next(iter(lens)),variable_std,pi_gapstate,[x for x,_ in rec]))
print('selected',len(names),'processed',len(rows),'failed',len(failures))
print('failures',failures)
print('alphabet',dict(alphabet))
print('top 15: pi_std file aln_len variable pi_gapstate taxa')
for x in sorted(rows,reverse=True)[:15]: print(*x)
print('max_std',max(x[0] for x in rows),'ties',sum(x[0]==max(y[0] for y in rows) for x in rows))
print('max_gapstate',max(x[4] for x in rows))
PY"
