from zipfile import ZipFile
from collections import Counter

AA=set('ACDEFGHIKLMNPQRSTVWY')

def fasta_records(data):
    records=[]
    label=None
    parts=[]
    for line in data.decode('ascii').splitlines():
        if line.startswith('>'):
            if label is not None:
                records.append((label, ''.join(parts).upper()))
            label=line[1:].split()[0]
            parts=[]
        elif line.strip():
            parts.append(line.strip())
    if label is not None:
        records.append((label, ''.join(parts).upper()))
    return records

results=[]
errors=[]
with ZipFile('inputs/scogs_animals.zip') as z:
    selected=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    for name in selected:
        recs=fasta_records(z.read(name))
        lengths={len(s) for _,s in recs}
        labels=[x for x,_ in recs]
        if not recs or len(lengths)"'!=1 or len(set(labels))!=len(labels):
            errors.append((name,len(recs),lengths,labels))
            continue
        seqs=[s for _,s in recs]
        counter_count=0
        partition_count=0
        sites=[]
        for pos,col in enumerate(zip(*seqs),start=1):
            observed=[x for x in col if x in AA]
            freqs=Counter(observed)
            informative=sum(n>=2 for n in freqs.values())>=2
            counter_count += informative
            # At most four taxa are present, so a PI site must be two different AA pairs.
            pattern=sorted(observed)
            pair_pattern=(len(pattern)==4 and pattern[0]==pattern[1] and
                          pattern[2]==pattern[3] and pattern[1]!=pattern[2])
            partition_count += pair_pattern
            if informative: sites.append((pos,'"''.join(col),dict(freqs)))
        if counter_count "'!= partition_count:
            errors.append((name,'"'method disagreement',counter_count,partition_count))
        results.append((counter_count,name,len(recs),next(iter(lengths)),sites))

print('selected=',len(selected),'processed=',len(results),'skipped=0 failed=',len(errors),sep='')
print('sequence_count_distribution=',dict(sorted(Counter(x[2] for x in results).items())))
print('method_disagreements_or_errors=',errors)
mx=max(x[0] for x in results)
print('maximum=',mx)
print('maxima=',[(name,nseq,length) for count,name,nseq,length,_ in results if count==mx])
for count,name,nseq,length,sites in results:
    if count==mx:
        print(name,'informative_columns=',sites)
PY"
