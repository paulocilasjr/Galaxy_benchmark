/bin/bash -c "python - <<'PY'
from pathlib import Path
from collections import Counter

def parse_fasta(path):
    seqs=[]
    name=None; buf=[]
    for line in path.read_text().splitlines():
        if line.startswith('>'):
            if name is not None:
                seqs.append((name,''.join(buf)))
            name=line[1:].strip(); buf=[]
        else:
            buf.append(line.strip())
    if name is not None:
        seqs.append((name,''.join(buf)))
    return seqs

def pis(seqs):
    n=len(seqs)
    if n<2 or not seqs:
        return 0
    L=len(seqs[0][1])
    if any(len(s)"'!=L for _,s in seqs):
        raise ValueError('"'unequal lengths')
    pi=0
    for i in range(L):
        c=Counter(s[i] for _,s in seqs if s[i]"'!='"'-')
        if sum(1 for v in c.values() if v>=2)>=2:
            pi+=1
    return pi

d=Path('/workspace/work/scogs_animals')
rows=[]
for f in d.glob('*.mafft'):
    try:
        seqs=parse_fasta(f)
        p=pis(seqs)
        rows.append((p,len(seqs),len(seqs[0][1]),f.name))
    except Exception as e:
        rows.append((-1,len(seqs),None,f.name,str(e)))
rows.sort(reverse=True)
print('top 30:')
for r in rows[:30]:
    print(r)
print('max', rows[0])
PY"
