/bin/bash -lc "python3 - <<'PY'
import zipfile, collections, statistics
z=zipfile.ZipFile('/workspace/inputs/scogs_animals.zip')
def parse_fasta(text):
    seqs=[]; name=None; cur=[]
    for line in text.splitlines():
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(cur)))
            name=line[1:].strip(); cur=[]
        else:
            cur.append(line.strip())
    if name is not None: seqs.append((name,''.join(cur)))
    return seqs
cnt=collections.Counter(); lens_by=collections.defaultdict(list)
sp=set(); examples={}
for n in z.namelist():
 if n.endswith('.mafft'):
  seqs=parse_fasta(z.read(n).decode())
  cnt[len(seqs)]+=1
  sp.update([s[0] for s in seqs])
  lens_by[len(seqs)].append(len(seqs[0][1]) if seqs else 0)
  examples.setdefault(len(seqs),(n,[s[0] for s in seqs]))
print('species set', sorted(sp))
for k,v in sorted(cnt.items()):
    print('taxa',k,'count',v,'length min/median/max',min(lens_by[k]),statistics.median(lens_by[k]),max(lens_by[k]),'example',examples[k])
PY"
