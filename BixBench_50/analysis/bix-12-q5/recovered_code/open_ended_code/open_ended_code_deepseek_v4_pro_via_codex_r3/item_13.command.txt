/bin/bash -lc "python3 - <<'PY'
import zipfile, io, collections
z=zipfile.ZipFile('/workspace/inputs/scogs_animals.zip')
from Bio import SeqIO
mafft_names=[n for n in z.namelist() if n.endswith('.mafft')]
print('mafft', len(mafft_names))
cnt=collections.Counter(); example={}
for n in mafft_names:
    data=z.read(n).decode()
    seqs=list(SeqIO.parse(io.StringIO(data),'fasta'))
    cnt[len(seqs)]+=1
    example.setdefault(len(seqs), [n, [s.id for s in seqs]])
print(cnt)
for k,v in sorted(cnt.items()):
    print(k, v, example[k])
PY"
