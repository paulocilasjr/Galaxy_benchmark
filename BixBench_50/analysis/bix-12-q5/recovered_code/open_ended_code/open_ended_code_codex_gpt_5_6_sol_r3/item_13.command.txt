/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

MISSING=set('-?*.XBJZ')
def parse(text):
    seqs=[]; cur=[]
    for raw in text.splitlines():
        line=raw.strip()
        if line.startswith('>'):
            if cur: seqs.append(''.join(cur).upper()); cur=[]
        elif line:
            cur.append(line)
    if cur: seqs.append(''.join(cur).upper())
    return seqs

counts=[]
with ZipFile('inputs/scogs_animals.zip') as z:
    members=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    for name in members:
        seqs=parse(z.read(name).decode())
        assert seqs and len({len(s) for s in seqs}) == 1, name
        pi=sum(sum(v >= 2 for v in Counter(c for c in col if c not in MISSING).values()) >= 2 for col in zip(*seqs))
        counts.append((pi,name,len(seqs),len(seqs[0])))
print('selected',len(members),'processed',len(counts),'failed',0)
print('maximum',max(x[0] for x in counts))
print('maxima',sorted(x for x in counts if x[0]==max(y[0] for y in counts)))
PY
find /workspace -maxdepth 1 -type d -name 'tmp_phylo.*' -print"
