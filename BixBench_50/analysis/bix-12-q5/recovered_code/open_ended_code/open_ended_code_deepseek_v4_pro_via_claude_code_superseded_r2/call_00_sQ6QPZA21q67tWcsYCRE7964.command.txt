    print(f'.faa (unaligned): {len(faa_only)}')
    print(f'.faa.mafft (aligned): {len(mafft_only)}')
    
    # Show some mafft entries
    for n in mafft_only[:10]:
        print(f'  {n}')
    
    # Read one mafft file to see format
    if mafft_only:
        with z.open(mafft_only[0]) as f:
            content = f.read().decode('utf-8')
            print(f'\nFirst mafft file: {mafft_only[0]}')
            print(content[:2000])
