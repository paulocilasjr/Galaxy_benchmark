    for n in aln[:10]:

    # Show top-level structure
    top = set()
    for n in names:
        parts = n.split('/')
        if len(parts) >= 1:
            top.add(parts[0])
    print(f'\\nTop-level: {sorted(top)}')
    for t in sorted(top):
        count = sum(1 for n in names if n.startswith(t + '/'))
        print(f'  {t}/: {count} entries')
