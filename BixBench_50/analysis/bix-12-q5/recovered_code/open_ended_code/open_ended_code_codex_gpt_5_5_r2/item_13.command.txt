/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter

def read_fasta(text):
    seqs=[]; name=None; parts=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(parts).upper())
            name=line[1:]; parts=[]
        else:
            parts.append(line)
    if name is not None: seqs.append(''.join(parts).upper())
    return seqs

def informative_count(seqs):
    if len({len(s) for s in seqs}) "'!= 1:
        raise SystemExit('"'unaligned input encountered')
    count=0
    for col in zip(*seqs):
        c=Counter(x for x in col if x "'!= '"'-')
        if sum(v >= 2 for v in c.values()) >= 2:
            count += 1
    return count

best=(-1,None)
checked=0
with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    for name in z.namelist():
        if not name.endswith('.faa.mafft'):
            continue
        checked += 1
        val=informative_count(read_fasta(z.read(name).decode()))
        if val > best[0]:
            best=(val,name)
print(checked, best[0], best[1])
PY"
