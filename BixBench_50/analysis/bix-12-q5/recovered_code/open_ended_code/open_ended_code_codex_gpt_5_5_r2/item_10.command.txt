/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter
zp='inputs/scogs_animals.zip'
chars=Counter()
unknown=[]
with zipfile.ZipFile(zp) as z:
    for n in z.namelist():
        if not n.endswith('.faa.mafft'): continue
        seq=''
        for line in z.read(n).decode().splitlines():
            if line.startswith('>'): continue
            seq += line.strip().upper()
        chars.update(seq)
        bad=sorted(set(seq)-set('ACDEFGHIKLMNPQRSTVWY-*?XBZUOJ'))
        if bad: unknown.append((n,bad))
print(chars)
print('X count', chars['X'], 'B', chars['B'], 'Z', chars['Z'], 'J', chars['J'], '*', chars['*'], '?', chars['?'])
print('unknown', unknown[:10])
PY"
