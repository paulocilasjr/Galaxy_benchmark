/bin/bash -lc "python - <<'PY'
import zipfile
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(zname) as z:
        names=z.namelist()
    print(zname)
    print('total', len(names), 'mafft', sum(n.endswith('.mafft') for n in names), 'faa_only', sum(n.endswith('.faa') and not n.endswith('.mafft') for n in names))
    print('first mafft', next(n for n in names if n.endswith('.mafft')))
PY"
