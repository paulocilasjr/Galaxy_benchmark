/bin/bash -lc "python - <<'PY'
import zipfile
from collections import Counter

missing=set('BJZX.*-?')

def parse(data):
    d={}; name=None
    for raw in data.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            name=line[1:]
            if name in d: raise ValueError('duplicate '+name)
            d[name]=[]
        else: d[name].append(line.upper())
    return {k:''.join(v) for k,v in d.items()}

summ=[]; taxon_hist=Counter(); errors=[]
with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    aligned=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
    raw={n.removesuffix('.faa') for n in z.namelist() if n.endswith('.faa')}
    for n in aligned:
        rec=parse(z.read(n)); taxon_hist[len(rec)]+=1
        lengths={len(v) for v in rec.values()}
        if len(lengths)"'!=1: errors.append((n,lengths)); continue
        seqs=list(rec.values()); sites=[]
        for pos,col in enumerate(zip(*seqs),1):
            counts=Counter(c for c in col if c not in missing)
            if sum(n>=2 for n in counts.values())>=2:
                sites.append((pos,'"''.join(col),dict(counts)))
        summ.append((len(sites),n,next(iter(lengths)),len(rec),sites))
    print('aligned members:',len(aligned))
    print('matching raw counterpart IDs:',sum(n.removesuffix('.faa.mafft') in raw for n in aligned))
    print('taxa per alignment:',dict(sorted(taxon_hist.items())))
    print('unequal-length alignments:',errors)
    print('max PI:',max(x[0] for x in summ))
    winners=[x for x in summ if x[0]==max(y[0] for y in summ)]
    print('winner count:',len(winners))
    for count,name,length,ntax,sites in winners:
        print(name,'length',length,'taxa',ntax,'PI',count)
        print('positions/patterns:',[(p,c) for p,c,_ in sites])
    print('next highest:',[(x[1],x[0]) for x in sorted(summ,reverse=True)[:5]])
PY"
