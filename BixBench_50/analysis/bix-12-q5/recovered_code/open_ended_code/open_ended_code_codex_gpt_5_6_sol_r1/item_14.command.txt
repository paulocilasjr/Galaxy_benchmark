/bin/bash -lc 'task_tmp=$(sed -n '"'1p' /workspace/.amas-check-path)
mkdir -p \""'$task_tmp/alns"
unzip -qq inputs/scogs_animals.zip '"'*.faa.mafft' -d \""'$task_tmp/alns"
python "$task_tmp/pkg/amas/AMAS.py" summary -f fasta -d aa -i "$task_tmp"/alns/*.faa.mafft -o "$task_tmp/summary.tsv"
awk -F '"'\\t' 'NR==1{for(i=1;i<=NF;i++) if("'$i=="Parsimony-informative sites") c=i; print "PI column",c; next} $c>m{m=$c; name=$1; ties=1; next} $c==m{ties++} END{print "AMAS max",m,"first",name,"ties",ties}'"' \""'$task_tmp/summary.tsv"
awk -F '"'\\t' 'NR==1 || "'$9==29'"' \""'$task_tmp/summary.tsv" | cut -f1-10
rm -r -- "$task_tmp"
rm -- /workspace/.amas-check-path'
