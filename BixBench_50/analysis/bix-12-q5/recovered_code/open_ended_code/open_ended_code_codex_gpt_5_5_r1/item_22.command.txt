/bin/bash -lc "python3 - <<'PY'
import zipfile, collections

def parse_fasta(b):
    seqs=[]; name=None; chunks=[]
    for raw in b.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else:
            chunks.append(line)
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs

def pi(seqs, gap_missing=True):
    if not seqs: return 0
    lens={len(s) for _,s in seqs}
    if len(lens)"'!=1: raise ValueError(lens)
    miss=set('"'-?*.Xx') if gap_missing else set('?*.Xx')
    total=0
    for col in zip(*(s for _,s in seqs)):
        counts=collections.Counter(c.upper() for c in col if c not in miss)
        if sum(v>=2 for v in counts.values()) >= 2:
            total += 1
    return total

with zipfile.ZipFile('inputs/scogs_animals.zip') as z:
    rows=[]; seq_count_dist=collections.Counter(); aln_len_dist=collections.Counter()
    for n in z.namelist():
        if not n.endswith('.faa.mafft'): continue
        seqs=parse_fasta(z.read(n))
        seq_count_dist[len(seqs)] += 1
        lens={len(s) for _,s in seqs}
        if len(lens)"'!=1: raise SystemExit(f'"'unequal aligned lengths in {n}: {lens}')
        rows.append({'name':n,'nseq':len(seqs),'len':next(iter(lens)),'pi_missing':pi(seqs, True),'pi_gapstate':pi(seqs, False)})
    print('alignments',len(rows))
    print('sequence_count_distribution', dict(sorted(seq_count_dist.items())))
    for key in ['pi_missing','pi_gapstate']:
        top=max(rows, key=lambda r:r[key])
        ties=[r['name'] for r in rows if r[key]==top[key]]
        print(key, top[key], top['name'], 'nseq', top['nseq'], 'len', top['len'], 'ties', len(ties))
    print('top5 standard')
    for r in sorted(rows, key=lambda r:(r['pi_missing'],r['name']), reverse=True)[:5]:
        print(r['pi_missing'], r['pi_gapstate'], r['name'], r['nseq'], r['len'])
PY"
