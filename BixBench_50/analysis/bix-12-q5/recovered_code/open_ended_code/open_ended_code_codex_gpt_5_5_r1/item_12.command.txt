/bin/bash -lc "python3 - <<'PY'
import zipfile, collections, re
from pathlib import Path
zp=Path('inputs/scogs_animals.zip')
with zipfile.ZipFile(zp) as z:
    names=[n for n in z.namelist() if not n.endswith('/')]
    print('total files', len(names))
    print(collections.Counter([('.faa.mafft' if n.endswith('.faa.mafft') else '.faa' if n.endswith('.faa') else Path(n).suffix) for n in names]))
    mafft=[n for n in names if n.endswith('.faa.mafft')]
    raw=[n for n in names if n.endswith('.faa')]
    print('mafft',len(mafft),'raw',len(raw))
    print('first mafft sample:')
    data=z.read(mafft[0]).decode()
    print('\\n'.join(data.splitlines()[:12]))
    print('first raw sample:')
    data=z.read(raw[0]).decode()
    print('\\n'.join(data.splitlines()[:8]))
PY"
