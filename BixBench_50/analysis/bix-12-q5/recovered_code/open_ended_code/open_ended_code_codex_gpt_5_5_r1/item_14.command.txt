/bin/bash -lc "python3 - <<'PY'
import zipfile, collections
from pathlib import Path

def read_fasta(text):
    seqs=[]; name=None; parts=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                seqs.append((name,''.join(parts)))
            name=line[1:].split()[0]
            parts=[]
        else:
            parts.append(line)
    if name is not None:
        seqs.append((name,''.join(parts)))
    return seqs

def pi_count(seqs, gap_missing=True, missing=set('-?*.Xx')):
    if not seqs: return 0
    lens={len(s) for _,s in seqs}
    if len(lens)"'!=1:
        raise ValueError(f'"'not aligned lengths {sorted(lens)}')
    n=0
    for col in zip(*(s for _,s in seqs)):
        if gap_missing:
            states=[c.upper() for c in col if c not in missing]
        else:
            states=[c.upper() for c in col if c not in set('?*.Xx')]
        counts=collections.Counter(states)
        if sum(1 for v in counts.values() if v>=2)>=2:
            n+=1
    return n

for archive in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(archive) as z:
        rows=[]
        bad=[]
        for name in z.namelist():
            if not name.endswith('.faa.mafft'): continue
            seqs=read_fasta(z.read(name).decode())
            try:
                rows.append((pi_count(seqs, True), pi_count(seqs, False), name, len(seqs), len(seqs[0][1]) if seqs else 0, [n for n,_ in seqs]))
            except Exception as e:
                bad.append((name,str(e)))
        print('\\n',archive,'n_aln',len(rows),'bad',len(bad))
        print('max gap_missing',max(rows)[:5])
        print('max gap_state',max((r[1],r[0],r[2],r[3],r[4]) for r in rows))
        print('top10 gap_missing')
        for r in sorted(rows, reverse=True)[:10]:
            print(r[:5])
        if bad: print('bad sample',bad[:3])
PY"
