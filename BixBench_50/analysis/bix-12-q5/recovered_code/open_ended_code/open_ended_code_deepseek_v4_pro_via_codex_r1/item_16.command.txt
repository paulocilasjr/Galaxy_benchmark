/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
from collections import Counter

def parse_fasta(path):
    seqs=[]; cur=[]
    for line in path.read_text().splitlines():
        if line.startswith('>'):
            if cur:
                seqs.append(''.join(cur)); cur=[]
        else:
            cur.append(line.strip())
    if cur: seqs.append(''.join(cur))
    return seqs

def pis(path):
    seqs=parse_fasta(path)
    n=len(seqs)
    if n == 0:
        return 0,0,n
    L=len(seqs[0])
    if any(len(s)"'!=L for s in seqs):
        raise ValueError(f'"'unbalanced {path}: {sorted(set(map(len,seqs)))}')
    count=0
    for col in zip(*seqs):
        chars=[c for c in col if c"'!='"'-']
        if len(chars)<2:
            continue
        cnt=Counter(chars)
        if sum(1 for v in cnt.values() if v>=2) >= 2:
            count+=1
    return count,L,n

best=[]
for path in sorted(Path('tmp/sca').glob('*.mafft')):
    c,L,n=pis(path)
    best.append((c,L,n,path.name))
best.sort(reverse=True)
print('top 30')
for row in best[:30]:
    print(row)
print('MAX', best[0][0])
print('all complete?', [(row) for row in best if row[2]==4][:1])
PY"
