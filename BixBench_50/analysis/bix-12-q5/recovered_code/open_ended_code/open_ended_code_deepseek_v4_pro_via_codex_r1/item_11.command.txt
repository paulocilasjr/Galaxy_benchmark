/bin/bash -c "cd /workspace && python3 - <<'PY'
import glob, os
for kind in ['tmp/sca','tmp/scf']:
    files=sorted(glob.glob(kind+'/*.mafft'))
    counts={}
    for f in files:
        seqs=[]; name=None
        for l in open(f):
            if l.startswith('>'):
                if name: pass
                name=l.strip()
                seqs.append([])
            else:
                s=l.strip()
                if s: seqs[-1].append(s)
        n=len(seqs)
        counts[n]=counts.get(n,0)+1
    print(kind, counts, 'total',len(files))
PY"
