/bin/bash -lc "python - <<'PY'
import pandas as pd
from scipy.stats import spearmanr
ge=pd.read_csv('inputs/CRISPRGeneEffect.csv',usecols=['Unnamed: 0','CDKN1A (1026)'],index_col=0)
ex=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=['Unnamed: 0','CDKN1A (1026)'],index_col=0)
model=pd.read_csv('inputs/Model.csv',usecols=['ModelID'])
rows=ge.index.intersection(ex.index,sort=False)
a=ex.loc[rows,'CDKN1A (1026)']; essentiality=-ge.loc[rows,'CDKN1A (1026)']
m=a.notna() & essentiality.notna()
r=spearmanr(a[m],essentiality[m])
model_ids=set(model['ModelID'])
print('n',int(m.sum()),'rho_expression_vs_essentiality',repr(r.statistic),'p',repr(r.pvalue))
print('shared_models_missing_from_Model',len(set(rows)-model_ids))
assert r.statistic < -0.52
PY"
