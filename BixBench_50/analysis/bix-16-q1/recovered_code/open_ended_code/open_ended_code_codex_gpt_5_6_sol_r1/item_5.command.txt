/bin/bash -lc "python - <<'PY'
import csv,re
ps=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
hs=[]
for p in ps:
 with open(p,newline='') as f: h=next(csv.reader(f))[1:]
 parsed=[]
 bad=[]
 for x in h:
  m=re.fullmatch(r'(.+) \\((\\d+)\\)',x)
  if m: parsed.append((m.group(2),m.group(1),x))
  else: bad.append(x)
 print(p,'parsed',len(parsed),'bad',len(bad),'unique ids',len({x[0] for x in parsed}),'id dups',len(parsed)-len({x[0] for x in parsed}))
 hs.append(parsed)
d0={i:(s,x) for i,s,x in hs[0]}; d1={i:(s,x) for i,s,x in hs[1]}
ids=set(d0)&set(d1)
exact=set(x for _,_,x in hs[0])&set(x for _,_,x in hs[1])
mm=[(i,d0[i],d1[i]) for i in ids if d0[i][0]"'!=d1[i][0]]
print('"'shared ids',len(ids),'exact shared labels',len(exact),'symbol mismatches by shared id',len(mm))
print('mismatches',mm[:30])
PY"
