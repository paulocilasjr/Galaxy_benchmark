/bin/bash -lc "python - <<'PY'
import pandas as pd
import numpy as np
from scipy.stats import rankdata
import re, os, math

ge_path = 'inputs/CRISPRGeneEffect.csv'
expr_path = 'inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
model_path = 'inputs/Model.csv'

ge_cols = list(pd.read_csv(ge_path, nrows=0).columns)
expr_cols = list(pd.read_csv(expr_path, nrows=0).columns)
common = [c for c in ge_cols[1:] if c in set(expr_cols[1:])]
print('shared_gene_labels', len(common), flush=True)

ge = pd.read_csv(ge_path, usecols=[ge_cols[0]] + common)
expr = pd.read_csv(expr_path, usecols=[expr_cols[0]] + common)
ge = ge.rename(columns={ge_cols[0]: 'ModelID'}).set_index('ModelID')
expr = expr.rename(columns={expr_cols[0]: 'ModelID'}).set_index('ModelID')

model_ids = set(pd.read_csv(model_path, usecols=['ModelID'])['ModelID'])
shared_models_all = ge.index.intersection(expr.index)
missing_meta = [m for m in shared_models_all if m not in model_ids]
print('shared_models_all', len(shared_models_all), 'missing_model_metadata', len(missing_meta), flush=True)
shared_models = pd.Index([m for m in shared_models_all if m in model_ids])
print('shared_models_used', len(shared_models), flush=True)

ge = ge.loc[shared_models, common]
expr = expr.loc[shared_models, common]

x_arr = expr.to_numpy(dtype=np.float64, copy=False)
y_arr = -ge.to_numpy(dtype=np.float64, copy=False)

best_col = None
best_rho = np.inf
best_n = 0
best_rows = []

for j, col in enumerate(common):
    x = x_arr[:, j]
    y = y_arr[:, j]
    mask = np.isfinite(x) & np.isfinite(y)
    n = int(mask.sum())
    if n < 3:
        continue
    xv = x[mask]
    yv = y[mask]
    rx = rankdata(xv, method='average')
    ry = rankdata(yv, method='average')
    rx -= rx.mean()
    ry -= ry.mean()
    denom = math.sqrt(float(np.dot(rx, rx) * np.dot(ry, ry)))
    if denom == 0.0:
        continue
    rho = float(np.dot(rx, ry) / denom)
    if rho < best_rho:
        best_rho = rho
        best_col = col
        best_n = n

symbol = re.sub(r' \\(["'^()]*'"\\)"'$'"', '', best_col)
print('best_col', best_col, flush=True)
print('best_symbol', symbol, flush=True)
print('best_rho', best_rho, flush=True)
print('best_n', best_n, flush=True)

os.makedirs('codex_output', exist_ok=True)
with open('codex_output/answer.txt', 'w', encoding='utf-8') as f:
    f.write(symbol + '\\n')
PY"
