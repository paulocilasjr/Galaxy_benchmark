/bin/bash -lc "python - <<'PY'
import pandas as pd
from pathlib import Path

crispr_path = Path('inputs/CRISPRGeneEffect.csv')
expr_path = Path('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')

# Header-only read to select exact same gene labels in both matrices.
crispr_cols = pd.read_csv(crispr_path, nrows=0).columns.tolist()
expr_cols = pd.read_csv(expr_path, nrows=0).columns.tolist()
common_genes = [c for c in expr_cols[1:] if c in set(crispr_cols[1:])]
usecols = [''] + common_genes

expr = pd.read_csv(expr_path, index_col=0, usecols=usecols)
crispr = pd.read_csv(crispr_path, index_col=0, usecols=usecols)

common_models = expr.index.intersection(crispr.index)
expr = expr.loc[common_models, common_genes]
essentiality = -crispr.loc[common_models, common_genes]

# pandas corrwith(method='spearman') uses pairwise complete observations and ranks within each pair.
corr = expr.corrwith(essentiality, axis=0, method='spearman')
corr = corr.dropna()
label = corr.idxmin()
value = corr.loc[label]
symbol = label.rsplit(' (', 1)[0]
print('common_models', len(common_models))
print('common_genes', len(common_genes))
print('valid_corrs', len(corr))
print('min_label', label)
print('symbol', symbol)
print('rho', value)
print('top10_negative')
print(corr.nsmallest(10).to_string())
PY"
