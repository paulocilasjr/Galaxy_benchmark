import csv, re, math
from collections import Counter
files=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
headers=[]
ids=[]
for path in files:
    with open(path,newline='') as f:
        r=csv.reader(f); h=next(r); headers.append(h[1:]); mids=[row[0] for row in r]; ids.append(mids)
    print(path, 'rows', len(mids), 'unique rows', len(set(mids)), 'cols', len(h)-1, 'unique cols', len(set(h[1:])))
    dup=[x for x,c in Counter(h[1:]).items() if c>1][:10]
    print('dup cols sample', dup)
print('common models', len(set(ids[0]) & set(ids[1])))
print('common exact gene labels', len(set(headers[0]) & set(headers[1])))
# symbol duplicates and id format
pat=re.compile(r'"'^(.*) '"\\((["'^()]*)'"\\)"'$'"')
for name,h in zip(['crispr','expr'],headers):
    symbols=[]; bad=[]
    for col in h:
        m=pat.match(col)
        if not m: bad.append(col)
        else: symbols.append(m.group(1))
    print(name,'bad labels',len(bad),'unique symbols',len(set(symbols)),'dup symbol count',sum(c>1 for c in Counter(symbols).values()))
