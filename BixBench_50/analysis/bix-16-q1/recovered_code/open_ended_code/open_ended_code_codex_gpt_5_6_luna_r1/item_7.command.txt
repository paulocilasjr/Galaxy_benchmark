/bin/bash -lc "python - <<'PY'
import csv
from pathlib import Path
files=['inputs/CRISPRGeneEffect.csv','inputs/Model.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
for fn in files:
    p=Path(fn)
    with p.open(newline='',encoding='utf-8',errors='replace') as f:
        rd=csv.reader(f)
        h=next(rd)
        n=1
        first=[]; last=[]
        for row in rd:
            n+=1
            if len(first)<3: first.append(row[0] if row else None)
            last=[row[0] if row else None]
    print(fn, 'rows incl header',n,'cols',len(h),'first ids',first,'last id',last[0] if last else None,'first header',repr(h[:3]),'last header',repr(h[-3:]))
PY"
