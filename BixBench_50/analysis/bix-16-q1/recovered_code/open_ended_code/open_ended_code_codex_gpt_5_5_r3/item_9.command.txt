import csv
files=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
headers=[]
rows=[]
for p in files:
    with open(p,newline='') as f:
        r=csv.reader(f)
        h=next(r)[1:]
        ids=[row[0] for row in r]
    headers.append(h); rows.append(ids)
    print(p, 'features', len(h), 'models', len(ids), 'unique models', len(set(ids)))
common_features=set(headers[0]).intersection(headers[1])
common_models=set(rows[0]).intersection(rows[1])
print('common features', len(common_features))
print('common models', len(common_models))
print('crispr duplicate labels', len(headers[0])-len(set(headers[0])))
print('expr duplicate labels', len(headers[1])-len(set(headers[1])))
print('first common sorted', sorted(common_features)[:5])
PY"
