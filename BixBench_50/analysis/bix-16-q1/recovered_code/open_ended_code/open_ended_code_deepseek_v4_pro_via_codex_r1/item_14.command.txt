/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, scipy.stats as st
x=pd.DataFrame({'a':[1,2,3,np.nan],'b':[4,5,np.nan,7]})
y=pd.DataFrame({'a':[2,3,4,5],'b':[1,2,3,4]})
print(x.corrwith(y,method='spearman'))
for c in ['a','b']:
    m=x[c].notna() & y[c].notna()
    print(c, st.spearmanr(x.loc[m,c], y.loc[m,c]).correlation)
PY"
