import numpy as np
from scipy.stats import rankdata, spearmanr

GE='inputs/CRISPRGeneEffect.csv'
EX='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'

def read_header(path):
    with open(path,newline='') as f: return next(csv.reader(f))[1:]
def read_models(path):
    out=[]
    with open(path,newline='') as f:
        r=csv.reader(f); next(r)
        for row in r: out.append(row[0])
    return out

gh=read_header(GE); eh=read_header(EX)
gmodels=set(read_models(GE)); emodels=set(read_models(EX)); mdmodels=set(read_models(MD))
models=sorted(gmodels & emodels & mdmodels)
model_to_i={m:i for i,m in enumerate(models)}

# Prefer exact full feature labels. If nomenclature differs between releases,
# align the remaining features by the stable Entrez ID.
eh_exact={s:i for i,s in enumerate(eh)}
eh_by_id={}
for i,s in enumerate(eh): eh_by_id.setdefault(s.rsplit(' (',1)[1][:-1],[]).append(i)
pairs=[]
for gi,s in enumerate(gh):
    if s in eh_exact:
        ei=eh_exact[s]
    else:
        entrez=s.rsplit(' (',1)[1][:-1]
        cand=eh_by_id.get(entrez,[])
        if len(cand)"'!=1: continue
        ei=cand[0]
    pairs.append((gi,ei,s))
print('"'analysis dimensions:',len(models),'models x',len(pairs),'genes',flush=True)

gidx=np.array([x[0] for x in pairs],dtype=np.intp)
eidx=np.array([x[1] for x in pairs],dtype=np.intp)
labels=[x[2] for x in pairs]

def load(path, expected_ncols, selected):
    out=np.empty((len(models),len(selected)),dtype=np.float64)
    seen=np.zeros(len(models),dtype=bool)
    with open(path,'r') as f:
        next(f)
        for lineno,line in enumerate(f,2):
            model,payload=line.rstrip('\\r\\n').split(',',1)
            oi=model_to_i.get(model)
            if oi is None: continue
            if payload.startswith(','): payload='nan'+payload
            if payload.endswith(','): payload=payload+'nan'
            while ',,' in payload: payload=payload.replace(',,',',nan,')
            vals=np.fromstring(payload,sep=',',dtype=np.float64)
            if vals.size "'!= expected_ncols:
                raise RuntimeError(f'"'{path}:{lineno}: parsed {vals.size}, expected {expected_ncols}')
            out[oi]=vals[selected]
            seen[oi]=True
    if not seen.all(): raise RuntimeError(f'{path}: missing {np.sum(~seen)} aligned models')
    print('loaded',path,'missing selected values',int(np.isnan(out).sum()),flush=True)
    return out

e=load(EX,len(eh),eidx)
