/bin/bash -lc "python3 - <<'PY'
import csv,time
for fn in ['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']:
 t=time.time(); miss=0; total=0; rows_with=0
 with open(fn,newline='') as f:
  r=csv.reader(f); next(r)
  for row in r:
   m=sum(1 for x in row[1:] if x=='')
   miss+=m; total+=len(row)-1
   if m: rows_with+=1
 print(fn,'missing cells',miss,'of',total,'rows_with_missing',rows_with,'time',time.time()-t)
PY"
