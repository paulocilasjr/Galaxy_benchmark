import numpy as np, time, csv
fn='inputs/CRISPRGeneEffect.csv'
with open(fn,newline='') as f:
 h=next(csv.reader(f))

t=time.time()
a=np.genfromtxt(fn,delimiter=',',skip_header=1,usecols=idx,dtype=np.float32,missing_values='',filling_values=np.nan)
print(a.shape, a.dtype, np.isnan(a).sum(), time.time()-t)
PY"
