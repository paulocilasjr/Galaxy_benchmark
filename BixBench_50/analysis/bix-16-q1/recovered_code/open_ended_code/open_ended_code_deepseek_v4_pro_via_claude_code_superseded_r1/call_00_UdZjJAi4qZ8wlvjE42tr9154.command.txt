    header = next(r)
    expr_genes = header[1:]
    for row in r:
        expr_rows.append(row[0])

print(f"Expression: {len(expr_rows)} rows, {len(expr_genes)} genes", flush=True)

import numpy as np

n_rows = len(aligned_rows)

# Load CRISPR data
print("Loading CRISPR data...", flush=True)
crispr_data = np.full((n_rows, n_genes), np.nan)
with open('/workspace/inputs/CRISPRGeneEffect.csv') as f:
    r = csv.reader(f)
    header = next(r)
    for row_idx, row in enumerate(r):
        rid = row[0]
        if rid not in crispr_row_idx:
            continue
        i = aligned_rows.index(rid)
        for g, j in crispr_gene_idx.items():
            col_idx_in_file = crispr_genes.index(g) + 1  # +1 for row label
            val = row[col_idx_in_file]
            if val.strip():
                crispr_data[i, j] = float(val)

print(f"CRISPR data shape: {crispr_data.shape}, NaNs: {np.sum(np.isnan(crispr_data))}", flush=True)

# Load Expression data
print("Loading Expression data...", flush=True)
expr_data = np.full((n_rows, n_genes), np.nan)
with open('/workspace/inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv') as f:
    r = csv.reader(f)
    header = next(r)
    for row_idx, row in enumerate(r):
        rid = row[0]
        if rid not in expr_row_idx:

        i = aligned_rows.index(rid)
        for g, j in expr_gene_idx.items():
            col_idx_in_file = expr_genes.index(g) + 1
            val = row[col_idx_in_file]
            if val.strip():
                expr_data[i, j] = float(val)

print(f"Expression data shape: {expr_data.shape}, NaNs: {np.sum(np.isnan(expr_data))}", flush=True)

from scipy.stats import spearmanr

best_gene = None
best_corr = float('inf')  # looking for most negative cor(expr, essentiality)
# essentiality = -gene_effect (crispr_data)

n_computed = 0
for j in range(n_genes):
    gene = common_genes[j]
    expr_vec = expr_data[:, j]
    crispr_vec = crispr_data[:, j]
    
    # Pairwise-complete: mask where both are present
    mask = ~np.isnan(expr_vec) & ~np.isnan(crispr_vec)
    n_valid = mask.sum()
    
    if n_valid < 5:  # need minimum observations for meaningful correlation
        continue
    
    x = expr_vec[mask]
    y_raw = crispr_vec[mask]
    y_essentiality = -y_raw  # negative-going → positive-going (essentiality)
    
    # Spearman correlation
    corr, pval = spearmanr(x, y_essentiality)
    
    if not np.isnan(corr) and corr < best_corr:
        best_corr = corr
        best_gene = gene
        # Extract just the gene symbol (without Entrez ID)
    
    n_computed += 1
    if n_computed % 2000 == 0:
        print(f"  Computed {n_computed}/{n_genes} genes, current best: {best_gene} rho={best_corr:.6f}", flush=True)

print(f"\n=== Results ===", flush=True)

gene_symbol = best_gene.split(' (')[0]
