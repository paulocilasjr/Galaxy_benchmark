/bin/bash -lc "PYTHONPATH=/workspace/.deps python - <<'PY'
import csv, time
import numpy as np
import pandas as pd
cris_path='inputs/CRISPRGeneEffect.csv'; expr_path='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
def read_header(path):
    with open(path,newline='') as f:return next(csv.reader(f))[1:]
def symbol(x): return x.rsplit(' (',1)[0]
cr_all=read_header(cris_path); ex_all=read_header(expr_path)
cr_by={symbol(x):x for x in cr_all}; ex_by={symbol(x):x for x in ex_all}
syms=[symbol(x) for x in cr_all if symbol(x) in ex_by]
cr_cols=[cr_by[s] for s in syms]; ex_cols=[ex_by[s] for s in syms]
print('genes',len(syms),flush=True)
t=time.time(); cr=pd.read_csv(cris_path,index_col=0,usecols=['Unnamed: 0']+cr_cols,dtype=np.float32)
print('read cris',cr.shape,'sec',round(time.time()-t,1),'nan',int(cr.isna().sum().sum()),'inf',int(np.isinf(cr.to_numpy()).sum()),flush=True)
t=time.time(); ex=pd.read_csv(expr_path,index_col=0,usecols=['Unnamed: 0']+ex_cols,dtype=np.float32)
print('read expr',ex.shape,'sec',round(time.time()-t,1),'nan',int(ex.isna().sum().sum()),'inf',int(np.isinf(ex.to_numpy()).sum()),flush=True)
common=cr.index.intersection(ex.index)
print('models',len(common),'cr-only',len(cr.index.difference(ex.index)),'expr-only',len(ex.index.difference(cr.index)),flush=True)
cr=cr.loc[common]; ex=ex.loc[common]
print('paired nan',int(cr.isna().sum().sum()),int(ex.isna().sum().sum()),'expr models any nan',int(ex.isna().any(axis=1).sum()),'max nan gene',int(ex.isna().sum(axis=0).max()),flush=True)
print('cr range',float(np.nanmin(cr.to_numpy())),float(np.nanmax(cr.to_numpy())),'expr range',float(np.nanmin(ex.to_numpy())),float(np.nanmax(ex.to_numpy())),flush=True)
PY"
