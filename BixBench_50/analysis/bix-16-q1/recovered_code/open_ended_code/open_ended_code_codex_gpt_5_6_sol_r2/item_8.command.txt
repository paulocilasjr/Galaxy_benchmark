/bin/bash -lc "python - <<'PY'
import csv, re
import pandas as pd
paths={'effect':'inputs/CRISPRGeneEffect.csv','expr':'inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'}
headers={}
for k,p in paths.items():
    with open(p,newline='') as f: headers[k]=next(csv.reader(f))[1:]
    syms=[re.sub(r' \\(\\d+\\)"'$'"','',x) for x in headers[k]]
    print(k,'genes',len(headers[k]),'unique exact',len(set(headers[k])),'unique symbols',len(set(syms)))
common=set(headers['effect']) & set(headers['expr'])
print('exact common genes',len(common))
for k,p in paths.items():
    ids=pd.read_csv(p,usecols=[0]).iloc[:,0].astype(str)
    print(k,'models',len(ids),'unique',ids.nunique(),'duplicate',ids.duplicated().sum(),'missing id',ids.isna().sum())
    if k=='effect': e=set(ids)
    else: x=set(ids)
print('common models',len(e&x),'effect only',len(e-x),'expr only',len(x-e))
PY"
