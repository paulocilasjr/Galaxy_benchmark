/bin/bash -c "python3 - <<'PY'
import csv, re, collections
for fn in ['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']:
    with open(fn) as f:
        header=next(csv.reader(f))[1:]
    syms=[re.sub(r'\\s*\\(.*\\)"'$'"','',h) for h in header]
    dup=[(s,n) for s,n in collections.Counter(syms).items() if n>1]
    print(fn, 'n header', len(header), 'unique syms', len(set(syms)), 'dup count', len(dup))
    print('first 20 dups:', dup[:20])
PY"
