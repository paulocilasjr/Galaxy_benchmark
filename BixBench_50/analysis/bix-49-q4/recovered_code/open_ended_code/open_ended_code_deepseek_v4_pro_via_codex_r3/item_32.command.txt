/bin/bash -lc "python3 - <<'PY'
import urllib.request, tarfile, io, os
url='https://micro.mamba.pm/api/micromamba/linux-aarch64/latest'
data=urllib.request.urlopen(url, timeout=30).read()
print('bytes',len(data), data[:10])
open('/workspace/tmp/micromamba.tar.bz2','wb').write(data)
# decompress with tarfile bz2
out='/workspace/tmp/mamba'
os.makedirs(out, exist_ok=True)
with tarfile.open('/workspace/tmp/micromamba.tar.bz2','r:bz2') as tf:
    tf.extract('bin/micromamba', path=out)
print('extracted')
PY
ls -l /workspace/tmp/mamba/bin/micromamba && /workspace/tmp/mamba/bin/micromamba --version"
