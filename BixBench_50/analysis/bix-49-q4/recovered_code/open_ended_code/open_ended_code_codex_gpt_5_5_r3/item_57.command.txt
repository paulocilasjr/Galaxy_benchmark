/bin/bash -lc "/workspace/micromamba_extract/bin/micromamba run -p /workspace/renv Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2)); suppressPackageStartupMessages(library(readxl))
counts <- read.table('inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt', header=TRUE, check.names=FALSE, row.names=1)
coldata <- as.data.frame(read_excel('inputs/Issy_ASXL1_blood_coldata_gender.xlsx'))
coldata <- coldata["'!is.na(coldata$sample), c('"'sample','condition','sex')]
rownames(coldata) <- coldata"'$sample
samples <- coldata$sample[coldata$sample %in% colnames(counts)]
coldata <- coldata[samples,,drop=FALSE]
counts <- counts[,samples,drop=FALSE]
coldata$condition <- factor(coldata$condition, levels=c('"'Control','ASXL1'))
coldata"'$sex <- factor(coldata$sex, levels=c('"'F','M'))
dds <- DESeq(DESeqDataSetFromMatrix(round(as.matrix(counts)), coldata, ~ sex + condition), quiet=TRUE)
res05 <- results(dds, name='condition_ASXL1_vs_Control', alpha=0.05)
res10 <- results(dds, name='condition_ASXL1_vs_Control')
res_noif <- results(dds, name='condition_ASXL1_vs_Control', independentFiltering=FALSE)
cat('alpha05', sum(res05"'$padj < 0.05, na.rm=TRUE), '"'nonNA', sum("'!is.na(res05$padj)), '"'\\n')
cat('defaultAlpha', sum(res10"'$padj < 0.05, na.rm=TRUE), '"'nonNA', sum("'!is.na(res10$padj)), '"'\\n')
cat('noIndependentFiltering', sum(res_noif"'$padj < 0.05, na.rm=TRUE), '"'nonNA', sum("'!is.na(res_noif$padj)), '"'\\n')
RS"
