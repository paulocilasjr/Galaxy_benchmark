/bin/bash -lc "Rscript - <<'RS'
.libPaths(c('/workspace/tmp/Rlib', .libPaths()))
suppressPackageStartupMessages({library(DESeq2);library(apeglm)})
o <- readRDS('/workspace/tmp/deseq_result.rds')
all_counts <- counts(o"'$dds)
cd <- as.data.frame(colData(o$dds))
ann <- read.csv('"'/workspace/inputs/gencode.v31.primary_assembly.genes.csv', stringsAsFactors=FALSE, check.names=FALSE)
stopifnot(identical(rownames(all_counts),ann"'$gene_id))
run <- function(keep,label) {
  d <- DESeqDataSetFromMatrix(all_counts[keep,,drop=FALSE], cd, design=~sex+condition)
  d <- DESeq(d, quiet=TRUE)
  r <- results(d, name='"'condition_ASXL1_vs_Control', alpha=.05)
  s <- lfcShrink(d, coef='condition_ASXL1_vs_Control', res=r, type='apeglm')
  cat(label,'rows',sum(keep),'sig',sum(s"'$padj<.05,na.rm=TRUE),'"'tested',sum("'!is.na(s$pvalue)),'"'padj_non_na',sum("'!is.na(s$padj)),'"'\\n')
}
run(ann"'$gene_type=='"'protein_coding','protein_coding')
run(rowSums(all_counts>=10)>=3,'rowSums10_ge3')
run(rowSums(all_counts>=10)>=5,'rowSums10_ge5')
RS"
