# Read row IDs without materializing counts
with open('inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt') as f:

print('gene_type',Counter(r['gene_type'] for r in rows).most_common())
print('protein_coding',sum(r['gene_type']=='protein_coding' for r in rows),'nonempty_hgnc',sum(bool(r['hgnc_id']) for r in rows))
