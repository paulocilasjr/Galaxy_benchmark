/bin/bash -c "/workspace/tools/envs/deseq/bin/Rscript - <<'RS'
suppressPackageStartupMessages(library(readxl))
suppressPackageStartupMessages(library(DESeq2))
suppressPackageStartupMessages(library(apeglm))
coldata <- as.data.frame(read_excel('inputs/Issy_ASXL1_blood_coldata_gender.xlsx'))
print(coldata)
counts <- read.table('inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt', header=TRUE, row.names=1, check.names=FALSE, sep='\\t', stringsAsFactors=FALSE)
cat('counts dim', dim(counts), '\\n')
cat('sample columns:'); print(colnames(counts))
cat('all metadata samples in cols?', all(coldata"'$sample %in% colnames(counts)), '"'\\n')
cat('colnames order equal?', identical(coldata"'$sample, colnames(counts)), '"'\\n')
cat('any duplicate count gene ids?', anyDuplicated(rownames(counts)), '\\n')
print(coldata"'$condition)
print(table(coldata$sex))
RS'
