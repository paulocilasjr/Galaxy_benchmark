/bin/bash -c "python3 - <<'PY'
import csv
p='inputs/gencode.v31.primary_assembly.genes.csv'
from collections import Counter
# parse header
with open(p) as f:
    header=f.readline().rstrip('\\n').split(',')
print(header)
ids=set(); rows=0; gene_rows=0
with open(p) as f:
    r=csv.DictReader(f)
    for row in r:
        rows+=1
        if row['type']=='gene':
            ids.add(row['gene_id']); gene_rows+=1
print('total lines', rows, 'gene rows', gene_rows, 'unique gene ids', len(ids))
# count file ids
with open('inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt') as f:
    next(f)
    cnt=0; overlap=0; missing=0
    for line in f:
        gid=line.split()[0]; cnt+=1
        if gid in ids: overlap+=1
        else: missing+=1
print('count ids',cnt,'overlap',overlap,'missing',missing)
PY"
