/bin/bash -c "/workspace/tools/envs/deseq/bin/Rscript - <<'RS'
suppressPackageStartupMessages(library(readxl))
coldata <- as.data.frame(read_excel('inputs/Issy_ASXL1_blood_coldata_gender.xlsx'))
print(dim(coldata))
counts <- read.table('inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt', header=TRUE, row.names=1, check.names=FALSE, sep='', stringsAsFactors=FALSE)
cat('counts dim', dim(counts), '\\n')
cat('sample columns:'); print(colnames(counts))
cat('all metadata samples in cols?', all(coldata"'$sample %in% colnames(counts)), '"'\\n')
cat('identical order?', identical(coldata"'$sample, colnames(counts)), '"'\\n')
cat('counts cols not in coldata:'); print(setdiff(colnames(counts), coldata"'$sample))
cat('"'coldata samples not in counts:'); print(setdiff(coldata"'$sample, colnames(counts)))
cat('"'any NA counts?', anyNA(counts), 'sum NA', sum(is.na(counts)), '\\n')
cat('non-integer?', any(counts "'!= round(counts)), '"'\\n')
cat('min max', range(counts), '\\n')
RS"
