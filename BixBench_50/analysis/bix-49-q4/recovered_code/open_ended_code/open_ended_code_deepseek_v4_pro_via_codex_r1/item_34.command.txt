/bin/bash -lc "python3 - <<'PY'
import urllib.request, gzip, io, re
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE230nnn/GSE230685/suppl/GSE230685_blood_RNAseq_rawCounts.txt.gz'
req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
try:
    data=urllib.request.urlopen(req,timeout=60).read()
    print('bytes',len(data))
    txt=gzip.decompress(data).decode('utf-8','ignore')
    lines=txt.splitlines()
    print('lines',len(lines))
    h=lines[0].split('\\t')
    print('header fields',len(h))
    print(h[:25])
    print('first rows')
    for l in lines[1:5]: print(l[:500])
except Exception as e:
    print('ERR',repr(e))
PY"
