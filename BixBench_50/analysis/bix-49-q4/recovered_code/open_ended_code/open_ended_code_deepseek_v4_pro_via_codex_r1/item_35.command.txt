/bin/bash -lc "Rscript - <<'RS'
suppressPackageStartupMessages({library(DESeq2); library(apeglm); library(openxlsx)})
counts <- read.table('/workspace/inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt', header=TRUE, row.names=1, check.names=FALSE)
coldata <- read.xlsx('/workspace/inputs/Issy_ASXL1_blood_coldata_gender.xlsx')
coldata <- coldata["'!is.na(coldata$sample),]
rownames(coldata) <- coldata$sample
keep <- intersect(coldata$sample, colnames(counts))
counts <- counts[,keep,drop=FALSE]; coldata <- coldata[colnames(counts),,drop=FALSE]
coldata$condition <- factor(coldata$condition, levels=c('"'Control','ASXL1'))
coldata"'$sex <- factor(coldata$sex, levels=c('"'F','M'))
# prefilter
keepg <- rowSums(counts) >= 10
cat('prefilter genes', sum(keepg), 'of', nrow(counts), '\\n')
for (pre in c(FALSE,TRUE)) {
  cnt <- if (pre) counts[keepg,,drop=FALSE] else counts
  dds <- DESeqDataSetFromMatrix(countData=as.matrix(cnt), colData=coldata, design=~sex+condition)
  dds <- DESeq(dds, quiet=TRUE)
  res <- results(dds, name='condition_ASXL1_vs_Control')
  rl <- lfcShrink(dds, coef='condition_ASXL1_vs_Control', type='apeglm', quiet=TRUE)
  cat('prefilter=',pre,'genes',nrow(cnt),'unshrunken sig',sum(res"'$padj<0.05,na.rm=TRUE),'"'shrunken sig',sum(rl"'$padj<0.05,na.rm=TRUE),'"'\\n')
}
RS"
