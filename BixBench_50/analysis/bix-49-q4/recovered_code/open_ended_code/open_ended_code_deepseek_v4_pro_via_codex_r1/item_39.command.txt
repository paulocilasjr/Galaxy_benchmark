/bin/bash -lc "python3 - <<'PY'
import urllib.request,gzip,io
# read workspace count selected sample values first few
sample='MGD1567B'
local={}
with open('/workspace/inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt') as f:
    header=f.readline().split()
    idx=header.index(sample)
    for line in f:
        parts=line.split()
        if len(parts)"'!=22: continue
        local[parts[0]]=int(parts[idx])
url='"'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE230nnn/GSE230685/suppl/GSE230685_blood_RNAseq_rawCounts.txt.gz'
data=gzip.decompress(urllib.request.urlopen(url,timeout=60).read()).decode()
pub={}
lines=data.splitlines()
h=lines[0].split('\\t'); idx=h.index(sample)
for l in lines[1:]:
    p=l.split('\\t')
    if len(p)"'!=20: continue
    pub[p[0]]=int(p[idx])
print('"'local genes',len(local),'pub genes',len(pub),'common',len(set(local)&set(pub)))
diff=sum(local.get(g)"'!=pub.get(g) for g in set(local)&set(pub))
print('"'diff',diff)
# overall sum local vs pub
print('sum local',sum(local.values()),'sum pub',sum(pub.values()))
PY"
