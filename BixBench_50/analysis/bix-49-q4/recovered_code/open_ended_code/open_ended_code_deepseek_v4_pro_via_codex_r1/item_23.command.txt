/bin/bash -lc "python3 - <<'PY'
import urllib.request, re
url='https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE230685&targ=self&form=text&view=brief'
data=urllib.request.urlopen(url, timeout=30).read().decode('utf-8','ignore')
# print all lines around sample title/source and characteristics
for line in data.splitlines():
    if re.search(r'Sample_(title|source_name|geo_accession|organism|molecule|characteristics|description)', line, re.I):
        print(line[:500])
PY"
