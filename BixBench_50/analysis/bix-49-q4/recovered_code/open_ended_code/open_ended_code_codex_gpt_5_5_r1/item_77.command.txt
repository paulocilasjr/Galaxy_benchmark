/bin/bash -lc "TMPDIR=/workspace/tmp Rscript - <<'RS'
.libPaths(c('/workspace/Rlib', .libPaths()))
suppressPackageStartupMessages(library(DESeq2))
suppressPackageStartupMessages(library(apeglm))
counts <- read.table('inputs/Issy_ASXL1_blood_featureCounts_GeneTable_final.txt', header=TRUE, row.names=1, check.names=FALSE)
coldata <- read.table('/workspace/tmp/coldata.tsv', header=TRUE, sep='\\t', stringsAsFactors=FALSE, check.names=FALSE)
rownames(coldata) <- coldata"'$sample
common <- intersect(coldata$sample, colnames(counts))
counts <- counts[, common, drop=FALSE]
coldata <- coldata[common, , drop=FALSE]
mat <- as.matrix(counts); storage.mode(mat) <- '"'integer'
coldata"'$sex <- factor(coldata$sex)
coldata$condition <- relevel(factor(coldata$condition), ref='"'Control')
dds <- DESeqDataSetFromMatrix(mat, coldata, ~ sex + condition)
dds <- DESeq(dds, quiet=TRUE)
for (a in c(0.05, 0.1)) {
  res <- results(dds, name='condition_ASXL1_vs_Control', alpha=a)
  shr <- lfcShrink(dds, coef='condition_ASXL1_vs_Control', res=res, type='apeglm', quiet=TRUE)
  cat('alpha', a, 'padj_lt_0.05', sum("'!is.na(shr$padj) & shr$padj < 0.05), '"'nonNA', sum("'!is.na(shr$padj)), '"'\\n')
}
shr_default <- lfcShrink(dds, coef='condition_ASXL1_vs_Control', type='apeglm', quiet=TRUE)
cat('lfcShrink_default', sum("'!is.na(shr_default$padj) & shr_default$padj < 0.05), '"'nonNA', sum("'!is.na(shr_default$padj)), '"'\\n')
RS"
