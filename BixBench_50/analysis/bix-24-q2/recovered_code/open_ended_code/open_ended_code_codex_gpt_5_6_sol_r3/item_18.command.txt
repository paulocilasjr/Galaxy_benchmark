/bin/bash -lc "python - <<'PY'
import requests,tarfile,io,os
os.makedirs('/workspace/.go_tmp',exist_ok=True)
urls={
'org':'https://bioconductor.org/packages/3.16/data/annotation/src/contrib/org.Hs.eg.db_3.16.0.tar.gz',
'go':'https://bioconductor.org/packages/3.16/data/annotation/src/contrib/GO.db_3.16.0.tar.gz'}
for key,url in urls.items():
 b=requests.get(url,timeout=60); b.raise_for_status()
 t=tarfile.open(fileobj=io.BytesIO(b.content),mode='r:gz')
 names=[n for n in t.getnames() if n.endswith('.sqlite')]
 print(key,names)
 member=t.getmember(names[0])
 with t.extractfile(member) as src, open(f'/workspace/.go_tmp/{key}.sqlite','wb') as dst: dst.write(src.read())
PY
python - <<'PY'
import sqlite3
for n in ['org','go']:
 c=sqlite3.connect(f'.go_tmp/{n}.sqlite')
 print('\\n',n)
 print([x[0] for x in c.execute(\"select name from sqlite_master where type='table'\")])
 for tbl in [x[0] for x in c.execute(\"select name from sqlite_master where type='table'\")]:
  if tbl in ['genes','ensembl','go','go_all','go_term','go_bp','go_bp_all']:
   print(tbl,c.execute(f'pragma table_info({tbl})').fetchall())
PY"
