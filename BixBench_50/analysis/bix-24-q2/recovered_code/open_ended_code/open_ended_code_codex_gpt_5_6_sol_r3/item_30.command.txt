/bin/bash -lc "python - <<'PY'
import pandas as pd,time
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
allc=pd.read_csv('inputs/counts_raw_unfiltered.csv',index_col=0)
m=pd.read_csv('inputs/sample_layout.csv'); m['key']=m.SampleID.str.replace('-','_',regex=False); m=m.set_index('key')
keep=(allc.ge(5).sum(axis=1)>=3)
sel=m.index[m.Group.isin(['CBD_IC50','DMSO'])]
c=allc.loc[keep,sel].T
meta=m.loc[sel].rename(columns={'Group':'condition'})[['condition']]
print('prefilter retained',keep.sum(),'pair samples',list(sel),flush=True)
t=time.time(); dds=DeseqDataSet(counts=c,metadata=meta,design='~condition',refit_cooks=True,n_cpus=4,quiet=True); dds.deseq2(); print('fit seconds',round(time.time()-t,1),flush=True)
ds=DeseqStats(dds,contrast=['condition','CBD_IC50','DMSO'],alpha=.1,independent_filter=True,cooks_filter=True,n_cpus=4,quiet=True); ds.summary(); z=ds.results_df
sig=z[(z.padj<.05)&(z.log2FoldChange.abs()>.5)]
print('padj nonNA',z.padj.notna().sum(),'sig',len(sig),'up',sum(sig.log2FoldChange>.5),'down',sum(sig.log2FoldChange<-.5),flush=True)
print(z.loc[z.index.intersection(['ENSG00000167772','ENSG00000165029','ENSG00000114268','ENSG00000125740','ENSG00000140465','ENSG00000138061']),['log2FoldChange','pvalue','padj']].to_string(),flush=True)
z[['baseMean','log2FoldChange','pvalue','padj']].to_csv('/workspace/tmp_pairwise_de.csv')
PY"
