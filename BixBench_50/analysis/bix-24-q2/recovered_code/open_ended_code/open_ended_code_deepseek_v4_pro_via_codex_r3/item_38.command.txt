/bin/bash -c "python3 - <<'PY'
import pandas as pd, gseapy
mp=pd.read_csv('/workspace/tmp/sig_symbols.tsv',sep='\\t').dropna(subset=['symbol']).drop_duplicates('symbol')
res=pd.read_csv('/workspace/tmp/deseq2_cbd_vs_dmso.tsv',sep='\\t',index_col=0)
sig=res[(res.padj<0.05)&(res.log2FoldChange.abs()>0.5)]
sig=sig.join(mp.set_index('ensembl')['symbol'], how='left').dropna(subset=['symbol'])
up=sig[sig.log2FoldChange>0]['symbol'].tolist()
down=sig[sig.log2FoldChange<0]['symbol'].tolist()
print(len(up), len(down), up[:10], down[:10])
for lib in ['GO_Biological_Process_2023','GO_Biological_Process_2021']:
    print('\\nLIB',lib)
    for direction, genes in [('up',up),('down',down)]:
        try:
            enr=gseapy.enrichr(gene_list=genes, gene_sets=lib, organism='human', outdir=None, cutoff=0.05)
            df=enr.results
            print(direction, df.shape)
            print(df[['Term','Overlap','P-value','Adjusted P-value','Genes']].head(10).to_string())
        except Exception as e:
            print(direction,'ERR',type(e).__name__,e)
PY"
