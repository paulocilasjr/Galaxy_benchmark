/bin/bash -c "python - <<'PY'
import requests
urls=[
'https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/table/ijms-24-14743-t002/',
'https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/table/ijms-24-14743-t003/',
'https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/table/ijms-24-14743-t004/',
'https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/table/ijms-24-14743-t005/',
]
headers={'User-Agent':'Mozilla/5.0'}
for u in urls:
    r=requests.get(u,headers=headers,timeout=30)
    print(u, r.status_code, len(r.text), r.url)
    print(r.text[:200].replace('\\n',' '))
PY"
