/bin/bash -c "python - <<'PY'
import requests, re, html
from bs4 import BeautifulSoup
u='https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/'
text=requests.get(u,headers={'User-Agent':'Mozilla/5.0'}).text
soup=BeautifulSoup(text,'lxml')
# remove scripts/styles
for tag in soup(['script','style','nav','header','footer']): tag.decompose()
content=soup.get_text('\\n')
content=re.sub(r'\\n+', '\\n', content)
# print snippets around metabolism/CBD/GO/method
for term in ['carbohydrate metabolism','Glucose metabolism','downregulat','upregulat','Gene Ontology','Biological Process','log2','padj','adjusted p','Reactome','Metascape','enrich']:
    print('\\n===',term,'===')
    starts=[m.start() for m in re.finditer(term, content, re.I)][:20]
    for s in starts:
        print(content[max(0,s-400):s+800].strip()[:1200])
        print('---')
PY"
