/bin/bash -c "python - <<'PY'
import requests, pandas as pd, io
for n in [2,3]:
 u=f'https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/table/ijms-24-14743-t00{n}/'
 headers={'User-Agent':'Mozilla/5.0'}
 text=requests.get(u,headers=headers).text
 tabs=pd.read_html(io.StringIO(text))
 print('table page', n, 'num tables', len(tabs))
 for i,t in enumerate(tabs):
  print('---',i,t.shape)
  print(t.to_string(max_colwidth=80)[:10000])
  print()
PY"
