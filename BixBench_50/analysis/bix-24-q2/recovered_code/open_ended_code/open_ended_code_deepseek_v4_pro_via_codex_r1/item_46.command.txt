/bin/bash -c "python - <<'PY'
import pandas as pd,json
res=pd.read_csv('/workspace/de_cbd_vs_dmso.csv',index_col=0)
mp=json.load(open('/workspace/ensembl_mapping.json'))
bg=res[res['padj'].notna()].index.tolist()
sig=res[(res['padj']<0.05)&(res['log2FoldChange'].abs()>0.5)]
up=sig[sig['log2FoldChange']>0].index.tolist()
down=sig[sig['log2FoldChange']<0].index.tolist()
def syms(ids):
 s=[]
 for i in ids:
  v=mp.get(i)
  if v and v.get('symbol'): s.append(v['symbol'])
 return sorted(set(s))
for name, ids in [('bg',bg),('sig',sig.index),('up',up),('down',down)]:
 s=syms(ids)
 print(name, len(ids), 'mapped', len(s), 'unmapped', sum(1 for i in ids if not (mp.get(i) or {}).get('symbol')))
 print('top', s[:10])
# intersect up/down mapped
ups=syms(up); downs=syms(down); bgs=syms(bg)
print('updown overlap', len(set(ups)&set(downs)))
print('not in bg', len(set(ups)-set(bgs)), len(set(downs)-set(bgs)))
PY"
