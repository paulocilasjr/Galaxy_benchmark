/bin/bash -c "python - <<'PY'
import pandas as pd,json,warnings
warnings.filterwarnings('ignore')
import gseapy
res=pd.read_csv('/workspace/de_cbd_vs_dmso.csv',index_col=0); mp=json.load(open('/workspace/ensembl_mapping.json'))
bg=sorted(set(v['symbol'] for i in res[res['padj'].notna()].index if (v:=mp.get(i)) and v.get('symbol')))
sig=res[(res['padj']<0.05)&(res['log2FoldChange'].abs()>0.5)]
def syms(ids): return sorted(set(v['symbol'] for i in ids if (v:=mp.get(i)) and v.get('symbol')))
up=syms(sig[sig['log2FoldChange']>0].index); down=syms(sig[sig['log2FoldChange']<0].index)
for version in ['GO_Biological_Process_2021','GO_Biological_Process_2023','GO_Biological_Process_2026']:
 gs=json.load(open(f'/workspace/{version}.json'))
 print('\\nVERSION',version)
 for lab,gl in [('UP',up),('DOWN',down)]:
  enr=gseapy.enrich(gene_list=gl,gene_sets=gs,background=bg,outdir=None,cutoff=0.05,no_plot=True,verbose=False)
  r=enr.results
  sigr=r[r['Adjusted P-value']<0.05].copy()
  print(lab,'total',len(r),'sig adj<.05',len(sigr),'min adj',r['Adjusted P-value'].min())
  print(sigr[['Term','Overlap','P-value','Adjusted P-value','Odds Ratio','Genes']].head(20).to_string(max_colwidth=80))
PY"
