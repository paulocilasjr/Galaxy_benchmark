/bin/bash -c "python - <<'PY'
import requests,re
from bs4 import BeautifulSoup
u='https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/'
text=requests.get(u,headers={'User-Agent':'Mozilla/5.0'}).text
soup=BeautifulSoup(text,'lxml')
for tag in soup(['script','style']): tag.decompose()
content=soup.get_text('\\n'); content=re.sub(r'\\n+','\\n',content)
for term in ['DESeq2','DESeq','Wald','counts','log2 fold change','0.59','paired','contrast','BiomaRt','adjusted p values below']:
 print('\\n===',term,'===')
 for m in list(re.finditer(term,content,re.I))[:10]:
  print(content[max(0,m.start()-500):m.start()+1000].strip()[:1500])
PY"
