/bin/bash -lc "Rscript - <<'RS'
counts <- read.csv('inputs/counts_raw_unfiltered.csv', check.names=FALSE, stringsAsFactors=FALSE)
meta <- read.csv('inputs/sample_layout.csv', stringsAsFactors=FALSE)
cat('counts shape', nrow(counts), ncol(counts), '\\n')
cat('count id header:', names(counts)[1], 'unique', length(unique(counts[[1]])), 'dups', sum(duplicated(counts[[1]])), '\\n')
cat('meta shape', nrow(meta), ncol(meta), '\\n')
print(with(meta, table(Group)))
cat('sample mapping all:', setequal(gsub('-', '_', meta"'$SampleID, fixed=TRUE), names(counts)[-1]), '"'\\n')
cat('counts not in metadata:', paste(setdiff(names(counts)[-1], gsub('-', '_', meta"'$SampleID, fixed=TRUE)), collapse='"','), '\\n')
cat('metadata not in counts:', paste(setdiff(gsub('-', '_', meta"'$SampleID, fixed=TRUE), names(counts)[-1]), collapse='"','), '\\n')
lib <- colSums(counts[-1])
print(data.frame(sample=names(lib), libsize=lib), row.names=FALSE)
cat('all-zero genes', sum(rowSums(counts[-1])==0), '\\n')
RS"
