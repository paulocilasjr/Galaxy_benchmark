/bin/bash -lc "PYTHONPATH=/workspace/tmp/pylib python - <<'PY'
import pandas as pd
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
counts.index = counts.index.astype(str)
meta = pd.read_csv('inputs/sample_layout.csv')
meta['count_id'] = meta['SampleID'].str.replace('-', '_', regex=False)
keep = meta['Group'].isin(['DMSO','CBD_IC50'])
meta2 = meta.loc[keep].copy().set_index('count_id')
meta2 = meta2.loc[counts.columns.intersection(meta2.index)]
# explicit order is control then target, with DMSO as reference
meta2['condition'] = pd.Categorical(meta2['Group'], categories=['DMSO','CBD_IC50'])
cts = counts.loc[:, meta2.index].astype('int64')
print('shape', cts.shape, meta2['condition'].tolist())
print('library', cts.sum(axis=0).to_dict())
# remove only all-zero structural rows, which cannot enter a count model
cts = cts.loc[cts.sum(axis=1)>0]
print('nonzero rows', len(cts))
dds = DeseqDataSet(counts=cts.T, metadata=meta2[['condition']], design='~condition', refit_cooks=True, n_cpus=1, quiet=True)
dds.deseq2()
stat = DeseqStats(dds, contrast=['condition','CBD_IC50','DMSO'], alpha=0.05, cooks_filter=True, independent_filter=True, quiet=True, n_cpus=1)
stat.summary()
r = stat.results_df
print('result cols', r.columns.tolist())
print(r.head())
print('n nonNA padj', r.padj.notna().sum(), 'sig', ((r.padj<0.05)&(r.log2FoldChange.abs()>0.5)).sum())
sig = r[(r.padj<0.05)&(r.log2FoldChange.abs()>0.5)]
print('up', (sig.log2FoldChange>0).sum(), 'down', (sig.log2FoldChange<0).sum())
print('top up')
print(sig[sig.log2FoldChange>0].sort_values('padj').head(20)[['log2FoldChange','pvalue','padj']].to_string())
print('top down')
print(sig[sig.log2FoldChange<0].sort_values('padj').head(20)[['log2FoldChange','pvalue','padj']].to_string())
PY"
