/bin/bash -lc "PYTHONPATH=/workspace/tmp/pylib python -u - <<'PY'
import pandas as pd, traceback, sys
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
from gprofiler import GProfiler
try:
 counts = pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
 meta = pd.read_csv('inputs/sample_layout.csv')
 meta['count_id'] = meta['SampleID'].str.replace('-', '_', regex=False)
 groups=['DMSO','CBD_IC50','DMSO_Serum_starvation','CBD_IC50_Serum_starvation_16h']
 m=meta[meta['Group'].isin(groups)].set_index('count_id')
 m=m.loc[counts.columns.intersection(m.index)]
 m['condition']=pd.Categorical(m['Group'],categories=groups)
 cts=counts.loc[:,m.index].astype('int64')
 cts=cts.loc[cts.sum(axis=1)>0]
 print('fitting', cts.shape, m['Group'].value_counts().to_dict(), flush=True)
 dds=DeseqDataSet(counts=cts.T,metadata=m[['condition']],design='~condition',refit_cooks=True,n_cpus=1,quiet=True)
 dds.deseq2()
 bg=cts.index.astype(str).tolist()
 gp=GProfiler(return_dataframe=True)
 for label, contrast in [('CBD_vs_DMSO',['condition','CBD_IC50','DMSO']),('CBD_starved_vs_DMSO_starved',['condition','CBD_IC50_Serum_starvation_16h','DMSO_Serum_starvation'])]:
  st=DeseqStats(dds,contrast=contrast,alpha=0.05,cooks_filter=True,independent_filter=True,quiet=True,n_cpus=1); st.summary()
  res=st.results_df; sig=res[(res.padj<0.05)&(res.log2FoldChange.abs()>0.5)]
  up=sig.index[sig.log2FoldChange>0].astype(str).tolist(); down=sig.index[sig.log2FoldChange<0].astype(str).tolist()
  print('\\n',label,'DE',len(sig),'up',len(up),'down',len(down),flush=True)
  for direction,genes in [('UP',up),('DOWN',down)]:
   tab=gp.profile(organism='hsapiens',query=genes,background=bg,sources=['GO:BP'],user_threshold=0.05,all_results=True,no_evidences=False)
   sg=tab[tab.p_value<0.05]
   met=sg[sg.name.str.contains('metabol|energy|oxidative phosphorylation|respirat|lipid|amino acid|carbohydrate|nucleotide|mitochond|fatty acid|organic acid|coenzyme|catabol|biosynthetic|small molecule',case=False,regex=True)]
   print(direction,'GO sig',len(sg),'metabolic-like',len(met),flush=True)
   if len(met): print(met.sort_values('p_value')[['native','name','p_value','intersection_size']].head(15).to_string(index=False),flush=True)
except Exception:
 traceback.print_exc(); sys.exit(1)
PY"
