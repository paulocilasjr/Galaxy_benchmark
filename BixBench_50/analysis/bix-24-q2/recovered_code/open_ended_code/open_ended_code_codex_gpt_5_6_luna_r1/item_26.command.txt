/bin/bash -lc "PYTHONPATH=/tmp/cbd_py_lib python - <<'PY'
import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts=pd.read_csv('inputs/counts_raw_unfiltered.csv', index_col=0)
meta0=pd.read_csv('inputs/sample_layout.csv').set_index('SampleID')
meta=meta0.loc[[x.replace('_','-') for x in counts.columns]]

def run(treat, control):
    samples=meta.index[meta.Group.isin([treat,control])]
    cols=[x for x in counts.columns if x.replace('_','-') in set(samples)]
    c0=counts[cols]
    keep=(c0>=10).sum(axis=1)>=3
    c=c0.loc[keep]
    m=meta.loc[[x.replace('_','-') for x in cols]].copy()
    m.index=cols
    m['condition']=pd.Categorical(m['Group'], categories=[control,treat])
    c=c.T
    print('\\nRUN',treat,'vs',control,'samples',list(m.index),'genes',c.shape[1],flush=True)
    dds=DeseqDataSet(counts=c, metadata=m[['condition']], design='~condition', refit_cooks=True, n_cpus=1, quiet=True)
    dds.deseq2()
    stat=DeseqStats(dds, contrast=['condition',treat,control], alpha=0.05, cooks_filter=True, independent_filter=True, quiet=True, n_cpus=1)
    stat.summary()
    r=stat.results_df.copy()
    sig=r[(r.padj<0.05)&(abs(r.log2FoldChange)>0.5)]
    print('DE rows',len(r),'finite padj',r.padj.notna().sum(),'sig',len(sig),'up',(sig.log2FoldChange>0).sum(),'down',(sig.log2FoldChange<0).sum(),flush=True)
    print('top up')
    print(sig[sig.log2FoldChange>0].sort_values('padj').head(10)[['log2FoldChange','pvalue','padj']].to_string())
    print('top down')
    print(sig[sig.log2FoldChange<0].sort_values('padj').head(10)[['log2FoldChange','pvalue','padj']].to_string())
    return r
r=run('CBD_IC50','DMSO')
r2=run('CBD_IC50_Serum_starvation_16h','DMSO_Serum_starvation')
PY"
