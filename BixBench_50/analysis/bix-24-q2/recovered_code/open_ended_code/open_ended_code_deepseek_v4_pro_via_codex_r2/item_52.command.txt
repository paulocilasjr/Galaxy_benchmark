/bin/bash -c "Rscript - <<'RS'
suppressPackageStartupMessages({library(DESeq2); library(org.Hs.eg.db); library(AnnotationDbi)})
counts <- read.csv('/workspace/inputs/counts_raw_unfiltered.csv', check.names=FALSE, row.names=1)
layout <- read.csv('/workspace/inputs/sample_layout.csv', stringsAsFactors=FALSE)
layout"'$SampleID <- gsub('"'-','_',layout"'$SampleID)
layout <- layout[layout$Group %in% c('"'DMSO','CBD_IC50'),]
layout <- layout[match(colnames(counts), layout"'$SampleID),]; layout <- layout[!is.na(layout$SampleID),]
layout$Group <- factor(layout$Group, levels=c('"'DMSO','CBD_IC50'))
dds <- DESeqDataSetFromMatrix(as.matrix(counts[,layout"'$SampleID]), data.frame(row.names=layout$SampleID, Group=layout$Group), ~Group)
dds <- DESeq(dds, quiet=TRUE)
res <- as.data.frame(results(dds, alpha=0.05)); res$gene=rownames(res)
down <- res[!is.na(res$padj)&res$padj<0.05&res$log2FoldChange<(-0.5),]
sym <- mapIds(org.Hs.eg.db, keys=down$gene, column='"'SYMBOL', keytype='ENSEMBL', multiVals='first')
down"'$symbol <- sym[down$gene]
print(down[order(down$padj), c('"'gene','symbol','log2FoldChange','padj')][1:30,], row.names=FALSE)
up <- res["'!is.na(res$padj)&res$padj<0.05&res$log2FoldChange>0.5,]
symup <- mapIds(org.Hs.eg.db, keys=up$gene, column='"'SYMBOL', keytype='ENSEMBL', multiVals='first')
up"'$symbol <- symup[up$gene]
print(up[order(up$padj), c('"'gene','symbol','log2FoldChange','padj')][1:30,], row.names=FALSE)
RS"
