/bin/bash -c "python3 - <<'PY'
import urllib.request, re, html
url='https://pmc.ncbi.nlm.nih.gov/articles/PMC10572413/?report=xml'
data=urllib.request.urlopen(url, timeout=30).read().decode('utf-8','ignore')
text=re.sub(r'<script.*?</script>|<style.*?</style>', ' ', data, flags=re.S|re.I)
text=re.sub(r'<["'^>]+>'"', '\\n', text)
text=html.unescape(text)
text=re.sub(r'\\s+', ' ', text)
for term in ['Gene Ontology','biological process','enrichGO','clusterProfiler','DAVID','g:Profiler','KEGG','glycolytic process','xenobiotic metabolic']:
 print(term, text.lower().count(term.lower()))
 for m in re.finditer(term, text, re.I):
  s=m.start(); print(text[max(0,s-300):s+700]); break
PY"
