import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/Proteomic_data .xlsx', data_only=True)
ws = wb['Tumor vs Normal']

for row in ws.iter_rows(min_row=1, max_row=ws.max_row, values_only=True):
    gene = row[2]
    if gene and str(gene).strip().upper() == 'ENO1':
        normal = float(row[4])
        tumor = float(row[5])
        fc = float(row[7])
        ratio = float(row[6])
        print(f"Normal abundance: {normal}")
        print(f"Tumor abundance: {tumor}")
        print(f"Tumor/Normal ratio: {tumor/normal}")
        print(f"FC column value: {fc}")
        print(f"Ratio column value: {ratio}")
