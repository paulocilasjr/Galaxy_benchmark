/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from pathlib import Path
p=Path('/workspace/inputs/Proteomic_data .xlsx')
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['Tumor vs Normal']
headers=[c for c in next(ws.iter_rows(min_row=1,max_row=1,values_only=True))]
print(headers)
matches=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    row=dict(zip(headers,r))
    gene=str(row.get('gene')).strip() if row.get('gene') is not None else ''
    desc=str(row.get('Description')) if row.get('Description') is not None else ''
    if gene == 'ENO1' or 'ENO1' in desc:
        matches.append(row)
print('matches', len(matches))
for row in matches:
    for k in ['protein','Description','gene','gene_id','Normal','Tumor','Ratio','FC','log2FC','p.value','adj.Pval','compare']:
        print(k, row.get(k))
    print('computed tumor/normal', row['Tumor']/row['Normal'] if row['Normal'] else None)
PY"
