/bin/bash -lc "python - <<'PY'
from pathlib import Path
import openpyxl
for p in Path('/workspace/inputs').glob('*.xlsx'):
    print('\\nFILE', p.name, 'bytes', p.stat().st_size)
    wb = openpyxl.load_workbook(p, read_only=True, data_only=False)
    print('SHEETS', wb.sheetnames)
    for ws in wb.worksheets:
        print(' SHEET', ws.title, 'rows', ws.max_row, 'cols', ws.max_column)
        for row in ws.iter_rows(min_row=1, max_row=min(6, ws.max_row), values_only=True):
            print('  ', row[:min(12, ws.max_column)])
        print()
PY"
