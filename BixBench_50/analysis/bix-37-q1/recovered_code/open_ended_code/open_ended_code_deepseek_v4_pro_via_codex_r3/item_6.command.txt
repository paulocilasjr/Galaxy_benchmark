/bin/bash -lc "python3 - <<'PY'
import openpyxl, math
wb=openpyxl.load_workbook('inputs/Proteomic_data .xlsx', data_only=True, read_only=True)
ws=wb.active
cnt=0
for i,row in enumerate(ws.iter_rows(values_only=True),1):
 if i==1: continue
 protein,desc,gene,gene_id,n,t,ratio,fc,logfc,p,padj,sig,comp,x,y=row[:15]
 if n and t and ratio is not None:
  exact=t/n
  # examine ratio vs exact and fc
  if abs(exact-ratio)>0.001 or abs(exact-fc)>0.001:
   print(i,gene,n,t,exact,ratio,fc, 'log2 exact',math.log2(abs(exact)) if exact>0 else None, 'logfc',logfc)
   cnt+=1
   if cnt>=20: break
print('count shown',cnt)
PY"
