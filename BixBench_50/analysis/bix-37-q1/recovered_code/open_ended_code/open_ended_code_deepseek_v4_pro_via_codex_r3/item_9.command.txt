/bin/bash -lc "python3 - <<'PY'
import openpyxl
wb=openpyxl.load_workbook('inputs/Proteomic_data .xlsx', data_only=False)
ws=wb.active
rows=[]
for i,row in enumerate(ws.iter_rows(min_row=2, values_only=False),2):
 ratio=row[6].value
 fmt=row[6].number_format
 # heuristic: ratio rounded to <=2 decimal places and not integer huge
 if isinstance(ratio,(int,float)):
  s=repr(ratio)
  if fmt=='General' and ratio not in (None,):
   # count decimals in repr
   if '.' in s:
    dec=len(s.split('.')[1].rstrip('0'))
   else: dec=0
   if dec<=2 and abs(ratio)>=1:
    rows.append((i,row[2].value,ratio,row[5].value/row[4].value if row[4].value else None,fmt))
print('General ratio <=2 dec abs>=1 count',len(rows))
for x in rows[:100]: print(x)
PY"
