/bin/bash -lc "python3 - <<'PY'
import openpyxl, os
for fn in os.listdir('inputs'):
 print(fn, os.path.getsize('inputs/'+fn))
 try:
  wb=openpyxl.load_workbook('inputs/'+fn, data_only=True, read_only=True)
  print(wb.sheetnames)
  for ws in wb.worksheets:
   print('sheet', ws.title, ws.max_row, ws.max_column)
 except Exception as e:
  print('ERR',e)
PY"
