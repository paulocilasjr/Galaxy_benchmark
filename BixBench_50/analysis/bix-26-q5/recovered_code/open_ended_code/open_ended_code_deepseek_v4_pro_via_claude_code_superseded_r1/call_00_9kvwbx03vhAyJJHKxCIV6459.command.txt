Rscript -e '
# Try reading RDS files without loading DESeq2
# The base R readRDS should work even without the class package
res1 <- tryCatch(
  readRDS("inputs/res_GluFevsGluFePlus.rds"),
  error = function(e) { cat("Error:", e$message, "\n"); NULL }
)
if (!is.null(res1)) {
  cat("Successfully read!\n")
  cat("Class:", class(res1), "\n")
  cat("Type:", typeof(res1), "\n")
  cat("Is S4:", isS4(res1), "\n")
  cat("Slot names:", slotNames(res1), "\n")
}
' 2>&1
