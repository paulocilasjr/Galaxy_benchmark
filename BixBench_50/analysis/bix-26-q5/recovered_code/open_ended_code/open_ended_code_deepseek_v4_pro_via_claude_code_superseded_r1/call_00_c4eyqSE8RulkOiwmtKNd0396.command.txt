Rscript -e '
obj1 <- suppressWarnings(readRDS("inputs/res_GluFevsGluFePlus.rds"))
a1 <- attributes(obj1)
lfc1 <- a1$listData$log2FoldChange
padj1 <- a1$listData$padj
names1 <- a1$rownames

# Check a few key genes
cat("Siderophore-related genes (pau01053):\n")
sidero_genes <- c("PA14_09240", "PA14_54920", "PA14_49340", "PA14_09270", 
                  "PA14_09290", "PA14_54940", "PA14_09280", "PA14_09220", "PA14_09210")
for(g in sidero_genes) {
  idx <- which(names1 == g)
  if(length(idx) > 0) {
    cat(sprintf("  %s: log2FC=%.3f, padj=%.3e\n", g, lfc1[idx], padj1[idx]))
  }
}

obj2 <- suppressWarnings(readRDS("inputs/res_SuccvsGluFePlus.rds"))

  if(length(idx) > 0) {
    cat(sprintf("  %s: log2FC=%.3f, padj=%.3e\n", g, lfc2[idx], padj2[idx]))
  }
}
' 2>&1
