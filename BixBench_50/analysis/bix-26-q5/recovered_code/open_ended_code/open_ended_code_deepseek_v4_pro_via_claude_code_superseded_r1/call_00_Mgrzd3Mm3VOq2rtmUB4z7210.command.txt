# First, let's check if gseapy supports P. aeruginosa KEGG
import gseapy
print("gseapy version:", gseapy.__version__)

# Check available organisms
from gseapy.utils import check_gmt
import requests

# Check KEGG organism codes
# P. aeruginosa PA14 is 'pau', PAO1 is 'pae'
# Let's test if we can access KEGG for pau
try:
    # Try to get KEGG gene list for pau
    r = requests.get("https://rest.kegg.jp/list/pau", timeout=30)
    if r.status_code == 200:
        print("KEGG pau accessible, entries:", len(r.text.split('\n')))
        print("First 5 lines:")
        for line in r.text.split('\n')[:5]:
            print(line)
    else:
        print("KEGG pau status:", r.status_code)
except Exception as e:
    print("Error:", e)

# Also try pae (PAO1)
try:
    r = requests.get("https://rest.kegg.jp/list/pae", timeout=30)

        print("First 5 lines:")
