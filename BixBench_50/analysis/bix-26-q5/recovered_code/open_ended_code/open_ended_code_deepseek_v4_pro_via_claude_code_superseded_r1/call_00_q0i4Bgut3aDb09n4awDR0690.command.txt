Rscript -e '
.libPaths(c("~/R/library", .libPaths()))

# Install CRAN dependencies first
cran_pkgs <- c("Rcpp", "RcppArmadillo", "ggplot2", "dplyr", "rlang", "tibble", 
               "magrittr", "tidyr", "purrr", "stringr", "reshape2", "scales",
               "igraph", "ggrepel", "viridis", "ggforce", "ggraph", "ggnewscale",
               "tidytree", "treeio", "ape", "aplot", "ggfun", "scatterpie", 
               "shadowtext", "gson", "plyr", "qvalue", "downloader", "jsonlite",
               "cli", "vctrs", "lifecycle", "memoise", "MASS", "lattice", 
               "locfit", "matrixStats", "XML", "RCurl", "httr", "curl", "xml2",
               "yaml", "RSQLite", "dbplyr", "digest", "progress", "rappdirs",
               "filelock", "BiocFileCache", "data.table")

for (pkg in cran_pkgs) {
  if (!requireNamespace(pkg, quietly = TRUE)) {
    cat("Installing", pkg, "...\n")
    try(install.packages(pkg, repos = "https://cloud.r-project.org", lib = "~/R/library", dependencies = TRUE))
  } else {
    cat(pkg, "already installed\n")
  }
}
' 2>&1 | tail -40
