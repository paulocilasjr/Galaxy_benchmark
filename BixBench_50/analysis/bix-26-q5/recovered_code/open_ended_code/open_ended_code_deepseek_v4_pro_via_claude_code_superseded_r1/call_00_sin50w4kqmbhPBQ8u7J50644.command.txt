from scipy.stats import hypergeom, fisher_exact
import numpy as np

# Verify Fisher vs hypergeometric
# Siderophore pathway in iron
N, K, n, k = 2041, 9, 98, 6
# Hypergeometric: P(X >= k)
hg_pval = hypergeom.sf(k-1, N, K, n)
# Fisher: table [[a,b],[c,d]]
a, b, c, d = 6, 3, 92, 1940
_, fish_pval = fisher_exact([[a, b], [c, d]], alternative='greater')
print(f"Siderophore: hypergeom pval={hg_pval:.6e}, Fisher pval={fish_pval:.6e}")

# Secondary metabolites in iron
N, K, n, k = 2041, 444, 98, 35
hg_pval2 = hypergeom.sf(k-1, N, K, n)
a2, b2, c2, d2 = 35, 409, 63, 1534
_, fish_pval2 = fisher_exact([[a2, b2], [c2, d2]], alternative='greater')
print(f"Secondary metabolites: hypergeom pval={hg_pval2:.6e}, Fisher pval={fish_pval2:.6e}")

# Also check: Biofilm formation in innate (1/108 with 252 FG)
N, K, n, k = 2041, 108, 252, 1
hg_pval3 = hypergeom.sf(k-1, N, K, n)
# For comparison, also enrichment direction:
exp = n * K / N
print(f"\nBiofilm formation: expected={exp:.1f}, observed={k}, enrichment? {k > exp}")
# This is clear depletion - observed << expected
a3, b3, c3, d3 = 1, 107, 251, 1682
_, fish_pval3 = fisher_exact([[a3, b3], [c3, d3]], alternative='greater')
print(f"Biofilm formation (one-sided enrichment): Fisher pval={fish_pval3:.6e}")
_, fish_pval3b = fisher_exact([[a3, b3], [c3, d3]], alternative='less')
print(f"Biofilm formation (one-sided depletion): Fisher pval={fish_pval3b:.6e}")

# Conclusion
print("\nConclusion: one-sided Fisher (greater) correctly identifies enrichment only")
PYEOF
