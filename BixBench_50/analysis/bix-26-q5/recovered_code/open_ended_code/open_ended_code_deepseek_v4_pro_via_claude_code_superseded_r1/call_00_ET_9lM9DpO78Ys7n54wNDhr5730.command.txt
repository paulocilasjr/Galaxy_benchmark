import requests

# Test KEGG API for pau (PA14) and pae (PAO1)
for org in ['pau', 'pae']:
    try:
        r = requests.get(f"https://rest.kegg.jp/list/{org}", timeout=30)
        if r.status_code == 200:
            lines = [l for l in r.text.strip().split('\n') if l]
            print(f"KEGG '{org}': {len(lines)} entries")
            print("First 5:")
            for line in lines[:5]:
                print(f"  {line}")
        else:
            print(f"KEGG '{org}' status: {r.status_code}")
    except Exception as e:
        print(f"KEGG '{org}' error: {e}")
    
    # Also check pathways
    try:
        r = requests.get(f"https://rest.kegg.jp/list/pathway/{org}", timeout=30)
        if r.status_code == 200:
            lines = [l for l in r.text.strip().split('\n') if l]
            print(f"KEGG pathways '{org}': {len(lines)} entries")
            print("First 3:")
            for line in lines[:3]:
                print(f"  {line}")
    except Exception as e:
        print(f"Pathways error: {e}")
    print()
PYEOF
