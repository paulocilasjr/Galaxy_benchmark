Rscript -e '
obj1 <- suppressWarnings(readRDS("inputs/res_GluFevsGluFePlus.rds"))
obj2 <- suppressWarnings(readRDS("inputs/res_SuccvsGluFePlus.rds"))

a1 <- attributes(obj1)
a2 <- attributes(obj2)

lfc1 <- a1$listData$log2FoldChange
padj1 <- a1$listData$padj
names1 <- a1$rownames

lfc2 <- a2$listData$log2FoldChange
padj2 <- a2$listData$padj
names2 <- a2$rownames

keep1 <- !is.na(padj1) & padj1 < 0.05 & abs(lfc1) > 1.5
keep2 <- !is.na(padj2) & padj2 < 0.05 & abs(lfc2) > 1.5

cat("N_TOTAL:", length(names1), "\n")
cat("N_IRON_DE:", sum(keep1), "\n")
cat("N_INNATE_DE:", sum(keep2), "\n")

writeLines(as.character(names1[keep1]), "/tmp/de_iron.txt")
writeLines(as.character(names2[keep2]), "/tmp/de_innate.txt")
writeLines(as.character(names1), "/tmp/all_genes.txt")

cat("DONE\n")
' 2>&1
