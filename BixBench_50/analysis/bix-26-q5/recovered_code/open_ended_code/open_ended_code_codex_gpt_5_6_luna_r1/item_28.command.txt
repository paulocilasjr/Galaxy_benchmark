read_tab<-function(u){z<-readLines(url(u),warn=F);z<-z[nzchar(z)];do.call(rbind,strsplit(z,'\\\\t'))};ln<-read_tab('https://rest.kegg.jp/link/pathway/pau');sets<-split(sub('"'^pau:'"','',ln[,1]),sub('"'^path:'"','',ln[,2]));
for(f in list.files('/workspace/inputs',pattern='rds"'$'"',full.names=T)){a<-attributes(readRDS(f));d<-a"'$listData;ids<-a$rownames;sig<-ids[abs(d$log2FoldChange)>1.5&d$padj<.05];q<-intersect(sig,unique(unlist(sets)));cat('"'\\n',basename(f),'\\n');for(bg in c('5828full','5828restricted','2041')){if(bg=='5828full'){u<-ids;ss<-sets};if(bg=='5828restricted'){u<-ids;ss<-lapply(sets,intersect,ids)};if(bg=='2041'){u<-unique(unlist(sets));ss<-lapply(sets,intersect,u)};q2<-intersect(sig,u);M<-lengths(ss);k<-vapply(ss,function(s)sum(s%in%q2),integer(1));N<-length(u);n<-length(q2);for(sz in c('all','10to500')){ix<-k>0;if(sz"'!='"'all')ix<-ix&M>=10&M<=500;p<-phyper(k[ix]-1,M[ix],N-M[ix],n,F);pa<-p.adjust(p,'BH');cat(bg,sz,'sig',sum(pa<.05),'ids',paste(names(ss)[ix][pa<.05],collapse=','),'\\n')}}}
RS"
