/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
read_tab<-function(u){con<-url(u,'rb');on.exit(close(con));z<-readLines(con,warn=FALSE);z<-z[nzchar(z)];do.call(rbind,strsplit(z,'\\\\t'))}
ln<-read_tab('https://rest.kegg.jp/link/pathway/pau');sets<-split(sub('"'^pau:'"','',ln[,1]),sub('"'^path:'"','',ln[,2]));allu<-unique(unlist(sets));
for(f in list.files('/workspace/inputs',pattern='rds"'$'"',full.names=T)){a<-attributes(readRDS(f));d<-a"'$listData;ids<-a$rownames;sig<-ids[abs(d$log2FoldChange)>1.5&d$padj<.05];q<-intersect(sig,allu);ss<-lapply(sets,intersect,allu);M<-lengths(ss); k<-vapply(ss,function(s)sum(s%in%q),integer(1)); N<-length(allu); n<-length(q);ix<-k>0 & M>=10 & M<=500;p<-phyper(k[ix]-1,M[ix],N-M[ix],n,lower.tail=F);pa<-p.adjust(p,'"'BH'); cat('\\n',basename(f),'N',N,'n',n,'terms',sum(ix),'sig',sum(pa<.05),'\\n'); print(data.frame(id=names(ss)[ix],M=M[ix],k=k[ix],p=p,padj=pa)[order(pa),][1:sum(pa<.05),]);}
RS"
