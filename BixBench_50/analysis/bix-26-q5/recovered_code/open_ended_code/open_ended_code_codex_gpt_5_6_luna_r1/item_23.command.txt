/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
read_tab <- function(u) { con<-url(u,'rb'); on.exit(close(con)); z<-readLines(con,warn=FALSE); z<-z[nzchar(z)]; do.call(rbind,strsplit(z,'\\\\t')) }
ln <- read_tab('https://rest.kegg.jp/link/pathway/pau'); pw <- read_tab('https://rest.kegg.jp/list/pathway/pau')
sets <- split(sub('"'^pau:'"','',ln[,1]),sub('"'^path:'"','',ln[,2])); allu<-unique(unlist(sets))
for (f in list.files('/workspace/inputs',pattern='rds"'$'"',full.names=TRUE)) {
 a<-attributes(readRDS(f)); d<-a"'$listData; ids<-a$rownames; sig<-ids[abs(d$log2FoldChange)>1.5 & d$padj<.05]; q<-intersect(sig,allu)
 cat('"'\\n',basename(f),'\\n')
 for (univ in c('all','measured')) {
  u<-if(univ=='all') allu else ids; q2<-intersect(q,u); ss<-lapply(sets,intersect,u); M<-lengths(ss); k<-vapply(ss,function(z)sum(z%in%q2),integer(1)); N<-length(u); n<-length(q2)
  ix<-k>0; p<-phyper(k[ix]-1,M[ix],N-M[ix],n,lower.tail=F)
  for (denom in c('overlap','all137')) {
   pa<-if(denom=='overlap')p.adjust(p,'BH') else {v<-rep(1,length(sets)); v[ix]<-p; p.adjust(v,'BH')[ix]}
   cat(univ,denom,'N',N,'n',n,'sig',sum(pa<.05),'ids',paste(names(ss)[ix][pa<.05],collapse=','),'\\n')
  }
 }
}
RS"
