/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
 x <- readRDS(f); a <- attributes(x); d <- a"'$listData; ids <- a$rownames
 sel <- is.finite(d$log2FoldChange) & is.finite(d$padj) & abs(d$log2FoldChange)>1.5 & d$padj<0.05
 cat('"'\\n',basename(f),'n=',length(ids),'selected=',sum(sel),'up=',sum(sel & d"'$log2FoldChange>1.5),'"'down=',sum(sel & d"'$log2FoldChange < -1.5),'"'NA lfc=',sum(is.na(d"'$log2FoldChange)),'"'NA padj=',sum(is.na(d"'$padj)),'"'\\n')
 cat('lfc threshold metadata=',a"'$metadata$lfcThreshold,'"'alpha=',a"'$metadata$alpha,'"'\\n')
 print(summary(d"'$log2FoldChange)); print(summary(d$padj))
 cat('"'selected IDs first 30:',paste(ids[sel][1:min(30,sum(sel))],collapse=','),'\\n')
}
RS"
