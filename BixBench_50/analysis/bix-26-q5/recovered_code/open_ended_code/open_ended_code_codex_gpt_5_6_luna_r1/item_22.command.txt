/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
read_tab <- function(u) { con<-url(u,'rb'); on.exit(close(con)); z<-readLines(con,warn=FALSE); z<-z[nzchar(z)]; do.call(rbind,strsplit(z,'\\\\t')) }
pw <- read_tab('https://rest.kegg.jp/list/pathway/pau'); rownames(pw)<-pw[,1];
for (id in c('pau00405','pau01053','pau00190','pau02024','pau01110','pau00860','pau00030','pau02010','pau01200','pau04146','pau01503')) print(pw[id,,drop=FALSE])
RS"
