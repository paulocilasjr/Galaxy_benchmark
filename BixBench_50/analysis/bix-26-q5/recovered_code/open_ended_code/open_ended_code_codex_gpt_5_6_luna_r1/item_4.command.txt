/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('/workspace/inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  cat('class:', paste(class(x), collapse=', '), '\\n')
  cat('typeof:', typeof(x), '\\n')
  cat('length:', length(x), '\\n')
  cat('names:', paste(names(x), collapse=' | '), '\\n')
  cat('dim:', paste(dim(x), collapse=' x '), '\\n')
  if (is.data.frame(x)) {
    cat('columns:\\n'); print(data.frame(name=names(x), class=vapply(x, function(z) paste(class(z), collapse=','), character(1)), row.names=FALSE))
    cat('head:\\n'); print(utils::head(x, 4))
  } else if (is.list(x)) {
    for (nm in names(x)) {
      z <- x[[nm]]
      cat('element', nm, 'class', paste(class(z), collapse=','), 'length', length(z), 'dim', paste(dim(z), collapse=' x '), '\\n')
      if (is.atomic(z) && length(z) <= 20) print(z)
      if (is.data.frame(z)) { print(utils::head(z, 3)); cat('cols', paste(names(z), collapse=','), '\\n') }
    }
  }
}
RS"
