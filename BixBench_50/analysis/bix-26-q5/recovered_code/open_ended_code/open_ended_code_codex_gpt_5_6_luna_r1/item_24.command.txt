/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
read_tab<-function(u){z<-readLines(url(u),warn=FALSE);z<-z[nzchar(z)];do.call(rbind,strsplit(z,'\\\\t'))}
ln<-read_tab('https://rest.kegg.jp/link/pathway/pau'); sets<-split(sub('"'^pau:'"','',ln[,1]),sub('"'^path:'"','',ln[,2])); allu<-unique(unlist(sets))
for(f in list.files('/workspace/inputs',pattern='rds"'$'"',full.names=T)){a<-attributes(readRDS(f));d<-a"'$listData;ids<-a$rownames;sig<-ids[abs(d$log2FoldChange)>1.5&d$padj<.05]; q<-intersect(sig,allu)
 k<-vapply(sets,function(s)sum(s%in%q),integer(1));M<-lengths(sets);N<-length(ids);n<-length(sig);ix<-k>0
 for(Mmode in c('"'full','measured')) {MM<-if(Mmode=='full')M else vapply(sets,function(s)sum(s%in%ids),integer(1)); p<-phyper(k[ix]-1,MM[ix],N-MM[ix],n,lower.tail=F);pa<-p.adjust(p,'BH');cat(basename(f),Mmode,'sig',sum(pa<.05),'\\n')}
}
RS"
