import pandas as pd
import numpy as np

# Read the CSV files we saved
df1 = pd.read_csv("df1_full.csv")
df2 = pd.read_csv("df2_full.csv")

print("df1 (GluFe vs GluFePlus) columns:", list(df1.columns))
print("df1 shape:", df1.shape)
print("df2 (Succ vs GluFePlus) columns:", list(df2.columns))
print("df2 shape:", df2.shape)

# Filter significant genes
sig1 = df1[(df1['padj'].notna()) & (df1['padj'] < 0.05) & (df1['log2FoldChange'].abs() > 1.5)]
sig2 = df2[(df2['padj'].notna()) & (df2['padj'] < 0.05) & (df2['log2FoldChange'].abs() > 1.5)]

print(f"\nSignificant genes GluFe vs GluFePlus (iron-depleted): {len(sig1)}")
print(f"Significant genes Succ vs GluFePlus (innate media): {len(sig2)}")

# Save gene lists
sig1['gene'].to_csv("genes_iron_depleted.txt", index=False, header=False)
sig2['gene'].to_csv("genes_innate_media.txt", index=False, header=False)

# Also save all genes as background
df1['gene'].to_csv("all_genes.txt", index=False, header=False)
