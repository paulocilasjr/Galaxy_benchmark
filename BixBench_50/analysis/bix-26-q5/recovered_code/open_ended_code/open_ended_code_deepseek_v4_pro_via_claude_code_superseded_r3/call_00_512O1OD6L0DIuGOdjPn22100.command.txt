Rscript -e '
con <- gzfile("inputs/res_GluFevsGluFePlus.rds", "rb")
data <- unserialize(con)
close(con)

attrs <- attributes(data)
cat("listData class:", class(attrs$listData), "\n")
cat("listData is list:", is.list(attrs$listData), "\n")
cat("listData names:", paste(names(attrs$listData), collapse=", "), "\n")

# Print first few elements
for (nm in names(attrs$listData)) {
  cat("\n", nm, ":", class(attrs$listData[[nm]]), "length:", length(attrs$listData[[nm]]), "\n")
  if (length(attrs$listData[[nm]]) > 0) {
    cat("  First few:", paste(head(attrs$listData[[nm]], 5), collapse=", "), "\n")
  }
}
' 2>&1
