from Bio.KEGG import REST
import time

# Check if we can access KEGG for P. aeruginosa PA14 (pau)
try:
    # List organisms
    result = REST.kegg_list("organism").read()
    pae_lines = [l for l in result.split('\n') if 'aeruginosa' in l.lower()]
    for line in pae_lines:
        print(line)
except Exception as e:
    print(f"Error: {e}")

# Try to get gene list
try:
    genes = REST.kegg_list("pau").read()
    lines = genes.strip().split('\n')
    print(f"\nTotal KEGG genes for pau: {len(lines)}")
    for line in lines[:10]:
        print(line)
except Exception as e:
    print(f"Error getting genes: {e}")
