import pandas as pd
import json

# Reload and verify
ora_iron = pd.read_csv("ora_iron.csv")
ora_innate = pd.read_csv("ora_innate.csv")

iron_sig = set(ora_iron[ora_iron['padj'] < 0.05]['pathway_id'].tolist())
innate_sig = set(ora_innate[ora_innate['padj'] < 0.05]['pathway_id'].tolist())

iron_only = iron_sig - innate_sig

print(f"Iron-depleted significant pathways: {len(iron_sig)}")
print(f"Innate media significant pathways: {len(innate_sig)}")
print(f"Iron-only (not in innate): {len(iron_only)}")
print(f"\nIron-only pathways:")
for pid in sorted(iron_only):
    r = ora_iron[ora_iron['pathway_id'] == pid].iloc[0]
    # Check innate
    ri = ora_innate[ora_innate['pathway_id'] == pid]
    inn_padj = ri['padj'].values[0] if len(ri) > 0 else "no overlap (N/A)"
    print(f"  {pid}: {r['pathway_name'][:80]}")
    print(f"    Iron padj={r['padj']:.4e}, Innate padj={inn_padj}")

print(f"\nAnswer: {len(iron_only)}")
