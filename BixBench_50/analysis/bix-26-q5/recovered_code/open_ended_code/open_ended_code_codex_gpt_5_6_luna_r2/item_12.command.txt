/bin/bash -lc "Rscript - <<'RS'
for (f in c('inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')) {
  cat('\\n===', basename(f), '===\\n')
  x <- readRDS(f)
  a <- attributes(x)
  cat('S4:', isS4(x), ' rows:', length(a"'$rownames), '"'\\n')
  cat('attr names:', paste(names(a), collapse=' | '), '\\n')
  ld <- a"'$listData
  cat('"'listData length:', length(ld), ' names:', paste(names(ld), collapse=' | '), '\\n')
  for (nm in names(ld)) {
    z <- ld[[nm]]
    cat('  ', nm, ': class=', paste(class(z), collapse='/'), ' length=', length(z), ' head=', paste(utils::head(z, 4), collapse=' | '), '\\n', sep='')
  }
  em <- a"'$elementMetadata
  cat('"'elementMetadata class raw:', paste(class(em), collapse='/'), '\\n')
  if (isS4(em)) {
    ea <- attributes(em)
    cat('elementMetadata attrs:', paste(names(ea), collapse=' | '), '\\n')
    emld <- ea"'$listData
    cat('"'elementMetadata listData:', paste(names(emld), collapse=' | '), '\\n')
  }
  md <- a"'$metadata
  cat('"'metadata class:', paste(class(md), collapse='/'), ' length:', length(md), '\\n')
  if (length(md)) print(md)
}
RS"
