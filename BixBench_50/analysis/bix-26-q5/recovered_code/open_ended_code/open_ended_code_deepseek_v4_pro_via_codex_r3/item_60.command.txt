/bin/bash -c "Rscript - <<'RS'
library(jsonlite)
D<-fromJSON('/workspace/tmp/kegg_pau.json'); g2p<-D"'$g2p
p2g<-split(rep(names(g2p),lengths(g2p)),unlist(g2p)); extID<-unique(unlist(p2g,use.names=FALSE))
files<-list(GluFe='"'/workspace/tmp/GluFe_degs.txt',Succ='/workspace/tmp/Succ_degs.txt')
run_ora<-function(genes,minG=1,maxG=Inf){
 genes<-unique(genes); q<-g2p[genes]; keep<-"'!is.na(names(q)) & lengths(q)>0; q<-q[keep]
 qterm<-unique(unlist(q,use.names=FALSE)); qdf<-unique(do.call(rbind,lapply(names(q),function(g)data.frame(gene=g,term=q[[g]],stringsAsFactors=FALSE))))
 q2e<-lapply(split(qdf$gene,qdf$term),intersect,extID); t2e<-lapply(qterm,function(t)intersect(p2g[[t]],extID)); names(t2e)<-qterm
 idx<-lengths(t2e)>=minG & lengths(t2e)<=maxG; t2e<-t2e[idx]; q2e<-q2e[names(t2e)]; qterm<-unique(names(q2e)); k<-lengths(q2e)[qterm]; M<-lengths(t2e)[qterm]; N<-length(extID); n<-length(q); p<-phyper(k-1,M,N-M,n,lower.tail=FALSE); padj<-p.adjust(p,'"'BH'); data.frame(ID=qterm,k,M,pvalue=p,padj=padj)
}
for(nm in names(files)){
 for(mn in c(1,10)){for(mx in c(Inf,500)){res<-run_ora(readLines(files[[nm]]),mn,mx); res<-res[order(res"'$pvalue),]; sig<-res[res$padj<.05,]; cat(nm,'"'min',mn,'max',mx,'tested',nrow(res),'sig',nrow(sig),'\\n'); if(nm=='GluFe' & mn==1 & mx==Inf) print(head(sig,15));}}
}
RS"
