/bin/bash -c "Rscript - <<'RS'
library(jsonlite)
D <- fromJSON('/workspace/tmp/kegg_pau.json')
g2p <- D"'$g2p
files <- list(GluFe=c('"'/workspace/tmp/GluFe_degs.txt','/workspace/tmp/GluFe_all.txt'), Succ=c('/workspace/tmp/Succ_degs.txt','/workspace/tmp/Succ_all.txt'))
p2g <- split(rep(names(g2p), lengths(g2p)), unlist(g2p))
for(nm in names(files)){
  genes <- unique(readLines(files[[nm]][1])); universe <- unique(readLines(files[[nm]][2]))
  extID <- intersect(unique(unlist(p2g, use.names=FALSE)), universe)
  qExtID2TermID <- g2p[genes]
  keep <- "'!is.na(names(qExtID2TermID)) & lengths(qExtID2TermID)>0
  qExtID2TermID <- qExtID2TermID[keep]
  qTermID <- unique(unlist(qExtID2TermID, use.names=FALSE))
  qdf <- do.call(rbind, lapply(names(qExtID2TermID), function(g) data.frame(gene=g, term=qExtID2TermID[[g]], stringsAsFactors=FALSE)))
  qdf <- unique(qdf)
  qTermID2ExtID <- split(qdf$gene, qdf$term); qTermID2ExtID <- lapply(qTermID2ExtID, intersect, extID)
  termID2ExtID <- lapply(qTermID, function(t) intersect(p2g[[t]], extID)); names(termID2ExtID)<-qTermID
  idx <- lengths(termID2ExtID)>=10 & lengths(termID2ExtID)<=500
  termID2ExtID <- termID2ExtID[idx]; qTermID2ExtID <- qTermID2ExtID[names(termID2ExtID)]; qTermID<-unique(names(qTermID2ExtID))
  k<-lengths(qTermID2ExtID)[qTermID]; M<-lengths(termID2ExtID)[qTermID]; N<-length(extID); n<-length(qExtID2TermID)
  p<-phyper(k-1,M,N-M,n,lower.tail=FALSE); padj<-p.adjust(p,'"'BH')
  res<-data.frame(ID=qTermID,k=k,M=M,pvalue=p,padj=padj,stringsAsFactors=FALSE); res<-res[order(res"'$pvalue),]
  sig<-res[res$padj<.05,]
  cat('"'\\n==',nm,'universe all==\\nN',N,'n',n,'tested',nrow(res),'sig',nrow(sig),'\\n'); print(sig[,c('ID','k','M','pvalue','padj')],row.names=FALSE)
  write.csv(res, paste0('/workspace/tmp/kegg_',nm,'_universe_all.csv'), row.names=FALSE)
}
RS"
