/bin/bash -c "python3 - <<'PY'
import urllib.request, json
def fetch(url):
    req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    return urllib.request.urlopen(req, timeout=60).read().decode()
link=fetch('https://rest.kegg.jp/link/pathway/pau')
paths=fetch('https://rest.kegg.jp/list/pathway/pau')
gene2paths={}
for line in link.splitlines():
    if '\\t' not in line: continue
    g,p=line.split('\\t')
    g=g.split(':',1)[1]
    p=p.split(':',1)[1]
    gene2paths.setdefault(g,set()).add(p)
pathname={}
for line in paths.splitlines():
    if '\\t' not in line: continue
    p,name=line.split('\\t',1)
    p=p.split(':',1)[-1]
    pathname[p]=name
print('mapped genes',len(gene2paths),'pathways',len(pathname))
json.dump({'g2p':{k:sorted(v) for k,v in gene2paths.items()},'pathname':pathname}, open('/workspace/tmp/kegg_pau.json','w'))
print('path gene count min/max', min(len(v) for v in gene2paths.values()), max(len(v) for v in gene2paths.values()))
print('pathway sizes min/max', min(len([g for g,v in gene2paths.items() if 'pau00010' in v]) for _ in [0]), len([g for g,v in gene2paths.items() if 'pau00010' in v]))
PY"
