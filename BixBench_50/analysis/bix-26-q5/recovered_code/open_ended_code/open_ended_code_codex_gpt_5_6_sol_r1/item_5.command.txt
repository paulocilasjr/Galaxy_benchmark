/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
 x <- readRDS(f)
 rn <- attr(x, 'rownames')
 ld <- attr(x, 'listData')
 em <- attr(x, 'elementMetadata')
 cat('\\nFILE:', basename(f), '\\n')
 cat('rows:', length(rn), ' columns:', paste(names(ld), collapse=', '), '\\n')
 cat('descriptions:', paste(attr(em, 'listData')"'$description, collapse='"' || '), '\\n')
 d <- data.frame(gene=rn, as.data.frame(ld, check.names=FALSE), check.names=FALSE)
 cat('DE selected:', sum("'!is.na(d$log2FoldChange) & abs(d$log2FoldChange)>1.5 & !is.na(d$padj) & d$padj<0.05), '"'\\n')
 cat('padj non-NA:', sum("'!is.na(d$padj)), '"'baseMean > 0:', sum(d"'$baseMean>0), '"'\\n')
 cat('selected up/down:', sum(d"'$log2FoldChange>1.5 & !is.na(d$padj) & d$padj<0.05), sum(d$log2FoldChange< -1.5 & !is.na(d$padj) & d$padj<0.05), '"'\\n')
 cat('selected genes head:', paste(head(d"'$gene[abs(d$log2FoldChange)>1.5 & !is.na(d$padj) & d$padj<0.05],20),collapse='"', '), '\\n')
}
RS
Rscript -e \"ip <- installed.packages(); cat(paste(ip[grep('clusterProfiler|KEGG|AnnotationDbi|org\\\\.Pa', rownames(ip)),1], collapse='\\n'))\""
