/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
  cat('\\nFILE', f, '\\n')
  x <- readRDS(f)
  cat('class:', paste(class(x), collapse=','), '\\n')
  cat('dim:', paste(dim(x), collapse=' x '), '\\n')
  cat('names:', paste(names(x), collapse=' | '), '\\n')
  cat('rownames head:', paste(head(rownames(x), 12), collapse=' | '), '\\n')
  cat('attributes:\\n'); print(attributes(x)[setdiff(names(attributes(x)), c('row.names','names'))])
  print(head(as.data.frame(x), 8))
  cat('column summaries:\\n'); print(summary(as.data.frame(x)))
}
RS"
