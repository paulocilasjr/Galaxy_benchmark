import pandas as pd
from scipy.stats import fisher_exact
from statsmodels.stats.multitest import multipletests

# Sanity checks

# Load data
glufe = pd.read_csv('/workspace/res_GluFevsGluFePlus.csv')
succ = pd.read_csv('/workspace/res_SuccvsGluFePlus.csv')

# Check overlap of DEGs
glufe_deg = set(glufe[(glufe['log2FoldChange'].abs() > 1.5) & (glufe['padj'] < 0.05)]['gene'])
succ_deg = set(succ[(succ['log2FoldChange'].abs() > 1.5) & (succ['padj'] < 0.05)]['gene'])

print(f'GluFe DEGs: {len(glufe_deg)}')
print(f'Succ DEGs: {len(succ_deg)}')
print(f'Overlap: {len(glufe_deg & succ_deg)}')
print(f'GluFe only: {len(glufe_deg - succ_deg)}')
print(f'Succ only: {len(succ_deg - glufe_deg)}')

# Check that all gene IDs are proper PA14_ IDs
print(f'Sample gene IDs (GluFe): {list(sorted(glufe_deg))[:5]}')
print(f'Sample gene IDs (Succ): {list(sorted(succ_deg))[:5]}')

# Check pathway gene intersection
pathway_links = pd.read_csv('/workspace/pau_pathway_links.txt', sep='\t', header=None, names=['pathway', 'gene'])
pathway_links['gene_id'] = pathway_links['gene'].str.replace('pau:', '')
pathway_links['pathway_id'] = pathway_links['pathway'].str.replace('path:', '')
kegg_genes = set(pathway_links['gene_id'])
all_genes = set(glufe['gene'])
print(f'KEGG genes total: {len(kegg_genes)}')
print(f'Overlap with data: {len(kegg_genes & all_genes)}')
print(f'Pathways: {len(pathway_links[\"pathway_id\"].unique())}')
