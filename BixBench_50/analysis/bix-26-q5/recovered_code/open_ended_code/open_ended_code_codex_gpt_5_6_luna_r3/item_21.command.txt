/bin/bash -lc "Rscript - <<'RS'
gettxt <- function(url) readLines(url, warn=FALSE)
link <- read.table(text=paste(gettxt('https://rest.kegg.jp/link/pathway/pau'), collapse='\\n'), sep='\\t', header=FALSE, col.names=c('gene','pathway'), quote='', comment.char='')
path <- read.table(text=paste(gettxt('https://rest.kegg.jp/list/pathway/pau'), collapse='\\n'), sep='\\t', header=FALSE, col.names=c('pathway','name'), quote='', comment.char='')
cat('link rows',nrow(link),'unique genes',length(unique(link"'$gene)),'"'unique pathways',length(unique(link"'$pathway)),'"'list pathways',nrow(path),'\\n')
cat('pathway size summary all genes:\\n'); print(summary(table(link"'$pathway)))
for (f in list.files('"'/workspace/inputs',full.names=TRUE)) {
 x<-readRDS(f); rn<-attr(x,'rownames',exact=TRUE); d<-x@listData
 sig<-abs(d"'$log2FoldChange)>1.5 & d$padj<0.05
 mapped <- paste0('"'pau:',rn)
 ov <- intersect(mapped,unique(link"'$gene))
 cat(basename(f), '"'universe',length(rn),'mapped',length(intersect(mapped,unique(link"'$gene))), '"'sig',sum(sig),'sig mapped',length(intersect(mapped[sig],unique(link"'$gene))), '"'unmapped sig',sum(sig & "'!mapped%in%unique(link$gene)), '"'\\n')
 cat('unmapped first',paste(rn[sig & "'!mapped%in%unique(link$gene)][1:20],collapse='"','),'\\n')
}
RS"
