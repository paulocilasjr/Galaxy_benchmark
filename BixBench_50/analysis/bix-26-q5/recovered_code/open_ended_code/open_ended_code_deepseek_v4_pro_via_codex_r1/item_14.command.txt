/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
 cat('\\nFILE', f, '\\n')
 x <- readRDS(f)
 a <- attributes(x)
 cat('attr names:\\n'); print(names(a))
 for (nm in names(a)) {
   cat('ATTR', nm, 'typeof=', typeof(a[[nm]]), 'length=', length(a[[nm]]), 'class=', paste(class(a[[nm]]),collapse=','), '\\n')
   if (nm %in% c('rownames','nrows','elementType','metadata')) { print(head(a[[nm]], 3)); print(a[[nm]]) }
 }
}
RS"
