/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
 x <- readRDS(f)
 a <- attributes(x)
 genes <- a"'$rownames
 ld <- a$listData
 lfc <- ld$log2FoldChange
 padj <- ld$padj
 df <- data.frame(gene=genes, log2FoldChange=lfc, padj=padj, stringsAsFactors=FALSE)
 out <- paste0('"'/workspace/', sub('\\\\.rds"'$'"','.csv', basename(f)))
 write.csv(df, out, row.names=FALSE)
 cat(f, 'n=',nrow(df),'pass=',sum(abs(lfc)>1.5 & "'!is.na(padj) & padj<0.05, na.rm=TRUE),'"'\\n')
}
RS
ls -l /workspace/*.csv"
