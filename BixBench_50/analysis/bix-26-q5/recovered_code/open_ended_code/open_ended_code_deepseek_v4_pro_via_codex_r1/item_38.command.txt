def r_bh(p):
    p=np.asarray(p, dtype=float)
    n=len(p)
    if n==0: return np.array([])
    i=np.arange(n,0,-1)
    o=np.argsort(p, kind='stable')[::-1] # decreasing p? Wait R o <- order(p, decreasing=TRUE)
    ro=np.argsort(o, kind='stable')
    adj=pmin_one(np.minimum.accumulate((n/i)*p[o]))[ro]
    return adj

def pmin_one(x):
    return np.minimum(1.0, x)
