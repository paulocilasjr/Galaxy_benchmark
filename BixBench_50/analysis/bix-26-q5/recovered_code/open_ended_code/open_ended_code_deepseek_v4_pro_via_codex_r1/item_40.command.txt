/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, requests, os
from scipy.stats import hypergeom

def read_csv(path):
    df=pd.read_csv(path)
    df['padj']=pd.to_numeric(df['padj'], errors='coerce')
    df['log2FoldChange']=pd.to_numeric(df['log2FoldChange'], errors='coerce')
    return df

def fetch_kegg(species='pau', cache_dir='/tmp'):
    cache_link=os.path.join(cache_dir, f'kegg_link_pathway_{species}.txt')
    cache_list=os.path.join(cache_dir, f'kegg_list_pathway_{species}.txt')
    if not os.path.exists(cache_link):
        r=requests.get(f'https://rest.kegg.jp/link/pathway/{species}', timeout=120); r.raise_for_status(); open(cache_link,'w').write(r.text)
    if not os.path.exists(cache_list):
        r=requests.get(f'https://rest.kegg.jp/list/pathway/{species}', timeout=120); r.raise_for_status(); open(cache_list,'w').write(r.text)
    return open(cache_link).read().splitlines(), open(cache_list).read().splitlines()

def parse_mapping(link, names):
    rows=[]
    for line in link:
        if not line.strip(): continue
        parts=line.split('\\t')
        if len(parts)<2: continue
        g=parts[0].split(':',1)[-1]; path=parts[1].split(':',1)[-1]
        rows.append((path,g))
    p2g=pd.DataFrame(rows, columns=['path','gene']).dropna().drop_duplicates()
    p2n={}
    for line in names:
        if not line.strip(): continue
        parts=line.split('\\t')
        if len(parts)>=2: p2n[parts[0]]=parts[1]
    p2g=p2g[p2g['path'].isin(p2n.keys())].copy()
    return p2g,p2n

def r_bh(p):
    p=np.asarray(p,dtype=float); n=len(p)
    if n==0: return np.array([])
    i=np.arange(n,0,-1); o=np.argsort(p,kind='stable')[::-1]; ro=np.argsort(o,kind='stable')
    return np.minimum(1.0,np.minimum.accumulate((n/i)*p[o]))[ro]

def ora(genes,p2g):
    genes=list(dict.fromkeys(str(g) for g in genes))
    g2p=p2g.groupby('gene')['path'].apply(list).to_dict()
    qmap={g:g2p[g] for g in genes if g in g2p}
    if not qmap: return None
    rows=[]
    for g,ps in qmap.items():
        for path in ps: rows.append((path,g))
    df=pd.DataFrame(rows,columns=['path','gene']).drop_duplicates()
    qterm2ext={path:list(grp['gene']) for path,grp in df.groupby('path')}
    extID=list(dict.fromkeys(p2g['gene']))
    qterm2ext={path:[g for g in gs if g in extID] for path,gs in qterm2ext.items()}
    qterms=list(qterm2ext.keys())
    term2ext={path:[g for g in p2g.loc[p2g.path==path,'gene'] if g in extID] for path in qterms}
    idx=[path for path in qterms if 10<=len(term2ext[path])<=500]
    qterms=[path for path in idx]
    qterm2ext={path:qterm2ext[path] for path in qterms}
    term2ext={path:term2ext[path] for path in qterms}
    qterms=list(dict.fromkeys(qterm2ext.keys()))
    if not qterms: return None
    k=np.array([len(qterm2ext[path]) for path in qterms],dtype=int)
    M=np.array([len(term2ext[path]) for path in qterms],dtype=int)
    N=len(extID); n=len(qmap)
    pvals=np.array([hypergeom.sf(kk-1,N,mm,n) for kk,mm in zip(k,M)],dtype=float)
    adj=r_bh(pvals)
    return pd.DataFrame({'ID':qterms,'Count':k,'Size':M,'pvalue':pvals,'p.adjust':adj,
                         'geneID':['/'.join(qterm2ext[path]) for path in qterms]})

def show(label,file,p2g,p2n):
    df=read_csv(file); sig=df[(df.log2FoldChange.abs()>1.5)&(df.padj<0.05)]
    genes=sig.gene.tolist(); print('\\n',label,'input',len(genes))
    res=ora(genes,p2g)
    if res is None: print('no result'); return
    print('tested',len(res),'sig',(res['p.adjust']<0.05).sum())
    for row in res[res['p.adjust']<0.05].sort_values('p.adjust').itertuples():
        print(row.ID, row.Count, row.Size, row.pvalue, row.p_adjust if '_' in '' else row[5], p2n.get(row.ID))
    print('nonsig top:')
    print(res.sort_values('p.adjust').head(20).to_string(index=False))

link,names=fetch_kegg('pau'); p2g,p2n=parse_mapping(link,names)
print('mapping rows',len(p2g),'genes',p2g.gene.nunique(),'terms',p2g.path.nunique())
show('GluFe','/workspace/res_GluFevsGluFePlus.csv',p2g,p2n)
show('Succ','/workspace/res_SuccvsGluFePlus.csv',p2g,p2n)
PY"
