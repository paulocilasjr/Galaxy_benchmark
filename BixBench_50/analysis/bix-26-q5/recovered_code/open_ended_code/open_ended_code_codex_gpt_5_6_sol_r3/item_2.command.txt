/bin/bash -lc "Rscript - <<'RS'
paths <- list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)
for (p in paths) {
  x <- readRDS(p)
  cat('\\nFILE', p, '\\n')
  cat('class:', paste(class(x), collapse=','), ' dim:', paste(dim(x), collapse='x'), '\\n')
  cat('names:', paste(names(x), collapse=', '), '\\n')
  cat('rownames head:', paste(head(rownames(x), 12), collapse=', '), '\\n')
  print(head(as.data.frame(x), 8))
  cat('attributes:\\n')
  print(attributes(x)[setdiff(names(attributes(x)), c('names','row.names','class'))])
}
RS"
