/bin/bash -lc "Rscript - <<'RS'
read_de <- function(p) { x <- readRDS(p); data.frame(gene=attr(x,'rownames'), attr(x,'listData'), check.names=FALSE) }
fg <- function(d) unique(d"'$gene[is.finite(d$log2FoldChange) & !is.na(d$padj) & abs(d$log2FoldChange)>1.5 & d$padj<0.05])
D1 <- read_de('"'inputs/res_GluFevsGluFePlus.rds'); D2 <- read_de('inputs/res_SuccvsGluFePlus.rds')
L <- read.delim('https://rest.kegg.jp/link/pathway/pau', header=FALSE, col.names=c('gene','path'))
L"'$gene <- sub('"'"'^pau:'"','',L"'$gene); L$path <- sub('"'"'^path:'"','',L"'$path); L <- unique(L)
ora <- function(g, universe) {
 universe <- intersect(unique(universe), unique(L$gene)); g <- intersect(g, universe)
 sets <- lapply(split(L$gene,L$path), function(z) intersect(unique(z),universe))
 sets <- sets[lengths(sets)>=10 & lengths(sets)<=500]
 hit <- vapply(sets, function(z) sum(z %in% g), integer(1)); hit <- hit[hit>0]; sets <- sets[names(hit)]
 pv <- phyper(hit-1, lengths(sets), length(universe)-lengths(sets), length(g), lower.tail=FALSE)
 setNames(p.adjust(pv, method='"'BH'), names(pv))
}
univ <- union(D1"'$gene,D2$gene)
p1 <- ora(fg(D1),univ); p2 <- ora(fg(D2),univ)
s1 <- names(p1)[p1<0.05]; s2 <- names(p2)[p2<0.05]; only <- setdiff(s1,s2)
stopifnot(length(fg(D1))==224L, length(fg(D2))==541L, identical(only,'"'pau01110'))
cat('foregrounds:',length(fg(D1)),length(fg(D2)),'\\n')
cat('significant pathway counts:',length(s1),length(s2),'\\n')
cat('iron-only:',paste(only,collapse=','),' count:',length(only),'\\n')
cat('adjusted p-values for iron-only pathway:',p1[only],p2[only],'\\n')
RS"
