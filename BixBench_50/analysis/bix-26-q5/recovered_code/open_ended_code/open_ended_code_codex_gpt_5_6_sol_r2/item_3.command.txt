/bin/bash -lc "Rscript - <<'RS'
files <- c('inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')
for (f in files) {
  x <- readRDS(f)
  cat('\\nFILE', f, '\\n')
  cat('class:', paste(class(x), collapse=', '), '\\n')
  cat('dim:', paste(dim(x), collapse=' x '), '\\n')
  cat('names:', paste(names(x), collapse=', '), '\\n')
  cat('rownames sample:', paste(head(rownames(x), 10), collapse=', '), '\\n')
  print(head(as.data.frame(x), 6))
  cat('attributes:', paste(names(attributes(x)), collapse=', '), '\\n')
  str(attributes(x), max.level=2)
}
RS"
