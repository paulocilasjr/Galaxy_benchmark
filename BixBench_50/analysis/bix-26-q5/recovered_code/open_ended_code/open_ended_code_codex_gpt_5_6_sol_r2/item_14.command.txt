/bin/bash -lc "python - <<'PY'
import pyreadr, rdata
files=['inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds']
for f in files:
 print('\\nFILE',f)
 try:
  o=pyreadr.read_r(f); print('pyreadr',o)
 except Exception as e: print('pyreadr error:',repr(e))
 try:
  p=rdata.parser.parse_file(f); print('rdata parsed:',type(p),repr(p)[:1000])
 except Exception as e: print('rdata parse error:',repr(e))
 try:
  c=rdata.conversion.convert(p); print('converted:',type(c),repr(c)[:2000])
 except Exception as e: print('rdata convert error:',repr(e))
PY"
