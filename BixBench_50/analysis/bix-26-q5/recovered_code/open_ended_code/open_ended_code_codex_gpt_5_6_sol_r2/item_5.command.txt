/bin/bash -lc "Rscript - <<'RS'
files <- c('inputs/res_GluFevsGluFePlus.rds','inputs/res_SuccvsGluFePlus.rds')
for (f in files) {
  x <- readRDS(f)
  cat('\\nFILE', f, '\\n')
  cat('typeof:', typeof(x), ' isS4:', isS4(x), ' length:', length(x), '\\n')
  cat('attribute names:', paste(names(attributes(x)), collapse=', '), '\\n')
  cat('raw names:', paste(names(x), collapse=', '), '\\n')
  print(str(unclass(x), max.level=2, give.attr=FALSE))
}
RS"
