/bin/bash -lc "python - <<'PY'
import zipfile, xml.etree.ElementTree as ET, re
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
ns='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
with zipfile.ZipFile(p) as z:
    root=ET.parse(z.open('xl/sharedStrings.xml')).getroot()
    ss=[''.join(t.text or '' for t in si.iter(ns+'t')) for si in root]
    print('shared strings',len(ss))
    with z.open('xl/worksheets/sheet3.xml') as f:
        for ev,e in ET.iterparse(f,events=('end',)):
            if e.tag==ns+'row':
                vals=[]
                for c in e.findall(ns+'c'):
                    ref=c.attrib.get('r'); typ=c.attrib.get('t')
                    v=c.find(ns+'v')
                    val=None if v is None else v.text
                    if typ=='s' and val is not None: val=ss[int(val)]
                    elif typ=='inlineStr': val=''.join(t.text or '' for t in c.iter(ns+'t'))
                    vals.append((ref,val))
                print('ROW',e.attrib.get('r'),vals)
                if int(e.attrib.get('r'))>=8: break
                e.clear()
PY"
