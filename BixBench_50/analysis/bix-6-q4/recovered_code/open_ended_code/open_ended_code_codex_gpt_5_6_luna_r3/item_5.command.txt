/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import math

p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
ids=[]; x=[]; y=[]; bad=[]
for i,row in enumerate(ws.iter_rows(min_row=2, values_only=True), start=2):
    if not row or row[0] is None:
        continue
    ident,a,b=row[:3]
    ids.append(ident)
    if isinstance(a,(int,float)) and isinstance(b,(int,float)) and math.isfinite(a) and math.isfinite(b):
        x.append(float(a)); y.append(float(b))
    else:
        bad.append((i,ident,a,b))

def average_ranks(vals):
    order=sorted(range(len(vals)), key=vals.__getitem__)
    ranks=[0.0]*len(vals)
    i=0
    while i<len(order):
        j=i+1
        v=vals[order[i]]
        while j<len(order) and vals[order[j]]==v:
            j+=1
        rank=(i+1+j)/2.0
        for k in order[i:j]:
            ranks[k]=rank
        i=j
    return ranks

def pearson(a,b):
    n=len(a)
    ma=sum(a)/n; mb=sum(b)/n
    da=[v-ma for v in a]; db=[v-mb for v in b]
    num=sum(u*v for u,v in zip(da,db))
    den=math.sqrt(sum(u*u for u in da)*sum(v*v for v in db))
    return num/den
rx=average_ranks(x); ry=average_ranks(y)
r=pearson(rx,ry)
print('n rows:', len(ids))
print('n valid:', len(x))
print('invalid:', len(bad), bad[:5])
print('duplicate IDs:', len(ids)-len(set(ids)))
print('unique S1 p-values:', len(set(x)))
print('unique S2 p-values:', len(set(y)))
print('S1 range:', min(x), max(x))
print('S2 range:', min(y), max(y))
print('spearman:', format(r,'.17g'))
PY"
