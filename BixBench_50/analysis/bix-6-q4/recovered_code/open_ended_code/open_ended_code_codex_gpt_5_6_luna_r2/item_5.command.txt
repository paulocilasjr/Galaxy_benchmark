/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import math

p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
headers=next(ws.iter_rows(min_row=1,max_row=1,values_only=True))
idx={h:i for i,h in enumerate(headers)}
ids=[]; x=[]; y=[]; missing=[]; nonnum=[]
for r,row in enumerate(ws.iter_rows(min_row=2, values_only=True), start=2):
    a=row[idx['Chronic Round1 S1']]; b=row[idx['Chronic Round1 S2']]; rid=row[idx['RefSeq ID']]
    if a is None or b is None:
        missing.append((r,rid,a,b)); continue
    if not isinstance(a,(int,float)) or not isinstance(b,(int,float)):
        nonnum.append((r,rid,a,b)); continue
    if not (math.isfinite(float(a)) and math.isfinite(float(b))):
        missing.append((r,rid,a,b)); continue
    ids.append(rid); x.append(float(a)); y.append(float(b))

def average_ranks(vals):
    # 1-based ranks; equal numeric values get their average rank.
    order=sorted(range(len(vals)), key=vals.__getitem__)
    ranks=[0.0]*len(vals)
    i=0
    while i<len(order):
        j=i+1
        v=vals[order[i]]
        while j<len(order) and vals[order[j]]==v:
            j+=1
        avg=(i+1+j)/2.0
        for k in order[i:j]:
            ranks[k]=avg
        i=j
    return ranks

def pearson(a,b):
    n=len(a); ma=sum(a)/n; mb=sum(b)/n
    cov=sum((u-ma)*(v-mb) for u,v in zip(a,b))
    va=sum((u-ma)**2 for u in a); vb=sum((v-mb)**2 for v in b)
    return cov/math.sqrt(va*vb)
rx=average_ranks(x); ry=average_ranks(y)
rho=pearson(rx,ry)
print('headers:', headers)
print('rows total:', ws.max_row-1)
print('used:', len(x), 'missing/non-numeric:', len(missing), len(nonnum))
print('unique RefSeq IDs:', len(set(ids)), 'duplicate rows:', len(ids)-len(set(ids)))
print('ties:', len(x)-len(set(x)), len(y)-len(set(y)))
print('rho:', format(rho,'.17g'))
print('rho_fixed_15:', format(rho,'.15f'))
PY"
