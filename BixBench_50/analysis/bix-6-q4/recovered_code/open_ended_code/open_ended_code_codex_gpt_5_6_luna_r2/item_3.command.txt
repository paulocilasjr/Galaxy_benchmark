/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from scipy.stats import spearmanr, rankdata
import math
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
headers=next(ws.iter_rows(min_row=1,max_row=1,values_only=True))
print(headers)
idx={h:i for i,h in enumerate(headers)}
ids=[]; x=[]; y=[]
missing=[]; nonnum=[]
for r,row in enumerate(ws.iter_rows(min_row=2,values_only=True), start=2):
    a=row[idx['Chronic Round1 S1']]; b=row[idx['Chronic Round1 S2']]
    rid=row[idx['RefSeq ID']]
    if a is None or b is None:
        missing.append((r,rid,a,b)); continue
    if not isinstance(a,(int,float)) or not isinstance(b,(int,float)):
        nonnum.append((r,rid,a,b)); continue
    if not (math.isfinite(a) and math.isfinite(b)):
        missing.append((r,rid,a,b)); continue
    ids.append(rid); x.append(float(a)); y.append(float(b))
print('rows total', ws.max_row-1, 'used',len(x),'missing',len(missing),'nonnum',len(nonnum))
print('unique IDs',len(set(ids)),'duplicate count',len(ids)-len(set(ids)))
print('x range',min(x),max(x),'y range',min(y),max(y))
print('ties x',len(x)-len(set(x)),'ties y',len(y)-len(set(y)))
r=spearmanr(x,y)
print('spearman',repr(r.statistic),'pvalue',repr(r.pvalue))
rx=rankdata(x,method='average'); ry=rankdata(y,method='average')
print('pearson ranks',repr(spearmanr(rx,ry).statistic))
# alternative scipy via covariance
mx=sum(rx)/len(rx); my=sum(ry)/len(ry)
num=sum((a-mx)*(b-my) for a,b in zip(rx,ry)); den=math.sqrt(sum((a-mx)**2 for a in rx)*sum((b-my)**2 for b in ry))
print('manual',repr(num/den))
PY"
