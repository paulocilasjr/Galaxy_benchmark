/bin/bash -lc "python - <<'PY'
import pandas as pd
from scipy.stats import spearmanr
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
df=pd.read_excel(p, sheet_name='MAGeCK P-values', engine='openpyxl')
print(df.shape)
print(df.columns.tolist())
cols=['Chronic Round1 S1','Chronic Round1 S2']
for c in cols:
    df[c]=pd.to_numeric(df[c], errors='coerce')
print(df[cols].isna().sum())
paired=df[cols].dropna()
print('n', len(paired))
print('duplicates RefSeq', df['RefSeq ID'].duplicated().sum())
print('pandas', paired.corr(method='spearman').iloc[0,1])
res=spearmanr(paired[cols[0]], paired[cols[1]])
print('scipy statistic', res.statistic)
print('scipy pvalue', res.pvalue)
print('minmax', paired.min().to_dict(), paired.max().to_dict())
PY"
