/bin/bash -lc "python - <<'PY'
import openpyxl, math
from collections import Counter

p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
header=next(ws.iter_rows(values_only=True))
cols={name:i for i,name in enumerate(header)}
i1=cols['Chronic Round1 S1']
i2=cols['Chronic Round1 S2']
refi=cols['RefSeq ID']
xs=[]; ys=[]; refs=[]
missing=0; nonnum=0
for row in ws.iter_rows(min_row=2, values_only=True):
    x=row[i1]; y=row[i2]
    if x is None or y is None:
        missing += 1
        continue
    try:
        x=float(x); y=float(y)
    except Exception:
        nonnum += 1
        continue
    if math.isnan(x) or math.isnan(y):
        missing += 1
        continue
    xs.append(x); ys.append(y); refs.append(row[refi])

def rankdata_avg(vals):
    n=len(vals)
    order=sorted(range(n), key=lambda i: vals[i])
    ranks=[0.0]*n
    pos=0
    while pos<n:
        start=pos
        v=vals[order[pos]]
        while pos+1<n and vals[order[pos+1]] == v:
            pos += 1
        end=pos
        avg=(start+1 + end+1)/2.0
        for j in range(start, end+1):
            ranks[order[j]]=avg
        pos += 1
    return ranks

def pearson(a,b):
    n=len(a)
    ma=sum(a)/n; mb=sum(b)/n
    cov=sum((u-ma)*(v-mb) for u,v in zip(a,b))
    va=sum((u-ma)**2 for u in a)
    vb=sum((v-mb)**2 for v in b)
    return cov/math.sqrt(va*vb)

rx=rankdata_avg(xs)
ry=rankdata_avg(ys)
rho=pearson(rx, ry)
print('n', len(xs), 'missing', missing, 'nonnum', nonnum)
print('duplicate refs', len(refs)-len(set(refs)))
print('ties x values', sum(1 for c in Counter(xs).values() if c>1), 'ties y values', sum(1 for c in Counter(ys).values() if c>1))
print('rho repr', repr(rho))
print('rho .15g', format(rho, '.15g'))
print('rho .17g', format(rho, '.17g'))
PY"
