/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from math import sqrt
path='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=load_workbook(path, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
header=next(ws.iter_rows(values_only=True))
i1=header.index('Chronic Round1 S1')
i2=header.index('Chronic Round1 S2')
x=[]; y=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    x.append(float(row[i1]))
    y.append(float(row[i2]))

def average_ranks(values):
    n=len(values)
    order=sorted(range(n), key=values.__getitem__)
    ranks=[0.0]*n
    i=0
    while i<n:
        j=i+1
        v=values[order[i]]
        while j<n and values[order[j]] == v:
            j += 1
        # ranks are 1-based; average of i+1 through j
        avg=(i+1 + j)/2.0
        for k in range(i,j):
            ranks[order[k]]=avg
        i=j
    return ranks
rx=average_ranks(x)
ry=average_ranks(y)
n=len(rx)
mx=sum(rx)/n
my=sum(ry)/n
num=sum((a-mx)*(b-my) for a,b in zip(rx,ry))
den=sqrt(sum((a-mx)**2 for a in rx)*sum((b-my)**2 for b in ry))
r=num/den
print(n)
print(format(r, '.17g'))
print(format(r, '.15f'))
# tie counts diagnostic
print('unique', len(set(x)), len(set(y)))
PY"
