/bin/bash -lc "python3 - <<'PY'
import openpyxl, math, statistics
p='/workspace/inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
rows=ws.iter_rows(values_only=True)
header=next(rows)
print(header)
idx=[header.index('Chronic Round1 S1'), header.index('Chronic Round1 S2')]
a=[];b=[]
for row in rows:
    if len(row)>max(idx):
        x=row[idx[0]]; y=row[idx[1]]
        if x is None or y is None or isinstance(x,str) or isinstance(y,str):
            continue
        a.append(float(x)); b.append(float(y))
print(len(a))
# ranks with average ties
def rankdata(vals):
    order=sorted(range(len(vals)), key=lambda i: vals[i])
    ranks=[0.0]*len(vals)
    i=0
    n=len(vals)
    while i<n:
        j=i
        while j+1<n and vals[order[j+1]]==vals[order[i]]:
            j+=1
        avg=(i+j+2)/2.0 # rank starts 1
        for k in range(i,j+1):
            ranks[order[k]]=avg
        i=j+1
    return ranks
ra=rankdata(a); rb=rankdata(b)
mean_a=sum(ra)/len(ra); mean_b=sum(rb)/len(rb)
cov=sum((ra[i]-mean_a)*(rb[i]-mean_b) for i in range(len(ra)))
sa=math.sqrt(sum((x-mean_a)**2 for x in ra)); sb=math.sqrt(sum((x-mean_b)**2 for x in rb))
r=cov/(sa*sb)
print('r', repr(r), format(r,'.15g'))
# also correlation raw? no
print('n',len(a))
print('minmax', min(a),max(a), min(b),max(b))
PY"
