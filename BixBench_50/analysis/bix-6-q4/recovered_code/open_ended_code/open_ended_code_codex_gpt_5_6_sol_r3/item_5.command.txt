/bin/bash -lc "python - <<'PY'
import openpyxl, math
from collections import Counter
from scipy.stats import spearmanr, rankdata, pearsonr
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
ids=[]; x=[]; y=[]; bad=[]
for rn,row in enumerate(ws.iter_rows(min_row=2, values_only=True),2):
    ident,a,b=row[0],row[5],row[6]
    ids.append(ident)
    if isinstance(a,(int,float)) and isinstance(b,(int,float)) and math.isfinite(a) and math.isfinite(b):
        x.append(float(a)); y.append(float(b))
    else:
        bad.append((rn,ident,a,b))
print('total data rows',len(ids),'valid pairs',len(x),'invalid pairs',len(bad))
print('invalid examples',bad[:20])
print('duplicate identifiers',sum(v-1 for v in Counter(ids).values() if v>1),'duplicate groups',sum(v>1 for v in Counter(ids).values()))
print('x min/max/zeros/ones/unique',min(x),max(x),sum(v==0 for v in x),sum(v==1 for v in x),len(set(x)))
print('y min/max/zeros/ones/unique',min(y),max(y),sum(v==0 for v in y),sum(v==1 for v in y),len(set(y)))
r=spearmanr(x,y)
print('scipy spearman',repr(r.statistic),'pvalue',repr(r.pvalue))
rx=rankdata(x,method='average'); ry=rankdata(y,method='average')
print('pearson average ranks',repr(pearsonr(rx,ry).statistic))
PY"
