import openpyxl
from scipy.stats import spearmanr
import numpy as np

wb = openpyxl.load_workbook('/workspace/inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx', read_only=True)
ws = wb['MAGeCK P-values']

# Read header
header = None
data_rows = []
for i, row in enumerate(ws.iter_rows(values_only=True)):
    if i == 0:
        header = list(row)
        print('Header:', header)
        # Find indices for Chronic Round1 S1 and S2
        idx_s1 = header.index('Chronic Round1 S1')
        idx_s2 = header.index('Chronic Round1 S2')
        print(f'Chronic Round1 S1 at col {idx_s1}, Chronic Round1 S2 at col {idx_s2}')
    else:
        data_rows.append(row)

print(f'Total gene rows: {len(data_rows)}')

# Extract P-values
s1_vals = []
s2_vals = []
missing = 0
for row in data_rows:
    v1 = row[idx_s1]
    v2 = row[idx_s2]
    if v1 is None or v2 is None:
        missing += 1
        continue
    s1_vals.append(float(v1))
    s2_vals.append(float(v2))

print(f'Missing pairs: {missing}')
print(f'Valid pairs: {len(s1_vals)}')

# Compute Spearman correlation
rho, pval = spearmanr(s1_vals, s2_vals)
print(f'Spearman rho: {rho}')
print(f'P-value: {pval}')
print(f'Spearman rho (full precision): {rho:.15f}')
