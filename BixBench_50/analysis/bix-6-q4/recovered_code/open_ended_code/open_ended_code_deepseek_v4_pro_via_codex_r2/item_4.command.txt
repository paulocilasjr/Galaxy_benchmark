/bin/bash -lc "cd /workspace && python - <<'PY'
import openpyxl, math, statistics
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
rows=ws.iter_rows(values_only=True)
header=next(rows)
print(header)
i1=header.index('Chronic Round1 S1')
i2=header.index('Chronic Round1 S2')
a=[]; b=[]; refs=[]
nulls=0; nonnumeric=[]
for r in rows:
    refs.append(r[0])
    x=r[i1]; y=r[i2]
    if x is None or y is None:
        nulls+=1; continue
    if isinstance(x,(int,float)) and isinstance(y,(int,float)) and math.isfinite(x) and math.isfinite(y):
        a.append(x); b.append(y)
    else:
        nonnumeric.append((r[0],x,y))
print('total', len(refs), 'pairs', len(a), 'nulls', nulls, 'nonnumeric', len(nonnumeric), nonnumeric[:3])
print('minmax a,b', min(a),max(a), min(b),max(b))
# duplicates? count distinct refs and duplicates
print('distinct refs', len(set(refs)), 'dups', len(refs)-len(set(refs)))
# check p-value ranges and any zero
print('zeros a,b', a.count(0), b.count(0), 'ones', a.count(1), b.count(1))
PY"
