/bin/bash -lc "cd /workspace && python - <<'PY'
import openpyxl, math
from decimal import Decimal, getcontext
getcontext().prec=50
p='inputs/JuliaJong_CRISPRa_BCL2_B3GNT_cacerResTcellCytotoxicity_supplData1_mageck.xlsx'
wb=openpyxl.load_workbook(p, read_only=True, data_only=True)
ws=wb['MAGeCK P-values']
rows=ws.iter_rows(values_only=True)
header=next(rows)
i1=header.index('Chronic Round1 S1'); i2=header.index('Chronic Round1 S2')
a=[];b=[]
for r in rows:
    a.append(r[i1]); b.append(r[i2])
n=len(a)

def rank_avg(vals):
    order=sorted(range(n), key=lambda i: vals[i])
    ranks=[0.0]*n
    j=0
    while j<n:
        k=j
        v=vals[order[j]]
        while k+1<n and vals[order[k+1]]==v:
            k+=1
        avg=(j+1 + k+1)/2.0 # rank average using 1-index
        for idx in order[j:k+1]:
            ranks[idx]=avg
        j=k+1
    return ranks
ra=rank_avg(a); rb=rank_avg(b)
# pearson corr
ma=sum(ra)/n; mb=sum(rb)/n
cov=sum((ra[i]-ma)*(rb[i]-mb) for i in range(n))
va=sum((x-ma)**2 for x in ra); vb=sum((x-mb)**2 for x in rb)
rho=cov/math.sqrt(va*vb)
print('rho float', repr(rho), format(rho,'.15f'), format(rho,'.17g'))
# check rank sums etc
print('rank sums', sum(ra), sum(rb), n*(n+1)/2)
# decimal exact perhaps
D=Decimal
ra_d=[D(str(x)) for x in ra]; rb_d=[D(str(y)) for y in rb]
ma_d=sum(ra_d)/D(n); mb_d=sum(rb_d)/D(n)
cov_d=sum((ra_d[i]-ma_d)*(rb_d[i]-mb_d) for i in range(n))
va_d=sum((x-ma_d)**2 for x in ra_d); vb_d=sum((x-mb_d)**2 for x in rb_d)
rho_d=cov_d/(va_d*vb_d).sqrt()
print('rho decimal', rho_d)
PY"
