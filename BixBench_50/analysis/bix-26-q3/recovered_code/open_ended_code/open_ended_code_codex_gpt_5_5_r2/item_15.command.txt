/bin/bash -lc "Rscript - <<'RS'
x<-readRDS('inputs/res_GluFevsGluFePlus.rds')
ld<-attr(x,'listData', exact=TRUE)
cat('ld typeof',typeof(ld),' len',length(ld),' names',paste(names(ld),collapse=','),'\\n')
for(i in seq_along(ld)){
  z<-ld[[i]]
  cat(i, names(ld)[i], typeof(z), length(z), paste(names(attributes(z)), collapse=','), '\\n')
  if(is.atomic(z)){
    print(head(z))
    cat('NAs', sum(is.na(z)),'\\n')
  }
}
md<-attr(x,'metadata', exact=TRUE)
cat('metadata names',paste(names(md),collapse=','),' len',length(md),'\\n')
print(md)
RS"
