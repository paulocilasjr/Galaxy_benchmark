/bin/bash -lc "Rscript - <<'RS'
x<-readRDS('inputs/res_GluFevsGluFePlus.rds')
ld<-attr(x,'listData', exact=TRUE)
df<-data.frame(gene=attr(x,'rownames', exact=TRUE), log2FoldChange=ld"'$log2FoldChange, padj=ld$padj)
fg<-df$gene[!is.na(df$padj) & df$padj < 0.05 & df$log2FoldChange > 1.5]
tf<-tempfile(); download.file('"'https://rest.kegg.jp/link/pathway/pau', tf, quiet=TRUE); links<-read.delim(tf, header=FALSE, stringsAsFactors=FALSE); colnames(links)<-c('gene','pathway'); links"'$gene<-sub('"'"'^pau:'"','',links"'$gene); links$pathway<-sub('"'"'^path:'"','',links"'$pathway)
tf2<-tempfile(); download.file('"'https://rest.kegg.jp/list/pathway/pau', tf2, quiet=TRUE); nm<-read.delim(tf2, header=FALSE, stringsAsFactors=FALSE); colnames(nm)<-c('pathway','description'); nm"'$pathway<-sub('"'"'^path:'"','',nm"'$pathway)
for(type in c('"'all_kegg','de_table')){
 universe<-if(type=='all_kegg') unique(links"'$gene) else intersect(df$gene, unique(links$gene))
 fgm<-intersect(fg, universe)
 res<-do.call(rbind,lapply(unique(links$pathway), function(pid){ gs<-intersect(unique(links$gene[links$pathway==pid]), universe); k<-sum(fgm%in%gs); data.frame(pathway=pid,n=length(gs),k=k,pvalue=phyper(k-1,length(gs),length(universe)-length(gs),length(fgm),lower.tail=FALSE)) }))
 res<-res[res$k>0 & res$n >= 10 & res$n <= 500,]
 res$padj<-p.adjust(res$pvalue,'"'BH'); res<-merge(res,nm,by='pathway',all.x=TRUE); res<-res[order(res"'$pvalue),]
 cat('"'\\n',type,'M',length(universe),'N',length(fgm),'terms',nrow(res),'\\n')
 print(res[grep('ABC transporters',res"'$description,fixed=TRUE),], row.names=FALSE)
 print(head(res,5), row.names=FALSE)
}
RS'
