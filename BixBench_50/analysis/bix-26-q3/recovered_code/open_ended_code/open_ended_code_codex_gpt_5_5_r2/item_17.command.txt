/bin/bash -lc "Rscript - <<'RS'
x<-readRDS('inputs/res_GluFevsGluFePlus.rds')
ld<-attr(x,'listData', exact=TRUE)
df<-data.frame(gene=attr(x,'rownames', exact=TRUE), baseMean=ld"'$baseMean, log2FoldChange=ld$log2FoldChange, lfcSE=ld$lfcSE, stat=ld$stat, pvalue=ld$pvalue, padj=ld$padj, stringsAsFactors=FALSE)
fg<-df$gene[!is.na(df$padj) & df$padj < 0.05 & !is.na(df$log2FoldChange) & df$log2FoldChange > 1.5]
cat('"'foreground n', length(fg), '\\n')
cat('first', paste(head(fg,20), collapse=' '), '\\n')
# Try KEGG organism code pau (P. aeruginosa PA14) and retrieve ABC transporters genes.
url <- 'https://rest.kegg.jp/link/pau/path:pau02010'
tf <- tempfile()
status <- tryCatch(download.file(url, tf, quiet=TRUE), error=function(e)e)
print(status)
if(file.exists(tf)){
  k <- read.delim(tf, header=FALSE, stringsAsFactors=FALSE)
  print(head(k)); cat('kegg rows', nrow(k),'\\n')
  pathway_genes <- sub('"'^pau:'"', '', k"'$V2)
  hits <- intersect(fg, pathway_genes)
  cat('"'hits n', length(hits), '\\n')
  cat(paste(sort(hits), collapse='\\n'), '\\n')
}
RS"
