/bin/bash -lc "Rscript - <<'RS'
options(timeout=60)
ln <- readLines('https://rest.kegg.jp/link/pau/pathway', warn=FALSE)
cat('links', length(ln), '\\n')
parts <- strsplit(ln, '\\t', fixed=TRUE)
path <- sub('"'^path:'"', '', vapply(parts, "'`[`, '"'', 1))
gene <- sub('"'^pau:'"', '', vapply(parts, "'`[`, '"'', 2))
tab <- data.frame(path=path,gene=gene,stringsAsFactors=FALSE)
sets <- split(tab"'$gene, tab$path)
sets <- lapply(sets, unique)
sets <- sets[lengths(sets)>=10 & lengths(sets)<=500]
cat('"'pathways tested',length(sets),'universe',length(unique(unlist(sets))), '\\n')
x <- readRDS('inputs/res_GluFevsGluFePlus.rds'); a <- attributes(x)
d <- data.frame(gene=a[['rownames']], a[['listData']], check.names=FALSE)
fg <- d"'$gene[!is.na(d$log2FoldChange)&!is.na(d$padj)&d$log2FoldChange > 1.5&d$padj < 0.05]
u <- intersect(unique(unlist(sets)), d$gene)
fg <- intersect(fg,u)
res <- lapply(names(sets), function(k) {
 s <- sets[[k]]; h <- sum(fg %in% s); M <- length(u); K <- length(s); n <- length(fg)
 data.frame(path=k,set=K,hits=h,p=phyper(h-1,K,M-K,n,lower.tail=FALSE))
})
r <- do.call(rbind,res); r$padj <- p.adjust(r$p,'"'BH')
print(r[r"'$path=='"'pau02010',],row.names=FALSE)
print(head(r[order(r"'$padj,r$p),],20),row.names=FALSE)
RS'
