/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n###', f, '\\n')
  x <- readRDS(f)
  cat('class:', paste(class(x), collapse=','), '\\n')
  cat('typeof:', typeof(x), ' length:', length(x), '\\n')
  str(x, max.level=2, give.attr=TRUE)
  if (is.data.frame(x)) {
    cat('dim:', paste(dim(x), collapse=' x '), '\\n')
    cat('names:', paste(names(x), collapse=' | '), '\\n')
    print(utils::head(x, 3))
  }
}
RS"
