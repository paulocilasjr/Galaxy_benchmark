/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
  cat('\\n###', f, '\\n')
  x <- readRDS(f); a <- attributes(x); ld <- a[['listData']]
  cat('listData typeof=', typeof(ld), 'length=', base::length(ld), 'names=', paste(base::names(ld), collapse=' | '), '\\n')
  for (nm in base::names(ld)) {
    z <- ld[[nm]]
    cat('column', nm, 'typeof=', typeof(z), 'class attr=', paste(base::attr(z,'class'), collapse=','), 'base_len=', base::length(z), '\\n')
    print(utils::head(z, 4))
  }
}
RS"
