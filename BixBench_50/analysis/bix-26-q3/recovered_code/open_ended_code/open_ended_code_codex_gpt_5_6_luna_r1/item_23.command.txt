/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_GluFevsGluFePlus.rds'); a <- attributes(x)
d <- data.frame(gene=a[['rownames']], a[['listData']], check.names=FALSE)
fg <- d"'$gene[!is.na(d$log2FoldChange)&!is.na(d$padj)&d$log2FoldChange > 1.5&d$padj < 0.05]
cat('"'foreground=', length(fg), '\\n')
ln <- tryCatch(readLines('https://rest.kegg.jp/link/pau/pathway:pau02010', warn=FALSE), error=function(e) {cat('ERR',conditionMessage(e),'\\n'); character()})
cat('KEGG lines=', length(ln), '\\n')
pathgenes <- sub('"'^path:pau02010'"\\\\tpau:', '', ln)
hits <- intersect(fg, pathgenes)
cat('ABC pathway genes=', length(pathgenes), '\\n')
cat('hits=', length(hits), '\\n')
print(hits)
RS"
