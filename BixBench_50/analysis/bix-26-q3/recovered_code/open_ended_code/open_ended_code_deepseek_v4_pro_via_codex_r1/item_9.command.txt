/bin/bash -lc "Rscript - <<'RS'
files <- c('/workspace/inputs/res_GluFevsGluFePlus.rds','/workspace/inputs/res_SuccvsGluFePlus.rds')
for (f in files) {
 x <- readRDS(f); a <- attributes(x); rn <- a"'$rownames; ld <- a$listData
 df <- data.frame(gene=rn, log2FC=ld$log2FoldChange, padj=ld$padj, stringsAsFactors=FALSE)
 cat('"'\\n', basename(f), 'n=', nrow(df), '\\n')
 cat('NA padj:', sum(is.na(df"'$padj)), '"' NA log2FC:', sum(is.na(df"'$log2FC)), '"'\\n')
 cat('padj<0.05:', sum(df"'$padj<0.05, na.rm=TRUE), '"'\\n')
 cat('up abs lfc>=1.5 padj<0.05:', sum(df"'$log2FC >= 1.5 & df$padj<0.05, na.rm=TRUE), '"'\\n')
 cat('down abs lfc>=1.5 padj<0.05:', sum(df"'$log2FC <= -1.5 & df$padj<0.05, na.rm=TRUE), '"'\\n')
 cat('all abs lfc>=1.5 padj<0.05:', sum(abs(df"'$log2FC)>=1.5 & df$padj<0.05, na.rm=TRUE), '"'\\n')
 cat('head up:\\n'); print(head(df[df"'$log2FC>=1.5 & df$padj<0.05,],10))
}
RS'
