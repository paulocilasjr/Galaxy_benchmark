/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_GluFevsGluFePlus.rds')
a <- attributes(x)
df <- data.frame(gene=a"'$rownames, a$listData, check.names=FALSE)
fg <- df$gene[!is.na(df$padj) & df$padj < 0.05 & !is.na(df$log2FoldChange) & df$log2FoldChange > 1.5]
links <- read.delim('"'https://rest.kegg.jp/link/pathway/pau', header=FALSE, stringsAsFactors=FALSE)
names(links) <- c('gene','pathway')
links"'$gene <- sub('"'"'^pau:'"', '', links"'$gene)
links$pathway <- sub('"'"'^path:'"', '', links"'$pathway)
paths <- read.delim('"'https://rest.kegg.jp/list/pathway/pau', header=FALSE, stringsAsFactors=FALSE)
names(paths) <- c('pathway','description')
for (mode in c('all_kegg','result_mapped')) {
  universe <- if (mode == 'all_kegg') unique(links"'$gene) else intersect(df$gene, unique(links$gene))
  fgm <- intersect(fg, universe)
  by_path <- split(links$gene, links$pathway)
  out <- do.call(rbind, lapply(names(by_path), function(pid) {
    genes <- intersect(unique(by_path[[pid]]), universe)
    k <- sum(fgm %in% genes)
    data.frame(pathway=pid, Count=k, pvalue=phyper(k-1, length(genes), length(universe)-length(genes), length(fgm), lower.tail=FALSE))
  }))
  out$padj <- p.adjust(out$pvalue, '"'BH')
  out <- merge(out, paths, by='pathway')
  abc <- out[grep('"'^ABC transporters'"', out"'$description),]
  cat(mode, '"'N=', length(universe), 'n=', length(fgm), 'Count=', abc"'$Count, '"'pvalue=', abc"'$pvalue, '"'padj=', abc"'$padj, '"'\\n')
}
RS"
