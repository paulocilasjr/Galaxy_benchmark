/bin/bash -lc "Rscript -e 'x<-readRDS(\"inputs/res_GluFevsGluFePlus.rds\"); ld<-attributes(x)"'$listData; ids<-attributes(x)$rownames; fg<-ids[ld$log2FoldChange>1.5 & ld$padj<0.05]; z<-readLines(pipe("curl -L --max-time 30 -sS https://rest.kegg.jp/link/pathway/pau")); tab<-do.call(rbind,strsplit(z,"'"\\\\t\")); genes<-sub(\""'^pau:","",tab[,1]); paths<-sub("''^path:","",tab[,2]); un<-intersect(ids,unique(genes)); fg2<-intersect(fg,un); u<-unique(paths); gs<-lapply(u,function(p)unique(genes[paths==p])); names(gs)<-u; gs<-lapply(gs,intersect,un); n<-length(un); k<-length(fg2); counts<-vapply(gs,function(g)sum(fg2%in%g),integer(1)); sizes<-vapply(gs,length,integer(1)); keep<-sizes>=10 & sizes<=500 & counts>0; pv<-rep(NA_real_,length(u)); pv[keep]<-vapply(which(keep),function(i)phyper(counts[i]-1,sizes[i],n-sizes[i],k,lower.tail=FALSE),numeric(1)); q<-p.adjust(pv,"BH"); abc<-match("pau02010",u); cat("mapped universe",n,"fg mapped",k,"pathways",length(u),"tested",sum(keep),"abc size",sizes[abc],"count",counts[abc],"p",pv[abc],"p.adjust all",q[abc],"p.adjust tested",p.adjust(pv[keep],"BH")[which(which(keep)==abc)],"'"\\\\n\"); ord<-order(pv); print(data.frame(id=u[ord][seq_len(min(10,sum("'!is.na(pv[ord]))))],count=counts[ord][seq_len(min(10,sum(!is.na(pv[ord])))],size=sizes[ord][seq_len(min(10,sum(!is.na(pv[ord])))],p=pv[ord][seq_len(min(10,sum(!is.na(pv[ord])))],q=q[ord][seq_len(min(10,sum(!is.na(pv[ord])))]) ))'"'"
