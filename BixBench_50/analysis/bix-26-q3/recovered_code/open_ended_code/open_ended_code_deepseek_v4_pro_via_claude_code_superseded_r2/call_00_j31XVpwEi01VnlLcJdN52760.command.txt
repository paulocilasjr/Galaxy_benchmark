Rscript -e '
x <- readRDS("inputs/res_GluFevsGluFePlus.rds")
ld <- attr(x, "listData")
rn <- attr(x, "rownames")

# Get elementMetadata as a simple list
em <- attr(x, "elementMetadata")
cat("ElementMetadata attributes:", names(attributes(em)), "\n")
em_ld <- attr(em, "listData")
if (!is.null(em_ld)) {
  cat("elementMetadata listData names:", names(em_ld), "\n")
  for (n in names(em_ld)) {
    cat("  ", n, ":", paste(head(em_ld[[n]], 5), collapse=", "), "\n")
  }
}

# Also check for any other useful attributes
cat("\nOther attributes:\n")
for (a in names(attributes(x))) {
  if (!a %in% c("listData", "elementMetadata", "rownames", "nrows", "class")) {
    cat("  ", a, ": class=", class(attr(x, a)), "\n")
  }
}
' 2>&1
