Rscript -e '
x <- readRDS("inputs/res_GluFevsGluFePlus.rds")
ld <- attr(x, "listData")
rn <- attr(x, "rownames")

# Create a dataframe and write to CSV
df <- data.frame(
  gene_id = rn,
  baseMean = ld$baseMean,
  log2FoldChange = ld$log2FoldChange,
  lfcSE = ld$lfcSE,
  stat = ld$stat,
  pvalue = ld$pvalue,
  padj = ld$padj,
  stringsAsFactors = FALSE
)
write.csv(df, "/workspace/res_GluFevsGluFePlus.csv", row.names = FALSE)
cat("Wrote", nrow(df), "rows to CSV\n")
cat("\nSummary stats:\n")
cat("Total genes:", nrow(df), "\n")
# Upregulated genes: log2FC > 1.5, padj < 0.05
up <- df[!is.na(df$padj) & df$padj < 0.05 & !is.na(df$log2FoldChange) & df$log2FoldChange > 1.5, ]
cat("Upregulated (log2FC>1.5, padj<0.05):", nrow(up), "\n")
cat("First 10 upregulated genes:\n")
print(head(up[, c("gene_id", "log2FoldChange", "padj")], 10))
' 2>&1
