Rscript -e '
x <- readRDS("inputs/res_GluFevsGluFePlus.rds")
# Access the listData without triggering S4 dispatch
ld <- attr(x, "listData")
cat("Names of listData:", names(ld), "\n")
cat("Length of each element:\n")
for (n in names(ld)) {
  cat("  ", n, ":", length(ld[[n]]), "\n")
}
# Get elementMetadata (column metadata)
em <- attr(x, "elementMetadata")
cat("\nElementMetadata:\n")
print(em)

# Show first few rows by accessing listData directly
cat("\nFirst 5 rows:\n")
nrow_val <- attr(x, "nrows")
for (i in 1:min(5, nrow_val)) {
  row_str <- paste(sapply(names(ld), function(n) ld[[n]][i]), collapse="\t")
  cat(row_str, "\n")
}
' 2>&1
