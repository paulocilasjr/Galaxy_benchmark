Rscript -e '
x <- readRDS("inputs/res_GluFevsGluFePlus.rds")
ld <- attr(x, "listData")
rn <- attr(x, "rownames")

# Filter for up: log2FC > 1.5, padj < 0.05
keep <- which(!is.na(ld$padj) & ld$padj < 0.05 & !is.na(ld$log2FoldChange) & ld$log2FoldChange > 1.5)
up_genes <- rn[keep]
cat("Upregulated genes:", length(up_genes), "\n")

# Read the ABC transporters pathway genes
abc_lines <- readLines("/workspace/pau_pathway_links.txt")
abc_genes_kegg <- unique(gsub("pau:", "", 
  sapply(strsplit(grep("pau02010", abc_lines, value=TRUE), "\t"), `[`, 2)))
cat("ABC transporters KEGG genes:", length(abc_genes_kegg), "\n")

# Overlap
overlap <- intersect(up_genes, abc_genes_kegg)
cat("Overlap count:", length(overlap), "\n")
cat("Overlapping genes:\n")
for (g in sort(overlap)) {
  idx <- which(rn == g)[1]
  cat(sprintf("  %s: log2FC=%.3f, padj=%.4e\n", g, ld$log2FoldChange[idx], ld$padj[idx]))
}
' 2>&1
