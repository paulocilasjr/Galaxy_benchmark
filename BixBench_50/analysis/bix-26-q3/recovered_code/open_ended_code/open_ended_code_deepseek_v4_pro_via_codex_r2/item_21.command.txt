import rdata, numpy as np, requests, warnings
warnings.filterwarnings('ignore')
obj=rdata.read_rds('inputs/res_GluFevsGluFePlus.rds')
rows=obj.rownames
lfc=np.asarray(obj.listData['log2FoldChange'])
padj=np.asarray(obj.listData['padj'])
up=set(rows[(lfc>1.5)&(padj<0.05)])
print('up',len(up))
r=requests.get('https://rest.kegg.jp/link/pau/path:pau02010',timeout=60)
pairs=[line.split('\\t') for line in r.text.strip().split('\\n')]
genes=set(x.split(':')[1] for a,x in pairs)
print('pathway genes',len(genes))
ov=up & genes
print('overlap',len(ov))
print(sorted(ov))
# print missing? no
# list up with lfc
idx=[i for i,g in enumerate(rows) if g in ov]
for i in sorted(idx, key=lambda i:-lfc[i]):
 print(rows[i],lfc[i],padj[i])
EOF"
