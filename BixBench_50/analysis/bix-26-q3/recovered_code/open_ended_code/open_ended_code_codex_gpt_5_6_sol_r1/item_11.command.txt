/bin/bash -lc "Rscript - <<'RS'
x<-readRDS('inputs/res_GluFevsGluFePlus.rds'); ld<-attr(x,'listData'); rn<-attr(x,'rownames')
up <- rn["'!is.na(ld$padj) & ld$padj < .05 & !is.na(ld$log2FoldChange) & ld$log2FoldChange > 1.5]
links <- read.delim('"'/tmp/pau_pathway_links.tsv',header=FALSE,stringsAsFactors=FALSE,col.names=c('gene','pathway'))
links"'$gene <- sub('"'"'^pau:'"','',links"'$gene); links$pathway <- sub('"'"'^path:'"','',links"'$pathway)
abc <- unique(links$gene[links$pathway=='"'pau02010'])
hits <- intersect(up,abc)
cat('foreground=',length(up),' abc_members=',length(abc),' overlap=',length(hits),'\\n',sep='')
cat(paste(hits,collapse='\\n'),'\\n')
# reconstruct ORA against all pathway-annotated PA14 genes; clusterProfiler default universe
universe<-unique(links"'$gene)
fg<-intersect(up,universe)
N<-length(universe); M<-length(intersect(abc,universe)); n<-length(fg); k<-length(intersect(fg,abc))
p<-phyper(k-1,M,N-M,n,lower.tail=FALSE)
# one-sided p for all pathways, BH among nonzero overlap terms, like tested terms
ps<-sapply(split(links$gene,links$pathway),function(g){mm<-length(unique(g)); kk<-length(intersect(fg,unique(g))); if(kk==0) NA_real_ else phyper(kk-1,mm,N-mm,n,lower.tail=FALSE)})
padj<-p.adjust(ps[!is.na(ps)],method='"'BH')
cat(sprintf('mapped foreground=%d universe=%d k=%d p=%.6g BH=%.6g tested_hit_terms=%d\\n',n,N,k,p,padj[['pau02010']],length(padj)))
RS"
