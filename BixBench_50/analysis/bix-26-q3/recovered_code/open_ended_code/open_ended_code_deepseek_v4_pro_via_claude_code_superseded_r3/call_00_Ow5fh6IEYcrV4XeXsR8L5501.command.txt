Rscript -e '
# Check what packages are available
ip <- installed.packages()
cat("System library packages:\n")
sys_lib <- ip[ip[, "LibPath"] == "/usr/local/lib/R/site-library", "Package"]
cat(paste(sys_lib, collapse=", "), "\n")
cat("\nR library packages:\n")
r_lib <- ip[ip[, "LibPath"] == "/usr/lib/R/library", "Package"]  
cat(paste(r_lib, collapse=", "), "\n")
' 2>&1 | head -20
