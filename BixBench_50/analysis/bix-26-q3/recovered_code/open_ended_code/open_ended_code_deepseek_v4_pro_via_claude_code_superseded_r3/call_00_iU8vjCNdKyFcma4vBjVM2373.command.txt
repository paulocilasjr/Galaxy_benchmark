source venv/bin/activate && python3 << 'EOF'
import pyreadr
import pandas as pd

# Read the RDS file
result = pyreadr.read_r('inputs/res_GluFevsGluFePlus.rds')
print("Keys:", list(result.keys()))
for key, val in result.items():
    print(f"\n--- {key} ---")
    print(f"Type: {type(val)}")
    print(f"Shape: {val.shape}")
    print(f"Columns: {list(val.columns)}")
    print(f"Index type: {type(val.index)}")
    print(f"First 5 rows:\n{val.head()}")
    print(f"\nSummary of log2FoldChange:\n{val['log2FoldChange'].describe()}")
    print(f"\nSummary of padj:\n{val['padj'].describe()}")
    
    # Count upregulated genes
    upregulated = val[(val['log2FoldChange'] > 1.5) & (val['padj'] < 0.05)]
    print(f"\nUpregulated (log2FC > 1.5 & padj < 0.05): {len(upregulated)}")
    
    # Show gene identifiers - check what the index/rownames look like
    print(f"\nFirst 20 gene IDs:\n{val.index[:20].tolist()}")
EOF
