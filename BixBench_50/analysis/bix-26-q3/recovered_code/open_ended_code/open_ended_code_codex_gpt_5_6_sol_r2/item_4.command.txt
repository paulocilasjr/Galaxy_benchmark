/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', full.names=TRUE)) {
  x <- readRDS(f)
  cat('\\nFILE', f, '\\n')
  cat('class:', paste(class(x), collapse=','), '\\n')
  cat('dim:', paste(dim(x), collapse=' x '), '\\n')
  cat('names:', paste(names(x), collapse=', '), '\\n')
  cat('rownames head:', paste(head(rownames(x)), collapse=', '), '\\n')
  print(head(as.data.frame(x), 3))
  cat('metadata names:', paste(names(attributes(x)), collapse=', '), '\\n')
  if ("'!is.null(metadata(x <- x)) && requireNamespace('"'S4Vectors', quietly=TRUE)) print(S4Vectors::metadata(x))
}
RS"
