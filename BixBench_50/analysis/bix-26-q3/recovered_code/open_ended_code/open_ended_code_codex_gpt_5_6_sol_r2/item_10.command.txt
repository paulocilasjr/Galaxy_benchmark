/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_GluFevsGluFePlus.rds')
d <- as.data.frame(attr(x, 'listData'))
rownames(d) <- attr(x, 'rownames')
fg <- rownames(d)["'!is.na(d$padj) & d$padj < 0.05 & d$log2FoldChange > 1.5]
abc.lines <- readLines('"'https://rest.kegg.jp/link/pau/pau02010', warn=FALSE)

hits <- intersect(fg, abc)
cat('foreground=', length(fg), '\\n', sep='')
cat('ABC membership=', length(abc), '\\n', sep='')
cat('hit_count=', length(hits), '\\n', sep='')
cat('hits=', paste(hits, collapse=','), '\\n', sep='')
cat('\\nHit DE values:\\n')
print(d[hits, c('log2FoldChange','padj')])
RS"
