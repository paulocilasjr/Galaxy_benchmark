/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_GluFevsGluFePlus.rds')
a <- attributes(x)
rn <- a"'$rownames
ld <- a$listData
cat('"'rownames length', length(rn), '\\n')
for (nm in names(ld)) cat(nm, length(ld[[nm]]), typeof(ld[[nm]]), '\\n')
print(head(rn,20))
print(head(ld"'$log2FoldChange,20))
print(head(ld$pvalue,20))
print(head(ld$padj,20))
cat('"'NA counts\\n')
for(nm in names(ld)) cat(nm, sum(is.na(ld[[nm]])), '\\n')
# thresholds
lfc <- ld"'$log2FoldChange
padj <- ld$padj
for(lfc_cut in c(1.5, -1.5)) {
 for(pcut in c(0.05)) {
  cat('"'padj<',pcut,' & lfc>',lfc_cut, sum("'!is.na(padj)&!is.na(lfc)&padj<pcut&lfc>lfc_cut), '"'\\n')
  cat('padj<',pcut,' & lfc>=',lfc_cut, sum("'!is.na(padj)&!is.na(lfc)&padj<pcut&lfc>=lfc_cut), '"'\\n')
  cat('padj<=',pcut,' & lfc>',lfc_cut, sum("'!is.na(padj)&!is.na(lfc)&padj<=pcut&lfc>lfc_cut), '"'\\n')
 }
}
cat('abs lfc >=1.5 & padj<.05', sum("'!is.na(padj)&!is.na(lfc)&padj<0.05&abs(lfc)>=1.5), '"'\\n')
cat('abs lfc >1.5 & padj<.05', sum("'!is.na(padj)&!is.na(lfc)&padj<0.05&abs(lfc)>1.5), '"'\\n')
cat('all genes', length(rn), '\\n')
RS"
