base='https://rest.kegg.jp'
def get(path):

        with urllib.request.urlopen(url, timeout=30) as r:
            data=r.read().decode('utf-8')
        return data
    except Exception as e:
        print('ERR',e); return None
# list pathways with ABC for pau/pae maybe
for org in ['pau','pae']:
    data=get('/list/pathway/'+org)
    if data:
        lines=data.splitlines()
        print(org, len(lines))
        for l in lines:
            if 'ABC transporters' in l: print(l)
# get pathway genes
for org in ['pau','pae']:
