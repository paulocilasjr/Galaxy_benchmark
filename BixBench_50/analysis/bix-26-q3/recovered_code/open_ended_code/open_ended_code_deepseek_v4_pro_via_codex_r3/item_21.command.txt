python3 - <<'PY'
import urllib.request,re
up=set(open('tmp/upgenes.txt').read().split())
allg=set(open('tmp/allgenes.txt').read().split())
data=urllib.request.urlopen('https://rest.kegg.jp/get/pau02010',timeout=30).read().decode()
# find GENE block
m=re.search(r'"'^GENE'"\\s+(.*?)(?="'^[A-Z_]+'"\\s+|\\Z)', data, re.S|re.M)
block=m.group(1)
genes=set()
for line in block.splitlines():
    line=line.strip()
    if not line: continue
    # first token before space
    g=line.split()[0]
    genes.add(g)
print('pathway genes',len(genes))
print('overlap up',len(up&genes))
print('overlap all',len(allg&genes))
print('up genes in path', sorted(up&genes))
from math import comb
N=len(allg); K=len(allg&genes); n=len(up); k=len(up&genes)
print('N K n k', N,K,n,k)
# hypergeom upper tail P(X>=k)
p=sum(comb(K,i)*comb(N-K,n-i) for i in range(k,min(K,n)+1))/comb(N,n)
print('p',p)
PY"
