/bin/bash -lc "python3 - <<'PY'
# compute overlaps for variants using R output maybe redo with all abs
import urllib.request
from collections import defaultdict
from math import comb
allg=open('tmp/allgenes.txt').read().split()
# parse values with R quickly
import subprocess, json, os
# use R to write sets
subprocess.run(['Rscript','-'],input='''x<-readRDS(\"inputs/res_GluFevsGluFePlus.rds\");a<-attributes(x);ld<-a"'$listData; rn<-a$rownames; lfc<-ld$log2FoldChange; padj<-ld$padj; writeLines(rn[padj<0.05 & abs(lfc)>=1.5],"tmp/absgenes.txt"); writeLines(rn[padj<0.05 & lfc<= -1.5],"tmp/downgenes.txt")'"''',text=True)
absgenes=set(open('tmp/absgenes.txt').read().split()); down=set(open('tmp/downgenes.txt').read().split())
link=urllib.request.urlopen('https://rest.kegg.jp/link/pathway/pau',timeout=30).read().decode()
pairs=[]
for l in link.splitlines():
 p=l.split('\\t')
 if len(p)==2: pairs.append((p[0].split(':')[-1],p[1].split(':')[-1]))
plist=urllib.request.urlopen('https://rest.kegg.jp/list/pathway/pau',timeout=30).read().decode()
ids=set(l.split('\\t')[0] for l in plist.splitlines() if '\\t' in l)
pairs=[x for x in pairs if x[1] in ids]
p2g=defaultdict(set)
for g,p in pairs:p2g[p].add(g)
ext={g for g,p in pairs}
abc=p2g['pau02010']
print('abs count',len(absgenes),'abs in ext',len(absgenes&ext),'abc overlap',len(absgenes&abc), sorted(absgenes&abc))
print('down count',len(down),'down in ext',len(down&ext),'down abc overlap',len(down&abc), sorted(down&abc))
PY"
