/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('inputs/res_GluFevsGluFePlus.rds')
d <- as.data.frame(attr(x,'listData'))
d"'$gene <- attr(x,'"'rownames')
up <- subset(d, "'!is.na(padj) & padj < 0.05 & log2FoldChange >= 1.5)
down <- subset(d, !is.na(padj) & padj < 0.05 & log2FoldChange <= -1.5)
cat('"'all',nrow(d),'up',nrow(up),'down',nrow(down),'padj_NA',sum(is.na(d"'$padj)),'"'\\n')
abc <- read.delim('https://rest.kegg.jp/link/pau/pathway:pau02010',header=FALSE,stringsAsFactors=FALSE)
abc_genes <- sub('"'^pau:'"','',abc[[2]])
hits <- intersect(up"'$gene,abc_genes)
cat('"'ABC annotated',length(unique(abc_genes)),'up hits',length(hits),'\\n')
print(data.frame(gene=hits,log2FC=up"'$log2FoldChange[match(hits,up$gene)],padj=up$padj[match(hits,up$gene)]))
# Full ORA, analogous to enrichKEGG default gene universe of all genes in KEGG pathway-gene links
links <- read.delim('"'https://rest.kegg.jp/link/pathway/pau',header=FALSE,stringsAsFactors=FALSE)
colnames(links)<-c('gene','path')
links"'$gene <- sub('"'"'^pau:'"','',links"'$gene); links$path <- sub('"'"'^path:'"','',links"'$path)
sets <- split(links$gene,links$path); sets <- lapply(sets,unique)
univ <- unique(links$gene); q <- intersect(up$gene,univ)
N<-length(univ); n<-length(q)
res <- do.call(rbind,lapply(names(sets),function(p){ s<-intersect(sets[[p]],univ); M<-length(s); k<-length(intersect(q,s)); data.frame(path=p,M=M,k=k,p=phyper(k-1,M,N-M,n,lower.tail=FALSE)) }))
res$padj<-p.adjust(res$p,'"'BH')
print(res[res"'$path=='"'pau02010',])
cat('universe',N,'query annotated',n,'terms',nrow(res),'\\n')
RS"
