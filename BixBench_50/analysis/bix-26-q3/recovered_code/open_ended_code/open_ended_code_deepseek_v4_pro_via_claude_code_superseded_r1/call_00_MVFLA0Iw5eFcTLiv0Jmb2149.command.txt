import requests

# Get the first few lines of the link to understand the format
url = "http://rest.kegg.jp/link/pau/pathway"
response = requests.get(url, timeout=60)
lines = response.text.strip().split('\n')
print("First 20 lines:")
for line in lines[:20]:
    print(line)
print(f"\nTotal lines: {len(lines)}")

# Check if all genes start with pau:
pau_genes = set()
pathways = set()
for line in lines:
    parts = line.split('\t')
    if len(parts) == 2:
        gene = parts[0].strip()
        path = parts[1].strip()
        pau_genes.add(gene)
        pathways.add(path)

print(f"Unique genes: {len(pau_genes)}")
print(f"Unique pathways: {len(pathways)}")
print(f"Sample genes: {list(pau_genes)[:10]}")
print(f"Sample pathways: {list(pathways)[:10]}")

# Check how many of our upregulated genes have KEGG mappings
# PA14 genes from our data
df_genes = set()
with open('/workspace/res_GluFevsGluFePlus.csv') as f:
    import csv
    reader = csv.DictReader(f)
    for row in reader:
        df_genes.add(row['gene'])

# Map: our gene IDs should match pau: prefix
kegg_genes_clean = set(g.replace('pau:', '') for g in pau_genes)
print(f"\nOur genes: {len(df_genes)}")
print(f"KEGG genes (cleaned): {len(kegg_genes_clean)}")
print(f"Intersection: {len(df_genes & kegg_genes_clean)}")

# Check pathway counts
path_counts = {}
for line in lines:
    parts = line.split('\t')
    if len(parts) == 2:
        path = parts[1].strip()
        path_counts[path] = path_counts.get(path, 0) + 1

for p, c in sorted(path_counts.items(), key=lambda x: -x[1])[:15]:
    print(f"  {p}: {c} genes")
