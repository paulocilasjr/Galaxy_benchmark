# Final verification
import pandas as pd
import requests

df = pd.read_csv("/workspace/res_GluFevsGluFePlus.csv")

# Verify upregulated count
up_mask = (df['log2FoldChange'] > 1.5) & (df['padj'] < 0.05)
up_genes = set(df.loc[up_mask, 'gene'].tolist())
print(f"Total upregulated: {len(up_genes)}")

# Verify the 11 genes are indeed upregulated
abc_genes_list = ['PA14_10160', 'PA14_10180', 'PA14_20030', 'PA14_33540', 'PA14_33550', 
                  'PA14_33560', 'PA14_33690', 'PA14_33760', 'PA14_48115', 'PA14_54980', 'PA14_62000']

for g in abc_genes_list:
    row = df[df['gene'] == g]
    if len(row) > 0:
        r = row.iloc[0]
        print(f"{g}: log2FC={r['log2FoldChange']:.4f}, padj={r['padj']:.4e}, baseMean={r['baseMean']:.1f}")
    else:
        print(f"{g}: NOT FOUND in dataset")

# Verify all 11 are in KEGG ABC transporters
url = "http://rest.kegg.jp/get/pau02010"
resp = requests.get(url, timeout=60)
kegg_genes_in_pathway = set()
for line in resp.text.split('\n'):
    if line.startswith('GENE'):
        # Format: GENE        pau:PA14_00360    gene_name
        parts = line.split()
        if len(parts) >= 2:
            gene_part = parts[1]
            if gene_part.startswith('pau:PA14_'):
                kegg_genes_in_pathway.add(gene_part.replace('pau:', ''))

print(f"\nKEGG ABC transporters genes count: {len(kegg_genes_in_pathway)}")
print(f"All 11 confirmed in pathway: {all(g in kegg_genes_in_pathway for g in abc_genes_list)}")

# Final answer
print(f"\nAnswer: {len(abc_genes_list)}")
