/bin/bash -lc "python - <<'PY'
import csv, collections
fn='inputs/Model.csv'
with open(fn,newline='') as f:
 r=csv.DictReader(f)
 vals={k:collections.Counter() for k in ['ModelType','DepmapModelType','OncotreeLineage','SourceType']}
 n=0
 for row in r:
  n+=1
  for k in vals: vals[k][row[k]]+=1
print('rows',n)
for k,c in vals.items(): print(k, c.most_common(20))
PY"
