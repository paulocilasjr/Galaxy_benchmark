import csv, gc, os, time
import numpy as np
import pandas as pd
from scipy.stats import rankdata, t, spearmanr

start_time=time.time()

def get_header(fn):
    with open(fn, newline='') as f:
        return next(csv.reader(f))

def get_ids(fn):
    with open(fn, newline='') as f:
        rd=csv.reader(f); next(rd)
        return [row[0] for row in rd]

cf='inputs/CRISPRGeneEffect.csv'
ef='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
hc=get_header(cf); he=get_header(ef)
expr_pos={g:i for i,g in enumerate(he) if i>0 and g}
genes=[g for i,g in enumerate(hc) if i>0 and g in expr_pos]
cpos=[i for i,g in enumerate(hc) if i>0 and g in expr_pos]
epos=[expr_pos[g] for g in genes]
print('shared genes',len(genes),flush=True)

# Read only shared genes from each table, retaining the original sample IDs.
cdtype={hc[i]:np.float64 for i in cpos}
edtype={he[i]:np.float64 for i in epos}
ct0=time.time()
cdf=pd.read_csv(cf, index_col=0, usecols=[0]+cpos, dtype=cdtype)

et0=time.time()

ids=[x for x in cdf.index if x in edf.index]
print('shared models',len(ids),'unique',len(set(ids)),flush=True)
# Explicitly align both matrices by ModelID and by identical shared-gene order.

counts=np.zeros(m,dtype=np.int32)
ps=np.full(m,np.nan,dtype=np.float64)

# With complete data, rank each gene column and compute Pearson correlation of ranks.
finite=np.isfinite(C)&np.isfinite(E)
counts[:]=finite.sum(axis=0)
if finite.all():
    block=1024
    for a in range(0,m,block):
        b=min(a+block,m)
        cr=rankdata(C[:,a:b],axis=0,method='average')
        er=rankdata(E[:,a:b],axis=0,method='average')
        cr-=cr.mean(axis=0); er-=er.mean(axis=0)
        denom=np.sqrt(np.sum(cr*cr,axis=0)*np.sum(er*er,axis=0))
        rho[a:b]=np.divide(np.sum(cr*er,axis=0),denom,out=np.full(b-a,np.nan),where=denom"'!=0)

    rr=np.clip(rho[valid],-1,1)
    with np.errstate(divide='"'ignore',invalid='ignore'):
        ts=rr*np.sqrt(dof/((1-rr)*(1+rr)))

    print('using pairwise-complete fallback',flush=True)
    for j in range(m):
