def read_ids(fn):

    rd=csv.reader(f); h=next(rd); ids=[r[0] for r in rd]

print('shared common model',len(sc&se&sm),'shared not model',len(sc&se-sm),'crispr/model',len(sc&sm),'expr/model',len(se&sm))
print('CRISPR not expr sample IDs:', sorted(sc-se)[:50])
print('Expression not CRISPR sample IDs:', sorted(se-sc)[:50])
print('genes CRISPR',len(hc)-1,'expr',len(he)-1,'intersection',len((set(hc)-{''})&(set(he)-{''})))

PY"
