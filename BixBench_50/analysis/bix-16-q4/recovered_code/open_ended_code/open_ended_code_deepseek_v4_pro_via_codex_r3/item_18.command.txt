/bin/bash -lc "/workspace/venv/bin/python - <<'PY'
import numpy as np, subprocess, tempfile, os
rng=np.random.default_rng(1); p=rng.random(5)
print('p',p)
def bh(p):
 p=np.asarray(p,float); m=len(p); order=np.argsort(p,kind='mergesort'); q=np.empty(m); q[order]=p[order]*m/np.arange(1,m+1); cm=1
 for i in range(m-1,-1,-1): cm=min(cm,q[order][i]); q[order][i]=cm
 return np.minimum(q,1)
print('python',bh(p))
with tempfile.NamedTemporaryFile('w',delete=False,dir='/tmp') as f:
 np.savetxt(f,p); fn=f.name
rscript=f'p=scan(\"{fn}\"); print(p); print(p.adjust(p,\"BH\"))'
res=subprocess.run(['Rscript','-e',rscript],capture_output=True,text=True)
print(res.stdout)
os.unlink(fn)
PY"
