/bin/bash -lc "/workspace/venv/bin/python - <<'PY'
import numpy as np, subprocess, tempfile, os
rng=np.random.default_rng(0)
p=rng.random(10000)
def bh(p):
    p=np.asarray(p,float); p[np.isnan(p)]=1
    m=len(p); order=np.argsort(p,kind='mergesort'); sp=p[order]
    q=np.empty(m); q[order]=sp*m/np.arange(1,m+1)
    cm=1.0
    for i in range(m-1,-1,-1):
        cm=min(cm,q[order][i]); q[order][i]=cm
    return np.minimum(q,1)
q=bh(p)
# write p to stdin R via temp
with tempfile.NamedTemporaryFile('w',delete=False,dir='/tmp') as f:
    np.savetxt(f,p)
    fn=f.name
rscript=f'''p=scan(\"{fn}\"); q=p.adjust(p,\"BH\"); cat(sum(q<.05), \"\\\\n\"); cat(sprintf(\"%.20f\", sum(q<.05)/length(q)), \"\\\\n\"); qq=quantile(q,c(.25,.5,.75,.9,.95,.99)); print(qq)'''
res=subprocess.run(['Rscript','-e',rscript],capture_output=True,text=True)
print(res.stdout)
print('python sig',(q<.05).sum(),'percent',(q<.05).mean())
print('maxdiff',np.max(np.abs(q-p.adjust?)))
# can't p.adjust python; use R output not values. just compare p raw stats
print('p order equal', np.allclose(np.sort(p), np.sort(p)))
os.unlink(fn)
PY"
