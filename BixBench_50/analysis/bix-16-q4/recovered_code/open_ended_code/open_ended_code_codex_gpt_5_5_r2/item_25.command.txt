/bin/bash -lc ".venv/bin/python - <<'PY'
import csv

def ids_from_matrix(path):
    with open(path, newline='') as f:
        r=csv.reader(f); next(r)
        return [row[0] for row in r]
with open('inputs/Model.csv', newline='') as f:
    r=csv.DictReader(f)
    model_ids={row['ModelID'] for row in r}
crispr=set(ids_from_matrix('inputs/CRISPRGeneEffect.csv'))
expr=set(ids_from_matrix('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'))
shared=crispr & expr
print('shared', len(shared))
print('shared_not_in_Model', len(shared-model_ids))
print('crispr_not_in_Model', len(crispr-model_ids))
print('expr_not_in_Model', len(expr-model_ids))
PY"
