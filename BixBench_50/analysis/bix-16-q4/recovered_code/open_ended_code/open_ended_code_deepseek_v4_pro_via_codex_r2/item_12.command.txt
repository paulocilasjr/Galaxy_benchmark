import pandas as pd, numpy as np, scipy
print('pandas',pd.__version__,'numpy',np.__version__,'scipy',scipy.__version__)
crispr_cols=pd.read_csv('inputs/CRISPRGeneEffect.csv', nrows=0).columns.tolist()

print('common gene columns exact',len(common))
# row ids from both quickly with low_memory
crids=pd.read_csv('inputs/CRISPRGeneEffect.csv', usecols=[0]).iloc[:,0].tolist()
exids=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', usecols=[0]).iloc[:,0].tolist()
common_ids=sorted(set(crids)&set(exids))
print('crispr rows',len(crids),'expr rows',len(exids),'common rows',len(common_ids))

PY"
