/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, time, gc
T=time.time()
ch=pd.read_csv('inputs/CRISPRGeneEffect.csv', nrows=0)
eh=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', nrows=0)
first_c=ch.columns[0]; first_e=eh.columns[0]
common=[c for c in ch.columns[1:] if c in eh.columns[1:]]
print('common',len(common),'load start',time.time()-T)
dt={c:np.float32 for c in common}
cd=pd.read_csv('inputs/CRISPRGeneEffect.csv', usecols=[first_c]+common, index_col=0, dtype=dt, low_memory=False)
print('crispr loaded',cd.shape,'mem',cd.memory_usage(deep=True).sum()/1e6,'time',time.time()-T)
ed=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', usecols=[first_e]+common, index_col=0, dtype=dt, low_memory=False)
print('expr loaded',ed.shape,'mem',ed.memory_usage(deep=True).sum()/1e6,'time',time.time()-T)
common_ids=sorted(set(cd.index)&set(ed.index))
cd=cd.loc[common_ids]; ed=ed.loc[common_ids]
print('aligned',cd.shape,ed.shape,'time',time.time()-T)
print('crispr nan total',int(cd.isna().sum().sum()),'expr nan total',int(ed.isna().sum().sum()))
print('crispr nan per col max',int(cd.isna().sum().max()),'expr max',int(ed.isna().sum().max()))
print('crispr zero variance cols',int((cd.nunique(dropna=True)<=1).sum()),'expr zero variance',int((ed.nunique(dropna=True)<=1).sum()))
valid=cd.notna().to_numpy() & ed.notna().to_numpy()
vc=valid.sum(axis=0)
print('pairwise valid n min/median/max',vc.min(),np.median(vc),vc.max(),'cols n<3',int((vc<3).sum()),'cols n<5',int((vc<5).sum()),'cols n==common rows',int((vc==len(common_ids)).sum()))
print('cd col missing counts top',cd.isna().sum().value_counts().head(10).to_dict())
print('ed col missing counts top',ed.isna().sum().value_counts().head(10).to_dict())
PY"
