/bin/bash -lc "Rscript - <<'RS'
options(stringsAsFactors=FALSE)
cat('reading crispr\\n')
crispr <- read.csv('inputs/CRISPRGeneEffect.csv', row.names=1, check.names=FALSE)
cat('reading expression\\n')
expr <- read.csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', row.names=1, check.names=FALSE)
models <- intersect(rownames(crispr), rownames(expr))
genes <- intersect(colnames(crispr), colnames(expr))
cat('models', length(models), 'genes', length(genes), '\\n')
C <- data.matrix(crispr[models, genes, drop=FALSE])
rm(crispr); gc()
E <- data.matrix(expr[models, genes, drop=FALSE])
rm(expr); gc()
cat('subset matrices', dim(E), dim(C), 'NA', sum(is.na(E)), sum(is.na(C)), '\\n')
m <- length(genes)
rho <- rep(NA_real_, m)
p <- rep(NA_real_, m)
nobs <- integer(m)
cat('testing\\n')
for (i in seq_len(m)) {
  ok <- "'!is.na(E[, i]) & !is.na(C[, i])
  n <- sum(ok)
  nobs[i] <- n
  if (n >= 3L) {
    ct <- suppressWarnings(cor.test(E[ok, i], -C[ok, i], method='"'spearman', exact=FALSE))
    rho[i] <- unname(ct"'$estimate)
    p[i] <- ct$p.value
  }
  if (i %% 2000L == 0L) cat('"'done', i, '\\n')
}
tested <- "'!is.na(p)
q <- rep(NA_real_, m)
q[tested] <- p.adjust(p[tested], method='"'BH')
sig <- sum(q[tested] < 0.05)
total <- sum(tested)
pct <- 100 * sig / total
cat('nobs_range', range(nobs[tested]), 'tested', total, '\\n')
cat('sig', sig, 'total', total, 'pct', format(pct, digits=17), '\\n')
cat('rho_range', range(rho[tested]), 'p_range', range(p[tested]), 'q_range', range(q[tested]), '\\n')
RS"
