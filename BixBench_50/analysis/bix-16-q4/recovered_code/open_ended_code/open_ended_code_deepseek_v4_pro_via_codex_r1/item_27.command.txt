/bin/bash -lc "pip install -q statsmodels && python - <<'PY'
import numpy as np
from statsmodels.stats.multitest import multipletests
D=np.load('/tmp/spearman_float64.npz')
p=D['p']; q_manual=D['q']; n=len(p)
_,q,bh_reject,_=multipletests(p,alpha=0.05,method='fdr_bh',is_sorted=False)
print('statsmodels reject',bh_reject.sum(),'pct',bh_reject.sum()/n*100)
print('manual reject',np.sum(q_manual<0.05),'max q diff',np.nanmax(np.abs(q-q_manual)))
PY"
