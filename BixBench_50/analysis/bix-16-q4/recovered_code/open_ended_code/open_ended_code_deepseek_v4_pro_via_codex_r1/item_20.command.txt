/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
from scipy import stats
E=pd.read_pickle('/tmp/effect.pkl')
X=pd.read_pickle('/tmp/expr.pkl')
common_samples=E.index.intersection(X.index)
common_genes=E.columns.intersection(X.columns)
e=E.loc[common_samples, common_genes].to_numpy(dtype=np.float64, copy=True)
x=X.loc[common_samples, common_genes].to_numpy(dtype=np.float64, copy=True)
n_genes=e.shape[1]
rho=np.empty(n_genes); p=np.empty(n_genes); valid=np.empty(n_genes,dtype=np.int64)
for j in range(n_genes):
    r,pj=stats.spearmanr(e[:,j], x[:,j], nan_policy='omit')
    rho[j]=r; p[j]=pj; valid[j]=np.sum(~np.isnan(e[:,j]))
pvals=p.copy(); order=np.argsort(pvals); n=len(pvals)
q=np.empty(n); running=np.inf
for idx in range(n-1,-1,-1):
    running=min(running, pvals[order[idx]]*n/(idx+1))
    q[idx]=running
q=np.minimum(q,1.0)
qfull=np.empty(n_genes); qfull[:]=np.nan; qfull[order]=q
sig=qfull<0.05
print('n',n,'valid min/max',valid.min(),valid.max())
print('sig',sig.sum(),'pct',sig.sum()/n*100)
print('positive',np.sum(sig & (rho>0)),'negative',np.sum(sig & (rho<0)))
print('zero',np.sum(sig & (rho==0)))
print('rho minmax',rho.min(),rho.max(),'p minmax',p.min(),p.max())
# use statsmodels if available; else install? 
try:
    import statsmodels.stats.multitest as smt
    q2=smt.multipletests(p, method='fdr_bh')[1]
    print('statsmodels sig',np.sum(q2<0.05), 'q max sig', q2[q2<0.05].max(), 'pct',np.sum(q2<0.05)/n*100)
    print('allclose q', np.allclose(qfull,q2,equal_nan=True,rtol=1e-12,atol=1e-15))
except Exception as e:
    print('statsmodels not available',e)
np.savez('/tmp/spearman_results.npz', genes=common_genes.to_numpy(), rho=rho, p=p, q=qfull, sig=sig, valid=valid)
print('saved')
PY"
