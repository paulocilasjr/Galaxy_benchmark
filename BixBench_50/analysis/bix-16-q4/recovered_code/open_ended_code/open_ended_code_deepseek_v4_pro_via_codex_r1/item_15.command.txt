/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
for f in ['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']:
    t=time.time()
    df=pd.read_csv(f, index_col=0, dtype=np.float32)
    print(f,'shape',df.shape,'read sec',round(time.time()-t,1),'mem MB',df.memory_usage(deep=True).sum()/1e6)
    print('nan count',int(df.isna().sum().sum()),'pct',df.isna().sum().sum()/df.size)
    # missing per gene/column distribution
    miss=df.isna().sum()
    print('cols any nan',int((miss>0).sum()),'max nan',int(miss.max()),'mean nan',float(miss.mean()))
    # ties per first few cols exact value counts? 
    if f.startswith('inputs/CRISPR'):
        for c in df.columns[:3]:
            v=df[c].dropna().values
            u,cnt=np.unique(v, return_counts=True)
            print(c,'n',len(v),'unique',len(u),'max count',cnt.max(),'zeros',(v==0).sum())
        df=df.copy()
        df.to_pickle('/tmp/effect.pkl')
        print('saved /tmp/effect.pkl')
    else:
        for c in df.columns[:3]:
            v=df[c].values
            u,cnt=np.unique(v, return_counts=True)
            print(c,'n',len(v),'unique',len(u),'max count',cnt.max())
        df.to_pickle('/tmp/expr.pkl')
        print('saved /tmp/expr.pkl')
PY"
