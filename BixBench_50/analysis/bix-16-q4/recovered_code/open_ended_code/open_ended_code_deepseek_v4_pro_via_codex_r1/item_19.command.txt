/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
from scipy import stats

E=pd.read_pickle('/tmp/effect.pkl')
X=pd.read_pickle('/tmp/expr.pkl')
common_samples=E.index.intersection(X.index)
common_genes=E.columns.intersection(X.columns)
print('samples',len(common_samples),'genes',len(common_genes), flush=True)
e=E.loc[common_samples, common_genes].to_numpy(dtype=np.float64, copy=True)
x=X.loc[common_samples, common_genes].to_numpy(dtype=np.float64, copy=True)
n_genes=e.shape[1]
rho=np.empty(n_genes, dtype=np.float64)
p=np.empty(n_genes, dtype=np.float64)
valid=np.empty(n_genes, dtype=np.int64)
t=time.time()
for j in range(n_genes):
    r,pj=stats.spearmanr(e[:,j], x[:,j], nan_policy='omit')
    rho[j]=r
    p[j]=pj
    valid[j]=np.sum(~np.isnan(e[:,j]))
print('corr loop sec',time.time()-t, flush=True)
# BH procedure (Benjamini-Hochberg 1995), handling nans conservatively? include finite p only.
p_finite=np.isfinite(p)
print('finite p',p_finite.sum(),'nan p',np.sum(~p_finite),'valid min/max',valid.min(),valid.max(), flush=True)
# Standard BH on all finite p-values, rank ascending
pvals=p[p_finite]
order=np.argsort(pvals)
n=len(pvals)
# q = min_{j>=i} p_(j)*n/j
q=np.empty(n, dtype=np.float64)
running=np.inf
for idx in range(n-1,-1,-1):
    pi=pvals[order[idx]]
    # ranks are 1-based: j=idx+1
    qcand=pi*n/(idx+1)
    if qcand < running:
        running=qcand
    q[idx]=running
# cap q at 1 (numeric rounding)
q=np.minimum(q,1.0)
qfull=np.empty(n_genes, dtype=np.float64); qfull[:]=np.nan
qfull[p_finite]=q[np.argsort(order)]
sig=qfull<0.05
print('significant q<0.05',int(sig.sum()),'percentage',sig.sum()/n_genes*100, flush=True)
print('sig positive',int((sig)&(rho>0)).sum(), 'sig negative',int((sig)&(rho<0)).sum(), flush=True)
print('rho range',np.nanmin(rho),np.nanmax(rho),'p range finite',pvals.min(),pvals.max(), flush=True)
print('top significant p <=',np.sort(pvals)[:10], flush=True)
print('genes sig count from p_bonf',int(np.sum(p[p_finite] < 0.05/n)), flush=True)
np.savez('/tmp/spearman_results.npz', genes=common_genes.to_numpy(), rho=rho, p=p, q=qfull, sig=sig, valid=valid)
print('saved /tmp/spearman_results.npz')
# report first 20 significant
inds=np.where(sig)[0]
print('first sig genes')
for i in inds[:20]:
    print(common_genes[i], 'rho',rho[i], 'p',p[i], 'q',qfull[i], 'valid',valid[i])
PY"
