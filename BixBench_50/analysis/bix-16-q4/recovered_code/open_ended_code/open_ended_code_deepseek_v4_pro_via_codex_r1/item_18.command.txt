/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
from scipy import stats
E=pd.read_pickle('/tmp/effect.pkl')
X=pd.read_pickle('/tmp/expr.pkl')
common_samples=E.index.intersection(X.index)
common_genes=E.columns.intersection(X.columns)
print('samples',len(common_samples),'genes',len(common_genes))
e=E.loc[common_samples, common_genes].to_numpy()
x=X.loc[common_samples, common_genes].to_numpy()
print(e.shape,x.shape)
# benchmark 200 genes first
idx=0; t=time.time()
for j in range(200):
    stats.spearmanr(e[:,j], x[:,j], nan_policy='omit')
print('200 spearmanr sec',time.time()-t)
# benchmark some with ties and missing
for j in [0,1,2,1000]:
    r,p=stats.spearmanr(e[:,j], x[:,j], nan_policy='omit')
    print(j,r,p)
PY"
