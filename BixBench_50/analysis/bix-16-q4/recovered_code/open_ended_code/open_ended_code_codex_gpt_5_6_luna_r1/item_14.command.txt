/bin/bash -lc "python3 - <<'PY'
import csv, time
import numpy as np
import pandas as pd

def header(fn):
    with open(fn,newline='') as f:
        return next(csv.reader(f))

t0=time.time()
cr_h=header('inputs/CRISPRGeneEffect.csv')
ex_h=header('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
ex_set=set(ex_h[1:])
common=[g for g in cr_h[1:] if g in ex_set]
cr_use=[0]+[i for i,g in enumerate(cr_h) if i and g in ex_set]
ex_pos={g:i for i,g in enumerate(ex_h)}
ex_use=[0]+[ex_pos[g] for g in common]
dtype={g:np.float32 for g in common}
print('common genes',len(common), 'read setup sec',round(time.time()-t0,2),flush=True)
cr=pd.read_csv('inputs/CRISPRGeneEffect.csv',usecols=cr_use,index_col=0,dtype=dtype)
print('cr loaded',cr.shape,'sec',round(time.time()-t0,2),'nbytes',cr.to_numpy(copy=False).nbytes,flush=True)
ex=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=ex_use,index_col=0,dtype=dtype)
print('ex loaded',ex.shape,'sec',round(time.time()-t0,2),'nbytes',ex.to_numpy(copy=False).nbytes,flush=True)
ids=cr.index.intersection(ex.index)
cr=cr.loc[ids,common]
ex=ex.loc[ids,common]
print('shared models',len(ids),'aligned',cr.shape,ex.shape,flush=True)
print('cr nan',int(cr.isna().sum().sum()),'ex nan',int(ex.isna().sum().sum()),flush=True)
print('constant columns cr',int((cr.nunique(dropna=True)<=1).sum()),'ex',int((ex.nunique(dropna=True)<=1).sum()),flush=True)
print('value ranges cr',float(cr.min().min()),float(cr.max().max()),'ex',float(ex.min().min()),float(ex.max().max()),flush=True)
print('done sec',round(time.time()-t0,2),flush=True)
PY"
