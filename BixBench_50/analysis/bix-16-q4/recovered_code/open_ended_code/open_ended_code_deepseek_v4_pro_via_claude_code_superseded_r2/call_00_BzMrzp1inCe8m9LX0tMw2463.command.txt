# Verify BH correction against statsmodels
import numpy as np
from statsmodels.stats.multitest import multipletests
import csv

# Load p-values computed by the main script
# We'll recompute a small verification

# Quick verification: generate some p-values and compare my BH impl with statsmodels
np.random.seed(42)
test_p = np.concatenate([
    np.random.uniform(0, 0.01, 50),  # 50 strong signals
    np.random.uniform(0, 1, 950)      # 950 noise
])
test_p = np.clip(test_p, 1e-300, 1.0)

# My BH implementation
n = len(test_p)
sorted_idx = np.argsort(test_p)
sorted_p = test_p[sorted_idx]
ranks = np.arange(1, n+1, dtype=float)
bh_crit = 0.05 * ranks / n
below = sorted_p <= bh_crit
if np.any(below):
    k_max = np.max(np.where(below)[0])
    my_nsig = k_max + 1
else:
    my_nsig = 0

# Statsmodels BH
reject, p_adj, _, _ = multipletests(test_p, alpha=0.05, method='fdr_bh')
sm_nsig = np.sum(reject)

print(f"My BH: {my_nsig} significant")
print(f"Statsmodels BH: {sm_nsig} significant")
print(f"Match: {my_nsig == sm_nsig}")
