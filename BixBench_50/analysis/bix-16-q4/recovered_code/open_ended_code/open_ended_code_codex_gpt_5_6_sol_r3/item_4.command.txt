/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
from pathlib import Path
files=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
info=[]
for fn in files:
    with open(fn,newline='') as f:
        rd=csv.reader(f); h=next(rd); ids=[r[0] for r in rd]
    c=Counter(ids); hc=Counter(h[1:])
    print(fn)
    print('rows',len(ids),'unique IDs',len(c),'duplicate IDs',sum(v>1 for v in c.values()),'blank IDs',c.get('',0))
    print('genes',len(h)-1,'unique genes',len(hc),'duplicate genes',sum(v>1 for v in hc.values()))
    info.append((h[1:],set(ids)))
g0,g1=info[0][0],info[1][0]
print('exact shared genes',len(set(g0)&set(g1)))
print('effect only',len(set(g0)-set(g1)),'expr only',len(set(g1)-set(g0)))
print('shared model IDs',len(info[0][1]&info[1][1]),'effect only',len(info[0][1]-info[1][1]),'expr only',len(info[1][1]-info[0][1]))
with open('inputs/Model.csv',newline='') as f:
    rd=csv.DictReader(f); mids=[r['ModelID'] for r in rd]
ms=set(mids)
print('Model rows',len(mids),'unique',len(ms))
print('shared IDs in Model',len(info[0][1]&info[1][1]&ms),'shared IDs not Model',len((info[0][1]&info[1][1])-ms))
PY"
