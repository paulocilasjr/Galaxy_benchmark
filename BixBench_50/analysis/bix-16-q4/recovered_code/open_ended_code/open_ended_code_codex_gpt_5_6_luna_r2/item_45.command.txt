/bin/bash -lc "python - <<'PY'
import csv, numpy as np, pandas as pd
from scipy.stats import rankdata, t, spearmanr
with open('inputs/CRISPRGeneEffect.csv',newline='') as f:h=next(csv.reader(f))
with open('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',newline='') as f:e=next(csv.reader(f))
common=[x for x in h[1:] if x in set(e[1:])]
# validate across a few columns after aligning IDs
readids=lambda p:[line.split(',',1)[0] for line in open(p).read().splitlines()[1:]]
di=readids('inputs/CRISPRGeneEffect.csv'); ei=readids('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
shared=[x for x in di if x in set(ei)]
for gene in [common[0],common[100],common[1000],common[-1]]:
 d=pd.read_csv('inputs/CRISPRGeneEffect.csv',usecols=[gene],dtype=float)[gene].to_numpy()
 ex=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=[gene],dtype=float)[gene].to_numpy()
 d=-d[[di.index(x) for x in shared]]; ex=ex[[ei.index(x) for x in shared]]
 ok=np.isfinite(d)&np.isfinite(ex); n=int(ok.sum())
 rx=rankdata(d[ok]); ry=rankdata(ex[ok]); rx-= (n+1)/2; ry-=(n+1)/2
 rho=np.dot(rx,ry)/np.sqrt(np.dot(rx,rx)*np.dot(ry,ry)); st=abs(rho)*np.sqrt((n-2)/((1-rho)*(1+rho))); p=2*t.sf(st,n-2)
 ss=spearmanr(d,ex,nan_policy='omit')
 print(gene,n,rho,p,ss.statistic,ss.pvalue, 'diff',p-ss.pvalue)
PY"
