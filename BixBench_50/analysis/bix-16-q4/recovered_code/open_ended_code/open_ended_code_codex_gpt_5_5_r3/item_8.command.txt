/bin/bash -lc "python - <<'PY'
import pandas as pd
from pathlib import Path
ge_path=Path('inputs/CRISPRGeneEffect.csv')
expr_path=Path('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
model_path=Path('inputs/Model.csv')
ge_cols=pd.read_csv(ge_path, nrows=0).columns.tolist()
expr_cols=pd.read_csv(expr_path, nrows=0).columns.tolist()
models=pd.read_csv(model_path, usecols=['ModelID'])['ModelID'].astype(str)
print('CRISPR cols incl index', len(ge_cols), 'genes', len(ge_cols)-1)
print('Expr cols incl index', len(expr_cols), 'genes', len(expr_cols)-1)
print('Shared gene labels', len(set(ge_cols[1:]) & set(expr_cols[1:])))
print('CRISPR duplicate gene labels', len(ge_cols[1:]) - len(set(ge_cols[1:])))
print('Expr duplicate gene labels', len(expr_cols[1:]) - len(set(expr_cols[1:])))
# row ids only first col
ge_ids=pd.read_csv(ge_path, usecols=[0]).iloc[:,0].astype(str)
expr_ids=pd.read_csv(expr_path, usecols=[0]).iloc[:,0].astype(str)
print('CRISPR rows', len(ge_ids), 'unique', ge_ids.nunique(), 'in Model', ge_ids.isin(set(models)).sum())
print('Expr rows', len(expr_ids), 'unique', expr_ids.nunique(), 'in Model', expr_ids.isin(set(models)).sum())
print('Shared rows', len(set(ge_ids) & set(expr_ids)))
print('Shared rows in Model', len(set(ge_ids) & set(expr_ids) & set(models)))
PY"
