import csv, math, os, time
from array import array

crispr_path = 'inputs/CRISPRGeneEffect.csv'
expr_path = 'inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
model_path = 'inputs/Model.csv'

start = time.time()

def read_header(path):
    with open(path, newline='') as f:
        return next(csv.reader(f))

def read_first_col_set(path):
    vals = []
    with open(path, 'r', newline='') as f:
        next(f)
        for line in f:
            vals.append(line.split(',', 1)[0])
    return vals, set(vals)

def read_model_ids(path):
    ids = []
    with open(path, newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        idx = header.index('ModelID')
        for row in reader:
            ids.append(row[idx])
    return set(ids)

print('Reading headers and IDs', flush=True)
ge_header = read_header(crispr_path)

ge_rows, ge_row_set = read_first_col_set(crispr_path)
expr_rows, expr_row_set = read_first_col_set(expr_path)
model_ids = read_model_ids(model_path)
common_model_set = ge_row_set & expr_row_set & model_ids
common_models = [rid for rid in ge_rows if rid in common_model_set]
model_pos = {rid:i for i, rid in enumerate(common_models)}
expr_gene_set = set(expr_header[1:])
common_genes = [g for g in ge_header[1:] if g in expr_gene_set]
print('shared models', len(common_models), flush=True)
print('shared genes', len(common_genes), flush=True)

ge_col_index = {name:i for i,name in enumerate(ge_header)}
expr_col_index = {name:i for i,name in enumerate(expr_header)}
ge_indices = [ge_col_index[g] for g in common_genes]
expr_indices = [expr_col_index[g] for g in common_genes]
ng = len(common_genes)

def init_cols():
    # array multiplication keeps storage compact and avoids Python float-object overhead.
    return [array('d', [math.nan]) * nm for _ in range(ng)]

def parse_float(s):
    if s == '' or s == 'NA' or s == 'NaN' or s == 'nan':
        return math.nan
    return float(s)

def load_columns(path, indices, negate=False):
    cols = init_cols()
    loaded_rows = 0
    with open(path, 'r', newline='') as f:
        next(f)
        for line_no, line in enumerate(f, start=1):

# Regularized incomplete beta via continued fraction; used for Student t survival.
def betacf(a, b, x):
    MAXIT = 200
    EPS = 3.0e-14
    FPMIN = 1.0e-300
    qab = a + b
    qap = a + 1.0
    qam = a - 1.0
    c = 1.0
    d = 1.0 - qab * x / qap
    if abs(d) < FPMIN:
        d = FPMIN
    d = 1.0 / d
    h = d
    for m in range(1, MAXIT + 1):
        m2 = 2 * m
        aa = m * (b - m) * x / ((qam + m2) * (a + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        h *= d * c
        aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        delta = d * c
        h *= delta
        if abs(delta - 1.0) <= EPS:
            break
    return h

def betai(a, b, x):
    if x <= 0.0:
        return 0.0
    if x >= 1.0:
        return 1.0
    bt = math.exp(math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b) + a * math.log(x) + b * math.log1p(-x))
    if x < (a + 1.0) / (a + b + 2.0):
        return bt * betacf(a, b, x) / a
    return 1.0 - bt * betacf(b, a, 1.0 - x) / b

def t_two_sided_p_from_r(r, n):
    ar = abs(r)
    if ar >= 1.0:
        return 0.0
    if n <= 2:
        return math.nan
    df = n - 2
    t2 = (r * r) * df / (1.0 - r * r)
    x = df / (df + t2)
    return betai(0.5 * df, 0.5, x)

def rank_for_indices(vals, idxs):
    m = len(idxs)
    order = list(range(m))
    order.sort(key=lambda k: vals[idxs[k]])
    ranks = [0.0] * m
    i = 0
    while i < m:
        j = i + 1
        vi = vals[idxs[order[i]]]
        while j < m and vals[idxs[order[j]]] == vi:
            j += 1
        avg = (i + 1 + j) / 2.0  # average of 1-based ranks i+1..j
        for k in range(i, j):
            ranks[order[k]] = avg
        i = j
    return ranks

def spearman_p(x, y):
    idxs = [i for i, (a, b) in enumerate(zip(x, y)) if math.isfinite(a) and math.isfinite(b)]
    n = len(idxs)
    if n < 3:
        return math.nan, math.nan, n
    rx = rank_for_indices(x, idxs)
    ry = rank_for_indices(y, idxs)
    mx = (n + 1) / 2.0
    my = mx
    sxy = sx2 = sy2 = 0.0
    for a, b in zip(rx, ry):
        da = a - mx
        db = b - my
        sxy += da * db
        sx2 += da * da
        sy2 += db * db
    if sx2 <= 0.0 or sy2 <= 0.0:
        return math.nan, math.nan, n
    r = sxy / math.sqrt(sx2 * sy2)
    # Clamp roundoff.
    if r > 1.0:
        r = 1.0
    elif r < -1.0:
        r = -1.0
    return r, t_two_sided_p_from_r(r, n), n

rhos = []
