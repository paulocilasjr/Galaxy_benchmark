from zipfile import ZipFile
from pathlib import Path
sets={}
for g in ['animals','fungi']:
 with ZipFile(Path('inputs')/f'scogs_{g}.zip') as z:
  sets[g]={n[:-len('.faa.mafft.clipkit.treefile')] for n in z.namelist() if n.endswith('.treefile')}
print({g:len(x) for g,x in sets.items()})
print('intersection',len(sets['animals']&sets['fungi']))
print('animals-only',len(sets['animals']-sets['fungi']), sorted(sets['animals']-sets['fungi']))
print('fungi-only',len(sets['fungi']-sets['animals']), sorted(sets['fungi']-sets['animals'])[:30])
PY"
