/bin/bash -lc "python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
from tempfile import TemporaryDirectory
from subprocess import check_output
from scipy.stats import mannwhitneyu

rates={}
with TemporaryDirectory() as td:
 for g in ['animals','fungi']:
  with ZipFile(Path('inputs')/f'scogs_{g}.zip') as z:
   items={n[:-len('.faa.mafft.clipkit.treefile')]:n for n in z.namelist() if n.endswith('.treefile')}
   rates[g]={k:float(check_output(['/opt/agent-venv/bin/pk_evo_rate',str(Path(td)/f'{g}_{i}.tre')], text=True).strip()) for i,(k,n) in enumerate(items.items()) if not (Path(td)/f'{g}_{i}.tre').write_bytes(z.read(n))}
# the dict comp side effect condition always false, so values computed
for scope, a, f in [('all',list(rates['animals'].values()),list(rates['fungi'].values())),('common',[rates['animals'][k] for k in rates['animals'] if k in rates['fungi']],[rates['fungi'][k] for k in rates['animals'] if k in rates['fungi']])]:
 r=mannwhitneyu(a,f,alternative='two-sided')
 print(scope,len(a),len(f),float(r.statistic),float(len(a)*len(f)-r.statistic),float(r.pvalue))
PY"
