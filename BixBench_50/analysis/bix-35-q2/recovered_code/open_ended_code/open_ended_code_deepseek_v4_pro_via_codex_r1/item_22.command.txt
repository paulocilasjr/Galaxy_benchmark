from Bio import SeqIO
import os, glob, re
for d in ['scogs_animals','scogs_fungi']:
    counts=[]
    ex={}
    for fn in glob.glob(d+'/*.faa'):
        recs=list(SeqIO.parse(fn,'fasta'))
        n=len(recs); counts.append(n); ex.setdefault(n,0); ex[n]+=1
        if n<4:
            print(d, os.path.basename(fn), n, [r.id for r in recs])
    print(d,'counts',ex,'total',len(counts))
