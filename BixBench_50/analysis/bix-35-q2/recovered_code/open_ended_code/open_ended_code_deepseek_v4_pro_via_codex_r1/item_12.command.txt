/bin/bash -lc "cd /workspace/work
python - <<'PY'
import os,re
for d in ['scogs_animals','scogs_fungi']:
    pat=re.compile(r'"'^('"\\d+at\\d+)\\.faa')
    bases=set(); ext_sets={}
    for fn in os.listdir(d):
        m=pat.match(fn)
        if not m: continue
        b=m.group(1); bases.add(b)
        # identify suffix after .faa
        suf=fn[len(b)+4:]  # '' or .mafft...
        key=suf or '.faa'
        ext_sets.setdefault(key,set()).add(b)
    print('\\n',d,'bases',len(bases))
    for k in sorted(ext_sets, key=lambda x:(x.count('.'),x)):
        print(k, len(ext_sets[k]))
    missing_tree = bases - ext_sets.get('.mafft.clipkit.treefile',set())
    print('missing treefile bases',len(missing_tree))
    print('first missing',sorted(missing_tree)[:20])
PY"
