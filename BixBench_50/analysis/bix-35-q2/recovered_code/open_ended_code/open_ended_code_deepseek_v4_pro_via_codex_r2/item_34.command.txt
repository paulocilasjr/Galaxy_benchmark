/bin/bash -lc "python3 - <<'PY'
import json, scipy.stats as st, numpy as np
sa=json.load(open('/workspace/tmp/sa/evo_rates.json'))
sf=json.load(open('/workspace/tmp/sf/evo_rates.json'))
A=list(sa.values()); F=list(sf.values())
print(len(A), len(F), np.mean(A), np.mean(F), np.median(A), np.median(F))
for alt in ['two-sided','less','greater']:
 u,p=st.mannwhitneyu(A,F,alternative=alt)
 print(alt,u,p)
print('order animals,fungi two-sided',st.mannwhitneyu(A,F,alternative='two-sided'))
print('order fungi,animals two-sided',st.mannwhitneyu(F,A,alternative='two-sided'))
# common
common=set(sa)&set(sf)
print('common',len(common))
Ac=[sa[g] for g in sorted(common)]; Fc=[sf[g] for g in sorted(common)]
print('common mean med',np.mean(Ac),np.mean(Fc),np.median(Ac),np.median(Fc))
u,p=st.mannwhitneyu(Ac,Fc,alternative='two-sided'); print('common U',u,p)
PY"
