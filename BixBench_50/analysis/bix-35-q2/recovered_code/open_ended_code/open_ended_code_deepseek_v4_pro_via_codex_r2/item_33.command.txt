/bin/bash -lc "cat > /workspace/tmp/compute_rates.py <<'PY'
import glob, os, argparse, json
from types import SimpleNamespace
from phykit.services.tree.evolutionary_rate import EvolutionaryRate

def rate(path):
    return float(str(EvolutionaryRate(SimpleNamespace(tree=path, json=False)).run()))
    # actually run returns None and prints; use service vars

def compute_rate(path):
    # call service object and run capture stdout
    import io, contextlib
    buf=io.StringIO()
    with contextlib.redirect_stdout(buf):
        EvolutionaryRate(SimpleNamespace(tree=path, json=False)).run()
    return float(buf.getvalue().strip())

for d in ['/workspace/tmp/sa','/workspace/tmp/sf']:
    out={}
    files=sorted(glob.glob(os.path.join(d,'*.treefile')))
    for f in files:
        gene=os.path.basename(f).replace('.faa.mafft.clipkit.treefile','')
        out[gene]=compute_rate(f)
    with open(os.path.join(d,'evo_rates.json'),'w') as fh:
        json.dump(out,fh,indent=2)
    print(d, len(out), sum(out.values()))
PY
python3 /workspace/tmp/compute_rates.py"
