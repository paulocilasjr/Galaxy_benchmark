/bin/bash -lc "python - <<'PY'
import zipfile,re
from collections import Counter
for path in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
  z=zipfile.ZipFile(path); vals=[]
  print('\\n',path)
  for n in z.namelist():
    if n.endswith('.treefile'):
      s=z.read(n).decode().strip()
      # names are known
      labels=re.findall(r'(?<"'![A-Za-z0-9_])(?:Animals|Fungi)_[A-Za-z0-9]+(?=[:),])'"',s)
      vals.append((n,len(labels),labels,s))
  print('leaf counts',Counter(v[1] for v in vals))
  print('examples per count:')
  for k in sorted(set(v[1] for v in vals)):
    v=next(x for x in vals if x[1]==k)
    print(k,v[0],v[2],v[3])
PY"
