/bin/bash -lc "PYTHONPATH=/workspace/.phykit_pkgs python - <<'PY'
from argparse import Namespace
from contextlib import redirect_stdout
from io import StringIO
from pathlib import Path
from phykit.services.tree.evolutionary_rate import EvolutionaryRate
from scipy.stats import mannwhitneyu

def rates(d):
  out={}
  for p in Path(d).glob('*.treefile'):
    b=p.name.replace('.faa.mafft.clipkit.treefile','')
    s=StringIO()
    with redirect_stdout(s): EvolutionaryRate(Namespace(tree=str(p),json=False)).run()
    out[b]=float(s.getvalue())
  return out
a=rates('/workspace/.rate_work/animals'); f=rates('/workspace/.rate_work/fungi')
common=sorted(a.keys()&f.keys())
print(len(common), mannwhitneyu([a[x] for x in common],[f[x] for x in common]))
print('all',mannwhitneyu(list(a.values()),list(f.values())))
# using unrounded exact total branch / leaves direct Newick numbers and same all set
import re

def raw(path):
 s=Path(path).read_text()
 # all branch lengths after : are every edge
 vals=[float(x) for x in re.findall(r':([0-9.eE+-]+)',s)]
 labels=re.findall(r'(?<"'![A-Za-z0-9_])(?:Animals|Fungi)_[A-Za-z0-9]+(?=[:),])'"',s)
 return sum(vals)/len(labels)
ar={p.name.replace('.faa.mafft.clipkit.treefile',''):raw(p) for p in Path('/workspace/.rate_work/animals').glob('*.treefile')}
fr={p.name.replace('.faa.mafft.clipkit.treefile',''):raw(p) for p in Path('/workspace/.rate_work/fungi').glob('*.treefile')}
print('raw',mannwhitneyu(list(ar.values()),list(fr.values())))
print('rounded/raw differing ranks?', mannwhitneyu([ar[x] for x in ar],[fr[x] for x in fr]))
PY"
