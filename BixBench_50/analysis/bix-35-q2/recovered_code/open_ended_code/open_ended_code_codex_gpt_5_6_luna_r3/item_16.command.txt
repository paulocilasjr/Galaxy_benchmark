/bin/bash -lc "PYTHONPATH=/workspace/.phykit_pkgs python - <<'PY'
import zipfile, re
from collections import Counter, defaultdict
for path in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\n===', path)
    z=zipfile.ZipFile(path)
    members=z.namelist()
    def base(n): return re.sub(r'\\.faa(?:\\.mafft(?:\\.clipkit)?)?(?:\\.(?:treefile|iqtree|bionj|mldist|log|model|ckp\\.gz))?"'$'"', '', n)
    # more direct suffix stripping to avoid weird
    sets={k:set() for k in ['faa','mafft','clipkit','treefile','iqtree','bionj','mldist','log','gz']}
    for n in members:
        for k,suf in [('treefile','.faa.mafft.clipkit.treefile'),('iqtree','.faa.mafft.clipkit.iqtree'),('bionj','.faa.mafft.clipkit.bionj'),('mldist','.faa.mafft.clipkit.mldist'),('clipkit','.faa.mafft.clipkit'),('mafft','.faa.mafft'),('faa','.faa'),('log','.faa.mafft.clipkit.log')]:
            if n.endswith(suf): sets[k].add(n[:-len(suf)])
    for k,v in sets.items():
        if v: print(f'{k:8} {len(v)}')
    print('tree but no clip', len(sets['treefile']-sets['clipkit']))
    print('clip but no tree', len(sets['clipkit']-sets['treefile']))
    print('raw but no clip', len(sets['faa']-sets['clipkit']))
    print('tree but no raw', len(sets['treefile']-sets['faa']))
    print('first clip-only', sorted(sets['clipkit']-sets['treefile'])[:10])
    print('first tree', sorted(sets['treefile'])[:5])
    # sequence counts for categories
    def fasta_info(data):
        names=[]; seqs=[]; cur=None; chunks=[]
        for line in data.decode().splitlines():
            if line.startswith('>'):
                if cur is not None: seqs.append(''.join(chunks))
                cur=line[1:].split()[0]; names.append(cur); chunks=[]
            elif cur is not None: chunks.append(line.strip())
        if cur is not None: seqs.append(''.join(chunks))
        return names, seqs
    for cat, suffix in [('tree','.faa.mafft.clipkit.treefile'),('clip','.faa.mafft.clipkit'),('mafft','.faa.mafft'),('faa','.faa')]:
        ns=[]
        for n in members:
            if n.endswith(suffix):
                names,seqs=fasta_info(z.read(n))
                ns.append((n,len(names),len(seqs[0]) if seqs else 0, len(set(map(len,seqs))) if seqs else 0, names))
        if ns:
            print(cat,'seqcount',Counter(x[1] for x in ns),'length range',min(x[2] for x in ns),max(x[2] for x in ns),'unequal',sum(x[3]"'!=1 for x in ns))
            if cat=='"'clip': print('clip examples',[(x[0],x[1],x[2],x[4]) for x in ns[:3]])
PY"
