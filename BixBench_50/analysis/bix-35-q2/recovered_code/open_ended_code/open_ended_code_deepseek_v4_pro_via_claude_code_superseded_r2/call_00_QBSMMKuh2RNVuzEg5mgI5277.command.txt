# Verify all 19 genes have 8 sequences in SCOS
SHARED_GENES = [
    '1014314at2759', '1018517at2759', '1193442at2759', '1249159at2759',
    '1269649at2759', '1342242at2759', '1346432at2759', '1405073at2759',
    '1428854at2759', '1504863at2759', '1525971at2759', '1620056at2759',
    '341721at2759', '453044at2759', '570797at2759', '664730at2759',
    '673132at2759', '974865at2759', '996662at2759'
]

ANIMAL_SPECIES = ['Animals_Cele', 'Animals_Ggal', 'Animals_Mmus', 'Animals_Xlae']
FUNGAL_SPECIES = ['Fungi_Ncra', 'Fungi_Scer', 'Fungi_Spom', 'Fungi_Umay']

import os
scogs_a = '/tmp/scogs_a'
scogs_f = '/tmp/scogs_f'

for gene in SHARED_GENES:
    a_faa = os.path.join(scogs_a, f"{gene}.faa")
    f_faa = os.path.join(scogs_f, f"{gene}.faa")
    
    a_seqs = set()
    f_seqs = set()
    
    if os.path.exists(a_faa):
        with open(a_faa) as fh:
            for line in fh:
                if line.startswith('>'):
                    a_seqs.add(line.strip()[1:])
    
    if os.path.exists(f_faa):
        with open(f_faa) as fh:
            for line in fh:
                if line.startswith('>'):
                    f_seqs.add(line.strip()[1:])
    
    all_present = (set(ANIMAL_SPECIES) == a_seqs) and (set(FUNGAL_SPECIES) == f_seqs)
    
    missing_animals = set(ANIMAL_SPECIES) - a_seqs
    missing_fungi = set(FUNGAL_SPECIES) - f_seqs
    
    status = "OK" if all_present else f"MISSING: animals={missing_animals}, fungi={missing_fungi}"
    print(f"{gene}: {len(a_seqs)} animals, {len(f_seqs)} fungi - {status}")
