# Check which species are in the SCOS files for failing genes
for gene in 1538526at2759 1552706at2759 1593937at2759 160593at2759 1645187at2759 176625at2759; do
  echo "=== $gene ==="
  echo "Animals:"
  grep '^>' /tmp/scogs_a/${gene}.faa 2>/dev/null || echo "  NOT FOUND in animals SCOS"
  echo "Fungi:"
  grep '^>' /tmp/scogs_f/${gene}.faa 2>/dev/null || echo "  NOT FOUND in fungi SCOS"
  echo ""
done
