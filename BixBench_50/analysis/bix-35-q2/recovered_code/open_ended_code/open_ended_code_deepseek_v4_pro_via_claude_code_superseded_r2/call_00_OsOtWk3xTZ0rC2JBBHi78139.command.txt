cd /tmp/busco_tables
python3 << 'EOF'
import csv

species_list = ['Animals_Cele', 'Animals_Ggal', 'Animals_Mmus', 'Animals_Xlae',
                'Fungi_Ncra', 'Fungi_Scer', 'Fungi_Spom', 'Fungi_Umay']

# Get complete single-copy genes for each species
complete_genes = {}
for sp in species_list:
    fpath = f"{sp}/run_eukaryota_odb10/full_table.tsv"
    genes = set()
    with open(fpath) as f:
        for line in f:
            if line.startswith('#') or not line.strip():
                continue
            parts = line.strip().split('\t')
            if len(parts) >= 2 and parts[1] == 'Complete':
                genes.add(parts[0])  # Busco id
    complete_genes[sp] = genes
    print(f"{sp}: {len(genes)} complete single-copy genes")

# Find intersection
shared = complete_genes[species_list[0]]
for sp in species_list[1:]:
    shared = shared & complete_genes[sp]

print(f"\nShared complete genes across all 8 species: {len(shared)}")
print("First 10:", sorted(list(shared))[:10])
EOF
