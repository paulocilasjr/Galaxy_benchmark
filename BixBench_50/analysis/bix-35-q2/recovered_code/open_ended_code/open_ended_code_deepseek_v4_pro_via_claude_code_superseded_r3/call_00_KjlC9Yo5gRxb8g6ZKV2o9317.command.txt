# Check if .treefile and .iqtree are paired or separate
unzip -l inputs/scogs_animals.zip | grep -E "\.treefile$" | head -5
echo "---"
unzip -l inputs/scogs_animals.zip | grep -E "\.iqtree$" | head -5
echo "---"
# Are genes with treefile the same as iqtree?
unzip -l inputs/scogs_animals.zip | grep -oP '\d+at\d+' | grep -E "treefile|iqtree" -B0 2>/dev/null
# Let me do this differently
unzip -l inputs/scogs_animals.zip | grep -E "\.treefile$" | awk '{print $NF}' | sed 's/\.faa\.mafft\.clipkit\.treefile//' | sort > /tmp/animals_treefile.txt
unzip -l inputs/scogs_animals.zip | grep -E "\.iqtree$" | awk '{print $NF}' | sed 's/\.faa\.mafft\.clipkit\.iqtree//' | sort > /tmp/animals_iqtree.txt
echo "=== Overlap ==="
comm -12 /tmp/animals_treefile.txt /tmp/animals_iqtree.txt | wc -l
echo "=== Treefile only ==="
comm -23 /tmp/animals_treefile.txt /tmp/animals_iqtree.txt | wc -l
echo "=== IQTREE only ==="
comm -13 /tmp/animals_treefile.txt /tmp/animals_iqtree.txt | wc -l
