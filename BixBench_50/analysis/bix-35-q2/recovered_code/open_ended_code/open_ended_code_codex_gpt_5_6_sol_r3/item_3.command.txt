/bin/bash -lc "find inputs -maxdepth 1 -type f -printf '%f\\t%s bytes\\n' | sort
for archive in inputs/scogs_animals.zip inputs/scogs_fungi.zip inputs/Animals_Cele.busco.zip inputs/busco_downloads.zip inputs/eukaryota_odb10.2024-01-08.tar.gz; do
  echo \""'$archive"
  case "$archive" in
    *.zip) unzip -l "$archive" | sed -n '"'1,18p' ;;
    *.tar.gz) tar -tzf \""'$archive" | sed -n '"'1,18p' ;;
  esac
done"
