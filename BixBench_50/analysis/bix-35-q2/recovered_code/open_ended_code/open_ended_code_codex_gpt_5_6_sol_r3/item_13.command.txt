/bin/bash -lc 'analysis_tmp=$(find /workspace -maxdepth 1 -type d -name '"'evo_rate.*' -print -quit)
python - \""'$analysis_tmp" <<'"'PY'
import sys, subprocess, pathlib, statistics, json
from scipy.stats import mannwhitneyu
root=pathlib.Path(sys.argv[1])
exe='/opt/agent-venv/bin/pk_evo_rate'
groups={}
for group in ('animals','fungi'):
    vals={}
    failed=[]
    for p in sorted((root/group).glob('*.treefile')):
        cp=subprocess.run([exe,str(p)],text=True,capture_output=True)
        if cp.returncode:
            failed.append((p.name,cp.stderr.strip()))
            continue
        try: v=float(cp.stdout.strip())
        except Exception as e:
            failed.append((p.name,repr((cp.stdout,cp.stderr,e))))
            continue
        vals[p.name.split('.faa',1)[0]]=v
    groups[group]=vals
    xs=list(vals.values())
    print(group,'selected',len(list((root/group).glob('*.treefile'))),'processed',len(xs),'failed',len(failed),
          'min',min(xs),'median',statistics.median(xs),'mean',statistics.mean(xs),'max',max(xs))
    if failed: print('failures',failed)
a=list(groups['animals'].values()); f=list(groups['fungi'].values())
for method in ('auto','asymptotic','exact'):
    res=mannwhitneyu(a,f,alternative='two-sided',method=method)
    print('full two-sided',method,'U_animals',res.statistic,'U_fungi',len(a)*len(f)-res.statistic,'p',res.pvalue)
common=sorted(groups['animals'].keys() & groups['fungi'].keys())
ac=[groups['animals'][k] for k in common]; fc=[groups['fungi'][k] for k in common]
res=mannwhitneyu(ac,fc,alternative='two-sided',method='auto')
print('common genes',len(common),'U_animals',res.statistic,'U_fungi',len(ac)*len(fc)-res.statistic,'p',res.pvalue)
PY"
