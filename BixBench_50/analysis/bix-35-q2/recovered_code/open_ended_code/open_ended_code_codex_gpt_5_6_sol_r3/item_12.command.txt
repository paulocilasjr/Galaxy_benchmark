/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from pathlib import PurePosixPath
sets=[]
for p in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 with ZipFile(p) as z:
  ids={PurePosixPath(n).name.split('.faa',1)[0] for n in z.namelist() if n.endswith('.treefile')}
 sets.append(ids)
print('animal',len(sets[0]),'fungi',len(sets[1]),'intersection',len(sets[0]&sets[1]))
print('animal_only',len(sets[0]-sets[1]),sorted(sets[0]-sets[1])[:20])
print('fungi_only',len(sets[1]-sets[0]),sorted(sets[1]-sets[0])[:20])
PY"
