/bin/bash -lc "python - <<'PY'
from contextlib import redirect_stdout
from io import StringIO
from pathlib import Path
from types import SimpleNamespace
from collections import Counter
from phykit.services.tree.evolutionary_rate import EvolutionaryRate

for group in ('animals','fungi'):
    vals=[]
    rows=[]
    for p in sorted((Path('/workspace/.work_rates')/group).glob('*.treefile')):
        buf=StringIO()
        EvolutionaryRate(SimpleNamespace(tree=str(p), json=False)).run()
        # class output goes to stdout; rerun capture correctly below
        with redirect_stdout(buf):
            EvolutionaryRate(SimpleNamespace(tree=str(p), json=False)).run()
        s=buf.getvalue().strip()
        v=float(s)
        vals.append(v); rows.append((p.name,v))
    print(group, 'n=',len(vals), 'min=',min(vals), 'max=',max(vals), 'mean=',sum(vals)/len(vals), 'median=',sorted(vals)[len(vals)//2] if len(vals)%2 else (sorted(vals)[len(vals)//2-1]+sorted(vals)[len(vals)//2])/2)
    print('unique=',len(set(vals)), 'ties=',sum(c-1 for c in Counter(vals).values() if c>1), 'top counts=',Counter(vals).most_common(10))
    print('first=',rows[:5])
PY"
