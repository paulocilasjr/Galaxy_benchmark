/bin/bash -lc "python - <<'PY'
from contextlib import redirect_stdout
from io import StringIO
from pathlib import Path
from types import SimpleNamespace
from phykit.services.tree.evolutionary_rate import EvolutionaryRate
from scipy.stats import mannwhitneyu, rankdata


def rates(group):
    out=[]
    for p in sorted((Path('/workspace/.work_rates')/group).glob('*.treefile')):
        b=StringIO()
        with redirect_stdout(b):
            EvolutionaryRate(SimpleNamespace(tree=str(p), json=False)).run()
        out.append(float(b.getvalue().strip()))
    return out
animals=rates('animals'); fungi=rates('fungi')
for alt in ('two-sided','less','greater'):
    r=mannwhitneyu(animals, fungi, alternative=alt, method='auto')
    print(alt, 'U_animals=',r.statistic, 'p=',r.pvalue)
print('manual U_animals=',sum((a > f) + 0.5*(a == f) for a in animals for f in fungi))
print('manual U_fungi=',sum((f > a) + 0.5*(f == a) for a in animals for f in fungi))
print('n',len(animals),len(fungi),'sum ranks animals',sum(rankdata(animals+fungi)[:len(animals)]))
print('means medians',sum(animals)/len(animals),sum(fungi)/len(fungi))
PY"
