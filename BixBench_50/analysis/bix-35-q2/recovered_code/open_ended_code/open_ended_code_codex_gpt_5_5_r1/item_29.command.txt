/bin/bash -lc "python - <<'PY'
import zipfile, collections
for zname in ['scogs_animals.zip','scogs_fungi.zip']:
    with zipfile.ZipFile('/workspace/inputs/'+zname) as z:
        raw=sorted(n for n in z.namelist() if n.endswith('.faa.mafft.clipkit'))
        tree=set(n[:-len('.treefile')] for n in z.namelist() if n.endswith('.faa.mafft.clipkit.treefile'))
        counts=collections.Counter()
        examples=[]
        for n in raw:
            data=z.read(n).decode(errors='replace')
            ntax=sum(1 for line in data.splitlines() if line.startswith('>'))
            has=n in tree
            counts[(ntax, has)]+=1
            if not has and len(examples)<8:
                examples.append((n, ntax))
        print('\\n', zname)
        for k,v in sorted(counts.items()): print(k, v)
        print('missing tree examples', examples)
PY"
