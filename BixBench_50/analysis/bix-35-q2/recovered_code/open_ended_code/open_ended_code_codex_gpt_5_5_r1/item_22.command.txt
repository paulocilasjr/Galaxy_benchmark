/bin/bash -lc "python - <<'PY'
import zipfile, os
sets={}
for name in ['scogs_animals.zip','scogs_fungi.zip']:
    with zipfile.ZipFile('/workspace/inputs/'+name) as z:
        genes={n.split('.faa.mafft.clipkit.treefile')[0] for n in z.namelist() if n.endswith('.faa.mafft.clipkit.treefile')}
        allgenes={n.split('.faa')[0] for n in z.namelist() if n.endswith('.faa')}
    sets[name]=genes
    print(name, 'tree genes', len(genes), 'all raw faa genes', len(allgenes))
    print('first10', sorted(list(genes))[:10])
inter=sets['scogs_animals.zip'] & sets['scogs_fungi.zip']
print('intersection tree genes', len(inter))
print('animal only', len(sets['scogs_animals.zip']-sets['scogs_fungi.zip']))
print('fungi only', len(sets['scogs_fungi.zip']-sets['scogs_animals.zip']))
print('intersection first20', sorted(inter)[:20])
PY"
