/bin/bash -lc "PYTHONPATH=work/phykit_vendor python - <<'PY'
import glob, os, subprocess, sys
from scipy.stats import mannwhitneyu, rankdata

def phykit_rates(group):
    files=sorted(glob.glob(f'work/scogs_{group}/*.treefile'))
    vals=[]
    for f in files:
        cp=subprocess.run([sys.executable,'-m','phykit','evo_rate',f],check=True,text=True,capture_output=True)
        vals.append(float(cp.stdout.strip()))
    return files,vals

af,a=phykit_rates('animals')
ff,f=phykit_rates('fungi')
r_auto=mannwhitneyu(a,f,alternative='two-sided')
r_asym=mannwhitneyu(a,f,alternative='two-sided',method='asymptotic')
# Independent U1 from rank sum for first sample.
combined=a+f
ranks=rankdata(combined,method='average')
u1=sum(ranks[:len(a)]) - len(a)*(len(a)+1)/2
u2=len(a)*len(f)-u1
print('versions scipy=',__import__('scipy').__version__)
print('counts',len(a),len(f))
print('animals min median max',min(a),__import__('numpy').median(a),max(a))
print('fungi min median max',min(f),__import__('numpy').median(f),max(f))
print('mannwhitneyu auto',r_auto)
print('mannwhitneyu asymptotic',r_asym)
print('manual U1 U2 smaller',u1,u2,min(u1,u2))
print('ties',len(combined)-len(set(combined)))
PY"
