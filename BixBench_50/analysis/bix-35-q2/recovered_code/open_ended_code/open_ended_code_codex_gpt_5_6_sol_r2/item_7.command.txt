/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter,defaultdict
for zfn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 z=ZipFile(zfn)
 c=Counter(); rec=[]
 for n in z.namelist():
  key=n.split('.faa',1)[0]
  suffix=n[len(key):]
  c[suffix]+=1
  if suffix=='.faa.mafft.clipkit':
   b=z.read(n)
   rec.append((key,b.count(b'>'),len(b)))
 print('\\n',zfn,'members',len(z.namelist()),'gene ids',len(set(x.split('.faa')[0] for x in z.namelist())))
 for k,v in sorted(c.items()): print(v,k)
 print('alignment header count',Counter(x[1] for x in rec))
PY
printf '%s\\n' 'ANIMAL MISSING TREE EXAMPLE'
name="'$(comm -23 <(unzip -Z1 inputs/scogs_animals.zip | sed -n '"'s/\\.faa\\.mafft\\.clipkit"'$//p'"' | sort) <(unzip -Z1 inputs/scogs_animals.zip | sed -n 's/\\.faa\\.mafft\\.clipkit\\.treefile"'$//p'"' | sort) | head -1)
printf '%s\\n' \""'$name"
unzip -p inputs/scogs_animals.zip "$name.faa.mafft.clipkit.log" | tail -80
printf '"'%s\\n' 'FUNGI MISSING TREE EXAMPLE'
name="'$(comm -23 <(unzip -Z1 inputs/scogs_fungi.zip | sed -n '"'s/\\.faa\\.mafft\\.clipkit"'$//p'"' | sort) <(unzip -Z1 inputs/scogs_fungi.zip | sed -n 's/\\.faa\\.mafft\\.clipkit\\.treefile"'$//p'"' | sort) | head -1)
printf '%s\\n' \""'$name"
unzip -p inputs/scogs_fungi.zip "$name.faa.mafft.clipkit.log" | tail -80'
