/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import subprocess
ws=load_workbook('inputs/data.xlsx',read_only=True,data_only=True).active
h=[c.value for c in ws[1]]; si=h.index('Sequence number'); yi=h.index('Efficacy'); xi=h.index('BMI')
rows=[r for r in ws.iter_rows(min_row=2,values_only=True) if r[si] is not None]
x=','.join(repr(float(r[xi])) for r in rows)
y=','.join('1' if r[yi]=='PR' else '0' for r in rows)
script=f'''x <- c({x})\\ny <- c({y})\\nfit <- glm(y ~ x, family=binomial(link=\"logit\"))\\ncat(\"n=\", nobs(fit), \"\\\\n\", sep=\"\")\\ncat(\"coef=\", paste(format(coef(fit), digits=16), collapse=\",\"), \"\\\\n\", sep=\"\")\\ncat(\"logLik=\", format(as.numeric(logLik(fit)), digits=16), \"\\\\n\", sep=\"\")\\ncat(\"AIC=\", format(AIC(fit), digits=16), \"\\\\n\", sep=\"\")\\ncat(\"deviance=\", format(deviance(fit), digits=16), \"\\\\n\", sep=\"\")\\n'''
p=subprocess.run(['Rscript','-'],input=script,text=True,capture_output=True,check=True)
print(p.stdout,end='')
if p.stderr: print(p.stderr,end='')
PY"
