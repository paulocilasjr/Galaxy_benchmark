from openpyxl import load_workbook
from collections import Counter, defaultdict
ws=load_workbook('inputs/data.xlsx', read_only=True, data_only=True).active
headers=[c.value for c in next(ws.iter_rows())]
data=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    data.append(dict(zip(headers,row)))
print('rows',len(data))
for key in ['Groups','Efficacy','BMI']:
    vals=[d[key] for d in data]
    print(key, 'missing',sum(v is None for v in vals), 'counts',Counter(vals))
print('cross efficacy/group')
c=defaultdict(Counter)
for d in data:c[d['Groups']][d['Efficacy']]+=1
for k,v in sorted(c.items()): print(k,v)
print('BMI by outcome')
for y in sorted(set(d['Efficacy'] for d in data)):
    x=[float(d['BMI']) for d in data if d['Efficacy']==y and d['BMI'] is not None]
    print(y,len(x),min(x),max(x),sum(x)/len(x))
PY
