R -e '
df <- read.csv("/tmp/data.csv", stringsAsFactors=FALSE)
df$response <- ifelse(df$Efficacy == "PR", 1, 0)

# Verify data
cat("Total rows:", nrow(df), "\n")
cat("PR count:", sum(df$response==1), "\n")
cat("non-PR count:", sum(df$response==0), "\n")

# Check for any missing BMI
cat("Missing BMI:", sum(is.na(df$BMI)), "\n")

# Fit model
model <- glm(response ~ BMI, data = df, family = binomial)

# Get exact values
cat("\nLog-likelihood:", as.numeric(logLik(model)), "\n")
cat("AIC:", AIC(model), "\n")
cat("AIC (model\$aic):", model$aic, "\n")
cat("Residual deviance:", model$deviance, "\n")
cat("Null deviance:", model$null.deviance, "\n")
cat("df residual:", model$df.residual, "\n")
cat("df null:", model$df.null, "\n")

# Manual AIC computation
n_params <- length(coef(model))
manual_aic <- -2 * as.numeric(logLik(model)) + 2 * n_params
cat("\nManual AIC:", manual_aic, "\n")

# Also verify with deviance
cat("AIC from deviance:", model$deviance + 2*n_params, "\n")

# Check model convergence
cat("\nConverged:", model$converged, "\n")
cat("Number of iterations:", model$iter, "\n")
' 2>&1
