R -e '
# Read the CSV data
df <- read.csv("/tmp/data.csv", stringsAsFactors=FALSE)
cat("Number of rows:", nrow(df), "\n")
cat("Columns:", names(df), "\n")

# Look at efficacy values
cat("\nEfficacy values:\n")
print(table(df$Efficacy))

# Look at BMI missingness
cat("\nBMI missing:", sum(is.na(df$BMI)), "\n")
cat("BMI summary:\n")
print(summary(df$BMI))

# Create binary outcome: 1 if PR, 0 otherwise
df$response <- ifelse(df$Efficacy == "PR", 1, 0)
cat("\nResponse table:\n")
print(table(df$response))

# Fit logistic regression: response ~ BMI
model <- glm(response ~ BMI, data = df, family = binomial)
cat("\nModel summary:\n")
print(summary(model))

# Get AIC
aic_val <- AIC(model)
cat("\nAIC:", aic_val, "\n")
cat("AIC (from model):", model$aic, "\n")
'
