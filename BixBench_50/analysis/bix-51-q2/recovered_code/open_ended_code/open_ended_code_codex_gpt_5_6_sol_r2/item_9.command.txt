/bin/bash -lc "R --vanilla --slave -e \"cat('readxl' %in% rownames(installed.packages()), '\\n')\"
python - <<'PY' | R --vanilla --slave
import openpyxl
ws=openpyxl.load_workbook('inputs/data.xlsx',read_only=True,data_only=True)['Sheet1']
it=ws.iter_rows(values_only=True); h=next(it); p={v:i for i,v in enumerate(h) if v is not None}
x=[]; y=[]
for r in it:
    if r[p['Efficacy']] is not None and r[p['BMI']] is not None:
        x.append(float(r[p['BMI']]))
        y.append(1 if r[p['Efficacy']]=='PR' else 0)
print('x <- c('+','.join(format(v,'.17g') for v in x)+')')
print('y <- c('+','.join(map(str,y))+')')
print('m <- glm(y ~ x, family=binomial())')
print(\"cat('n=',nobs(m),' coefficients=',paste(format(coef(m),digits=16),collapse=','),' logLik=',format(as.numeric(logLik(m)),digits=16),' df=',attr(logLik(m),'df'),' AIC=',format(AIC(m),digits=16),' deviance=',format(deviance(m),digits=16),'\\\\n',sep='')\")
PY"
