/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
wb=load_workbook('/workspace/inputs/data.xlsx', data_only=True, read_only=True)
ws=wb['Sheet1']
headers=list(next(ws.iter_rows(values_only=True)))
cols={h:i for i,h in enumerate(headers) if h is not None}
rows=[]
for idx,r in enumerate(ws.iter_rows(min_row=2, values_only=True), start=2):
    vals=[r[cols[c]] for c in ('Sequence number','Efficacy','BMI')]
    if any(v is not None for v in vals):
        rows.append((idx, vals))
print('rows with seq/eff/bmi data', len(rows))
print('last 10')
for item in rows[-10:]: print(item)
complete=[vals for _, vals in rows if vals[1] is not None and vals[2] is not None]
print('complete', len(complete), Counter(v[1] for v in complete))
for eff in sorted(Counter(v[1] for v in complete)):
    vals=[float(v[2]) for v in complete if v[1]==eff]
    print(eff, len(vals), round(sum(vals)/len(vals),6), min(vals), max(vals))
PY"
