import openpyxl
wb = openpyxl.load_workbook('inputs/data.xlsx')
ws = wb['Sheet1']

# Get all unique values in key columns
efficacy_vals = set()
groups_vals = set()
bmi_vals = []

for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    if row[2] is not None:
        efficacy_vals.add(row[2])
    if row[1] is not None:
        groups_vals.add(row[1])
    if row[10] is not None:
        bmi_vals.append(row[10])

print('Efficacy values:', efficacy_vals)
print('Groups values:', groups_vals)
print(f'Total rows (data): {ws.max_row - 1}')
print(f'BMI count: {len(bmi_vals)}')
print(f'BMI range: {min(bmi_vals):.1f} - {max(bmi_vals):.1f}')

# Count efficacy by category
from collections import Counter
efficacy_counts = Counter()
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    if row[2] is not None:
        efficacy_counts[row[2]] += 1
print('Efficacy counts:', dict(efficacy_counts))
