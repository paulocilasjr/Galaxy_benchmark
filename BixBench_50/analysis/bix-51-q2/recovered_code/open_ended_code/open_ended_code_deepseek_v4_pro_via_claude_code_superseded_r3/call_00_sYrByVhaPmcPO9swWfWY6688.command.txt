python3 -c "
import numpy as np
import pandas as pd
import statsmodels.api as sm

# Read data
df = pd.read_excel('inputs/data.xlsx')
print('Columns:', list(df.columns))
print('Shape:', df.shape)

# Relevant columns
print('BMI column:', df.columns[10])
print('Efficacy column:', df.columns[2])

# Filter to rows with BMI
df_model = df[df['BMI'].notna()].copy()
print(f'Rows with BMI: {len(df_model)}')

# Create binary outcome: PR = 1, else = 0
df_model['response'] = (df_model['Efficacy'] == 'PR').astype(int)
print('Response distribution:', df_model['response'].value_counts().to_dict())

# Add intercept
X = sm.add_constant(df_model['BMI'])
y = df_model['response']

# Fit logistic regression
model = sm.Logit(y, X)
result = model.fit(disp=0)

print(f'AIC: {result.aic:.6f}')
print(f'LLF: {result.llf:.6f}')
print(f'Params: {result.params.values}')
print(result.summary())
