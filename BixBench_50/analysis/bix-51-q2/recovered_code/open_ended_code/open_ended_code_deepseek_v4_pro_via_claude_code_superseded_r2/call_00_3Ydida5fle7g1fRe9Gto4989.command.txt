import openpyxl
wb = openpyxl.load_workbook('inputs/data.xlsx')
ws = wb.active

# Print all unique efficacy values
efficacy_vals = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    efficacy_vals.append(row[2])
from collections import Counter
print('Efficacy value counts:')
for k, v in Counter(efficacy_vals).items():
    print(f'  {k!r}: {v}')
print()

# Check BMI for missing values
bmi_vals = []
for row in ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True):
    bmi_vals.append(row[10])
print(f'BMI values: {bmi_vals}')
print(f'BMI missing: {sum(1 for b in bmi_vals if b is None)}')
print(f'Total rows: {len(bmi_vals)}')
