import openpyxl
wb = openpyxl.load_workbook('inputs/data.xlsx')
ws = wb.active

rows = list(ws.iter_rows(min_row=2, max_row=ws.max_row, values_only=True))

# Check alignment of missing values
eff_missing = [i for i, r in enumerate(rows) if r[2] is None]
bmi_missing = [i for i, r in enumerate(rows) if r[10] is None]
print(f'Rows with missing efficacy: {len(eff_missing)}')
print(f'Rows with missing BMI: {len(bmi_missing)}')
print(f'Intersection: {len(set(eff_missing) & set(bmi_missing))}')
print(f'eff missing indices: {eff_missing}')
print(f'bmi missing indices: {bmi_missing}')

# Check how many have both non-missing
both_present = [(i, r) for i, r in enumerate(rows) if r[2] is not None and r[10] is not None]
print(f'Rows with both efficacy and BMI present: {len(both_present)}')
for i, r in both_present:
    print(f'  Row {i+2}: efficacy={r[2]}, BMI={r[10]}')

# Create the analysis dataset
eff_map = {'PR': 1, 'SD': 0, 'PD': 0}
y = []
bmi = []
for r in rows:
    if r[2] is not None and r[10] is not None:
        y.append(eff_map[r[2]])
        bmi.append(r[10])

print(f'\nAnalysis dataset: {len(y)} rows')
print(f'PR=1: {sum(y)}, Non-PR=0: {len(y)-sum(y)}')
