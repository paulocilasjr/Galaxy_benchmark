/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
wb=load_workbook('inputs/data.xlsx', data_only=True, read_only=True)
ws=wb.active
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0])
# trim headers non-null
for i,h in enumerate(headers):
    if h is not None:
        print(i+1, repr(h))
idx_eff=headers.index('Efficacy')
idx_bmi=headers.index('BMI')
vals=[]
missing=[]
for rnum,row in enumerate(rows[1:], start=2):
    eff=row[idx_eff] if idx_eff < len(row) else None
    bmi=row[idx_bmi] if idx_bmi < len(row) else None
    if eff is not None or bmi is not None:
        vals.append((rnum,eff,bmi))
        if eff is None or bmi is None:
            missing.append((rnum,eff,bmi))
print('nonempty eff/bmi rows', len(vals))
print('eff counts', Counter(e for _,e,_ in vals))
print('missing', missing)
print('bmi minmax', min(b for _,_,b in vals if b is not None), max(b for _,_,b in vals if b is not None))
# maybe group/effect counts
idx_grp=headers.index('Groups')
print('group x eff')
c=Counter((row[idx_grp], row[idx_eff]) for row in rows[1:] if row[idx_eff] is not None)
for k,v in sorted(c.items(), key=lambda kv: (str(kv[0][0]), str(kv[0][1]))): print(k,v)
PY"
