/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np, statsmodels.api as sm
from openpyxl import load_workbook
# parse
wb=load_workbook('/workspace/inputs/data.xlsx', data_only=True)
ws=wb['Sheet1']
rows=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None:
        continue
    rows.append(r[:12])
df=pd.DataFrame(rows, columns=['Sequence number','Groups','Efficacy','Age','Family history','ECOG-PS','Child-Pugh','Chemotherapy experience','HBV infection','Gender','BMI','Toxic and side effect'])
print(df.shape)
print(df['Efficacy'].value_counts(dropna=False))
print(df['BMI'].describe())
print('missing BMI', df['BMI'].isna().sum())
y=(df['Efficacy']=='PR').astype(int)
X=sm.add_constant(df[['BMI']].astype(float))
mod=sm.Logit(y,X).fit(disp=False)
print(mod.summary())
print('params', mod.params)
print('llf', mod.llf, 'aic', mod.aic, 'bic', mod.bic, 'nobs', mod.nobs, 'df_model', mod.df_model, 'df_resid', mod.df_resid)
print('aic manual', -2*mod.llf + 2*(mod.df_model+1))
print('bic manual', -2*mod.llf + np.log(mod.nobs)*(mod.df_model+1))
PY"
