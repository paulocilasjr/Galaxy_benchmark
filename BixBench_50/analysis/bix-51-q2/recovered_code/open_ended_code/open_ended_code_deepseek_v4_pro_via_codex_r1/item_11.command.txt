/bin/bash -c "python - <<'PY'
from openpyxl import load_workbook
import numpy as np, math
wb=load_workbook('/workspace/inputs/data.xlsx', data_only=True)
ws=wb['Sheet1']
ys=[]; xs=[]
for r in ws.iter_rows(min_row=2, values_only=True):
    if r[0] is None: continue
    if r[2] is None or r[10] is None: continue
    ys.append(1.0 if r[2]=='PR' else 0.0)
    xs.append(float(r[10]))
y=np.array(ys); x=np.array(xs)
b0=-3.631934; b1=0.167746
for _ in range(100):
    eta=b0+b1*x
    p=1/(1+np.exp(-eta))
    res=y-p
    g0=res.sum(); g1=(res*x).sum()
    w=p*(1-p)
    H00=w.sum(); H01=(w*x).sum(); H11=(w*x*x).sum()
    det=H00*H11-H01**2
    db0=(H11*g0-H01*g1)/det
    db1=(-H01*g0+H00*g1)/det
    b0+=db0; b1+=db1
    if max(abs(db0),abs(db1))<1e-12: break
eta=b0+b1*x
p=1/(1+np.exp(-eta))
p=np.clip(p,1e-300,1-1e-300)
ll=np.sum(y*np.log(p)+(1-y)*np.log(1-p))
print('n', len(y))
print('coefs', b0,b1)
print('ll',repr(float(ll)))
print('AIC',repr(float(-2*ll+2*2)))
PY"
