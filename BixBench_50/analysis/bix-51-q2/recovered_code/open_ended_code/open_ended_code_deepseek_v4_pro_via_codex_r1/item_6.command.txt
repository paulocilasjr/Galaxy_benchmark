/bin/bash -c "python - <<'PY'
from openpyxl import load_workbook
wb=load_workbook('/workspace/inputs/data.xlsx', data_only=True)

# print dimensions and all nonempty cells beyond
for i,row in enumerate(ws.iter_rows(values_only=False),1):
    vals=[c.value for c in row if c.value is not None]
    if vals:
        print(i, [(c.coordinate, repr(c.value)) for c in row if c.value is not None])
PY"
