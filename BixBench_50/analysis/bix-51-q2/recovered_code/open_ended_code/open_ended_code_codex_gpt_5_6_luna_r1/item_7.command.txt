/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
from collections import Counter
p='/workspace/inputs/data.xlsx'
wb=load_workbook(p,data_only=True,read_only=True)
ws=wb.active
rows=list(ws.iter_rows(values_only=True))
headers=list(rows[0])
# only through last nonempty header
last=max(i for i,h in enumerate(headers) if h is not None)
headers=headers[:last+1]
data=[dict(zip(headers[:last+1],r[:last+1])) for r in rows[1:]]
print('headers:',headers)
for k in ['Efficacy','BMI']:
    vals=[d[k] for d in data]
    print(k,'n',len(vals),'missing',sum(v is None for v in vals),'unique/counts',Counter(vals))
print('eff x BMI usable:')
print(Counter((d['Efficacy'],d['BMI'] is None) for d in data))
print('rows with nonnumeric BMI:',[(i+2,d['BMI']) for i,d in enumerate(data) if d['BMI'] is not None and not isinstance(d['BMI'],(int,float))])
PY"
