/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import math

wb=load_workbook('inputs/data.xlsx', data_only=True, read_only=True)
ws=wb['Sheet1']
headers=[c.value for c in next(ws.iter_rows(min_row=1,max_row=1))]
headers=[h for h in headers if h is not None]
rows=[]
for row in ws.iter_rows(min_row=2, values_only=True):
    vals=list(row[:len(headers)])
    if all(v is None or (isinstance(v,str) and v.strip()=='') for v in vals):
        continue
    d=dict(zip(headers, vals))
    if d['Efficacy'] is None or d['BMI'] is None:
        continue
    rows.append(d)

y=[1.0 if r['Efficacy']=='PR' else 0.0 for r in rows]
x=[float(r['BMI']) for r in rows]
# Newton-Raphson for two-parameter logistic regression.
b0=0.0; b1=0.0
for it in range(100):
    g0=g1=0.0
    h00=h01=h11=0.0
    ll=0.0
    for yi,xi in zip(y,x):
        eta=b0+b1*xi
        if eta>=0:
            p=1.0/(1.0+math.exp(-eta))
            ll += yi*eta - math.log1p(math.exp(eta))
        else:
            p=math.exp(eta)/(1.0+math.exp(eta))
            ll += yi*eta - math.log1p(math.exp(eta))
        w=p*(1-p)
        g0 += yi-p
        g1 += (yi-p)*xi
        h00 -= w
        h01 -= w*xi
        h11 -= w*xi*xi
    # solve H delta = g; beta_new = beta - delta for maximizing with Hessian H
    det=h00*h11-h01*h01
    d0=( h11*g0 - h01*g1)/det
    d1=(-h01*g0 + h00*g1)/det
    b0_new=b0-d0
    b1_new=b1-d1
    if max(abs(b0_new-b0), abs(b1_new-b1)) < 1e-12:
        b0,b1=b0_new,b1_new
        break
    b0,b1=b0_new,b1_new
# final ll
ll=0.0
for yi,xi in zip(y,x):
    eta=b0+b1*xi
    if eta>=0:
        ll += yi*eta - math.log1p(math.exp(eta))
    else:
        ll += yi*eta - math.log1p(math.exp(eta))
aic=-2*ll+2*2
print('n',len(y),'events',sum(y),'non_events',len(y)-sum(y))
print('coef',b0,b1)
print('logLik',ll)
print('AIC',aic)
PY"
