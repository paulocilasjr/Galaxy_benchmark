/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, statsmodels.api as sm
from scipy.optimize import minimize
p='/workspace/inputs/data.xlsx'
df=pd.read_excel(p)
df=df.iloc[:80]
print(df.head())
print(df.shape)
print(df['Efficacy'].value_counts(dropna=False))
print(df['BMI'].isna().sum(), df['BMI'].dtype)
y=(df['Efficacy']=='PR').astype(int)
X=sm.add_constant(df['BMI'], has_constant='add')
model=sm.Logit(y,X).fit(disp=0)
print(model.summary())
print('params', model.params)
print('llf',model.llf)
print('aic',model.aic)
print('bic',model.bic)
print('df_model',model.df_model, 'df_resid', model.df_resid, 'nobs', int(model.nobs))
# manual mle with intercept and BMI to cross-check
x=df['BMI'].to_numpy(float); yy=y.to_numpy(float)
def negll(b):
    z=b[0]+b[1]*x
    # log1pexp stable
    return np.sum(np.logaddexp(0,z)-yy*z)
r=minimize(negll, [0,0], method='BFGS')
print('manual params',r.x, 'nll',r.fun)
print('manual aic', 2*r.fun+2*2)
# no intercept variant just for info
model0=sm.Logit(y, df['BMI']).fit(disp=0)
print('no intercept params',model0.params,'llf',model0.llf,'aic',model0.aic)
PY"
