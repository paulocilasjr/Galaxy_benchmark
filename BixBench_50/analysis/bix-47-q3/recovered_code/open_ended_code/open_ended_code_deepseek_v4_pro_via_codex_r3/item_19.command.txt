/bin/bash -lc "mkdir -p codex_output && python3 - <<'PY'
import openpyxl, glob, os, re, collections
# Determine oldest male carrier from Trio_Status
wb = openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx', data_only=True)
ws = wb.active
rows = list(ws.iter_rows(values_only=True))
hdr = rows[0]
idx = {h:i for i,h in enumerate(hdr)}
male_carriers = []
for r in rows[1:]:
    sid, status, sex, blm, age = r[0], r[1], r[2], r[3], r[4]
    if sex == 'M' and blm == 'Carrier' and isinstance(age, (int, float)):
        male_carriers.append((age, sid, status))
oldest = max(male_carriers)
print('oldest male carrier:', oldest)
# Locate file by sample ID in filename
pat = f'inputs/CHIP_DP10_GQ20_PASS/*CHIP_{oldest[1]}-*.xlsx'
files = glob.glob(pat)
print('matched', files)
f = files[0]
wb = openpyxl.load_workbook(f, data_only=True)
ws = wb.active
# Gene list
chip = set()
for r in openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx', data_only=True).active.iter_rows(values_only=True):
    if r[0]:
        chip.add(str(r[0]).strip())
counts = collections.Counter()
for r in ws.iter_rows(min_row=3, values_only=True):
    zyg = str(r[4])
    if zyg in ('Reference', 'None'):
        continue
    seen = set()
    for tok in re.split('[,;]', str(r[12] or '')):
        tok = tok.strip()
        if tok in chip:
            seen.add(tok)
    for g in seen:
        counts[g] += 1
print('top gene counts:')
for g,c in counts.most_common(10):
    print(g,c)
ans = counts.most_common(1)[0][0]
print('ANSWER', ans)
with open('codex_output/answer.txt','w') as out:
    out.write(ans + '\\n')
PY"
