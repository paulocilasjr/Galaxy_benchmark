/bin/bash -lc "python3 - <<'PY'
import openpyxl, collections, re
chip=set()
for r in openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx', data_only=True).active.iter_rows(values_only=True):
 if r[0]: chip.add(str(r[0]).strip())
print('chip',sorted(chip))
f='inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx'
wb=openpyxl.load_workbook(f, data_only=True); ws=wb.active
rows=list(ws.iter_rows(values_only=True))
genes=collections.Counter(); missing=[]
for r in rows[2:]:
    gs=str(r[12] or '')
    for g in re.split('[,;]',gs):
        g=g.strip()
        if g: genes[g]+=1
        if g and g not in chip and g not in missing: missing.append(g)
print('total unique genes in file',len(genes))
print('missing from CHIP list',missing)
for g,c in genes.most_common():
    mark='*' if g not in chip else ' '
    print(f'{mark} {g:12s} {c}')
PY"
