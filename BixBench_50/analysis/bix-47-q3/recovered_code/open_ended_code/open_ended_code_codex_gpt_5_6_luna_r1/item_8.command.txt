/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter
import openpyxl

base=Path('inputs')
# CHIP genes
wb=openpyxl.load_workbook(base/'230214_Schenz_et_al_2022_CHIP_Genes.xlsx',data_only=True)
chip={str(wsrow[0].value).strip() for wsrow in wb.active.iter_rows() if wsrow[0].value is not None}
wb.close()
# metadata
wb=openpyxl.load_workbook(base/'230215_Trio_Status.xlsx',data_only=True)
ws=wb.active
headers=[c.value for c in next(ws.iter_rows())]
records=[]
for row in ws.iter_rows(min_row=2,values_only=True):
    rec=dict(zip(headers,row))
    rec={k:(v.strip() if isinstance(v,str) else v) for k,v in rec.items()}
    records.append(rec)
wb.close()
carriers=[r for r in records if r.get('BLM Mutation Status')=='Carrier' and r.get('Sex')=='M' and isinstance(r.get('Age'),(int,float))]
print('male carriers sorted by age:')
for r in sorted(carriers,key=lambda x:x['Age'],reverse=True): print(r['Sample ID'],r['Status'],r['Sex'],r['Age'])
old=max(carriers,key=lambda x:x['Age'])
sid=str(old['Sample ID'])
files=list((base/'CHIP_DP10_GQ20_PASS').glob(f'*CHIP_{sid}-*.xlsx'))
print('oldest',old,'files',[p.name for p in files])
assert len(files)==1
p=files[0]
wb=openpyxl.load_workbook(p,data_only=True)
ws=wb.active
print('file header row1', [ws.cell(1,c).value for c in range(1,ws.max_column+1) if ws.cell(1,c).value is not None])
headers=[ws.cell(2,c).value for c in range(1,ws.max_column+1)]
hi={h:i for i,h in enumerate(headers)}
print('rows',ws.max_row-2)
nonref=[]
for rr in range(3,ws.max_row+1):
    vals=[ws.cell(rr,c).value for c in range(1,ws.max_column+1)]
    z=vals[hi['Zygosity']]
    if z in {'Heterozygous','Homozygous Variant'}:
        nonref.append(vals)
wb.close()
print('nonref',len(nonref))
# Exact annotation count and split count among listed CHIP genes
exact=Counter(v[hi['Gene Names']] for v in nonref)
split=Counter()
for v in nonref:
    raw=v[hi['Gene Names']]
    if raw is None: continue
    for g in str(raw).replace(';',',').split(','):
        g=g.strip()
        if g in chip: split[g]+=1
print('exact top',exact.most_common(20))
print('split CHIP top',split.most_common(20))
print('top exact', exact.most_common(1))
print('top split', split.most_common(1))
# Freq nonref statuses and nonempty VAF
print('all zygosity counter',Counter(ws.cell(rr,hi['Zygosity']+1).value for rr in range(3,ws.max_row+1)))
PY"
