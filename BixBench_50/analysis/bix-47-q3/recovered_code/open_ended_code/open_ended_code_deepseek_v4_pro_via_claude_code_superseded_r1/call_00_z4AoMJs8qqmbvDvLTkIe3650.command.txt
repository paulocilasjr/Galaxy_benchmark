import openpyxl
from collections import Counter

# 1. Load CHIP genes
wb_chip = openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx')
ws_chip = wb_chip.active
chip_genes = set()
for row in range(1, ws_chip.max_row + 1):
    gene = ws_chip.cell(row=row, column=1).value
    if gene:
        chip_genes.add(gene.strip())
print(f"CHIP genes: {len(chip_genes)}")
print(f"Sample: {sorted(list(chip_genes))[:5]}...")

# 2. Load variant file for sample 556
wb = openpyxl.load_workbook('inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_556-PF.xlsx')
ws = wb.active

# Print column indices
header_row = [cell.value for cell in ws[1]]
for i, h in enumerate(header_row):
    print(f"  Col {i}: {h}")

# Columns:
# Col 4 (E): Zygosity
# Col 12 (M): Gene Names
# Col 29 (AD): In_CHIP

# 3. Count non-reference variants per CHIP gene
gene_counter = Counter()
total_variants = 0
non_ref = 0
chip_non_ref = 0

for row in range(2, ws.max_row + 1):
    zygosity = ws.cell(row=row, column=5).value  # Col E
    gene_names = ws.cell(row=row, column=13).value  # Col M
    in_chip = ws.cell(row=row, column=30).value  # Col AD
    
    total_variants += 1
    
    if zygosity != 'Reference':
        non_ref += 1
        if in_chip == 'True' or (gene_names and gene_names.strip() in chip_genes):
            chip_non_ref += 1
            # Could be multiple genes separated by comma/semicolon
            genes = [g.strip() for g in str(gene_names).split(',')]
            for g in genes:
                g = g.strip()
                if g in chip_genes:
                    gene_counter[g] += 1

print(f"\nTotal variants: {total_variants}")
print(f"Non-reference: {non_ref}")
print(f"CHIP gene non-reference: {chip_non_ref}")

print(f"\nTop 20 genes by non-reference variant count:")
for gene, count in gene_counter.most_common(20):
    print(f"  {gene}: {count}")

print(f"\nGene with most non-reference variants: {gene_counter.most_common(1)}")
