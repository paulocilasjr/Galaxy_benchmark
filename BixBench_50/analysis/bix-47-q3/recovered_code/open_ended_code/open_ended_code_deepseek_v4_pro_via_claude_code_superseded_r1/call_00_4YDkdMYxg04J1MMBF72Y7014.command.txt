import openpyxl

# Read CHIP Genes file
wb = openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx')

print("Active sheet:", ws.title)

# Print headers and all rows
headers = [cell.value for cell in ws[1]]
print("\nHeaders:", headers)
for row in range(2, min(ws.max_row + 1, 100)):
    vals = [ws.cell(row=row, column=c).value for c in range(1, ws.max_column + 1)]
    print(f"Row {row}:", vals)
