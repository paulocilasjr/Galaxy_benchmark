import openpyxl

# Final verification: re-check oldest male
wb = openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx')
ws = wb.active

best_male = None
best_age = -1

for row in range(2, ws.max_row + 1):
    s_id = ws.cell(row=row, column=1).value
    sex = ws.cell(row=row, column=3).value
    blm = ws.cell(row=row, column=4).value
    age_raw = ws.cell(row=row, column=5).value
    
    if sex == 'M':
        try:
            age = float(age_raw) if age_raw is not None else -1
        except:
            age = -1
        
        if age > best_age:
            best_age = age
            best_male = (s_id, sex, blm, age, ws.cell(row=row, column=2).value)
            
print(f"Oldest male: ID={best_male[0]}, Sex={best_male[1]}, BLM={best_male[2]}, Age={best_male[3]}, Status={best_male[4]}")
print(f"Age: {best_age}")
print(f"Variant file: 230209_Exome_GRCh38_CHIP_{int(best_male[0])}-PF.xlsx")
