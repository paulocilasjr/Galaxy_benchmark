import openpyxl

wb = openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx')
ws = wb.active
headers = [cell.value for cell in ws[1]]
print(headers)

# Find all males
males = []
for row in range(2, ws.max_row + 1):
    vals = [ws.cell(row=row, column=c).value for c in range(1, ws.max_column + 1)]
    sample_id, status, sex, blm, age, cancer, _ = vals
    if sex == 'M':
        males.append((sample_id, status, sex, blm, age, cancer))
        
males_sorted = sorted(males, key=lambda x: x[4] if x[4] is not None else -1, reverse=True)
for m in males_sorted:
    print(m)

print(f"\nTotal males: {len(males)}")
print(f"Oldest male: {males_sorted[0]}")
