/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
import re
# Independent pass using normal-mode worksheet parsing.
meta=load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True).active
carrier={str(r[0].value).strip() for r in meta.iter_rows(min_row=2) if r[3].value=='Carrier'}
num=den=0; per=[]
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    token=p.stem.split('CHIP_',1)[1].split('-',1)[0]
    if token not in carrier: continue
    ws=load_workbook(p,data_only=True).active
    headers={ws.cell(2,c).value:c for c in range(1,ws.max_column+1)}
    n=d=0
    for rr in range(3,ws.max_row+1):
        val=lambda k: ws.cell(rr,headers[k]).value
        if val('In_CHIP') is not True: continue
        if str(val('Zygosity')).strip().casefold()=='reference': continue
        try: vaf=float(val('Variant Allele Freq'))
        except (TypeError,ValueError): continue
        if vaf >= 0.3: continue
        so=str(val('Sequence Ontology (Combined)') or '')
        if any(x in so for x in ('intron_variant','intergenic_variant','3_prime_UTR_variant','5_prime_UTR_variant')): continue
        cl=str(val('Classification') or '').strip()
        if not cl: continue
        d+=1
        if cl.casefold() in {'benign','likely benign'}: n+=1
    num+=n; den+=d; per.append((token,n,d))
print('per_sample',per)
print('numerator',num,'denominator',den,'proportion',num/den)
PY"
