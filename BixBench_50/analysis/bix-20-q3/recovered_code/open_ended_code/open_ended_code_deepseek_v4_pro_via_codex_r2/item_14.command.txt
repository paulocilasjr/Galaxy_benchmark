/bin/bash -c "python - <<'PY'
import openpyxl,glob,os,re,collections
# Read trio statuses
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True)
ws=wb.active
trio=[]
for r in ws.iter_rows(min_row=2,values_only=True):
    if r[0] is None: continue
    trio.append((str(r[0]).strip(), r[1], r[3]))
carriers=set()
for sid,status,blm in trio:
    if blm == 'Carrier':
        carriers.add(sid)
print('carrier IDs',sorted(carriers, key=lambda x:(x.isdigit(), x)))

files=glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')
carrier_files=[]
for p in files:
    b=os.path.basename(p)
    m=re.search(r'CHIP_(\\d+)-', b)
    if m and m.group(1) in carriers:
        carrier_files.append(p)
print('carrier files',len(carrier_files))
for p in sorted(carrier_files): print(os.path.basename(p))

exclude_terms = {'intron_variant','intergenic_variant','3_prime_UTR_variant','5_prime_UTR_variant','5_prime_UTR_premature_start_codon_gain_variant'}

def is_excluded(so):
    s = str(so)
    # exact or split by delimiters
    parts=[x.strip() for x in re.split(r'[,&|/]',s)]
    return any(t in exclude_terms for t in parts)

counts=collections.Counter(); rows=[]
for p in carrier_files:
    wb=openpyxl.load_workbook(p,data_only=True)
    wsv=wb.active
    for r in wsv.iter_rows(min_row=3,values_only=True):
        if len(r)<33: continue
        zyg=r[4]; vaf=r[9]; cls=r[25]; so=r[13]; refalt=r[1]
        # normalize
        sample=os.path.basename(p)
        rows.append((sample,r[0],zyg,vaf,so,cls,r[2],r[3],r[29],r[13]))
        # broad categories
        counts['all']+=1
        if zyg=='Reference': counts['reference']+=1
        else: counts['nonref']+=1
        # Apply task filters
        if zyg=='Reference': continue
        if vaf is None:
            counts['no_vaf']+=1; continue
        try: v=float(vaf)
        except: 
            counts['vaf_bad']+=1; continue
        if v>=0.3:
            counts['vaf_ge_0.3']+=1; continue
        if is_excluded(so):
            counts['region_excluded']+=1; continue
        if cls is None or str(cls).strip()=='':
            counts['no_clinvar']+=1; continue
        counts['denominator']+=1
        if cls in ('Benign','Likely Benign'):
            counts['benign_likely']+=1
print('\\ncounts',dict(counts))
print('proportion', counts['benign_likely']/counts['denominator'] if counts['denominator'] else None)
# print rows after filters maybe all
print('\\nfiltered rows:')
for x in rows:
    sample,pos,zyg,vaf,so,cls,*_=x
    if zyg=='Reference' or vaf is None: continue
    try: v=float(vaf)
    except: continue
    if v>=0.3: continue
    if is_excluded(so): continue
    if cls is None or str(cls).strip()=='': continue
    print(sample,pos,zyg,vaf,so,cls)
PY"
