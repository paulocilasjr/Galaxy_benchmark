/bin/bash -c "python - <<'PY'
import openpyxl,glob,os,collections,re,math
files=glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')
allN=collections.Counter(); allZ=collections.Counter(); allIn=collections.Counter(); allE=collections.Counter(); allJblank=0; nonref_blank=[]
for p in files:
    wb=openpyxl.load_workbook(p,data_only=True)
    ws=wb.active
    n=0
    for r in ws.iter_rows(min_row=3,values_only=True):
        n+=1
        if len(r)>13: allN[str(r[13])]+=1
        if len(r)>25: allZ[str(r[25])]+=1
        if len(r)>29: allIn[str(r[29])]+=1
        if len(r)>4: allE[str(r[4])]+=1
        # count non-reference J blank
        if len(r)>9 and str(r[4])"'!='"'Reference' and r[9] is None:
            allJblank+=1; nonref_blank.append((os.path.basename(p),r[0],r[4],r[9],r[13],r[25]))
    print(os.path.basename(p), n)
print('\\nN')
for k,v in allN.most_common(): print(repr(k),v)
print('\\nZ Classification')
for k,v in allZ.most_common(): print(repr(k),v)
print('\\nIn_CHIP')
for k,v in allIn.most_common(): print(repr(k),v)
print('\\nZygosity')
for k,v in allE.most_common(): print(repr(k),v)
print('\\nnonref J blank',allJblank)
for x in nonref_blank[:30]: print(x)
PY"
