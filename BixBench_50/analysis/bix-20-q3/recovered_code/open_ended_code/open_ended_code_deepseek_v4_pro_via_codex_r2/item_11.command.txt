/bin/bash -c "python - <<'PY'
import openpyxl,glob,os,re,collections
files=glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')
print('files',len(files))
allvals=collections.Counter()
allgenes=collections.Counter()
allO=collections.Counter()
allAD=collections.Counter()
allZ=collections.Counter()
for p in files:
    wb=openpyxl.load_workbook(p,read_only=True,data_only=True)
    ws=wb.active
    # read all rows with values only from row3
    it=ws.iter_rows(min_row=3,values_only=True)
    n=0
    for r in it:
        n+=1
        if len(r)>13: allvals[str(r[13])]+=1
        if len(r)>14: allO[str(r[14])]+=1
        if len(r)>28: allAD[str(r[28])]+=1
        if len(r)>25: allZ[str(r[25])]+=1
        if len(r)>12: allgenes[str(r[12])]+=1
    print(os.path.basename(p), n)
print('\\nSequence Ontology')
for k,v in allvals.most_common(): print(k,v)
print('\\nEffect')
for k,v in allO.most_common(): print(k,v)
print('\\nIn_CHIP')
for k,v in allAD.most_common(): print(k,v)
print('\\nClassification')
for k,v in allZ.most_common(): print(k,v)
PY"
