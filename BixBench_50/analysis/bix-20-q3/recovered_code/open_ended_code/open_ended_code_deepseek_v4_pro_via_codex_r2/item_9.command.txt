/bin/bash -c "python - <<'PY'
import openpyxl,glob,collections,math
p=glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')[0]
wb=openpyxl.load_workbook(p,read_only=False,data_only=True)
ws=wb.active
rows=list(ws.iter_rows(min_row=3,values_only=True))
print('file',p,'rows',len(rows))
# map headers list from row2
headers=[c for c in next(ws.iter_rows(min_row=2,max_row=2,values_only=True))]
print(headers)
for key in ['E','J','C','D','N','O','Z']:
    # E index4, J9, C2,D3,N13,O14,Z25
    idx={'E':4,'J':9,'C':2,'D':3,'N':13,'O':14,'Z':25}[key]
    vals=collections.Counter(str(r[idx]) for r in rows if idx<len(r))
    print('\\n',key,headers[idx] if idx<len(headers) else '?',len(vals))
    for v,c in vals.most_common(30): print(repr(v),c)
PY"
