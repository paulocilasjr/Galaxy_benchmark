import pandas as pd

sample = pd.read_excel("inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_185-PF.xlsx", header=None)

# Check VAF values
print("=== Variant Allele Freq (Col 9) stats ===")
vaf = pd.to_numeric(sample.iloc[2:, 9], errors='coerce')
print(f"Count: {vaf.count()}, NaN: {vaf.isna().sum()}")
print(vaf.describe())

# Check rows where Zygosity is NOT Reference and has a VAF < 0.3
data = sample.iloc[2:].copy()
data.columns = ['ChrPos', 'RefAlt', 'PrimaryFindings', 'IncidentalFindings', 'Zygosity', 
                'MendelError', 'InheritedFrom', 'ReadDepths', 'GenotypeQual', 'VAF',
                'InACMG_SF', 'Match_ACMG', 'GeneNames', 'SeqOntology', 'Effect',
                'Transcript', 'ExonNum', 'HGVSc', 'HGVSp', 'Filter',
                'gnomAD_AF', 'gnomAD_AC', 'gnomAD_AN', 'gnomAD_Hom', 'VariantID',
                'Classification', 'AggregateInterpretations', 'ReviewStatus', 'Conditions',
                'In_CHIP', 'Match_CHIP', 'In_DNAm', 'Match_DNAm']

data['VAF'] = pd.to_numeric(data['VAF'], errors='coerce')

# Non-reference variants with VAF info
nonref = data[data['Zygosity'] != 'Reference']
print(f"\nNon-reference variants: {len(nonref)}")
print(f"Non-ref with VAF < 0.3: {len(nonref[nonref['VAF'] < 0.3])}")
print(f"Non-ref with VAF NaN: {nonref['VAF'].isna().sum()}")

# Show some examples
print("\nSample non-ref rows with VAF:")
print(nonref[['ChrPos', 'RefAlt', 'Zygosity', 'VAF', 'GeneNames', 'SeqOntology', 'Classification']].head(10).to_string())
