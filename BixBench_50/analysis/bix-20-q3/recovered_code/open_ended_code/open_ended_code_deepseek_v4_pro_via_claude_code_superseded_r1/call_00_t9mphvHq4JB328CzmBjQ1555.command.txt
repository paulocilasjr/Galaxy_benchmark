import pandas as pd
import os
import numpy as np

# Final verification with all edge cases checked
trio = pd.read_excel("inputs/230215_Trio_Status.xlsx")
carriers = trio[trio['BLM Mutation Status'] == 'Carrier']

variant_dir = "inputs/CHIP_DP10_GQ20_PASS"
all_files = os.listdir(variant_dir)

exclude_seq = ['intron_variant', 'intergenic_variant', 'intergenic_region', 
               '3_prime_UTR_variant', '5_prime_UTR_variant',
               '5_prime_UTR_premature_start_codon_gain_variant']

total_with_clinvar = 0
total_benign = 0
total_non_benign = 0
non_benign_details = []

for _, row in carriers.iterrows():
    sid = row['Sample ID']
    matching = [f for f in all_files if f'CHIP_{sid}-' in f]
    if not matching:
        continue
    fpath = os.path.join(variant_dir, matching[0])
    df = pd.read_excel(fpath, header=None)
    data = df.iloc[2:]
    
    zygosity = data[4]
    seq_ontology = data[13]
    vaf = pd.to_numeric(data[9], errors='coerce')
    classification = data[25]
    
    combined = data[(zygosity != 'Reference') & 
                    (~seq_ontology.isin(exclude_seq)) & 
                    (vaf < 0.3) & 
                    (classification.notna())]
    
    for _, v in combined.iterrows():
        total_with_clinvar += 1
        cls = str(v[25])
        if cls in ['Benign', 'Likely Benign']:
            total_benign += 1
        else:
            total_non_benign += 1
            non_benign_details.append((sid, v[0], v[25]))

print(f"FINAL RESULTS:")
print(f"  Total with ClinVar classification: {total_with_clinvar}")
print(f"  Benign or Likely Benign: {total_benign}")
print(f"  Other classifications: {total_non_benign}")
if total_non_benign > 0:
    for d in non_benign_details:
        print(f"    Sample {d[0]}, {d[1]}: {d[2]}")
if total_with_clinvar > 0:
    proportion = total_benign / total_with_clinvar
    print(f"  Proportion: {proportion}")
