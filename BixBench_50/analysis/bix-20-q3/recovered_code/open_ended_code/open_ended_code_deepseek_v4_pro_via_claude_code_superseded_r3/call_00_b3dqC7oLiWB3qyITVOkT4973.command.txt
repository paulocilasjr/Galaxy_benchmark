import openpyxl

wb = openpyxl.load_workbook('/workspace/inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_185-PF.xlsx')
ws = wb[wb.sheetnames[0]]

# Check: for VAF=None variants, what is their zygosity?
vaf_none_count = 0
vaf_none_zyg = {}
vaf_none_so = {}
for row in ws.iter_rows(min_row=3, values_only=True):
    if row[9] is None:
        vaf_none_count += 1
        z = row[4]
        so = row[13]
        vaf_none_zyg[z] = vaf_none_zyg.get(z, 0) + 1
        vaf_none_so[so] = vaf_none_so.get(so, 0) + 1

print(f"Total VAF=None: {vaf_none_count}")
print(f"VAF=None by zygosity: {vaf_none_zyg}")
print(f"VAF=None by SO (top 10): {sorted(vaf_none_so.items(), key=lambda x:-x[1])[:10]}")

# Check VAF < 0.3 versus >= 0.3
vaf_lt_03 = 0
vaf_ge_03 = 0
vaf_none = 0
for row in ws.iter_rows(min_row=3, values_only=True):
    if row[9] is None:
        vaf_none += 1
    else:
        vaf = float(row[9])
        if vaf < 0.3:
            vaf_lt_03 += 1
        else:
            vaf_ge_03 += 1

print(f"\nVAF distribution: <0.3: {vaf_lt_03}, >=0.3: {vaf_ge_03}, None: {vaf_none}")
