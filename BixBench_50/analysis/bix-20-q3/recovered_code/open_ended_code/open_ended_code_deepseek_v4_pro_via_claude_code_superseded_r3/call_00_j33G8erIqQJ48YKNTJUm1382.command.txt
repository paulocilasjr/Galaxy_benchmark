import openpyxl
import os
import glob
from collections import Counter

# Get carrier samples
wb_trio = openpyxl.load_workbook('/workspace/inputs/230215_Trio_Status.xlsx')
ws_trio = wb_trio['Sheet1']
carrier_samples = set()
for row in ws_trio.iter_rows(min_row=2, values_only=True):
    if row[3] == 'Carrier':
        carrier_samples.add(str(row[0]).strip())

# Map sample IDs to files
variant_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS/'
all_files = glob.glob(os.path.join(variant_dir, '*.xlsx'))
sample_to_file = {}
for fpath in all_files:
    fname = os.path.basename(fpath)
    parts = fname.replace('.xlsx', '').split('_')
    last_part = parts[-1]
    if '-' in last_part:
        sid = last_part.split('-')[0]
    else:
        sid = last_part
    sample_to_file[sid] = fpath

# Collect ALL SO values across all files (just sample a few for speed)
all_so = set()
clinvar_classifications = set()
for fpath in all_files[:5]:  # sample 5 files
    wb = openpyxl.load_workbook(fpath)
    ws = wb[wb.sheetnames[0]]
    for row in ws.iter_rows(min_row=3, values_only=True):
        if row[13] is not None:
            all_so.add(str(row[13]))
        if row[25] is not None:
            clinvar_classifications.add(str(row[25]))

print(f"All SO values across sampled files: {sorted(all_so)}")
print(f"\nAll ClinVar classifications: {sorted(clinvar_classifications)}")
