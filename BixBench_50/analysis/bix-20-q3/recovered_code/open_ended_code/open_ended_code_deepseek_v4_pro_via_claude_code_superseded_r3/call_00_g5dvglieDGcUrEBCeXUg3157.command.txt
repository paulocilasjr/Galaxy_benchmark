import openpyxl
import os
import glob

# Step 1: Get carrier samples from Trio Status
wb_trio = openpyxl.load_workbook('/workspace/inputs/230215_Trio_Status.xlsx')
ws_trio = wb_trio['Sheet1']

carrier_samples = set()
for row in ws_trio.iter_rows(min_row=2, values_only=True):
    if row[3] == 'Carrier':
        # Sample ID might have trailing whitespace (like 'SRR5463149\t')
        sample_id = str(row[0]).strip()
        carrier_samples.add(sample_id)

print(f"Carrier samples: {sorted(carrier_samples)}")
print(f"Total carriers: {len(carrier_samples)}")

# Step 2: Build mapping from sample ID to file path
variant_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS/'
all_files = glob.glob(os.path.join(variant_dir, '*.xlsx'))

# Extract sample ID from filename: "230209_Exome_GRCh38_CHIP_185-PF.xlsx" -> "185"
sample_to_file = {}
for fpath in all_files:
    fname = os.path.basename(fpath)
    # Extract the sample identifier between "CHIP_" and "-" or ".xlsx"
    # Pattern: ...CHIP_<ID>-...xlsx
    parts = fname.replace('.xlsx', '').split('_')
    # Last part contains ID like "185-PF" or "SRR5456220"
    last_part = parts[-1]
    if '-' in last_part:
        # Like "185-PF" -> sample ID is the part before the dash
        sid = last_part.split('-')[0]
    else:
        sid = last_part
    sample_to_file[sid] = fpath

# Check which carrier samples have variant files
print(f"\nCarrier samples with variant files:")
for s in sorted(carrier_samples):
    if s in sample_to_file:
        print(f"  {s}: {os.path.basename(sample_to_file[s])}")
    else:
        print(f"  {s}: NO FILE FOUND")

# Also check for mismatches
missing = [s for s in carrier_samples if s not in sample_to_file]
print(f"\nMissing carrier files: {missing}")
