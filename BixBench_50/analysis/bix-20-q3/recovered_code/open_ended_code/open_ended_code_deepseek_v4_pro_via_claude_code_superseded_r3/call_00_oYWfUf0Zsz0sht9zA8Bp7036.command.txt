import openpyxl

# Check a few more files for In_CHIP values and ClinVar classifications
files_to_check = [
    '230209_Exome_GRCh38_CHIP_185-PF.xlsx',
    '230209_Exome_GRCh38_CHIP_184-PM.xlsx',
    '230209_Exome_GRCh38_CHIP_380-P.xlsx',
]

for fname in files_to_check:
    wb = openpyxl.load_workbook(f'/workspace/inputs/CHIP_DP10_GQ20_PASS/{fname}')
    ws = wb[wb.sheetnames[0]]
    
    in_chip_vals = set()
    clinvar_vals = set()
    total_rows = 0
    
    for row in ws.iter_rows(min_row=3, values_only=True):
        total_rows += 1
        if row[29] is not None:
            in_chip_vals.add(str(row[29]))
        if row[25] is not None:
            clinvar_vals.add(str(row[25]))
    
    print(f"\n{fname}:")
    print(f"  Total rows: {total_rows}")
    print(f"  In_CHIP values: {in_chip_vals}")
    print(f"  ClinVar values: {sorted(clinvar_vals)}")
