# Check multiple files to see if In_CHIP is always True
import os

    ws = wb[wb.sheetnames[0]]
    for row in ws.iter_rows(min_row=3, max_row=ws.max_row, values_only=True):
        if row[29] is not None: chip_vals.add(str(row[29]))
        if row[13] is not None: ont_vals.add(str(row[13]))
print('In_CHIP values:', sorted(chip_vals))
print()
print('All ontologies:', sorted(ont_vals))
