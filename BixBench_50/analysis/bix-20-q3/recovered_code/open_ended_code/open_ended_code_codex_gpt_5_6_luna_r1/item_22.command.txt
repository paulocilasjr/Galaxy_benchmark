/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter
from openpyxl import load_workbook
base=Path('inputs');carriers=set()
wb=load_workbook(base/'230215_Trio_Status.xlsx',read_only=True,data_only=True);ws=wb.active
for row in ws.iter_rows(min_row=2,values_only=True):
 if row[0] is not None and str(row[3]).strip()=='Carrier': carriers.add(str(row[0]).strip())
def fid(p):
 t=p.stem.split('_CHIP_',1)[1]; return t.split('-',1)[0] if t.split('-',1)[0].isdigit() else t
c=Counter(); details=[]
for p in sorted((base/'CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
 if fid(p) not in carriers:continue
 wb=load_workbook(p,read_only=True,data_only=True);ws=wb.active;ws.reset_dimensions();it=ws.iter_rows(values_only=True);next(it);hdr=next(it);idx={v:i for i,v in enumerate(hdr) if v}
 for rn,row in enumerate(it,start=3):
  try:v=float(row[idx['Variant Allele Freq']])
  except:continue
  if v<.3:
   so=str(row[idx['Sequence Ontology (Combined)']] or '').lower()
   if row[idx['Zygosity']]"'!='"'Reference' and not any(x in so for x in ['intron','intergenic','utr']):
    cl=row[idx['Classification']]
    c[(row[idx['Zygosity']],cl)]+=1
    if cl not in ('Benign','Likely Benign') and cl not in (None,''):
     details.append((p.name,rn,row[idx['Chr:Pos']],row[idx['Gene Names']],so,v,cl))
print(c)
print('nonBLB classified',details)
PY"
