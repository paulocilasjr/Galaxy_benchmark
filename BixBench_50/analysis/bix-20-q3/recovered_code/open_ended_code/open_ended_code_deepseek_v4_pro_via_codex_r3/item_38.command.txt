/bin/bash -c "cd /workspace && python - <<'PY'
from pypdf import PdfReader
r=PdfReader('paper.pdf')
print('pages',len(r.pages))
text='\\n'.join((p.extract_text() or '') for p in r.pages)
open('paper.txt','w').write(text)
for term in ['VAF','non-reference','intronic','intergenic','UTR','ClinVar','Benign','Likely Benign','CHIP genes','VarSeq']:
    print('\\nTERM',term)
    import re
    for m in re.finditer(re.escape(term), text, re.I):
        s=max(0,m.start()-300); e=min(len(text),m.end()+500)
        print(text[s:e].replace('\\n',' '))
        break
PY"
