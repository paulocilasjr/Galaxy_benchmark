/bin/bash -lc "python - <<'PY'
from pathlib import Path
from collections import Counter, defaultdict
import openpyxl, re, math

# status
wb=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True,read_only=False)
ws=wb.active
rows=list(ws.iter_rows(min_row=2,values_only=True))
carrier=[]
for r in rows:
    sid=str(r[0]).strip() if r[0] is not None else ''
    if r[3]=='Carrier': carrier.append(sid)
wb.close()
print('carrier IDs:',carrier,'n=',len(carrier))
files=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))
byid={p.stem.split('_CHIP_',1)[1]:p for p in files}
print('n variant files',len(files))
print('carrier files:',[(x, bool(byid.get(x)), byid.get(x).name if byid.get(x) else None) for x in carrier])

# load all once and collect structures
all_counts=defaultdict(Counter)
carrier_counts=defaultdict(Counter)
carrier_rows=[]
for p in files:
    sid=p.stem.split('_CHIP_',1)[1]
    wb=openpyxl.load_workbook(p,data_only=True,read_only=False)
    ws=wb.active
    vals=list(ws.iter_rows(values_only=True))
    headers=list(vals[1])
    d={h:i for i,h in enumerate(headers) if h is not None}
    dat=vals[2:]
    c=carrier_counts if sid in carrier else all_counts
    c['rows'] += len(dat)
    c['nonempty_vaf'] += sum(r[d['Variant Allele Freq']] is not None for r in dat)
    c['inchip_true'] += sum(str(r[d['In_CHIP']]).strip().lower()=='true' for r in dat)
    c['zygosity_ref'] += sum(str(r[d['Zygosity']]).strip().lower()=='reference' for r in dat)
    c['clinvar_nonnull'] += sum(r[d['Classification']] not in (None,'') for r in dat)
    for key in ['Zygosity','Sequence Ontology (Combined)','Effect (Combined)','Classification','In_CHIP','Filter']:
        all_counts[key].update([r[d[key]] for r in dat])
        if sid in carrier: carrier_counts[key].update([r[d[key]] for r in dat])
    if sid in carrier:
        carrier_rows.extend((sid,p.name,r,d) for r in dat)
    wb.close()
print('\\nALL summaries',dict(all_counts))
print('\\nCARRIER summaries')
for k,v in carrier_counts.items(): print(k, v if isinstance(v,Counter) else v)
# print unique ontology and class top
for label,c in [('all',all_counts),('carrier',carrier_counts)]:
    print('\\n',label,'ontology',c['Sequence Ontology (Combined)'])
    print(label,'classification',c['Classification'])
    print(label,'zygosity',c['Zygosity'])
    print(label,'inchip',c['In_CHIP'])
    print(label,'filter',c['Filter'])
PY"
