from pathlib import Path
import openpyxl
for p in sorted(Path('inputs').glob('*.xlsx')):
    print('\\nFILE', p)

    print('SHEETS', wb.sheetnames)
    for ws in wb.worksheets:
        print(' SHEET', ws.title, 'DIM', ws.max_row, ws.max_column)
        for row in ws.iter_rows(min_row=1, max_row=min(6, ws.max_row), values_only=True):
            print('  ', row[:min(12, ws.max_column)])
    wb.close()

for ws in wb.worksheets:
    print('SHEET',ws.title,'DIM',ws.max_row,ws.max_column)
    for row in ws.iter_rows(min_row=1,max_row=min(10,ws.max_row),values_only=True):
