/bin/bash -lc "python - <<'PY'
# Independent check of the documented LBS formula from the provided Newick tree.
import zipfile, itertools, statistics
from Bio import Phylo
from io import StringIO
newick=zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip').read('996662at2759.faa.mafft.clipkit.treefile').decode().strip()
t=Phylo.read(StringIO(newick), 'newick')
tips=t.get_terminals()
labels=[x.name for x in tips]
pairs=[]
for a,b in itertools.combinations(tips,2):
    pairs.append(t.distance(a,b))
avg=sum(pairs)/len(pairs)
print('tree', newick)
print('avg_pairwise', avg)
vals=[]
for a in tips:
    mean=sum(t.distance(a,b) for b in tips if b is not a)/(len(tips)-1)
    score=mean/avg
    vals.append(score)
    print(a.name, mean, score)
print('median', statistics.median(vals))
PY"
