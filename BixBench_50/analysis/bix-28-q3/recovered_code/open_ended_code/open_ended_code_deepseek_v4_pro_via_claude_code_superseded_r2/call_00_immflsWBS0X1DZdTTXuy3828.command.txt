import dendropy
import math

tree_str = "(Fungi_Ncra:0.4916927411,(Fungi_Scer:0.4216513655,Fungi_Spom:0.4407010522):0.1387171356,Fungi_Umay:1.0828368619);"
tree = dendropy.Tree.get(data=tree_str, schema="newick")

taxa = sorted([t.label for t in tree.taxon_namespace])
print("Taxa:", taxa)

pdm = tree.phylogenetic_distance_matrix()

mean_per_taxon = {}
all_pairs = []
for t1 in taxa:
    dists = []
    for t2 in taxa:
        if t1 != t2:
            d = pdm(taxon1=t1, taxon2=t2)
            dists.append(d)
            all_pairs.append((d, t1, t2))
    mean_per_taxon[t1] = sum(dists) / len(dists)
    print(f"Mean dist {t1}: {mean_per_taxon[t1]:.10f}")

grand_mean = sum(d for d, _, _ in all_pairs) / len(all_pairs)
print(f"Grand mean: {grand_mean:.10f}")

# Struck 2014 LB scores = mean_i / grand_mean
lb_scores = {}
for t in taxa:
    lb = mean_per_taxon[t] / grand_mean
    lb_scores[t] = lb
    print(f"LB Struck {t}: {lb:.10f}")

print("\nAll pairwise patristic distances:")
for d, t1, t2 in all_pairs:
    print(f"  {t1} <-> {t2}: {d:.10f}")

# Compute correlation with phykit values
phykit_vals = {"Fungi_Ncra": -29.4826, "Fungi_Scer": -32.1541, "Fungi_Spom": -31.4275, "Fungi_Umay": -6.9357}
print("\nComparison:")
for t in taxa:
    print(f"  {t}: Struck={lb_scores[t]:.6f}, PhyKIT={phykit_vals[t]:.6f}, ratio={lb_scores[t]/phykit_vals[t]:.6f}")

# Now compute median of actual Struck 2014 scores

n = len(sorted_scores)

print(f"\nMedian LB score (Struck 2014): {median:.10f}")
