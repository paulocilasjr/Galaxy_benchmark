/bin/bash -lc "PYTHONPATH=/workspace/.phykit_runtime python - <<'PY'
from io import StringIO
from Bio import Phylo
from itertools import combinations
import statistics, zipfile
with zipfile.ZipFile('inputs/scogs_fungi.zip') as z:
    tree=Phylo.read(StringIO(z.read('996662at2759.faa.mafft.clipkit.treefile').decode()), 'newick')
names=[tip.name for tip in tree.get_terminals()]
pair_avg=statistics.mean(tree.distance(a,b) for a,b in combinations(names,2))
# PhyKIT compatibility semantics include self-distance 0 in each taxon's mean.
scores=[]
for a in names:
    taxon_avg=statistics.mean(tree.distance(a,b) for b in names)
    scores.append(taxon_avg/pair_avg*100-100)
print('taxa_count',len(names))
print('scores_unrounded',*[repr(x) for x in scores],sep='\\n')
print('median_unrounded',repr(statistics.median(scores)))
print('median_phykit_text',round(statistics.median(scores),4))
PY
