/bin/bash -lc "python - <<'PY'
from collections import Counter
vcf='inputs/SRR35233585_raw_variants.vcf'
counts=Counter(); gts=Counter(); multi=0; malformed=0
for line in open(vcf):
    if line.startswith('#'): continue
    chrom,pos,id_,ref,alts,qual,fil,info,fmt,sample=line.rstrip('\\n').split('\\t')[:10]
    gt=sample.split(':',1)[0]
    gts[gt]+=1
    alt_list=alts.split(',')
    if len(alt_list)>1: multi+=1
    # allele-level sample-specific count: only ALT alleles present in nonref genotype, once per ALT allele present
    allele_idxs=set()
    if gt not in ('.','./.','.|.'):
        for x in gt.replace('|','/').split('/'):
            if x=='.': continue
            try: i=int(x)
            except ValueError: malformed+=1; continue
            if i>0: allele_idxs.add(i)
    for i in allele_idxs:
        alt=alt_list[i-1]
        if len(ref)==1 and len(alt)==1 and ref in 'ACGT' and alt in 'ACGT':
            if (ref,alt) in {('A','G'),('G','A'),('C','T'),('T','C')}:
                counts[(fil,'ts')]+=1
            else:
                counts[(fil,'tv')]+=1
        else:
            counts[(fil,'non_snv')]+=1
print('genotypes')
for k,v in gts.most_common(): print(k,v)
print('multi',multi,'malformed',malformed)
print('counts')
for k in sorted(counts): print(k,counts[k])
for filtset,name in [({'.','PASS'}, 'dot_pass'), ({'LowQual'}, 'lowqual'), (None,'all')]:
    ts=tv=0
    for (f,t),v in counts.items():
        if filtset is None or f in filtset:
            if t=='ts': ts+=v
            elif t=='tv': tv+=v
    print(name, ts, tv, ts/tv if tv else None)
PY"
