/bin/bash -lc "python - <<'PY'
from collections import Counter
vcf='inputs/SRR35233585_raw_variants.vcf'
transition={('A','G'),('G','A'),('C','T'),('T','C')}
counts=Counter()
examples=[]
for line in open(vcf):
    if line.startswith('#'): continue
    fields=line.rstrip('\\n').split('\\t')
    ref=fields[3]; alts=fields[4].split(','); gt=fields[9].split(':',1)[0]
    allele_idxs=[]
    for x in gt.replace('|','/').split('/'):
        if x=='.': continue
        i=int(x)
        if i>0 and i not in allele_idxs:
            allele_idxs.append(i)
    snv_alleles=[]
    for i in allele_idxs:
        alt=alts[i-1]
        if len(ref)==len(alt)==1 and ref in 'ACGT' and alt in 'ACGT':
            snv_alleles.append((ref,alt))
    if len(alts)>1 and snv_alleles[:5]:
        examples.append((fields[1],ref,fields[4],gt,snv_alleles))
    for ref,alt in snv_alleles:
        counts['ts' if (ref,alt) in transition else 'tv']+=1
print(counts)
print('ratio_raw', counts['ts']/counts['tv'])
print('ratio_2dp', format(counts['ts']/counts['tv'],'.2f'))
print('multi_examples')
for e in examples[:10]: print(e)
PY"
