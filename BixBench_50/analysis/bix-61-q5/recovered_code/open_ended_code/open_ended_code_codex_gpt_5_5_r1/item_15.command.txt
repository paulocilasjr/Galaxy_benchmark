/bin/bash -lc "python - <<'PY'
from collections import Counter
tr={('A','G'),('G','A'),('C','T'),('T','C')}
for mode in ['all_alt_alleles','gt_unique_alt_alleles','hom_alt_sites_only','biallelic_snv_sites']:
    c=Counter()
    for line in open('inputs/SRR35233585_raw_variants.vcf'):
        if line.startswith('#'): continue
        f=line.rstrip('\\n').split('\\t')
        ref=f[3]; alts=f[4].split(','); gt=f[9].split(':',1)[0]
        if mode=='all_alt_alleles':
            idxs=range(1,len(alts)+1)
        elif mode=='gt_unique_alt_alleles':
            idxs=sorted({int(x) for x in gt.replace('|','/').split('/') if x"'!='"'.' and int(x)>0})
        elif mode=='hom_alt_sites_only':
            parts=gt.replace('|','/').split('/')
            if len(set(parts))==1 and parts[0] not in ('0','.'):
                idxs=[int(parts[0])]
            else:
                idxs=[]
        elif mode=='biallelic_snv_sites':
            if len(alts)==1:
                idxs=[1]
            else:
                idxs=[]
        for i in idxs:
            alt=alts[i-1]
            if len(ref)==1 and len(alt)==1 and ref in 'ACGT' and alt in 'ACGT':
                c['ts' if (ref,alt) in tr else 'tv']+=1
    print(mode, c['ts'], c['tv'], format(c['ts']/c['tv'],'.6f'), format(c['ts']/c['tv'],'.2f'))
PY"
