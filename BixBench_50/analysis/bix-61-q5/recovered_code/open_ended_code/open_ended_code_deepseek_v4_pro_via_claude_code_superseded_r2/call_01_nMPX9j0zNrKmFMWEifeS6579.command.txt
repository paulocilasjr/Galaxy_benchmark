# Let me also check what bcftools does differently by comparing
# Let me look at specific edge cases

# Check for multiallelic records with mix of SNP and indel
with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        ref = parts[3]
        alt = parts[4]
        
        alts = alt.split(',')
        if len(alts) > 1:
            # Check if it's a mix of types
            types = set()
            for a in alts:
                if len(ref) == 1 and len(a) == 1:
                    types.add('SNP')
                else:
                    types.add('indel')
            if len(types) > 1:
                print(f"POS={parts[1]} REF={ref} ALT={alt} - mixed types: {types}")
EOF
