# More precise investigation
# Let me compare my counts to bcftools by checking edge cases

# Count ALL records by type
snp_ref_single = 0
indel_records = 0  # records where ref is not single char
total_records = 0

with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    for line in f:
        if line.startswith('#'):
            continue
        total_records += 1
        parts = line.strip().split('\t')
        ref = parts[3]
        alt = parts[4]
        
        if len(ref) == 1:
            snp_ref_single += 1
        else:
            indel_records += 1

print(f"Total records: {total_records}")
print(f"Records with single-char REF: {snp_ref_single}")
print(f"Records with multi-char REF (indels/MNPs): {indel_records}")

# Now let me check: for records with single-char REF, how many have ONLY single-char ALTs?
# And how many have mixed?
snp_only = 0
mixed = 0
indel_only = 0
with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        ref = parts[3]
        alt = parts[4]
        if len(ref) != 1:
            continue
        
        alts = alt.split(',')
        snp_count = sum(1 for a in alts if len(a) == 1 and a != '*' and not any(c not in 'ACGT' for c in a))
        non_snp_count = len(alts) - snp_count
        
        if snp_count > 0 and non_snp_count == 0:
            snp_only += 1
        elif snp_count > 0 and non_snp_count > 0:
            mixed += 1
        elif snp_count == 0:
            indel_only += 1

print(f"\nRecords with single-char REF:")
print(f"  SNP-only ALTs: {snp_only}")
print(f"  Mixed SNP/indel ALTs: {mixed}")
print(f"  Indel-only ALTs: {indel_only}")

# Check for * alleles
star_count = 0
with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        alt = parts[4]
        if '*' in alt.split(','):
            star_count += 1
            if star_count <= 10:
                print(f"  * allele: POS={parts[1]} REF={parts[3]} ALT={alt}")

print(f"Records with * allele: {star_count}")
EOF
