python3 << 'EOF'
# Compute Ts/Tv from the VCF file
# Transitions: A<->G, C<->T
# Count only single-nucleotide substitutions (SNPs)
# Count each ALT allele separately

transitions = 0
transversions = 0
indel_count = 0
multiallelic = 0
other = 0

with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        if len(parts) < 5:
            continue
        ref = parts[3]
        alt = parts[4]
        
        # Skip non-SNP ref or alt
        if len(ref) != 1:
            # Ref is not a single base - could be MNP or indel
            # Check each alt allele separately
            for a in alt.split(','):
                if len(a) == 1 and len(ref) == 1:
                    # Actually ref is single, continue
                    pass
                else:
                    other += 1
            continue
        
        for a in alt.split(','):
            if len(a) != 1:
                indel_count += 1
                continue
            
            # Both ref and alt are single nucleotides - this is a SNP
            pair = ref + a
            if pair in ('AG', 'GA', 'CT', 'TC'):
                transitions += 1
            else:
                transversions += 1

print(f"Transitions: {transitions}")
print(f"Transversions: {transversions}")
if transversions > 0:
    print(f"Ts/Tv: {transitions/transversions:.6f}")
    print(f"Ts/Tv (rounded to 2 decimal places): {round(transitions/transversions, 2)}")
else:
    print("Ts/Tv: undefined (no transversions)")

print(f"Indels (skipped): {indel_count}")
print(f"Other (skipped): {other}")

# Also show site-level (1st ALT only) counts
ts_first = 0
tv_first = 0
with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        ref = parts[3]
        alt = parts[4].split(',')[0]  # First ALT only
        
        if len(ref) != 1 or len(alt) != 1:
            continue
        
        pair = ref + alt
        if pair in ('AG', 'GA', 'CT', 'TC'):
            ts_first += 1
        else:
            tv_first += 1

print(f"\nSite-level (1st ALT only):")
print(f"Transitions: {ts_first}")
print(f"Transversions: {tv_first}")
if tv_first > 0:
    print(f"Ts/Tv: {ts_first/tv_first:.6f}")
    print(f"Ts/Tv (rounded to 2 decimal places): {round(ts_first/tv_first, 2)}")
EOF
