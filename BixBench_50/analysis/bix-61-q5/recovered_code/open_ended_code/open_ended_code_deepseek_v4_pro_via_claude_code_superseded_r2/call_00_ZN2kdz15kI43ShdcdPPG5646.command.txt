bcftools query -f '%REF\t%ALT\n' /workspace/inputs/SRR35233585_raw_variants.vcf > /tmp/vcf_alleles.txt && python3 << 'EOF'
transitions = 0
transversions = 0
valid_bases = set('ACGT')

with open('/tmp/vcf_alleles.txt') as f:
    for line in f:
        line = line.strip()
        if not line:
            continue
        parts = line.split('\t')
        if len(parts) < 2:
            continue
        ref = parts[0]
        alt = parts[1].split(',')[0]  # first ALT only
        
        if len(ref) != 1 or ref not in valid_bases:
            continue
        if len(alt) != 1 or alt not in valid_bases or alt == '*':
            continue
        
        pair = ref + alt
        if pair in ('AG', 'GA', 'CT', 'TC'):
            transitions += 1
        else:
            transversions += 1

print(f"bcftools query, first ALT only:")
print(f"Transitions: {transitions}")
print(f"Transversions: {transversions}")
if transversions > 0:
    print(f"Ts/Tv: {transitions/transversions:.6f}")
    print(f"Ts/Tv (rounded to 2 decimal places): {round(transitions/transversions, 2)}")
EOF
