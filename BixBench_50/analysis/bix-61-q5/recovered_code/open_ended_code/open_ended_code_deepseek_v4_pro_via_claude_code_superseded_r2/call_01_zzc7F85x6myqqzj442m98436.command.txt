bcftools query -f '%REF %ALT\n' /workspace/inputs/SRR35233585_raw_variants.vcf 2>&1 | python3 << 'EOF'
import sys

transitions = 0
transversions = 0
valid_bases = set('ACGT')

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    parts = line.split()
    if len(parts) < 2:
        continue
    ref = parts[0]
    alt = parts[1]  # first ALT only
    
    if len(ref) != 1 or ref not in valid_bases:
        continue
    if len(alt) != 1 or alt not in valid_bases or alt == '*':
        continue
    
    pair = ref + alt
    if pair in ('AG', 'GA', 'CT', 'TC'):
        transitions += 1
    else:
        transversions += 1

print(f"bcftools query, first ALT only:")
print(f"Transitions: {transitions}")
print(f"Transversions: {transversions}")
if transversions > 0:
    print(f"Ts/Tv: {transitions/transversions:.6f}")
EOF
