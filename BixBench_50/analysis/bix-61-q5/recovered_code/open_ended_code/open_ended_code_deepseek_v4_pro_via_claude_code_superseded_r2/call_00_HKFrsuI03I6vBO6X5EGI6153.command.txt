# Investigate records where ref or alt is not single-nucleotide
with open('/workspace/inputs/SRR35233585_raw_variants.vcf') as f:
    count = 0
    for line in f:
        if line.startswith('#'):
            continue
        parts = line.strip().split('\t')
        ref = parts[3]
        alt = parts[4]
        
        if len(ref) != 1:
            if count < 20:
                print(f"REF={ref!r}, ALT={alt!r}, POS={parts[1]}")
            count += 1

print(f"\nTotal records with non-single-char REF: {count}")
EOF
