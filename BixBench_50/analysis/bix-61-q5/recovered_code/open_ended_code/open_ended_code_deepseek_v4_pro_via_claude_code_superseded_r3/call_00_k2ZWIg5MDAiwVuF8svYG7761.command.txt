# Check * allele records
vcf_path = "/workspace/inputs/SRR35233585_raw_variants.vcf"
f = open(vcf_path, 'r')

star_records = []
for line in f:
    if line.startswith('#'):
        continue
    fields = line.strip().split('\t')
    ref = fields[3]
    alt_field = fields[4]
    if len(ref) == 1 and '*' in alt_field.split(','):
        star_records.append((fields[0], fields[1], ref, alt_field))

f.close()
print(f"Records with * allele at SNV site: {len(star_records)}")
for r in star_records:
    print(f"  {r[0]}:{r[1]} REF={r[2]} ALT={r[3]}")
