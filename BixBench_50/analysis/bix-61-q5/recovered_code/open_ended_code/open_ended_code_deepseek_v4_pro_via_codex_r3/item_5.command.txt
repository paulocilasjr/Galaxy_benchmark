/bin/bash -c "python3 - <<'PY'
from collections import Counter
p='inputs/SRR35233585_raw_variants.vcf'
c=Counter(); filt_type=Counter(); pass_low=Counter(); pass_ratio=Counter()
# count all SNVs (single-char ref and all single-char alts? use bcftools-like row SNP)
for line in open(p):
 if line.startswith('#'): continue
 f=line.rstrip('\\n').split('\\t'); filt=f[6]; ref=f[3]; alts=f[4].split(',')
 c[filt]+=1
 # SNP row if ref len 1 and at least one alt len1, and no weird?
 has_snp=len(ref)==1 and any(len(a)==1 for a in alts)
 if has_snp:
  filt_type[filt]+=1
  # first alt bcftools 1st ALT
  a=alts[0]
  if len(a)==1:
   pair={ref,a}
   key='ts' if pair in ({'A','G'},{'C','T'}) else 'tv'
   pass_ratio[(filt,key)]+=1
print(c)
print(filt_type)
print(pass_ratio)
for filt in ['.','LowQual']:
 ts=pass_ratio[(filt,'ts')]; tv=pass_ratio[(filt,'tv')]
 print(filt,'SNV first alt',ts,tv,'ratio',ts/tv if tv else None)
PY"
