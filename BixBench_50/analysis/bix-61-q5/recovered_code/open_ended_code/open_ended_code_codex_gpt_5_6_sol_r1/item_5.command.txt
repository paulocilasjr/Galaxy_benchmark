/bin/bash -lc "awk -F '\\t' '
BEGIN { OFS=\"\\t\" }
/"'^#/ { next }
{
  nrec++
  if ($7=="PASS") pass++; else if ($7==".") dot++; else filt++
  split($5,a,","); split($10,f,":"); split(f[1],g,"[/|]")
  delete used; for (j in g) if (g[j] ~ /''^[1-9][0-9]*$/) used[g[j]]=1
  if (length($4)==1) {
    snvsite=0
    for (i=1;i<=length(a);i++) if (length(a[i])==1 && $4 ~ /''^[ACGT]$/ && a[i] ~ /''^[ACGT]$/) {
      snvsite=1; pair=$4 a[i]
      if (pair ~ /''^(AG|GA|CT|TC)$/) { ats++; if (i in used) gts++ } else { atv++; if (i in used) gtv++ }
    }
    if (snvsite) ssites++
    if (length(a)==1 && snvsite) { if ($4 a[1] ~ /''^(AG|GA|CT|TC)$/) bts++; else btv++ }
  }
}
END { print "records",nrec,"dot",dot,"PASS",pass,"filtered",filt; print "SNV_sites",ssites,"allele_TS",ats,"allele_TV",atv,"ratio",ats/atv; print "GT_carried_TS",gts,"GT_carried_TV",gtv,"ratio",gts/gtv; print "biallelic_TS",bts,"biallelic_TV",btv,"ratio",bts/btv }
'"' inputs/SRR35233585_raw_variants.vcf
bcftools query -f '%GT\\n' inputs/SRR35233585_raw_variants.vcf | sort | uniq -c
bcftools query -f '%FILTER\\n' inputs/SRR35233585_raw_variants.vcf | sort | uniq -c"
