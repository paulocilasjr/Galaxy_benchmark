/bin/bash -lc "python3 - <<'PY'
from collections import Counter
vcf='inputs/SRR35233585_raw_variants.vcf'
ts_pairs={('A','G'),('G','A'),('C','T'),('T','C')}
for mode in ['all_nonref','hom_alt_only','site_any_alt']:
    c=Counter()
    with open(vcf) as f:
        for line in f:
            if line.startswith('#'): continue
            fields=line.rstrip('\\n').split('\\t')
            ref, alts, sample = fields[3], fields[4].split(','), fields[9]
            gt=sample.split(':',1)[0].replace('|','/')
            if gt in {'.','./.'}: continue
            alleles=gt.split('/')
            if mode=='hom_alt_only' and not (len(set(alleles))==1 and alleles[0] not in {'.','0'}):
                continue
            if mode=='site_any_alt':
                idxs=[]
                for x in alleles:
                    if x not in {'.','0'}:
                        idxs=[int(x)]
                        break
            else:
                idxs=sorted({int(x) for x in alleles if x not in {'.','0'}})
            for idx in idxs:
                alt=alts[idx-1]
                if len(ref)==1 and len(alt)==1 and ref in 'ACGT' and alt in 'ACGT':
                    if (ref,alt) in ts_pairs: c['ts']+=1
                    else: c['tv']+=1
    print(mode, c['ts'], c['tv'], c['ts']/c['tv'])
PY"
