/bin/bash -lc "python3 - <<'PY'
from collections import Counter
vcf='inputs/SRR35233585_raw_variants.vcf'
ts_pairs={('A','G'),('G','A'),('C','T'),('T','C')}
counts=Counter()
gts=Counter()
examples=[]
with open(vcf) as f:
    for line in f:
        if line.startswith('#'): continue
        chrom,pos,vid,ref,alt,qual,fil,info,fmt,sample=line.rstrip('\\n').split('\\t')[:10]
        gt=sample.split(':',1)[0]
        gts[gt]+=1
        if gt in {'.','./.','.|.'}: continue
        alleles=[]
        for sep in ('/','|'):
            if sep in gt:
                alleles=gt.replace('|','/').split('/')
                break
        if not alleles: alleles=[gt]
        present=sorted({int(x) for x in alleles if x not in {'.','0'}})
        alts=alt.split(',')
        for idx in present:
            if idx<1 or idx>len(alts):
                continue
            a=alts[idx-1]
            if len(ref)==1 and len(a)==1 and ref in 'ACGT' and a in 'ACGT':
                counts['snv_alt_alleles']+=1
                if (ref,a) in ts_pairs: counts['ts']+=1
                else: counts['tv']+=1
            else:
                counts['non_snv_alt_alleles']+=1
                if len(examples)<5: examples.append((pos,ref,a,gt))
print('counts', counts)
print('ratio', counts['ts']/counts['tv'])
print('top_gts', gts.most_common(10))
print('non_snv_examples', examples)
PY"
