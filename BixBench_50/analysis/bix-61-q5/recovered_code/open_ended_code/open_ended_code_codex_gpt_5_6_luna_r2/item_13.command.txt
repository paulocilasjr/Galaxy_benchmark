/bin/bash -lc "python3 - <<'PY'
from collections import Counter
p='inputs/SRR35233585_raw_variants.vcf'
trans={frozenset(x) for x in ('AG','CT')}
for mode in ('all','pass'):
    c=Counter(); rec=0; selected=0; alleles=0; genotypes=Counter(); multi=0
    with open(p) as fh:
        for line in fh:
            if line.startswith('#'): continue
            f=line.rstrip('\\n').split('\\t'); rec+=1
            if mode=='pass' and f[6] not in ('.','PASS'): continue
            gt=f[9].split(':')[0]; genotypes[gt]+=1
            for ai in {int(x) for x in gt.replace('|','/').split('/') if x not in ('.','') and x"'!='"'0'}:
                alts=f[4].split(',')
                if ai>len(alts): raise ValueError((f[0],f[1],gt))
                ref,alt=f[3].upper(),alts[ai-1].upper()
                if len(ref)==len(alt)==1 and ref in 'ACGT' and alt in 'ACGT' and ref"'!=alt:
                    alleles += 1; selected += 1
                    c['"'Ts' if frozenset((ref,alt)) in trans else 'Tv'] += 1
            if ',' in f[4]: multi+=1
    print(mode, 'records',rec,'selected_snv_alt_alleles',selected,'Ts',c['Ts'],'Tv',c['Tv'],'ratio',c['Ts']/c['Tv'] if c['Tv'] else None,'multi',multi,'GTs',genotypes)
PY"
