/bin/bash -lc "bcftools stats -s SRR35233585 inputs/SRR35233585_raw_variants.vcf | rg '"'^(PSC|TSTV|SN)'"'
bcftools query -f '%POS\\t%REF\\t%ALT\\t[%GT]\\n' inputs/SRR35233585_raw_variants.vcf | awk 'BEGIN{FS=\"\\t\"} "'$3 ~ /,/ {n=split($3,a,","); s=0; ind=0; for(i=1;i<=n;i++){if(length($2)==1 && length(a[i])==1)s++; else ind++} c[s "/" ind]++} END{for(k in c) print k,c[k]}'"' | sort
bcftools norm -f inputs/GCF_000005845.2_ASM584v2_genomic.fna -m -any inputs/SRR35233585_raw_variants.vcf 2>/dev/null | bcftools stats - | rg '"'^(SN|TSTV)'"'"
