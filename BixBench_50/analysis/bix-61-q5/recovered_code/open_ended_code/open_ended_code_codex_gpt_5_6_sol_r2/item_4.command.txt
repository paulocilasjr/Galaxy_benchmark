/bin/bash -lc "bcftools stats inputs/SRR35233585_raw_variants.vcf | rg '"'^(SN|TSTV)'"' 
bcftools query -f '%FILTER\\t%REF\\t%ALT[\\t%GT]\\n' inputs/SRR35233585_raw_variants.vcf | awk 'BEGIN{FS=\"\\t\"} {f["'$1]++; gt[$4]++; n++; if(length($2)==1 && $3 !~ /,|'"\\*|"'^</ && length($3)==1) snv++} END{print "records",n,"simple_biallelic_snv",snv; for(x in f) print "FILTER",x,f[x]; for(x in gt) print "GT",x,gt[x]}'"' | sort
bcftools view -H -v snps -m2 -M2 inputs/SRR35233585_raw_variants.vcf | head -n 8"
