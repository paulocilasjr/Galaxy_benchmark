/bin/bash -c "python - <<'PY'
vcf='inputs/SRR35233585_raw_variants.vcf'
trans=set([('A','G'),('G','A'),('C','T'),('T','C')])
allele_ts=allele_tv=first_ts=first_tv=bi_ts=bi_tv=0
records=snps=indels=0
for line in open(vcf):
    if line.startswith('#'): continue
    records+=1
    f=line.rstrip('\\n').split('\\t')
    ref=f[3]; alts=f[4].split(',')
    def isbase(x): return len(x)==1 and x in 'ACGT'
    any_snp=isbase(ref) and any(isbase(a) for a in alts)
    any_indel=(not isbase(ref)) or any(not isbase(a) for a in alts)
    if any_snp: snps+=1
    if any_indel: indels+=1
    for a in alts:
        if isbase(ref) and isbase(a) and ref"'!=a:
            if (ref,a) in trans: allele_ts+=1
            else: allele_tv+=1
    a=alts[0]
    if isbase(ref) and isbase(a) and ref!=a:
        if (ref,a) in trans: first_ts+=1
        else: first_tv+=1
    if '"',' not in f[4] and isbase(ref) and isbase(a) and ref"'!=a:
        if (ref,a) in trans: bi_ts+=1
        else: bi_tv+=1
print('"'records',records,'snpsites',snps,'indels',indels)
print('allele',allele_ts,allele_tv,allele_ts/allele_tv,round(allele_ts/allele_tv,2))
print('first',first_ts,first_tv,first_ts/first_tv,round(first_ts/first_tv,2))
print('biallelic',bi_ts,bi_tv,bi_ts/bi_tv,round(bi_ts/bi_tv,2))
PY"
