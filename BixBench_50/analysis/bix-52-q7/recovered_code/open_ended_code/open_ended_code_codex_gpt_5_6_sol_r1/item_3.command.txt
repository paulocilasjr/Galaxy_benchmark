/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
p='inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
with open(p, newline='') as f:
    rows=list(csv.DictReader(f))
vals=[]
missing=0
for row in rows:
    s=row['MethylationPercentage'].strip()
    if not s:
        missing += 1
    else:
        vals.append(float(s))
kept=sum(v>90 or v<10 for v in vals)
removed_numeric=sum(not (v>90 or v<10) for v in vals)
print({'rows':len(rows),'numeric':len(vals),'missing':missing,'kept':kept,'removed_numeric':removed_numeric,'removed_total_if_missing_fail':removed_numeric+missing})
print('boundaries', Counter(v for v in vals if v in (10.0,90.0)))
PY
