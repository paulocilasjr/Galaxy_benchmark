import csv
import os

os.chdir('/workspace')

filepath = './inputs/ZF_AgeRelatedCpG_noMT_Final.csv'
print(f"Exists: {os.path.exists(filepath)}")

try:
    with open(filepath, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        print(f"Header: {header}")

        total = 0
        kept = 0
        for row in reader:
            total += 1
            try:
                meth_pct = float(row[7])
                if meth_pct > 90 or meth_pct < 10:
                    kept += 1
            except (ValueError, IndexError):
                pass

        removed = total - kept
        print(f"Total rows: {total}")
        print(f"Kept rows (meth > 90% or < 10%): {kept}")
        print(f"Removed rows (10% <= meth <= 90%): {removed}")
except Exception as e:
    print(f"Error: {e}")
