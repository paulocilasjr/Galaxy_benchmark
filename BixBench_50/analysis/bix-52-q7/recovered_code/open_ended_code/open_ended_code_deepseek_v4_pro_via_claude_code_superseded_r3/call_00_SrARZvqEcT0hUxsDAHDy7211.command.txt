import csv, sys
with open(sys.argv[1]) as f:
    reader = csv.reader(f)
    header = next(reader)
    print('Header:', header)
    total = 0
    kept = 0
    for row in reader:
        total += 1
        try:
            meth = float(row[7])
            if meth > 90 or meth < 10:
                kept += 1
        except:
            pass
    removed = total - kept
    print(f'Total data rows: {total}')
    print(f'Kept: {kept}')
    print(f'Removed: {removed}')
" {} \;
