/bin/bash -lc "python3 - <<'PY'
import csv
from pathlib import Path
p=Path('inputs/ZF_AgeRelated_CpG_noMT_Final.csv')
total=kept=removed=missing=0
exact10=exact90=0
with p.open(newline='') as f:
    r=csv.DictReader(f)
    for row in r:
        total += 1
        s=row['MethylationPercentage']
        if s == '' or s is None:
            missing += 1
            removed += 1
            continue
        v=float(s)
        if v == 10: exact10 += 1
        if v == 90: exact90 += 1
        if v > 90 or v < 10:
            kept += 1
        else:
            removed += 1
print('total', total)
print('kept', kept)
print('removed', removed)
print('missing', missing)
print('exact10', exact10)
print('exact90', exact90)
PY"
