/bin/bash -lc "python3 - <<'PY'
import csv
from decimal import Decimal
p = 'inputs/ZF_AgeRelated_CpG_noMT_Final.csv'
total = retained = removed = 0
boundary = {'10':0,'90':0}
with open(p, newline='') as f:
    r = csv.DictReader(f)
    for row in r:
        total += 1
        m = Decimal(row['MethylationPercentage'])
        if m > Decimal('90') or m < Decimal('10'):
            retained += 1
        else:
            removed += 1
        if m == Decimal('10'):
            boundary['10'] += 1
        if m == Decimal('90'):
            boundary['90'] += 1
print('total', total)
print('retained', retained)
print('removed', removed)
print('boundary', boundary)
PY"
