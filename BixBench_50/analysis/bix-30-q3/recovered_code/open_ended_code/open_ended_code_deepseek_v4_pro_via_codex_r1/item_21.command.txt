/bin/bash -c "python - <<'PY'
import xlrd, numpy as np
from scipy import stats
from statsmodels.stats.multitest import multipletests
p='/workspace/inputs/pone.0150501.s002.xls'
sh=xlrd.open_workbook(p).sheet_by_index(0)
headers=sh.row_values(1)[1:]
A=np.array([[float(v) if isinstance(v,(int,float)) else np.nan for v in sh.row_values(r)[1:]] for r in range(2,sh.nrows)],float)
names=[sh.row_values(r)[0] for r in range(2,sh.nrows)]
P=[i for i,h in enumerate(headers) if h.startswith('P_') and h"'!='"'P_3']
C=[i for i,h in enumerate(headers) if h.startswith('C_') and h"'!='"'C_18']
means=np.nanmean(A,axis=0) # includes all samples or use included only?
# try included only mean
B=means[None,:]-A
pvals=[]
for k in range(A.shape[0]):
 x=B[k,P]; y=B[k,C]; x=x[~np.isnan(x)]; y=y[~np.isnan(y)]
 pvals.append(stats.ttest_ind(x,y,equal_var=True).pvalue)
pv=np.array(pvals)
for method in ['bonferroni','fdr_by']:
 res=multipletests(pv,alpha=.05,method=method)
 print(method, np.sum(res[0]), list(np.array(names)[res[0]]))
 if method=='bonferroni':
  print([(n,p,a) for n,p,a in zip(names,pv,res[1]) if res[0]])
PY"
