import xlrd, numpy as np
from scipy import stats
from statsmodels.stats.multitest import multipletests
p='/workspace/inputs/pone.0150501.s002.xls'
sh=xlrd.open_workbook(p).sheet_by_index(0)
headers=sh.row_values(1)[1:]
mat=[]; names=[]
for r in range(2,sh.nrows):
    vals=sh.row_values(r); names.append(vals[0]); nums=[]
    for v in vals[1:]:
        nums.append(float(v) if isinstance(v,(int,float)) else np.nan)
    mat.append(nums)
A=np.array(mat,dtype=float)
P=[i for i,h in enumerate(headers) if h.startswith('P_')]
C=[i for i,h in enumerate(headers) if h.startswith('C_')]
exP={'P_3'}; exC={'C_18'}

def analysis(center_mode='allmean', exclude=False, threshold=None, sign='sample-gene', equal_var=True):
    B=A.copy()
    if threshold is not None:
        B[B>=threshold]=np.nan
    pidx=[i for i in P if not exclude or headers[i] not in exP]
    cidx=[i for i in C if not exclude or headers[i] not in exC]
    if center_mode=='allmean':
        sample_means=np.nanmean(B, axis=0)
    elif center_mode=='detected_mean':
        # mean of detected (<threshold) raw Ct per sample (when threshold provided)
        sample_means=np.nanmean(np.where(A<threshold if threshold else A, A, np.nan), axis=0)
    else:
        sample_means=np.zeros(B.shape[1])
    if center_mode"'!='"'none':
        B=sample_means[None,:]-B if sign=='sample-gene' else B-sample_means[None,:]
    pvals=[]; details=[]
    for k,name in enumerate(names):
        x=B[k,pidx]; y=B[k,cidx]
        x=x[~np.isnan(x)]; y=y[~np.isnan(y)]
        if len(x)<2 or len(y)<2:
            pvals.append(np.nan); details.append((name,np.nan,len(x),len(y)))
        else:
            t,pv=stats.ttest_ind(x,y,equal_var=equal_var); pvals.append(pv); details.append((name,pv,len(x),len(y)))
    pvals=np.array(pvals)
    valid=~np.isnan(pvals)
    pv=pvals[valid]; nm=np.array(names)[valid]
    res=multipletests(pv,alpha=0.05,method='bonferroni')
    by=multipletests(pv,alpha=0.05,method='fdr_by')
    raw_sig=np.sum(pv<=0.05)
    bonf_sig=int(np.sum(res[0])); by_sig=int(np.sum(by[0]))
    print('center',center_mode,'sign',sign,'thr',threshold,'excl',exclude,'valid',valid.sum(),'raw',raw_sig,'bonf',bonf_sig,'BY',by_sig)
    if bonf_sig or by_sig:
        print('  bonf_names', list(nm[res[0]]))
        print('  by_names', list(nm[by[0]]))
    return pvals

for cm in ['none','allmean','detected_mean']:

      for thr in [None,34]:
