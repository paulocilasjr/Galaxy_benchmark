/bin/bash -c "python - <<'PY'
import xlrd, numpy as np, csv
from scipy import stats
p='/workspace/inputs/pone.0150501.s002.xls'
sh=xlrd.open_workbook(p).sheet_by_index(0)
headers=sh.row_values(1)[1:]
A=np.array([[float(v) if isinstance(v,(int,float)) else np.nan for v in sh.row_values(r)[1:]] for r in range(2,sh.nrows)],float)
names=[sh.row_values(r)[0] for r in range(2,sh.nrows)]
P=[i for i,h in enumerate(headers) if h.startswith('P_')]
C=[i for i,h in enumerate(headers) if h.startswith('C_')]
means=np.nanmean(A,axis=0)
B=means[None,:]-A
rows=[]
for k,n in enumerate(names):
 x=B[k,P]; y=B[k,C]; x=x[~np.isnan(x)]; y=y[~np.isnan(y)]
 pv=stats.ttest_ind(x,y,equal_var=True).pvalue
 rows.append((n,pv))
with open('/tmp/pvals.csv','w',newline='') as f:
 w=csv.writer(f); w.writerow(['miRNA','pvalue'])
 for r in rows: w.writerow(r)
print('wrote')
PY
Rscript - <<'RS'
d<-read.csv('/tmp/pvals.csv')
p<-d"'$pvalue
bonf<-p.adjust(p,method='"'bonferroni')
by<-p.adjust(p,method='BY')
cat('R bonf sig',sum(bonf<=0.05),'BY sig',sum(by<=0.05),'\\n')
print(d[bonf<=0.05,])
print(d[by<=0.05,])
cat('min p',min(p),'bonf min',min(bonf),'BY min',min(by),'\\n')
RS"
