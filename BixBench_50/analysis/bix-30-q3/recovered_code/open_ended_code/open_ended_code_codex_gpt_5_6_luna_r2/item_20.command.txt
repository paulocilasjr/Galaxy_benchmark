/bin/bash -lc "python - <<'PY'
import xlrd, subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
lines=['\\t'.join(str(x) for x in sh.row_values(1))]
for i in range(2,sh.nrows): lines.append('\\t'.join(x if isinstance(x,str) else repr(x) for x in sh.row_values(i)))
code=r'''
d<-read.delim(file(\"stdin\"),header=T,sep=\"\\t\",check.names=F,na.strings=c(\"NAN\",\"\"),quote=\"\",comment.char=\"\")
P<-d[,grep(\""'^P_",names(d)),drop=F];C<-d[,grep("''^C_",names(d)),drop=F]
p<-vapply(1:nrow(d),function(i)t.test(as.numeric(P[i,]),as.numeric(C[i,]),var.equal=T)$p.value,numeric(1))
cat("raw",sum(p<=.05),"BH",sum(p.adjust(p,"BH")<=.05),"BY",sum(p.adjust(p,"BY")<=.05),"bonf",sum(p.adjust(p,"bonferroni")<=.05),"'"\\n\")
'''
res=subprocess.run(['Rscript','--vanilla','-e',code],input='\\n'.join(lines).encode(),stdout=subprocess.PIPE,stderr=subprocess.PIPE)
print(res.stdout.decode(),res.stderr.decode())
PY"
