/bin/bash -lc "python - <<'PY'
import xlrd, subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
lines=['\\t'.join(str(x) for x in sh.row_values(1))]
for i in range(2,sh.nrows): lines.append('\\t'.join(x if isinstance(x,str) else repr(x) for x in sh.row_values(i)))
r_code=r'''
d <- read.delim(file(\"stdin\"), header=T, sep=\"\\t\", check.names=F, na.strings=c(\"NAN\",\"\"), quote=\"\", comment.char=\"\")
P <- d[,grep(\""'^P_",names(d)),drop=F]; C <- d[,grep("''^C_",names(d)),drop=F]
for (scale in c("Ct","2''^-Ct","-Ct")) {
 raw <- vapply(1:nrow(d),function(i) {x<-as.numeric(P[i,]);y<-as.numeric(C[i,]); if(scale=="2''^-Ct"){x<-2''^-x;y<-2''^-y}; if(scale=="-Ct"){x<- -x;y<- -y}; t.test(x,y,var.equal=TRUE)$p.value},numeric(1))
 b<-p.adjust(raw,"bonferroni"); by<-p.adjust(raw,"BY"); o<-order(raw)
 cat(scale,"min",raw[o[1]],"Bonf",sum(b<=.05,na.rm=T),"BY",sum(by<=.05,na.rm=T),"top",d[o[1],1],"'"\\n\")
}
'''
res=subprocess.run(['Rscript','--vanilla','-e',r_code],input='\\n'.join(lines).encode(),stdout=subprocess.PIPE,stderr=subprocess.PIPE)
print(res.stdout.decode());print(res.stderr.decode())
PY"
