/bin/bash -lc "python - <<'PY'
import xlrd,numpy as np
from scipy.stats import ttest_ind
s=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
names=[s.cell_value(i,0) for i in range(2,s.nrows)]
X=np.array([[float(s.cell_value(i,j)) if isinstance(s.cell_value(i,j),(int,float)) else np.nan for j in range(1,21)] for i in range(2,s.nrows)])
for cutoff in [34,35,40,None]:
 Y=X.copy()
 if cutoff: Y[Y>=cutoff]=np.nan
 means=np.nanmean(Y,axis=0)
 Z=means-Y
 for drop in [True,False]:
  P=[j for j in range(10) if not(drop and j==2)]; C=[j for j in range(10,20) if not(drop and j==17)]
  p=np.array([ttest_ind(r[P],r[C],nan_policy='omit',equal_var=True).pvalue for r in Z])
  print('cut',cutoff,'drop',drop,'nominal',np.sum(p<=.05),'bonf',np.sum(p*len(p)<=.05),'min',names[np.nanargmin(p)],np.nanmin(p))
PY"
