import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_excel('inputs/pone.0150501.s002.xls')

patient_cols_data = [0, 1, 3, 4, 5, 6, 7, 8, 9]
control_cols_data = [10, 11, 12, 13, 14, 15, 16, 18, 19]

data = df.iloc[1:, 1:].apply(pd.to_numeric, errors='coerce')
mir_names = df.iloc[1:, 0].values

# Step 1: Per-sample mean of detected miRNAs (Ct < 34)
all_data_cols = patient_cols_data + control_cols_data
sample_means = {}
for c in all_data_cols:
    vals = data.iloc[:, c].dropna().values.astype(float)
    detected = vals[vals < 34]
    sample_means[c] = np.mean(detected)

# Step 2: ΔCt t-test
p_values = []
analysis_results = []

for i in range(data.shape[0]):
    if pd.isna(mir_names[i]):
        continue
    mir = mir_names[i]
    
    p_delta = [float(data.iloc[i, c]) - sample_means[c] for c in patient_cols_data if pd.notna(data.iloc[i, c])]
    c_delta = [float(data.iloc[i, c]) - sample_means[c] for c in control_cols_data if pd.notna(data.iloc[i, c])]
    
    if len(p_delta) < 2 or len(c_delta) < 2:
        continue
    
    t_stat, p_val = stats.ttest_ind(np.array(p_delta), np.array(c_delta), equal_var=True)
    p_values.append(p_val)
    analysis_results.append((mir, p_val, t_stat))

p_values = np.array(p_values)
n_tests = len(p_values)

# Verify miR-21
for mir, pv, ts in analysis_results:
    if 'mir-21' == mir.lower().replace('hsa-', ''):
        print(f'{mir}: raw p = {pv:.10f}')
        print(f'Bonferroni: {pv * n_tests:.10f}')
        harmonic = np.sum(1.0 / np.arange(1, n_tests + 1))
        # BY for most significant test
        by_val = pv * n_tests * harmonic / 1.0
        print(f'Harmonic number: {harmonic:.6f}')
        print(f'BY (rank 1): {by_val:.10f}')
        break

# Bonferroni
bonf_p = np.minimum(p_values * n_tests, 1.0)
bonf_sig = int(np.sum(bonf_p <= 0.05))

# BY
sorted_idx = np.argsort(p_values)
sorted_p = p_values[sorted_idx]
harmonic = np.sum(1.0 / np.arange(1, n_tests + 1))

by_p = np.ones(n_tests)
for i in range(n_tests):
    rank = i + 1
    by_p[i] = min(sorted_p[i] * n_tests * harmonic / rank, 1.0)

for i in range(n_tests - 2, -1, -1):
    by_p[i] = min(by_p[i], by_p[i + 1])

by_sig = int(np.sum(by_p <= 0.05))

print(f'\nBonferroni significant: {bonf_sig}')
print(f'BY significant: {by_sig}')
print(f'Ratio: {bonf_sig}:{by_sig}')

# Also show the smallest BY-adjusted p-values
print(f'\nSmallest BY adj p-values (top 10):')
for i in range(min(10, n_tests)):
    orig_idx = sorted_idx[i]
    mir = analysis_results[orig_idx][0]
    print(f'  {mir}: raw p={sorted_p[i]:.6f}, BY adj={by_p[i]:.6f}')
