import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_excel('inputs/pone.0150501.s002.xls')

data = df.iloc[1:, 1:].apply(pd.to_numeric, errors='coerce')
mir_names = df.iloc[1:, 0].values

# Find miR-21
for i in range(data.shape[0]):
    if 'mir-21' in str(mir_names[i]).lower() and 'mir-21' == str(mir_names[i]).lower().replace('hsa-', '').replace('*', '').split('*')[0]:
        print(f'Index {i}: {mir_names[i]}')
        p_vals = data.iloc[i, 0:10].dropna().values.astype(float)
        c_vals = data.iloc[i, 10:20].dropna().values.astype(float)
        print(f'  Patient Ct: {p_vals}')
        print(f'  Control Ct: {c_vals}')
        print(f'  Mean patient: {np.mean(p_vals):.3f}')
        print(f'  Mean control: {np.mean(c_vals):.3f}')
        t_stat, p_val = stats.ttest_ind(p_vals, c_vals, equal_var=True)
        print(f'  Student t-test p={p_val:.6f}')
        t_stat_w, p_val_w = stats.ttest_ind(p_vals, c_vals, equal_var=False)
        print(f'  Welch t-test p={p_val_w:.6f}')

# Also find mir-21-5p and any variant
for i in range(data.shape[0]):
    if 'mir-21' in str(mir_names[i]).lower():
        print(f'  All miR-21: Index {i}: {mir_names[i]}')
