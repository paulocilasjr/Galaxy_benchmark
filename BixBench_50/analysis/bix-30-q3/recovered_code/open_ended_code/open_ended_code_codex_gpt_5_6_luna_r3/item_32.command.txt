/bin/bash -lc "python - <<'PY'
import xlrd,csv,io,subprocess
sh=xlrd.open_workbook('inputs/pone.0150501.s002.xls').sheet_by_index(0)
out=io.StringIO();w=csv.writer(out,delimiter='\\t',lineterminator='\\n')
w.writerow(['miRNA']+[sh.cell_value(1,c) for c in range(1,sh.ncols)])
for r in range(2,sh.nrows):
 w.writerow([sh.cell_value(r,0)]+['NA' if isinstance(sh.cell_value(r,c),str) else sh.cell_value(r,c) for c in range(1,sh.ncols)])
rscript=r'''
d<-read.delim(file(\"stdin\"),check.names=FALSE,na.strings=\"NA\"); P<-grep(\""'^P_",names(d)); C<-grep("''^C_",names(d));
for(drop in c(FALSE,TRUE)) {pp<-P;cc<-C;if(drop){pp<-pp[names(d)[pp]!="P_3"];cc<-cc[names(d)[cc]!="C_18"]};X<-as.matrix(d[c(pp,cc)]);means<-apply(X,2,function(z){z<-as.numeric(z);mean(z[is.finite(z)&z<34])});E<-sweep(X,2,means,"-")*(-1);p<-apply(E,1,function(z){x<-z[seq_along(pp)];y<-z[length(pp)+seq_along(cc)];ok<-is.finite(x)&is.finite(y);if(sum(ok)<2)return(NA);t.test(x[ok],y[ok],paired=TRUE)$p.value});i<-which(d$miRNA=="hsa-miR-21");cat('"'drop',drop,'raw',sum(p<=.05,na.rm=T),'bonf',sum(p.adjust(p,'bonferroni')<=.05,na.rm=T),'BY',sum(p.adjust(p,'BY')<=.05,na.rm=T),'p21',p[i],'min',min(p,na.rm=T),'\\\\n')}
'''
res=subprocess.run(['Rscript','-e',rscript],input=out.getvalue(),text=True,capture_output=True)
print(res.stdout);print(res.stderr)
PY"
