/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
# Read model metadata for common IDs without loading numeric matrices again.
with open('inputs/CRISPRGeneEffect.csv', newline='') as f:
    r=csv.reader(f); next(r); dep_ids={row[0] for row in r}
with open('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', newline='') as f:
    r=csv.reader(f); next(r); expr_ids={row[0] for row in r}
common=dep_ids & expr_ids
with open('inputs/Model.csv', newline='') as f:
    r=csv.DictReader(f)
    rows=[row for row in r if row['ModelID'] in common]
print('common dep expr IDs', len(common))
print('common IDs found in Model.csv', len(rows))
print('DepmapModelType counts', Counter(row['DepmapModelType'] for row in rows))
print('ModelType counts', Counter(row.get('ModelType','') for row in rows).most_common(10))
missing=common - {row['ModelID'] for row in rows}
print('missing from Model.csv', len(missing))
PY"
