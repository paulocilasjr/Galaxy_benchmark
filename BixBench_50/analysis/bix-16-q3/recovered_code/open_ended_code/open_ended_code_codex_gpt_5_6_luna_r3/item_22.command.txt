/bin/bash -lc "python - <<'PY'
import csv, time
from pathlib import Path
import numpy as np
import pandas as pd
base=Path('inputs')
def header(name):
    with (base/name).open(newline='') as f:return next(csv.reader(f))
ch=header('CRISPRGeneEffect.csv'); eh=header('OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
expr_pos={g:i for i,g in enumerate(eh)}
shared=[g for g in ch[1:] if g in expr_pos]
ci=[0]+[i for i,g in enumerate(ch[1:],1) if g in expr_pos]
ei=[0]+[expr_pos[g] for g in shared]
dtypes={g:np.float64 for g in shared}
print('shared genes',len(shared),flush=True)
t=time.time(); cr=pd.read_csv(base/'CRISPRGeneEffect.csv',usecols=ci,dtype=dtypes,index_col=0); cr=cr.loc[:,shared]
print('loaded/reordered crispr',cr.shape,'sec',round(time.time()-t,2),flush=True)
t=time.time(); ex=pd.read_csv(base/'OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',usecols=ei,dtype=dtypes,index_col=0); ex=ex.loc[:,shared]
print('loaded/reordered expression',ex.shape,'sec',round(time.time()-t,2),flush=True)
assert list(cr.columns)==shared and list(ex.columns)==shared
ids=[x for x in cr.index if x in set(ex.index)]
A=cr.loc[ids].to_numpy(copy=True) # gene effect; lower means more essential
B=ex.loc[ids].to_numpy(copy=True) # expression, now same gene order
mask=np.isfinite(A)&np.isfinite(B)
n=mask.sum(axis=0).astype(float)
print('cell lines',len(ids),'genes',len(shared),'valid n min/median/max',int(n.min()),float(np.median(n)),int(n.max()),'effect missing',int(np.isnan(A).sum()),flush=True)
t=time.time()
Ar=pd.DataFrame(np.where(mask,A,np.nan)).rank(axis=0,method='average',na_option='keep').to_numpy()
Br=pd.DataFrame(np.where(mask,B,np.nan)).rank(axis=0,method='average',na_option='keep').to_numpy()
mu=(n+1.0)/2.0
da=Ar-mu; db=Br-mu
num=np.nansum(da*db,axis=0); den=np.sqrt(np.nansum(da*da,axis=0)*np.nansum(db*db,axis=0))
r=num/den
re=-r
print('rank/corr sec',round(time.time()-t,2),flush=True)
print('essentiality (-gene effect) count >=0.6',int(np.sum(re>=0.6)),'min/max',float(np.nanmin(re)),float(np.nanmax(re)),flush=True)
print('raw gene effect count >=0.6',int(np.sum(r>=0.6)),'min/max',float(np.nanmin(r)),float(np.nanmax(r)),flush=True)
order=np.argsort(-re)
print('top10')
for j in order[:10]:print(shared[j], 'essentiality_r',float(re[j]),'raw_r',float(r[j]),'n',int(n[j]))
sel=np.where((re>=0.58)&(re<=0.62))[0]
print('near +0.6 [0.58,0.62]',len(sel))
for j in sel[np.argsort(re[sel])]: print('near',shared[j],float(re[j]),int(n[j]))
# Independent direct checks with these arrays for top and a threshold-near gene.
from scipy.stats import spearmanr
for j in order[:5]:
    valid=mask[:,j]
    rr=spearmanr(A[valid,j],B[valid,j]).statistic
    print('check',shared[j],int(valid.sum()),float(rr),float(re[j]),'delta',float(rr-r[j]))
PY"
