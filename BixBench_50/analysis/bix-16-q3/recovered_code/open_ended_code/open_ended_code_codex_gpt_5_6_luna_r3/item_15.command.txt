/bin/bash -lc "python - <<'PY'
import csv, time
from pathlib import Path
import numpy as np
import pandas as pd

base=Path('inputs')
def header(name):
    with (base/name).open(newline='') as f:
        return next(csv.reader(f))
ch=header('CRISPRGeneEffect.csv')
eh=header('OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
expr_pos={g:i for i,g in enumerate(eh)}
shared=[g for g in ch[1:] if g in expr_pos]
ci=[0]+[i for i,g in enumerate(ch[1:],1) if g in expr_pos]
ei=[0]+[expr_pos[g] for g in shared]
print('shared genes',len(shared), 'indices',len(ci),len(ei), flush=True)
t=time.time()
cr=pd.read_csv(base/'CRISPRGeneEffect.csv', usecols=ci, dtype=np.float64, index_col=0)
print('loaded crispr',cr.shape,'sec',round(time.time()-t,2),'nan',int(cr.isna().sum().sum()),'inf',int(np.isinf(cr.to_numpy()).sum()), flush=True)
t=time.time()
ex=pd.read_csv(base/'OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', usecols=ei, dtype=np.float64, index_col=0)
print('loaded expression',ex.shape,'sec',round(time.time()-t,2),'nan',int(ex.isna().sum().sum()),'inf',int(np.isinf(ex.to_numpy()).sum()), flush=True)
crids=set(cr.index); exids=set(ex.index); common_ids=crids & exids
print('common IDs',len(common_ids),'cr-only',len(crids-exids),'ex-only',len(exids-crids), flush=True)
print('duplicate indices',int(cr.index.duplicated().sum()),int(ex.index.duplicated().sum()), flush=True)
ids=[x for x in cr.index if x in common_ids]
A=cr.loc[ids].to_numpy(copy=True); B=ex.loc[ids].to_numpy(copy=True)
print('aligned',A.shape,B.shape,'pair nan',int(np.isnan(A).sum()),int(np.isnan(B).sum()), flush=True)
print('effect ranges',np.nanmin(A),np.nanmax(A),'expr ranges',np.nanmin(B),np.nanmax(B), flush=True)
print('effect cols all nan',int(np.all(np.isnan(A),axis=0).sum()),'expr all nan',int(np.all(np.isnan(B),axis=0).sum()), flush=True)
PY"
