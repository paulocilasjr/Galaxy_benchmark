/bin/bash -lc "python - <<'PY'
import csv,re
from collections import Counter
import pandas as pd, numpy as np
from scipy.stats import spearmanr
ge_fn='inputs/CRISPRGeneEffect.csv'; ex_fn='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
with open(ge_fn,newline='') as f: gh=next(csv.reader(f))[1:]
with open(ex_fn,newline='') as f: eh=next(csv.reader(f))[1:]

def eid(s):return re.search(r'\\((\\d+)\\)"'$'"',s).group(1)
c=Counter(map(eid,eh)); ambig_exact=sorted(set(gh)&set(eh)&{x for x in eh if c[eid(x)]>1})
candidates=['CCND1 (595)','FERMT2 (10979)','KLF5 (688)']
cols=sorted(set(ambig_exact+candidates)); wanted=set(cols)
def load(fn):
 return pd.read_csv(fn,usecols=lambda x:x=='Unnamed: 0' or x in wanted,index_col=0)
g=load(ge_fn); x=load(ex_fn); ix=g.index.intersection(x.index)
print('shared models',len(ix),'ambiguous-ID exact columns checked',len(ambig_exact))
for col in cols:
 ok=g.loc[ix,col].notna() & x.loc[ix,col].notna()
 rho=spearmanr(x.loc[ix,col][ok],-g.loc[ix,col][ok]).statistic
 if col in candidates or rho>=.6: print(col,'n',int(ok.sum()),'rho',format(rho,'.12f'))
print('ambiguous columns reaching threshold',sum(spearmanr(x.loc[ix,z][(g.loc[ix,z].notna()&x.loc[ix,z].notna())],-g.loc[ix,z][(g.loc[ix,z].notna()&x.loc[ix,z].notna())]).statistic>=.6 for z in ambig_exact))
PY"
