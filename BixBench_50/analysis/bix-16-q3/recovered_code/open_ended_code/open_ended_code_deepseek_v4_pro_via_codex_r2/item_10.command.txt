/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import spearmanr, pearsonr
# small
x=pd.DataFrame({'a':[1,2,np.nan,4,5], 'b':[2,1,4,np.nan,5], 'c':[np.nan,np.nan,3,4,5]}, dtype=float)
y=pd.DataFrame({'a':[2,3,4,5,6], 'b':[1,2,3,4,5], 'c':[1,2,3,4,5]}, dtype=float)
print('corrwith df')
print(x.corrwith(y))
print('rank corrwith')
rx=x.rank(axis=0); ry=y.rank(axis=0)
print(rx.corrwith(ry))
for col in ['a','b','c']:
 print(col, spearmanr(x[col],y[col],nan_policy='omit')[0], pearsonr(rx[col],ry[col])[0])
PY"
