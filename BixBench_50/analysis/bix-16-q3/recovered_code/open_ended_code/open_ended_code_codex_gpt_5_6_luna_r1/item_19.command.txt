import csv, collections, time, warnings
import numpy as np
from scipy.stats import rankdata

cp='inputs/CRISPRGeneEffect.csv'

def read_header_ids(path):
    with open(path, newline='', encoding='utf-8', errors='replace') as f:
        rd=csv.reader(f)
        header=next(rd)
        ids=[row[0] for row in rd]
    return header, ids

def feature_id(x):
    p=x.rfind('(')
    return x[p+1:-1] if p >= 0 and x.endswith(')') else x

start=time.time()
ch,cids=read_header_ids(cp)
eh,eids=read_header_ids(ep)
e_row={x:i for i,x in enumerate(eids)}
shared_ids=[x for x in cids if x in e_row]
c_rows=np.array([i for i,x in enumerate(cids) if x in e_row],dtype=np.int64)
e_rows=np.array([e_row[x] for x in shared_ids],dtype=np.int64)
e_id_cols=collections.defaultdict(list); e_label_col={}
for j,x in enumerate(eh[1:],1):
    e_id_cols[feature_id(x)].append(j); e_label_col[x]=j
pairs=[]
for cj,x in enumerate(ch[1:],1):
    fid=feature_id(x)
    if fid in e_id_cols:
        ej=e_label_col.get(x,e_id_cols[fid][0])
        pairs.append((cj,ej,x,eh[ej],fid,x==eh[ej]))
assert len({p[0] for p in pairs})==len(pairs)
assert len({p[1] for p in pairs})==len(pairs)
ccols=[p[0] for p in pairs]; ecols=[p[1] for p in pairs]
exact=np.array([p[5] for p in pairs],dtype=bool)

def read_matrix(path,usecols):
    with warnings.catch_warnings():
        warnings.simplefilter('ignore')
        x=np.genfromtxt(path,delimiter=',',skip_header=1,usecols=usecols,dtype=np.float64,
                         missing_values=['','NA','NaN','nan'],filling_values=np.nan,
                         invalid_raise=True,autostrip=True)
    if x.ndim==1: x=x.reshape((-1,1))
    return x

t=time.time(); C_all=read_matrix(cp,ccols); print('loaded_CRISPR',C_all.shape,'sec',round(time.time()-t,2),flush=True)

C=C_all[c_rows,:]; E=E_all[e_rows,:]

ordidx=np.flatnonzero(finite)[np.argsort(-bio[finite])[:20]]
