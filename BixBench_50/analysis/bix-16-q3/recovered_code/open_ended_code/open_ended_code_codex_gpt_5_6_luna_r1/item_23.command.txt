/bin/bash -lc "PYTHONPATH=/workspace/.local_packages python3 - <<'PY'
import csv, math
from scipy.stats import spearmanr
cp='inputs/CRISPRGeneEffect.csv'; ep='inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'
targets=['CCND1 (595)','FERMT2 (10979)','KLF5 (688)','WWTR1 (25937)']

def load(path):
    with open(path,newline='',encoding='utf-8',errors='replace') as f:
        rd=csv.reader(f); h=next(rd); cols=[h.index(t) for t in targets]
        out={}
        for row in rd:
            vals=[]
            for j in cols:
                try: vals.append(float(row[j]))
                except ValueError: vals.append(float('nan'))
            out[row[0]]=vals
    return out
C=load(cp); E=load(ep)
ids=sorted(set(C)&set(E))
print('independent_shared_ids',len(ids))
for j,t in enumerate(targets):
    xs=[]; ys=[]
    for mid in ids:
        x=C[mid][j]; y=E[mid][j]
        if math.isfinite(x) and math.isfinite(y): xs.append(y); ys.append(-x)
    r=spearmanr(xs,ys).statistic
    print(t,'n=',len(xs),'rho_expression_vs_essentiality=',repr(float(r)),'meets=',bool(r>=0.6))
# Validate all matrix models represented as cell lines in supplied Model metadata.
with open('inputs/Model.csv',newline='',encoding='utf-8',errors='replace') as f:
    models={r['ModelID']:r for r in csv.DictReader(f)}
print('metadata_shared',len(ids),'all_cell_line=',all(models[x].get('ModelType')=='Cell Line' for x in ids))
PY"
