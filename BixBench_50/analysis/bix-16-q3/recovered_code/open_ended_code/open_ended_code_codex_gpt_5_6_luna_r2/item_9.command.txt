import csv
from pathlib import Path

def ids(path):
    with open(path,newline='') as f:
        rd=csv.reader(f); h=next(rd); vals=[row[0] for row in rd]
    return h, vals
files={
 'crispr':'inputs/CRISPRGeneEffect.csv',
 'expr':'inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv',
 'model':'inputs/Model.csv',
}
for k,p in files.items():
    h,v=ids(p)
    print(k,'rows',len(v),'unique',len(set(v)),'first',v[:5],'last',v[-5:])
    print('header first/last',h[:5],h[-5:])
c={x for x in ids(files['crispr'])[1]}; e={x for x in ids(files['expr'])[1]}; m={x for x in ids(files['model'])[1]}
print('CRISPR∩expr',len(c&e),'CRISPR only',len(c-e),'expr only',len(e-c))
print('CRISPR∩expr∩model',len(c&e&m),'shared not model',len((c&e)-m))
print('model∩crispr',len(m&c),'model∩expr',len(m&e))
print('missing/blank IDs', {k:sum(x=='' for x in ids(p)[1]) for k,p in files.items()})
PY"
