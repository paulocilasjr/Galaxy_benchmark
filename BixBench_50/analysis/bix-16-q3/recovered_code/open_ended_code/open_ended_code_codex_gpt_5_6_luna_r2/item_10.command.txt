/bin/bash -lc "python - <<'PY'
import csv
from collections import Counter
paths=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
h=[]
for p in paths:
    with open(p,newline='') as f: row=next(csv.reader(f))[1:]
    c=Counter(row)
    print(p,'features',len(row),'unique',len(c),'duplicates',sum(n-1 for n in c.values() if n>1))
    h.append(row)
inter=set(h[0])&set(h[1])
print('same exact gene labels',len(inter))
print('crispr-only labels',len(set(h[0])-set(h[1])),'expr-only labels',len(set(h[1])-set(h[0])))
print('examples shared',sorted(inter)[:10])
print('examples no expr',sorted(set(h[0])-set(h[1]))[:10])
PY"
