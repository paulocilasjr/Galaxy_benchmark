/bin/bash -lc "PYTHONPATH=/workspace/.local_deps python3 - <<'PY'
import csv
from pathlib import Path
import numpy as np
import pandas as pd
crispr_path=Path('inputs/CRISPRGeneEffect.csv'); expr_path=Path('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'); model_path=Path('inputs/Model.csv')
def header(path):
    with path.open(newline='') as f:return next(csv.reader(f))
def first_col(path):
    with path.open(newline='') as f:
        rd=csv.reader(f);next(rd);return [r[0] for r in rd]
ch=header(crispr_path);eh=header(expr_path);epos={x:i for i,x in enumerate(eh)}
common=[x for x in ch[1:] if x in epos]; common_set=set(common)
cp=[0]+[i for i,x in enumerate(ch) if x in common_set]; ep=[0]+[epos[x] for x in common]
cs=set(first_col(crispr_path));es=set(first_col(expr_path));ms=first_col(model_path);shared=[x for x in ms if x in cs&es]
dtypes={x:np.float32 for x in common}
print('reading',flush=True)
c=pd.read_csv(crispr_path,usecols=cp,index_col=0,dtype=dtypes)
e=pd.read_csv(expr_path,usecols=ep,index_col=0,dtype=dtypes)
print('loaded',c.shape,e.shape,flush=True)
c=c.loc[shared,common];e=e.loc[shared,common]
print('aligned',c.shape,e.shape,'same indexes',c.index.equals(e.index),'same cols',c.columns.equals(e.columns),flush=True)
print('missing CRISPR total/maxcol',int(c.isna().sum().sum()),int(c.isna().sum(axis=0).max()),'expr',int(e.isna().sum().sum()),int(e.isna().sum(axis=0).max()),flush=True)
print('constant cols CRISPR/expression',int((c.nunique(dropna=True)<=1).sum()),int((e.nunique(dropna=True)<=1).sum()),flush=True)
print('raw score range',float(np.nanmin(c.to_numpy())),float(np.nanmax(c.to_numpy())),'expr range',float(np.nanmin(e.to_numpy())),float(np.nanmax(e.to_numpy())),flush=True)
PY"
