import pandas as pd, numpy as np, time, warnings
from scipy.stats import rankdata
warnings.filterwarnings('ignore')
ce=pd.read_csv('inputs/CRISPRGeneEffect.csv', index_col=0)
ex=pd.read_csv('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv', index_col=0)

for g,n,r in sorted(rows,key=lambda z:-z[2])[:20]:
