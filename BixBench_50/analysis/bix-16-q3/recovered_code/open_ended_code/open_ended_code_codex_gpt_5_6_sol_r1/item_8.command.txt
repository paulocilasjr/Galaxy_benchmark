/bin/bash -lc "python - <<'PY'
import csv,re
from pathlib import Path
import pandas as pd
D={}
for name,p in [('effect',Path('inputs/CRISPRGeneEffect.csv')),('expr',Path('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv'))]:
    with p.open(newline='') as f: h=next(csv.reader(f))[1:]
    ids=pd.read_csv(p,usecols=[0]).iloc[:,0]
    D[name]=(h,ids)
    print(name,'rows',len(ids),'unique_rows',ids.nunique(),'genes',len(h),'unique_labels',len(set(h)),'missing_ids',ids.isna().sum())
E,X=D['effect'],D['expr']
print('shared exact gene labels',len(set(E[0])&set(X[0])))
print('shared model IDs',len(set(E[1])&set(X[1])))
print('effect-only models',len(set(E[1])-set(X[1])),'expr-only models',len(set(X[1])-set(E[1])))
def parse(x):
 m=re.fullmatch(r'(.+) \\((\\d+)\\)',x); return (m.group(1),m.group(2)) if m else (x,None)
e={parse(x)[1]:x for x in E[0]}; x={parse(y)[1]:y for y in X[0]}
shared_ids=set(e)&set(x)-{None}; conf=[(e[i],x[i]) for i in shared_ids if e[i]"'!=x[i]]
print('"'shared Entrez IDs',len(shared_ids),'same exact',sum(e[i]==x[i] for i in shared_ids),'conflicting labels',len(conf))
print('conflict examples',conf[:20])
m=pd.read_csv('inputs/Model.csv',usecols=['ModelID','ModelType'])
for label,s in [('shared',set(E[1])&set(X[1])),('effect_only',set(E[1])-set(X[1])),('expr_only',set(X[1])-set(E[1]))]:
 z=m[m.ModelID.isin(s)]
 print(label,'in_model',len(z),'types',z.ModelType.value_counts(dropna=False).to_dict())
PY"
