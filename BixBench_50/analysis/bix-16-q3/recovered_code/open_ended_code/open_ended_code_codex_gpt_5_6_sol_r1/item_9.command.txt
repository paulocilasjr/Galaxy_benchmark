/bin/bash -lc "python - <<'PY'
import csv,re,collections
for path in ['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']:
 with open(path,newline='') as f: h=next(csv.reader(f))[1:]
 parsed=[]; bad=[]
 for z in h:
  m=re.fullmatch(r'(.+) \\((\\d+)\\)',z)
  if m: parsed.append(m.group(2))
  else: bad.append(z)
 c=collections.Counter(parsed); dup={k:v for k,v in c.items() if v>1}
 print(path,'bad',bad,'duplicate Entrez IDs',len(dup),list(dup.items())[:20])
PY"
