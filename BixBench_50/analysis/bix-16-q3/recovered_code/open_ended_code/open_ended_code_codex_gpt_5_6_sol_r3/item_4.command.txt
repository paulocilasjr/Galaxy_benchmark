/bin/bash -lc "python - <<'PY'
import csv, re
import pandas as pd
paths=['inputs/CRISPRGeneEffect.csv','inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv']
headers=[]
for p in paths:
    with open(p,newline='') as f: h=next(csv.reader(f))[1:]
    headers.append(h)
    ids=[]; syms=[]; bad=[]
    for x in h:
        m=re.match(r'"'^(.*) '"\\((["'^()]*)'"\\)"'$'"',x)
        if m: syms.append(m.group(1)); ids.append(m.group(2))
        else: bad.append(x)
    print(p)
    print('features',len(h),'unique labels',len(set(h)),'unique symbols',len(set(syms)),'unique IDs',len(set(ids)),'unparsed',len(bad))
common=set(headers[0]) & set(headers[1])
print('exact common feature labels',len(common))
print('effect not expression',len(set(headers[0])-set(headers[1])))
print('expression not effect',len(set(headers[1])-set(headers[0])))
for p in paths:
    x=pd.read_csv(p,usecols=[0],dtype=str).iloc[:,0]
    print(p,'rows',len(x),'unique IDs',x.nunique(),'duplicate IDs',x.duplicated().sum(),'first/last',x.iloc[0],x.iloc[-1])
    if p==paths[0]: a=set(x)
    else: b=set(x)
m=pd.read_csv('inputs/Model.csv',usecols=['ModelID'])['ModelID']
print('Model rows',len(m),'unique',m.nunique(),'dups',m.duplicated().sum())
print('matrix row overlap',len(a&b),'effect-only',len(a-b),'expr-only',len(b-a))
print('overlap in Model',len((a&b)&set(m)),'shared not Model',len((a&b)-set(m)))
PY"
