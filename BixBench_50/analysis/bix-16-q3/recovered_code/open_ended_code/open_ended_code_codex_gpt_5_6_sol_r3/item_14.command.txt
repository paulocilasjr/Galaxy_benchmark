/bin/bash -lc "python - <<'PY'
import csv, math
names=['CCND1 (595)','FERMT2 (10979)','KLF5 (688)','WWTR1 (25937)']

def read_selected(path):
    out={}
    with open(path,newline='') as f:
        rd=csv.reader(f); h=next(rd); pos=[h.index(x) for x in names]
        for row in rd:
            out[row[0]]=[float(row[i]) if row[i] else math.nan for i in pos]
    return out

def ranks(v):
    order=sorted(range(len(v)),key=v.__getitem__); r=[0.0]*len(v); i=0
    while i<len(order):
        j=i+1
        while j<len(order) and v[order[j]]==v[order[i]]: j+=1
        avg=(i+1+j)/2
        for k in range(i,j): r[order[k]]=avg
        i=j
    return r

def pearson(a,b):
    aa=sum(a)/len(a); bb=sum(b)/len(b)
    xy=sum((x-aa)*(y-bb) for x,y in zip(a,b))
    xx=sum((x-aa)**2 for x in a); yy=sum((y-bb)**2 for y in b)
    return xy/math.sqrt(xx*yy)
ge=read_selected('inputs/CRISPRGeneEffect.csv')
ex=read_selected('inputs/OmicsExpressionProteinCodingGenesTPMLogp1BatchCorrected.csv')
for j,name in enumerate(names):
    pairs=[(ex[m][j],-ge[m][j]) for m in ge.keys()&ex.keys() if math.isfinite(ge[m][j]) and math.isfinite(ex[m][j])]
    x,y=zip(*pairs)
    rho=pearson(ranks(x),ranks(y))
    print(name,len(pairs),format(rho,'.15f'))
PY"
