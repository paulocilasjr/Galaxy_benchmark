fg=pd.read_pickle('/tmp/foreground.pkl').gene_name.dropna().drop_duplicates().tolist()
print('foreground',len(fg),'background',len(bg),'foreground in bg',len(set(fg)&set(bg)))
enr=gp.enrichr(gene_list=fg,gene_sets='Reactome_2022',organism='human',background=bg,outdir=None,cutoff=1.0,verbose=True)
r=enr.results
row=r[r.Term.eq('TP53 Regulates Transcription Of Cell Cycle Genes R-HSA-6791312')]
print(row.to_string(index=False))
for c in ['Overlap','P-value','Adjusted P-value','Odds Ratio','Combined Score']:
