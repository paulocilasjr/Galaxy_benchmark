/bin/bash -c "python - <<'PY'
import pandas as pd, gseapy, json
sig=pd.read_csv('/workspace/tmp/sig_deg.csv',index_col=0)
mp=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t')
s2g=mp.set_index('ENSG')['gene_name']
symbols=sig.index.map(s2g).dropna().astype(str)
symbols=symbols[symbols.str.len()>0]
symbols=pd.Series(symbols.unique())
print('n foreground unique symbols',len(symbols))
# show first few
print(symbols.head(20).tolist())
enr = gseapy.enrichr(gene_list=symbols.tolist(), gene_sets='Reactome_2022', organism='human', outdir=None, no_plot=True)
res=enr.results
print('result shape',res.shape)
print(res.columns.tolist())
print(res.head())
# search term
term='p53-mediated cell cycle gene regulation'
matches=res[res.Term.astype(str).str.contains('p53', case=False, na=False)]
print('p53 matches')
print(matches[['Term','Overlap','P-value','Adjusted P-value','Odds Ratio','Combined Score','Genes']].to_string(index=False))
print('exact match count', (res.Term.astype(str)==term).sum())
print('lower exact', res[res.Term.astype(str).str.lower()==term.lower()][['Term','Overlap','P-value','Adjusted P-value','Odds Ratio','Combined Score','Genes']].to_string(index=False))
res.to_csv('/workspace/tmp/gseapy_reactome2022_results.csv',index=False)
PY"
