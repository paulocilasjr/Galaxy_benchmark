/bin/bash -c "python - <<'PY'
import pandas as pd, gseapy
sig=pd.read_csv('/workspace/tmp/sig_deg.csv',index_col=0)
# Try raw ensembl IDs
try:
    enr=gseapy.enrichr(gene_list=sig.index.tolist(), gene_sets='Reactome_2022', organism='human', outdir=None, no_plot=True)
    res=enr.results
    print('raw ensembl result shape',res.shape)
    print(res.head())
    term='TP53 Regulates Transcription Of Cell Cycle Genes R-HSA-6791312'
    print(res[res.Term==term][['Term','Overlap','Odds Ratio']].to_string(index=False))
except Exception as e:
    print('error',e)
PY"
