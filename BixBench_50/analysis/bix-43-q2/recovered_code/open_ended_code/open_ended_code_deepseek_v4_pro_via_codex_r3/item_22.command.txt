/bin/bash -c "python - <<'PY'
import pandas as pd, gseapy
sig=pd.read_csv('/workspace/tmp/sig_deg.csv',index_col=0)
mp=pd.read_csv('inputs/ensg_to_gene_name.tsv',sep='\\t')
s2g=mp.set_index('ENSG')['gene_name']
raw=sig.index.map(s2g)
raw=raw[raw.notna() & (raw.astype(str).str.len()>0)]
print('raw mapped count',len(raw),'unique',raw.nunique(),'dups',raw.duplicated().sum())
# gseapy with duplicates
enr = gseapy.enrichr(gene_list=raw.astype(str).tolist(), gene_sets='Reactome_2022', organism='human', outdir=None, no_plot=True)
res=enr.results
term='TP53 Regulates Transcription Of Cell Cycle Genes R-HSA-6791312'
row=res[res.Term==term]
print(row[['Term','Overlap','P-value','Adjusted P-value','Odds Ratio','Combined Score','Genes']].to_string(index=False))
print('shape',res.shape)
PY"
