/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd
from gseapy.enrichr import Enrichr
deg=pd.read_csv('/workspace/tmp/deg_full.csv', index_col=0)
ann=pd.read_csv('/workspace/inputs/ensg_to_gene_name.tsv', sep='\\t')
gm=ann.set_index('ENSG')['gene_name']
gl=deg.index.to_series().map(gm).dropna().tolist()
E=Enrichr(gene_list=gl, gene_sets='Reactome_2022', outdir=None, no_plot=True)
# send and view
payload={\"list\":(None,\"\\n\".join(gl)), \"description\":(None,\"\")}
job=E.send_genes(payload, f\"{E.base_url}/addList\")
print('job', job)
view=E.view_list(job['userListId'])
print('keys', view.keys())
print('ngenes returned', len(view.get('genes',[])))
print('genes sample', view.get('genes',[])[:20])
print('maybe unrecognized sample', [x for x in view.get('genes',[]) if x in gl][:10])
# recognize count exactly: returnedL may be uppercase etc
ret=view.get('genes',[])
print('recognized count intersection exact', sum(1 for g in gl if g in ret))
# print gene list length and unique
print('gl len', len(gl), 'unique', len(set(gl)))
# list elements not in returned
notin=[g for g in gl if g not in ret]
print('not recognized', len(notin), notin[:100])
PY"
