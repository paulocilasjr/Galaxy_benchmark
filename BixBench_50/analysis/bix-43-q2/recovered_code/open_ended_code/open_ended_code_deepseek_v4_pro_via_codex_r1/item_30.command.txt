/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
counts=pd.read_csv('/workspace/inputs/counts_raw_unfiltered.csv', index_col=0)
layout=pd.read_csv('/workspace/inputs/sample_layout.csv')
layout['sid']=layout['SampleID'].str.replace('-','_')
sel=['DMSO','DMSO_Serum_starvation','Cisplatin_IC50_CBD_IC50','Cisplatin_IC50_CBD_IC50_Serum_starvation_16h']
meta=layout[layout['Group'].isin(sel)].set_index('sid')
sub=counts[meta.index].T
keep=(counts > 10).any(axis=1)
sub=sub.loc[:, keep].iloc[:, :500]
meta=meta[['Group']].rename(columns={'Group':'group'})
print(sub.shape, meta.shape)
dds=DeseqDataSet(counts=sub, metadata=meta, design='~group', quiet=True)
dds.deseq2(quiet=True)
st=DeseqStats(dds, contrast=['group','Cisplatin_IC50_CBD_IC50','DMSO'], alpha=0.05, quiet=True)
st.summary(quiet=True)
print(st.results_df.head())
print(st.results_df.columns)
PY"
