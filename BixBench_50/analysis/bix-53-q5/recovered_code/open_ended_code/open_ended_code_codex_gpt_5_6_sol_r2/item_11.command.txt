/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
x=pd.read_csv('inputs/count.csv',index_col=0).T
meta=pd.DataFrame({'condition':['KL']*3+['WL']*3}, index=x.index)
dds=DeseqDataSet(counts=x, metadata=meta, design='~condition', ref_level=['condition','WL'], refit_cooks=True, n_cpus=4, quiet=False)
dds.deseq2()
print('LFC columns', list(dds.varm['LFC'].columns))
st=DeseqStats(dds, contrast=['condition','KL','WL'], alpha=0.05, n_cpus=4, quiet=True)
st.summary()
coeff='condition[T.KL]'
print('before', ((st.results_df.pvalue<.05)&(st.results_df.log2FoldChange.abs()>1)&(st.results_df.baseMean>10)).sum())
st.lfc_shrink(coeff=coeff)
r=st.results_df
m=(r.pvalue<.05)&(r.log2FoldChange.abs()>1)&(r.baseMean>10)
print('after',m.sum(), 'up',(m&(r.log2FoldChange>0)).sum(),'down',(m&(r.log2FoldChange<0)).sum())
print(r.loc[m].sort_values('pvalue').head(20).to_string())
PY"
