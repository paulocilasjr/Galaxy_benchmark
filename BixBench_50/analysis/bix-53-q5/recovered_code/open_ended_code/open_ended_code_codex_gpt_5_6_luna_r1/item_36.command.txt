/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, time
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

raw=pd.read_csv('inputs/count.csv')
counts=raw.set_index('geneID').T
counts.index=pd.Index(counts.index, name='sample')
metadata=pd.DataFrame({'condition':['KL']*3+['WL']*3}, index=counts.index)
print(counts.shape, metadata)

t=time.time()
dds=DeseqDataSet(counts=counts, metadata=metadata, design='~condition', ref_level=['condition','WL'], refit_cooks=True, quiet=True, n_cpus=1)
print('init',time.time()-t)
t=time.time(); dds.deseq2(); print('deseq',time.time()-t)
print('var cols', dds.var.columns.tolist()); print(dds.var.head())
print('obs cols', dds.obs.columns.tolist()); print(dds.obsm.keys())

t=time.time(); stat=DeseqStats(dds, contrast=['condition','KL','WL'], alpha=0.05, cooks_filter=True, independent_filter=True, quiet=True, n_cpus=1); stat.summary(); print('stats',time.time()-t)
print('results cols',stat.results_df.columns.tolist())
print(stat.results_df.head())
print('dds var sample',dds.var.head())
print('lfc cols',stat.LFC.columns.tolist())
print('coeffs',stat.LFC.head())
# shrink
co=[c for c in stat.LFC.columns if 'condition' in str(c)]
print('co',co)
t=time.time();stat.lfc_shrink(co[0],adapt=True); print('shrink',time.time()-t)
print(stat.results_df.head())
res=stat.results_df
print('baseMean available',res['baseMean'].describe())
print('p<.05',((res.pvalue<.05)).sum(),'abs lfc>1',((res.log2FoldChange.abs()>1)).sum(),'base>10',(res.baseMean>10).sum(),'all',((res.pvalue<.05)&(res.log2FoldChange.abs()>1)&(res.baseMean>10)).sum())
print(res[(res.pvalue<.05)&(res.log2FoldChange.abs()>1)&(res.baseMean>10)].sort_values('pvalue').head(20).to_string())
PY"
