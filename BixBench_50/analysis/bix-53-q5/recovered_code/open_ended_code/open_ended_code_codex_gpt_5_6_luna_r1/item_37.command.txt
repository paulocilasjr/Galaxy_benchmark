/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, time, traceback, sys
try:
 from pydeseq2.dds import DeseqDataSet
 from pydeseq2.ds import DeseqStats
 raw=pd.read_csv('inputs/count.csv')
 counts=raw.set_index('geneID').T
 metadata=pd.DataFrame({'condition':['KL']*3+['WL']*3}, index=counts.index)
 print('start',counts.shape,flush=True)
 dds=DeseqDataSet(counts=counts, metadata=metadata, design='~condition', ref_level=['condition','WL'], refit_cooks=True, quiet=False, n_cpus=1)
 print('init',flush=True)
 dds.deseq2()
 print('deseq done',flush=True)
 stat=DeseqStats(dds, contrast=['condition','KL','WL'], alpha=0.05, cooks_filter=True, independent_filter=True, quiet=False, n_cpus=1)
 print('stat init',flush=True)
 stat.summary()
 print('summary done',flush=True)
 print(stat.results_df.head(),flush=True)
except BaseException as e:
 print('ERROR',repr(e),flush=True); traceback.print_exc(); sys.exit(1)
PY"
