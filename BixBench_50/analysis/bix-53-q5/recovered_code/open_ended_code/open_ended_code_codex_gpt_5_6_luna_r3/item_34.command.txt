/bin/bash -lc "python -u - <<'PY'
import pandas as pd, time, sys
from gseapy import Biomart
res=pd.read_csv('/tmp/pydeseq2_res.csv')
sel=res['pvalue'].lt(.05)&res['log2FoldChange'].abs().gt(1)&res['baseMean'].gt(10)
ids=res.loc[sel,'geneID'].tolist()
print('ids',len(ids),ids[:3],flush=True)
bm=Biomart(verbose=True)
print('host',bm.host,flush=True)
t=time.time()
mp=bm.query(dataset='mmusculus_gene_ensembl',attributes=['ensembl_gene_id','external_gene_name'],filters={'ensembl_gene_id':ids})
print('elapsed',time.time()-t,'shape',None if mp is None else mp.shape,flush=True)
print(mp.head() if mp is not None else None,flush=True)
if mp is not None: mp.to_csv('/tmp/mapping.csv',index=False)
PY"
