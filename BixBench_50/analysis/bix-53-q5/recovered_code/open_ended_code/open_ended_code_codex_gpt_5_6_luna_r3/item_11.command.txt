/bin/bash -lc "Rscript - <<'RS'
x <- read.csv('inputs/count.csv', check.names=FALSE)
cat('dim', nrow(x), ncol(x), '\\n')
cat('cols', paste(names(x), collapse=', '), '\\n')
cat('unique gene IDs', length(unique(x[[1]])), 'duplicates', sum(duplicated(x[[1]])), '\\n')
cat('all ENSMUSG', all(grepl('"'^ENSMUSG[0-9]+$'"', x[[1]])), '\\n')
cat('totals\\n'); print(colSums(x[-1]))
cat('nonzero per sample\\n'); print(colSums(x[-1] > 0))
cat('range\\n'); print(sapply(x[-1], range))
RS"
