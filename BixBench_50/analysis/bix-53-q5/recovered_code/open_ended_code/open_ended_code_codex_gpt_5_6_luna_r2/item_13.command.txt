/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats

counts=pd.read_csv('/workspace/inputs/count.csv')
counts=counts.set_index('geneID')
meta=pd.DataFrame({'condition':['KL']*3+['WL']*3},index=counts.columns)
# PyDESeq2 expects samples in rows and genes in columns.
cts=counts.T
print('counts',cts.shape,'metadata',meta.shape)
start=time.time()
dds=DeseqDataSet(counts=cts,metadata=meta,design='~condition',ref_level=['condition','WL'],quiet=True,n_cpus=1)
dds.deseq2()
print('deseq2 secs',time.time()-start)
print('LFC columns',dds.varm['LFC'].columns if hasattr(dds.varm['LFC'],'columns') else getattr(dds.varm['LFC'],'shape',None))
# use stats object, contrast KL vs WL
stat=DeseqStats(dds,contrast=['condition','KL','WL'],quiet=True,n_cpus=1)
stat.summary()
print('results cols',stat.results_df.columns.tolist())
print('pre shrink')
print(stat.results_df[['baseMean','log2FoldChange','pvalue','padj']].describe().to_string())
# coeff from LFC columns
print('lfc stats columns',stat.LFC.columns.tolist())
# shrink the condition coefficient
coeff=[c for c in stat.LFC.columns if c"'!='"'intercept'][0]
print('shrink coeff',coeff)
stat.lfc_shrink(coeff=coeff,adapt=True)
res=stat.results_df.copy()
print('post shrink')
print(res[['baseMean','log2FoldChange','lfcSE','pvalue','padj']].describe().to_string())
for pcol in ['pvalue','padj']:
  sel=res[(res[pcol]<.05)&(res.log2FoldChange.abs()>1)&(res.baseMean>10)]
  print(pcol,'selected',len(sel),'up',sum(sel.log2FoldChange>1),'down',sum(sel.log2FoldChange < -1))
  print(sel.sort_values(pcol)[['baseMean','log2FoldChange','pvalue','padj']].head(10).to_string())
PY"
