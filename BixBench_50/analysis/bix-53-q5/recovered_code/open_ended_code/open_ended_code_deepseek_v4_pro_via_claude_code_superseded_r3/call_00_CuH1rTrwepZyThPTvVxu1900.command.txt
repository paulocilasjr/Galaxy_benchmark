from pydeseq2.dds import DeseqDataSet
from pydeseq2.ds import DeseqStats
import pandas as pd

counts = pd.read_csv('inputs/count.csv', index_col=0)
counts_t = counts.T
metadata = pd.DataFrame({'condition': ['KL','KL','KL','WL','WL','WL']}, index=counts_t.index)

dds = DeseqDataSet(counts=counts_t, metadata=metadata, design='~condition')
dds.deseq2()

# Try to see what LFC names are available
print('dds results_df columns:', dds.varm.keys() if hasattr(dds, 'varm') else 'N/A')
print('dds layers:', dds.layers.keys() if hasattr(dds, 'layers') else 'N/A')

# Try to access the contrast directly
stat_res = DeseqStats(dds, contrast=['condition', 'KL', 'WL'])
stat_res.summary()

print('\\nTrying lfc_shrink with coeff...')
