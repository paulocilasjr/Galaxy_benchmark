/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, mygene
res = pd.read_csv('tmp/deseq_results.csv', index_col=0)
mask = (res['padj'] < 0.05) & (res['log2FoldChange'].abs()>1) & (res['baseMean']>10)
ids = res.index[mask].tolist()
print('n ids', len(ids), ids[:5])
mg = mygene.MyGeneInfo()
out = mg.querymany(ids, scopes='ensembl.gene', fields='symbol', species='mouse', verbose=False)
# inspect dup/missing
print('out len', len(out))
missing=[x for x in out if 'notfound' in x]
print('missing count', len(missing), missing[:5])
# build
pairs=[]
for d in out:
 if 'symbol' in d:
  pairs.append((d['query'], d['symbol']))
print('symbol pairs', len(pairs), pairs[:10])
# duplicates per query
from collections import defaultdict
mp=defaultdict(list)
for q,s in pairs: mp[q].append(s)
dups={q:v for q,v in mp.items() if len(v)>1}
print('queries with >1', len(dups), list(dups.items())[:20])
# choose first unique
symbols=[]
for q in ids:
 v=mp.get(q)
 if v and len(v)==1: symbols.append(v[0])
print('unique symbols', len(symbols))
open('tmp/sig_symbols_padj.txt','w').write('\\n'.join(symbols))
PY"
