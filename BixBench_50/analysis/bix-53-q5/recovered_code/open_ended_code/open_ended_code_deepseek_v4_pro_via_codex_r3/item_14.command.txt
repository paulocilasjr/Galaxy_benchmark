/bin/bash -c "python - <<'PY'
import os
for pid in os.listdir('/proc'):
    if not pid.isdigit(): continue
    try:
        with open(f'/proc/{pid}/cmdline','rb') as f: cmd=f.read().replace(b'\\0',b' ').decode(errors='ignore')
    except Exception: continue
    if any(s in cmd for s in ['Rscript','BiocManager','INSTALL','R CMD']):
        print(pid, cmd)
PY"
