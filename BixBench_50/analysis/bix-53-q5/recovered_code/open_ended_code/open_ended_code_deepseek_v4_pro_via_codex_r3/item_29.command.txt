/bin/bash -c "Rscript - <<'RS'
suppressPackageStartupMessages(library(DESeq2))
counts <- read.csv('/workspace/inputs/count.csv', row.names=1, check.names=FALSE)
counts <- as.matrix(counts)
mode(counts) <- 'integer'
samples <- data.frame(row.names=colnames(counts),
                      condition=factor(c('KL','KL','KL','WL','WL','WL'), levels=c('KL','WL')))
dds <- DESeqDataSetFromMatrix(countData=counts, colData=samples, design=~condition)
dds <- DESeq(dds, quiet=TRUE)
cat('Results names:\\n'); print(resultsNames(dds))
res <- lfcShrink(dds, contrast=c('condition','WL','KL'), type='normal', quiet=TRUE)
res <- as.data.frame(res)
res"'$geneID <- rownames(res)
write.csv(res, '"'/workspace/tmp/deseq2_results.csv', row.names=FALSE)
sig <- res["'!is.na(res$padj) & res$padj < 0.05 & !is.na(res$log2FoldChange) & abs(res$log2FoldChange) > 1 & res$baseMean > 10, ]
sig <- sig[order(sig$padj),]
write.table(sig[,c('"'geneID','baseMean','log2FoldChange','lfcSE','pvalue','padj')], '/workspace/tmp/sig_genes.tsv', sep='\\t', quote=FALSE, row.names=FALSE, col.names=TRUE)
cat('sig genes:', nrow(sig), '\\n')
cat('up/down:', sum(sig"'$log2FoldChange>0), sum(sig$log2FoldChange<0), '"'\\n')
cat('raw pvalue significant genes:', sum("'!is.na(res$pvalue) & res$pvalue<0.05 & !is.na(res$log2FoldChange) & abs(res$log2FoldChange)>1 & res$baseMean>10), '"'\\n')
print(head(sig, 20))
RS"
