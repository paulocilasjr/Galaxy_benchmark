/bin/bash -c "python3 - <<'PY'
import zipfile, tempfile, os, subprocess, pathlib, statistics, json
root=pathlib.Path('/workspace/tmp')
root.mkdir(exist_ok=True)
results={}
for group, zpath in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
    td=tempfile.mkdtemp(prefix=f'{group}_trees_', dir=str(root))
    vals=[]
    with zipfile.ZipFile(zpath) as zf:
        for n in zf.namelist():
            b=n.split('/')[-1]
            if not b.endswith('.treefile'):
                continue
            p=pathlib.Path(td)/b
            p.write_bytes(zf.read(n))
            proc=subprocess.run(['phykit','total_tree_length',str(p)],capture_output=True,text=True,check=True)
            lines=[l for l in proc.stdout.splitlines() if l.strip()]
            val=float(lines[0].strip().split()[0])
            vals.append((b,val))
    results[group]=vals
    print(group, 'count', len(vals), 'median', statistics.median(v for _,v in vals))
    out=pathlib.Path('/workspace/tmp')/f'{group}_tree_lengths.tsv'
    out.write_text('\\n'.join(f'{b}\\t{v}' for b,v in vals)+'\\n')
    print('sample first 5', vals[:5])
# fold changes
am=statistics.median(v for _,v in results['animals'])
fm=statistics.median(v for _,v in results['fungi'])
print('animals_median',am,'fungi_median',fm,'fungi/animals',fm/am,'animals/fungi',am/fm)
PY"
