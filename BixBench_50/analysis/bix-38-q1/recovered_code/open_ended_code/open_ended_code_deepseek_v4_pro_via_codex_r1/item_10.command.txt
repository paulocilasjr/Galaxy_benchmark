/bin/bash -c "python3 - <<'PY'
import zipfile, os, collections
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as zf:
        names=zf.namelist()
    print(z, len(names))
    bysuf=collections.Counter()
    for n in names:
        # basename after slash
        b=n.split('/')[-1]
        if '.treefile' in b: bysuf['treefile']+=1
        elif b.endswith('.iqtree'): bysuf['iqtree']+=1
        elif '.mafft.clipkit' in b and not any(x in b for x in ['.model.gz','.ckp.gz','.log','.mldist','.bionj','.treefile','.iqtree']): bysuf['clipkit']+=1
        elif b.endswith('.mafft'): bysuf['mafft']+=1
        elif not any(x in b for x in ['.mafft','.clipkit','.model','.ckp','.log','.mldist','.bionj','.treefile','.iqtree']): bysuf['faa']+=1
        else: bysuf['other']+=1
    print(bysuf)
PY"
