/bin/bash -c "python3 - <<'PY'
import zipfile, tempfile, os, subprocess, pathlib, statistics
root=pathlib.Path('/workspace/tmp'); root.mkdir(exist_ok=True)
for group,zpath in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
    td=tempfile.mkdtemp(prefix=f'round_{group}_', dir=str(root))
    raw=[]; rounded=[]; errors=[]
    with zipfile.ZipFile(zpath) as zf:
        for n in zf.namelist():
            b=n.split('/')[-1]
            if not b.endswith('.treefile'): continue
            p=pathlib.Path(td)/b; p.write_bytes(zf.read(n))
            r=subprocess.run(['phykit','total_tree_length',str(p)],capture_output=True,text=True)
            if r.returncode"'!=0:
                errors.append((b,r.stderr)); continue
            v=float(r.stdout.strip().split()[0])
            raw.append((b,v)); rounded.append((b,round(v,4)))
    print('"'\\n',group,'n',len(raw),'errors',len(errors))
    print('raw median', statistics.median(v for _,v in raw))
    print('rounded median', statistics.median(v for _,v in rounded))
    print('raw min/max',min(v for _,v in raw),max(v for _,v in raw))
    print('rounded min/max',min(v for _,v in rounded),max(v for _,v in rounded))
    print('values sorted rounded first/last', sorted(rounded)[:3], sorted(rounded)[-3:])
PY"
