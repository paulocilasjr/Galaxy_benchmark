/bin/bash -lc "python - <<'PY'
import zipfile, tempfile, pathlib, subprocess
with tempfile.TemporaryDirectory(dir='/workspace') as d:
    with zipfile.ZipFile('/workspace/inputs/scogs_animals.zip') as z:
        n=next(x for x in z.namelist() if x.endswith('.treefile'))
        p=pathlib.Path(d)/n
        p.write_bytes(z.read(n))
    res=subprocess.run(['python','-m','phykit','total_tree_length',str(p)],text=True,capture_output=True,check=True)
    print(n)
    print('STDOUT:',repr(res.stdout))
    print('STDERR:',repr(res.stderr))
PY"
