/bin/bash -lc "rm -rf tmp/tree_lengths && mkdir -p tmp/tree_lengths/animals tmp/tree_lengths/fungi && python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
pairs = [('inputs/scogs_animals.zip', 'tmp/tree_lengths/animals'), ('inputs/scogs_fungi.zip', 'tmp/tree_lengths/fungi')]
for archive, outdir in pairs:
    out = Path(outdir)
    with ZipFile(archive) as z:
        n = 0
        for info in z.infolist():
            if info.filename.endswith('.treefile'):
                target = out / Path(info.filename).name
                target.write_bytes(z.read(info))
                n += 1
        print(archive, n)
PY"
