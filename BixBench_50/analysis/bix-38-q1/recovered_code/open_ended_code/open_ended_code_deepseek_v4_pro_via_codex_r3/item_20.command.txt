/bin/bash -lc "cd /workspace && python - <<'PY'
from pathlib import Path
import collections
for g in ['scogs_animals','scogs_fungi']:
    files=list(Path('work',g).iterdir())
    bybase=collections.defaultdict(list)
    for f in files:
        name=f.name
        for suf in ['.faa.mafft.clipkit.treefile','.faa.mafft.clipkit.iqtree','.faa.mafft.clipkit','.faa.mafft','.faa']:
            if name.endswith(suf):
                base=name[:-len(suf)]
                bybase[base].append(suf)
                break
    print(g, 'bases',len(bybase))
    for suf in ['.faa','.faa.mafft','.faa.mafft.clipkit','.faa.mafft.clipkit.iqtree','.faa.mafft.clipkit.treefile']:
        n=sum(1 for l in bybase.values() if suf in l)
        print(' ',suf,n)
    print('bases tree',len([b for b,l in bybase.items() if '.faa.mafft.clipkit.treefile' in l]))
    if g=='scogs_animals':
        animal=set(bybase)
    else: fungi=set(bybase)
print('overlap bases',len(animal & fungi))
print('animal-only',len(animal-fungi), 'fungi-only', len(fungi-animal))
PY"
