/bin/bash -lc "python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
from io import StringIO
from Bio import Phylo
from statistics import median

for group in ('animals','fungi'):
    vals=[]; taxon_counts={}; no_branch=0; invalid=0
    with ZipFile(Path('inputs') / f'scogs_{group}.zip') as z:
        names=sorted(n for n in z.namelist() if n.endswith('.treefile'))
        for n in names:
            tree=Phylo.read(StringIO(z.read(n).decode()), 'newick')
            branches=[c.branch_length for c in tree.find_clades() if c is not tree.root]
            if any(x is None for x in branches): no_branch+=1
            if any(x is not None and x < 0 for x in branches): invalid+=1
            vals.append(round(sum(x for x in branches if x is not None), 4))
            taxon_counts[len(tree.get_terminals())]=taxon_counts.get(len(tree.get_terminals()),0)+1
    print(group, len(vals), taxon_counts, 'missing_branch_trees', no_branch, 'negative_branch_trees', invalid, 'median', median(vals))
PY"
