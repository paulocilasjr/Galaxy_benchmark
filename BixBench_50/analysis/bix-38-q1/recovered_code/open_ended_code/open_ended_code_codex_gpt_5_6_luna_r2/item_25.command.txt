/bin/bash -lc "python - <<'PY'
from pathlib import Path
from zipfile import ZipFile
from tempfile import TemporaryDirectory
from subprocess import run
from statistics import median
import json, sys

results = {}
with TemporaryDirectory(prefix='.phykit_', dir='/workspace') as tmp:
    tmp = Path(tmp)
    for group in ('animals','fungi'):
        archive = Path('inputs') / f'scogs_{group}.zip'
        members = []
        with ZipFile(archive) as z:
            members = sorted(n for n in z.namelist() if n.endswith('.treefile'))
            vals = []
            for i, member in enumerate(members):
                path = tmp / f'{group}_{i}.treefile'
                path.write_bytes(z.read(member))
                cp = run([sys.executable, '-m', 'phykit', 'total_tree_length', str(path), '--json'],
                         text=True, capture_output=True, check=True)
                line = cp.stdout.strip().splitlines()[-1]
                vals.append(float(json.loads(line)['total_tree_length']))
        results[group] = vals
        print(group, 'n=', len(vals), 'min=', min(vals), 'median=', median(vals), 'max=', max(vals))

animals = median(results['animals'])
fungi = median(results['fungi'])
print('animals_median', repr(animals))
print('fungi_median', repr(fungi))
print('fungi_over_animals', repr(fungi/animals))
print('animals_over_fungi', repr(animals/fungi))
PY"
