/bin/bash -lc "sed -n '1,100p' .venv_phykit/lib/python3.11/site-packages/phykit/services/tree/total_tree_length.py
.venv_phykit/bin/python - <<'PY'
from pathlib import Path
from statistics import median
from subprocess import run
from Bio import Phylo
exe = '.venv_phykit/bin/phykit'
for group in ('animals','fungi'):
    paths=sorted(Path('work_phykit',group).glob('*.treefile'))
    vals=[]; exact=[]; failures=[]
    for p in paths:
        q=run([exe,'tree_len',str(p)],capture_output=True,text=True)
        if q.returncode or not q.stdout.strip():
            failures.append((p.name,q.returncode,q.stderr))
            continue
        vals.append(float(q.stdout.strip()))
        exact.append(Phylo.read(p,'newick').total_branch_length())
    print(group, 'selected',len(paths),'processed',len(vals),'failed',len(failures))
    print('phykit median',repr(median(vals)),'min',min(vals),'max',max(vals))
    print('exact median',repr(median(exact)))
    globals()[group]=vals
print('fungi/animals using PhyKIT outputs:',repr(median(fungi)/median(animals)))
print('animals/fungi using PhyKIT outputs:',repr(median(animals)/median(fungi)))
PY"
