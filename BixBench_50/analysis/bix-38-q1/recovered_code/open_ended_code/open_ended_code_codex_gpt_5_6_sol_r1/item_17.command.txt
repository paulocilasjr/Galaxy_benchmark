/bin/bash -lc ".venv_phykit/bin/python - <<'PY'
from pathlib import Path
from statistics import median
from subprocess import check_output
exe='.venv_phykit/bin/phykit'
def load(g):
 d={}
 for p in Path('work_phykit',g).glob('*.treefile'):
  k=p.name.split('.')[0]
  d[k]=float(check_output([exe,'tree_len',str(p)],text=True))
 return d
a=load('animals'); f=load('fungi'); common=sorted(a.keys() & f.keys())
print('common inferred locus count',len(common))
print('common-loci medians and fungi/animals',median(f[k] for k in common),median(a[k] for k in common),median(f[k] for k in common)/median(a[k] for k in common))
for name,d in [('animals',a),('fungi',f)]:
 # branch lengths: 3-taxon Newick has 3 colons; 4 has 5 due internal branch
 by={3:[],4:[]}
 for k,v in d.items():
  s=next(Path('work_phykit',name).glob(k+'*.treefile')).read_text()
  ntips=s.count(',')+1
  by[ntips].append(v)
 print(name,[(n,len(x),median(x)) for n,x in by.items() if x])
PY"
