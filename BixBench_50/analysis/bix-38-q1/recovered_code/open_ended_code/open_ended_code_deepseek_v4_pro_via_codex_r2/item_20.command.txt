/bin/bash -c "/workspace/tmp/phykit_venv/bin/python - <<'PY'
import glob, re, statistics
from collections import Counter
# parse newick total branch length robustly via simple regex numbers after colon
def tree_len(path):
    s=open(path).read().strip()
    # all branch lengths after colon before comma/paren
    nums=re.findall(r':([-+]?\\d*\\.?\\d+(?:[eE][-+]?\\d+)?)', s)
    return sum(float(x) for x in nums)
for grp,path in [('animals','/workspace/tmp/scogs_animals_extracted'),('fungi','/workspace/tmp/scogs_fungi_extracted')]:
    vals=[]
    for f in glob.glob(path+'/*.treefile'):
        vals.append(tree_len(f))
    vals=sorted(vals)
    print(grp,'n',len(vals),'min',min(vals),'max',max(vals))
    # quantile med
    n=len(vals)
    med=(vals[n//2] if n%2 else (vals[n//2-1]+vals[n//2])/2)
    print(' exact median',repr(med))
    # rounded phykit median
    rvals=[round(v,4) for v in vals]
    rvals=sorted(rvals)
    n=len(rvals); rmed=(rvals[n//2] if n%2 else (rvals[n//2-1]+rvals[n//2])/2)
    print(' rounded median',repr(rmed))
# store exacts
all={}
for grp,path in [('animals','/workspace/tmp/scogs_animals_extracted'),('fungi','/workspace/tmp/scogs_fungi_extracted')]:
    vals=[tree_len(f) for f in glob.glob(path+'/*.treefile')]
    vals=sorted(vals); n=len(vals); med=(vals[n//2] if n%2 else (vals[n//2-1]+vals[n//2])/2); all[grp]=(vals,med)
print('ratio exact fungi/animal',all['fungi'][1]/all['animals'][1])
# using phykit rounded medians
print('ratio rounded medians',3.0962/1.4219)
# per-phykit outputs stored
PY"
