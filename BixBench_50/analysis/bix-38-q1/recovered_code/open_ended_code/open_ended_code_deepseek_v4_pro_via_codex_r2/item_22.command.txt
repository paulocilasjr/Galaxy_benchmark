/bin/bash -c "/workspace/tmp/phykit_venv/bin/python - <<'PY'
import glob, os, re, zipfile, collections
from Bio import SeqIO
for grp,path in [('animals','/workspace/tmp/scogs_animals_extracted'),('fungi','/workspace/tmp/scogs_fungi_extracted')]:
    tree_ids={os.path.basename(f).split('.')[0] for f in glob.glob(path+'/*.treefile')}
    counts=collections.Counter(); tree_count_status=collections.Counter()
    for f in glob.glob(path+'/*.faa'):
        gid=os.path.basename(f).split('.')[0]
        n=sum(1 for _ in SeqIO.parse(f,'fasta'))
        counts[n]+=1
        tree_count_status[(n,gid in tree_ids)]+=1
    print('\\n',grp,'raw faa',len(glob.glob(path+'/*.faa')))
    print(' seq count distribution',sorted(counts.items()))
    print(' by n and tree status',sorted(tree_count_status.items()))
    # clipkit counts similarly
    counts2=collections.Counter(); status2=collections.Counter()
    for f in glob.glob(path+'/*.clipkit'):
        gid=os.path.basename(f).split('.')[0]
        n=sum(1 for _ in SeqIO.parse(f,'fasta'))
        counts2[n]+=1; status2[(n,gid in tree_ids)]+=1
    print(' clipkit count distribution',sorted(counts2.items()))
    print(' clipkit status',sorted(status2.items()))
PY"
