/bin/bash -lc "python - <<'PY'
import re, zipfile, statistics

def branch_sum(s):
    return sum(float(x) for x in re.findall(r':([+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?)', s))
for label, archive in [('animals','inputs/scogs_animals.zip'), ('fungi','inputs/scogs_fungi.zip')]:
    vals=[]
    with zipfile.ZipFile(archive) as z:
        for n in sorted(x for x in z.namelist() if x.endswith('.treefile')):
            vals.append(round(branch_sum(z.read(n).decode()), 4))
    print(label, len(vals), format(statistics.median(vals), '.12g'))
print('ratio', format(3.0962/1.4219, '.12g'))
PY"
