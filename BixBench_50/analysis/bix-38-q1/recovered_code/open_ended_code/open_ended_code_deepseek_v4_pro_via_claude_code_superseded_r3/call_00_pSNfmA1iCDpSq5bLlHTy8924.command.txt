python3 << 'EOF'
import statistics

with open('/workspace/phykit_animals.txt') as f:
    animals = sorted([float(line.strip()) for line in f if line.strip()])

with open('/workspace/phykit_fungi.txt') as f:
    fungi = sorted([float(line.strip()) for line in f if line.strip()])

animals_median = statistics.median(animals)
fungi_median = statistics.median(fungi)

print(f"Animals (n={len(animals)}): median={animals_median}")
print(f"Fungi (n={len(fungi)}): median={fungi_median}")
print(f"Fold-change fungi/animals: {fungi_median / animals_median}")

# Show some distribution details
print(f"\nAnimals percentiles: 25th={animals[len(animals)//4]:.4f}, 75th={animals[3*len(animals)//4]:.4f}")
print(f"Fungi percentiles: 25th={fungi[len(fungi)//4]:.4f}, 75th={fungi[3*len(fungi)//4]:.4f}")
EOF
