import subprocess
import glob
import statistics
import os

phykit = "/tmp/phykit_env/bin/phykit"

def get_tree_length(treefile):
    result = subprocess.run([phykit, "total_tree_length", treefile], 
                          capture_output=True, text=True, timeout=30)
    for line in result.stdout.strip().split('\n'):
        try:
            return float(line.strip())
        except ValueError:
            continue
    return None

animals_files = sorted(glob.glob("/workspace/animals_trees/*.treefile"))
fungi_files = sorted(glob.glob("/workspace/fungi_trees/*.treefile"))

animals_lengths = []
for f in animals_files:
    tl = get_tree_length(f)
    if tl is not None:
        animals_lengths.append(tl)

fungi_lengths = []
for f in fungi_files:
    tl = get_tree_length(f)
    if tl is not None:
        fungi_lengths.append(tl)

animals_median = statistics.median(animals_lengths)
fungi_median = statistics.median(fungi_lengths)

print(f"Animals median: {animals_median:.10f}")
print(f"Fungi median: {fungi_median:.10f}")
print(f"Fold change (fungi/animals): {fungi_median/animals_median:.10f}")

# Also list all values to verify
print(f"\nAnimals lengths ({len(animals_lengths)}): {sorted(animals_lengths)}")
print(f"\nFungi lengths ({len(fungi_lengths)}): {sorted(fungi_lengths)}")
