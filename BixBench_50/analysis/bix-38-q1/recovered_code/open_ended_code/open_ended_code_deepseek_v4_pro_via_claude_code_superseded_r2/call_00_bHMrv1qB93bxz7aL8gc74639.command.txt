import subprocess
import glob
import statistics
import sys

phykit = "/tmp/phykit_env/bin/phykit"

def get_tree_length(treefile):
    result = subprocess.run([phykit, "total_tree_length", treefile], 
                          capture_output=True, text=True, timeout=30)
    # Parse output
    for line in result.stdout.strip().split('\n'):
        try:
            return float(line.strip())
        except ValueError:
            continue
    return None

def main():
    for label, pattern in [("animals", "/workspace/animals_trees/*.treefile"),
                           ("fungi", "/workspace/fungi_trees/*.treefile")]:
        files = sorted(glob.glob(pattern))
        lengths = []
        for f in files:
            l = get_tree_length(f)
            if l is not None:
                lengths.append(l)
        
        median = statistics.median(lengths)
        print(f"{label}: {len(lengths)} trees, median length = {median:.6f}")
        print(f"{label} lengths: {lengths[:5]}...")
        print()

if __name__ == "__main__":
    main()
