/bin/bash -lc "python - <<'PY'
from pathlib import Path
from statistics import median
import subprocess

root = Path('/workspace/.treecalc.P3EFFv')
for group in ('animals', 'fungi'):
    paths = sorted((root / group).glob('*.treefile'))
    vals = []
    for path in paths:
        out = subprocess.check_output(['/opt/agent-venv/bin/phykit', 'tree_len', str(path)], text=True).strip()
        vals.append(float(out))
    vals_sorted = sorted(vals)
    med = median(vals)
    print(group, 'n=', len(vals), 'min=', vals_sorted[0], 'q1=', median(vals_sorted[:len(vals)//2]), 'median=', repr(med), 'q3=', median(vals_sorted[(len(vals)+1)//2:]), 'max=', vals_sorted[-1])
    print('median positions=', vals_sorted[(len(vals)-1)//2:(len(vals)//2)+1])
animals = []
for path in sorted((root / 'animals').glob('*.treefile')):
    animals.append(float(subprocess.check_output(['/opt/agent-venv/bin/phykit', 'tree_len', str(path)], text=True).strip()))
fungi = []
for path in sorted((root / 'fungi').glob('*.treefile')):
    fungi.append(float(subprocess.check_output(['/opt/agent-venv/bin/phykit', 'tree_len', str(path)], text=True).strip()))
print('fungi/animals=', format(median(fungi)/median(animals), '.17g'))
print('animals/fungi=', format(median(animals)/median(fungi), '.17g'))
PY"
