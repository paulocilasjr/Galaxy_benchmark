/bin/bash -lc "python - <<'PY'
import pandas as pd, re
from pathlib import Path
trio=pd.read_excel('inputs/230215_Trio_Status.xlsx',dtype={'Sample ID':str})
trio['Sample ID']=trio['Sample ID'].str.strip()
car=set(trio.loc[trio['BLM Mutation Status'].eq('Carrier'),'Sample ID'])
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
    token=p.stem.replace('230209_Exome_GRCh38_CHIP_','')
    sid=re.match(r'(\\d+|SRR\\d+)',token).group(1)
    if sid not in car: continue
    df=pd.read_excel(p,header=1)
    print('\\n',sid,p.name,'rows',len(df))
    for c in ['Zygosity','Sequence Ontology (Combined)','Effect (Combined)','In_CHIP','Variant Allele Freq']:
        print(c,df[c].value_counts(dropna=False).to_dict())
PY"
