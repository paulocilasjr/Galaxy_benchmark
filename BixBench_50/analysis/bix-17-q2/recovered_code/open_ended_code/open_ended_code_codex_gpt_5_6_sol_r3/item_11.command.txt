/bin/bash -lc "python - <<'PY'
import pandas as pd,re,numpy as np
from pathlib import Path
trio=pd.read_excel('inputs/230215_Trio_Status.xlsx',dtype={'Sample ID':str})
trio['Sample ID']=trio['Sample ID'].str.strip()
car=set(trio.loc[trio['BLM Mutation Status'].eq('Carrier'),'Sample ID'])
rows=[]
for p in sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx')):
 token=p.stem.replace('230209_Exome_GRCh38_CHIP_',''); sid=re.match(r'(\\d+|SRR\\d+)',token).group(1)
 if sid not in car: continue
 df=pd.read_excel(p,header=1)
 so=df['Sequence Ontology (Combined)'].fillna('').astype(str)
 excl=so.str.contains(r'intron|intergenic|UTR',case=False,regex=True)
 keep=(df['Zygosity'].ne('Reference') & ~excl & pd.to_numeric(df['Variant Allele Freq'],errors='coerce').lt(.3))
 sub=df.loc[keep,['Chr:Pos','Ref/Alt','Zygosity','Variant Allele Freq','Gene Names','Sequence Ontology (Combined)','Effect (Combined)']].copy()
 rows.append((sid,len(sub)))
 print('\\n',sid,'count',len(sub))
 print(sub.to_string(index=False))
print('\\nCOUNTS',rows)
vals=[n for _,n in rows]
print('sorted',sorted(vals),'N',len(vals),'median',np.median(vals),'mean',np.mean(vals))
print('missing carriers',sorted(car-{x for x,_ in rows}))
PY"
