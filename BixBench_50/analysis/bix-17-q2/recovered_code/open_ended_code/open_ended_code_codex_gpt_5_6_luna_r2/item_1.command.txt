/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook

root = Path('/workspace')
for p in sorted(root.glob('inputs/*.xlsx')):
    wb = load_workbook(p, read_only=True, data_only=True)
    print(f'FILE\\t{p.relative_to(root)}')
    for ws in wb.worksheets:
        print(f' SHEET\\t{ws.title}\\trows={ws.max_row}\\tcols={ws.max_column}')
        rows = ws.iter_rows(min_row=1, max_row=min(ws.max_row,4), values_only=True)
        for i,row in enumerate(rows,1):
            print('  ', i, repr(row[:min(len(row),20)]))
    wb.close()
PY"
