/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook

root = Path('/workspace')
files = sorted(root.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))
print('NFILES', len(files))
for p in files[:6] + files[-2:]:
    wb = load_workbook(p, read_only=True, data_only=True)
    print(f'FILE\\t{p.name}')
    for ws in wb.worksheets:
        print(f' SHEET\\t{ws.title}\\trows={ws.max_row}\\tcols={ws.max_column}')
        for i,row in enumerate(ws.iter_rows(min_row=1, max_row=min(ws.max_row,4), values_only=True),1):
            print('  ', i, repr(row))
    wb.close()
PY"
