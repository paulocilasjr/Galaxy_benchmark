/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
from collections import Counter, defaultdict
import re, statistics

root=Path('/workspace')
files=sorted((root/'inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))
# status rows
wb=load_workbook(root/'inputs/230215_Trio_Status.xlsx',data_only=True)
ws=wb.active
status={str(row[0]).strip(): row for row in ws.iter_rows(min_row=2,values_only=True) if row[0] is not None}
wb.close()
carrier_ids=[sid for sid,row in status.items() if str(row[3]).strip().lower()=='carrier']
print('carrier IDs',len(carrier_ids),carrier_ids)
file_by_id={p.name.split('_CHIP_',1)[1].rsplit('.',1)[0].split('-',1)[0]: p for p in files}
print('files for carriers',len([x for x in carrier_ids if x in file_by_id]),'missing',[x for x in carrier_ids if x not in file_by_id])

# read data
ont_all=Counter(); zyg_all=Counter(); vaf_types=Counter(); row_lengths=Counter();
allstats={}
for p in files:
    wb=load_workbook(p,read_only=True,data_only=True); ws=wb.active; ws.reset_dimensions()
    rows=ws.iter_rows(min_row=3,max_col=33,values_only=True)
    vals=[]; vals_no_chip=[]; raw_elig=[]
    c=Counter()
    for row in rows:
        row_lengths[len(row)]+=1
        z=row[4]
        so=row[13]
        v=row[9]
        ont_all[str(so)]+=1; zyg_all[str(z)]+=1; vaf_types[type(v).__name__]+=1
        # In_CHIP encoded col AD index29
        in_chip=str(row[29]).strip().lower()=='true'
        c['rows']+=1
        if in_chip: c['chip']+=1
        try:
            vf=float(v)
        except (TypeError,ValueError):
            vf=None
        if vf is not None and vf < 0.3: c['vaflt']+=1
        if vf is not None and vf < 0.3 and in_chip:
            # Broad exclusion based on sequence ontology text as presented
            sot=str(so).strip().lower() if so is not None else ''
            zt=str(z).strip().lower() if z is not None else ''
            excluded = ('intron' in sot or 'intergenic' in sot or 'utr' in sot or zt=='reference')
            if not excluded:
                c['eligible']+=1
                raw_elig.append((row[0],row[1],row[4],row[9],row[12],row[13],row[14],row[29],row[30]))
            if not excluded and zt"'!='"'reference': vals.append(vf)
        if vf is not None and vf < 0.3 and not in_chip:
            vals_no_chip.append(row)
    wb.close()
    allstats[p]= (c, raw_elig)

print('row lengths',row_lengths)
print('all ontologies')
for k,n in ont_all.most_common(): print(n,repr(k))
print('zyg',zyg_all)
print('vaf types',vaf_types)
print('\\ncarrier stats')
counts=[]
for sid in carrier_ids:
    p=file_by_id.get(sid)
    if p is None: continue
    c,eligible=allstats[p]
    counts.append(c['eligible'])
    print(sid, p.name, dict(c), 'eligible rows:', eligible[:5])
print('counts',counts)
print('sorted',sorted(counts),'n',len(counts),'median',statistics.median(counts))
# Include missing as zero for sensitivity only
print('with missing zero n',len(carrier_ids),'median',statistics.median(counts+[0]))
PY"
