import pandas as pd
import numpy as np

# Detailed check on sample 381
df = pd.read_excel('inputs/CHIP_DP10_GQ20_PASS/230209_Exome_GRCh38_CHIP_381-PM.xlsx', header=None)
data = df.iloc[2:]

exclude_so = {
    'intron_variant',
    'intergenic_variant',
    '3_prime_UTR_variant',
    '5_prime_UTR_variant',
    '5_prime_UTR_premature_start_codon_gain_variant',
}

# Step by step
non_ref = data[data[4] != 'Reference']
print(f"Non-reference variants: {len(non_ref)}")

non_ref_coding = non_ref[~non_ref[13].isin(exclude_so)]
print(f"Non-reference, non-intronic/intergenic/UTR: {len(non_ref_coding)}")

# Check VAF
vaf = pd.to_numeric(non_ref_coding[9], errors='coerce')
print(f"\nVAF distribution (coding, non-ref):")
print(f"  NaN: {vaf.isna().sum()}")
print(f"  < 0.3: {(vaf < 0.3).sum()}")
print(f"  >= 0.3: {(vaf >= 0.3).sum()}")

# Show the variants that pass all filters
final = non_ref_coding[vaf < 0.3]
print(f"\nFinal variants ({len(final)}):")
for i, row in final.iterrows():
    print(f"  {row[0]} {row[1]} {row[12]} {row[13]} VAF={row[9]} Zyg={row[4]}")

# Also check: what's the VAF distribution of all non-ref variants in 381?
vaf_all = pd.to_numeric(non_ref[9], errors='coerce')
print(f"\nAll non-ref VAF distribution:")
print(f"  NaN: {vaf_all.isna().sum()}")
print(f"  < 0.3: {(vaf_all < 0.3).sum()}")
print(f"  >= 0.3: {(vaf_all >= 0.3).sum()}")
print(f"  VAF >= 0.3 and < 0.5: {((vaf_all >= 0.3) & (vaf_all < 0.5)).sum()}")
print(f"  VAF >= 0.5: {(vaf_all >= 0.5).sum()}")
