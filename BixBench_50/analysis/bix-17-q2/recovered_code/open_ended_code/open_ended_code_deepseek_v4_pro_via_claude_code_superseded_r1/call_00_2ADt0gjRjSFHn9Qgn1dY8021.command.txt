import pandas as pd
import os
import re
import numpy as np

# Read Trio Status
trio = pd.read_excel('inputs/230215_Trio_Status.xlsx')
print("Trio Status columns:", trio.columns.tolist())

# Get BLM carriers
carriers = trio[trio['BLM Mutation Status'] == 'Carrier']
print(f"\nBLM carriers ({len(carriers)}):")
print(carriers[['Sample ID', 'Status', 'BLM Mutation Status']].to_string())

# List all CHIP files
chip_dir = 'inputs/CHIP_DP10_GQ20_PASS'
chip_files = sorted(os.listdir(chip_dir))
print(f"\nCHIP files ({len(chip_files)}):")
for f in chip_files:
    print(f"  {f}")

# Extract sample IDs from CHIP filenames
print("\nSample IDs extracted from CHIP filenames:")
for f in chip_files:
    # Patterns: 179-P1, 184-PM, 185-PF, SRR5456220, etc.
    match = re.search(r'CHIP_(\d+)-', f)
    if match:
        print(f"  {f} -> sample {match.group(1)}")
    else:
        match = re.search(r'CHIP_(SRR\d+)', f)
        if match:
            print(f"  {f} -> sample {match.group(1)}")
