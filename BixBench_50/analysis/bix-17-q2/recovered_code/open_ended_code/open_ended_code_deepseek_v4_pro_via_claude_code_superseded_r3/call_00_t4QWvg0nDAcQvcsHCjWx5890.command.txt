from openpyxl import load_workbook
import os
import re

chip_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'
files = sorted(os.listdir(chip_dir))

# Get ALL Sequence Ontology values across ALL files
all_seq = set()
all_effect = set()

for f in files:
    wb = load_workbook(os.path.join(chip_dir, f))
    ws = wb.active
    for row in ws.iter_rows(min_row=3, values_only=True):
        if row[13]:
            all_seq.add(str(row[13]))
        if row[14]:
            all_effect.add(str(row[14]))

print("All Sequence Ontology values:")
for v in sorted(all_seq):
    print(f"  {v}")
print(f"\nAll Effect values:")
for v in sorted(all_effect):
    print(f"  {v}")

# Parse file names to sample IDs
print("\n=== File to Sample ID mapping ===")
for f in files[:5]:
    # Pattern: 230209_Exome_GRCh38_CHIP_{sample_id}.xlsx
    match = re.search(r'CHIP_(.+)\.xlsx', f)
    if match:
        sample_str = match.group(1)
        # Remove suffixes like -P1, -PM, -PF, -P, -P2
        sample_id = re.sub(r'-[A-Z0-9]+$', '', sample_str)
        print(f"  {f} -> {sample_id}")
