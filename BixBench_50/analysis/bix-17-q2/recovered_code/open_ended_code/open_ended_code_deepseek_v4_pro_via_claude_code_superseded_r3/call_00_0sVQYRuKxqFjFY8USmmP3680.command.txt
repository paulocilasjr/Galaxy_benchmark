from openpyxl import load_workbook
import os

chip_dir = '/workspace/inputs/CHIP_DP10_GQ20_PASS'
files = sorted(os.listdir(chip_dir))

# Check values using correct column indices (row 2 has headers)
seq_onto = set()
in_chip_vals = set()
zyg_vals = set()
effect_vals = set()

for f in files[:10]:
    wb = load_workbook(os.path.join(chip_dir, f))
    ws = wb.active
    # Row 2 has the column headers
    header2 = [cell.value for cell in ws[2]]
    
    for row in ws.iter_rows(min_row=3, values_only=True):
        if row[13]:
            seq_onto.add(str(row[13]))
        if row[29] is not None:
            in_chip_vals.add(str(row[29]))
        if row[4] is not None:
            zyg_vals.add(str(row[4]))
        if row[14] is not None:
            effect_vals.add(str(row[14]))

print("Zygosity values:", zyg_vals)
print()
print("In_CHIP values:", in_chip_vals)
print()

# Find UTR/intronic/intergenic values
print("=== UTR/intronic/intergenic related Sequence Ontology values ===")
for v in sorted(seq_onto):
    vl = v.lower()
    if 'utr' in vl or 'intron' in vl or 'intergenic' in vl:
        print(f"  {v}")
print(f"\nTotal unique Sequence Ontology values: {len(seq_onto)}")
print("\n=== Effect values ===")
for v in sorted(effect_vals):
    print(f"  {v}")
