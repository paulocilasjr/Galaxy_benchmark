import pandas as pd
import numpy as np
import os, glob

# Quick verification: check VAF edge cases and ensure consistency
chip_files = glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')

vaf_exact_03 = 0
total_variants = 0

for f in chip_files[:5]:  # Check first 5 files
    df = pd.read_excel(f)
    zyg_col = df.columns[4]
    vaf_col = df.columns[9]
    
    data_df = df[df[zyg_col] != 'Zygosity'].copy()
    vafs = pd.to_numeric(data_df[vaf_col], errors='coerce')
    
    # Check for VAF exactly 0.3
    exact = (vafs == 0.3).sum()
    if exact > 0:
        vaf_exact_03 += exact
        print(f"File {os.path.basename(f)}: {exact} variants with VAF == 0.3")
    total_variants += len(data_df)

print(f"\nChecked {total_variants} variants across 5 files")
print(f"Variants with VAF == 0.3: {vaf_exact_03}")
