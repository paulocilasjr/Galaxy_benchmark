/bin/bash -c "python3 - <<'PY'
import openpyxl, glob, os, re, statistics
# reproduce carrier counts and list rows
chip=set(); ws=openpyxl.load_workbook('inputs/230214_Schenz_et_al_2022_CHIP_Genes.xlsx',data_only=True).active
for r in ws.iter_rows(values_only=True):
 if r[0]: chip.add(str(r[0]))
trio=[]
ws=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx',data_only=True).active
for r in ws.iter_rows(min_row=2,values_only=True):
 sid=str(r[0]).strip().replace('\\t','').strip(); status=str(r[1]).strip(); blm=str(r[3]).strip()
 if blm=='Carrier': trio.append((sid,status,blm))
files={os.path.basename(f).split('_')[-1].split('.')[0]:f for f in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx')}
def match(sid):
 if sid in files:return sid
 cand=[t for t in files if re.fullmatch(re.escape(sid)+r'-[A-Z0-9]+',t)]
 return cand[0] if len(cand)==1 else None
excl={'intron_variant','intergenic_variant','3_prime_UTR_variant','5_prime_UTR_variant','5_prime_UTR_premature_start_codon_gain_variant'}
allc=[]
for sid,status,blm in trio:
 tok=match(sid)
 if not tok: continue
 w=openpyxl.load_workbook(files[tok],data_only=True).active
 passed=[]
 for r in w.iter_rows(min_row=3,values_only=True):
  # optionally gene list restriction no effect
  zyg=str(r[4]); ont=str(r[13])
  if zyg=='Reference': continue
  if ont in excl: continue
  try: v=float(r[9])
  except: continue
  if v<0.3:
   passed.append((r[0],r[1],zyg,v,r[12],ont,r[14],r[16],r[17]))
 print(sid,tok,'count',len(passed))
 for p in passed: print('   ',p)
 allc.append(len(passed))
print('ALL counts',allc,'median',statistics.median(allc))
PY"
