/bin/bash -c "python3 - <<'PY'
import openpyxl, glob, os, collections
for f in glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'):
 w=openpyxl.load_workbook(f,data_only=True).active
 positions=[str(r[0]) for r in w.iter_rows(min_row=3,values_only=True)]
 c=collections.Counter(positions)
 dups=[(p,n) for p,n in c.items() if n>1]
 if dups:
  print(os.path.basename(f),'total',len(positions),'unique',len(c),'dups',dups[:20])
PY"
