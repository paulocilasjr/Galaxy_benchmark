/bin/bash -c "python3 - <<'PY'
import openpyxl, glob, os, re, math, statistics, collections
# parse trio
trio=[]
ws=openpyxl.load_workbook('inputs/230215_Trio_Status.xlsx', data_only=True).active
for r in ws.iter_rows(min_row=2, values_only=True):
    sid=str(r[0]).strip().replace('\\t','').strip()
    status=str(r[1]).strip() if r[1] is not None else ''
    blm=str(r[3]).strip() if r[3] is not None else ''
    trio.append({'sid':sid,'status':status,'blm':blm})

files=sorted(glob.glob('inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))
filemap={}
for f in files:
    token=os.path.basename(f).split('_')[-1].split('.')[0]
    filemap[token]=f

def match_trio(rec):
    sid=rec['sid']
    if sid in filemap: return sid
    if re.fullmatch(r'\\d+',sid):
        cands=[t for t in filemap if re.fullmatch(re.escape(sid)+r'-[A-Z0-9]+', t)]
        if len(cands)==1: return cands[0]
        if len(cands)>1: print('multi', sid, cands)
    return None

# parse one file filtered count
excl_ont = {'intron_variant','intergenic_variant','3_prime_UTR_variant','5_prime_UTR_variant','5_prime_UTR_premature_start_codon_gain_variant'}

def count_file(f, excl_ref=True, excl_missing_vaf=True, min_vaf=0.3):
    ws=openpyxl.load_workbook(f, data_only=True).active
    n=0
    kept=[]
    for r in ws.iter_rows(min_row=3, values_only=True):
        zyg=str(r[4]).strip() if r[4] is not None else 'None'
        ont=str(r[13]).strip() if r[13] is not None else ''
        # exclude ontology terms if term present? exact entire value? Use set split
        ont_parts=set(ont.split(',')) if ont else set()
        if excl_ref and (zyg=='Reference'):
            continue
        if ont_parts & excl_ont:
            continue
        if zyg not in ('Heterozygous','Homozygous Variant'):
            # maybe skip None; count separately
            pass
        vaf=r[9]
        try:
            v=float(vaf)
        except:
            if excl_missing_vaf:
                continue
            v=None
        if v is not None and v < min_vaf:
            n+=1
            kept.append(r)
    return n,kept

def median_counts(recs):
    counts=[]
    for rec in recs:
        tok=match_trio(rec)
        if tok is None:
            #print('no file',rec)
            continue
        n,_=count_file(filemap[tok])
        counts.append(n)
        print(rec['sid'], rec['status'], rec['blm'], 'file', tok, 'count', n)
    print('N with files',len(counts),'counts',counts,'median',statistics.median(counts))
    return counts

for label, pred in [
    ('blm Carrier exact', lambda r:r['blm']=='Carrier'),
    ('blm Affected', lambda r:r['blm']=='Affected'),
    ('blm Carrier or Affected', lambda r:r['blm'] in ('Carrier','Affected')),
    ('Unaffected', lambda r:r['blm']=='Unaffected'),
]:
    print('\\n###',label)
    recs=[r for r in trio if pred(r)]
    print('records',len(recs))
    median_counts(recs)
PY"
