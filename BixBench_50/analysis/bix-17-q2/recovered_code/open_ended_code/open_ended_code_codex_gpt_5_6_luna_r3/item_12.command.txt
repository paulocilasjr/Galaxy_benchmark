/bin/bash -lc "python - <<'PY'
from pathlib import Path
from openpyxl import load_workbook
import re

# parse trio IDs preserving strings and strip whitespace/tabs
ws=load_workbook('inputs/230215_Trio_Status.xlsx',read_only=False,data_only=True).active
trio=[]
for row in ws.iter_rows(min_row=2,values_only=True):
    sid=row[0]
    if sid is not None:
        sid=str(sid).strip()
        trio.append((sid,row[1],row[3]))
print('trio rows',len(trio), 'unique',len({x[0] for x in trio}))
print('trio numeric',sum(x[0].isdigit() for x in trio),'SRR',sum(x[0].startswith('SRR') for x in trio))
files=sorted(Path('inputs/CHIP_DP10_GQ20_PASS').glob('*.xlsx'))
fileids=[]
for p in files:
    m=re.search(r'CHIP_(.+)\\.xlsx"'$'"',p.name)
    fid=m.group(1)
    fileids.append(fid)
print('files',len(fileids),'unique',len(set(fileids)), 'numeric',sum(x[0].isdigit() for x in fileids),'SRR',sum(x.startswith('SRR') for x in fileids))
print('trio IDs no file', sorted(set(s for s,_,_ in trio)-set(fileids)))
print('file IDs no trio', sorted(set(fileids)-set(s for s,_,_ in trio)))
print('carrier IDs',[(s,stat) for s,stat,b in trio if b=='Carrier'])
print('carrier no file',[(s,stat) for s,stat,b in trio if b=='Carrier' and s not in set(fileids)])
PY"
