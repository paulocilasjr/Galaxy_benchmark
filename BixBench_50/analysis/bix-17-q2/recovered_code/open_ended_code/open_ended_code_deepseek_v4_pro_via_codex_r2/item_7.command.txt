/bin/bash -lc "python3 - <<'PY'
import openpyxl, glob, os
from collections import Counter, defaultdict
files=sorted(glob.glob('/workspace/inputs/CHIP_DP10_GQ20_PASS/*.xlsx'))
# inspect all files, load normal mode maybe expensive but fine
cnt=Counter(); colvals=defaultdict(Counter); nrows=[]
for f in files:
    wb=openpyxl.load_workbook(f, read_only=False, data_only=True)
    ws=wb.active
    rows=list(ws.iter_rows(min_row=3, values_only=True))
    nrows.append(len(rows))
    for r in rows:
        # indexes 0 based from row: 4 zygosity, 9 VAF, 12 genes,13 SO, 29 In_CHIP, 30 gene list?, 31 false?
        cnt['rows']+=1
        for idx,name in [(4,'zygosity'),(9,'vaf'),(12,'gene_names'),(13,'so'),(29,'in_chip'),(30,'match_gene'),(31,'match_bool')]:
            val=r[idx] if idx<len(r) else None
            # truncate values
            if name in ('gene_names','so'):
                val=str(val)[:80]
            colvals[name][val]+=1
print('files',len(files),'total rows',cnt['rows'])
print('row counts min max sample:',min(nrows),max(nrows), Counter(nrows).most_common(10))
for name in ['zygosity','in_chip','match_gene','match_bool']:
 print('\\n',name)
 for k,v in colvals[name].most_common(30): print(repr(k),v)
print('\\nvaf sample unique/range')
vs=[k for k,v in colvals['vaf'].items()]
print(len(vs), 'min',min([x for x in vs if isinstance(x,(int,float))],default=None),'max',max([x for x in vs if isinstance(x,(int,float))],default=None))
print('vaf types',Counter(type(k).__name__ for k in colvals['vaf']).most_common())
print('SO counts:')
for k,v in colvals['so'].most_common(30): print(repr(k),v)
print('gene names unique',len(colvals['gene_names']))
print('top genes')
for k,v in colvals['gene_names'].most_common(80): print(repr(k),v)
PY"
