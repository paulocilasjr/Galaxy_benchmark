/bin/bash -c "python3 - <<'PY'
import csv, os
from collections import defaultdict
p='/workspace/inputs/Swarm_1.csv'
areas=defaultdict(list); circs=defaultdict(list)
with open(p, newline='') as f:
    r=csv.DictReader(f)
    for row in r:
        g=row['Genotype']
        areas[g].append(float(row['Area']))
        circs[g].append(float(row['Circularity']))
means={g: sum(v)/len(v) for g,v in areas.items()}
for g in sorted(means):
    print(g, means[g], sum(circs[g])/len(circs[g]))
geno=max(means, key=means.get)
val=sum(circs[geno])/len(circs[geno])
print('largest mean area genotype:', geno)
print('mean circularity:', repr(val))
os.makedirs('/workspace/codex_output', exist_ok=True)
with open('/workspace/codex_output/answer.txt','w') as f:
    f.write(str(val)+'\\n')
PY
cat /workspace/codex_output/answer.txt"
