import csv

rows = []
with open('inputs/Swarm_1.csv') as f:
    reader = csv.DictReader(f)
    for r in reader:
        rows.append(r)

# Group by Genotype, compute mean Area
from collections import defaultdict
geno_areas = defaultdict(list)
for r in rows:
    geno_areas[r['Genotype']].append(float(r['Area']))

geno_mean_area = {g: sum(a)/len(a) for g, a in geno_areas.items()}
print('Mean areas by genotype:')
for g, m in sorted(geno_mean_area.items(), key=lambda x: -x[1]):
    print(f'  {g}: {m}')

top_geno = max(geno_mean_area, key=geno_mean_area.get)
print(f'\nGenotype with largest mean area: {top_geno} (mean area = {geno_mean_area[top_geno]})')

# Mean circularity for that genotype
circ_vals = [float(r['Circularity']) for r in rows if r['Genotype'] == top_geno]
mean_circ = sum(circ_vals) / len(circ_vals)
print(f'Mean circularity for {top_geno}: {mean_circ}')
