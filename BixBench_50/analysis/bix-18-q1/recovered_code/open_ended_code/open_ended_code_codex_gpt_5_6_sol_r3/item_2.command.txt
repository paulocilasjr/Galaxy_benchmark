/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
rows=list(csv.DictReader(open('inputs/Swarm_1.csv', newline='')))
g=defaultdict(list)
for row in rows:
    g[row['Genotype']].append(row)
for genotype, rs in g.items():
    print(genotype, sum(float(x['Area']) for x in rs)/len(rs), sum(float(x['Circularity']) for x in rs)/len(rs))
PY"
