python3 -c "
import csv

with open('inputs/Swarm_1.csv') as f:
    reader = csv.DictReader(f)
    rows = list(reader)

# Group by genotype and compute mean area
from collections import defaultdict
genotype_areas = defaultdict(list)
genotype_circ = defaultdict(list)

for row in rows:
    gt = row['Genotype']
    genotype_areas[gt].append(float(row['Area']))
    genotype_circ[gt].append(float(row['Circularity']))

# Find genotype with largest mean area
mean_areas = {gt: sum(v)/len(v) for gt, v in genotype_areas.items()}
for gt, ma in sorted(mean_areas.items(), key=lambda x: -x[1]):
    print(f'{gt}: mean_area={ma:.4f}, mean_circ={sum(genotype_circ[gt])/len(genotype_circ[gt]):.6f}')

best = max(mean_areas, key=mean_areas.get)
print(f'\nGenotype with largest mean area: {best}')
print(f'Mean circularity: {sum(genotype_circ[best])/len(genotype_circ[best])}')
