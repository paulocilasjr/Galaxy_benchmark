/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict

with open('inputs/Swarm_1.csv', newline='') as f:
    rows = list(csv.DictReader(f))

groups = defaultdict(list)
for row in rows:
    groups[row['Genotype']].append(row)

summary = {}
for genotype, rs in groups.items():
    mean_area = sum(float(r['Area']) for r in rs) / len(rs)
    mean_circularity = sum(float(r['Circularity']) for r in rs) / len(rs)
    summary[genotype] = (mean_area, mean_circularity)

for genotype, (area, circularity) in summary.items():
    print(f'{genotype}: mean area={area}, mean circularity={circularity}')

winner = max(summary, key=lambda g: summary[g][0])
print('winner=', winner)
print('answer=', format(summary[winner][1], 'g'))
PY"
