/bin/bash -lc "python - <<'PY'
import csv
from collections import defaultdict
p='inputs/Swarm_1.csv'
groups=defaultdict(list)
with open(p, newline='') as f:
    for row in csv.DictReader(f):
        groups[row['Genotype']].append(row)
for genotype, rows in groups.items():
    mean_area=sum(float(r['Area']) for r in rows)/len(rows)
    mean_circ=sum(float(r['Circularity']) for r in rows)/len(rows)
    print(genotype, len(rows), mean_area, mean_circ)
PY"
