import csv

rows = []
with open('inputs/Swarm_1.csv') as f:
    reader = csv.DictReader(f)
    for r in reader:
        rows.append(r)

# Group by genotype
from collections import defaultdict
area_by_geno = defaultdict(list)
circ_by_geno = defaultdict(list)

for r in rows:
    geno = r['Genotype']
    area_by_geno[geno].append(float(r['Area']))
    circ_by_geno[geno].append(float(r['Circularity']))

print('Mean Area by Genotype:')
for geno, areas in area_by_geno.items():
    mean_a = sum(areas) / len(areas)
    print(f'  {geno}: {mean_a}')

# Find genotype with largest mean area
mean_areas = {g: sum(a)/len(a) for g, a in area_by_geno.items()}
largest_geno = max(mean_areas, key=mean_areas.get)
print(f'\nLargest mean area: {largest_geno} ({mean_areas[largest_geno]})')

# Mean circularity for that genotype
circs = circ_by_geno[largest_geno]
mean_circ = sum(circs) / len(circs)
print(f'Mean circularity for {largest_geno}: {mean_circ}')
