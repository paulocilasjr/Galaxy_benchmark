/bin/bash -lc "python - <<'PY'
import csv, os
from collections import defaultdict
path='/workspace/inputs/Swarm_1.csv'
areas=defaultdict(list)
circs=defaultdict(list)
with open(path, newline='') as f:
    for row in csv.DictReader(f):
        g=row['Genotype']
        areas[g].append(float(row['Area']))
        circs[g].append(float(row['Circularity']))
mean_area={g:sum(v)/len(v) for g,v in areas.items()}
best=max(mean_area, key=mean_area.get)
answer=sum(circs[best])/len(circs[best])
os.makedirs('/workspace/codex_output', exist_ok=True)
with open('/workspace/codex_output/answer.txt','w') as f:
    f.write(f'{answer:g}\\n')
print(best, mean_area[best], answer)
PY
cat /workspace/codex_output/answer.txt"
