/bin/bash -lc "python - <<'PY'
import zipfile, os
sets={}
for group, files in [('animals',['Animals_Cele','Animals_Ggal','Animals_Mmus','Animals_Xlae']),('fungi',['Fungi_Ncra','Fungi_Scer','Fungi_Spom','Fungi_Umay'])]:
 u=set(); counts={}
 for sp in files:
  with zipfile.ZipFile(f'inputs/{sp}.busco.zip') as z:
   vals=set()
   for n in z.namelist():
    pref='run_eukaryota_odb10/busco_sequences/single_copy_busco_sequences/'
    if n.startswith(pref) and n.endswith('.faa'):
     vals.add(os.path.basename(n)[:-4])
   counts[sp]=len(vals); u |= vals
  
 sets[group]=u
 print(group,counts,'union',len(u))
for fn, key in [('inputs/scogs_animals.zip','animals'),('inputs/scogs_fungi.zip','fungi')]:
 with zipfile.ZipFile(fn) as z: s={os.path.basename(n)[:-10] for n in z.namelist() if n.endswith('.faa.mafft')}
 print(fn,'SCOG',len(s),'SCOG-union',len(s-sets[key]),'union-SCOG',len(sets[key]-s))
 print('sample diff',sorted(sets[key]-s)[:10])
PY"
