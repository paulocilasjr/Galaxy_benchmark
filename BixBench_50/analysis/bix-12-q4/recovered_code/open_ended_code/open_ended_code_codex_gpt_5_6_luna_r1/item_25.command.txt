import zipfile,os

def seqcount(fn):
 out={}
 with zipfile.ZipFile(fn) as z:
  for n in z.namelist():
   if n.endswith('.mafft'):
    ss=[]; seqs=[]
    for l in z.read(n).decode().splitlines():
     if l.startswith('>'):
      if ss: seqs.append(''.join(ss))
      ss=[]
     else:ss.append(l.strip())
    if ss:seqs.append(''.join(ss))
    out[n]=len(seqs)
 return out
A=seqcount('inputs/scogs_animals.zip'); F=seqcount('inputs/scogs_fungi.zip')
print('animal 4',sum(n==4 for n in A.values()),'fungi 4',sum(n==4 for n in F.values()))
print('A4 in F4',sum(A[k]==4 and F.get(k)==4 for k in A),'A4 F<4',sum(A[k]==4 and F.get(k,0)<4 for k in A))
print('F4 in A4',sum(F[k]==4 and A.get(k)==4 for k in F))
print('A4 ids',sorted(k.split('.')[0] for k,v in A.items() if v==4))
PY"
