/bin/bash -lc "python - <<'PY'
import zipfile,collections

def load(fn):
 out=[]
 with zipfile.ZipFile(fn) as z:
  for name in z.namelist():
   if name.endswith('.mafft'):
    ss=[]; seqs=[]
    for l in z.read(name).decode().splitlines():
     if l.startswith('>'):
      if ss: seqs.append(''.join(ss))
      ss=[]
     else: ss.append(l.strip())
    if ss: seqs.append(''.join(ss))
    out.append((name,seqs))
 return out
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 print('\\n',fn)
 for name,seqs in load(fn):
  pi_m=[]; pi_g=[]; comp=[]
  for c in zip(*seqs):
   cm=collections.Counter(x for x in c if x"'!='"'-')
   cg=collections.Counter(c)
   im=sum(n>=2 for n in cm.values())>=2
   ig=sum(n>=2 for n in cg.values())>=2
   if im: pi_m.append(c)
   if ig: pi_g.append(c)
   if not any(x=='-' for x in c): comp.append(c)
  if len(pi_m)"'!=len(pi_g):
   print(name,'"'n',len(seqs),'L',len(seqs[0]),'missingPI',len(pi_m),'gapPI',len(pi_g),'extra',len(pi_g)-len(pi_m),'examples gap-only', [''.join(c) for c in pi_g if c not in pi_m][:5])
PY"
