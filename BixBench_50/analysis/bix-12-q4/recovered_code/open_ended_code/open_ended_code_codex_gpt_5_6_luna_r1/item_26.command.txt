# quick compute variants for complete/common scopes
import zipfile,collections

def load(fn):
 out={}
 with zipfile.ZipFile(fn) as z:
  for n in z.namelist():
   if n.endswith('.mafft'):
    seqs=[]; cur=None
    for l in z.read(n).decode().splitlines():
     if l.startswith('>'):
      if cur is not None: seqs.append(''.join(cur))
      cur=[]
     else: cur.append(l.strip())
    seqs.append(''.join(cur)); out[n]=seqs
 return out
A=load('inputs/scogs_animals.zip');F=load('inputs/scogs_fungi.zip')
def p(seqs, gap):
 cols=zip(*seqs); pi=0; L=len(seqs[0])
 for c in cols:
  vals=list(c) if gap else [x for x in c if x"'!='"'-']
  co=collections.Counter(vals)
  pi += sum(v>=2 for v in co.values())>=2
 return pi/L*100

def U(x,y):
 z=sorted([(v,0) for v in x]+[(v,1) for v in y]);r=[];i=0
 while i<len(z):
  j=i
  while j<len(z) and z[j][0]==z[i][0]:j+=1
  r += [(i+j+2)/2]*(j-i);i=j
 return sum(r[k] for k,(v,g) in enumerate(z) if g==0)-len(x)*(len(x)+1)/2
for idsname,ids in [('all4',set(k for k,v in A.items() if len(v)==4),),('common4',set(k for k,v in A.items() if len(v)==4 and len(F.get(k,[]))==4))]:
 for gap in [False,True]:
  av=[p(A[k],gap) for k in ids]; fv=[p(F[k],gap) for k in ids]
  print(idsname,'gap',gap,'n',len(av),'U',U(av,fv),'A',av,'F',fv)
PY"
