/bin/bash -lc "python - <<'PY'
import zipfile, collections, re
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\n===',fn)
    with zipfile.ZipFile(fn) as z:
        rows=[]
        for name in z.namelist():
            if not name.endswith('.mafft'): continue
            text=z.read(name).decode()
            seqs=[]; h=None; s=[]
            for line in text.splitlines():
                if line.startswith('>'):
                    if h is not None: seqs.append((h,''.join(s)))
                    h=line[1:].strip(); s=[]
                else: s.append(line.strip())
            if h is not None: seqs.append((h,''.join(s)))
            rows.append((name,len(seqs),len(seqs[0][1]) if seqs else 0,tuple(x[0] for x in seqs)))
        print('alignments',len(rows))
        print('record counts',collections.Counter(x[1] for x in rows))
        print('length count',collections.Counter(x[2] for x in rows).most_common(10))
        print('taxon sets')
        print(collections.Counter(x[3] for x in rows).most_common())
        print('first 5',*rows[:5],sep='\\n')
PY"
