/bin/bash -lc "python - <<'PY'
import zipfile
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 with zipfile.ZipFile(fn) as z:
  ids={n[:-5] for n in z.namelist() if n.endswith('.mafft')}
 print(fn, len(ids), sorted(ids)[:5],sorted(ids)[-5:])
 if fn.endswith('animals.zip'): a=ids
 else: f=ids
print('intersection',len(a&f),'animals-only',len(a-f),'fungi-only',len(f-a))
print('animals-only',sorted(a-f))
print('fungi-only',sorted(f-a))
PY"
