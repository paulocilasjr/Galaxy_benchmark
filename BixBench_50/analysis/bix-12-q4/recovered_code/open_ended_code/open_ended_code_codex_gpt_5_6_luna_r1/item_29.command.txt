import zipfile,collections

def load(fn):
 out={}
 with zipfile.ZipFile(fn) as z:
  for n in z.namelist():
   if n.endswith('.mafft'):
    seq=[];cur=None
    for l in z.read(n).decode().splitlines():
     if l.startswith('>'):
      if cur is not None:seq.append(''.join(cur))
      cur=[]
     else:cur.append(l.strip())
    seq.append(''.join(cur));out[n]=seq
 return out
def pct(s, gap=True, complete_den=False):
 cols=list(zip(*s))
 if complete_den: cols=[c for c in cols if not any(x=='-' for x in c)]
 pi=0
 for c in cols:
  vals=list(c) if gap else [x for x in c if x"'!='"'-']
  co=collections.Counter(vals); pi += sum(v>=2 for v in co.values())>=2
 return 100*pi/len(cols) if cols else 0
def U(a,b):return sum((x>y)+.5*(x==y) for x in a for y in b)

for scope,ids in [('all4',set(k for k,v in A.items() if len(v)==4)),('common4',set(k for k,v in A.items() if len(v)==4 and len(F.get(k,[]))==4))]:

  for comp in [False,True]:
   a=[pct(A[k],gap,comp) for k in ids]; b=[pct(F[k],gap,comp) for k in ids]
