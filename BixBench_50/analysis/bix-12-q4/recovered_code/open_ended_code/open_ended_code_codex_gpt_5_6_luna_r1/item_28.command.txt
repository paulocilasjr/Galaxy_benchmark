/bin/bash -lc "python - <<'PY'
import zipfile,collections

def load(fn):
 out=[]
 with zipfile.ZipFile(fn) as z:
  for n in z.namelist():
   if n.endswith('.mafft'):
    seq=[];cur=None
    for l in z.read(n).decode().splitlines():
     if l.startswith('>'):
      if cur is not None:seq.append(''.join(cur))
      cur=[]
     else:cur.append(l.strip())
    seq.append(''.join(cur));out.append(seq)
 return out
def pct(s, ignore_gap=True):
 pi=0
 for c in zip(*s):
  vals=[x for x in c if x"'!='"'-'] if ignore_gap else list(c)
  co=collections.Counter(vals)
  pi += sum(v>=2 for v in co.values())>=2
 return 100*pi/len(s[0])
def Upair(a,b):
 return sum((x>y)+(0.5*(x==y)) for x in a for y in b)
def Urank(a,b):
 z=sorted([(x,0) for x in a]+[(y,1) for y in b])
 R=0.;i=0
 while i<len(z):
  j=i+1
  while j<len(z) and z[j][0]==z[i][0]:j+=1
  rank=(i+1+j)/2
  R += rank*sum(g==0 for _,g in z[i:j])
  i=j
 return R-len(a)*(len(a)+1)/2
A=load('inputs/scogs_animals.zip');F=load('inputs/scogs_fungi.zip')
for ig in [True,False]:
 a=[pct(s,ig) for s in A];b=[pct(s,ig) for s in F]
 print('ignore',ig,'pair',Upair(a,b),'rank',Urank(a,b),'prod',len(a)*len(b),'min',min(Upair(a,b),len(a)*len(b)-Upair(a,b)))
# denominator complete for all alignments, compare
for ig in [True,False]:
 def pc(s):
  cols=[c for c in zip(*s) if not (ig and any(x=='-' for x in c))]
  pi=0
  for c in cols:
   vals=[x for x in c if not (ig and x=='-')]
   co=collections.Counter(vals);pi+=sum(v>=2 for v in co.values())>=2
  return 100*pi/len(cols) if cols else 0
 a=[pc(s) for s in A];b=[pc(s) for s in F]
 print('complete denominator ignore',ig,'U',Upair(a,b),'rank',Urank(a,b))
PY"
