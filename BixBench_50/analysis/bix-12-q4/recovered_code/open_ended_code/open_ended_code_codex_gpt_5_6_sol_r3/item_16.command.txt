/bin/bash -lc "python - <<'PY'
# Recalculate U for all fungi versus shared-ID-only fungi for scope sensitivity.
from zipfile import ZipFile
from collections import Counter
from fractions import Fraction

def get(p):
 d={}
 with ZipFile(p) as z:
  for n in z.namelist():
   if not n.endswith('.faa.mafft'): continue
   seq=[]; cur=[]
   for line in z.read(n).decode().splitlines():
    if line.startswith('>'):
     if cur: seq.append(''.join(cur));cur=[]
    else:cur.append(line.strip())
   if cur:seq.append(''.join(cur))
   k=sum(sum(v>=2 for v in Counter(x for x in col if x"'!='"'-').values())>=2 for col in zip(*seq))
   d[n.removesuffix('.faa.mafft')]=Fraction(k,len(seq[0]))
 return d
a=get('inputs/scogs_animals.zip');f=get('inputs/scogs_fungi.zip')
def U(x,y):return sum(Fraction(i>j)+Fraction(i==j,2) for i in x for j in y)
print('all',U(a.values(),f.values()))
print('shared',U(a.values(),[f[k] for k in a]),'complement',len(a)**2-U(a.values(),[f[k] for k in a]))
print('fungi extra vals',[(k,float(f[k]*100)) for k in f.keys()-a.keys()])
PY"
