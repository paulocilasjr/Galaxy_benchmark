/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
for zpath in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with ZipFile(zpath) as z:
        names=[x.filename for x in z.infolist() if not x.is_dir()]
        aligned=[n for n in names if n.endswith('.faa.mafft')]
        raw=[n for n in names if n.endswith('.faa')]
        print(zpath, 'members',len(names),'aligned',len(aligned),'raw',len(raw), 'unique_aligned_ids',len({n.removesuffix('.faa.mafft') for n in aligned}))
        print('aligned-only ids', len({n.removesuffix('.faa.mafft') for n in aligned}-{n.removesuffix('.faa') for n in raw}), 'raw-only ids',len({n.removesuffix('.faa') for n in raw}-{n.removesuffix('.faa.mafft') for n in aligned}))
        for n in aligned[:3]:
            s=z.read(n).decode()
            heads=[x for x in s.splitlines() if x.startswith('>')]
            seqs=[]
            cur=''
            for line in s.splitlines():
                if line.startswith('>'):
                    if cur: seqs.append(cur)
                    cur=''
                else: cur += line.strip()
            if cur: seqs.append(cur)
            print(n, len(seqs), sorted(set(map(len,seqs))), heads)
PY"
