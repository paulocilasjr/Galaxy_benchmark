# Check a real animal alignment to understand gap patterns
def parse_fasta(filepath):
    seqs = []
    cur_h, cur_s = None, []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if cur_h:
                    seqs.append((cur_h, ''.join(cur_s)))
                cur_h = line[1:]
                cur_s = []
            else:
                cur_s.append(line.upper())
        if cur_h:
            seqs.append((cur_h, ''.join(cur_s)))
    return seqs

# Pick a 4-species OG
seqs = parse_fasta('/workspace/tmp/animals/1014314at2759.faa.mafft')
print(f'Species: {len(seqs)}')
length = len(seqs[0][1])
print(f'Alignment length: {length}')

# Compute PIS with two gap treatments
pis_gap_state = 0
pis_gap_missing = 0
total_cols = length

for col in range(length):
    chars = [s[1][col] for s in seqs]
    
    # With gaps as state
    freq = {}
    for c in chars:
        freq[c] = freq.get(c, 0) + 1
    inf_gap_state = sum(1 for v in freq.values() if v >= 2)
    if inf_gap_state >= 2:
        pis_gap_state += 1
    
    # With gaps as missing data
    freq2 = {}
    for c in chars:
        if c != '-':
            freq2[c] = freq2.get(c, 0) + 1
    inf_gap_missing = sum(1 for v in freq2.values() if v >= 2)
    if inf_gap_missing >= 2:
        pis_gap_missing += 1

print(f'PIS (gaps as state): {pis_gap_state}/{total_cols} = {100*pis_gap_state/total_cols:.2f}%')
print(f'PIS (gaps as missing): {pis_gap_missing}/{total_cols} = {100*pis_gap_missing/total_cols:.2f}%')

# Also check gap fraction
gap_cols = sum(1 for col in range(length) if any(s[1][col]=='-' for s in seqs))
print(f'Columns with gaps: {gap_cols}/{total_cols} = {100*gap_cols/total_cols:.2f}%')
