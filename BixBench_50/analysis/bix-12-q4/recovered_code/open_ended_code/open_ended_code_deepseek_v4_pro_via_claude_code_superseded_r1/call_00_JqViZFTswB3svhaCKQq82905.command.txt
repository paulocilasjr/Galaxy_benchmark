# Validate PIS calculation with a small example
# Consider 3 sequences, alignment length 4
seqs = [
    ('s1', 'ACGT'),
    ('s2', 'ACGT'),
    ('s3', 'AAGT'),
]
# Col 1: A=3 -> 1 state >=2, NOT inf
# Col 2: C=2, A=1 -> 1 state >=2, NOT inf  
# Col 3: G=3 -> NOT inf
# Col 4: T=3 -> NOT inf
# PIS = 0

# Now with informative sites:
seqs2 = [
    ('s1', 'ACGT'),
    ('s2', 'ACGG'),
    ('s3', 'AAGT'),
    ('s4', 'AAGG'),
]
# Col 1: A=4 -> NOT inf
# Col 2: C=2, A=2 -> 2 states >=2, INF
# Col 3: G=4 -> NOT inf
# Col 4: T=2, G=2 -> 2 states >=2, INF
# PIS = 2/4 = 50%

# Let's manually check
for col in range(4):
    chars = [s[1][col] for s in seqs2]
    freq = {}
    for c in chars:
        freq[c] = freq.get(c, 0) + 1
    inf = sum(1 for v in freq.values() if v >= 2)
    print(f'Col {col}: {chars}, freq={freq}, inf_chars={inf} -> {\"INF\" if inf >= 2 else \"not\"}')
print('Expected: cols 2 and 4 are inf')
