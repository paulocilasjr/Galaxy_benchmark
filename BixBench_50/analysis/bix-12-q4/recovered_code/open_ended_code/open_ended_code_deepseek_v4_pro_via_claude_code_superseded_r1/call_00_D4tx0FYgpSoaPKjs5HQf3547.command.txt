# Compare gap-as-state vs gap-as-missing for all alignments
import os

def parse_fasta(filepath):
    seqs = []
    cur_h, cur_s = None, []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if cur_h:
                    seqs.append((cur_h, ''.join(cur_s)))
                cur_h = line[1:]
                cur_s = []
            else:
                cur_s.append(line.upper())
        if cur_h:
            seqs.append((cur_h, ''.join(cur_s)))
    return seqs

def pis_pct(seqs, gaps_as_missing=True):
    if len(seqs) < 2:
        return 0.0
    chars_seqs = [s for _, s in seqs if len(s) > 0]
    if not chars_seqs:
        return 0.0
    n = len(chars_seqs[0])
    pis = 0
    for col in range(n):
        chars = [s[col] for s in chars_seqs if col < len(s)]
        freq = {}
        for c in chars:
            if gaps_as_missing and c == '-':
                continue
            freq[c] = freq.get(c, 0) + 1
        inf = sum(1 for v in freq.values() if v >= 2)
        if inf >= 2:
            pis += 1
    return (pis / n) * 100.0 if n > 0 else 0.0

# Compare a few files
for d in ['/workspace/tmp/animals', '/workspace/tmp/fungi']:
    print(f'\\n=== {d} ===')
    mafft_files = sorted([f for f in os.listdir(d) if f.endswith('.mafft')])[:5]
    for mf in mafft_files:
        seqs = parse_fasta(os.path.join(d, mf))
        ps = pis_pct(seqs, gaps_as_missing=False)
        pm = pis_pct(seqs, gaps_as_missing=True)
        print(f'  {mf}: {len(seqs)} spp, gap_state={ps:.2f}%, gap_missing={pm:.2f}%')
