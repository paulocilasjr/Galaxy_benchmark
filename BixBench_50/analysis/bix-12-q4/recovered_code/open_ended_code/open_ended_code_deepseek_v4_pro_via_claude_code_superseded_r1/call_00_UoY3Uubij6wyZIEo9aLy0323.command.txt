def parse_fasta(filepath):
    seqs = []
    cur_h, cur_s = None, []
    with open(filepath) as f:
        for line in f:
            line = line.strip()
            if line.startswith('>'):
                if cur_h:
                    seqs.append((cur_h, ''.join(cur_s)))
                cur_h = line[1:]
                cur_s = []
            else:
                cur_s.append(line.upper())
        if cur_h:
            seqs.append((cur_h, ''.join(cur_s)))
    return seqs
