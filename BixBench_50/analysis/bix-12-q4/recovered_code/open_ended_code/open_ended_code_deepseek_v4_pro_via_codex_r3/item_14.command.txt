from collections import Counter

def countseq(p):
    return sum(1 for l in p.read_text().splitlines() if l.startswith('>'))
for d in ['tmp/scogs_a','tmp/scogs_f']:
    c=Counter(countseq(p) for p in Path(d).glob('*.mafft'))
    print(d,c)
