/bin/bash -c "cd /workspace && python3 - <<'PY'
from pathlib import Path
from collections import Counter

def read_fasta(p):
    seqs=[]; name=None; cur=[]
    for line in p.read_text().splitlines():
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(cur)))
            name=line[1:].strip(); cur=[]
        else: cur.append(line.strip())
    if name is not None: seqs.append((name,''.join(cur)))
    return seqs

def calc(p, gap_as_state=False):
    seqs=read_fasta(p); n=len(seqs); L=len(seqs[0][1])
    pis=0
    for j in range(L):
        col=[s[j] for n,s in seqs]
        if gap_as_state: syms=col
        else: syms=[c for c in col if c not in ('-','.')]
        counts=Counter(syms)
        states=[c for c,v in counts.items() if v>=2]
        if len(set(states))>=2: pis+=1
    return pis/L*100

def mw_u(A,B):
    # U for A vs B: number of A_i < B_j plus half ties? scipy uses rank sum ties correction? statistic U = R1 - n1*(n1+1)/2 using average ranks.
    vals=sorted([(x,0) for x in A]+[(y,1) for y in B], key=lambda t:t[0])
    n=len(vals); ranks=[0]*n
    i=0
    while i<n:
        j=i
        while j<n and vals[j][0]==vals[i][0]: j+=1
        avg=(i+1 + j)/2
        for k in range(i,j): ranks[k]=avg
        i=j
    R1=sum(ranks[k] for k in range(n) if vals[k][1]==0)
    n1=len(A)
    U=R1-n1*(n1+1)/2
    return U

for gap_as_state in [False, True]:
  print('gap_as_state',gap_as_state)
  for d in ['tmp/scogs_a','tmp/scogs_f']:
      vals=[calc(p, gap_as_state=gap_as_state) for p in Path(d).glob('*.mafft')]
      vals.sort()
      print(d,'n',len(vals),'min',min(vals),'median',vals[len(vals)//2],'max',max(vals),'mean',sum(vals)/len(vals))
  A=sorted(calc(p, gap_as_state=gap_as_state) for p in Path('tmp/scogs_a').glob('*.mafft'))
  F=sorted(calc(p, gap_as_state=gap_as_state) for p in Path('tmp/scogs_f').glob('*.mafft'))
  print('U A,F', mw_u(A,F), 'U F,A', mw_u(F,A), 'n1*n2',len(A)*len(F))
PY"
