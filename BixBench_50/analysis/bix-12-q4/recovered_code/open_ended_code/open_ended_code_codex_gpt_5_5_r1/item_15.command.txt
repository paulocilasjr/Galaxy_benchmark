/bin/bash -lc "python3 - <<'PY'
import zipfile, collections

def read_heads(b):
    return [line[1:].strip() for line in b.decode().splitlines() if line.startswith('>')]
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    species=collections.Counter(); combos=collections.Counter()
    with zipfile.ZipFile(fn) as z:
        for name in z.namelist():
            if not name.endswith('.faa.mafft'): continue
            heads=read_heads(z.read(name))
            labs=[]
            for h in heads:
                # show likely prefix before separators
                lab=h.split('|')[0].split()[0]
                labs.append(lab)
                species[lab]+=1
            combos[tuple(sorted(labs))]+=1
    print('\\n',fn,'species',species)
    print('top combos')
    for combo,c in combos.most_common(10): print(c, combo)
PY"
