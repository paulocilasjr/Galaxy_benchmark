/bin/bash -lc "python3 - <<'PY'
import zipfile, collections, statistics

def read(b):
    seqs=[]; chunks=[]; name=None
    for line in b.decode().splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(chunks))
            name=line[1:]; chunks=[]
        else: chunks.append(line)
    if name is not None: seqs.append(''.join(chunks))
    return seqs

def ids_with_n(fn, n):
    out={}
    with zipfile.ZipFile(fn) as z:
        for name in z.namelist():
            if name.endswith('.faa.mafft'):
                base=name[:-10]
                seqs=read(z.read(name))
                if len(seqs)==n: out[base]=seqs
    return out

def pi(seqs):
    c=0
    for col in zip(*seqs):
        cnt=collections.Counter(ch for ch in col if ch not in '-?.')
        if sum(v>=2 for v in cnt.values()) >= 2: c += 1
    return 100*c/len(seqs[0])
def U(x,y): return sum(1 if a>b else .5 if a==b else 0 for a in x for b in y)
A=ids_with_n('inputs/scogs_animals.zip',4)
F=ids_with_n('inputs/scogs_fungi.zip',4)
common=sorted(set(A)&set(F))
xa=[pi(A[i]) for i in common]; yf=[pi(F[i]) for i in common]
print(len(A),len(F),len(common))
print('U_animals',U(xa,yf),'U_fungi',len(xa)*len(yf)-U(xa,yf),'means',statistics.mean(xa),statistics.mean(yf),'medians',statistics.median(xa),statistics.median(yf))
PY"
