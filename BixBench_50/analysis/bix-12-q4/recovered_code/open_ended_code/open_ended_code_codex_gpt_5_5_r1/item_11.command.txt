/bin/bash -lc "python3 - <<'PY'
import zipfile, collections

def read_fasta(text):
    seqs=[]; name=None; chunks=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chunks)))
            name=line[1:].split()[0]; chunks=[]
        else:
            chunks.append(line)
    if name is not None: seqs.append((name,''.join(chunks)))
    return seqs
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    nseq_counts=collections.Counter(); len_counts=[]; bad=[]
    with zipfile.ZipFile(fn) as z:
        for name in sorted(i.filename for i in z.infolist() if i.filename.endswith('.faa.mafft')):
            seqs=read_fasta(z.read(name).decode())
            nseq_counts[len(seqs)] += 1
            lens={len(s) for _,s in seqs}
            if len(lens)"'!=1: bad.append((name,lens))
            else: len_counts.append(next(iter(lens)))
    print(fn, '"'nseq_counts', dict(nseq_counts), 'bad', len(bad), 'min_len', min(len_counts), 'max_len', max(len_counts))
    print('first_bad', bad[:3])
PY"
