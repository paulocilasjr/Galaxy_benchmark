/bin/bash -lc "python3 - <<'PY'
import zipfile, collections

def read_fasta_bytes(b):
    seqs=[]; name=None; chunks=[]
    for raw in b.decode().splitlines():
        line=raw.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(chunks))
            name=line[1:]; chunks=[]
        else:
            chunks.append(line)
    if name is not None: seqs.append(''.join(chunks))
    return seqs

def pi_pct(seqs):
    n=0
    for col in zip(*seqs):
        counts=collections.Counter(c for c in col if c not in '-?.')
        if sum(v >= 2 for v in counts.values()) >= 2:
            n += 1
    return 100 * n / len(seqs[0])

def vals(path):
    out=[]
    with zipfile.ZipFile(path) as z:
        for name in sorted(z.namelist()):
            if name.endswith('.faa.mafft'):
                out.append(pi_pct(read_fasta_bytes(z.read(name))))
    return out
A=vals('inputs/scogs_animals.zip')
F=vals('inputs/scogs_fungi.zip')
U_pair=sum(1 if a>f else 0.5 if a==f else 0 for a in A for f in F)
combined=sorted([(v,0) for v in A] + [(v,1) for v in F], key=lambda x:x[0])
ranks=[None]*len(combined)
i=0
while i<len(combined):
    j=i+1
    while j<len(combined) and combined[j][0]==combined[i][0]: j+=1
    avg=(i+1+j)/2
    for k in range(i,j): ranks[k]=avg
    i=j
rank_sum_A=sum(r for r,(v,g) in zip(ranks,combined) if g==0)
U_rank=rank_sum_A - len(A)*(len(A)+1)/2
print(len(A), len(F), U_pair, U_rank, len(A)*len(F)-U_pair)
PY"
