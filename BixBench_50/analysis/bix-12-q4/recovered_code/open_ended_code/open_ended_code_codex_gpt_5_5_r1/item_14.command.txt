/bin/bash -lc "python3 - <<'PY'
import zipfile
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\n',fn)
    with zipfile.ZipFile(fn) as z:
        for name in [n for n in z.namelist() if n.endswith('.faa.mafft')][:8]:
            text=z.read(name).decode().splitlines()
            heads=[l for l in text if l.startswith('>')]
            print(name, len(heads), heads[:5])
PY"
