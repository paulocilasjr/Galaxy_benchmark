import os, glob, math
from collections import Counter, defaultdict

def read_fasta(path):
    seqs=[]; name=None; parts=[]
    with open(path) as f:
        for line in f:
            if line.startswith('>'):
                if name is not None:
                    seqs.append(''.join(parts).replace('\\n',''))
                name=line[1:].strip(); parts=[]
            else:
                parts.append(line.strip())
        if name is not None: seqs.append(''.join(parts))
    return seqs

def stats(path):
    seqs=read_fasta(path)
    n=len(seqs)
    L=len(seqs[0]) if n else 0
    allchars=set('-Xx?.*')
    pi=0; pi_ambig=0; variable=0; occupied_cols=0
    for j in range(L):
        col=[s[j] for s in seqs]
        cnt=Counter(c for c in col if c not in ['-','.','?','X','x','*'])
        # gap-only?
        if not cnt:
            continue
        occupied_cols += 1
        # standard two chars each >=2
        if sum(v>=2 for v in cnt.values()) >= 2:
            pi += 1
        # variable
        if len(cnt)>1:
            variable += 1
    return pi, L, occupied_cols, variable, n

for label, root in [('animal','/workspace/tmp/scogs_animal'),('fungi','/workspace/tmp/scogs_fungi')]:
    files=glob.glob(root+'/*.mafft')
    arr=[]
    print('\\n',label,len(files))
    for f in files:
        pi,L,occ,var,n=stats(f)
        pct=100.0*pi/L
        arr.append((pct,pi,L,occ,var,n,os.path.basename(f)))
    # means denominator L
    print('mean pct over L',sum(x[0] for x in arr)/len(arr))
    print('sum pi/sum L',sum(x[1] for x in arr)/sum(x[2] for x in arr)*100)
    print('range',min(x[0] for x in arr),max(x[0] for x in arr))
    print('zeros',sum(x[1]==0 for x in arr),'all?',arr[:3])
    globals()[label+'_arr']=arr
PY"
