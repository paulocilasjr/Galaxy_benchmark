from scipy.stats import mannwhitneyu
# Simple test: x has smaller values, y has larger values
x = [1, 2, 3]
y = [10, 20, 30]
res = mannwhitneyu(x, y)
print(f'x=[1,2,3], y=[10,20,30]')
print(f'U statistic: {res.statistic}')
print(f'p-value: {res.pvalue}')
# U should be: all 3*3=9 pairs have x_i < y_j, so U = 9
# Or min convention: U = 0

res2 = mannwhitneyu(y, x)
print(f'y=[10,20,30], x=[1,2,3]')
print(f'U statistic: {res2.statistic}')
print(f'p-value: {res2.pvalue}')
