common = animal_files & fungi_files
print(f'Animal mafft files: {len(animal_files)}')

print(f'Common mafft files: {len(common)}')

# Also check non-mafft files
with zipfile.ZipFile('/workspace/inputs/scogs_animals.zip') as za:
    animal_all = set(f for f in za.namelist())

with zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip') as zf:
    fungi_all = set(f for f in zf.namelist())

common_all = animal_all & fungi_all
