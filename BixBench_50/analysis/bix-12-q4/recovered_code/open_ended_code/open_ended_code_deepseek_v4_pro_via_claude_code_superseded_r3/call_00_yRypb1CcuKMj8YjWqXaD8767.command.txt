from collections import Counter

# Test: 4 sequences, 10 positions
seqs = {
    's1': 'AABBC--DEE',
    's2': 'AACCB--DED',
    's3': 'BBAA---DEF',
    's4': 'BBACC--DEF',
}

# Manual analysis:
# pos 0: A(2), B(2) -> 2 chars with >=2 each -> INFORMATIVE
# pos 1: A(2), B(2) -> INFORMATIVE
# pos 2: A(1), B(1), C(2) -> only C has >=2 -> not informative
# pos 3: B(2), C(2) -> INFORMATIVE (gaps?)
# pos 4: all gaps -> not informative (gaps excluded)
# pos 5: D(2), gap(2) -> D(2) only -> not informative (gaps are not characters)
# pos 6: - (all gaps)
# pos 7: D(3), E(1) -> D(3) only -> not informative
# pos 8: E(4) -> not informative
# pos 9: E(2), F(2) -> INFORMATIVE

# Expected: 4 informative / 10 total = 40%

# Now with code:
informative = 0
seq_list = list(seqs.values())
for pos in range(len(seq_list[0])):
    column = [s[pos] for s in seq_list]
    counts = Counter(c for c in column if c != '-')
    chars_with_min2 = [c for c, cnt in counts.items() if cnt >= 2]
    if len(chars_with_min2) >= 2:
        informative += 1
        print(f'Pos {pos}: {column} -> informative (chars with >=2: {chars_with_min2})')
    else:
        print(f'Pos {pos}: {column} -> NOT informative (chars with >=2: {chars_with_min2})')

print(f'Informative: {informative}/{len(seq_list[0])} = {100*informative/len(seq_list[0]):.1f}%')
