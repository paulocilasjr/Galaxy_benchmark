import sys
sys.path.insert(0, '/tmp/venv/lib/python3.11/site-packages')
from scipy.stats import rankdata, mannwhitneyu
import os

ANIMALS_DIR = '/tmp/scogs_animals'
FUNGI_DIR = '/tmp/scogs_fungi'

def parse_mafft(filepath):
    sequences = []
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_seq:
                    sequences.append(''.join(current_seq))
                    current_seq = []
            else:
                current_seq.append(line)
        if current_seq:
            sequences.append(''.join(current_seq))
    return sequences

def count_parsimony_informative_sites(sequences):
    if not sequences:
        return 0, 0
    lengths = set(len(s) for s in sequences)
    if len(lengths) != 1:
        raise ValueError(f'Sequences have different lengths: {lengths}')
    n_sites = lengths.pop()
    n_parsimony_informative = 0
    for col in range(n_sites):
        chars = [seq[col] for seq in sequences]
        counts = {}
        for c in chars:
            counts[c] = counts.get(c, 0) + 1
        states_with_min2 = [c for c, cnt in counts.items() if cnt >= 2]
        if len(states_with_min2) >= 2:
            n_parsimony_informative += 1
    return n_parsimony_informative, n_sites

animal_mafft = set(f for f in os.listdir(ANIMALS_DIR) if f.endswith('.mafft'))
fungi_mafft = set(f for f in os.listdir(FUNGI_DIR) if f.endswith('.mafft'))
shared = sorted(animal_mafft & fungi_mafft)

animal_pct = []
fungi_pct = []

for ortholog in shared:
    animal_path = os.path.join(ANIMALS_DIR, ortholog)
    fungi_path = os.path.join(FUNGI_DIR, ortholog)
    animal_seqs = parse_mafft(animal_path)
    fungi_seqs = parse_mafft(fungi_path)
    animal_inf, animal_total = count_parsimony_informative_sites(animal_seqs)
    fungi_inf, fungi_total = count_parsimony_informative_sites(fungi_seqs)
    if animal_total > 0:
        animal_pct.append(100.0 * animal_inf / animal_total)
    if fungi_total > 0:
        fungi_pct.append(100.0 * fungi_inf / fungi_total)

n1 = len(animal_pct)
n2 = len(fungi_pct)
combined = animal_pct + fungi_pct
ranks = rankdata(combined)
R1 = sum(ranks[:n1])
R2 = sum(ranks[n1:])
U1 = R1 - n1*(n1+1)/2
U2 = R2 - n2*(n2+1)/2

print(f'n1={n1}, n2={n2}')
print(f'R1={R1}, R2={R2}')
print(f'U1={U1}, U2={U2}')
print(f'U1+U2={U1+U2}, n1*n2={n1*n2}')
print()

# scipy result
result = mannwhitneyu(animal_pct, fungi_pct, alternative='two-sided')
print(f'scipy U: {result.statistic}')
print(f'scipy p: {result.pvalue}')
print(f'scipy method: {result.method if hasattr(result, \"method\") else \"N/A\"}')

# Also try with exact method
result_exact = mannwhitneyu(animal_pct, fungi_pct, alternative='two-sided', method='exact')
print(f'exact U: {result_exact.statistic}')
print(f'exact p: {result_exact.pvalue}')
