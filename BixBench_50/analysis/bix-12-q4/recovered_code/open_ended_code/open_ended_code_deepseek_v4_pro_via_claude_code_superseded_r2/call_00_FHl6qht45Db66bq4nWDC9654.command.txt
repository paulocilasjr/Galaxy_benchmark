# Quick spot check on a few alignments
def parse_mafft(filepath):
    sequences = []
    current_seq = []
    with open(filepath, 'r') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith('>'):
                if current_seq:
                    sequences.append(''.join(current_seq))
                    current_seq = []
            else:
                current_seq.append(line)
        if current_seq:
            sequences.append(''.join(current_seq))
    return sequences

def count_parsimony_informative_sites(sequences):
    if not sequences:
        return 0, 0
    n_sites = len(sequences[0])
    n_parsimony_informative = 0
    for col in range(n_sites):
        chars = [seq[col] for seq in sequences]
        counts = {}
        for c in chars:
            counts[c] = counts.get(c, 0) + 1
        states_with_min2 = [c for c, cnt in counts.items() if cnt >= 2]
        if len(states_with_min2) >= 2:
            n_parsimony_informative += 1
    return n_parsimony_informative, n_sites

# Check specific orthologs with non-zero PI%
for ortholog in ['1454155at2759.faa.mafft', '1530008at2759.faa.mafft', '1474000at2759.faa.mafft']:
    animal_path = f'/tmp/scogs_animals/{ortholog}'
    fungi_path = f'/tmp/scogs_fungi/{ortholog}'
    
    animal_seqs = parse_mafft(animal_path)
    fungi_seqs = parse_mafft(fungi_path)
    
    a_inf, a_tot = count_parsimony_informative_sites(animal_seqs)
    f_inf, f_tot = count_parsimony_informative_sites(fungi_seqs)
    
    print(f'{ortholog}:')
    print(f'  Animals: {a_inf}/{a_tot} = {100*a_inf/a_tot:.4f}% PI, {len(animal_seqs)} seqs, len={a_tot}')
    print(f'  Fungi:   {f_inf}/{f_tot} = {100*f_inf/f_tot:.4f}% PI, {len(fungi_seqs)} seqs, len={f_tot}')
