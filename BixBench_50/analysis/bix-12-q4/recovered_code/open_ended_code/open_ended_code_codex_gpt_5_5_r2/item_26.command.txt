/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse_fasta(text):
    records=[]; name=None; chunks=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None:
                records.append((name, ''.join(chunks).upper()))
            name=line[1:].split()[0]; chunks=[]
        else:
            chunks.append(line)
    if name is not None:
        records.append((name, ''.join(chunks).upper()))
    return records

def pi_pct(records):
    seqs=[seq for _,seq in records]
    if not seqs:
        raise ValueError('empty alignment')
    L=len(seqs[0])
    if any(len(seq)"'!=L for seq in seqs):
        raise ValueError('"'unaligned sequences')
    informative=0
    for col in zip(*seqs):
        counts=Counter(c for c in col if c not in '-?X')
        if sum(1 for count in counts.values() if count >= 2) >= 2:
            informative += 1
    return informative / L * 100

def archive_values(path):
    vals=[]
    with ZipFile(path) as z:
        for name in sorted(z.namelist()):
            if name.endswith('.mafft') and not name.endswith('/'):
                vals.append(pi_pct(parse_fasta(z.read(name).decode())))
    return vals

def mann_whitney_u_first(x, y):
    # U for the first sample: count x > y plus half of ties.
    return sum(1.0 if a > b else 0.5 if a == b else 0.0 for a in x for b in y)

animals=archive_values('inputs/scogs_animals.zip')
fungi=archive_values('inputs/scogs_fungi.zip')
print(len(animals), len(fungi))
print(mann_whitney_u_first(animals, fungi))
print(len(animals)*len(fungi)-mann_whitney_u_first(animals, fungi))
PY"
