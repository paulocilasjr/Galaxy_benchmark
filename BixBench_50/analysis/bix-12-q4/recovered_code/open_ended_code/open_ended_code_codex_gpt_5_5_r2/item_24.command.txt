/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter,defaultdict
for path in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    chars=Counter(); files=defaultdict(int)
    with ZipFile(path) as z:
        for n in z.namelist():
            if n.endswith('.mafft'):
                text=z.read(n).decode()
                for line in text.splitlines():
                    if line and not line.startswith('>'):
                        for c in line.strip().upper():
                            chars[c]+=1
                            if c in 'XNBZJUO?*': files[(c,n)]+=1
    print(path)
    print('ambig counts', {c:chars[c] for c in 'XNBZJUO?*-' if chars[c]})
    per=Counter(c for c,n in files)
    print('ambig file counts', per)
PY"
