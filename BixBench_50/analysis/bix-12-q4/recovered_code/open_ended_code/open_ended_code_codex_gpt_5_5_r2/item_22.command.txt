/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse_fasta(text):
    rec=[]; name=None; seq=[]
    for line in text.splitlines():
        if line.startswith('>'):
            if name is not None: rec.append((name,''.join(seq)))
            name=line[1:].strip(); seq=[]
        elif line.strip(): seq.append(line.strip())
    if name is not None: rec.append((name,''.join(seq)))
    return rec

def pct(records):
    seqs=[s.upper() for _,s in records]
    L=len(seqs[0])
    pi=0
    for col in zip(*seqs):
        counts=Counter(c for c in col if c not in '-?XxNn')
        if sum(v>=2 for v in counts.values()) >= 2:
            pi+=1
    return pi/L*100

def vals(path, only4=False):
    out=[]
    with ZipFile(path) as z:
        for n in sorted(z.namelist()):
            if n.endswith('.mafft'):
                r=parse_fasta(z.read(n).decode())
                if only4 and len(r)"'!=4: continue
                out.append(pct(r))
    return out

def u(x,y):
    return sum(1 if a>b else 0.5 if a==b else 0 for a in x for b in y)
for only4 in [False, True]:
    a=vals('"'inputs/scogs_animals.zip', only4); f=vals('inputs/scogs_fungi.zip', only4)
    print('only4',only4,'n',len(a),len(f),'U_animals',u(a,f),'U_fungi',len(a)*len(f)-u(a,f),'means',sum(a)/len(a),sum(f)/len(f))
PY"
