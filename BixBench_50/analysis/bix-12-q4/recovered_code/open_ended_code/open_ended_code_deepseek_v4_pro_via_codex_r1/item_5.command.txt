/bin/bash -lc "cd /workspace && python3 - <<'PY'
import zipfile, re, os
for name in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    z=zipfile.ZipFile(name)
    names=z.namelist()
    ma=[n for n in names if n.endswith('.faa.mafft')]
    faa=[n for n in names if n.endswith('.faa')]
    print(name, 'total',len(names),'mafft',len(ma),'faa',len(faa), 'nonmatch', set(ma)-set(f[:-1]+'.mafft' for f in faa))
    # inspect sequence count per first files and lengths etc
    import collections
    seqcounts=collections.Counter()
    for n in ma[:10]:
        data=z.read(n).decode()
        seqcounts[len(re.findall(r'"'^>'"', data, re.M))]+=1
    print('seq counts first 10',seqcounts)
PY"
