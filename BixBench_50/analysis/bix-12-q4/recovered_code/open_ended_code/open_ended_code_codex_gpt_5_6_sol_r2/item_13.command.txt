import zipfile,collections
STD=set('ACDEFGHIKLMNPQRSTVWY')
def vals(fn):
 out=[]
 with zipfile.ZipFile(fn) as z:

   if not n.endswith('.mafft'):continue

   for l in z.read(n).decode().splitlines():
    if l.startswith('>'):

   if s:ss.append(s)
   pi=sum(sum(v>=2 for v in collections.Counter(x for x in c if x in STD).values())>=2 for c in zip(*ss))
   out.append(100*pi/len(ss[0]))
 return out

  while j<len(a)and a[j][0]==a[i][0]:j+=1
  rank=((i+1)+j)/2
  for k in range(i,j):r[a[k][1]]+=rank
  i=j
 return r[0]-len(x)*(len(x)+1)/2
