import zipfile, collections, fractions
STD=set('ACDEFGHIKLMNPQRSTVWY')

def parse(data):
 out=[]; s=[]
 for line in data.decode().splitlines():
  if line.startswith('>'):
   if s: out.append(''.join(s).upper()); s=[]
  elif line.strip(): s.append(line.strip())
 if s: out.append(''.join(s).upper())
 return out

def vals(fn, mode='std', den='all'):
 out=[]; meta=[]; chars=collections.Counter()
 with zipfile.ZipFile(fn) as z:
  for n in sorted(x for x in z.namelist() if x.endswith('.mafft')):
   ss=parse(z.read(n)); L={len(x) for x in ss}
   if len(L)"'!=1: raise Exception((n,L))
   L=L.pop(); chars.update('"''.join(ss))
   pi=0; included=0
   for col in zip(*ss):
    if mode=='std': c=collections.Counter(x for x in col if x in STD)
    elif mode=='nogap': c=collections.Counter(x for x in col if x"'!='"'-')
    elif mode=='all': c=collections.Counter(col)
    else: raise Exception
    if den=='all' or len(c)>0: included+=1
    if sum(v>=2 for v in c.values())>=2: pi+=1
   D=L if den=='all' else included
   out.append(fractions.Fraction(pi,D) if D else fractions.Fraction(0))
   meta.append((n,len(ss),L,pi,D,float(out[-1])*100))
 return out,meta,chars

def mw(x,y):
 # exact rational ordering + average ranks; returns Ux, Uy, tie summary
 a=[(v,0,i) for i,v in enumerate(x)]+[(v,1,i) for i,v in enumerate(y)]
 a.sort(key=lambda z:z[0])
 rank_sum=[fractions.Fraction(0),fractions.Fraction(0)]
 ties=[]; i=0
 while i<len(a):
  j=i+1
  while j<len(a) and a[j][0]==a[i][0]:j+=1
  rank=fractions.Fraction((i+1)+j,2)
  for k in range(i,j):rank_sum[a[k][1]]+=rank
  if j-i>1: ties.append((a[i][0],j-i))
  i=j
 n=len(x);m=len(y)
 ux=rank_sum[0]-fractions.Fraction(n*(n+1),2)
 return ux, fractions.Fraction(n*m)-ux, ties
for mode in ['std','nogap','all']:
 for den in ['all','nongap']:
  a,am,ac=vals('inputs/scogs_animals.zip',mode,den)
  f,fm,fc=vals('inputs/scogs_fungi.zip',mode,den)
  u1,u2,t=mw(a,f)
  print(mode,den,'n',len(a),len(f),'zero',sum(v==0 for v in a),sum(v==0 for v in f),'U animal/fungi',u1,u2,'ties groups',len(t))
print('chars animals',ac)
print('chars fungi',fc)
PY"
