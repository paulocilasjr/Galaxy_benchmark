/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from pathlib import PurePosixPath
from collections import Counter
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\n###',fn)
    with ZipFile(fn) as z:
        names=z.namelist()
        c=Counter(PurePosixPath(n).suffix for n in names)
        print('n',len(names),'suffixes',c)
        for suff in sorted(c):
            ns=[n for n in names if PurePosixPath(n).suffix==suff]
            print(suff,len(ns), 'first', ns[:5], 'last',ns[-3:])
        print('first names')
        print('\\n'.join(names[:10]))
        for n in [x for x in names if x.endswith('.mafft')][:3]:
            print('\\n---',n,'---')
            print(z.read(n).decode()[:1200])
PY"
