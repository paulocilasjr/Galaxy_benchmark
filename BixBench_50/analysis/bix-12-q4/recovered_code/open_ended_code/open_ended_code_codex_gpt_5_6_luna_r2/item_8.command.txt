/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
from pathlib import PurePosixPath
import statistics

def fasta(data):
    rec=[]; name=None; seq=[]
    for ln in data.decode().splitlines():
        if ln.startswith('>'):
            if name is not None: rec.append((name,''.join(seq)))
            name=ln[1:].strip(); seq=[]
        else: seq.append(ln.strip())
    if name is not None: rec.append((name,''.join(seq)))
    return rec
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('\\n###',fn)
    with ZipFile(fn) as z:
      nms=[n for n in z.namelist() if n.endswith('.mafft')]
      stats=Counter(); labelsets=Counter(); lengths=Counter(); records=[]
      for n in nms:
        rs=fasta(z.read(n)); labels=tuple(x[0] for x in rs)
        lens=tuple(len(x[1]) for x in rs)
        stats[len(rs)]+=1; labelsets[labels]+=1; lengths[lens]+=1
        records.append((n,rs))
      print('n alignments',len(records),'seq counts',stats)
      print('label sets',len(labelsets))
      for labels,c in labelsets.most_common(): print(c,labels)
      inconsistent=[]
      for n,rs in records:
        if len({len(s) for _,s in rs})>1: inconsistent.append(n)
      print('length inconsistent',len(inconsistent),'of',len(records))
      Ls=[len(rs[0][1]) for _,rs in records]
      print('alignment length summary',min(Ls),max(Ls),statistics.median(Ls),sum(Ls)/len(Ls))
      for n,rs in records:
        if len(rs)"'!=4:
          print('"'unusual',n,'n=',len(rs),'labels',[x[0] for x in rs],'L',[len(x[1]) for x in rs]);
      raw={PurePosixPath(n).name[:-4] for n in z.namelist() if n.endswith('.faa')}
      ma={PurePosixPath(n).name[:-11] for n in nms}
      print('raw',len(raw),'ma',len(ma),'overlap',len(raw&ma),'raw only',len(raw-ma),'ma only',len(ma-raw))
      print('raw-only sample',sorted(raw-ma)[:10])
PY"
