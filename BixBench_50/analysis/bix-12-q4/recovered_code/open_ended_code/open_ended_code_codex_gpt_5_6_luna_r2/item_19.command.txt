# Independent calculation of candidate using direct pair counts for gap-excluded, all alignments.
from zipfile import ZipFile
from collections import Counter
from pathlib import PurePosixPath

def parse(b):
 out=[]; h=None;s=[]
 for l in b.decode().splitlines():
  if l.startswith('>'):
   if h is not None: out.append(''.join(s))
   h=l[1:];s=[]
  else:s.append(l.strip())
 if h is not None:out.append(''.join(s))
 return out

def pct(ss):
 L=len(ss[0]); p=0
 for col in zip(*ss):
  c=Counter(x for x in col if x"'!='"'-')
  if len(c)>=2 and sum(v>=2 for v in c.values())>=2:p+=1
 return 100*p/L
arr=[]
for f in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 vals=[]
 with ZipFile(f) as z:
  for n in z.namelist():
   if n.endswith('.mafft'): vals.append(pct(parse(z.read(n))))
 arr.append(vals)
a,b=arr
print(len(a),len(b),sum(x==0 for x in a),sum(x==0 for x in b))
print('direct >',sum(x>y for x in a for y in b),'=',sum(x==y for x in a for y in b),'<',sum(x<y for x in a for y in b))
print('U1',sum(x>y for x in a for y in b)+.5*sum(x==y for x in a for y in b),'U2',sum(x<y for x in a for y in b)+.5*sum(x==y for x in a for y in b))
print('check product',len(a)*len(b))
print('animals nonzero values',sorted(set(round(x,12) for x in a if x))[:30])
print('fungal smallest positive',sorted(set(round(x,12) for x in b if x))[:10])
PY"
