/bin/bash -lc "python - <<'PY'
import scipy
from scipy.stats import mannwhitneyu
print('scipy',scipy.__version__)
from zipfile import ZipFile
from collections import Counter
AA=set('ACDEFGHIKLMNPQRSTVWY')

def fasta(txt):
 d={}; k=None
 for line in txt.splitlines():
  if line.startswith('>'): k=line[1:].split()[0]; d[k]=''
  elif k: d[k]+=line.strip().upper()
 return d

def vals(zp, mode):
 out=[]; desc=[]
 with ZipFile(zp) as z:
  for n in sorted(x for x in z.namelist() if x.endswith('.faa.mafft')):
   d=fasta(z.read(n).decode())
   lens={len(s) for s in d.values()}
   if len(lens)"'!=1: raise ValueError((n,lens))
   L=lens.pop(); pi=0
   for col in zip(*d.values()):
    if mode=='"'strict': c=Counter(x for x in col if x in AA)
    elif mode=='gapstate': c=Counter(col)
    elif mode=='nongap_any': c=Counter(x for x in col if x not in '-?X')
    if sum(v>=2 for v in c.values())>=2: pi+=1
   out.append(100*pi/L)
   desc.append((n,len(d),L,pi,100*pi/L))
 return out,desc
for mode in ['strict','nongap_any','gapstate']:
 a,ad=vals('inputs/scogs_animals.zip',mode)
 f,fd=vals('inputs/scogs_fungi.zip',mode)
 u=mannwhitneyu(a,f,alternative='two-sided',method='asymptotic')
 print('\\n',mode,'n',len(a),len(f),'U1',u.statistic,'U2',len(a)*len(f)-u.statistic,'p',u.pvalue)
 import numpy as np
 print('animal mean med range',np.mean(a),np.median(a),(min(a),max(a)))
 print('fungal mean med range',np.mean(f),np.median(f),(min(f),max(f)))
 print('diff entries strict relevance?')
 print('seq count dist A',Counter(x[1] for x in ad),'F',Counter(x[1] for x in fd))
PY"
