/bin/bash -lc "Rscript - <<'RS'
read_member <- function(zipfile, member) {
 con <- unz(zipfile, member, open=\"rb\")
 raw <- readBin(con, what=\"raw\", n=100000000L)
 close(con)
 strsplit(rawToChar(raw), \"\\\\r?\\\\n\")[[1]]
}
read_group <- function(zipfile) {
  members <- unzip(zipfile, list=TRUE)"'$Name
  members <- members[grepl("'"\\\\.faa\\\\.mafft"'$", members)]
  vals <- numeric(length(members)); ntaxa <- integer(length(members))
  for (i in seq_along(members)) {
    lines <- read_member(zipfile, members[i])
    headers <- which(startsWith(lines, ">")); ntaxa[i] <- length(headers)
    ends <- c(headers[-1]-1L, length(lines))
    seqs <- mapply(function(a,b) paste0(lines[(a+1L):b], collapse=""), headers, ends, USE.NAMES=FALSE)
    stopifnot(length(unique(nchar(seqs))) == 1L)
    mat <- do.call(rbind, strsplit(seqs, "", fixed=TRUE))
    valid <- c("A","C","D","E","F","G","H","I","K","L","M","N","P","Q","R","S","T","V","W","Y")
    informative <- apply(mat, 2, function(col) sum(table(factor(col[col %in% valid], levels=valid)) >= 2L) >= 2L)
    vals[i] <- 100 * sum(informative) / ncol(mat)
  }
  list(values=vals, ntaxa=ntaxa, members=members)
}
a <- read_group("inputs/scogs_animals.zip"); f <- read_group("inputs/scogs_fungi.zip")
result <- wilcox.test(a$values, f$values, alternative="two.sided", exact=FALSE, correct=TRUE)
cat("n:", length(a$values), length(f$values), "'"\\n\")
cat(\"taxa animal:\", paste(names(table(a"'$ntaxa)), table(a$ntaxa), collapse=", "), "'"\\n\")
cat(\"taxa fungi:\", paste(names(table(f"'$ntaxa)), table(f$ntaxa), collapse=", "), "'"\\n\")
cat(\"R statistic:\", format(result"'$statistic, scientific=FALSE), "'"\\n\")
cat(\"rank-derived U:\", sum(rank(c(a"'$values,f$values))[seq_along(a$values)]) - length(a$values)*(length(a$values)+1)/2, "'"\\n\")
cat(\"p-value:\", format(result"'$p.value, digits=16), "'"\\n\")
RS"
