/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
AA=set('ACDEFGHIKLMNPQRSTVWY')
def fasta(t):
 d={};k=None
 for l in t.splitlines():
  if l.startswith('>'):k=l[1:].split()[0];d[k]=''
  else:d[k]+=l.strip().upper()
 return d
def read(zp):
 out={}
 with ZipFile(zp) as z:
  for n in z.namelist():
   if not n.endswith('.mafft'):continue
   d=fasta(z.read(n).decode());L=len(next(iter(d.values())))
   pi=sum(sum(v>=2 for v in Counter(x for x in col if x in AA).values())>=2 for col in zip(*d.values()))
   out[n.removesuffix('.mafft')]=(len(d),pi,L,100*pi/L)
 return out
def U(x,y):
 q=sorted([(v,0)for v in x]+[(v,1)for v in y]);rs=0;i=0
 while i<len(q):
  j=i+1
  while j<len(q) and q[j][0]==q[i][0]:j+=1
  rs+=(i+1+j)/2*sum(g==0 for _,g in q[i:j]);i=j
 return rs-len(x)*(len(x)+1)/2
A=read('inputs/scogs_animals.zip');F=read('inputs/scogs_fungi.zip')
scopes={
'all':(list(A),list(F)),
'intersection':(sorted(A.keys()&F.keys()),sorted(A.keys()&F.keys())),
'four_taxa_each':([k for k,v in A.items() if v[0]==4],[k for k,v in F.items() if v[0]==4]),
'paired_both_four':([k for k in A.keys()&F.keys() if A[k][0]==F[k][0]==4],[k for k in A.keys()&F.keys() if A[k][0]==F[k][0]==4]),
'atleast2':([k for k,v in A.items() if v[0]>=2],[k for k,v in F.items() if v[0]>=2]),
}
for scope,(ak,fk) in scopes.items():
 print('\\n'+scope,len(ak),len(fk))
 for label,fun in [('exact',lambda x:x),('AMAS_prop3',lambda x:round(x/100,3)),('pct3',lambda x:round(x,3)),('pct2',lambda x:round(x,2)),('pct1',lambda x:round(x,1))]:
  x=[fun(A[k][3]) for k in ak];y=[fun(F[k][3]) for k in fk]
  u=U(x,y)
  print(label,'U1',u,'minU',min(u,len(x)*len(y)-u))
PY"
