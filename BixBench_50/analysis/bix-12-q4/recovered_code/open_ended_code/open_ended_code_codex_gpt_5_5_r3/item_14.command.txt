/bin/bash -lc "python - <<'PY'
import zipfile
for z in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with zipfile.ZipFile(z) as f:
        names=sorted(n for n in f.namelist() if n.endswith('.mafft'))
        lens=[]; ntax=[]; bad=[]
        for n in names:
            seqs=[]; s=[]
            for line in f.read(n).decode().splitlines():
                if line.startswith('>'):
                    if s: seqs.append(''.join(s)); s=[]
                else: s.append(line.strip())
            if s: seqs.append(''.join(s))
            ntax.append(len(seqs)); lens.append(len(seqs[0]) if seqs else 0)
            if len(set(map(len,seqs)))"'!=1: bad.append(n)
        print(z,'"'n',len(names),'ntaxa',sorted(set(ntax)),'len min/max',min(lens),max(lens),'bad',len(bad))
PY"
