import zipfile
from collections import Counter

def read_fasta(text):
    seqs=[]; parts=[]; name=None
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(parts)))
            name=line[1:].split()[0]; parts=[]
        else: parts.append(line)
    if name is not None: seqs.append((name,''.join(parts)))
    return seqs

def pct(seqs, gap_state=False, denom_mode='alignment'):
    ss=[s for _,s in seqs]; L=len(ss[0])
    pis=0; den=0
    for chars in zip(*ss):
        if denom_mode=='ungapped' and '-' in chars: continue
        if denom_mode=='nonempty' and all(c in '-?X' for c in chars): continue
        den += 1
        states=[c for c in chars if gap_state or c not in '-?X']
        if not gap_state: states=[c for c in states if c not in '-?X']
        else: states=[c for c in states if c"'!='"'?']
        c=Counter(states)
        if sum(v>=2 for v in c.values())>=2: pis += 1
    return 100*pis/den if den else 0.0

def load(z, complete4, gap_state, denom):
    out=[]
    with zipfile.ZipFile(z) as f:
        for n in sorted(f.namelist()):
            if n.endswith('.mafft'):
                s=read_fasta(f.read(n).decode())
                if complete4 and len(s)"'!=4: continue
                out.append(pct(s,gap_state,denom))
    return out

def u(x,y):
    return sum(1 if a>b else 0.5 if a==b else 0 for a in x for b in y)
for denom in ['"'alignment','ungapped','nonempty']:
  for complete in [False,True]:
    a=load('inputs/scogs_animals.zip',complete,False,denom)
    f=load('inputs/scogs_fungi.zip',complete,False,denom)
    print(denom, complete, len(a), len(f), u(a,f))
PY"
