/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from fractions import Fraction

def read(z, fn):
    rec=[]; seq=[]; seen=False
    for line in z.read(fn).decode().splitlines():
        if line.startswith('>'):
            if seen: rec.append(''.join(seq))
            seen=True; seq=[]
        else: seq.append(line.strip())
    if seen: rec.append(''.join(seq))
    return rec

def calc(rec):
    cols=list(zip(*rec)); L=len(cols); n=len(rec)
    pi=pi_no_gap=complete_pi=0; complete=0
    for col in cols:
        c=Counter(col)
        if sum(v>=2 for v in c.values())>=2: pi+=1
        ng=[x for x in col if x not in '-?']
        cc=Counter(ng)
        if sum(v>=2 for v in cc.values())>=2: pi_no_gap+=1
        if len(ng)==n:
            complete+=1
            if sum(v>=2 for v in cc.values())>=2: complete_pi+=1
    return {'id':None,'n':n,'L':L,'pi':pi,'pi_no_gap':pi_no_gap,'complete':complete,'complete_pi':complete_pi}

def load(g):
    d={}
    with ZipFile('inputs/scogs_'+g+'.zip') as z:
        for fn in z.namelist():
            if fn.endswith('.faa.mafft'):
                m=calc(read(z,fn)); m['id']=fn[:-10]; d[m['id']]=m
    return d

A=load('animals'); F=load('fungi')

def u(x,y):
    return sum(1 if a>b else .5 if a==b else 0 for a in x for b in y)

def fp(m, key='pi'):
    return Fraction(m[key]*100,m['L'])

for desc, aa, ff in [
    ('all',A,F),
    ('n4', {k:v for k,v in A.items() if v['n']==4}, {k:v for k,v in F.items() if v['n']==4}),
    ('common', {k:v for k,v in A.items() if k in F}, {k:v for k,v in F.items() if k in A}),
    ('common_n4', {k:v for k,v in A.items() if k in F and v['n']==4}, {k:v for k,v in F.items() if k in A and v['n']==4}),
]:
    for key in ['pi','pi_no_gap','complete_pi']:
        x=[fp(m,key) for m in aa.values()]; y=[fp(m,key) for m in ff.values()]
        if x and y:
            st=u(x,y); print(desc,key,'n',len(x),len(y),'Uanimal',st,'Ufugi',len(x)*len(y)-st,'min',min(st,len(x)*len(y)-st),'zero',sum(v==0 for v in x),sum(v==0 for v in y))
    print()

# exact summary for all ignoring gaps
for label,d in [('A',A),('F',F)]:
    vals=[fp(m,'pi_no_gap') for m in d.values()]
    print(label,'sum PI',sum(m['pi_no_gap'] for m in d.values()),'sum L',sum(m['L'] for m in d.values()),'mean percent',float(sum(vals)/len(vals)),'median sorted',sorted(vals)[len(vals)//2], 'nonzero',sum(v>0 for v in vals))
    print('counts pi:',Counter(m['pi_no_gap'] for m in d.values()).most_common(15))
PY"
