/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from fractions import Fraction

def fasta(raw):
    seqs=[]; cur=[]
    for line in raw.decode('ascii').splitlines():
        if line.startswith('>'):
            if cur: seqs.append(''.join(cur))
            cur=[]
        else: cur.append(line.strip())
    if cur: seqs.append(''.join(cur))
    return seqs

def percentages(archive):
    out=[]
    with ZipFile(archive) as z:
        names=sorted(n for n in z.namelist() if n.endswith('.faa.mafft'))
        assert names
        for name in names:
            seqs=fasta(z.read(name))
            lengths={len(s) for s in seqs}
            assert len(lengths)==1, (name,lengths)
            pi=0
            for column in zip(*seqs):
                counts=Counter(a for a in column if a "'!= '"'-')
                if sum(c >= 2 for c in counts.values()) >= 2:
                    pi += 1
            out.append(Fraction(100*pi, len(seqs[0])))
    return out

animals=percentages('inputs/scogs_animals.zip')
fungi=percentages('inputs/scogs_fungi.zip')
assert (len(animals),len(fungi)) == (241,255)
u=sum(Fraction(1) if a>b else Fraction(1,2) if a==b else Fraction(0) for a in animals for b in fungi)
u_other=len(animals)*len(fungi)-u
print('animals_alignments=',len(animals),'fungi_alignments=',len(fungi))
print('animals_zero=',sum(x==0 for x in animals),'fungi_zero=',sum(x==0 for x in fungi))
print('U_animals=',u,'U_fungi=',u_other,'U_min=',min(u,u_other))
assert u == 6948
PY"
