/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
from collections import Counter
p=Path('/workspace/tmp/animals/1428854at2759.faa.mafft')
print(p.exists())
seqs=[]
for block in p.read_text().strip().split('>'):
    if not block: continue
    lines=block.splitlines()
    name=lines[0]
    s=''.join(lines[1:]).replace('\\n','')
    seqs.append((name,s))
print(len(seqs), [(n,len(s)) for n,s in seqs])
L=len(seqs[0][1])
for i in [0,1,2]:
 col=[s[i] for _,s in seqs]
 print(i,col)
for EX in [set('-'),set('-?X*')]:
 pis=0
 for i in range(L):
  col=[s[i] for _,s in seqs if s[i] not in EX]
  if len(col)<4: continue
  cnt=Counter(col)
  if sum(1 for v in cnt.values() if v>=2)>=2: pis+=1
 print(EX,pis)
PY"
