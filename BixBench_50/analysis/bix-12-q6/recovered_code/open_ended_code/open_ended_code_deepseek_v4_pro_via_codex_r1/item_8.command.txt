from pathlib import Path
from collections import Counter

def parse(path):
    seqs=[]; name=None; chars=[]
    for line in path.read_text().splitlines():
        if line.startswith('>'):
            if name is not None: seqs.append((name,''.join(chars)))
            name=line[1:].strip(); chars=[]
        else:
            chars.append(line.strip())
    if name is not None: seqs.append((name,''.join(chars)))
    return seqs

def pis(seqs, gap='-'):
    n=len(seqs)
    L=len(seqs[0])
    c=0
    for j in range(L):
        chars=[s[j] for _,s in seqs if s[j]"'!=gap]
        if len(chars)<4: continue
        counts=Counter(chars)
        if len(counts)>=2 and sum(v>=2 for v in counts.values())>=2:
            c+=1
    return c
for group in ['"'animals','fungi']:
    files=sorted(Path('/workspace/tmp',group).glob('*.mafft'))
    vals=[]
    for f in files:
        vals.append((f.name,pis(parse(f))))
    Path('/workspace/tmp',f'{group}_pis.tsv').write_text('\\n'.join(f'{a}\\t{b}' for a,b in vals)+'\\n')
    print(group, len(vals), 'mean', sum(b for a,b in vals)/len(vals), 'minmax', min(b for a,b in vals), max(b for a,b in vals), 'zeros', sum(b==0 for a,b in vals))

an={a:int(b) for a,b in (l.split('\\t') for l in Path('/workspace/tmp/animals_pis.tsv').read_text().splitlines())}
fu={a:int(b) for a,b in (l.split('\\t') for l in Path('/workspace/tmp/fungi_pis.tsv').read_text().splitlines())}
common=sorted(set(an)&set(fu))
print('common n',len(common))
print('an n',len(an),'fu n',len(fu))
# rank function with average ties
def ranks(vals):
    order=sorted(range(len(vals)), key=lambda i: (vals[i], i))
    r=[0]*len(vals); i=0
    while i<len(order):
        j=i
        while j+1<len(order) and vals[order[j+1]]==vals[order[i]]:
            j+=1
        avg=(i+j)/2+1
        for k in range(i,j+1): r[order[k]]=avg
        i=j+1
    return r
for label,keys1,keys2 in [('all',list(an),list(fu)),('common',common,common)]:
    x=[an[k] for k in keys1]; y=[fu[k] for k in keys2]
    n1=len(x); n2=len(y)
    r=ranks(x+y); R1=sum(r[:n1])
    U1=n1*n2 + n1*(n1+1)/2 - R1
    U2=n1*n2 - U1
    print(label,'n1,n2',n1,n2,'R1',R1,'U1',U1,'U2',U2,'min',min(U1,U2))
PY"
