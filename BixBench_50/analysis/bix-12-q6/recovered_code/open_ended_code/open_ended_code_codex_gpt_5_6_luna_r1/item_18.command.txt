/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse(t):
 o=[]; h=None; s=[]
 for l in t.decode().splitlines():
  if l.startswith('>'):
   if h is not None:o.append((h,''.join(s)))
   h=l[1:].split()[0];s=[]
  elif l.strip():s.append(l.strip())
 if h is not None:o.append((h,''.join(s)))
 return o

def get(p):
 with ZipFile(p) as z:return [(n,parse(z.read(n))) for n in z.namelist() if n.endswith('.faa.mafft')]
def pi(ss, gap=True):
 tot=0
 for col in zip(*ss):
  c=Counter(col)
  if not gap:c.pop('-',None)
  if sum(v>=2 for v in c.values())>=2:tot+=1
 return tot
def u(a,b):return sum(1 if x>y else .5 if x==y else 0 for x in a for y in b)
A=get('inputs/scogs_animals.zip'); F=get('inputs/scogs_fungi.zip')
for gap in [True,False]:
 av=[(len(s),pi([x for _,x in ss],gap)) for _,ss in A]; fv=[(len(s),pi([x for _,x in ss],gap)) for _,ss in F]
 print('\\ngap',gap)
 for filtname,filt in [('all',lambda n,x:True),('>=4',lambda n,x:n>=4),('nonzero',lambda n,x:x>0),('>=4_nonzero',lambda n,x:n>=4 and x>0)]:
  a=[x for n,x in av if filt(n,x)];b=[x for n,x in fv if filt(n,x)]
  print(filtname,'n',len(a),len(b),'U1',u(a,b),'U2',len(a)*len(b)-u(a,b),'sum',sum(a),sum(b))
PY"
