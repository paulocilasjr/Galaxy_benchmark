from collections import Counter, defaultdict

def parse(t):
 o=[]; h=None; s=[]
 for l in t.decode().splitlines():
  if l.startswith('>'):
   if h is not None:o.append((h,''.join(s)))
   h=l[1:].split()[0];s=[]
  elif l.strip():s.append(l.strip())
 if h is not None:o.append((h,''.join(s)))
 return o
for p in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 print('\\n===',p)
 with ZipFile(p) as z:
  shown=0
  for n in z.namelist():
   if not n.endswith('.faa.mafft'):continue
   ss=parse(z.read(n));
   if len(ss)"'!=4:continue
   pats=[]
   for col in zip(*(s for _,s in ss)):
    c=Counter(col)
    if sum(v>=2 for v in c.values())>=2 and '"'-' in c:
     pats.append(''.join(col))
   if pats:
    print(n, 'gapPI',len(pats),'examples',pats[:8], 'headers',[h for h,_ in ss])
