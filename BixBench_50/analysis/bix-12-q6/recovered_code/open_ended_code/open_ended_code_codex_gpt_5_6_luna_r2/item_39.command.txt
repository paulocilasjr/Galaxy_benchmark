/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def parse(d):
 r=[];b=[]
 for l in d.decode().splitlines():
  if l.startswith('>'):
   if b:r.append(''.join(b))
   b=[]
  else:b.append(l.strip())
 if b:r.append(''.join(b))
 return r

def pi(s,gap=False):
 c=0
 for col in zip(*s):
  if gap and '-' in col:continue
  q=Counter(col if gap else [x for x in col if x"'!='"'-'])
  c += sum(v>=2 for v in q.values())>=2
 return c

def us(a,b):
 z=sorted([(x,0) for x in a]+[(x,1) for x in b]); ranks=[0.,0.];i=0
 while i<len(z):
  j=i+1
  while j<len(z) and z[j][0]==z[i][0]: j+=1
  rr=(i+1+j)/2
  for k in range(i,j): ranks[z[k][1]]+=rr
  i=j
 return ranks[0]-len(a)*(len(a)+1)/2, ranks[1]-len(b)*(len(b)+1)/2
D=[]
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 d={}
 with ZipFile(fn) as z:
  for n in z.namelist():
   if n.endswith('.faa.mafft'):
    d[n[:-10]]=pi(parse(z.read(n)))
 D.append(d)
for scope in ['all','common']:
 keys=sorted(D[0]) if scope=='all' else sorted(set(D[0])&set(D[1]))
 a=[D[0][k] for k in keys]; b=[D[1][k] for k in keys] if scope=='common' else list(D[1].values())
 print(scope,len(a),len(b),us(a,b))
PY"
