from zipfile import ZipFile
from collections import Counter


def parse(d):
    r=[]; name=None; buf=[]
    for l in d.decode().splitlines():
        if l.startswith('>'):
            if name is not None:r.append(''.join(buf))
            name=l[1:];buf=[]
        else:buf.append(l.strip())
    if name is not None:r.append(''.join(buf))
    return r

def pi(s, gaps_missing=True):
    n=0
    for col in zip(*s):
        if gaps_missing: col=[x for x in col if x"'!='"'-']
        c=Counter(col)
        if sum(x>=2 for x in c.values())>=2:n+=1
    return n

def stats(x,y):
    # observations are (value, group), group 0 animal, 1 fungi
    v=sorted([(z,0) for z in x]+[(z,1) for z in y])
    rank_sum=[0.0,0.0];i=0
    while i<len(v):
        j=i+1
        while j<len(v) and v[j][0]==v[i][0]:j+=1
        r=(i+1+j)/2
        for k in range(i,j):rank_sum[v[k][1]]+=r
        i=j
    u1=rank_sum[0]-len(x)*(len(x)+1)/2
    u2=rank_sum[1]-len(y)*(len(y)+1)/2
    return rank_sum,u1,u2
for gm in [True,False]:
    gs=[]
    for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
        v=[]
        with ZipFile(fn) as z:
            for nm in z.namelist():
                if nm.endswith('.faa.mafft'):
                    v.append(pi(parse(z.read(nm)),gm))
        gs.append(v)
    (ra,rf),u1,u2=stats(*gs)
    print('gaps_missing',gm,'n',len(gs[0]),len(gs[1]),'rank sums',ra,rf,'U animal',u1,'U fungi',u2,'min',min(u1,u2),'sum',u1+u2)
    # Show full summary counts zero, mean/med
    for label,a in zip(['a','f'],gs):
      print(label,'sum',sum(a),'mean',sum(a)/len(a),'median',sorted(a)[len(a)//2],'zeros',a.count(0),'top',Counter(a).most_common())
PY
