import zipfile
from collections import Counter

def count_seqs_in_mafft(zip_path, label):
    counts = Counter()
    with zipfile.ZipFile(zip_path, 'r') as zf:
        for name in zf.namelist():
            if name.endswith('.faa.mafft'):
                content = zf.read(name).decode('utf-8')
                n_seqs = content.count('>')
                counts[n_seqs] += 1
    print(f'{label}: {dict(sorted(counts.items()))}')

count_seqs_in_mafft('/workspace/inputs/scogs_animals.zip', 'Animals')
count_seqs_in_mafft('/workspace/inputs/scogs_fungi.zip', 'Fungi')
