import zipfile

def parse_mafft_alignment(text):
    sequences = []
    current_header = None
    current_seq_parts = []
    for line in text.strip().split('\n'):
        line = line.rstrip('\r')
        if line.startswith('>'):
            if current_header is not None:
                sequences.append((current_header, ''.join(current_seq_parts)))
            current_header = line[1:].strip()
            current_seq_parts = []
        elif line.strip():
            current_seq_parts.append(line.strip())
    if current_header is not None:
        sequences.append((current_header, ''.join(current_seq_parts)))
    return sequences

def count_pis(sequences):
    """Count parsimony-informative sites. Gaps treated as character state."""
    if len(sequences) < 4:
        return 0  # Need at least 4 sequences for any PIS
    seqs = [s for _, s in sequences]
    aln_len = len(seqs[0])
    pis = 0
    for col in range(aln_len):
        char_counts = {}
        for s in seqs:
            c = s[col]
            char_counts[c] = char_counts.get(c, 0) + 1
        states_ge2 = sum(1 for cnt in char_counts.values() if cnt >= 2)
        if states_ge2 >= 2:
            pis += 1
    return pis

def read_all_mafft(zip_path):
    results = {}
    with zipfile.ZipFile(zip_path, 'r') as zf:
        for name in zf.namelist():
            if name.endswith('.faa.mafft'):
                ortho_id = name.replace('.faa.mafft', '').split('/')[-1]
                content = zf.read(name).decode('utf-8')
                sequences = parse_mafft_alignment(content)
                results[ortho_id] = sequences
    return results

# Mann-Whitney U computation
# Method 1: Rank-sum approach
n1, n2 = len(x), len(y)
combined = [(v, 0) for v in x] + [(v, 1) for v in y]
combined.sort(key=lambda t: t[0])

# Assign ranks (mid-ranks for ties)
ranks = [0.0] * len(combined)
i = 0
while i < len(combined):
    j = i
    while j < len(combined) and combined[j][0] == combined[i][0]:
        j += 1
    mid_rank = (i + 1 + j) / 2.0
    for k in range(i, j):
        ranks[k] = mid_rank
    i = j

R1 = sum(ranks[k] for k in range(len(combined)) if combined[k][1] == 0)
R2 = sum(ranks[k] for k in range(len(combined)) if combined[k][1] == 1)

U1_rank = n1 * n2 + n1 * (n1 + 1) / 2.0 - R1
U2_rank = n1 * n2 + n2 * (n2 + 1) / 2.0 - R2
U_min_rank = min(U1_rank, U2_rank)

print(f"\nMann-Whitney U (rank-sum method):")
print(f"  U1 (Animals) = {U1_rank}")
print(f"  U2 (Fungi)   = {U2_rank}")
print(f"  U_min        = {U_min_rank}")

# Method 2: Direct pairwise comparison
U1_direct = 0.0
for a in x:
    for b in y:
        if a > b:
            U1_direct += 1.0
        elif a == b:
            U1_direct += 0.5
U2_direct = n1 * n2 - U1_direct

print(f"\nMann-Whitney U (pairwise method):")
print(f"  U1 (Animals) = {U1_direct}")
print(f"  U2 (Fungi)   = {U2_direct}")
print(f"  U_min        = {min(U1_direct, U2_direct)}")

    return pis
