import zipfile

# Check a fungi alignment with 4 sequences to verify PIS counting
with zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip', 'r') as zf:
    for name in sorted(zf.namelist()):
        if name.endswith('.faa.mafft'):
            content = zf.read(name).decode('utf-8')
            n_seqs = content.count('>')
            if n_seqs == 4:
                ortho = name.replace('.faa.mafft','').split('/')[-1]
                # Parse alignment
                seqs = []
                cur_hdr = None
                cur_seq = []
                for line in content.strip().split('\n'):
                    if line.startswith('>'):
                        if cur_hdr:
                            seqs.append((cur_hdr, ''.join(cur_seq)))
                        cur_hdr = line[1:].strip()
                        cur_seq = []
                    elif line.strip():
                        cur_seq.append(line.strip())
                if cur_hdr:
                    seqs.append((cur_hdr, ''.join(cur_seq)))
                
                # Count PIS
                pis = 0
                for col in range(len(seqs[0][1])):
                    chars = {}
                    for _, s in seqs:
                        c = s[col]
                        chars[c] = chars.get(c, 0) + 1
                    states_ge2 = sum(1 for v in chars.values() if v >= 2)
                    if states_ge2 >= 2:
                        pis += 1
                print(f'fungi {ortho}: {n_seqs} seqs, len={len(seqs[0][1])}, PIS={pis}')
                break  # Just look at first one
