import zipfile

# Check species in animal vs fungi alignments
with zipfile.ZipFile('/workspace/inputs/scogs_animals.zip', 'r') as zf:
    content = zf.read('1014314at2759.faa.mafft').decode('utf-8')
    print("Animal ortholog 1014314at2759 species:")
    for line in content.strip().split('\n'):
        if line.startswith('>'):
            print(f"  {line}")

with zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip', 'r') as zf:
    content = zf.read('1001705at2759.faa.mafft').decode('utf-8')
    print("\nFungi ortholog 1001705at2759 species:")
    for line in content.strip().split('\n'):
        if line.startswith('>'):
            print(f"  {line}")

# Get all species in each group
print("\n\nAll species in animal scogs:")
species_a = set()
with zipfile.ZipFile('/workspace/inputs/scogs_animals.zip', 'r') as zf:
    for name in zf.namelist():
        if name.endswith('.faa.mafft'):
            content = zf.read(name).decode('utf-8')
            for line in content.strip().split('\n'):
                if line.startswith('>'):
                    species_a.add(line[1:].strip())
print(f"  {sorted(species_a)}")

print("\nAll species in fungi scogs:")
species_f = set()
with zipfile.ZipFile('/workspace/inputs/scogs_fungi.zip', 'r') as zf:
    for name in zf.namelist():
        if name.endswith('.faa.mafft'):
            content = zf.read(name).decode('utf-8')
            for line in content.strip().split('\n'):
                if line.startswith('>'):
                    species_f.add(line[1:].strip())
print(f"  {sorted(species_f)}")
