import zipfile

def parse_mafft_alignment(text):
    seqs = []
    cur_hdr = None
    cur_seq = []
    for line in text.strip().split('\n'):
        line = line.rstrip('\r')
        if line.startswith('>'):
            if cur_hdr: seqs.append((cur_hdr, ''.join(cur_seq)))
            cur_hdr = line[1:].strip()
            cur_seq = []
        elif line.strip(): cur_seq.append(line.strip())
    if cur_hdr: seqs.append((cur_hdr, ''.join(cur_seq)))
    return seqs

def count_pis(sequences):
    if len(sequences) < 4: return 0
    seqs = [s for _, s in sequences]
    aln_len = len(seqs[0])
    pis = 0
    for col in range(aln_len):
        cc = {}
        for s in seqs:
            c = s[col]
            cc[c] = cc.get(c, 0) + 1
        if sum(1 for v in cc.values() if v >= 2) >= 2:
            pis += 1
    return pis

def read_all(zp):
    results = {}
    with zipfile.ZipFile(zp, 'r') as zf:
        for name in zf.namelist():
            if name.endswith('.faa.mafft'):
                oid = name.replace('.faa.mafft','').split('/')[-1]
                results[oid] = count_pis(parse_mafft_alignment(zf.read(name).decode('utf-8')))
    return results
