/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
VALID=set('ACDEFGHIKLMNPQRSTVWY')
def read_fasta(b):
    out=[]; parts=[]; name=None
    for l in b.decode().splitlines():
        l=l.strip()
        if not l: continue
        if l[0]=='>':
            if name is not None: out.append((name,''.join(parts)))
            name=l[1:]; parts=[]
        else: parts.append(l)
    if name is not None: out.append((name,''.join(parts)))
    return out

def pi(seqs, gap=False):
    n=0
    for col in zip(*(s for _,s in seqs)):
        if gap:
            chars=[c.upper() for c in col if c.upper() not in '?X*']
        else:
            chars=[c.upper() for c in col if c.upper() in VALID]
        cnt=Counter(chars)
        if sum(v>=2 for v in cnt.values())>=2: n+=1
    return n

def U(x,y):
    vals=sorted([(v,0) for v in x]+[(v,1) for v in y])
    rsum=0; i=0
    while i<len(vals):
        j=i+1
        while j<len(vals) and vals[j][0]==vals[i][0]: j+=1
        avg=(i+1+j)/2
        rsum += avg*sum(1 for k in range(i,j) if vals[k][1]==0)
        i=j
    n=len(x)
    return rsum-n*(n+1)/2
with ZipFile('inputs/scogs_animals.zip') as za, ZipFile('inputs/scogs_fungi.zip') as zf:
    common=sorted(set(n for n in za.namelist() if n.endswith('.mafft')) & set(n for n in zf.namelist() if n.endswith('.mafft')))
    for gap in [False, True]:
        a=[pi(read_fasta(za.read(n)), gap) for n in common]
        f=[pi(read_fasta(zf.read(n)), gap) for n in common]
        print('gap',gap,'n',len(a),len(f),'U a',U(a,f),'U f',U(f,a),'sum',U(a,f)+U(f,a), 'prod',len(a)*len(f))
        print('means', sum(a)/len(a), sum(f)/len(f), 'medians', sorted(a)[len(a)//2], sorted(f)[len(f)//2])
PY"
