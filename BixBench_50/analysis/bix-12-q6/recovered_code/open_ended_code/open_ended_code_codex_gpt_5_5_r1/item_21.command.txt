/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter,defaultdict

def read_count(b):
    return sum(1 for l in b.decode().splitlines() if l.startswith('>'))
for fn in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    with ZipFile(fn) as z:
        d=defaultdict(list)
        for name in sorted(n for n in z.namelist() if n.endswith('.mafft')):
            d[read_count(z.read(name))].append(name)
        print('\\n',fn)
        for k in sorted(d): print(k, len(d[k]), d[k][:20])
PY"
