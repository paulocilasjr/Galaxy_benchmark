/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
VALID=set('ACDEFGHIKLMNPQRSTVWY')

def read_fasta(data):
    seqs=[]; name=None; parts=[]
    for line in data.decode().splitlines():
        line=line.strip()
        if not line:
            continue
        if line.startswith('>'):
            if name is not None:
                seqs.append((name,''.join(parts)))
            name=line[1:].strip()
            parts=[]
        else:
            parts.append(line)
    if name is not None:
        seqs.append((name,''.join(parts)))
    return seqs

def raw_pi_count(seqs):
    lens={len(s) for _,s in seqs}
    if len(seqs) < 4 or len(lens) "'!= 1:
        raise ValueError((len(seqs), lens))
    count=0
    for col in zip(*(s for _,s in seqs)):
        states=Counter(c.upper() for c in col if c.upper() in VALID)
        if sum(1 for n in states.values() if n >= 2) >= 2:
            count += 1
    return count

def archive_counts(path):
    vals=[]
    with ZipFile(path) as z:
        for name in sorted(n for n in z.namelist() if n.endswith('"'.faa.mafft')):
            vals.append(raw_pi_count(read_fasta(z.read(name))))
    return vals

def mann_whitney_u_x(x, y):
    # U for x: pairwise wins plus half ties, equivalent to average ranks with ties.
    u=0.0
    for xv in x:
        for yv in y:
            if xv > yv:
                u += 1.0
            elif xv == yv:
                u += 0.5
    return u
animals=archive_counts('inputs/scogs_animals.zip')
fungi=archive_counts('inputs/scogs_fungi.zip')
u=mann_whitney_u_x(animals, fungi)
print('animal_n', len(animals), 'fungi_n', len(fungi))
print('animal_sum_min_max', sum(animals), min(animals), max(animals))
print('fungi_sum_min_max', sum(fungi), min(fungi), max(fungi))
print('U_animals_vs_fungi', u)
PY"
