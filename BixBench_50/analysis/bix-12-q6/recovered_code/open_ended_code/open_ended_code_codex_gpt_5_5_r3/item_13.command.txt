import zipfile
from collections import Counter

def parse(text):
    seqs=[]; cur=[]; headers=[]
    for line in text.splitlines():
        line=line.strip()
        if not line: continue
        if line.startswith('>'):
            if cur: seqs.append(''.join(cur)); cur=[]
            headers.append(line[1:])
        else: cur.append(line)
    if cur: seqs.append(''.join(cur))
    return headers,seqs

def pi(seqs):
    total=0
    for col in zip(*seqs):
        c=Counter(x for x in col if x"'!='"'-')
        if sum(v>=2 for v in c.values())>=2: total+=1
    return total

def records(z):
    rec={}
    with zipfile.ZipFile(z) as f:
        for name in f.namelist():
            if name.endswith('.faa.mafft'):
                h,s=parse(f.read(name).decode())
                stem=name[:-len('.faa.mafft')]
                rec[stem]={'name':name,'nseq':len(s),'count':pi(s),'headers':h}
    return rec

def mann_u(x,y):
    vals=sorted([(v,0) for v in x]+[(v,1) for v in y])
    rs=0; i=0
    while i<len(vals):
        j=i+1
        while j<len(vals) and vals[j][0]==vals[i][0]: j+=1
        avg=(i+1+j)/2
        rs+=avg*sum(g==0 for _,g in vals[i:j])
        i=j
    nx=len(x)
    return rs-nx*(nx+1)/2
A=records('inputs/scogs_animals.zip'); F=records('inputs/scogs_fungi.zip')
print('A,F,common',len(A),len(F),len(set(A)&set(F)), 'A-F',len(set(A)-set(F)), 'F-A',len(set(F)-set(A)))
for scope in ['all','complete4','common','common_complete4_both','complete4_common_ids_individual']:
    if scope=='all':
        ax=[r['count'] for r in A.values()]; fy=[r['count'] for r in F.values()]
    elif scope=='complete4':
        ax=[r['count'] for r in A.values() if r['nseq']==4]; fy=[r['count'] for r in F.values() if r['nseq']==4]
    elif scope=='common':
        ids=sorted(set(A)&set(F)); ax=[A[i]['count'] for i in ids]; fy=[F[i]['count'] for i in ids]
    elif scope=='common_complete4_both':
        ids=sorted(i for i in set(A)&set(F) if A[i]['nseq']==4 and F[i]['nseq']==4); ax=[A[i]['count'] for i in ids]; fy=[F[i]['count'] for i in ids]
    elif scope=='complete4_common_ids_individual':
        ids=set(A)&set(F); ax=[A[i]['count'] for i in ids if A[i]['nseq']==4]; fy=[F[i]['count'] for i in ids if F[i]['nseq']==4]
    print(scope, 'n', len(ax), len(fy), 'sum', sum(ax), sum(fy), 'zero', ax.count(0), fy.count(0), 'U_A', mann_u(ax,fy), 'U_F', mann_u(fy,ax), 'nxny', len(ax)*len(fy))
print('A nseq counts', Counter(r['nseq'] for r in A.values()))
print('F nseq counts', Counter(r['nseq'] for r in F.values()))
print('first A-only', sorted(set(A)-set(F))[:20])
print('first F-only', sorted(set(F)-set(A))[:20])
PY"
