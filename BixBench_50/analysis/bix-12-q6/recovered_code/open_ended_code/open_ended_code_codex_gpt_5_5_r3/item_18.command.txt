/bin/bash -lc "python - <<'PY'
import os, zipfile
from collections import Counter

ANIMALS = 'inputs/scogs_animals.zip'
FUNGI = 'inputs/scogs_fungi.zip'

def parse_fasta(text):
    seqs=[]
    cur=[]
    for raw in text.splitlines():
        line=raw.strip()
        if not line:
            continue
        if line.startswith('>'):
            if cur:
                seqs.append(''.join(cur))
                cur=[]
        else:
            cur.append(line)
    if cur:
        seqs.append(''.join(cur))
    return seqs

def parsimony_informative_sites(seqs):
    if not seqs:
        return 0
    lengths={len(s) for s in seqs}
    if len(lengths) "'!= 1:
        raise ValueError(f'"'not aligned: {lengths}')
    count=0
    for column in zip(*seqs):
        states=Counter(ch for ch in column if ch not in '-?')
        if sum(n >= 2 for n in states.values()) >= 2:
            count += 1
    return count

def archive_counts(path):
    counts=[]
    with zipfile.ZipFile(path) as zf:
        for name in zf.namelist():
            if name.endswith('.faa.mafft'):
                seqs=parse_fasta(zf.read(name).decode())
                counts.append(parsimony_informative_sites(seqs))
    return counts

def mann_whitney_u_first(x, y):
    ranked=sorted([(value, 0) for value in x] + [(value, 1) for value in y], key=lambda item: item[0])
    rank_sum_x=0.0
    i=0
    while i < len(ranked):
        j=i+1
        while j < len(ranked) and ranked[j][0] == ranked[i][0]:
            j += 1
        avg_rank=(i + 1 + j) / 2.0
        rank_sum_x += avg_rank * sum(group == 0 for _, group in ranked[i:j])
        i=j
    n=len(x)
    return rank_sum_x - n * (n + 1) / 2.0

animals=archive_counts(ANIMALS)
fungi=archive_counts(FUNGI)
u=mann_whitney_u_first(animals, fungi)
print('animal n sum zero min max:', len(animals), sum(animals), animals.count(0), min(animals), max(animals))
print('fungi n sum zero min max:', len(fungi), sum(fungi), fungi.count(0), min(fungi), max(fungi))
print('U animals first:', u)
os.makedirs('codex_output', exist_ok=True)
with open('codex_output/answer.txt', 'w') as fh:
    fh.write(str(int(u) if u.is_integer() else u) + '\\n')
PY
printf 'answer: '; cat codex_output/answer.txt"
