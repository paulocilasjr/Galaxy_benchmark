/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
from pathlib import Path

def fasta(data):
    out=[]; h=None; s=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if h is not None: out.append((h,''.join(s)))
            h=line[1:]; s=[]
        else: s.append(line.strip())
    if h is not None: out.append((h,''.join(s)))
    return out

def count_pis(seqs, gap_missing):
    x=0
    for j in range(len(seqs[0][1])):
        col=[s[j] for _,s in seqs]
        if gap_missing: col=[c for c in col if c not in '-?X']
        if sum(v>=2 for v in Counter(col).values())>=2: x+=1
    return x

def vals(group,gap_missing):
    out=[]
    with ZipFile(Path('inputs')/f'scogs_{group}.zip') as z:
        for name in z.namelist():
            if name.endswith('.mafft'): out.append(count_pis(fasta(z.read(name)),gap_missing))
    return out

def mw(x,y):
    allv=sorted(x+y)
    # average rank 1 based exact ties
    ranks={}
    i=0
    while i<len(allv):
        j=i+1
        while j<len(allv) and allv[j]==allv[i]: j+=1
        r=(i+1+j)/2
        ranks[allv[i]]=r
        i=j
    rx=sum(ranks[v] for v in x)
    n=len(x); m=len(y)
    u=rx-n*(n+1)/2
    return u,n*m-u,rx
for gm in [True,False]:
    x=vals('animals',gm); y=vals('fungi',gm)
    u1,u2,rx=mw(x,y)
    print('gap_missing',gm,'n',len(x),len(y),'U animals',u1,'U fungi',u2,'min',min(u1,u2),'rank sum',rx)
    print('animal freq',sorted(Counter(x).items()))
    print('fungi freq',sorted(Counter(y).items())[:20], '... distinct',len(set(y)))
PY"
