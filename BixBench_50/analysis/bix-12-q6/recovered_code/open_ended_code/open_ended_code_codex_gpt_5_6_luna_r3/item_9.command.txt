/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter, defaultdict
from pathlib import Path

def fasta(data):
    out=[]; h=None; s=[]
    for line in data.decode().splitlines():
        if line.startswith('>'):
            if h is not None: out.append((h,''.join(s)))
            h=line[1:].strip(); s=[]
        else: s.append(line.strip())
    if h is not None: out.append((h,''.join(s)))
    return out
for group in ['animals','fungi']:
  print('\\n###',group)
  with ZipFile(Path('inputs')/f'scogs_{group}.zip') as z:
    names=sorted(n for n in z.namelist() if n.endswith('.mafft'))
    d=defaultdict(list)
    for n in names:
      d[len(fasta(z.read(n)))].append(n)
    for k in sorted(d): print(k,len(d[k]),'examples',d[k][:5])
    for k in sorted(d):
      if k < 4:
        print('--- n=',k,'examples headers')
        for n in d[k][:3]: print(n,[h for h,_ in fasta(z.read(n))])
PY
