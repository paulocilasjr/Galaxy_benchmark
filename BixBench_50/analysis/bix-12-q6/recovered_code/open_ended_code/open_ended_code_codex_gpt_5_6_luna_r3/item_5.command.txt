/bin/bash -lc 'for f in inputs/scogs_animals.zip inputs/scogs_fungi.zip; do echo "### $f"; unzip -Z1 "$f" | awk '"'
  /\\.mafft"'$/ {a++}
  /'"\\.faa"'$/ && !/'"\\.mafft"'$/ {u++}
  END {print "aligned",a,"unaligned",u,"total",a+u}'"'
  unzip -Z1 \""'$f" | tail -n 20; done
printf '"'%s\\n' '--- sample aligned files ---'
for f in inputs/scogs_animals.zip inputs/scogs_fungi.zip; do echo \"### "'$f"; n=$(unzip -Z1 "$f" | awk '"'/\\.mafft"'$/{print; exit}'"'); echo \""'$n"; unzip -p "$f" "$n" | sed -n '"'1,24p'; done
printf '%s\\n' '--- manifests ---'
sed -n '1,240p' inputs_manifest.json
printf '%s\\n' '--- prompt ---'
sed -n '1,240p' prompt.txt"
