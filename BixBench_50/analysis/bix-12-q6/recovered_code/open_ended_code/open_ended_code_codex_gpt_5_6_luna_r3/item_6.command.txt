/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter
from pathlib import Path

for path in [Path('inputs/scogs_animals.zip'), Path('inputs/scogs_fungi.zip')]:
    print(path)
    with ZipFile(path) as z:
        names = sorted(n for n in z.namelist() if n.endswith('.mafft'))
        lens = Counter(); ntaxa = Counter(); chars=Counter(); bad=[]
        for name in names:
            text=z.read(name).decode()
            seqs=[]; h=None; s=[]
            for line in text.splitlines():
                if line.startswith('>'):
                    if h is not None: seqs.append((h,''.join(s)))
                    h=line[1:].strip(); s=[]
                else: s.append(line.strip())
            if h is not None: seqs.append((h,''.join(s)))
            lens[len(seqs[0][1])] += 1
            ntaxa[len(seqs)] += 1
            if any(len(x[1]) "'!= len(seqs[0][1]) for x in seqs): bad.append(name)
            chars.update('"''.join(x[1] for x in seqs))
        print('aligned files',len(names),'taxa counts',ntaxa,'lengths',lens,'bad',len(bad))
        print('characters',sorted(chars.items()))
        print('first ids', [x[1:] for x in names[:10]])
        # taxa names from first
        print('first alignment',names[0], [line for line in z.read(names[0]).decode().splitlines() if line.startswith('>')])
PY"
