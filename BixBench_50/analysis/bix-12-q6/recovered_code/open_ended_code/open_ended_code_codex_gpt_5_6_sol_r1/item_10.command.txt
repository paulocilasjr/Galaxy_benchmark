/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from collections import Counter

def readfa(txt):
    seqs=[]
    for line in txt.splitlines():
        if line.startswith('>'): seqs.append('')
        elif line.strip(): seqs[-1]+=line.strip()
    return seqs

def midranks(v):
    order=sorted(range(len(v)),key=v.__getitem__); r=[0.0]*len(v); i=0
    while i<len(v):
        j=i+1
        while j<len(v) and v[order[j]]==v[order[i]]: j+=1
        mr=((i+1)+j)/2
        for k in range(i,j): r[order[k]]=mr
        i=j
    return r

def ustat(a,b):
    ranks=midranks(a+b); ua=sum(ranks[:len(a)])-len(a)*(len(a)+1)/2
    return ua,len(a)*len(b)-ua,sum(ranks[:len(a)])

def get(zpath, gaps_missing=True):
    out=[]; ntaxa=Counter(); invalid=[]
    with ZipFile(zpath) as z:
        for name in z.namelist():
            if not name.endswith('.mafft'): continue
            ss=readfa(z.read(name).decode()); ntaxa[len(ss)]+=1
            if not ss or len({len(s) for s in ss})"'!=1:
                invalid.append(name); continue
            npi=0
            for col in zip(*ss):
                c=Counter(x.upper() for x in col if not (gaps_missing and x in '"'-?Xx*'))
                if sum(v>=2 for v in c.values())>=2: npi+=1
            out.append(npi)
    return out,ntaxa,invalid
for gapmode in (True,False):
    vals={}
    print('GAPS_MISSING',gapmode)
    for g in ('animals','fungi'):
        v,n,inv=get('inputs/scogs_'+g+'.zip',gapmode); vals[g]=v
        print(g,'N',len(v),'ntaxa',dict(sorted(n.items())),'invalid',inv,'zeros',v.count(0),'positive',sum(x>0 for x in v),'sum',sum(v),'mean',sum(v)/len(v),'range',min(v),max(v))
        print('frequency first',sorted(Counter(v).items())[:20])
    ua,uf,rs=ustat(vals['animals'],vals['fungi'])
    print('U animals',ua,'U fungi',uf,'min',min(ua,uf),'product',len(vals['animals'])*len(vals['fungi']),'rank sum animals',rs)
    # write temp values to stdout as R literals compact
    if gapmode:
        print('A=',','.join(map(str,vals['animals'])))
        print('F=',','.join(map(str,vals['fungi'])))
PY"
