/bin/bash -c "cd /workspace/tmp/scogs.ktaXc8 && python3 - <<'PY'
from pathlib import Path
from collections import Counter
import scipy.stats as st

def read_align(path):
    seqs=[]
    cur=[]
    name=None
    for line in open(path):
        line=line.rstrip('\\n')
        if line.startswith('>'):
            if name is not None:
                seqs.append(''.join(cur))
            name=line[1:]
            cur=[]
        else:
            cur.append(line.strip())
    if name is not None: seqs.append(''.join(cur))
    return seqs

def pic(seqs, ignore_gaps=True):
    L=min(map(len,seqs)) if seqs else 0
    if L==0: return 0
    count=0
    for col in zip(*seqs):
        counts=Counter(c for c in col if not (ignore_gaps and c in '-?.'))
        # parsimony informative: at least two distinct chars each count >=2
        if sum(v>=2 for v in counts.values()) >=2:
            count+=1
    return count

for group in ['animals','fungi']:
    arr=[]
    for f in sorted(Path(group).glob('*.mafft')):
        seqs=read_align(f)
        arr.append(pic(seqs))
    print(group, len(arr), sum(arr), min(arr), max(arr))
    if group=='animals': A=arr
    else: F=arr
res=st.mannwhitneyu(A,F,alternative='two-sided')
print('U',res.statistic,'p',res.pvalue)
print('n1 n2', len(A),len(F))
print('sum ranks animals', sum(st.rankdata(A+F)[:len(A)]), 'U manual', sum(st.rankdata(A+F)[:len(A)])-len(A)*(len(A)+1)//2)
print(st.mannwhitneyu(A,F,alternative='less').statistic, st.mannwhitneyu(A,F,alternative='greater').statistic)
PY"
