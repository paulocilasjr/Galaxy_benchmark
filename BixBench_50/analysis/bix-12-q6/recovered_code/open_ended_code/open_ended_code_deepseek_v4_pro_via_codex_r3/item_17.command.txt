/bin/bash -c "cd /workspace/tmp/scogs.ktaXc8 && python3 - <<'PY'
from pathlib import Path
from collections import Counter

def read_align(path):
    seqs=[]; cur=[]; name=None
    for line in open(path):
        line=line.rstrip('\\n')
        if line.startswith('>'):
            if name is not None: seqs.append(''.join(cur))
            name=line[1:]; cur=[]
        else: cur.append(line.strip())
    if name is not None: seqs.append(''.join(cur))
    return seqs

def pic(seqs, mode):
    cols=zip(*seqs)
    n=0
    for col in cols:
        if mode=='all':
            c=Counter(col)
        elif mode=='no_gap':
            c=Counter(x for x in col if x"'!='"'-')
        elif mode=='no_gap_x':
            c=Counter(x for x in col if x not in '-Xx')
        elif mode=='no_gap_q':
            c=Counter(x for x in col if x not in '-?')
        if sum(v>=2 for v in c.values())>=2: n+=1
    return n

def mw(A,B):
    vals=A+B
    pairs=sorted((v,i) for i,v in enumerate(vals))
    ranks=[0]*len(vals); i=0
    while i<len(pairs):
        j=i
        while j+1<len(pairs) and pairs[j+1][0]==pairs[i][0]: j+=1
        r=(i+1+j+1)/2
        for k in range(i,j+1): ranks[pairs[k][1]]=r
        i=j+1
    return sum(ranks[:len(A)])-len(A)*(len(A)+1)/2
arrs={g:[] for g in ['animals','fungi']}
for g in arrs:
    for f in sorted(Path(g).glob('*.mafft')):
        arrs[g].append(read_align(f))
for mode in ['all','no_gap','no_gap_x','no_gap_q']:
    A=[pic(s,mode) for s in arrs['animals']]
    F=[pic(s,mode) for s in arrs['fungi']]
    print(mode,sum(A),sum(F),mw(A,F))
PY"
