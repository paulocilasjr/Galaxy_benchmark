from pathlib import Path
from collections import Counter
from scipy.stats import mannwhitneyu

def parse(path):
 seqs=[]; n=None; b=[]
 for l in path.read_text().splitlines():
  if l.startswith('>'):
   if n is not None:seqs.append(''.join(b))
   n=l[1:];b=[]
  else:b.append(l.strip())
 if n is not None:seqs.append(''.join(b))
 return seqs
def count(seqs):
 L=max(map(len,seqs)); arr=[s.ljust(L,'-') for s in seqs]; ok=set('ABCDEFGHIKLMNPQRSTVWYZ'); info=0
 for col in zip(*arr):
  c=Counter(v.upper() for v in col if v.upper() in ok)
  if sum(1 for z in c.values() if z>=2)>=2:info+=1
 return info,L
for pct in [False, True]:
 vals={}
 for group in ['animals','fungi']:
  base=Path('/workspace/tmp')/f'inspect_{group}'/group
  arr=[]
  for f in sorted(base.glob('*.mafft')):
   seqs=parse(f); c,L=count(seqs)
   arr.append(c/L*100 if pct else c)
  vals[group]=arr
 print('pct',pct,'U animals fungi',mannwhitneyu(vals['animals'],vals['fungi'],alternative='two-sided').statistic)
 print('means',sum(vals['animals'])/len(vals['animals']),sum(vals['fungi'])/len(vals['fungi']))
PY"
