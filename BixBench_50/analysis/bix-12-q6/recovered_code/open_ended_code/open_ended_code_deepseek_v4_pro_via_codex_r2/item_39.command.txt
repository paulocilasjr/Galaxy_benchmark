/bin/bash -lc "python - <<'PY'
import csv
from pathlib import Path
for group in ['animals','fungi']:
 p=Path(f'/workspace/tmp/{group}_amas_summary.txt')
 lines=p.read_text().splitlines()
 print(group,'header',lines[0][:300])
 # parse tabs
 rows=list(csv.DictReader(lines, delimiter='\\t'))
 print('n rows',len(rows))
 if rows:
  print('keys',list(rows[0].keys()))
  vals=[int(r['Parsimony_informative_sites']) for r in rows]
  print('sum',sum(vals),'first10',vals[:10])
  Path(f'/workspace/tmp/{group}_amas_counts.txt').write_text('\\n'.join(map(str,vals)))
PY
python - <<'PY'
from pathlib import Path
from scipy.stats import mannwhitneyu

print(len(x),len(y),'U animals fungi',mannwhitneyu(x,y,alternative='two-sided').statistic)
