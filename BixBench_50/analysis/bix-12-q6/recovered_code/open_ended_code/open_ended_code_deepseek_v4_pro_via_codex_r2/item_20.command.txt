from pathlib import Path
from collections import Counter

def parse(path):
 seqs=[]; name=None; buf=[]
 for line in path.read_text().splitlines():
  if line.startswith('>'):
   if name is not None: seqs.append(''.join(buf))
   name=line[1:]; buf=[]
  else: buf.append(line.strip())
 if name is not None: seqs.append(''.join(buf))
 return seqs
for group in ['animals','fungi']:
 p=Path('/workspace/tmp')/f'inspect_{group}'/group/'1003258at2759.faa.mafft'
 seqs=parse(p)
 print(group, len(seqs), [len(s) for s in seqs])
 # sample columns differences at positions
 from itertools import zip_longest
 diffs=sum(1 for col in zip_longest(*seqs, fillvalue='-') if len(set(c.upper() for c in col))>1)
 print('diff cols', diffs)
 # count standard ignoring gaps and X, requiring two states each >=2
 ok=set('ABCDEFGHIKLMNPQRSTVWYZ')
 cnt=0
 for col in zip_longest(*seqs, fillvalue='-'):
  c=Counter()
  for x in col:
   u=x.upper()
   if u in ok: c[u]+=1
  states=[v for v in c.values() if v>=2]
  if len(states)>=2: cnt+=1
 print('cnt',cnt)
PY"
