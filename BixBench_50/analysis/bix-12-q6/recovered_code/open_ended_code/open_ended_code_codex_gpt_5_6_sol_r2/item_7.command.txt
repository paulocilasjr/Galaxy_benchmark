/bin/bash -lc "command -v R || true; command -v iqtree2 || command -v iqtree || true; command -v amas || true; python - <<'PY'
import zipfile, collections, statistics

def read_fasta(s):
    seqs=[]; cur=[]
    for line in s.decode().splitlines():
        if line.startswith('>'):
            if cur: seqs.append(''.join(cur)); cur=[]
        else: cur.append(line.strip())
    if cur:seqs.append(''.join(cur))
    return seqs

def pis(seqs,gap_state=False):
    n=0
    for col in zip(*seqs):
        c=collections.Counter(col)
        if not gap_state: c.pop('-',None)
        if sum(v>=2 for v in c.values()) >= 2: n += 1
    return n

def ustat(x,y):
    # U_x = number of y lower than x + half ties
    return sum((a>b) + .5*(a==b) for a in x for b in y)
for gap in [False,True]:
    vals=[]
    print('GAPS_AS_STATE',gap)
    for zp in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
        with zipfile.ZipFile(zp) as z:
            ss=[]; nd=collections.Counter()
            for n in z.namelist():
                if n.endswith('.faa.mafft'):
                    seqs=read_fasta(z.read(n)); nd[len(seqs)]+=1; ss.append(pis(seqs,gap))
            vals.append(ss)
            print(zp, 'n',len(ss),'nseqdist',dict(sorted(nd.items())), 'sum',sum(ss),'mean',statistics.mean(ss),'median',statistics.median(ss),'zero',ss.count(0),'range',(min(ss),max(ss)))
    a,f=vals; u=ustat(a,f)
    print('U animals first',u,'U fungi first',ustat(f,a),'smaller',min(u,ustat(f,a)),'nprod',len(a)*len(f))
PY"
