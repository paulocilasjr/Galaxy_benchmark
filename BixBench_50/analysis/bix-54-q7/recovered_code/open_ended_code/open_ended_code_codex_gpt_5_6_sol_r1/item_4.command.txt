/bin/bash -lc "Rscript - <<'RS'
d <- read.csv('inputs/Swarm_2.csv', check.names=FALSE)
ratio_p <- function(x) vapply(strsplit(as.character(x), ':', fixed=TRUE), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
mix <- subset(d, StrainNumber == '287_98')
mix"'$p287 <- ratio_p(mix$Ratio)
ends <- subset(d, StrainNumber %in% c('"'98','287'))
ends"'$p287 <- ifelse(ends$StrainNumber == '"'287', 1, 0)
sets <- list(interior=mix, complete=rbind(ends, mix))
for (nm in names(sets)) {
  z <- sets[[nm]]
  fits <- list(
    quadratic=lm(Area ~ p287 + I(p287"'^2), data=z),
    cubic=lm(Area ~ p287 + I(p287''^2) + I(p287''^3), data=z),
    ns4=lm(Area ~ splines::ns(p287, df=4), data=z)
  )
  cat('"'\\nSCOPE',nm,'n=',nrow(z),'range=',paste(range(z"'$p287),collapse='"'..'),'\\n')
  for (fn in names(fits)) {
    f <- fits[[fn]]
    opt <- optimize(function(p) -predict(f,newdata=data.frame(p287=p)), interval=range(z"'$p287), tol=.Machine$double.eps''^.5)
    cat(fn, '"'df=', attr(logLik(f),'df'), 'AIC=',AIC(f),'adjR2=',summary(f)"'$adj.r.squared,'"'optp=',opt"'$minimum,'"'predmax=',-opt"'$objective,'"'\\n')
  }
}

# aggregated frequency means (for sensitivity only)
for (nm in names(sets)) {
 z <- aggregate(Area~p287, sets[[nm]], mean)
 fits <- list(quadratic=lm(Area~p287+I(p287"'^2),z),cubic=lm(Area~p287+I(p287''^2)+I(p287''^3),z),ns4=lm(Area~splines::ns(p287,df=4),z))
 cat('"'\\nAGG',nm,'n=',nrow(z),'\\n')
 for(fn in names(fits)) {
  f<-fits[[fn]]; opt<-optimize(function(p)-predict(f,data.frame(p287=p)),range(z"'$p287),tol=.Machine$double.eps''^.5)
  cat(fn,'"'AIC=',AIC(f),'adjR2=',summary(f)"'$adj.r.squared,'"'optp=',opt"'$minimum,'"'predmax=',-opt"'$objective,'"'\\n')
 }
}
RS"
