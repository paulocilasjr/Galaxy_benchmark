/bin/bash -lc "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv')
parse_prop <- function(x) {
  parts <- strsplit(as.character(x), ':')
  vapply(parts, function(z) {
    vals <- as.numeric(z)
    vals[1] / sum(vals)
  }, numeric(1))
}
mix <- subset(d, StrainNumber == '287_98')
mix"'$freq <- parse_prop(mix$Ratio)
models <- list(
  quadratic = lm(Area ~ poly(freq, 2, raw = TRUE), data = mix),
  cubic = lm(Area ~ poly(freq, 3, raw = TRUE), data = mix),
  ns4 = lm(Area ~ ns(freq, df = 4), data = mix)
)
best_name <- names(which.min(vapply(models, AIC, numeric(1))))
best <- models[[best_name]]
lo <- min(mix$freq)
hi <- max(mix$freq)
pred_at <- function(x) as.numeric(predict(best, newdata = data.frame(freq = x)))
opt <- optimize(function(x) -pred_at(x), interval = c(lo, hi))
answer <- -opt$objective
dir.create('"'codex_output', showWarnings = FALSE)
writeLines(format(answer, digits = 10, scientific = FALSE, trim = TRUE), 'codex_output/answer.txt')
cat(best_name, format(opt"'$minimum, digits = 10), format(answer, digits = 10, scientific = FALSE), '"'\\n')
RS
cat codex_output/answer.txt"
