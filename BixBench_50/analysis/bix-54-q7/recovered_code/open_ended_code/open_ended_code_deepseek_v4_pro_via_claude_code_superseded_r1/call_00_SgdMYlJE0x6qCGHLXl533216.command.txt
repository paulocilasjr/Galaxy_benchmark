Rscript -e '
library(splines)
d <- read.csv("inputs/Swarm_2.csv")
compute_proportion <- function(strain, ratio_str) {
  if (strain == 287 && ratio_str == "1:0") return(1.0)
  else if (strain == 98 && ratio_str == "1:0") return(0.0)
  else if (strain == "287_98") {
    parts <- as.numeric(strsplit(ratio_str, ":")[[1]])
    return(parts[1] / (parts[1] + parts[2]))
  } else return(NA)
}
d$prop287 <- mapply(compute_proportion, d$StrainNumber, d$Ratio)
d <- d[!is.na(d$prop287), ]
m_ns <- lm(Area ~ ns(prop287, df=4), data=d)
obj <- function(p) predict(m_ns, newdata=data.frame(prop287=p))
opt <- optimise(obj, interval=c(0, 1), maximum=TRUE)
cat(sprintf("%.6f\n", opt$objective))
' 2>&1
