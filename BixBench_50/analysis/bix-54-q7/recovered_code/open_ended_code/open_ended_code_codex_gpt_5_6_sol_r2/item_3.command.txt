/bin/bash -lc "Rscript -e '
d <- read.csv(\"inputs/Swarm_2.csv\", check.names=FALSE)
d <- subset(d, StrainNumber %in% c(\"98\", \"287\", \"287_98\"))
d"'$frequency <- ifelse(d$StrainNumber == "98", 0, ifelse(d$StrainNumber == "287", 1, vapply(strsplit(d$Ratio, ":", fixed=TRUE), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))))
library(splines)
mods <- list(quadratic=lm(Area ~ frequency + I(frequency''^2), d), cubic=lm(Area ~ frequency + I(frequency''^2) + I(frequency''^3), d), natural_spline_df4=lm(Area ~ ns(frequency, df=4), d))
print(data.frame(model=names(mods), n=vapply(mods,nobs,numeric(1)), k=vapply(mods,function(m) length(coef(m)),numeric(1)), AIC=vapply(mods,AIC,numeric(1)), BIC=vapply(mods,BIC,numeric(1)), R2=vapply(mods,function(m) summary(m)$r.squared,numeric(1)), adjR2=vapply(mods,function(m) summary(m)$adj.r.squared,numeric(1)), RMSE=vapply(mods,function(m) sqrt(mean(resid(m)''^2)),numeric(1))))
for (nm in names(mods)) {
  m <- mods[[nm]]
  oo <- optimize(function(x) -predict(m, newdata=data.frame(frequency=x)), interval=c(0,1), tol=.Machine$double.eps''^0.25)
  cat("'"\\n\", nm, \" optimum_frequency=\", format(oo"'$minimum, digits=17), " predicted_max=", format(-oo$objective, digits=17), "'"\\n\", sep=\"\")
  print(coef(m), digits=17)
}
'"
