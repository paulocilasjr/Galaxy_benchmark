/bin/bash -lc "Rscript -e '
d <- read.csv(\"inputs/Swarm_2.csv\", check.names=FALSE)
d <- subset(d, StrainNumber %in% c(\"98\", \"287\", \"287_98\"))
d"'$frequency <- ifelse(d$StrainNumber == "98", 0, ifelse(d$StrainNumber == "287", 1, vapply(strsplit(d$Ratio, ":", fixed=TRUE), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))))
library(splines)
m <- lm(Area ~ ns(frequency, df=4), data=d)
g <- seq(0, 1, length.out=100001)
y <- predict(m, newdata=data.frame(frequency=g))
i <- which.max(y)
br <- c(g[max(1,i-2)], g[min(length(g),i+2)])
o <- optimize(function(x) -as.numeric(predict(m, newdata=data.frame(frequency=x))), interval=br, tol=.Machine$double.eps)
stopifnot(nobs(m) == 42, o$minimum > 0, o$minimum < 1, -o$objective >= max(c(predict(m, data.frame(frequency=0)), predict(m, data.frame(frequency=1)))))
cat(sprintf("frequency=%.12f'"\\nmaximum=%.12f\\ngrid_max=%.12f\\n\", o"'$minimum, -o$objective, max(y)))
'"'"
