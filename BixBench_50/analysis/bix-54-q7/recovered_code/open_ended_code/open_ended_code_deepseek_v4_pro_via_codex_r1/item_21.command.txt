/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv')
dd <- d[d"'$StrainNumber=='"'287_98',]
# alternative second component as focal
dd"'$p2 <- vapply(strsplit(dd$Ratio, '"':'), function(z) as.numeric(z[2])/sum(as.numeric(z)), numeric(1))
fit <- lm(Area ~ ns(p2, df=4), data=dd)
pr <- function(x) predict(fit, newdata=data.frame(p2=x))
o <- optimize(function(x) pr(x), interval=range(dd"'$p2), maximum=TRUE)
print(c(p2max=o$maximum, area=o$objective), digits=20)
# AIC compare
f2=lm(Area~p2+I(p2''^2),data=dd);f3=lm(Area~p2+I(p2''^2)+I(p2''^3),data=dd)
cat(AIC(f2),AIC(f3),AIC(fit),'"'\\n')
RS"
