/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv'); dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio,'"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
# default boundary knots range
f1 <- lm(Area~ns(p,df=4),data=dd)
# fixed boundary 0,1
f2 <- lm(Area~ns(p,df=4,Boundary.knots=c(0,1)),data=dd)
# fixed boundary min/max same
f3 <- lm(Area~ns(p,df=4,Boundary.knots=range(dd"'$p)),data=dd)
for(nm in c('"'default','bk01','bkrange')) {
 f <- switch(nm,default=f1,bk01=f2,bkrange=f3)
 pr <- function(x) predict(f,newdata=data.frame(p=x))
 o <- optimize(pr,interval=range(dd"'$p),maximum=TRUE,tol=1e-12)
 cat(nm,'"'AIC',AIC(f),'maxp',o"'$maximum,'"'maxy',o"'$objective,'"'\\n')
}
# compare predictions at 0.9076577
cat('pred default',predict(f1,newdata=data.frame(p=.9076577)),'bk01',predict(f2,newdata=data.frame(p=.9076577)),'\\n')
# inspect knots
print(attr(ns(dd"'$p,df=4),'"'knots'))
print(attr(ns(dd"'$p,df=4,Boundary.knots=c(0,1)),'"'knots'))
RS"
