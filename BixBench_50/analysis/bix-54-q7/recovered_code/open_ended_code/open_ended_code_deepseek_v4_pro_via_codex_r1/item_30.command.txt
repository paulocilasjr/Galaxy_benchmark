/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv'); dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio,'"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
fit <- lm(Area~ns(p,df=4), data=dd)
for(p in c(0.9076577,0.9019310)) cat(p,predict(fit,newdata=data.frame(p=p)),'\\n')
# perhaps fit spline on logit of p
dd"'$logitp <- log(dd$p/(1-dd$p))
fitl <- lm(Area~ns(logitp,df=4), data=dd)
pr <- function(x) predict(fitl,newdata=data.frame(logitp=x))
o <- optimize(function(x) pr(x), interval=range(dd$logitp), maximum=TRUE)
cat('"'logit max',o"'$maximum, '"'p',exp(o"'$maximum)/(1+exp(o$maximum)), '"'area',o"'$objective,'"'\\n')
# maybe cubic on p; max p .818 not .907
# fit natural spline df=5?
for(df in 3:8){
 fitx<-lm(Area~ns(p,df=df),data=dd); prx<-function(x) predict(fitx,newdata=data.frame(p=x)); ox<-optimize(prx,range(dd"'$p),maximum=TRUE); cat(df,ox$maximum,ox$objective,AIC(fitx),'"'\\n')
}
RS"
