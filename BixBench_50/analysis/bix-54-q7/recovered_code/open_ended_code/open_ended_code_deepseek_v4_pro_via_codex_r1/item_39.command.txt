/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv'); dd<-d[d"'$StrainNumber=='"'287_98',]; dd"'$p<-vapply(strsplit(dd$Ratio,'"':'),function(z)as.numeric(z[1])/sum(as.numeric(z)),numeric(1))
fits<-list(q=lm(Area~p+I(p"'^2),data=dd),c=lm(Area~p+I(p''^2)+I(p''^3),data=dd),s=lm(Area~ns(p,df=4),data=dd))
for(nm in names(fits)){n=nrow(dd);k=length(coef(fits[[nm]])); aicc=AIC(fits[[nm]])+2*k*(k+1)/(n-k-1);cat(nm,'"'AIC',AIC(fits[[nm]]),'AICc',aicc,'\\n')}
RS"
