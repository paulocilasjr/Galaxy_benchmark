/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv')
dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio, '"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
fit <- lm(Area ~ ns(p, df=4), data=dd)
pred <- function(p) as.numeric(predict(fit, newdata=data.frame(p=p)))
o <- optimize(function(x) pred(x), interval=range(dd"'$p), maximum=TRUE, tol=.Machine$double.eps''^0.25)
print(o$maximum, digits=22)
print(o$objective, digits=22)
print(sprintf('"'%.20f', o"'$maximum))
print(sprintf('"'%.20f', o"'$objective))
# evaluate at exact rounded optimum maybe sensitivity
for(x in seq(o$maximum-1e-6,o$maximum+1e-6,length.out=5)) print(sprintf('"'x %.20f y %.20f',x,pred(x)))
RS"
