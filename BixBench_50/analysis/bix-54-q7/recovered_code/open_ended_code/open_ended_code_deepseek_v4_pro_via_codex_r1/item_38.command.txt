/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv')
# create p for 287/98 system, exclude strain1
d <- d[d"'$StrainNumber %in% c('"'98','287','287_98'),]
d"'$p <- ifelse(d$StrainNumber=='"'98',0, ifelse(d"'$StrainNumber=='"'287',1, vapply(strsplit(d"'$Ratio,'"':'),function(z) as.numeric(z[1])/sum(as.numeric(z)),numeric(1))))
fit <- lm(Area~ns(p,df=4),data=d)
pr<-function(x) predict(fit,newdata=data.frame(p=x))
for(dom in list(c(0,1),c(0.25,0.999001),c(0.25,1))) {
 o<-optimize(pr,dom,maximum=TRUE,tol=1e-12); cat(dom[1],dom[2],o"'$maximum,o$objective,'"'\\n')
}
# fit mixture only but predict on 0-1 maybe default ns
RS"
