/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv'); dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio,'"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
trans <- list(p=dd"'$p, logp=log(dd$p), log10p=log10(dd$p), logit=log(dd$p/(1-dd$p)), odds=dd$p/(1-dd$p), ratio=vapply(strsplit(dd$Ratio,'"':'),function(z) as.numeric(z[1])/as.numeric(z[2]),numeric(1)))
# For each transform fit ns(df=4), optimize back to p if monotonic
for(nm in names(trans)) {
 x <- trans[[nm]]; fit <- lm(Area~ns(x,df=4), data=data.frame(Area=dd"'$Area,x=x))
 prx <- function(v) predict(fit,newdata=data.frame(x=v))
 o <- optimize(prx, range(x), maximum=TRUE, tol=1e-10)
 # map xopt back to p
 xopt <- o$maximum
 if(nm=='"'p') popt <- xopt
 if(nm=='logp') popt <- exp(xopt)
 if(nm=='log10p') popt <- 10"'^xopt
 if(nm=='"'logit') popt <- exp(xopt)/(1+exp(xopt))
 if(nm=='odds') popt <- xopt/(1+xopt)
 if(nm=='ratio') popt <- xopt/(1+xopt)
 cat(nm,'xopt',xopt,'popt',popt,'area',o"'$objective,'"'AIC',AIC(fit),'\\n')
}
RS"
