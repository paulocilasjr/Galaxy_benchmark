/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('inputs/Swarm_2.csv'); dd <- d[d"'$StrainNumber=='"'287_98',]
dd"'$p <- vapply(strsplit(dd$Ratio,'"':'), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
formulas <- list(q=Area~p+I(p"'^2), c=Area~p+I(p''^2)+I(p''^3), s=Area~ns(p,df=4))
# LOOCV rmse
for(nm in names(formulas)) {
 errs <- sapply(seq_len(nrow(dd)), function(i) {
   fit <- lm(formulas[[nm]], data=dd[-i,]); predict(fit,newdata=dd[i,,drop=FALSE])-dd$Area[i]
 })
 cat(nm, '"'RMSE', sqrt(mean(errs"'^2)), '"'MAE',mean(abs(errs)),'\\n')
}
RS"
