/bin/bash -lc "Rscript - <<'RS'
d <- read.csv('inputs/Swarm_2.csv', stringsAsFactors=FALSE)
d <- d[d"'$StrainNumber == '"'287_98', , drop=FALSE]
d"'$freq287 <- vapply(strsplit(d$Ratio, '"':', fixed=TRUE), function(z) {q <- as.numeric(z); q[1]/sum(q)}, numeric(1))
m <- lm(Area ~ splines::ns(freq287, df=4), data=d)
f <- function(x) as.numeric(predict(m, newdata=data.frame(freq287=x)))
for (tol in c(1e-4,1e-8,1e-12,1e-14,1e-16)) {
  o <- optimize(function(z) -f(z), c(.89,.92), tol=tol)
  cat('tol',tol,'x',format(o"'$minimum, digits=17),'"'y',format(-o"'$objective,digits=17),'"'\\n')
}
# finite derivative and uniroot in local interval
h <- 1e-5
der <- function(x) (f(x+h)-f(x-h))/(2*h)
for (x in c(.9018,.9019,.90192,.90193,.9020)) cat('der',x,der(x),'\\n')
rt <- uniroot(der,c(.901,.903),tol=1e-12)"'$root
cat('"'root',format(rt,digits=17),'y',format(f(rt),digits=17),'der',der(rt),'\\n')
# spline knots
print(attr(splines::ns(d"'$freq287, df=4), '"'knots'))
RS"
