/bin/bash -lc "Rscript - <<'RS'
d <- read.csv('inputs/Swarm_2.csv', stringsAsFactors=FALSE)
mi <- subset(d, StrainNumber == '287_98')
mi"'$x <- vapply(strsplit(mi$Ratio, '"':', fixed=TRUE), function(z) as.numeric(z[1])/sum(as.numeric(z)), numeric(1))
m <- lm(Area ~ splines::ns(x, df=4), data=mi)
means <- aggregate(Area ~ x + Ratio, mi, mean)
means"'$pred <- predict(m, newdata=means)
print(means[order(-means$pred),], row.names=FALSE)
# use root/derivative for exact interior maxima from spline piecewise polynomial? dense + optimize enough
cat(sprintf('"'max=%.15f x=%.15f\\n', max(predict(m, data.frame(x=seq(0,1,length.out=1000001)))), seq(0,1,length.out=1000001)[which.max(predict(m, data.frame(x=seq(0,1,length.out=1000001))))]))
RS"
