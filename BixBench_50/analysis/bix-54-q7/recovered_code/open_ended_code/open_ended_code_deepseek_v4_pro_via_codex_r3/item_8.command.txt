/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv')
m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':')
num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2)); m$p <- num/(num+den)
fit <- lm(Area ~ ns(p, df=4), data=m)
cat('"'coef:\\n'); print(coef(fit)); print(summary(fit)); print(AIC(fit))
# continuous grid
pseq <- seq(min(m"'$p), max(m$p), length.out=1e6)
pred <- predict(fit, newdata=data.frame(p=pseq))
i <- which.max(pred)
cat('"'grid opt p', format(pseq[i],digits=16), 'area', format(pred[i],digits=16),'\\n')
# optimize
neg <- function(p) -predict(fit, newdata=data.frame(p=p))
o <- optimize(neg, interval=c(min(m"'$p), max(m$p)), tol=1e-12)
cat('"'optimize opt p', format(o"'$minimum,digits=16), '"'area', format(-o"'$objective,digits=16),'"'\\n')
# observed points predictions
obs <- sort(unique(m"'$p))
pr <- predict(fit,newdata=data.frame(p=obs))
print(data.frame(p=obs,pred=pr),digits=16)
# endpoints and pure if full
cat('"'\\nfull boundaries model\\n')
b <- d[d"'$StrainNumber %in% c('"'98','287'),]
b"'$p <- ifelse(b$StrainNumber=='"'287',1,0)
full <- rbind(m[,c('Area','p')], b[,c('Area','p')])
fitf <- lm(Area ~ ns(p, df=4), data=full)
o2 <- optimize(function(p) -predict(fitf,newdata=data.frame(p=p)), c(0,1), tol=1e-12)
cat('full opt p', format(o2"'$minimum,digits=16), '"'area', format(-o2"'$objective,digits=16),'"'\\n')
RS"
