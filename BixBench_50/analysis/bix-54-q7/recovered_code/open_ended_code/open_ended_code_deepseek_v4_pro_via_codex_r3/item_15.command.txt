/bin/bash -c "Rscript - <<'RS'
library(splines)
d <- read.csv('/workspace/inputs/Swarm_2.csv'); m <- d[d"'$StrainNumber=='"'287_98',]
sp <- strsplit(m"'$Ratio, '"':'); num <- as.numeric(sapply(sp, "'`[`,1)); den <- as.numeric(sapply(sp, `[`,2)); m$p <- num/(num+den)
fit <- lm(Area ~ ns(p, df=4), data=m)
pred <- function(p) predict(fit,newdata=data.frame(p=p))
for (rng in list(c(0,1),c(.25,.999000999),c(.25,1))) {
  g <- seq(rng[1],rng[2],length.out=1e6); y <- pred(g); i <- which.max(y)
  cat('"'range',rng[1],rng[2],'p',format(g[i],digits=16),'area',format(y[i],digits=16),'\\n')
}
RS"
