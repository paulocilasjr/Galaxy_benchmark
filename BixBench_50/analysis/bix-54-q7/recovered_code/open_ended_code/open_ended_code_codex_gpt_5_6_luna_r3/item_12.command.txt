/bin/bash -lc "Rscript -e '
d <- read.csv(\"inputs/Swarm_2.csv\", stringsAsFactors=FALSE); m <- subset(d, StrainNumber==\"287_98\"); z <- strsplit(m"'$Ratio,":",fixed=TRUE); m$freq <- vapply(z,function(x) as.numeric(x[1])/sum(as.numeric(x)),numeric(1)); mods <- list(quadratic=lm(Area~freq+I(freq''^2),m), cubic=lm(Area~freq+I(freq''^2)+I(freq''^3),m), spline_df4=lm(Area~splines::ns(freq,df=4),m)); for(nm in names(mods)){f<-mods[[nm]]; for(int in list(c(min(m$freq),max(m$freq)),c(0,1))){o<-optimize(function(x)as.numeric(predict(f,data.frame(freq=x))),interval=int,maximum=TRUE,tol=.Machine$double.eps''^0.5); cat(nm,"interval",paste(int,collapse=","),"x",format(o$maximum,digits=15),"y",format(o$objective,digits=15),"'"\\n\")}; grid<-seq(min(m"'$freq),max(m$freq),length.out=100001); pr<-as.numeric(predict(f,data.frame(freq=grid))); ii<-which.max(pr); cat("grid",grid[ii],pr[ii],"'"\\n\") }
'"
