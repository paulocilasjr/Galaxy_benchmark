/bin/bash -lc "Rscript --vanilla - <<'RS'
suppressPackageStartupMessages(library(splines))
d0 <- read.csv('inputs/Swarm_2.csv', stringsAsFactors=FALSE)
keep <- d0"'$StrainNumber %in% c('"'98','287','287_98')
d <- d0[keep,]
d"'$freq287 <- NA_real_
d$freq287[d$StrainNumber=='"'98'] <- 0
d"'$freq287[d$StrainNumber=='"'287'] <- 1
mix <- d"'$StrainNumber=='"'287_98'
p <- do.call(rbind,strsplit(d"'$Ratio[mix],'"':',fixed=TRUE))
d"'$freq287[mix] <- as.numeric(p[,1])/(as.numeric(p[,1])+as.numeric(p[,2]))
mods <- list(quadratic=lm(Area~poly(freq287,2,raw=TRUE),d), cubic=lm(Area~poly(freq287,3,raw=TRUE),d), natural_spline_df4=lm(Area~ns(freq287,df=4),d))
print(table(d$StrainNumber))
print(data.frame(model=names(mods),AIC=sapply(mods,AIC),BIC=sapply(mods,BIC),RSS=sapply(mods,deviance),adjR2=sapply(mods,function(x)summary(x)$adj.r.squared)))
for(nm in names(mods)) {
 f <- function(x)as.numeric(predict(mods[[nm]],newdata=data.frame(freq287=x)))
 grid<-seq(0,1,length.out=1000001); pv<-predict(mods[[nm]],data.frame(freq287=grid)); i<-which.max(pv)
 # optimize bracket around global grid max to refine
 lo<-max(0,grid[max(1,i-2)]); hi<-min(1,grid[min(length(grid),i+2)])
 oo<-if(i==1||i==length(grid)) list(maximum=grid[i],objective=pv[i]) else optimize(f,interval=c(lo,hi),maximum=TRUE)
 cat(nm,'"'grid x',format(grid[i],digits=15),'area',format(pv[i],digits=15),'opt x',format(oo"'$maximum,digits=15),'"'area',format(oo"'$objective,digits=15),'"'\\n')
}
RS"
