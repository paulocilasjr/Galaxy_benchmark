/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
counts=pd.read_csv('/workspace/inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
meta=pd.read_csv('/workspace/inputs/GeneMetaInfo_Zenodo.csv', index_col=0)
samp=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv')
pc_idx=meta.index[meta['gene_biotype']=='protein_coding']
length=meta.loc[pc_idx,'Length'].astype(float).values
print('genes',len(pc_idx))
results={}
for ct in ['CD4','CD8','CD14','CD19']:
    cols=samp.loc[samp['celltype']==ct,'sample']
    sub=counts.loc[pc_idx, cols].astype(float)
    # variants
    avg=sub.mean(axis=1).values
    log_avg=np.log2(avg+1)
    avg_log=np.log2(sub+1).mean(axis=1).values
    med=np.median(sub, axis=1)
    log_med=np.log2(med+1)
    variants={'raw_mean':avg,'log2(mean+1)':log_avg,'mean_log2(x+1)':avg_log,'median':med,'log2(median+1)':log_med}
    for name,y in variants.items():
        r=np.corrcoef(length,y)[0,1]
        results.setdefault(name,{})[ct]=r
for name,d in results.items():
    weakest=min(d, key=lambda k: abs(d[k]))
    print(name, {k:round(v,6) for k,v in d.items()}, 'weakest', weakest)
PY"
