/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
counts=pd.read_csv('/workspace/inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
meta=pd.read_csv('/workspace/inputs/GeneMetaInfo_Zenodo.csv', index_col=0)
samp=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv')
# exact
pc=meta['gene_biotype']=='protein_coding'
# contains any protein_coding token
pc_any=meta['gene_biotype'].str.split(';').map(lambda x: 'protein_coding' in x)
print('exact',pc.sum(),'any',pc_any.sum(),'diff',np.where(pc_any.values & ~pc.values)[0], meta.index[pc_any & ~pc])
for label, mask in [('exact',pc),('contains',pc_any)]:
    idx=meta.index[mask]
    length=meta.loc[idx,'Length'].astype(float)
    print(label,len(idx))
    for ct in ['CD4','CD8','CD14','CD19']:
        cols=samp.loc[samp['celltype']==ct,'sample']
        sub=counts.loc[idx,cols].astype(float)
        for name,y in [('raw',sub.mean(axis=1)),('log',np.log2(sub.mean(axis=1)+1))]:
            print(ct,name,round(np.corrcoef(length,y)[0,1],6))
PY"
