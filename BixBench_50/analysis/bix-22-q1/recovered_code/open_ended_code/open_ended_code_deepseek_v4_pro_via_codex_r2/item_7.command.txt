/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
start=time.time()
counts=pd.read_csv('/workspace/inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
print('counts', counts.shape, 'load time', time.time()-start)
meta=pd.read_csv('/workspace/inputs/GeneMetaInfo_Zenodo.csv', index_col=0)
samp=pd.read_csv('/workspace/inputs/Sample_annotated_Zenodo.csv')
# align meta order maybe count index same
print('index equal?', counts.index.equals(meta.index))
# protein coding simple exact
pc = meta['gene_biotype']=='protein_coding'
print('pc', pc.sum())
# ensure in counts
pc_idx = meta.index[pc]
print('all pc in counts?', pc_idx.isin(counts.index).sum(), len(pc_idx))
# cell columns
ct_order=['CD4','CD8','CD14','CD19']
for ct in ct_order:
    cols=samp.loc[samp['celltype']==ct,'sample'].tolist()
    # all cols exist
    sub=counts.loc[pc_idx, cols]
    print(ct, 'cols', len(cols), 'genes', sub.shape[0], 'zero frac', (sub==0).mean().mean())
    avg=sub.mean(axis=1)
    length=meta.loc[pc_idx,'Length']
    r=np.corrcoef(length, avg)[0,1]
    rlog=np.corrcoef(length, np.log2(avg+1))[0,1]
    rlog10=np.corrcoef(length, np.log10(avg+1))[0,1]
    # spearman raw
    from scipy.stats import pearsonr, spearmanr
    print('raw pearson', pearsonr(length, avg)[0], 'abs', abs(pearsonr(length, avg)[0]), 'log2', pearsonr(length,np.log2(avg+1))[0], 'spearman raw', spearmanr(length,avg)[0])
print('done', time.time()-start)
PY"
