/bin/bash -lc "python - <<'PY'
import csv, collections
n=0; pc=0; dup=[]; seen=set(); bio=collections.Counter()
with open('inputs/GeneMetaInfo_Zenodo.csv', newline='') as f:
    for r in csv.DictReader(f):
        n+=1; bio[r['gene_biotype']]+=1
        g=r['Geneid']
        if g in seen: dup.append(g)
        seen.add(g)
        if r['gene_biotype']=='protein_coding': pc+=1
print('meta genes', n, 'protein_coding', pc, 'dups', len(dup), dup[:10])
print('top biotypes', bio.most_common(10))
with open('inputs/BatchCorrectedReadCounts_Zenodo.csv', newline='') as f:
    rr=csv.reader(f); next(rr)
    ncounts=0; dupc=[]; seenc=set()
    for row in rr:
        ncounts+=1; g=row[0]
        if g in seenc: dupc.append(g)
        seenc.add(g)
print('count genes', ncounts, 'dups', len(dupc), dupc[:10])
print('overlap', len(seen & seenc))
PY"
