/bin/bash -lc "python - <<'PY'
try:
 import numpy as np
 print('numpy',np.__version__)
except Exception as e: print('no numpy',e)
PY
python - <<'PY'
import csv, collections
ann_path='inputs/Sample_annotated_Zenodo.csv'; meta_path='inputs/GeneMetaInfo_Zenodo.csv'; expr_path='inputs/BatchCorrectedReadCounts_Zenodo.csv'
with open(ann_path,newline='') as f:
 r=csv.DictReader(f); anns=list(r)
print('annotation rows',len(anns))
print('celltypes',dict(collections.Counter(x['celltype'] for x in anns)))
print('samtype x celltype',dict(collections.Counter((x['samtype'],x['celltype']) for x in anns)))
samples=[x['sample'] for x in anns]
print('annotation unique samples',len(set(samples)),'dupes',len(samples)-len(set(samples)))
with open(expr_path,newline='') as f: header=next(csv.reader(f))
cols=header[1:]
print('matrix sample cols',len(cols),'unique',len(set(cols)))
print('matrix-not-annotation',len(set(cols)-set(samples)),sorted(set(cols)-set(samples))[:10])
print('annotation-not-matrix',len(set(samples)-set(cols)),sorted(set(samples)-set(cols))[:10])
with open(meta_path,newline='') as f:
 r=csv.DictReader(f); metas=list(r)
idx=[x[''] for x in metas]; gids=[x['Geneid'] for x in metas]
print('metadata rows',len(metas),'index unique',len(set(idx)),'Geneid unique',len(set(gids)))
print('index equals Geneid all',all(a==b for a,b in zip(idx,gids)))
print('protein_coding count',sum(x['gene_biotype']=='protein_coding' for x in metas))
print('biotype top',collections.Counter(x['gene_biotype'] for x in metas).most_common(12))
print('length missing',sum(not x['Length'] for x in metas),'nonpositive',sum(float(x['Length'])<=0 for x in metas if x['Length']))
PY"
