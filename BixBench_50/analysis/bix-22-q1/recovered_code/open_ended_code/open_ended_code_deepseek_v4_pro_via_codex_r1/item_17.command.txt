/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import pearsonr
meta=pd.read_csv('inputs/GeneMetaInfo_Zenodo.csv').set_index('Unnamed: 0')
ann=pd.read_csv('inputs/Sample_annotated_Zenodo.csv')
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv').set_index('Unnamed: 0')
pc=meta.index[meta.gene_biotype.eq('protein_coding')]
length=meta.loc[pc,'Length'].astype(float)
for ct in ['CD4','CD8','CD14','CD19']:
 cols=ann.loc[ann.celltype.eq(ct),'sample']
 avg=counts.loc[pc, cols].mean(axis=1)
 r=pearsonr(length, avg).statistic
 print(ct, f'{r:.12f}', f'{abs(r):.12f}')
PY"
