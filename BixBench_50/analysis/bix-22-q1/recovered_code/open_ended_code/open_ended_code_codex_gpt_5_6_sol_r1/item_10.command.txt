/bin/bash -lc "Rscript - <<'RS'
expr <- read.csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', check.names=FALSE, row.names=1)
gmeta <- read.csv('inputs/GeneMetaInfo_Zenodo.csv', check.names=FALSE, row.names=1, stringsAsFactors=FALSE)
smeta <- read.csv('inputs/Sample_annotated_Zenodo.csv', check.names=FALSE, stringsAsFactors=FALSE)
stopifnot(identical(rownames(expr), rownames(gmeta)))
stopifnot(identical(colnames(expr), smeta"'$sample))
stopifnot(!anyDuplicated(rownames(expr)), !anyDuplicated(colnames(expr)), !anyNA(expr))
keep <- gmeta$gene_biotype == '"'protein_coding'
cts <- c('CD4','CD8','CD14','CD19')
r <- sapply(cts, function(ct) cor(gmeta"'$Length[keep], rowMeans(expr[keep, smeta$celltype == ct, drop=FALSE]), method='"'pearson'))
print(r, digits=17)
print(abs(r), digits=17)
cat('weakest', names(which.min(abs(r))), '\\n')
RS"
