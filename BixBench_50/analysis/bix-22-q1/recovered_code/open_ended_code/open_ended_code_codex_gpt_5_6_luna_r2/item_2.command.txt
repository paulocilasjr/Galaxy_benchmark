import csv, os
from collections import Counter
import pandas as pd
base='inputs'
for fn in ['GeneMetaInfo_Zenodo.csv','Sample_annotated_Zenodo.csv']:
    p=os.path.join(base,fn)
    df=pd.read_csv(p)
    print(fn, 'shape=', df.shape)
    print('columns=', list(df.columns))
    print('nulls=', df.isna().sum().to_dict())
    print('duplicates=', int(df.iloc[:,0].duplicated().sum()))
    if 'celltype' in df: print('celltype counts=', df['celltype'].value_counts(dropna=False).to_dict())
    if 'gene_biotype' in df: print('biotype counts top=', df['gene_biotype'].value_counts(dropna=False).head(15).to_dict())

p=os.path.join(base,'BatchCorrectedReadCounts_Zenodo.csv')
with open(p,newline='') as f:
    rd=csv.reader(f)
    header=next(rd)
    n=0; first=[]; last=[]
    for row in rd:
        n+=1
        if len(first)<3: first.append(row[0])
        last=row[0]
print('counts rows=',n,'columns=',len(header)-1,'first genes=',first,'last gene=',last)
print('header duplicates=',len(header[1:])-len(set(header[1:])))
print('count first col empty=',sum(1 for x in [header[0]] if x==''))
PY"
