import csv, os, importlib.util
from collections import Counter, defaultdict
base='inputs'
for fn in ['GeneMetaInfo_Zenodo.csv','Sample_annotated_Zenodo.csv']:
    p=os.path.join(base,fn)
    with open(p,newline='') as f:
        rows=list(csv.DictReader(f))
    print(fn, 'rows=',len(rows))
    print('columns=', list(rows[0]) if rows else [])
    for k in (rows[0] if rows else []):
        print(' ',k,'nulls=',sum(1 for r in rows if r[k]==''))
    key=list(rows[0])[0] if rows else None
    if key: print(' duplicates=',sum(v-1 for v in Counter(r[key] for r in rows).values() if v>1))
    if 'celltype' in (rows[0] if rows else {}): print('celltype counts=',dict(Counter(r['celltype'] for r in rows)))
    if 'gene_biotype' in (rows[0] if rows else {}): print('biotype counts top=',Counter(r['gene_biotype'] for r in rows).most_common(15))

p=os.path.join(base,'BatchCorrectedReadCounts_Zenodo.csv')
with open(p,newline='') as f:
    rd=csv.reader(f)
    header=next(rd)
    n=0; first=[]; last=''
    for row in rd:
        n+=1
        if len(first)<3: first.append(row[0])
        last=row[0]
print('counts rows=',n,'columns=',len(header)-1,'first genes=',first,'last gene=',last)
print('header duplicates=',len(header[1:])-len(set(header[1:])))
print('header sample empty=',sum(1 for x in header[1:] if x==''))

PY"
