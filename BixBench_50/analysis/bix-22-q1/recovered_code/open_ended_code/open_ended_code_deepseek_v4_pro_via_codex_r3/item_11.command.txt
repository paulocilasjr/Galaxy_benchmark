/bin/bash -lc "python3 - <<'PY'
import csv
p='inputs/BatchCorrectedReadCounts_Zenodo.csv'
with open(p,newline='') as f:
    r=csv.reader(f)
    header=next(r)
    print('header length',len(header))
    print('first',header[:5])
    print('last',header[-5:])
    rows=0
    firstrow=next(r,None)
    rows+=1
    print('firstrow len',len(firstrow), 'first', firstrow[:5], 'last',firstrow[-5:])
    # count total rows and maybe errors
    for row in r:
        rows+=1
    print('rows',rows)
PY"
