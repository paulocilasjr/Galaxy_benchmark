/bin/bash -lc "python3 - <<'PY'
import csv, collections
p='inputs/GeneMetaInfo_Zenodo.csv'
with open(p,newline='') as f:
    r=csv.reader(f)
    header=next(r)
    print(header)
    bio=collections.Counter()
    rows=0; dupids=0
    seen=set(); dups=[]
    for row in r:
        rows+=1
        bio[row[4] if len(row)>4 else '']+=1
        g=row[1] if len(row)>1 else row[0]
        if g in seen: dups.append(g)
        seen.add(g)
print('rows',rows,'unique geneid',len(seen),'dups count',len(dups),'dups',dups[:10])
for k,v in bio.most_common(40): print(k,v)
PY"
