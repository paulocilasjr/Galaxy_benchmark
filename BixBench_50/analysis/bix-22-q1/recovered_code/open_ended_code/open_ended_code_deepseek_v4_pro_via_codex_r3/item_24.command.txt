/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import pearsonr
ann=pd.read_csv('inputs/Sample_annotated_Zenodo.csv')
gene=pd.read_csv('inputs/GeneMetaInfo_Zenodo.csv')
pc=gene.loc[gene.gene_biotype.eq('protein_coding'), ['Geneid','Length']].set_index('Geneid')['Length'].astype(float)
# read counts fully, index first unnamed col
counts=pd.read_csv('inputs/BatchCorrectedReadCounts_Zenodo.csv', index_col=0)
print(counts.shape, len(pc), counts.columns.isin(ann['sample']).all())
counts_pc=counts.loc[counts.index.isin(pc.index)].copy()
# align pc order
pc2=pc.loc[counts_pc.index]
for t in ['CD4','CD8','CD14','CD19']:
    cols=ann.loc[ann.celltype.eq(t),'sample'].tolist()
    mean=counts_pc.loc[:,cols].mean(axis=1)
    r,p=pearsonr(pc2, mean)
    print(t, r, p, np.corrcoef(pc2,mean)[0,1])
PY"
