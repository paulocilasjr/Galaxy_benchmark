/bin/bash -lc "python3 - <<'PY'
import csv, math, statistics
ann={}
with open('inputs/Sample_annotated_Zenodo.csv',newline='') as f:
    for d in csv.DictReader(f): ann[d['sample']]=d['celltype']
lengths={}; pc=[]
with open('inputs/GeneMetaInfo_Zenodo.csv',newline='') as f:
    for d in csv.DictReader(f):
        if d['gene_biotype']=='protein_coding':
            pc.append(d['Geneid']); lengths[d['Geneid']]=float(d['Length'])
idx={g:i for i,g in enumerate(pc)}
with open('inputs/BatchCorrectedReadCounts_Zenodo.csv',newline='') as f:
    r=csv.reader(f); header=next(r); cols=header[1:]
    pos={t:[] for t in ['CD4','CD8','CD14','CD19']}
    for i,c in enumerate(cols,1):
        t=ann[c]
        if t in pos: pos[t].append(i)
    n=len(pc)
    rawsum={t:[0.0]*n for t in pos}; nonzero={t:[0]*n for t in pos}; npos={t:len(pos[t]) for t in pos}
    for row in r:
        g=row[0]; i=idx.get(g)
        if i is None: continue
        for t,ps in pos.items():
            s=0.0; nn=0
            for p in ps:
                v=float(row[p]); s+=v
                if v>0: nn+=1
            rawsum[t][i]=s; nonzero[t][i]=nn
L=[lengths[g] for g in pc]
def pearson_pairs(ps):
    a=[x for x,y in ps]; b=[y for x,y in ps]
    ma=statistics.fmean(a); mb=statistics.fmean(b)
    return sum((x-ma)*(y-mb) for x,y in zip(a,b))/math.sqrt(sum((x-ma)**2 for x in a)*sum((y-mb)**2 for y in b)), len(a)
for t in ['CD4','CD8','CD14','CD19']:
    means=[rawsum[t][i]/npos[t] for i in range(n)]
    pairs_all=list(zip(L,means))
    pairs_mean_pos=[(L[i],means[i]) for i in range(n) if means[i]>0]
    pairs_all_samples_pos=[(L[i],means[i]) for i in range(n) if nonzero[t][i]==npos[t]]
    print(t,'all',pearson_pairs(pairs_all))
    print(' mean>0',pearson_pairs(pairs_mean_pos))
    print(' allsamples>0',pearson_pairs(pairs_all_samples_pos))
PY"
