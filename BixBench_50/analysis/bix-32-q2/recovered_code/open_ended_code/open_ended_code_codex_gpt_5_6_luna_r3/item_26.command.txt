/bin/bash -lc "Rscript - <<'RS'
paths <- c(\"/workspace/inputs/res_1vs97.rds\", \"/workspace/inputs/res_1vs98.rds\", \"/workspace/inputs/res_1vs99.rds\")
res <- lapply(paths, function(p) {
  x <- readRDS(p)
  ld <- attr(x, \"listData\")
  data.frame(gene=attr(x, \"rownames\"), lfc=ld"'$log2FoldChange, padj=ld$padj, stringsAsFactors=FALSE)
})
names(res) <- sub("res_", "", sub("'"\\\\.rds"'$", "", basename(paths)))
links <- read.delim("https://rest.kegg.jp/link/pathway/pau", header=FALSE, sep="'"\\\\t\", quote=\"\", stringsAsFactors=FALSE)
colnames(links) <- c(\"gene\", \"pathway\")
links"'$gene <- sub("''^pau:", "", links$gene)
links$pathway <- sub("''^path:", "", links$pathway)
pathgenes <- split(links$gene, links$pathway)
pathgenes <- lapply(pathgenes, unique)
cat("KEGG link rows", nrow(links), "mapped genes", length(unique(links$gene)), "pathways", length(pathgenes), "'"\\n\")
allgenes <- lapply(res, "'`[[`, "gene")
cat("result genes", paste(sapply(allgenes, length), collapse=","), "common", length(Reduce(intersect, allgenes)), "'"\\n\")
for (nm in names(res)) {
  d <- res[[nm]]
  cat(nm, \"mapped\", sum(d"'$gene %in% unique(links$gene)), "abs", sum(abs(d$lfc)>1.5), "pos", sum(d$lfc>1.5), "neg", sum(d$lfc < -1.5), "both padj", sum(abs(d$lfc)>1.5 & d$padj<0.05), "'"\\n\")
}
RS"
