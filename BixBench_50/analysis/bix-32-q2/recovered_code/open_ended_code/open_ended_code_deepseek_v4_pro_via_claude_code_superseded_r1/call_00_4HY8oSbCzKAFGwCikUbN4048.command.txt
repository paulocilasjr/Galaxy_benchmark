R -e '
x <- readRDS("/workspace/inputs/res_1vs97.rds")
ld <- attr(x, "listData")
cat("listData class:", class(ld), "\n")
cat("listData names:", names(ld), "\n")
cat("listData lengths:\n")
for (nm in names(ld)) {
  cat(" ", nm, ":", length(ld[[nm]]), " - class:", class(ld[[nm]]), "\n")
}
cat("\nFirst 10 rows of each column:\n")
n_rows <- length(ld[[1]])
show_rows <- min(10, n_rows)
for (nm in names(ld)) {
  cat("\n", nm, ":\n", sep="")
  print(head(ld[[nm]], show_rows))
}
' 2>&1
