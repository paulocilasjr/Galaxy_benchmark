import requests
import time

headers = {'User-Agent': 'Python/3.11'}

# Get all gene-to-pathway links for pau organism
r = requests.get('https://rest.kegg.jp/link/pathway/pau', headers=headers)
gene_to_pathway = {}
for line in r.text.strip().split('\n'):
    if not line.strip():
        continue
    parts = line.split('\t')
    gene = parts[0].strip()  # e.g., pau:PA14_07190
    pathway = parts[1].strip()  # e.g., path:pau00010
    if gene not in gene_to_pathway:
        gene_to_pathway[gene] = []
    gene_to_pathway[gene].append(pathway)

print(f'Total genes with pathways: {len(gene_to_pathway)}')

# Get pathway names
r = requests.get('https://rest.kegg.jp/list/pathway/pau', headers=headers)
pathway_names = {}
for line in r.text.strip().split('\n'):
    if not line.strip():
        continue
    parts = line.split('\t')
    pid = parts[0].strip()  # e.g., pau01100
    name = parts[1].strip() if len(parts) > 1 else ''
    pathway_names[pid] = name

print(f'Total pathways: {len(pathway_names)}')

# Get all PA14 genes from KEGG
r = requests.get('https://rest.kegg.jp/list/pau', headers=headers)
all_kegg_genes = set()
for line in r.text.strip().split('\n'):
    if not line.strip():
        continue
    parts = line.split('\t')
    gene_id = parts[0].strip()
    all_kegg_genes.add(gene_id)

print(f'Total PA14 genes in KEGG: {len(all_kegg_genes)}')

# Save the mappings
import json
with open('/workspace/gene_to_pathway.json', 'w') as f:
    json.dump(gene_to_pathway, f)
with open('/workspace/pathway_names.json', 'w') as f:
    json.dump(pathway_names, f)
with open('/workspace/all_kegg_genes.json', 'w') as f:
    json.dump(list(all_kegg_genes), f)
