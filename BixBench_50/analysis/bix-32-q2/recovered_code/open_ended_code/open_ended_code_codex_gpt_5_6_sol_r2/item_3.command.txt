/bin/bash -lc "Rscript - <<'RS'
for (f in Sys.glob('inputs/*.rds')) {
  x <- readRDS(f)
  cat('\\nFILE', f, '\\n')
  cat('class:', paste(class(x), collapse=','), ' dim:', paste(dim(x), collapse='x'), '\\n')
  cat('names:', paste(names(x), collapse=' | '), '\\n')
  cat('rownames head:', paste(head(rownames(x),10), collapse=' | '), '\\n')
  cat('attributes:', paste(names(attributes(x)), collapse=' | '), '\\n')
  print(summary(x))
  print(head(as.data.frame(x),3))
}
RS"
