/bin/bash -lc "Rscript - <<'RS'
for(f in Sys.glob('inputs/*.rds')) {
 x<-readRDS(f); a<-attributes(x)
 cat('\\n',f,'\\n')
 md <- a[['metadata']]
 cat('metadata typeof',typeof(md),'len',length(md),'names',names(md),'\\n')
 str(md, max.level=3)
 em <- attributes(a[['elementMetadata']])
 cat('EM attrs:',names(em),'\\n')
 ld <- em[['listData']]
 cat('EM fields:',names(ld),'\\n')
 for(i in seq_along(ld)) {
   z<-ld[[i]]; cat(i,names(ld)[i],typeof(z),length(z),'\\n'); if(is.character(z)) print(z)
 }
}
RS"
