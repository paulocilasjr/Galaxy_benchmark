/bin/bash -lc "Rscript - <<'RS'
options(timeout=120)
files <- c('97'='inputs/res_1vs97.rds','98'='inputs/res_1vs98.rds','99'='inputs/res_1vs99.rds')
res <- lapply(files, function(f) {x<-readRDS(f);a<-attributes(x);d<-as.data.frame(a[['listData']]);rownames(d)<-a[['rownames']];d})
cat('dims:',sapply(res,nrow),'row sets identical:',all(vapply(res[-1],function(x) setequal(rownames(res[[1]]),rownames(x)),logical(1))),'\\n')
ln<-readLines('https://rest.kegg.jp/link/pathway/pau',warn=FALSE);z<-strsplit(ln,'\\\\t')
links<-data.frame(gene=sub('"'^pau:'"','',vapply(z,"'`[`,character(1),1)),pathway=sub('"'"'^path:'"','',vapply(z,"'`[`,character(1),2)),stringsAsFactors=FALSE)
all_rows<-Reduce(intersect,lapply(res,rownames));shared_univ<-intersect(all_rows,unique(links$gene));kegg_univ<-unique(links$gene)
cat('"'links',nrow(links),'genes',length(kegg_univ),'pathways',length(unique(links"'$pathway)),'"'all rows',length(all_rows),'shared mapped',length(shared_univ),'\\n')
ora<-function(fg,universe,minGS=10,maxGS=500){lk<-links[links"'$gene%in%universe,,drop=F];universe<-unique(lk$gene);fg<-intersect(fg,universe);sets<-lapply(split(lk$gene,lk$pathway),unique);sets<-sets[lengths(sets)>=minGS&lengths(sets)<=maxGS];N<-length(universe);n<-length(fg);k<-vapply(sets,function(s)sum(fg%in%s),integer(1));M<-lengths(sets);keep<-k>0;p<-phyper(k[keep]-1,M[keep],N-M[keep],n,lower.tail=F);data.frame(path=names(k)[keep],k=k[keep],M=M[keep],n=n,N=N,pvalue=p,padj=p.adjust(p,'"'BH'))}
run<-function(gene_rule,universe_type,path_cut){univ<-if(universe_type=='shared')shared_univ else kegg_univ;sig<-list();fgcounts<-c()
 for(nm in names(res))for(dir in c('up','down')){d<-res[[nm]];ok<-if(dir=='up')d"'$log2FoldChange>1.5 else d$log2FoldChange< -1.5;if(gene_rule=='"'padj05')ok<-ok&"'!is.na(d$padj)&d$padj<.05;if(gene_rule=='"'padj10')ok<-ok&"'!is.na(d$padj)&d$padj<.10;fgcounts<-c(fgcounts,sum(ok));o<-ora(rownames(d)[ok],univ);hit<-if(path_cut=='"'padj05')rownames(o)[o"'$padj<.05] else rownames(o)[o$pvalue<.05];# row names are paths inherited
 sig[[paste(nm,dir,sep='"'_')]]<-hit}
 up<-Reduce(intersect,sig[c('97_up','98_up','99_up')]);down<-Reduce(intersect,sig[c('97_down','98_down','99_down')]);
 cat(gene_rule,universe_type,path_cut,'fg',paste(fgcounts,collapse=','),'sig',paste(lengths(sig),collapse=','),'up',length(up),paste(up,collapse=','),'down',length(down),paste(down,collapse=','),'union',length(union(up,down)),'\\n')}
for(gr in c('lfc','padj05','padj10'))for(ut in c('shared','kegg'))for(pc in c('padj05','p05'))run(gr,ut,pc)
RS"
