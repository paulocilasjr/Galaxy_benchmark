def bh(ps):
    n=len(ps)
    if n==0:return []
    # sorted ascending index
    s=sorted(range(n), key=lambda i:ps[i])
    # order decreasing p (same as rev(s)), rank i descending
    # produce raw adjusted in original order
    out=[0.0]*n

        raw=ps[i]*n/r
