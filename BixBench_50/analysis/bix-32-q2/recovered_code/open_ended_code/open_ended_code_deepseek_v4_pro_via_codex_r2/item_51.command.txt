/bin/bash -lc "Rscript - <<'RS'
link <- read.table('/workspace/tmp/pau_link.tsv', sep='\\t', header=FALSE, stringsAsFactors=FALSE, quote='')
link[,1] <- sub('["'^:]+:'"', '', link[,1])
link[,2] <- sub('["'^:]+:'"', '', link[,2])
paths <- read.table('/workspace/tmp/pau_pathways.tsv', sep='\\t', header=FALSE, stringsAsFactors=FALSE, quote='')
link <- link[link[,2] %in% paths[,1], ]
p2g <- split(link[,1], link[,2])
g2p <- split(link[,2], link[,1])
N <- length(unique(link[,1]))
cat('N',N,'paths',length(p2g),'\\n')
ora <- function(gs) {
  gs <- intersect(unique(gs), names(g2p))
  if(length(gs)==0) return(character())
  mapped <- g2p[gs]
  qts <- unique(unlist(mapped))
  qts <- qts[sapply(p2g[qts], length) >= 10 & sapply(p2g[qts], length) <= 500]
  n <- length(mapped)
  ans <- list()
  for(t in qts) {
    k <- sum(p2g[[t]] %in% gs)
    M <- length(p2g[[t]])
    p <- phyper(k-1, M, N-M, n, lower.tail=FALSE)
    ans[[t]] <- c(k=k, M=M, n=n, p=p)
  }
  ps <- sapply(ans, function(x) x['p'])
  pa <- p.adjust(ps, method='BH')
  names(ans)[pa <= 0.05 & ps <= 0.05]
}
for(st in c('97','98','99')) {
 d <- read.csv(sprintf('/workspace/tmp/res_%s.csv',st), stringsAsFactors=FALSE)
 cat('\\n',st,'\\n')
 for(dir in c('up','down')) {
   if(dir=='up') g <- d"'$gene[d$lfc > 1.5] else g <- d$gene[d$lfc < -1.5]
   ids <- ora(g)
   cat(dir,length(ids),paste(ids,collapse='"','),'\\n')
   assign(paste0(dir,st), ids)
 }
}
up <- Reduce(intersect,list(up97,up98,up99))
down <- Reduce(intersect,list(down97,down98,down99))
cat('up common',length(up),up,'\\n')
cat('down common',length(down),down,'\\n')
cat('union',length(union(up,down)),union(up,down),'\\n')
RS"
