/bin/bash -lc "Rscript - <<'RS'
# Print exact significant pathways and p/adjust for each
link <- read.table('/workspace/tmp/pau_link.tsv',sep='\\t',header=FALSE,stringsAsFactors=FALSE,quote='')
link[,1]<-sub('["'^:]+:'"','',link[,1]);link[,2]<-sub('["'^:]+:'"','',link[,2])
paths<-read.table('/workspace/tmp/pau_pathways.tsv',sep='\\t',header=FALSE,stringsAsFactors=FALSE,quote='')
link<-link[link[,2]%in%paths[,1],]
p2g<-split(link[,1],link[,2]);g2p<-split(link[,2],link[,1]);N<-length(unique(link[,1]))
ora_details<-function(gs){
 gs<-intersect(unique(gs),names(g2p)); mapped<-g2p[gs]; qts<-unique(unlist(mapped)); qts<-qts[sapply(p2g[qts],length)>=10 & sapply(p2g[qts],length)<=500]; n<-length(mapped)
 ans<-list()
 for(t in qts){k<-sum(p2g[[t]]%in%gs);M<-length(p2g[[t]]);p<-phyper(k-1,M,N-M,n,lower.tail=FALSE);ans[[t]]<-c(k=k,M=M,n=n,p=p)}
 ps<-sapply(ans,function(x)x['p']);pa<-p.adjust(ps,'BH'); ids<-names(ps)[pa<=.05 & ps<=.05]
 data.frame(ID=ids,pvalue=ps[ids],padj=pa[ids],stringsAsFactors=FALSE)
}
for(st in c('97','98','99')){
 d<-read.csv(sprintf('/workspace/tmp/res_%s.csv',st),stringsAsFactors=FALSE)
 cat('\\nSTRAIN',st,'\\n')
 for(dir in c('up','down')){
  g<-if(dir=='up') d"'$gene[d$lfc>1.5] else d$gene[d$lfc< -1.5]
  print(ora_details(g))
 }
}
RS'
