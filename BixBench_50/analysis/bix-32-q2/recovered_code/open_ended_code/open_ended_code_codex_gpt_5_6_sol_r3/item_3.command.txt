/bin/bash -lc "Rscript - <<'RS'
files <- Sys.glob('inputs/*.rds')
for (f in files) {
  x <- readRDS(f)
  cat('\\nFILE:', f, '\\n')
  cat('class:', paste(class(x), collapse=', '), '\\n')
  cat('dim:', paste(dim(x), collapse=' x '), '\\n')
  cat('names:', paste(names(x), collapse=', '), '\\n')
  cat('rownames head:', paste(head(rownames(x), 8), collapse=', '), '\\n')
  print(head(as.data.frame(x), 3))
  cat('attributes:\\n')
  print(attributes(x)[setdiff(names(attributes(x)), c('names','row.names'))])
  cat('NA counts:\\n')
  print(colSums(is.na(as.data.frame(x))))
  cat('numeric summaries:\\n')
  print(summary(as.data.frame(x)))
}
RS"
