/bin/bash -lc "Rscript - <<'RS'
for(f in Sys.glob('inputs/*.rds')) {
 x<-readRDS(f); d<-as.data.frame(attr(x,'listData')); rownames(d)<-attr(x,'rownames')
 cat('\\n',f,'dim',paste(dim(d),collapse='x'),'\\n')
 print(head(d,10))
 cat('metadata names:',paste(names(attr(x,'metadata')),collapse=', '),'\\n')
 print(attr(x,'metadata'))
 cat('elementMetadata listData:\\n')
 em<-attr(x,'elementMetadata'); print(attr(em,'listData'))
 cat('rownames tail:',paste(tail(rownames(d)),collapse=', '),'\\n')
 cat('lfc abs>1.5:',sum(abs(d"'$log2FoldChange)>1.5,na.rm=T),'"'up:',sum(d"'$log2FoldChange>1.5,na.rm=T),'"'down:',sum(d"'$log2FoldChange< -1.5,na.rm=T),'"' padj<.05:',sum(d"'$padj<.05,na.rm=T),'"'both:',sum(abs(d"'$log2FoldChange)>1.5 & d$padj<.05,na.rm=T),'"'\\n')
}
RS"
