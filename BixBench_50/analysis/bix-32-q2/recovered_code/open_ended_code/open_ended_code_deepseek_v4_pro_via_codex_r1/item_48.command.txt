/bin/bash -lc "python3 - <<'PY'
import urllib.request, csv
from collections import defaultdict
link=urllib.request.urlopen('https://rest.kegg.jp/link/pathway/pau').read().decode().splitlines()
g2p=defaultdict(set)
for l in link:
 if '\\t' not in l: continue
 g,p=l.split('\\t'); g=g.split(':')[1]; p=p[5:]; g2p[g].add(p)
data={}
for f in ['tmp/res_1vs97.csv','tmp/res_1vs98.csv','tmp/res_1vs99.csv']:
 with open(f) as fh: data[f]={r['gene']:float(r['log2FoldChange']) for r in csv.DictReader(fh)}
vals=list(data.values())
common_up=[g for g in data[list(data)[0]] if all(v[g]>1.5 for v in vals)]
common_down=[g for g in data[list(data)[0]] if all(v[g]<-1.5 for v in vals)]
print('common_up',len(common_up)); print('common_down',len(common_down))
from collections import Counter
pcount=Counter()
for g in common_down:
 for p in g2p[g]: pcount[p]+=1
print('down common genes annotated',sum(1 for g in common_down if g in g2p))
for p,c in pcount.most_common(20): print(p,c, 'total', sum(1 for x in g2p if p in g2p[x]))
PY"
