/bin/bash -lc "Rscript -e '
for (f in list.files(\"inputs\", full.names=TRUE)) {
 cat(\"FILE\", f, \"\\n\")
 x <- readRDS(f)
 cat(\"class:\", paste(class(x), collapse=\",\"), \" length/dim:\", paste(dim(x), collapse=\"x\"), \"\\n\")
 if (is.data.frame(x) || is.matrix(x)) {
  cat(\"cols:\", paste(colnames(x), collapse=\", \"), \"\\n\")
  print(head(x,3))
  cat(\"rownames head:\", paste(head(rownames(x)), collapse=\", \"), \"\\n\")
 } else if (is.list(x)) {
  cat(\"names:\", paste(names(x), collapse=\", \"), \"\\n\")
  for (n in names(x)) {
   y <- x[[n]]
   cat(\"  elem\", n, \"class\", paste(class(y), collapse=\",\"), \"dim\", paste(dim(y), collapse=\"x\"), \"\\n\")
   if (is.data.frame(y) || is.matrix(y)) {
    cat(\"  cols:\", paste(colnames(y), collapse=\", \"), \"\\n\")
    print(head(y,2))
   } else cat(\"  head:\", paste(head(y), collapse=\", \"), \"\\n\")
  }
 }
 cat(\"---\\n\")
}
'"
