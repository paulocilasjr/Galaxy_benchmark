/bin/bash -lc "Rscript - <<'RS'
for (f in list.files('inputs', pattern='\\\\.rds"'$'"', full.names=TRUE)) {
 con <- gzfile(f,'rb'); x <- unserialize(con); close(con)
 a <- attributes(x); ld <- a"'$listData
 df <- data.frame(gene=a$rownames, baseMean=ld$baseMean, log2FoldChange=ld$log2FoldChange, lfcSE=ld$lfcSE, stat=ld$stat, pvalue=ld$pvalue, padj=ld$padj, stringsAsFactors=FALSE)
 out <- sub('"'\\\\.rds"'$'"','.csv',basename(f)); out <- file.path('tmp',out)
 write.csv(df, out, row.names=FALSE)
 cat('wrote',out,'rows',nrow(df),'\\n')
}
RS
ls -l tmp"
