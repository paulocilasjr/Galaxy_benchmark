/bin/bash -lc "python3 - <<'PY'
# use previous pvals to count raw p<0.05 intersections
import glob, pandas as pd
sets={}
for f in glob.glob('/workspace/tmp/*_pvals.txt'):
    key=f.split('/')[-1].split('_pvals')[0]
    d=pd.read_csv(f,sep='\\t')
    sets[key]=set(d.loc[d['p']<0.05,'path'])
for direction in ['up','down']:
    keys=[f'{direction}97',f'{direction}98',f'{direction}99']
    common=set.intersection(*[sets[k] for k in keys])
    print(direction, 'raw p<.05', len(common), sorted(common))
PY"
