/bin/bash -lc "cat > /workspace/tmp/check_pvals.R <<'RS'
N <- 2055
files <- c('up97','down97','up98','down98','up99','down99')
for (f in files) {
  fn <- file.path('tmp',paste0(f,'_pvals.txt'))
  p <- read.table(fn, header=TRUE, sep='\\t')
  p"'$r_padj <- p.adjust(p$p, method='"'BH')
  p"'$r_p <- phyper(p$k-1, p$K, N-p$K, p$n, lower.tail=FALSE)
  cat(f, '"'n', nrow(p), 'py sig', sum(p"'$py_padj<0.05), '"'R sig', sum(p"'$r_padj<0.05), '"'max p diff', max(abs(p"'$r_p-p$p)), '"'max padj diff', max(abs(p"'$r_padj-p$py_padj)), '"'\\n')
  if (sum(p"'$r_padj<0.05) != sum(p$py_padj<0.05)) print(head(p[order(p$r_padj),],15))
}
RS
Rscript /workspace/tmp/check_pvals.R'
