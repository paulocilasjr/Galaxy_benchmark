import rdata
import numpy as np
import warnings
warnings.filterwarnings('ignore')

result = rdata.read_rda('/workspace/inputs/res_1vs97.rds')
ld = result.listData
print('listData type:', type(ld))
print('listData keys:', list(ld.keys()) if hasattr(ld, 'keys') else dir(ld))

for k in ld.keys() if hasattr(ld, 'keys') else []:
    v = ld[k]
    print(f'{k}: type={type(v).__name__}', end='')
    if isinstance(v, np.ndarray):
        print(f' shape={v.shape} dtype={v.dtype}', end='')
        if v.ndim == 1:
            # Print first few non-NaN values
            non_nan = v[~np.isnan(v)] if v.dtype.kind == 'f' else v
            print(f' sample={non_nan[:5]}', end='')
    elif hasattr(v, '__dict__'):
        print(f' attrs={list(v.__dict__.keys())}', end='')
    print()
