Rscript -e '
.libPaths(c("/workspace/R_libs", .libPaths()))
# Install packages with no/few dependencies first
pkgs <- c("rlang", "cli", "glue", "lifecycle", "magrittr", "vctrs", "pillar", 
          "tibble", "tidyselect", "dplyr", "purrr", "tidyr", "stringr", "Rcpp", 
          "RcppArmadillo", "matrixStats", "BiocGenerics")
for (p in pkgs) {
  cat("\n=== Installing", p, "===\n")
  try(install.packages(p, repos="https://cloud.r-project.org", lib="/workspace/R_libs", dependencies=FALSE))
}
' 2>&1 | grep -E "=== |installed|ERROR|Warning.*package"
