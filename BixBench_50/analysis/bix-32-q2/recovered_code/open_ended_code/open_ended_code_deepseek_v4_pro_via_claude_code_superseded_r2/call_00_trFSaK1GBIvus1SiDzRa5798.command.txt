import rdata, numpy as np, requests, warnings
from scipy.stats import hypergeom
from statsmodels.stats.multitest import multipletests
warnings.filterwarnings('ignore')

# Load data
res97 = rdata.read_rda('/workspace/inputs/res_1vs97.rds')
res98 = rdata.read_rda('/workspace/inputs/res_1vs98.rds')
res99 = rdata.read_rda('/workspace/inputs/res_1vs99.rds')

def get_results(res):
    return {
        'genes': [str(g) for g in res.rownames],
        'lfc': res.listData['log2FoldChange'],
        'pvalue': res.listData['pvalue'],
        'padj': res.listData['padj'],
    }

data97 = get_results(res97)
data98 = get_results(res98)
data99 = get_results(res99)

# Get pathway data
r_link = requests.get('https://rest.kegg.jp/link/pau/pathway', timeout=30)
links = r_link.text.strip().split('\n')
pathway_genes = {}
for line in links:
    parts = line.strip().split('\t')
    if len(parts) == 2:
        pid = parts[0].replace('path:', '')
        gene = parts[1].replace('pau:', '')
        pathway_genes.setdefault(pid, set()).add(gene)

r_list = requests.get('https://rest.kegg.jp/list/pathway/pau', timeout=30)
pathway_names = {}
for line in r_list.text.strip().split('\n'):
    parts = line.split('\t')
    if len(parts) >= 2:
        pathway_names[parts[0].replace('path:', '')] = parts[1]

all_genes_set = set(str(g) for g in res97.rownames)
kegg_mapped = set()
for gs in pathway_genes.values():
    kegg_mapped |= gs
universe = kegg_mapped & all_genes_set

for pid in pathway_genes:
    pathway_genes[pid] = pathway_genes[pid] & universe
pathway_genes = {pid: gs for pid, gs in pathway_genes.items() if len(gs) >= 5}

# Verify: manual hypergeometric test for a specific case
# 97 up: pau00405 (Phenazine biosynthesis), k=6, n=24, K=13, N=2041
N = len(universe)
K97_up = 13
n_phenazine = len(pathway_genes['pau00405'])
k_phenazine = 6
pval = hypergeom.sf(k_phenazine - 1, N, n_phenazine, K97_up)
print(f'Manual check pau00405 97 up: N={N}, K={K97_up}, n={n_phenazine}, k={k_phenazine}, pval={pval:.2e}')

# Let me also check if any pathway appears in the top results for all three
# even if not significant at 0.05
print('\n=== Pathways approaching significance in all 3 ===')
all_pids = set(pathway_genes.keys())
results_store = {}

for label, data in [('97', data97), ('98', data98), ('99', data99)]:
    for direction in ['up', 'down']:
        genes_arr = np.array(data['genes'])
        lfc = data['lfc']
        if direction == 'up':
            mask = lfc > 1.5
        else:
            mask = lfc < -1.5
        fg = set(genes_arr[mask]) & universe
        K = len(fg)
        
        for pid in all_pids:
            pw_set = pathway_genes[pid]
            n = len(pw_set)
            k = len(fg & pw_set)
            if k == 0:
                pval = 1.0
            else:
                pval = hypergeom.sf(k - 1, N, n, K)
            results_store[(label, direction, pid)] = {'pval': pval, 'k': k, 'n': n, 'K': K}

# Apply FDR within each comparison+dir
for label in ['97', '98', '99']:
    for direction in ['up', 'down']:
        pids_list = []
        pvals_list = []
        for pid in all_pids:
            r = results_store[(label, direction, pid)]
            pids_list.append(pid)
            pvals_list.append(r['pval'])
        _, padjs, _, _ = multipletests(pvals_list, method='fdr_bh')
        for pid, padj in zip(pids_list, padjs):
            results_store[(label, direction, pid)]['padj'] = padj

# Check crossover
for direction in ['up', 'down']:
    print(f'\n--- {direction} ---')
    for pid in sorted(all_pids):
        p97 = results_store[('97', direction, pid)]['padj']
        p98 = results_store[('98', direction, pid)]['padj']
        p99 = results_store[('99', direction, pid)]['padj']
        if max(p97, p98, p99) < 0.1:  # Show borderline cases
            name = pathway_names.get(pid, '')
            print(f'  {pid}: 97_padj={p97:.4f}, 98_padj={p98:.4f}, 99_padj={p99:.4f} ({name})')
            print(f'    k: 97={results_store[(chr(34)+chr(57)+chr(55), direction, pid)][chr(34)+chr(107)+chr(34)]}, 98={results_store[(chr(34)+chr(57)+chr(56), direction, pid)][chr(34)+chr(107)+chr(34)]}, 99={results_store[(chr(34)+chr(57)+chr(57), direction, pid)][chr(34)+chr(107)+chr(34)]}')
