/bin/bash -lc "Rscript - <<'RS'
parse <- function(f){a<-attributes(readRDS(f));list(g=a"'$rownames,l=a$listData$log2FoldChange,p=a$listData$padj)}
rr<-lapply(list.files("inputs",pattern="''^res_.*'"\\\\.rds"'$",full.names=TRUE),parse)
sp<-strsplit(readLines("https://rest.kegg.jp/link/pathway/pau",warn=FALSE),"'"\\t\",fixed=TRUE); mp<-data.frame(g=sub(\""'^pau:","",vapply(sp,`[`,"",1)),p=sub("''^path:","",vapply(sp,`[`,"",2))); ss<-lapply(split(mp$g,mp$p),unique); bg<-unique(mp$g)
for (use_padj in c(FALSE,TRUE)) { z<-list(); for (i in 1:3) {r<-rr[[i]]; b<-intersect(r$g,bg); sets<-lapply(ss,intersect,y=b); keep<-lengths(sets)>=10&lengths(sets)<=500;sets<-sets[keep]; crit<-abs(r$l)>1.5;if(use_padj)crit<-crit&r$p<.05; sel<-intersect(r$g[crit],b); N<-length(b);n<-length(sel);K<-lengths(sets);k<-vapply(sets,function(x)sum(sel%in%x),integer(1));pv<-phyper(k-1,K,N-K,n,lower.tail=FALSE);ad<-p.adjust(pv,"BH");z[[i]]<-data.frame(path=names(sets),k=k,p=pv,adj=ad)};for(cutname in c("p","adj"))for(cut in c(.05,.1,.2)){sig<-lapply(z,function(x)x$path[x[[cutname]]<cut]);cat("padjgenes",use_padj,cutname,cut,"per",paste(vapply(sig,length,integer(1)),collapse=","),"inter",length(Reduce(intersect,sig)),if(length(Reduce(intersect,sig)))paste(Reduce(intersect,sig),collapse=","),"'"\\n\")}}
RS"
