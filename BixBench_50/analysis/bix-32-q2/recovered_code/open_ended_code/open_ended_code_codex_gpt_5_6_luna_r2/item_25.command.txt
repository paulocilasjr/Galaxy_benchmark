/bin/bash -lc "Rscript - <<'RS'
files <- c(\"inputs/res_1vs97.rds\", \"inputs/res_1vs98.rds\", \"inputs/res_1vs99.rds\")
read_result <- function(f) {
  a <- attributes(readRDS(f))
  list(gene=a[[\"rownames\"]], lfc=a[[\"listData\"]][[\"log2FoldChange\"]])
}
res <- lapply(files, read_result)
links <- strsplit(readLines(\"https://rest.kegg.jp/link/pathway/pau\", warn=FALSE), \"\\t\", fixed=TRUE)
map <- data.frame(
  gene=sub(\""'^pau:", "", vapply(links, `[`, "", 1)),
  path=sub("''^path:", "", vapply(links, `[`, "", 2)),
  stringsAsFactors=FALSE
)
term2gene <- lapply(split(map$gene, map$path), unique)
universe <- unique(map$gene)
ora <- function(x, direction) {
  genes <- intersect(universe, x$gene[if (direction == "up") x$lfc > 1.5 else x$lfc < -1.5])
  sets <- lapply(term2gene, intersect, y=universe)
  keep <- lengths(sets) >= 10 & lengths(sets) <= 500
  sets <- sets[keep]
  N <- length(universe); n <- length(genes); K <- lengths(sets)
  k <- vapply(sets, function(s) sum(genes %in% s), integer(1))
  p <- phyper(k - 1, K, N - K, n, lower.tail=FALSE)
  names(p)[p.adjust(p, method="BH") < 0.05]
}
sig_up <- lapply(res, ora, direction="up")
sig_down <- lapply(res, ora, direction="down")
common_up <- Reduce(intersect, sig_up)
common_down <- Reduce(intersect, sig_down)
cat("up counts:", paste(lengths(sig_up), collapse=","), "common:", length(common_up), "'"\\n\")
cat(\"down counts:\", paste(lengths(sig_down), collapse=\",\"), \"common:\", length(common_down), \"\\n\")
cat(\"answer:\", length(unique(c(common_up, common_down))), \"\\n\")
stopifnot(length(unique(c(common_up, common_down))) == 0)
RS"
