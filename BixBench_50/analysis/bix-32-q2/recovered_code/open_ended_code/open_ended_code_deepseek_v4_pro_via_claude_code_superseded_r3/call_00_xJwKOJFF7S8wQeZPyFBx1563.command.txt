Rscript -e '
# Extract all three datasets as CSV
process_rds <- function(fname, outname) {
  x <- readRDS(fname)
  a <- attributes(x)
  ld <- a$listData
  df <- data.frame(
    gene = a$rownames,
    baseMean = ld$baseMean,
    log2FoldChange = ld$log2FoldChange,
    lfcSE = ld$lfcSE,
    stat = ld$stat,
    pvalue = ld$pvalue,
    padj = ld$padj,
    stringsAsFactors = FALSE
  )
  write.csv(df, outname, row.names = FALSE)
  cat(sprintf("%s: %d rows\n", outname, nrow(df)))
  
  # Summary stats
  cat(sprintf("  DE genes (|lfc|>1.5 & padj<0.05): %d up, %d down\n",
    sum(df$padj < 0.05 & df$log2FoldChange > 1.5, na.rm = TRUE),
    sum(df$padj < 0.05 & df$log2FoldChange < -1.5, na.rm = TRUE)))
}

dir.create("/tmp/analysis", showWarnings = FALSE)
process_rds("/workspace/inputs/res_1vs97.rds", "/tmp/analysis/res_97.csv")
process_rds("/workspace/inputs/res_1vs98.rds", "/tmp/analysis/res_98.csv")
process_rds("/workspace/inputs/res_1vs99.rds", "/tmp/analysis/res_99.csv")
'
