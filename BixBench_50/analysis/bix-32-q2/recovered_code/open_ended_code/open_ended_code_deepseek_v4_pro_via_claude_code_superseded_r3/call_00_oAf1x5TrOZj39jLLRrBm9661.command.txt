def run_ora_nominal(res_df, pathway_genes, lfc_threshold=1.5):
    all_genes = set(res_df["gene"])
    all_pw_genes = set()
    for genes in pathway_genes.values():
        all_pw_genes.update(genes)
    universe = all_genes & all_pw_genes
    
    de_up = set(res_df[res_df["log2FoldChange"] > lfc_threshold]["gene"]) & universe
    de_down = set(res_df[res_df["log2FoldChange"] < -lfc_threshold]["gene"]) & universe
    
    results = {'up': {}, 'down': {}}
    
    for pid, pw_genes in pathway_genes.items():
        pw_in_u = pw_genes & universe
        if len(pw_in_u) < 2:
            continue
        
        for direction, de_genes in [('up', de_up), ('down', de_down)]:
            overlap = pw_in_u & de_genes
            a = len(overlap)
            if a == 0:
                continue
            b = len(de_genes) - a
            c = len(pw_in_u) - a
            d = len(universe) - len(de_genes) - c
            
            table = [[a, b], [c, d]]
            _, pval = fisher_exact(table, alternative='greater')
            results[direction][pid] = pval
    
    return results

ora_97 = run_ora_nominal(res_97, pathway_genes_raw)
ora_98 = run_ora_nominal(res_98, pathway_genes_raw)
ora_99 = run_ora_nominal(res_99, pathway_genes_raw)

# Find pathways with nominal p<0.05 in same direction across all three
