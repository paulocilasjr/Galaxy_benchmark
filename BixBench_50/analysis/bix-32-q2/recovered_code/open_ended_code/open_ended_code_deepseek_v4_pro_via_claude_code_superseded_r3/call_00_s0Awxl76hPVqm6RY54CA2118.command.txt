import pandas as pd
import numpy as np
from collections import defaultdict
from scipy.stats import hypergeom
from statsmodels.stats.multitest import multipletests
import pickle

# Load data
res_97 = pd.read_csv("/tmp/analysis/res_97.csv")
res_98 = pd.read_csv("/tmp/analysis/res_98.csv")
res_99 = pd.read_csv("/tmp/analysis/res_99.csv")

with open("/tmp/analysis/kegg_data.pkl", "rb") as f:
    kegg_data = pickle.load(f)
pathway_genes_raw = {k: set(v) for k, v in kegg_data["pathway_genes"].items()}
pathways = kegg_data["pathways"]

all_genes = set(res_97["gene"])
all_kegg_genes = set()
for gs in pathway_genes_raw.values():
    all_kegg_genes.update(gs)
universe = all_genes & all_kegg_genes
N = len(universe)

# Run ORA with BH correction ACROSS both directions combined (more conservative)
def run_ora_combined(res_df, pathway_genes, lfc_threshold=1.5):
    de_up = set(res_df[res_df["log2FoldChange"] > lfc_threshold]["gene"]) & universe
    de_down = set(res_df[res_df["log2FoldChange"] < -lfc_threshold]["gene"]) & universe
    
    all_results = []
    for direction, de_set in [('up', de_up), ('down', de_down)]:
        K = len(de_set)
        for pid, pw_genes in pathway_genes.items():
            pw_in_u = pw_genes & universe
            n = len(pw_in_u)
            if n < 2: continue
            x = len(pw_in_u & de_set)
            if x == 0: continue
            pval = hypergeom.sf(x - 1, N, K, n)
            fe = (x / n) / (K / N)
            all_results.append((pid, direction, pval, fe, x, n, K))
    
    if all_results:
        pvals = [r[2] for r in all_results]
        _, padjs, _, _ = multipletests(pvals, method='fdr_bh')
        for i, padj in enumerate(padjs):
            all_results[i] = all_results[i] + (padj,)
    
    return all_results

# Run for each strain
for label, res_df in [('97', res_97), ('98', res_98), ('99', res_99)]:
    results = run_ora_combined(res_df, pathway_genes_raw)
    sig = {}
    for r in results:
        pid, direction, pval, fe, x, n, K, padj = r
        if padj < 0.05:
            if direction not in sig:
                sig[direction] = set()
            sig[direction].add(pid)
    
    print(f"Strain {label}:")
    for d in ['up', 'down']:
        if d in sig:
            print(f"  Sig {d} (combined BH): {len(sig[d])} pathways")
            for pid in sorted(sig[d]):
                r = [x for x in results if x[0]==pid and x[1]==d][0]
                print(f"    {pid}: padj={r[7]:.4f}, fe={r[3]:.2f}, overlap={r[4]}")
        else:
            print(f"  Sig {d} (combined BH): 0 pathways")
    
    print()

# Now find consistent pathways (padj < 0.05 across all three, same direction)
