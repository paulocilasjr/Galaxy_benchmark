import pyreadr
import pandas as pd

# Read the RDS files
for fname in ["res_1vs97.rds", "res_1vs98.rds", "res_1vs99.rds"]:
    result = pyreadr.read_r(f"inputs/{fname}")
    print(f"\n=== {fname} ===")
    print(f"Keys: {list(result.keys())}")
    for key, df in result.items():
        if df is not None:
            print(f"  {key}: shape={df.shape}, columns={list(df.columns)}")
            print(f"  First 3 rows:\n{df.head(3)}")
        else:
            print(f"  {key}: None")
