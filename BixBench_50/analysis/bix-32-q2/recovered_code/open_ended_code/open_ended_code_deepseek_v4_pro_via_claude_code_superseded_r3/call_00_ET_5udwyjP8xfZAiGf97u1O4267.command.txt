import requests

# Check KEGG organism codes for Pseudomonas aeruginosa
# Try known codes
org_codes = ["pae", "pau", "paer", "pap", "paf", "pag", "pah", "pai", "paj", "pal", "pam", "pan", "pao", "paq", "par", "pas", "pat", "pau", "pav", "paw", "pax", "pay", "paz"]

for code in org_codes[:5]:
    url = f"https://rest.kegg.jp/list/{code}"
    try:
        r = requests.get(url, timeout=10)
        if r.status_code == 200 and len(r.text) > 100:
            lines = r.text.strip().split("\n")
            print(f"=== {code}: {lines[0][:120]} ===")
            # Check first gene to see if it matches PA14 format
            for line in lines[:5]:
                print(f"  {line[:120]}")
        else:
            print(f"{code}: not found or empty")
    except Exception as e:
        print(f"{code}: error - {e}")
