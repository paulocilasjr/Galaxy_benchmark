    pvals=np.asarray(pvals)
    order=np.argsort(pvals)
    sorted_p=pvals[order]
    n=len(pvals)
    adj=np.empty(n)
    adj[order]=np.minimum.accumulate(sorted_p*n/np.arange(1,n+1))

    pvals=np.asarray(pvals, dtype=float)
    order=np.argsort(pvals)
    sorted_p=pvals[order]
    n=len(pvals)
    scaled=sorted_p*n/np.arange(1,n+1)
    # BH adjusted p-value is reverse cumulative minimum of scaled p-values
    adj_sorted=np.minimum.accumulate(scaled[::-1])[::-1]
    adj=np.empty(n)
    adj[order]=adj_sorted
    return np.minimum(1, adj)
