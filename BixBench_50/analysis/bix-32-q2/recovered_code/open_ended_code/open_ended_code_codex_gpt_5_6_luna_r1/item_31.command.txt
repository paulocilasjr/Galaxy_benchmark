mp<-strsplit(readLines(\"https://rest.kegg.jp/link/pathway/pau\",warn=FALSE),\"\\\\t\"); map<-data.frame(gene=sub(\""'^pau:","",vapply(mp,`[[`,"",1)),path=sub("''^path:","",vapply(mp,`[[`,"",2))); sets<-lapply(split(map$gene,map$path),unique); ids_by<-list(); lfc_by<-list(); for(f in list.files("inputs",pattern="'"\\\\.rds"'$",full.names=TRUE)){x<-readRDS(f);a<-attributes(x);ids_by[[basename(f)]]<-a$rownames;lfc_by[[basename(f)]]<-a$listData$log2FoldChange}
ora<-function(g,u){g<-intersect(unique(g),unique(u));u<-intersect(unique(u),map$gene);ts<-names(sets)[vapply(sets,function(s){K<-sum(s%in%u);K>=10&&K<=500},logical(1))];p<-vapply(ts,function(t){s<-sets[[t]];K<-sum(s%in%u);k<-sum(s%in%g);phyper(k-1,K,length(u)-K,length(g),lower.tail=FALSE)},numeric(1));data.frame(path=ts,k=vapply(ts,function(t)sum(sets[[t]]%in%g),integer(1)),K=vapply(ts,function(t)sum(sets[[t]]%in%u),integer(1)),p=p,padj=p.adjust(p,"BH"))}
for(nm in names(ids_by)){ids<-ids_by[[nm]];l<-lfc_by[[nm]];cat("###",nm,"###'"\\n\");for(dir in c(\"UP\",\"DOWN\")){g<-if(dir==\"UP\")ids[l>1.5] else ids[l< -1.5];e<-ora(g,map"'$gene); e<-e[order(e$p),]; cat(dir,"n",sum(g%in%map$gene),"'"\\n\"); print(e[e"'$p<.2,],row.names=FALSE)} }
