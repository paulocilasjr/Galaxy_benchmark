/bin/bash -lc "Rscript - <<'RS'
read_res <- function(f) { x <- readRDS(f); data.frame(gene=slot(x,'rownames'), lfc=slot(x,'listData')[[\"log2FoldChange\"]], stringsAsFactors=FALSE) }
res <- lapply(c('97'='inputs/res_1vs97.rds','98'='inputs/res_1vs98.rds','99'='inputs/res_1vs99.rds'), read_res)
ln <- readLines('https://rest.kegg.jp/link/pathway/pau', warn=FALSE)
p <- strsplit(ln,'\\t'); map <- data.frame(gene=sub('"'^pau:'"','',vapply(p,"'`[`,'"'',1)), path=sub('"'^path:'"','',vapply(p,"'`[`,'"'',2)))
ora <- function(fg, universe) {
  universe <- intersect(unique(universe), unique(map"'$gene)); fg <- intersect(unique(fg), universe)
  t2g <- split(map$gene[map$gene %in% universe], map$path[map$gene %in% universe]); t2g <- lapply(t2g, unique)
  sz <- vapply(t2g, length, 1L); t2g <- t2g[sz >= 10 & sz <= 500]
  out <- do.call(rbind, lapply(names(t2g), function(path) { genes <- t2g[[path]]; k <- length(intersect(fg, genes)); M <- length(genes); N <- length(universe); n <- length(fg); data.frame(path=path,k=k,pvalue=phyper(k-1,M,N-M,n,lower.tail=FALSE)) }))
  out <- out[out$k>0,]; out$padj <- p.adjust(out$pvalue,'"'BH'); out[out"'$padj<0.05,]
}
up_common <- Reduce(intersect, lapply(res, function(d) d$gene[d$lfc > 1.5]))
down_common <- Reduce(intersect, lapply(res, function(d) d$gene[d$lfc < -1.5]))
cat('"'common up/down genes', length(up_common), length(down_common), 'mapped', length(intersect(up_common,map"'$gene)), length(intersect(down_common,map$gene)), '"'\\n')
cat('common-gene ORA sig up/down', nrow(ora(up_common, res[[1]]"'$gene)), nrow(ora(down_common, res[[1]]$gene)), '"'\\n')
cat('paths up', paste(ora(up_common, res[[1]]"'$gene)$path, collapse='"','), '\\n')
cat('paths down', paste(ora(down_common, res[[1]]"'$gene)$path, collapse='"','), '\\n')
RS"
