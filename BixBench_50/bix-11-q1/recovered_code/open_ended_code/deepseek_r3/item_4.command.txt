/bin/bash -c "python3 - <<'PY'
import zipfile, collections, os
for zname in ['scogs_animals.zip','scogs_fungi.zip']:
    p=f'/workspace/inputs/{zname}'
    with zipfile.ZipFile(p) as z:
        names=z.namelist()
    ext=collections.Counter()
    for n in names:
        # suffix after last dot
        base=n.rsplit('.',1)[-1] if '.' in n else ''
        ext[base]+=1
    print('\\n',zname,'files',len(names))
    for k,v in sorted(ext.items()):
        print(k,v)
    # list tree-like files
    for pat in ['.treefile','.bionj','.mldist','.iqtree','.ckp.gz','.model.gz']:
        print(pat, sum(n.endswith(pat) for n in names))
    print('sample treefile', [n for n in names if n.endswith('.treefile')][:5])
    print('sample bionj', [n for n in names if n.endswith('.bionj')][:5])
    print('sample mldist', [n for n in names if n.endswith('.mldist')][:5])
PY"
