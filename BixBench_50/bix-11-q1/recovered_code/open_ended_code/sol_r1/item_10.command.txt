/bin/bash -lc "unzip -tq inputs/scogs_animals.zip
unzip -tq inputs/scogs_fungi.zip
python - <<'PY'
from zipfile import ZipFile
from io import StringIO
from statistics import median
from Bio import Phylo
for group in ('animals','fungi'):
    rows=[]
    with ZipFile(f'inputs/scogs_{group}.zip') as z:
        names=[n for n in z.namelist() if n.endswith('.treefile')]
        assert len(names)==len(set(names))
        for n in names:
            t=Phylo.read(StringIO(z.read(n).decode()),'newick')
            clades=list(t.find_clades())
            assert all(c.branch_length is not None for c in clades if c is not t.root)
            total=sum(c.branch_length or 0 for c in clades)
            internal=sum(c.branch_length or 0 for c in t.get_nonterminals())
            rows.append((internal/total,n,total,internal))
    rows.sort()
    m=median(x[0] for x in rows)
    center=rows[(len(rows)-1)//2] if len(rows)%2 else rows[len(rows)//2-1:len(rows)//2+1]
    print(group, 'median=',format(m,'.17g'), 'center=',center)
PY"
