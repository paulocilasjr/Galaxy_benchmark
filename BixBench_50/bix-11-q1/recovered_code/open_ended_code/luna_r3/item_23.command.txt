for archive in (\"inputs/scogs_animals.zip\", \"inputs/scogs_fungi.zip\"):

                continue

            internal = float(mi.group(1))
            pct = float(mi.group(2))
            vals.append(pct/100)
            rows.append((n, total, internal, pct/100, internal/total))
        print(\"\\n\", archive)
        print(\"reports\", len(reports), \"parsed\", len(vals))
        print(\"median_report_decimal\", statistics.median(vals))
