/bin/bash -lc "python - <<'PY'
import zipfile, collections, re

def fasta_n(text):
    return sum(1 for line in text.splitlines() if line.startswith('>'))
for path in ('inputs/scogs_animals.zip','inputs/scogs_fungi.zip'):
    print('\\n',path)
    c=collections.Counter(); examples={}
    with zipfile.ZipFile(path) as z:
        names=set(z.namelist())
        for name in sorted(n for n in names if n.endswith('.faa')):
            n=fasta_n(z.read(name).decode())
            c[n]+=1; examples.setdefault(n,(name,z.read(name).decode()[:180]))
    print('original fasta counts',dict(sorted(c.items())))
    for n in sorted(examples): print('n=',n,'example=',examples[n][0],repr(examples[n][1]))
PY"
