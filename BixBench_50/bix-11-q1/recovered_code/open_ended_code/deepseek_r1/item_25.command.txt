/bin/bash -c "python - <<'PY'
from phykit.services.tree.base import Tree
import glob, statistics
animals=[Tree(tree_file_path=f).calculate_treeness() for f in glob.glob('tmp/scogs_animals/*.treefile')]
fungi=[Tree(tree_file_path=f).calculate_treeness() for f in glob.glob('tmp/scogs_fungi/*.treefile')]
ma=statistics.median(animals); mf=statistics.median(fungi)
print(f'ma={ma"'!r} mf={mf!r}'"')
print(f'fungi-animals={mf-ma"'!r}'"')
print(f'animals-fungi={ma-mf"'!r}'"')
print(f'abs={abs(mf-ma)"'!r}'"')
print(f'round4 f-a={round(mf-ma,4)} abs={round(abs(mf-ma),4)}')
print(f'round6={round(mf-ma,6)} round8={round(mf-ma,8)} round10={round(mf-ma,10)}')
print('n',len(animals),len(fungi))
PY"
