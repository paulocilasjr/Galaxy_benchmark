/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from io import StringIO
try:
 from Bio import Phylo
 print('biopython', __import__('Bio').__version__)
except Exception as e: print('no biopython',e)
for zpath in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
 with ZipFile(zpath) as z:
  ts=[n for n in z.namelist() if n.endswith('.treefile')]
  bad=[]; treeness=[]
  for n in ts:
   s=z.read(n).decode()
   t=Phylo.read(StringIO(s),'newick')
   edges=[c.branch_length for c in t.find_clades(order='level') if c is not t.root]
   terminals=[c.branch_length for c in t.get_terminals()]
   internal=[c.branch_length for c in t.get_nonterminals() if c is not t.root]
   if any(x is None for x in edges) or not edges or sum(edges)<=0:
    bad.append((n,s.strip(),edges))
   else:
    treeness.append((sum(internal)/sum(edges), n, sum(internal),sum(edges),len(terminals),len(internal)))
 print(zpath,'trees',len(ts),'valid',len(treeness),'bad',len(bad),'tip/internal edges',sorted(set((x[4],x[5]) for x in treeness)))
 import statistics
 vals=[x[0] for x in treeness]
 print('median',statistics.median(vals),'range',min(vals),max(vals))
 print('middle',sorted(treeness)[len(vals)//2-2:len(vals)//2+2])
PY"
