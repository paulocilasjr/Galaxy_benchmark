/bin/bash -lc "python3 - <<'PY'
import zipfile, re
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    print('---',zname)
    with zipfile.ZipFile(zname) as z:
        for n in z.namelist():
            if n.endswith('.iqtree'):
                txt=z.read(n).decode(errors='replace')
                for pat in ['Treeness','treeness','Tree length','Sum of branch lengths']:
                    if pat in txt:
                        print(n, pat, 'found')
                        m=re.search(r'.{0,80}'+re.escape(pat)+r'.{0,160}', txt, re.I|re.S)
                        print(m.group(0).replace('\\n',' ')[:260])
                        raise SystemExit
                print('no treeness in first iqtree', n)
                print('\\n'.join(txt.splitlines()[:30]))
                break
PY"
