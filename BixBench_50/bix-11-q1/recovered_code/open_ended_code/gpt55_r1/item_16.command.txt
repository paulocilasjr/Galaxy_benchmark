/bin/bash -lc "python3 - <<'PY'
import zipfile, re
for zname in ['inputs/scogs_animals.zip','inputs/scogs_fungi.zip']:
    pats={}
    with zipfile.ZipFile(zname) as z:
        for n in z.namelist():
            if n.endswith('.iqtree'):
                txt=z.read(n).decode(errors='ignore')
                for line in txt.splitlines():
                    low=line.lower()
                    if 'tree length' in low or 'treeness' in low or 'total tree' in low:
                        pats[line]=pats.get(line,0)+1
    print(zname, 'matched lines', len(pats))
    for line,count in list(pats.items())[:20]:
        print(count, line)
PY"
