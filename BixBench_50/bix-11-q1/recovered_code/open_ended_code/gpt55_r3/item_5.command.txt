/bin/bash -lc "python - <<'PY'
import zipfile, os
for fn in sorted(os.listdir('/workspace/inputs')):
    if fn.endswith('.zip'):
        p='/workspace/inputs/'+fn
        try:
            with zipfile.ZipFile(p) as z:
                names=z.namelist()[:25]
                print('\\n###',fn, 'n=', len(z.namelist()))
                for n in names: print(n)
        except Exception as e: print(fn,e)
PY"
