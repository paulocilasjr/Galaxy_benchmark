/bin/bash -lc "python3 - <<'PY'
from zipfile import ZipFile
from statistics import median
import re

for label, path in [('animals','inputs/scogs_animals.zip'),('fungi','inputs/scogs_fungi.zip')]:
    vals=[]
    totals=[]
    missing=[]
    with ZipFile(path) as z:
        names=sorted(n for n in z.namelist() if n.endswith('.iqtree'))
        for n in names:
            txt=z.read(n).decode('utf-8', errors='replace')
            mt=re.search(r'Total tree length \\(sum of branch lengths\\):\\s*([0-9.eE+-]+)', txt)
            mi=re.search(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)\\s*\\(([0-9.eE+-]+)% of tree length\\)', txt)
            if not (mt and mi):
                missing.append(n)
                continue
            total=float(mt.group(1)); internal=float(mi.group(1))
            if total <= 0:
                missing.append(n)
                continue
            vals.append(internal/total)
            totals.append((n,total,internal,internal/total,float(mi.group(2))/100))
    print(label, 'iqtree', len(names), 'parsed', len(vals), 'missing', len(missing))
    print('median', median(vals), 'min', min(vals), 'max', max(vals))
    print('median_pct_reported', median([x[4] for x in totals]))
    print('first', totals[0])
print('difference fungi-animals', median([float(re.search(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)', ZipFile('inputs/scogs_fungi.zip').read(n).decode()).group(1))/float(re.search(r'Total tree length \\(sum of branch lengths\\):\\s*([0-9.eE+-]+)', ZipFile('inputs/scogs_fungi.zip').read(n).decode()).group(1)) for n in sorted(ZipFile('inputs/scogs_fungi.zip').namelist()) if n.endswith('.iqtree')]) - median([float(re.search(r'Sum of internal branch lengths:\\s*([0-9.eE+-]+)', ZipFile('inputs/scogs_animals.zip').read(n).decode()).group(1))/float(re.search(r'Total tree length \\(sum of branch lengths\\):\\s*([0-9.eE+-]+)', ZipFile('inputs/scogs_animals.zip').read(n).decode()).group(1)) for n in sorted(ZipFile('inputs/scogs_animals.zip').namelist()) if n.endswith('.iqtree')]))
PY"
