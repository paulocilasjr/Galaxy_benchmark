/bin/bash -lc "python3 - <<'PY'
from zipfile import ZipFile
from statistics import median
import re

num = r'[+-]?(?:\\d+(?:\\.\\d*)?|\\.\\d+)(?:[eE][+-]?\\d+)?'
branch_re = re.compile(r':(' + num + r')')
internal_re = re.compile(r'\\)["'^:(),;]*:('"' + num + r')')
summary_total_re = re.compile(r'Total tree length \\(sum of branch lengths\\):\\s*(' + num + r')')
summary_internal_re = re.compile(r'Sum of internal branch lengths:\\s*(' + num + r')')

def tree_vals(path):
    vals=[]
    with ZipFile(path) as z:
        for n in sorted(x for x in z.namelist() if x.endswith('.treefile')):
            nw=z.read(n).decode().strip()
            total=sum(map(float, branch_re.findall(nw)))
            internal=sum(map(float, internal_re.findall(nw)))
            vals.append(internal / total)
    return vals

def summary_vals(path):
    vals=[]
    with ZipFile(path) as z:
        for n in sorted(x for x in z.namelist() if x.endswith('.iqtree')):
            txt=z.read(n).decode(errors='replace')
            total=float(summary_total_re.search(txt).group(1))
            internal=float(summary_internal_re.search(txt).group(1))
            vals.append(internal / total)
    return vals

animals_tree=tree_vals('inputs/scogs_animals.zip')
fungi_tree=tree_vals('inputs/scogs_fungi.zip')
animals_summary=summary_vals('inputs/scogs_animals.zip')
fungi_summary=summary_vals('inputs/scogs_fungi.zip')
print('tree medians:', median(fungi_tree), median(animals_tree), median(fungi_tree)-median(animals_tree))
print('summary medians:', median(fungi_summary), median(animals_summary), median(fungi_summary)-median(animals_summary))
print('counts:', len(fungi_tree), len(animals_tree))
PY"
