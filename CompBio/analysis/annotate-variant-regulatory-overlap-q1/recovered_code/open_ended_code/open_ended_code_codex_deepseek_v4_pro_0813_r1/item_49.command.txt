/bin/bash -lc "python - <<'PY'
import pyBigWig, collections
bw=pyBigWig.open('workspace_ccre/encodeCcreRegistry.bb')
counts=collections.Counter()
# iterate all entries from all chroms maybe slow; but can sample unique classes by iterating chr
for chrom in bw.chroms():
    size=bw.chroms(chrom)
    # skip alt? 
    try:
        for e in bw.entries(chrom,0,size):
            rest=e[2]
            cls=rest.split('\\t')[6] if '\\t' in rest else rest
            counts[cls]+=1
    except Exception as ex:
        pass
    if len(counts)>=10:
        break
print(counts)
PY"
