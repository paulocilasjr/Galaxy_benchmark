/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
from scipy.cluster.hierarchy import linkage, dendrogram
from scipy.spatial.distance import pdist, squareform
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
X=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
log=np.log2(X+1)
pca=PCA(n_components=2).fit(log.T)
print(pca.explained_variance_ratio_)
print(log.T.index.tolist())
print('pc',pca.transform(log.T))
# correlations
print('corr')
print(log.corr().round(4))
# linkage
Z=linkage(pdist(log.T), method='ward')
print(Z)
plt.figure(figsize=(10,8))
dn=dendrogram(Z, labels=log.columns, leaf_rotation=90)
plt.savefig('dendro.png', bbox_inches='tight')
# top genes PCA loadings
load=pd.DataFrame(pca.components_.T,index=log.index,columns=['PC1','PC2'])
print('top pc1')
print(load.reindex(load.PC1.abs().sort_values(ascending=False).index).head(30))
print('top pc2')
print(load.reindex(load.PC2.abs().sort_values(ascending=False).index).head(30))
PY"
