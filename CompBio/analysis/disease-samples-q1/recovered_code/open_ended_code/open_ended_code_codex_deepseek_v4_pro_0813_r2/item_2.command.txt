/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
p='disease.samples.q1.tsv'
df=pd.read_csv(p,sep='\\t',index_col=0)
X=np.log2(df.values+1)
# center by gene
Xc=X-X.mean(axis=1,keepdims=True)
U,S,Vt=np.linalg.svd(Xc, full_matrices=False)
# scores = U*S, loadings = V
print('S',S[:10])
print('V rows samples first 3')
for i,c in enumerate(df.columns): print(c, Vt[0,i], Vt[1,i], Vt[2,i])
genes=df.index
# PC1 loadings are Vt[0] sample coefficients; gene projections U*S? Actually Xc = U S Vt. Gene loading for PC1 = U[:,0]*S[0], high positive associated with samples positive.
ld=U[:,0]*S[0]
idx=np.argsort(ld)[::-1]
print('top positive genes')
for i in idx[:50]: print(genes[i], df.iloc[i].values, ld[i])
print('top negative genes')
for i in idx[-50:]: print(genes[i], df.iloc[i].values, ld[i])
PY"
