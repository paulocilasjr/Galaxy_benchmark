/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.cluster.hierarchy import linkage, fcluster
from scipy.spatial.distance import pdist
p='disease.samples.q1.tsv'
df=pd.read_csv(p,sep='\\t',index_col=0)
X=np.log2(df.values+1)
Z=linkage(pdist(X.T), method='average')
print(pd.DataFrame({'sample':df.columns,'cluster':fcluster(Z,2,criterion='maxclust')}))
# markers
markers=['S100A7','S100A8','S100A9','SERPINB4','SERPINB3','DEFB4A','PI3','SPRR2A','KRT16','KRT6A','IL36G','CXCL8','LCN2','FLG2','DCD','ADH1B','CLEC3B']
print(df.loc[df.index.isin(markers)])
PY"
