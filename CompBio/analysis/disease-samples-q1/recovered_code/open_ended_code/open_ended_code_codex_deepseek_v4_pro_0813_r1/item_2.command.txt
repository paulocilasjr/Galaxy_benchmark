/bin/bash -lc "python3 - <<'PY'
import numpy as np, pandas as pd
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
p='disease.samples.q1.tsv'
df=pd.read_csv(p,sep='\\t',index_col=0)
X=df.values.T
print('samples', df.columns.tolist())
print('log10 total', np.log10(X.sum(axis=1)))
# PCA on log1p standardized
Y=np.log1p(X)
# Use top variable genes maybe
var=Y.var(axis=0)
for k in [100,500,1000,3000,10000]:
    idx=np.argsort(var)[::-1][:k]
    Z=StandardScaler().fit_transform(Y[:,idx])
    pc=PCA(n_components=2).fit_transform(Z)
    print('\\nk',k,'var explained', PCA(n_components=2).fit(Z).explained_variance_ratio_)
    for s,(a,b) in zip(df.columns,pc):
        print(f'{s}: {a:.4f} {b:.4f}')
PY"
