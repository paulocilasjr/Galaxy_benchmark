/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('disease.samples.q1.tsv', sep='\\t', index_col=0)
print(x.shape, flush=True)
log=np.log1p(x)
print('corr', flush=True)
print(log.corr().round(3), flush=True)
from sklearn.decomposition import PCA
p=PCA(n_components=5).fit(log.T)
print('var',p.explained_variance_ratio_, flush=True)
print(pd.DataFrame(p.transform(log.T),index=x.columns).round(2), flush=True)
PY"
