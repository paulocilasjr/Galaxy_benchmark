/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
markers=['S100A7','S100A7A','S100A8','S100A9','S100A12','SERPINB3','SERPINB4','SPRR2A','SPRR2B','SPRR2D','SPRR2F','DEFB4A','DEFB4B','IL36A','IL36G','CXCL1','CXCL8','NOS2','LCN2','KRT6A','KRT6B','KRT16','IFI27','OAS2','MX1','CD274','PI3','TCN1','TMPRSS11D']
print(x.reindex(markers).round(2).to_string())
# standardized score of markers each sample, only genes present
present=[m for m in markers if m in x.index]
z=(np.log1p(x.loc[present])-np.log1p(x.loc[present]).mean(axis=1).values[:,None])/np.log1p(x.loc[present]).std(axis=1).replace(0,np.nan).values[:,None]
print('\\nmarker z-score mean (within gene)')
print(z.mean(axis=0).round(2).sort_values(ascending=False).to_string())
# robust disease score using rank / log values
print('\\nper sample marker median log1p')
print(np.log1p(x.loc[present]).median(axis=0).round(2).sort_values(ascending=False).to_string())
# genes with high A relative B and sample consistency: minimum A vs max B perhaps
la=np.log1p(x)
for s in x.columns:
    pass
print('\\nA signature score by sample (mean z among top A markers)')
# manually select induced genes with A-B > 2 and mean A > 2
b=['Sample_2','Sample_3','Sample_5','Sample_7'];a=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
diff=la[a].mean(axis=1)-la[b].mean(axis=1)
sel=diff[(diff>2)&(la[a].mean(axis=1)>2)].index
zall=(la.loc[sel]-la.loc[sel].mean(axis=1).values[:,None])/la.loc[sel].std(axis=1).replace(0,np.nan).values[:,None]
print('selected genes',len(sel))
print(zall.mean(axis=0).round(2).to_string())
PY"
