/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import ttest_ind, mannwhitneyu
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
a=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
b=['Sample_2','Sample_3','Sample_5','Sample_7']
# log1p statistics; filter genes with enough expression and fold changes
la=np.log1p(x[a]); lb=np.log1p(x[b])
res=pd.DataFrame(index=x.index)
res['meanA']=la.mean(axis=1); res['meanB']=lb.mean(axis=1)
res['diff']=res.meanA-res.meanB
res['medA']=la.median(axis=1);res['medB']=lb.median(axis=1)
res['fracA']= (x[a]>0).mean(axis=1);res['fracB']=(x[b]>0).mean(axis=1)
res['p']= [ttest_ind(la.loc[g,a],lb.loc[g,b],equal_var=False).pvalue for g in x.index]
res['q']=res.p.rank(method='first')/len(res)
print('TOP A by diff (require mean TPM > 1 in either)')
print(res.loc[(np.expm1(res.meanA)>1)|(np.expm1(res.meanB)>1)].sort_values('diff',ascending=False).head(80)[['meanA','meanB','diff','medA','medB','fracA','fracB']].to_string())
print('\\nTOP B by diff')
print(res.loc[(np.expm1(res.meanA)>1)|(np.expm1(res.meanB)>1)].sort_values('diff').head(80)[['meanA','meanB','diff','medA','medB','fracA','fracB']].to_string())
print('\\nTop t p')
print(res.sort_values('p').head(50)[['meanA','meanB','diff','p']].to_string())
PY"
