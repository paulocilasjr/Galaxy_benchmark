/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import spearmanr
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, leaves_list, dendrogram
from sklearn.decomposition import PCA
x=pd.read_csv('disease.samples.q1.tsv', sep='\\t', index_col=0)
print(x.shape)
print('columns', list(x.columns))
print('nonzero per sample')
print((x>0).sum().to_string())
print('sum TPM', x.sum().to_string())
print('median', x.median().to_string())
print('mean', x.mean().to_string())
log=np.log1p(x)
print('\\nPearson log corr')
print(log.corr().round(3).to_string())
print('\\nSpearman corr')
print(x.corr(method='spearman').round(3).to_string())
# PCA centered log expression, genes rows samples cols
p=PCA().fit(log.T)
print('\\nPCA var', p.explained_variance_ratio_[:8])
print(pd.DataFrame(p.transform(log.T)[:,:5],index=x.columns,columns=['PC1','PC2','PC3','PC4','PC5']).round(2).to_string())
# hierarchical euclidean standardized? sample distances log raw; corr distance
for name, data in [('log',log),('z', (log-log.mean(axis=1).values[:,None])/log.std(axis=1).replace(0,np.nan).values[:,None])]:
    d=pdist(data.T,metric='euclidean')
    print('\\n',name,'euclidean dist')
    print(pd.DataFrame(squareform(d), index=x.columns, columns=x.columns).round(1).to_string())
    z=linkage(d,method='average')
    print('leaves',x.columns[leaves_list(z)].tolist())
PY"
