/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import ttest_ind, mannwhitneyu
D=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
A=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']
B=['Sample_2','Sample_3','Sample_5','Sample_7']
# log means and fold changes
la=np.log2(D[A]+0.1).mean(axis=1); lb=np.log2(D[B]+0.1).mean(axis=1)
res=pd.DataFrame({'gene':D.index,'A_mean':D[A].mean(axis=1),'B_mean':D[B].mean(axis=1),'log2FC':la-lb})
# t test log
res['p']=ttest_ind(np.log1p(D[A]),np.log1p(D[B]),axis=1,equal_var=False).pvalue
res['q']=res.p.rank(method='average') # placeholder
print('A vs B top A')
for _,r in res.sort_values('log2FC',ascending=False).head(100).iterrows(): print(f\"{r.gene}\\tA={r.A_mean:.2f}\\tB={r.B_mean:.2f}\\tlog2FC={r.log2FC:.2f}\\tp={r.p:.2g}\")
print('\\nB top')
for _,r in res.sort_values('log2FC').head(100).iterrows(): print(f\"{r.gene}\\tA={r.A_mean:.2f}\\tB={r.B_mean:.2f}\\tlog2FC={r.log2FC:.2f}\\tp={r.p:.2g}\")
PY"
