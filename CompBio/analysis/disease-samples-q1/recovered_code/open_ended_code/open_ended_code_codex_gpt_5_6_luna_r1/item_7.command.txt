/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.cluster.hierarchy import linkage, leaves_list, dendrogram
from scipy.spatial.distance import squareform
from sklearn.decomposition import PCA
p='disease.samples.q1.tsv'
df=pd.read_csv(p,sep='\\t',index_col=0)
s=list(df.columns)
x=np.log1p(df.to_numpy())
# center genes, sample Pearson on log TPM
cor=np.corrcoef(x.T)
print('samples',s)
print('correlation')
print(pd.DataFrame(cor,index=s,columns=s).round(5).to_string())
# Euclidean distances after gene centering and scale by gene sd? common PCA
xc=x-x.mean(axis=1,keepdims=True)
# PCA on genes x samples centered; transpose
pca=PCA().fit(xc.T)
print('PCA variance',pca.explained_variance_ratio_.round(5).tolist())
print('PCA coords')
print(pd.DataFrame(pca.transform(xc.T)[:,:5],index=s,columns=['PC1','PC2','PC3','PC4','PC5']).round(3).to_string())
# top gene loadings PCs
for k in range(3):
    load=pca.components_[k]
    ix=np.argsort(np.abs(load))[::-1][:15]
    print('PC',k+1,'top abs',[(df.index[i],round(load[i],4)) for i in ix])
# correlations/distance linkage
D=1-cor
for method in ['average','complete','single']:
    z=linkage(squareform(D,checks=False),method=method)
    print(method,'order',[s[i] for i in leaves_list(z)])
# top variable genes on log scale
v=x.var(axis=1)
ix=np.argsort(v)[::-1][:50]
print('top variable genes')
for i in ix[:50]:
    print(df.index[i], 'var',round(v[i],3), 'vals', ['%.2f'%a for a in df.iloc[i].to_numpy()])
PY"
