/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from scipy.cluster.hierarchy import linkage, leaves_list
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
print(x.shape, x.sum().round(1).to_dict())
y=np.log2(x+1)
# top variable genes
v=y.var(1).sort_values(ascending=False)
print('top variable')
print(pd.DataFrame({'var':v.head(60),'vals':[','.join(f'{z:.1f}' for z in y.loc[g]) for g in v.head(60).index]}).to_string())
# sample corr and pca
sel=y.loc[v.head(3000).index]
print('\\ncorr')
print(sel.corr().round(2).to_string())
Z=linkage(sel.T,method='ward')
print('leaves', list(sel.columns[leaves_list(Z)]))
p=PCA(5).fit_transform(sel.T)
print('PCA',pd.DataFrame(p,index=x.columns).round(2).to_string())
PY"
