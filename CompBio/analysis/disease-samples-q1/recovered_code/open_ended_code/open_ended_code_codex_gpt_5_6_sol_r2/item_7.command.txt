/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import ttest_ind
from sklearn.cluster import KMeans
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
y=np.log2(x+1)
a=['Sample_1','Sample_4','Sample_6','Sample_8','Sample_9']; b=['Sample_2','Sample_3','Sample_5','Sample_7']
d=y[a].mean(axis=1)-y[b].mean(axis=1)
print('TOP HIGH IN 5-SAMPLE CLUSTER')
print(pd.DataFrame({'log2diff':d.nlargest(50)}).round(2).to_string())
print('\\nTOP HIGH IN 4-SAMPLE CLUSTER')
print(pd.DataFrame({'log2diff':d.nsmallest(40)}).round(2).to_string())
print('\\nMarker-score checks')
panels={
'inflammation':['S100A7','S100A7A','S100A8','S100A9','DEFB4A','PI3','SERPINB3','SERPINB4','IL36A','IL36G','CXCL1','CXCL8','LCN2','NOS2','KRT6A','KRT16'],
'normal_diff':['FLG','LOR','CASP14','FLG2','KRT2']}
for name,g in panels.items():
 g=[q for q in g if q in y.index]
 score=y.loc[g].apply(lambda row:(row-row.mean())/row.std(),axis=1).mean()
 print(name,score.round(2).to_dict())
# stability over top variable selection sizes
for n in [100,250,500,1000,2000,5000,10000]:
 z=y.loc[y.var(axis=1).nlargest(n).index].T
 lab=KMeans(2,random_state=1,n_init=50).fit_predict(z)
 print(n,{z.index[i]:int(lab[i]) for i in range(9)})
PY"
