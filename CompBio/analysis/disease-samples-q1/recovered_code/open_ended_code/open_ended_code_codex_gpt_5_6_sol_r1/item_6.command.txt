/bin/bash -lc "OPENBLAS_NUM_THREADS=1 python - <<'PY'
import pandas as pd,numpy as np
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
A=['Sample_2','Sample_3','Sample_5','Sample_7']; B=[c for c in x if c not in A]
# pseudocount modest based TPM-ish sum million (values CPM scale actually)
ma=x[A].mean(1); mb=x[B].mean(1)
lf=np.log2((ma+0.1)/(mb+0.1))
# avg and per-sample values for top changes, require mean >2
for name,s in [('A_up',lf[(ma>2)].nlargest(100)),('B_up',lf[(mb>2)].nsmallest(100))]:
 print('\\n'+name)
 for g,z in s.head(80).items(): print(f'{g}\\t{z:.2f}\\t{ma[g]:.1f}\\t{mb[g]:.1f}')
PY"
