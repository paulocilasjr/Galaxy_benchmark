/bin/bash -lc "OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 python - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
x=pd.read_csv('disease.samples.q1.tsv',sep='\\t',index_col=0)
y=np.log2(x+1)
v=y.var(1); yy=y.loc[v.nlargest(5000).index]
p=PCA().fit(yy.T); pcs=p.transform(yy.T)
print('var',p.explained_variance_ratio_[:6])
for s,row in zip(x.columns,pcs): print(s,*(f'{z:.3f}' for z in row[:5]))
print('\\ncorr log')
print(y.corr().round(2).to_string())
PY"
