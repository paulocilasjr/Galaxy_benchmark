/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
fn='disease.samples.q1.tsv'
df=pd.read_csv(fn,sep='\\t',index_col=0)
print(df.shape, df.columns.tolist(), flush=True)
log=np.log2(df+1)
print('totals')
print(df.sum().sort_values().to_string(), flush=True)
print('detected_gt1')
print((df>1).sum().sort_values().to_string(), flush=True)
print('corr')
print(log.corr().round(3).to_string(), flush=True)
# eigen PCA simple np SVD on top variable genes
var=log.var(axis=1)
for n in [500,2000,10000,log.shape[0]]:
    idx=var.nlargest(n).index if n<log.shape[0] else log.index
    X=log.loc[idx].T.values
    X=X-X.mean(axis=0)
    U,S,Vt=np.linalg.svd(X, full_matrices=False)
    print('PCA top',n,'varratio', (S*S/(S*S).sum())[:5])
    for s, coords in zip(df.columns, U[:,:5]*S[:5]):
        print(s, ' '.join(f'{v:.3f}' for v in coords))
PY"
