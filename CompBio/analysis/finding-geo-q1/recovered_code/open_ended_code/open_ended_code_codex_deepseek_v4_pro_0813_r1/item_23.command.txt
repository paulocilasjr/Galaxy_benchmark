/bin/bash -lc "python3 - <<'PY'
import gzip,h5py,os
bc=[]
with gzip.open('check125620/barcodes.tsv.gz','rt') as f:
    for line in f:
        line=line.strip()
        if line: bc.append(line)
print('barcodes',len(bc),bc[:10],bc[-10:])
with h5py.File('finding.geo.q1.h5ad','r') as f:
    obs=[x.decode() for x in f['obs/_index'][:]]
print('obs',len(obs),obs[:10],obs[-10:])
print('sets equal', set(obs)==set(bc))
print('first equal', obs[:10]==bc[:10], 'last', obs[-10:]==bc[-10:])
# missing/extras
s=set(bc); o=set(obs)
print('missing',len(s-o), list(s-o)[:10]); print('extra',len(o-s), list(o-s)[:10])
PY"
