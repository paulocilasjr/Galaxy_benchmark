/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    genes=f['var/gene_symbols/categories'][:]
    codes=f['var/gene_symbols/codes'][:]
    symbols=genes[codes]
    print('first/last symbols', symbols[:20], symbols[-20:])
    # compute counts per gene using csr
    from scipy.sparse import csr_matrix
    data=f['X/data'][:]
    indices=f['X/indices'][:]
    indptr=f['X/indptr'][:]
    X=csr_matrix((data,indices,indptr), shape=(f['X'].attrs['shape'][0], f['X'].attrs['shape'][1]))
    sums=np.asarray(X.sum(axis=0)).ravel()
    n_cells=np.asarray((X>0).sum(axis=0)).ravel()
    order=np.argsort(-sums)
    for i in order[:100]:
        print(i, symbols[i], int(sums[i]), int(n_cells[i]))
PY"
