import gzip, h5py
with gzip.open('geo_check/GSM5134084_filtered_barcodes.tsv.gz','rt') as f: bc=f.read().splitlines()
with gzip.open('geo_check/GSM5134084_filtered_features.tsv.gz','rt') as f: feat=f.read().splitlines()
print('filtered barcodes',len(bc), bc[:5], bc[-5:])
print('filtered features',len(feat), feat[:3], feat[-3:])
with gzip.open('geo_check/GSM5134084_raw_barcodes.tsv.gz','rt') as f: rbc=f.read().splitlines()
print('raw barcodes',len(rbc), rbc[:3], rbc[-3:])
with h5py.File('finding.geo.q1.h5ad','r') as f:
    obs=list(f['obs/_index'])
    var=list(f['var/_index'])
    print('h5ad obs',len(obs), obs[:5], obs[-5:])
    print('h5ad var',len(var), var[:5], var[-5:])
print('obs equal', obs==bc if len(obs)==len(bc) else False)
print('var equal', var==feat if len(var)==len(feat) else False)
PY"
