/bin/bash -lc "python3 - <<'PY'
import h5py
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    print('uns children:', list(f['uns'].keys()))
    for k in f['uns'].keys():
        obj=f['uns'][k]
        print('uns',k, type(obj).__name__, obj.shape if isinstance(obj,h5py.Dataset) else '', obj.dtype if isinstance(obj,h5py.Dataset) else '')
        if isinstance(obj,h5py.Dataset):
            try: print(repr(obj[()])[:1000])
            except Exception as e: print('err',e)
    print('obs columns attrs:', f['obs'].attrs.get('column-order'))
    print('obs index first/last:', f['obs/_index'][:10], f['obs/_index'][-10:])
    print('var columns attrs:', f['var'].attrs.get('column-order'))
PY"
