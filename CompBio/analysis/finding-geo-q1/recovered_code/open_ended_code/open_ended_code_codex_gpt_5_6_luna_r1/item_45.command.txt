/bin/bash -lc "python - <<'PY'
import urllib.parse, urllib.request, json, re, time, sys
term='(\"Mus musculus\"[Organism]) AND (\"single cell\"[All Fields]) AND (\"10X\"[All Fields]) AND GSE[Filter] AND (\"2017/01/01\"[PDAT] : \"2020/12/31\"[PDAT])'
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
def get(params):
 u=base+'esearch.fcgi?'+urllib.parse.urlencode(params) if params.get('db') and 'term' in params else base+'esummary.fcgi?'+urllib.parse.urlencode(params)
 return json.load(urllib.request.urlopen(u))
ids=get({'db':'gds','term':term,'retmode':'json','retmax':'6000'})['esearchresult']['idlist']
print('NIDS',len(ids),file=sys.stderr)
allr={}
for st in range(0,len(ids),250):
 chunk=ids[st:st+250]
 obj=get({'db':'gds','id':','.join(chunk),'retmode':'json'})['result']
 allr.update(obj); time.sleep(.4)
pat=re.compile(r'brain|neuro|cort|hippoc|embry|develop|retin|spinal|cerebell|neuron|ganglion|glia|cns|dopamin|olfact|forebrain|midbrain|cortex|organoid',re.I)
for i in ids:
 x=allr.get(i,{})
 text=' '.join(str(x.get(k,'')) for k in ['title','summary','seriestitle'])
 if pat.search(text):
  print(x.get('accession'),x.get('pdat'),x.get('title'), '| samples:', '; '.join(s.get('title','') for s in x.get('samples',[])[:4]), '| rel',x.get('relations'), 'ss',x.get('ssinfo'))
PY"
