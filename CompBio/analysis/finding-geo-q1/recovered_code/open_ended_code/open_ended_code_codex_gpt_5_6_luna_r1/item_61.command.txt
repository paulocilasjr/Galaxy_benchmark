def call(endpoint,params):
 u='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'+endpoint+'?'+urllib.parse.urlencode(params)
 for a in range(5):
  try:
   raw=urllib.request.urlopen(u).read(); return json.loads(raw)
  except Exception as e:
   print('retry',endpoint,e,file=sys.stderr); time.sleep(3+a*2)
 raise RuntimeError
ids=call('esearch.fcgi',{'db':'gds','term':q,'retmode':'json','retmax':'500'})['esearchresult']['idlist']; print('n',len(ids),file=sys.stderr)
for st in range(0,len(ids),100):
 obj=call('esummary.fcgi',{'db':'gds','id':','.join(ids[st:st+100]),'retmode':'json'})['result']
 for i in ids[st:st+100]:
  x=obj[i]
  print(x.get('accession'),x.get('pdat'),x.get('title'),'|',x.get('suppfile'),'|',','.join(s.get('title','') for s in x.get('samples',[])[:3]))
 time.sleep(3)
PY"
