/bin/bash -lc "python - <<'PY'
import urllib.parse,urllib.request,json,time,sys
q='(\"Mus musculus\"[Organism]) AND (\"H5\"[All Fields]) AND (brain OR neural OR cortex OR neuron OR spinal)[All Fields] AND GSE[Filter] AND (\"2016/01/01\"[PDAT]:\"2020/12/31\"[PDAT])'
b='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
u=b+'esearch.fcgi?'+urllib.parse.urlencode({'db':'gds','term':q,'retmode':'json','retmax':500})
raw=urllib.request.urlopen(u,timeout=60).read(); print('search bytes',len(raw),file=sys.stderr); ids=json.loads(raw)['esearchresult']['idlist']; print('ids',len(ids),file=sys.stderr);time.sleep(12)
for st in range(0,min(len(ids),120),20):
 p={'db':'gds','id':','.join(ids[st:st+20]),'retmode':'json'}
 try:
  x=urllib.request.urlopen(b+'esummary.fcgi?'+urllib.parse.urlencode(p),timeout=90).read(); print('raw',st,len(x),file=sys.stderr); d=json.loads(x)['result'];
  for i in d['uids']:print(d[i].get('accession',''),'\\t',d[i].get('title',''))
 except Exception as e: print('ERR',st,e,file=sys.stderr)
 time.sleep(12)
PY"
