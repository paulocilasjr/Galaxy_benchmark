import glob,h5py,os
for p in glob.glob('GSM*.h5'):
 try:
  with h5py.File(p,'r') as f:
   def info(g):
    if 'barcodes' in g and 'data' in g:
     feat=g.get('genes',g.get('features')); return len(g['barcodes']),len(feat),list(g.keys())
    for v in g.values():
     if isinstance(v,h5py.Group):
      x=info(v)
      if x:return x
   print(os.path.basename(p),info(f))
 except Exception as e: print('ERR',p,e)
PY"
