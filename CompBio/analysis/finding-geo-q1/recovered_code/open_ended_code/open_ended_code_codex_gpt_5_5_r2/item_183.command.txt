/bin/bash -lc "mkdir -p geo_check/gse113576 && cd geo_check/gse113576 && curl -L 'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE113nnn/GSE113576/suppl/GSE113576_barcodes.tsv.gz' -o barcodes.tsv.gz && curl -L 'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE113nnn/GSE113576/suppl/GSE113576_genes.tsv.gz' -o genes.tsv.gz && python - <<'PY'
import gzip, os
ours=open('../fingerprints/obs.txt').read().splitlines(); os=set(ours)
for fn in ['genes.tsv.gz','barcodes.tsv.gz']:
    try:
        rows=[line.strip() for line in gzip.open(fn,'rt')]
        print(fn, os.path.getsize(fn), len(rows), rows[:5], rows[-5:])
    except Exception as e: print(fn,e)
print('overlap', len(os & set(rows)), 'equal', rows==ours)
PY"
