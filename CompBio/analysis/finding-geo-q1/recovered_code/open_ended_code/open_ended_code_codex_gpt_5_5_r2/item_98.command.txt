/bin/bash -lc "python - <<'PY'
import urllib.request, urllib.parse, json, time, re
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
terms=[
 '\"28692\" AND \"Mus musculus\" AND \"Expression profiling by high throughput sequencing\"',
 '\"ENSMUSG00000051951\" AND GEO',
 '\"Cell Ranger\" AND \"Mus musculus\" AND \"2.1.0\" AND GEO',
 '\"Cell Ranger\" AND \"Mus musculus\" AND \"10x\" AND sensory',
 '\"Cell Ranger\" AND \"Mus musculus\" AND \"10x\" AND neuron',
 '\"Cell Ranger\" AND \"Mus musculus\" AND \"10x\" AND cortex',
 '\"Cell Ranger\" AND \"Mus musculus\" AND \"10x\" AND ganglion',
 '\"Cell Ranger\" AND \"Mus musculus\" AND \"10x\" AND \"neural crest\"',
]
for term in terms:
    params={'db':'gds','term':term,'retmax':'50','retmode':'json'}
    url=base+'esearch.fcgi?'+urllib.parse.urlencode(params)
    with urllib.request.urlopen(url, timeout=30) as r: data=json.load(r)
    ids=data['esearchresult']['idlist']; print('\\nTERM',term,'count',data['esearchresult']['count'],'ids',ids[:10])
    if ids:
        params={'db':'gds','id':','.join(ids[:10]),'retmode':'json'}
        url=base+'esummary.fcgi?'+urllib.parse.urlencode(params)
        with urllib.request.urlopen(url, timeout=30) as r: summ=json.load(r)
        for id in ids[:10]:
            it=summ['result'].get(id,{})
            print(id, it.get('accession'), it.get('title')[:120] if it.get('title') else '')
    time.sleep(.35)
PY"
