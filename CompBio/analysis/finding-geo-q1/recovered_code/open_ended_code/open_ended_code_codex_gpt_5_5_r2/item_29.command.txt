/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Pax6 Sox2 Sox11 Sox4 Neurod1 Neurod2 Neurog2 Eomes Tbr1 Fezf2 Satb2 Cux1 Cux2 Foxp2 Reln Gad1 Gad2 Slc17a6 Slc17a7 Olig1 Olig2 Pdgfra Mog Mbp Gfap Aldh1l1 Aqp4 Nes Top2a Mki67 Hbb-bs Hba-a1 Epcam Krt8 Krt18 Pecam1 Cdh5 Ptprc Col1a1 Myod1 Myog Dcx Mapt Snap25 Rbfox3'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/gene_symbols/categories'][:]]
    codes=f['var/gene_symbols/codes'][:]
    syms=np.array([cats[c] if c>=0 else '' for c in codes])
    for m in markers:
        idx=np.where(syms==m)[0]
        if len(idx):
            v=X[:,idx[0]].toarray().ravel()
            print(f'{m}\\t{int(v.sum())}\\t{int((v>0).sum())}')
PY"
