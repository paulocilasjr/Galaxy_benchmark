/bin/bash -lc "python - <<'PY'
import urllib.request, urllib.parse, json, time
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
terms=['\"refdata-cellranger-mm10-2.1.0\"','\"mm10-2.1.0\" AND \"Mus musculus\"','\"Cell Ranger 2.1.1\" AND \"Mus musculus\"','\"Cell Ranger 2.1.0\" AND \"Mus musculus\"','\"28692 features\"','\"28692 genes\" AND \"GEO\"']
for term in terms:
    url=base+'esearch.fcgi?'+urllib.parse.urlencode({'db':'gds','term':term,'retmax':'200','retmode':'json'})
    data=json.load(urllib.request.urlopen(url, timeout=30))
    ids=data['esearchresult']['idlist']; print('\\nTERM',term,'count',data['esearchresult']['count'],'nids',len(ids))
    for chunk_start in range(0,min(len(ids),50),20):
        chunk=ids[chunk_start:chunk_start+20]
        if not chunk: continue
        summ=json.load(urllib.request.urlopen(base+'esummary.fcgi?'+urllib.parse.urlencode({'db':'gds','id':','.join(chunk),'retmode':'json'}), timeout=30))
        for id in chunk:
            it=summ['result'].get(id,{})
            print(id, it.get('accession'), it.get('gdsType'), (it.get('title') or '')[:100])
        time.sleep(.34)
PY"
