/bin/bash -lc "cd geo_check/gse122357 && python - <<'PY'
import tarfile,gzip,csv,io
ours=open('../fingerprints/obs.txt').read().splitlines(); ours_no=[x[:-2] for x in ours]
ours_set=set(ours_no)
with tarfile.open('GSE122357_RAW.tar') as tar:
    for m in tar.getmembers():
        f=gzip.GzipFile(fileobj=tar.extractfile(m))
        text=io.TextIOWrapper(f)
        header=next(csv.reader(text))
        bars=header[1:]
        nrows=sum(1 for _ in text)
        print(m.name, 'cells',len(bars),'genes',nrows,'first',bars[:5], 'overlap',len(ours_set&set(bars)), 'equal',bars==ours_no)
PY"
