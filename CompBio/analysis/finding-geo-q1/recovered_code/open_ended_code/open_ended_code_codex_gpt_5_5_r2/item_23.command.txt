import h5py, gzip
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    ours=[x.decode() if isinstance(x,bytes) else str(x) for x in f['obs/_index'][:]]
ours_set=set(ours)
with gzip.open('geo_check/GSM3302829_embryo3_SeuratBarcodes.tsv.gz','rt') as fh:
    cand=[line.strip() for line in fh]
print(len(cand), len(ours), 'intersection', len(set(cand)&ours_set))
print('cand_subset_ours', set(cand).issubset(ours_set), 'ours_subset_cand', ours_set.issubset(set(cand)))
print('first ours not cand', next((x for x in ours if x not in set(cand)), None))
print('first cand not ours', next((x for x in cand if x not in ours_set), None))
PY"
