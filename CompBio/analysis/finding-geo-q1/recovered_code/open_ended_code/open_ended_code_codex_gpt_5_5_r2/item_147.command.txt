/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
from pathlib import Path
ours=Path('geo_check/fingerprints/obs.txt').read_text().splitlines()
# cell sums for GSE132355 E12 raw matrix, no scipy load needed
mtx='geo_check/gse132355/GSM3860735_E12_matrix.mtx.gz'
with gzip.open(mtx,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrow,ncol,nnz=map(int,line.split()); break
    sums=np.zeros(ncol,dtype=np.int64)
    for line in f:
        r,c,v=line.split(); sums[int(c)-1]+=int(v)
with gzip.open('geo_check/gse132355/GSM3860735_E12_barcodes.tsv.gz','rt') as f: bars=[line.strip() for line in f]
for th in [0,100,200,250,273,274,300,500,1000]:
    sel=[bars[i] for i,s in enumerate(sums) if s>=th]
    print(th, len(sel), sel[:5], 'equals ours', sel==ours, 'overlap', len(set(sel)&set(ours)))
print('ours sums stats in E12', min(sums[bars.index(b)] for b in ours), max(sums[bars.index(b)] for b in ours), sum(sums[bars.index(b)] for b in ours))
PY"
