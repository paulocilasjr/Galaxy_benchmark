    if any(key in line for key in ['"'^SERIES'"','"'!Series_title'"','"'!Series_summary'"','"'!Series_overall_design'"','"'!Series_supplementary_file'"','"'!Series_sample_id'"','"'!Series_type'"','"'!Series_relation'"','"'!Series_platform_organism'"','"'!Series_sample_organism'"']):
        print(line)
PY"
