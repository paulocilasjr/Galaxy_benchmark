/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
markers='Prph Isl1 Isl2 Pou4f1 Pou4f2 Ntrk1 Ntrk2 Ntrk3 Ret Runx1 Runx3 Prrxl1 Tlx3 Lhx9 Phox2b Hoxb8 Hoxb9 Hoxa5 P2rx3 Scn9a Scn10a Trpv1 Trpa1 Calca Calcb Tac1 Npy Sst Gal Th Thbs4 Tubb3 Dcx Neurog1 Neurog2 Neurod4 Neurod1 Ebf1 Onecut1 Onecut2 Onecut3 Tfap2a Tfap2b Pax3 Pax7 Sox10 Fabp7'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    cats=[x.decode() for x in f['var/gene_symbols/categories'][:]]
    syms=np.array([cats[c] if c>=0 else '' for c in f['var/gene_symbols/codes'][:]])
    data=f['X/data'][:]; idx=f['X/indices'][:]; indptr=f['X/indptr'][:]
    cs=np.zeros(len(syms),np.int64); cn=np.zeros(len(syms),np.int64); np.add.at(cs,idx,data); np.add.at(cn,idx,1)
    n=len(indptr)-1
    for m in markers:
        wh=np.where(syms==m)[0]
        if len(wh):
            i=wh[0]; print(f'{m}\\t{cs[i]}\\t{cn[i]}\\t{cn[i]/n:.3f}')
        else: print(m,'missing')
PY"
