import urllib.parse, urllib.request, json, xml.etree.ElementTree as ET
queries=[
'(\"Neurog2\"[All Fields] AND \"Onecut2\"[All Fields] AND \"10x\"[All Fields])',
'(\"P2rx3\"[All Fields] AND \"Onecut2\"[All Fields])',
'(\"hSeO\"[All Fields])',
'(\"human ascending somatosensory\"[All Fields])',
'(\"sensory organoid\"[All Fields] AND \"GEO\"[All Fields])',
]
for q in queries:
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gds&retmax=20&term='+urllib.parse.quote(q)
    print('QUERY',q)
    txt=urllib.request.urlopen(url, timeout=20).read().decode()
    root=ET.fromstring(txt)
    ids=[e.text for e in root.findall('.//Id')]
    print('count', root.findtext('Count'), ids)
    if ids:
        url2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=gds&id='+','.join(ids)
        data=urllib.request.urlopen(url2, timeout=20).read().decode()
        print(data[:3000])
PY"
