import urllib.request, math, gzip, io, re
sizes=[67243192,27895704,28438468,29862324,26946742,27648989,27317701,30056263]
names=['GSM5401072_E9_5_raw_count.csv.gz','GSM5401073_E11_5_1_raw_count.csv.gz','GSM5401074_E11_5_2_raw_count.csv.gz','GSM5401075_E11_5_3_raw_count.csv.gz','GSM5401076_E13_5_1_raw_count.csv.gz','GSM5401077_E13_5_2_raw_count.csv.gz','GSM5401078_E13_5_3_raw_count.csv.gz','GSM5401079_E13_5_4_raw_count.csv.gz']
base='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE178nnn/GSE178931/suppl/GSE178931_RAW.tar'
off=0
for name,sz in zip(names,sizes):
    header_off=off
    content_off=off+512
    end=content_off+min(sz,500000)-1
    req=urllib.request.Request(base, headers={'Range': f'bytes={content_off}-{end}'})
    data=urllib.request.urlopen(req, timeout=30).read()
    try:
        txt=gzip.GzipFile(fileobj=io.BytesIO(data)).read(2000).decode('utf-8','replace')
    except Exception as e:
        txt='ERR '+repr(e)
    print('\\n##',name,'offset',content_off,'peek')
    print(txt[:1000].replace('\\n','\\n'))
    off=content_off+math.ceil(sz/512)*512
print('total calc',off)
PY"
