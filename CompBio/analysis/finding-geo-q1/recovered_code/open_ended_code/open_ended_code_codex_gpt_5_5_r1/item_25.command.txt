import h5py, numpy as np
markers='Rax Vsx2 Chx10 Atoh7 Math5 Pou4f2 Pou4f1 Rbpms Isl1 Isl2 Eomes Crx Otx2 Neurod1 Neurod4 Onecut1 Onecut2 Prox1 Lhx1 Lhx9 Tfap2a Tfap2b Ap2a Ap2b Pax6 Six3 Six6 Prdm13 Prdm8 Hes1 Hes5 Ascl1 Neurog2 Dll1 Dll3 Dkk3 Sstr2 Slc17a6 Slc17a7 Gad1 Gad2 Calb1 Calb2 Ptf1a Pcp2 Grm6 Rho Opn1mw Opn1sw Rcvrn Recoverin Prph Nefl Nefm Gap43 Tubb3 Stmn2 Sox2 Mki67 Top2a Ccnd1 Ccnd2'.split()

    cats=[x.decode() for x in f['var/gene_symbols/categories'][:]]
    syms=np.array([cats[c] if c>=0 else '' for c in f['var/gene_symbols/codes'][:]])
    data=f['X/data'][:]; idx=f['X/indices'][:]; indptr=f['X/indptr'][:]

    n=len(indptr)-1
    seen=set()
    for m in markers:
        if m in seen: continue
        seen.add(m)
        wh=np.where(syms==m)[0]
        if len(wh):
            i=wh[0]; print(f'{m}\\t{cs[i]}\\t{cn[i]}\\t{cn[i]/n:.3f}')
