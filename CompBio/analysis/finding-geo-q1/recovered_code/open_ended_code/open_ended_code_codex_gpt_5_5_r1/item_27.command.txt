/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
markers='Eya1 Eya2 Six1 Six2 Six4 Sox10 Foxd3 Tfap2a Tfap2b Phox2a Phox2b Gata2 Gata3 Neurod4 Neurog1 Neurog2 Isl1 Pou4f1 Tlx3 Prdm12 Nhlh1 Nhlh2 Myt1 Myt1l Lmx1b Lmx1a Maf Bnc2 Elavl3 Elavl4 Uchl1 Ntrk1 Ntrk2 Ntrk3 P2rx3 Scn9a Tmem233 Fam19a4 Rspo2'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    cats=[x.decode() for x in f['var/gene_symbols/categories'][:]]; syms=np.array([cats[c] if c>=0 else '' for c in f['var/gene_symbols/codes'][:]])
    data=f['X/data'][:]; idx=f['X/indices'][:]; indptr=f['X/indptr'][:]
    cs=np.zeros(len(syms),np.int64); cn=np.zeros(len(syms),np.int64); np.add.at(cs,idx,data); np.add.at(cn,idx,1); n=len(indptr)-1
    for m in markers:
        wh=np.where(syms==m)[0]
        if len(wh): i=wh[0]; print(f'{m}\\t{cs[i]}\\t{cn[i]}\\t{cn[i]/n:.3f}')
        else: print(m,'missing')
PY"
