/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, scipy.sparse as sp
f=h5py.File('finding.geo.q1.h5ad','r')
data=f['X/data'][:]; indices=f['X/indices'][:]; indptr=f['X/indptr'][:]
X=sp.csr_matrix((data,indices,indptr), shape=tuple(f['X'].attrs['shape']))
print('X',X.shape,'nnz',X.nnz)
genes=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/_index'][:]])
codes=f['var/gene_symbols/codes'][:]; cats=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/gene_symbols/categories'][:]])
syms=cats[codes]
print(genes[:5],syms[:5])
tot=np.asarray(X.sum(axis=0)).ravel()
csc=X.tocsc(); nc=np.diff(csc.indptr)
idx=np.argsort(-tot)[:100]
for i in idx:
 print(f'{i:6d} {genes[i]:20s} {syms[i]:20s} total={tot[i]:10d} cells={nc[i]:5d}')
print('\\nmarkers')
markers=['Pecam1','Cdh5','Kdr','Ptprc','Cd3d','Cd3e','Cd4','Cd8a','Cd19','Ms4a1','Cd68','Adgre1','Itgam','Nkg7','Ncr1','Col1a1','Col3a1','Pdgfra','Epcam','Krt8','Krt18','Acta2','Myh11','Mbp','Plp1','Sox10','Hbb-bs','Hba-a1','Hba-a2','Cst3','Apoe','S100a8','S100a9','Lyz2','Cd74','H2-Aa','H2-Ab1','Gzmb','Foxp3']
s2i={s:i for i,s in enumerate(syms)}
for m in markers:
 i=s2i.get(m)
 if i is not None: print(m, 'tot',int(tot[i]),'cells',int(nc[i]))
PY"
