/bin/bash -lc "python - <<'PY'
import urllib.request
urls=['https://ftp.ncbi.nlm.nih.gov/geo/series/GSE118nnn/GSE118953/','https://ftp.ncbi.nlm.nih.gov/geo/series/GSE102nnn/GSE102934/','https://ftp.ncbi.nlm.nih.gov/geo/series/GSE125nnn/GSE125788/','https://ftp.ncbi.nlm.nih.gov/geo/series/GSE116nnn/GSE116140/','https://ftp.ncbi.nlm.nih.gov/geo/series/GSE89nnn/GSE89798/','https://ftp.ncbi.nlm.nih.gov/geo/series/GSE275nnn/GSE275643/']
for u in urls:
 print('\\nURL',u)
 try:
  with urllib.request.urlopen(u,timeout=30) as r:
   txt=r.read().decode()
  print(txt[:5000])
 except Exception as e: print('ERR',e)
PY"
