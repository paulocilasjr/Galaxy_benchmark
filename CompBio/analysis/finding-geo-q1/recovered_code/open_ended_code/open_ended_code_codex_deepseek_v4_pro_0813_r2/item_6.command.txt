/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np
a=ad.read_h5ad('finding.geo.q1.h5ad', backed='r')
X=a.X
# compute per cell sums and nonzero genes from sparse in chunks
n=X.shape[0]
sums=np.zeros(n,dtype=np.int64)
ngenes=np.zeros(n,dtype=np.int64)
for i in range(n):
 r=X[i]
 sums[i]=r.sum()
 ngenes[i]=r.nnz
print('total UMI',sums.sum(),'mean',sums.mean(),'median',np.median(sums))
print('min max cell sums',sums.min(),sums.max())
print('percentiles',np.percentile(sums,[0,1,5,10,25,50,75,90,95,99,100]))
print('mean genes/cell',ngenes.mean(),'median',np.median(ngenes), 'min max',ngenes.min(),ngenes.max())
print('percentiles genes',np.percentile(ngenes,[0,1,5,10,25,50,75,90,95,99,100]))
PY"
