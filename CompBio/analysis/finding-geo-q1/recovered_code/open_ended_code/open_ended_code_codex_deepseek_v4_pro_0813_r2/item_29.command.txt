/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, urllib.request, gzip, io
# get our syms
f=h5py.File('finding.geo.q1.h5ad','r')
codes=f['var/gene_symbols/codes'][:]; cats=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/gene_symbols/categories'][:]])
syms=cats[codes]
print('our vars',len(syms),'unique syms',len(np.unique(syms)))
# download GSE102934 gene names
u='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE102nnn/GSE102934/suppl/GSE102934_gene_names_10x.txt.gz'
data=urllib.request.urlopen(u,timeout=60).read(); txt=gzip.decompress(data).decode(); gl=txt.splitlines()
print('gse genes',len(gl),'unique',len(set(gl)))
print('our first 30',syms[:30])
print('gse first 30',gl[:30])
print('our last 30',syms[-30:])
print('gse last 30',gl[-30:])
# compare if gse list is prefix/suffix or exact subset
from collections import Counter
c1=Counter(syms); c2=Counter(gl)
print('symbols in our not gse count',sum((c1-c2).values()), 'examples', list((c1-c2).items())[:20])
print('symbols in gse not our count',sum((c2-c1).values()), 'examples', list((c2-c1).items())[:20])
# longest common prefix/suffix
lp=0
while lp<min(len(syms),len(gl)) and syms[lp]==gl[lp]: lp+=1
ls=0
while ls<min(len(syms),len(gl)) and syms[-1-ls]==gl[-1-ls]: ls+=1
print('lcp',lp,'lcs',ls)
# is gl subsequence of syms?
it=iter(syms); cnt=0
for x in gl:
 for y in it:
  if y==x: cnt+=1; break
print('gse subsequence matched',cnt)
PY"
