/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
 cats=np.array([x.decode() for x in f['var/gene_symbols/categories'][:]])
 syms=cats[f['var/gene_symbols/codes'][:]]
 total=np.asarray(X.sum(0)).ravel(); nz=np.asarray((X>0).sum(0)).ravel()
 ci=np.asarray(X.sum(1)).ravel(); nn=np.diff(X.indptr)
 print('count quantiles',np.quantile(ci,[0,.01,.1,.25,.5,.75,.9,.95,.99,1]))
 print('cells counts <= thresholds',[(t,int((ci<=t).sum())) for t in [100,200,300,500,1000,2000,5000,10000]])
 for i in np.argsort(total)[-100:][::-1]: print(syms[i],total[i],nz[i])
PY"
