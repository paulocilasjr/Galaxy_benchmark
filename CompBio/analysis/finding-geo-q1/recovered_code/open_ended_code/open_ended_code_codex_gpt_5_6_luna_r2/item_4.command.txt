/bin/bash -lc "python - <<'PY'
import h5py, json, os
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    print('root attrs:')
    for k,v in f.attrs.items(): print(' ',k,repr(v))
    def walk(name,obj):
        if isinstance(obj,h5py.Group):
            print('GROUP',name, 'attrs=', {k:repr(v)[:300] for k,v in obj.attrs.items()})
        else:
            print('DATA',name, 'shape=',obj.shape,'dtype=',obj.dtype,'attrs=', {k:repr(v)[:300] for k,v in obj.attrs.items()})
    f.visititems(walk)
PY"
