/bin/bash -lc "python - <<'PY'
import requests,gzip,io
u='https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM5212nnn/GSM5212829/suppl/GSM5212829_Wound_tissue_count_matrix.tsv.gz'
b=requests.get(u).content
with gzip.open(io.BytesIO(b),'rt') as f:
 line=f.readline().rstrip('\\n').split('\\t')
 print('cols incl gene',len(line),'cells',len(line)-1,'first',line[1:5], 'last',line[-5:])
 n=0
 for line in f:n+=1
 print('genes',n)
PY"
