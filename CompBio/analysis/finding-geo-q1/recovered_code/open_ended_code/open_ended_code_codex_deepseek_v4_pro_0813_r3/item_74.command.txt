/bin/bash -lc "python - <<'PY'
import json,re
S=json.load(open('broad_summaries.json'))
out=[]
for uid,x in S.items():
    supp=(x.get('suppfile') or '').upper()
    if not any(k in supp for k in ['MTX','TSV','H5','TAR']):
        continue
    low=(x.get('title','')+' '+(x.get('summary') or '')).lower()
    # score developmental neural terms
    score=0
    terms=['develop','embry','spinal','neural tube','hindbrain','midbrain','forebrain','retina','cortex','progenitor','neur','cns','brain','nervous system','e9','e10','e11','e12','e13','e14','e15','temporal','spatial']
    for t in terms:
        if t in low: score+=1
    # exclude obvious adult disease/immune/cancer? maybe keep
    exclude=['als','amyotrophic','multiple sclerosis','injur','ischem','alzheimer','parkinson','tumor','glioma','leukemia','cancer','microglia','macrophage','t cell','b cell','immune','lymph','blood','bone marrow','schwann','pain','sensory neuron']
    ex=sum(1 for w in exclude if w in low)
    if score>=3 and ex<=2:
        out.append((score,uid,x))
out.sort(key=lambda t:-t[0])
print('candidates',len(out))
for score,uid,x in out[:300]:
    print(f'{score:2d}',uid,x.get('accession'),'n',x.get('n_samples'),'supp',x.get('suppfile'),'pdat',x.get('pdat'))
    print('   ',x.get('title'))
PY"
