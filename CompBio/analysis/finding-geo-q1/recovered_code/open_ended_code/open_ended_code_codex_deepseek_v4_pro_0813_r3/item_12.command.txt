/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
f=h5py.File('finding.geo.q1.h5ad','r')
var_idx=f['var/_index'][:].astype(str)
cats=f['var/gene_symbols/categories'][:].astype(str)
codes=f['var/gene_symbols/codes'][:]
sym=np.where(codes>=0,cats[codes],'NA')
indices=f['X/indices'][:]; indptr=f['X/indptr'][:]; data=f['X/data'][:]
# build sym->gene index (first)
sym2idx={}
for i,s in enumerate(sym):
    if s not in sym2idx: sym2idx[s]=i
markers=['Snap25','Syt1','Slc17a7','Gad1','Gad2','Slc32a1','Aif1','Cx3cr1','Mbp','Mog','Plp1','Olig1','Olig2','Pdgfra','Sox10','Gfap','Aqp4','Vim','Flt1','Cldn5','Pecam1','Rbfox3','Tubb3','Nnat','Stmn2','Map1b','Mki67','Sox2','Pax6','Eomes','Neurod1','Dcx','Foxp2','Pvalb','Sst','Vip','Lamp5','Npy','Th','Chat','Slc6a3','Drd2','Drd1a','Tac1','Penk','Reln','Calb1','Calb2','Nrgn','Camk2a','Dlg4','Grin1','Gabra1','Gabbr1','Grm2','Grm5','Gria1','Gria2']
# sum counts per marker
for m in markers:
    idx=sym2idx.get(m,-1)
    if idx>=0:
        total=int(np.bincount(indices,weights=data,minlength=len(var_idx))[idx])
        ncells=int((np.diff(indptr)>0) & np.bincount(indices,minlength=len(var_idx))[idx]) # wrong
        gc=int(np.bincount(indices,minlength=len(var_idx))[idx])
        print(f'{m:12s} total={total:8d} ncells={gc:5d}')
    else:
        print(f'{m:12s} MISSING')
PY"
