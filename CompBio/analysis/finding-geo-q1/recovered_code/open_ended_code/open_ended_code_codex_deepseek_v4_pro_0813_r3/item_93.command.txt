/bin/bash -lc "python - <<'PY'
import json,re,math
L=json.load(open('candidate_filelists.json'))
rows=[]
for item in L:
    acc=item['acc']; text=item.get('text') or ''
    if not text: continue
    entries=[]
    for line in text.splitlines():
        if line.startswith('#') or not line.strip(): continue
        parts=line.split('\\t')
        if len(parts)>=4:
            try: size=int(parts[3])
            except: continue
            entries.append({'name':parts[1],'size':size,'type':parts[4] if len(parts)>4 else ''})
    feats=[e for e in entries if re.search(r'(features|genes)',e['name'],re.I) and not re.search(r'(atac|peaks)',e['name'],re.I)]
    bars=[e for e in entries if re.search(r'barcodes',e['name'],re.I)]
    # compute best feature closeness to 231294 among features with size 180k-280k
    cand_feat=[e for e in feats if 180000<=e['size']<=280000]
    if not cand_feat: continue
    best=min(cand_feat,key=lambda e:abs(e['size']-231294))
    # any bar with size >60000
    bigbars=[e for e in bars if e['size']>=60000]
    if not bigbars: continue
    rows.append((abs(best['size']-231294),acc,best,bigbars,feats,bars))
rows.sort(key=lambda x:x[0])
print('candidates close feature size',len(rows))
for diff,acc,best,bigbars,feats,bars in rows[:100]:
    print(f'diff={diff:6d} {acc:12s} bestfeat {best[\"name\"]} {best[\"size\"]}')
    print('   bigbars:',[(e['name'],e['size']) for e in bigbars[:6]])
    print('   allfeats:',[(e['name'],e['size']) for e in feats[:4]])
PY"
