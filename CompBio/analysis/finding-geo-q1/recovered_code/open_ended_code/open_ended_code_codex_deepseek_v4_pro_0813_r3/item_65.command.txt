/bin/bash -lc "python - <<'PY'
import h5py,gzip
f=h5py.File('finding.geo.q1.h5ad','r')
ours=set(x.decode() for x in f['var/_index'][:])
std=[]
with gzip.open('candidates/GSE156023/GSM4720584_genes.tsv.gz','rt') as fh:
    for l in fh:
        p=l.rstrip('\\n').split('\\t'); std.append(p[0])
stdset=set(std)
print('ours',len(ours),'std',len(stdset),'our missing from std',len(ours-stdset),'std missing from ours',len(stdset-ours))
print('examples ours not std', list(ours-stdset)[:20])
print('examples std not ours', list(stdset-ours)[:50])
# preserve order
our_order=[x.decode() for x in f['var/_index'][:]]
print('first 20 std:',std[:20])
print('first 20 ours:',our_order[:20])
PY"
