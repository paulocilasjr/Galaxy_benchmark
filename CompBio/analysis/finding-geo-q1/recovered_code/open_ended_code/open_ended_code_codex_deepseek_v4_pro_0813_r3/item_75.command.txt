/bin/bash -lc "python - <<'PY'
import requests,json
for uid in ['200171891','200171890','200171892']:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':uid,'retmode':'json'})
 if r.status_code==200 and uid in r.json().get('result',{}):
  x=r.json()['result'][uid]
  print('\\n',uid,x.get('accession'),x.get('entrytype'),x.get('n_samples'),x.get('suppfile'),x.get('pdat'))
  print(x.get('title'))
  print((x.get('summary') or '')[:500])
  print('ftp',x.get('ftplink'))
 else:
  print('failed',uid,r.text[:200])
PY"
