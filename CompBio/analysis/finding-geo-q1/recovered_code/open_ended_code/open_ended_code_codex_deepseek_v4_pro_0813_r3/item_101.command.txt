/bin/bash -lc "python - <<'PY'
import json
S=json.load(open('broad_summaries.json'))
terms=['spinal cord','neural tube','hindbrain','midbrain','forebrain','neural crest','neural plate','cranial','embryonic brain','developing brain','brain development','nervous system development','cortex development','retina development','neurogenesis','motor neuron','progenitor','neuronal diversity','cns development']
items=[]
for uid,x in S.items():
 supp=(x.get('suppfile') or '').upper()
 if not any(k in supp for k in ['MTX','TSV','TAR','H5','H5AD']): continue
 low=(x.get('title','')+' '+(x.get('summary') or '')).lower()
 hits=[t for t in terms if t in low]
 if hits:
  items.append((len(hits),uid,x,hits))
items.sort(key=lambda t:-t[0])
print('total',len(items))
for n,uid,x,hits in items:
 print(f'{n:2d}',uid,x.get('accession'),'n',x.get('n_samples'),'supp',x.get('suppfile'),'pdat',x.get('pdat'), 'hits=',hits)
 print('   ',x.get('title'))
PY"
