/bin/bash -lc "python - <<'PY'
import requests,json,re
uids=['200335515','200280655','200310909','200293269','200337989','200172101','200125620','200156609','200245105','200183759','200156023','200171891']
for uid in uids:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':uid,'retmode':'json'})
 if uid not in r.json().get('result',{}): print('missing',uid); continue
 x=r.json()['result'][uid]; acc=x['accession']; prefix=acc[:6]+'nnn'
 print('\\n===',uid,acc,'n',x.get('n_samples'),'supp',x.get('suppfile'))
 print(x.get('title'))
 url=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{prefix}/{acc}/suppl/filelist.txt'
 rl=requests.get(url)
 if rl.status_code==200:
  txt=rl.text
  # print first 30 lines
  lines=txt.splitlines()
  for l in lines[:40]: print(l)
 else: print('filelist',rl.status_code)
PY"
