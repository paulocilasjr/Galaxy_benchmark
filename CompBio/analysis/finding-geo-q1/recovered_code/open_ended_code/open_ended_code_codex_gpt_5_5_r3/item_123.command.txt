/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Atoh1 Ptf1a Lhx1 Lhx5 Tlx3 Barhl1 Barhl2 Meis2 Lmx1a Lmx1b Zic1 Zic2 Pax7 Gbx2 En1 En2 Otx2 Wnt1 Fgf8 Foxp2 Foxp1 Skor1 Skor2 Ebf1 Ebf2 Ebf3 Tfap2b Olig2 Ascl1 Neurog1 Neurog2 Ptf1a Gsx1 Gsx2 Hoxa1 Hoxa2 Hoxb1 Hoxb2 Hoxb3 Hoxb4 Hoxc4 Hoxd4 Phox2b Lbx1 Pax3 Pax6 Dbx1 Dbx2 Nhlh1 Nhlh2'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]; codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes]); gmap={g:i for i,g in enumerate(genes)}
    sums=np.asarray(X.sum(axis=0)).ravel(); nz=np.asarray((X>0).sum(axis=0)).ravel()
    for g in markers:
        if g in gmap:
            i=gmap[g]; print(f'{g}\\t{int(sums[i])}\\t{int(nz[i])}')
PY"
