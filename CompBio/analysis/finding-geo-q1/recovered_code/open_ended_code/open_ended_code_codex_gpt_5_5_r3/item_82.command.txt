/bin/bash -lc 'for gse in GSE317146 GSE317114 GSE318565; do prefix=${gse:0:6}nnn; echo "$gse $prefix"; curl -sL "https://ftp.ncbi.nlm.nih.gov/geo/series/${prefix}/${gse}/suppl/filelist.txt" -o "${gse}_filelist.txt"; if head -1 "${gse}_filelist.txt" | grep -q '"'#Archive'; then sed -n '1,80p' \""'${gse}_filelist.txt"; else curl -sL "https://ftp.ncbi.nlm.nih.gov/geo/series/${prefix}/${gse}/suppl/" -o "${gse}_listing.html"; sed -n '"'1,80p' \""'${gse}_listing.html"; fi; done'
