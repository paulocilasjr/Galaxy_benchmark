/bin/bash -lc "curl -sL 'https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM3302nnn/GSM3302829/suppl/GSM3302829_embryo3_SeuratBarcodes.tsv.gz' -o GSM3302829_embryo3_SeuratBarcodes.tsv.gz
curl -sL 'https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM3302nnn/GSM3302829/suppl/GSM3302829_embryo3_SeuratGenes.tsv.gz' -o GSM3302829_embryo3_SeuratGenes.tsv.gz
curl -sL 'https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM3302nnn/GSM3302829/suppl/GSM3302829_embryo3_SeuratNorm.mtx.gz' -o GSM3302829_embryo3_SeuratNorm.mtx.gz
printf 'barcodes '; gzip -dc GSM3302829_embryo3_SeuratBarcodes.tsv.gz | wc -l
printf 'genes '; gzip -dc GSM3302829_embryo3_SeuratGenes.tsv.gz | wc -l
printf 'first barcodes\\n'; gzip -dc GSM3302829_embryo3_SeuratBarcodes.tsv.gz | head -10
printf 'first genes\\n'; gzip -dc GSM3302829_embryo3_SeuratGenes.tsv.gz | head -10
printf 'mtx header\\n'; gzip -dc GSM3302829_embryo3_SeuratNorm.mtx.gz | sed -n '1,5p'"
