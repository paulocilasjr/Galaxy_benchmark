/bin/bash -lc "python3 - <<'PY'
import h5py
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    def show_attrs(path):
        obj=f[path]
        print('\\nATTRS', path)
        for k,v in obj.attrs.items(): print(k, repr(v))
    for path in ['/', 'obs', 'var', 'uns', 'X', 'var/gene_symbols']:
        show_attrs(path)
    print('\\nobs keys', list(f['obs'].keys()))
    print('var keys', list(f['var'].keys()))
    print('uns keys', list(f['uns'].keys()))
    def dec(x): return x.decode() if isinstance(x,bytes) else str(x)
    print('\\nfirst obs index')
    print([dec(x) for x in f['obs/_index'][:20]])
    print('\\nlast obs index')
    print([dec(x) for x in f['obs/_index'][-20:]])
    print('\\nfirst var index')
    print([dec(x) for x in f['var/_index'][:20]])
    cats=f['var/gene_symbols/categories'][:]
    codes=f['var/gene_symbols/codes'][:20]
    print('\\nfirst gene symbols')
    print([dec(cats[i]) if i>=0 else None for i in codes])
PY"
