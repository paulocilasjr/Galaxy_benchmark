/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
p='finding.geo.q1.h5ad'
with h5py.File(p,'r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]
    codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
    cell_counts=np.asarray(X.sum(axis=1)).ravel()
    cell_genes=np.diff(X.indptr)
    gene_counts=np.asarray(X.sum(axis=0)).ravel()
    gene_cells=np.asarray((X>0).sum(axis=0)).ravel()
    print('shape', X.shape, 'nnz', X.nnz)
    print('cell counts min median mean max', cell_counts.min(), np.median(cell_counts), cell_counts.mean(), cell_counts.max())
    print('cell genes min median mean max', cell_genes.min(), np.median(cell_genes), cell_genes.mean(), cell_genes.max())
    print('total UMI', int(cell_counts.sum()))
    print('top genes by total:')
    for i in np.argsort(-gene_counts)[:100]:
        print(genes[i], int(gene_counts[i]), int(gene_cells[i]))
PY"
