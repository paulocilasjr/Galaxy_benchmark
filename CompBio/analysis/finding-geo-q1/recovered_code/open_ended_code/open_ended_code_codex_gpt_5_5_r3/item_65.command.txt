/bin/bash -lc "python3 - <<'PY'
import gzip, csv, time, os, numpy as np
fn='GSM5401072_E9_5_raw_count.csv.gz'
start=time.time()
with gzip.open(fn,'rt') as f:
    header=f.readline().rstrip('\\n').split(',')
    ncell=len(header)-1
    totals=np.zeros(ncell,dtype=np.int64)
    genes=[]
    nnz=0
    for r,line in enumerate(f,1):
        parts=line.rstrip('\\n').split(',')
        genes.append(parts[0])
        vals=np.fromiter((int(x) for x in parts[1:]), dtype=np.int32, count=ncell)
        totals += vals
        nnz += np.count_nonzero(vals)
        if r<=5: print('gene', r, parts[0], 'firstvals', vals[:5].tolist())
print('rows', len(genes), 'cells', ncell, 'nnz', nnz, 'time', time.time()-start)
print('totals min med mean max', totals.min(), np.median(totals), totals.mean(), totals.max())
for th in [0,1,10,50,100,150,200,250,274,300,400,500,1000]:
    print('>=',th, int((totals>=th).sum()))
print('first kept >=274', [header[i+1] for i in np.where(totals>=274)[0][:10]])
print('last kept >=274', [header[i+1] for i in np.where(totals>=274)[0][-10:]])
PY"
