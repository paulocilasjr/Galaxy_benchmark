/bin/bash -lc "python3 - <<'PY'
import csv, urllib.request, concurrent.futures, re, socket
socket.setdefaulttimeout(8)
rows=list(csv.DictReader(open('geo_candidates_mouse_10x_all.tsv'), delimiter='\\t'))

def filelist_url(acc):
    n=int(acc[3:]); return f'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE{n//1000:03d}nnn/{acc}/suppl/filelist.txt'

def fetch(r):
    acc=r['accession']
    try:
        txt=urllib.request.urlopen(filelist_url(acc), timeout=8).read(300000).decode('utf-8','ignore')
    except Exception: return []
    res=[]
    for line in txt.splitlines():
        if ('genes.tsv.gz' in line or 'features.tsv.gz' in line):
            parts=line.split('\\t')
            if len(parts)>=5:
                try: size=int(parts[3])
                except: continue
                if 220000 <= size <= 225000 or 210000 <= size <= 216000:
                    res.append((acc,r['title'],r['taxon'],r['pdat'],parts[1],size,line))
    return res
hits=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:
    futs=[ex.submit(fetch,r) for r in rows]
    for i,f in enumerate(concurrent.futures.as_completed(futs),1):
        h=f.result(); hits.extend(h)
        if i%1000==0: print('done',i,'hits',len(hits), flush=True)
print('total',len(hits))
with open('broad_gene_size_hits.tsv','w') as out:
    for h in hits: out.write('\\t'.join(map(str,h))+'\\n')
for h in hits[:80]: print('\\t'.join(map(str,h[:6])))
PY"
