/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, xml.etree.ElementTree as ET
terms=[
 'otic 10x Genomics Mus musculus',
 'inner ear 10x Genomics Mus musculus single-cell',
 'cochlea 10x Genomics Mus musculus single-cell',
 'vestibular ganglion 10x Genomics Mus musculus',
 'spiral ganglion 10x Genomics Mus musculus',
 'Pax2 Foxg1 Ai9 10x single-cell',
 'Shox2 Myt1 Casz1 Sall3',
]
for t in terms:
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gds&retmax=20&term='+urllib.parse.quote(t)
    data=urllib.request.urlopen(url).read()
    root=ET.fromstring(data)
    ids=[x.text for x in root.findall('.//Id')]
    print('\\nTERM',t,'count',root.findtext('Count'),'ids',ids[:10])
    if ids:
        url2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=gds&id='+','.join(ids[:10])
        r=ET.fromstring(urllib.request.urlopen(url2).read())
        for doc in r.findall('DocSum'):
            acc=doc.find(\"Item[@Name='Accession']\").text
            title=doc.find(\"Item[@Name='title']\").text
            gse=doc.find(\"Item[@Name='GSE']\").text if doc.find(\"Item[@Name='GSE']\") is not None else ''
            typ=doc.find(\"Item[@Name='entryType']\").text
            tax=doc.find(\"Item[@Name='taxon']\").text
            print(acc,typ,'GSE'+gse,tax,title[:120])
PY"
