/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Ntrk1 Ntrk2 Ntrk3 Trpv1 Scn10a Scn11a Scn9a Prph Pou4f1 Pou4f2 Isl1 Isl2 Tac1 Calca Calcb Mrgprd Nefh Ngfr Sox10 Plp1 Mpz Krt8 Krt18 Sox2 Pax3 Pax7 Tlx3 Lmx1b Phox2b Chat Slc17a6 Slc17a7 Slc32a1'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]; codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
    sums=np.asarray(X.sum(axis=0)).ravel(); cells=np.asarray((X>0).sum(axis=0)).ravel()
    for m in markers:
        idx=np.where(genes==m)[0]
        if len(idx): print(m, int(sums[idx[0]]), int(cells[idx[0]]))
PY"
