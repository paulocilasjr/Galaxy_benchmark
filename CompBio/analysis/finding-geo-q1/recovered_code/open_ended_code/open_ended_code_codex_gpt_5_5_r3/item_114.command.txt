/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
markers='Neurog1 Neurod1 Neurod4 Neurod6 Ntrk2 Ntrk3 Bdnf Gata3 Gata2 Gfi1 Fgfr3 Sox10 Plp1 Mpz Pou4f1 Pou4f2 Pou4f3 Isl1 Isl2 Tlx3 Runx1 Runx3 Brn3a Brn3b Phox2a Phox2b Ret Hand2 Dbh Th Chat Slc17a6 Slc17a7 Slc17a8 Slc6a5 Gad1 Gad2 Ebf1 Ebf2 Ebf3 Tfap2a Tfap2b'.split()
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]; codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
    gmap={g:i for i,g in enumerate(genes)}
    sums=np.asarray(X.sum(axis=0)).ravel(); nz=np.asarray((X>0).sum(axis=0)).ravel()
    for g in markers:
        if g in gmap:
            i=gmap[g]; print(f'{g}\\t{int(sums[i])}\\t{int(nz[i])}')
PY"
