/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np, re
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=tuple(f['X'].attrs['shape']))
    cats=f['var/gene_symbols/categories'][:]; codes=f['var/gene_symbols/codes'][:]
    genes=np.array([(cats[i].decode() if i>=0 else '') for i in codes])
    sums=np.asarray(X.sum(axis=0)).ravel(); cells=np.asarray((X>0).sum(axis=0)).ravel()
for pat in ['"'^Hox'"','"'^Fox'"','"'^Pou'"','"'^Ntrk'"','"'^Scn'"','"'^Krt'"','"'^Col'"','"'^Sox'"','"'^Pax'"','"'^Tfap'"','"'^Mafb'"','"'^Onecut'"','"'^Eya'"','"'^Six'"','"'^Isl'"','"'^Lhx'"']:
    idx=[i for i,g in enumerate(genes) if re.match(pat,g) and sums[i]>0]
    idx=sorted(idx,key=lambda i:-sums[i])[:20]
    print('\\n',pat)
    for i in idx: print(genes[i], int(sums[i]), int(cells[i]))
PY"
