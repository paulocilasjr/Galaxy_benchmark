/bin/bash -lc 'for fn in GSM9497652_exp1_Otic_filtered_feature_bc_matrix.h5 GSM9497656_exp2_otic1_filtered_feature_bc_matrix.h5 GSM9497657_exp2_otic2_filtered_feature_bc_matrix.h5; do gsm=${fn%%_*}; curl -L "https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM9497nnn/${gsm}/suppl/${fn}" -o "$fn"; echo "$fn"; python3 - <<PY
import h5py
fn='"'"'$fn'"'
with h5py.File(fn,'r') as f:
    def keys(g): return list(f[g].keys()) if g in f else []
    print('root', list(f.keys()))
    if 'matrix' in f:
        m=f['matrix']; print('shape', m['shape'][:], 'barcodes', m['barcodes'].shape, 'features', m['features/id'].shape)
        def dec(x): return x.decode() if isinstance(x,bytes) else str(x)
        print('firstbc', [dec(x) for x in m['barcodes'][:5]])
        print('lastbc', [dec(x) for x in m['barcodes'][-5:]])
        print('firstfeatures', [(dec(m['features/id'][i]), dec(m['features/name'][i])) for i in range(5)])
PY
done"
