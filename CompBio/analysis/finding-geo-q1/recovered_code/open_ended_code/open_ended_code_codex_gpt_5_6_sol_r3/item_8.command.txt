/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
 d=f['X/data'][:]; ix=f['X/indices'][:]; ip=f['X/indptr'][:]
 ens=f['var/_index'][:].astype(str)
 cats=f['var/gene_symbols/categories'][:].astype(str); codes=f['var/gene_symbols/codes'][:]
 syms=cats[codes]
 X=csr_matrix((d,ix,ip),shape=(len(ip)-1,len(ens)))
 sums=np.asarray(X.sum(0)).ravel(); cells=np.asarray((X>0).sum(0)).ravel()
 for j in np.argsort(sums)[-100:][::-1]: print(f'{syms[j]:20} {ens[j]} counts={sums[j]} cells={cells[j]}')
PY"
