/bin/bash -lc "python - <<'PY'
import urllib.request
for acc in ['GSE114176','GSE114172','GSE114173','GSE114174','GSE114175']:
 u=f'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc={acc}&targ=self&form=text&view=quick'
 try:
  d=urllib.request.urlopen(u,timeout=30).read().decode(errors='replace')
  open(acc+'.txt','w').write(d)
  print(acc,len(d),d[:100].replace('\\n',' '))
 except Exception as e: print(e)
PY"
