/bin/bash -lc "python - <<'PY'
import h5py,numpy as np
from scipy.sparse import csr_matrix
f=h5py.File('finding.geo.q1.h5ad');x=csr_matrix((f['X/data'][:],f['X/indices'][:],f['X/indptr'][:]),shape=f['X'].attrs['shape'])
g=f['var/gene_symbols']; genes=g['categories'][:].astype(str)[g['codes'][:]]
sums=np.asarray(x.sum(0)).ravel();cells=np.asarray(x.sum(1)).ravel()
print('counts quantiles',np.quantile(cells,[0,.1,.25,.5,.75,.9,.95,.99,1]))
print('genes quantiles',np.quantile(np.diff(x.indptr),[0,.1,.5,.9,1]))
for i in np.argsort(sums)[-100:][::-1]:print(genes[i],sums[i],x[:,i].getnnz())
np.savez('summary.npz',genes=genes,sums=sums,cells=cells)
PY"
