import h5py, numpy as np
from collections import Counter
src='GSM3578982.h5'; tar='finding.geo.q1.h5ad'
with h5py.File(src,'r') as f:
 g=f['output_genome2']
 b=[x.decode() for x in g['barcodes'][:]]
 gid=[x.decode() for x in g['genes'][:]]
 gn=[x.decode() for x in g['gene_names'][:]]
 print('source shape',g['shape'][:], 'barcodes first/last',b[:5],b[-5:],'unique',len(set(b)))
 print('genes first',list(zip(gid[:15],gn[:15])))
 print('genes last',list(zip(gid[-15:],gn[-15:])))
 with h5py.File(tar,'r') as f2:
  obs=[x.decode() for x in f2['obs']['_index'][:]]
  vi=[x.decode() for x in f2['var']['_index'][:]]
  # symbols target
  sy=f2['var']['gene_symbols']['categories'][:]
  codes=f2['var']['gene_symbols']['codes'][:]
  vs=[sy[c].decode() if c>=0 else '' for c in codes]
  print('target obs first/last',obs[:5],obs[-5:],'unique',len(set(obs)))

  print('gene exact order first?', vi==gid[:len(vi)])

  print('mapped all',all(x in mi for x in vi),'identity map first 20',[mi[x] for x in vi[:20]],'last',[mi[x] for x in vi[-20:]])
