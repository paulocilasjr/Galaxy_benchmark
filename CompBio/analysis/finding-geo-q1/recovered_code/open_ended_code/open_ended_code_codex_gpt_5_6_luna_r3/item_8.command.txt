/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
from scipy.sparse import csr_matrix
with h5py.File('finding.geo.q1.h5ad','r') as f:
    X=csr_matrix((f['X/data'][:], f['X/indices'][:], f['X/indptr'][:]), shape=tuple(f['X'].attrs['shape']))
    genes=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/gene_symbols/categories'][:]], dtype=object)
    codes=f['var/gene_symbols/codes'][:]
    genes=genes[codes]
    cells=np.array([x.decode() if isinstance(x,bytes) else x for x in f['obs/_index'][:]], dtype=object)
print('shape',X.shape,'nnz',X.nnz,'dtype',X.dtype,'max',X.data.max(),'min',X.data.min(),'sum',X.data.sum())
tot=np.asarray(X.sum(axis=1)).ravel(); nfeat=np.diff(X.indptr)
print('tot quantiles',np.quantile(tot,[0,.001,.01,.05,.1,.25,.5,.75,.9,.95,.99,.999,1]))
print('nfeat quantiles',np.quantile(nfeat,[0,.001,.01,.05,.1,.25,.5,.75,.9,.95,.99,.999,1]))
for thr in [0,100,200,500,1000,1500,2000,2500,3000,4000,5000,10000]: print(thr,(tot<thr).sum(),(tot>=thr).sum())
for prefix in ['mt-','Rpl','Rps','Gm','Mki67','Sox2','Pdgfra','Pcp4','Snap25','Foxp2','Cux2','Gad1','Gad2','Slc17a7','Aldoc','Aqp4','Olig1','Olig2','Itgam','Cx3cr1','P2ry12','C1qa','C1qb','Cldn5','Kdr','Pecam1','Col1a1','Dcn','Actb','Gapdh']:
 idx=np.flatnonzero(np.char.startswith(genes.astype(str),prefix))
 if len(idx): print(prefix, [(int(i),genes[i]) for i in idx[:5]], 'total',int(X[:,idx].sum()),'cells',int((X[:,idx].sum(axis=1)>0).sum()))
print('first cells totals',list(zip(cells[:20],tot[:20],nfeat[:20])))
print('last cells totals',list(zip(cells[-20:],tot[-20:],nfeat[-20:])))
PY"
