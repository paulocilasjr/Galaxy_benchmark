/bin/bash -lc "python - <<'PY'
import subprocess
samples=[('GSM3934450','GSM3934450_ET_CT_12'),('GSM3934451','GSM3934451_ET_CT_15'),('GSM3934452','GSM3934452_PT_CT_0'),('GSM3934453','GSM3934453_C57BL6-P3-cortex'),('GSM3934454','GSM3934454_C57BL6-P6-cortex'),('GSM3934455','GSM3934455_ET_PO_12')]
for gsm,pref in samples:
 base=f'https://ftp.ncbi.nlm.nih.gov/geo/samples/GSM3934nnn/{gsm}/suppl/'
 for typ in ['genes','barcodes','matrix']:
  fn=pref+'_'+typ+'.mtx.gz' if typ=='matrix' else pref+'_'+typ+'.tsv.gz'
  head=subprocess.check_output(['curl','-L','-sS','-I',base+fn],text=True)
  size=next((x.split(':',1)[1].strip() for x in head.splitlines() if x.lower().startswith('content-length:')),'?')
  if typ in ['genes','barcodes']:
   n=subprocess.check_output(\"curl -L --fail --silent '\"+base+fn+\"' | gzip -cd | wc -l\",shell=True,text=True).strip()
  else: n=''
  print(gsm,typ,'size',size,'n',n)
PY"
