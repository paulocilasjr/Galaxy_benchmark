import h5py
with h5py.File('finding.geo.q1.h5ad','r') as f:
    for x in f['obs/_index'][:]:
        print(x.decode() if isinstance(x,bytes) else x)
PY
bc <- scan(file='stdin', what=character(), quiet=TRUE)
cat('finding cells',length(bc),'\\n')
files <- list.files('gse188528_raw', pattern='filtered_RNA_counts.RDS.gz"'$'"', full.names=TRUE)
for (f in files) {
 z <- sub('\\\\.gz"'$'"','',f); system2('gzip',c('-cd',f),stdout=z); x <- readRDS(z); unlink(z)
 cn <- sub('"'^[''^_]+_'"','',colnames(x))
 cn <- paste0(cn,'-1')
 ov <- intersect(bc,cn)
 if (length(ov)>0) cat('MATCH',basename(f),'overlap',length(ov),'of',length(cn),'example',ov[1],'\\n')
}
RS"
