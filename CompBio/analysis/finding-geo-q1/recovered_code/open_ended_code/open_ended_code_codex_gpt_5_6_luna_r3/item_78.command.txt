/bin/bash -lc "python - <<'PY'
import json, subprocess, time, sys
ids=json.load(open('geo_ids2.json'))['esearchresult']['idlist']
allres={}
for j in range(0,len(ids),100):
    s=','.join(ids[j:j+100])
    out=subprocess.check_output(['curl','-L','--silent','--show-error','-X','POST','https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi','--data-urlencode','db=gds','--data-urlencode','id='+s,'--data-urlencode','retmode=json'])
    try: z=json.loads(out)
    except Exception: print('bad json',j,out[:500]); time.sleep(2); continue
    if 'result' not in z: print('no result',j,z); time.sleep(2); continue
    allres.update({k:v for k,v in z['result'].items() if k"'!='"'uids'})
    print(j+len(ids[j:j+100]),file=sys.stderr)
    time.sleep(.34)
json.dump({'result':allres},open('geo_summary.json','w'))
print('summaries',len(allres))
PY
python - <<'PY'
import json
x=json.load(open('geo_summary.json'))['result']; rows=[]

 acc=v.get('accession','')
 if acc.startswith('GSE'): rows.append((acc,v.get('title',''),v.get('entrytype',''),v.get('n_samples','')))
print('GSE',len(rows))
for r in rows: print('\\t'.join(map(str,r)))
PY"
