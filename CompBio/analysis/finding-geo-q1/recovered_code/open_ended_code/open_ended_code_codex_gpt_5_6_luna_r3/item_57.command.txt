/bin/bash -lc "mkdir -p gse188528_raw && tar -xf GSE188528_RAW.tar -C gse188528_raw; ls -lh gse188528_raw | head -10; Rscript - <<'RS'
files <- list.files('gse188528_raw', pattern='filtered_RNA_counts.RDS.gz"'$'"', full.names=TRUE)
for (f in files) {
  z <- sub('\\\\.gz"'$'"','',f)
  system2('gzip',c('-cd',f),stdout=z)
  x <- readRDS(z)
  cat(basename(f), 'dim=',paste(x@Dim,collapse='x'),'nnz=',length(x@x),'sum=',sum(x@x),'range=',min(x@x),max(x@x),'\\n')
  unlink(z)
}
RS"
