/bin/bash -lc "python3 - <<'PY'
import requests,json,urllib.parse,time
base='https://zenodo.org/api/records'
queries=['rheumatoid factor GWAS summary statistics','immunoglobulin levels GWAS summary statistics','IgG levels GWAS summary statistics','IgA IgG IgM GWAS summary statistics','rheumatoid factor UK Biobank GWAS']
for q in queries:
 r=requests.get(base,params={'q':q,'size':10},timeout=30)
 js=r.json()
 print('\\nQUERY',q,'total',js.get('hits',{}).get('total'))
 for hit in js.get('hits',{}).get('hits',[])[:10]:
  print(hit.get('id'), hit.get('metadata',{}).get('title'), hit.get('metadata',{}).get('publication_date'))
 time.sleep(0.3)
PY"
