/bin/bash -lc "python3 - <<'PY'
import requests, json, time
variants=[('1',161530617,'CTT','C'),('1',161531006,'C','G'),('6',32067917,'C','T'),('17',16939677,'G','A'),('19',49525049,'CCTGCTG','C'),('4',3445429,'C','T'),('4',102486185,'C','T'),('2',111471251,'T','C'),('13',108308037,'T','A'),('2',27508073,'T','C'),('6',33546790,'T','A')]
for chrom,pos,ref,alt in variants:
    url=f'https://myvariant.info/v1/variant?q=hg38.chr:{chrom} AND hg38.start:{pos} AND hg38.end:{pos}'
    try:
        r=requests.get(url,timeout=30)
        js=r.json()
        hits=js.get('hits',[])
        print('\\n',chrom,pos,ref,alt,'hits',js.get('total'))
        for h in hits[:5]:
            print(' rsid',h.get('_id'), 'dbsnp', h.get('dbSNP',{}).get('rsid') if isinstance(h.get('dbSNP'),dict) else h.get('dbSNP'), 'clinvar?')
    except Exception as e: print('err',e)
PY"
