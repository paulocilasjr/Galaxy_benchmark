import requests, json
variants=['chr1:161530617:CTT:C','chr1:161531006:C:G','chr6:32067917:C:T','chr1:161540726:C:G','chr6:32092090:A:G']

    r=requests.get(url, params={'fields':'dbsnp.rsid,snpedia,clinvar,gwas'})
    print(v, r.status_code)
    print(json.dumps(r.json(), indent=2)[:1000])
PY"
