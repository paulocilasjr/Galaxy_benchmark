/bin/bash -lc "python3 - <<'PY'
import json
s=json.load(open('[REDACTED LOCAL PATH]'))
terms=['IgG','immunoglobulin','glycosylation','Fc','CD16','CD32','CD64','TACI','TNFRSF13B','B cell','B-cell','plasma','antibody','rheumatoid factor','IgA','IgM','IgE','Fc gamma','FCGR']
# search trait field only
matches=[]
for id,v in s['datasets_compressed'].items():
    arr=v[0]; meta=v[1] if len(v)>1 else {}
    trait=arr[0]
    if any(t.lower() in trait.lower() for t in terms):
        matches.append((id,arr[0],arr[6],arr[7],arr[10],meta.get('nsnp'),meta.get('pmid'),meta.get('ncase'),meta.get('ncontrol')))
print('matches',len(matches))
for m in matches[:500]:
    print(m)
PY"
