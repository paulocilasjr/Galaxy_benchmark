/bin/bash -c "python3 - <<'PY'
import urllib.request,json,time
url='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByFullPvalueSet?fullPvalueSet=true&size=100&page=0'
seen=set(); found=[]; page=0
while True:
 u=f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByFullPvalueSet?fullPvalueSet=true&size=100&page={page}'
 try:
  d=json.load(urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'}),timeout=60))
 except Exception as e:
  print('ERR page',page,e); break
 studies=d.get('_embedded',{}).get('studies',[])
 if not studies: break
 for st in studies:
  if st['accessionId'] in seen: continue
  seen.add(st['accessionId'])
  n=st.get('snpCount')
  if n and abs(n-12774703)<1000000:
   pi=st.get('publicationInfo') or {}
   found.append((st['accessionId'], n, st['diseaseTrait']['trait'], pi.get('pubmedId'), st.get('fullPvalueSet')))
 page+=1
 total=d['page']['totalElements']
 print('page',page,'total seen',len(seen),'of',total,'found',len(found))
 if page*100>=total: break
 if page>200: break
print('FOUND')
for x in sorted(found,key=lambda x:abs(x[1]-12774703))[:100]: print(x)
PY"
