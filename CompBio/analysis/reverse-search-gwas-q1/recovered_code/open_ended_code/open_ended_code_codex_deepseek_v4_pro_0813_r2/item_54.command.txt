/bin/bash -lc "python3 - <<'PY'
import requests, json
leads=[('chr1:161530617:CTT:C','deletion'),('chr17:16939677:G:A','snv'),('chr19:49525049:CCTGCTG:C','del?'),('chr4:3445429:C:T','snv'),('chr6:29686923:C:T','snv'),('chr6:30355154:GT:G','del'),('chr6:30931873:G:A','snv'),('chr6:31481775:T:C','snv'),('chr6:32067917:C:T','snv'),('chr6:32668221:G:T','snv')]
# Build HGVS rough for SNVs: chr:g.POSREF>ALT. indels omitted maybe
for vid,typ in leads:
    chrom,pos,ref,alt=vid.split(':')
    if typ=='snv':
        hgvs=f'{chrom}:g.{pos}{ref}>{alt}'
    elif typ=='deletion':
        # ref CTT alt C, deletion positions pos+1 to pos+len(ref)-1
        start=int(pos)+1; end=int(pos)+len(ref)-1
        if end==start:
            hgvs=f'{chrom}:g.{start}del'
        else:
            hgvs=f'{chrom}:g.{start}_{end}del'
    else:
        hgvs=None
    print('\\n',vid,hgvs)
    if not hgvs: continue
    r=requests.get('https://myvariant.info/v1/variant/'+hgvs, params={'assembly':'hg38','fields':'dbsnp.rsid,cadd.gene.genename,cadd.gene.gene_id,dbnsfp.genename,dbnsfp.ensembl.geneid,gwas'})
    print(r.status_code, json.dumps(r.json())[:2000])
PY"
