/bin/bash -lc "python3 - <<'PY'
import requests, json, time
rsids=['rs72717048','rs144126567','rs1150756','rs1150753']
session=requests.Session()
for rs in rsids:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations'
    r=session.get(url,headers={'Accept':'application/json'},params={'size':100})
    if not r.ok:
        print(rs,'err',r.status_code); continue
    assoc=r.json()['_embedded']['associations']
    print('\\nRS',rs,'n',len(assoc))
    for a in assoc[:20]:
        p=a.get('pvalue')
        or_=a.get('orPerCopyNum')
        beta=a.get('betaNum')
        study_url=a['_links']['study']['href']
        trait_url=a['_links']['efoTraits']['href']
        sr=session.get(study_url,headers={'Accept':'application/json'})
        tr=session.get(trait_url,headers={'Accept':'application/json'})
        study=sr.json() if sr.ok else {}
        traits=[]
        if tr.ok:
            try:
                emb=tr.json().get('_embedded',{})
                # may be efoTraits list
                if 'efoTraits' in emb:
                    traits=[t.get('trait') for t in emb['efoTraits']]
                elif 'traits' in emb:
                    traits=[t.get('trait') for t in emb['traits']]
            except Exception as e: pass
        print(f\"  p={p} beta={beta} or={or_} study={study.get('accessionId')} pmid={study.get('pubmedId')} title={study.get('title')} traits={traits}\")
        time.sleep(.1)
PY"
