for chrom,pos,radius in [('1',161530617,200000),('19',49525049,100000),('4',3445429,100000),('6',29686923,200000),('6',30355154,200000)]:

                print(x.get('external_name'),x.get('seq_region_name'),x.get('start'),x.get('end'))
    else: print(r.text[:100])
