/bin/bash -lc "cd [REDACTED LOCAL PATH] && python3 - <<'PY'
import csv,re
p='gwas-catalog-download-studies-v1.0.3.1.txt'
with open(p) as f: rows=list(csv.DictReader(f,delimiter='\\t'))
terms=re.compile(r'("'^|[''^A-Za-z])(IgA|IgG|IgM|immunoglob'"\\w*|rheumatoid factor|autoantibod\\w*|antibod\\w*|B-?cell|B cell|plasma cell|humoral|multiple myeloma|MGUS|monoclonal gammopathy|free light chain|glycoprotein|N-glycosyl|glycosylation|glycan|sialyl|fucosyl)',re.I)
c=[]
for r in rows:
 if r.get('FULL SUMMARY STATISTICS','').strip().lower()=='yes':
  trait=(r.get('DISEASE/TRAIT') or '')
  mapped=(r.get('MAPPED_TRAIT') or '')
  study=(r.get('STUDY') or '')
  s=trait+' '+mapped+' '+study
  if terms.search(s):
   c.append(r)
print('candidates',len(c))
with open('candidate_immune.tsv','w',newline='') as out:
 w=csv.DictWriter(out,fieldnames=list(c[0].keys()),delimiter='\\t')
 w.writeheader(); w.writerows(c)
for r in c[:100]:
 print(r.get('STUDY ACCESSION'), r.get('PUBMED ID'), r.get('DISEASE/TRAIT'), '|', r.get('INITIAL SAMPLE SIZE'))
PY"
