/bin/bash -lc "python3 - <<'PY'
import requests,json,urllib.parse,time
base='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByDiseaseTrait?diseaseTrait={}'
for term in ['immunoglobulin','rheumatoid factor','antibody','IgG','IgA','IgM','glycan']:
 url=base.format(urllib.parse.quote(term))
 try:
  r=requests.get(url,timeout=60)
  js=r.json()
  studies=js.get('_embedded',{}).get('studies',[])
  print('\\nTERM',term,'total',js.get('page',{}).get('totalElements'),'got',len(studies))
  for s in studies[:50]:
   print(s.get('accessionId'), s.get('publicationInfo',{}).get('pubmedId'), s.get('diseaseTrait',{}).get('trait'), '|full',s.get('fullPvalueSet'),'|snps',s.get('snpCount'))
 except Exception as e: print('ERR',term,e)
PY"
