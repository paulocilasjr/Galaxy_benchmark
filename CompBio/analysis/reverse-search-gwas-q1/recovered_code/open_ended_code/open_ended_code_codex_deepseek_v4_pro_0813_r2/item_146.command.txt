/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
ids=['1_161531006_C_G','1_161540726_C_G','17_16939677_G_A','4_3445429_C_T','6_29686923_C_T','6_31481775_T_C','6_32668221_G_T']
for vid in ids:
 q={'query':f'''query {{ variant(variantId: \"{vid}\") {{ id rsIds credibleSets {{ rows {{ pValueMantissa pValueExponent studyId studyType study {{ traitFromSource pubmedId publicationFirstAuthor }} }} }} }} }}'''}
 r=requests.post(url,json=q)
 data=r.json().get('data',{}).get('variant')
 print('\\n',vid, data and data.get('rsIds'))
 if not data: continue
 rows=data['credibleSets']['rows']
 # filter gwas, sort p
 gw=[]
 for row in rows:
  if row.get('studyType')=='gwas' and row.get('pValueMantissa') is not None:
   p=row['pValueMantissa']*10**row['pValueExponent']
   study=row.get('study') or {}
   gw.append((p,row['studyId'],study.get('traitFromSource'),study.get('pubmedId'),study.get('publicationFirstAuthor')))
 gw.sort()
 for x in gw[:15]: print(' ',x)
PY"
