/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
q='''query { variant(variantId:\"1_161530617_CTT_C\") { credibleSets { count rows { studyId studyType pValueMantissa pValueExponent } } } }'''
r=requests.post(url,json={'query':q},timeout=30); d=r.json()
rows=d['data']['variant']['credibleSets']['rows']
ids=[]
for row in rows:
 if row['studyType']=='gwas': ids.append(row['studyId'])
print('unique',sorted(set(ids)))
for sid in sorted(set(ids)):
    rr=requests.get(f'https://www.ebi.ac.uk/gwas/rest/api/studies/{sid}',timeout=30)
    if rr.ok:
        s=rr.json()
        print(sid, s.get('snpCount'), s.get('fullPvalueSet'), s.get('diseaseTrait',{}).get('trait'), s.get('publicationInfo',{}).get('pubmedId'), s.get('initialSampleSize'))
    else:
        print(sid,'err',rr.status_code)
PY"
