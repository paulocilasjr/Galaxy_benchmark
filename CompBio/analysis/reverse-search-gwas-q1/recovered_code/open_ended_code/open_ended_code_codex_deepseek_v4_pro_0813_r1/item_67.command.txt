/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
q='''{ __type(name:\"Variant\") { fields { name description type { name kind ofType { name kind ofType { name kind } } } } } }'''
r=requests.post(url,json={'query':q},timeout=30)
print(r.status_code)
data=r.json()
for f in data['data']['__type']['fields']:
    print(f['name'], '=>', f['type'])
PY"
