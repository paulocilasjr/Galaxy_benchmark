/bin/bash -lc "python3 - <<'PY'
import requests,json
base='https://www.ebi.ac.uk/gwas/rest/api'
for rs in ['rs372779368']:
    r=requests.get(f'{base}/singleNucleotidePolymorphisms/{rs}/associations',params={'size':100,'projection':'associationByEfoTrait'},timeout=30)
    print(r.status_code, r.url)
    if r.ok:
      d=r.json(); assocs=d.get('_embedded',{}).get('associations',[])
      print(len(assocs))
      for a in assocs:
        st=a.get('study',{})
        print(a.get('pvalue'), st.get('accessionId'), st.get('diseaseTrait',{}).get('trait'), '|', st.get('publicationInfo',{}).get('pubmedId'), '|', st.get('initialSampleSize'))
PY"
