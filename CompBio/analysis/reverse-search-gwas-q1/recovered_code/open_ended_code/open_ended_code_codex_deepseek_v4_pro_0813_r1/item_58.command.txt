/bin/bash -lc "python3 - <<'PY'
import gzip
bins={}
with gzip.open('reverse.search.gwas.q1.tsv.gz','rt') as f:
    next(f)
    for line in f:
        parts=line.rstrip().split('\\t')
        if len(parts)<8: continue
        try:p=float(parts[7])
        except:continue
        if p<1e-8:
            chrom=parts[0]; pos=int(parts[1]); key=(chrom,pos//1000000)
            bins.setdefault(key,[0,None])
            if bins[key][1] is None or p<bins[key][1]: bins[key]=[bins[key][0]+1,p]
            else: bins[key][0]+=1
print('chrom','mb','count','minp')
for k in sorted(bins, key=lambda x:(int(x[0]) if x[0].isdigit() else 99,x[1])):
    print(k[0],k[1],bins[k][0],bins[k][1])
PY"
