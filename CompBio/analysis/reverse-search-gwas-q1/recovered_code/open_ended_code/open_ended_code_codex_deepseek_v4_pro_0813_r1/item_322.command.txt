/bin/bash -lc "python3 - <<'PY'
import openpyxl, glob, os
for f in sorted(glob.glob('supp/*.xlsx')):
    try:
        wb=openpyxl.load_workbook(f, read_only=True, data_only=True)
    except Exception as e:
        print(f, 'ERR', e); continue
    print('\\n===',os.path.basename(f), 'sheets', wb.sheetnames)
    for sn in wb.sheetnames:
        ws=wb[sn]
        print(' sheet',sn, 'dims',ws.max_row,ws.max_column)
        rows=0
        for row in ws.iter_rows(values_only=True):
            vals=[str(x)[:40] if x is not None else '' for x in row[:10]]
            print('  ', '\\t'.join(vals))
            rows+=1
            if rows>=3: break
    wb.close()
PY"
