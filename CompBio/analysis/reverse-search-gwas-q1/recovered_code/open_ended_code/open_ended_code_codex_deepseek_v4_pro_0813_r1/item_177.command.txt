import urllib.request, json, urllib.parse, math
base='https://www.ebi.ac.uk/gwas/rest/api/associations/search/findByPubmedId'
params={'pubmedId':'40593539','projection':'associationByStudy'}
with urllib.request.urlopen(base+'?'+urllib.parse.urlencode(params), timeout=60) as r: data=json.load(r)
ass=data['_embedded']['associations']
targets=[7.81024e-23,4.85572e-19,1.4221e-16,2e-30,7e-18]
for a in ass:
    p=a.get('pvalue')
    if p and any(abs(math.log10(p)-math.log10(t))<0.02 for t in targets):
        s=a['study']; rs=[]
        for loc in a.get('loci',[]):
          for al in loc.get('strongestRiskAlleles',[]): rs.append(al.get('riskAlleleName'))
        print('MATCH?',p,a.get('pvalueMantissa'),a.get('pvalueExponent'),s['accessionId'],s['diseaseTrait']['trait'],rs)
PY"
