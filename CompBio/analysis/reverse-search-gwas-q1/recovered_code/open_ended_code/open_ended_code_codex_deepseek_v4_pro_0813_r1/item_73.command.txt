/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
ids=['GCST90479508','GCST90475122','GCST90691895','GCST90501254','GCST90301984','GCST90474339','GCST90480702','GCST90497328','GCST90019501','GCST90137412','GCST90668505','GCST90499705','GCST90668508','GCST90474314','GCST90239658','GCST90692815']
for sid in ids:
 q='''query { study(studyId:\"%s\") { id studyType traitFromSource pubmedId publicationTitle publicationFirstAuthor publicationDate initialSampleSize nCases nControls nSamples hasSumstats summarystatsLocation } }'''%sid
 r=requests.post(url,json={'query':q},timeout=30)
 d=r.json().get('data',{}).get('study')
 if d:
  print(sid, '|', d.get('traitFromSource'), '|', d.get('pubmedId'), '|', d.get('nCases'), d.get('nControls'), d.get('nSamples'), '|', d.get('publicationTitle'))
 else:
  print(sid,'no')
PY"
