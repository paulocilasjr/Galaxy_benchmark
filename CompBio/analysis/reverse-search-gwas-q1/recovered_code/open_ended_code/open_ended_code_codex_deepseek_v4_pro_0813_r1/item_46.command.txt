/bin/bash -lc "python3 - <<'PY'
import requests,json
base='https://www.ebi.ac.uk/gwas/rest/api'
for rs in ['rs1801274','rs9273373','rs3130342','rs9272598']:
    r=requests.get(f'{base}/singleNucleotidePolymorphisms/{rs}/associations',params={'size':1000},timeout=30)
    print('---',rs,r.status_code)
    if r.ok:
        d=r.json(); assoc=d.get('_embedded',{}).get('associations',[])
        for a in assoc:
            traits=', '.join(t['trait'] for t in a.get('efoTraits',[]))
            study=a.get('study',{}).get('accessionId','')
            pmid=a.get('study',{}).get('publicationInfo',{}).get('pubmedId','')
            pv=a.get('pvalue','')
            print(a.get('rsId'), traits, study, pmid, pv, a.get('_links',{}).get('self',{}).get('href'))
PY"
