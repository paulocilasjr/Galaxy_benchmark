/bin/bash -lc "python3 - <<'PY'
import requests, json
base='https://www.ebi.ac.uk/gwas/rest/api'
for chrom,start,end in [('6',32000000,32700000),('1',161400000,161700000)]:
    url=f'{base}/singleNucleotidePolymorphisms/search/findByChromBpLocationRange'
    params={'chrom':chrom,'bpStart':start,'bpEnd':end,'size':1000}
    r=requests.get(url,params=params,timeout=30)
    print('status',r.status_code, 'url', r.url)
    data=r.json()
    em=data.get('_embedded',{}).get('singleNucleotidePolymorphisms',[])
    print(chrom,len(em))
    for x in em[:50]:
        print(x.get('rsId'), x.get('locations',[{}])[0].get('chromosomePosition'))
    print('...')
PY"
