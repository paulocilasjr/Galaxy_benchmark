/bin/bash -lc "python3 - <<'PY'
import subprocess, concurrent.futures, math, time, os
accs=[]
with open('summary_statistics_table_export.tsv') as f:
    next(f)
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)>15 and p[15]=='40593539':
            path=p[11].strip('\"').replace('http://ftp.ebi.ac.uk','https://ftp.ebi.ac.uk')
            accs.append((p[1],f'{path}/harmonised/{p[1]}.h.tsv.gz'))
print('n',len(accs),flush=True)
regions=['6:32067917-32067917','17:16939677-16939677']
target6=4.85572e-19; target17=1.4221e-16

def log10safe(x):
    try:
        v=float(x)
        if v<=0: return -320
        return math.log10(v)
    except: return -999

def query(item):
    acc,url=item
    vals={}
    for reg in regions:
        try:
            cp=subprocess.run(['tabix',url,reg],capture_output=True,text=True,timeout=90)
            for line in cp.stdout.splitlines():
                cols=line.split('\\t')
                if len(cols)>=8 and cols[1]==reg.split(':')[1]:
                    vals[reg]=cols[7]
                    break
        except Exception as e:
            vals[reg]='ERR'
    return acc,vals

results=[]
start=time.time()
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:
    futs=[ex.submit(query,item) for item in accs]
    for fut in concurrent.futures.as_completed(futs):
        results.append(fut.result())
# match
matches=[]
for acc,vals in results:
    p6=vals.get(regions[0]); p17=vals.get(regions[1])
    l6=log10safe(p6); l17=log10safe(p17)
    d6=abs(l6-log10safe(target6)); d17=abs(l17-log10safe(target17))
    if d6<0.02 and d17<0.02:
        matches.append((acc,p6,p17,d6,d17))
print('elapsed',time.time()-start,'matches',len(matches),flush=True)
for m in matches: print(m)
# print closest by combined for top 30
arr=[]
for acc,vals in results:
    p6=vals.get(regions[0]); p17=vals.get(regions[1])
    l6=log10safe(p6); l17=log10safe(p17)
    score=abs(l6-log10safe(target6))+abs(l17-log10safe(target17))
    arr.append((score,acc,p6,p17))
arr.sort()
for x in arr[:30]: print(x)
PY"
