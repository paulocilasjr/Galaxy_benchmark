/bin/bash -lc "python3 - <<'PY'
import urllib.request, xml.etree.ElementTree as ET
for pmid in ['33710309','28878392','32128391']:
 # get pmcid
 import json
 with urllib.request.urlopen(f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=pubmed&id={pmid}&retmode=json') as r:d=json.load(r)
 pmc=d['result'][pmid].get('articleids')
 pmcid=next((x['value'] for x in pmc if x['idtype']=='pmc'),None)
 print('PMID',pmid,'PMC',pmcid)
 if pmcid:
  try:
   with urllib.request.urlopen(f'https://www.ebi.ac.uk/europepmc/webservices/rest/{pmcid}/fullTextXML',timeout=60) as r: xml=r.read()
   text=' '.join(ET.fromstring(xml).itertext())
   for term in ['rs34562254','161530617','32067917','FCGR2A','FCGR3A','HGFAC','FCGRT','TNFRSF13B']:
    print(term, text.find(term))
  except Exception as e: print('ERR',e)
PY"
