/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://api.platform.opentargets.org/api/v4/graphql'
for typ in ['CredibleSets','Evidences','CredibleSet']:
 q=f'{{ __type(name:\"{typ}\") {{ fields {{ name type {{ name kind ofType {{ name kind ofType {{ name kind }} }} }} }} }} }}'
 r=requests.post(url,json={'query':q},timeout=30)
 print('\\nTYPE',typ,r.status_code)
 d=r.json()
 try:
  for f in d['data']['__type']['fields']:
   print(f['name'],'=>',f['type'])
 except Exception as e: print(r.text[:1000])
PY"
