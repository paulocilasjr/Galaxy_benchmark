/bin/bash -lc "python3 - <<'PY'
import requests,json,time,sys
base='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByFullPvalueSet'
params={'fullPvalueSet':'true','page':0,'size':1000}
allstudies=[]
while True:
    r=requests.get(base,params=params,timeout=30)
    if not r.ok:
        print('err',r.status_code,r.text[:200]); break
    d=r.json(); em=d.get('_embedded',{}).get('studies',[])
    if not em: break
    allstudies += em
    print('page',params['page'],'got',len(em),'total so far',len(allstudies))
    page=d.get('page',{})
    if page.get('number',0)+1 >= page.get('totalPages',0):
        break
    params['page']=params['page']+1
print('total studies',len(allstudies))
# filter snpCount 12,000,000-14,000,000
hits=[]
for s in allstudies:
    snp=s.get('snpCount')
    if isinstance(snp,int) and 12000000 <= snp <= 14000000:
        hits.append((snp,s.get('accessionId'),s.get('diseaseTrait',{}).get('trait'),s.get('publicationInfo',{}).get('pubmedId'),s.get('initialSampleSize')))
print('hits',len(hits))
for h in sorted(hits):
    print(h)
PY"
