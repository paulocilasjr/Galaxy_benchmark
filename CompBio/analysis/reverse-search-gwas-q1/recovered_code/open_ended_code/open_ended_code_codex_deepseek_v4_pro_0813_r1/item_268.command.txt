/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for pmid in ['32128391','28878392','33710309','40593539']:
 print('\\nPMID',pmid)
 # studies
 try:
  with urllib.request.urlopen(f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByPublicationIdPubmedId?pubmedId={pmid}&size=1000',timeout=60) as r:d=json.load(r)
  studies=d.get('_embedded',{}).get('studies',[])
  print('studies',len(studies))
  for s in studies[:20]: print(' ',s['accessionId'],s['snpCount'],s['fullPvalueSet'],s['diseaseTrait']['trait'])
 except Exception as e: print('err studies',e)
 # associations count and top
 try:
  with urllib.request.urlopen(f'https://www.ebi.ac.uk/gwas/rest/api/associations/search/findByPubmedId?pubmedId={pmid}&projection=associationByStudy',timeout=60) as r:a=json.load(r)
  ass=a.get('_embedded',{}).get('associations',[])
  print('assoc count',len(ass))
  for x in ass[:20]:
   rs=[]
   for loc in x.get('loci',[]):
    for al in loc.get('strongestRiskAlleles',[]):rs.append(al.get('riskAlleleName'))
   print(' ',x['study']['accessionId'],x['study']['diseaseTrait']['trait'],x.get('pvalue'),rs)
 except Exception as e: print('err assoc',e)
PY"
