/bin/bash -lc "python3 - <<'PY'
import csv,re
path='gwas_studies_new.tsv'
rows=[]
with open(path, encoding='utf-8-sig', newline='') as f:
    r=csv.reader(f,delimiter='\\t')
    header=next(r)
    for row in r:
        if len(row)<11: continue
        pmid=row[1]; study=row[6]; trait=row[7]; initial=row[8]; platform=row[10]
        # extract numbers from platform
        nums=re.findall(r'[\\d,]{4,}', platform)
        vals=[]
        for n in nums:
            try: vals.append(int(n.replace(',','')))
            except: pass
        snp=max(vals) if vals else None
        if snp and 12000000 <= snp <= 14000000:
            rows.append((snp,study,trait,pmid,initial,platform))
print('hits',len(rows))
for x in sorted(rows):
    print(x)
PY"
