/bin/bash -lc "python - <<'PY'
import urllib.request,json
u='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main/data?expand=true&recursive=true'
all=[]
while u:
 req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
 resp=urllib.request.urlopen(req,timeout=120)
 x=json.load(resp); all.extend(x)
 links=resp.headers.get('Link','')
 nxt=None
 for part in links.split(','):
  if 'rel=\"next\"' in part:
   nxt=part.split(';')[0].strip()[1:-1]
 u=nxt
 print('page',len(x),'total',len(all),'next',bool(u))
for a in all:
 if 'reverse' in a.get('path','') or 'gwas' in a.get('path',''):
  print(json.dumps(a,sort_keys=True))
PY"
