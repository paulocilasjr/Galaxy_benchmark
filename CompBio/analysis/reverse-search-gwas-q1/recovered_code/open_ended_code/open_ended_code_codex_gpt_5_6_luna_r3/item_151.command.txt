/bin/bash -lc "python - <<'PY'
import urllib.request,io,re
u='https://europepmc.org/api/getPdf?pmcid=PMC3484648'
d=urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'curl/8.0'}),timeout=120).read();print('bytes',len(d))
for mod in ['pypdf','PyPDF2','pdfplumber']:
 try:
  m=__import__(mod);print('module',mod)
  if mod=='pypdf': R=m.PdfReader(io.BytesIO(d))
  elif mod=='PyPDF2':R=m.PdfReader(io.BytesIO(d))
  else:R=None
  if R:
   s='\\n'.join(p.extract_text() or '' for p in R.pages);print('textlen',len(s))
   for term in ['PMID','total protein','TNFRSF13B','FCGRT','rs4985726','rs34562254','Summary','Data availability']:
    print('\\nTERM',term)
    for x in list(re.finditer(term,s,re.I))[:5]:print(x.start(),s[x.start()-150:x.start()+600].replace('\\n',' '))
  break
 except Exception as e:print('err',mod,e)
PY"
