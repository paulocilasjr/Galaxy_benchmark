/bin/bash -lc "python - <<'PY'
import urllib.request,gzip,io,heapq,collections
name='reverse.search.gwas.q2.gz'
u='https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/data/'+name+'?download=true'
raw=urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'}),timeout=300).read()
print('compressed',len(raw))
f=gzip.GzipFile(fileobj=io.BytesIO(raw))
header=f.readline().decode().strip().split('\\t'); print(header)
# keep smallest p
best=[]; n=0; ch=collections.Counter();
for b in f:
 a=b.decode().rstrip().split('\\t');
 if len(a)"'!=len(header): continue
 n+=1; ch[a[2]]+=1
 try:p=float(a[-1])
 except:continue
 if len(best)<30:heapq.heappush(best,(-p,a))
 elif p < -best[0][0]:heapq.heapreplace(best,(-p,a))
print('"'n',n,'chrom',ch)
for _,a in sorted(best,key=lambda z:float(z[1][-1])):print('\\t'.join(a))
PY"
