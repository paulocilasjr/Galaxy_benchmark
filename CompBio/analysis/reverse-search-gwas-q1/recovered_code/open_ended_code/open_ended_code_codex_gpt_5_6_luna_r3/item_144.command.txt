/bin/bash -lc "python - <<'PY'
import urllib.request,json
for rs in ['rs10007186','rs1260326','rs4985726','rs34562254','rs3803800','rs11552708','rs4792800','rs3751987']:
 urls=['https://rest.ensembl.org/variation/human/'+rs+'?content-type=application/json','https://api.ncbi.nlm.nih.gov/variation/v0/beta/refsnp/'+rs[2:]]
 for base in urls:
  try:
   req=urllib.request.Request(base,headers={'User-Agent':'curl/8.0','Accept':'application/json'})
   x=json.load(urllib.request.urlopen(req,timeout=60));
   print('\\n',rs,base.split('/')[2]);
   if 'ensembl' in base: print([(m.get('seq_region_name'),m.get('start'),m.get('allele_string'),m.get('assembly_name')) for m in x.get('mappings',[])])
   else: print([(m.get('seq_id'),m.get('start'),m.get('allele'),m.get('placement_annot',{}).get('seq_type')) for p in x.get('primary_snapshot_data',{}).get('placements_with_allele',[]) for m in p.get('alleles',[])[:1]])
   break
  except Exception as e: print(rs,type(e).__name__,e)
PY"
