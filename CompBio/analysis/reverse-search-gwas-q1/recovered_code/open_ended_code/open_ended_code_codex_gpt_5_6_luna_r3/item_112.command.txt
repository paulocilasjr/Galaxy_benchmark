/bin/bash -lc "python - <<'PY'
import urllib.request,json
for rec in ['17370173','11111186']:
 print('\\nREC',rec)
 u='https://zenodo.org/api/records/'+rec
 req=urllib.request.Request(u,headers={'User-Agent':'curl/8.0'})
 x=json.load(urllib.request.urlopen(req,timeout=60))
 print(x.get('title'),x.get('metadata',{}).get('publication_date'))
 for f in x.get('files',[]):
  print(f.get('key'),f.get('size'),f.get('checksum'),f.get('links'))
PY"
