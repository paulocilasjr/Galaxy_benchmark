/bin/bash -lc "python - <<'PY'
import urllib.request,re
u='https://www.ebi.ac.uk/europepmc/webservices/rest/PMC3484648/fullTextXML'
s=urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'curl/8.0'}),timeout=60).read().decode()
print('len',len(s))
# metadata/pmed
for term in ['PMID','PubMed','total protein','TNFRSF13B','FCGRT','GWAS','Data Availability','supplementary-material']:
 print('\\n',term)
 for m in list(re.finditer(term,s,re.I))[:8]:print(m.start(),re.sub('<["'^>]+>'"',' ',s[m.start()-100:m.start()+500]))
# supp
for m in re.finditer(r'<supplementary-material["'^>]*?id="([''^"]+)".*?</supplementary-material>'"',s,re.S):
 x=m.group();print('SUPP',m.group(1),re.findall(r'xlink:href=\"(["'^"]+)"'"',x),re.findall(r'\\?suppdata-name (["'^?]+)'"\\?',x))
PY"
