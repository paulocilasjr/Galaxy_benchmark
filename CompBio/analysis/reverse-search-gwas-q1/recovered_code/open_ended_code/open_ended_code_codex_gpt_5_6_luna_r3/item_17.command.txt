import urllib.request,json
ids=['rs4985726','rs3803800','rs11552708','rs1260326','rs1049174','rs1012411','rs1801274','rs10516487','rs7318477']
for id in ids:
 u='https://rest.ensembl.org/variation/human/'+id+'?content-type=application/json'
 try:
  d=json.load(urllib.request.urlopen(urllib.request.Request(u,headers={'Content-Type':'application/json','User-Agent':'x'})))
  arr=[]
  for m in d.get('mappings',[]):
   if m.get('assembly_name')=='GRCh38': arr.append((m.get('seq_region_name'),m.get('start'),m.get('allele_string'),m.get('strand')))
  print(id,arr)
 except Exception as e: print(id,e)
PY"
