/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for gene in ['FCAR','TNFSF13B','TNFRSF13B','TNFSF13','TRAF3','FCRL3','FCGR2A','FUT2','HGFAC','CD28','REL','LYN','ANXA3','IRF8']:
    url='https://rest.ensembl.org/lookup/symbol/homo_sapiens/'+gene+'?content-type=application/json'
    try:
        data=json.loads(urllib.request.urlopen(url,timeout=10).read())
        print(gene,data.get('seq_region_name'),data.get('start'),data.get('end'))
    except Exception as e: print(gene,'ERR',e)
PY"
