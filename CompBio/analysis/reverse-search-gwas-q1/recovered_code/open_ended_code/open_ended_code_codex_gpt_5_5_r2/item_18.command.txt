/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse,time
coords=[('1',161530617),('6',32067917),('17',16939677),('19',49525049),('4',3445429),('4',102486185),('2',111471251),('13',108308037),('2',27508073),('15',43464986),('14',102789124),('14',105657511),('17',7536794),('17',37701537)]
for ch,pos in coords:
    url=f'https://rest.ensembl.org/overlap/region/human/{ch}:{pos}-{pos}?feature=variation'
    req=urllib.request.Request(url,headers={'Content-Type':'application/json','Accept':'application/json'})
    try:
        data=json.loads(urllib.request.urlopen(req,timeout=20).read())
        print(ch,pos, [(d.get('id'), d.get('alleles')) for d in data[:5]])
    except Exception as e:
        print(ch,pos,'ERR',e)
    time.sleep(0.1)
PY"
