import requests
u='https://ftp.ebi.ac.uk/pub/databases/gwas/releases/latest/gwas-catalog-studies.tsv'

if r.ok:
 with open('catalog_studies_ftp.tsv','wb') as f:
  for b in r.iter_content(1048576):f.write(b)
 print('done',flush=True)
