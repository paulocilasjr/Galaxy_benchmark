/bin/bash -lc "cat > search_locus.py <<'PY'
import requests,concurrent.futures,json
qs=['\"HGFAC\" AND \"TNFRSF13B\"','\"FCGRT\" AND \"total protein\"','\"rs34562254\"','\"rs72717048\"','\"161530617\"','\"7.81024\"','\"1.4221\" AND genetics']
def f(q):
 try:
  d=requests.get('https://www.ebi.ac.uk/europepmc/webservices/rest/search',params={'query':q,'format':'json','pageSize':1000},timeout=60).json(); print(q,d['hitCount'],flush=True)
  open('locus_search_'+str(qs.index(q))+'.json','w').write(json.dumps(d))
  for r in d['resultList']['result']: print(q,r.get('id'),r.get('pmcid'),r['title'],flush=True)
 except Exception as e:print(q,e,flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=7) as e:list(e.map(f,qs))
PY
python search_locus.py > search_locus.txt"
