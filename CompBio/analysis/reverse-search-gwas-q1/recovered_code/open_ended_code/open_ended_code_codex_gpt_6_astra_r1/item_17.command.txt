/bin/bash -lc "cat > lead_lookup.py <<'PY'
import requests,concurrent.futures,json
loci=[('1',161540726),('1',161595363),('1',161558222),('1',161533223),('19',49525049),('4',102486185),('2',27508073)]
def f(cp):
 c,p=cp
 r=requests.get(f'https://rest.ensembl.org/overlap/region/human/{c}:{p}-{p+1}',params={'feature':'variation','content-type':'application/json'})
 if not r.ok:return
 snps=[x['id'] for x in r.json() if x['start']==p or (len(x['alleles'][0])>1 and x['start']==p+1)]
 for s in snps:
  r=requests.get('https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/'+s+'/associations',params={'projection':'associationBySnp','size':1000})
  if not r.ok:continue
  open(s+'.json','w').write(r.text)
  for a in r.json().get('_embedded',{}).get('associations',[]):print(c,p,s,a['pvalue'],[t['trait'] for t in a.get('efoTraits',[])],a['_links']['study']['href'],flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=6) as e:list(e.map(f,loci))
PY
python -u lead_lookup.py > lead_associations.txt"
