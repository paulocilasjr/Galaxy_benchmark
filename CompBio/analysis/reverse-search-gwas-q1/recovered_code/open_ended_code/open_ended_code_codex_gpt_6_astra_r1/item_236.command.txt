/bin/bash -lc "python - <<'PY'
import requests,concurrent.futures,re
urls=['https://ukb-ppp.gwas.eu/','https://www.ebi.ac.uk/europepmc/webservices/rest/PMC10567551/fullTextXML']
def f(u):
 r=requests.get(u,timeout=30);print(u,r.status_code,len(r.content),flush=True);open('ukbppp_'+str(urls.index(u))+'.txt','w').write(r.text)
 if 'XML' in u:
  s=re.sub('<["'^>]+>'"',' ',r.text);s=re.sub(r'\\s+',' ',s)
  for x in ['Data availability','summary statistics','CD27','FCGR2B','PLINK']:
   for m in list(re.finditer(x,s,re.I))[:4]:print(x,s[max(0,m.start()-100):m.end()+900],flush=True)
 else:print(r.text[:2000],flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as e:list(e.map(f,urls))
PY"
