import requests,pysam,concurrent.futures
ids=['GCST90019522','GCST90019515','GCST90310183','GCST90018756','GCST90474423']
leads=[('1',161531006),('6',32067917),('17',16939677),('19',49534760),('4',3445429),('4',102486185),('2',111471251),('13',108308037),('2',27508073)]
def f(i):
 try:
  n=int(i[4:]);lo=(n-1)//1000*1000+1;u=f'https://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/GCST{lo:08d}-GCST{lo+999:08d}/{i}/harmonised/{i}.h.tsv.gz'
  r=requests.get(u+'.tbi',timeout=30)
  if not r.ok:print(i,r.status_code);return
  open(i+'.tbi','wb').write(r.content)
  t=pysam.TabixFile(u,index=i+'.tbi')
  for c,p in leads:
   for s in t.fetch(c,p-1,p):print(i,s,flush=True)
 except Exception as e:print(i,str(e),flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as e:list(e.map(f,ids))
PY
python compare_leads.py > compare_leads.txt"
