/bin/bash -lc "python - <<'PY'
import gzip,heapq
h=[];n=0
with gzip.open('SAS_totalprotein.gz','rt') as f:
 hd=next(f).strip().split('\\t');print(hd,flush=True)
 pi=next((i for i,x in enumerate(hd) if x.upper() in ['PVALUE','P','PVAL']),-1)
 print('Pindex',pi,flush=True)
 for l in f:
  a=l.split();n+=1
  if ('17:16842991' in l or ('17'==a[0] and a[1]=='16842991')):print('MATCH17',l,flush=True)
  if pi<0:break
  try:p=float(a[pi])
  except:continue
  if len(h)<10:heapq.heappush(h,(-p,n,l.strip()))
  elif -p>h[0][0]:heapq.heapreplace(h,(-p,n,l.strip()))
 print('COUNT',n,flush=True)
 for t in sorted(h,reverse=True):print(t[2])
PY"
