import requests,pysam,concurrent.futures
ids=['GCST90319622','GCST90319650','GCST90161454']
def f(i):
 try:
  n=int(i[4:]);lo=(n-1)//1000*1000+1;hi=lo+999
  u=f'https://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/GCST{lo:08d}-GCST{hi:08d}/{i}/harmonised/{i}.h.tsv.gz'
  r=requests.get(u+'.tbi',timeout=30)
  if not r.ok:print(i,r.status_code);return
  open(i+'.tbi','wb').write(r.content)
  t=pysam.TabixFile(u,index=i+'.tbi')
  print(i,list(t.header),flush=True)
  for c,p in [('1',14671),('17',16939677),('1',161531006)]:
   for s in t.fetch(c,p-1,p):print(i,s,flush=True)
 except Exception as e:print(i,str(e),flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as e:list(e.map(f,ids))
PY
python -u probe_tabix.py"
