/bin/bash -lc "cat > candidate_assoc.py <<'PY'
import requests,concurrent.futures,json
ids=['GCST90161454','GCST90161703','GCST90248015','GCST90247561','GCST90101125','GCST008576','GCST90013335']
def f(i):
 try:
  r=requests.get('https://www.ebi.ac.uk/gwas/rest/api/studies/'+i+'/associations',params={'projection':'associationBySnp','size':1000},timeout=45)
  if not r.ok:print(i,r.status_code);return
  open(i+'_assoc.json','w').write(r.text)
  print(i)
  for a in r.json().get('_embedded',{}).get('associations',[]):print(i,a['pvalue'],[(s['rsId'],[(l['chromosomeName'],l['chromosomePosition']) for l in s.get('locations',[])]) for s in a.get('snps',[])],flush=True)
 except Exception as e:print(i,str(e))
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as e:list(e.map(f,ids))
PY
python -u candidate_assoc.py"
