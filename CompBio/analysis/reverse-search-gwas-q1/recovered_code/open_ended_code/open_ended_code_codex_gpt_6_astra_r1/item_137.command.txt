/bin/bash -lc "python - <<'PY'
import requests
u='https://www.ebi.ac.uk/gwas/api/search/downloads/studies/v1.0.3'

if r.ok:
 with open('catalog_studies.tsv','wb') as f:
  for chunk in r.iter_content(1048576):f.write(chunk)
 print('done')
PY"
