/bin/bash -lc "python - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
import re
M='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
for fn in ['sharapov_MOESM3.xlsx','sharapov_MOESM4.xlsx','sharapov_MOESM5.xlsx']:
 z=ZipFile(fn); ss=[]
 if 'xl/sharedStrings.xml' in z.namelist():
  rt=ET.fromstring(z.read('xl/sharedStrings.xml'))
  ss=[''.join(t.text or '' for t in x.iter(M+'t')) for x in rt.findall(M+'si')]
 wb=ET.fromstring(z.read('xl/workbook.xml')); sheets=list(wb.find(M+'sheets'))
 print('\\nFILE',fn)
 for i,s in enumerate(sheets,1):
  print('SHEET',s.get('name'))
  root=ET.fromstring(z.read(f'xl/worksheets/sheet{i}.xml'))
  for row in list(root.iter(M+'row'))[:12]:
   vals=[]
   for c in row.findall(M+'c'):
    t=c.get('t'); v=c.find(M+'v'); x='' if v is None else v.text
    if t=='s' and x: x=ss[int(x)]
    elif t=='inlineStr': x=''.join(a.text or '' for a in c.iter(M+'t'))
    vals.append(c.get('r')+'='+x)
   print(' | '.join(vals))
PY"
