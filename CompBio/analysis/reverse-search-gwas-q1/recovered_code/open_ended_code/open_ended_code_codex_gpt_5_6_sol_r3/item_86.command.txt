/bin/bash -lc "python - <<'PY'
import csv,math,collections
by=collections.defaultdict(list)
with open('q_catalog_position_hits.tsv') as f:
 for row in csv.reader(f,delimiter='\\t'):
  if len(row)<10: continue
  qp,pmid,title,trait,sra,snp,cp,ptxt,acc,pos=row[:10]
  try:d=abs(math.log10(float(qp))-math.log10(float(cp)))
  except:continue
  by[(acc,pmid,title,trait)].append((pos,float(qp),float(cp),d,snp))
sc=[]
for k,v in by.items():
 best={}
 for x in v:
  if x[0] not in best or x[3]<best[x[0]][3]:best[x[0]]=x
 vv=list(best.values()); close=[x for x in vv if x[3]<0.3]; close1=[x for x in vv if x[3]<1]; close2=[x for x in vv if x[3]<2]
 score=sum(max(0,3-x[3]) for x in vv)
 sc.append((len(close),len(close1),len(close2),len(vv),score,k,vv))
for c03,c1,c2,n,s,k,v in sorted(sc,reverse=True)[:100]:
 print('\\nRANK',c03,c1,c2,n,round(s,2),k)
 for x in sorted(v,key=lambda z:z[3])[:15]: print(' ',x)
PY"
