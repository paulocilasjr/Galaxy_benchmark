/bin/bash -lc "python3 - <<'PY'
# use data range quick, manually print first/last text lines
import urllib.request,zlib
url='https://pan-ukb-us-east-1.s3.amazonaws.com/sumstats_flat_files/biomarkers-30600-both_sexes-irnt.tsv.bgz'
start=101365798;end=start+100000
raw=urllib.request.urlopen(urllib.request.Request(url,headers={'Range':f'bytes={start}-{end-1}'})).read();p=0; out=[]
while p+18<=len(raw) and raw[p:p+2]==b'\\x1f\\x8b':
 n=int.from_bytes(raw[p+16:p+18],'little')+1
 if p+n>len(raw):break
 out.append(zlib.decompress(raw[p+18:p+n-8],-15));p+=n
text=b''.join(out).decode(errors='replace').splitlines()
print('blocks',len(out),'lines',len(text),'bytes',p)
print('\\n'.join(text[:3])); print('---'); print('\\n'.join(text[-3:]));
PY"
