for fn in ['sheet1.xml','sheet2.xml','sheet3.xml','sheet4.xml','sheet5.xml','sheet6.xml','sheet7.xml','sheet8.xml','sheet9.xml']:
 root=ET.fromstring(z.read('xl/worksheets/'+fn)); rows=root.findall('.//m:sheetData/m:row',ns)
 print('\\n',fn,'rows',len(rows))
 for row in rows[:3]:
  vals=[]
  for c in row.findall('m:c',ns):
   v=c.find('m:v',ns); x='' if v is None else v.text
   if c.attrib.get('t')=='s': x=ss[int(x)]
   vals.append((c.attrib.get('r'),x))
  print(vals[:20])
