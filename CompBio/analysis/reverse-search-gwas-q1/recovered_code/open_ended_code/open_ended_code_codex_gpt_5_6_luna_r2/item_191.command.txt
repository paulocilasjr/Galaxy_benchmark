/bin/bash -lc "python3 - <<'PY'
import struct,gzip
p='pan-albumin.tbi'
data=gzip.open(p,'rb').read()
o=0
print(data[:4])
magic,n_ref,fmt,cs,cb,ce,meta,skip,l_nm=struct.unpack_from('<4s8i',data,o); o=36
names=data[o:o+l_nm].rstrip(b'\\0').split(b'\\0'); o+=l_nm
print(n_ref,fmt,cs,cb,ce,names[:5],names[-2:])
for i,name in enumerate(names):
    if i>=n_ref: break
    n_bin=struct.unpack_from('<i',data,o)[0]; o+=4
    bins=[]
    for _ in range(n_bin):
        b,nc=struct.unpack_from('<II',data,o); o+=8
        ch=[]
        for j in range(nc):
            beg,end=struct.unpack_from('<QQ',data,o);o+=16;ch.append((beg,end))
        bins.append((b,ch))
    n_intv=struct.unpack_from('<i',data,o)[0];o+=4
    ints=struct.unpack_from('<'+'Q'*n_intv,data,o) if n_intv else [];o+=8*n_intv
    if i==0:
        print('chr1 bins',n_bin,'intervals',n_intv)
        print('first bins',[(b,len(ch),ch[:2]) for b,ch in bins[:5]])
        print('last intervals',ints[-5:])
print('parsed',o,len(data))
PY"
