import json,glob,urllib.request,concurrent.futures,os,time
fs=['rs7554873','rs9273902','rs4273077','rs4561508','rs4792800','rs4985726','rs34562254','rs3752442','rs3803800','rs1801274','rs1260326']

 for a in d.get('_embedded',{}).get('associations',[]):
  h=a.get('_links',{}).get('study',{}).get('href')

 except Exception as e: return {'variant':x,'assoc_p':p,'url':h,'error':str(e)}
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as ex:
 out=list(ex.map(get,uniq.items()))
json.dump(out,open('key-study-meta.json','w'))
print('ok',sum('study' in x for x in out),'err',sum('error' in x for x in out))
for x in out:
