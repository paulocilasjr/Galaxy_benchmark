/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET
r=ET.parse('antibody.xml').getroot()
for s in r.findall('.//sec'):
 t=' '.join(s.findtext('title','').split())
 if t in ('GWAS of serum IgG and IgM in the EPIC Cohort','Acquisition of public GWAS data sets','Harmonisation and standardisation of GWAS data sets'):
  print('\\n',t)
  for p in s.findall('./p'):
   print(' '.join(''.join(p.itertext()).split()))
for ref in r.findall('.//ref'):
 lab=ref.findtext('label')
 if lab in [str(i) for i in range(1,12)]:
  print('REF',lab,' '.join(''.join(ref.itertext()).split()))
PY"
