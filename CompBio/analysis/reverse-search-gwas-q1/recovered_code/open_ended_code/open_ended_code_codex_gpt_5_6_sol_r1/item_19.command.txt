import struct
b=open('afr_tail.bin','rb').read()
pos=0
while True:
 p=b.find(b'PK\\x01\\x02',pos)

 h=struct.unpack_from('<4s6H3I5H2I',b,p)
 fnl,xl,cl=h[10],h[11],h[12]
 name=b[p+46:p+46+fnl].decode()
 extra=b[p+46+fnl:p+46+fnl+xl]
 print(name,'c32',h[8],'u32',h[9],'off32',h[16],'extra',extra.hex())
 pos=p+46+fnl+xl+cl
PY"
