/bin/bash -lc "python - <<'PY'
import xml.etree.ElementTree as ET
root=ET.parse('antibody.xml').getroot()
for e in root.iter():
    tag=e.tag.split('}')[-1]
    if tag in ('supplementary-material','media','ext-link'):
        attrs={k.split('}')[-1]:v for k,v in e.attrib.items()}
        txt=' '.join(''.join(e.itertext()).split())[:160]
        print(tag, attrs, txt)
PY"
