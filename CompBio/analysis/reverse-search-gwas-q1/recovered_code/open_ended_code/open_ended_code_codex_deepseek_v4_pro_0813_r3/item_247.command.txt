/bin/bash -lc "sleep 2; python3 - <<'PY'
import urllib.parse, urllib.request, json, time
terms=[
 '(TNFRSF13B[Title/Abstract] AND TNFSF13B[Title/Abstract] AND HGFAC[Title/Abstract])',
 '(SLAMF7[Title/Abstract] AND TNFRSF13B[Title/Abstract] AND FCGR2A[Title/Abstract])',
 '(GCKR[Title/Abstract] AND HGFAC[Title/Abstract] AND FUT2[Title/Abstract] AND GWAS[Title/Abstract])',
 '(FCGR2A[Title/Abstract] AND FUT2[Title/Abstract] AND TNFRSF13B[Title/Abstract])',
 '(FCGR2A[Title/Abstract] AND GCKR[Title/Abstract] AND HGFAC[Title/Abstract])',
]
for term in terms:
 url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=pubmed&term='+urllib.parse.quote(term)+'&retmode=json&retmax=20'
 try:
  data=json.loads(urllib.request.urlopen(url).read())
 except Exception as e:
  print('ERR',e); time.sleep(5); continue
 ids=data['esearchresult'].get('idlist',[])
 print('\\nTERM',term,'count',data['esearchresult'].get('count'),'ids',ids)
 if ids:
  time.sleep(1)
  url2='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=pubmed&id='+','.join(ids)+'&retmode=json'
  try:
   j=json.loads(urllib.request.urlopen(url2).read())
   for uid in ids:
    r=j['result'][uid]; print(' ',uid,r.get('title'))
  except Exception as e: print('summary err',e)
 time.sleep(2)
PY"
