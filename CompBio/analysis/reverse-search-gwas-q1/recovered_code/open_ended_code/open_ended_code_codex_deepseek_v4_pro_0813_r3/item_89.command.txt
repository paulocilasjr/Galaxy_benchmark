/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, concurrent.futures, time, sys, os

def vep(row):
    p,ch,pos,ref,alt,vid=row.rstrip('\\n').split('\\t')
    hgvs=f'{ch}:g.{pos}{ref}>{alt}'
    url='https://rest.ensembl.org/vep/human/hgvs/'+urllib.parse.quote(hgvs,safe='')+'?content-type=application/json&dbSNP=1'
    req=urllib.request.Request(url,headers={'Accept':'application/json','User-Agent':'Mozilla/5.0'})
    for attempt in range(3):
        try:
            data=json.load(urllib.request.urlopen(req,timeout=60))
            x=data[0]
            col=x.get('colocated_variants',[])
            rs=[c.get('id') for c in col if c.get('id','').startswith('rs')]
            return (p,ch,pos,ref,alt,vid,rs)
        except Exception as e:
            if attempt==2: return (p,ch,pos,ref,alt,vid,['ERR:'+repr(e)])
            time.sleep(1)
rows=open('[REDACTED LOCAL PATH]').readlines()
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as ex:
    futs=[ex.submit(vep,r) for r in rows]
    for fut in concurrent.futures.as_completed(futs):
        results.append(fut.result())
# sort by p numeric
results.sort(key=lambda x:float(x[0]))
with open('[REDACTED LOCAL PATH]','w') as out:
    for r in results:
        out.write('\\t'.join([r[0],r[1],r[2],r[3],r[4],r[5],','.join(r[6])])+'\\n')
        print(r)
PY"
