/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://api.genetics.opentargets.org/graphql'
query='query { variantInfo(variantId: \"1_161530617_CTT_C\") { id rsId chromosome position refAllele altAllele } }'
req=urllib.request.Request(url,data=json.dumps({'query':query}).encode(),headers={'Content-Type':'application/json','User-Agent':'Mozilla/5.0'})
try:
 d=urllib.request.urlopen(req,timeout=30).read()
 print(d[:2000].decode())
except Exception as e:
 print('ERR',repr(e))
 try:
  print(e.read().decode()[:2000])
 except: pass
PY"
