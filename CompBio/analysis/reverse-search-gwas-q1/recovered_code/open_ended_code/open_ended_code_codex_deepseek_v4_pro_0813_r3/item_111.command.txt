/bin/bash -lc "python3 - <<'PY'
import urllib.request,urllib.parse,json,time
rows=['4 3445429 C T','17 16939677 G A','19 49525049 CCTGCTG C','2 111471251 T C','13 108308037 T A']
for row in rows:
 ch,pos,ref,alt=row.split()
 # SNVs only; indel skip
 if len(ref)==1 and len(alt)==1:
  hgvs=f'{ch}:g.{pos}{ref}>{alt}'
  url='https://rest.ensembl.org/vep/human/hgvs/'+urllib.parse.quote(hgvs,safe='')+'?content-type=application/json&dbSNP=1'
  req=urllib.request.Request(url,headers={'Accept':'application/json','User-Agent':'Mozilla/5.0'})
  try:
   data=json.load(urllib.request.urlopen(req,timeout=30))[0]
   print(row,'rsids',[c.get('id') for c in data.get('colocated_variants',[])])
   print('consequences', [tc.get('gene_symbol') for tc in data.get('transcript_consequences',[])[:5]], data.get('most_severe_consequence'))
  except Exception as e: print(row,'ERR',e)
  time.sleep(0.2)
 else: print(row,'indel')
PY"
