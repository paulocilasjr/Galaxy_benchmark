/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,math
for rs,target in [('rs374039502',1.3194e-8),('rs200748895',1.3194e-8),('rs11097787',1.13487e-8),('rs4690098',1.61837e-10),('rs4849748',1.25972e-8)]:
 print('\\nRS',rs)
 try:
  d=json.load(urllib.request.urlopen(f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?projection=associationBySnp&size=100',timeout=30))
  rows=[]
  for a in d.get('_embedded',{}).get('associations',[]):
   pv=a.get('pvalue') or (a.get('pvalueMantissa') or 1)*10**(a.get('pvalueExponent') or 0)
   traits=[t.get('trait') for t in a.get('efoTraits',[])]
   href=a.get('_links',{}).get('study',{}).get('href','')
   rows.append((abs(math.log10(pv)-math.log10(target)) if pv and target else 999,pv,traits,href))
  print('n',len(rows))
  for r in sorted(rows,key=lambda x:x[0])[:30]: print(r)
 except Exception as e: print('ERR',e)
PY"
