/bin/bash -lc "python3 - <<'PY'
import gzip
s=[]
with gzip.open('reverse.search.gwas.q1.tsv.gz','rt') as f:
    next(f)
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        pv=float(p[7])
        if pv<5e-8:
            s.append((p[0], int(p[1]), pv, p[2]))
# clusters within 1Mb per chr
clusters=[]
for chrom in sorted(set(x[0] for x in s), key=lambda c: (int(c) if c.isdigit() else 999, c)):
    arr=sorted([x for x in s if x[0]==chrom], key=lambda x:x[1])
    cur=[]; start=None; last=None
    for x in arr:
        if not cur or x[1]-last<=1000000:
            cur.append(x); last=x[1]; start=cur[0][1]
        else:
            clusters.append((chrom,start,last,cur)); cur=[x]; last=x[1]
    if cur: clusters.append((chrom,cur[0][1],cur[-1][1],cur))
for chrom,start,end,cur in clusters:
    best=min(cur,key=lambda x:x[2])
    print(chrom,start,end,len(cur),best[2],best[3])
PY"
