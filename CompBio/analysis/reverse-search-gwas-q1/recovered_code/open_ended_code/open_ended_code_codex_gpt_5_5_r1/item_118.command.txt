/bin/bash -lc "python3 - <<'PY' > orru_studies.tsv
import urllib.request,json
studies=[]
for page in range(5):
 url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByPublicationIdPubmedId?pubmedId=32929287&size=1000&page={page}'
 try:
  d=json.load(urllib.request.urlopen(url,timeout=60))
 except Exception as e:
  print('ERR',e); break
 batch=d.get('_embedded',{}).get('studies',[])
 studies+=batch
 if page+1 >= d.get('page',{}).get('totalPages',page+1): break
for s in studies:
 print(s.get('accessionId'), s.get('diseaseTrait',{}).get('trait',''), sep='\\t')
PY
wc -l orru_studies.tsv
head -n 20 orru_studies.tsv"
