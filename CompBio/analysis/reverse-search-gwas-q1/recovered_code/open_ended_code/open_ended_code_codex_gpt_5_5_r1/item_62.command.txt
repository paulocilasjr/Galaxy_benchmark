/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
rsids=['rs7554873','rs34562254','rs1260326']
for rs in rsids:
 print('\\nRS',rs)
 url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?projection=associationBySnp&size=200'
 data=json.load(urllib.request.urlopen(url,timeout=40))
 ass=data.get('_embedded',{}).get('associations',[])
 rows=[]
 for a in ass:
  pv=a.get('pvalue') or (a.get('pvalueMantissa') or 1)*10**(a.get('pvalueExponent') or 0)
  traits=';'.join(t.get('trait','') for t in a.get('efoTraits',[]))
  href=a.get('_links',{}).get('study',{}).get('href','')
  rows.append((pv,traits,href,a.get('betaNum'),a.get('orPerCopyNum')))
 for row in sorted(rows,key=lambda x:x[0])[:80]:
  print(row)
PY"
