/bin/bash -lc "python3 - <<'PY' > loya_top_coords.tsv
import urllib.request,json
rsids=['rs75417510','rs73017399','rs78349639','rs140797462','rs12078104','rs527665651','rs186021206','rs7554873','rs6697139','rs67418890','rs687621','rs72717028','rs367853839']
for rs in rsids:
  try:
    d=json.load(urllib.request.urlopen(f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}',timeout=20))
    for loc in d.get('locations',[]): print(rs,loc['chromosomeName'],loc['chromosomePosition'],sep='\\t')
  except Exception as e: print(rs,'ERR',e,sep='\\t')
PY
cat loya_top_coords.tsv
gzip -dc reverse.search.gwas.q1.tsv.gz | awk 'NR==FNR{key["'$2":"$3]=$1; next} (($1":"$2) in key){print key[$1":"$2]"'"\\t\""'$0}'"' loya_top_coords.tsv -"
