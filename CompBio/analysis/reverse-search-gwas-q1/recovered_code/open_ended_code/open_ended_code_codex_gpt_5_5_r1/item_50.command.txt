/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
rsids=['rs72704050','rs7554873','rs2133037','rs2511713','rs735665','rs1611637','rs2269710','rs117775520','rs11627594','rs9405084']
for rs in rsids:
  try:
    d=json.load(urllib.request.urlopen(f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}',timeout=20))
    locs=[(l['chromosomeName'],l['chromosomePosition']) for l in d.get('locations',[])]
    print(rs,locs)
  except Exception as e: print(rs,e)
PY"
