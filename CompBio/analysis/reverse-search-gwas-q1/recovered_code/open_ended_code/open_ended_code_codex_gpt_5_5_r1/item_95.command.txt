/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByPubmedId?pubmedId=39789286&size=1000'
try:
 d=json.load(urllib.request.urlopen(url,timeout=60))
 studies=d.get('_embedded',{}).get('studies',[])
 print('n',len(studies), d.get('page'))
 for s in studies[:10]: print(s.get('accessionId'), s.get('diseaseTrait',{}).get('trait'), s.get('snpCount'), s.get('initialSampleSize'))
except Exception as e: print('ERR',e)
PY"
