import urllib.request
urls = [
 'https://www.ncbi.nlm.nih.gov/clinvar/variation/2506651/',
 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=clinvar&id=2506651&retmode=json',
 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=clinvar&term=2506651%5Buid%5D&retmode=json'
]
for u in urls:
    print('\\nURL',u)
    try:
        with urllib.request.urlopen(u, timeout=30) as f:
            b=f.read()
        print('bytes',len(b))
        print(b[:1000].decode('utf-8','replace'))
    except Exception as e: print(type(e).__name__,e)
PY"
