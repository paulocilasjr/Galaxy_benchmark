/bin/bash -lc "python3 - <<'PY'
import statistics
rows=[]
for line in open('cov100k_filtered.tsv'):
 c,s,e,v=line.strip().split('\\t'); s=int(s); e=int(e); v=float(v)
 rows.append((s,e,v))
h=sum(v for s,e,v in rows if 36000000<=s<=36300000)/4
# all bins with v>3 and not high, excluding first/last gaps? 
vals=[v for s,e,v in rows if v>3 and not (36000000<=s<=36300000)]
print('high',h,'all v>3 median/mean',statistics.median(vals),statistics.mean(vals),'ratio mean',h/statistics.mean(vals),'ratio median',h/statistics.median(vals))
# using mode bins 3.5-4.5
vals2=[v for s,e,v in rows if 3.5<=v<=4.5 and not (36000000<=s<=36300000)]
print('vals 3.5-4.5 median mean',statistics.median(vals2),statistics.mean(vals2),'ratio',h/statistics.mean(vals2),h/statistics.median(vals2))
PY"
