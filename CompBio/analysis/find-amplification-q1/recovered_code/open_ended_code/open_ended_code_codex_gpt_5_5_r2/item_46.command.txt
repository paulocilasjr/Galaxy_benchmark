/bin/bash -lc "samtools view -f 66 -F 3844 chr22.sorted.bam | awk 'BEGIN{bin=100000} {t="'$9; if(t==0) next; a=t<0?-t:t; s=(t>0?$4:$4+t); mid=s+a/2; b=int((mid-1)/bin); c[b]++} END{for (b in c) printf "%d'"\\t%d\\t%d\\n\", b*bin+1, (b+1)*bin, c[b]}' | sort -n > chr22_100kb_counts.tsv
awk '{sum+="'$3; n++; if($3>max){max=$3; maxs=$1; maxe=$2}} END{print "bins",n,"sum",sum,"mean",sum/n,"max",max,maxs,maxe}'"' chr22_100kb_counts.tsv
sort -k3,3nr chr22_100kb_counts.tsv | head -20"
