/bin/bash -lc "awk '"'$1=="chr22"{print $1"'"\\t\""'$2}'"' hg38.chrom.sizes > chr22.genome
bedtools makewindows -g chr22.genome -w 100000 > chr22_100kb_windows.bed
bedtools coverage -a chr22_100kb_windows.bed -b chr22.sorted.bam -mean > chr22_100kb_mean_depth.tsv
awk '"'$2>=34000000 && $2<=38000000{printf "%d'"\\t%d\\t%.4f\\n\", "'$2+1,$3,$4}'"' chr22_100kb_mean_depth.tsv
awk 'BEGIN{in=0; fl=0; nin=0; nfl=0} ("'$2>=36000000 && $3<=36400000){in+=$4; nin++} (($2>=35000000 && $3<=36000000)||($2>=36400000 && $3<=37400000)){fl+=$4; nfl++} END{print "amp_mean",in/nin,"flank_mean",fl/nfl,"ratio",(in/nin)/(fl/nfl)}'"' chr22_100kb_mean_depth.tsv"
