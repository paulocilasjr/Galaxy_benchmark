import numpy as np
# 10kb windows in q-arm 15M-50.8M, exclude zero gaps? compute median and MAD
vals=[]
with open('cov10k.tsv') as f:
    for line in f:
        c,s,e,v=line.split(); s=int(s); v=float(v)
        if 15000000<=s and v>0:
            vals.append((s,v))
a=np.array([v for s,v in vals])
med=np.median(a); mad=np.median(np.abs(a-med))
print('med',med,'mad',mad,'sd',a.std())
# top segments > med + 2*1.4826*mad maybe
thr=med+3*1.4826*mad
print('thr',thr)
# contiguous runs of 10k bins above med+1.5*sd?
thr2=med+1.5*a.std()
print('thr2',thr2)
runs=[]
cur=[]
for s,v in vals:
    if v>thr2:
        if not cur or s-cur[-1][1]==10000:
            cur.append((s,v))
        else:
            if len(cur)>=5: runs.append(cur)
            cur=[(s,v)]
    else:
        if len(cur)>=5: runs.append(cur)
        cur=[]
if len(cur)>=5: runs.append(cur)
for r in runs:
    print('run',r[0][0],r[-1][0]+10000,'len',len(r),'mean',np.mean([v for s,v in r]),'min',min(v for s,v in r),'max',max(v for s,v in r))
PY"
