/bin/bash -lc "python3 - <<'PY'
import requests, time, re, json
rid='7X03DXY9016'
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi'
for i in range(20):
    params={'CMD':'Get','FORMAT_TYPE':'JSON2_S','RID':rid}
    r=requests.get(url,params=params,timeout=60)
    text=r.text
    print('poll',i,'len',len(text), 'status snippet', text[:120].replace('\\n',' '))
    # try parse JSON
    try:
        data=json.loads(text)
        print('JSON keys', list(data.keys())[:20])
        break
    except Exception as e:
        pass
    m=re.search(r'Status=(\\w+)',text)
    print('status', m.group(1) if m else None)
    if 'There are no hits' in text or 'QBlastInfoEnd' in text and 'Status=READY' not in text and 'Status=WAITING' not in text:
        pass
    time.sleep(5)
PY"
