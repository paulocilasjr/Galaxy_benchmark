/bin/bash -lc "python3 - <<'PY'
import csv, statistics
rows=[]
with open('depth_1kb.csv') as f:
    for r in csv.DictReader(f):
        rows.append((int(r['start']),float(r['mean'])))
means=[m for s,m in rows]
nz=[m for m in means if m>0]
med=statistics.median(nz)
print('med',med,'mean nz',statistics.mean(nz))
for thrmult in [1.1,1.15,1.2,1.25,1.3]:
    thr=med*thrmult
    seg=[]; start=None
    for idx,(s,m) in enumerate(rows):
        if m>thr:
            if start is None: start=idx
            end=idx
        else:
            if start is not None and end-start+1>=50:
                seg.append((start,end))
            start=None
    if start is not None and len(rows)-1-start+1>=50:
        seg.append((start,len(rows)-1))
    print('thrmult',thrmult,'thr',round(thr,2),'long segs')
    for a,b in seg:
        print(' ',rows[a][0],rows[b][0]+999,'len',(b-a+1)*1000,'mean',statistics.mean(means[a:b+1]),'max',max(means[a:b+1]))
PY"
