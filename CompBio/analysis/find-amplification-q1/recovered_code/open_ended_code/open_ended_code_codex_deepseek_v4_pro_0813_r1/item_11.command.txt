/bin/bash -lc "python3 - <<'PY'
import requests, re
seq=open('read1.txt').read().strip()
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi'
params={'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,'ENTREZ_QUERY':'Homo sapiens[orgn]','HITLIST_SIZE':'5','EXPECT':'1e-10','FILTER':'F'}
r=requests.post(url,data=params,timeout=60)
text=r.text
for pat in ['RID','rid','QBlastInfo','Status','There was','error']:
    for m in re.finditer(pat, text, re.I):
        s=max(0,m.start()-200); print(pat, text[s:m.start()+300].replace('\\n',' ')[:500])
        break
open('blast_put.html','w').write(text)
PY"
