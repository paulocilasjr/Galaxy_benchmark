import pandas as pd, numpy as np, scipy.io
sc=pd.read_csv('data/spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('data/spatial_q_sc_metadata.csv',index_col=0).squeeze('columns')

# spot x cells exact sum of one cell
exact=[]
for i in range(mat.shape[1]):
    x=np.asarray(mat[:,i].todense()).ravel()
    eq=(sc.values == x).all(axis=1)
    if eq.any():
        exact.append((i, sc.index[eq].tolist(), x.sum()))
print('spots exactly equal one cell:',len(exact))
print(exact[:20])
# exact equality maybe scale?
# check for spot 710 if any single cell equal
x=np.asarray(mat[:,710].todense()).ravel()
print('spot sum',x.sum(),'exact any', (sc.values==x).all(axis=1).sum())
# nearest single cells by L1
l1=np.abs(sc.values-x).sum(axis=1)
print('top L1 single')
print(pd.Series(l1,index=sc.index).nsmallest(10), meta[sc.index[np.argsort(l1)[:10]]].tolist())
PY"
