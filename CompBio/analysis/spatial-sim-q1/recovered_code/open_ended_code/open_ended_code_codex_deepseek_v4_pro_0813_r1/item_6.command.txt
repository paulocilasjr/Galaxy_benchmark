/bin/bash -lc "python3 - <<'PY'
import gzip, scipy.io, numpy as np, pandas as pd
# read mtx
mat=scipy.io.mmread('data/matrix.mtx.gz')
print(type(mat),mat.shape)
# convert csr
mat=mat.tocsr()
bc=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')
feat=pd.read_csv('data/features.tsv.gz',header=None,sep='\\t',compression='gzip')
idx=bc[bc[0]=='Spot_710-1'].index[0]
x=np.asarray(mat[idx,:].todense()).ravel()
print('idx',idx,'nnz',np.sum(x>0),'sum',x.sum(),'max',x.max())
# top genes
order=np.argsort(-x)
for i in order[:30]: print(feat.iloc[i,1], int(x[i]))
PY"
