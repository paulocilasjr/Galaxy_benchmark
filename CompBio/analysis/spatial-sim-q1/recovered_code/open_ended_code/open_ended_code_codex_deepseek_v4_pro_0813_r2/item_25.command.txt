/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
from itertools import product
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
mat=mmread('matrix.mtx.gz').tocsc()
t=mat[:,710].toarray().ravel().astype(float)
means=sc.groupby(meta.cell_type).mean()
cts=means.index.tolist()
# all nonneg integer compositions of 3 among 6
combos=[]
def rec(rem, prefix, idx):
    if idx==len(cts)-1:
        combos.append(prefix+[rem]); return
    for n in range(rem+1):
        rec(rem-n,prefix+[n],idx+1)
rec(3,[],0)
# For each combo, predicted = sum n_i * mean_i, SSE, corr
rows=[]
for comb in combos:
    pred=np.zeros(1000)
    for c,n in zip(cts,comb):
        pred += n*means.loc[c].values
    sse=((t-pred)**2).sum()
    corr=np.corrcoef(t,pred)[0,1]
    rows.append((comb,sse,corr))
rows.sort(key=lambda x:x[1])
for r in rows[:15]:
    print(r)
PY"
