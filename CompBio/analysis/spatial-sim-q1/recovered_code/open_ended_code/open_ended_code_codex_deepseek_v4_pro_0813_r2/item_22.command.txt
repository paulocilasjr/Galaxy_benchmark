/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
from scipy.stats import spearmanr, pearsonr
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
mat=mmread('matrix.mtx.gz').tocsc()
t=mat[:,710].toarray().ravel().astype(float)
# normalize? correlation pearson on raw counts
A=sc.values.astype(float)
corrs=np.array([np.corrcoef(t,A[i])[0,1] for i in range(A.shape[0])])
idx=np.argsort(corrs)[::-1][:30]
for i in idx:
    print(i, meta.index[i], meta.iloc[i].cell_type, corrs[i])
print('mean corr by type')
print(pd.Series(corrs, index=meta.cell_type).groupby(meta.cell_type).agg(['mean','max']))
# spearman top maybe costly but okay
# normalized correlation via log1p
X=np.log1p(A)
y=np.log1p(t)
corrs2=np.array([np.corrcoef(y,X[i])[0,1] for i in range(A.shape[0])])
idx2=np.argsort(corrs2)[::-1][:20]
print('log corr top')
for i in idx2:
    print(meta.index[i], meta.iloc[i].cell_type, corrs2[i])
PY"
