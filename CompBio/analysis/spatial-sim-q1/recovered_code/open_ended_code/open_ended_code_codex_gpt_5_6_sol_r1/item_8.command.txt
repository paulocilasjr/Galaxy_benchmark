/bin/bash -lc "python -u - <<'PY'
import csv,gzip,math,itertools
# metadata
meta={}
with open('spatial_q_sc_metadata.csv',newline='') as f:
 for row in list(csv.reader(f))[1:]: meta[row[0]]=row[1]
types=sorted(set(meta.values())); ti={t:i for i,t in enumerate(types)}
# profiles sums
sums=[[0.0]*1000 for _ in types]; ns=[0]*len(types)
with open('spatial_q_sc_counts.csv',newline='') as f:
 r=csv.reader(f); header=next(r); genes=header[1:]
 for row in r:
  k=ti[meta[row[0]]]; ns[k]+=1
  vals=map(float,row[1:])
  sk=sums[k]
  for i,v in enumerate(vals): sk[i]+=v
M=[[sums[k][g]/ns[k] for k in range(len(types))] for g in range(1000)]
print('types',types,'ns',ns,flush=True)
# target y
Y=[0.0]*1000
with gzip.open('matrix.mtx.gz','rt') as f:
 for line in f:
  if line.startswith('%'): continue
  p=line.split()
  if len(p)==3 and int(p[1])==711: Y[int(p[0])-1]=float(p[2])
print('ysum',sum(Y),'ynz',sum(v"'!=0 for v in Y),flush=True)
# Gram and cross products
G=[[sum(M[g][i]*M[g][j] for g in range(1000)) for j in range(6)] for i in range(6)]
b=[sum(M[g][i]*Y[g] for g in range(1000)) for i in range(6)]
y2=sum(v*v for v in Y)
# solve subset gaussian
def solve(A,b):
 n=len(b); a=[A[i][:]+[b[i]] for i in range(n)]
 for c in range(n):
  p=max(range(c,n),key=lambda r:abs(a[r][c])); a[c],a[p]=a[p],a[c]
  if abs(a[c][c])<1e-12:return None
  q=a[c][c]
  for j in range(c,n+1):a[c][j]/=q
  for r in range(n):
   if r==c:continue
   q=a[r][c]
   for j in range(c,n+1):a[r][j]-=q*a[c][j]
 return [a[i][n] for i in range(n)]
res=[]
for mask in range(1,64):
 inds=[i for i in range(6) if mask>>i&1]
 co=solve([[G[i][j] for j in inds] for i in inds],[b[i] for i in inds])
 if co and min(co)>=0:
  ss=y2-2*sum(co[a]*b[i] for a,i in enumerate(inds))+sum(co[a]*co[c]*G[i][j] for a,i in enumerate(inds) for c,j in enumerate(inds))
  res.append((ss,inds,co))
for ss,inds,co in sorted(res)[:15]:print('"'fit',ss/y2,[types[i] for i in inds],[round(x,3) for x in co],flush=True)
# marker genes: ratio profile to all others, print target values and expected
for i,t in enumerate(types):
 scores=[]
 for g in range(1000):
  other=sum(M[g][j] for j in range(6) if j"'!=i)/5
  scores.append(((M[g][i]+.1)/(other+.1),g))
 top=sorted(scores,reverse=True)[:15]
 print(t,[(genes[g],round(s,1),round(M[g][i],1),Y[g]) for s,g in top],flush=True)
PY'
