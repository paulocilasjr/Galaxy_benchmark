/bin/bash -lc "OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 python -u - <<'PY'
import pandas as pd, numpy as np
print('imports1',flush=True)
from scipy.io import mmread
from scipy.optimize import nnls
print('imports2',flush=True)
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
md=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
X=mmread('matrix.mtx.gz').tocsr()
bc=pd.read_csv('barcodes.tsv.gz',header=None)[0]
types=md.cell_type.unique().tolist()
print('shape',sc.shape,X.shape,'types',md.cell_type.value_counts().to_dict())
j=np.where(bc.values=='Spot_710-1')[0][0]
y=X[:,j].toarray().ravel()
print('spot index,total,nonzero',j,y.sum(),(y>0).sum())
means=np.column_stack([sc.loc[md.index[md.cell_type==t]].mean(axis=0).values for t in types])
coef,res=nnls(means,y)
print('raw mean res',res,'coef',dict(zip(types,np.round(coef,4))))
print('shares',dict(zip(types,np.round(coef/coef.sum(),4))))
normmeans=np.column_stack([(sc.loc[md.index[md.cell_type==t]].div(sc.loc[md.index[md.cell_type==t]].sum(1),axis=0)).mean(axis=0).values for t in types])
coef,res=nnls(normmeans,y)
print('norm mean res',res,'coef',dict(zip(types,np.round(coef,4))))
print('shares',dict(zip(types,np.round(coef/coef.sum(),4))))
for transformed in ['raw','log']:
 yy=y if transformed=='raw' else np.log1p(y)
 A=means if transformed=='raw' else np.log1p(means)
 co={t:np.corrcoef(A[:,i],yy)[0,1] for i,t in enumerate(types)}
 print(transformed,'corr',dict(sorted(co.items(),key=lambda x:-x[1])))
PY"
