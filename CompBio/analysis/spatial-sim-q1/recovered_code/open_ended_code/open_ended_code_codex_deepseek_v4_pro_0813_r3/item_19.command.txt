/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
import gzip
from scipy.optimize import nnls
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
with gzip.open('matrix.mtx.gz','rt') as f:
 X=mmread(f).toarray().T
types=sorted(meta.cell_type.unique())
ref=pd.DataFrame(sc.values,columns=sc.columns,index=sc.index).groupby(meta['cell_type']).mean()
A=ref.loc[types].values.T
for si in [710,709,711,680,681,682,739,740,741]:
 y=X[si]
 c,_=nnls(A,y)
 print(si,'sum',y.sum(),'raw coefs',dict(zip(types,np.round(c,3))),'round',dict(zip(types,np.round(c))))
PY"
