/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
import gzip
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0).values
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
with gzip.open('matrix.mtx.gz','rt') as f:
 X=mmread(f).toarray().T
# exact single per spot
exact_single=np.array([(sc==y).all(axis=1).any() for y in X])
print('spots exactly equal single cell',exact_single.sum())
# exact pair using candidate sum <= spot sum and meet hash by sum then exact
single={tuple(sc[i]):i for i in range(len(sc))}
count=0
for si,y in enumerate(X):
    if exact_single[si]: count+=1; continue
    found=False
    # iterate all cells with sum <= ysum
    for i in range(len(sc)):
        need=y-sc[i]
        if need.min()>=0:
            j=single.get(tuple(need))
            if j is not None:
                found=True; break
    if found: count+=1
print('spots exactly equal single or pair sc cells',count)
PY"
