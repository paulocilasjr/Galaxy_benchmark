/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
import gzip
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
with gzip.open('matrix.mtx.gz','rt') as f:
 X=mmread(f).toarray().T
types=sorted(meta.cell_type.unique())
means=pd.DataFrame(index=types,columns=sc.columns)
for t in types: means.loc[t]=sc.values[meta.cell_type==t].mean(axis=0)
# find top marker genes by ratio to max others or mean
for t in types:
 others=[u for u in types if u"'!=t]
 ratio=means.loc[t]/(means.loc[others].mean(axis=0)+0.1)
 top=ratio.sort_values(ascending=False).head(15)
 print('"'\\n',t,'top ratio')
 print(list(zip(top.index,[round(v,2) for v in top.values])))
 print('spot counts', [int(X[710,sc.columns.get_loc(g)]) for g in top.index])
 print('mean expr in t', [round(means.loc[t,g],2) for g in top.index])
 # log fold
 # maybe unique genes expression values
PY"
