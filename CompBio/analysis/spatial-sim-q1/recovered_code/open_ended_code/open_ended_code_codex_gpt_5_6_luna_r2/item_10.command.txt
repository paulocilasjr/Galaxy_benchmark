/bin/bash -lc "python3 - <<'PY'
import numpy as np, csv
from scipy.io import mmread
from scipy.optimize import nnls
sc=np.loadtxt('spatial_q_sc_counts.csv',delimiter=',',skiprows=1,dtype=float,usecols=range(1,1001))
sp=mmread('matrix.mtx.gz').tocsr()
target=sp[:,710].toarray().ravel()
with open('spatial_q_sc_metadata.csv',newline='') as f: types=np.array([r['cell_type'] for r in csv.DictReader(f)])
labels=list(dict.fromkeys(types))
profiles=np.array([sc[types==t].mean(0) for t in labels]).T
for name, idx in [('all',np.arange(1000)),('marker',np.arange(300)),('blocks',np.arange(6))]:
 if name=='blocks':
  y=np.array([target[a:a+50].sum() for a in range(0,300,50)])
  A=np.array([[profiles[a:a+50,k].sum() for k in range(6)] for a in range(0,300,50)])
 else:
  y=target[idx]; A=profiles[idx]
 coef,res=nnls(A,y)
 pred=A@coef
 print(name)
 print('coef',dict(zip(labels,np.round(coef,4))))
 print('resnorm',round(res,3),'relative',round(res/np.linalg.norm(y),5),'predtotal',round(pred.sum(),3),'ytotal',round(y.sum(),3))
y=np.array([target[a:a+50].sum() for a in range(0,300,50)])
A=np.array([[profiles[a:a+50,k].sum() for k in range(6)] for a in range(0,300,50)])
print('block observed',y.tolist())
print('block matrix')
print(np.round(A,1))
print('solution block',np.round(nnls(A,y)[0],4))
PY"
