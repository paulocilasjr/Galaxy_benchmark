/bin/bash -lc "OPENBLAS_NUM_THREADS=4 python - <<'PY'
import gzip,pandas as pd,numpy as np
from scipy.io import mmread
from scipy import sparse
g=pd.read_csv('differential.composition.q1.genes.txt.gz',sep='\\t').gene_symbols.astype(str).to_numpy(); d={z:i for i,z in enumerate(g)}
for fn in ['differential.composition.q1.1.mtx.gz','differential.composition.q1.2.mtx.gz']:
 with gzip.open(fn,'rb') as f:x=mmread(f).tocsr().T.astype('float32')
 xn=sparse.diags(1e4/np.maximum(np.asarray(x.sum(1)).ravel(),1))@x; xn.data=np.log1p(xn.data); gm=np.asarray(xn.mean(0)).ravel()
 print('\\n',fn)
 for name,gs,q in [('fib',['COL1A1','COL1A2','DCN','LUM','COL3A1','PDGFRA'],2),('horiz',['ONECUT1','ONECUT2','PROX1','CALB1','LHX1'],3),('lhx1',['LHX1'],1),('endo',['PECAM1','VWF','CLDN5','EMCN','KDR','RAMP2'],2),('peri',['RGS5','PDGFRB','CSPG4','MCAM','DES','ACTA2'],2)]:
  ids=[d[z] for z in gs]; keep=np.asarray((x[:,ids]>0).sum(1)).ravel()>=q
  cm=np.asarray(xn[keep].mean(0)).ravel(); pct=np.asarray((xn[keep]>0).mean(0)).ravel(); sc=(cm-gm)*np.sqrt(pct);sc[pct<.15]=-np.inf
  top=np.argsort(sc)[-20:][::-1]
  print(name,keep.sum(),','.join(g[top]))
PY"
