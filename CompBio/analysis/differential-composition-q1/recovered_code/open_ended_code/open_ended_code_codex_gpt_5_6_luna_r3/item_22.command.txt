/bin/bash -lc "python -u - <<'PY'
import gzip,numpy as np
from scipy.io import mmread
from scipy.sparse import csc_matrix
with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
 next(f); genes=[line.rstrip('\\n').split('\\t')[1] for line in f]
idx={s:i for i,s in enumerate(genes)}
Ms=[]
for fn in ['differential.composition.q1.1.mtx.gz','differential.composition.q1.2.mtx.gz']:
 with gzip.open(fn,'rb') as f: Ms.append(csc_matrix(mmread(f)))
for si,M in enumerate(Ms,1):
 lib=np.asarray(M.sum(0)).ravel(); X=M.multiply(1e4/lib).tocsr().astype(np.float32); X.data=np.log1p(X.data); X=X.T.tocsr()
 print('\\nSAMPLE',si)
 for g in ['MERTK','RPE65','BEST1','SERPINA3','RDH5','MITF','TTR','PMEL','TYR','DCT','LRAT','RLBP1','TRPM3','RPE65']:
  ix=np.where(M[idx[g],:].toarray().ravel()>0)[0]
  if len(ix)==0: continue
  cm=np.asarray(X[ix,:].mean(0)).ravel(); top=np.argsort(cm)[-25:][::-1]
  # count coexpression RPE anchors
  anc=['MERTK','RPE65','BEST1','SERPINA3','RDH5','MITF','TTR','PMEL','TYR','DCT','LRAT','RLBP1']
  co=[]
  for h in anc:
   co.append(f'{h}:{M[idx[h],ix].getnnz(axis=1)}')
  print('GROUP',g,'n',len(ix),'medlib',np.median(lib[ix]),'co',' '.join(co),'top',','.join(f'{genes[j]}:{cm[j]:.1f}' for j in top))
PY"
