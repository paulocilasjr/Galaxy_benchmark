/bin/bash -lc "python -u - <<'PY'
import gzip, numpy as np
from scipy.io import mmread
from scipy.sparse import csc_matrix
Ms=[]
for fn in ['differential.composition.q1.1.mtx.gz','differential.composition.q1.2.mtx.gz']:
 with gzip.open(fn,'rb') as f: Ms.append(csc_matrix(mmread(f)))
for k,M in enumerate(Ms,1):
 totals=np.asarray(M.sum(axis=0)).ravel(); ng=M.getnnz(axis=0)
 print(k,'cells',M.shape[1],'total quantiles',np.percentile(totals,[0,1,5,25,50,75,95,99,100]),'genes',np.percentile(ng,[0,1,5,25,50,75,95,99,100]))
 print('cells total<1000',sum(totals<1000),'genes<300',sum(ng<300))
PY"
