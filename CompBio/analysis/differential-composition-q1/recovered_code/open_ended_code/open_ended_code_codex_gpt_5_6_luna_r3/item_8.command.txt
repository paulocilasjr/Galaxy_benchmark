import gzip, time
from scipy.io import mmread
for f in ['differential.composition.q1.1.mtx.gz','differential.composition.q1.2.mtx.gz']:

    with gzip.open(f,'rb') as h: x=mmread(h)
    print(type(x),x.shape,x.nnz, x.dtype, 'sec',time.time()-t,flush=True)
    print('sum',x.sum(), 'max',x.max(),flush=True)
