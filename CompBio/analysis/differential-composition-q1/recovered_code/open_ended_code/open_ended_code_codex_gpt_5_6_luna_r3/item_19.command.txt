/bin/bash -lc "python -u - <<'PY'
import gzip,numpy as np
from scipy.io import mmread
from scipy.sparse import csc_matrix
with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
 next(f); genes=[line.rstrip('\\n').split('\\t')[1] for line in f]
idx={s:i for i,s in enumerate(genes)}
Ms=[]
for fn in ['differential.composition.q1.1.mtx.gz','differential.composition.q1.2.mtx.gz']:
 with gzip.open(fn,'rb') as f: Ms.append(csc_matrix(mmread(f)))
for si,M in enumerate(Ms,1):
 lib=np.asarray(M.sum(0)).ravel()
 X=M.multiply(1e4/lib).tocsr().astype(np.float32); X.data=np.log1p(X.data)
 print('\\nSAMPLE',si)
 for g in ['LHX1','LHX2','ONECUT1','ONECUT2','CALB1','PROX1','VSX1','GAD1','GAD2','TFAP2A','TFAP2B','FOXP2']:
  i=idx[g]; ix=np.where(M[i,:].toarray().ravel()>0)[0]
  print(g,'n',len(ix),'sum',M[i,:].sum(),'coexpress:',end=' ')
  # percentage of cells with each marker
  for h in ['LHX1','LHX2','ONECUT1','ONECUT2','CALB1','PROX1','VSX1','GAD1','GAD2','TFAP2A','TFAP2B','FOXP2']:
   j=idx[h]; print(h+':'+str(int(M[j,ix].getnnz(axis=1))),end=' ')
  print()
 # top average genes in LHX1+ and each gene? 
 for g in ['LHX1','LHX2','ONECUT1','PROX1','VSX1','CALB1','TFAP2A']:
  ix=np.where(M[idx[g],:].toarray().ravel()>0)[0]
  if len(ix)==0: continue
  cm=np.asarray(X[ix,:].mean(axis=0)).ravel()
  # only top 100 all genes but X is normalized log; compute full means
  top=np.argsort(cm)[-35:][::-1]
  print('GROUP',g,'n',len(ix),'top:',','.join(f'{genes[j]}:{cm[j]:.1f}' for j in top))
PY"
