/bin/bash -lc "python -u - <<'PY'
import gzip, numpy as np
from scipy.io import mmread
from scipy.sparse import csc_matrix
with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
 next(f); genes=[line.rstrip('\\n').split('\\t')[1] for line in f]
Ms=[]
for fn in ['differential.composition.q1.1.mtx.gz','differential.composition.q1.2.mtx.gz']:
 with gzip.open(fn,'rb') as f: Ms.append(csc_matrix(mmread(f)))
# use library-size normalized sums (not log); also prevalence
for metric in ['mean_norm','det']:
 vals=[]
 for M in Ms:
  lib=np.asarray(M.sum(axis=0)).ravel()
  norm=M.multiply(1e4/lib)
  vals.append(np.asarray(norm.mean(axis=1)).ravel() if metric=='mean_norm' else np.asarray(M.getnnz(axis=1)).ravel()/M.shape[1])
 a,b=vals
 # log fold b/a; restrict enough
 score=np.log2((b+1e-4)/(a+1e-4))
 for label,sc in [('sample2 higher',score),('sample1 higher',-score)]:
  inds=np.argsort(sc)[-50:][::-1]
  print('\\n',metric,label)
  print(' '.join(f'{genes[i]}:{sc[i]:.2f}' for i in inds[:50]))
PY"
