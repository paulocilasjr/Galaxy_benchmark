/bin/bash -lc "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('processed.h5ad')
# raw counts? adata.X is normalized log1p. We'll count positive threshold
markers=['ONECUT1','ONECUT2','LHX1','TNR','SEMA5A','CALB1','C1QL1','PRDM13']
sub=adata[:, [g for g in markers if g in adata.var_names]]
X=sub.X.toarray()
for g in sub.var_names:
    d=pd.DataFrame({'x':X[:,sub.var_names==g].ravel() if False else X[:,list(sub.var_names).index(g)], 'ind':adata.obs.individual.values})
    print(g, (d['x']>0).groupby(d['ind']).sum().to_dict(), 'cells with >0.5', (d['x']>0.5).groupby(d['ind']).sum().to_dict())
# score sum normalized markers
score=X.sum(axis=1)
d=pd.DataFrame({'score':score,'ind':adata.obs.individual.values,'leiden':adata.obs.leiden.values})
for thr in [0.5,1,2,3,5]:
 print('score>=',thr,d[d.score>=thr].groupby('ind').size().to_dict(), d[d.score>=thr].groupby('ind').apply(lambda x:x.leiden.value_counts().to_dict(), include_groups=False).to_dict())
PY"
