/bin/bash -lc "python -u - <<'PY'
import gzip,numpy as np
from scipy.io import mmread
p='differential.composition.q1.'
g=[]
with gzip.open(p+'genes.txt.gz','rt') as f:
 next(f)
 for l in f:g.append(l.rstrip().split('\\t')[1])
for k in ['1','2']:
 x=mmread(p+k+'.mtx.gz').tocsr().astype(np.float64)
 s=np.asarray(x.sum(1)).ravel(); lib=x.sum(); rate=s/lib*1e6
 print('\\nSAMPLE',k,'total',lib)
 ix=np.argsort(rate)[::-1]
 print('top abundant')
 for i in ix[:100]: print(g[i],round(rate[i],1),int(s[i]))
# diff pseudobulk
x1=mmread(p+'1.mtx.gz').tocsr().astype(np.float64); x2=mmread(p+'2.mtx.gz').tocsr().astype(np.float64)
s1=np.asarray(x1.sum(1)).ravel();s2=np.asarray(x2.sum(1)).ravel();
# log CPM with pseudocount around 0.1 read per million? log2((s2+1)/(lib2+...)/(s1+1)...)
lib1=float(x1.sum());lib2=float(x2.sum())
logfc=np.log2((s2+0.5)/(lib2+0.5e6*1e-6)/(s1+0.5)/(lib1+0.5e6*1e-6)) # wrong denominators? ratio rate: (s2+0.5)/(lib2+0.5) / ((s1+.5)/(lib1+.5))
logfc=np.log2((s2+0.5)/(lib2+0.5)*((lib1+0.5)/(s1+0.5)))
# rank among genes with at least 10 counts across
mask=(s1+s2)>=20
for lab,ix in [('sample2 higher',np.where(mask)[0][np.argsort(logfc[mask])[::-1]]),('sample1 higher',np.where(mask)[0][np.argsort(logfc[mask])])]:
 print('\\n',lab)
 for i in ix[:100]:print(g[i], 'logfc',round(logfc[i],2),'c1',int(s1[i]),'c2',int(s2[i]),'rate1',round(s1[i]/lib1*1e6,1),'rate2',round(s2[i]/lib2*1e6,1))
PY"
