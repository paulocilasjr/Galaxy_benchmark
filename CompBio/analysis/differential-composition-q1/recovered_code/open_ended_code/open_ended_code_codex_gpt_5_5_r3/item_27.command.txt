'microglial cell':['P2RY12','CX3CR1','TMEM119','AIF1','C1QA','C1QB','TYROBP','SPI1'],

}

 rows.append((ct,med,mean,'; '.join(f'{m}:{lr:.2f}' for m,a,b,lr in sorted(rr,key=lambda x:x[3])[:5])))
