/bin/bash -lc "python3 -u - <<'PY'
import gzip, scipy.io, numpy as np, re
from scipy.sparse import vstack
from sklearn.decomposition import TruncatedSVD
from sklearn.cluster import KMeans
base='differential.composition.q1'
with gzip.open(base+'.genes.txt.gz','rt') as f:
 f.readline(); genes=np.array([line.rstrip('\\n').split('\\t')[1] for line in f])
A=[]; batches=[]
for j in [1,2]:
 m=scipy.io.mmread(base+f'.{j}.mtx.gz').tocsr().astype(np.float32); lib=np.asarray(m.sum(axis=0)).ravel(); x=m.multiply(1e4/(lib+1e-9)).tocsr(); x.data=np.log1p(x.data); A.append(x.T.tocsr()); batches += [j]*m.shape[1]
X=vstack(A).tocsr(); batches=np.array(batches)
mu=np.asarray(X.mean(axis=0)).ravel(); mu2=np.asarray(X.multiply(X).mean(axis=0)).ravel(); var=mu2-mu*mu
valid=(mu>0.02)&(mu<8)&~np.array([bool(re.match(r'"'^(MT-|RPS|RPL|MALAT1$)'"',g)) for g in genes])
hvg=np.argsort(np.where(valid,var,-np.inf))[-3000:]
pc=TruncatedSVD(n_components=40,random_state=1,n_iter=3).fit_transform(X[:,hvg])
pc=pc/np.std(pc,axis=0)
markers={
'astrocyte':['GFAP','AQP4','SLC1A3','CLDN11','VIM','GJA1','ALDOC','S100B'],
'microglial cell':['AIF1','C1QA','C1QB','C1QC','TYROBP','P2RY12','TMEM119','CX3CR1','CSF1R','C3'],
'RPE cell':['RPE65','BEST1','RLBP1','PMEL','TYR','MERTK','LRAT','MITF','SERPINF1','TTR'],
'cone cell':['ARR3','OPN1SW','OPN1MW','OPN1LW','GNGT2','PDE6H','GUCA1C','CNGB3','GNAT2','RXRG'],
'rod cell':['RHO','SAG','GNAT1','PDE6A','PDE6B','NRL','NR2E3','ROM1','GPR37L1','GUCA1B'],
'horizontal cell':['LHX1','ONECUT1','ONECUT2','CALB1','VSX1','GAD1','GAD2','SLC32A1'],
'bipolar cell':['VSX2','GRM6','CABP5','PRKCA','PCP4','TRPM1','GNG13','ISL1','SLC24A2'],
'RGC':['RBPMS','SNCG','POU4F1','POU4F2','THY1','NEFL','NEFM','ISL1','BRN3A','OPN4'],
'T cells':['CD3D','CD3E','TRBC1','TRBC2','CD2','IL7R','CCL5','LTB'],
'macrophage':['LYZ','CTSD','FCER1G','TYROBP','CD68','C1QC','LST1','MS4A7','AIF1','LGALS3'],
'Schwann cell':['S100B','SOX10','MPZ','PLP1','GFAP','NCAM1','FABP7','CLDN11'],
'Pericyte':['RGS5','CSPG4','MCAM','PDGFRB','ACTA2','COL4A1','COL4A2','NOTCH3','ABCC9','KCNJ8'],
'B cell':['CD79A','MS4A1','CD37','CD74','HLA-DRA','CD79B','CD83','CD22','CD69','CD48'],
'fibroblast':['COL1A1','COL1A2','DCN','LUM','COL3A1','COL6A1','COL6A2','PDGFRA','FBLN1'],
'endothelial cell':['EMCN','KDR','ESAM','PECAM1','VWF','CLDN5','RAMP2','ENG','PLVAP'],
'muller glia cell':['RLBP1','GLUL','SLC1A3','SOX9','AQP4','VIM','KCNJ10','SLC1A6','FABP7']}
mi={g:i for i,g in enumerate(genes)}; typenames=list(markers)
scores=np.zeros((X.shape[0],len(markers)),np.float32)
for k,(typ,gs) in enumerate(markers.items()):
 ids=[mi[g] for g in gs if g in mi]; scores[:,k]=X[:,ids].toarray().mean(axis=1)
for K in [25,40,60]:
 lab=KMeans(n_clusters=K,n_init=5,random_state=1,max_iter=100).fit_predict(pc[:,:30])
 counts=np.array([[np.sum((lab==z)&(batches==b)) for b in [1,2]] for z in range(K)])
 cscore=np.array([scores[lab==z].mean(axis=0) for z in range(K)])
 print('\\nK',K,'total')
 for z in np.argsort(-counts.sum(axis=1)):
  best=np.argsort(-cscore[z])[:5]
  print(z,counts[z].tolist(),'|',','.join(f'{typenames[q]}={cscore[z,q]:.2f}' for q in best))
PY"
