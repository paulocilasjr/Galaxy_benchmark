/bin/bash -lc "cat > load_and_qc.R <<'EOF'
suppressMessages({library(Matrix); library(Seurat)})
genes <- read.table(gzfile('differential.composition.q1.genes.txt.gz'), header=TRUE, sep='\\t', stringsAsFactors=FALSE, comment.char='')
rownames(genes) <- genes"'$gene_ids
cat('"'genes dim', dim(genes), '\\n')
X1 <- readMM(gzfile('differential.composition.q1.1.mtx.gz'))
X2 <- readMM(gzfile('differential.composition.q1.2.mtx.gz'))
cat('X dims', dim(X1), dim(X2), '\\n')
# ensure row order same as genes
stopifnot(nrow(X1)==nrow(genes), nrow(X2)==nrow(genes))
rn <- genes"'$gene_ids
colnames(X1) <- paste0('"'s1_', seq_len(ncol(X1)))
colnames(X2) <- paste0('s2_', seq_len(ncol(X2)))
rownames(X1) <- rn; rownames(X2) <- rn
# Basic per-cell summaries
cs1 <- Matrix::colSums(X1); cs2 <- Matrix::colSums(X2)
cat('sample counts mean/median/total:', mean(cs1), median(cs1), sum(cs1), ' / ', mean(cs2), median(cs2), sum(cs2), '\\n')
# gene detection
cat('genes detected sample1 >=1:', sum(Matrix::rowSums(X1)>0), ' sample2:', sum(Matrix::rowSums(X2)>0), '\\n')
# create seurat object maybe with raw matrices, keep all genes? Create can filter features with min.cells=3
s1 <- CreateSeuratObject(counts=X1, project='s1', min.cells=0, min.features=0)
s2 <- CreateSeuratObject(counts=X2, project='s2', min.cells=0, min.features=0)
s1"'$sample <- '"'s1'; s2"'$sample <- '"'s2'
s1"'$percent.mt <- PercentageFeatureSet(s1, pattern='"'"'^MT-'"')
s2"'$percent.mt <- PercentageFeatureSet(s2, pattern='"'"'^MT-'"')
print(summary(s1"'$nCount_RNA)); print(summary(s1$nFeature_RNA)); print(summary(s1$percent.mt))
print(summary(s2$nCount_RNA)); print(summary(s2$nFeature_RNA)); print(summary(s2$percent.mt))
saveRDS(list(X1=X1,X2=X2,s1=s1,s2=s2,genes=genes), file='"'merged_input.rds')
cat('saved\\n')
EOF
Rscript load_and_qc.R"
