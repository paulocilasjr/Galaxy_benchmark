/bin/bash -lc "python - <<'PY'
import gzip, scipy.io, numpy as np, scipy.sparse as sp
# load genes
genes=[]
with gzip.open('differential.composition.q1.genes.txt.gz','rt') as f:
    next(f)
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        genes.append(p[1] if len(p)>1 else p[0])
genes=np.array(genes)
X1=sp.load_npz('X1.npz').tocsc(); X2=sp.load_npz('X2.npz').tocsc()
sets={
'astrocyte':['GFAP','AQP4','SLC1A3','GJA1','S100B'],
'microglial cell':['C1QA','C1QB','C1QC','CSF1R','PTPRC','AIF1','TYROBP','CX3CR1'],
'RPE cell':['RPE65','BEST1','RLBP1','TTR','SERPINF1','PMEL','SLC16A8','MITF'],
'cone cell':['ARR3','PDE6H','GNAT2','OPN1SW','OPN1MW','OPN1LW','GUCA1A','GNGT2','CNGB3'],
'rod cell':['RHO','NRL','PDE6A','PDE6B','CNGA1','NR2E3','GNAT1','SAG','RCVRN'],
'horizontal cell':['ONECUT1','LHX1','CALB2','SLC17A8','TNR','SNTG2','CARTPT','PRKG1','LRRTM4','SPOCK3','GAD1','GAD2'],
'bipolar cell':['VSX2','CABP5','GRIK1','TRPM1','GRM6','ISL1','PRKCA','GNG13','CHAT','SLC17A6','OTX2'],
'RGC':['RBPMS','SNCG','NEFL','NEFM','POU4F2','THY1','SLIT2','ISL1','SLC17A6','MEG3'],
'T cells':['CD3D','CD3E','CD3G','TRAC','IL7R','CCL5','CD2'],
'macrophage':['CD68','LYZ','C1QA','C1QB','C1QC','CD163','ITGAX'],
'Schwann cell':['MPZ','MBP','SOX10','PLP1','NGFR','S100B'],
'Pericyte':['RGS5','NOTCH3','PDGFRB','ACTA2','TAGLN','COX4I2'],
'B cell':['CD79A','MS4A1','CD19','BANK1','PAX5','CD74'],
'fibroblast':['DCN','COL1A1','LUM','COL1A2','SERPINE1','PDGFRA'],
'endothelial cell':['PECAM1','VWF','CLDN5','FLT1','KDR','CDH5','EGFL7'],
'muller glia cell':['RLBP1','GLUL','SOX9','VIM','SLC1A3','DKK3','APOE','CLU','TF']}
for name, marks in sets.items():
    idx=[i for i,g in enumerate(genes) if g in marks]
    if not idx: continue
    for X,lab in [(X1,'s1'),(X2,'s2')]:
        sub=X[idx,:]
        # percent cells with >=1 count in any marker, and mean UMIs per cell (any marker)
        anyexpr=(sub.sum(axis=0)>0)
        total=sub.sum()
        print(f'{name:20s} {lab}: pct_expr={np.asarray(anyexpr).mean()*100:.2f}% meanUMI={total/X.shape[1]:.3f} totalUMI={int(total)}')
    print()
PY"
