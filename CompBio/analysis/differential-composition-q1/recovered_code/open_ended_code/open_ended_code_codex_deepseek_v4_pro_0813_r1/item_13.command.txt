/bin/bash -lc "python - <<'PY'
import csv, collections
# read markers_all with pandas unavailable? use csv
rows=[]
with open('markers_all.csv') as f:
    r=csv.reader(f)
    header=next(r)
    for row in r:
        rows.append(row)
print('header', header)
print('rows', len(rows))
# identify columns
for i,h in enumerate(header): print(i,h)
# group top 20 per cluster by avg_log2FC sorted
clusters=collections.defaultdict(list)
for row in rows:
    cluster=row[header.index('cluster')]
    clusters[cluster].append(row)
for cl in sorted(clusters, key=lambda x:int(x)):
    rows2=sorted(clusters[cl], key=lambda r: float(r[header.index('avg_log2FC')]), reverse=True)[:20]
    print('\\nCLUSTER', cl, 'n markers', len(clusters[cl]))
    for row in rows2:
        print(row[header.index('gene')], row[header.index('avg_log2FC')], row[header.index('pct.1')], row[header.index('pct.2')], row[header.index('p_val_adj')])
PY"
