/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
integrated[['RNA']] <- JoinLayers(integrated[['RNA']])
DefaultAssay(integrated) <- 'RNA'
markers <- c('RPE65','BEST1','TTR','PMEL','MLANA','DCT','TYRP1','SERPINF1','RGR','OTX1')
avg <- AverageExpression(integrated, assays='RNA', features=markers, group.by='seurat_clusters', slot='data')
mat <- as.matrix(avg"'$RNA)
print(round(mat,3))
# Cells with RPE65>0 and MLANA>0 etc
expr <- GetAssayData(integrated, assay='"'RNA', layer='data')
mask <- expr['RPE65',] > 0 & expr['MLANA',] > 0
cat('\\nRPE65>0 & MLANA>0 counts by sample/cluster\\n')
print(table(integrated"'$sample[mask], integrated$seurat_clusters[mask]))
mask2 <- expr['"'RPE65',] > 0.5 & (expr['MLANA',] > 0.5 | expr['PMEL',] > 0.5)
cat('\\nRPE65>.5 & (MLANA>.5|PMEL>.5)\\n')
print(table(integrated"'$sample[mask2], integrated$seurat_clusters[mask2]))
# expression values for all RPE65 positive cells sample and cluster
sub <- which(expr['"'RPE65',] > 0)
cat('RPE65 positive cells by sample', table(integrated"'$sample[sub]), '"'\\n')
print(round(head(sort(expr['RPE65',sub], decreasing=TRUE)),2))
EOF
Rscript [REDACTED LOCAL PATH]"
