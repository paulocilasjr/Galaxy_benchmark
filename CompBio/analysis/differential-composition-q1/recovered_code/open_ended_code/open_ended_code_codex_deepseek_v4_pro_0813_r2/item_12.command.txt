/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
suppressPackageStartupMessages(library(Matrix))
x <- readRDS('mats.rds')
obj1 <- CreateSeuratObject(counts=x"'$mat1, project='"'s1', min.cells=0, min.features=0)
obj2 <- CreateSeuratObject(counts=x"'$mat2, project='"'s2', min.cells=0, min.features=0)
obj1"'$sample <- '"'sample1'; obj2"'$sample <- '"'sample2'
obj1 <- NormalizeData(obj1, verbose=FALSE)
obj2 <- NormalizeData(obj2, verbose=FALSE)
obj1 <- FindVariableFeatures(obj1, selection.method='vst', nfeatures=2000, verbose=FALSE)
obj2 <- FindVariableFeatures(obj2, selection.method='vst', nfeatures=2000, verbose=FALSE)
features <- SelectIntegrationFeatures(object.list=list(obj1,obj2), nfeatures=2000)
cat('integration features', length(features), '\\n')
# use reciprocal PCA integration (fast)
obj1 <- ScaleData(obj1, features=features, verbose=FALSE)
obj2 <- ScaleData(obj2, features=features, verbose=FALSE)
obj1 <- RunPCA(obj1, features=features, npcs=30, verbose=FALSE)
obj2 <- RunPCA(obj2, features=features, npcs=30, verbose=FALSE)
anchors <- FindIntegrationAnchors(object.list=list(obj1,obj2), anchor.features=features, reduction='rpca', dims=1:30, k.anchor=5, verbose=TRUE)
integrated <- IntegrateData(anchorset=anchors, dims=1:30, verbose=TRUE)
DefaultAssay(integrated) <- 'integrated'
integrated <- ScaleData(integrated, verbose=FALSE)
integrated <- RunPCA(integrated, npcs=30, verbose=FALSE)
integrated <- RunUMAP(integrated, reduction='pca', dims=1:30, verbose=FALSE)
integrated <- FindNeighbors(integrated, reduction='pca', dims=1:30, verbose=FALSE)
integrated <- FindClusters(integrated, resolution=0.8, verbose=FALSE)
saveRDS(integrated, 'integrated.rds')
cat('clusters\\n')
print(table(integrated"'$seurat_clusters, integrated$sample))
cat('"'done\\n')
EOF
Rscript [REDACTED LOCAL PATH]"
