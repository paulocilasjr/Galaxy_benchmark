/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
DefaultAssay(integrated) <- 'RNA'
sets <- list(
 Rod=c('RHO','NRL','NR2E3','GNAT1','PDE6B','SAG'),
 Cone=c('ARR3','GNAT2','PDE6H','OPN1SW','OPN1MW','OPN1LW'),
 Bipolar=c('VSX2','GRM6','TRPM1','ISL1','GRIK1','VSX1','LHX4'),
 RGC=c('RBPMS','SLC17A6','POU4F1','POU4F2','THY1','NEFL','NEFM','SNCG'),
 Muller=c('RLBP1','GLUL','AQP4','SOX9','SLC1A3','CLU','DKK3'),
 Horizontal=c('ONECUT1','ONECUT2','LHX1','CALB1'),
 Microglia=c('C1QA','C1QB','C1QC','CSF1R','P2RY12','TYROBP','CX3CR1','AIF1'),
 Tcell=c('CD3D','CD3E','CD3G','TRAC','LCK','IL7R'),
 Bcell=c('MS4A1','CD79A','CD19','BANK1'),
 Endothelial=c('PECAM1','VWF','CLDN5','FLT1','KDR','EGFL7'),
 Pericyte=c('PDGFRB','RGS5','NOTCH3','ACTA2','CSPG4'),
 Fibroblast=c('DCN','COL1A1','COL1A2','LUM','COL3A1','COL6A1','COL6A2','PDGFRA'),
 RPE=c('RPE65','BEST1','TTR','PMEL','MLANA','DCT','SERPINF1','RGR'),
 Schwann=c('MPZ','PLP1','MBP','SOX10','GAP43'),
 Astrocyte=c('GFAP','GJA1','ALDH1L1','SLC1A2','VIM')
)
# remove genes not present
sets <- lapply(sets, function(g) g[g %in% rownames(integrated)])
for(nm in names(sets)) integrated[[paste0('score_',nm)]] <- NULL
for(nm in names(sets)){
 integrated <- AddModuleScore(integrated, features=sets[nm], name=paste0('score_',nm), assay='RNA', nbin=24, ctrl=100, seed=42)
 # AddModuleScore name adds 1 suffix
}
# gather columns, names are score_name1
scorecols <- paste0('score_', names(sets), '1')
# ensure no missing; replace NA
meta <- integrated@meta.data
# classify max score
smat <- as.matrix(meta[, scorecols, drop=FALSE])
colnames(smat) <- names(sets)
best <- colnames(smat)[max.col(smat, ties.method='first')]
meta"'$pred <- best
tab <- table(meta$pred, meta$sample)
print(tab)
print(round(prop.table(tab,2)*100,2))
# assign cluster to pred mode for confusion
print(table(meta$seurat_clusters, meta$pred))
# save meta
write.csv(meta, '"'pred_meta.csv')
# show horizontal score distribution and counts per sample
cat('\\nHorizontal score summary by sample\\n')
print(tapply(meta"'$score_Horizontal1, meta$sample, summary))
cat('"'\\nCells with horizontal score > 0.5?\\n')
print(table(meta"'$pred == '"'Horizontal', meta"'$sample))
cat('"'\\nCluster 11 preds\\n')
print(table(meta"'$pred[meta$seurat_clusters==11]))
cat('"'\\nCluster 4 preds\\n')
print(table(meta"'$pred[meta$seurat_clusters==4]))
cat('"'\\nCluster 6 preds\\n')
print(table(meta"'$pred[meta$seurat_clusters==6]))
EOF
Rscript [REDACTED LOCAL PATH]'
