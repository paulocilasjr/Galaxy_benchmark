/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
DefaultAssay(integrated) <- 'RNA'
# markers used
markers <- c('ONECUT1','ONECUT2','LHX1','CALB1','SLC17A8','GAD1','TFAP2A','SLC6A9','CALB2','SLC32A1',
 'RBPMS','SLC17A6','POU4F1','THY1','NEFL','NEFM','SNCG',
 'RLBP1','GLUL','AQP4','SOX9','SLC1A3','CLU','DKK3','S100A6',
 'RHO','NRL','NR2E3','GNAT1','PDE6B','SAG',
 'ARR3','GNAT2','PDE6H','OPN1LW','OPN1SW','OPN1MW',
 'VSX2','GRM6','TRPM1','ISL1','GRIK1','VSX1','LHX4','OTX2',
 'C1QA','C1QB','C1QC','CSF1R','P2RY12','TYROBP','CX3CR1','AIF1',
 'CD3D','CD3E','CD3G','TRAC','LCK','IL7R',
 'MS4A1','CD79A','CD19','BANK1',
 'PECAM1','VWF','CLDN5','FLT1','KDR','EGFL7',
 'PDGFRB','RGS5','NOTCH3','ACTA2','CSPG4',
 'DCN','COL1A1','COL1A2','LUM','COL3A1','COL6A1','COL6A2','PDGFRA',
 'RPE65','BEST1','TTR','PMEL','MLANA','DCT','SERPINF1','RGR',
 'MPZ','PLP1','MBP','SOX10','GAP43','S100B',
 'GFAP','GJA1','ALDH1L1','SLC1A2','VIM')
present <- markers[markers %in% rownames(integrated)]
avg <- AverageExpression(integrated, assays='RNA', features=present, group.by='seurat_clusters', slot='data')
mat <- as.matrix(avg"'$RNA)
write.csv(mat,'"'avg_markers_all.csv')
# For each cluster print top 20 by average expression among present, with values
clusters <- levels(integrated"'$seurat_clusters)
for(cl in clusters){
 col <- paste0('"'g',cl)
 vals <- mat[,col]
 ord <- order(vals, decreasing=TRUE)
 cat('\\n### Cluster', cl, 'size', sum(integrated"'$seurat_clusters==cl), '"'sample', paste(names(table(integrated"'$sample[integrated$seurat_clusters==cl])), table(integrated$sample[integrated$seurat_clusters==cl]), collapse='"' '), '\\n')
 print(round(vals[ord][1:20],3))
}
EOF
Rscript [REDACTED LOCAL PATH]"
