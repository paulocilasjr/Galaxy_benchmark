/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
suppressPackageStartupMessages(library(Seurat))
integrated <- readRDS('integrated.rds')
DefaultAssay(integrated) <- 'RNA'
# use normalized data matrix for genes of interest, clusters
markers <- c(
 'RHO','NRL','NR2E3','GNAT1','PDE6B','SAG', # rods
 'ARR3','GNAT2','PDE6H','OPN1SW','OPN1MW','OPN1LW', # cones
 'VSX2','OTX2','GRM6','TRPM1','ISL1','GRIK1','VSX1','GAD2','LHX4', # bipolar? 
 'RBPMS','SLC17A6','POU4F1','POU4F2','THY1','NEFL','NEFM','SNCG', # RGC
 'RLBP1','GLUL','AQP4','SOX9','SLC1A3','CLU','DKK3','S100A6', # muller
 'TFAP2A','GAD1','SLC6A9','CALB2','SLC32A1', # amacrine (not list)
 'ONECUT1','ONECUT2','LHX1','CALB1','SLC17A8', # horizontal
 'C1QA','C1QB','C1QC','CSF1R','P2RY12','TYROBP','CX3CR1','AIF1', # microglia
 'CD3D','CD3E','CD3G','TRAC','LCK','IL7R', # T
 'MS4A1','CD79A','CD19','BANK1', # B
 'PECAM1','VWF','CLDN5','FLT1','KDR','EGFL7', # endothelial
 'PDGFRB','RGS5','NOTCH3','ACTA2','CSPG4','DES', # pericyte
 'DCN','COL1A1','COL1A2','LUM','COL3A1','COL6A1','COL6A2','PDGFRA', # fibroblast
 'RPE65','BEST1','TTR','PMEL','MLANA','DCT','SERPINF1','RGR','OTX1','TYRP1', # RPE
 'MPZ','S100B','PLP1','MBP','SOX10','GAP43', # Schwann
 'GFAP','GJA1','SLC1A2','ALDH1L1','S100B','VIM' # astrocyte
)
present <- markers[markers %in% rownames(integrated)]
cat('missing:', setdiff(markers, present), '\\n')
avg <- AverageExpression(integrated, assays='RNA', features=present, group.by='seurat_clusters', slot='data')
mat <- as.matrix(avg"'$RNA)
# print transposed rounded? markers rows, clusters cols. Show top cluster patterns perhaps.
# write csv
write.csv(mat, '"'avg_markers.csv')
# print selected clusters 4,6,11 for all markers, sorted
for(cl in c('4','6','11','0','1','2','3','5')){
 cat('\\n### Cluster', cl, '\\n')
 vals <- mat[,cl,drop=FALSE]
 ord <- order(vals[,1], decreasing=TRUE)
 print(round(vals[ord,,drop=FALSE][1:min(40,nrow(vals)),,drop=FALSE],3))
}
cat('\\nSizes\\n')
print(table(integrated"'$seurat_clusters, integrated$sample))
EOF
