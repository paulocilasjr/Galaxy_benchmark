/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bc={}
for line in open('work/id_bc.tsv'):
 p=line.rstrip().split('\\t'); bc[p[0]]=p[1]
# assign cell species using primary best as before
cc=collections.defaultdict(lambda: collections.Counter())
# store top transcripts per species
trans=collections.defaultdict(lambda: collections.Counter())
for rec in pysam.AlignmentFile('work/aln.sam','r'):
 if rec.is_unmapped or rec.is_supplementary or rec.is_secondary: continue
 sp=rec.reference_name.split('|')[0]
 b=bc.get(rec.query_name)
 if b and sp in ('hg','mm','ss'):
  cc[b][sp]+=1
  trans[sp][rec.reference_name]+=1
assign={}
for b,c in cc.items(): assign[b]=c.most_common(1)[0][0]
for sp in ['hg','mm','ss']:
 print('SPECIES',sp,'cells',sum(1 for x in assign.values() if x==sp))
 print('top transcripts:')
 for name,cnt in trans[sp].most_common(15):
  print(' ',cnt,name)
PY"
