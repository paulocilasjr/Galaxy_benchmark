/bin/bash -lc "python3 - <<'PY'
import pysam
counts={'total':0,'unmapped':0,'mapped_primary':0,'secondary':0,'supp':0}
for rec in pysam.AlignmentFile('work/aln.sam','r'):
 counts['total']+=1
 if rec.is_unmapped: counts['unmapped']+=1
 else:
  if rec.is_secondary: counts['secondary']+=1
  else: counts['mapped_primary']+=1
  if rec.is_supplementary: counts['supp']+=1
print(counts)
PY"
