/bin/bash -lc "python3 - <<'PY'
import collections
bc={}; umi={}
for line in open('work/id_bc.tsv'):
 p=line.rstrip().split('\\t'); bc[p[0]]=p[1]
for line in open('work/id_umi.tsv'):
 p=line.rstrip().split('\\t'); umi[p[0]]=p[1]
cc=collections.defaultdict(lambda: collections.defaultdict(set))
for line in open('work/aln.sam'):
 if line.startswith('@'): continue
 f=line.rstrip().split('\\t'); flag=int(f[1])
 if flag & 0x4: continue
 if flag & 0x800: continue
 if flag & 0x100: continue
 sp=f[2].split('|')[0]
 b=bc.get(f[0]); u=umi.get(f[0])
 if b and u and sp in ('hg','mm','ss'):
  cc[b][sp].add(u)
assign=[]
for b,d in cc.items():
 counts={sp:len(s) for sp,s in d.items()}
 sp=max(counts, key=counts.get)
 assign.append(sp)
print('cells',len(assign),'counts',collections.Counter(assign))
print('lowest UMI cells')
low=sorted(cc.items(), key=lambda kv: sum(len(s) for s in kv[1].values()))[:10]
for b,d in low:
 print(b, {sp:len(s) for sp,s in d.items()})
PY"
