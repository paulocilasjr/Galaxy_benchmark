/bin/bash -c "python3 - <<'PY'
from collections import Counter
for f in ['cell_counts_i80_m20.tsv','cell_counts_i70_m20.tsv','cell_counts_i90_m20.tsv','cell_counts_i95_m40.tsv','cell_species_counts.tsv']:
    rows=[]
    for line in open(f):
        if line.startswith('barcode'): continue
        b,h,m,p=line.split(); rows.append((int(h),int(m),int(p)))
    assign=Counter(); amb=0; zero=0
    for h,m,p in rows:
        arr=[h,m,p]; mx=max(arr)
        ties=[i for i,x in enumerate(arr) if x==mx]
        if mx==0: zero+=1
        elif len(ties)>1: amb+=1
        else: assign[['human','mouse','pig'][ties[0]]]+=1
    print(f, 'n',len(rows), 'assign',dict(assign),'amb',amb,'zero',zero,
          'pct', {k:round(v/len(rows)*100) for k,v in assign.items()},
          'total class',sum(h+m+p for h,m,p in rows))
PY"
