/bin/bash -c "python3 - <<'PY'
import gzip, collections
bc={}
with gzip.open('three.way.barnyard.q1.R1.fq.gz','rt') as f:
    for i,line in enumerate(f):
        if i%4==0:
            name=line[1:].strip().split()[0]
            if name.endswith('/1'): name=name[:-2]
        elif i%4==1:
            bc[name]=line.strip()[:16]
# group PAF by read, retain best score by species and mapq
from collections import defaultdict, Counter
best_species=defaultdict(lambda: -1)
best_mapq=defaultdict(int)
seen=set()
for line in open('mapped.paf'):
    f=line.split('\\t')
    if len(f)<12: continue
    q=f[0]
    q=q[:-2] if q.endswith('/2') else q
    sp=f[5].split('|')[0]
    if sp not in ('human','mouse','pig'): continue
    matches=int(f[9]); block=int(f[10]); mapq=int(f[11])
    if mapq<=0: continue
    score=matches
    if score>best_species[q]:
        best_species[q]=score; best_mapq[q]=mapq; best_sp[q]=sp
# assignment cell
cell=defaultdict(Counter)
for q,sp in best_sp.items():
    b=bc.get(q)
    if b: cell[b][sp]+=1
assign=Counter()
amb=0; zero=0
for b,c in cell.items():
    arr=[c['human'],c['mouse'],c['pig']]
    mx=max(arr); ties=[i for i,x in enumerate(arr) if x==mx]
    if mx==0: zero+=1
    elif len(ties)>1: amb+=1
    else: assign[['human','mouse','pig'][ties[0]]]+=1
print('classified reads',sum(cell[b]['human']+cell[b]['mouse']+cell[b]['pig'] for b in cell), 'cells',len(cell))
print('assign',assign,'amb',amb,'zero',zero,'pct', {k:round(v/len(cell)*100) for k,v in assign.items()})
PY"
