/bin/bash -c "python3 - <<'PY'
import collections
rows=[]
for line in open('cell_species_counts.tsv'):
    if line.startswith('barcode'): continue
    b,h,m,p=line.split()
    rows.append((b,int(h),int(m),int(p)))
print('cells',len(rows),'total classified',sum(h+m+p for _,h,m,p in rows))
for name,idx in [('human',1),('mouse',2),('pig',3)]:
    vals=[r[idx] for r in rows]
    print(name,'total reads',sum(vals),'mean',sum(vals)/len(rows),'max',max(vals))
# majority assignment
from collections import Counter
assign=Counter(); ambiguous=[]
for b,h,m,p in rows:
    arr=[h,m,p]
    mx=max(arr)
    ties=[i for i,x in enumerate(arr) if x==mx]
    if len(ties)>1 or mx==0:
        ambiguous.append((b,h,m,p))
    else:
        assign[['human','mouse','pig'][ties[0]]]+=1
print('majority',assign,'ambiguous',len(ambiguous),'pct', {k:round(v/len(rows)*100) for k,v in assign.items()})
print('ambig examples',ambiguous[:10])
# top barcode counts
print('top total', sorted(rows,key=lambda r:-(r[1]+r[2]+r[3]))[:10])
PY"
