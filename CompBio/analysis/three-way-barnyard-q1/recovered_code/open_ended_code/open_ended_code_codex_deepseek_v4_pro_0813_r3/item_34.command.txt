/bin/bash -c "python3 - <<'PY'
from collections import Counter
rows=[]
for line in open('cell_species_counts.tsv'):
    if line.startswith('barcode'): continue
    b,h,m,p=line.split()
    rows.append((b,int(h),int(m),int(p)))
assign=Counter(); ambiguous=[]; zero=0
for b,h,m,p in rows:
    arr=[h,m,p]; mx=max(arr)
    ties=[i for i,x in enumerate(arr) if x==mx]
    if mx==0: zero+=1
    elif len(ties)>1: ambiguous.append((b,arr))
    else: assign[['human','mouse','pig'][ties[0]]]+=1
print('cells',len(rows),'assigned',sum(assign.values()),'zero',zero,'ambig',len(ambiguous))
print(assign, {k:round(v/len(rows)*100) for k,v in assign.items()})
print('ambiguous',ambiguous[:20])
PY"
