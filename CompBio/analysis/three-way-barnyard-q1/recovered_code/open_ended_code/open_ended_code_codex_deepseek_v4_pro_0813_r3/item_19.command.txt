/bin/bash -c "python3 - <<'PY'
import gzip, os
species = {'human':'homo_sapiens','mouse':'mus_musculus','pig':'sus_scrofa'}
paths = [('human','refs/human.cdna.fa.gz'),('mouse','refs/mouse.cdna.fa.gz'),('pig','refs/pig.cdna.fa.gz')]
out = open('refs/combined.cdna.fa','w')
for sp,path in paths:
    print('processing',sp,flush=True)
    n=0
    with gzip.open(path,'rt') as f:
        for line in f:
            if line.startswith('>'):
                out.write('>' + sp + '|' + line[1:])
                n+=1
            else:
                out.write(line)
    print(sp,'records',n,flush=True)
out.close()
print('done')
PY
ls -lh refs/combined.cdna.fa"
