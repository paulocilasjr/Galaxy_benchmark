import urllib.request, concurrent.futures, os, time
refs={'human':('homo_sapiens','Homo_sapiens.GRCh38'),'mouse':('mus_musculus','Mus_musculus.GRCm39'),'pig':('sus_scrofa','Sus_scrofa.Sscrofa11.1')}
os.makedirs('references/parts',exist_ok=True)
def run(sp,pair):
 d,f=pair;url=f'https://ftp.ensembl.org/pub/release-115/fasta/{d}/cdna/{f}.cdna.all.fa.gz'
 with urllib.request.urlopen(urllib.request.Request(url,method='HEAD'),timeout=60) as r:size=int(r.headers['Content-Length'])

 chunk=2_000_000
 def part(i):
  start=i*chunk;end=min(size,start+chunk)-1;path=f'references/parts/{sp}.{i}'
  if os.path.exists(path) and os.path.getsize(path)==end-start+1:return path
  for attempt in range(5):
   try:
    req=urllib.request.Request(url,headers={'Range':f'bytes={start}-{end}'})
    with urllib.request.urlopen(req,timeout=180) as r:
     if r.status"'!=206:raise RuntimeError(f'"'Range failed: {r.status}')
     data=r.read()
    if len(data)"'!=end-start+1:raise RuntimeError('"'wrong length')
    with open(path,'wb') as o:o.write(data)
    return path
   except Exception as e:
    print(sp,i,str(e),flush=True)
    if attempt==4:raise
    time.sleep(2)
 with concurrent.futures.ThreadPoolExecutor(max_workers=16) as pool:
  paths=list(pool.map(part,range((size+chunk-1)//chunk)))
 with open(f'references/{sp}.complete.fa.gz','wb') as out:
  for p in paths:
   with open(p,'rb') as f:out.write(f.read())
