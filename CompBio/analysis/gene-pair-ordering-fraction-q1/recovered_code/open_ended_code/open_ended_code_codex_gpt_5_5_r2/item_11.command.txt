/bin/bash -lc "python3 - <<'PY'
import anndata as ad
import numpy as np
from scipy import sparse
adata = ad.read_h5ad('10x_pbmc68k_reduced.h5ad')
mask = (adata.obs['bulk_labels'].astype(str) == 'CD19+ B') & (adata.obs['percent_mito'] < 0.04)
for name, obj in [('X', adata), ('raw', adata.raw.to_adata())]:
    sub = obj[mask, ['SRM','MRPS21']].X
    if sparse.issparse(sub): sub = sub.toarray()
    sub = np.asarray(sub)
    diff = sub[:,0] - sub[:,1]
    ans = (int((diff > 1).sum()), int((diff < -1).sum()), int(((diff >= -1)&(diff <= 1)).sum()))
    print(name, sub.shape, ans, 'minmax', float(diff.min()), float(diff.max()))
PY"
