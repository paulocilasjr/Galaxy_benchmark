/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
f=h5py.File('10x_pbmc68k_reduced.h5ad','r')
obs=f['obs'][...]
var=f['var'][...]
print('bulk labels cat:', [x.decode() if isinstance(x,bytes) else x for x in f['uns/bulk_labels_categories'][...]])
print('phase cat:', [x.decode() if isinstance(x,bytes) else x for x in f['uns/phase_categories'][...]])
print('louvain cat:', [x.decode() if isinstance(x,bytes) else x for x in f['uns/louvain_categories'][...]])
# counts bulk_labels
labels=obs['bulk_labels']; uniq, counts=np.unique(labels,return_counts=True)
cat=[x.decode() if isinstance(x,bytes) else x for x in f['uns/bulk_labels_categories'][...]]
for u,c in zip(uniq,counts):
    print(u, c, cat[u] if u<len(cat) else '?')
# gene indices for SRM MRPS21
indices=var['index']
names=[x.decode() if isinstance(x,bytes) else x for x in indices]
for g in ['SRM','MRPS21','CD19','MS4A1','CD79A']:
    hits=[i for i,n in enumerate(names) if n==g]
    print(g, hits, names[hits[0]] if hits else None)
# percent_mito stats
p=obs['percent_mito']
print('percent_mito min max mean', p.min(), p.max(), p.mean())
print('cells percent_mito<0.04', np.sum(p<0.04))
# maybe labels + p<0.04
for u in uniq:
    sel=(labels==u)&(p<0.04)
    print(cat[u] if u<len(cat) else u, np.sum(sel), 'total', np.sum(labels==u))
f.close()
PY"
