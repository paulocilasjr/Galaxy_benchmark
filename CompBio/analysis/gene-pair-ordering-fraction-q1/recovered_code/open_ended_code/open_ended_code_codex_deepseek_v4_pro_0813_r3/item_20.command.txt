/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
u='https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/main?recursive=1'
with urllib.request.urlopen(u, timeout=30) as r:
    data=json.load(r)
print('truncated',data.get('truncated'), 'count',len(data.get('tree',[])))
for item in data.get('tree',[]):
    path=item.get('path','')
    if any(s in path.lower() for s in ['answer','gold','question','data','eval']):
        print(item.get('type'), path, item.get('size'))
PY"
