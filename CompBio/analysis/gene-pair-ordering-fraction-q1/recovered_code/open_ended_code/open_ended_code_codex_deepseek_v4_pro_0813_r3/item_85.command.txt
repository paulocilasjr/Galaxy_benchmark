/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for rid in ['19443185','19443186']:
 u=f'https://zenodo.org/api/records/{rid}'
 with urllib.request.urlopen(u,timeout=30) as r: d=json.load(r)
 print('REC',rid,d.get('metadata',{}).get('title'))
 for f in d.get('files',[]):
  print(f.get('key'),f.get('size'))
PY"
