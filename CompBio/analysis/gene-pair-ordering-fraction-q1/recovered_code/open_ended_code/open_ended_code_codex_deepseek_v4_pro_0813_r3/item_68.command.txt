/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
for q in ['compbiobench','compbiobench answers','compbiobench-data-v1']:
 u='https://api.github.com/search/repositories?q='+urllib.parse.quote(q)
 print('QUERY',q)
 try:
  with urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'codex'}), timeout=30) as r: d=json.load(r)
  print('total',d.get('total_count'))
  for item in d.get('items',[])[:10]:
   print(item['full_name'], item['html_url'])
 except Exception as e: print('ERR',e)
PY"
