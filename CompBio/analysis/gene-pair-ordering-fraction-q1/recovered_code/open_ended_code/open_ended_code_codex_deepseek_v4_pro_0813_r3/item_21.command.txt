/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
u='https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/main?recursive=1'
with urllib.request.urlopen(u, timeout=30) as r:
    data=json.load(r)
for item in data.get('tree',[]):
    print(item.get('type'), item.get('path'), item.get('size'))
PY"
