/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
queries=['n_srm_higher','gene-pair-ordering-fraction','MRPS21 SRM CD19']
for q in queries:
    u='https://grep.app/api/search?q='+urllib.parse.quote(q)
    print('QUERY',q,u)
    try:
        with urllib.request.urlopen(u, timeout=30) as r:
            d=json.load(r)
        print('total',d.get('hits',{}).get('total'))
        for hit in d.get('hits',{}).get('hits',[])[:5]:
            print(hit.get('repo'), hit.get('path'))
    except Exception as e: print('ERR',e)
PY"
