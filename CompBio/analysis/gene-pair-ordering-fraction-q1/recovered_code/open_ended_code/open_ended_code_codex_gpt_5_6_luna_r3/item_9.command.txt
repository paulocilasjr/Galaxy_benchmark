/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
 obs=f['obs'][:]
 genes=[x.decode() for x in f['var']['index'][:]]
 x=f['X'][:]
 mask=(obs['bulk_labels']==6) & (obs['percent_mito'] < 0.04)
 diff=x[:,genes.index('SRM')]-x[:,genes.index('MRPS21')]
 a=mask & (diff>1.0)
 b=mask & ((-diff)>1.0)
 c=mask & (diff>=-1.0) & (diff<=1.0)
 print('filtered', int(mask.sum()))
 print('counts', int(a.sum()), int(b.sum()), int(c.sum()), 'sum', int((a|b|c).sum()))
 print('mito excluded CD19', int(((obs['bulk_labels']==6)&~(obs['percent_mito']<.04)).sum()))
 print('diff filtered', diff[mask])
 print('boundary exact', diff[mask][np.isclose(np.abs(diff[mask]),1)])
 print('formatted', f'{a.sum():.2f};{b.sum():.2f};{c.sum():.2f}')
PY"
