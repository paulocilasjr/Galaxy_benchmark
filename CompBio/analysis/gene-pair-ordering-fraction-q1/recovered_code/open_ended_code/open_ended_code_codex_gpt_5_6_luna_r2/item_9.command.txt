/bin/bash -lc "python - <<'PY'
import h5py
import numpy as np
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
 for path in ['/', '/X','/obs','/var','/raw.X','/raw.var']:
  o=f[path]
  print(path, 'attrs=', {k:o.attrs[k] for k in o.attrs.keys()})
 print('bulk categories', f['/uns/bulk_labels_categories'][:])
 print('phase categories', f['/uns/phase_categories'][:])
 print('first obs', f['/obs'][:10])
 print('obs unique bulk ids', np.unique(f['/obs']['bulk_labels'], return_counts=True))
 print('var first', f['/var'][:5])
 genes=np.array([x.decode() for x in f['/var']['index'][:]])
 for g in ['SRM','MRPS21']:
  inds=np.flatnonzero(genes==g)
  print(g, inds)
 X=f['/X']
 print('X sample', X[:3,:10])
 print('X min max mean nonzero', X[:].min(), X[:].max(), X[:].mean(), np.count_nonzero(X[:]))
 d=f['/raw.X']
 print('raw attrs', dict(d.attrs))
 print('raw nnz', len(d['data']), 'raw data sample', d['data'][:20])
PY"
