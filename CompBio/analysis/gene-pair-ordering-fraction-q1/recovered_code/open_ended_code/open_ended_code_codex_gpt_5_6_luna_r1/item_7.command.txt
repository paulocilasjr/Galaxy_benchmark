/bin/bash -lc "python -u - <<'PY'
import anndata as ad, numpy as np, pandas as pd
p='10x_pbmc68k_reduced.h5ad'
a=ad.read_h5ad(p)
obs=a.obs
mask=(obs['bulk_labels'].astype(str)=='CD19+ B') & (obs['percent_mito']<0.04)
srm=np.asarray(a[:, 'SRM'].X).reshape(-1).astype(float)
mrp=np.asarray(a[:, 'MRPS21'].X).reshape(-1).astype(float)
d=srm-mrp
print('CD19+', (obs['bulk_labels'].astype(str)=='CD19+ B').sum())
print('mito<.04', mask.sum(), 'max selected mito', obs.loc[mask,'percent_mito'].max())
print('counts >1, <-1, within inclusive', (d[mask]>1).sum(), (d[mask]<-1).sum(), ((d[mask]>=-1)&(d[mask]<=1)).sum())
print('sum', ((d[mask]>1).sum()+(d[mask]<-1).sum()+((d[mask]>=-1)&(d[mask]<=1)).sum()))
print('difference selected describe')
print(pd.Series(d[mask]).describe().to_string())
print('boundary values near +/-1')
sel=np.where(mask & (np.abs(d-1)<0.05))[0]
for i in sel: print(i,obs.iloc[i]['bulk_labels'],obs.iloc[i]['percent_mito'],srm[i],mrp[i],d[i])
sel=np.where(mask & (np.abs(d+1)<0.05))[0]
for i in sel: print(i,obs.iloc[i]['bulk_labels'],obs.iloc[i]['percent_mito'],srm[i],mrp[i],d[i])
print('all selected rows')
for i in np.where(mask)[0]: print(i, f\"mito={obs.iloc[i]['percent_mito']:.8f}\", f\"srm={srm[i]:.3f}\",f\"mrp={mrp[i]:.3f}\",f\"d={d[i]:.3f}\")
PY"
