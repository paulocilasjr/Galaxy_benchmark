/bin/bash -lc "python3 - <<'PY'
import h5py
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
    def visit(name,obj):
        if isinstance(obj,h5py.Dataset):
            print(name, obj.shape, obj.dtype)
        else:
            print(name+'/', dict(obj.attrs))
    f.visititems(visit)
PY"
