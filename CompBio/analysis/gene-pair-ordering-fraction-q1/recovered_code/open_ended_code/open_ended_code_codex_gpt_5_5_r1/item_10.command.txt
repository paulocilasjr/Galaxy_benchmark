/bin/bash -lc "python3 - <<'PY'
import h5py
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
    def show_group(g, prefix=''):
        for k in g.keys():
            obj=g[k]
            print(prefix+k, type(obj).__name__, getattr(obj,'shape',None), getattr(obj,'dtype',None), dict(obj.attrs))
            if isinstance(obj,h5py.Group) and prefix.count('/') < 2:
                show_group(obj, prefix+k+'/')
    show_group(f['uns'], 'uns/')
PY"
