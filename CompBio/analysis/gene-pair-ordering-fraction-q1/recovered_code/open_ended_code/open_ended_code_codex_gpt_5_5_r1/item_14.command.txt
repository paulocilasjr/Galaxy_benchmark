/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
p='10x_pbmc68k_reduced.h5ad'
with h5py.File(p,'r') as f:
    cats=[x.decode() for x in f['uns/bulk_labels_categories'][:]]
    obs=f['obs'][:]
    mask=np.where((obs['bulk_labels']==cats.index('CD19+ B')) & (obs['percent_mito'] < 0.04))[0]
    var=[x.decode() for x in f['raw.var']['index'][:]]
    sidx=var.index('SRM'); midx=var.index('MRPS21')
    g=f['raw.X']; data=g['data'][:]; indices=g['indices'][:]; indptr=g['indptr'][:]
    vals=[]
    for row in mask:
        start,end=indptr[row],indptr[row+1]
        cols=indices[start:end]
        rowdata=data[start:end]
        s=rowdata[cols==sidx][0] if np.any(cols==sidx) else 0.0
        m=rowdata[cols==midx][0] if np.any(cols==midx) else 0.0
        vals.append(s-m)
    diff=np.array(vals)
    print('raw counts', int(np.sum(diff>1.0)), int(np.sum(diff<-1.0)), int(np.sum((diff>=-1.0)&(diff<=1.0))))
    print('raw first vals', diff[:20])
PY"
