/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
with h5py.File('10x_pbmc68k_reduced.h5ad','r') as f:
 obs=f['obs'][:]; genes=f['var'][:]['index']; labels=f['uns/bulk_labels_categories'][:]
 mask=(obs['bulk_labels']==np.flatnonzero(labels==b'CD19+ B')[0]) & (obs['percent_mito']<0.04)
 s=np.flatnonzero(genes==b'SRM')[0]; m=np.flatnonzero(genes==b'MRPS21')[0]
 x=f['X'][:][mask]
 d=x[:,s]-x[:,m]
 counts=[int(np.count_nonzero(d>1.0)),int(np.count_nonzero(d < -1.0)),int(np.count_nonzero((d>=-1.0)&(d<=1.0)))]
 assert sum(counts)==int(mask.sum())
 print('Selected cells:',mask.sum())
 print(';'.join(map(str,counts)))
PY"
