/bin/bash -lc "python -u - <<'PY'
import anndata as ad, numpy as np
p='10x_pbmc68k_reduced.h5ad'
a=ad.read_h5ad(p)
mask=(a.obs['bulk_labels']=='CD19+ B') & (a.obs['percent_mito']<0.04)
print('filtered n',int(mask.sum()),'all B',int((a.obs.bulk_labels=='CD19+ B').sum()))
for name,mat,varnames in [('X',a.X,a.var_names),('raw',a.raw.X,a.raw.var_names)]:
 idx=[varnames.get_loc(g) for g in ['SRM','MRPS21']]
 arr=np.asarray(mat[mask][:,idx])
 d=arr[:,0]-arr[:,1]
 vals=(int(np.sum(d>1.0)),int(np.sum(d < -1.0)),int(np.sum((d>=-1.0)&(d<=1.0))))
 print(name,'shape',mat.shape,'indices',idx,'counts',vals,'sum',sum(vals))
 print('range genes',arr.min(axis=0),arr.max(axis=0),'diff',d.min(),d.max())
 print('boundary distances',sorted(d, key=lambda x:min(abs(x-1),abs(x+1)))[:10])
 print('diffs',np.sort(d).tolist())
PY"
