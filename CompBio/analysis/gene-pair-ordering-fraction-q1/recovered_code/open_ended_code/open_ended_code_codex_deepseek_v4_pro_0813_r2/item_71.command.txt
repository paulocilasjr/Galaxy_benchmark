/bin/bash -lc "python - <<'PY'
import urllib.request, re
u='https://www.biorxiv.org/content/biorxiv/early/2026/04/09/2026.04.06.716850.source.xml'
req=urllib.request.Request(u, headers={'User-Agent':'Mozilla/5.0'})
data=urllib.request.urlopen(req, timeout=30).read().decode()
for pat in ['supplement','Supplementary','xlsx','tsv','ground truth','answer']:
    for m in re.finditer(pat, data, re.I):
        s=max(0,m.start()-200); e=min(len(data),m.end()+300)
        print('---',pat, m.start())
        print(data[s:e].replace('\\n',' ')[:600])
PY"
