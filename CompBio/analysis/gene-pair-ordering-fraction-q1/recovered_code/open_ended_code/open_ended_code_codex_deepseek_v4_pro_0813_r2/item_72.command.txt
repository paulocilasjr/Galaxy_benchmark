import urllib.request
base='https://www.biorxiv.org/content/biorxiv/early/2026/04/09/2026.04.06.716850/'
# The xlink href is supplements/716850_file02.tsv relative to article URL? Try variants
urls=[
 'https://www.biorxiv.org/content/biorxiv/early/2026/04/09/2026.04.06.716850/supplements/716850_file02.tsv',
 'https://www.biorxiv.org/content/biorxiv/early/2026/04/09/2026.04.06.716850v1/supplements/716850_file02.tsv',

]

    try:
        req=urllib.request.Request(u, headers={'User-Agent':'Mozilla/5.0'})
        r=urllib.request.urlopen(req, timeout=30)

        if r.status==200 and len(data)>1000:

            break
    except Exception as e:
