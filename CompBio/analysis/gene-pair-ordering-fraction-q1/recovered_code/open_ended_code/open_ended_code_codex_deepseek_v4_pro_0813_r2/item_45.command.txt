/bin/bash -lc "python - <<'PY'
import urllib.request, csv, io
u='https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/compbiobench.v1.tsv'
data=urllib.request.urlopen(u, timeout=30).read().decode('utf-8')
for row in csv.DictReader(io.StringIO(data), delimiter='\\t'):
    q=row.get('question','')
    if 'fractions' in q.lower() or 'fraction' in q.lower():
        print(row['question_id'], '::', q[:500].replace('\\n',' '))
        print('---')
PY"
