/bin/bash -lc "python - <<'PY'
import urllib.request, json
u='https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/main?recursive=1'
data=urllib.request.urlopen(u, timeout=30).read().decode()
items=json.loads(data).get('tree',[])
for it in items:
    print(it['path'], it.get('size'))
PY"
