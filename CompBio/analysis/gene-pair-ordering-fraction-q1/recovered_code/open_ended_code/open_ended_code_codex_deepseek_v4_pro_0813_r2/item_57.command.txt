/bin/bash -lc "python - <<'PY'
import urllib.request, urllib.parse, json
queries=['\"MRPS21\" \"SRM\"', '\"n_srm_higher\"', '\"gene-pair-ordering-fraction\"']
for q in queries:
    u='https://api.github.com/search/code?q='+urllib.parse.quote(q)
    req=urllib.request.Request(u, headers={'Accept':'application/vnd.github+json','User-Agent':'codex'})
    try:
        data=urllib.request.urlopen(req, timeout=30).read().decode()
        js=json.loads(data)
        print(q, js.get('total_count'), [it.get('html_url') for it in js.get('items',[])][:5])
        print(js.get('message'))
    except Exception as e:
        print(q,'ERR',e)
PY"
