/bin/bash -lc "python - <<'PY'
import h5py,numpy as np
from scipy.sparse import csr_matrix
f=h5py.File('10x_pbmc68k_reduced.h5ad','r')
obs=f['obs'][:]; genes=np.char.decode(f['var'][:]['index'])
i=np.where(genes=='SRM')[0][0]; j=np.where(genes=='MRPS21')[0][0]
mask=(obs['bulk_labels']==6)&(obs['percent_mito']<0.04)
print('filtered n',mask.sum(),'excluded B',((obs['bulk_labels']==6)&~(obs['percent_mito']<0.04)).sum())
for name, a,b in [('X',f['X'][:,i],f['X'][:,j])]:
 d=a[mask]-b[mask]
 print(name, 'counts',np.sum(d>1),np.sum(d < -1),np.sum((d>=-1)&(d<=1)),'sum',np.sum(d>1)+np.sum(d < -1)+np.sum((d>=-1)&(d<=1)))
 print('boundary/near',np.sort(d[np.abs(np.abs(d)-1)<.05]))
 print('diffs',np.unique(d,return_counts=True))
g=f['raw.X']; raw=csr_matrix((g['data'][:],g['indices'][:],g['indptr'][:]),shape=tuple(g.attrs['h5sparse_shape']))
d=np.asarray(raw[:,i].toarray()).ravel()[mask]-np.asarray(raw[:,j].toarray()).ravel()[mask]
print('raw counts',np.sum(d>1),np.sum(d < -1),np.sum((d>=-1)&(d<=1)),'sum',len(d))
print('raw near',np.sort(d[np.abs(np.abs(d)-1)<.05]))
PY"
