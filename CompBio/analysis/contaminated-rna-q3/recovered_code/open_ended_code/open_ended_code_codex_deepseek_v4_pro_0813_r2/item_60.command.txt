/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob,os
for fn in sorted(glob.glob('qblast_common_*.xml')):
    if os.path.getsize(fn)==0: continue
    try:
        recs=list(NCBIXML.parse(open(fn)))
    except Exception as e:
        print(fn,e); continue
    for rec in recs:
        print('\\n',fn)
        for al in rec.alignments[:5]:
            h=al.hsps[0]
            print(al.hit_def[:150], h.expect, f'{h.identities}/{h.align_length}')
PY"
