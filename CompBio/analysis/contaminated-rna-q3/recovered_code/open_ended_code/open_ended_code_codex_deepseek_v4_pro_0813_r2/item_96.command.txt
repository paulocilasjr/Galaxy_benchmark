/bin/bash -c "python3 - <<'PY'
from Bio.SeqIO.QualityIO import FastqGeneralIterator
import random
random.seed(11)
reads=[]
with open('q3_unmapped_hg38_unmapped.fq') as f:
    for title,seq,qual in FastqGeneralIterator(f):
        reads.append((title,seq))
s=random.sample(reads,min(60,len(reads)))
with open('q3_candidates60.fa','w') as out:
    for t,seq in s:
        out.write(f'>{t}\\n{seq}\\n')
print(len(reads))
PY
head q3_candidates60.fa"
