/bin/bash -c "python3 - <<'PY'
import glob, os
from Bio import SeqIO
# sample first 5 common unmapped sequences from fastq
from Bio.SeqIO.QualityIO import FastqGeneralIterator
reads=[]
with open('common_all_unmapped.fq') as f:
    for title,seq,qual in FastqGeneralIterator(f):
        reads.append((title,seq))
        if len(reads)>=5: break
with open('common_unmapped5.fa','w') as out:
    for t,s in reads: out.write(f'>{t}\\n{s}\\n')
print('wrote',len(reads))
PY
ls -l common_unmapped5.fa"
