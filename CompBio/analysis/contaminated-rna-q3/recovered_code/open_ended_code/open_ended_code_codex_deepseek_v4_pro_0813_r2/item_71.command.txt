/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIWWW
from Bio import SeqIO
import concurrent.futures, os

def blast_rec(rec):
    fn=f'qblast_q3_{rec.id}.xml'
    if os.path.exists(fn) and os.path.getsize(fn)>0:
        return (rec.id, fn, 'cached')
    try:
        handle=NCBIWWW.qblast('blastn','nt',rec.seq,entrez_query='',hitlist_size=5,expect=0.001)
        data=handle.read(); handle.close()
        open(fn,'w').write(data)
        return (rec.id, fn, 'ok')
    except Exception as e:
        return (rec.id, None, str(e))

recs=list(SeqIO.parse('q3_unmapped12.fa','fasta'))
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
    futs=[ex.submit(blast_rec,r) for r in recs]
    for f in concurrent.futures.as_completed(futs):
        print(f.result(), flush=True)
PY"
