/bin/bash -c "python3 - <<'PY'
import random
# sample 20 unmapped reads
random.seed(1)
reads=[]
with open('q3_unmapped.fq') as f:
    while True:
        name=f.readline().rstrip()
        if not name: break
        seq=f.readline().rstrip()
        plus=f.readline()
        qual=f.readline()
        reads.append((name,seq))
s=random.sample(reads,20)
with open('q3_unmapped20.fa','w') as out:
    for name,seq in s:
        out.write(f'>{name[1:]}\\n{seq}\\n')
print(len(reads))
PY
blastn -remote -query q3_unmapped20.fa -db nt -out q3_unmapped20.blast.tsv -outfmt '6 qseqid sseqid staxids pident length mismatch gapopen qstart qend sstart send evalue bitscore sscinames scomnames stitle' -max_target_seqs 5 -evalue 1e-10 -num_threads 4"
