/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIXML
import glob, os, re, collections
rows=[]
for fn in glob.glob('qblast_cand_*.xml'):
    if os.path.getsize(fn)==0: continue
    try: recs=list(NCBIXML.parse(open(fn)))
    except Exception as e: continue
    for rec in recs:
        if not rec.alignments: 
            rows.append((os.path.basename(fn), 'NO HIT','','',''))
            continue
        al=rec.alignments[0]
        h=al.hsps[0]
        rows.append((os.path.basename(fn), al.hit_def, h.expect, h.identities, h.align_length))
for r in sorted(rows):
    print(r[0], '=>', r[1][:160], r[2], f'{r[3]}/{r[4]}')
PY"
