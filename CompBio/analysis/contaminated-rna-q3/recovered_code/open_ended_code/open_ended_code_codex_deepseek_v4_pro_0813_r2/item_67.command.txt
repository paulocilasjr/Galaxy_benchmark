/bin/bash -c "python3 - <<'PY'
from Bio.SeqIO.QualityIO import FastqGeneralIterator
import random
random.seed(4)
reads=[]
with open('q2_unmapped.fq') as f:
    for title,seq,qual in FastqGeneralIterator(f):
        reads.append((title,seq))
s=random.sample(reads,10)
with open('q2_unmapped10.fa','w') as out:
    for t,s in s:
        out.write(f'>{t}\\n{s}\\n')
print(len(reads))
PY
head q2_unmapped10.fa"
