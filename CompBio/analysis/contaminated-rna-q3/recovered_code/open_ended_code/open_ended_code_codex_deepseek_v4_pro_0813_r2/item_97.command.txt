/bin/bash -c "python3 - <<'PY'
from Bio.Blast import NCBIWWW
from Bio import SeqIO
import concurrent.futures, os, time

def blast_rec(rec):
    fn=f'qblast_cand_{rec.id}.xml'
    if os.path.exists(fn) and os.path.getsize(fn)>0:
        return (rec.id, fn, 'cached')
    for attempt in range(3):
        try:
            handle=NCBIWWW.qblast('blastn','nt',rec.seq,entrez_query='',hitlist_size=3,expect=0.001)
            data=handle.read(); handle.close()
            open(fn,'w').write(data)
            return (rec.id, fn, 'ok')
        except Exception as e:
            last=e; time.sleep(2)
    return (rec.id, None, str(last))

recs=list(SeqIO.parse('q3_candidates60.fa','fasta'))
with concurrent.futures.ThreadPoolExecutor(max_workers=6) as ex:
    futs=[ex.submit(blast_rec,r) for r in recs]
    done=0
    for f in concurrent.futures.as_completed(futs):
        res=f.result(); done+=1
        if done%5==0 or res[2]"'!='"'ok':
            print(done,res,flush=True)
print('finished',done)
PY"
