/bin/bash -c "python3 - <<'PY'
import json, collections
j=json.load(open('blast_topdup30.json'))
tax=collections.Counter(); ex={}
for rep in j['BlastOutput2']:
    r=rep['report']['results']['search']; hits=r.get('hits',[])
    if not hits: continue
    d=hits[0]['description'][0]; sc=d.get('sciname','unknown')
    tax[sc]+=1; ex.setdefault(sc,(r.get('query_title'), hits[0]['hsps'][0].get('identity'), hits[0]['hsps'][0].get('align_len')))
for k,v in tax.most_common(): print(v,k,ex[k])
PY"
