/bin/bash -c "python3 - <<'PY'
import urllib.parse, urllib.request, re, time
rid='82T7Z9P0014'
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi'
params={'CMD':'Get','RID':rid,'FORMAT_TYPE':'JSON2_S'}
for i in range(20):
    time.sleep(10)
    req=urllib.request.Request(url+'?'+urllib.parse.urlencode(params), headers={'User-Agent':'Mozilla/5.0'})
    try:
        text=urllib.request.urlopen(req, timeout=30).read().decode('utf-8','ignore')
    except Exception as e:
        print('ERR',e); break
    if 'Status=READY' in text or 'There was a problem' in text or 'qBlastOutput' in text or 'QBlastInfo' in text and 'READY' in text:
        print('ready len',len(text)); print(text[:500]); break
    status = re.search(r'QBlastInfoBegin.*?Status=(\\w+)', text, re.S)
    print(i, 'status', status.group(1) if status else '?', 'len', len(text))
    if status and status.group(1) in ['READY','FAILED','UNKNOWN']:
        print(text[:2000]); break
PY"
