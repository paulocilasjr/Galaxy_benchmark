/bin/bash -c "python3 - <<'PY'
# parse readids from top_nonhuman_readids
ids=[]
for line in open('top_nonhuman_readids.txt'):
    name, idlist=line.rstrip('\\n').split('\\t')
    ids += idlist.split(',') if idlist else []
print('total ids', len(ids))
want=set(ids)
# read fastq via gzip and write fasta for wanted
import gzip
out=open('top_nonhuman.fa','w')
count=0
with gzip.open('contaminated.rna.q3.fq.gz','rt') as f:
    while True:
        h=f.readline().strip()
        if not h: break
        s=f.readline().strip(); plus=f.readline().strip(); q=f.readline().strip()
        rid=h[1:].split()[0]
        if rid in want:
            out.write(f'>{rid}\\n{s}\\n'); count+=1
out.close(); print('wrote',count)
PY
wc -l top_nonhuman.fa; head -4 top_nonhuman.fa"
