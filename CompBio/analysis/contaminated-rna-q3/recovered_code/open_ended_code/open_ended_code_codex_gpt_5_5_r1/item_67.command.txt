/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
files = {
'human':'human_best_nm.tsv',
'callithrix':'best_Callithrix_jacchus.mCalJac1.pat.X.cdna.all.tsv',
'chlorocebus':'best_Chlorocebus_sabaeus.ChlSab1.1.cdna.all.tsv',
'gorilla':'best_Gorilla_gorilla.gorGor4.cdna.all.tsv',
'macaca_f':'best_Macaca_fascicularis.Macaca_fascicularis_6.0.cdna.all.tsv',
'macaca_m':'best_Macaca_mulatta.Mmul_10.cdna.all.tsv',
'nomascus':'best_Nomascus_leucogenys.Nleu_3.0.cdna.all.tsv',
'pan':'best_Pan_troglodytes.Pan_tro_3.0.cdna.all.tsv',
'papio':'best_Papio_anubis.Panubis1.0.cdna.all.tsv',
'pongo':'best_Pongo_abelii.Susie_PABv2.cdna.all.tsv',
}
data={}
reads=set()
for name,fn in files.items():
    d={}
    with open(fn) as f:
        for line in f:
            r,n=line.split()[:2]
            d[r]=int(n)
            reads.add(r)
    data[name]=d

names=[n for n in files if n"'!='"'human']
strict={n:0 for n in names}
tie={n:0 for n in names}
better_h={n:0 for n in names}
better_h3={n:0 for n in names}
better_h5={n:0 for n in names}
strict_better_h={n:0 for n in names}
exact_bad={n:0 for n in names}
for r in reads:
    vals={n:data[n].get(r,99) for n in files}
    h=vals['human']
    non={n:vals[n] for n in names}
    mn=min(non.values())
    winners=[n for n,v in non.items() if v==mn]
    for n,v in non.items():
        if v < h: better_h[n]+=1
        if h-v >=3: better_h3[n]+=1
        if h-v >=5: better_h5[n]+=1
        if v==0 and h>=8: exact_bad[n]+=1
    if len(winners)==1:
        strict[winners[0]]+=1
        if mn < h:
            strict_better_h[winners[0]]+=1
    for n in winners:
        tie[n]+=1
print('name\\tstrict_nonhuman_winner\\tstrict_winner_better_than_human\\ttie_for_best_nonhuman\\tbetter_than_human\\tbetter_h3\\tbetter_h5\\texact_hbad')
for n in sorted(names, key=lambda x: strict_better_h[x], reverse=True):
    print(n, strict[n], strict_better_h[n], tie[n], better_h[n], better_h3[n], better_h5[n], exact_bad[n], sep='\\t')
PY"
