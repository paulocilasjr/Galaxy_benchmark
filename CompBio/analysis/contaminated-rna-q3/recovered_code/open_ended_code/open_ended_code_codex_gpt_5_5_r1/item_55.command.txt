/bin/bash -lc 'sleep 35
> qblast_candidate_hits.txt
while IFS=$'"'\\t' read id rid rtoe; do
  curl -s \"https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&FORMAT_TYPE=Text&RID="'$rid" > "${id}.${rid}.blast.txt"
  echo "== $id $rid ==" >> qblast_candidate_hits.txt
  grep -A25 -m1 '"'"'^Sequences producing significant alignments'"' \""'${id}.${rid}.blast.txt" >> qblast_candidate_hits.txt || sed -n '"'1,60p' \""'${id}.${rid}.blast.txt" >> qblast_candidate_hits.txt
done < qblast_candidate_rids.tsv
cat qblast_candidate_hits.txt'
