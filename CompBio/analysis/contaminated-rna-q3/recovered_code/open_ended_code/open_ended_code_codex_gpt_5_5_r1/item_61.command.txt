/bin/bash -lc "printf 'reference\\tbest_better5\\tbest_better3\\tnonhuman_nm_le2_human_bad\\tnonhuman_nm0_human_bad\\n' > primate_better_than_human.tsv
for ref in refs/panel/Callithrix_jacchus*.fa.gz refs/panel/Chlorocebus_sabaeus*.fa.gz refs/panel/Gorilla_gorilla*.fa.gz refs/panel/Macaca_fascicularis*.fa.gz refs/panel/Macaca_mulatta*.fa.gz refs/panel/Nomascus_leucogenys*.fa.gz refs/panel/Pan_troglodytes*.fa.gz refs/panel/Papio_anubis*.fa.gz refs/panel/Pongo_abelii*.fa.gz; do
  name="'$(basename "$ref" .fa.gz)
  minimap2 -t 4 -x sr --secondary=no -a "$ref" contaminated.rna.q3.fq.gz 2> "${name}.primate.map.log" | awk -v name="$name" '"'BEGIN{OFS=\"\\t\"} FNR==NR{h["'$1]=$2; next} /''^@/{next} {unmapped=int($2/4)%2; if(!unmapped){nm=999; for(i=12;i<=NF;i++) if($i ~ /''^NM:i:/){nm=substr($i,6)+0}; if(!($1 in b) || nm<b[$1]) b[$1]=nm}} END{for(r in b){hn=(r in h?h[r]:99); if(hn-b[r]>=5)c5++; if(hn-b[r]>=3)c3++; if(b[r]<=2 && hn>=8)c2bad++; if(b[r]==0 && hn>=8)c0bad++} print name,c5+0,c3+0,c2bad+0,c0bad+0}'"' human_best_nm.tsv - >> primate_better_than_human.tsv
  tail -n 1 primate_better_than_human.tsv
done
cat primate_better_than_human.tsv"
