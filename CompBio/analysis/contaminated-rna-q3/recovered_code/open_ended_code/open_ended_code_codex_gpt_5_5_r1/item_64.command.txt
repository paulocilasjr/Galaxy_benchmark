/bin/bash -lc 'for ref in refs/panel/Callithrix_jacchus*.fa.gz refs/panel/Chlorocebus_sabaeus*.fa.gz refs/panel/Gorilla_gorilla*.fa.gz refs/panel/Macaca_fascicularis*.fa.gz refs/panel/Macaca_mulatta*.fa.gz refs/panel/Nomascus_leucogenys*.fa.gz refs/panel/Pan_troglodytes*.fa.gz refs/panel/Papio_anubis*.fa.gz refs/panel/Pongo_abelii*.fa.gz; do
  name=$(basename "$ref" .fa.gz)
  out="best_${name}.tsv"
  minimap2 -t 4 -x sr --secondary=no -a "$ref" contaminated.rna.q3.fq.gz 2> "${name}.bestfile.map.log" | awk '"'BEGIN{OFS=\"\\t\"} /"'^@/{next} {unmapped=int($2/4)%2; if(!unmapped){nm=999; for(i=12;i<=NF;i++) if($i ~ /''^NM:i:/){nm=substr($i,6)+0}; if(!($1 in best) || nm<best[$1]) best[$1]=nm}} END{for(k in best) print k,best[k]}'"' > \""'$out"
  printf '"'%s\\t' \""'$name"; wc -l < "$out"
done'
