/bin/bash -lc "i=0
> qblast_pan_rids.tsv
awk 'NR<=6{print "'$1"'"\\t\""'$4}'"' pan_strict_hiconf.tsv | while IFS="'$'"'\\t' read id seq; do
  i="'$((i+1))
  curl -s '"'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data-urlencode CMD=Put --data-urlencode PROGRAM=blastn --data-urlencode DATABASE=nt --data-urlencode MEGABLAST=on --data-urlencode EXPECT=1e-5 --data-urlencode HITLIST_SIZE=20 --data-urlencode QUERY=\""'$seq" > "qblast_pan_${i}_put.txt"
  rid=$(grep -m1 '"'RID =' \"qblast_pan_"'${i}_put.txt" | awk '"'{print "'$3}'"')
  rtoe="'$(grep -m1 '"'RTOE =' \"qblast_pan_"'${i}_put.txt" | awk '"'{print "'$3}'"')
  printf '%s\\t%s\\t%s\\n' \""'$id" "$rid" "$rtoe" >> qblast_pan_rids.tsv
done
cat qblast_pan_rids.tsv'
