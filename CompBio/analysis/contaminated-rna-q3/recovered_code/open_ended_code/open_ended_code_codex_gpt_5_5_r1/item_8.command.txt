/bin/bash -lc "gzip -dc contaminated.rna.q3.fq.gz | awk 'NR%4==1{h=substr("'$0,2)} NR%4==2{print ">"h"'"\\n\""'$0}'"' | head -n 200 > sample100.fa
blastn -remote -db nt -task blastn -query sample100.fa -max_target_seqs 10 -evalue 1e-10 -outfmt '6 qseqid staxids sscinames pident length evalue bitscore stitle' > sample100.nt.tsv"
