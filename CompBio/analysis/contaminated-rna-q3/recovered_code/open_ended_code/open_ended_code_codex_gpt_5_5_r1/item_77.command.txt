/bin/bash -lc 'sleep 60
> qblast_pan_hits2.txt
while IFS=$'"'\\t' read id rid rtoe; do
  curl -s \"https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&FORMAT_TYPE=Text&RID="'$rid" > "${id}.${rid}.blast2.txt"
  echo "== $id $rid ==" >> qblast_pan_hits2.txt
  grep -A20 -m1 '"'"'^Sequences producing significant alignments'"' \""'${id}.${rid}.blast2.txt" >> qblast_pan_hits2.txt || grep -E '"'Status=|No significant' \""'${id}.${rid}.blast2.txt" | head -n 5 >> qblast_pan_hits2.txt
done < qblast_pan_rids.tsv
cat qblast_pan_hits2.txt'
