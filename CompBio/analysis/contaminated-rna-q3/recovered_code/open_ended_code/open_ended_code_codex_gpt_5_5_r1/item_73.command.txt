/bin/bash -lc 'sleep 35
> qblast_pan_hits.txt
while IFS=$'"'\\t' read id rid rtoe; do
  curl -s \"https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&FORMAT_TYPE=Text&RID="'$rid" > "${id}.${rid}.blast.txt"
  echo "== $id $rid ==" >> qblast_pan_hits.txt
  grep -A20 -m1 '"'"'^Sequences producing significant alignments'"' \""'${id}.${rid}.blast.txt" >> qblast_pan_hits.txt || sed -n '"'1,60p' \""'${id}.${rid}.blast.txt" >> qblast_pan_hits.txt
done < qblast_pan_rids.tsv
cat qblast_pan_hits.txt'
