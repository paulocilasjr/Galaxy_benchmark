/bin/bash -lc "printf 'reference\\tany_mapped\\tnm_le3\\tmapq20_nm_le5\\n' > panel_mapping_counts.tsv
for ref in refs/human_cdna.fa.gz refs/panel/*.fa.gz; do
  name="'$(basename "$ref" .fa.gz)
  minimap2 -t 4 -x sr --secondary=no -a "$ref" contaminated.rna.q3.fq.gz 2> "${name}.map.log" | awk -v name="$name" '"'BEGIN{OFS=\"\\t\"} /"'^@/{next} {unmapped=int($2/4)%2; if(!unmapped){any[$1]=1; nm=999; for(i=12;i<=NF;i++) if($i ~ /''^NM:i:/){nm=substr($i,6)+0}; if(nm<=3) nm3[$1]=1; if($5>=20 && nm<=5) conf[$1]=1}} END{for(k in any)c1++; for(k in nm3)c2++; for(k in conf)c3++; print name,c1+0,c2+0,c3+0}'"' >> panel_mapping_counts.tsv
  tail -n 1 panel_mapping_counts.tsv
done
cat panel_mapping_counts.tsv"
