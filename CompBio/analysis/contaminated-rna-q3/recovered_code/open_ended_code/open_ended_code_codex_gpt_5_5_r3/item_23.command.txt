/bin/bash -lc "gzip -dc contaminated.rna.q3.fq.gz | awk 'NR%4==2{count["'$0]++} END{n=0; for(s in count) print count[s] "'"\\t\" s}' | sort -nr | head -100 | awk '{print \">top\" NR \"_count\" "'$1 "'"\\n\" "'$2}'"' > top100.fa
blastn -remote -db refseq_rna -query top100.fa -task blastn -max_target_seqs 10 -outfmt '6 qseqid sacc stitle pident length evalue bitscore staxids sscinames scomnames' -out top100.refseq_rna.tsv"
