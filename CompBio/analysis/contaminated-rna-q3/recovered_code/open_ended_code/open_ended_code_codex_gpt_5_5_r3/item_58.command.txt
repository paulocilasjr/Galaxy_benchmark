/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import Counter

def read_fa(path):
    s=[]
    for line in open(path):
        if line.startswith('>'): continue
        s.append(line.strip().upper())
    return ''.join(s)
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
def kmers_circular(s,k=75):
    ss=s+s[:k-1]
    out=set()
    for i in range(len(s)):
        kmer=ss[i:i+k]
        out.add(kmer); out.add(rc(kmer))
    return out
hset=kmers_circular(read_fa('refs/human_mt_NC_012920.fa'))
pset=kmers_circular(read_fa('refs/pan_paniscus_mt_NC_001644.fa'))
counts=Counter(); ex={}
with gzip.open('contaminated.rna.q3.fq.gz','rt') as f:
    while True:
        name=f.readline().strip()
        if not name: break
        seq=f.readline().strip().upper(); f.readline(); f.readline()
        key=(seq in hset, seq in pset)
        counts[key]+=1
        ex.setdefault(key,(name,seq))
print(counts)
for k,v in sorted(ex.items()): print(k,v)
PY"
