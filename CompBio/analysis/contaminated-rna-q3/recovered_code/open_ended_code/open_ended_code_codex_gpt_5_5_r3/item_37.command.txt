/bin/bash -lc "gzip -dc contaminated.rna.q3.fq.gz | awk 'NR%4==2{++i; if(i%8000==0) print \">read\" i \"\\n\" "'$0}'"' > qbatch30.fa
curl -s 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data-urlencode CMD=Put --data-urlencode PROGRAM=blastn --data-urlencode DATABASE=nt --data-urlencode MEGABLAST=on --data-urlencode HITLIST_SIZE=5 --data-urlencode QUERY@qbatch30.fa > qbatch30_put.txt
grep -E 'RID =|RTOE =' qbatch30_put.txt | head -5"
