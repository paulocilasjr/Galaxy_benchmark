/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import Counter

def read_fa(path):
    s=[]
    for line in open(path):
        if line.startswith('>'): continue
        s.append(line.strip().upper())
    return ''.join(s)
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
h=read_fa('refs/human_mt_NC_012920.fa')
p=read_fa('refs/pan_paniscus_mt_NC_001644.fa')
counts=Counter()
examples={}
with gzip.open('contaminated.rna.q3.fq.gz','rt') as f:
    while True:
        name=f.readline().strip()
        if not name: break
        seq=f.readline().strip().upper(); f.readline(); f.readline()
        hm=(seq in h) or (rc(seq) in h)
        pm=(seq in p) or (rc(seq) in p)
        key=(hm,pm)
        counts[key]+=1
        examples.setdefault(key,(name,seq))
print(counts)
for k,v in examples.items(): print(k,v)
PY"
