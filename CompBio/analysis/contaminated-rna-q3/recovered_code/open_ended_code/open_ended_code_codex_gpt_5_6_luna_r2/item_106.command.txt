/bin/bash -lc "samtools view -F 2308 human.ensembl.map.bam | awk '{as=0;nm=999;for(i=12;i<=NF;i++){if("'$i ~ /''^AS:i:/){as=substr($i,6)} if($i ~ /''^NM:i:/){nm=substr($i,6)}}; print $1"'"\\t\"as\"\\t\"nm}' | sort -k1,1 > human.as.tsv
samtools view -F 2308 macaque.map.bam | awk '{as=0;nm=999;for(i=12;i<=NF;i++){if("'$i ~ /''^AS:i:/){as=substr($i,6)} if($i ~ /''^NM:i:/){nm=substr($i,6)}}; print $1"'"\\t\"as\"\\t\"nm}' | sort -k1,1 > macaque.as.tsv
join -t '\\t' -a1 -a2 -e 0 -o 0,1.2,1.3,2.2,2.3 human.as.tsv macaque.as.tsv > human_macaque.as.tsv
awk -F '\\t' 'BEGIN{hbetter=mbetter=tie=both=0;honly=monly=0} {h="'$2;m=$4;if(h>0&&m>0)both++;if(h>m)hbetter++;else if(m>h)mbetter++;else tie++;if(h>0&&m==0)honly++;if(m>0&&h==0)monly++} END{print "hbetter",hbetter,"mbetter",mbetter,"tie",tie,"both",both,"honly",honly,"monly",monly}'"' human_macaque.as.tsv
printf '%s\\n' '--- high macaque advantage ---'; awk -F '\\t' '"'$4-$2>=10{print}'"' human_macaque.as.tsv | head -30
printf '%s\\n' '--- highest macaque-only/advantage count ---'; awk -F '\\t' '"'$4-$2>=5{n++} END{print n+0}'"' human_macaque.as.tsv
wc -l human.as.tsv macaque.as.tsv human_macaque.as.tsv"
