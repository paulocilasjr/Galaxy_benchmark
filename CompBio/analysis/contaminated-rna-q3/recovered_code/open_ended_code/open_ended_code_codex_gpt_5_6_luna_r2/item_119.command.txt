/bin/bash -lc "head -20 candidate.ids > candidate20.ids
awk 'NR==FNR{ids["'$1]=1;next} NR%4==1{h=$0} NR%4==2{if(ids[substr(h,2)]) print ">"substr(h,2)"'"\\n\""'$0}'"' candidate20.ids human.ensembl.unmapped.fq > candidate20.fa
wc -l candidate20.fa
query="'$(awk '"'{print}' candidate20.fa)
curl -L --fail --silent --show-error --retry 2 --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' --data-urlencode 'MEGABLAST=on' --data-urlencode 'HITLIST_SIZE=5' --data-urlencode 'FORMAT_TYPE=XML' --data-urlencode \"QUERY="'$query" '"'https://blast.ncbi.nlm.nih.gov/Blast.cgi' > candidate.blast.put.html
grep -E 'RID =|RTOE =|value=\"[A-Z0-9]+\" id=\"rid\"' candidate.blast.put.html | head -10"
