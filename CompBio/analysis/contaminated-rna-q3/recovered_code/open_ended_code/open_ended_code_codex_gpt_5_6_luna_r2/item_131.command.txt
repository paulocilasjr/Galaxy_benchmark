/bin/bash -lc "samtools fastq -F 4 -n mesomycoplasma.map.bam | awk 'NR%4==1{h="'$0} NR%4==2{n++; if(n<=20){print ">"substr(h,2);print}}'"' > myco.mapped20.fa
cat myco.mapped20.fa | head -8
query="'$(awk '"'{print}' myco.mapped20.fa)
curl -L --fail --silent --show-error --retry 2 --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' --data-urlencode 'MEGABLAST=on' --data-urlencode 'HITLIST_SIZE=5' --data-urlencode 'FORMAT_TYPE=XML' --data-urlencode \"QUERY="'$query" '"'https://blast.ncbi.nlm.nih.gov/Blast.cgi' > myco.blast.put.html
grep -E 'RID =|RTOE =|value=\"[A-Z0-9]+\" id=\"rid\"' myco.blast.put.html | head -10"
