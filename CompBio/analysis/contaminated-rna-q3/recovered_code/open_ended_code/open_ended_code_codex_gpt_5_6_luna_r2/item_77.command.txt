/bin/bash -lc 'query=$(awk '"'NR<=100{print}' human.ensembl.unmapped.fq | awk 'NR%4==1{print \">\"substr("'$0,2)} NR%4==2{print}'"' | head -100)
curl -L --fail --silent --show-error --retry 2 --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' --data-urlencode 'MEGABLAST=on' --data-urlencode 'HITLIST_SIZE=5' --data-urlencode 'FORMAT_TYPE=XML' --data-urlencode \"QUERY="'$query" '"'https://blast.ncbi.nlm.nih.gov/Blast.cgi' > ensembl.blast50.put.html
grep -E 'RID =|RTOE =|value=\"[A-Z0-9]+\" id=\"rid\"' ensembl.blast50.put.html | head -10"
