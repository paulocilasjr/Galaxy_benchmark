/bin/bash -lc "for spec_bam in 'H human.ensembl.map.bam' 'M macaque.map.bam' 'C chimp.map.bam' 'G gorilla.map.bam'; do set -- "'$spec_bam; sp=$1; bam=$2; samtools view -F 2308 "$bam" | awk -v sp="$sp" '"'{as=0;nm=999;for(i=12;i<=NF;i++){if("'$i ~ /''^AS:i:/){as=substr($i,6)} if($i ~ /''^NM:i:/){nm=substr($i,6)}}; print $1"'"\\t\"sp\"\\t\"as\"\\t\"nm}' >> primate.scores.tsv; done
sort -k1,1 -k2,2 primate.scores.tsv > primate.scores.sorted.tsv
awk -F '\\t' 'BEGIN{OFS=\"\\t\"}
{q="'$1;sp=$2;as=$3;nm=$4+0; score[q,sp]=as+0; mism[q,sp]=nm; seen[q]=1}
END{for(q in seen){h=score[q,"H"]+0;m=score[q,"M"]+0;c=score[q,"C"]+0;g=score[q,"G"]+0;best="H";b=h;if(m>b){best="M";b=m}if(c>b){best="C";b=c}if(g>b){best="G";b=g} print q,h,m,c,g,best,b,h,m,c,g}}'"' primate.scores.sorted.tsv | sort -k1,1 > primate.compare.tsv
printf '%s\\n' '--- best species counts ---'; awk -F '\\t' '{n["'$6]++} END{for(k in n)print k,n[k]}'"' primate.compare.tsv | sort
printf '%s\\n' '--- score margins over human ---'; awk -F '\\t' '"'$7-$2>=5{n++} END{print ">=5",n+0} $7-$2>=10{n10++} END{print ">=10",n10+0} $7-$2>=20{n20++} END{print ">=20",n20+0}'"' primate.compare.tsv
printf '%s\\n' '--- top nonhuman advantage ---'; awk -F '\\t' '"'$6!="H" && $7-$2>=10{print}'"' primate.compare.tsv | sort -t '\\t' -k7,7nr | head -50"
