/bin/bash -lc "python3 -u - <<'PY'
import urllib.parse, urllib.request, json
rid='7WKC4R76016'
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','FORMAT_TYPE':'JSON2_S','RID':rid})
text=urllib.request.urlopen(url,timeout=60).read().decode()
open('sample20_blast.json','w').write(text)
print('saved',len(text))
data=json.loads(text)
for report in data.get('BlastOutput2',[]):
    print('\\nQUERY', report.get('report',{}).get('results',{}).get('search',{}).get('query_title'))
    hits=report.get('report',{}).get('results',{}).get('search',{}).get('hits',[])
    for h in hits[:3]:
        print(' desc', h.get('description',[{}])[0].get('title'))
        print(' taxid', h.get('description',[{}])[0].get('taxid'), 'sciname', h.get('description',[{}])[0].get('sciname'))
        for hsp in h.get('hsps',[])[:1]:
            print('  pident',hsp.get('identity'),'len',hsp.get('align_len'),'evalue',hsp.get('evalue'))
PY"
