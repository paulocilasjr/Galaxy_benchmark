/bin/bash -lc "python3 - <<'PY'
import gzip, random
N=300
random.seed(42)
res=[]; n=0
with gzip.open('contaminated.rna.q3.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h: break
        seq=f.readline().strip(); plus=f.readline(); qual=f.readline()
        n+=1
        name=h.rstrip()[1:]
        if len(res)<N:
            res.append((name,seq))
        else:
            j=random.randrange(n)
            if j<N:
                res[j]=(name,seq)
print('reads',n,'sampled',len(res))
with open('sample300.fa','w') as out:
    for name,seq in res:
        out.write(f'>{name}\\n{seq}\\n')
PY
wc -l sample300.fa && head -4 sample300.fa && tail -4 sample300.fa"
