/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, time
seqs = open('sample20.fa').read().strip().split('>')[1:]
query=''
for s in seqs[:5]:
    lines=s.split('\\n',1)
    query += '>'+lines[0]+'\\n'+lines[1].strip()+'\\n'
params = {
 'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','MEGABLAST':'on',
 'QUERY':query,'FORMAT_TYPE':'JSON','HITLIST_SIZE':'5','DESCRIPTIONS':'5','ALIGNMENTS':'0'
}
data=urllib.parse.urlencode(params).encode()
req=urllib.request.Request('https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi', data=data)
try:
    resp=urllib.request.urlopen(req, timeout=120)
    print(resp.read().decode()[:2000])
except Exception as e:
    print('ERR',repr(e))
PY"
