/bin/bash -lc "python3 - <<'PY'
import json
from collections import Counter,defaultdict
data=json.load(open('unmapped1000_blast.json'))
cnt=Counter(); taxcnt=Counter(); genus_counter=Counter(); family_counter=Counter(); nohits=0; total=0
# We only have descriptions with sciname maybe taxid.
for rep in data['BlastOutput2']:
    res=rep['report']['results']; search=res['search']; hits=search.get('hits',[]); total+=1
    if not hits: nohits+=1; continue
    h=hits[0]; desc=h.get('description',[{}])[0]; sc=desc.get('sciname',''); tax=desc.get('taxid','')
    cnt[sc]+=1; taxcnt[tax]+=1
    genus=sc.split()[0].lower() if sc else 'none'
    genus_counter[genus]+=1
    # family not available directly
print('total',total,'nohits',nohits)
print('\\nTop species counts (top hit):')
for sc,n in cnt.most_common(60): print(n,sc)
print('\\nTop taxids:')
for t,n in taxcnt.most_common(30): print(n,t)
print('\\nTop genus:')
for g,n in genus_counter.most_common(60): print(n,g)
PY"
