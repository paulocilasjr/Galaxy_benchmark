/bin/bash -lc "python3 -u - <<'PY'
import json, glob
for path in sorted(glob.glob('blast_out/chunk_*.json')):
    data=json.load(open(path))
    for rep in data['BlastOutput2']:
        search=rep['report']['results']['search']; hits=search.get('hits',[])
        if not hits: continue
        desc0=hits[0].get('description',[{}])[0]
        if desc0.get('sciname')=='Mesomycoplasma hyorhinis':
            print('\\nQUERY',search.get('query_title'))
            for h in hits:
                d=h.get('description',[{}])[0]
                hsp=h['hsps'][0]
                print(' ',d.get('sciname'),d.get('taxid'),'id',d.get('accession'),d.get('title'),'identity',hsp.get('identity'),'align',hsp.get('align_len'),'evalue',hsp.get('evalue'))
            if search.get('query_title')=='READ_163728': break
    else:
        continue
    break
PY"
