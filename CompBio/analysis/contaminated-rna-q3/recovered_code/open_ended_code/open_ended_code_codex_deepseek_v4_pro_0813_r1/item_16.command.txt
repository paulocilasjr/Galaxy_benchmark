/bin/bash -lc "python3 - <<'PY'
import gzip, random
N=300
random.seed(42)
res=[]
with gzip.open('contaminated.rna.q3.fq.gz','rt') as f:
    i=0
    for line in f:
        i+=1
        if i%4==1:
            name=line.rstrip()[1:]
        elif i%4==2:
            seq=line.rstrip()
        elif i%4==4:
            if len(res)<N:
                res.append((name,seq))
            else:
                j=random.randrange(i//4)
                if j<N:
                    res[j]=(name,seq)
print('reads',i//4,'sampled',len(res))
with open('sample300.fa','w') as out:
    for name,seq in res:
        out.write(f'>{name}\\n{seq}\\n')
PY
wc -l sample300.fa && head -4 sample300.fa && tail -4 sample300.fa"
