/bin/bash -lc "python3 - <<'PY'
import json
from collections import Counter,defaultdict
data=json.load(open('sample300_blast.json'))
cnt=Counter(); taxcnt=Counter(); genus_counter=Counter(); humans=0; total=0
records=[]
for rep in data['BlastOutput2']:
    res=rep['report']['results']; search=res['search']; hits=search.get('hits',[])
    total+=1
    rec=[]
    for h in hits:
        desc=h.get('description',[{}])[0]
        rec.append((desc.get('sciname',''),desc.get('taxid',''),h['hsps'][0].get('identity'),h['hsps'][0].get('align_len')))
    records.append((search.get('query_title'),rec))
    if rec:
        sc=rec[0][0]; tax=rec[0][1]
        cnt[sc]+=1; taxcnt[tax]+=1
        if sc and 'Homo sapiens' in sc: humans+=1
        genus=sc.split()[0].lower() if sc else 'none'
        genus_counter[genus]+=1
print('total queries',total,'top human',humans)
print('\\nTop hit species counts:')
for sc,n in cnt.most_common(30): print(n,sc)
print('\\nTop hit taxid counts:')
for t,n in taxcnt.most_common(20): print(n,t)
print('\\nTop genus counts:')
for g,n in genus_counter.most_common(30): print(n,g)
# print records with non-human top
print('\\nNon-human top records:')
for q,rec in records:
    sc=rec[0][0] if rec else ''
    if 'Homo sapiens' not in sc:
        print(q, rec[:3])
PY"
