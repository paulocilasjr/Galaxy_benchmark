/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, re
seqs = open('sample20.fa').read().strip().split('>')[1:]
query=''
for s in seqs[:5]:
    lines=s.split('\\n',1); query += '>'+lines[0]+'\\n'+lines[1].strip()+'\\n'
params={'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','MEGABLAST':'on','QUERY':query,'FORMAT_TYPE':'JSON','HITLIST_SIZE':'5','DESCRIPTIONS':'5','ALIGNMENTS':'0'}
data=urllib.parse.urlencode(params).encode()
resp=urllib.request.urlopen(urllib.request.Request('https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi',data=data),timeout=120)
text=resp.read().decode()
print('len',len(text))
for pat in ['RID','Request ID','Error','error','QBlastInfo']:
    for m in re.finditer(pat,text):
        print(pat, m.start(), text[max(0,m.start()-150):m.start()+250].replace('\\n',' ')[:400])
        break
print(text[-1000:])
PY"
