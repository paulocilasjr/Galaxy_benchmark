import collections
n=collections.Counter();hits=[]
for l in open('candidate.sam'):
 if l.startswith('@'):continue
 x=l.split('\\t')
 if x[2]=='*': continue
 tags={y.split(':',2)[0]:y.strip().split(':',2)[-1] for y in x[11:]}
 n[(x[5],tags.get('NM'))]+=1
 if int(tags.get('NM','99'))<=2 and '75M'==x[5]:hits.append(x)
print('cigars',n.most_common(10)); print('full high quality hits',len(hits));print('bins',collections.Counter(int(x[3])//10000 for x in hits))
with open('candidate.reads.fa','w') as f:
 for x in hits:f.write('>'+x[0]+'\\n'+x[9]+'\\n')
PY"
