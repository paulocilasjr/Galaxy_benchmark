import gzip,collections,random
random.seed(42)
c=collections.Counter(); lens=collections.Counter(); sample=[]; n=0
with gzip.open('contaminated.rna.q3.fq.gz','rt') as f:
 while True:
  h=f.readline().strip()
  if not h: break
  s=f.readline().strip(); f.readline(); q=f.readline().strip()
  lens[len(s)]+=1;c[s]+=1;n+=1
  if len(sample)<200: sample.append((h,s))
  else:
   j=random.randrange(n)
   if j<200: sample[j]=(h,s)
print('reads',n,'lengths',lens,'unique',len(c))
print('first sample',sample[:5]); print('top',c.most_common(15))
with open('sample.fa','w') as f:
 for i,(h,s) in enumerate(sample):f.write(f'>q{i}\\n{s}\\n')
with open('abundant.fa','w') as f:
 for i,(s,cnt) in enumerate(c.most_common(100)):f.write(f'>a{i}_count{cnt}\\n{s}\\n')
PY"
