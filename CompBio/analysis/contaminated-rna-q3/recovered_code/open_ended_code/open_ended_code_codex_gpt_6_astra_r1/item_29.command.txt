/bin/bash -lc "python3 - <<'PY'
import collections,random
c=collections.Counter();unmapped=[];poor=[];cand={l[1:].strip() for l in open('candidate.reads.fa') if l.startswith('>')};cc=collections.Counter()
for l in open('human.sam'):
 if l.startswith('@'):continue
 x=l.rstrip().split('\\t'); flag=int(x[1]);
 if flag&256 or flag&2048:continue
 if flag&4:
  c['unmapped']+=1;unmapped.append((x[0],x[9]));
  if x[0] in cand:cc['unmapped']+=1
  continue
 tags={y.split(':',2)[0]:y.strip().split(':',2)[-1] for y in x[11:]}
 nm=int(tags.get('NM','99')); c['mapped']+=1
 if x[5]=='75M':c['full_NM_'+str(nm)]+=1
 else: c['clipped_or_indel']+=1
 if x[0] in cand:cc[(x[5],nm)]+=1
 if nm>=3 or x[5]"'!='"'75M':poor.append((x[0],x[9]))
print(c);print('candidate human matches',cc);print('poor',len(poor))
random.seed(443)
with open('unmapped.sample.fa','w') as f:
 for h,s in random.sample(unmapped,min(len(unmapped),500)):f.write(f'>{h}\\n{s}\\n')
with open('poor.sample.fa','w') as f:
 for h,s in random.sample(poor,min(len(poor),300)):f.write(f'>{h}\\n{s}\\n')
PY
python3 blast_submit.py unmapped.sample.fa"
