/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as E,collections,re
r=E.parse('sample2.xml');c=collections.Counter();bad=[]
primates=['Homo','Human','Pan','Pongo','Gorilla','Macaca','Theropithecus','Chlorocebus','Symphalangus','Nomascus','Papio','Aotus','Callithrix','Saimiri','Trachypithecus','Rhinopithecus','Piliocolobus','Colobus','Cebus','Cercocebus','Cercopithecus','Erythrocebus','Mandrillus','Mandrillus','Hylobates','Miopithecus','Semnopithecus','Microcebus']
for it in r.findall('.//Iteration'):
 q=it.findtext('Iteration_query-def'); hits=it.findall('.//Hit');hs=[]
 for hit in hits:
  h=hit.find('.//Hsp'); desc=hit.findtext('Hit_def');acc=hit.findtext('Hit_accession');genus=re.sub('PREDICTED: ','',desc).split()[0]
  hs.append((float(h.findtext('Hsp_bit-score')),int(h.findtext('Hsp_identity')),int(h.findtext('Hsp_align-len')),genus,desc,acc))
 if not hs:bad.append((q,'NO_HIT'));continue
 c[hs[0][3]]+=1
 if not any(x[3] in primates for x in hs):bad.append((q,hs))
print(c)
for q,hs in bad:print(q,str(hs)[:1600])
PY"
