/bin/bash -lc "set -o pipefail
curl --max-time 120 -sS -A 'Mozilla/5.0' -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' \\
  --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' \\
  --data-urlencode 'MEGABLAST=on' --data-urlencode 'HITLIST_SIZE=20' --data-urlencode 'QUERY@work/sample_1000.fa' > work/blast1000_put.html
awk 'BEGIN{IGNORECASE=1} /RID =|RTOE =|NUM_QUERIES/ {print}' work/blast1000_put.html | sed -n '1,20p'
wc -c work/blast1000_put.html"
