/bin/bash -lc "unzip -p plasmodium.zip 'ncbi_dataset/data/GCF_001602025.1/*_genomic.fna' > plasmodium_gaboni.fna
unzip -p mycoplasma.zip 'ncbi_dataset/data/GCF_001705605.1/*_genomic.fna' > mycoplasma_hyorhinis.fna
for ref in plasmodium_gaboni mycoplasma_hyorhinis; do
  minimap2 -ax sr --secondary=no \""'${ref}.fna" contaminated.rna.q3.fq.gz 2> "${ref}.map.log" | samtools view -F 4 -o "${ref}.mapped.sam" -
  awk '"'BEGIN{n=0;hi=0;al=0;nm=0} {n++; cigar="'$6; x=cigar; a=0; while(match(x,/([0-9]+)[M=X]/,m)){a+=m[1]; x=substr(x,RSTART+RLENGTH)}; e=0; for(i=12;i<=NF;i++) if($i~/''^NM:i:/){split($i,z,":");e=z[3]}; al+=a;nm+=e; if(a>=60 && e<=3)hi++} END{print "'"'\""'$ref"'"'\", \"mapped=\"n, \"high_conf=\"hi, \"mean_aln=\"al/n, \"mean_nm=\"nm/n}' \""'${ref}.mapped.sam"
done'
