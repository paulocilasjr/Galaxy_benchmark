/bin/bash -lc "python - <<'PY'
import json,re
from collections import Counter,defaultdict
s=json.load(open('cistrome_samples.json'))
rows=[]
with open('borzoi_targets_human.txt') as f:
 h=f.readline().rstrip().split('\\t')
 for l in f: rows.append(dict(zip(h,l.rstrip().split('\\t'))))
# Extract ENCODE experiment ids from file and ATAC GSM ids
borzoi_encsr=set(); borzoi_gsm=set(); borzoi_ids=set()
for r in rows:
 borzoi_ids.add(r['identifier'])
 for x in re.findall(r'ENCSR[A-Z0-9]+',r['file']+' '+r['identifier']): borzoi_encsr.add(x)
 for x in re.findall(r'GSM\\d+',r['file']+' '+r['identifier']): borzoi_gsm.add(x)
print('Borzoi ENCSR experiments',len(borzoi_encsr),'GSM',len(borzoi_gsm))
# Cistrome matches
matches=[]
for x in s.values():
 eid=x.get('external_id','')
 norm=re.sub(r'_\\d+"'$'"','',eid)
 if norm in borzoi_encsr or eid in borzoi_gsm:
  matches.append(x)
print('matches',len(matches),'unique Cistrome IDs',len({x['id'] for x in matches}))
print('by ext type',Counter(x['external_id_type'] for x in matches))
print('by borzoi assay inferred')
for x in matches[:20]: print(x['id'],x['external_id'],x['title'],x['experiment_type'])
print('matched ENCSR ids',len({re.sub(r'_\\d+"'$'"','',x['external_id']) for x in matches if x['external_id_type']=='ENCODE'}))
print('matched GSM ids',len({x['external_id'] for x in matches if x['external_id_type']=='GEO'}))
# list unmatched Cistrome ENCODE vs Borzoi
ce=[x for x in s.values() if x['external_id_type']=='ENCODE']
print('Cistrome ENCODE total',len(ce),'unique normalized',len({re.sub(r'_\\d+"'$'"','',x['external_id']) for x in ce}),'matches',len([x for x in ce if re.sub(r'_\\d+"'$'"','',x['external_id']) in borzoi_encsr]))
print('matched IDs count distributions')
print(Counter(re.sub(r'_\\d+"'$'"','',x['external_id']) for x in matches if x['external_id_type']=='ENCODE').most_common(20))
PY"
