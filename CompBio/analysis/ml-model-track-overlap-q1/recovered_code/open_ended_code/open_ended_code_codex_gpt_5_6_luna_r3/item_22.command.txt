/bin/bash -lc "gzip -cd borzoi/data/targets_human.txt.gz > borzoi_targets_human.txt && python - <<'PY'
from collections import Counter
p='borzoi_targets_human.txt'
rows=[]
with open(p) as f:
 h=f.readline().rstrip('\\n').split('\\t')
 for l in f:
  x=l.rstrip('\\n').split('\\t'); rows.append(dict(zip(h,x)))
print('n',len(rows),h)
print('descriptions prefix',Counter(x['description'].split(':',1)[0] for x in rows).most_common())
for k,v in Counter(x['description'].split(':',1)[0] for x in rows).most_common(): print(k,v)
print('file prefix',Counter(x['file'].split('/')[5] if len(x['file'].split('/'))>5 else x['file'] for x in rows).most_common(20))
print('\\nlast 20')
for x in rows[-20:]: print(x)
PY"
