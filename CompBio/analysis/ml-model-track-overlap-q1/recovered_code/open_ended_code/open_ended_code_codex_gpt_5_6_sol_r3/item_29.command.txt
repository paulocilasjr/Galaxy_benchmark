/bin/bash -lc "python - <<'PY'
import json,re,csv,collections
samples={a['id']:a for a in json.load(open('cistrome_all.json'))['samples']}
sei=[]
for idx,s in enumerate(open('sei-framework/model/target.names')):
 if m:=re.search(r'ID:(\\d+)\\s*"'$'"',s):
  a=samples[int(m.group(1))]; sei.append((idx,s.strip(),a))
bor=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
# Borzoi source accession taken as GSM identifier or ENCSR pathname.
bor_gsm=set()
bor_encsr=set()
for r in bor:
 bor_gsm.update(re.findall(r'GSM\\d+',r['identifier']+' '+r['file']))
 bor_encsr.update(re.findall(r'ENCSR[A-Z0-9]+',r['file']+' '+r['identifier']))
print('bor gsm',len(bor_gsm),'encsr',len(bor_encsr))
matched=[]
for idx,name,a in sei:
 ext=a['external_id']
 key=re.match(r'(ENCSR[A-Z0-9]+)',ext)
 hit=ext in bor_gsm or (key and key.group(1) in bor_encsr)
 if hit: matched.append((idx,name,a,key.group(1) if key else ext))
print('matched tracks',len(matched),'rounded',round(len(matched)/100)*100)
print('by ext type',collections.Counter(z[2]['external_id_type'] for z in matched))
print('unique key',len(set(z[3] for z in matched)),collections.Counter(z[2]['experiment_type'] for z in matched))
print('matched first')
for z in matched[:20]:print(z)
print('bor gsm not sei direct', len(bor_gsm-set(a['external_id'] for _,_,a in sei)))
print('match GEO',sorted(bor_gsm & set(a['external_id'] for _,_,a in sei))[:10])
PY"
