/bin/bash -lc "python3 - <<'PY'
import json,csv,re,collections
sei=json.load(open('sei_cistrome_meta.json'))
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
# Build Borzoi source GSM -> source type set
borzoi_gsm_src=collections.defaultdict(set)
# direct
for r in rows:
 if re.fullmatch(r'GSM\\d+',r['identifier']):
  p=r['file'].split('/')
  borzoi_gsm_src[r['identifier']].add(f\"{p[6]}:{p[7]}\")
# ENCODE experiments
enc=json.load(open('borzoi_encode_experiments.json'))
for d in enc:
 if d['gsms']:
  for gsm in d['gsms'].split(','): borzoi_gsm_src[gsm].add(d['assay'])
# Sei matched GSM with factor
id2track={}
for l in open('sei-framework/model/target.names'):
 p=[x.strip() for x in l.rstrip().split('|')]
 if len(p)>=3 and p[2].startswith('ID:'): id2track[p[2][3:]]=(p[0],p[1])
c=collections.Counter()
examples=[]
for s in sei.values():
 if s.get('external_id_type')"'!='"'ENCODE' and s.get('external_id') in borzoi_gsm_src:
  gsm=s['external_id']; srcs=tuple(sorted(borzoi_gsm_src[gsm])); factor=id2track[s['id']][1]
  c[(factor,srcs)]+=1
  examples.append((s['id'],factor,gsm,srcs))
print('total',sum(c.values()))
print('by factor/src')
for k,v in c.most_common(30): print(k,v)
print('RNA match examples')
for e in examples:
 if any('RNA' in x or ':rna' in x for x in e[3]): print(e)
PY"
