import csv,re,json,collections
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))

 en=f[f.index('encode')+1]
 acc=mm.group(1); ds=fm.get(acc,{}).get('dataset',''); cur=ds.rstrip('/').split('/')[-1] if ds else ''
 reps=tuple(fm.get(acc,{}).get('biological_replicates',[]))
 if en in rep_by_exp and rep_by_exp[en][:2] "'!= (cur,reps):
  print('"'conflict',en,rep_by_exp[en],(cur,reps))
 rep_by_exp[en]=(acc,cur,reps)
c=collections.Counter(v[2] for v in rep_by_exp.values())
print('exp',len(rep_by_exp))
for reps,n in c.most_common(20): print(reps,n)
print('examples reps [1,2]', [ (k,v) for k,v in list(rep_by_exp.items())[:20] if len(v[2])>1][:10])
