/bin/bash -lc "python3 - <<'PY'
from collections import Counter
rows=[l.rstrip('\\n') for l in open('sei-framework/model/target.names')]
print('rows',len(rows))
parsed=[ [x.strip() for x in l.split('|')] for l in rows]
cat=Counter(); track_ids=[]; other=[]
for p in parsed:
    if len(p)>=3 and p[2].startswith('ID:'):
      cat['cistrome']+=1; track_ids.append(p[2][3:])
    else:
      cat[p[2] if len(p)>2 else p[0]]+=1; other.append(p)
print('cat',cat)
print('track_ids',len(track_ids),'unique',len(set(track_ids)))
c=Counter(track_ids)
print('duplicate track ids',sum(1 for v in c.values() if v>1),'max',max(c.values()))
for k,v in c.most_common(10): print(k,v)
print('other first',other[:20])
PY"
