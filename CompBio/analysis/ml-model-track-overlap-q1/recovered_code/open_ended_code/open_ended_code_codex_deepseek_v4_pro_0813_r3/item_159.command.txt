/bin/bash -lc "python3 - <<'PY'
import json,csv,re,collections
# Sei meta
meta=json.load(open('sei_cistrome_meta.json'))
sei_geo={}; sei_enc={}
for cid,v in meta.items():
    et=v.get('external_id_type'); ext=v.get('external_id')
    if et=='GEO' and ext: sei_geo[cid]=ext
    elif et=='ENCODE' and ext: sei_enc[cid]=ext
print('sei geo tracks',len(sei_geo),'enc',len(sei_enc))
print('enc examples',list(sei_enc.items())[:20])
# Borzoi path ENCSR
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
bor_encsrs=set()
for r in rows:
    f=r['file'].strip().split('/')
    if 'encode' in f:
        i=f.index('encode')
        if i+1<len(f): bor_encsrs.add(f[i+1])
print('borzoi path encsrs',len(bor_encsrs))
# direct GSM
bor_gsm_direct=set()
for r in rows:
    ident=r['identifier'].strip()
    if re.match(r'GSM\\d+"'$'"',ident): bor_gsm_direct.add(ident)
print('bor direct gsm',len(bor_gsm_direct))
# existing exp
exp=json.load(open('borzoi_encode_experiments.json'))
print('exp entries',len(exp))
exp_acc=set(e['accession'] for e in exp)
print('missing path not exp',bor_encsrs-exp_acc)
# Sei encode base forms
for ext in list(sei_enc.values())[:20]: print(ext)
print('base unique',len(set(ext.split('_')[0] for ext in sei_enc.values())))
print('base_rep unique',len(set(ext for ext in sei_enc.values())))
PY"
