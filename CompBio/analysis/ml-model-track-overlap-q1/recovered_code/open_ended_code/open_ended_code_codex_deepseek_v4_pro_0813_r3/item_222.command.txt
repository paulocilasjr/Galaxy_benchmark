/bin/bash -lc "python3 - <<'PY'
import gzip,csv,re,json
meta=json.load(open('sei_cistrome_meta.json'))
sei_geo=set(v['external_id'] for v in meta.values() if v.get('external_id_type')=='GEO')
sei_enc=set(v['external_id'].split('_')[0] for v in meta.values() if v.get('external_id_type')=='ENCODE')
mouse_gsm=set(); mouse_enc=set()
with gzip.open('borzoi/data/targets_mouse.txt.gz','rt') as f:
 f.readline()
 for l in f:
  p=l.rstrip('\\n').split('\\t'); ident=p[1].strip(); file=p[4].strip() if len(p)>4 else ''
  if re.match(r'GSM\\d+"'$'"',ident): mouse_gsm.add(ident)
  elif re.match(r'ENCFF',ident) and 'encode' in file:
   parts=file.split('/'); mouse_enc.add(parts[parts.index('encode')+1] if 'encode' in parts else '')
print('overlap gsm',len(sei_geo & mouse_gsm),'enc',len(sei_enc & mouse_enc))
print('examples gsm',list(sei_geo & mouse_gsm)[:20])
print('examples enc',list(sei_enc & mouse_enc)[:20])
PY"
