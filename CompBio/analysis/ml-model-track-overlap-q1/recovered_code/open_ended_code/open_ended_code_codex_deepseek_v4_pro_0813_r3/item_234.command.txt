import json
reps=json.load(open('borzoi_exp_reps.json'))
libs=set()

 for r in e.get('replicates',[]):
  if r.get('library'): libs.add(r['library'])
mp=json.load(open('borzoi_lib_gsm_map.json'))
missing=sorted(libs-set(mp))
print('missing libs',missing)
# print experiments/reps for missing
for e in reps:
 for r in e.get('replicates',[]):
  if r.get('library') in missing:
   print(e['accession'],r)
PY"
