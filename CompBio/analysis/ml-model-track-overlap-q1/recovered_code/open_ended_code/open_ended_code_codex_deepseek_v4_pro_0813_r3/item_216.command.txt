/bin/bash -lc "python3 - <<'PY'
import requests,json
# query ENCSR418RRF with target and replicates
for acc in ['ENCSR418RRF','ENCSR994NPK','ENCSR746ZXP']:
 params=[('type','Experiment'),('format','json'),('limit','all'),('field','accession'),('field','assay_title'),('field','target.label'),('field','dbxrefs'),('field','replicates.library.biosample.accession'),('field','replicates.antibody.accession'),('field','possible_controls'),('field','controls')]
 params.append(('accession',acc))
 r=requests.get('https://www.encodeproject.org/search/',params=params,timeout=120)
 print('\\n',acc,r.status_code)
 for g in r.json().get('@graph',[]):
  print('assay',g.get('assay_title'),'target',g.get('target',{}).get('label') if isinstance(g.get('target'),dict) else g.get('target'),'dbxrefs',g.get('dbxrefs'))
  print('reps',[(x.get('library',{}).get('biosample',{}).get('accession'), x.get('antibody',{}).get('accession') if x.get('antibody') else None) for x in g.get('replicates',[])])
  print('possible_controls',g.get('possible_controls'),'controls',g.get('controls'))
PY"
