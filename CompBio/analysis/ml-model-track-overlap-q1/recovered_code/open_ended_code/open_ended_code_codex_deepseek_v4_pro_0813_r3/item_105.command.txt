/bin/bash -lc "python3 - <<'PY'
import json,csv,re,collections
sei=json.load(open('sei_cistrome_meta.json'))
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
borzoi_enc=set(); direct_gsm=set()
for r in rows:
 m=re.search(r'/ENCSR[0-9A-Z]+/',r['file'])
 if m: borzoi_enc.add(m.group(0).strip('/'))
 if re.fullmatch(r'GSM\\d+',r['identifier']): direct_gsm.add(r['identifier'])
enc=json.load(open('borzoi_encode_experiments.json'))
borzoi_gsm=set(direct_gsm)
for d in enc:
 if d['gsms']: borzoi_gsm.update(d['gsms'].split(','))
# counts and overlapping groups by track factor?
matched=[]
for s in sei.values():
 if s.get('external_id_type')=='ENCODE':
  if s['external_id'].split('_')[0] in borzoi_enc: matched.append((s,'enc'))
 elif s.get('external_id') in borzoi_gsm: matched.append((s,'gsm'))
print('total matched',len(matched))
print('by ext type',collections.Counter(m[1] for m in matched))
# match by assay? we don't have Sei factor perhaps from target names, join by id
id2track={}
for l in open('sei-framework/model/target.names'):
 p=[x.strip() for x in l.rstrip().split('|')]
 if len(p)>=3 and p[2].startswith('ID:'): id2track[p[2][3:]]=(p[0],p[1])
fac=collections.Counter(id2track[m[0]['id']][1] for m in matched)
print(fac.most_common(20))
PY"
