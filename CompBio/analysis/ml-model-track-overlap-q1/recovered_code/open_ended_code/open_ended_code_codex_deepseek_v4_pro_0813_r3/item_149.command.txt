from collections import Counter,defaultdict
rows=[l.rstrip('\\n').split('|') for l in open('sei-framework/model/target.names')]
print('rows',len(rows))
# categories
cat=Counter()
track_ids=[]
for p in rows:
    src=p[0].strip() if p else ''
    if len(p)>=3 and p[2].startswith('ID:'):
      cat['cistrome']+=1
      track_ids.append(p[2][3:].strip())
    else:
      # maybe source field? print first
      cat[p[2].strip() if len(p)>2 else src]+=1
print('cat',cat)
print('track_ids',len(track_ids),'unique',len(set(track_ids)))
# duplicates frequency
c=Counter(track_ids)
print('dup ids',sum(1 for v in c.values() if v>1),'max',max(c.values()))
for k,v in c.most_common(20): print(k,v)
# inspect no cistrome rows
for p in rows:
    if not (len(p)>=3 and p[2].startswith('ID:')):
        print('NON',p); 
        if len([q for q in rows if not(len(q)>=3 and q[2].startswith('ID:'))])>20: pass
# only first 20
PY"
