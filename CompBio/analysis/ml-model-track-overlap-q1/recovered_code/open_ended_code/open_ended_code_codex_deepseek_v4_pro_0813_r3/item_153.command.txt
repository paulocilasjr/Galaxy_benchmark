/bin/bash -lc "python3 - <<'PY'
import csv,re,collections
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
accs1=[]; accs2=[]; bad=[]
for r in rows:
 ident=r['identifier'].strip()
 m=re.match(r'(ENCFF\\d{3}[A-Z]{3})[+-]?"'$'"', ident)
 if m: accs1.append(m.group(1))
 m2=re.match(r'(ENCFF\\d{3}[A-Z]{3})',ident)
 if m2:
  accs2.append(m2.group(1))
  if not (m2 and (m2.end()==len(ident) or (m2.end()<len(ident) and ident[m2.end():] in '+-'))):
   bad.append((ident,m2.group(1)))
print('strict',len(set(accs1)),'broad',len(set(accs2)))
print('broad-strict',sorted(set(accs2)-set(accs1))[:20],len(set(accs2)-set(accs1)))
print('bad',bad[:20])
PY"
