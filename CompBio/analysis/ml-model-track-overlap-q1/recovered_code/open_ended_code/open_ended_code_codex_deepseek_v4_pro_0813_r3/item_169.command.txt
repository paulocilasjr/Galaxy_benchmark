/bin/bash -lc "python3 - <<'PY'
import json,requests,time
meta=json.load(open('sei_cistrome_meta.json'))
enc=[(k,v) for k,v in meta.items() if v.get('external_id_type')=='ENCODE']

# query details for a few ids
ids=[v['id'] for _,v in enc[:5]]
r=requests.get('https://db3.cistrome.org/cistrome/samples',params={'ids':','.join(ids)},timeout=60)
print(r.status_code)
print(json.dumps(r.json(),indent=2)[:5000])
PY"
