/bin/bash -lc "python3 - <<'PY'
import csv,re,collections,os
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
print('n',len(rows))
# classify identifiers/file paths
cnt=collections.Counter()
examples={}
for r in rows:
    ident=r['identifier'].strip(); file=r['file'].strip()
    m=re.match(r'ENCFF\\d{3}[A-Z]{3}',ident)
    if m:
        k='ENCFF'; cnt[k]+=1
    elif ident.startswith('CNhs'):
        k='FANTOM_CAGE'; cnt[k]+=1
    elif re.match(r'GSM\\d+',ident):
        k='GEO_GSM'; cnt[k]+=1
    elif ident.startswith('ENCSR'):
        k='ENCSR'; cnt[k]+=1
    elif 'gtex' in file.lower() or 'recount' in file.lower() or ident.startswith('GTEX'):
        k='GTEX_RECOUNT'; cnt[k]+=1
    elif 'catlas' in file.lower() or 'atac' in file.lower():
        k='CATlas_ATAC'; cnt[k]+=1
    else:
        k='OTHER'; cnt[k]+=1
    examples.setdefault(k,(ident,file))
print(cnt)
for k,v in examples.items(): print(k,v)
# files path first segments
seg=collections.Counter()
for r in rows:
    seg[r['file'].strip().split('/')[0]]+=1
print('file first seg',seg.most_common(20))
PY"
