python3 - <<'PY'
import csv,collections,re
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'), delimiter='\\t'))
print(len(rows))
# parse path components, descriptions
cats=collections.Counter()
for r in rows:
 p=r['file']
 cats[p.split('/')[5] if len(p.split('/'))>5 else '?'] +=1
print(cats)
# identifier types
ids=collections.Counter()
for r in rows:
 ident=r['identifier']
 # classify
 if ident.startswith('CNhs'): ids['FANTOM_CAGE'] +=1
 elif ident.startswith('GTEX-'): ids['GTEX_RNA'] +=1
 elif ident.startswith('ENCFF'): ids['ENCODE_FILE'] +=1
 elif ident.startswith('ENC'): ids['ENCODE_other'] +=1
 elif ident.startswith('S'):
   ids['maybe_other'] +=1
 else:
   ids[ident[:20]] +=1
print(ids.most_common(50))
print('description cats')
print(collections.Counter(r['description'] for r in rows).most_common(80))
PY"
