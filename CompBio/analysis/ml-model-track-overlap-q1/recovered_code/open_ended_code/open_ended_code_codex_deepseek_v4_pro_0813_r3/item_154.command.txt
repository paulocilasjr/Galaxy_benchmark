/bin/bash -lc "python3 - <<'PY'
import json,csv,re
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
encsrs=set()
for r in rows:
 file=r['file'].strip()
 parts=file.split('/')
 if 'encode' in parts:
  idx=parts.index('encode')
  if idx+1<len(parts): encsrs.add(parts[idx+1])
print('borzoi encsr',len(encsrs))
# strict unique encff
encffs=set()
for r in rows:
 ident=r['identifier'].strip()
 m=re.match(r'ENCFF\\d{3}[A-Z]{3}',ident)
 if m: encffs.add(m.group(0))
print('encff',len(encffs))
exp=json.load(open('borzoi_encode_experiments.json'))
print('exp',len(exp), 'keys', exp[0].keys())
exp_acc=set(e['accession'] for e in exp)
print('missing encsr in exp', sorted(encsrs-exp_acc), len(encsrs-exp_acc))
print('exp acc not in target',len(exp_acc-encsrs), list(exp_acc-encsrs)[:5])
# missing encff
meta=json.load(open('borzoi_encff_meta.json'))
print('meta',len(meta))
print('missing encff', sorted(encffs-set(meta)), len(encffs-set(meta)))
PY"
