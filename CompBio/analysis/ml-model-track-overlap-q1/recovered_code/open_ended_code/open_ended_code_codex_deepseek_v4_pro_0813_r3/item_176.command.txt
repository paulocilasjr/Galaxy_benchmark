/bin/bash -lc "python3 - <<'PY'
import json,csv,re
meta=json.load(open('sei_cistrome_meta.json'))
sei_geo=set()
for cid,v in meta.items():
 if v.get('external_id_type')=='GEO':
  sei_geo.add(v['external_id'])
print('sei geo',len(sei_geo))
rows=list(csv.DictReader(open('borzoi_targets_human.tsv'),delimiter='\\t'))
bor_direct=set()
for r in rows:
 ident=r['identifier'].strip()
 if re.match(r'GSM\\d+"'$'"',ident): bor_direct.add(ident)
print('bor direct gsm',len(bor_direct))
exp=json.load(open('borzoi_encode_experiments.json'))
bor_enc_gsm=set()
for e in exp:
 for g in e.get('gsms','').split(','):
  g=g.strip()
  if g: bor_enc_gsm.add(g)
print('bor encode gsm',len(bor_enc_gsm))
print('intersection sei vs direct',len(sei_geo & bor_direct))
print('intersection sei vs enc',len(sei_geo & bor_enc_gsm))
print('union bor all gsm',len(bor_direct|bor_enc_gsm))
print('intersection sei vs all',len(sei_geo & (bor_direct|bor_enc_gsm)))
# examples
print('example direct matches',list(sei_geo & bor_direct)[:20])
print('example enc matches',list(sei_geo & bor_enc_gsm)[:20])
PY"
