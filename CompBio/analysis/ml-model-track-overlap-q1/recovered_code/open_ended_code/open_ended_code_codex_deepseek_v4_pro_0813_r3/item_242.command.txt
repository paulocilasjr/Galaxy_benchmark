/bin/bash -lc "python3 - <<'PY'
import json
# Compare broad vs library target GSMs and Sei overlap counts
meta=json.load(open('sei_cistrome_meta.json'))
sei=set(v['external_id'] for v in meta.values() if v.get('external_id_type')=='GEO')
exp=json.load(open('borzoi_encode_experiments.json'))
broad=set()
for e in exp:
 for g in e.get('gsms','').split(','):
  if g: broad.add(g)
lib_map=json.load(open('borzoi_lib_gsm_map.json'))
reps=json.load(open('borzoi_exp_reps.json'))
target=set()
for e in reps:
 for r in e.get('replicates',[]):
  if r.get('library') in lib_map: target.update(lib_map[r.get('library')])
extra=broad-target
print('extra total',len(extra),'extra in Sei',len(extra&sei), 'sei target library',len(target&sei),'sei broad',len(broad&sei))
print('extra sei examples',sorted(extra&sei)[:50])
PY"
