import urllib.request, json, re, time, csv
from collections import defaultdict
lines=open('sei_repo/model/target.names').read().splitlines()
sei=[]
for i,line in enumerate(lines):
 m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if m: sei.append((i,int(m.group(1)),line))
print('sei tracks',len(sei),'unique',len(set(x[1] for x in sei)))
ids=sorted(set(x[1] for x in sei))
print('id range',min(ids),max(ids))
# test chunks sizes by fetching first chunks
for chunk in [100,500,1000]:
 c=ids[:chunk]
 url='https://db3.cistrome.org/cistrome/samples?ids='+','.join(map(str,c))+'&limit='+str(chunk)
 try:
  t=time.time(); data=urllib.request.urlopen(url,timeout=60).read().decode(); j=json.loads(data)
  print('chunk',chunk,'samples',len(j.get('samples',[])),'time',round(time.time()-t,2))
 except Exception as e:
  print('chunk',chunk,'ERR',repr(e)[:200])
PY"
