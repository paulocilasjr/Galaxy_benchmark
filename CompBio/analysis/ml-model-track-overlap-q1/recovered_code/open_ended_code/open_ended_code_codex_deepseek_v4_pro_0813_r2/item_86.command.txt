/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for acc in ['ENCFF076JMD','ENCFF723FDT','ENCFF862JVF']:
 url=f'https://www.encodeproject.org/files/{acc}/?format=json'
 print('---',acc)
 try:
  data=json.loads(urllib.request.urlopen(url,timeout=30).read().decode())
  for key in ['accession','dataset','replicate','biological_replicates','technical_replicates','experiment','file_format','output_type','derived_from']:
   if key in data: print(key, data[key])
  print('experiment accession', data.get('dataset'))
  print('replicate', data.get('replicate'))
 except Exception as e: print('ERR',e)
PY"
