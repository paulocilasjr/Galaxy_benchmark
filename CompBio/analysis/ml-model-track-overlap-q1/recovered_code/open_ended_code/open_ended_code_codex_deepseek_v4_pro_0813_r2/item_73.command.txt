/bin/bash -lc "python3 - <<'PY'
import csv,re
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 vals=[]
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  p=r['file']; m=re.search(r'/(?:encode|geo|fantom|recount3)/(["'^/]+)/'"',p)
  tok=m.group(1) if m else r['identifier']
  if tok.startswith('ENCSR'):
   vals.append(tok)
 print(fn,'ENCSR count',len(vals),'with underscore',sum('_' in v for v in vals), 'unique with suffix', len(set(vals)))
PY"
