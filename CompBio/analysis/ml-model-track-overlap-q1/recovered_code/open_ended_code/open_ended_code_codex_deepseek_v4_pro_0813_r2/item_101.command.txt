/bin/bash -lc "python3 - <<'PY'
import csv,re
from collections import defaultdict
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 d=defaultdict(set)
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  if not r['description'].startswith(('CHIP:','DNASE:','ATAC:')): continue
  m=re.search(r'/(?:encode|geo)/(["'^/]+)/'"',r['file'])
  if m and r['identifier'].startswith('ENCFF'):
   d[m.group(1)].add(r['identifier'].rstrip('+-'))
 print(fn,'non-RNA ENCSR unique',len(d),'with >1',sum(len(v)>1 for v in d.values()))
 for k,v in d.items():
  if len(v)>1:
   print('multi',k,v); break
PY"
