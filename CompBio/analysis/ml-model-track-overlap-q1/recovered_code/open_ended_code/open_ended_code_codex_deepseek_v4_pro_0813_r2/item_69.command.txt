/bin/bash -lc "python3 - <<'PY'
import json,csv,re
mapping=json.load(open('sei_cistrome_map.json'))
borz={}
for fn in ['borzoi_targets_human.tsv','borzoi_targets_mouse.tsv']:
 for r in csv.DictReader(open(fn),delimiter='\\t'):
  p=r['file']; m=re.search(r'/(?:encode|geo|fantom|recount3)/(["'^/]+)/'"',p)
  key=m.group(1) if m else r['identifier']
  if key.startswith('ENCSR') or key.startswith('GSM'):
   borz.setdefault(key,[]).append((fn,r['identifier'],r['description']))
# print some matched ENCODE and GEO
import random
matched_enc=[]; matched_geo=[]
for line in open('sei_repo/model/target.names'):
 line=line.rstrip('\\n'); m=re.search(r'ID:(\\d+)\\s*"'$'"',line)
 if not m: continue
 s=mapping[m.group(1)]; typ=s['external_id_type']; ext=s['external_id']
 norm=re.sub(r'_\\d+"'$'"','',ext) if typ=='ENCODE' else ext
 if norm in borz:
  (matched_enc if typ=='ENCODE' else matched_geo).append((m.group(1),ext,norm,line,borz[norm][:2]))
print('matched ENCODE',len(matched_enc),'GEO',len(matched_geo))
for x in matched_enc[:5]: print('ENC',x)
for x in matched_geo[:5]: print('GEO',x)
PY"
