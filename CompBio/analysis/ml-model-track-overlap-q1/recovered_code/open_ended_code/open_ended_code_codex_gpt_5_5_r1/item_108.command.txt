/bin/bash -lc "python - <<'PY'
import json, collections, re
matches=[json.loads(l) for l in open('matched_sei_tracks.jsonl')]
print('matches',len(matches))
print('by external type',collections.Counter(m['external_id_type'] for m in matches))
print('by hit prefix',collections.Counter(re.match(r'[A-Z]+', m['hit_keys'][0]).group(0) for m in matches))
print('prefix any hit',collections.Counter(re.match(r'[A-Z]+', k).group(0) for m in matches for k in m['hit_keys']))
for pref in ['GSE','GSM','ENCSR']:
 print('\\n',pref)
 n=0
 for m in matches:
  if any(k.startswith(pref) for k in m['hit_keys']):
   print(m['cistrome_id'], m['external_id'], m['hit_keys'][:5], m['target'])
   n+=1
   if n>=8: break
PY"
