/bin/bash -lc "python - <<'PY'
import re,json,collections
from openpyxl import load_workbook
wb=load_workbook('sei_supp_tables.xlsx',read_only=True,data_only=True)
ids=[]
for row in wb['Supp Table 1'].iter_rows(min_row=3,max_col=1,values_only=True):
 s=row[0]
 if s and 'ID:' in s: ids.append(int(re.search(r'ID:(\\d+)"'$'"',s).group(1)))
ss={s['id'] for s in json.load(open('cistrome_all_samples.json'))['samples']}
print(min(ids),max(ids),len(set(ids)))
print('id bins',collections.Counter(i//10000 for i in ids))
print('missing bins',collections.Counter(i//10000 for i in ids if i not in ss))
print('mapped ranges',[(b,sum(i in ss for i in ids if i//10000==b),sum(i//10000==b for i in ids)) for b in sorted(set(i//10000 for i in ids))])
print('missing minmax',min(i for i in ids if i not in ss),max(i for i in ids if i not in ss))
print('missing sample tail',ids[-30:])
PY"
