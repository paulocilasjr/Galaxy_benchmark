/bin/bash -lc "python - <<'PY'
from openpyxl import load_workbook
import re,collections
wb=load_workbook('sei_supp_tables.xlsx',read_only=True,data_only=True)
ws=wb['Supp Table 1']
vals=[]
for row in ws.iter_rows(min_row=3,max_col=3,values_only=True):
    s=row[0]
    if s: vals.append(s)
print('profiles',len(vals))
print('ID suffix',sum(bool(re.search(r'\\bID:\\s*["'^|]+$'"',s)) for s in vals), 'contains ID',sum('ID:' in s for s in vals))
print('no ID examples', [s for s in vals if 'ID:' not in s][:20])
print('last examples',vals[-20:])
print('contains ID counts',collections.Counter('ID:' in s for s in vals))
PY"
