/bin/bash -lc "python - <<'PY'
import re,requests
from openpyxl import load_workbook
wb=load_workbook('sei_supp_tables.xlsx',read_only=True,data_only=True)
ids=[]
for row in wb['Supp Table 1'].iter_rows(min_row=3,max_col=1,values_only=True):
 if row[0] and 'ID:' in row[0]: ids.append(int(re.search(r'ID:(\\d+)"'$'"',row[0]).group(1)))
for n in [10,50,100,200,500]:
 a=ids[:n]
 r=requests.get('https://db3.cistrome.org/cistrome/samples',params={'ids':','.join(map(str,a))},timeout=60)
 print(n,r.status_code,len(r.text),len(r.json().get('samples',[])),[x['id'] for x in r.json().get('samples',[])[:3]], [x['id'] for x in r.json().get('samples',[])[-3:]])
PY"
