/bin/bash -lc "python - <<'PY'
import json,re,collections
from openpyxl import load_workbook
wb=load_workbook('sei_supp_tables.xlsx',read_only=True,data_only=True)
ids=[]; profiles={}
for row in wb['Supp Table 1'].iter_rows(min_row=3,max_col=1,values_only=True):
 s=row[0]
 if s and 'ID:' in s:
  i=int(re.search(r'ID:(\\d+)"'$'"',s).group(1)); ids.append(i); profiles[i]=s
samples={s['id']:s for name in ['cistrome_all_samples.json','cistrome_all_samples_2.json'] for s in json.load(open(name))['samples']}
print('mapped',sum(i in samples for i in ids),'missing',sum(i not in samples for i in ids))
print('missing',[(i,profiles[i]) for i in ids if i not in samples][:20])
print('external types',collections.Counter(samples[i].get('external_id_type') for i in ids if i in samples))
print('species',collections.Counter(samples[i].get('species') for i in ids if i in samples))
print('prefixes',collections.Counter(re.match(r'[A-Za-z]+',samples[i].get('external_id','')).group(0) if re.match(r'[A-Za-z]+',samples[i].get('external_id','')) else '' for i in ids if i in samples))
PY"
