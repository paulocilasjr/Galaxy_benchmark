/bin/bash -lc "python3 - <<'PY'
import urllib.request,re,html
url='https://www.nature.com/articles/s41588-024-02053-6'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read().decode('utf-8','ignore')
# Strip tags around relevant sections
text=re.sub('<script.*?</script>','',data,flags=re.S)
text=re.sub('<style.*?</style>','',text,flags=re.S)
text=re.sub('<["'^>]+>'"',' ',text)
text=html.unescape(text)
text=re.sub(r'\\s+',' ',text)
for pat in ['Cistrome','ENCODE','FANTOM','Supplementary Data','Data availability','training tracks','7611','21,907']:
 print('\\n###',pat)
 for m in re.finditer(pat,text,re.I):
  s=max(0,m.start()-300); e=min(len(text),m.end()+500)
  print(text[s:e][:1000])
  break
PY"
