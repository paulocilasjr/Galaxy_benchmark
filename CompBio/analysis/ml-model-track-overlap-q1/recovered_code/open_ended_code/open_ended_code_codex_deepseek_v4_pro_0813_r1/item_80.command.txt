/bin/bash -lc "python3 - <<'PY'
from pypdf import PdfReader
import os
for fn in os.listdir('data/borzoi_supp'):
 path='data/borzoi_supp/'+fn
 try:
  reader=PdfReader(path)
  text='\\n'.join(page.extract_text() or '' for page in reader.pages)
  open('data/borzoi_supp/'+fn+'.txt','w').write(text)
  print(fn,'pages',len(reader.pages),'chars',len(text),'cistrome',text.lower().count('cistrome'))
  for i,page in enumerate(reader.pages):
   p=page.extract_text() or ''
   if 'cistrome' in p.lower() or 'Cistrome' in p:
    print(' page',i,p[:500])
 except Exception as e: print('ERR',fn,e)
PY"
