# query file metadata
for f in ['ENCFF078OIB']:
 d=json.load(urllib.request.urlopen(f'https://www.encodeproject.org/files/{f}/?format=json',timeout=30))
 print('file',f,'bio',d['biological_replicates'],'tech',d['technical_replicates'],'derived_from',d.get('derived_from'))
PY"
