import urllib.request,re,html
url='https://www.nature.com/articles/s41588-022-01102-2'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read().decode('utf-8','ignore')
for m in re.finditer(r'href=\"(["'^"]*(?:MOESM|supp|41588)[''^"]*)"'"',data,re.I):
 print(m.group(1))
# data availability snippet
text=re.sub('<script.*?</script>','',data,flags=re.S); text=re.sub('<style.*?</style>','',text,flags=re.S); text=re.sub('<["'^>]+>'"',' ',text); text=html.unescape(text); text=re.sub(r'\\s+',' ',text)
for pat in ['Cistrome','Data availability','Code availability','track','21,907']:
 m=re.search(pat,text,re.I)
 if m: print('\\n',pat,text[max(0,m.start()-300):m.start()+800])
PY"
