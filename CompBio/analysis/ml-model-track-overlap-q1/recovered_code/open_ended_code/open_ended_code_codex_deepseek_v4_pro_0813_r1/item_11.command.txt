/bin/bash -lc "python3 - <<'PY'
import gzip, collections
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 print('\\n###',path)
 if path.endswith('.gz'):
  f=gzip.open(path,'rt')
 else: f=open(path)
 header=f.readline().rstrip('\\n').split('\\t')
 print('header',header)
 descs=collections.Counter()
 prefixes=collections.Counter()
 idpref=collections.Counter()
 n=0
 for line in f:
  n+=1
  parts=line.rstrip('\\n').split('\\t')
  if len(parts)>=8:
   desc=parts[7] if len(parts)>7 else ''
   ident=parts[0]
   p=desc.split(':')[0] if ':' in desc else 'NOCOLON'
   descs[p]+=1
   # identifier patterns
   if ident.startswith('ENC'): idpref['ENC']+=1
   elif ident.startswith('CNhs'): idpref['CNhs']+=1
   elif ident.startswith('EN'): idpref['EN*']+=1
   elif ident.startswith('GSM'): idpref['GSM']+=1
   elif ident.startswith('SRR'): idpref['SRR']+=1
   else: idpref['other']+=1
 f.close()
 print('n tracks',n)
 print('description prefixes',descs.most_common(30))
 print('id prefixes',idpref.most_common(20))
PY"
