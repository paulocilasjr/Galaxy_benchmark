/bin/bash -lc "python3 - <<'PY'
import json,collections,re
rows=[json.loads(l) for l in open('data/sei_cistrome_samples.jsonl')]
for r in rows:
 if r.get('species')=='' and r.get('external_id_type')=='GEO':
  print(r)
  break
# count external prefixes
c=collections.Counter()
for r in rows:
 ext=r.get('external_id','')
 p=re.match(r'[A-Z]+',ext).group(0) if re.match(r'[A-Z]+',ext) else ext[:20]
 c[(r.get('external_id_type'),p)]+=1
print(c.most_common(30))
# print ENCODE ext examples
for r in rows:
 if r.get('external_id_type')=='ENCODE':
  print(r['external_id']); 
  break
PY"
