/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, re, html
queries=['Cistrome DB API Cistrome ID download metadata','cistrome.org api sample id','Cistrome ID mapping ENCODE GEO GSM','Sei Cistrome ID download']
for q in queries:
 print('\\n###',q)
 url='https://www.google.com/search?q='+urllib.parse.quote(q)
 req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
 try:
  data=urllib.request.urlopen(req,timeout=20).read().decode('utf-8','ignore')
  # crude links
  for m in re.finditer(r'href=\"(http["'^"]+)"'"',data):
   href=html.unescape(m.group(1))
   if 'google' not in href and 'gstatic' not in href:
    print(href[:300])
 except Exception as e: print('ERR',e)
PY"
