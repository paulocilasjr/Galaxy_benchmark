/bin/bash -lc "python3 - <<'PY'
import json,collections
files=[json.loads(l) for l in open('data/borzoi_encff_metadata_full.jsonl')]
exps={g['accession']:g for g in [json.loads(l) for l in open('data/borzoi_experiments_reps_metadata.jsonl')]}
unmatched_total=0; files_unmatched=0; total_tech=0; missing_rep=0; examples=[]
for r in files:
 acc=r.get('dataset','').rstrip('/').split('/')[-1]; ex=exps.get(acc)
 tech=set(r.get('technical_replicates') or [])
 total_tech+=len(tech)
 if not ex:
  missing_rep+=len(tech); continue
 keys={f\"{rep.get('biological_replicate_number')}_{rep.get('technical_replicate_number')}\" for rep in ex.get('replicates',[])}
 unmatched=tech-keys
 if unmatched:
  files_unmatched+=1; unmatched_total+=len(unmatched); examples.append((r['accession'],acc,tech,keys))
print('files',len(files),'with unmatched tech',files_unmatched,'unmatched tech count',unmatched_total,'total tech',total_tech,'missing exp',missing_rep)
print('examples',examples[:20])
PY"
