for acc in ['ENCSR000DSP','ENCSR000DHK','ENCSR000EDS']:

 for r in exp.get('replicates',[])[:10]:

  print(r['biological_replicate_number'],r['technical_replicate_number'],'dbxrefs',lib.get('dbxrefs'),'notes',lib.get('notes'))
 # cistrome ext enscr
 for n in [1,2]:
  data=json.load(urllib.request.urlopen(f'https://db3.cistrome.org/cistrome/samples?external_ids={acc}_{n}',timeout=30))
  if data['samples']: print(' cistrome ENCODE',acc,n,data['samples'])
PY"
