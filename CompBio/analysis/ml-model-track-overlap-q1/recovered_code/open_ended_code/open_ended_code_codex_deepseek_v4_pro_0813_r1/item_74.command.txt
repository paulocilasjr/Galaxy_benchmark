import json,collections
exps=[json.loads(l) for l in open('data/borzoi_experiments_metadata.jsonl')]

examples={}

 n=sum(1 for d in e.get('dbxrefs',[]) if d.startswith('GEO:GSM'))
 cnt[n]+=1
 if n>1: examples.setdefault(n,[]).append((e['accession'],e.get('dbxrefs')))
print(cnt)
for n,vals in examples.items():
 print(n,vals[:3])
PY"
