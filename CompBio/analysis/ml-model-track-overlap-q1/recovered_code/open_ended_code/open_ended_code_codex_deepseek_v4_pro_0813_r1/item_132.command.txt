/bin/bash -lc "python3 - <<'PY'
# print D matched Sei line examples and counts by assay prefix from target.names
import json,re,gzip,collections
# Build precise sets as D
direct=set()
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  ident=re.sub(r'[+-]"'$'"','',line.rstrip('\\n').split('\\t')[1])
  if ident.startswith('GSM'): direct.add(ident)
files=[json.loads(l) for l in open('data/borzoi_encff_metadata_full.jsonl')]
exps={g['accession']:g for g in [json.loads(l) for l in open('data/borzoi_experiments_reps_metadata.jsonl')]}
geo=set(direct); encode=set()
for r in files:
 acc=r.get('dataset','').rstrip('/').split('/')[-1]; ex=exps.get(acc); tech=set(r.get('technical_replicates') or [])
 if not ex: continue
 for b in r.get('biological_replicates') or []: encode.add(f'{acc}_{b}')
 if tech:
  for rep in ex.get('replicates',[]):
   if f\"{rep.get('biological_replicate_number')}_{rep.get('technical_replicate_number')}\" in tech:
    for d in (rep.get('library') or {}).get('dbxrefs',[]):
     if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
 else:
  for d in ex.get('dbxrefs',[]):
   if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
sei=[json.loads(l) for l in open('data/sei_cistrome_samples.jsonl')]
seilines={}
for line in open('[REDACTED LOCAL PATH]'):
 m=re.search(r'ID:(\\d+)"'$'"',line.rstrip())
 if m: seilines[int(m.group(1))]=line.rstrip()
matched=[]
for r in sei:
 ext=r.get('external_id'); typ=r.get('external_id_type')
 if (typ=='GEO' and ext in geo) or (typ=='ENCODE' and ext in encode): matched.append(r['id'])
print('matched',len(matched))
print('first 20:')
for sid in matched[:20]: print(sid,seilines[sid])
# distribution of assay
c=collections.Counter()
for sid in matched:
 line=seilines[sid]
 parts=line.split(' | ')
 c[parts[1]]+=1
print('assay top',c.most_common(20))
PY"
