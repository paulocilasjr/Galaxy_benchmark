/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,time
url='https://db3.cistrome.org/cistrome/samples?ids=1,2,3,4,5'
for i in range(3):
 t=time.time()
 try:
  data=json.load(urllib.request.urlopen(url,timeout=30))
  print(i,time.time()-t, len(data['samples']))
  print(data)
  break
 except Exception as e:
  print('ERR',i,e,time.time()-t)
PY"
