import subprocess, struct, time, sys, os

url='https://zenodo.org/api/records/19443186/files/compbiobench_v1_data.tar/content'

def curl_range(off,n):
    cmd=['curl','-s','-L','-r',f'{off}-{off+n-1}','-o','[REDACTED LOCAL PATH]',url]
    for attempt in range(5):
        subprocess.run(cmd,check=False,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,timeout=120)
        data=open('[REDACTED LOCAL PATH]','rb').read()
        if len(data)==n:
            return data
        # sometimes 416? print and retry
        print('short',off,n,len(data),flush=True)
        time.sleep(2*(attempt+1))
    raise RuntimeError(f'failed {off}')

def parse_header(b):
    if all(x==0 for x in b): return {'zero':True}
    name=b[0:100].split(b'\\0',1)[0].decode('utf-8','replace')
    size_field=b[124:136].split(b'\\0',1)[0].strip() or b'0'
    try: size=int(size_field,8)
    except: size=0
    typeflag=chr(b[156]) if b[156] else '0'
    prefix=b[345:500].split(b'\\0',1)[0].decode('utf-8','replace')
    return {'name':name,'size':size,'type':typeflag,'prefix':prefix,'zero':False}

offset=0
entries=[]
longname=''
for idx in range(10000):
    b=curl_range(offset,512)
    h=parse_header(b)
    if h.get('zero'):
        print('zero block at',offset); break
    name=(h['prefix']+'/'+h['name'] if h['prefix'] else h['name']).lstrip('/')
    if name.startswith('./'): name=name[2:]
    if h['type']=='L':
        data=curl_range(offset+512,h['size']) if h['size'] else b''
        longname=data.split(b'\\0',1)[0].decode('utf-8','replace')
        entries.append(('L',longname,h['size']))
        offset += 512 + ((h['size']+511)//512)*512
        continue
    if longname:
        name=longname; longname=''
    entries.append((h['type'],name,h['size']))
    if idx%10==0: print('entry',idx,'offset',offset,'type',h['type'],'size',h['size'],'name',name[:120],flush=True)
    offset += 512 + ((h['size']+511)//512)*512
print('total entries',len(entries),'final offset',offset)
with open('data/zenodo_tar_entries.txt','w') as f:
    for typ,name,size in entries:
        f.write(f'{typ}\\t{size}\\t{name}\\n')
# print relevant names
for typ,name,size in entries:
    low=name.lower()
    if any(s in low for s in ['answer','ground','truth','expected','ml-model','borzoi','sei','cistrome']) or (size<1000000 and not any(name.lower().endswith(ext) for ext in ['.h5ad','.bam','.bigwig','.fastq','.fq.gz','.mtx.gz','.tar.gz','.fna'])):
        print('RELEVANT',typ,size,name)
PY
