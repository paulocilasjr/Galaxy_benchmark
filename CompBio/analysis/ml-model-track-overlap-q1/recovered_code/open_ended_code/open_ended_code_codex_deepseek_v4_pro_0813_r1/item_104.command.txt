/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for q in ['borzoi','borzoi Linder']:
 url='https://zenodo.org/api/records?q='+urllib.parse.quote(q)+'&size=20'
 try:
  data=json.load(urllib.request.urlopen(url,timeout=30))
  print(q,len(data.get('hits',{}).get('hits',[])))
  for h in data.get('hits',{}).get('hits',[]):
   print(h.get('id'),h.get('metadata',{}).get('title'),h.get('links',{}).get('self'))
 except Exception as e: print('ERR',e)
PY"
