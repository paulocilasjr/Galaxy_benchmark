/bin/bash -lc "python3 - <<'PY'
import gzip, urllib.request, json, urllib.parse, re
ids=set()
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  ident=line.rstrip('\\n').split('\\t')[1]
  ident=re.sub(r'[+-]"'$'"','',ident)
  if ident.startswith('ENCFF'): ids.add(ident)
ids=sorted(ids)
print('unique',len(ids))
batch=ids[:100]
params=[('type','File'),('format','json'),('field','accession'),('field','dataset'),('field','biological_replicates'),('limit','all')] + [('accession',a) for a in batch]
url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode(params)
data=json.load(urllib.request.urlopen(url,timeout=60))
print('total',data['total'],'got',len(data['@graph']))
print('missing',set(batch)-{g['accession'] for g in data['@graph']})
PY"
