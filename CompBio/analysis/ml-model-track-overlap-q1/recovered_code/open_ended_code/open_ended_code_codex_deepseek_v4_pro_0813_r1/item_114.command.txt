sei=[json.loads(l) for l in open('data/sei_cistrome_samples.jsonl')]
files=[json.loads(l) for l in open('data/borzoi_encff_metadata_full.jsonl')]

# direct gsm set
direct=set()
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  ident=re.sub(r'[+-]"'$'"','',line.rstrip('\\n').split('\\t')[1])
  if ident.startswith('GSM'): direct.add(ident)

# E: for each Borzoi track's experiment, include all GEO dbxrefs and all ENCSR_rep (all bio reps in exp)

for r in files:

 if acc: exp_accs.add(acc)

 ex=exps.get(acc)
 if not ex: continue

 reps=ex.get('replicates',[])
 bios=set(rep.get('biological_replicate_number') for rep in reps)
 for b in bios:
  encodeE.add(f'{acc}_{b}')
countE=0
for r in sei:
 ext=r.get('external_id'); typ=r.get('external_id_type')
 if (typ=='GEO' and ext in geoE) or (typ=='ENCODE' and ext in encodeE): countE+=1
print('E count',countE,'round',round(countE/100)*100,'geo int',sum(1 for r in sei if r['external_id_type']=='GEO' and r['external_id'] in geoE),'enc int',sum(1 for r in sei if r['external_id_type']=='ENCODE' and r['external_id'] in encodeE))

# G: direct + all geo dbxrefs but ENCODE only if Borzoi file's bio reps, same C geo and encode

 acc=r.get('dataset','').rstrip('/').split('/')[-1]; ex=exps.get(acc); tech=set(r.get('technical_replicates') or [])
 if not ex: continue
 if tech:
  for rep in ex.get('replicates',[]):
   key=f\"{rep.get('biological_replicate_number')}_{rep.get('technical_replicate_number')}\"

    for d in (rep.get('library') or {}).get('dbxrefs',[]):
     if d.startswith('GEO:GSM'): geoH.add(d.split(':',1)[1])
 else:
  for d in ex.get('dbxrefs',[]):
   if d.startswith('GEO:GSM'): geoH.add(d.split(':',1)[1])
