/bin/bash -lc "python3 - <<'PY'
import re,collections
p='[REDACTED LOCAL PATH]'
n=0; noid=0; ids=set(); nonint=[]
for line in open(p):
 line=line.rstrip('\\n')
 if not line: continue
 n+=1
 m=re.search(r'ID:(\\S+)"'$'"', line)
 if m:
  val=m.group(1)
  if val.isdigit(): ids.add(int(val))
  else: nonint.append(line)
 else:
  noid+=1; nonint.append(line)
print('n',n,'with id',n-noid,'unique ids',len(ids),'no id/nonint examples',nonint[:10])
print('min max',min(ids),max(ids))
PY"
