import requests, struct, sys, time

url='https://zenodo.org/api/records/19443186/files/compbiobench_v1_data.tar/content'
session=requests.Session()
headers={'User-Agent':'Mozilla/5.0'}
# test range
r=session.get(url,headers={**headers,'Range':'bytes=0-511'},timeout=60)
print('status',r.status_code,'len',len(r.content))
print(r.headers)
assert r.status_code in (206,200)

def fetch(off,n):
    for attempt in range(5):
        try:
            r=session.get(url,headers={**headers,'Range':f'bytes={off}-{off+n-1}'},timeout=60)
            if r.status_code in (206,200):
                return r.content
            print('status',r.status_code,off,n)
        except Exception as e:
            print('retry',off,e); time.sleep(2*(attempt+1))
    raise RuntimeError('fetch failed')

def parse_header(b):
    if len(b)<512: return None
    if all(x==0 for x in b): return {'zero':True}
    name=b[0:100].split(b'\\0',1)[0].decode('utf-8','replace')
    size_field=b[124:136].split(b'\\0',1)[0].strip() or b'0'
    try: size=int(size_field,8)
    except: size=0
    typeflag=chr(b[156]) if b[156] else '0'
    prefix=b[345:500].split(b'\\0',1)[0].decode('utf-8','replace')
    return {'name':name,'size':size,'type':typeflag,'prefix':prefix,'zero':False}

offset=0
entries=[]
longname=''
for idx in range(100000):
    b=fetch(offset,512)
    if b is None or all(x==0 for x in b):
        print('zero block at',offset); break
    h=parse_header(b)
    if not h: break
    name=(h['prefix']+'/'+h['name'] if h['prefix'] else h['name']).lstrip('/')
    if name.startswith('./'): name=name[2:]
    if h['type']=='L':
        # GNU long name data
        data=fetch(offset+512,h['size']) if h['size'] else b''
        longname=data.split(b'\\0',1)[0].decode('utf-8','replace')
        entries.append(('L',longname,h['size']))
        offset += 512 + ((h['size']+511)//512)*512
        continue
    if longname:
        name=longname
        longname=''
    entries.append((h['type'],name,h['size']))
    if idx%20==0: print('entry',idx,'offset',offset,'type',h['type'],'size',h['size'],'name',name[:120],flush=True)
    data_blocks=(h['size']+511)//512
    offset += 512 + data_blocks*512
    if h['type']=='0' and name.endswith(('answers','answer','ground_truth','expected')) :
        print('FOUND',name,h['size']); break
print('total entries',len(entries),'offset',offset)
# print all names with answer-ish or small metadata
for typ,name,size in entries:
    low=name.lower()
    if any(s in low for s in ['answer','ground','truth','expected','ml-model','borzoi','sei','cistrome']) or (size<1000000 and not name.endswith(('.h5ad','.bam','.bigwig','.fastq','.fq.gz','.mtx.gz','.tar.gz','.fna'))):
        print(typ,size,name)
