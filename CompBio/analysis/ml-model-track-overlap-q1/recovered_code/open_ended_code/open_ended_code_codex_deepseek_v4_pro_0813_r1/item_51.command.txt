/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,re,time
ids=[]
for line in open('[REDACTED LOCAL PATH]'):
 m=re.search(r'ID:(\\d+)"'$'"',line.rstrip())
 if m: ids.append(int(m.group(1)))
# query 200 in one URL
batch=ids[:200]
url='https://db3.cistrome.org/cistrome/samples?ids='+','.join(map(str,batch))
t=time.time()
try:
 data=json.load(urllib.request.urlopen(url,timeout=120))
 print('time',time.time()-t,'returned',len(data['samples']))
 print(data['samples'][0], data['samples'][-1])
except Exception as e: print('ERR',e,time.time()-t)
PY"
