/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# get one borzoi ENCFF
import gzip
f=gzip.open('[REDACTED LOCAL PATH]','rt'); f.readline()
encff=None
for line in f:
 parts=line.rstrip('\\n').split('\\t')
 if parts[1].startswith('ENCFF'):
  encff=parts[1]; break
print('encff',encff)
url=f'https://www.encodeproject.org/files/{encff}/?format=json'
print(url)
try:
 data=json.load(urllib.request.urlopen(url,timeout=30))
 print(data.keys())
 for k in ['accession','dataset','experiment','file_format','output_type','biological_replicates','replicate','derived_from','href']:
  print(k,data.get(k))
except Exception as e:
 print('ERR',e)
 try: print(e.read().decode()[:1000])
 except: pass
PY"
