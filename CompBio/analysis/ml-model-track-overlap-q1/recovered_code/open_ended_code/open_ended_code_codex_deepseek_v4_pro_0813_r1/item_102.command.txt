/bin/bash -lc "python3 - <<'PY'
import json,re,gzip,collections
# load file full and experiment reps
files=[json.loads(l) for l in open('data/borzoi_encff_metadata_full.jsonl')]
exps={g['accession']:g for g in [json.loads(l) for l in open('data/borzoi_experiments_reps_metadata.jsonl')]}
# direct Borzoi GSM
direct=set()
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  ident=re.sub(r'[+-]"'$'"','',line.rstrip('\\n').split('\\t')[1])
  if ident.startswith('GSM'): direct.add(ident)
# build precise sets
geo=set(direct); encode=set()
no_tech=0; no_bio=0; unmatched_tech=0; missing_exp=0; fallback=0
for r in files:
 acc=r.get('dataset','').rstrip('/').split('/')[-1]
 ex=exps.get(acc)
 if not ex:
  missing_exp+=1; continue
 reps=ex.get('replicates',[])
 bio=r.get('biological_replicates') or []
 tech=set(r.get('technical_replicates') or [])
 # encode IDs by biological replicate
 for b in bio:
  encode.add(f'{acc}_{b}')
 if tech:
  for rep in reps:
   br=rep.get('biological_replicate_number'); tr=rep.get('technical_replicate_number')
   key=f'{br}_{tr}'
   if key in tech:
    for d in (rep.get('library') or {}).get('dbxrefs',[]):
     if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
  # check if any tech had no matching reps (maybe because reps missing tech number)
 else:
  fallback+=1
  # fallback: use experiment-level dbxrefs (or all rep library dbxrefs)
  for d in ex.get('dbxrefs',[]):
   if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
  for rep in reps:
   for d in (rep.get('library') or {}).get('dbxrefs',[]):
    if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
print('files',len(files),'missing exp',missing_exp,'fallback no tech',fallback)
print('direct',len(direct),'geo precise',len(geo),'encode precise',len(encode))
# compare to old broad
geo_old=set(direct)
for r in files:
 acc=r.get('dataset','').rstrip('/').split('/')[-1]; ex=exps.get(acc)
 if ex:
  for d in ex.get('dbxrefs',[]):
   if d.startswith('GEO:GSM'): geo_old.add(d.split(':',1)[1])
print('geo old',len(geo_old),'removed',len(geo_old-geo),'added',len(geo-geo_old))
# Sei count
sei=[json.loads(l) for l in open('data/sei_cistrome_samples.jsonl')]
count=0; by=collections.Counter(); matched=[]
for r in sei:
 ext=r.get('external_id'); typ=r.get('external_id_type')
 if typ=='GEO' and ext in geo:
  count+=1; by['GEO']+=1; matched.append(r['id'])
 elif typ=='ENCODE' and ext in encode:
  count+=1; by['ENCODE']+=1; matched.append(r['id'])
print('count precise',count,by,'round',round(count/100)*100)
# intersection counts
print('geo intersect',len({r['external_id'] for r in sei if r['external_id_type']=='GEO'}&geo))
print('encode intersect',len({r['external_id'] for r in sei if r['external_id_type']=='ENCODE'}&encode))
# matched IDs difference vs old
matched_prec=set(matched)
# recompute old matched
old_count=0; old_matched=[]
for r in sei:
 ext=r.get('external_id'); typ=r.get('external_id_type')
 if typ=='GEO' and ext in geo_old: old_matched.append(r['id'])
 elif typ=='ENCODE' and ext in encode: old_matched.append(r['id'])
old_matched=set(old_matched)
print('old matched not precise',len(old_matched-matched_prec),'precise not old',len(matched_prec-old_matched))
print('old not precise sample',list(old_matched-matched_prec)[:20])
PY"
