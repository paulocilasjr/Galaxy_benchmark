/bin/bash -lc "cd [REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-281652bfe651d812e2dd4d4b/workspace && cat > fetch_borzoi_experiments.py <<'PY'
import json, urllib.request, urllib.parse, time, os, sys

workspace='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-281652bfe651d812e2dd4d4b/workspace'
sets=json.load(open(os.path.join(workspace,'data','borzoi_encff_datasets.json')))
print('datasets',len(sets))
out_path=os.path.join(workspace,'data','borzoi_experiments_metadata.jsonl')
if os.path.exists(out_path): os.remove(out_path)
batch_size=100
rows=[]
for start in range(0,len(sets),batch_size):
    batch=sets[start:start+batch_size]
    params=[('type','Experiment'),('format','json'),('field','accession'),('field','dbxrefs'),('limit','all')]
    params += [('accession',a) for a in batch]
    url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode(params)
    data=None
    for attempt in range(5):
        try:
            data=json.load(urllib.request.urlopen(url,timeout=120))
            break
        except Exception as e:
            print('retry',start,attempt,e,flush=True)
            time.sleep(2*(attempt+1))
    if data is None:
        print('FAILED batch',start,file=sys.stderr); sys.exit(1)
    graph=data.get('@graph',[])
    with open(out_path,'a') as f:
        for g in graph:
            f.write(json.dumps(g)+'\\n')
            rows.append(g)
    got={g.get('accession') for g in graph}
    missing=set(batch)-got
    if missing:
        print('WARN missing',start,missing,flush=True)
    print('batch',start//batch_size+1,'got',len(graph),'total',len(rows),'time',time.strftime('%H:%M:%S'),flush=True)
    time.sleep(0.05)
print('done',len(rows))
PY
python3 fetch_borzoi_experiments.py"
