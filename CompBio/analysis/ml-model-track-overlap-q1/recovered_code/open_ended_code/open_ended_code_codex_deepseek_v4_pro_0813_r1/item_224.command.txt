/bin/bash -lc "python3 - <<'PY'
import json,re,gzip,collections
direct=set()
for path in ['[REDACTED LOCAL PATH]','[REDACTED LOCAL PATH]']:
 f=gzip.open(path,'rt'); f.readline()
 for line in f:
  ident=re.sub(r'[+-]"'$'"','',line.rstrip('\\n').split('\\t')[1])
  if ident.startswith('GSM'): direct.add(ident)
files=[json.loads(l) for l in open('data/borzoi_encff_metadata_full.jsonl')]
exps={g['accession']:g for g in [json.loads(l) for l in open('data/borzoi_experiments_reps_metadata.jsonl')]}
geo=set(direct); encode=set(); fallback_files=0
for r in files:
 acc=r.get('dataset','').rstrip('/').split('/')[-1]; ex=exps.get(acc)
 if not ex: continue
 for b in r.get('biological_replicates') or []: encode.add(f'{acc}_{b}')
 tech=set(r.get('technical_replicates') or [])
 if not tech:
  fallback_files+=1
  for d in ex.get('dbxrefs',[]):
   if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
  continue
 any_missing=False
 for rep in ex.get('replicates',[]):
  key=f\"{rep.get('biological_replicate_number')}_{rep.get('technical_replicate_number')}\"
  if key in tech:
   lib=rep.get('library') or {}
   db=lib.get('dbxrefs',[])
   geos=[d for d in db if d.startswith('GEO:GSM')]
   if geos:
    geo.update(d.split(':',1)[1] for d in geos)
   else:
    any_missing=True
 if any_missing:
  fallback_files+=1
  for d in ex.get('dbxrefs',[]):
   if d.startswith('GEO:GSM'): geo.add(d.split(':',1)[1])
print('fallback files',fallback_files,'geo',len(geo),'encode',len(encode))
sei=[json.loads(l) for l in open('data/sei_cistrome_samples.jsonl')]
count=0; by=collections.Counter()
for r in sei:
 ext=r.get('external_id'); typ=r.get('external_id_type')
 if typ=='GEO' and ext in geo:
  count+=1; by['GEO']+=1
 elif typ=='ENCODE' and ext in encode:
  count+=1; by['ENCODE']+=1
print('hybrid count',count,by,'round',round(count/100)*100)
print('geo intersect',sum(1 for r in sei if r['external_id_type']=='GEO' and r['external_id'] in geo))
print('encode intersect',sum(1 for r in sei if r['external_id_type']=='ENCODE' and r['external_id'] in encode))
PY"
