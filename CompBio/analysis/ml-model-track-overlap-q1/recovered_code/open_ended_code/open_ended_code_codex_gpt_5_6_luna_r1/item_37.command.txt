/bin/bash -lc "python - <<'PY'
import urllib.request,urllib.parse,json,re
lines=open('borzoi/examples/targets_human.txt').read().splitlines()[1:]
ids=sorted({x.split('\\t')[1].rstrip('+-') for x in lines if x.split('\\t')[1].startswith('GSM')})
sei={int(m.group(1)) for line in open('sei-framework/model/target.names') if (m:=re.search(r'\\| ID:(\\d+)"'$'"',line))}
for lim in [100,500,1000]:
 u='https://db3.cistrome.org/cistrome/samples?external_ids='+urllib.parse.quote(','.join(ids),safe=',')+'&limit='+str(lim)
 with urllib.request.urlopen(u,timeout=60) as f: data=json.load(f)
 xs=data['samples']; print('limit',lim,'returned',len(xs),'max?',[x['id'] for x in xs[-3:]])
 hits=[x for x in xs if x['id'] in sei]
 print('hits',len(hits),[(x['id'],x['external_id']) for x in hits])
PY"
