/bin/bash -lc "python - <<'PY'
import urllib.request,json,urllib.parse,re
lines=open('borzoi/examples/targets_human.txt').read().splitlines()[1:]
ids=sorted({x.split('\\t')[1].rstrip('+-') for x in lines if x.split('\\t')[1].startswith('GSM')})
u='https://db3.cistrome.org/cistrome/samples?external_ids='+','.join(ids)+'&limit=1000'
xs=json.load(urllib.request.urlopen(u))['samples']
from collections import Counter
print(len(ids),len(xs),len({x['external_id'] for x in xs}),len({x['id'] for x in xs}))
print('duplicate external',[(k,v) for k,v in Counter(x['external_id'] for x in xs).items() if v>1])
# Exact final calculation no API network needed now
sei=set()
for line in open('sei-framework/model/target.names'):
 m=re.search(r'\\| ID:(\\d+)"'$'"',line)
 if m: sei.add(int(m.group(1)))
hit=sorted(x['id'] for x in xs if x['id'] in sei)
print('Cistrome-linked Sei total',len(sei),'matching raw',len(hit),'rounded',round(len(hit)/100)*100)
PY"
