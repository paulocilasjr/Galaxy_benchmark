/bin/bash -lc "python - <<'PY'
import urllib.request,urllib.parse,json,re,time
lines=open('borzoi/examples/targets_human.txt').read().splitlines()[1:]
ids=sorted({x.split('\\t')[1].rstrip('+-') for x in lines if x.split('\\t')[1].startswith('ENCFF')})
print('query',len(ids))
# URL may exceed server; do chunks of 200 and sequential with delay, collect only IDs
allx=[]
for i in range(0,len(ids),200):
 q=','.join(ids[i:i+200]); u='https://db3.cistrome.org/cistrome/samples?external_ids='+q+'&limit=1000'
 for a in range(10):
  try:
   with urllib.request.urlopen(u,timeout=60) as f: x=json.load(f).get('samples',[])
   allx.extend(x); break
  except Exception as e:
   if '429' in str(e): time.sleep(2+a)
   elif a==9: print('err',i,e)
 if (i//200+1)%5==0: print('done',i+len(q.split(',')),'returned',len(allx),flush=True)
print('total returned',len(allx))
print(allx[:5])
sei={int(m.group(1)) for line in open('sei-framework/model/target.names') if (m:=re.search(r'\\| ID:(\\d+)"'$'"',line))}
print('in sei',sum(x['id'] in sei for x in allx))
PY"
