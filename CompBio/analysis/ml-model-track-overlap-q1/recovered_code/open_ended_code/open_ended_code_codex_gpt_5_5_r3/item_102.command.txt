/bin/bash -lc "python3 - <<'PY'
import json,collections
m=json.load(open('provenance_overlap_matches.json'))
p=collections.Counter()
for x in m:
 for t in x['overlap']:
  if t.startswith('GSM'): p['GSM']+=1
  elif t.startswith('GSE'): p['GSE']+=1
  elif t.startswith('ENCSR'): p['ENCSR']+=1
  elif t.startswith('ENCFF'): p['ENCFF']+=1
  else: p[t[:4]]+=1
print('matches', len(m))
print(p)
for x in m[:20]: print(x)
PY"
