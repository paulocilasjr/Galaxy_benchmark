/bin/bash -lc "python3 - <<'PY'
import re
ids=[]
for l in open('sei_target.names'):
    m=re.search(r'ID:(\\d+)',l)
    if m: ids.append(int(m.group(1)))
print(len(ids), len(set(ids)), min(ids), max(ids))
from collections import Counter
c=Counter(ids)
print('dups', sum(v>1 for v in c.values()), 'maxdup', max(c.values()))
print('first dup', [x for x,v in c.items() if v>1][:10])
PY"
