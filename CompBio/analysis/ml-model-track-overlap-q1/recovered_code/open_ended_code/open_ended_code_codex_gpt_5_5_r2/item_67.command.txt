/bin/bash -lc "python3 - <<'PY'
import json, urllib.request
for acc in ['ENCSR000AKB','ENCFF279CYY','ENCSR000DZN','ENCFF578TBN']:
    url=f'https://www.encodeproject.org/{acc}/?format=json'
    try:
        req=urllib.request.Request(url, headers={'Accept':'application/json'})
        data=json.load(urllib.request.urlopen(req, timeout=30))
        print(acc, data.get('accession'), data.get('dbxrefs'), data.get('aliases')[:5] if data.get('aliases') else None, data.get('description'), data.get('date_released'))
        if 'replicates' in data:
            for r in data['replicates'][:2]:
                lib=r.get('library',{})
                biosample=lib.get('biosample',{}) if isinstance(lib,dict) else {}
                print(' rep', r.get('aliases'), lib.get('accession') if isinstance(lib,dict) else None, biosample.get('accession') if isinstance(biosample,dict) else None, biosample.get('dbxrefs') if isinstance(biosample,dict) else None)
        print()
    except Exception as e: print(acc, e)
PY"
