import json,collections,re,pandas as pd
s=json.load(open('sei_cistrome_metadata.json'))['samples']

print('external types',collections.Counter(x['external_id_type'] for x in s))

print('missing external',sum(not x['external_id'] for x in s))
print('nonGSM examples')
for x in [x for x in s if not x['external_id'].startswith('GSM')][:15]: print(x)

 q=b[b.file.str.contains('/'+src+'/',regex=False)]
 print(src,len(q),'id prefixes',collections.Counter(re.match('[A-Za-z]+',str(v)).group() if re.match('[A-Za-z]+',str(v)) else '' for v in q.identifier).most_common(8))
PY"
