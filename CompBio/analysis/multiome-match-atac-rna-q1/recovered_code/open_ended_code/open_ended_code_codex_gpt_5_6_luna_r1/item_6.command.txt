/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
from scipy.stats import rankdata

def load(fn):
    with gzip.open(fn,'rt') as f:
        f.readline()
        a=np.loadtxt(f,delimiter='\\t',usecols=range(1,9))
    return a
R=load('multiome.match.atac.rna.q1.rna.tsv.gz')
A=load('multiome.match.atac.rna.q1.atac.tsv.gz')

def corr(x): return np.corrcoef(x.T)
def cosine(x):
    return (x.T@x)/np.sqrt((x*x).sum(0)[:,None]*(x*x).sum(0)[None,:])
def show(name,x):
    print(name)
    print(np.array2string(x,precision=4,suppress_small=True))
for label, f in [('raw',lambda x:x),('log',lambda x:np.log1p(x)),('sqrt',lambda x:np.sqrt(x)),('rank',lambda x:np.apply_along_axis(rankdata,0,x))]:
    print('\\n###',label)
    show('RNA corr',corr(f(R)))
    show('ATAC corr',corr(f(A)))
    show('RNA cos',cosine(f(R)))
    show('ATAC cos',cosine(f(A)))
PY"
