/bin/bash -lc "python3 - <<'PY'
import gzip, itertools, numpy as np
from scipy.stats import rankdata, spearmanr

def load(fn):
    with gzip.open(fn,'rt') as f:
        f.readline()
        return np.loadtxt(f,delimiter='\\t',usecols=range(1,9))
R=load('multiome.match.atac.rna.q1.rna.tsv.gz')
A=load('multiome.match.atac.rna.q1.atac.tsv.gz')

def corr(x): return np.corrcoef(x.T)
def cosine(x): return (x.T@x)/np.sqrt((x*x).sum(0)[:,None]*(x*x).sum(0)[None,:])
def rankcols(x): return np.apply_along_axis(rankdata,0,x)

def off(M): return M[np.triu_indices(8,1)]
perms=list(itertools.permutations(range(8)))
methods={}
for label, f in [('rawcorr',lambda x:corr(x)),('logcorr',lambda x:corr(np.log1p(x))),('sqrtcorr',lambda x:corr(np.sqrt(x))),('rankcorr',lambda x:corr(rankcols(x))),('rawcos',lambda x:cosine(x)),('logcos',lambda x:cosine(np.log1p(x))),('sqrtcos',lambda x:cosine(np.sqrt(x))),('rankcos',lambda x:cosine(rankcols(x)))]:
    SR=f(R); SA=f(A)
    vr=off(SR)
    out=[]
    for p in perms:
        va=SA[np.ix_(p,p)]
        x=off(va)
        # corr and mse standardized
        pc=np.corrcoef(vr,x)[0,1]
        mse=np.mean((vr-x)**2)
        # rank corr
        rc=spearmanr(vr,x).statistic
        out.append((pc, -mse, rc, p))
    methods[label]=out
    print('\\n',label)
    for score in [0,1,2]:
        print('score',score)
        for row in sorted(out,key=lambda z:z[score],reverse=True)[:10]: print(f'{row[score]:.8f}',row[3])
PY"
