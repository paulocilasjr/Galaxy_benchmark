/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np

def load(f):
 with gzip.open(f,'rt') as h:
  h.readline(); return np.loadtxt(h,delimiter='\\t',usecols=range(1,9))
R=load('multiome.match.atac.rna.q1.rna.tsv.gz'); A=load('multiome.match.atac.rna.q1.atac.tsv.gz')
for name,x in [('R',R),('A',A)]:
 p=x/x.sum(0)
 stats={
 'sd':x.std(0), 'cv':x.std(0)/x.mean(0),
 'entropy':-(p*np.log(p+1e-300)).sum(0),
 'simpson':(p*p).sum(0),
 'gini':np.array([np.abs(v[:,None]-v[None,:]).sum()/(2*len(v)*v.sum()) for v in x.T]),
 'top10':np.sort(x,axis=0)[-10:].sum(0),
 'top100':np.sort(x,axis=0)[-100:].sum(0),
 'q99':np.quantile(x,.99,axis=0),
 'zero':(x==0).sum(0),
 }
 print(name)
 for k,v in stats.items(): print(k, np.round(v,6))
PY"
