/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np
with gzip.open('multiome.match.atac.rna.q1.atac.tsv.gz','rt') as f:
 f.readline(); names=[]; rows=[]
 for line in f:
  q=line.rstrip().split('\\t'); names.append(q[0]); rows.append([float(v) for v in q[1:]])
a=np.asarray(rows); names=np.asarray(names)
# log fold vs mean other and zscore across columns; add tiny pseudocount to avoid zeros
for mode in ['ratio','diff','z']:
 print('\\nMODE',mode)
 mean=(a.sum(1,keepdims=True)-a)/7
 if mode=='ratio': score=np.log2((a+1e-3)/(mean+1e-3))
 elif mode=='diff': score=a-mean
 else: score=(a-a.mean(1,keepdims=True))/(a.std(1,keepdims=True)+1e-9)
 for j in range(8):
  idx=np.argsort(score[:,j])[::-1][:35]
  print('\\nCOL',j)
  print(' '.join(f'{names[i]}={a[i,j]:.1f}[{score[i,j]:.2f}]' for i in idx))
PY"
