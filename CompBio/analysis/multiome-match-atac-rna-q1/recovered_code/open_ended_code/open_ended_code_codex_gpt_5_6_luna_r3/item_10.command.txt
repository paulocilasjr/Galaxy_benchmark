/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
print('start',flush=True)
def load(f):
  with gzip.open(f,'rt') as h:
    hdr=h.readline()
    a=np.loadtxt(h,delimiter='\\t',dtype=np.float64,usecols=range(1,9))
  return a
r=load('multiome.match.atac.rna.q1.rna.tsv.gz'); print('rna',r.shape,flush=True)
a=load('multiome.match.atac.rna.q1.atac.tsv.gz'); print('atac',a.shape,flush=True)
for name,x in [('RNA',r),('ATAC',a)]:
  print(name,'sums',x.sum(axis=0))
  for trans in ['raw','log']:
    y=np.log1p(x) if trans=='log' else x
    print(name,trans,np.round(np.corrcoef(y.T),3))
PY"
