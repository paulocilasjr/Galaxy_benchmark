/bin/bash -lc "python3 - <<'PY'
import gzip,numpy as np

def load(f):
 with gzip.open(f,'rt') as h:
  h.readline(); return np.loadtxt(h,delimiter='\\t',dtype=np.float64,usecols=range(1,9))
r=load('multiome.match.atac.rna.q1.rna.tsv.gz'); a=load('multiome.match.atac.rna.q1.atac.tsv.gz')
for name,x in [('RNA',r),('ATAC',a)]:
 log=np.log1p(x)
 print('\\n',name)
 print('nonzero', (x>0).sum(0).tolist())
 print('zero_frac', np.round((x==0).mean(0),4).tolist())
 print('mean',x.mean(0).tolist(),'sd',x.std(0).tolist())
 print('log sum',log.sum(0).round(2).tolist(),'log sd',log.std(0).round(4).tolist())
 print('L2',np.sqrt((x*x).sum(0)).round(1).tolist())
 print('top10sum',np.sort(x,axis=0)[-10:,:].sum(0).round(1).tolist())
 print('quantiles',np.round(np.quantile(x,[.5,.9,.95,.99,.999],axis=0),4))
PY"
