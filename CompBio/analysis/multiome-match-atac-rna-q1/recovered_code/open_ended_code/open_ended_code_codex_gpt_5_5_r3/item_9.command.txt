/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np

def summarize(path):
    n=0; sums=np.zeros(8); sums2=np.zeros(8); nz=np.zeros(8,dtype=int)
    vals=[]
    with gzip.open(path,'rt') as f:
        header=next(f).strip().split('\\t')
        for line in f:
            parts=line.rstrip('\\n').split('\\t')
            x=np.fromiter((float(v) for v in parts[1:]), dtype=float, count=8)
            sums+=x; sums2+=x*x; nz+=x>0; n+=1
            if n<=0: vals.append(x)
    mean=sums/n; std=(sums2/n-mean*mean)**0.5
    print(path, n)
    print('sum', np.round(sums,3))
    print('nz', nz)
    print('mean', np.round(mean,5))
    print('std', np.round(std,5))

summarize('multiome.match.atac.rna.q1.rna.tsv.gz')
summarize('multiome.match.atac.rna.q1.atac.tsv.gz')
PY"
