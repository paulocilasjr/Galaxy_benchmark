/bin/bash -lc "python3 - <<'PY'
import pandas as pd
import numpy as np
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t')
atac=pd.read_csv('multiome.match.atac.rna.q1.atac.tsv.gz',sep='\\t')
print(rna.shape, atac.shape)
for name,df in [('rna',rna),('atac',atac)]:
    X=df.iloc[:,1:].to_numpy(float)
    print(name)
    print('sum', np.round(X.sum(axis=0),3))
    print('nonzero', (X>0).sum(axis=0))
    print('mean', X.mean(axis=0))
    print('std', X.std(axis=0))
    print('q99', np.quantile(X,0.99,axis=0))
PY"
