def corr_matrix(A,B,mask):
    X=A[mask].astype(np.float64); Y=B[mask].astype(np.float64)
    C=np.zeros((8,8))
    for i in range(8):
        x=X[:,i]; x=x-x.mean(); xs=np.sqrt((x*x).sum())
        for j in range(8):
            y=Y[:,j]; y=y-y.mean(); ys=np.sqrt((y*y).sum())
            C[i,j]=(x@y)/(xs*ys) if xs and ys else np.nan
    return C

    return best
