/bin/bash -lc "python3 - <<'PY'
import gzip, json, time
f='Homo_sapiens.GRCh38.116.gtf.gz'
pos={}
t0=time.time()
with gzip.open(f,'rt') as fh:
    for line in fh:
        if line.startswith('#'): continue
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<9: continue
        if parts[2]"'!='"'gene': continue
        chrom=parts[0]
        # accept chr? GTF has 1.. no chr; RNA/ATAC use chr. store without? add chr later
        start=int(parts[3]); end=int(parts[4]); strand=parts[6]
        attr=parts[8]
        d={}
        for kv in attr.split(';'):
            kv=kv.strip()
            if not kv: continue
            if ' ' in kv:
                k,v=kv.split(' ',1)
            else:
                k=kv; v=''
            v=v.strip().strip('\"')
            d[k]=v
        gid=d.get('gene_id','')
        gname=d.get('gene_name',gid)
        # store all? gene_name collisions (unique mostly)
        if gname and gname not in pos:
            pos[gname]={'chr':'chr'+chrom if not chrom.startswith('chr') else chrom,'start':start,'end':end,'strand':strand,'gene_id':gid}
        elif gname in pos:
            # merge min/max if multiple genes same name? choose first or merge
            old=pos[gname]
            old['start']=min(old['start'],start); old['end']=max(old['end'],end)
    print('parsed',len(pos),'time',time.time()-t0)
with open('gene_positions_gtf.json','w') as fh:
    json.dump(pos,fh)
# check markers
for g in ['CD19','CD3D','CD14','MS4A1','NKG7','FCGR3A','CD8A','IL7R','POU5F1','SOX2','TAGLN','KRT14']:
 print(g,pos.get(g))
PY"
