/bin/bash -lc "python3 - <<'PY'
import gzip
f='Homo_sapiens.GRCh38.116.gtf.gz'
for target in ['AL627309.1','AL627309.5','AL669831.2','TMEM88B']:
    found=[]
    with gzip.open(f,'rt') as fh:
        for line in fh:
            if line.startswith('#'): continue
            if target in line:
                found.append(line.rstrip()[:300])
                if len(found)>=2: break
    print(target, found)
PY"
