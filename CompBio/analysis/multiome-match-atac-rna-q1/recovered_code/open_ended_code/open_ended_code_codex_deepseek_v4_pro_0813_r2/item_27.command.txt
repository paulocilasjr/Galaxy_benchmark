/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, gzip
from scipy.stats import spearmanr
# load
genes=[]; rna=[]
with gzip.open('multiome.match.atac.rna.q1.rna.tsv.gz','rt') as f:
 h=f.readline().rstrip().split('\\t')[1:]
 for line in f:
  p=line.rstrip().split('\\t'); genes.append(p[0]); rna.append([float(x) for x in p[1:9]])
R=pd.DataFrame(rna,index=genes,columns=h)
act=pd.read_pickle('act_tss.pkl')
common=act.index.intersection(R.index)
X=np.log1p(R.loc[common].values); Y=np.log1p(act.loc[common].values)
keep=(R.loc[common].values.sum(1)>0.1)&(act.loc[common].values.sum(1)>0)
idx=np.argsort(-(np.var(X[keep],axis=1)+np.var(Y[keep],axis=1)))[:2000]
X=X[keep][idx]; Y=Y[keep][idx]
C=np.empty((8,8))
for i in range(8):
 for j in range(8): C[i,j]=spearmanr(X[:,i],Y[:,j])[0]
print('C matrix')
for i in range(8):
 print([f'{C[i,j]:.3f}' for j in range(8)])
print('best',C.argmax(1))
for i in range(8):
 order=np.argsort(-C[i])[:4]
 print('row',i,[(j,round(C[i,j],3),round(C[i,j]-C[i,C[i].argmax()],3)) for j in order])
PY"
