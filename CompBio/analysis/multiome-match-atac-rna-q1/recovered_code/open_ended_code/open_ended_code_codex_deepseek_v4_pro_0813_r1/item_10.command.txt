/bin/bash -lc "python3 - <<'PY'
import gzip, re, pandas as pd
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz',sep='\\t',index_col=0,compression='gzip')
symbols=set(rna.index)
gene_info={}
with gzip.open('gencode.v32.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9 or p[2]"'!='"'gene': continue
        m=re.search(r'gene_name \"(["'^"]+)"'"',p[8])
        if not m: continue
        g=m.group(1)
        if g in symbols and g not in gene_info:
            gene_info[g]=(p[0],int(p[3]),int(p[4]),p[6])
with open('gene_info.tsv','w') as f:
    for g,(c,s,e,strand) in gene_info.items():
        f.write(f'{g}\\t{c}\\t{s}\\t{e}\\t{strand}\\n')
print(len(gene_info), list(symbols-set(gene_info)))
PY"
