/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
rna='multiome.match.atac.rna.q1.rna.tsv.gz'
atac='multiome.match.atac.rna.q1.atac.tsv.gz'
R=pd.read_csv(rna,sep='\\t',index_col=0)
A=pd.read_csv(atac,sep='\\t',index_col=0)
print(R.shape,A.shape)
# transforms
for name,X in [('rna',R),('atac',A)]:
    print(name, X.sum().to_numpy(), X.mean().to_numpy(), (X>0).sum().to_numpy())

def corrmat(X, method='pearson', log=True, topvar=None):
    Y=np.log1p(X.to_numpy(dtype=float)) if log else X.to_numpy(dtype=float)
    if topvar:
        v=Y.var(axis=1)
        idx=np.argsort(v)[-topvar:]
        Y=Y[idx]
    return np.corrcoef(Y.T)
for log in [False, True]:
  for top in [None, 5000, 10000, 20000]:
    Cr=corrmat(R,log=log,topvar=top)
    Ca=corrmat(A,log=log,topvar=top)
    print('\\nlog',log,'top',top)
    print('RNA corr\\n',np.round(Cr,3))
    print('ATAC corr\\n',np.round(Ca,3))
PY"
