import numpy as np
from PIL import Image, ImageDraw
from scipy.ndimage import binary_dilation, distance_transform_edt, gaussian_filter
from pathlib import Path
letters=list('BDFHJLNPRTVXZ')

def template(letter):
    img=Image.new('L',(N,N),0); d=ImageDraw.Draw(img)
    l=margin; r=N-margin; t=margin; b=N-margin; mid=N//2
    w=stroke
    def line(a,b): d.line([a,b],fill=255,width=w,joint='curve')
    def rect(x0,y0,x1,y1): d.rounded_rectangle([x0,y0,x1,y1], radius=2, fill=255)
    if letter=='H': line((l,t),(l,b)); line((r,t),(r,b)); line((l,mid),(r,mid))
    elif letter=='N': line((l,b),(l,t)); line((l,t),(r,b)); line((r,b),(r,t))
    elif letter=='Z': line((l,t),(r,t)); line((r,t),(l,b)); line((l,b),(r,b))
    elif letter=='T': line((l,t),(r,t)); line((N//2,t),(N//2,b))
    elif letter=='L': line((l,t),(l,b)); line((l,b),(r,b))
    elif letter=='F': line((l,t),(l,b)); line((l,t),(r,t)); line((l,mid),(r-8,mid))
    elif letter=='V': line((l,t),(N//2,b)); line((r,t),(N//2,b))
    elif letter=='X': line((l,t),(r,b)); line((r,t),(l,b))
    elif letter=='J': line((l,t),(r,t)); line((r,t),(r,b-18)); d.arc([l,b-42,r,b+8],0,180,fill=255,width=w)
    elif letter in 'BDPR':
        line((l,t),(l,b))
        if letter=='D':
            d.arc([l-w//2,t,r,b],-90,90,fill=255,width=w); line((l,t),(r-14,t)); line((l,b),(r-14,b))
        elif letter=='P':
            d.arc([l-w//2,t,r,mid+8],-90,90,fill=255,width=w); line((l,t),(r-14,t)); line((l,mid+8),(r-14,mid+8))
        elif letter=='R':
            d.arc([l-w//2,t,r,mid+8],-90,90,fill=255,width=w); line((l,t),(r-14,t)); line((l,mid+8),(r-14,mid+8)); line((l+18,mid),(r,b))

            line((l,t),(r-14,t)); line((l,mid),(r-14,mid)); line((l,b),(r-14,b))
    return np.array(img)>0

tmpl={L:template(L) for L in letters}
td={L:distance_transform_edt(~m) for L,m in tmpl.items()}
# also allow mirrored templates
variants=[]
for L,m in tmpl.items():
    for flip in [False, True]:
        mm=np.fliplr(m) if flip else m
        variants.append((L,mm,distance_transform_edt(~mm)))

coords=[]
for line in open('protein.shape.q1.pdb'):
    if line.startswith('ATOM'):
        coords.append([float(line[30:38]),float(line[38:46]),float(line[46:54])])
coords=np.array(coords); coords-=coords.mean(0)
# sub-sample not needed
rng=np.random.default_rng(4)
# include PCA bases plus random
_,_,vh=np.linalg.svd(coords,full_matrices=False); C=vh.T
bases=[]
for pair in [(0,1),(0,2),(1,2)]: bases.append(C[:,pair])
for i in range(2500):
    M=rng.normal(size=(3,3)); Q,_=np.linalg.qr(M); bases.append(Q[:,:2])

def mask_for(B, anis=True):
    uv=coords@B
    mn=uv.min(0); mx=uv.max(0); span=mx-mn
    if anis:
        p=(uv-mn)/(span+1e-9)*(N-2*margin)+margin
    else:

    img[pp[ok,1],pp[ok,0]]=1
    img=binary_dilation(img, iterations=2)
    return img
