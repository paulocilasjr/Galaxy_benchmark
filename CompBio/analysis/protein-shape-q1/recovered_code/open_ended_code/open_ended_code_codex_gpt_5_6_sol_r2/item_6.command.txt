/bin/bash -lc "python - <<'PY'
import matplotlib.pyplot as plt
import numpy as np
from collections import defaultdict
ch=defaultdict(list)
for line in open('protein.shape.q1.pdb'):
 if line.startswith('ATOM') and line[12:16].strip()=='CA':
  ch[line[21]].append([float(line[30:38]),float(line[38:46]),float(line[46:54])])
fig,axs=plt.subplots(2,3,figsize=(18,10))
for j,(a,b,nm) in enumerate([(0,1,'XY'),(0,2,'XZ'),(1,2,'YZ')]):
 for ax,eq in [(axs[0,j],False),(axs[1,j],True)]:
  for c,p in ch.items():
   p=np.array(p); ax.scatter(p[:,a],p[:,b],s=2,label=c)
  ax.set_title(nm+(' equal' if eq else '')); ax.legend();
  if eq: ax.set_aspect('equal')
plt.tight_layout(); plt.savefig('projections.png',dpi=160)
PY"
