/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict
import numpy as np
coords=[]; cas=[]; chains=Counter(); residues=defaultdict(set); atoms=Counter()
with open('protein.shape.q1.pdb') as f:
    for line in f:
        if line.startswith(('ATOM  ','HETATM')):
            name=line[12:16].strip(); chain=line[21]; res=int(line[22:26])
            x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54])
            coords.append((x,y,z)); chains[chain]+=1; residues[chain].add(res); atoms[name]+=1
            if name=='CA': cas.append((x,y,z))
arr=np.array(coords); ca=np.array(cas)
print('atoms',arr.shape,'CA',ca.shape)
print('chains',chains)
print('residues', {c:len(s) for c,s in residues.items()})
print('atom names common', atoms.most_common(10))
for name,a in [('all',arr),('ca',ca)]:
    print(name,'min',a.min(axis=0),'max',a.max(axis=0),'range',a.ptp(axis=0),'center',a.mean(axis=0))
    w,v=np.linalg.eigh(np.cov((a-a.mean(0)).T)); idx=np.argsort(w)[::-1]
    print('eig',w[idx], 'ratios', w[idx]/w[idx].sum())
    print('axes',v[:,idx])
PY"
