/bin/bash -c "cat > plot_pca.py <<'PY'
import numpy as np, matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from matplotlib.collections import LineCollection
coords=np.load('ca_coords.npy'); chains=np.load('ca_chains.npy')
# center
c=coords-coords.mean(0)
# PCA
u,s,vt=np.linalg.svd(c, full_matrices=False)
proj=c@vt.T
print('singular',s,'ratios',s/s[0])
# plot first three PC pairs
fig,axs=plt.subplots(1,3,figsize=(18,6))
pairs=[(0,1),(0,2),(1,2)]
for ax,(a,b) in zip(axs,pairs):
    colors={'A':'r','B':'g','C':'b','D':'c','E':'m','F':'y'}
    for ch in np.unique(chains):
        m=chains==ch
        ax.scatter(proj[m,a],proj[m,b],s=2,label=ch,color=colors.get(ch))
    ax.set_aspect('equal'); ax.set_title(f'PC{a+1} vs PC{b+1}')
plt.legend(); plt.tight_layout(); plt.savefig('pca_scatter.png',dpi=100)
print('saved')
PY
python plot_pca.py"
