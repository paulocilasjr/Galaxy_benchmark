def dice(a,b):
    a=a.astype(bool); b=b.astype(bool)
    inter=np.logical_and(a,b).sum(); denom=a.sum()+b.sum()
    return (2*inter)/denom if denom else 0

def iou(a,b):
    inter=np.logical_and(a,b).sum(); union=np.logical_or(a,b).sum()
    return inter/union if union else 0

def basis_from_n(n):
    n=n/np.linalg.norm(n)
    # choose e1
    if abs(n[0])<0.9:
        e1=np.cross(n,[1,0,0]); e1=e1/np.linalg.norm(e1)
    else:
        e1=np.cross(n,[0,1,0]); e1=e1/np.linalg.norm(e1)
    e2=np.cross(n,e1)
    return np.vstack([e1,e2])

def fib_sphere(n):
