/bin/bash -c "cat > ascii_best_chamfer.py <<'PY'
import numpy as np
from PIL import Image
from scipy.ndimage import distance_transform_edt, binary_dilation
# load saved best images for letters
for ch in ['L','Z','X','T','J','V','N']:
    img=np.array(Image.open(f'chamfer_best_{ch}.png').convert('L'))>127
    # skeletonize and print downsample
    from skimage.morphology import skeletonize
    sk=skeletonize(binary_dilation(img,iterations=1))
    print('\\n===',ch,'sum',img.sum())
    H,W=img.shape; outH=32; outW=80
    for r in range(outH):
        line=''
        y0=int(r*H/outH); y1=max(y0+1,int((r+1)*H/outH))
        for c in range(outW):
            x0=int(c*W/outW); x1=max(x0+1,int((c+1)*W/outW))
            v=sk[y0:y1,x0:x1].sum()
            line += '#' if v>0 else ' '
        print(line)
PY
python ascii_best_chamfer.py"
