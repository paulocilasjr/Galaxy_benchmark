/bin/bash -c "cat > ascii_sticks.py <<'PY'
import numpy as np
coords=np.load('ca_coords.npy'); chains=np.load('ca_chains.npy'); coords-=coords.mean(0)
names=np.unique(chains)
# define end-to-start line for each chain
segs=[]; labels=[]
for ch in names:
 p=coords[chains==ch]; segs.append((p[0],p[-1])); labels.append(ch)
# project axes
pairs=[('XY',(0,1)),('XZ',(0,2)),('YZ',(1,2))]
for name,(a,b) in pairs:
    print('\\n===',name)
    pts=[s for seg in segs for s in seg]
    uv=np.array([(p[a],p[b]) for p in pts])
    umin,umax=uv[:,0].min(),uv[:,0].max(); vmin,vmax=uv[:,1].min(),uv[:,1].max()
    W=110;H=30
    span=max(umax-umin,vmax-vmin); scaleu=(W-1)/span; scalev=(H-1)/span
    grid=[[' ']*W for _ in range(H)]
    chars={ch:c for ch,c in zip(names,'ABCDEF')}
    for idx,(ch,(s,e)) in enumerate(zip(labels,segs)):
        u0=(s[a]-(umin+umax)/2)*scaleu+(W-1)/2; v0=(s[b]-(vmin+vmax)/2)*scalev+(H-1)/2
        u1=(e[a]-(umin+umax)/2)*scaleu+(W-1)/2; v1=(e[b]-(vmin+vmax)/2)*scalev+(H-1)/2
        steps=max(1,int(max(abs(u1-u0),abs(v1-v0))*3)+1)
        for t in range(steps+1):
            u=int(round(u0+(u1-u0)*t/steps)); v=int(round(v0+(v1-v0)*t/steps))
            if 0<=u<W and 0<=v<H:
                grid[v][u]=chars[ch]
        # mark endpoints with lower letters? use same
        for u,v in [(u0,v0),(u1,v1)]:
            u=int(round(u));v=int(round(v))
            if 0<=u<W and 0<=v<H: grid[v][u]='*'
    for row in grid: print(''.join(row))
PY
python ascii_sticks.py"
