import numpy as np
import matplotlib.pyplot as plt
p=[]; ch=[]
for line in open('protein.shape.q1.pdb'):
 if line.startswith('ATOM') and line[12:16].strip()=='CA':
  p.append([float(line[30:38]),float(line[38:46]),float(line[46:54])]);ch.append(line[21])
p=np.array(p); q=p-p.mean(0)
w,v=np.linalg.eigh(np.cov(q.T)); v=v[:,np.argsort(w)[::-1]]; r=q@v
print('eig',w[np.argsort(w)[::-1]],'vecs',v)
fig,axs=plt.subplots(2,3,figsize=(15,9))
pairs=[(0,1),(0,2),(1,2)]
for ax,(i,j) in zip(axs[0],pairs):
 for c in sorted(set(ch)):
  m=np.array(ch)==c; ax.scatter(p[m,i],p[m,j],s=4,label=c)
 ax.axis('equal'); ax.set_title('xyz'[i]+' vs '+'xyz'[j])
for ax,(i,j) in zip(axs[1],pairs):
 for c in sorted(set(ch)):
  m=np.array(ch)==c; ax.scatter(r[m,i],r[m,j],s=4,label=c)
 ax.axis('equal'); ax.set_title('PC'+str(i+1)+' vs PC'+str(j+1))
axs[0,0].legend(); plt.tight_layout(); plt.savefig('projections.png',dpi=180)
PY"
