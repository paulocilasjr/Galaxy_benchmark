def project_image(pts,n):
    n=np.asarray(n,float); n/=np.linalg.norm(n)

def fib_sphere(k):
    i=np.arange(k)
    z=1 - (2*i+1)/k
    phi=np.arccos(np.clip(z,-1,1))
    theta=math.pi*(1+5**0.5)*i
    return np.stack([np.sin(phi)*np.cos(theta), np.sin(phi)*np.sin(theta), z],axis=1)
