/bin/bash -lc "python3 - <<'PY'
import numpy as np
# parse CA chains
ca=[]
for line in open('protein.shape.q1.pdb'):
 if line.startswith('ATOM') and line[12:16].strip()=='CA':
  chain=line[21]; resi=int(line[22:26]); x=float(line[30:38]); y=float(line[38:46]); z=float(line[46:54]); ca.append((chain,resi,np.array([x,y,z])))
chains={}
for ch,ri,p in ca: chains.setdefault(ch,[]).append((ri,p))
for ch in chains: chains[ch].sort()
allpts=np.array([p for _,_,p in ca]); c=allpts.mean(0)

def trace_for(n, W=80,H=30):
 n=np.asarray(n,float); n/=np.linalg.norm(n); a=np.array([0.,0.,1.]) if abs(n[2])<.9 else np.array([1.,0.,0.]); u=np.cross(n,a); u/=np.linalg.norm(u); v=np.cross(n,u); v/=np.linalg.norm(v)
 lines=[]
 for ch,res in chains.items():
  pts=[p for ri,p in res]
  for p,q in zip(pts,pts[1:]):
   lines.append((p@u,p@v,q@u,q@v))
 xs=[]; ys=[]
 for x1,y1,x2,y2 in lines: xs += [x1,x2]; ys += [y1,y2]
 xmin=min(xs); xmax=max(xs); ymin=min(ys); ymax=max(ys)
 sx=(W-1)/(xmax-xmin); sy=(H-1)/(ymax-ymin); scale=min(sx,sy); W2=int((xmax-xmin)*scale)+1; H2=int((ymax-ymin)*scale)+1
 img=np.zeros((H2,W2),bool)
 def draw(x1,y1,x2,y2):
  ix1=int(round((x1-xmin)*scale)); iy1=int(round((y1-ymin)*scale)); ix2=int(round((x2-xmin)*scale)); iy2=int(round((y2-ymin)*scale))
  dx=abs(ix2-ix1); sx=1 if ix1<ix2 else -1; dy=-abs(iy2-iy1); sy=1 if iy1<iy2 else -1; err=dx+dy
  while True:
   if 0<=ix1<W2 and 0<=iy1<H2: img[iy1,ix1]=True
   if ix1==ix2 and iy1==iy2: break
   e=2*err
   if e>=dy: err+=dy; ix1+=sx
   if e<=dx: err+=dx; iy1+=sy
 for x1,y1,x2,y2 in lines: draw(x1,y1,x2,y2)
 return img
for n in ([0.151,-0.919,-0.365],[-0.377,0.872,0.312],[0.891,-0.42,-0.172],[-0.845,0.535,-0.002]):
 print('\\nCA trace n',n)
 img=trace_for(n,W=70,H=28)
 for row in img: print(''.join('#' if v else ' ' for v in row))
PY"
