/bin/bash -lc "python3 - <<'PY'
from PIL import Image, ImageDraw, ImageFont
import numpy as np
font=ImageFont.truetype('/System/Library/Fonts/Supplemental/Arial Bold.ttf', 120)
letters=list('BDFHJLNPRTVXZ')
N=128
for ch in letters:
 img=Image.new('L',(N,N),0)
 d=ImageDraw.Draw(img)
 # center text using bbox
 bbox=d.textbbox((0,0),ch,font=font)
 w=bbox[2]-bbox[0]; h=bbox[3]-bbox[1]
 d.text(((N-w)/2-bbox[0],(N-h)/2-bbox[1]),ch,font=font,fill=255)
 a=np.array(img)>40
 # print downsample 48x32
 H,W=a.shape
 ys=np.linspace(0,H,32,endpoint=False,dtype=int); xs=np.linspace(0,W,48,endpoint=False,dtype=int)
 small=a[np.ix_(ys,xs)] # wrong sampling but ok
 print(ch)
 for row in small:
  print(''.join('#' if v else ' ' for v in row))
 print()
PY"
