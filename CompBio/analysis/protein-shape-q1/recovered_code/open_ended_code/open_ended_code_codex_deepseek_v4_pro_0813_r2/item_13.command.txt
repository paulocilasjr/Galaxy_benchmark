def project_image(points, n, N=96, sigma_pix=None):
    # n unit view direction. pick u,v perpendicular.
    n=np.asarray(n,float); n/=np.linalg.norm(n)
    # arbitrary basis
    a=np.array([0.,0.,1.]) if abs(n[2])<0.9 else np.array([1.,0.,0.])
    u=np.cross(n,a); u/=np.linalg.norm(u)
    v=np.cross(n,u); v/=np.linalg.norm(v)
    p=points-points.mean(0)
    x=p@u; y=p@v
    margin=4
    span=max(x.max()-x.min(), y.max()-y.min())
    if span==0: span=1
    scale=(N-2*margin)/span
    xi=((x-x.min())*scale + margin).astype(int)
    yi=((y-y.min())*scale + margin).astype(int)
    img=np.zeros((N,N),np.float32)
    # binary occupancy
    for xx,yy in zip(xi,yi):
        if 0<=xx<N and 0<=yy<N: img[yy,xx]=1.0
    # blur slightly and threshold to thicken
    if sigma_pix:
        from scipy.ndimage import gaussian_filter
        img=gaussian_filter(img,sigma_pix)
        img=(img>0.1).astype(np.float32)
    return img

def dice_rot(imgp, imgt, angles=None):
    # best translation dice between binary p and rotated t
    if angles is None: angles=np.arange(0,180,5)
    best=(-1,None,None,None)
    for ang in angles:
        tr=rotate(imgt, ang, order=1, mode='constant', cval=0, preserve_range=True)
        tr=tr>0.5
        corr=fftconvolve(imgp, tr[::-1,::-1], mode='same')
        maxcorr=corr.max()
        area_t=tr.sum(); area_p=imgp.sum()
        dice=2*maxcorr/(area_p+area_t) if area_p+area_t>0 else 0
        if dice>best[0]: best=(dice,ang,maxcorr,tr.sum())
    return best
