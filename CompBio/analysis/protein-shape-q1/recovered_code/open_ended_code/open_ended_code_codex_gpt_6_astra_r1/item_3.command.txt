/bin/bash -lc "cat > inspect_shape.py <<'PY'
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from collections import Counter
rows=[l for l in open('protein.shape.q1.pdb') if l.startswith('ATOM')]
ca=[l for l in rows if l[12:16].strip()=='CA']
x=np.array([[float(l[30:38]),float(l[38:46]),float(l[46:54])] for l in ca])
print('atoms',len(rows),'CA',len(x),'chains',Counter(l[21] for l in ca))
x-=x.mean(0)
u,s,v=np.linalg.svd(x,full_matrices=False)
print('singular',s,'bounds',np.ptp(x,axis=0))
p=x@v.T
np.save('coords.npy',p)
fig,axs=plt.subplots(1,3,figsize=(15,6))
for ax,(i,j) in zip(axs,[(0,1),(0,2),(1,2)]):
 ax.scatter(p[:,i],p[:,j],s=3,c=np.arange(len(p)),cmap='turbo')
 ax.set_aspect('equal'); ax.set_title(f'PC{i+1}, PC{j+1}')
fig.tight_layout();fig.savefig('projections.png',dpi=140)
PY
python3 inspect_shape.py"
