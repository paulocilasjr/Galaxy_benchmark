/bin/bash -lc "mkdir -p .mplconfig; cat > views.py <<'PY'
import os
os.environ['MPLCONFIGDIR']=os.path.abspath('.mplconfig')
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
p=np.load('coords.npy')
fig,axs=plt.subplots(4,6,figsize=(18,12))
for row,tilt in enumerate([0,30,55,75]):
 for col,az in enumerate(range(0,180,30)):
  a,b=np.deg2rad([az,tilt]); up=np.array([0,np.cos(a),np.sin(a)]); right=np.array([np.cos(b),-np.sin(b)*np.sin(a),np.sin(b)*np.cos(a)])
  q=np.column_stack([p@right,p@up]);ax=axs[row,col]
  ax.scatter(q[:,0],q[:,1],s=2,color='black');ax.set_aspect('equal');ax.set_title(f'az {az}, tilt {tilt}');ax.axis('off')
fig.tight_layout();fig.savefig('rotations.png',dpi=130)
PY
python3 views.py"
