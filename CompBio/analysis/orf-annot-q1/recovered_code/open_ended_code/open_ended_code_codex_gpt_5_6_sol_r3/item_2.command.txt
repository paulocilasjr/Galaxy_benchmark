/bin/bash -lc "python - <<'PY'
from pathlib import Path
s=''.join(Path('orf.annot.q1.fa').read_text().splitlines()[1:]).upper().replace('U','T')
print('length',len(s))
stops={'TAA','TAG','TGA'}
for frame in range(3):
    orfs=[]
    for i in range(frame,len(s)-2,3):
        if s[i:i+3]=='ATG':
            for j in range(i+3,len(s)-2,3):
                if s[j:j+3] in stops:
                    orfs.append((i,(j-i)//3,s[j:j+3],j))
                    break
            else: orfs.append((i,(len(s)-i)//3,'none',None))
    print('frame',frame,sorted(orfs,key=lambda x:x[1],reverse=True)[:20])
PY"
