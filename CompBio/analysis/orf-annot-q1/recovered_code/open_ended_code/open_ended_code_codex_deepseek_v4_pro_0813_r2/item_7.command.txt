/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
from Bio.Seq import Seq
rec = SeqIO.read('orf.annot.q1.fa','fasta')
seq = str(rec.seq).upper().replace('T','U')
print('len',len(seq), 'first', seq[:20])
# find all ATG and longest
best=[]
for i in range(len(seq)-2):
    if seq[i:i+3]=='AUG':
        aa = str(Seq(seq[i:]).translate(to_stop=True))
        # ensure starts M and length until stop; translate(to_stop=True) stops at first stop
        best.append((i,len(aa),aa[:30]))
best=sorted(best,key=lambda x:(-x[1],x[0]))
print(best[:5])
print('BEST',best[0])
PY"
