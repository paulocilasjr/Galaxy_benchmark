/bin/bash -lc "python3 - <<'PY'
from urllib.parse import urlencode
from urllib.request import urlopen
import time,re
rid='762X5DR2014'
u='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi?'+urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML','HITLIST_SIZE':'10'})
for n in range(8):
 x=urlopen(u,timeout=60).read().decode()
 m=re.search(r'Status[= ]+(\\w+)',x)
 print('poll',n,m.group(1) if m else 'results',len(x))
 if 'Status=WAITING' not in x: break
 time.sleep(3)
if x.lstrip().startswith('<?xml'):
 print(x[:20000])
else:
 for pat in ['Hit_def>["'^<]+'"','Hit_accession>["'^<]+'"','Hsp_qseq>["'^<]+'"','Hsp_hseq>["'^<]+'"','Status[= ]+\\\\w+']:
  print(pat,re.findall(pat,x)[:20])
PY"
