/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
s=''.join(l.strip() for l in Path('orf.annot.q1.fa').read_text().splitlines() if not l.startswith('>')).upper()
comp=str.maketrans('AUGC','UACG')
rc=s.translate(comp)[::-1]
stops={'UAA','UAG','UGA'}
for name,seq in [('plus',s),('minus_rc',rc)]:
    best=None
    for i in range(len(seq)-2):
        if seq[i:i+3]=='AUG':
            for j in range(i+3,len(seq)-2,3):
                if seq[j:j+3] in stops:
                    aa=(j-i)//3
                    cand=(aa,i,j)
                    if best is None or cand>best: best=cand
                    break
    print(name,best)
PY"
