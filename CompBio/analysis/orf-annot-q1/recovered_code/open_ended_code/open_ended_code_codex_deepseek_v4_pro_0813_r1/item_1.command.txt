/bin/bash -c "python3 - <<'PY'
from pathlib import Path
s=Path('orf.annot.q1.fa').read_text().splitlines()[1].strip()
print(len(s))
print(s)
print('first AUG',s.find('AUG'))
# find all ATGs and stop codons
stops={'UAA','UAG','UGA'}
for i in range(len(s)-2):
    if s[i:i+3]=='AUG':
        for j in range(i+3,len(s)-2,3):
            if s[j:j+3] in stops:
                print(i,j,(j-i)//3,s[i:j+3][:30],s[i:j+3][-10:])
                break
PY"
