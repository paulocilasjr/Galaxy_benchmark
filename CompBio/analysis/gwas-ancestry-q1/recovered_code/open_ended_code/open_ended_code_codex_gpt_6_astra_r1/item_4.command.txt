/bin/bash -lc "python - <<'PY'
import requests
urls=['https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5a.20130502.genotypes.vcf.gz.tbi','https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/integrated_call_samples_v3.20130502.ALL.panel','https://1000genomes.s3.amazonaws.com/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5a.20130502.genotypes.vcf.gz.tbi']
for url in urls:
 try:
  r=requests.get(url,timeout=30);print(r.status_code,len(r.content),url)
  if r.status_code==200:open(url.split('/')[-1],'wb').write(r.content)
 except Exception as e:print(e)
PY"
