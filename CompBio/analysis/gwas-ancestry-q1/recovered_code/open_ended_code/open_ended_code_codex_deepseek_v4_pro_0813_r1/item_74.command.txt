/bin/bash -lc "python3 - <<'PY'
import requests,json,gzip
rows=[]
with gzip.open('gwas.ancestry.q1.tsv.gz','rt') as f:
 f.readline()
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  if p[6]=='NA': rows.append((p[1],p[4],p[5]))
ids=[r[0] for r in rows]
r=requests.post('https://myvariant.info/v1/variant', json={'ids':ids,'fields':'gnomad_genome'}, timeout=120)
d=r.json()
by={}
for x in d:
 by.setdefault(x.get('query'),[]).append(x)
for rs,effect,other in rows:
 xs=by.get(rs,[])
 matches=[x for x in xs if (x.get('gnomad_genome') or {}).get('alt')==effect]
 print('\\n'+rs+' effect='+effect+' other='+other)
 for x in (matches or xs):
  g=x.get('gnomad_genome') or {}
  af=g.get('af') or {}
  print(' ',g.get('ref'),g.get('alt'),'AFR',af.get('af_afr'),'AMR',af.get('af_amr'),'ASJ',af.get('af_asj'),'EAS',af.get('af_eas'),'NFE',af.get('af_nfe'),'OTH',af.get('af_oth'))
PY"
