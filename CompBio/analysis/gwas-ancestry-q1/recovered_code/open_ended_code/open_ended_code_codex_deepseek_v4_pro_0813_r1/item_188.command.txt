/bin/bash -lc "python3 - <<'PY'
from playwright.sync_api import sync_playwright
import json,time
with sync_playwright() as p:
    browser=p.chromium.launch(headless=True)
    page=browser.new_page()
    url='https://dataverse.harvard.edu/api/datasets/:persistentId/?persistentId=doi:10.7910/DVN/FAWEQK'
    page.goto(url, wait_until='domcontentloaded', timeout=60000)
    time.sleep(15)  # challenge auto reload
    print('url',page.url)
    body=page.content()
    print(body[:500])
    if 'awsWaf' not in body[:500]:
        try:
            data=page.evaluate('() => JSON.parse(document.body.innerText)')
            print(json.dumps(data, indent=2)[:5000])
        except Exception as e:
            print('eval err',e)
            print(body[:2000])
    browser.close()
PY"
