/bin/bash -lc "python3 - <<'PY'
import re,subprocess
entries=[]
for line in open('GIA_Vitals_toc.txt'):
 m=re.match(r'([-d])\\S+\\s+\\S+\\s+(\\d+)\\s+\\S+\\s+\\S+\\s+(.*)"'$'"', line.strip())
 if m:
  typ=m.group(1); size=int(m.group(2)); name=m.group(3).strip(); entries.append((typ,size,name))
# compute offset map
off=0
pos={}
for typ,size,name in entries:
    data_start=off+512
    pos[name]=data_start
    blocks=(size+511)//512
    off += 512 + blocks*512
print('total',off)
name='MVP_R4.1000G_AGR.GIA.Surveys_Vitals_Quantitative/MVP_R4.1000G_AGR.BMI_Mean_INT.EAS.GIA.dbGaP.metadata.txt'
typ,size,_=next(e for e in entries if e[2]==name)
start=pos[name]; end=start+size-1
url='https://ftp.ncbi.nlm.nih.gov/dbgap/studies/phs002453/phs002453.v1.p1/analyses/GIA/phs002453.MVP_R4.1000G_AGR.GIA.Surveys_Vitals_Quantitative.analysis-PI.MULTI.tar'
out=subprocess.check_output(['curl','-L','--max-time','30','-s','-r',f'{start}-{end}',url])
print(name,size,len(out),out[:300].decode(errors='replace'))
PY"
