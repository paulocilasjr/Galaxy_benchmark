/bin/bash -lc "python3 - <<'PY'
import requests, json, gzip
ids=[]
with gzip.open('gwas.ancestry.q1.tsv.gz','rt') as f:
    header=f.readline().strip().split('\\t')
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if p[6]=='NA':
            ids.append(p[1])
print(len(ids))
r=requests.post('https://myvariant.info/v1/variant', json={'ids': ids, 'fields':'gnomad_genome.af,gnomad_exome.af'}, timeout=60)
print(r.status_code)
d=r.json()
for rs in ids:
    item=d.get(rs)
    if item:
        af=(item.get('gnomad_genome') or {}).get('af') or {}
        ea=None
        # effect allele from file list not here, just print AFs
        print(rs, 'AFR', af.get('af_afr'), 'AMR', af.get('af_amr'), 'ASJ', af.get('af_asj'), 'EAS', af.get('af_eas'), 'NFE', af.get('af_nfe'), 'OTH', af.get('af_oth'))
    else:
        print(rs, 'notfound')
PY"
