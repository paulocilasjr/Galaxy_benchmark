/bin/bash -lc "python3 - <<'PY'
import re,subprocess,os
entries=[]
for line in open('GIA_Vitals_toc.txt'):
 m=re.match(r'([-d])\\S+\\s+\\S+\\s+(\\d+)\\s+\\S+\\s+\\S+\\s+(.*)"'$'"', line.strip())
 if m: entries.append((m.group(1),int(m.group(2)),m.group(3).strip()))
off=0; pos={}; sizes={}
for typ,size,name in entries:
    if len(name.encode())>100:
        off += 512 + ((len(name.encode())+511)//512)*512
    pos[name]=off+512; sizes[name]=size
    off += 512 + ((size+511)//512)*512
base='https://ftp.ncbi.nlm.nih.gov/dbgap/studies/phs002453/phs002453.v1.p1/analyses/GIA/phs002453.MVP_R4.1000G_AGR.GIA.Surveys_Vitals_Quantitative.analysis-PI.MULTI.tar'
for name in entries:
    if 'Height_' in name[2] and name[2].endswith('.metadata.txt'):
        start=pos[name[2]]; end=start+sizes[name[2]]-1
        out=subprocess.check_output(['curl','-L','--max-time','30','-s','-r',f'{start}-{end}',base])
        text=out.decode(errors='replace')
        print('\\n',name[2].split('/')[-1])
        print(text[:600])
PY"
