/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
snp='rs9435734'
url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{snp}/associations?projection=associationBySnp'
data=json.load(urllib.request.urlopen(url, timeout=30))
assocs=data.get('_embedded',{}).get('associations',[])
for a in assocs:
    sid=a.get('_links',{}).get('study',{}).get('href','')
    # fetch study accession
    try:
        s=json.load(urllib.request.urlopen(sid, timeout=30))
        acc=s.get('accessionId'); sample=s.get('initialSampleSize')
    except Exception as e:
        acc='ERR'; sample=''
    print('p',a.get('pvalue'),'trait',[t.get('trait') for t in a.get('efoTraits',[])],'study',acc,'sample',sample)
PY"
