/bin/bash -lc "python3 - <<'PY'
import requests, json, time
for rs in ['rs9435734','rs4920608','rs2871776','rs17361789','rs10429950','rs3766763','rs11164665','rs2744666','rs1892425','rs11205303','rs1014719']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}'
    try:
        r=requests.get(url,headers={'Accept':'application/json'},timeout=20)
        print('\\n',rs,r.status_code)
        if r.ok:
            d=r.json()
            print('location',d.get('locations'))
            for a in (d.get('associations') or [])[:5]:
                print('assoc',a.get('pValue'),a.get('pValueMantissa'),a.get('pValueExponent'),a.get('riskAllele'),a.get('orPerCopyNum'),a.get('betaNum'),a.get('studyAccession'))
                print('trait',[t.get('trait') for t in a.get('_embedded',{}).get('traitNames',[])][:3] if a.get('_embedded',{}).get('traitNames') else None)
                print('pop',a.get('_embedded',{}).get('initialSampleDescription') if a.get('_embedded') else None)
    except Exception as e: print('ERR',e)
PY"
