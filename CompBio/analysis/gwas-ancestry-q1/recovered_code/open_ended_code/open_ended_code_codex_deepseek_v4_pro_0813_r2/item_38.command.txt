/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
snps=['rs761423','rs10429950']
for snp in snps:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{snp}/associations?projection=associationBySnp'
    data=json.load(urllib.request.urlopen(url, timeout=30))
    assocs=data.get('_embedded',{}).get('associations',[])
    print('\\n====',snp)
    for a in assocs:
        study=a.get('study')
        print('pvalue',a.get('pvalue'),'mant',a.get('pvalueMantissa'),'exp',a.get('pvalueExponent'))
        print('studyAcc?', study)
        print('trait', [t.get('trait') for t in a.get('efoTraits',[])])
        print('riskFreq', a.get('riskFrequency'))
        # maybe links to study
        links=a.get('_links',{})
        print('links', links.get('study',{}).get('href'))
        print()
PY"
