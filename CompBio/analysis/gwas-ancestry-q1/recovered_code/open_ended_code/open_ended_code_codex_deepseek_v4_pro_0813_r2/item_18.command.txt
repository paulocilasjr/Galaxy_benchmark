/bin/bash -lc "python3 - <<'PY'
import urllib.request
for url in [
'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/raw/main/README.md',
'https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/README.md',
'https://datasets-server.huggingface.co/info?dataset=Genentech%2Fcompbiobench-data-v1',
]:
    print('URL',url)
    try:
        req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
        data=urllib.request.urlopen(req, timeout=30).read().decode('utf-8','ignore')
        print(data[:2000])
    except Exception as e:
        print(type(e).__name__, e)
    print('\\n---')
PY"
