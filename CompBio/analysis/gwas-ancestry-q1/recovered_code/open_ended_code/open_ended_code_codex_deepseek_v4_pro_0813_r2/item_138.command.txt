/bin/bash -lc "python3 - <<'PY'
import requests, json
url='https://gnomad.broadinstitute.org/api'
query='''query { __schema { queryType { fields { name args { name type { kind name ofType { kind name ofType { kind name } } } } } } } }'''
r=requests.post(url,json={'query':query},timeout=30)
print(r.status_code)
d=r.json()
# print fields of Query
for f in d['data']['__schema']['queryType']['fields']:
    if f['name'] in ['variant','gene','region','variant_by_rsid','rsid']:
        print(f['name'])
        for a in f['args']:
            print(' ',a['name'], a['type'])
PY"
