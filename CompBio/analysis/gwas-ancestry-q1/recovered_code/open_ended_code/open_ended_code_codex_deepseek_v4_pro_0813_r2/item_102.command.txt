/bin/bash -lc "python3 - <<'PY'
import requests, json
rs='rs9435734'
r=requests.get(f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}?projection=associationBySnp')
d=r.json()
print('keys',d.keys())
for a in d.get('associations',[]):
    print(json.dumps({k:a.get(k) for k in ['pValue','pValueMantissa','pValueExponent','riskAllele','riskFrequency','orPerCopyNum','betaNum','betaUnit','range','studyAccession','lastUpdateDate']},indent=2))
    # traverse embedded
    if '_embedded' in a: print('emb keys',a['_embedded'].keys())
    # print trait names separately
    print('traits',a.get('traitNames'))
    print('pvalue text',a.get('pValue'))
PY"
