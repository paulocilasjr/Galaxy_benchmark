import urllib.request, json
for id in ['90706388','90611733','90808440','90782451','21711706','21716320','21708911']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/associations/{id}'
    try:
        a=json.load(urllib.request.urlopen(url, timeout=30))
        print('\\nID',id)
        print('study', a.get('study'))
        print('pvalue', a.get('pvalue'), a.get('pvalueMantissa'), a.get('pvalueExponent'))
        print('trait', [t.get('trait') for t in a.get('efoTraits',[])])
        print('riskFreq', a.get('riskFrequency'))
        # study link
        print('_links', a.get('_links'))
    except Exception as e:
        print(id,e)
PY"
