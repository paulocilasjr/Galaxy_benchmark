/bin/bash -lc "python3 - <<'PY'
import pandas as pd
import numpy as np
p='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-0bb2941c0318233f7b766bc0/workspace/gwas.ancestry.q1.tsv'

print(df.shape)
print(df.dtypes)
print(df['P'].head(), df['P'].isna().sum(), (df['P']=='NA').sum())
df['pnum']=pd.to_numeric(df['P'],errors='coerce')
print(df['pnum'].min(), df.loc[df.pnum.idxmin()].to_dict())

# top independent by distance > 1e6 perhaps
sig=df[df.pnum<5e-8].copy().sort_values('pnum')
print('num sig',len(sig))
# simple clump: choose min p then remove within 1Mb
remaining=sig.copy(); chosen=[]
while len(remaining):
    i=remaining.pnum.idxmin(); row=remaining.loc[i]; chosen.append(row)
    remaining=remaining[(remaining.POS-row.POS).abs()>1000000]
print('independent ~1Mb',len(chosen))
for r in chosen[:50]:
    print(r.SNPID,r.RSID,r.POS,r.pnum)
PY"
