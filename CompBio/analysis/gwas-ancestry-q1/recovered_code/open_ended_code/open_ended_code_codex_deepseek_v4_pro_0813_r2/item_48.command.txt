/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for path in ['tasks','tasks/harbor_tasks','traces','derived']:
    url=f'https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/{path}'
    print('PATH',path)
    try:
        data=json.load(urllib.request.urlopen(url, timeout=30))
        for d in data[:200]:
            print(d['type'], d.get('path'), d.get('size'))
    except Exception as e: print(e)
PY"
