import requests,json,math
snp_list=['rs9435734','rs4920608','rs17361789','rs10429950','rs11164665','rs1892425','rs11205303','rs1014719']
# target p mapping rs-> (pos,p)

    matches=[]
    for i,a in enumerate(data):

        if p is not None and target[rs] is not None:

    print('n assoc',len(data),'matches',matches)
    # show first 10 self hrefs p
    for a in data[:10]:
        print(a['_links']['self']['href'].split('/')[-1], a.get('pvalue'), a.get('pvalueMantissa'),a.get('pvalueExponent'),a.get('betaNum'),a.get('betaUnit'))
