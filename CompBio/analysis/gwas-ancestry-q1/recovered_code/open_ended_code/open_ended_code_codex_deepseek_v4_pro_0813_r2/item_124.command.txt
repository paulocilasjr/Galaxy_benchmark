/bin/bash -lc "python3 - <<'PY'
import json
lines=open('codex_raw_stdout.jsonl').read().splitlines()
print('events',len(lines))
# print human-readable sequence of assistant messages / tool commands summaries
for i,l in enumerate(lines):
    try: d=json.loads(l)
    except: continue
    typ=d.get('type')
    if typ=='item.completed':
        item=d.get('item',{})
        if item.get('type')=='command_execution':
            cmd=item.get('command','')
            out=item.get('aggregated_output','')
            print('\\n--- CMD',i, cmd[:500].replace('\\n',' '))
            print(out[:2000])
    elif typ in ['message','assistant_message','turn.completed']:
        print('\\n=== EVENT',typ,i,str(d)[:2000])
PY"
