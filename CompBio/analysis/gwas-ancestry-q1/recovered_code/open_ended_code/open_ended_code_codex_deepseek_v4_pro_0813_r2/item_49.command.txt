/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
base='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/traces/compbiobench'
def rec(path):
    data=json.load(urllib.request.urlopen(base+path, timeout=30))
    for d in data:
        print(d['type'], d.get('path'))
        if d['type']=='directory':
            rec(path+'/'+d['path'].split('/')[-1])  # but path relative issue; pass full
# simpler use path from API; need absolute subpath
def rec2(fullpath):
    data=json.load(urllib.request.urlopen(fullpath, timeout=30))
    for d in data:
        p=d.get('path')
        print(d['type'], p, d.get('size'))
        if d['type']=='directory':
            rec2('https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/'+p)
rec2('https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/traces/compbiobench')
PY"
