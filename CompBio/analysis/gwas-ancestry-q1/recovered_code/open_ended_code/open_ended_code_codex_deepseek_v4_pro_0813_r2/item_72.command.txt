/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for trait in ['body height','body mass index']:
    q=urllib.parse.quote(trait)
    url=f'https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByEfoTrait?efoTrait={q}&size=100'
    data=json.load(urllib.request.urlopen(url, timeout=30))
    emb=data.get('_embedded',{}).get('studies',[])
    print('\\nTRAIT',trait,'total',data.get('page',{}).get('totalElements'))
    for s in emb:
        print(s.get('accessionId'), s.get('initialSampleSize'))
PY"
