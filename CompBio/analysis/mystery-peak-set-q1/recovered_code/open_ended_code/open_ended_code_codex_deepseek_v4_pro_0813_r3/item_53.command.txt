/bin/bash -lc "python3 - <<'PY'
import pyBigWig, glob, os
for f in glob.glob('[REDACTED LOCAL PATH]'):
    out=f.replace('.bb','.bed')
    bw=pyBigWig.open(f)
    o=open(out,'w')
    for chrom,size in bw.chroms().items():
        if not chrom.startswith('chr'): continue
        for e in bw.entries(chrom,0,size):
            o.write(chrom+'\\t'+str(e[0])+'\\t'+str(e[1])+'\\n')
    o.close()
    print(out)
PY
