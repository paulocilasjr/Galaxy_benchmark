/bin/bash -lc "printf \"set\\tn\\tn_any_ccre\\tfrac_any\\tbp_overlap\\tfrac_bp\\n\" > results/ccre_overlap.tsv
for s in S1 S2 S3 S4 S5 S6; do
  n="'$(wc -l < split/${s}.bed | tr -d '"' ')
  c="'$(bedtools intersect -u -a split/${s}.bed -b refs/encodeCcreCombined.sorted.bed | wc -l | tr -d '"' ')
  bp="'$(bedtools coverage -a split/${s}.bed -b refs/encodeCcreCombined.sorted.bed | awk '"'{sum+="'$6} END{print sum+0}'"')
  awk -v set="'$s -v n=$n -v c=$c -v bp=$bp '"'BEGIN{printf \"%s\\t%d\\t%d\\t%.4f\\t%d\\t%.4f\\n\",set,n,c,c/n,bp,bp/(n*500)}' >> results/ccre_overlap.tsv
done
cat results/ccre_overlap.tsv"
