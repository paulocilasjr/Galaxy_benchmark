/bin/bash -lc "gzip -dc refs/DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz | awk 'NR>1{print "'$1"'"\\t\""'$2"'"\\t\""'$3"'"\\t\""'$4"'"\\t\""'$5"'"\\t\""'$6"'"\\t\""'$10}'"' | sort -k1,1 -k2,2n > refs/dhs_index.sorted.bed
wc -l refs/dhs_index.sorted.bed
head -3 refs/dhs_index.sorted.bed
printf \"set\\tn\\tn_any_dhs\\tfrac_any\\tbp_overlap\\tfrac_bp\\tmean_overlapping_dhs_numsamples\\tmedian_numsamples\\n\" > results/dhs_overlap.tsv
for s in S1 S2 S3 S4 S5 S6; do
  n="'$(wc -l < split/${s}.bed | tr -d '"' ')
  c="'$(bedtools intersect -u -a split/${s}.bed -b refs/dhs_index.sorted.bed | wc -l | tr -d '"' ')
  bp="'$(bedtools coverage -a split/${s}.bed -b refs/dhs_index.sorted.bed | awk '"'{sum+="'$6} END{print sum+0}'"')
  stats="'$(bedtools intersect -wa -wb -a split/${s}.bed -b refs/dhs_index.sorted.bed | awk '"'{print "'$10}'"' | sort -n | awk '{a[NR]="'$1; sum+=$1} END{if(NR==0){print "0'"\\t0\"} else {med=(NR%2?a[(NR+1)/2]:(a[NR/2]+a[NR/2+1])/2); print sum/NR\"\\t\"med}}')
  mean="'$(printf "%s" "$stats" | cut -f1); med=$(printf "%s" "$stats" | cut -f2)
  awk -v set=$s -v n=$n -v c=$c -v bp=$bp -v mean=$mean -v med=$med '"'BEGIN{printf \"%s\\t%d\\t%d\\t%.4f\\t%d\\t%.4f\\t%.3f\\t%.1f\\n\",set,n,c,c/n,bp,bp/(n*500),mean,med}' >> results/dhs_overlap.tsv
done
cat results/dhs_overlap.tsv"
