import gzip,collections,numpy as np,json
ref=collections.defaultdict(list)
with gzip.open('DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz','rt') as f:
 next(f)
 for l in f:
  v=l.split('\\t');ref[v[0]].append((int(v[1]),int(v[2])))

 i=np.searchsorted(starts,b,side='left')-1
 if i>=0 and ends[i]>a:hits[s]+=1

 if i>=0 and ends[i]>m:centers[s]+=1

print(json.dumps(results,indent=2));json.dump(results,open('dhs_overlap_results.json','w'),indent=2)
PY
