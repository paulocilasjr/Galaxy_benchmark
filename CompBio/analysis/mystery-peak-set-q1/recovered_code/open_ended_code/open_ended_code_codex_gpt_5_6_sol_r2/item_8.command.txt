/bin/bash -lc "gzip -cd mystery.peak.set.q1.bed.gz | awk 'BEGIN{OFS=\"\\t\"}{print "'$1,$2,$3,$4,NR}'"' > mystery.peaks.bed
gzip -cd DHS_Index_and_Vocabulary_hg38.txt.gz | tail -n +2 | cut -f1-6 > dhs_index.bed
bedtools intersect -a mystery.peaks.bed -b dhs_index.bed -wao > peaks_vs_dhs.tsv
awk 'BEGIN{OFS=\"\\t\"} {id="'$5; set=$4; ov=$12+0; ns=$11+0; if(!(id in seen)){seen[id]=1; n[set]++; max[id]=0; outset[id]=set} if(ov>0){hit[id]=1; if(ns>max[id])max[id]=ns}} END{for(id in seen){s=outset[id]; h[s]+=hit[id]; sm[s]+=max[id]; if(max[id]>=2)h2[s]++; if(max[id]>=5)h5[s]++; if(max[id]>=10)h10[s]++; if(max[id]>=50)h50[s]++; if(max[id]>=100)h100[s]++} for(i=1;i<=6;i++){s="S"i; print s,n[s],h[s],h[s]/n[s],h2[s]/n[s],h5[s]/n[s],h10[s]/n[s],h50[s]/n[s],h100[s]/n[s],sm[s]/n[s]}}'"' peaks_vs_dhs.tsv | sort -k1,1V"
