/bin/bash -lc "python3 -u - <<'PY'
import requests
for assay in ['ATAC-seq','DNase-seq']:
 u='https://www.encodeproject.org/search/'
 p={'type':'File','file_format':'bed','assembly':'GRCh38','status':'released','assay_title':assay,'limit':100,'format':'json'}
 r=requests.get(u,params=p); print(assay,r.status_code,len(r.content),r.text[:100],flush=True)
 try:
  j=r.json(); print('n',len(j.get('@graph',[])),j.get('total'),flush=True)
 except Exception as e: print(e,flush=True)
PY"
