/bin/bash -lc "python3 - <<'PY'
import gzip,bisect,collections,statistics
bychr=collections.defaultdict(list)
for line in gzip.open('refGene.txt.gz','rt'):
 f=line.rstrip('\\n').split('\\t')
 if len(f)<13: continue
 chrom,strand=f[2],f[3]; txs,txe=int(f[4]),int(f[5]); name=f[12]
 tss=txs if strand=='+' else txe
 bychr[chrom].append((tss,name,txs,txe,strand))
for c in bychr: bychr[c].sort()
sets=collections.defaultdict(list)
for line in gzip.open('mystery.peak.set.q1.bed.gz','rt'):
 c,s,e,lab=line.rstrip().split('\\t'); sets[lab].append((c,int(s),int(e)))
for lab,peaks in sets.items():
 d=[]; prom=0; body=0; genes=collections.Counter()
 for c,s,e in peaks:
  center=(s+e)//2; arr=bychr.get(c,[])
  if not arr: continue
  i=bisect.bisect_left(arr,(center,))
  cand=[]
  for j in range(max(0,i-3),min(len(arr),i+3)):
   t,n,ts,te,st=arr[j]; cand.append((abs(t-center),n,ts,te))
  md,n,ts,te=min(cand); d.append(md); genes[n]+=1
  if md<=2000: prom+=1
  if s<te and e>ts: body+=1
 print(lab,'n',len(peaks),'median_tss_dist',statistics.median(d),'within2kb',round(prom/len(d),3),'within10kb',round(sum(x<=10000 for x in d)/len(d),3),'overlap_tx',round(body/len(d),3),'topgenes',genes.most_common(10))
PY"
