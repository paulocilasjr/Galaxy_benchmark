/bin/bash -lc "python3 - <<'PY'
import requests,os
marks=['H3K4me3','H3K27ac','H3K4me1','H3K36me3','H3K27me3','H3K9me3']
for mark in marks:
 p={'type':'File','file_format':'bed','assembly':'GRCh38','status':'released','assay_title':'Histone ChIP-seq','target.label':mark,'output_type':'replicated peaks','limit':1,'format':'json'}
 x=requests.get('https://www.encodeproject.org/search/',params=p).json()['@graph'][0]
 u='https://www.encodeproject.org'+x['href']
 out='encode_'+mark+'.bed.gz'
 print(mark,x['accession'],x['dataset'],x['file_size'],flush=True)
 with requests.get(u,stream=True) as rr:
  rr.raise_for_status()
  with open(out,'wb') as f:
   for chunk in rr.iter_content(1024*1024): f.write(chunk)
PY
ls -lh encode_H3*.bed.gz"
