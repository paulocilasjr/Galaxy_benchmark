/bin/bash -lc 'set -e
gzip -dc DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz | tail -n +2 | cut -f1-3 | sort -k1,1 -k2,2n > dhs_index_sorted.bed
head dhs_index_sorted.bed
for s in S1 S2 S3 S4 S5 S6; do
  n=$(wc -l < set_$s.bed | tr -d '"' ')
  ov="'$(bedtools intersect -u -a set_$s.bed -b dhs_index_sorted.bed | wc -l | tr -d '"' ')
  python3 - <<PY
print('"'$s'"', '"'$n'"', '"'$ov'"', round("'$ov/$n,4))
PY
done'
