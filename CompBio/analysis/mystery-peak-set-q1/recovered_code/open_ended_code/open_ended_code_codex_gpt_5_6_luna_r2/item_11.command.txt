/bin/bash -lc "python3 - <<'PY'
import gzip, collections, statistics
fn='mystery.peak.set.q1.bed.gz'
D=collections.defaultdict(list)
for line in gzip.open(fn,'rt'):
    c,s,e,n=line.rstrip().split('\\t')
    D[n].append((c,int(s),int(e)))
for n in sorted(D):
    x=D[n]
    lens=[e-s for c,s,e in x]
    starts=[s for c,s,e in x]
    print(n,len(x),'length',min(lens),max(lens),'median_start',statistics.median(starts),'chrY',sum(c=='chrY' for c,s,e in x),'chrX',sum(c=='chrX' for c,s,e in x),'first',x[:3],'last',x[-3:])
# overlaps exact or reciprocal among sets using interval sweep
for n in sorted(D):
  for m in sorted(D):
    if n>=m: continue
    hits=0
    # by chromosome sorted
    j=0
    for c,s,e in D[n]:
      while j<len(D[m]) and (D[m][j][0]<c or (D[m][j][0]==c and D[m][j][2]<=s)): j+=1
      # j invalid global due chromosome lexical same? just brute nearby all use bisect grouped
    # do grouped
    bym=collections.defaultdict(list)
    for c,s,e in D[m]: bym[c].append((s,e))
    for c,s,e in D[n]:
      for a,b in bym[c]:
        if a>=e: break
        if b>s: hits+=1; break
    print('overlap',n,m,hits,round(hits/len(D[n]),3),round(hits/len(D[m]),3))
PY"
