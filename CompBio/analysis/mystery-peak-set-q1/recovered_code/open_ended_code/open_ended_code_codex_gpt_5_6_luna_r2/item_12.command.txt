/bin/bash -lc "python3 - <<'PY'
import gzip, collections, statistics, bisect
fn='mystery.peak.set.q1.bed.gz'
D=collections.defaultdict(list)
for line in gzip.open(fn,'rt'):
    c,s,e,n=line.rstrip().split('\\t')
    D[n].append((c,int(s),int(e)))
for n in sorted(D):
    x=D[n]
    lens=[e-s for c,s,e in x]
    starts=[s for c,s,e in x]
    print(n,len(x),'length',min(lens),max(lens),'median_start',statistics.median(starts),'chrY',sum(c=='chrY' for c,s,e in x),'chrX',sum(c=='chrX' for c,s,e in x),'first',x[:3],'last',x[-3:])
by={}
for n,x in D.items():
  by[n]={}
  for c,s,e in x: by[n].setdefault(c,[]).append((s,e))
  for c in by[n]: by[n][c].sort()
for n in sorted(D):
  for m in sorted(D):
    if n>=m: continue
    hitsn=hitsm=0
    for c,s,e in D[n]:
      a=by[m].get(c,[]); k=bisect.bisect_left(a,(e,-1))
      # only previous intervals that have end > s; use local backwards until start<e, potentially overlap density small
      for aa,bb in reversed(a[:k]):
        if aa>=e: continue
        if bb>s: hitsn+=1; break
        if aa < s-1000: break
    for c,s,e in D[m]:
      a=by[n].get(c,[]); k=bisect.bisect_left(a,(e,-1))
      for aa,bb in reversed(a[:k]):
        if aa>=e: continue
        if bb>s: hitsm+=1; break
        if aa < s-1000: break
    print('overlap',n,m,hitsn,hitsm,round(hitsn/len(D[n]),3),round(hitsm/len(D[m]),3))
PY"
