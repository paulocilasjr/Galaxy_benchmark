/bin/bash -lc "python3 - <<'PY'
import gzip, collections, statistics, os, re
bed=collections.defaultdict(list)
for line in gzip.open('mystery.peak.set.q1.bed.gz','rt'):
 c,s,e,n=line.rstrip().split('\\t'); bed[c].append((int(s),int(e),n))
stat=collections.defaultdict(lambda: [0,0,0,collections.Counter(),0,0])
# n, GC, CpG, basecounts, N, runs etc
for c,rows in bed.items():
 p=c+'.fa.gz'
 if not os.path.exists(p): continue
 with gzip.open(p,'rt') as f:
  header=next(f); seq=''.join(x.strip() for x in f).upper()
 for s,e,n in rows:
  q=seq[s:e]
  x=stat[n]; x[0]+=1; x[1]+=sum(a in 'GC' for a in q); x[2]+=sum(q[i:i+2]=='CG' for i in range(len(q)-1)); x[3].update(q); x[4]+=q.count('N'); x[5]+=sum(1 for i in range(len(q)-5) if q[i:i+6] in ('AAAAAA','CCCCCC','GGGGGG','TTTTTT'))
for n in sorted(stat):
 x=stat[n]; total=x[0]*500; print(n,x[0], 'GC%',round(100*x[1]/total,2),'CpG/500',round(x[2]/x[0],2),'N',x[4], '6merhom',x[5], 'bases',' '.join(f'{b}:{x[3][b]/total:.3f}' for b in 'ACGT'))
PY"
