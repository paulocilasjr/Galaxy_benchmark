/bin/bash -lc "gzip -cd data/DHS_Index_and_Vocabulary_hg38_WM20190703.txt.gz | awk 'NR>1{print "'$1"'"\\t\""'$2"'"\\t\""'$3}'"' | sort -k1,1 -k2,2n > work/dhs_index.hg38.sorted.bed && wc -l work/dhs_index.hg38.sorted.bed && bedtools intersect -a work/mystery.sorted.bed -b work/dhs_index.hg38.sorted.bed -c -sorted | awk '{total["'$4]++; if($5>0) hit[$4]++; sum[$4]+=$5} END{for(k in total) printf "%s'"\\t%d\\t%d\\t%.4f\\t%.3f\\n\", k,total[k],hit[k],hit[k]/total[k],sum[k]/total[k]}' | sort && bedtools coverage -a work/mystery.sorted.bed -b work/dhs_index.hg38.sorted.bed -sorted | awk '{n["'$4]++; frac[$4]+=$8} END{for(k in n) printf "%s'"\\tmeanFrac=%.4f\\n\", k,frac[k]/n[k]}' | sort"
