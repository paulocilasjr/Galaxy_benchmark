/bin/bash -lc "python3 - <<'PY'
import pandas as pd
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
print(x.shape)
print(list(x.columns))
print(x.describe().T[['mean','std','min','max']])
# top 15 specificity by column: high value and high ratio to next highest
for c in x.columns:
    ratios=[]
    for g,row in x.iterrows():
        vals=row.sort_values(ascending=False)
        if vals.index[0]==c and vals.iloc[0]>1:
            ratio=(vals.iloc[0]+1)/(vals.iloc[1]+1)
            ratios.append((ratio, vals.iloc[0], g, vals.index[1], vals.iloc[1]))
    ratios.sort(reverse=True)
    print('\\n',c)
    for r in ratios[:20]: print(r)
PY"
