/bin/bash -lc "python3 - <<'PY'
import gzip, csv, math, statistics
import pandas as pd
fn='sample.swap.rna.q2.tsv.gz'
df=pd.read_csv(fn,sep='\\t',index_col=0)
print(df.shape)
print(df.columns.tolist())
print(df.sum().round(1).to_dict())
print(df.describe().loc[['mean','50%','max']])
# correlations log1p
log=(df+1).applymap(math.log1p)
print('corr')
print(log.corr().round(3))
# top markers by logFC vs second best or rest mean
for col in df.columns:
    vals=df[col]
    others=df.drop(columns=[col])
    rest=others.mean(axis=1)
    maxother=others.max(axis=1)
    score=(vals+1)/(rest+1)
    # require expr >5
    top=score[vals>5].sort_values(ascending=False).head(10)
    print('\\n',col)
    for g,s in top.items():
        print(g, vals[g], 'score', round(s,1), 'secondmax', maxother[g], 'restmean', round(rest[g],2))
PY"
