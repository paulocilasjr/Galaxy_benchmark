/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
from collections import Counter
p='sample.swap.rna.q2.tsv.gz'
with gzip.open(p,'rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    n=0
    vals=[]
    for line in f:
        if line.strip():
            x=line.rstrip('\\n').split('\\t')
            n+=1
            vals.append([float(v) for v in x[1:]])
A=np.array(vals)
print('genes',n,'celltypes',len(header)-1)
print(header)
print('min max mean median nonzero')
for j,h in enumerate(header[1:]):
    v=A[:,j]
    print(h, v.min(),v.max(),v.mean(),np.median(v),np.count_nonzero(v),np.percentile(v,[50,75,90,95,99]))
print('column Pearson correlations')
C=np.corrcoef(A.T)
print('\\t'+'\\t'.join(header[1:]))
for h,row in zip(header[1:],C): print(h+'\\t'+'\\t'.join(f'{z:.4f}' for z in row))
PY"
