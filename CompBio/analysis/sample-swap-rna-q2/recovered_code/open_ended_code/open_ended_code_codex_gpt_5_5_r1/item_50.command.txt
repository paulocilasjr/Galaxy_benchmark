/bin/bash -lc "python3 - <<'PY'
import csv, collections
p='wheat_root_atlas_repo/GRN_regulon_analysis/input/miniex_regulons/tae_rankedRegulons.csv'
ct=collections.defaultdict(collections.Counter)
with open(p) as f:
    r=csv.DictReader(f, delimiter=';')
    for row in r:
        cluster=row['cluster']
        celltype=row['celltype']
        # cluster field like Celltype_Cluster_5
        cid=cluster.split('_Cluster_')[-1]
        ct[cid][celltype]+=1
for cid in sorted(ct, key=lambda x:int(x) if x.isdigit() else 999):
    print(cid, ct[cid].most_common())
PY"
