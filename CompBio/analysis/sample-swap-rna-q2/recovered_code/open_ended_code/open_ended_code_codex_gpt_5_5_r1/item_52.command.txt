/bin/bash -lc "python3 - <<'PY'
import csv,gzip,math,statistics
from collections import defaultdict, Counter
# cluster celltype mapping from regulon
cid2ct={}
with open('wheat_root_atlas_repo/GRN_regulon_analysis/input/miniex_regulons/tae_rankedRegulons.csv') as f:
    r=csv.DictReader(f, delimiter=';')
    for row in r:
        cid=row['cluster'].split('_Cluster_')[-1]
        cid2ct[cid]=row['celltype']
print('cid2ct')
for cid in sorted(cid2ct, key=lambda x:int(x)):
    print(cid, cid2ct[cid])
# read markers, top by avg_log2FC per cluster
markers=defaultdict(list)
with open('wheat_root_atlas_repo/conserved_marker_analysis/input/tae_log2FC0.5_FDR0.05.csv') as f:
    r=csv.DictReader(f)
    for row in r:
        markers[row['cluster']].append((row['gene ID'], float(row['avg_log2FC']), float(row['pct.1']), float(row['pct.2'])))
for c in markers: markers[c].sort(key=lambda x:x[1], reverse=True)
want={g for ms in markers.values() for g,_,_,_ in ms}
expr={}
with gzip.open('sample.swap.rna.q2.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t')
    header=next(r)[1:]
    for row in r:
        if row[0] in want:
            expr[row[0]]=[float(x) for x in row[1:]]
print('header', header)
print('matrix markers', len(expr), 'of', len(want))
# score by mean log1p expression z within gene across columns, top N markers
def zscores(vals):
    xs=[math.log1p(v) for v in vals]
    m=sum(xs)/len(xs)
    sd=(sum((x-m)**2 for x in xs)/len(xs))**0.5
    return [(x-m)/sd if sd else 0 for x in xs]
for N in [10,20,50,100,9999]:
    print('\\nN',N)
    for cid in sorted(markers, key=lambda x:int(x)):
        scores=[0]*len(header); n=0; maxcounts=Counter()
        used=[]
        for g,fc,p1,p2 in markers[cid][:N]:
            if g in expr:
                zs=zscores(expr[g])
                scores=[a+b for a,b in zip(scores,zs)]
                n+=1; used.append(g)
                maxcounts[header[max(range(len(header)), key=lambda i: expr[g][i])]]+=1
        if n:
            scores=[s/n for s in scores]
            best=header[max(range(len(header)), key=lambda i:scores[i])]
            top=sorted(zip(header,scores), key=lambda x:x[1], reverse=True)[:4]
            print(f\"cluster {cid:>2} {cid2ct.get(cid,'?'):<25} n={n:3} best={best:<12} maxcnt={dict(maxcounts.most_common(3))} top=\" + ' '.join(f'{h}:{s:.2f}' for h,s in top))
PY"
