/bin/bash -lc "python3 - <<'PY'
import openpyxl, glob, os
for f in sorted(glob.glob('supp/*.xlsx')):
    try:
        wb=openpyxl.load_workbook(f, read_only=True, data_only=True)
        print(os.path.basename(f), wb.sheetnames)
        wb.close()
    except Exception as e:
        print('ERR', f, e)
PY"
