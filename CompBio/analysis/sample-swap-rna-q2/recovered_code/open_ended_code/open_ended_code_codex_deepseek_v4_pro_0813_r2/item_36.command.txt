/bin/bash -lc "python3 - <<'PY'
from bs4 import BeautifulSoup
import glob
markers=[]
for f in ['wheatomics_markers.html']+[f'page_{p}.html' for p in range(2,11)]:
    soup=BeautifulSoup(open(f).read(),'html.parser')
    for table in soup.find_all('table'):
        rows=table.find_all('tr')
        for r in rows[1:]:
            cells=[c.get_text(' ', strip=True) for c in r.find_all(['td','th'])]
            if len(cells)>=6 and cells[4].startswith('Traes'):
                markers.append((cells[4], cells[5], cells[0], cells[1]))
# de-dupe
seen=set(); out=[]
for m in markers:
    if m not in seen:
        seen.add(m); out.append(m)
print('total',len(out))
from collections import Counter
print(Counter(m[1] for m in out))
for m in out:
    print('\\t'.join(m))
PY"
