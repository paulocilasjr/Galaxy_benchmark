/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from bs4 import BeautifulSoup
import glob
markers=[]
for f in ['wheatomics_markers.html']+[f'page_{p}.html' for p in range(2,11)]:
    soup=BeautifulSoup(open(f).read(),'html.parser')
    for table in soup.find_all('table'):
        for r in table.find_all('tr')[1:]:
            c=[x.get_text(' ', strip=True) for x in r.find_all(['td','th'])]
            if len(c)>=6 and c[4].startswith('Traes'):
                markers.append((c[4],c[5]))
seen=set(); mk=[]
for x in markers:
    if x not in seen: seen.add(x); mk.append(x)
print('markers',len(mk))
idx=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0,usecols=[0]).index
mset=set(idx)
for ct in sorted(set(ct for g,ct in mk)):
    genes=[g for g,c in mk if c==ct]
    inter=[g for g in genes if g in mset]
    print(ct, 'total',len(genes),'overlap',len(inter),'missing',len(genes)-len(inter))
    print('missing sample', [g for g in genes if g not in mset][:10])
PY"
