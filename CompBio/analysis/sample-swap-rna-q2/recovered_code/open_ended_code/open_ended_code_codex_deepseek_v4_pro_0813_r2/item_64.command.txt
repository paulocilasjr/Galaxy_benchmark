/bin/bash -lc "curl -sL 'https://zenodo.org/api/records/14524519' -o zenodo1.json; curl -sL 'https://zenodo.org/api/records/14524587' -o zenodo2.json; python3 - <<'PY'
import json
for f in ['zenodo1.json','zenodo2.json']:
    try:
        d=json.load(open(f))
        print(f, d.get('metadata',{}).get('title'))
        for fl in d.get('files',[]):
            print(fl.get('key'), fl.get('size'), fl.get('links',{}).get('self'))
    except Exception as e: print(e)
PY"
