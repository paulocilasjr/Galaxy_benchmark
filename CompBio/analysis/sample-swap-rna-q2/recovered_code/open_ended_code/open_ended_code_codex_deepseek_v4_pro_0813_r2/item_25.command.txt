import openpyxl
for f, sheets in [('supp/mmc10.xlsx',['Table Legend','A. thaliana','T. aestivum','O.sativa','Z.mays','wheat triad DEG information']), ('supp/mmc7.xlsx',['Table Legend','snTae','Ath','Osa','Zma']), ('supp/mmc9.xlsx',['Table Legend'])]:
