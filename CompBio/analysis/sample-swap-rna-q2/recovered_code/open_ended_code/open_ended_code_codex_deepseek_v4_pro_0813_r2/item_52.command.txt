/bin/bash -lc "python3 - <<'PY'
import openpyxl
for f in ['supp/mmc2.xlsx','supp/mmc3.xlsx','supp/mmc4.xlsx','supp/mmc5.xlsx','supp/mmc6.xlsx','supp/mmc8.xlsx','supp/mmc9.xlsx']:
    wb=openpyxl.load_workbook(f, read_only=True, data_only=True)
    print('\\nFILE',f)
    for sn in wb.sheetnames:
        ws=wb[sn]
        print(' SHEET',sn,'dims',ws.max_row,ws.max_column)
        n=0
        for row in ws.iter_rows(values_only=True):
            print([str(c)[:50] if c is not None else None for c in row[:10]])
            n+=1
            if n>=5: break
    wb.close()
PY"
