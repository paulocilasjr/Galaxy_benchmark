/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
cat('start\\n')
obj <- readRDS(gzfile('GSE270342_seuratObj_for_publication.rds.gz'))
cat('class', class(obj), '\\n')
cat('length', length(obj), '\\n')
cat('names', paste(names(obj), collapse=','), '\\n')
cat('slots', paste(slotNames(obj), collapse=','), '\\n')
if (inherits(obj, 'Seurat')) {
  cat('dims', paste(dim(obj), collapse='x'), '\\n')
  md <- obj@meta.data
  cat('meta dims', paste(dim(md), collapse='x'), '\\n')
  cat('meta cols', paste(colnames(md), collapse=' | '), '\\n')
  for (cn in colnames(md)) {
    cat('COL', cn, 'class', class(md[[cn]]), 'nuniq', length(unique(md[[cn]])), '\\n')
    print(head(unique(md[[cn]]), 30))
  }
  # active assay, layers
  print(names(obj@assays))
  for (an in names(obj@assays)) {
    assay <- obj@assays[[an]]
    cat('assay', an, 'class', class(assay), 'slots', paste(slotNames(assay), collapse=','), '\\n')
    print(assay)
  }
}
EOF
