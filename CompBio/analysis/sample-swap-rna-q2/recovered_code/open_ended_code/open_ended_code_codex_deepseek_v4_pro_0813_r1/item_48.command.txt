import requests, json, sys, concurrent.futures, time

def get(gene):
    for attempt in range(3):
        try:
            r=requests.get(f'https://rest.ensembl.org/xrefs/id/{gene}?content-type=application/json',headers={'Content-Type':'application/json'},timeout=20)
            if r.status_code==200:
                return gene, r.json()
            elif r.status_code==400:
                return gene, []
            time.sleep(1)
        except Exception as e:
            time.sleep(2)
    return gene, []
if __name__=='__main__':
    genes=[l.strip() for l in sys.stdin if l.strip()]
    out={}
    with concurrent.futures.ThreadPoolExecutor(max_workers=12) as ex:
        for gene,res in ex.map(get,genes):
            out[gene]=res
    json.dump(out,sys.stdout,indent=2)
