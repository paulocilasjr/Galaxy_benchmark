/bin/bash -lc "python3 - <<'PY'
import re,html
s=open('article.html',encoding='utf-8').read()
text=html.unescape(re.sub(r'<["'^>]+>'"',' ',s)); text=re.sub(r'\\s+',' ',text)
# find all occurrences of 'cluster X' and print nearby
for n in range(0,22):
 print('\\n### cluster',n)
 count=0
 for m in re.finditer(rf'cluster\\s+{n}\\b', text, flags=re.I):
  st=max(0,m.start()-350); en=min(len(text),m.end()+450)
  sn=text[st:en]
  # only print those with likely annotation words
  if re.search(r'cluster|cell|meristem|pericycle|cortex|endodermis|epidermis|xylem|phloem|root|columella|stele|companion|sieve|border|cap|provascul', sn, re.I):
   print(sn)
   count+=1
   if count>=4: break
PY"
