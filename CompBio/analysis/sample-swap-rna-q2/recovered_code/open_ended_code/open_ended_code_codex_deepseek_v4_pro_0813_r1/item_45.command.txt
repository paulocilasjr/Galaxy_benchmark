/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np, math
expr=[]
genes=[]
with gzip.open('sample.swap.rna.q2.tsv.gz','rt') as f:
 r=csv.reader(f,delimiter='\\t'); h=next(r)
 for row in r:
  genes.append(row[0]); expr.append(list(map(float,row[1:])))
A=np.array(expr,dtype=float)
print(A.shape)
cols=h[1:]
# compute specificity ratio / zscore
for ci,col in enumerate(cols):
 vals=A[:,ci]
 # avoid zeros, zscore across genes maybe
 # top raw mean and top ratio col/rowsum
 ratio=vals/(A.sum(axis=1)+1e-9)
 idx=np.argsort(vals)[::-1][:30]
 print('\\n===',col,'top raw mean')
 for i in idx[:20]:
  print(genes[i],round(vals[i],2),'rowsum',round(A[i].sum(),2),'ratio',round(ratio[i],3),'other max',round(np.delete(A[i],ci).max(),2))
PY"
