/bin/bash -lc "python3 - <<'PY'
from PIL import Image, ImageOps, ImageEnhance
im=Image.open('Fig1_HTML.jpg')
# Crop legend left area x700-1220, y350-1420, upscale 4
box=(680,340,1240,1420)
crop=im.crop(box).resize(((box[2]-box[0])*4,(box[3]-box[1])*4), Image.LANCZOS)
crop=crop.convert('L')
crop=ImageEnhance.Contrast(crop).enhance(2)
crop.save('legend_crop.png')
PY
tesseract legend_crop.png stdout --psm 6 2>/dev/null"
