/bin/bash -lc "python3 - <<'PY'
import requests
for key in ['5ed8','10074895','14','56','083a','11860762']:
 for pmcid in ['PMC10074895','PMC11860762']:
  url=f'https://europepmc.org/ftp/suppl/OA/{key}/{pmcid}.zip'
  try:
   r=requests.get(url,timeout=10)
   if r.status_code==200:
    print('FOUND',url,len(r.content),r.headers.get('content-type'))
    open(pmcid+'.zip','wb').write(r.content)
   else:
    print(r.status_code,url)
  except Exception as e: print('err',url,e)
PY"
