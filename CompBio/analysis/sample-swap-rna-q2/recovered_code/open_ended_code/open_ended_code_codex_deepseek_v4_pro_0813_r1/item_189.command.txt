/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'EOF'
obj <- readRDS('GSE270342_seuratObj_for_publication.rds')
md <- obj@meta.data
groups <- list(
 cortex='Cortex',
 endodermis='Endodermis',
 epidermis='Epidermis',
 meristems=c('Meristems','Dividing Cells'),
 pericycle='Pericycle',
 phloem=c('Phloem','Endodermis/Phloem'),
 provascular='Provascular cells',
 root_cap='Root Cap',
 root_hair='Root Hair',
 xylem='Xylem'
)
# RNA counts and data
counts <- GetAssayData(obj, assay='RNA', slot='counts')
data <- GetAssayData(obj, assay='RNA', slot='data')
features <- rownames(counts)
# function sum and mean
calc <- function(mat, fun=c('sum','mean')) {
  out <- matrix(0, nrow=length(features), ncol=length(groups), dimnames=list(features, names(groups)))
  for (j in seq_along(groups)) {
    cells <- rownames(md)[md"'$annotation %in% groups[[j]]]
    cat(names(groups)[j], length(cells), '"'\\n')
    sub <- mat[, cells, drop=FALSE]
    if (fun=='sum') out[,j] <- Matrix::rowSums(sub)
    else out[,j] <- Matrix::rowMeans(sub)
  }
  out
}
sum_counts <- calc(counts, 'sum')
mean_counts <- calc(counts, 'mean')
mean_data <- calc(data, 'mean')
write.table(sum_counts, 'seurat_pseudobulk_sum_counts.tsv', sep='\\t', quote=FALSE, col.names=NA)
write.table(mean_counts, 'seurat_pseudobulk_mean_counts.tsv', sep='\\t', quote=FALSE, col.names=NA)
write.table(mean_data, 'seurat_pseudobulk_mean_data.tsv', sep='\\t', quote=FALSE, col.names=NA)
cat('done\\n')
EOF
Rscript [REDACTED LOCAL PATH]"
