/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
d=pd.read_csv('wheat_root_atlas/conserved_marker_analysis/input/tae_log2FC0.5_FDR0.05.csv')
print('clusters',sorted(d.cluster.unique()), d.groupby('cluster').size().to_dict())
# log expression standardize genes across cols
lx=np.log1p(x)
z=lx.sub(lx.mean(1),axis=0).div(lx.std(1).replace(0,np.nan),axis=0)
for n in [10,20,50,100,200,10000]:
 score={}
 for cl,g in d.sort_values('avg_log2FC',ascending=False).groupby('cluster'):
  genes=[q for q in g['gene ID'].head(n) if q in z.index]
  # weight FC optionally
  score[cl]=z.loc[genes].mean()
 s=pd.DataFrame(score).T
 print('\\nN',n)
 print(s.round(2).to_string())
 print('best observed label per cluster',s.idxmax(1).to_dict())
 print('best cluster per observed label',s.idxmax(0).to_dict())
PY"
