/bin/bash -lc "Rscript - <<'RS'
library(SeuratObject); library(Matrix)
z<-readRDS('gse270342_seurat.rds'); m<-z@assays"'$RNA@data; md<-z@meta.data; g<-factor(md$annotation)
I<-sparse.model.matrix(~0+g); colnames(I)<-levels(g)
a<-as.matrix(m %*% I); a<-sweep(a,2,colSums(I),'"'/');
cat('dims',dim(a),'groups',colnames(a),'\\n')
for (gene in c('TraesCSU03G0001100','TraesCS1D03G0787300','TraesCS5A03G0388600','TraesCS5D03G0377000','TraesCS4D03G0146400')) { if(gene %in% rownames(a)) print(cbind(gene=gene,round(a[gene,],3))) else print(paste('missing',gene)) }
cat('ranges per col\\n'); print(round(apply(a,2,quantile,probs=c(0,.5,.9,.99,.999,1)),3))
RS"
