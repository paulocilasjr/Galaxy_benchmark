/bin/bash -lc "Rscript - <<'RS'
library(SeuratObject)
z<-readRDS('gse270342_seurat.rds')
cat('class:',paste(class(z),collapse=','),' isS4:',isS4(z),'\\n')
for (expr in list(quote(attributes(z)),quote(isS4(z)),quote(slotNames(z)),quote(slot(z,'meta.data')),quote(z@meta.data),quote(unclass(z)))) {
 cat('---',deparse(expr),'---\\n')
 tryCatch({v<-eval(expr); print(if(is.data.frame(v)) dim(v) else if(is.list(v)) names(v)[1:min(10,length(v))] else v)},error=function(e) cat('ERR',conditionMessage(e),'\\n'))
}
RS"
