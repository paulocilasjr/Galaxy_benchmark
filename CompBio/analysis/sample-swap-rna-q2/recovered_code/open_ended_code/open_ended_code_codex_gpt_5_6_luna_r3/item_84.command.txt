/bin/bash -lc "Rscript - <<'RS'
library(SeuratObject)
z<-readRDS('gse270342_seurat.rds')
md<-z@meta.data
cat('metadata columns:\\n'); print(names(md)); print(head(md));
for(n in names(md)) { if(is.character(md[[n]]) || is.factor(md[[n]])) { cat('\\n',n,'\\n'); print(table(md[[n]],useNA='ifany')) } }
cat('\\nassay names:',names(z@assays),'\\n')
r<-z@assays"'$RNA
cat('"'RNA class',class(r),'slots',slotNames(r),'\\n')
for(n in slotNames(r)) {v<-slot(r,n); cat(n, class(v), if(is.matrix(v)||inherits(v,'Matrix')) paste(dim(v),collapse='x') else if(is.data.frame(v)) paste(dim(v),collapse='x') else '', '\\n')}
RS"
