/bin/bash -lc "PYTHONPATH=./pydeps python - <<'PY'
import pandas as pd,numpy as np
x=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0)
d=pd.read_excel('mmc5.xlsx',sheet_name='Table S4',engine='openpyxl')
L=np.log1p(x); Z=L.sub(L.mean(axis=1),axis=0).div(L.std(axis=1).replace(0,np.nan),axis=0)
out=[]
for lab,gp in d.groupby('cluster',sort=False):
 gp=gp.sort_values('avg_log2FC',ascending=False)
 for n in [20,50,100,200,99999]:
  gs=gp.head(n).Gene_ID;gs=gs[gs.isin(x.index)].unique()
  row=Z.loc[gs].mean(); row['cluster']=lab;row['topn']=n;row['ngenes']=len(gs);out.append(row)
pd.DataFrame(out).set_index(['cluster','topn','ngenes']).to_csv('atlas_marker_scores.tsv',sep='\\t')
PY
column -t -s "'$'"'\\t' atlas_marker_scores.tsv"
