import urllib.request, json
ids=['TraesCS1D02G011200','TraesCS6D02G370100','TraesCS2D02G349900']

for endpoint in ['https://rest.ensembl.org/lookup/id','https://rest.ensembl.org/xrefs/id']:
 req=urllib.request.Request(endpoint, data=body.encode(), headers={'Content-Type':'application/json','Accept':'application/json'})
 try:
  r=urllib.request.urlopen(req); print(endpoint,r.status,r.read().decode()[:1000])
 except Exception as e: print(endpoint,'ERR',e)
PY"
