/bin/bash -lc "python3 - <<'PY'
import openpyxl, gzip, pandas as pd
import numpy as np
wb=openpyxl.load_workbook('refs/13059_2023_2908_MOESM2_ESM.xlsx', read_only=True, data_only=True)
ws=wb['Table S2']
markers={}
for row in ws.iter_rows(values_only=True):
    cluster,gene,p,hom,ann=row[:5]
    if cluster is None or gene is None or not str(cluster).isdigit(): continue
    markers.setdefault(str(cluster), []).append((gene,ann))
wb.close()
print({k:len(v) for k,v in sorted(markers.items(), key=lambda kv:int(kv[0]))})
df=pd.read_csv('sample.swap.rna.q2.tsv.gz', sep='\\t', index_col=0)
mapping={
'cortex':['12'], # likely
'endodermis':['10','17'],
'epidermis':['0'],
'meristems':['2','7','3','18'],
'pericycle':['1','4','9'],
'phloem':['14','16','21'],
'provascular':['5'],
'root_cap':['15','20'],
'root_hair':['6','8'],
'xylem':['11','13'],
}
for ct, clusters in mapping.items():
    genes=[]
    for c in clusters:
        genes += [g for g,a in markers.get(c,[])]
    present=[g for g in genes if g in df.index]
    if not present: continue
    mat=df.loc[present]
    means=mat.mean(axis=0)
    print('\\n', ct, 'clusters', clusters, 'markers present', len(present), '/', len(genes))
    print('mean expr per col:')
    print(means.round(2).to_dict())
    top=means.nlargest(5)
    print('top columns', list(top.index), top.round(2).tolist())
PY"
