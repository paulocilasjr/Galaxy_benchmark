/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np,re,itertools
df=pd.read_csv('sample.swap.rna.q2.tsv.gz',sep='\\t',index_col=0); cols=df.columns.tolist()
stab=pd.read_csv('refs/wheat_root_atlas/GRN_regulon_analysis/input/stability_and_specificity_info.csv',sep=';')
ms=stab[(stab['Specificity']=='specific')&(stab['Stability']=='stable')&stab['is_Marker_of_Cluster'].notna()].copy()
def calc(map5,map11):
 exact_map={'1Epidermis':'epidermis','2Cortex':'cortex','3Meristems':'meristems','4ProvascularCells':'provascular','5Unknown':map5,'6Pericycle':'pericycle','7Epidermis':'epidermis','8RootHair':'root_hair','9Endodermis':'endodermis','10Xylem':'xylem','11Endodermis/Phloem':map11,'12DividingCells':'meristems','13Phloem':'phloem','14RootCap':'root_cap'}
 sets={ct:set() for ct in cols}
 for lab,bt in exact_map.items(): sets[bt].update(ms.loc[ms['is_Marker_of_Cluster'].str.contains(re.escape(lab),na=False),'Gene'].tolist())
 pres={ct:[g for g in genes if g in df.index] for ct,genes in sets.items()}
 def sc(c,gs):
  m=df.loc[gs]; mx=m.max(axis=1); p=m.div(mx.where(mx>0),axis=0).fillna(0); return p[c].mean()
 obs={ct:sc(ct,pres[ct]) for ct in cols}
 return obs,pres
for map5 in ['root_cap','meristems']:
 for map11 in ['root_cap','phloem']:
  obs,pres=calc(map5,map11)
  for pair in [('meristems','root_cap'),('meristems','root_hair'),('phloem','root_hair'),('cortex','root_hair')]:
   a,b=pair; loss=(obs[a]-sc(b,pres[a]))+(obs[b]-sc(a,pres[b]))
   print(map5,map11,pair,'loss',round(loss,3),'obs',round(obs[a],3),round(obs[b],3))
PY"
