/bin/bash -lc "python3 - <<'PY'
import openpyxl, urllib.request, json, time, collections
wb=openpyxl.load_workbook('refs/13059_2023_2908_MOESM2_ESM.xlsx', read_only=True, data_only=True)
ws=wb['Table S2']
genes=[]
for row in ws.iter_rows(values_only=True):
    c,g,p,h,a=row[:5]
    if c is not None and str(c).isdigit() and g:
        genes.append(str(g))
unique=list(dict.fromkeys(genes))
print('unique',len(unique))
# test 10
res={}
for g in unique[:10]:
    url=f'https://rest.ensembl.org/xrefs/id/{urllib.parse.quote(g)}?content-type=application/json'
    try:
        r=urllib.request.urlopen(url, timeout=10)
        data=json.loads(r.read().decode())
        entrez=[x for x in data if x.get('dbname')=='EntrezGene']
        res[g]=[x.get('primary_id') for x in entrez]
    except Exception as e:
        res[g]=['ERR '+str(e)]
    time.sleep(0.05)
print(res)
PY"
