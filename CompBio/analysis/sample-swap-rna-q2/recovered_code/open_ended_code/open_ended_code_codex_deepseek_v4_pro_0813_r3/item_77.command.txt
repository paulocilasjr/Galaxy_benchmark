import pandas as pd, numpy as np, re
stab=pd.read_csv('refs/wheat_root_atlas/GRN_regulon_analysis/input/stability_and_specificity_info.csv', sep=';')
ms=stab[(stab['Specificity']=='specific') & (stab['Stability']=='stable') & stab['is_Marker_of_Cluster'].notna()].copy()
df=pd.read_csv('sample.swap.rna.q2.tsv.gz', sep='\\t', index_col=0)
exact_map={
'1Epidermis':'epidermis',
'2Cortex':'cortex',
'3Meristems':'meristems',
'4ProvascularCells':'provascular',
'5Unknown':'root_cap',
'6Pericycle':'pericycle',
'7Epidermis':'epidermis',
'8RootHair':'root_hair',
'9Endodermis':'endodermis',
'10Xylem':'xylem',
'11Endodermis/Phloem':'root_cap',
'12DividingCells':'meristems',
'13Phloem':'phloem',
'14RootCap':'root_cap',
}
sets={ct:set() for ct in ['cortex','endodermis','epidermis','meristems','pericycle','phloem','provascular','root_cap','root_hair','xylem']}
for lab,bt in exact_map.items():
    genes=ms.loc[ms['is_Marker_of_Cluster'].str.contains(re.escape(lab), na=False),'Gene'].tolist()
    sets[bt].update(genes)
for ct in sets:
    present=[g for g in sets[ct] if g in df.index]
    print('\\n',ct,len(sets[ct]),'present',len(present))
    mat=df.loc[present]
    prop=(mat.div(mat.max(axis=1),axis=0)).mean(axis=0)
    print(prop.round(2).to_dict())
    print('top3',prop.nlargest(3).index.tolist())
