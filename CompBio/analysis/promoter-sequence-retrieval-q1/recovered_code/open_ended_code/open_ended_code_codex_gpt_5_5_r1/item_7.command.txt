/bin/bash -lc "python3 - <<'PY'
import json, re
j=json.load(open('shha_lookup.json'))
t=[x for x in j['Transcript'] if x.get('is_canonical')][0]
tss=t['start'] if t['strand']==1 else t['end']
start=tss-500 if t['strand']==1 else tss-100
end=tss+100 if t['strand']==1 else tss+500
seq=''.join(line.strip() for line in open('shha_promoter.fa') if not line.startswith('>'))
print(j['assembly_name'], j['seq_region_name'], j['strand'], t['id'], tss, start, end, len(seq), bool(re.fullmatch('[ACGTN]+', seq)))
print(seq)
PY"
