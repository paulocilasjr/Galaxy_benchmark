/bin/bash -lc "python3 - <<'PY'
import gzip,collections,csv
assign={}
with open('results/barcode_species.tsv') as f:
 for d in csv.DictReader(f,delimiter='\\t'):
  counts=[int(d[s]) for s in ['human','mouse','pig']]
  if max(counts)/max(1,sum(counts))>=.8:assign[d['barcode']]=['human','mouse','pig'][counts.index(max(counts))]
a=collections.Counter();cells=collections.Counter()
with gzip.open('results/cell_gene_umis.tsv.gz','rt') as f:
 for d in csv.DictReader(f,delimiter='\\t'):
  if assign.get(d['barcode'])==d['species']:
   a[d['species'],d['gene']]+=int(d['umis']);cells[d['species'],d['gene']]+=1
for sp in ['human','mouse','pig']:
 print(sp,[(g,n,cells[s,g]) for (s,g),n in a.most_common() if s==sp][:50])
PY"
