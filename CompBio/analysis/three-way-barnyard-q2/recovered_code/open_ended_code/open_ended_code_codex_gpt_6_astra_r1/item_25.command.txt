/bin/bash -lc "python3 - <<'PY'
import numpy as np,pandas as pd
z=np.load('results/species_counts.npz');a=z['counts'];s=z['strict'];b=z['barcodes'];f=a/np.maximum(1,a.sum(1))[:,None]
for i,sp in enumerate(['human','mouse','pig']):
 inds=np.where(a.argmax(1)==i)[0]
 print(sp,len(inds),'fractions',np.percentile(f[inds,i],[0,10,25,50,75,90,100]))
 print('bins',np.histogram(f[inds,i],bins=[0,.5,.6,.7,.8,.9,.95,.99,1.001])[0])
 for j in inds[np.argsort(f[inds,i])][:15]:print(b[j],a[j],s[j],np.round(f[j],3))
print('20 human top fraction cells')
for j in np.argsort(-f[:,0])[:20]:print(b[j],a[j],s[j],np.round(f[j],3))
PY"
