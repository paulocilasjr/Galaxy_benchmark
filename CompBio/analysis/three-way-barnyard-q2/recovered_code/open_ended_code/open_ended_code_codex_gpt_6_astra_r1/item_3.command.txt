/bin/bash -lc "python3 - <<'PY'
import gzip,collections
for fn in ['three.way.barnyard.q2.R1.fq.gz','three.way.barnyard.q2.R2.fq.gz']:
 c=collections.Counter(); b=collections.Counter(); n=0
 with gzip.open(fn,'rt') as f:
  for h in f:
   s=next(f).strip(); next(f);next(f);n+=1;c[len(s)]+=1
   if 'R1' in fn:b[s[:16]]+=1
   if n<=3: print(fn,h.strip(),s)
 print('reads',n,'lengths',c,'barcodes',len(b),'top',b.most_common(10))
 if b:
  with open('barcode_counts.tsv','w') as o:
   for k,v in b.most_common():o.write(f'{k}\\t{v}\\n')
PY"
