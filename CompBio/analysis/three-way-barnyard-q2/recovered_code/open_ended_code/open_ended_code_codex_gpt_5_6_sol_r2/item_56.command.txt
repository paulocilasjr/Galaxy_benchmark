/bin/bash -lc "python - <<'PY'
import pandas as pd
from sklearn.mixture import GaussianMixture
from sklearn.cluster import KMeans
import numpy as np
files={s:f'{s}.bc.ratio.tsv' for s in ['human','mouse','pig']}
d=None
for s,f in files.items():
 x=pd.read_csv(f,sep='\\t',header=None,names=['bc','reads',s+'_n',s])
 x=x[['bc',s+'_n',s]]
 d=x if d is None else d.merge(x,on='bc',how='outer')
d=d.fillna(0)
for features in [['human','mouse','pig'],['human_n','mouse_n','pig_n']]:
 X=d[features].to_numpy()
 if features[0].endswith('_n'): X=np.log1p(X/d[['human_n','mouse_n','pig_n']].sum(axis=1).to_numpy()[:,None]*100)
 for model in [KMeans(n_clusters=3,random_state=1,n_init=50),GaussianMixture(n_components=3,random_state=1,n_init=20)]:
  z=model.fit_predict(X)
  print(features,type(model).__name__,np.bincount(z))
  print(pd.DataFrame(d.groupby(z)[['human','mouse','pig']].mean()).round(3))
  print('mins/max labels')
print(d.sort_values('human',ascending=False).head(20).to_string(index=False))
PY"
