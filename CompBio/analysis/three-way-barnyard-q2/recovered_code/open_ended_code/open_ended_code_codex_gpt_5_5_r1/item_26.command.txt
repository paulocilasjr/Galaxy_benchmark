import gzip,re,os
refs=[('human','refs/human.cdna.fa.gz'),('mouse','refs/mouse.cdna.fa.gz'),('pig','refs/pig.cdna.fa.gz')]
with open('refs/combined.cdna.fa','w') as out, open('refs/transcript_gene.tsv','w') as mapout:
    mapout.write('species\\ttranscript\\tgene\\n')
    for sp,fn in refs:
        with gzip.open(fn,'rt') as f:
            for line in f:
                if line.startswith('>'):
                    header=line[1:].strip()
                    tid=header.split()[0]
                    m=re.search(r'gene_symbol:(["'^'"\\s]+)',header)
                    gene=m.group(1) if m else ''
                    out.write(f'>{sp}|{tid}|{gene}\\n')
                    mapout.write(f'{sp}\\t{tid}\\t{gene}\\n')
                else:
                    out.write(line)
print('done')
PY
