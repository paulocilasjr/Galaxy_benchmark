from collections import Counter, defaultdict
total=Counter()
with open('work/read_barcodes.tsv') as f:
    for l in f:
        rid,b=l.rstrip().split('\\t'); total[b]+=1

    for l in f:
        b,g,c=l.rstrip().split('\\t'); m[b][g]+=int(c)
rows=[]
for b,genes in m.items():
    s=sum(genes.values()); rows.append((s,b,total[b],s/total[b]))
print('barcodes with mouse reads',len(m))
for r in sorted(rows,reverse=True)[:25]: print(r[1],'mouse',r[0],'total',r[2],'frac',round(r[3],3))
for thresh in [0.6,0.7,0.8]:
    sel=[b for b,genes in m.items() if sum(genes.values())/total[b]>=thresh]
    agg=Counter()
    for b in sel: agg.update(m[b])
    print('\\nthresh',thresh,'n',len(sel),'reads',sum(agg.values()))
