/bin/bash -lc "python3 - <<'PY'
import gzip
with gzip.open('refs/pig.cdna.fa.gz','rt') as fi, open('work/pig.cdna.fa','w') as fo:
    for line in fi:
        if line.startswith('>'):
            parts=line.strip().split()
            tid=parts[0][1:]
            gene=''
            for p in parts:
                if p.startswith('gene_symbol:'):
                    gene=p.split(':',1)[1]
                    break
            if not gene:
                for p in parts:
                    if p.startswith('gene:'):
                        gene=p.split(':',1)[1]
                        break
            fo.write(f'>{gene}|{tid}\\n')
        else:
            fo.write(line)
print('done')
PY
ls -lh work/pig.cdna.fa; head -1 work/pig.cdna.fa"
