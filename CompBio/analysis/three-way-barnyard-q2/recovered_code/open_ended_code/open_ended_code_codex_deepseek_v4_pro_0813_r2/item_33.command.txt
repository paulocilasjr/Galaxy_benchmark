/bin/bash -lc "python3 - <<'PY'
import gzip
with gzip.open('three.way.barnyard.q2.R1.fq.gz','rt') as fi, open('work/read_barcodes.tsv','w') as fo:
    n=0
    for line in fi:
        n+=1
        if n%4==1:
            rid=line.strip().split()[0]
        elif n%4==2:
            bc=line.strip()[:16]
            fo.write(f'{rid}\\t{bc}\\n')
print(n//4)
PY
head -3 work/read_barcodes.tsv; tail -3 work/read_barcodes.tsv"
