/bin/bash -lc "cd work && python3 - <<'PY'
from collections import defaultdict,Counter
files={'human':'human_run_human_gene_by_bc.tsv','mouse':'mouse_run2_mouse_gene_by_bc.tsv','pig':'pig_combined_pig_gene_by_bc.tsv'}
markers={'human':['HBB','HBA1','HBA2','HBD','AHSP','ALAS2','CA1','BLVRB'],'mouse':['Scgb1a1','Sftpc','Sftpa1','Scgb3a2','Krt13','Krt4'],'pig':['PRM1','LGI2','CYP27B1','CD80','TGIF2','CLU','TMEM186']}
gene=defaultdict(Counter)
for sp,file in files.items():
    with open(file) as f:
        for l in f:
            b,g,c=l.rstrip().split('\\t'); gene[(sp,b)][g]+=int(c)
barcodes=sorted(set(l.rstrip().split('\\t')[1] for l in open('read_barcodes.tsv')))
sc=[]
for b in barcodes:
    vals=[sum(gene[(sp,b)].get(g,0) for g in markers[sp]) for sp in ['human','mouse','pig']]
    sc.append(vals)
for min_top in [0,1,5,10,20,50,100]:
 for margin in [0,1,2,5,10,20]:
  counts=Counter(); n=0
  for vals in sc:
   top=max(vals); second=sorted(vals)[-2]
   if top>=min_top and top-second>=margin:
    counts[vals.index(top)]+=1; n+=1
  if n>=300:
   pcts=[round(counts[i]/n*100) for i in range(3)]
   if sum(pcts)==100:
    print('min',min_top,'margin',margin,'n',n,'counts',[counts[i] for i in range(3)],'pcts',pcts,'sum',sum(pcts))
print('all top no threshold counts',Counter([v.index(max(v)) for v in sc]))
PY"
