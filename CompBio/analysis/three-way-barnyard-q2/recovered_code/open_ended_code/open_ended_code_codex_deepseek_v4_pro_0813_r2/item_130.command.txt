/bin/bash -lc "cd work && python3 - <<'PY'
from collections import defaultdict,Counter
# load gene counts per species
files={'human':'human_run_human_gene_by_bc.tsv','mouse':'mouse_run2_mouse_gene_by_bc.tsv','pig':'pig_combined_pig_gene_by_bc.tsv'}
markers={
 'human':['HBB','HBA1','HBA2','HBD','AHSP','ALAS2','CA1','BLVRB'],
 'mouse':['Scgb1a1','Sftpc','Sftpa1','Scgb3a2','Krt13','Krt4'],
 'pig':['PRM1','LGI2','CYP27B1','CD80','TGIF2','CLU','TMEM186'],
}
gene=defaultdict(Counter)
for sp,file in files.items():
    with open(file) as f:
        for l in f:
            b,g,c=l.rstrip().split('\\t'); gene[(sp,b)][g]+=int(c)
# score each barcode
barcodes=set()
with open('read_barcodes.tsv') as f:
    for l in f: barcodes.add(l.rstrip().split('\\t')[1])
scores=defaultdict(Counter)
for b in barcodes:
    for sp in ['human','mouse','pig']:
        sc=sum(gene[(sp,b)].get(g,0) for g in markers[sp])
        scores[b][sp]=sc
# classify top species if min score>=10 and margin
assign={}
for b,sc in scores.items():
    arr=sorted(sc.items(), key=lambda x:-x[1])
    top,topc=arr[0]; second=arr[1][1]
    if topc>=20 and topc-second>=5:
        assign[b]=top
print('assigned',len(assign),Counter(assign.values()))
# cross-tab top genes for assigned groups
for sp in ['human','mouse','pig']:
    sel=[b for b,x in assign.items() if x==sp]
    agg=Counter()
    for b in sel:
        for g,c in gene[(sp,b)].items(): agg[g]+=c
    print('\\n',sp,'n',len(sel),'top genes:')
    print(agg.most_common(30))
PY"
