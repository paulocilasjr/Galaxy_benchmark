/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import Counter
c=Counter(); n=0
with gzip.open('three.way.barnyard.q2.R1.fq.gz','rt') as f:
    for line in f:
        n+=1
        if n%4==2:
            c[line.strip()[:16]]+=1
print('reads',n//4,'unique barcodes',len(c))
for lim in [1,10,50,100,200,500,1000]:
    print(lim, sum(1 for v in c.values() if v>=lim))
print(c.most_common(20))
PY"
