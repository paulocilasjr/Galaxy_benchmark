/bin/bash -lc "cd work && python3 - <<'PY'
import numpy as np
from collections import Counter
bc_list=[]
with open('read_barcodes.tsv') as f:
    for l in f: bc_list.append(l.rstrip().split('\\t')[1])
barcodes=sorted(set(bc_list)); bidx={b:i for i,b in enumerate(barcodes)}; ridx=np.array([bidx[b] for b in bc_list],dtype=np.int32)
h=np.load('human_scores.npy');m=np.load('mouse_scores.npy');p=np.load('pig_scores.npy')
S=np.vstack([h,m,p]).T
sums=np.zeros((len(barcodes),3)); np.add.at(sums,ridx,S)
arr=np.sort(sums,axis=1); best=arr[:,2]; second=arr[:,1]; top=sums.argmax(axis=1)
rel=(best-second)/np.maximum(best,1)
print('rel margin distribution',np.percentile(rel,[0,5,10,25,50,75,90,95,100]))
for t in [0,0.01,0.02,0.05,0.1,0.2]:
    conf=rel>=t; c=Counter(top[conf]); print(t,'n',conf.sum(),[c[i] for i in range(3)])
# print ambiguous barcodes (lowest rel)
idx=np.argsort(rel)[:30]
for i in idx:
    print(i,barcodes[i],sums[i],rel[i],top[i])
PY"
