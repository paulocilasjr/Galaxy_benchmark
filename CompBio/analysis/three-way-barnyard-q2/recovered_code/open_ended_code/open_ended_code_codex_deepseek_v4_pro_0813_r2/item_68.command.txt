/bin/bash -lc "python3 - <<'PY'
from collections import Counter, defaultdict
# total reads per bc
total=Counter()
with open('work/read_barcodes.tsv') as f:
    for l in f:
        rid,b=l.rstrip().split('\\t'); total[b]+=1
pig=defaultdict(Counter)
with open('work/pig_run_pig_gene_by_bc.tsv') as f:
    for l in f:
        b,g,c=l.rstrip().split('\\t'); pig[b][g]+=int(c)
print('barcodes with pig reads',len(pig),'total barcodes',len(total))
# top by pig read count
rows=[]
for b,genes in pig.items():
    s=sum(genes.values()); rows.append((s,b,total[b],s/total[b]))
for r in sorted(rows, reverse=True)[:30]: print(r[1], 'pig',r[0], 'total',r[2], 'frac',round(r[3],3))
# choose barcodes with >=90% pig? print count, aggregate genes
for thresh in [0.8,0.9,0.95,0.99]:
    sel=[b for b,genes in pig.items() if sum(genes.values())/total[b]>=thresh]
    agg=Counter()
    for b in sel:
        agg.update(pig[b])
    print('\\nthresh',thresh,'n',len(sel),'totalpigreads',sum(agg.values()))
    print(agg.most_common(40))
PY"
