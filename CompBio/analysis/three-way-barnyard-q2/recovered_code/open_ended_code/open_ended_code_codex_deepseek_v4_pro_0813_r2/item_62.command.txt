/bin/bash -lc "cd work && bwa mem -t 8 pig.cdna.fa R2.fq | samtools view -F 2304 - | python3 - <<'PY'
import sys, collections
# load barcodes
bc={}
with open('read_barcodes.tsv') as f:
    for l in f:
        rid,b=l.rstrip().split('\\t')
        rid=rid[1:].split('/')[0]
        bc[rid]=b
gene_by_bc=collections.defaultdict(collections.Counter)
species_gene=collections.Counter()
n=0
for line in sys.stdin:
    p=line.split('\\t')
    if len(p)<12: continue
    q=p[0]; flag=int(p[1]); rname=p[2]
    if flag & 4: continue
    gene=rname.split('|')[0]
    b=bc.get(q)
    if b:
        gene_by_bc[b][gene]+=1
        species_gene[gene]+=1
    n+=1
print('pig mapped primary reads',n, file=sys.stderr)
print('top pig genes overall:')
for g,c in species_gene.most_common(80):
    print(g,c)
PY"
