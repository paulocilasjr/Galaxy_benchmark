/bin/bash -lc "python3 - <<'PY'
import gzip,collections
# Collect primary species-specific gene counts per barcode from combined mapping.
g=collections.defaultdict(collections.Counter)
with open('gene_counts.tsv') as f:
 next(f)
 for line in f:
  bc,sp,gene,n=line.rstrip().split('\\t');
  if sp=='ss': g[bc][gene]+=int(n)
# classify pig labels by CD80/testis/other marker sums
sets={
'CD80':['CD80'],
'testis':['PRM1','PRM2','TNP1','TNP2','SPO11','DDX4','HSPA4L','INSL3','TDRD1','AKAP4','ODF1'],
'spleen':['HLA-DRA','CD74','CD3D','CD3E','PTPRC','MS4A1','CD79A','CD80','TYROBP','LST1'],
'liver':['APOA1','APOA2','ALB','TTR','HP','FGB'],
}
for name,ms in sets.items():
 print('\\n',name)
 vals=[]
 for bc,c in g.items():
  vals.append((sum(c[m] for m in ms),bc,[(m,c[m]) for m in ms if c[m]]))
 for x in sorted(vals,reverse=True)[:20]: print(x)
 print('nonzero',sum(v[0]>0 for v in vals),'total',sum(v[0] for v in vals))
# cross tabs thresholds
print('\\ncrosstab')
for cd in (0,1,5,10,20):
 for te in (0,1,2,5,10):
  n=sum(c['CD80']>cd and sum(c[m] for m in sets['testis'])>te for c in g.values())
  print(cd,te,n,end='; ')
 print()
PY"
