import gzip,collections
wanted={'CD80','PRM1','TNP1','TNP2','PRM2','SPO11','DDX4','HLA-DRA','CD74','APOA1'}
idx_to_key={}
with gzip.open('read_scores.tsv.gz','rt') as f:
 next(f)
 for line in f:
  x=line.rstrip().split('\\t'); idx=int(x[0]); sp=x[5]; g=x[6]
  if sp=='ss' and g in wanted: idx_to_key[idx]=(x[1],g)
umis=collections.defaultdict(set); seqs=collections.defaultdict(collections.Counter); n=0
with gzip.open('three.way.barnyard.q2.R1.fq.gz','rt') as f:
 i=0
 while True:
  if not f.readline(): break
  s=f.readline().strip(); f.readline(); f.readline(); i+=1
  if i in idx_to_key:
   key=idx_to_key[i]; umis[key].add(s[16:28]); seqs[key][s]+=1
for g in sorted(wanted):
 keylist=[k for k in umis if k[1]==g]
 print(g,'reads',sum(seqs[k].total() for k in keylist),'unique_umi',sum(len(umis[k]) for k in keylist),'unique_seq',sum(len(seqs[k]) for k in keylist),'top_seq_dup',max((max(seqs[k].values()) for k in keylist),default=0))
PY"
