/bin/bash -lc "python3 - <<'PY'
import gzip,collections
cell=collections.defaultdict(collections.Counter)
with gzip.open('read_scores.tsv.gz','rt') as f:
 next(f)
 for line in f:
  x=line.rstrip().split('\\t'); bc=x[1]; vals={'hs':int(x[2]),'mm':int(x[3]),'ss':int(x[4])}; av=sorted(((v,s) for s,v in vals.items() if v>=0),reverse=True)
  if not av: continue
  top,sp=av[0]; sec=av[1][0] if len(av)>1 else -1
  if top>sec: cell[bc][sp]+=1
for bc,c in cell.items():
 tot=sum(c.values()); sp,n=max(c.items(),key=lambda z:z[1]); frac=n/tot
 if frac<.8:
  print(bc,dict(c),'total',tot,'frac',round(frac,3))
PY"
