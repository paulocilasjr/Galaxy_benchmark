/bin/bash -lc "python3 - <<'PY'
import gzip, collections, math
p='read_scores.tsv.gz'
S=[]
per=collections.defaultdict(lambda:collections.Counter())
diffs=collections.Counter()
rows=0
with gzip.open(p,'rt') as f:
    next(f)
    for line in f:
        q,bc,*x=line.rstrip().split('\\t')
        hs,mm,ss=map(int,x[:3])
        vals={'hs':hs,'mm':mm,'ss':ss}
        avail=sorted(((v,k) for k,v in vals.items() if v>=0), reverse=True)
        if not avail: continue
        rows+=1
        top,sp=avail[0]
        second=avail[1][0] if len(avail)>1 else -1
        d=top-second
        diffs[d if d<50 else 50]+=1
        for th in (0,2,4,6,10,20,30):
            if d>th: per[th][(bc,sp)]+=1
print('rows',rows)
print('diffs',sorted(diffs.items())[:40], '>=50',diffs[50])
for th in (0,2,4,6,10,20,30):
    c=collections.Counter()
    for (bc,sp),n in per[th].items(): c[sp]+=n
    print('th',th,'read_assign',c)
    cells=collections.defaultdict(lambda:collections.Counter())
    for (bc,sp),n in per[th].items(): cells[bc][sp]=n
    for rule in ('major','frac80','frac90','frac95'):
        cc=collections.Counter(); conf=0
        for bc,x in cells.items():
            total=sum(x.values())
            if not total: continue
            a,sp=max((n,s) for s,n in x.items())
            frac=a/total
            ok={'major':True,'frac80':frac>=.8,'frac90':frac>=.9,'frac95':frac>=.95}[rule]
            if ok: cc[sp]+=1; conf+=1
        print(' ',rule,conf,dict(cc))
PY"
