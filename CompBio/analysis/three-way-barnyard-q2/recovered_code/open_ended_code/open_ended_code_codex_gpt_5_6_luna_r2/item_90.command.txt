/bin/bash -lc "python3 - <<'PY'
import collections,gzip,re
# robust pig labels
cell=collections.defaultdict(collections.Counter)
with gzip.open('read_scores.tsv.gz','rt') as f:
 next(f)
 for line in f:
  x=line.rstrip().split('\\t'); bc=x[1]; v=list(map(int,x[2:5])); a=sorted(v,reverse=True); sp=['hs','mm','ss'][v.index(a[0])]
  if a[0]>a[1]:cell[bc][sp]+=1
labels={b:max(c,key=c.get) for b,c in cell.items() if max(c.values())/sum(c.values())>=.8}
agg=collections.Counter();
with open('gene_counts.tsv') as f:
 next(f)
 for line in f:
  bc,sp,g,n=line.rstrip().split('\\t');
  if labels.get(bc)=='ss' and sp=='ss':agg[g]+=int(n)
# filters
pat=re.compile(r'"'^(RPL|RPS|MRPL|MRPS|MT-|ND|COX|ATP6$|CYT|HSP|HNRN|EEF|EIF|UB|UBA|FAU$|PFDN|PSM|CCT|NUP|SEC|ST13$|TPT1$|CALM|ACTG|ACTB$|GAPDH$|FTH|FTL|HMGB|SNR|RRP|POLR|PRPF|CST3$|PTMA$|YBX|KLC|KRT18$|KRT8$)'"')
arr=[(n,g) for g,n in agg.items() if g"'!='"'NA' and not pat.match(g)]
for n,g in sorted(arr,reverse=True)[:250]: print(f'{g}\\t{n}')
PY"
