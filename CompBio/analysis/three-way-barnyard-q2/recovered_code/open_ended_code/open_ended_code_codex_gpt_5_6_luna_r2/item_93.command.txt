/bin/bash -lc "python3 - <<'PY'
import gzip,collections
cell=collections.defaultdict(collections.Counter)
with gzip.open('read_scores.tsv.gz','rt') as f:
 next(f)
 for l in f:
  x=l.rstrip().split('\\t'); v=list(map(int,x[2:5])); a=sorted(v,reverse=True); 
  if a[0]>a[1]: cell[x[1]][['hs','mm','ss'][v.index(a[0])]]+=1
for th in [0.5,0.6,0.65,0.7,0.75,0.8,0.85,0.9,0.95,0.99]:
 c=collections.Counter()
 for bc,z in cell.items():
  sp,n=max(z.items(),key=lambda q:q[1]);
  if n/sum(z.values())>=th:c[sp]+=1
 print(th,sum(c.values()),dict(c),{s:round(c[s]/sum(c.values())*100) for s in ('hs','mm','ss')})
PY"
