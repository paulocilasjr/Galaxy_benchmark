/bin/bash -lc "python3 - <<'PY'
import gzip, collections, re
rows=[]
cell=collections.defaultdict(lambda:collections.Counter())
with gzip.open('read_scores.tsv.gz','rt') as f:
    next(f)
    for line in f:
        q,bc,hs,mm,ss,ps,gene,pas,pmapq=line.rstrip().split('\\t')
        vals={'hs':int(hs),'mm':int(mm),'ss':int(ss)}
        av=sorted(((v,s) for s,v in vals.items() if v>=0),reverse=True)
        if not av: continue
        top,sp=av[0]; second=av[1][0] if len(av)>1 else -1
        if top>second: cell[bc][sp]+=1
        rows.append((bc,sp,top-second if len(av)>1 else 999,ps,gene))
labels={}
for bc,c in cell.items():
    sp,n=max(c.items(),key=lambda x:x[1]); tot=sum(c.values())
    labels[bc]=(sp,n/tot,tot)
print('label counts all cells',collections.Counter(x[0] for x in labels.values()))
print('confidence counts',collections.Counter(x[0] for x in labels.values() if x[1]>=.8))
for sp in ('hs','mm','ss'):
    print('\\nSPECIES',sp)
    genes=collections.Counter(); cells=collections.defaultdict(set); reads=0
    for bc,topsp,diff,ps,gene in rows:
        lab=labels.get(bc)
        if not lab or lab[1]<.8 or lab[0]"'!=sp or topsp!=sp or diff<=2: continue
        if ps!=sp or gene=='"'NA': continue
        genes[gene]+=1; cells[gene].add(bc); reads+=1
    print('reads',reads,'genes',len(genes))
    for g,n in genes.most_common(80):
        print(g,n,len(cells[g]))
PY"
