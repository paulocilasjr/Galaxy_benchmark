/bin/bash -c "python3 - <<'PY'
import gzip,collections,statistics
cell_umi_sp=collections.defaultdict(lambda: collections.defaultdict(set))
with gzip.open('read_unique_assign.tsv.gz','rt') as f:
 for line in f:
  p=line.rstrip().split('\\t'); bc,u,sp=p[0],p[1],p[2]; cell_umi_sp[bc][u].add(sp)
# UMI species counts per cell
cell_sp=collections.defaultdict(collections.Counter)
for bc,d in cell_umi_sp.items():
 for u,spp in d.items():
  if len(spp)==1: cell_sp[bc][next(iter(spp))]+=1
for th in [0.5,0.7,0.8,0.9]:
 ass={}
 for bc,c in cell_sp.items():
  tot=sum(c.values()); sp,n=c.most_common(1)[0]
  if tot and n/tot>=th: ass[bc]=sp
 print('\\nth',th)
 for sp in ['hs','mm','ss']:
  cells=[bc for bc,s in ass.items() if s==sp]
  tots=[sum(cell_sp[bc].values()) for bc in cells]
  print(sp,'n',len(cells),'totalU',sum(tots),'median',statistics.median(tots),'min',min(tots),'max',max(tots))
PY"
