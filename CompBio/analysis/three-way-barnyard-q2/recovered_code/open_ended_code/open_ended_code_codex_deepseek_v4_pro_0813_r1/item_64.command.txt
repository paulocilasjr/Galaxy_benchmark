/bin/bash -c "python3 - <<'PY'
import gzip,collections
raw=collections.defaultdict(collections.Counter); umi=collections.defaultdict(lambda: collections.defaultdict(set))
with gzip.open('read_assign.tsv.gz','rt') as f:
 for line in f:
  bc,u,sp,g,gid=line.rstrip().split('\\t'); raw[bc][sp]+=1; umi[bc][u].add(sp)
for sp in ['hs','mm','ss']:
 arr=[]
 for bc,c in raw.items():
  if c.most_common(1)[0][0]"'!=sp: continue
  other='"'mm' if sp=='hs' else 'hs'
  # umi frac other
  s=collections.Counter(); 
  for u,spp in umi[bc].items():
   if len(spp)==1: s[next(iter(spp))]+=1
  tot=sum(s.values()); f=s[other]/tot if tot else 0
  arr.append(f)
 arr.sort()
 print('SP',sp,'n',len(arr),'min',min(arr),'q25',arr[len(arr)//4],'med',arr[len(arr)//2],'q75',arr[3*len(arr)//4],'max',max(arr))
 print('top 10',[round(x,2) for x in arr[-10:]])
 print('bot 10',[round(x,2) for x in arr[:10]])
PY"
