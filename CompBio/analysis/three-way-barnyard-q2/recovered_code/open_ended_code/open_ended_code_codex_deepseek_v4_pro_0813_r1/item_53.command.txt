/bin/bash -c "python3 - <<'PY'
import gzip, collections
raw=collections.defaultdict(collections.Counter); umi=collections.defaultdict(lambda: collections.defaultdict(set)); totalraw=collections.Counter(); totalumi=collections.Counter()
with gzip.open('read_assign.tsv.gz','rt') as f:
 for line in f:
  bc,u,sp,g,gid=line.rstrip().split('\\t')
  raw[bc][sp]+=1; totalraw[sp]+=1; umi[bc][u].add(sp)
# umi species by cell
rows=[]
for bc,c in raw.items():
 sp_umi=collections.Counter()
 for u,species in umi[bc].items():
  if len(species)==1: sp_umi[next(iter(species))]+=1
 tot=sum(sp_umi.values())
 fracs={sp:sp_umi[sp]/tot for sp in ['hs','mm','ss']}
 # raw frac
 rawtot=sum(c.values())
 rfrac={sp:c[sp]/rawtot for sp in ['hs','mm','ss']}
 rows.append((bc,c.most_common(1)[0][0],tot,c,sp_umi,fracs,rfrac))
print('raw totals',totalraw)
print('top raw species counts',collections.Counter(r[1] for r in rows))
# sort by assigned species and show some human/mouse low pig/high pig
for sp in ['hs','mm','ss']:
 sub=[r for r in rows if r[1]==sp]
 print('\\nSP',sp,'n',len(sub))
 for r in sorted(sub, key=lambda x: x[5]['ss'] if sp"'!='"'ss' else x[5]['ss'], reverse=True)[:10]:
  bc,top,tot,c,spumi,fracs,rfrac=r
  print(bc, 'totU',tot,'raw',dict(c),'umi',dict(spumi),'ufrac',{k:round(v,2) for k,v in fracs.items()},'rfrac',{k:round(v,2) for k,v in rfrac.items()})
 # also bottom contamination
 print('least contam')
 for r in sorted(sub, key=lambda x: x[5]['ss'] if sp"'!='"'ss' else (x[5]['hs']+x[5]['mm']))[:5]:
  bc,top,tot,c,spumi,fracs,rfrac=r
  print(bc, 'totU',tot,'raw',dict(c),'umi',dict(spumi),'ufrac',{k:round(v,2) for k,v in fracs.items()},'rfrac',{k:round(v,2) for k,v in rfrac.items()})
PY"
