M=scipy.io.mmread('kallisto_countoutput.mtx').tocsr()

for i,bc in enumerate(barcodes):
    row=M.getrow(i)
    counts=collections.Counter()

        sp=genes[j].split('|')[0]
        counts[sp]+=int(v)
    sp_counts.append(counts)
print('total UMI per sp', sum((collections.Counter({sp:c[sp] for sp in ['hs','mm','ss']}) for c in sp_counts), collections.Counter()))

        tot=sum(c.values())
        if tot==0: no+=1; continue
        top_sp,top=c.most_common(1)[0]
        if top/tot>=thresh: counts[top_sp]+=1
        else: amb+=1
    print('thr',thresh,'counts',counts,'amb',amb,'no',no,'pct',{k:round(v/sum(counts.values())*100) for k,v in counts.items()})
