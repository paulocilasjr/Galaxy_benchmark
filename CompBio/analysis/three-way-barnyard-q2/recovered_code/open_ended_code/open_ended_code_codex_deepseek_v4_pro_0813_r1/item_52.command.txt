/bin/bash -c "python3 - <<'PY'
import gzip, collections
raw=collections.defaultdict(collections.Counter)
umi=collections.defaultdict(lambda: collections.defaultdict(set))
with gzip.open('read_assign.tsv.gz','rt') as f:
    for line in f:
        bc, u, sp, gene, gid = line.rstrip().split('\\t')
        raw[bc][sp]+=1
        umi[bc][u].add(sp)
print('barcodes',len(raw))
for thresh in [0.5,0.6,0.7,0.8,0.9,0.95,1.0]:
    counts=collections.Counter(); ambiguous=0; unassigned_total=0
    for bc,c in raw.items():
        total=sum(c.values())
        # umi species counts unique UMI
        sp_umi=collections.Counter()
        for u,species in umi[bc].items():
            if len(species)==1:
                sp_umi[next(iter(species))]+=1
        tot=sum(sp_umi.values())
        if tot==0:
            unassigned_total+=1; continue
        top_sp, top=sp_umi.most_common(1)[0]
        if top/tot>=thresh:
            counts[top_sp]+=1
        else:
            ambiguous+=1
    print('thresh',thresh,'counts',counts,'ambig',ambiguous,'noUmi',unassigned_total,'pct', {k:round(v/sum(counts.values())*100) for k,v in counts.items()})
# distribution raw species fractions per barcode
print('top species by raw:')
raw_counts=collections.Counter()
for bc,c in raw.items():
    sp=c.most_common(1)[0][0]; raw_counts[sp]+=1
print(raw_counts)
PY"
