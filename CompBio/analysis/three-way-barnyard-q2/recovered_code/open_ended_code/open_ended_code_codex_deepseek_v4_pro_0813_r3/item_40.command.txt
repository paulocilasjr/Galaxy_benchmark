/bin/bash -lc "python3 - <<'PY'
import collections,csv
conf=set()
with open('barcode_species.tsv') as f:
 for l in f:
  p=l.rstrip('\\n').split('\\t')
  if float(p[4])>=0.8: conf.add((p[0],p[2]))
spcnt=collections.defaultdict(collections.Counter)
with open('species_barcode_gene_umi.tsv') as f:
 for line in f:
  sp,bc,g,c=line.rstrip('\\n').split('\\t')
  if (bc,sp) in conf:
   spcnt[sp][g]+=int(c)
for sp in ['HUMAN','MOUSE','PIG']:
 print('\\n###',sp,'genes',len(spcnt[sp]),'umi',sum(spcnt[sp].values()),'cells',sum(1 for b,s in conf if s==sp))
 for g,c in spcnt[sp].most_common(80): print(g,c)
PY"
