/bin/bash -lc "python3 - <<'PY'
import collections
conf=set()
with open('barcode_species.tsv') as f:
 for l in f:
  p=l.rstrip('\\n').split('\\t')
  if float(p[4])>=0.8: conf.add((p[0],p[2]))
cnt=collections.Counter()
with open('species_barcode_gene_umi.tsv') as f:
 for line in f:
  sp,bc,g,c=line.rstrip('\\n').split('\\t')
  if (bc,sp) in conf and sp=='HUMAN': cnt[g]+=int(c)
for g in ['CD34','KIT','GATA1','GATA2','MPO','ELANE','AZU1','PRTN3','CTSG','CD19','MS4A1','CD79A','CD3D','CD3E','CD4','CD8A','NKG7','GNLY','FCGR3A','CD14','FCER1G','LYZ','S100A8','S100A9','FCN1','MNDA','HBA1','HBA2','HBB','ALAS2','AHSP','BPGM','GYPA','GYPC','HEMGN','BLVRB']:
 print(g,cnt.get(g,0))
PY"
