/bin/bash -lc "python3 - <<'PY'
import collections
# load high-conf species gene counts
conf=set()
with open('barcode_species.tsv') as f:
 for l in f:
  p=l.rstrip('\\n').split('\\t')
  if float(p[4])>=0.8: conf.add((p[0],p[2]))
spcnt=collections.defaultdict(collections.Counter)
with open('species_barcode_gene_umi.tsv') as f:
 for line in f:
  sp,bc,g,c=line.rstrip('\\n').split('\\t')
  if (bc,sp) in conf: spcnt[sp][g]+=int(c)
markers={
'bone_marrow':['HBB','HBA1','HBA2','ALAS2','AHSP','BPGM','GYPC','HBD','S100A8','S100A9','LYZ','MPO','ELANE'],
'spleen':['HBB','HBA1','HBA2','CD8A','CD4','CD19','MS4A1','GNLY','NKG7','FCGR3A'],
'liver':['ALB','APOA1','APOA2','FGA','FGB','FGG','HP','TTR','CYP3A4'],
'heart':['TNNT2','TNNI3','MYH6','MYH7','ACTC1','NPPA','NPPB'],
'testis':['PRM1','PRM2','TNP1','ACR','HSPA1L','DDX4','DAZL','SYCP3','PGK2'],
'kidney':['UMOD','SLC12A1','NPHS2','PODXL','AQP2','SLC34A1'],
'cortex':['STMN2','TUBB3','SYP','RBFOX3','SLC17A7','GAD1','MBP'],
'retina':['RHO','OPN1SW','OPN1LW','NRL','PDE6B','RBP3'],
'lung':['SFTPC','SCGB1A1','AGER','SFTPB','MUC1','FOXJ1'],
'skin':['KRT5','KRT14','KRT1','KRT10','FLG','LOR'],
}
for sp in ['HUMAN','MOUSE','PIG']:
 print('\\n###',sp)
 cnt=spcnt[sp]
 total=sum(cnt.values())
 for tis,genes in markers.items():
  found=[]; score=0
  for g in genes:
   # case-insensitive for mouse
   val=cnt.get(g) or cnt.get(g.capitalize()) or cnt.get(g.upper()) or cnt.get(g.lower())
   if val:
    found.append(f'{g}:{val}')
    score+=val
  print(f'{tis:12s} score={score:6d} {found[:8]}')
 # top tissue markers specifically any genes in marker list sorted
PY"
