/bin/bash -lc 'set -e
for sp in human mouse pig; do
  case $sp in human) prefix=HUMAN; file=human.cdna.fa.gz;; mouse) prefix=MOUSE; file=mouse.cdna.fa.gz;; pig) prefix=PIG; file=pig.cdna.fa.gz;; esac
  gzip -cd $file | awk -v p=$prefix '"'/"'^>/{print ">" p "_" substr($0,2); next}{print}'"' 
done > combined.cdna.fa
wc -l combined.cdna.fa; grep -c '"'^>'"' combined.cdna.fa; ls -lh combined.cdna.fa"
