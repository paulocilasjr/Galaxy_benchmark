/bin/bash -lc "python3 - <<'PY'
import collections
rows=[]
with open('barcode_species_umi.tsv') as f:
 for l in f:
  p=l.rstrip('\\n').split('\\t'); rows.append((p[0],p[2],float(p[4])))
for th in [0.75,0.8,0.9]:
 c=collections.Counter()
 for bc,sp,fr in rows:
  if fr>=th: c[sp]+=1
 tot=sum(c.values())
 print(th,c,tot, {sp:round(n/tot*100/10)*10 for sp,n in c.items()})
PY"
