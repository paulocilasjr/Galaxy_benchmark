/bin/bash -lc "python3 - <<'PY'
import gzip, os, re
files=[('human','refs/cds/Homo_sapiens.GRCh38.cds.all.fa.gz'),('mouse','refs/cds/Mus_musculus.GRCm39.cds.all.fa.gz'),('pig','refs/cds/Sus_scrofa.Sscrofa11.1.cds.all.fa.gz')]
out='refs/cds/combined_cds.fa'
t2g='refs/cds/t2gene.tsv'
with open(out,'w') as fo, open(t2g,'w') as tg:
    for sp,path in files:
        with gzip.open(path,'rt') as f:
            for line in f:
                if line.startswith('>'):
                    h=line[1:].strip()
                    tid=h.split()[0]
                    gene=re.search(r'gene:(["'^'"\\s]+)',h)
                    sym=re.search(r'gene_symbol:(["'^'"\\s]+)',h)
                    desc=re.search(r'description:(.*?)(?: \\[Source:|"'$)'"',h)
                    gene_id=gene.group(1) if gene else ''
                    symbol=sym.group(1) if sym else gene_id
                    description=desc.group(1) if desc else ''
                    new=f'{sp}|{tid}'
                    fo.write(f'>{new}\\n')
                    tg.write('\\t'.join([new,sp,gene_id,symbol,description])+'\\n')
                else:
                    fo.write(line)
print(os.path.getsize(out), os.path.getsize(t2g))
PY
minimap2 -d refs/cds/combined_cds.mmi refs/cds/combined_cds.fa"
