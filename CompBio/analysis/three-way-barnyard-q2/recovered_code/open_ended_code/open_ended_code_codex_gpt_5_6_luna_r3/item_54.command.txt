/bin/bash -lc "python3 - <<'PY'
import csv
from collections import defaultdict,Counter
sums=defaultdict(lambda:Counter()); counts=defaultdict(Counter)
with open('read_species_scores.tsv') as f:
 for r in csv.DictReader(f,delimiter='\\t'):
  bc=r['barcode']
  for sp,col in [('human','human_AS'),('mouse','mouse_AS'),('pig','pig_AS')]:
   v=int(r[col]);
   if v>=0:sums[bc][sp]+=v;counts[bc][sp]+=1
for metric in ['sum','count']:
 c=Counter()
 for bc in sums:
  vals=sums[bc] if metric=='sum' else counts[bc]; c[max(('human','mouse','pig'),key=lambda x:vals[x])]+=1
 print(metric,c)
# score differences / top
for bc in list(sums)[:3]: print(bc,sums[bc],counts[bc])
PY"
