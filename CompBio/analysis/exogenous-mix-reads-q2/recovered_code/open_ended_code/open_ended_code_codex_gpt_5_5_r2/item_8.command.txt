/bin/bash -lc "python3 - <<'PY'
from collections import Counter
files=['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']
def read_fastq(path):
    rec=[]
    with open(path) as f:
        while True:
            h=f.readline().rstrip()
            if not h: break
            s=f.readline().rstrip(); p=f.readline().rstrip(); q=f.readline().rstrip()
            rec.append((h,s,q))
    return rec
R={p:read_fastq(p) for p in files}
for p,rec in R.items():
    print(p, 'records', len(rec), 'unique_seq', len(set(s for _,s,_ in rec)), 'unique_seqqual', len(set((s,q) for _,s,q in rec)))
    print('dups seq', sum(c-1 for c in Counter(s for _,s,_ in rec).values() if c>1))
# overlap of exact seq and seqqual sets
for a in files:
  for b in files:
    if a<b:
      Sa=set(s for _,s,_ in R[a]); Sb=set(s for _,s,_ in R[b])
      Qa=set((s,q) for _,s,q in R[a]); Qb=set((s,q) for _,s,q in R[b])
      print('overlap',a,b,'seq',len(Sa&Sb),'seqqual',len(Qa&Qb))
# multiset containment classify mix exact seqqual unique to refs
exo=Counter((s,q) for _,s,q in R[files[0]])
endo=Counter((s,q) for _,s,q in R[files[1]])
mix=Counter((s,q) for _,s,q in R[files[2]])
print('mix seqqual in exo not endo',sum(c for k,c in mix.items() if k in exo and k not in endo))
print('mix seqqual in endo not exo',sum(c for k,c in mix.items() if k in endo and k not in exo))
print('mix seqqual in both',sum(c for k,c in mix.items() if k in exo and k in endo))
print('mix seqqual in neither',sum(c for k,c in mix.items() if k not in exo and k not in endo))
# sequence-only
exoS=set(s for _,s,_ in R[files[0]]); endoS=set(s for _,s,_ in R[files[1]])
print('mix seq in exo not endo',sum(1 for _,s,_ in R[files[2]] if s in exoS and s not in endoS))
print('mix seq in endo not exo',sum(1 for _,s,_ in R[files[2]] if s in endoS and s not in exoS))
print('mix seq in both',sum(1 for _,s,_ in R[files[2]] if s in exoS and s in endoS))
print('mix seq neither',sum(1 for _,s,_ in R[files[2]] if s not in exoS and s not in endoS))
PY"
