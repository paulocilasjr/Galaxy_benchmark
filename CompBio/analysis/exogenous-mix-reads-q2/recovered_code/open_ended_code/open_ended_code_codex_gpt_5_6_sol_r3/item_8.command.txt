/bin/bash -lc 'bwa index NM_004235.6.fa >/dev/null 2>&1
for f in exogenous.mix.reads.q2.exo.fq exogenous.mix.reads.q2.endo.fq exogenous.mix.reads.q2.mix.fq; do
  bwa mem -t 2 NM_004235.6.fa "$f" 2>/dev/null | samtools view -b -o "$f.bam"
  samtools sort -o "$f.sorted.bam" "$f.bam"
  samtools index "$f.sorted.bam"
  printf '"'%s ' \""'$f"
  samtools view -c "$f.bam"
  printf '"'mapped '
  samtools view -c -F 4 \""'$f.bam"
  printf '"'unique '
  samtools view -c -F 4 -q 20 \""'$f.bam"
done
samtools depth -aa exogenous.mix.reads.q2.exo.fq.sorted.bam > exo.depth.txt
samtools depth -aa exogenous.mix.reads.q2.endo.fq.sorted.bam > endo.depth.txt
samtools depth -aa exogenous.mix.reads.q2.mix.fq.sorted.bam > mix.depth.txt
samtools view -F 4 exogenous.mix.reads.q2.mix.fq.bam | cut -f1,3,4,5,6 | head -20'
