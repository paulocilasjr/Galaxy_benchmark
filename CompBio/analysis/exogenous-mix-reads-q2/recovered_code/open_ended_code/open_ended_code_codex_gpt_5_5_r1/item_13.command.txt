/bin/bash -lc "python3 - <<'PY'
# Try fetching a KLF4 RefSeq mRNA into workspace for local alignment checks.
# If network fails, this exits nonzero but does not affect existing files.
import urllib.request
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235&rettype=fasta&retmode=text'
try:
    data=urllib.request.urlopen(url, timeout=20).read().decode()
    open('NM_004235_KLF4.fasta','w').write(data)
    print(data.split('\\n',1)[0])
    print('bases', sum(len(l.strip()) for l in data.splitlines() if not l.startswith('>')))
except Exception as e:
    print('FETCH_FAILED', type(e).__name__, e)
    raise
PY"
