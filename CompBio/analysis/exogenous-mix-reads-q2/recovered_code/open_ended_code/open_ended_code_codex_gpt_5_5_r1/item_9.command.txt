# Write no files. Summarize record ranges and whether records match exactly by index.
files=['exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.mix.fq']

for i,f in enumerate(files):
    print(f, len(recs[i]), recs[i][0][0], recs[i][-1][0])
print('endo == exo prefix:', recs[0]==recs[1][:len(recs[0])])
print('exo == mix prefix:', recs[1]==recs[2][:len(recs[1])])
print('mix extra headers:', recs[2][len(recs[1])][0], recs[2][-1][0])
# Unique by ranges
for name,lo,hi in [('common_1_395',0,395),('exo_extra_396_592',395,592),('mix_extra_593_691',592,691)]:
    seqs=[r[1] for r in recs[2][lo:hi]]
    print(name, len(seqs), 'unique', len(set(seqs)), 'dups', len(seqs)-len(set(seqs)))
    print('GC', sum(s.count('G')+s.count('C') for s in seqs)/(len(seqs)*50))
