/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
files=['exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.mix.fq','exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.mix.fq','exogenous.mix.reads.q2.endo.fq']
sets={}; counts={}
for f in files:
 c=Counter()
 with open(f) as fh:
  while True:
   h=fh.readline()
   if not h:break
   s=fh.readline().strip(); fh.readline(); fh.readline(); c[s]+=1
 key=f.split('/')[-1]
 counts[key]=c; sets[key]=set(c)
 print(key,'reads',sum(c.values()),'unique',len(c))
for a in ['exogenous.mix.reads.q2.mix.fq','exogenous.mix.reads.q2.exo.fq']:
 for b in ['exogenous.mix.reads.q1.mix.fq','exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.endo.fq']:
  print(a,'reads in',b,'set:',sum(n for s,n in counts[a].items() if s in sets[b]),'/',sum(counts[a].values()),'unique inter',len(sets[a]&sets[b]))
PY"
