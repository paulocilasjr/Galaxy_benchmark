/bin/bash -c "python3 - <<'PY'
from pathlib import Path
seq=''.join(l.strip() for l in open('klf4_nm.fa') if not l.startswith('>'))
comp=str.maketrans('ACGT','TGCA')
rc=seq.translate(comp)[::-1]
reads=[]
for name in ['exo','endo','mix']:
 with open(f'exogenous.mix.reads.q2.{name}.fq') as f:
  while True:
   h=f.readline()
   if not h: break
   s=f.readline().strip(); f.readline(); f.readline()
   reads.append((name,s))
# print matches for first few
for name,s in reads[:10]:
 print(name,s)
 for lab,t in [('+',seq),('-',rc)]:
  i=t.find(s)
  if i>=0: print(' ',lab,i,i+len(s), 'or rc', t.find(s.translate(comp)[::-1]))
# all match stats
for name in ['exo','endo','mix']:
 n=0; plus=minus=both=0; no=0
 seen=set()
 with open(f'exogenous.mix.reads.q2.{name}.fq') as f:
  while True:
   h=f.readline()
   if not h: break
   s=f.readline().strip(); f.readline(); f.readline(); n+=1
   if s in seen: continue
   seen.add(s)
   p=seq.find(s); m=rc.find(s)
   if p>=0: plus+=1
   if m>=0: minus+=1
   if p>=0 and m>=0: both+=1
   if p<0 and m<0: no+=1
 print(name,'reads',n,'unique',len(seen),'plus',plus,'minus',minus,'both',both,'no',no)
PY"
