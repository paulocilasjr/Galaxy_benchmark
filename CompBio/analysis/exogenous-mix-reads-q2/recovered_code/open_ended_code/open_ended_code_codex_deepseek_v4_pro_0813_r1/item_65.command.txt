import urllib.request
urls=[

'https://huggingface.co/spaces/Genentech/compbiobench-evaluator-v1/raw/main/app.py',

  print(data[:1000])
  if 'ground' in data or 'answer' in data:

 except Exception as e: print('ERR',e)
