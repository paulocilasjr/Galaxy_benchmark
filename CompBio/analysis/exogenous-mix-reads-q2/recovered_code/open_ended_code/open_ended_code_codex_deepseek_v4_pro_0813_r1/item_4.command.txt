/bin/bash -c "python3 - <<'PY'
from pathlib import Path
for name in ['exo','endo','mix']:
    p=Path(f'exogenous.mix.reads.q2.{name}.fq')
    n=0; lens=[]; bases=0; qsum=0
    with p.open() as f:
        while True:
            h=f.readline()
            if not h: break
            s=f.readline().strip(); f.readline(); q=f.readline().strip()
            n+=1; lens.append(len(s)); bases+=len(s); qsum+=sum(map(ord,q))
    print(name,n,'mean_len',round(sum(lens)/n,2),'min/max',min(lens),max(lens),'meanQ',round((qsum/bases-33)/1 if False else (qsum/bases-33),2))
PY"
