/bin/bash -lc "python3 - <<'PY'
import pysam
for f in ['exogenous.mix.reads.q2.exo.bam','exogenous.mix.reads.q2.endo.bam','exogenous.mix.reads.q2.mix.bam']:
    b=pysam.AlignmentFile(f,'rb')
    n=0; flags={}; mapq=[]; strands=[]
    for r in b:
        n+=1
        flags[r.flag]=flags.get(r.flag,0)+1
        if not r.is_unmapped:
            mapq.append(r.mapping_quality)
            strands.append('-' if r.is_reverse else '+')
    print(f,n,'flags',flags,'mapped',len(mapq),'strands',strands.count('+'),strands.count('-'))
PY"
