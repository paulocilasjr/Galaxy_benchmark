/bin/bash -lc "python3 - <<'PY'
from collections import Counter
ref=''.join(open('klf4_nm.fa').read().splitlines()[1:])
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s[::-1].translate(comp)
def get(s):
    if s in ref: return ref.find(s),'+'
    r=rc(s)
    if r in ref: return ref.find(r),'-'
    # best hamming 1
    for i in range(len(ref)-50+1):
        if sum(a"'!=b for a,b in zip(s,ref[i:i+50]))<=1: return i,'"'+*'
    for i in range(len(ref)-50+1):
        if sum(a"'!=b for a,b in zip(r,ref[i:i+50]))<=1: return i,'"'-*'
    return None
seqs={}
for f in ['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']:
    lines=open(f).read().splitlines(); seqs[f]=[lines[i+1] for i in range(0,len(lines),4)]
exo=seqs['exogenous.mix.reads.q2.exo.fq']; mix=seqs['exogenous.mix.reads.q2.mix.fq']
print('verify mix prefix', all(a==b for a,b in zip(exo,mix[:len(exo)])))
extra=mix[len(exo):]
print('extra count',len(extra),'unique',len(set(extra)),'counts',Counter(extra).most_common(20))
exo_set=set(exo)
print('extra in exo set',sum(s in exo_set for s in extra))
for s in extra:
    print(s, get(s))
PY"
