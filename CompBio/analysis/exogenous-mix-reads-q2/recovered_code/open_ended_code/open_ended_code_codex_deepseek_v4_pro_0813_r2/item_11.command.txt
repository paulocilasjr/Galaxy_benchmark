/bin/bash -lc "python3 - <<'PY'
from collections import Counter
ref=''.join(open('klf4_nm.fa').read().splitlines()[1:])
comp=str.maketrans('ACGT','TGCA')
def rc(s): return s[::-1].translate(comp)
def pos(s):
    if s in ref: return ref.find(s),'+'
    r=rc(s)
    if r in ref: return ref.find(r),'-'
    return None
for f in ['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']:
    lines=open(f).read().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
    ps=[]
    for s in seqs:
        p=pos(s); ps.append(p)
    print(f,'first 20 pos/strand',ps[:20])
    print('last 20',ps[-20:])
    print('num exact/position',sum(p is not None for p in ps))
PY"
