/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
from collections import Counter,defaultdict
for f in Path('.').glob('*.fq'):
    seqs=[]; quals=[]
    with f.open() as fh:
        lines=fh.read().splitlines()
    for i in range(0,len(lines),4):
        seqs.append(lines[i+1])
        quals.append(lines[i+3])
    print(f, 'reads', len(seqs), 'uniq seqs', len(set(seqs)), 'len min/max', min(map(len,seqs)), max(map(len,seqs)))
    c=Counter(seqs)
    print('top', c.most_common(10))
    print('unique counts distribution', Counter(c.values()).most_common(20))
PY"
