/bin/bash -lc "python3 - <<'PY'
from collections import Counter
def load(f):
 lines=open(f).read().splitlines(); return [lines[i+1] for i in range(0,len(lines),4)]
files=['exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q1.mix.fq']
sets={}; counts={}
for f in files:
 seqs=load(f); sets[f]=set(seqs); counts[f]=Counter(seqs)
exo,endo,mix=[f'exogenous.mix.reads.q1.{x}.fq' for x in ['exo','endo','mix']]
E=sets[exo]; N=sets[endo]
exo_only=E-N; endo_only=N-E; both=E&N
for label,s in [('exo_only',exo_only),('endo_only',endo_only),('both',both)]:
 print(label,'uniq',len(s))
 print(' exo reads',sum(counts[exo][x] for x in s),'rate',sum(counts[exo][x] for x in s)/sum(counts[exo].values()))
 print(' endo reads',sum(counts[endo][x] for x in s),'rate',sum(counts[endo][x] for x in s)/sum(counts[endo].values()))
 print(' mix reads',sum(counts[mix][x] for x in s),'rate',sum(counts[mix][x] for x in s)/sum(counts[mix].values()))
# simple estimates
a=sum(counts[exo][x] for x in exo_only)/sum(counts[exo].values())
b=sum(counts[endo][x] for x in endo_only)/sum(counts[endo].values())
am=sum(counts[mix][x] for x in exo_only)/sum(counts[mix].values())
bm=sum(counts[mix][x] for x in endo_only)/sum(counts[mix].values())
print('a,b,am,bm',a,b,am,bm)
print('p from exo marker',am/a)
print('p from endo marker',1-bm/b)
print('combined maybe (am/a + 1-bm/b)/2',(am/a+1-bm/b)/2)
PY"
