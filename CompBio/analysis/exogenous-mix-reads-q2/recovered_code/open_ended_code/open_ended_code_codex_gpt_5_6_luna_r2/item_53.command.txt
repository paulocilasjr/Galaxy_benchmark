/bin/bash -lc "python3 - <<'PY'
import subprocess, math
from collections import Counter

def feats(fn,bs):
 p=subprocess.run(['minimap2','-a','-x','sr','--secondary=no','klf4_ref.fa',fn],capture_output=True,text=True,check=True)
 z=[]
 for l in p.stdout.splitlines():
  if l.startswith('@'):continue
  f=l.split('\\t');
  if f[2]=='*': continue
  z.append((int(f[3])-1)//bs)
 return z
for q in ['q1','q2']:
 print('\\n'+q,flush=True)
 for bs in [1,10,50,100,200]:
  E=Counter(feats(f'exogenous.mix.reads.{q}.endo.fq',bs)); X=Counter(feats(f'exogenous.mix.reads.{q}.exo.fq',bs)); M=feats(f'exogenous.mix.reads.{q}.mix.fq',bs)
  nE=sum(E.values());nX=sum(X.values());keys=set(E)|set(X)|set(M);k=len(keys);pc=1
  pe={s:(E[s]+pc)/(nE+pc*k) for s in keys};px={s:(X[s]+pc)/(nX+pc*k) for s in keys}
  vals=[]
  for j in range(101):
   a=j/100
   vals.append(sum(math.log(a*px[s]+(1-a)*pe[s]) for s in M))
  best=max(range(101),key=lambda j:vals[j])/100
  print(bs,len(M),round(best,2),flush=True)
PY"
