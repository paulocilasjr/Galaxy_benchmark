/bin/bash -lc "python3 - <<'PY'
import subprocess
from collections import Counter

def pos(fn):
 p=subprocess.run(['minimap2','-a','-x','sr','--secondary=no','klf4_ref.fa',fn],capture_output=True,text=True,check=True)
 z=[]
 for l in p.stdout.splitlines():
  if l.startswith('@'):continue
  f=l.split('\\t')
  if f[2]"'!='"'*':z.append((int(f[3]),int(f[1])&16))
 return z
for q in ['q1','q2']:
 print('\\n',q)
 for name in ['endo','exo','mix']:
  z=pos(f'exogenous.mix.reads.{q}.{name}.fq');
  print(name,len(z),'plus',sum(not s for p,s in z),'minus',sum(s for p,s in z),'mean',round(sum(p for p,s in z)/len(z),1),'median',sorted(p for p,s in z)[len(z)//2])
 for bs in [50,100,200,500]:
  cs={n:Counter((p-1)//bs for p,s in pos(f'exogenous.mix.reads.{q}.{n}.fq')) for n in ['endo','exo','mix']}
  print('bs',bs)
  for n in ['endo','exo','mix']:
   print(n,' '.join(str(cs[n][i]) for i in range((2967+bs-1)//bs)))
PY"
