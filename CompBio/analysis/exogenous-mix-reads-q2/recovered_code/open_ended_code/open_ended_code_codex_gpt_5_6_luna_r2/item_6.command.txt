/bin/bash -lc 'tmpdir=$(mktemp -d); for n in exo endo mix; do f="exogenous.mix.reads.q2.$n.fq"; awk '"'NR%4==2' \""'$f" | sort | uniq -c | awk '"'{print "'$2"'"\\t\""'$1}'"' | sort > \""'$tmpdir/$n.tsv"; done; echo '"'exo vs endo count pattern'; join -t "'$'"'\\t' \""'$tmpdir/exo.tsv" "$tmpdir/endo.tsv" | awk -F '"'\\t' '{p["'$2","$3]++} END{for(k in p) print k,p[k]}'"' | sort; echo 'endo sequences counts absent/smaller/equal/larger in exo'; join -t "'$'"'\\t' -a2 -e 0 -o 2.1,1.2,2.2 \""'$tmpdir/exo.tsv" "$tmpdir/endo.tsv" | awk -F '"'\\t' '{if("'$2==0)abs++; else if($2<$3)smaller++; else if($2==$3)eq++; else larger++} END{print "absent",abs,"exo<endo",smaller,"equal",eq,"exo>endo",larger}'"'; echo 'mix sequence membership and counts'; join -t "'$'"'\\t' -a1 -a2 -e 0 -o 0,1.2,2.2,3.2 \""'$tmpdir/exo.tsv" "$tmpdir/endo.tsv" | sort -k1,1 > "$tmpdir/both.tsv"; join -t $'"'\\t' -a1 -a2 -e 0 -o 0,1.2,2.2 \""'$tmpdir/both.tsv" "$tmpdir/mix.tsv" | awk -F '"'\\t' '{if("'$2>0&&$3==0) exonly++; else if($2==0&&$3>0) endonly++; else if($2>0&&$3>0) both++; else un++} END{print "exonly",exonly,"endonly",endonly,"both",both,"un",un}'"'"
