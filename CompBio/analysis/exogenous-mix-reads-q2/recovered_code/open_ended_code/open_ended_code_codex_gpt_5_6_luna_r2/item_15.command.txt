/bin/bash -lc "python3 - <<'PY'
from pathlib import Path

def seqs(fn):
    a=Path(fn).read_text().splitlines()
    return [a[i] for i in range(1,len(a),4)]
exo=seqs('exogenous.mix.reads.q2.exo.fq'); endo=seqs('exogenous.mix.reads.q2.endo.fq'); mix=seqs('exogenous.mix.reads.q2.mix.fq')
for name,a,b in [('exo-endo',exo,endo),('mix-exo',mix,exo),('mix-endo',mix,endo)]:
    print(name)
    for n in [10,20,50,100,200,300,395]:
        if n<=len(a) and n<=len(b): print(n,sum(x==y for x,y in zip(a[:n],b[:n])))
    print('exact same sequence positions',sum(x==y for x,y in zip(a,b)))
# find whether endo sequences appear in exo in same order (indices)
idx={}
for i,s in enumerate(exo): idx.setdefault(s,[]).append(i)
for label,b in [('endo',endo),('mix',mix)]:
    inds=[]; missing=0
    for s in b:
        if s in idx: inds.append(idx[s][0])
        else: missing+=1
    print(label,'matches',len(inds),'missing',missing,'first idx',inds[:20],'nondecreasing',all(x<=y for x,y in zip(inds,inds[1:])))
PY"
