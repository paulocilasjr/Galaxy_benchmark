/bin/bash -lc "python3 - <<'PY'
import subprocess, math
from collections import Counter
from scipy.optimize import minimize_scalar

def feats(fn,bs,include_strand=False):
 p=subprocess.run(['minimap2','-a','-x','sr','--secondary=no','klf4_ref.fa',fn],capture_output=True,text=True,check=True)
 z=[]
 for l in p.stdout.splitlines():
  if l.startswith('@'):continue
  f=l.split('\\t');
  if f[2]=='*': continue
  pos=(int(f[3])-1)//bs
  z.append((pos, f[1]=='16') if include_strand else pos)
 return z
for q in ['q1','q2']:
 print('\\n',q)
 for bs in [1,5,10,20,50,100,200,500]:
  E=Counter(feats(f'exogenous.mix.reads.{q}.endo.fq',bs)); X=Counter(feats(f'exogenous.mix.reads.{q}.exo.fq',bs)); M=feats(f'exogenous.mix.reads.{q}.mix.fq',bs)
  nE=sum(E.values());nX=sum(X.values());
  # smoothed p with pseudocount 1/number bins? For multinomial source empirical with pseudocount alpha0=1
  keys=set(E)|set(X)|set(M); k=len(keys)
  for pc in [0.1,1,5]:
   pe={s:(E[s]+pc)/(nE+pc*k) for s in keys};px={s:(X[s]+pc)/(nX+pc*k) for s in keys}
   def nll(a):
    return -sum(math.log(a*px[s]+(1-a)*pe[s]) for s in M)
   res=minimize_scalar(nll,bounds=(0,1),method='bounded')
   if pc==1:best=res.x
  print('bs',bs,'mapped',len(M),'MLE pc1',round(best,3),end='; ')
 print()
PY"
