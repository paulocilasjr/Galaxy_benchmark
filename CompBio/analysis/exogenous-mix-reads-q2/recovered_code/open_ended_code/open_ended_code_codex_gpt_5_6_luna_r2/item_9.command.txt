/bin/bash -lc "python3 - <<'PY'
from collections import Counter, defaultdict
from pathlib import Path

def seqs(fn):
    lines=Path(fn).read_text().splitlines()
    return [lines[i] for i in range(1,len(lines),4)]
files=['exogenous.mix.reads.q2.exo.fq','exogenous.mix.reads.q2.endo.fq','exogenous.mix.reads.q2.mix.fq']
S={f:seqs(f) for f in files}
C={f:Counter(S[f]) for f in files}
exo=set(C[files[0]]); endo=set(C[files[1]]); mix=set(C[files[2]])
un= [s for s in mix if s not in exo and s not in endo]
print('counts', {f:(len(S[f]),len(C[f])) for f in files})
print('mix class unique/count', len(un), sum(C[files[2]][s] for s in un))
# exact max suffix-prefix overlap with reference reads
for s in sorted(un, key=lambda z:z)[:10]:
    best=[]
    mx=0
    for t in exo|endo:
        for k in range(49,9,-1):
            if s[:k]==t[-k:] or s[-k:]==t[:k]:
                if k>mx: mx=k; best=[(k,t[:12],t[-12:], 'prefix' if s[:k]==t[-k:] else 'suffix')]
                elif k==mx: best.append((k,t[:12],t[-12:], 'prefix' if s[:k]==t[-k:] else 'suffix'))
                break
    print('un',s,'maxoverlap',mx,'best',best[:3])
# for each unmatched, max exact substring relation to all exo; aggregate
bins=Counter(); examples={}
for s in un:
    mx=0; bt=None; typ=None
    for t in exo:
        # max prefix/prefix or suffix/suffix? shared reads from same transcript can have shifted start, overlap
        for k in range(49,9,-1):
            if s[:k] in t or s[-k:] in t:
                mx=k; bt=t; typ='contains'; break
        if mx: break
    bins[mx]+=1; examples.setdefault(mx,(s,bt))
print('max substring bins',bins)
for k in sorted(examples,reverse=True)[:10]: print(k,examples[k])
PY"
