/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
lines=Path('exogenous.mix.reads.q2.mix.fq').read_text().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
extra=seqs[592:691]
c=Counter(extra)
print('extra total',len(extra),'unique',len(c))
for s,n in c.most_common(20):
    if n>1: print(n,s)
# duplicates in exo and endo for comparison
for name,ss in [('endo',seqs[:395]),('exo_extra',seqs[395:592]),('exo_all',seqs[:592])]:
    cc=Counter(ss)
    print(name,'dup reads',sum(v for v in cc.values() if v>1),'unique',len(cc))
    print([(v,k) for k,v in cc.most_common() if v>1][:10])
PY"
