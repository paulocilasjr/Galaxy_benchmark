/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import defaultdict
lines=Path('exogenous.mix.reads.q2.mix.fq').read_text().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
exo=set(seqs[:592]); endo=set(seqs[:395]); mix=seqs
for k in [10,15,20,25,30,40,50]:
    # kmer sets from exo and endo reads
    def kmers_set(seqlist):
        s=set()
        for seq in seqlist:
            for i in range(0,50-k+1): s.add(seq[i:i+k])
        return s
    ek=kmers_set(seqs[:592]); nk=kmers_set(seqs[:395])
    exo_specific=ek-nk; endo_specific=nk-ek; shared=ek&nk
    print('\\nk',k,'exo_specific kmers',len(exo_specific),'endo_specific',len(endo_specific),'shared',len(shared))
    # classify each mix read: contains any exo-specific kmer, any endo-specific (none), etc
    counts={'exo_specific':0,'only_shared_or_none':0,'has_exo_specific_and_endospecific':0}
    for s in seqs:
        ks={s[i:i+k] for i in range(0,50-k+1)}
        if ks & exo_specific:
            if ks & endo_specific: counts['has_exo_specific_and_endospecific']+=1
            else: counts['exo_specific']+=1
        else: counts['only_shared_or_none']+=1
    print(counts, 'pct exo_specific', counts['exo_specific']/len(seqs))
PY"
