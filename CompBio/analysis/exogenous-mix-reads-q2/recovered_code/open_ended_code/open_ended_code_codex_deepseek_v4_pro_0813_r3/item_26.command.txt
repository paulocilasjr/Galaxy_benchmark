from pathlib import Path
from collections import Counter
for f in ['exogenous.mix.reads.q2.endo.sam','exogenous.mix.reads.q2.exo.sam','exogenous.mix.reads.q2.mix.sam']:
    flags=[]

        if line.startswith('@'): continue
        flags.append(int(line.split('\\t')[1]))
    c=Counter(f & 16 for f in flags)
    print(f,'plus(0)',c[0],'minus(16)',c[16])
# groups
for name,file,start,end in [('endo','exogenous.mix.reads.q2.endo.sam',0,395),('exo_extra','exogenous.mix.reads.q2.exo.sam',395,592),('mix_extra','exogenous.mix.reads.q2.mix.sam',592,691)]:
    fs=[]
    for i,line in enumerate(Path(file).read_text().splitlines()):
        if line.startswith('@'): continue
        if start<=i<end: fs.append(int(line.split('\\t')[1]))
    c=Counter(f & 16 for f in fs)
