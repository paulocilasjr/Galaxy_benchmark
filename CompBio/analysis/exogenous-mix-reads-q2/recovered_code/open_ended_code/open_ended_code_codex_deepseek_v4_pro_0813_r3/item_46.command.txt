/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter,defaultdict
import math
lines=Path('exogenous.mix.reads.q2.mix.fq').read_text().splitlines(); seqs=[lines[i+1] for i in range(0,len(lines),4)]
exo=seqs[:592]; endo=seqs[:395]
for k in [10,15,20,25,30]:
    exoc=Counter(); endoc=Counter()
    for s in exo:
        for i in range(0,50-k+1): exoc[s[i:i+k]]+=1
    for s in endo:
        for i in range(0,50-k+1): endoc[s[i:i+k]]+=1
    total_exo=sum(exoc.values()); total_endo=sum(endoc.values())
    vocab=set(exoc)|set(endoc)
    # log ratio with pseudocount .5
    logodds={}
    for km in vocab:
        pe=(exoc[km]+0.5)/(total_exo+0.5*len(vocab))
        pn=(endoc[km]+0.5)/(total_endo+0.5*len(vocab))
        logodds[km]=math.log(pe/pn)
    exo_count=0
    scores=[]
    for s in seqs:
        score=sum(logodds.get(s[i:i+k],0) for i in range(0,50-k+1))
        scores.append(score)
        if score>0: exo_count+=1
    print('k',k,'classified exo',exo_count,'pct',exo_count/len(seqs),'mean score groups')
    for name,inds in [('endo',range(395)),('exo_extra',range(395,592)),('mix_extra',range(592,691))]:
        sc=[scores[i] for i in inds]
        print(name,sum(1 for x in sc if x>0)/len(sc),sum(sc)/len(sc))
PY"
