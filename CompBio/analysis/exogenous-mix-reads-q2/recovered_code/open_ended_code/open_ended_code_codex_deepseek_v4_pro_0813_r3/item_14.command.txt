/bin/bash -c "python3 - <<'PY'
from pathlib import Path
def parse(f):
    lines=Path(f).read_text().splitlines()
    return [(lines[i][1:],lines[i+1],lines[i+3]) for i in range(0,len(lines),4) if lines[i].startswith('@')]
exo=parse('exogenous.mix.reads.q2.exo.fq')
endo=parse('exogenous.mix.reads.q2.endo.fq')
mix=parse('exogenous.mix.reads.q2.mix.fq')
# compare names
for f,a,b in [('exo-endo',exo,endo),('exo-mix',exo,mix),('endo-mix',endo,mix)]:
    names_a=[x[0] for x in a]; names_b=[x[0] for x in b]
    common=set(names_a)&set(names_b)
    print(f,'names',len(names_a),len(names_b),'common',len(common),'first20equal',names_a[:20]==names_b[:20])
    # Are common records identical seq?
    dict_b={x[0]:x[1] for x in b}
    diff=sum(1 for x in a if x[0] in dict_b and x[1]"'!=dict_b[x[0]])
    print('"'common names seq diff',diff)
# Check endo subset in order of exo
exo_names=[x[0] for x in exo]
endo_names=[x[0] for x in endo]
# positions in exo
pos=[exo_names.index(n) for n in endo_names if n in exo_names]
print('endo names in exo?', all(n in exo_names for n in endo_names), 'pos sorted?', pos==sorted(pos), 'pos first',pos[:20], 'last',pos[-10:])
mix_names=[x[0] for x in mix]
print('mix names in exo?', sum(n in exo_names for n in mix_names), 'of',len(mix_names))
# duplicates names? 
from collections import Counter
print('exo duplicate names',sum(1 for n,c in Counter(exo_names).items() if c>1), 'endo',sum(1 for n,c in Counter(endo_names).items() if c>1))
PY"
