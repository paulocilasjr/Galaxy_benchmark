/bin/bash -c "python3 - <<'PY'
from pathlib import Path
import statistics
for name,file,start,end in [('endo','exogenous.mix.reads.q2.endo.fq',0,395),('exo_extra','exogenous.mix.reads.q2.exo.fq',395,592),('mix_extra','exogenous.mix.reads.q2.mix.fq',592,691)]:
    lines=Path(file).read_text().splitlines(); quals=[lines[i+3] for i in range(0,len(lines),4)][start:end]
    per=[statistics.mean([ord(c)-33 for c in q]) for q in quals]
    print(name,'mean of per-read mean',statistics.mean(per),'sd',statistics.pstdev(per),'median',statistics.median(per),'minmax',min(per),max(per))
    # distribution low quality <30
    print('frac bases q<30', sum(1 for q in quals for c in q if ord(c)-33<30)/sum(len(q) for q in quals))
PY"
