/bin/bash -lc 'tmpdir=$(find . -maxdepth 1 -type d -name '"'refcheck.*' | head -n 1)
for f in exogenous.mix.reads.q2.endo.fq exogenous.mix.reads.q2.exo.fq exogenous.mix.reads.q2.mix.fq; do b="'${f%.fq}; minimap2 -a -x sr "$tmpdir/KLF4_NM_004235.6.fa" "$f" 2>/dev/null | samtools view -b -o "$tmpdir/$(basename "$b").bam" -; samtools sort -o "$tmpdir/$(basename "$b").sorted.bam" "$tmpdir/$(basename "$b").bam"; samtools index "$tmpdir/$(basename "$b").sorted.bam"; done
for b in endo exo mix; do echo "== $b mapping"; samtools view "$tmpdir/exogenous.mix.reads.q2.$b.sorted.bam" | awk '"'BEGIN{m=0;nm=0}{m++; if("'$5>=20)q20++; if($6~/''^[0-9]+M$/)full++;}END{print "mapped="m,"MAPQ20="q20+0,"50M="full+0}'"'; echo 'start counts (top)'; samtools view \""'$tmpdir/exogenous.mix.reads.q2.$b.sorted.bam" | awk '"'{print "'$4}'"' | sort -n | uniq -c | sort -k2n | head -n 15; done"
