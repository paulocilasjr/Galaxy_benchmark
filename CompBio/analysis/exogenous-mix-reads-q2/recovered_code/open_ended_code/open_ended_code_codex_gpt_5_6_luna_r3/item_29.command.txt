/bin/bash -lc 'tmpdir=$(find . -maxdepth 1 -type d -name '"'hfcheck.*' | head -n 1); for pair in 'endo.fq ../exogenous.mix.reads.q2.endo.fq' 'exo.fq ../exogenous.mix.reads.q2.exo.fq' 'mix.fq ../exogenous.mix.reads.q2.mix.fq'; do set -- "'$pair; a="$tmpdir/$1"; b="$2"; echo "$(basename "$a") vs $(basename "$b")"; diff -q "$a" "$b" >/dev/null; echo status=$?; done
# compare sequence intersections q1/q2 each type
for k in exo endo mix; do echo -n "$k q1∩q2="; comm -12 <(awk '"'NR%4==2' \""'$tmpdir/$k.fq" | sort -u) <(awk '"'NR%4==2' \"exogenous.mix.reads.q2."'$k.fq" | sort -u) | wc -l; done'
