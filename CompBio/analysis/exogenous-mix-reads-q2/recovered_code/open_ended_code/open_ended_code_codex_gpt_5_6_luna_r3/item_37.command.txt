/bin/bash -lc 'tmpdir=$(find . -maxdepth 1 -type d -name '"'refcheck.*' | head -n 1); hfdir="'$(find . -maxdepth 1 -type d -name '"'hfcheck.*' | head -n 1)
for f in \""'$hfdir"/*.fq; do b=$(basename "${f%.fq}"); minimap2 -a -x sr "$tmpdir/KLF4_NM_004235.6.fa" "$f" 2>/dev/null | samtools view -b -o "$tmpdir/q1_${b}.bam" -; samtools sort -o "$tmpdir/q1_${b}.sorted.bam" "$tmpdir/q1_${b}.bam"; samtools index "$tmpdir/q1_${b}.sorted.bam"; done
for b in endo exo mix; do echo "== q1 $b mapping"; samtools view "$tmpdir/q1_${b}.sorted.bam" | awk '"'BEGIN{m=0}{m++;if("'$5>=20)q20++;if($6~/''^[0-9]+M$/)full++}END{print "mapped="m,"MAPQ20="q20+0,"50M="full+0}'"'; done
for b in endo exo mix; do echo \"== q1 "'$b bins"; samtools view "$tmpdir/q1_${b}.sorted.bam" | awk '"'{bin=int(("'$4-1)/100)*100; c[bin]++}END{for(i in c) print i,c[i]}'"' | sort -n; done"
