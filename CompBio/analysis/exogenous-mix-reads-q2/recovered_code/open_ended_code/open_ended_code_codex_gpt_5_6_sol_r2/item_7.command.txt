/bin/bash -lc "curl -L --fail --silent --show-error 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text' -o KLF4_NM_004235.6.fa
head -2 KLF4_NM_004235.6.fa
wc -c KLF4_NM_004235.6.fa
minimap2 -a -x sr KLF4_NM_004235.6.fa exogenous.mix.reads.q2.mix.fq > mix.klf4.sam 2> minimap.log
samtools flagstat mix.klf4.sam
samtools view mix.klf4.sam | awk '{if(and("'$2,4)) u++; else {m++; pos[int($4/100)*100]++}} END{print "mapped",m,"unmapped",u; for(p in pos) print p,pos[p]}'"' | sort -n"
