/bin/bash -c "python3 - <<'PY'
import requests
base='https://rest.ensembl.org'
for region in ['11:57000000-58000000','11:57700000-57950000']:
    r=requests.get(f'{base}/overlap/region/mus_musculus/{region}?feature=gene;content-type=application/json', headers={'Content-Type':'application/json'}, timeout=20)
    print('REGION',region, r.status_code)
    genes=r.json()
    for g in sorted([x for x in genes if x.get('biotype')=='protein_coding'], key=lambda x:x['start']):
        print(g['id'], g['external_name'], g['strand'], g['start'], g['end'])
PY"
