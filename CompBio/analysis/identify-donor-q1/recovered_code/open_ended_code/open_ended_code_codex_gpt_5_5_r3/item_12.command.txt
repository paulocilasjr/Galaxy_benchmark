import gzip, urllib.parse, urllib.request, json, time
coords=[]
with gzip.open('identify.donor.R1.fq.gz','rt') as f:
    for i in range(0,2000):
        name=f.readline().strip(); seq=f.readline().strip(); f.readline(); f.readline()
        if i%200==0:
            url='https://genome.ucsc.edu/cgi-bin/hgBlat?'+urllib.parse.urlencode({'userSeq':seq,'type':'DNA','db':'hg38','output':'json'})
            req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
            txt=urllib.request.urlopen(req, timeout=30).read().decode('utf-8','replace')
            try:
                data=json.loads(txt)
            except Exception as e:
                print('json fail',i,repr(txt[:100])); continue
            hits=data.get('blat',[])
            hits=sorted(hits, key=lambda x:(-x[0], x[1]))
            if hits:
                h=hits[0]
                print(i, h[13], h[15], h[16], 'matches',h[0],'mis',h[1],'strand',h[8])
                coords.append((h[13],h[15],h[16]))
            time.sleep(0.2)
print('span')
from collections import Counter
print(Counter(c for c,_,__ in coords))
for chrom in sorted(set(c for c,_,__ in coords)):
    vals=[(s,e) for c,s,e in coords if c==chrom]
    print(chrom,min(s for s,e in vals), max(e for s,e in vals), len(vals))
PY"
