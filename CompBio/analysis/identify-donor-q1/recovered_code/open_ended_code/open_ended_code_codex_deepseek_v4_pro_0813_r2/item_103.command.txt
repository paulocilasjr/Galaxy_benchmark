/bin/bash -lc "bcftools view -h calls.vcf.gz | awk 'BEGIN{keep=1} /"'^##contig=/{ if($0 !~ /''^##contig=<ID=chr22[>,]/) next } /''^#CHROM/{print; exit} {print}'"' > calls_min_header_full.txt
bcftools reheader -h calls_min_header_full.txt -o calls_min.vcf.gz calls.vcf.gz
bcftools index -f calls_min.vcf.gz
ls -lh calls_min.vcf.gz*; bcftools index -n calls_min.vcf.gz; bcftools query -l calls_min.vcf.gz"
