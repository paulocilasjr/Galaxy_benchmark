/bin/bash -lc "bcftools view -h calls_one_sample.vcf.gz | awk 'BEGIN{keep=1} /"'^##contig=/{ if($0 !~ /''^##contig=<ID=chr22[>,]/) next } /''^#CHROM/{print; exit} {print}'"' > calls_min_header.txt
# Wait awk exits at #CHROM and prints it; ensure only one #CHROM
tail -1 calls_min_header.txt | head -c 200; echo
bcftools reheader -h calls_min_header.txt -o calls_rehead.vcf.gz calls_one_sample.vcf.gz
bcftools index -f calls_rehead.vcf.gz
bcftools gtcheck -g panel_one_sample.vcf.gz calls_rehead.vcf.gz 2>&1 | grep -E 'sites-|DCv2'"
