/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
bam=pysam.AlignmentFile('donor.bam','rb')
pos=[]
for read in bam.fetch('chr22'):
    if read.is_unmapped or read.is_secondary or read.is_supplementary or read.mapping_quality<30:
        continue
    pos.append(read.reference_start)
arr=np.array(pos)
print('count',len(arr),'min',arr.min(),'max',arr.max())
# depth using pileup
bam.close()
# histogram coarse
bins=np.arange(arr.min()//1000000*1000000, arr.max()+2000000, 1000000)
hist,_=np.histogram(arr,bins=bins)
for b,h in zip(bins,hist):
    print(b,h)
PY"
