/bin/bash -lc "mkdir -p tmp
cat > get_panel.py <<'PY'
import json,urllib.request,subprocess,os
r=json.load(open('region.json'));chrom=r['chrom']
start=max(r['start'],min(x[2] for x in r['hits'] if x[1]==chrom));end=start+1000000
base='https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20201028_3202_phased/'
filename='CCDG_14151_B01_GRM_WGS_2020-08-05_'+chrom+'.filtered.shapeit2-duohmm-phased.vcf.gz'
url=base+filename
urllib.request.urlretrieve(url+'.tbi','panel.vcf.gz.tbi')
reg=f'{chrom}:{start+1}-{end}'
print('Fetching',reg,url,flush=True)
os.environ['TMPDIR']=os.path.abspath('tmp')
subprocess.run(['bcftools','view','-r',reg,url+'##idx##panel.vcf.gz.tbi','-Ob','-o','panel.bcf'],check=True)
subprocess.run(['bcftools','index','panel.bcf'],check=True)
print('Panel ready',flush=True)
PY"
