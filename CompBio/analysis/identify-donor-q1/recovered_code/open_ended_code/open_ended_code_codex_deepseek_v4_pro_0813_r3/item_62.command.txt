/bin/bash -lc "samtools view -b -F 2304 -q 20 donor.bam | samtools depth -a - | python3 -c 'import sys; total=0; n=0; bs=be=None; blocks=[]
for line in sys.stdin:
 p,d=int(line.split()[1]),int(line.split()[2])
 if d<=0: continue
 n+=1; total+=d
 if bs is None: bs=be=p
 elif p==be+1: be=p
 else:
  if be-bs+1>=1000: blocks.append((bs,be,be-bs+1))
  bs=be=p
if bs is not None and be-bs+1>=1000: blocks.append((bs,be,be-bs+1))
print(\"covered_bases\",n,\"avg_depth\",total/n if n else 0,\"min\",min(b[0] for b in blocks),\"max\",max(b[1] for b in blocks),\"nblocks\",len(blocks))
print(\"first\",blocks[:5]); print(\"last\",blocks[-5:])'"
