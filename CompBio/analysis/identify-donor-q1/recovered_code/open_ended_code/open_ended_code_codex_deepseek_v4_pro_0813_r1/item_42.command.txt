/bin/bash -lc "bcftools annotate -x FORMAT/PL,FORMAT/AD donor_snps.vcf.gz -Oz -o donor_gt_only.vcf.gz && bcftools index -t donor_gt_only.vcf.gz && printf '##fileformat=VCFv4.3\\n##contig=<ID=chr22>\\n##FORMAT=<ID=GT,Number=1,Type=String,Description=\"Genotype\">\\n#CHROM\\tPOS\\tID\\tREF\\tALT\\tQUAL\\tFILTER\\tINFO\\tFORMAT\\treads.bam\\n' > hdr_minimal.txt && bcftools reheader -h hdr_minimal.txt donor_gt_only.vcf.gz -o donor_minimal.vcf.gz 2>/dev/null && bcftools index -t donor_minimal.vcf.gz && bcftools gtcheck -u GT,GT -g test_both.vcf.gz donor_minimal.vcf.gz 2>&1 | grep -E 'skipping|sites-compared|DCv2' | head -20"
