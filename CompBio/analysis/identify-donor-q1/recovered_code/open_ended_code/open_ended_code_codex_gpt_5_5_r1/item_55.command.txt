/bin/bash -lc "python3 - <<'PY'
import gzip, math
from work.score_donor import parse_bases
obs={}
for line in open('work/panel_sites.mpileup'):
    chrom,pos,ref,depth,bases,quals=line.rstrip('\\n').split('\\t')[:6]
    obs[(chrom,int(pos))]=bases
err=0.005
with gzip.open('work/reads.panel_likelihoods.vcf.gz','wt') as out, gzip.open('work/region.1kg.vcf.gz','rt') as vcf:
    out.write('##fileformat=VCFv4.2\\n')
    out.write('##FORMAT=<ID=PL,Number=G,Type=Integer,Description=\"Phred-scaled genotype likelihoods\">\\n')
    out.write('##FORMAT=<ID=DP,Number=1,Type=Integer,Description=\"Read depth\">\\n')
    out.write('#CHROM\\tPOS\\tID\\tREF\\tALT\\tQUAL\\tFILTER\\tINFO\\tFORMAT\\tquery\\n')
    for line in vcf:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t',9)
        chrom,pos,vid,ref,alt=p[0],int(p[1]),p[2],p[3].upper(),p[4].upper()
        if len(ref)"'!=1 or len(alt)!=1: continue
        bases=obs.get((chrom,pos))
        if bases is None: continue
        r,a,o=parse_bases(bases,ref,alt); total=r+a
        if total<4: continue
        probs=[(1-err,err),(0.5,0.5),(err,1-err)]
        lls=[r*math.log10(pr)+a*math.log10(pa) for pr,pa in probs]
        mx=max(lls)
        pls=[min(255, int(round(-10*(ll-mx)))) for ll in lls]
        out.write(f'"'{chrom}\\t{pos}\\t{vid}\\t{ref}\\t{alt}\\t.\\t.\\t.\\tPL:DP\\t{pls[0]},{pls[1]},{pls[2]}:{total}\\n')
PY
bcftools index -f -t work/reads.panel_likelihoods.vcf.gz
bcftools gtcheck -g work/region.1kg.vcf.gz work/reads.panel_likelihoods.vcf.gz > work/gtcheck_panel_likelihoods.txt
grep '"'^DC'"' work/gtcheck_panel_likelihoods.txt | sort -k4,4n | head -n 15"
