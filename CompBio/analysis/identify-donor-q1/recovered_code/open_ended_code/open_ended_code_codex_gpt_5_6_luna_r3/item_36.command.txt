/bin/bash -lc "samtools idxstats reads.mm.sorted.bam | sort -k3,3nr | head -20; echo 'Mapped coordinate summary:'; samtools view -F 4 -F 256 -F 2048 reads.mm.sorted.bam | awk 'BEGIN{OFS=\"\\t\"}{c["'$3]++; if(!lo[$3]||$4<lo[$3])lo[$3]=$4; if($4>hi[$3])hi[$3]=$4} END{for(x in c)print x,c[x],lo[x],hi[x]}'"' | sort -k2,2nr | head -20"
