/bin/bash -lc "samtools view -F 4 -q 20 reads.mm.sorted.bam | awk 'BEGIN{OFS=\"\\t\"}{c["'$3]++; if(!lo[$3]||$4<lo[$3])lo[$3]=$4; if($4>hi[$3])hi[$3]=$4} END{for(x in c)print x,c[x],lo[x],hi[x]}'"' | sort -k2,2nr | head -20; echo 'chr22 high-quality extrema/count'; samtools view -F 4 -q 20 reads.mm.sorted.bam 22:1-60000000 | awk 'BEGIN{n=0;min=1e12;max=0}{n++;if("'$4<min)min=$4;if($4>max)max=$4}END{print n,min,max}'"'; echo 'chr22 100kb bins q20'; samtools view -F 4 -q 20 reads.mm.sorted.bam 22:29000000-37000000 | awk 'BEGIN{OFS=\"\\t\"}{print int("'$4/100000)}'"' | sort | uniq -c | sort -k2n | head -100"
