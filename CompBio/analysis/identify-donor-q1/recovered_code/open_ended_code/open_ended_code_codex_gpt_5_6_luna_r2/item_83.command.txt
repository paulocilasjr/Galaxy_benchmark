import pysam
k=pysam.VariantFile('chr22.merged.vcf.gz'); q=pysam.VariantFile('reads.merged.called.vcf.gz')
for cand in ('HG03861','HG03851'):
  kk=pysam.VariantFile('chr22.merged.vcf.gz'); qq=pysam.VariantFile('reads.merged.called.vcf.gz')
  n=match=dis=0
  for kr,qr in zip(kk,qq):
    if (qr.samples[0].get('DP') or 0)<2: continue
    a=qr.samples[0].get('GT'); b=kr.samples[cand].get('GT')
    if a is None or b is None or None in a or None in b: continue
    n+=1
    if sorted(a)==sorted(b): match+=1
    else: dis+=1
  print(cand,'sites',n,'matches',match,'discordant',dis)
