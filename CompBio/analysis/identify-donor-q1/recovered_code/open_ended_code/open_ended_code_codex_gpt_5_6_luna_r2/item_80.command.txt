/bin/bash -lc "python3 -u - <<'PY'
import pysam, sys, time
known=pysam.VariantFile('chr22.merged.vcf.gz'); query=pysam.VariantFile('reads.merged.called.vcf.gz')
samples=list(known.header.samples); scores=[0.0]*len(samples); used=[0]*len(samples); qit=iter(query)
start=time.time(); n=0; scored=0; mismatch=0
for kr in known:
    qr=next(qit); n+=1
    if (kr.pos,kr.ref,kr.alts)"'!=(qr.pos,qr.ref,qr.alts): mismatch+=1; continue
    qs=qr.samples[0]; pl=qs.get('"'PL'); dp=qs.get('DP') or 0
    if dp<2 or pl is None: continue
    scored+=1; nalleles=1+len(kr.alts)
    # local sample proxy lookups are expensive but robust
    for i,samp in enumerate(samples):
        gt=kr.samples[samp].get('GT')
        if gt is None or len(gt)"'!=2 or gt[0] is None or gt[1] is None: continue
        a,b=gt
        if a<0 or b<0 or a>=nalleles or b>=nalleles: continue
        if a>b: a,b=b,a
        idx=b*(b+1)//2+a
        if idx>=len(pl): continue
        scores[i]+=float(pl[idx]); used[i]+=1
    if n%10000==0: print('"'processed',n,'scored',scored,'sec',round(time.time()-start,1),flush=True)
order=sorted(range(len(samples)),key=lambda i:scores[i])
print('SUMMARY records',n,'mismatch',mismatch,'scored',scored,'sec',round(time.time()-start,1),flush=True)
for rank,i in enumerate(order[:20],1): print(rank,samples[i],scores[i],used[i],scores[i]/used[i],flush=True)
print('margin',scores[order[1]]-scores[order[0]],flush=True)
PY"
