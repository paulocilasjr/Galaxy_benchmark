/bin/bash -lc "python3 - <<'PY'
import pysam, sys, math
known=pysam.VariantFile('chr22.merged.vcf.gz')
query=pysam.VariantFile('reads.merged.called.vcf.gz')
samples=list(known.header.samples); scores=[0.0]*len(samples); used=[0]*len(samples); qn=0; sites=0; dps=0; mismatch=0
qit=iter(query)
for kr in known:
    try: qr=next(qit)
    except StopIteration: print('query ended',kr.pos); break
    qn+=1
    if (kr.pos,kr.ref,kr.alts)"'!=(qr.pos,qr.ref,qr.alts):
        if mismatch<5: print('"'mismatch',kr.pos,qr.pos,kr.ref,qr.ref,kr.alts,qr.alts)
        mismatch+=1; continue
    qs=qr.samples[0]; dp=qs.get('DP') or 0; pl=qs.get('PL')
    if dp<2 or pl is None: continue
    dps+=1
    nalleles=1+len(kr.alts)
    for i,samp in enumerate(samples):
        gt=kr.samples[samp].get('GT')
        if gt is None or len(gt)"'!=2 or gt[0] is None or gt[1] is None: continue
        a,b=gt
        if a<0 or b<0 or a>=nalleles or b>=nalleles: continue
        if a>b: a,b=b,a
        idx=b*(b+1)//2+a
        if idx>=len(pl): continue
        scores[i]+=float(pl[idx]); used[i]+=1
    sites+=1
print('"'samples',len(samples),'known_query_records',qn,'mismatch',mismatch,'scored_sites',sites,'dp>=2',dps)
order=sorted(range(len(samples)),key=lambda i:scores[i])
for rank,i in enumerate(order[:20],1):
    print(rank,samples[i],f'{scores[i]:.1f}',used[i],f'avg={scores[i]/used[i]:.4f}')
print('margin',scores[order[1]]-scores[order[0]])
PY"
