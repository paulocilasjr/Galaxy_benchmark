/bin/bash -lc "samtools view -q 20 donor.hg38.bam chr22 | awk 'BEGIN{min=1e12;max=0} {if("'$4<min)min=$4; e=$4+length($10)-1; if(e>max)max=e; n++} END{print n,min,max,max-min+1}'"' && bcftools mpileup -f hg38.fa -r chr22:30000000-35000000 -q 20 -Q 20 -a FORMAT/DP,FORMAT/AD -Ou donor.hg38.bam | bcftools call -mv -Oz -o donor.calls.vcf.gz && bcftools index -t donor.calls.vcf.gz && bcftools view -H donor.calls.vcf.gz | wc -l && bcftools stats donor.calls.vcf.gz | grep '"'^SN'"'"
