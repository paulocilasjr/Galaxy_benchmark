import gzip, heapq
candidate='AACATGGAGCGCCGGG-1'
active=[]
events=[]
cur=None

    for line in f:
        chrom,s,e,bc,c=line.rstrip('\\n').split('\\t')
        if bc"'!=candidate: continue
        s=int(s); e=int(e)
        if chrom!=cur:
            active=[]; cur=chrom
        while active and active[0][0] <= s:
            heapq.heappop(active)
        depth=len(active)+1
        if depth>2:
            events.append((chrom,s,e,depth,[(a,b) for a,b in active[:5]]))
            if len(events)>=20: break
        heapq.heappush(active,(e,s))
print('"'events_shown',len(events))
for x in events:
    print(x)
