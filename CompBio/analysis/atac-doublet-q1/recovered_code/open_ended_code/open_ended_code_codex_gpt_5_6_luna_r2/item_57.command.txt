/bin/bash -lc "python3 - <<'PY'
import numpy as np
from sklearn.neighbors import NearestNeighbors
p=np.load('profile_1000.npz',allow_pickle=True); z=p['Z']; b=p['barcodes'];n=len(b);rng=np.random.default_rng(123)
for ns in [10000,50000,200000]:
 a=rng.integers(0,n,ns); c=rng.integers(0,n,ns); same=a==c
 while same.any(): c[same]=rng.integers(0,n,same.sum());same=a==c
 d=z[a]+z[c];d/=np.linalg.norm(d,axis=1,keepdims=True)
 # normalize real + synth, nearest euclidean
 allz=np.vstack([z,d]); nn=NearestNeighbors(n_neighbors=51,metric='euclidean',n_jobs=-1).fit(allz)
 dist,ix=nn.kneighbors(z)
 # exclude self real from first neighbor; count synthetic labels >=n
 scores=(ix[:,1:]>=n).sum(1)/50
 # synthetic fraction after density correction perhaps ratio to real neighbors within same radius
 print('\\nNS',ns,'score stats',np.percentile(scores,[0,50,90,95,99,100]))
 for i in np.argsort(-scores)[:30]:print(b[i],round(float(scores[i]),3),round(float(dist[i,1]),3),p['fragments'][i])
PY"
