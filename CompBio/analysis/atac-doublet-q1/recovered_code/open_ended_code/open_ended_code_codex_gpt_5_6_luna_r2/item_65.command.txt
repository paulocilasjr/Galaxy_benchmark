/bin/bash -lc "python3 - <<'PY'
import gzip,re,numpy as np,collections
p=np.load('profile_1000.npz',allow_pickle=True); bcs=p['barcodes']; bd={b:i for i,b in enumerate(bcs)}; nb=len(bcs)
# parse one promoter per gene name
prom=[]; seen=set(); attr_re=re.compile(r'gene_name \"(["'^"]+)"'"')
with gzip.open('gencode.v47.basic.annotation.gtf.gz','rt',errors='ignore') as h:
 for line in h:
  if not line or line[0]=='#': continue
  f=line.rstrip().split('\\t')
  if len(f)<9 or f[2]"'!='"'gene' or f[0] not in set(['chr'+str(i) for i in range(1,23)]+['chrX','chrY']): continue
  m=attr_re.search(f[8]);
  if not m: continue
  name=m.group(1)
  if name in seen: continue
  seen.add(name); s=int(f[3])-1; e=int(f[4]); t=s if f[6]=='+' else e-1
  prom.append((f[0],max(0,t-2000),t+2001,name))
print('genes',len(prom))
# bin index for intervals
binmap=collections.defaultdict(lambda:collections.defaultdict(list))
for gi,(ch,lo,hi,name) in enumerate(prom):
 for z in range(lo//1000, (hi-1)//1000+1): binmap[ch][z].append(gi)
C=np.zeros((nb,len(prom)),dtype=np.int16)
with gzip.open('atac.doublet.q1.bed.gz','rt') as h:
 for line in h:
  f=line.rstrip().split('\\t'); ch=f[0]; start=int(f[1]);end=int(f[2]); bi=bd[f[3]]; mid=(start+end)//2
  for gi in binmap.get(ch,{}).get(mid//1000,[]):
   lo,hi=prom[gi][1:3]
   if lo<=mid<hi: C[bi,gi]+=1
print('nonzero',np.count_nonzero(C),'max',C.max())
# marker summary and per cell module scores
markers=['MS4A1','CD79A','CD37','CD3D','CD3E','TRBC1','TRBC2','IL7R','LTB','MALAT1','NKG7','GNLY','GZMB','GZMH','LST1','LYZ','S100A8','S100A9','CTSS','FCN1','LGALS3','TYROBP','FCER1G','CLEC10A','FCER1A','CST3','HLA-DRA','HLA-DPB1','HLA-DPA1','GZMK','CCL5','CD8A','CD8B','CCR7','MKI67','PPBP','PF4','GATA3','KLRD1','CD14','CD74','CD68','B2M','IRF7','IFITM3']
gd={x:i for i,x in enumerate([x[3] for x in prom])}
for x in markers:
 if x in gd:
  j=gd[x]; vals=C[:,j]; print(x,'mean',round(float(vals.mean()),2),'max',int(vals.max()),'top',[(bcs[i],int(vals[i])) for i in np.argsort(-vals)[:3]])
# module cell scores: log1p counts standardized per gene then sums
mods={'B':['MS4A1','CD79A','CD37','CD74','HLA-DRA'],'T':['CD3D','CD3E','TRBC1','TRBC2','IL7R','CCR7'],'NK':['NKG7','GNLY','GZMB','KLRD1','CCL5','GZMK'],'Mono':['LST1','LYZ','S100A8','S100A9','CTSS','FCN1','TYROBP','FCER1G','LGALS3'],'DC':['FCER1A','CLEC10A','CST3','HLA-DPA1','HLA-DPB1']}
S={}
for mod,genes in mods.items():
 inds=[gd[x] for x in genes if x in gd]; X=np.log1p(C[:,inds].astype(float));
 # gene-wise z to handle promoter accessibility differences
 X=(X-X.mean(0))/(X.std(0)+1e-6);S[mod]=X.mean(1)
 print('\\n',mod,'top',[(bcs[i],round(float(S[mod][i]),2),int(C[i,inds].sum())) for i in np.argsort(-S[mod])[:15]])
# combinations: high scores in two modules
for m1,m2 in [('B','T'),('B','Mono'),('T','Mono'),('T','NK'),('Mono','DC')]:
 x=S[m1]+S[m2]
 print('COMBO',m1,m2,[(bcs[i],round(float(S[m1][i]),2),round(float(S[m2][i]),2),round(float(x[i]),2)) for i in np.argsort(-x)[:15]])
PY"
