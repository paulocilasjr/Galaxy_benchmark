/bin/bash -lc "python3 - <<'PY'
# repeat equal sum excluding self and report top best and ratios
import numpy as np
p=np.load('profile_1000.npz',allow_pickle=True); z=p['Z']; b=p['barcodes']; n=len(b); s=z@z.T; np.fill_diagonal(s,-1); single=s.max(1)
ii,jj=np.triu_indices(n,1); mask=s[ii,jj]<0.7;ii=ii[mask];jj=jj[mask]
best=np.full(n,-1.);arg=np.zeros((n,2),int)
for st in range(0,len(ii),5000):
 a=ii[st:st+5000];c=jj[st:st+5000];d=z[a]+z[c];d/=np.linalg.norm(d,axis=1)[:,None];v=z@d.T; q=np.arange(n)[:,None];v[(q==a)|(q==c)]=-1; ix=v.argmax(1);vv=v[np.arange(n),ix];m=vv>best;best[m]=vv[m];arg[m]=np.c_[a[ix[m]],c[ix[m]]]
for metric,x in [('best',best),('diff',best-single),('ratio',best/(single+1e-9))]:
 print('\\n',metric)
 for i in np.argsort(-x)[:50]:print(b[i],round(float(x[i]),4),round(float(best[i]),4),round(float(single[i]),4),b[arg[i,0]],b[arg[i,1]])
PY"
