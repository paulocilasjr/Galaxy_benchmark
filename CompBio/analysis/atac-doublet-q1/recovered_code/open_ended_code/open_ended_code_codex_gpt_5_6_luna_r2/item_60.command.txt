/bin/bash -lc "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 python3 - <<'PY'
import numpy as np
from sklearn.cluster import KMeans
p=np.load('profile_1000.npz',allow_pickle=True);z=p['Z']; b=p['barcodes']; f=p['fragments']
s=z@z.T;np.fill_diagonal(s,-1);o=np.argsort(-s,axis=1)
for k in [5,10,20]:
 c=KMeans(k,n_init=3,random_state=1).fit_predict(z)
 print('\\nK',k,flush=True)
 for r in [10,30]:
  ent=[]
  for i in range(len(b)):
   h=np.bincount(c[o[i,:r]],minlength=k);q=h/h.sum();ent.append(-(q[q>0]*np.log2(q[q>0])).sum())
  ent=np.array(ent)
  print('r',r,flush=True)
  for i in np.argsort(-ent)[:15]:print(b[i],round(float(ent[i]),3),np.bincount(c[o[i,:r]],minlength=k).tolist(),f[i],flush=True)
PY"
