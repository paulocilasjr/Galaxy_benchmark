/bin/bash -lc "python3 - <<'PY'
import numpy as np
from sklearn.cluster import KMeans
p=np.load('profile_1000.npz',allow_pickle=True); z=p['Z']; b=p['barcodes']; n=len(b)
s=z@z.T; np.fill_diagonal(s,-1)
ss=np.max(s,axis=1)
# all pairs with similarity below thresholds, calculate synthetic sum in LSI
rng=np.random.default_rng(4)
for threshold in [0.5,0.6,0.7,0.75]:
 ii,jj=np.triu_indices(n,1); mask=s[ii,jj]<threshold
 ii=ii[mask];jj=jj[mask]
 # all pairs can fit approx 5e5 x 40; batches
 best=np.full(n,-1.0); arg=np.empty((n,2),int)
 for st in range(0,len(ii),10000):
  a=z[ii[st:st+10000]]; c=z[jj[st:st+10000]]
  d=a+c; d/=np.linalg.norm(d,axis=1,keepdims=True)
  vals=z@d.T
  ix=np.argmax(vals,axis=1); vv=vals[np.arange(n),ix]
  m=vv>best; best[m]=vv[m]; arg[m,0]=ii[st:st+10000][ix[m]];arg[m,1]=jj[st:st+10000][ix[m]]
 score=best-ss
 print('\\nthreshold',threshold,'pairs',len(ii),'candidate',best[369],ss[369],score[369],b[arg[369,0]],b[arg[369,1]])
 print('top score')
 for i in np.argsort(-score)[:20]: print(b[i],round(float(score[i]),4),round(float(best[i]),4),round(float(ss[i]),4),b[arg[i,0]],b[arg[i,1]])
PY"
