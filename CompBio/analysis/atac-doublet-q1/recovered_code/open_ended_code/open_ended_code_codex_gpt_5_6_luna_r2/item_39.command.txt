/bin/bash -lc "python3 - <<'PY'
import numpy as np
p=np.load('profile_1000.npz',allow_pickle=True); z=p['Z']; b=p['barcodes']; n=len(b)
s=z@z.T; np.fill_diagonal(s,-1); ss=np.max(s,axis=1)
ii,jj=np.triu_indices(n,1)
for threshold in [0.5,0.6,0.7]:
 mask=s[ii,jj]<threshold; ii0=ii[mask]; jj0=jj[mask]
 best=np.full(n,-1.0); arg=np.empty((n,2),int)
 for st in range(0,len(ii0),10000):
  ia=ii0[st:st+10000]; ja=jj0[st:st+10000]
  d=z[ia]+z[ja]; d/=np.linalg.norm(d,axis=1,keepdims=True)
  vals=z@d.T
  vals[ia,np.arange(len(ia))]=-1
  vals[ja,np.arange(len(ja))]=-1
  ix=np.argmax(vals,axis=1); vv=vals[np.arange(n),ix]
  m=vv>best; best[m]=vv[m]; arg[m,0]=ia[ix[m]];arg[m,1]=ja[ix[m]]
 score=best-ss
 print('\\nthreshold',threshold,'pairs',len(ii0),'candidate',best[369],ss[369],score[369],b[arg[369,0]],b[arg[369,1]])
 for i in np.argsort(-score)[:30]: print(b[i],round(float(score[i]),4),round(float(best[i]),4),round(float(ss[i]),4),b[arg[i,0]],b[arg[i,1]])
PY"
