/bin/bash -lc "python3 - <<'PY'
import gzip, collections, statistics
p='atac.doublet.q1.bed.gz'
c=collections.Counter()
with gzip.open(p,'rt') as f:
    for line in f:
        x=line.split('\\t')
        c[x[3]] += 1
v=sorted(c.values())
print('cells',len(v),'fragments',sum(v),'min',v[0],'max',v[-1])
for q in [0,.001,.005,.01,.05,.1,.25,.5,.75,.9,.95,.99,.995,.999,1]:
    print(q, v[min(len(v)-1,int(q*(len(v)-1)))])
print('top')
for b,n in c.most_common(100): print(n,b)
PY"
