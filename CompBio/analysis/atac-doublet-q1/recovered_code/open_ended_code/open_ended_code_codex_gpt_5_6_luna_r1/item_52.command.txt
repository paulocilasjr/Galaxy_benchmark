/bin/bash -lc "python3 -u - <<'PY'
from doublet_scan import load_fragments
import numpy as np
x,b,n=load_fragments('atac.doublet.q1.bed.gz',5000,'mid')
S=np.asarray((x@x.T).toarray(),dtype=np.float32); np.fill_diagonal(S,-1)
N=len(b);p,q=np.triu_indices(N,1);pd=S[p,q];den=np.sqrt(2+2*pd)
# actual fragment counts from row nnz isn't counts; use supplied total via known rows? derive not needed
for fq in [.01,.05,.1,.2,.5,.8,1.0]:
 th=np.quantile(pd,fq) if fq<1 else 2
 valid=pd<=th
 fit=np.full(N,-1.,np.float32); pp=np.zeros((N,2),np.int32)
 for i in range(N):
  z=(S[i,p]+S[i,q])/den;z[~valid]=-2;z[(p==i)|(q==i)]=-2
  j=z.argmax();fit[i]=z[j];pp[i]=p[j],q[j]
 order=np.argsort(fit)[::-1];rank=np.empty(N);rank[order]=np.arange(1,N+1)
 print('\\nfq',fq,'th',round(float(th),5),'fit quant',np.round(np.quantile(fit,[.01,.5,.9,.95,.99,1]),5))
 for name in ['CCGTAATCATGTAAAA-1','CGCACGCGTACTGCAG-1','CACGGACAGCCCGCCT-1','TACTCCTATGTTCTGC-1','GCTTGCACGCCTTCTC-1','ACTTCTTACGCTCTCA-1','TCATCCCAAACTTCAC-1','GTTGATGTAGGGGCGG-1']:
  i=b.index(name);print(name,'rank',int(rank[i]),'fit',round(float(fit[i]),5),'parents',b[pp[i,0]],b[pp[i,1]],'pd',round(float(S[pp[i,0],pp[i,1]]),5))
 print('top',[(b[i],round(float(fit[i]),5)) for i in order[:10]])
PY"
