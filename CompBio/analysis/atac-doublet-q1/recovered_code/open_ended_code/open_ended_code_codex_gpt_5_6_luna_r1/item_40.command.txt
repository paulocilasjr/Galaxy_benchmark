/bin/bash -lc "python3 -u - <<'PY'
from doublet_scan import load_fragments
import numpy as np
x,b,n=load_fragments('atac.doublet.q1.bed.gz',1000,'start')
df=np.bincount(x.indices,minlength=x.shape[1])
for lo,hi in [(1,1000),(2,1000),(5,1000),(10,1000),(20,1000),(10,200),(20,200),(50,200),(2,50)]:
 mask=(df>=lo)&(df<=hi); y=x[:,mask]; y=y.multiply(1/np.maximum(np.sqrt(np.asarray(y.multiply(y).sum(1)).ravel()),1e-12)[:,None]).tocsr()
 g=np.asarray((y@y.T).toarray());np.fill_diagonal(g,np.nan)
 print(lo,hi,'feat',mask.sum(),'nnz',y.nnz,'q',np.round(np.nanquantile(g,[.01,.1,.5,.9,.99]),4),'topmean q',np.round(np.quantile(np.nanmean(np.sort(g,axis=1)[:,-11:-1],1),[0,.1,.5,.9,1]),4),flush=True)
 vals=np.linalg.eigvalsh(np.nan_to_num(g,nan=0)); print('eig',np.round(vals[-8:],3),flush=True)
PY"
