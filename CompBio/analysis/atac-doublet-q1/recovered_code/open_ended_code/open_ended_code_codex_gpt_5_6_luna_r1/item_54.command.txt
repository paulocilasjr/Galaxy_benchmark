/bin/bash -lc "curl -L --fail --silent --show-error -o pbmc10k_peaks.h5 https://cf.10xgenomics.com/samples/cell-atac/1.0.1/atac_v1_pbmc_10k/atac_v1_pbmc_10k_filtered_peak_bc_matrix.h5 && ls -lh pbmc10k_peaks.h5 && python3 - <<'PY'

with h5py.File('pbmc10k_peaks.h5','r') as f:

 def rec(g,p=''):
  for k,v in g.items():
   if hasattr(v,'shape'): print(p+k,v.shape,v.dtype)
   else: rec(v,p+k+'/')
 rec(f)
