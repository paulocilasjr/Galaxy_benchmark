/bin/bash -lc "python3 - <<'PY'
import gzip,collections,numpy as np
c=collections.Counter()
with gzip.open('atac.doublet.q1.bed.gz','rt') as f:
 for l in f:c[l.split('\\t')[3]]+=1
v=np.array(sorted(c.values()))
print('unique counts',len(set(v)),'range',v.min(),v.max())
print('diff q',np.quantile(np.diff(v),[0,.1,.5,.9,1]),'duplicates counts',collections.Counter(v).most_common(20))
print('first sorted',v[:100]); print('last',v[-100:])
PY"
