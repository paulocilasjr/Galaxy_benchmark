/bin/bash -lc "python3 -u - <<'PY'
from doublet_scan import load_fragments
import numpy as np
x,b,n=load_fragments('atac.doublet.q1.bed.gz',1000,'start')
G=np.asarray((x@x.T).toarray());np.fill_diagonal(G,-1)
# summary of similarity and rank
mean=G.mean(1); top=np.sort(G,axis=1)[:,-11:].mean(1); mx=G.max(1); q=np.quantile(G,[.1,.5,.9],axis=1).T
u=np.diff(x.indptr)
print('candidate summaries')
for i in np.argsort(u)[-20:][::-1]: print(b[i],'u',u[i],'mean',round(mean[i],5),'top10',round(top[i],5),'max',round(mx[i],5),'q',np.round(q[i],5),'peer',b[G[i].argmax()])
print('low top / low max')
for name,z in [('top',top),('max',mx),('mean',mean)]:
 print(name,[(b[i],round(float(z[i]),5),int(u[i])) for i in np.argsort(z)[:20]])
PY"
