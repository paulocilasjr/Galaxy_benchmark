/bin/bash -lc "python3 - <<'PY'
import snapatac2 as snap, numpy as np
adata=snap.read('atac.h5ad')
names=adata.obs_names
clusters=np.array(adata.obs['leiden'].to_list())
scores=np.array(adata.obs['doublet_score'].to_list())
D=adata.obsp['distances'][:].tocsr()
# build nearest neighbors for selected candidates
for target in ['GTCAAAGCCCGTTCGG-1','TGAATTTGACCACCAC-1','GTTGCGGTTCCGCTCG-1','ACGGAGCCTGACTGGA-1','GCCGAGCGCTAGCCAT-1','TTTTTCGTTGGGAATT-1','CCGTAATCATGTAAAA-1']:
 if target in names:
  i=names.index(target)
  row=D.getrow(i); idx=row.indices
  from collections import Counter
  cnt=Counter(clusters[idx].tolist())
  print('\\n',target,'i',i,'clust',clusters[i],'score',scores[i],'neighbors')
  for k,v in sorted(cnt.items(), key=lambda x:-x[1]): print('  c',k,v)
  # print neighbor names distances
  for j in idx:
   print('   ',names[j],clusters[j],round(row.data[list(idx).index(j)] if False else D[i,j],3))
  break
PY"
