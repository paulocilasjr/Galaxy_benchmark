/bin/bash -lc "python3 - <<'PY'
import gzip, collections, time
f='atac.doublet.q1.bed.gz'
barc=collections.defaultdict(int); chroms=collections.defaultdict(lambda:[10**18,0])
t=time.time()
n=0
with gzip.open(f,'rt') as fh:
 for line in fh:
  parts=line.rstrip('\\n').split('\\t')
  ch=parts[0]; s=int(parts[1]); e=int(parts[2]); b=parts[3]
  barc[b]+=1
  cmin,cmax=chroms[ch]
  if s<cmin: cmin=s
  if e>cmax: cmax=e
  chroms[ch]=(cmin,cmax)
  n+=1
  if n%2000000==0: print(n,time.time()-t)
print('lines',n,'barcodes',len(barc),'chroms',len(chroms))
print('chrom lengths derived')
for ch,(mn,mx) in sorted(chroms.items(), key=lambda x:(len(x[0]),x[0])):
 print(ch,mn,mx,mx-mn+1)
PY"
