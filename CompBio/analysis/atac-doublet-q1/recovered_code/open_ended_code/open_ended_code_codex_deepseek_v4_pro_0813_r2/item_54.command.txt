/bin/bash -lc "python3 - <<'PY'
import snapatac2 as snap, numpy as np, matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
adata=snap.read('atac.h5ad')
X=adata.obsm['X_umap']
clusters=np.array(adata.obs['leiden'].to_list())
scores=np.array(adata.obs['doublet_score'].to_list())
names=adata.obs_names
fig,ax=plt.subplots(1,2,figsize=(16,6))
for c in sorted(set(clusters)):
 idx=clusters==c
 ax[0].scatter(X[idx,0],X[idx,1],s=3,label=str(c))
ax[0].legend(fontsize=6,bbox_to_anchor=(1,1))
sc=ax[1].scatter(X[:,0],X[:,1],c=scores,s=3,cmap='viridis')
plt.colorbar(sc,ax=ax[1])
# annotate top score 20
pairs=sorted(zip(scores,names,range(len(names))), reverse=True)
for s,n,i in pairs[:15]:
 ax[1].annotate(n, (X[i,0],X[i,1]), fontsize=5, alpha=.8)
# annotate high count top and low same GTC
for target in ['GTCAAAGCCCGTTCGG-1','CCGTAATCATGTAAAA-1']:
 i=names.index(target)
 ax[0].annotate(target, (X[i,0],X[i,1]), fontsize=6, color='red')
 ax[1].annotate(target, (X[i,0],X[i,1]), fontsize=6, color='red')
plt.savefig('umap.png', dpi=150, bbox_inches='tight')
print('saved umap.png')
PY"
