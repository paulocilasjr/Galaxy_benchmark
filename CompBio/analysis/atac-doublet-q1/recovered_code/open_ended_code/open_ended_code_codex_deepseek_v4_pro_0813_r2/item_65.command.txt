/bin/bash -lc "python3 - <<'PY'
import snapatac2 as snap, time
adata=snap.read('atac_tile5k.h5ad')
t=time.time()
snap.pp.select_features(adata, n_features=200000, filter_lower_quantile=0.005, filter_upper_quantile=0.005, max_iter=1, inplace=True, n_jobs=8)
print('select',time.time()-t)
print('n selected', adata.var['selected'].sum())
t=time.time()
snap.pp.scrublet(adata, features='selected', n_comps=15, sim_doublet_ratio=2.0, expected_doublet_rate=0.05, random_state=0, inplace=True, n_jobs=8, verbose=False)
print('scrub',time.time()-t)
names=adata.obs_names
scores=adata.obs['doublet_score'].to_list(); probs=adata.obs['doublet_probability'].to_list()
pairs=sorted(zip(scores,probs,names), reverse=True)
for s,p,n in pairs[:30]: print(n,s,p)
print('sim stats')
sim=adata.uns['scrublet_sim_doublet_score']
import numpy as np
print(sim.min(), sim.max(), np.quantile(sim,[.1,.5,.9]))
print('obs max',max(scores))
PY"
