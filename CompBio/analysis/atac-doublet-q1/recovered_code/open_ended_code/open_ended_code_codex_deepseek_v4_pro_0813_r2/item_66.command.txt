/bin/bash -lc "python3 - <<'PY'
import numpy as np, snapatac2 as snap
adata=snap.read('atac.h5ad')
names=adata.obs_names
for seed in [0,1,2]:
 a=np.load(f'scores_obs_seed{seed}.npy')
 pairs=sorted(zip(a,names), reverse=True)
 print('seed',seed,'top10')
 for s,n in pairs[:10]: print(' ',n,s)
# correlations
a0=np.load('scores_obs_seed0.npy');a1=np.load('scores_obs_seed1.npy');a2=np.load('scores_obs_seed2.npy')
print('corr01',np.corrcoef(a0,a1)[0,1],'corr02',np.corrcoef(a0,a2)[0,1],'corr12',np.corrcoef(a1,a2)[0,1])
# rank correlations
from scipy.stats import spearmanr
print('spearman',spearmanr(a0,a1)[0],spearmanr(a0,a2)[0],spearmanr(a1,a2)[0])
# max score and consistency
mx=np.maximum.reduce([a0,a1,a2])
pairs=sorted(zip(mx,names,a0,a1,a2), reverse=True)
print('max across seeds top')
for m,n,x,y,z in pairs[:20]: print(n,m,x,y,z)
PY"
