/bin/bash -lc "python3 - <<'PY'
import snapatac2 as snap, numpy as np, math
from collections import Counter
adata=snap.read('atac.h5ad')
names=adata.obs_names
clusters=np.array(adata.obs['leiden'].to_list())
scores=np.array(adata.obs['doublet_score'].to_list())
probs=np.array(adata.obs['doublet_probability'].to_list())
counts=np.array(adata.obs['n_fragment'].to_list())
D=adata.obsp['distances'][:].tocsr()
rows=[]
for i in range(len(names)):
 row=D.getrow(i); idx=row.indices
 cnt=Counter(clusters[idx].tolist())
 total=len(idx)
 ent=0
 for c,v in cnt.items():
  p=v/total; ent-=p*math.log(p+1e-12)
 # normalized entropy by log(total)
 ent/=math.log(total)
 # concentration max cluster
 maxp=max(cnt.values())/total
 # cluster size
 # second max
 second=sorted(cnt.values(), reverse=True)[1] if len(cnt)>1 else 0
 rows.append((ent,maxp,scores[i],probs[i],counts[i],names[i],clusters[i]))
# sort by entropy then score
for r in sorted(rows, key=lambda t:(-t[0],-t[2]))[:50]:
 print(r[5], 'cl',r[6], 'ent',round(r[0],3),'maxp',round(r[1],2),'scrub',round(r[2],3),'count',r[4])
print('\\nhigh entropy and high scrub')
for r in sorted(rows, key=lambda t:(-t[0]*t[2]))[:50]:
 print(r[5],r[6],round(r[0],3),round(r[2],3),r[3])
PY"
