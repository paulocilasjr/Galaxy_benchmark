/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
adata=snap.read('atac.snap.h5ad')
names=np.array(adata.obs_names)
score=np.array(adata.obs['doublet_score'])
leiden=np.array(adata.obs['leiden'])
# print top 40 with cluster
idx=np.argsort(-score)
for i in idx[:50]:
    print(names[i], score[i], leiden[i])
# distribution of scores in each cluster
for cl in np.unique(leiden):
    vals=score[leiden==cl]
    print(cl, len(vals), vals.mean().round(4), np.quantile(vals,[.5,.75,.9,.95,1]).round(4))
PY"
