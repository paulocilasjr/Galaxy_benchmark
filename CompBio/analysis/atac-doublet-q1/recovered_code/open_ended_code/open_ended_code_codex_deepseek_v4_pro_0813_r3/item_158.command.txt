/bin/bash -c "cat > snap_single.py <<'PY'
import snapatac2 as snap, numpy as np, time
frags='atac.doublet.q1.bed.gz'
adata=snap.pp.import_fragments(frags, chrom_sizes=snap.genome.GRCh38.chrom_sizes, is_paired=False, min_num_fragments=0, sorted_by_barcode=False, file='atac.snap.single.h5ad')
snap.pp.add_tile_matrix(adata, bin_size=500, inplace=True)
snap.pp.select_features(adata, n_features=50000, filter_lower_quantile=0.005, filter_upper_quantile=0.005)
snap.pp.scrublet(adata, expected_doublet_rate=0.06, sim_doublet_ratio=2.0, n_comps=30, random_state=0)
p=np.array(adata.obs['doublet_probability']); s=np.array(adata.obs['doublet_score']); names=np.array(adata.obs_names)
print('max p',p.max(),'p>0.5',(p>0.5).sum(),'p>0.1',(p>0.1).sum(),'s max',s.max(),'s q',np.quantile(s,[.5,.9,.95,.99]))
idx=np.argsort(-s)
for i in idx[:40]: print(names[i],p[i],s[i])
PY
python3 snap_single.py"
