/bin/bash -c "python3 - <<'PY'
import numpy as np
p=np.loadtxt('scrublet_peaks_scores.txt',dtype=str)
t=np.loadtxt('scrublet_tile_scores.txt',dtype=str)
ps=p[:,0]; pp=p[:,1].astype(float)
ts=t[:,0]; tp=t[:,1].astype(float)
common=set(ps) & set(ts)
rows=[]
for b in common:
    i=np.where(ps==b)[0][0]; j=np.where(ts==b)[0][0]
    rows.append((b,pp[i],tp[j],pp[i]+tp[j]))
for r in sorted(rows,key=lambda x:-x[3]): print(r)
PY"
