/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np, time
adata=snap.read('atac.snap.h5ad')
# ensure selected features? select_features already ran and stored. KNN uses X_spectral by default, computed by scrublet? maybe uns['spectral']?
print('obsm keys', list(adata.obsm.keys()))

snap.pp.knn(adata, n_neighbors=30, use_dims=30, inplace=True)
print('knn done',time.time()-t)
snap.tl.leiden(adata, resolution=1.0, inplace=True)
print('leiden done')
# inspect obs keys
for key in ['leiden']:
 try:
  arr=np.array(adata.obs[key]); print(key, np.unique(arr, return_counts=True))
 except Exception as e: print(key,'err',e)
# top score per cluster
score=np.array(adata.obs['doublet_score'])
leiden=np.array(adata.obs['leiden'])
for cl in np.unique(leiden):
 mask=leiden==cl
 print(cl, mask.sum(), score[mask].mean(), score[mask].max())
PY"
