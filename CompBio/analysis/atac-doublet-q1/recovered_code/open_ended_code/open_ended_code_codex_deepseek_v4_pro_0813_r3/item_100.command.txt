/bin/bash -c "cat > scrublet_tile.py <<'PY'
import numpy as np, snapatac2 as snap, scrublet as scr
adata=snap.read('atac.snap.h5ad')
feat=np.array(adata.var['selected'])
print('selected sum',feat.sum())
X=adata.X[:, feat]
print('X shape',X.shape,'nnz',X.nnz)
# run scrublet raw counts
scrub=scr.Scrublet(X, expected_doublet_rate=0.06, sim_doublet_ratio=2.0)
scores,pred=scrub.scrub_doublets(min_counts=1,min_cells=2,min_gene_variability_pctl=85,use_approx_neighbors=False,n_prin_comps=30)
print('pred rate',pred.mean(),'score q',np.quantile(scores,[0,.25,.5,.75,.9,.95,.99,1]),'max',scores.max())
names=np.array(adata.obs_names)
idx=np.argsort(-scores)
for i in idx[:80]: print(names[i], scores[i], pred[i])
np.savetxt('scrublet_tile_scores.txt',np.c_[names,scores,pred.astype(int)],fmt='%s')
PY
python3 scrublet_tile.py"
