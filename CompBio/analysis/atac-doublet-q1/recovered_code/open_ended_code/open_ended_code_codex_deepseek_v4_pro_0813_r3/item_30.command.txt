/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
adata=snap.read('atac.snap.h5ad')
obs=adata.obs
p=np.array(obs['doublet_probability'])
s=np.array(obs['doublet_score'])
names=np.array(adata.obs_names)
idx=np.argsort(-s)
for i in idx[:100]:
    print(names[i], p[i], s[i])
print('score quantiles', np.quantile(s,[.5,.75,.9,.95,.99,1]))
PY"
