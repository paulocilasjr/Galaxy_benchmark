/bin/bash -c "python3 - <<'PY'
import snapatac2 as snap, numpy as np
from scipy.io import mmwrite
adata=snap.read('atac_peaks_scrub.h5ad')
X=adata.X[:].tocsr().astype(np.float64)
mmwrite('peak_matrix.mtx', X)
np.savetxt('peak_barcodes.txt', np.array(adata.obs_names), fmt='%s')
# genes/peaks names
var_names=list(adata.var_names)
with open('peak_features.txt','w') as f:
 for v in var_names: f.write(v+'\\n')
print('saved',X.shape,X.nnz)
PY
ls -lh peak_matrix.mtx peak_barcodes.txt peak_features.txt"
