/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, snapatac2 as snap
D=pd.read_csv('scDblFinder_results.tsv',sep='\\t')
print(D.head())
print('called',D['scDblFinder.class'].value_counts())
print(D.sort_values('scDblFinder.score',ascending=False).head(80))

print('\\nTop 100 scDblFinder with other scores:')
print(D.sort_values('scDblFinder.score',ascending=False).head(100).to_string(index=False))
# Check our candidate barcodes
for b in ['GCGACACGCCGGGCAC-1','CGAGCAATAGAGGTGG-1','CGCACGCGTACTGCAG-1','GACTATAATGATTTTT-1','TGAGCCTCACGCTTCG-1','GGCGACCTCTACGTAG-1','CGTGTGGATCTGGCAG-1','TTCGAGACGTAGAAAA-1','CACGGTTTTCATACAA-1','CATCTGGAAAGTAGGC-1']:
 row=D[D.barcode==b]
 print('candidate',b,row[['scDblFinder.score','scDblFinder.class','nf','snap_s','peak_scrub','tile_scrub']].to_dict('records'))
PY"
