/bin/bash -c "python3 - <<'PY'
import numpy as np, snapatac2 as snap
from scipy.sparse import csr_matrix
# load peak matrix residual and scrub scores
ap=snap.read('atac_peaks_scrub.h5ad'); X=ap.X[:].tocsr().astype(np.float32); X.data[:]=1; N=np.asarray(X.sum(axis=1)).ravel(); names=np.array(ap.obs_names)
at=snap.read('atac.snap.h5ad'); names2=np.array(at.obs_names); nf=np.array(at.obs['n_fragment']); idx=[np.where(names2==n)[0][0] for n in names]; nf=nf[idx]
coef=np.polyfit(nf,N,1); res=N-np.polyval(coef,nf)
p=np.loadtxt('scrublet_peaks_scores.txt',dtype=str); ps={b:float(s) for b,s in zip(p[:,0],p[:,1])}
t=np.loadtxt('scrublet_tile_scores.txt',dtype=str); ts={b:float(s) for b,s in zip(t[:,0],t[:,1])}
# ranks (higher better) for each metric
def rank(arr): return arr.argsort().argsort()/(len(arr)-1)
r_res=rank(res); r_ps=rank(np.array([ps.get(b,np.nan) for b in names])); r_ts=rank(np.array([ts.get(b,np.nan) for b in names]))
comb=(r_res+r_ps+r_ts)/3
order=np.argsort(-comb)
for i in order[:50]:
 b=names[i]
 print(b,'comb',round(comb[i],3),'res',round(res[i],1),'peak_scrub',round(ps.get(b,np.nan),3),'tile_scrub',round(ts.get(b,np.nan),3),'nf',nf[i])
PY"
