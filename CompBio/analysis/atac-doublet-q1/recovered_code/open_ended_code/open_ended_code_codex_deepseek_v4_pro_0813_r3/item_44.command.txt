/bin/bash -c "python3 - <<'PY'
import requests
url='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main'
r=requests.get(url, timeout=30)
print(r.status_code)
data=r.json()
for item in data:
    if 'answer' in item.get('path','').lower() or 'ground' in item.get('path','').lower() or 'eval' in item.get('path','').lower() or 'solution' in item.get('path','').lower():
        print(item)
print('total',len(data))
print(data[:5])
PY"
