/bin/bash -c "python3 - <<'PY'
import numpy as np, snapatac2 as snap
configs={}
for b in [500,1000,5000,10000]:
    if b==500:
        adata=snap.read('atac.snap.h5ad'); names=np.array(adata.obs_names); s=np.array(adata.obs['doublet_score'])
    else:
        adata=snap.read(f'atac.snap.bin{b}.h5ad'); names=np.array(adata.obs_names); s=np.array(adata.obs['doublet_score'])
    # rank descending, smaller better
    order=np.argsort(-s)
    rank={name:i+1 for i,name in enumerate(names[order])}
    configs[b]=rank
ap=snap.read('atac_peaks_scrub.h5ad'); names=np.array(ap.obs_names); s=np.array(ap.obs['doublet_score']); order=np.argsort(-s); configs['peak']={name:i+1 for i,name in enumerate(names[order])}
common=set.intersection(*[set(r) for r in configs.values()])
rows=[]
for b in common:
    avg=np.mean([configs[k][b] for k in configs])
    rows.append((avg,b,[configs[k][b] for k in ['peak',500,1000,5000,10000]]))
for avg,b,ranks in sorted(rows)[:30]:
    print(b, avg, ranks)
PY"
