/bin/bash -c "python3 - <<'PY'
import numpy as np, snapatac2 as snap
adata=snap.read('atac.snap.h5ad')
names=np.array(adata.obs_names); leiden_raw=np.array(adata.obs['leiden'])
uniq,leiden=np.unique(leiden_raw,return_inverse=True)
D=adata.obsp['distances'].tocsr()
def ent(b):
 j=np.where(names==b)[0][0]
 inds=D.indices[D.indptr[j]:D.indptr[j+1]]; labs=leiden[inds]
 counts=np.bincount(labs,minlength=len(uniq)); p=counts/counts.sum()
 return -(p[p>0]*np.log(p[p>0])).sum(), p.max(), counts
for b in ['CGAGCAATAGAGGTGG-1','CGCGTCGCCACGTTAG-1','CATCTGGAAAGTAGGC-1','ACTACCAACGACTCTG-1','CGATATGGACGAGCGG-1','GCGACACGCCGGGCAC-1','CGCACGCGTACTGCAG-1','GACTATAATGATTTTT-1','TGAGCCTCACGCTTCG-1','GGCGACCTCTACGTAG-1','CGTGTGGATCTGGCAG-1','TTCGAGACGTAGAAAA-1']:
 e,p,c=ent(b)
 print(b,'entropy',round(e,3),'purity',round(p,3),'counts',c)
PY"
