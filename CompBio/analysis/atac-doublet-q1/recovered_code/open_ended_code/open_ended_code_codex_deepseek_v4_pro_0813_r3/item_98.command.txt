/bin/bash -c "python3 - <<'PY'
import numpy as np, pandas as pd
# load snap peak scores from saved? We didn't save snap peak scores. But atac_peaks_scrub.h5ad has snap scores.
import snapatac2 as snap
adata=snap.read('atac_peaks_scrub.h5ad')
snap_names=np.array(adata.obs_names); snap_s=np.array(adata.obs['doublet_score']); snap_p=np.array(adata.obs['doublet_probability'])
# load scrublet scores
d=np.loadtxt('scrublet_peaks_scores.txt',dtype=str)
scr_names=d[:,0]; scr_s=d[:,1].astype(float); scr_pred=d[:,2].astype(int)
# map snap scores for scrub top 100
scr_top_idx=np.argsort(-scr_s)
print('Scrublet top 50 with SnapATAC2 score/prob:')
for i in scr_top_idx[:80]:
    b=scr_names[i]
    j=np.where(snap_names==b)[0][0]
    print(b, 'scrub',scr_s[i], 'pred',scr_pred[i], 'snap_s',round(snap_s[j],3), 'snap_p',round(snap_p[j],3))
# overlap top 50 scrub predicted vs top 50 snap score
scr_pred_barcodes=set(scr_names[scr_pred=='1'])
print('num scrub pred',len(scr_pred_barcodes))
snap_top50=set(snap_names[np.argsort(-snap_s)[:50]])
print('overlap scrub pred vs snap top50',len(scr_pred_barcodes & snap_top50))
print(list(scr_pred_barcodes & snap_top50)[:30])
# What are snap top50 names
print('snap top50 by score:')
for i in np.argsort(-snap_s)[:50]: print(snap_names[i], round(snap_s[i],3))
PY"
