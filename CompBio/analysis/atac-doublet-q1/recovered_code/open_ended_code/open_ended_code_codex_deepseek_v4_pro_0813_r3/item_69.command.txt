/bin/bash -c "cat > snap_doublets2.py <<'PY'
import snapatac2 as snap, numpy as np, time, warnings
warnings.filterwarnings('ignore')
frags='atac.doublet.q1.bed.gz'
t=time.time()
adata = snap.pp.import_fragments(frags, chrom_sizes=snap.genome.GRCh38.chrom_sizes, is_paired=True, min_num_fragments=0, sorted_by_barcode=False, file='atac.snap2.h5ad')
print('import', time.time()-t, flush=True)
snap.pp.add_tile_matrix(adata, bin_size=500, inplace=True)
print('tile', adata.shape, time.time()-t, flush=True)
snap.pp.select_features(adata, n_features=50000, filter_lower_quantile=0.005, filter_upper_quantile=0.005, inplace=True)
print('select', time.time()-t, flush=True)
snap.pp.scrublet(adata, expected_doublet_rate=0.1, sim_doublet_ratio=2.0, n_comps=30, random_state=0, inplace=True)
print('scrub', time.time()-t, flush=True)
obs=adata.obs
p=np.array(obs['doublet_probability']); s=np.array(obs['doublet_score']); names=np.array(adata.obs_names)
print('max p', p.max(), 'p>0.5', (p>0.5).sum(), 'p>0.1',(p>0.1).sum(), 'p>0.01',(p>0.01).sum())
idx=np.argsort(-s)
for i in idx[:30]: print(names[i],p[i],s[i])
PY
python3 snap_doublets2.py"
