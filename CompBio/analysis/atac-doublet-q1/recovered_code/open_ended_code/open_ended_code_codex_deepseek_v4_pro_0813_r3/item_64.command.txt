/bin/bash -c "python3 - <<'PY'
import requests
url='https://zenodo.org/api/records/19443186'
r=requests.get(url, timeout=30)
print(r.status_code)
data=r.json()
print(data.keys())
print(data.get('metadata',{}).keys())
for f in data.get('files',[]): print(f.get('key'), f.get('size'), f.get('links',{}).get('self'))
PY"
