/bin/bash -lc "python3 - <<'PY'
import numpy as np
z=np.load('doublet_profiles.npz',allow_pickle=True); G=z['sim'].astype(float); b=z['barcodes']; n=len(b)
# sim uses normalized tfidf, diag reset. reconstruct diag 1
np.fill_diagonal(G,1)
i,j=np.triu_indices(n,1); den=np.sqrt(2+2*G[i,j]);
print('pairs',len(i),'den range',den.min(),den.max())
# For every target find best pair excluding target and pair whose components have balanced target sims.
for mode in ['all','balanced']:
 best=np.full(n,-1.); bp=[]; bal=np.full(n,-1.)
 for t in range(n):
  val=(G[t,i]+G[t,j])/den
  if mode=='balanced':
   # Require both component similarities >= 80% of max component? use difference
   val=np.where(np.minimum(G[t,i],G[t,j])>0.75*np.maximum(G[t,i],G[t,j]),val,-1)
  # exclude target in pair
  val[(i==t)|(j==t)]=-1
  q=np.argmax(val); best[t]=val[q]; bp.append((i[q],j[q]))
 print('\\n',mode,'range',best.min(),best.max())
 for t in np.argsort(best)[-40:][::-1]: print(b[t],round(best[t],5),b[bp[t][0]],b[bp[t][1]],'sims',round(G[t,bp[t][0]],4),round(G[t,bp[t][1]],4))
PY"
