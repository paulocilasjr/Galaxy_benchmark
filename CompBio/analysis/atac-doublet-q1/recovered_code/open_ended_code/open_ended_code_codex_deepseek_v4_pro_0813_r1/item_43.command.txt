def count_regions(ints):
 import collections
 ev=collections.defaultdict(int)
 for s,e in ints: ev[s]+=1; ev[e]-=1
 cov=0; reg=0
 for pos in sorted(ev):
  old=cov; cov+=ev[pos]
  if old<3 and cov>=3: reg+=1
 return reg
