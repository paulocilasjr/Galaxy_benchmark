/bin/bash -lc "Rscript - <<'RS'
suppressPackageStartupMessages(library(IRanges))
suppressPackageStartupMessages(library(GenomicRanges))
# read candidate intervals via gzip file? use data.table? use read.delim from gz
f <- gzfile('atac.doublet.q1.bed.gz','rt')
lines <- readLines(f)
close(f)
cat('read lines',length(lines),'\\n')
fields <- strsplit(lines,'\\t')
chrs <- sapply(fields, "'`[`, 1)
starts <- as.integer(sapply(fields, `[`, 2))
ends <- as.integer(sapply(fields, `[`, 3))
bcs <- sapply(fields, `[`, 4)
cand <- '"'AACATGGAGCGCCGGG-1'
sel <- which(bcs==cand & ends-starts<=1000 & "'!(chrs %in% c('"'chrM','chrX','chrY','M','X','Y')))
cat('cand frags',length(sel),'\\n')
grl <- split(GRanges(chrs[sel], IRanges(starts[sel], ends[sel]-1L)), bcs[sel])
# count nAbove2 like getFragmentOverlaps
d <- data.frame(nFrags=lengths(grl))
d"'$nAbove2 <- 0L
gr2 <- unlist(GRangesList(lapply(grl, function(x){
  x <- slice(coverage(x), lower=3L, rangesOnly=TRUE)
  GRanges(rep(seqlevels(grl)[1], length(x)), unlist(x, use.names=FALSE))
})))
# Actually above names may be wrong; simpler: coverage per chrom and count slice ranges
gr2 <- GRanges()
for(nm in names(grl)){
  cov <- coverage(grl[[nm]])
  # cov is RleList per seqlevel; we can slice each
  for(ch in names(cov)){
    x <- slice(cov[[ch]], lower=3L, rangesOnly=TRUE)
    if(length(x)>0) gr2 <- c(gr2, GRanges(ch, x))
  }
}
cat('"'nAbove2 R',length(gr2),'\\n')
cat('range first',head(gr2),'\\n')
RS"
