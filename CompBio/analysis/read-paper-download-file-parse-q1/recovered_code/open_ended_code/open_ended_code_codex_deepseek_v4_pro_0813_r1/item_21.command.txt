/bin/bash -lc "curl -L -s 'https://api.github.com/repos/kundajelab/scATAC-reprog/git/trees/main?recursive=1' -o repo_tree.json && python3 - <<'PY'
import json
j=json.load(open('repo_tree.json'))
print('truncated',j.get('truncated'),'files',len(j.get('tree',[])))
for item in j.get('tree',[]):
    path=item['path']
    if any(x in path.lower() for x in ['peak','cluster','dynamic','archetype','coc','table']):
        print(item['type'], item.get('size'), path)
PY"
