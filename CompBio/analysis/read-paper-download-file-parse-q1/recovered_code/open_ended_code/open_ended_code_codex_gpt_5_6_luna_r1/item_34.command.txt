/bin/bash -lc "curl -L --fail -sS -r 1599216676-1604413645 -o peaks_member.bin 'https://zenodo.org/api/records/8313962/files/scATAC.zip/content' && python3 - <<'PY'
import struct
b=open('peaks_member.bin','rb').read()
print(len(b),b[:4])
vals=struct.unpack_from('<4s5H3L2H',b,0)
print(vals)
_,ver,flags,comp,mt,md,crc,csize,usize,nlen,xlen=vals
print('nlen',nlen,'xlen',xlen,'csize',csize,'usize',usize,'data',nlen+xlen+30)
name=b[30:30+nlen]; print(name)
import zlib
raw=b[30+nlen+xlen:30+nlen+xlen+csize]
print(len(raw),raw[:10])
out=zlib.decompress(raw,-15)
print(len(out),out[:200])
open('peaks.bed','wb').write(out)
PY
head -5 peaks.bed; wc -l peaks.bed; grep -F 'COC/L1' peaks.bed | head"
