/bin/bash -lc "python3 - <<'PY'
from remotezip import RemoteZip
url='https://zenodo.org/api/records/8313962/files/scATAC.zip/content'
with RemoteZip(url) as z:
    infos=z.infolist()
    print('nfiles',len(infos))
    for i in infos:
        name=i.filename
        if any(s.lower() in name.lower() for s in ['peak','metadata','cluster','barcodes','features','h5ad','mtx','readme','bed']):
            print(i.file_size, i.compress_size, name)
PY"
