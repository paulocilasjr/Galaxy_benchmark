/bin/bash -lc "python - <<'PY'
import xml.etree.ElementTree as ET
root=ET.parse('article.xml').getroot()
for e in root.iter():
    txt=' '.join(''.join(e.itertext()).split())
    if 'COC' in txt or 'CO/L' in txt or 'CO/E' in txt:
        tag=e.tag.split('}')[-1]
        if tag in ('p','title','caption','td','th','label'):
            print(tag, txt[:1200])
PY"
