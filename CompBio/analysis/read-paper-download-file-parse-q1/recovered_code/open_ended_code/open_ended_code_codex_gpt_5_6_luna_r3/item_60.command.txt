import sys,urllib.request,struct,zlib,gzip,os
url=sys.argv[1]; off=int(sys.argv[2]); csize=int(sys.argv[3]); out=sys.argv[4]
req=urllib.request.Request(url,headers={'Range':f'bytes={off}-{off+30+1024+csize-1}'})
with urllib.request.urlopen(req) as r: b=r.read()
assert b[:4]==b'PK\\x03\\x04', (b[:4],len(b))
_,ver,flag,method,tm,dt,crc,cs,us,nl,xl=struct.unpack_from('<4s5H3L2H',b,0)
start=30+nl+xl
print('local',method,nl,xl,'start',start,'csize',csize,'read',len(b))
raw=b[start:start+csize]
if method==0: data=raw
elif method==8: data=zlib.decompress(raw,-15)
else: raise Exception(method)
open(out,'wb').write(data); print('wrote',len(data))
PY
