/bin/bash -lc "python - <<'PY'
import urllib.request
url='https://zenodo.org/api/records/8313962/files/scATAC_clusters.zip/content'
size=4351228465
start=size-2_000_000
req=urllib.request.Request(url,headers={'Range':f'bytes={start}-{size-1}'})

print('done')
PY
python - <<'PY'
import struct
b=open('zip_tail.bin','rb').read()
for sig in [b'PK\\x05\\x06',b'PK\\x06\\x06',b'PK\\x01\\x02']:
 print(sig,b.rfind(sig))
pos=b.rfind(b'PK\\x05\\x06')
print(b[pos:pos+22])
if pos>=0:
 vals=struct.unpack('<4s4H2LH',b[pos:pos+22]); print(vals)
 print('cd offset',vals[6],'cd size',vals[5],'entries',vals[4])
PY"
