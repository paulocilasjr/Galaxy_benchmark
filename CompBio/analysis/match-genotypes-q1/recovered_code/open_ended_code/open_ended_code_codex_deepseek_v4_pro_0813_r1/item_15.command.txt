def read_pos(fn):
 s=set(); cnt=0
 with open(fn) as f:
  for line in f:
   if line.startswith('#'): continue
   p=line.split('\\t')[1]
   s.add(p); cnt+=1
 print(fn,cnt,len(s))
 return s
s1=read_pos('match.genotypes.1.1.pvar')
s3=read_pos('match.genotypes.1.3.pvar')
print('only1',len(s1-s3), sorted((int(x),x) for x in s1-s3)[:20])
print('only3',len(s3-s1), sorted((int(x),x) for x in s3-s1)[:20])
PY"
