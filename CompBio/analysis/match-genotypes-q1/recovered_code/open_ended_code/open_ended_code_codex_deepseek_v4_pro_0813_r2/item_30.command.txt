/bin/bash -c "python3 - <<'PY'
import pgenlib, pandas as pd, numpy as np

def read_pvar(pre):
 return pd.read_csv(pre+'.pvar',sep='\\t',comment='#',names=['CHROM','POS','ID','REF','ALT'],dtype={'CHROM':str,'POS':int})
def read_genos(pre, subset):
 r=pgenlib.PgenReader(pre.encode()+b'.pgen')
 ids=[]
 with open(pre+'.psam') as f:
  for line in f:
   line=line.strip()
   if line.startswith('#') or not line: continue
   ids.append(line)
 idx=np.array(subset,dtype=np.uint32)
 buf=np.empty([idx.size,r.get_raw_sample_ct()],dtype=np.int32)
 r.read_list(idx,buf)
 return ids,buf
p1=read_pvar('match.genotypes.1.1'); p3=read_pvar('match.genotypes.1.3')
# target positions for selected O-related shared markers
positions=[133261703,133261662,133256625,133260743,133258141,133253159,133261737,133261730,133261560,133257486,133257567,133257570,133257647,133256074,133256028]
for pos in positions:
 i1=p1.index[p1.POS==pos]
 if len(i1)==0: continue
 # read target one variant
 ids1,g1=read_genos('match.genotypes.1.1',i1.to_numpy())
 i3=p3.index[p3.POS==pos]
 ids3,g3=read_genos('match.genotypes.1.3',i3.to_numpy())
 row3=p3.iloc[i3[0]]
 row1=p1.iloc[i1[0]]
 print(pos,row1.ID,row1.REF,row1.ALT,'target',dict(zip(ids1,g1[0])),'ref row',row3.REF,row3.ALT)
PY"
