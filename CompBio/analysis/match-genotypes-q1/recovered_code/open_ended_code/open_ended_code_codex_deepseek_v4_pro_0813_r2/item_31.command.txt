/bin/bash -c "python3 - <<'PY'
import pgenlib, pandas as pd, numpy as np

def read_pvar(pre):
 return pd.read_csv(pre+'.pvar',sep='\\t',comment='#',names=['CHROM','POS','ID','REF','ALT'],dtype={'CHROM':str,'POS':int})
def read_genos(pre, subset):
 r=pgenlib.PgenReader(pre.encode()+b'.pgen')
 ids=[]
 with open(pre+'.psam') as f:
  for line in f:
   line=line.strip()
   if line.startswith('#') or not line: continue
   ids.append(line)
 idx=np.array(subset,dtype=np.uint32)
 buf=np.empty([idx.size,r.get_raw_sample_ct()],dtype=np.int32)
 r.read_list(idx,buf)
 return ids,buf

p1=read_pvar('match.genotypes.1.1'); p3=read_pvar('match.genotypes.1.3')
lo,hi=133250000,133262000
idx3=p3.index[(p3.POS>=lo)&(p3.POS<=hi)].to_numpy(); ids3,g3=read_genos('match.genotypes.1.3',idx3); sub3=p3.iloc[idx3].reset_index(drop=True)
idx1=p1.index[(p1.POS>=lo)&(p1.POS<=hi)].to_numpy(); ids1,g1=read_genos('match.genotypes.1.1',idx1); sub1=p1.iloc[idx1].reset_index(drop=True)
# match shared: target variant -> ref index same pos/ref/alt
tref=[]
for j,row in sub1.iterrows():
 m=np.where((sub3.POS==row.POS)&(sub3.REF==row.REF)&(sub3.ALT==row.ALT))[0]
 if len(m)==1: tref.append((j,int(m[0]),row.POS))
print('shared',len(tref))
T=g1[np.array([a for a,b,c in tref])] # variants x targets
R=g3[np.array([b for a,b,c in tref])]
# key positions indices
keyO=np.where(sub3.POS.to_numpy()==133257521)[0][0]
keyA=np.where(sub3.POS.to_numpy()==133255935)[0][0]
keyB=np.where(sub3.POS.to_numpy()==133255928)[0][0]
# target genotypes as columns
for ti,tid in enumerate(ids1):
    # distances across shared variants (genotype 0/1/2, hamming-like sum absolute)
    dist=np.abs(R-T[:,ti][:,None]).sum(axis=0)
    # also correlation?
    order=np.argsort(dist)
    print('\\nTARGET',tid,'total',dist[order[:10]])
    for k in order[:10]:
        print(' ref',ids3[k],'dist',dist[k],'O',g3[keyO,k],'A',g3[keyA,k],'B',g3[keyB,k])
PY"
