/bin/bash -c "python3 - <<'PY'
import pgenlib, pandas as pd, numpy as np

def read_genos(pre, variant_subset):
    reader=pgenlib.PgenReader(pre.encode()+b'.pgen')
    ids=[]
    with open(pre+'.psam') as f:
        for line in f:
            line=line.strip()
            if line.startswith('#') or not line: continue
            ids.append(line)
    ns=reader.get_raw_sample_ct(); nv=reader.get_variant_ct()
    idx=np.array(variant_subset, dtype=np.int64)
    buf=np.empty([ns, idx.size], dtype=np.int32)
    reader.read_list(idx, buf)
    return ids,buf

pvar=pd.read_csv('match.genotypes.1.1.pvar', sep='\\t', comment='#', names=['CHROM','POS','ID','REF','ALT'], dtype={'CHROM':str,'POS':int})
for lo,hi in [(133255850,133256150),(133257450,133257650),(133254500,133261500)]:
    idx=pvar.index[(pvar['POS']>=lo)&(pvar['POS']<=hi)].to_numpy()
    ids,buf=read_genos('match.genotypes.1.1', idx)
    sub=pvar.iloc[idx].copy()
    sub['idx']=idx
    print('\\nREGION',lo,hi,'n',len(idx))
    for j,(_,row) in enumerate(sub.iterrows()):
        g=buf[:,j]
        print(row['POS'], row['ID'], row['REF'],'>',row['ALT'], dict(zip(ids,g)))
PY"
