/bin/bash -lc "python3 - <<'PY'
import urllib.request
for url in ['https://www.snpedia.com/index.php/Gs129','https://www.snpedia.com/index.php/Gs130','https://www.snpedia.com/index.php/Rs8176746','https://www.snpedia.com/index.php/Rs8176747','https://www.snpedia.com/index.php/Rs8176719']:
    try:
        req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
        text=urllib.request.urlopen(req,timeout=20).read().decode('utf-8','ignore')
        print('\\nURL',url,'len',len(text))
        import re
        text=re.sub('<["'^>]+>'"',' ',text)
        text=re.sub(r'\\s+',' ',text)
        print(text[:2000])
    except Exception as e: print(e)
PY"
