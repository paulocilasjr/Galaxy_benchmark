/bin/bash -lc "python - <<'PY'
import pyBigWig, numpy as np
bw=pyBigWig.open('ENCFF822FDB.bigWig')
for c,s,e in [('chr10',80_000_000,81_000_000),('chr11',69_000_000,70_000_000),('chr8',70_000_000,71_000_000),('chr7',45_000_000,46_000_000)]:
    stat=bw.stats(c,s,e,nBins=1,type='sum',exact=True)[0]
    vals=np.asarray(bw.values(c,s,e,numpy=True),dtype=np.float64)
    direct=np.nansum(vals)
    print(f'{c}:{s}-{e} stats={stat:.12f} direct={direct:.12f} finite={np.isfinite(vals).sum()}')
bw.close()
PY"
