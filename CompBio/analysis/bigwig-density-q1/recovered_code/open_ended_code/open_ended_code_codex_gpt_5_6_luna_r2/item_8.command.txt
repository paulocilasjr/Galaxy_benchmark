/bin/bash -lc "python - <<'PY'
import pyBigWig, time
p='ENCFF822FDB.bigWig'; bw=pyBigWig.open(p)
rows=[]
t=time.time()
for i in range(1,20):
    c=f'chr{i}'; L=bw.chroms(c); n=L//1_000_000
    vals=bw.stats(c, 0, n*1_000_000, nBins=n, type='sum', exact=True)
    assert len(vals)==n
    for j,v in enumerate(vals):
        rows.append((0.0 if v is None else v,c,j*1_000_000,(j+1)*1_000_000))
bw.close()
rows.sort(key=lambda x:x[0], reverse=True)
print('bins',len(rows),'elapsed',time.time()-t)
for rank,row in enumerate(rows[:20],1): print(rank, row)
PY"
