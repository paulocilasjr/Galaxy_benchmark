/bin/bash -lc "python - <<'PY'
import pyBigWig, time
bw=pyBigWig.open('ENCFF822FDB.bigWig')
chrom='chr1'; L=bw.chroms(chrom); n=L//1000000
for exact in (False, True):
 t=time.time(); x=bw.stats(chrom,0,n*1000000,nBins=n,type='sum',exact=exact); print(exact, len(x), x[:5], x[-2:], time.time()-t)
bw.close()
PY"
