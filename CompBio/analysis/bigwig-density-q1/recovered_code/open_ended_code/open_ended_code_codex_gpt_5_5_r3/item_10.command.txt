/bin/bash -lc "python3 - <<'PY'
import pyBigWig
BIN=1_000_000
bw=pyBigWig.open('ENCFF822FDB.bigWig')
rows=[]
for i in range(1,20):
    chrom=f'chr{i}'
    size=bw.chroms()[chrom]
    for start in range(0, size - BIN + 1, BIN):
        end=start+BIN
        s=bw.stats(chrom, start, end, type='sum', exact=True)[0]
        if s is None:
            s=0.0
        rows.append((s, chrom, start, end))
rows_sorted=sorted(rows, reverse=True)
print('n_bins', len(rows_sorted))
for rank,(s,c,st,en) in enumerate(rows_sorted[:20], start=1):
    print(rank, c, st, en, format(s,'.12g'))
PY"
