/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
checks=[('chr10',80000000,81000000),('chr11',69000000,70000000),('chr8',70000000,71000000),('chr7',45000000,46000000),('chr11',120000000,121000000)]
for c,st,en in checks:
    stat=bw.stats(c,st,en,type='sum',exact=True)[0] or 0.0
    intervals=bw.intervals(c,st,en) or []
    manual=sum((min(e,en)-max(s,st))*v for s,e,v in intervals)
    covered=sum(min(e,en)-max(s,st) for s,e,v in intervals)
    print(f'{c}:{st}-{en}\\tstats={stat:.12f}\\tmanual={manual:.12f}\\tcovered={covered}\\tn_intervals={len(intervals)}\\tdiff={stat-manual:.6g}')
PY"
