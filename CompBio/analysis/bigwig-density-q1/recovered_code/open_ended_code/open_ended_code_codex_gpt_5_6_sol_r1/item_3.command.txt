/bin/bash -lc "python3 - <<'PY'
import pyBigWig, math, time
path='ENCFF822FDB.bigWig'
bw=pyBigWig.open(path)
chroms=bw.chroms()
print('chrom count',len(chroms))
print({f'chr{i}':chroms.get(f'chr{i}') for i in range(1,20)})
rows=[]
t=time.time()
for i in range(1,20):
    chrom=f'chr{i}'
    length=chroms[chrom]
    n=length//1_000_000
    vals=bw.stats(chrom,0,n*1_000_000,type='sum',nBins=n,exact=True)
    if len(vals)"'!=n: raise RuntimeError((chrom,n,len(vals)))
    for j,v in enumerate(vals):
        rows.append((float('"'-inf') if v is None else v,chrom,j*1_000_000,(j+1)*1_000_000))
    print(chrom,n,'done',round(time.time()-t,1))
rows.sort(reverse=True)
print('TOP20')
for rank,x in enumerate(rows[:20],1): print(rank,*x)
bw.close()
PY"
