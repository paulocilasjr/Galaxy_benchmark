/bin/bash -lc "python3 - <<'PY'
import pyBigWig, math
bw=pyBigWig.open('ENCFF822FDB.bigWig')
cands=[('chr10',80000000,81000000),('chr11',69000000,70000000),('chr8',70000000,71000000),('chr7',45000000,46000000),('chr11',120000000,121000000)]
for chrom,start,end in cands:
    segs=bw.intervals(chrom,start,end) or []
    raw=sum((min(e,end)-max(s,start))*v for s,e,v in segs if min(e,end)>max(s,start))
    stat=bw.stats(chrom,start,end,type='sum',exact=True)[0]
    vals=bw.values(chrom,start,end,numpy=True)
    dense=float(__import__('numpy').nansum(vals,dtype=__import__('numpy').float64))
    covered=sum(min(e,end)-max(s,start) for s,e,v in segs if min(e,end)>max(s,start))
    print(chrom,start,end,'segments',len(segs),'covered',covered,'raw',repr(raw),'stat',repr(stat),'dense',repr(dense),'diff',raw-stat)
bw.close()
PY"
