/bin/bash -lc "python3 - <<'PY'
import math
import numpy as np
import pyBigWig
regions=[('chr10',80000000,81000000),('chr11',69000000,70000000),('chr8',70000000,71000000),('chr7',45000000,46000000),('chr11',120000000,121000000)]
with pyBigWig.open('ENCFF822FDB.bigWig') as bw:
 for chrom,start,end in regions:
  iv=bw.intervals(chrom,start,end)
  weighted=sum((min(e,end)-max(s,start))*v for s,e,v in iv if e>start and s<end)
  perbase=float(np.nansum(bw.values(chrom,start,end,numpy=True),dtype=np.float64))
  exact=bw.stats(chrom,start,end,type='sum',exact=True)[0]
  print(f'{chrom}:{start}-{end}\\tinterval={weighted:.12f}\\tperbase={perbase:.12f}\\tstats={exact:.12f}\\truns={len(iv)}')
PY"
