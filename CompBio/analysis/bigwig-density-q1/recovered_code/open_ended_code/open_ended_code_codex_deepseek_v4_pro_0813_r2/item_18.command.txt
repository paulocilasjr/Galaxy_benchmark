# sort by sum (col4) desc, output top 10
rows=[]
for line in open('avg.tab'):
    parts=line.rstrip('\\n').split('\\t')
    if len(parts)<5: continue
    name,size,covered,sumv,mean0,mean=parts[0],int(parts[1]),int(parts[2]),float(parts[3]),float(parts[4]),float(parts[5])
    chrom,idx=name.rsplit('_',1)
    st=int(idx)*1000000
    rows.append((sumv,chrom,st,st+1000000,covered,mean0,mean))
rows.sort(key=lambda x:x[0], reverse=True)
for rank,r in enumerate(rows[:10],1):
    print(rank, f'{r[1]}:{r[2]}-{r[3]}', r)
