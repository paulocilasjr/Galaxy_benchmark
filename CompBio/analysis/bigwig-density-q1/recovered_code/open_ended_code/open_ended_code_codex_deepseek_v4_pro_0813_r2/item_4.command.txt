/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
st=2900000; en=3000000
for t in ['mean','min','max','sum','coverage','std']:
    try: print(t,bw.stats('chr1',st,en,type=t))
    except Exception as e: print('err',t,e)
ivs=bw.intervals('chr1',st,en)
print('n ivs',len(ivs))
for x in ivs[:20]: print(x)
non=[x for x in ivs if x[2]>0]
print('non',len(non), non[:20])
print('sum len?', sum(x[1]-x[0] for x in ivs))
print('area nonzero', sum((e-s)*v for s,e,v in non))
bw.close()
PY"
