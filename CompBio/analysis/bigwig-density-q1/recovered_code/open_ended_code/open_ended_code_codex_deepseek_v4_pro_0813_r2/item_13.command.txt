/bin/bash -lc "python3 - <<'PY'
import pyBigWig, time
bw=pyBigWig.open('ENCFF822FDB.bigWig')
bins=[('chr10',80000000),('chr11',69000000),('chr8',70000000),('chr7',45000000),('chr11',120000000)]
M=1000000
for chrom,st in bins:
    t=time.time()
    ivs=bw.intervals(chrom, st, st+M)
    area=sum((e-s)*v for s,e,v in ivs)
    nonzero=sum(v>0 for s,e,v in ivs)
    exact=bw.stats(chrom,st,st+M,type='sum',exact=True)[0]
    default=bw.stats(chrom,st,st+M,type='sum',exact=False)[0]
    mean_exact=bw.stats(chrom,st,st+M,type='mean',exact=True)[0]
    mean_default=bw.stats(chrom,st,st+M,type='mean',exact=False)[0]
    print(chrom, st, 'ivs_area', area, 'nonzero',nonzero,'exact_sum',exact,'default_sum',default,'mean_exact',mean_exact,'mean_default',mean_default,'time',time.time()-t)
bw.close()
PY"
