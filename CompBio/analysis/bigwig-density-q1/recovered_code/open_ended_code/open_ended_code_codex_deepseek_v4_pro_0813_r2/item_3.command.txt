/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
for chrom in ['chr1','chr2']:
    found=False
    # scan some intervals via chrom intervals? use stats mean in chunks to find nonzero
    size=bw.chroms(chrom)
    for st in range(0, size, 100000):
        m=bw.stats(chrom, st, min(st+100000,size), type='mean')[0]
        if m and m>0:
            s=bw.stats(chrom, st, min(st+100000,size), type='sum')[0]
            ivs=bw.intervals(chrom, st, min(st+100000,size))
            area=sum((e-s)*v for s,e,v in ivs)
            print(chrom,st,m,'stats_sum',s,'intervals_area',area)
            found=True
            break
    if not found: print(chrom,'no nonzero in chunks?')
    if found: break
bw.close()
PY"
