/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
st=2900000; en=3000000
for exact in [False,True]:
 print('exact',exact)
 for t in ['mean','max','sum','coverage']:
  try: print(t,bw.stats('chr1',st,en,type=t,exact=exact))
  except Exception as e: print('ERR',t,e)
ivs=bw.intervals('chr1',st,en)
print('intervals area',sum((e-s)*v for s,e,v in ivs),'nonzero',sum(v>0 for s,e,v in ivs))
bw.close()
PY"
