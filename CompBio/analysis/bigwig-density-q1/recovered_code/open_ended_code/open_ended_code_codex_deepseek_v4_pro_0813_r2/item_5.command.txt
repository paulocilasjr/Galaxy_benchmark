/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
st=2900000; en=3000000
# find max stats location via smaller windows
mx=0; mxst=None
for s in range(st,en,1000):
    e=min(s+1000,en)
    vals=bw.stats('chr1',s,e,type='max',nBins=1)
    v=vals[0] if vals else 0
    if v and v>mx:
        mx=v;mxst=s
print('approx max st',mxst,mx)
if mxst is not None:
  vals=bw.values('chr1',mxst,mxst+1000)
  print('values len',len(vals),'max',max(vals),'nonzero count',sum(v>0 for v in vals), 'first nonzero')
  for i,v in enumerate(vals):
    if v>0:
      print(mxst+i,v);break
  ivs=bw.intervals('chr1',mxst,mxst+1000)
  print('ivs',ivs[:5], 'nonzero', [x for x in ivs if x[2]>0][:5])
bw.close()
PY"
