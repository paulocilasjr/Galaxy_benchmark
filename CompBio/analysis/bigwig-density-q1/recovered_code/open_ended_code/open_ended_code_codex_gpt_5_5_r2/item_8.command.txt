/bin/bash -lc "python3 - <<'PY'
import pyBigWig, math
bw = pyBigWig.open('ENCFF822FDB.bigWig')
chroms = bw.chroms()
print({k: chroms.get(k) for k in [f'chr{i}' for i in range(1,20)]})
rows=[]
for i in range(1,20):
    chrom=f'chr{i}'
    size=chroms[chrom]
    for start in range(0, size - 1_000_000 + 1, 1_000_000):
        end=start+1_000_000
        val = bw.stats(chrom, start, end, type='sum', exact=True)[0]
        if val is None or math.isnan(val):
            val=0.0
        rows.append((val, chrom, start, end))
rows_sorted=sorted(rows, key=lambda x: x[0], reverse=True)
for rank, (val, chrom, start, end) in enumerate(rows_sorted[:20], 1):
    print(rank, chrom, start, end, repr(val))
print('total bins', len(rows_sorted))
bw.close()
PY"
