/bin/bash -lc "python3 - <<'PY'
import pyBigWig, json, time
bw=pyBigWig.open('ENCFF822FDB.bigWig')
wanted=sorted([f'chr{i}' for i in range(1,20)], key=lambda x:int(x[3:]))
results=[]
start_time=time.time()
for idx,c in enumerate(wanted,1):
    l=bw.chroms()[c]
    n=l//1000000
    for i in range(n):
        st=i*1000000; en=st+1000000
        its=bw.intervals(c,st,en)
        total=0.0
        if its:
            # exact sum values*len; intervals expected non-overlap, but use overlap anyway
            for s,e,v in its:
                total += v*(min(e,en)-max(s,st))
        results.append((total,c,st,en))
    print(f'{idx:2d} {c} done n={n} elapsed={time.time()-start_time:.1f}s', flush=True)
bw.close()
results_sorted=sorted(results,key=lambda x:-x[0])
print('total bins',len(results))
for rank,(total,c,st,en) in enumerate(results_sorted[:20],1):
    print(rank, f'{c}:{st}-{en}', repr(total))
with open('bin_totals_direct.json','w') as f:
    json.dump([{'total':t,'chrom':c,'start':st,'end':en} for t,c,st,en in results_sorted], f)
print('elapsed',time.time()-start_time)
PY"
