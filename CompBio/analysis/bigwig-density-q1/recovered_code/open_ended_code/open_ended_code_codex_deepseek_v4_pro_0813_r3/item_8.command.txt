/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
bins=[('chr10',80000000,81000000),('chr11',69000000,70000000),('chr7',45000000,46000000),('chr11',120000000,121000000),('chr8',70000000,71000000),('chr7',44000000,45000000),('chr19',5000000,6000000)]
for c,st,en in bins:
    its=bw.intervals(c,st,en)
    if its:
      total=sum(v*(min(e,en)-max(s,st)) for s,e,v in its)
    else:
      total=0.0
    print(f'{c}:{st}-{en}', 'n',len(its) if its else 0,'exact_total',repr(total),'mean_stats_total',repr(bw.stats(c,st,en,type='mean')[0]*1e6))
bw.close()
PY"
