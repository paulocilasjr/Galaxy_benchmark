/bin/bash -lc "python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF822FDB.bigWig')
# inspect coverage for several random-ish intervals and count gaps
for c in ['chr1','chr2','chr19']:
    l=bw.chroms()[c]
    cov=bw.stats(c,0,l,type='coverage',nBins=1)[0]
    print(c,'len',l,'coverage',cov)
# check first 100kb around boundaries
for c,st in [('chr1',0),('chr1',100000),('chr1',195000000),('chr19',61000000)]:
    en=min(st+100000,bw.chroms()[c])
    its=bw.intervals(c,st,en)
    gaps=[]
    pos=st
    for s,e,v in its or []:
      if s>pos: gaps.append((pos,s))
      pos=max(pos,e)
    if pos<en: gaps.append((pos,en))
    print(c,st,en,'n',len(its or []),'gaps',gaps[:10], 'first', (its[0] if its else None),'last',(its[-1] if its else None))
bw.close()
PY"
