/bin/bash -lc "cat > rank_signal.py <<'PY'
import pyBigWig
import json
from math import fsum

width = 1_000_000
bw = pyBigWig.open('ENCFF822FDB.bigWig')
rows = []
for i in range(1, 20):
    chrom = f'chr{i}'
    length = bw.chroms(chrom)
    count = length // width
    sums = bw.stats(chrom, 0, count * width, nBins=count, type='sum', exact=True)
    rows.extend((value or 0.0, chrom, j*width, (j+1)*width) for j, value in enumerate(sums))
rows.sort(reverse=True)
with open('window_signal_ranking.tsv', 'w') as out:
    out.write('rank\\tchrom\\tstart\\tend\\ttotal_signal\\n')
    for rank, (value, chrom, start, end) in enumerate(rows, 1):
        out.write(f'{rank}\\t{chrom}\\t{start}\\t{end}\\t{value:.12g}\\n')
print('Complete windows:', len(rows))
for value, chrom, start, end in rows[:10]:
    intervals = bw.intervals(chrom, start, end) or []
    checked = fsum((min(b,end)-max(a,start))*v for a,b,v in intervals)
    assert abs(value - checked) < max(1e-6, abs(checked)*1e-10), (value, checked)
    print(f'{chrom}:{start}-{end}\\t{value:.12g}\\tverified={checked:.12g}')
bw.close()
PY
python rank_signal.py"
