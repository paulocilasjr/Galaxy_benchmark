/bin/bash -lc "head -n 11 window_signal_ranking.tsv; python - <<'PY'
import pyBigWig
from math import fsum
bw=pyBigWig.open('ENCFF822FDB.bigWig')
with open('window_signal_ranking.tsv') as inp:
    next(inp)
    for line in list(inp)[:10]:
        rank, chrom, a, b, total = line.strip().split('\\t')
        a,b=int(a),int(b)
        checked=fsum((min(y,b)-max(x,a))*v for x,y,v in bw.intervals(chrom,a,b))
        assert abs(float(total)-checked) < abs(checked)*1e-8
        print(rank, f'{chrom}:{a}-{b}', checked)
PY"
