/bin/bash -lc "python3 - <<'PY'
import pyBigWig, time
bw=pyBigWig.open('ENCFF822FDB.bigWig')
chrom='chr8'; size=bw.chroms()[chrom]; n=size//1_000_000
# compute all nBins
arr=bw.stats(chrom,0,n*1_000_000,type='sum',exact=True,nBins=n)
# check first 5 and last 5 and top bin individual
for idx in list(range(5))+list(range(65,75))+[80]:
    one=bw.stats(chrom, idx*1_000_000, (idx+1)*1_000_000, type='sum', exact=True)[0]
    print(idx, one, arr[idx], one-arr[idx])
# also raw intervals sum for top bin 70M-71M
start=70000000; end=71000000
ints=bw.intervals(chrom,start,end)
manual=0.0
for s,e,v in ints:
    a=max(s,start); b=min(e,end)
    if b>a: manual += v*(b-a)
print('manual top',manual,arr[70])
print('n intervals',len(ints))
print('first last ints', ints[:5], ints[-5:])
PY"
