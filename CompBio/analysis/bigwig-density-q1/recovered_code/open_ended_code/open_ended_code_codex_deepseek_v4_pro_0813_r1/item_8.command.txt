/bin/bash -lc "python3 - <<'PY'
import pyBigWig, json, time
bw=pyBigWig.open('ENCFF822FDB.bigWig')
chroms={f'chr{i}':s for i in range(1,20) for s in [bw.chroms()[f'chr{i}']]}
rows=[]
t=time.time()
for chrom,size in chroms.items():
    n=size//1_000_000
    if n<=0: continue
    sums=bw.stats(chrom,0,n*1_000_000,type='sum',exact=True,nBins=n)
    # verify len
    for j,s in enumerate(sums):
        rows.append((chrom,j*1_000_000,(j+1)*1_000_000,s))
    print(chrom,n,time.time()-t)
print('total bins',len(rows),'time',time.time()-t)
rows_sorted=sorted(rows, key=lambda x:x[3], reverse=True)
for i,r in enumerate(rows_sorted[:20]):
    print(i+1, r[0], r[1], r[2], repr(r[3]))
# Save all rows
with open('bin_sums.json','w') as f:
    json.dump(rows, f)
PY"
