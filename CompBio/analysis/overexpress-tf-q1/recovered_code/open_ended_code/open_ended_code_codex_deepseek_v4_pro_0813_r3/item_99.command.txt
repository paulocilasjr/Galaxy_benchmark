/bin/bash -c "python3 - <<'PY'
import pandas as pd, statsmodels.api as sm, numpy as np
f='work/features.tsv'; df=pd.read_csv(f,sep='\\t')
tf=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
mot=[c for c in df.columns if '_' in c and c.split('_')[0] in tf]
for bg in ['common','lost']:
    sub=df[df.label.isin(['fg',bg])].copy()
    sub['y']=(sub.label=='fg').astype(int)
    print('\\n### fg vs',bg,'n=',len(sub),'mean gc fg/back',sub[sub.y==1].gc.mean(),sub[sub.y==0].gc.mean())
    rows=[]
    for feat in tf+mot:
        X=sm.add_constant(sub[['gc',feat]])
        model=sm.Logit(sub.y,X).fit(disp=0)
        coef=model.params[feat]; z=coef/model.bse[feat]; p=model.pvalues[feat]; or_=np.exp(coef)
        rows.append((feat,coef,z,p,or_))
    rows=sorted(rows,key=lambda x:-x[2])
    for feat,coef,z,p,or_ in rows[:10]:
        print(f'{feat:22s} coef {coef:+.4f} z {z:+.2f} p {p:.3e} OR {or_:.3f}')
PY"
