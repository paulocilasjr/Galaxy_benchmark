/bin/bash -c "python3 - <<'PY'
import MOODS.scan as s, MOODS.tools as t
mat=[[0.8,0.05,0.05,0.1],[0.05,0.8,0.05,0.1],[0.05,0.05,0.8,0.1]]
# rows A,C,G,T
mat2=t.log_odds(mat, t.flat_bg(4), 0.01)
print('mat2', mat2)
print('max', t.max_score(mat2), 'min', t.min_score(mat2))
thr=t.threshold_from_p(mat2, t.flat_bg(4), 0.0001)
print('thr', thr)
seq='AAACCCGGGAAACCCGGG'
res=s.scan_dna(seq, [mat2], t.flat_bg(4), [thr])
print(type(res), res)
print('len', len(res))
for x in res:
 print(type(x), len(x))
 print(list(x)[:5])
PY"
