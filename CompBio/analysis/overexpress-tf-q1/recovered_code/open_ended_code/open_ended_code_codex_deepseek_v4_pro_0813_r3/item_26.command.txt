/bin/bash -c "python3 - <<'PY'
from pyjaspar import jaspardb
jdb=jaspardb()
for n in ['RUNX1','PAX7','SPI1','GATA3','FOXA1']:
  for coll in ['CORE','CNE']:
    try:
      ms=jdb.fetch_motifs(collection=coll, tf_name=n)
      print(coll,n,len(ms))
      for m in ms[:20]: print(' ',m.matrix_id,m.name,len(m),m.species if hasattr(m,'species') else '')
    except Exception as e: print('ERR',coll,n,e)
PY"
