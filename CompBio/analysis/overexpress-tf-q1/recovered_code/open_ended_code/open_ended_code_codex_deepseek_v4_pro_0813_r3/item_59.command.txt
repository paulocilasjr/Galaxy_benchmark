/bin/bash -c "python3 - <<'PY'
import MOODS.scan as s, MOODS.tools as t
# 4xL rows A,C,G,T
m=[[0.8,0.05,0.05],[0.05,0.8,0.05],[0.05,0.05,0.8],[0.1,0.1,0.1]]
lo=t.log_odds(m, t.flat_bg(4), 0.01)
print('lo', lo)
thr=t.threshold_from_p(lo, t.flat_bg(4), 0.001)
print('thr',thr)
res=s.scan_dna('AAACCCGGGAAACCCGGG',[lo],t.flat_bg(4),[thr])
print(type(res), len(res), res)
for x in res: print(list(x)[:10])
PY"
