/bin/bash -c "python3 - <<'PY'
from pyjaspar import jaspardb
jdb=jaspardb()
names=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
for n in names:
    try:
        motifs=jdb.fetch_motifs(collection='CORE', tax_group='vertebrates', name=n)
        print(n, len(motifs))
        for m in motifs[:10]:
            print(' ', m.matrix_id, m.name, len(m))
    except Exception as e:
        print('ERR', n, e)
PY"
