/bin/bash -c "python3 - <<'PY'
import pandas as pd, statsmodels.api as sm, numpy as np
# load features fg only and scores
feat=pd.read_csv('work/features.tsv',sep='\\t')
fg=feat[feat.label=='fg'].copy()
scores=pd.read_csv('work/perturb_gained_summits.bed',sep='\\t',header=None)
print(scores.head(), scores.shape)
fg['score']=scores.iloc[:len(fg),4].values
print(fg.score.describe())
# top quartile vs bottom quartile by score
q=fg.score.quantile([0.25,0.75])
print('quartiles',q.values)
top=fg[fg.score>=q[0.75]].copy(); bot=fg[fg.score<=q[0.25]].copy()
top['y']=1; bot['y']=0
sub=pd.concat([top,bot])
tf=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
mot=[c for c in feat.columns if '_' in c and c.split('_')[0] in tf]
print('n top/bot',len(top),len(bot),'gc',top.gc.mean(),bot.gc.mean())
rows=[]
for f in tf+mot:
 X=sm.add_constant(sub[['gc',f]])
 m=sm.Logit(sub.y,X).fit(disp=0)
 coef=m.params[f]; z=coef/m.bse[f]; p=m.pvalues[f]; or_=np.exp(coef)
 rows.append((f,coef,z,p,or_))
rows=sorted(rows,key=lambda x:-x[2])
print('Top vs bottom gained peaks adjusted GC:')
for f,coef,z,p,or_ in rows:
 print(f'{f:22s} coef {coef:+.4f} z {z:+.2f} p {p:.3e} OR {or_:.3f}')
PY"
