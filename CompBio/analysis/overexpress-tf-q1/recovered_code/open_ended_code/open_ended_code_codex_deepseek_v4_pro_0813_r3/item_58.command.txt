/bin/bash -c "python3 - <<'PY'
import MOODS.scan as s, MOODS.tools as t
# orientation 1: rows A,C,G,T, columns positions (4xL)
mat=[[0.8,0.05,0.05,0.1],[0.05,0.8,0.05,0.1],[0.05,0.05,0.8,0.1]]
# orientation 2: rows positions, columns A,C,G,T (Lx4) 
matT=[list(x) for x in zip(*mat)]
for name, m in [('rowACGT',mat),('rowPos',matT)]:
 print('---',name, m)
 try:
  lo=t.log_odds(m, t.flat_bg(4), 0.01)
  thr=t.threshold_from_p(lo, t.flat_bg(4), 0.001)
  res=s.scan_dna('AAACCCGGGAAACCCGGG',[lo],t.flat_bg(4),[thr])
  print('ok', len(res), list(res[0])[:5])
 except Exception as e:
  print('err', type(e).__name__, e)
PY"
