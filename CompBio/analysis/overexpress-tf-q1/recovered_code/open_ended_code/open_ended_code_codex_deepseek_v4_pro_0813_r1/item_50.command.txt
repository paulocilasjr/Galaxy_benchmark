/bin/bash -lc "python3 - <<'PY'
import MOODS.scan, MOODS.tools
mat=[[0.8,0.066,0.066,0.066],[0.8,0.066,0.066,0.066],[0.066,0.8,0.066,0.066],[0.066,0.066,0.8,0.066]]
bg=MOODS.tools.flat_bg(4)
lo=MOODS.tools.log_odds(mat,bg,1)
th=MOODS.tools.threshold_from_p(lo,bg,0.01)
for seq in ['AACGTTTT','AACG','AACGAAAA','TTTTAACG','TTTTCGTTTT','CGTT','AACGT']:
 for ws in [1,4,7]:
  try:
   r=MOODS.scan.scan_dna(seq,[lo],bg,[th],ws)
   print(seq,'ws',ws,'n',len(r[0]), [(m.pos,m.score,seq[m.pos:m.pos+4]) for m in r[0]])
  except Exception as e: print('err',e)
 print()
PY"
