/bin/bash -lc "python3 - <<'PY'
import json,glob,os
# Create HOMER custom motif file from JASPAR PFMs. Matrix rows ACGT probabilities.
with open('candidate_homer.motifs','w') as out:
 for f in sorted(glob.glob('motifs/*.json')):
  d=json.load(open(f)); pfm=d['pfm']; name=os.path.basename(f)[:-5]
  cons=[]
  probs=[]
  n=len(pfm['A'])
  for i in range(n):
   vals=[pfm[nuc][i] for nuc in 'ACGT']; total=sum(vals)
   p=[v/total for v in vals]; probs.append(p); cons.append('ACGT'[max(range(4), key=lambda j:p[j])])
  out.write('>'+''.join(cons)+'\\t'+name+'\\t0\\t0\\t0\\t0\\n')
  for p in probs:
   out.write('\\t'.join(f'{x:.6g}' for x in p)+'\\n')
print('written', os.path.getsize('candidate_homer.motifs'))
PY
head -10 candidate_homer.motifs"
