/bin/bash -lc "python3 - <<'PY'
import random
from collections import Counter,defaultdict
def dinuc_shuffle(seq):
    if len(seq)<=1: return seq
    first=seq[0]
    edges=Counter(zip(seq,seq[1:]))
    out=defaultdict(list)
    for (a,b),cnt in edges.items(): out[a].extend([b]*cnt)
    for lst in out.values(): random.shuffle(lst)
    stack=[first]; path=[]
    while stack:
        v=stack[-1]
        if out[v]: stack.append(out[v].pop())
        else: path.append(stack.pop())
    path.reverse()
    return ''.join(path)
seq='AACCGGTTAACCGGTTACGT'
for _ in range(5):
 s=dinuc_shuffle(seq)
 print(seq, s)
 print(Counter(zip(seq,seq[1:]))==Counter(zip(s,s[1:])), Counter(seq)==Counter(s), s[0]==seq[0])
PY"
