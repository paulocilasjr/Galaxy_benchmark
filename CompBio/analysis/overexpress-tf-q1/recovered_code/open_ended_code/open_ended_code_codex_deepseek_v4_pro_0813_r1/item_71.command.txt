/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from scipy.stats import fisher_exact
sets=['cond1','common','cond2']; dfs={s:pd.read_csv(f'macs/bdgdiff/{s}.motif_scan.tsv',sep='\\t') for s in sets}
for suffix in ['_p1e-4','_p1e-5']:
 cols=[c for c in dfs['cond2'].columns if c.endswith(suffix)]
 rows=[]
 for col in cols:
  motif=col[:-len(suffix)]
  a=(dfs['cond2'][col]>0).sum(); b=len(dfs['cond2'])-a; c=(dfs['common'][col]>0).sum(); d=len(dfs['common'])-c
  odds,p=fisher_exact([[a,b],[c,d]],alternative='greater')
  frac2=(dfs['cond2'][col]>0).mean(); frac0=(dfs['common'][col]>0).mean()
  rows.append((motif,frac0,frac2,odds,p))
 res=pd.DataFrame(rows,columns=['motif','common_frac','cond2_frac','odds','p']).sort_values('odds',ascending=False)
 print('\\n',suffix)
 print(res.to_string(index=False, float_format=lambda x:f'{x:,.3g}'))
PY"
