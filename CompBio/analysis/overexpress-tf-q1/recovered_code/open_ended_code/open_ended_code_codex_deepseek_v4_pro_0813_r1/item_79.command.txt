/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from Bio import SeqIO
import MOODS.parsers, MOODS.tools, MOODS.scan
sets=['cond1','common','cond2']
for s in sets:
 d=pd.read_csv(f'macs/bdgdiff/{s}.motif_scan.tsv',sep='\\t')
 c4='ASCL1_MA1631.2_p1e-4'; c5='ASCL1_MA1631.2_p1e-5'
 idx=d.index[(d[c5]>d[c4])].tolist()
 print(s,'anomalies',len(idx))
 if idx:
  i=idx[0]
  seq=str(list(SeqIO.parse(f'macs/bdgdiff/{s}.fa','fasta'))[i].seq).upper()
  print('i',i,'len',len(seq),'tsv p4 p5',d.at[i,c4],d.at[i,c5])
  pf='motif_pfms/ASCL1_MA1631.2.pfm'; bg=MOODS.tools.flat_bg(4)
  lo=MOODS.parsers.pfm_to_log_odds(pf,bg,1); rc=MOODS.tools.reverse_complement(lo)
  th_lo=MOODS.tools.threshold_from_p(lo,bg,1e-4); th_hi=MOODS.tools.threshold_from_p(lo,bg,1e-5)
  print('th',th_lo,th_hi)
  res=MOODS.scan.scan_dna(seq,[lo,rc],bg,[th_lo,th_lo],1)
  n4=len(res[0])+len(res[1]); n5=sum(m.score>=th_hi for m in res[0])+sum(m.score>=th_hi for m in res[1])
  print('recalc n4 n5',n4,n5)
  print('lo matches',[(m.pos,m.score,seq[m.pos:m.pos+4]) for m in res[0]][:5])
  print('rc matches',[(m.pos,m.score,seq[m.pos:m.pos+4]) for m in res[1]][:5])
  break
PY"
