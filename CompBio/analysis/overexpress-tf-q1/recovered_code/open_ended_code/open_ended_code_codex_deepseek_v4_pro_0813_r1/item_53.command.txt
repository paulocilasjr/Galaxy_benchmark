/bin/bash -lc "python3 - <<'PY'
import MOODS.scan, MOODS.tools
# desired positions A,A,C,G -> columns; MOODS matrix rows ACGT
mat_rows = [
 [0.8,0.8,0.066,0.066],  # A across positions
 [0.066,0.066,0.8,0.066],# C
 [0.066,0.066,0.066,0.8],# G
 [0.066,0.066,0.066,0.066],# T
]
bg=MOODS.tools.flat_bg(4)
lo=MOODS.tools.log_odds(mat_rows,bg,1)
print('lo',lo)
print('maxscore',MOODS.tools.max_score(lo))
th=MOODS.tools.threshold_from_p(lo,bg,0.01)
print('thr',th)
for seq in ['AACG','AACGAACG','CGTT','TTTTCGTTTTTT','AACGT']:
 r=MOODS.scan.scan_dna(seq,[lo],bg,[th],7)
 print(seq, [(m.pos,m.score,seq[m.pos:m.pos+4]) for m in r[0]])
PY"
