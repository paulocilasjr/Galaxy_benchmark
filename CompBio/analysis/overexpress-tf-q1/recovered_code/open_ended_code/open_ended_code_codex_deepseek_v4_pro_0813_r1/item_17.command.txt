import json, urllib.request, time, os
ids = {
'FOXA1':['MA0148.5'],
'ASCL1':['MA1100.3','MA1631.2'],
'TFAP2A':['MA0872.1','MA0810.2','MA0003.5'],

'GATA3':['MA0037.5'],
}
os.makedirs('motifs', exist_ok=True)
for tf, mids in ids.items():
    for mid in mids:
        out=f'motifs/{tf}_{mid}.json'
        if os.path.exists(out): continue
        url=f'https://jaspar.elixir.no/api/v1/matrix/{mid}/'
        for attempt in range(3):
            try:
                with urllib.request.urlopen(url, timeout=30) as r:

                print(tf,mid,'ok')
                break
            except Exception as e:
                print('retry',tf,mid,e); time.sleep(5)
