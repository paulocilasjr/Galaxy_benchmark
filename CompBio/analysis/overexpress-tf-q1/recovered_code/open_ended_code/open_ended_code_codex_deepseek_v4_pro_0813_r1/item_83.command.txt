/bin/bash -lc "python3 - <<'PY'
import glob,os,MOODS.parsers,MOODS.tools,MOODS.scan
from Bio import SeqIO
seqs=[]
for rec in SeqIO.parse('macs/bdgdiff/cond1.fa','fasta'):
 seqs.append(str(rec.seq).upper())
 if len(seqs)>=2: break
pfiles=sorted(glob.glob('motif_pfms/*.pfm')); bg=MOODS.tools.flat_bg(4); pseudo=1.0; p_lo=1e-4; p_hi=1e-5
models=[]; mat_list=[]; th_lo_list=[]; th_hi_list=[]
for pf in pfiles:
 name=os.path.basename(pf)[:-4]; lo=MOODS.parsers.pfm_to_log_odds(pf,bg,pseudo); rc=MOODS.tools.reverse_complement(lo)
 th_lo=MOODS.tools.threshold_from_p(lo,bg,p_lo); th_hi=MOODS.tools.threshold_from_p(lo,bg,p_hi)
 models.append((name,th_lo,th_hi)); mat_list += [lo,rc]; th_lo_list += [th_lo,th_lo]; th_hi_list += [th_hi,th_hi]
for idx,seq in enumerate(seqs):
 res=MOODS.scan.scan_dna(seq,mat_list,bg,th_lo_list,1)
 vals=[]
 for j in range(len(models)):
  fw=res[2*j]; rv=res[2*j+1]
  n4=len(fw)+len(rv); n5=sum(1 for m in fw if m.score>=th_hi_list[2*j])+sum(1 for m in rv if m.score>=th_hi_list[2*j+1])
  vals += [n4,n5]
 j=models.index(next(m for m in models if m[0]=='ASCL1_MA1631.2'))
 print('seq',idx,'vals idx',2*j,2*j+1,'=',vals[2*j],vals[2*j+1])
 print('raw file corresponding would be fields',3+2*j+1,3+2*j+1+1)
PY"
