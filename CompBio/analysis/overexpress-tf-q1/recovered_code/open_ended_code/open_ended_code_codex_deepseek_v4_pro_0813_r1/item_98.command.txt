/bin/bash -lc "cat > scan_union_motifs.py <<'PY'
import os,glob,time
from Bio import SeqIO
import MOODS.parsers,MOODS.tools,MOODS.scan
pfiles=sorted(glob.glob('motif_pfms/*.pfm')); bg=MOODS.tools.flat_bg(4); pseudocount=1.0; p=1e-4
models=[]; mats=[]; ths=[]
for pf in pfiles:
 name=os.path.basename(pf)[:-4]; lo=MOODS.parsers.pfm_to_log_odds(pf,bg,pseudocount); rc=MOODS.tools.reverse_complement(lo); th=MOODS.tools.threshold_from_p(lo,bg,p)
 models.append(name); mats += [lo,rc]; ths += [th,th]
out=open('macs/union_motif_scan.tsv','w'); out.write('idx\\tlength\\t'+'\\t'.join(models)+'\\n')
for i,rec in enumerate(SeqIO.parse('macs/union_peaks.fa','fasta')):
 seq=str(rec.seq).upper()
 res=MOODS.scan.scan_dna(seq,mats,bg,ths,1)
 vals=[]
 for j in range(len(models)):
  vals.append(len(res[2*j])+len(res[2*j+1]))
 out.write(str(i)+'\\t'+str(len(seq))+'\\t'+'\\t'.join(str(v) for v in vals)+'\\n')
 if (i+1)%20000==0: print(i+1,flush=True)
out.close(); print('done')
PY
python3 scan_union_motifs.py"
