/bin/bash -lc "python3 - <<'PY'
import json, os, statistics, collections
C=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
def matches(name):
    u=name.upper().replace('::',' ').replace('/',' ')
    parts=u.split()
    return [c for c in C if c in parts or c in u]
for limit in [300,500]:
    vals={c:[] for c in C}; scores=collections.Counter()
    for i,line in enumerate(open('work/top_gained_1000.bed')):
        if i>=limit: break
        chrom,s,e=line.split()[:3]; s=int(s); e=int(e); center=(s+e)/2
        p=f'work/jaspar_cache/{chrom}_{s}_{e}.json'
        if not os.path.exists(p): continue
        data=json.load(open(p))
        for item in data.get('jaspar2024',[]):
            if int(item['score'])<400: continue
            mcenter=(int(item['chromStart'])+int(item['chromEnd']))/2
            for c in matches(item['TFName']):
                vals[c].append(abs(mcenter-center)); scores[c]+=int(item['score'])
    print('limit',limit)
    for c in C:
        arr=vals[c]
        if not arr: continue
        print(c, 'n',len(arr),'median_abs',round(statistics.median(arr),1),'frac100',round(sum(x<=100 for x in arr)/len(arr),3),'frac200',round(sum(x<=200 for x in arr)/len(arr),3),'score',scores[c])
PY"
