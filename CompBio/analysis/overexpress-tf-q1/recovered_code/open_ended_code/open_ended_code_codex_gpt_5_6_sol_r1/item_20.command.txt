/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
import numpy as np
x=np.loadtxt('results/union_counts.tsv',usecols=(3,4))
r=x[:,0]; p=x[:,1]
lfc=np.log2((p+5)/(11257343+5*len(p)))-np.log2((r+5)/(10141848+5*len(r)))
for q in [0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]: print(q, np.quantile(lfc,q))
for th in [.5,1,1.5,2,3]: print('th',th,'gain',((lfc>th)&(r+p>=30)).sum(),'loss',((lfc<-th)&(r+p>=30)).sum())
PY"
