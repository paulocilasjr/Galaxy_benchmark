/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
import numpy as np
files={'ref':'motif_endpoint_profile.ref.tsv','pert':'motif_endpoint_profile.perturb.tsv'}
agg={k:defaultdict(lambda:np.zeros(200,dtype=float)) for k in files}; nh={}
for group,path in files.items():
 with open(path) as f:
  for line in f:
   a=line.split(); name=a[3]; tf=name.split('|')[1]; strand=a[4]; pos=int(a[5])-1; val=float(a[6])
   if strand=='-': pos=199-pos
   agg[group][tf][pos]+=val
   nh[(group,tf)]=nh.get((group,tf),0)+(1 if int(a[5])==1 else 0)
scale=10141848/11257343
print('TF\\tn\\tcenter_ref\\tcenter_pert\\tflank_ref\\tflank_pert\\tcenter/flank_ref\\tcenter/flank_pert\\tdelta_center_minus_flank')
for tf in sorted(agg['ref']):
 n=nh[('ref',tf)]; r=agg['ref'][tf]/n; p=agg['pert'][tf]/n*scale
 rf=r[80:95].mean(); pf=p[80:95].mean(); rf2=r[105:120].mean(); pf2=p[105:120].mean();
 print(tf,n,*(f'{v:.4f}' for v in [r[99],p[99],(rf+rf2)/2,(pf+pf2)/2,r[99]/((rf+rf2)/2),p[99]/((pf+pf2)/2),p[99]/((pf+pf2)/2)-r[99]/((rf+rf2)/2)]),sep='\\t')
PY"
