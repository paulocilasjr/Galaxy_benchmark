from collections import defaultdict
import numpy as np
agg={g:defaultdict(lambda:np.zeros(200,dtype=float)) for g in ['ref','pert']}; nh=defaultdict(int)
for g,path in [('ref','all_motif_profile.ref.tsv'),('pert','all_motif_profile.perturb.tsv')]:
 with open(path) as f:
  for line in f:
   a=line.split(); prefix=a[3].split('|')[0]; group=prefix.split('_',1)[0]; tf=a[3].split('|')[1]; pos=int(a[5])-1; strand=a[4];val=float(a[6]);
   if strand=='-':pos=199-pos
   agg[g][(group,tf)][pos]+=val
   if int(a[5])==1: nh[(group,tf)]+=1
scale=10141848/11257343
print('group\\tTF\\tn\\tref_center\\tpert_center\\tref_flank\\tpert_flank\\tref_ratio\\tpert_ratio\\tlog2_fc_center\\tlog2_fc_flank\\tfootprint_change')
for group in ['gain','loss','neutral']:
 for tf in sorted({k[1] for k in agg['ref'] if k[0]==group}):
  n=nh[(group,tf)];r=agg['ref'][(group,tf)]/n;p=agg['pert'][(group,tf)]/n*scale
  rf=(r[80:95].mean()+r[105:120].mean())/2;pf=(p[80:95].mean()+p[105:120].mean())/2
  print(group,tf,n,*[f'{v:.4f}' for v in [r[99],p[99],rf,pf,r[99]/rf if rf else 0,p[99]/pf if pf else 0,np.log2((p[99]+1e-5)/(r[99]+1e-5)),np.log2((pf+1e-5)/(rf+1e-5)),np.log2((p[99]/pf)/(r[99]/rf)) if r[99]>0 and pf>0 else 0]],sep='\\t')
PY"
