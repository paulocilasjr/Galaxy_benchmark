/bin/bash -lc "python3 - <<'PY'
import json, numpy as np
p=json.load(open('candidate_motifs.json')); lens={k:len(v['pfm']['A']) for k,v in p.items()}
agg={g:{} for g in ['ref','pert']}; nh={}
for g,path in [('ref','all_motif_profile.ref.tsv'),('pert','all_motif_profile.perturb.tsv')]:
 from collections import defaultdict
 d=defaultdict(lambda: np.zeros(200))
 with open(path) as f:
  for line in f:
   a=line.split(); group=a[3].split('|')[0].split('_',1)[0];tf=a[3].split('|')[1];pos=int(a[5])-1
   if a[4]=='-':pos=199-pos
   d[(group,tf)][pos]+=float(a[6]);
   if int(a[5])==1:nh[(group,tf)]=nh.get((group,tf),0)+1
 agg[g]=d
scale=10141848/11257343
print('TF\\tgain_n\\tneutral_n\\tgain_log2_center_flank_change\\tneutral_change\\tchange_diff')
for tf,L in lens.items():
 def vals(g,group):
  n=nh.get((group,tf),0); arr=agg[g].get((group,tf),np.zeros(200))/n
  lo=100-L//2; hi=lo+L
  center=arr[lo:hi].mean(); flank=np.r_[arr[50:80],arr[120:150]].mean()
  return center,flank
 rg,pg=vals('ref','gain'),vals('pert','gain');rn,pn=vals('ref','neutral'),vals('pert','neutral')
 def ch(r,p): return np.log2((p[0]/p[1])/(r[0]/r[1])) if min(r+p)>0 else np.nan
 print(tf,nh.get(('gain',tf),0),nh.get(('neutral',tf),0),*[f'{x:.3f}' for x in [ch(rg,pg),ch(rn,pn),ch(rg,pg)-ch(rn,pn)]],sep='\\t')
PY"
