/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
import numpy as np
files={'ref':'motif_profile.ref.tsv','pert':'motif_profile.perturb.tsv'}
# per tf, pos profile sum; identify by name token
agg={k:defaultdict(lambda:np.zeros(200,dtype=float)) for k in files}
nh={}
for group,path in files.items():
 with open(path) as f:
  for line in f:
   a=line.split(); name=a[3]; tf=name.split('|')[1]; strand=a[4]; pos=int(a[5])-1; val=float(a[6])
   if strand=='-': pos=199-pos
   agg[group][tf][pos]+=val
   nh[(group,tf)]=nh.get((group,tf),0)+ (1 if int(a[5])==1 else 0)
for tf in sorted(agg['ref']):
 n=nh[('ref',tf)]
 r=agg['ref'][tf]/n; p=agg['pert'][tf]/n*10141848/11257343
 print(tf,'n',n,'center ref/pert',round(r[99],2),round(p[99],2),'flank ref/pert',round(r[80:95].mean(),2),round(p[80:95].mean(),2),round(r[105:120].mean(),2),round(p[105:120].mean(),2),'center/flank',round(r[99]/r[80:95].mean(),3),round(p[99]/p[80:95].mean(),3))
 print('profiles', ' '.join(f'{i-99}:{p[i]:.1f}/{r[i]:.1f}' for i in range(85,115)))
PY"
