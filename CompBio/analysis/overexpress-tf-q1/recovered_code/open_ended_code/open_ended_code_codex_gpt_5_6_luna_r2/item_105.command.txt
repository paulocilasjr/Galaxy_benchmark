/bin/bash -lc "gzip -cd overexpress.tf.q1.perturb.bed.gz | awk -v OFS='\\t' '{if("'$1 ~ /''^chr([1-9]$|1[0-9]$|2[0-2]$|X$|Y$)/){print $1,$2,$2+1; print $1,$3-1,$3}}'"' | bedtools coverage -a motif_windows_KLF4.bed -b stdin -d > KLF4.pert.cuts.cov; python3 - <<'PY'
import numpy as np
for s in ['ref','pert']:
 a=np.zeros(101); n=0
 for line in open(f'KLF4.{s}.cuts.cov'):
  f=line.split('\\t'); rel=int(f[5])-int(f[4])-1; a[rel+50]+=int(f[6])
  if rel==0:n+=1
 a/=n
 flank=np.mean(np.r_[a[:20],a[-20:]])
 print(s,'flank',flank,'center',a[50],'ratio',a[50]/flank,'min',a.min(),'max',a.max())
 print('profile', ' '.join(f'{i}:{a[i+50]:.2f}' for i in range(-10,11)))
PY"
