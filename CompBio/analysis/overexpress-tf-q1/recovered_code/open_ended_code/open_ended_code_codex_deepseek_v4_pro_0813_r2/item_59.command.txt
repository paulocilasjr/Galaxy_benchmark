/bin/bash -c "python3 - <<'EOF'
import json
from MOODS import tools
motifs=json.load(open('candidate_motifs.json'))
bg=[0.295,0.205,0.205,0.295]
for tf,m in list(motifs.items())[:3]:
    pfm=m['pfm']; cols=len(pfm['A'])
    mat=[[pfm[b][i] for i in range(cols)] for b in 'ACGT']
    th=tools.threshold_from_p(mat,bg,1e-4)
    lo=tools.log_odds(mat,bg)
    print(tf,'threshold raw',th)
    print('log_odds type',type(lo))
    print(lo)
    try:
        th2=tools.threshold_from_p(lo,bg,1e-4)
        print('threshold from lo',th2)
    except Exception as e: print('lo err',e)
    print('max score raw',tools.max_score(mat,bg),'max lo',tools.max_score(lo,bg))
EOF"
