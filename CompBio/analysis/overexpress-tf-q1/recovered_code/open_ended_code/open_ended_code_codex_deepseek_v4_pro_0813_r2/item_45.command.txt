/bin/bash -c "Rscript - <<'EOF'
suppressMessages(library(DESeq2))
ref <- read.table('ref_counts.txt', sep='\\t', header=FALSE, col.names=c('chr','start','end','count'))
pert <- read.table('perturb_counts.txt', sep='\\t', header=FALSE, col.names=c('chr','start','end','count'))
counts <- cbind(ref=ref"'$count, perturb=pert$count)
rownames(counts) <- paste0(ref$chr, '"':', ref"'$start, '"'-', ref"'$end)
cond <- factor(c('"'ref','perturb'))
dds <- DESeqDataSetFromMatrix(countData=counts, colData=data.frame(condition=cond), design=~condition)
dds <- estimateSizeFactors(dds)
# try DESeq with no replicates
tryCatch({
 dds <- DESeq(dds)
 res <- results(dds)
 write.table(as.data.frame(res), file='DESeq_results.tsv', sep='\\t', quote=FALSE)
 cat('n', nrow(res), '\\n')
 print(head(res))
}, error=function(e) {cat('ERROR:', conditionMessage(e),'\\n')})
EOF"
