/bin/bash -lc "python - <<'PY'
import numpy as np,pandas as pd
from scipy.stats import fisher_exact
s=pd.read_csv('results/motif_scores.tsv',sep='\\t')
s['gcbin']=(s.gc/0.025).astype(int)
g=s[(s.group=='gain')&(s['rank']<3000)]
l=s[(s.group=='loss')&(s['rank']<10000)]
for col in [c for c in s.columns if '_MA' in c]:
 threshold=s.loc[s.group=='stable',col].quantile(.9)
 expected=0; covered=0
 for b in g.gcbin.unique():
  ng=(g.gcbin==b).sum(); bg=l[l.gcbin==b]
  if len(bg)<10: continue
  expected+=ng*(bg[col]>=threshold).mean();covered+=ng
 rate=(g[col]>=threshold).mean()
 print(col, 'gain',round(rate,3),'matched_loss',round(expected/covered,3),'enrichment',round(rate/(expected/covered),3))
PY"
