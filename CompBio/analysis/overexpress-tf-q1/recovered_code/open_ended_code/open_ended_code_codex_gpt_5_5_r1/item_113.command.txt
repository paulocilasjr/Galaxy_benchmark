/bin/bash -lc "python3 - <<'PY'
import numpy as np
patterns=['ATGCAAAT','CATTCCACCA','ATCAAAG','TGATAA','CACCC','CAGCTG']
RC=str.maketrans('ACGT','TGCA')
cur=None; rows={}
cons=[]
def emit(cur, rows):
 if not cur or len(rows)"'!=4: return
 mid,name=cur
 mat=np.vstack([rows[b] for b in '"'ACGT'])
 c=''.join('ACGT'[i] for i in mat.argmax(axis=0))
 cons.append((mid,name,c,c.translate(RC)[::-1]))
with open('JASPAR2024_CORE_vertebrates_non-redundant_pfms_jaspar.txt') as f:
 for line in f:
  line=line.strip()
  if not line: continue
  if line.startswith('>'):
   emit(cur,rows); fs=line[1:].split(); cur=(fs[0],fs[1]); rows={}
  else:
   rows[line[0].upper()]=np.array([float(x) for x in line[line.find('[')+1:line.find(']')].split()])
emit(cur,rows)
for pat in patterns:
 print('PAT',pat)
 for mid,name,c,rc in cons:
  if pat in c or pat in rc or c in pat or rc in pat:
   print(mid,name,c,rc)
PY"
