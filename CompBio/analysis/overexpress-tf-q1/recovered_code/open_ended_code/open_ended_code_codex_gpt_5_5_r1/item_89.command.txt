/bin/bash -lc "python3 - <<'PY'
import numpy as np
cands={'FOXA1','ASCL1','TFAP2A','RUNX1','Runx1','KLF4','PAX7','Pax7','TCF7','REST','SPI1','Spi1','GATA3','Gata3'}
cur=None; rows={}
def emit(cur, rows):
    if not cur or len(rows)"'!=4: return
    mid,name=cur
    if name not in cands: return
    mat=np.vstack([rows[b] for b in '"'ACGT'])
    cons=''.join('ACGT'[i] for i in mat.argmax(axis=0))
    print(mid, name, cons)
with open('JASPAR2024_CORE_vertebrates_non-redundant_pfms_jaspar.txt') as f:
 for line in f:
  line=line.strip()
  if not line: continue
  if line.startswith('>'):
   emit(cur,rows)
   fs=line[1:].split(); cur=(fs[0],fs[1]); rows={}
  else:
   rows[line[0].upper()]=np.array([float(x) for x in line[line.find('[')+1:line.find(']')].split()])
emit(cur,rows)
PY"
