from scipy.stats import fisher_exact
patterns={'TCF7_7mer':'ATCAAAG','TCF7_rc':'CTTTGAT','FOXA1':'GTAAACAT','FOXA1_rc':'ATGTTTAC','PAX7':'TAATCAATTA','PAX7_rc':'TAATTGATTA','GATA3':'AGATAAGA','GATA3_rc':'TCTTATCT','KLF4':'CCCCACCC','KLF4_rc':'GGGTGGGG','OCT4':'ATGCAAAT','OCT4_rc':'ATTTGCAT'}
def read(path,limit=None):
 seqs=[]; parts=[]
 for l in open(path):
  l=l.strip()
  if not l: continue
  if l.startswith('>'):
   if parts:
    seqs.append(''.join(parts).upper()); parts=[]
    if limit and len(seqs)>=limit: return seqs
  else: parts.append(l)
 if parts and (not limit or len(seqs)<limit): seqs.append(''.join(parts).upper())
 return seqs
s=read('stable.fa')
for top in [500,1000,2000,5000]:
 g=read('gained.top5000.fa',top)
 print('TOP',top)
 for name,pat in patterns.items():
  gh=sum(pat in x for x in g); sh=sum(pat in x for x in s)
  odds,p=fisher_exact([[gh,top-gh],[sh,len(s)-sh]],alternative='greater')
  print(name,pat,'OR',round(odds,3),'p','%.2g'%p,'frac',round(gh/top,4),round(sh/len(s),4),'hits',gh,sh)
 print()
