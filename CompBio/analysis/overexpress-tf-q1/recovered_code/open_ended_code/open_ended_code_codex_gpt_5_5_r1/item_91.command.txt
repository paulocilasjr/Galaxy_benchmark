/bin/bash -lc "python3 - <<'PY'
# Show FOXA1 PFM probabilities and top 8-mers by PWM score
import itertools, numpy as np, math
rows={}; cur=None
with open('JASPAR2024_CORE_vertebrates_non-redundant_pfms_jaspar.txt') as f:
 for line in f:
  line=line.strip()
  if line.startswith('>'):
   cur=line[1:].split()[1]
   rows={}
  elif cur=='FOXA1' and line:
   rows[line[0]]=np.array([float(x) for x in line[line.find('[')+1:line.find(']')].split()])
   if len(rows)==4:
    mat=np.vstack([rows[b] for b in 'ACGT'])
    probs=(mat+0.5)/(mat.sum(axis=0)+2)
    print('probs')
    for b,p in zip('ACGT',probs): print(b, ['%.2f'%x for x in p])
    log=np.log2(probs/0.25)
    km=[]
    for s in itertools.product('ACGT', repeat=8):
     seq=''.join(s); sc=sum(log['ACGT'.index(ch),i] for i,ch in enumerate(seq))
     km.append((sc,seq))
    for sc,seq in sorted(km, reverse=True)[:20]: print(seq,sc)
PY"
