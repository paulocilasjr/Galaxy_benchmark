/bin/bash -lc "paste summit_ref_counts.bed summit_pert_counts.bed | awk 'BEGIN{OFS=\"\\t\"} {r="'$4;p=$8;l=log((p+2)/(r+2)/(11257343/10141848))/log(2); print $1,$2,$3,r,p,l,r+p}'"' > summit_diff.tsv; awk '{if("'$7>=20)n++;if($7>=20&&$6>1)n1++;if($7>=20&&$6>2)n2++;if($7>=20&&$6<.2&&$6>-.2)st++}END{print n,n1,n2,st}'"' summit_diff.tsv; sort -k6,6gr summit_diff.tsv | awk '"'$5>=20'"' | head -n 15000 | cut -f1-3 > gain15k.bed; awk '"'$5>=20 && $6<0.15 && $6>-0.15'"' summit_diff.tsv | head -n 15000 | cut -f1-3 > stable15k.bed; awk '"'$5>=20 && $6< -0.5'"' summit_diff.tsv | sort -k6,6g | head -n 15000 | cut -f1-3 > loss_from_pert.bed; wc -l gain15k.bed stable15k.bed loss_from_pert.bed; bedtools getfasta -fi hg38.fa -bed gain15k.bed -fo gain15k.fa; bedtools getfasta -fi hg38.fa -bed stable15k.bed -fo stable15k.fa; bedtools getfasta -fi hg38.fa -bed loss_from_pert.bed -fo loss_from_pert.fa"
