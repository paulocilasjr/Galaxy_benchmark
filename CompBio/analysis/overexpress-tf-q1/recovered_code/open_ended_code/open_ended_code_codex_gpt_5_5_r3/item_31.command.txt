/bin/bash -lc "python3 - <<'PY'
import zipfile,re
cands=['FOXA1','ASCL1','TFAP2A','RUNX1','KLF4','PAX7','TCF7','REST','SPI1','GATA3']
z=zipfile.ZipFile('PFMs.zip')
for name in z.namelist():
    if not name.endswith('.jaspar'): continue
    txt=z.read(name).decode().splitlines()[0]
    m=re.match(r'>\\s*(\\S+)\\s+(\\S+)',txt)
    if not m: continue
    mid,gene=m.groups()
    for c in cands:
        if c.lower()==gene.lower() or c.lower() in gene.lower():
            print(c, name, txt)
PY"
