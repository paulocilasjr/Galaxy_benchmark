/bin/bash -lc "python3 - <<'PY'
import zipfile,re
z=zipfile.ZipFile('PFMs.zip')
for mid in ['MA0680.3','MA0148.5','MA0037.5','MA0769.3']:
    txt=z.read(mid+'.jaspar').decode().strip().splitlines()
    print(txt[0])
    mat=[]
    for line in txt[1:]:
        nums=[float(x) for x in re.findall(r'[0-9.]+',line)]; mat.append(nums)
    cons=''
    for col in zip(*mat):
        cons+='ACGT'[max(range(4), key=lambda i: col[i])]
    print(cons)
PY"
