/bin/bash -lc "python3 - <<'PY'
import gzip, collections, statistics
for fn in ['deleterious.mutation.q1.R1.fq.gz','deleterious.mutation.q1.R2.fq.gz']:
    lens=[]; gc=[]; n=0
    with gzip.open(fn,'rt') as f:
        while True:
            h=f.readline()
            if not h: break
            s=f.readline().strip(); f.readline(); f.readline()
            n += 1; lens.append(len(s)); gc.append((s.count('G')+s.count('C'))/len(s))
    print(fn, 'n',n,'lengths',collections.Counter(lens).most_common(10),'minmax',min(lens),max(lens),'median',statistics.median(lens),'GCmean',sum(gc)/n)
PY
printf '\\nConda info:\\n'; conda --version 2>&1; conda info --envs 2>&1 | head -30"
