/bin/bash -lc "command -v jq || true
awk '{print \"9 \" "'$2 " " $2 " " $3 "/" $4}'"' hom_snvs.tsv | jq -R -s '{variants:(split(\"\\n\")|map(select(length>0)))}' > vep_request.json
head -c 300 vep_request.json

wc -c vep_response.json
jq -r '.[] | .input as "'$i | .transcript_consequences[]? | select((.consequence_terms|index("stop_gained"))) | [$i,.gene_symbol,.gene_id,.transcript_id,.biotype,.canonical,.mane_select,.impact,.hgvsc,.hgvsp] | @tsv'"' vep_response.json"
