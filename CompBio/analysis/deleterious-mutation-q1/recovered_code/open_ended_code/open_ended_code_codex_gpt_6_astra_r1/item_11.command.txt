/bin/bash -lc "cat > run_alignment.sh <<'SH'
#"'!/bin/bash
set -euo pipefail
export TMPDIR="$PWD/tmp"
bwa mem -t 4 -R '"'@RG\\tID:sample\\tSM:sample\\tPL:ILLUMINA' reference/hg38.chr9.fa deleterious.mutation.q1.R1.fq.gz deleterious.mutation.q1.R2.fq.gz 2> results/bwa_mem.log | samtools sort -n -@ 2 -T tmp/names -o results/name_sorted.bam
samtools fixmate -m results/name_sorted.bam results/fixmate.bam
samtools sort -@ 2 -T tmp/coord -o results/sorted.bam results/fixmate.bam
samtools markdup results/sorted.bam results/aligned.bam 2> results/markdup.log
samtools index results/aligned.bam
samtools flagstat results/aligned.bam > results/flagstat.txt
bcftools mpileup -Ou -f reference/hg38.chr9.fa -q 20 -Q 20 -a FORMAT/DP,FORMAT/AD,FORMAT/ADF,FORMAT/ADR results/aligned.bam 2> results/mpileup.log | bcftools call -mv -Ov -o results/variants.vcf 2> results/call.log
python3 annotate.py > results/stop_gains.txt
SH"
