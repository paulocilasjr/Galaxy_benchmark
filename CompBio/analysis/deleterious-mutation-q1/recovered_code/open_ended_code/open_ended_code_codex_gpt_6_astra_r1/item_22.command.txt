/bin/bash -lc "python3 - <<'PY'
import pysam,collections,json
bam=pysam.AlignmentFile('results/aligned.bam'); pos=127661125-1
reads=[]
for r in bam.fetch('chr9',pos,pos+1):
 if r.is_unmapped or r.is_secondary or r.is_supplementary or r.is_duplicate or r.mapping_quality<20: continue
 for q,p in r.get_aligned_pairs(matches_only=True):
  if p==pos:
   reads.append(dict(name=r.query_name,base=r.query_sequence[q],BQ=r.query_qualities[q],MQ=r.mapping_quality,strand='-' if r.is_reverse else '+',readpos=q+1,length=r.query_length,start=r.reference_start,cigar=r.cigarstring))
print('Reads:',len(reads),'fragments:',len(set(r['name'] for r in reads)),'bases:',dict(collections.Counter(r['base'] for r in reads)))
print(json.dumps(reads,indent=2));json.dump(reads,open('results/candidate_reads.json','w'),indent=2)
PY"
