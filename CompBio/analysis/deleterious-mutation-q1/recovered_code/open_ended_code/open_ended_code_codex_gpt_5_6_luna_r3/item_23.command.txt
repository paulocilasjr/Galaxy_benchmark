/bin/bash -lc "bcftools query -i 'TYPE=\"snp\" && GT=\"1/1\" && INFO/BCSQ~\"stop_gained\"' -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%GT\\t%DP\\t%AD]\\t%INFO/BCSQ\\n' chr9.hg38.q1.csq.vcf.gz 2>/dev/null > hom_stop_all.tsv
python3 - <<'PY'
from pathlib import Path
rows=[]
for line in Path('hom_stop_all.tsv').read_text().splitlines():
    f=line.split('\\t')
    csq=f[8] if len(f)>8 else ''
    cons=[]
    for c in csq.split(','):
        x=c.split('|')
        if x and x[0].startswith('stop_gained') and len(x)>=6:
            cons.append((x[1],x[2],x[3],x[5]))
    if cons:
        rows.append((f[:8],list(dict.fromkeys(cons))))
print('records',len(rows))
for f,cs in rows:
    print('\\t'.join(f), ';'.join('|'.join(x) for x in cs[:5]), ('...'+str(len(cs)) if len(cs)>5 else ''))
PY"
