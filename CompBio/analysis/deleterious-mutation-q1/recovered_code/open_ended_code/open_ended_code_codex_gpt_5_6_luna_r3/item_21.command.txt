/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
p=Path('hom_stop_candidates.tsv')
lines=p.read_text().splitlines()
print('candidate_count',len(lines))
for line in lines:
    f=line.split('\\t')
    print('variant', '\\t'.join(f[:7]))
    csq=f[7] if len(f)>7 else ''
    genes=[]
    for c in csq.split(','):
        if c.startswith('stop_gained'):
            x=c.split('|')
            genes.append((x[1],x[2],x[3],x[5] if len(x)>5 else ''))
    print('stop_consequences',len(genes))
    for x in dict.fromkeys(genes): print('\\t', '\\t'.join(x))
PY"
