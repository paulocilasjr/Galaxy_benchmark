# build candidate list via bcftools
cmd = \"bcftools view -v snps q1.raw.vcf -Ou | bcftools query -f '%CHROM\\\\t%POS\\\\t%REF\\\\t%ALT\\\\t%QUAL[\\\\t%DP\\\\t%AD\\\\t%GT]\\\\n'\"
out = subprocess.check_output(cmd, shell=True, text=True, stderr=subprocess.DEVNULL)
variants=[]

    if len(f)<8: continue

    dp=int(dp); adp=ad.split(',')

    # include SNVs high confidence homo alt, or all high-quality SNVs
    if gt=='1/1' and dp>=8 and af>=0.8:
        variants.append((chrom,pos,ref,alt,qual,dp,ad,gt,af))

# POST in batches of 200

        records.extend(r.json())
