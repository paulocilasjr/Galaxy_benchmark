/bin/bash -lc 'for asm in hg38 hg19; do
  echo ${asm}_all_variants
  bcftools view -H calls/${asm}.vcf.gz | wc -l
  echo ${asm}_hom_highconf
  bcftools query -f '"'%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%GT\\t%DP\\t%AD\\t%GQ]\\n' calls/"'${asm}.vcf.gz '"\\
    | awk -F'\\t' 'length("'$3)==1 && length($4)==1 && $6=="1/1" && $7>=10 && $8 ~ /''^[0-9]+,[0-9]+$/ {split($8,a,","); if (a[2]/(a[1]+a[2])>=0.9 && $5>=30) print}'"' \\
    | tee calls/"'${asm}.hom.highconf.tsv | wc -l
  sed -n '"'1,200p' calls/"'${asm}.hom.highconf.tsv
done'
