/bin/bash -lc 'curl -L --fail --retry 3 -o chr9.hg38.fa.gz https://hgdownload.soe.ucsc.edu/goldenPath/hg38/chromosomes/chr9.fa.gz
ls -lh chr9.hg38.fa.gz
gzip -cd chr9.hg38.fa.gz > chr9.hg38.fa
bwa index chr9.hg38.fa
samtools faidx chr9.hg38.fa
bwa mem -t 4 chr9.hg38.fa deleterious.mutation.q1.R1.fq.gz deleterious.mutation.q1.R2.fq.gz | samtools sort -@ 4 -o aligned.hg38.bam
samtools index aligned.hg38.bam
samtools flagstat aligned.hg38.bam'
