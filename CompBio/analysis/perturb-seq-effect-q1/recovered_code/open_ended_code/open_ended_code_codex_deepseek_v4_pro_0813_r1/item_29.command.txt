/bin/bash -lc "python - <<'PY'
import gzip, numpy as np, pandas as pd
from scipy.io import mmread
from scipy import stats
from statsmodels.stats.multitest import multipletests

features=[]
with gzip.open('filtered_feature_bc_matrix/features.tsv.gz','rt') as f:
    for line in f: features.append(line.rstrip('\\n').split('\\t')[1])
barcodes=[]
with gzip.open('filtered_feature_bc_matrix/barcodes.tsv.gz','rt') as f:
    barcodes=[line.strip() for line in f]
mat=mmread('filtered_feature_bc_matrix/matrix.mtx.gz').tocsc()
calls=pd.read_csv('protospacer_calls_per_cell.csv')
ref=pd.read_csv('feature_reference.csv')
calls['target']=calls.feature_call.map(dict(zip(ref.id, ref.target_gene_name)))
bc2col={b:i for i,b in enumerate(barcodes)}
calls['col']=calls.cell_barcode.map(bc2col)
lib=np.asarray(mat.sum(axis=0)).ravel()
path=features[:20]
idx=[features.index(g) for g in path]
log=np.log2(1 + np.asarray(mat[idx,:].todense())*(10000/lib[None,:]))
# zscore each gene across cells
z=(log-log.mean(axis=1,keepdims=True))/log.std(axis=1,keepdims=True)
score=z.mean(axis=0)
df=pd.DataFrame({'score':score,'cell':barcodes}).merge(calls[['cell_barcode','target','col']], left_on='cell', right_on='cell_barcode')
nt=df[df.target=='Non-Targeting'].score
print('NT mean zscore',nt.mean())
rows=[]
for t,g in df[df.target"'!='"'Non-Targeting'].groupby('target'):
    p=stats.mannwhitneyu(g.score,nt,alternative='two-sided').pvalue
    rows.append((t,g.score.mean()-nt.mean(),p,len(g),g.score.mean()))
res=pd.DataFrame(rows,columns=['target','diff','p','n','mean']).sort_values('diff')
res['bh']=multipletests(res.p, method='fdr_bh')[1]
print(res.head(15).to_string(index=False))
PY"
