/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
p='extracted'
f=pd.read_csv(p+'/filtered_feature_bc_matrix/features.tsv.gz',sep='\\t',header=None)
c=pd.read_csv(p+'/protospacer_calls_per_cell.csv')
r=pd.read_csv(p+'/feature_reference.csv')
assert len(f)==1000 and len(c)==2455 and len(r)==50
assert np.array_equal(pd.read_csv(p+'/filtered_feature_bc_matrix/barcodes.tsv.gz',header=None).iloc[:,0].values,c.cell_barcode.values)
assert r.loc[r.id=='guide_011','target_gene_name'].iat[0]=='STAT1'
X=mmread(p+'/filtered_feature_bc_matrix/matrix.mtx.gz').tocsr(); assert X.shape==(1000,2455)
lib=np.asarray(X.sum(axis=0)).ravel(); Z=X.multiply(1e4/lib).tocsr(); Z.data=np.log1p(Z.data)
ref=r.set_index('id').target_gene_name.to_dict(); c['target']=c.feature_call.map(ref)
panel=np.asarray(Z[:20].mean(axis=0)).ravel(); ctrl=panel[c.target.eq('Non-Targeting').values]
res=[]
for gid,g in c.groupby('feature_call'):
 v=panel[g.index]
 res.append((v.mean(),np.median(v),gid,g.target.iloc[0],len(v),np.sum(np.asarray(Z[:20,g.index].mean(axis=0)).ravel()<np.quantile(ctrl,.25))))
res.sort()
print('lowest:',res[:4])
print('top target',res[0][3])
PY"
