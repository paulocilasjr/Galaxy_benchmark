/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
from scipy.io import mmread
from pathlib import Path
p=Path('extracted')
f=pd.read_csv(p/'filtered_feature_bc_matrix/features.tsv.gz',sep='\\t',header=None,names=['id','name','type'])
c=pd.read_csv(p/'protospacer_calls_per_cell.csv')
ref=pd.read_csv(p/'feature_reference.csv'); mp=ref.set_index('id').target_gene_name.to_dict()
c['target']=c.feature_call.map(mp)
X=mmread(p/'filtered_feature_bc_matrix/matrix.mtx.gz').tocsr(); lib=np.asarray(X.sum(0)).ravel(); N=X.multiply(1e4/lib).tocsr(); L=N.copy(); L.data=np.log1p(L.data)
ctrl=np.flatnonzero(c.target.values=='Non-Targeting')
rows=[]
for gid,g in c.groupby('feature_call'):
 idx=g.index.to_numpy(); target=g.target.iloc[0]
 j=np.flatnonzero(f.name.values==target)
 if len(j)==1:
  vals=np.asarray(N[j[0],idx].todense()).ravel(); cv=np.asarray(N[j[0],ctrl].todense()).ravel()
  lv=np.asarray(L[j[0],idx].todense()).ravel(); lcv=np.asarray(L[j[0],ctrl].todense()).ravel()
  rows.append((gid,target,len(idx),vals.mean(),np.median(vals),cv.mean(),np.mean(lv)-np.mean(lcv),np.sum(vals==0),np.std(lv,ddof=1) if len(lv)>1 else np.nan))
o=pd.DataFrame(rows,columns=['guide','target','n','target_mean','target_med','ctrl_mean','target_logfc','zeros','target_log_sd']).sort_values('target_logfc')
print(o.to_string(index=False,float_format=lambda x:f'{x:.4f}'))
PY"
