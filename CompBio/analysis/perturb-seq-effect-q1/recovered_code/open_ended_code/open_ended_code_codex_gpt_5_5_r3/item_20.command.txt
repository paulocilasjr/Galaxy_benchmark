/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from statsmodels.stats.multitest import multipletests
for file in ['first20_guide_effects.csv','hallmark_subset_guide_effects.csv','first20_no_targetgenes_guide_effects.csv']:
    df=pd.read_csv(file)
    df['fdr']=multipletests(df['p'], method='fdr_bh')[1]
    print('\\n',file)
    print(df.sort_values('diff').head(8)[['target','guide','n','diff','p','fdr']].to_string(index=False,float_format=lambda x:f'{x:.4g}'))
    print('robust n>=20 fdr<0.05 diff<0:', df[(df.n>=20)&(df.fdr<0.05)&(df['diff']<0)][['target','n','diff','p','fdr']].sort_values('diff').to_string(index=False,float_format=lambda x:f'{x:.4g}'))
PY"
