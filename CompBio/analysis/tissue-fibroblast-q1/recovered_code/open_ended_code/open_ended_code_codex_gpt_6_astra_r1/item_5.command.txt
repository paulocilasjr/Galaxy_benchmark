/bin/bash -lc "cat > analyze.R <<'EOF'
suppressPackageStartupMessages(library(Seurat))
x<-readRDS('tissue.fibroblast.q1.rds')
s<-CreateSeuratObject(x)
s<-NormalizeData(s,verbose=FALSE)
s<-FindVariableFeatures(s,nfeatures=2500,verbose=FALSE)
s<-ScaleData(s,verbose=FALSE)
s<-RunPCA(s,npcs=40,verbose=FALSE)
e<-Embeddings(s,'pca')
for(b in c('Cell_11249_MCI','Cell_10369_WXV')){
 d<-rowSums(sweep(e,2,e[b,])"'^2)
 nn<-names(sort(d))[1:60]
 cat('"'\\nTARGET',b,'\\n'); print(head(nn,15))
 a<-GetAssayData(s,layer='data'); av<-Matrix::rowMeans(a[,nn]); bg<-Matrix::rowMeans(a)
 score<-av-bg
 print(head(sort(score,decreasing=TRUE),65))
 write.csv(data.frame(gene=rownames(a),mean=av,bg=bg,score=score),paste0(b,'_neighbors.csv'),row.names=FALSE)
}
saveRDS(s,'analyzed.rds')
EOF
Rscript analyze.R"
