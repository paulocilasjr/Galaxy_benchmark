/bin/bash -lc "Rscript -e '
suppressMessages(library(Seurat)); suppressMessages(library(Matrix));
cat(\"loading query\\n\"); q <- readRDS(\"tissue.fibroblast.q1.rds\");
cat(\"loading full\\n\"); full <- readRDS(\"Mouse_SS_Fibro.RDS\");
cat(\"extracting counts\\n\"); counts <- GetAssayData(full[[\"RNA\"]], layer=\"counts\");
cat(\"counts dim:\", paste(dim(counts), collapse=\"x\"), \"\\n\");
targets <- c(\"Cell_11249_MCI\",\"Cell_10369_WXV\");
for (bc in targets) {
  qi <- which(colnames(q)==bc); cat(\"\\nTarget\", bc, \"col\", qi, \"\\n\");
  qv <- q[, qi, drop=FALSE];
  rows <- qv@i + 1; vals <- qv@x; total <- sum(vals);
  cat(\"n nonzero\", length(rows), \"total\", total, \"\\n\");
  sub <- counts[rows, , drop=FALSE];
  cat(\"sub class\", class(sub), \"dim\", paste(dim(sub), collapse=\"x\"), \"\\n\");
  eq <- sub == vals;
  cat(\"eq class\", class(eq), \"\\n\");
  matched_nz <- Matrix::colSums(eq) == length(rows);
  totals <- Matrix::colSums(counts);
  matched_total <- totals == total;
  cand <- which(matched_nz & matched_total);
  cat(\"candidates\", length(cand), \"\\n\");
  if (length(cand) > 0) {
    print(head(data.frame(original=colnames(counts)[cand], total=totals[cand], nFeature=full"'$nFeature_RNA[cand], Tissue=full$Tissue[cand], Cluster=full$ClustName[cand]), 30))
  } else {
    cat("no exact match'"\\n\");
  }
}
cat(\"done\\n\")
'"
