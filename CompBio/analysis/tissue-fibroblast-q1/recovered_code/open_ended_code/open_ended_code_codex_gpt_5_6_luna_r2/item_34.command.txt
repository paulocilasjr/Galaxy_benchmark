/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
x<-readRDS('tissue.fibroblast.q1.rds'); genes<-rownames(x); cells<-colnames(x)
# target2 cosine tf-idf neighbors
cs<-colSums(x); df<-rowSums(x>0); w<-log1p(ncol(x)/pmax(df,1)); z<-x*w; z<-z%*%Diagonal(x=1/sqrt(colSums(z"'^2)))
j<-match('"'Cell_10369_WXV',cells); sim<-as.numeric(crossprod(z[,j],z)); sim[j]<--Inf
for (n in c(20,50,100,200,500)) {
 ii<-order(sim,decreasing=T)[1:n]
 # log normalized mean in cohort vs all, rank by logFC
 norm<-x%*%Diagonal(x=1/colSums(x))*10000
 a<-log1p(rowMeans(norm[,ii])); b<-log1p(rowMeans(norm[,-ii])); fc<-a-b
 o<-order(fc,decreasing=T)
 cat('\\nN',n,'sim range',sim[ii[n]],'\\n')
 print(data.frame(gene=genes[o[1:60]],logdiff=round(fc[o[1:60]],2),pct=round(100*rowMeans(x[,ii]>0)[o[1:60]],1)),row.names=F)
}
RS"
