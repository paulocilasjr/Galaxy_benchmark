/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
x <- readRDS(\"tissue.fibroblast.q1.rds\")
cat(\"dim\", paste(dim(x), collapse=\" x \"), \"\\n\")
for (bc in c(\"Cell_11249_MCI\", \"Cell_10369_WXV\")) {
  j <- match(bc, colnames(x))
  v <- x[,j]
  idx <- which(v > 0)
  ord <- idx[order(v[idx], decreasing=TRUE)]
  cat(\"\\n\", bc, \"column\", j, \"total\", sum(v), \"detected\", length(idx), \"\\n\", sep=\"\")
  print(data.frame(gene=rownames(x)[ord][seq_len(min(100,length(ord)))], count=as.numeric(v[ord][seq_len(min(100,length(ord)))]), row.names=NULL), row.names=FALSE)
}
RS"
