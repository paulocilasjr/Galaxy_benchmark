/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
x <- readRDS(\"tissue.fibroblast.q1.rds\")
cs <- Matrix::colSums(x)
df <- Matrix::rowSums(x > 0)
w <- log1p(ncol(x) / pmax(df, 1))
z <- x * w; z <- z %*% Diagonal(x=1/sqrt(Matrix::colSums(z"'^2)))
for (bc in c("Cell_11249_MCI", "Cell_10369_WXV")) {
 j <- match(bc,colnames(x)); sim <- as.numeric(crossprod(z[,j], z)); sim[j] <- -Inf
 ii <- order(sim,decreasing=TRUE)[1:20]
 cat("'"\\nTARGET\",bc,\"\\n\")
 for (k in seq_along(ii)) {
   jj<-ii[k]; v<-x[,jj]; ix<-which(v>0); o<-ix[order(v[ix],decreasing=TRUE)]; top<-rownames(x)[o][1:min(15,length(o))]
   cat(k,colnames(x)[jj],round(sim[jj],3),paste(top,collapse=\",\"),\"\\n\")
 }
}
RS"
