/bin/bash -lc "python3 - <<'PY'
from zipfile import ZipFile
from xml.etree import ElementTree as ET
p='supplement_table1.xlsx'
ns='{http://schemas.openxmlformats.org/spreadsheetml/2006/main}'
with ZipFile(p) as z:
    ss=[]
    root=ET.fromstring(z.read('xl/sharedStrings.xml'))
    for si in root.findall(ns+'si'):
        ss.append(''.join(t.text or '' for t in si.iter(ns+'t')))
    wb=ET.fromstring(z.read('xl/workbook.xml'))
    rel=ET.fromstring(z.read('xl/_rels/workbook.xml.rels'))
    relns='{http://schemas.openxmlformats.org/package/2006/relationships}'
    mp={r.attrib['Id']:r.attrib['Target'] for r in rel}
    for sh in wb.find(ns+'sheets')[:10]:
        name=sh.attrib['name']; f='xl/'+mp[sh.attrib['{http://schemas.openxmlformats.org/officeDocument/2006/relationships}id']]
        rt=ET.fromstring(z.read(f))
        rows=[]
        for row in rt.iter(ns+'row'):
            vals=[]
            for c in row.findall(ns+'c'):
                v=c.find(ns+'v'); val='' if v is None else v.text
                if c.attrib.get('t')=='s' and val: val=ss[int(val)]
                vals.append((c.attrib.get('r'),val))
            if vals: rows.append(vals)
        print('\\nSHEET',name,'rows',len(rows))
        for row in rows[:12]: print(row)
PY"
