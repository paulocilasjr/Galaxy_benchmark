/bin/bash -lc "Rscript - <<'RS'
suppressPackageStartupMessages(library(Matrix))
x <- readRDS('tissue.fibroblast.q1.rds')
q <- c('Cell_11249_MCI','Cell_10369_WXV')
for (cell in q) {
  v <- x[,cell]
  z <- sort(as.numeric(v), decreasing=TRUE, index.return=TRUE)
  ix <- z"'$ix[z$x>0][1:min(100,sum(z$x>0))]
  cat('"'\\nCELL',cell,'\\n')
  print(data.frame(gene=rownames(x)[ix], count=z"'$x[seq_along(ix)]),row.names=FALSE)
  cat('"'total=',sum(v),'detected=',sum(v>0),'\\n')
}
RS"
