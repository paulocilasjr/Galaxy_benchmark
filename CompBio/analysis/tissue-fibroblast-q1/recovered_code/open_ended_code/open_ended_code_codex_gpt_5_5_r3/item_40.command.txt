/bin/bash -lc "Rscript -e '
library(Matrix)
local <- readRDS(\"tissue.fibroblast.q1.rds\")
obj <- readRDS(\"Mouse_SS_Fibro.RDS\")
md <- attr(obj, \"meta.data\")
ref <- attr(attr(obj, \"assays\")"'$RNA, "counts")
targets <- c("Cell_11249_MCI","Cell_10369_WXV")
for (t in targets) {
  v <- local[, t, drop=FALSE]
  nc <- sum(v); nf <- sum(v != 0)
  cand <- rownames(md)[md$nCount_RNA == nc & md$nFeature_RNA == nf]
  cat("TARGET", t, "nCount", nc, "nFeature", nf, "candidates", length(cand), "'"\\n\")
  exact <- character()
  for (cc in cand) {
    rv <- ref[, cc, drop=FALSE]
    if (length(rv@x) == length(v@x) && all(rv@i == v@i) && all(rv@x == v@x)) exact <- c(exact, cc)
  }
  cat(\"exact\", length(exact), paste(exact, collapse=\",\"), \"\\n\")
  if (length(exact)) print(md[exact,,drop=FALSE])
}
'"
