/bin/bash -lc "Rscript - <<'RS'
obj <- readRDS('mouse_steady_atlas.rds')
att <- attributes(obj)
md <- att"'$meta.data
cat('"'metadata columns:\\n'); print(colnames(md)); print(head(md, 10)); cat('\\ncolumn value summaries:\\n')
for (cn in colnames(md)) { cat('\\n',cn,'\\n'); print(head(sort(table(md[[cn]]), decreasing=TRUE), 30)) }
cat('\\nassays structure\\n')
print(str(att"'$assays, max.level=3))
RNA <- att$assays[[1]]
cat('"'RNA attrs:', paste(names(attributes(RNA)), collapse=','), '\\n')
print(str(attributes(RNA), max.level=2))
RS"
