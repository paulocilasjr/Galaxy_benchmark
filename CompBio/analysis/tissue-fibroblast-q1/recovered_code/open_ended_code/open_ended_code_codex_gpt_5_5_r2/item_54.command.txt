/bin/bash -lc "Rscript - <<'RS'
obj <- readRDS('mouse_steady_atlas.rds')
RNA <- attributes(obj)"'$assays[[1]]
cat('"'RNA isS4', isS4(RNA), 'class', paste(class(RNA), collapse=','), '\\n')
cat('RNA attribute names:', paste(names(attributes(RNA)), collapse=','), '\\n')
for (nm in names(attributes(RNA))) {
  x <- attributes(RNA)[[nm]]
  cat('ATTR', nm, 'class', paste(class(x), collapse=','), 'length', length(x), '\\n')
  if (nm %in% c('counts','data','scale.data')) { cat('  dim attr:', paste(attr(x,'Dim'), collapse=' x '), '\\n') }
}
RS"
