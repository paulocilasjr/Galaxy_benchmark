/bin/bash -lc "Rscript - <<'RS'
library(Matrix)
obj <- readRDS('tissue.fibroblast.q1.rds')
dn <- attr(obj,'Dimnames'); genes <- dn[[1]]; cells <- dn[[2]]
targets <- c('Cell_11249_MCI','Cell_10369_WXV')
col_norm <- sqrt(Matrix::colSums(obj"'^2))
for (tar in targets) {
  ti <- match(tar, cells)
  targ <- obj[,ti]
  idx <- which(targ != 0)
  dots <- as.numeric(Matrix::crossprod(targ[idx], obj[idx,,drop=FALSE]))
  sim <- dots/(col_norm[ti]*col_norm)
  sim[ti] <- NA
  top <- order(sim, decreasing=TRUE, na.last=NA)[1:200]
  nums <- as.integer(sub('"'Cell_([0-9]+)_.*','\\\\1', cells[top]))
  cat('\\nTARGET', tar, 'expressed genes', length(idx), 'top sim', round(sim[top[1]],3), round(sim[top[200]],3), '\\n')
  cat('top20:', paste(cells[top[1:20]], collapse=', '), '\\n')
  cat('nums summary:', paste(capture.output(print(summary(nums))), collapse=' '), '\\n')
  # top expressed genes in target
  tx <- sort(as.numeric(targ[idx]), decreasing=TRUE)
  names(tx) <- genes[idx][order(as.numeric(targ[idx]), decreasing=TRUE)]
  cat('target top genes:', paste(paste0(names(tx)[1:40],':',tx[1:40]), collapse=', '), '\\n')
  markers <- c('Adipoq','Lpl','Fabp4','Plin1','Cfd','Dlk1','Ccl19','Ccl21a','Ccl21b','Ccl21c','Cxcl13','Il7','Pdpn','Lyve1','Tnmd','Scx','Mkx','Col22a1','Thbs4','Myod1','Myog','Myl1','Tnnt3','Tnnt2','Myh6','Myh7','Nppa','Myl2','Alb','Apoa1','Cyp2e1','Rbp4','Sftpc','Sftpa1','Scgb1a1','Krt14','Krt5','Krt10','Krt17','Epcam','Muc2','Reg3g','Fabp1','Amy2a5','Prss2','Pnlip','Ins1','Ins2','Sost','Bglap','Ibsp','Mmp13','Spp1','Mgp','Vcam1','Icam1','Msln','Wt1','Upk3b','Pdgfra','Dcn','Col1a1','Col1a2','Lum','Clec3b','Pi16','Dpt','Col15a1','Cxcl14','Postn','Acta2','Tagln','Rgs5','Myh11','Wnt2','Hand2','Hic1')
  markers <- markers[markers %in% genes]
  raw <- obj[match(markers,genes), c(ti,top), drop=FALSE]
  det <- Matrix::rowMeans(raw[, -1, drop=FALSE] > 0)
  mean_expr <- Matrix::rowMeans(raw[, -1, drop=FALSE])
  tar_expr <- as.numeric(raw[,1])
  names(det) <- markers; names(mean_expr) <- markers; names(tar_expr) <- markers
  ord <- order(det, mean_expr, decreasing=TRUE)
  print(data.frame(gene=markers[ord][1:30], target=tar_expr[ord][1:30], det=round(det[ord][1:30],2), mean=round(mean_expr[ord][1:30],2)), row.names=FALSE)
}
RS"
