/bin/bash -lc "python3 - <<'PY'
import re,html
s=open('buechler_nature.pdf',encoding='utf-8',errors='ignore').read()
term='Data availability'
for m in re.finditer(re.escape(term),s,re.I):
    chunk=s[m.start():m.start()+8000]
    txt=re.sub(r'<["'^>]+>'"',' ',chunk)
    txt=html.unescape(txt)
    txt=re.sub(r'\\s+',' ',txt)
    print(txt[:5000])
    break
PY"
