/bin/bash -lc "python - <<'PY'
import xml.etree.ElementTree as ET, re, html
root=ET.parse('contig_result.xml').getroot()
rows=[]
for it in root.findall('.//Iteration'):
 q=it.findtext('Iteration_query-def') or it.findtext('Iteration_query-ID') or ''
 hits=it.find('Iteration_hits')
 if hits is None or not list(hits): continue
 hit=list(hits)[0]
 desc=html.unescape(hit.findtext('Hit_def') or '')
 acc=hit.findtext('Hit_accession') or ''
 hsps=hit.find('Hit_hsps')
 hsp=list(hsps)[0] if hsps is not None and list(hsps) else None
 vals=[hsp.findtext(x) if hsp is not None else '' for x in ('Hsp_identity','Hsp_align-len','Hsp_evalue')]
 rows.append((q,acc,desc,*vals))
print('iterations with hits',len(rows))
for x in rows[:80]: print('\\t'.join(x))
print('\\nfirst-hit broad labels:')
from collections import Counter
c=Counter()
for q,a,d,i,l,e in rows:
 m=re.match(r'(?:PREDICTED: )?([A-Z][A-Za-z]+ [a-z]+)',d)
 c[m.group(1) if m else d[:60]] += 1
for x,n in c.most_common(30): print(n,x)
PY"
