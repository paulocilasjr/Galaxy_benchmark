/bin/bash -lc "python - <<'PY'
import xml.etree.ElementTree as ET, re, html
from collections import Counter
root=ET.parse('blast500.xml').getroot(); c=Counter(); n=0
for it in root.findall('.//Iteration'):
 hs=it.find('Iteration_hits')
 if hs is None or not list(hs): continue
 n+=1
 d=html.unescape(list(hs)[0].findtext('Hit_def') or '')
 m=re.match(r'(?:PREDICTED: )?([A-Z][A-Za-z]+ [a-z]+)',d)
 c[m.group(1) if m else d[:70]]+=1
print('iterations',n)
for k,v in c.most_common(50): print(v,k)
PY"
