/bin/bash -lc "gzip -cd unmapped.fq.gz | awk 'NR%4==1{print \">\" substr("'$0,2); next} NR%4==2{print}'"' | awk 'BEGIN{n=0} /"'^>/{n++; if(n>500) exit} {print}'"' > unmapped500.fa; wc -l unmapped500.fa; curl -L --max-time 30 -s -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' -H 'User-Agent: compbiobench-agent/1.0' --data-urlencode CMD=Put --data-urlencode PROGRAM=blastn --data-urlencode DATABASE=nt --data-urlencode MEGABLAST=off --data-urlencode WORD_SIZE=7 --data-urlencode EXPECT=1000 --data-urlencode HITLIST_SIZE=5 --data-urlencode FORMAT_TYPE=XML --data-urlencode QUERY@unmapped500.fa > blast500_put.html; grep -Eo 'RID = [A-Z0-9]+' blast500_put.html | tail -1; grep -Eo 'RTOE = [0-9]+' blast500_put.html | tail -1"
