/bin/bash -lc "python3 - <<'PY'
import requests
rid='76ANJJ8W016'
for fmt in ['Tabular','Text','XML']:
    r=requests.get('https://blast.ncbi.nlm.nih.gov/Blast.cgi',params={'CMD':'Get','RID':rid,'FORMAT_TYPE':fmt,'DESCRIPTIONS':'50','ALIGNMENTS':'50'},timeout=60)
    print('FMT',fmt,'status',r.status_code,'len',len(r.text))
    print(r.text[:1000].replace('\\n',' | '))
    open(f'blast_api_top1_{fmt}.txt','w').write(r.text)
PY"
