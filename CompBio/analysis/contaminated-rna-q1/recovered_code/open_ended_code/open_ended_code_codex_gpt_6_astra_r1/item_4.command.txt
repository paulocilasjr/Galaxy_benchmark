/bin/bash -lc "python3 - <<'PY'
import random
lines=open('reads.fa').readlines();random.seed(11)
with open('sample.fa','w') as f:
 for j in random.sample(range(len(lines)//2),100):f.writelines(lines[2*j:2*j+2])
PY
blastn -remote -db nt -query sample.fa -out sample.blast.tsv -outfmt '6 qseqid sacc pident length evalue bitscore staxids sscinames stitle' -max_target_seqs 5 -evalue 0.0001"
