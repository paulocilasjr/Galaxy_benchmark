/bin/bash -lc "python3 - <<'PY'
import gzip,collections
c=collections.Counter(); lens=collections.Counter(); names=[]
with gzip.open('contaminated.rna.q1.fq.gz','rt') as f:
 for i,line in enumerate(f):
  if i%4==0 and len(names)<10:names.append(line.strip())
  if i%4==1:c[line.strip()]+=1;lens[len(line.strip())]+=1
print('reads',sum(c.values()),'unique',len(c),'lengths',lens)
print('names',names)
print('top seqs',c.most_common(20))
with open('reads.fa','w') as f:
 for i,(seq,n) in enumerate(c.items()):f.write(f'>r{i}_n{n}\\n{seq}\\n')
PY"
