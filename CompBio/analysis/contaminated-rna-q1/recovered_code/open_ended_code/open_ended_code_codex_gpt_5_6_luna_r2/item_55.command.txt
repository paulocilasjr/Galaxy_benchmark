/bin/bash -lc "python3 - <<'PY'
import re,xml.etree.ElementTree as ET
from collections import Counter
root=ET.parse('unmapped_sample.nt.xml').getroot()
for key in ['Hydra vulgaris','Homo sapiens']:
    c=Counter()
    n=0
    for it in root.findall('.//Iteration'):
        h=it.find('./Iteration_hits/Hit')
        if h is None: continue
        d=h.findtext('Hit_def') or ''
        if key in d:
            n+=1;c[d]+=1
    print('\\n',key,n)
    for d,k in c.most_common(25): print(k,d)
PY"
