/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as ET
from collections import Counter
p='unmapped_sample.nt.xml'
r=ET.parse(p).getroot()
its=r.findall('.//Iteration')
print('iterations',len(its),'hits',sum(len(i.findall('./Iteration_hits/Hit')) for i in its))
for it in its[:5]:
    q=it.findtext('Iteration_query-def')
    h=it.find('./Iteration_hits/Hit')
    if h is None: print(q,'NOHIT'); continue
    print(q, h.findtext('Hit_def'), h.findtext('.//Hsp_identity'),h.findtext('.//Hsp_align-len'),h.findtext('.//Hsp_evalue'))
print('\\nTop hit defs:')
for d,n in Counter(it.find('./Iteration_hits/Hit/Hit_def').text if it.find('./Iteration_hits/Hit/Hit_def') is not None else 'NOHIT' for it in its).most_common(30): print(n,d)
PY"
