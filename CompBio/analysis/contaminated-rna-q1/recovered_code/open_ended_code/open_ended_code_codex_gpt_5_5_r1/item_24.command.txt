/bin/bash -lc "awk 'BEGIN{srand(2)} /"'^>/{h=$0; getline s; if(rand()<0.03){print h"'"\\n\"s}}' results/nonhuman_candidates.fa > results/candidates_rand.fa && grep -c '"'^>'"' results/candidates_rand.fa && awk 'BEGIN{n=0} /"'^>/{n++} n<=300{print}'"' results/candidates_rand.fa > results/candidates_300.fa && grep -c '"'^>'"' results/candidates_300.fa
blastn -remote -db nt -query results/candidates_300.fa -task megablast -max_target_seqs 10 -evalue 1e-10 -outfmt '6 qseqid saccver pident length evalue bitscore staxids stitle' > results/candidates_300.blast.tsv"
