/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
import random
recs=list(SeqIO.parse('q1_unmapped_unclass_unique.fa','fasta'))
random.seed(42)
random.shuffle(recs)
SeqIO.write(recs[:10], 'sample10.fa', 'fasta')
print(len(recs))
PY
time blastn -remote -db nt -query sample10.fa -out sample10.out -outfmt '6 qseqid sseqid staxids sscinames scomnames pident length mismatch gapopen qstart qend sstart send evalue bitscore' -max_target_seqs 3 -evalue 1e-5 -task megablast 2>&1 | tail -20; cat sample10.out | head -50"
