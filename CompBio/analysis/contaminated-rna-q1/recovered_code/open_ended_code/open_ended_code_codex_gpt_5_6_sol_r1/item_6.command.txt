/bin/bash -lc 'gzip -dc human_refseq_rna.fna.gz > human_refseq_rna.fna
minimap2 -d human_refseq_rna.mmi human_refseq_rna.fna
minimap2 -ax sr -t 8 human_refseq_rna.mmi contaminated.rna.q1.fq.gz | samtools view -@ 4 -b -o human_rna_map.bam -
samtools fastq -f 4 human_rna_map.bam | gzip > unmapped_human_rna.fq.gz
samtools flagstat human_rna_map.bam
ls -lh human_refseq_rna.fna human_refseq_rna.mmi human_rna_map.bam unmapped_human_rna.fq.gz'
