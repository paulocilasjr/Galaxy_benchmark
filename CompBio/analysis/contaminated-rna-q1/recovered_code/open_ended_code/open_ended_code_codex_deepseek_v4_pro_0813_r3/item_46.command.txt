/bin/bash -c "python3 - <<'PY'
from collections import defaultdict
hits=defaultdict(list)
for line in open('blast_hydra_100.tsv'):
 p=line.rstrip().split('\\t'); q=p[0]; ident=float(p[2]); hits[q].append(ident)
print('queries with Hydra hit', len(hits))
bins={}
for q,ids in hits.items():
 best=max(ids)
 for lo in [100,99,98,97,95,90,80]:
  if best>=lo:
   bins.setdefault(f'>={lo}',0); bins[f'>= {lo}']=bins.get(f'>= {lo}',0)+1
   break
for k,v in sorted(bins.items()): print(k,v)
PY"
