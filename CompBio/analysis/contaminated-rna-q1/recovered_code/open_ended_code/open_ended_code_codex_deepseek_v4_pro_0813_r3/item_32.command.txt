/bin/bash -c "python3 - <<'PY'
from collections import Counter, defaultdict
names={}
with open('names.dmp') as f:
    for line in f:
        parts=line.rstrip('\\n').split('\\t|\\t')
        # fields: tax_id, name_txt, unique name, name class
        if len(parts)>=4:
            tid=parts[0].strip(); name=parts[1].strip(); cls=parts[3].strip()
            if cls=='scientific name':
                names[tid]=name
best=[]
for line in open('blast_sample_100.tsv'):
    parts=line.rstrip('\\n').split('\\t')
    q=parts[0]; taxid=parts[12].split(';')[0]; name=names.get(taxid, taxid)
    best.append((q,taxid,name,parts[2],parts[3],parts[12],parts[1]))
# count top taxid per query
seen=set(); c=Counter(); per_tax=defaultdict(list)
for q,taxid,name,pident,length,staxids,sseqid in best:
    if q not in seen:
        seen.add(q); c[(taxid,name)]+=1; per_tax[(taxid,name)].append((q,pident,length,sseqid))
print('top-hit taxid counts for 100 sample:')
for (taxid,name),cnt in c.most_common(30):
    print(cnt, taxid, name)
    for item in per_tax[(taxid,name)][:5]: print('   ',item)
PY"
