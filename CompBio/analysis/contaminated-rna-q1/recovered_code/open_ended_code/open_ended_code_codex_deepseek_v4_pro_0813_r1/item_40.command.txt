/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for t in ['10376','3050299','10375','1396','29382','492670','1404']:
    url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=taxonomy&id={t}&retmode=json'
    d=json.load(urllib.request.urlopen(url))
    r=d['result'][t]
    print(t, r.get('scientificname'), r.get('rank'))
PY"
