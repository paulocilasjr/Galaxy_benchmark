def best(sam):

        for line in f:
            if line.startswith('@'): continue

            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None; mapq=int(p[4])

                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm,mapq)
    return d
