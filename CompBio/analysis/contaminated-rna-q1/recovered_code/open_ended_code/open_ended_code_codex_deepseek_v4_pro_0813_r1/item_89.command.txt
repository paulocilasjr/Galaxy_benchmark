/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None; mapq=int(p[4])
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm,mapq)
    return d
h=best('hg38.sam'); m=best('mouse.sam'); r=best('rat.sam')
for margin in [10,20,30,40]:
    c=Counter()
    for n in h.keys()|m.keys()|r.keys():
        hv=h.get(n,(None,None,None))[0] or -999
        mv=m.get(n,(None,None,None))[0] or -999
        rv=r.get(n,(None,None,None))[0] or -999
        # require one non-human score high and human much lower
        if max(mv,rv)-hv>=margin:
            if mv>=rv+margin:
                c['mouse']+=1
            elif rv>=mv+margin:
                c['rat']+=1
            else:
                c['ambiguous_nonhuman']+=1
    print('margin',margin,c)
PY"
