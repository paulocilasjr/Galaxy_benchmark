/bin/bash -lc "python3 - <<'PY'
import gzip
names=set()
with open('human.sam') as f:
    for line in f:
        if line.startswith('@'): continue
        parts=line.split('\\t')
        flag=int(parts[1])
        if flag & 4:
            names.add(parts[0])
print('unmapped names',len(names))
out=open('unmapped.fq','w')
count=0
with gzip.open('contaminated.rna.q1.fq.gz','rt') as f:
    while True:
        h=f.readline().strip(); s=f.readline().strip(); p=f.readline().strip(); q=f.readline().strip()
        if not h: break
        rid=h[1:]
        if rid in names:
            out.write(h+'\\n'+s+'\\n'+p+'\\n'+q+'\\n')
            count+=1
out.close()
print('written',count)
PY
ls -lh unmapped.fq; head -4 unmapped.fq"
