/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm)
    return d
h=best('hg38.sam'); m=best('mouse.sam'); r=best('rat.sam')
for label,ref in [('mouse',m),('rat',r)]:
    c=Counter()
    for n,mv in ref.items():
        if mv[0]==150 and mv[1]==0:
            hv=h.get(n)
            if hv is None or hv[0] is None or hv[0]<=130: c['human_worse']+=1
            elif hv[0]>130: c['human_good']+=1
    print(label,c)
# intersection perfect mouse and perfect rat?
mset={n for n,mv in m.items() if mv[0]==150 and mv[1]==0 and (h.get(n) is None or h[n][0]<=130)}
rset={n for n,mv in r.items() if mv[0]==150 and mv[1]==0 and (h.get(n) is None or h[n][0]<=130)}
print('mouse only perfect',len(mset-rset),'rat only perfect',len(rset-mset),'both',len(mset&rset))
PY"
