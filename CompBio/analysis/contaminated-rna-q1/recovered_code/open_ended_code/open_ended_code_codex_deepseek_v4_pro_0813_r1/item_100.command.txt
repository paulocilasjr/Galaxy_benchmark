/bin/bash -lc "python3 - <<'PY'
import gzip, os
os.makedirs('customdb/library', exist_ok=True)
# seqid2taxid map file
mapf=open('customdb/taxonomy/seqid2taxid.map','w')
files=[
 ('Homo_sapiens.GRCh38.cdna.all.fa.gz','human',9606,'custom_human.fa'),
 ('Mus_musculus.GRCm39.cdna.all.fa.gz','mouse',10090,'custom_mouse.fa'),
 ('Rattus_norvegicus.mRatBN7.2.cdna.all.fa.gz','rat',10116,'custom_rat.fa'),
 ('ebv.fa','ebv',3050299,'custom_ebv.fa'),  # species taxid for Lymphocryptovirus humangamma4
]
for gz,prefix,taxid,outname in files:
    out=open(os.path.join('customdb/library',outname),'w')
    n=0
    with gzip.open(gz,'rt') as f:
        seq=[]
        for line in f:
            if line.startswith('>'):
                if seq:
                    n+=1
                    hid=f'{prefix}|{n}'
                    out.write('>'+hid+'\\n'+''.join(seq)+'\\n')
                    mapf.write(f'{hid}\\t{taxid}\\n')
                    seq=[]
            else:
                seq.append(line.strip())
        if seq:
            n+=1
            hid=f'{prefix}|{n}'
            out.write('>'+hid+'\\n'+''.join(seq)+'\\n')
            mapf.write(f'{hid}\\t{taxid}\\n')
    out.close()
    print(outname,n)
mapf.close()
PY
ls -lh customdb/library; wc -l customdb/taxonomy/seqid2taxid.map; head -3 customdb/taxonomy/seqid2taxid.map"
