def mapnames(sam, q=60):
    s=set()
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.split('\\t')
            if not (int(p[1])&4) and int(p[4])>=q: s.add(p[0])
    return s
human=mapnames('human.sam')
mouse=mapnames('mouse.sam')
