/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, time
for t in ['3050299','10376','10375','9606','1396','29382','1404']:
    try:
        d=json.load(urllib.request.urlopen(f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=taxonomy&id={t}&retmode=json'))
        r=d['result'][t]
        print(t, 'rank=',r.get('rank'), 'name=',r.get('scientificname'), 'common=',r.get('commonname'))
    except Exception as e:
        print('err',t,e)
    time.sleep(0.5)
PY"
