/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def align_dict(sam):
    d=defaultdict(list)
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]); mapq=int(p[4]); nm=None
            for x in p[11:]:
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            d[name].append((flag,mapq,nm,p[9]))
    return d
h=align_dict('human.sam'); m=align_dict('mouse.sam')
# primary best by mapq then nm
mouse60=set(n for n,alns in m.items() if any(a[1]>=60 and not(a[0]&4) for a in alns))
# mouse not human mapq60
cand=set()
for n in mouse60:
    hbest=[a for a in h.get(n,[]) if not(a[0]&4)]
    if not hbest or max(a[1] for a in hbest)<60:
        cand.add(n)
print('cand',len(cand))
for n in sorted(cand, key=lambda x:int(x.split('_')[1])):
    mbest=max([a for a in m[n] if not(a[0]&4) and a[1]>=60], key=lambda a:a[1])
    hbest=max([a for a in h.get(n,[]) if not(a[0]&4)], key=lambda a:a[1], default=None)
    print(n,'mouse',mbest[1],mbest[2],'human', (hbest[1],hbest[2]) if hbest else None, 'seq',mbest[3][:75])
PY"
