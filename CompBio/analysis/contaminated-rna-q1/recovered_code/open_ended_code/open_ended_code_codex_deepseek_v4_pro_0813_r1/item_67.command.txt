from collections import defaultdict, Counter

def best(sam):
    d={}
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            p=line.rstrip().split('\\t')
            name=p[0]; flag=int(p[1]);
            if flag&4: continue
            as_=None; nm=None; mapq=int(p[4])
            for x in p[11:]:
                if x.startswith('AS:i:'): as_=int(x.split(':')[2])
                if x.startswith('NM:i:'): nm=int(x.split(':')[2])
            if name not in d or (as_ is not None and (d[name][0] is None or as_>d[name][0])):
                d[name]=(as_,nm,mapq)
    return d
h=best('hg38.sam'); m=best('mouse.sam')
cats=Counter(); examples=[]
for n,mv in m.items():
    mas,mnm,mmapq=mv
    if mas is None or mas<100 or (mnm is None or mnm>3): continue
    hv=h.get(n)
    if hv is None:
        cats['mouse_only_strong_no_hg38']+=1
    else:
        has,hnm,hmapq=hv
        if has is None or mas>has+5:
            cats['mouse_better_strong_vs_hg38']+=1
            if len(examples)<20: examples.append((n,mv,hv))
print(cats)
print('examples')
for e in examples: print(e)
# intersection with prior mouse_only_strong vs human cDNA
PY"
