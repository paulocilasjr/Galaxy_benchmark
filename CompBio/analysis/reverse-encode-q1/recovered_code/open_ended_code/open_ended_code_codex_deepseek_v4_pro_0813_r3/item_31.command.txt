/bin/bash -c "python3 - <<'PY'
import decimal, math
vals=[]
with open('track.tsv') as f:
 for line in f:
  c,s,e,v=line.split()
  vals.append(decimal.Decimal(v))
print(len(vals), sum(vals), sum(float(x) for x in vals))
# how many decimals precision counts
from collections import Counter
c=Counter()
for v in vals:
    s=str(v)
    if '.' in s: c[len(s.split('.')[1])]+=1
    else: c[0]+=1
print(c)
# max/min
print(max(vals), min(vals))
# count zeros/nonzero
print(sum(1 for v in vals if v==0), sum(1 for v in vals if v>0))
PY"
