/bin/bash -c "curl -s 'https://www.encodeproject.org/search/?type=File&file_format=bigWig&output_type=read-depth%20normalized%20signal&assembly=GRCh38&assay_title=DNase-seq&status=released&format=json&limit=all&field=accession&field=dataset&field=analysis_step_version.name&field=biological_replicates&field=technical_replicates&field=file_size&field=href&field=cloud_metadata.url&field=derived_from' -o dnase_v1_bigwigs.json && python3 - <<'PY'
import json,collections
j=json.load(open('dnase_v1_bigwigs.json'))
g=j['@graph']; print('total',j.get('total'),'len',len(g))
c=collections.Counter((x.get('analysis_step_version') or {}).get('name') for x in g)
print(c)
# filter v1 peak calling step? names
for name,n in c.most_common(): print(n,name)
v1=[x for x in g if (x.get('analysis_step_version') or {}).get('name')=='dnase-peak-calling-step-v-1-0']
print('v1 peak-calling bigwigs',len(v1),'exps',len(set(x['dataset'] for x in v1)))
print(v1[:3])
PY"
