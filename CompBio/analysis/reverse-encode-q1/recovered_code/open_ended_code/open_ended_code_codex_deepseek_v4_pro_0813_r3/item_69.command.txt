/bin/bash -c "python3 - <<'PY'
import pyBigWig, json, urllib.request, math, statistics, random
with urllib.request.urlopen('https://www.encodeproject.org/files/ENCFF995SHL/?format=json') as f:j=json.load(f)
url=j['cloud_metadata']['url']
bw=pyBigWig.open(url)
vals=[]
with open('track.tsv') as f:
 for line in f:
  c,s,e,v=line.split()
  if c=='chr10': vals.append(float(v))
st=bw.stats('chr10',0,bw.chroms('chr10'),type='mean',nBins=len(vals))
st=[0 if x is None else x for x in st]
# only nonzero both
pairs=[(v,s) for v,s in zip(vals,st) if v>0 and s>0]
print('pairs',len(pairs))
# log corr
lv=[math.log(v) for v,s in pairs]; ls=[math.log(s) for v,s in pairs]
print('log pearson',statistics.correlation(lv,ls))
print('rank all spearman?')
# top 500 bins by track
idx=sorted(range(len(vals)), key=lambda i:vals[i], reverse=True)[:500]
print('top track mean bigwig')
for i in idx[:10]: print(i, vals[i], st[i])
PY"
