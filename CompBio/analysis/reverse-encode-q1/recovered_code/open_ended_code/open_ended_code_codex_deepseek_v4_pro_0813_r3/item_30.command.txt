/bin/bash -c "python3 - <<'PY'
import json, collections
j=json.load(open('dnase_bams_compact.json'))['@graph']
# filter v1 step
cand=[x for x in j if (x.get('analysis_step_version') or {}).get('name')=='dnase-alignment-filtering-step-v-1-0']
print('v1 files', len(cand))
exp=collections.Counter(x['dataset'] for x in cand)
print('unique exps', len(exp))
print('distribution replicates')
reps=collections.Counter(len([y for y in cand if y['dataset']==d]) for d in exp)
print(reps)
# print some exps with >1
for d,n in exp.most_common(20): print(d,n)
PY"
