/bin/bash -c "python3 - <<'PY'
import pyBigWig, json, math, statistics
urls=[]
for acc in ['ENCFF995SHL','ENCFF524RHM']:
 import urllib.request
 with urllib.request.urlopen(f'https://www.encodeproject.org/files/{acc}/?format=json') as f:j=json.load(f)
 urls.append(j['cloud_metadata']['url'])
vals=[]
with open('track.tsv') as f:
 for line in f:
  c,s,e,v=line.split()
  if c=='chr10': vals.append(float(v))
def spearman(a,b):
 ra={x:i for i,x in enumerate(sorted(set(a)))}
 rb={x:i for i,x in enumerate(sorted(set(b)))}
 n=len(a)
 ra=[ra[x] for x in a]; rb=[rb[x] for x in b]
 ma=sum(ra)/n; mb=sum(rb)/n
 cov=sum((x-ma)*(y-mb) for x,y in zip(ra,rb))
 sd1=math.sqrt(sum((x-ma)**2 for x in ra)); sd2=math.sqrt(sum((y-mb)**2 for y in rb))
 return cov/(sd1*sd2)
for url in urls:
 bw=pyBigWig.open(url)
 st=bw.stats('chr10',0,bw.chroms('chr10'),type='mean',nBins=len(vals))
 st=[0 if x is None else x for x in st]
 print('pearson',statistics.correlation(vals,st),'spearman',spearman(vals,st))
PY"
