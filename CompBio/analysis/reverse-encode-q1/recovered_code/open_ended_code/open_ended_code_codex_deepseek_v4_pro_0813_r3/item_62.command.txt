/bin/bash -c "python3 - <<'PY'
import pyBigWig
url='https://encode-public.s3.amazonaws.com/2020/09/16/780343e3-c4ed-446c-9e57-ddb9b5d4910a/ENCFF995SHL.bigWig'

intervals=[('chr10',i,i+10000) for i in range(0,200000,10000)]
print(bw.stats(*zip(*[(c,s,e) for c,s,e in intervals]),type='mean'))
PY"
