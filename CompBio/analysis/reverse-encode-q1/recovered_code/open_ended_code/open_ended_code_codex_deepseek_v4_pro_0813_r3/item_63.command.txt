/bin/bash -c "python3 - <<'PY'
import pyBigWig
url='https://encode-public.s3.amazonaws.com/2020/09/16/780343e3-c4ed-446c-9e57-ddb9b5d4910a/ENCFF995SHL.bigWig'
bw=pyBigWig.open(url)
ints=bw.intervals('chr10',0,200000)
print(ints[:20])
print(len(ints))
# values first bins average manually
for i in range(0,200000,10000):
    vals=bw.intervals('chr10',i,i+10000)
    mean=sum(v[2] for v in vals)/(10000) if vals else 0
    print(i,mean)
PY"
