import json, pyBigWig, math, concurrent.futures, time, sys, os
# Load all track bins with coordinates, choose top N nonzero bins (by track value) for scoring
bins=[]
with open('track.tsv') as f:
    for line in f:
        c,s,e,v=line.split()
        bins.append((c,int(s),int(e),float(v)))
# choose top bins, ensure not too clustered? Just top 120; include some zeros? For signal similarity, top values enough.
bins_sorted=sorted([b for b in bins if b[3]>0], key=lambda x:x[3], reverse=True)
# select top 120 but maybe deduplicate to limit per chromosome and add random low
sel=bins_sorted[:120]
# maybe also random 30 from nonzero moderate
import random
random.seed(42)
others=[b for b in bins if 0<b[3]<5]
sel += random.sample(others, min(30,len(others)))
# remove duplicates
seen=set(); selected=[]
for b in sel:
    key=(b[0],b[1])
    if key not in seen:
        seen.add(key); selected.append(b)
# sort by chrom order for remote query locality
chrom_order=['chr'+str(i) for i in range(1,23)]+['chrX','chrY']
selected.sort(key=lambda b:(chrom_order.index(b[0]) if b[0] in chrom_order else 999,b[1]))
track_vals=[b[3] for b in selected]
print('selected',len(selected),'bins')

def score_one(item):
    try:
        url=item['cloud_metadata']['url']
        bw=pyBigWig.open(url)
        st=[]
        for c,s,e,v in selected:
            m=bw.stats(c,s,e,type='mean')[0]
            st.append(0.0 if m is None else m)
        # correlation
        meanv=sum(track_vals)/len(track_vals); means=sum(st)/len(st)
        sdv=math.sqrt(sum((x-meanv)**2 for x in track_vals)); sds=math.sqrt(sum((x-means)**2 for x in st))
        if sdv==0 or sds==0: pear=0.0
        else: pear=sum((a-meanv)*(b-means) for a,b in zip(track_vals,st))/(sdv*sds)
        # spearman using rank with scipy? avoid scipy
        def rank(a):
            order=sorted(range(len(a)), key=lambda i:a[i]); r=[0]*len(a)
            for rank,i in enumerate(order): r[i]=rank
            return r
        rv=rank(track_vals); rs=rank(st)
        mrv=sum(rv)/len(rv); mrs=sum(rs)/len(rs)
        sdrv=math.sqrt(sum((x-mrv)**2 for x in rv)); sdrs=math.sqrt(sum((x-mrs)**2 for x in rs))
        spear=sum((x-mrv)*(y-mrs) for x,y in zip(rv,rs))/(sdrv*sdrs) if sdrv and sdrs else 0.0
        return (item, pear, spear, None)
    except Exception as e:
        return (item, None, None, repr(e))

if __name__=='__main__':
    data=json.load(open('dnase_v1_bigwigs.json'))['@graph']
    cand=[x for x in data if (x.get('analysis_step_version') or {}).get('name')=='dnase-peak-calling-step-v-1-0']
    if len(sys.argv)>1: cand=cand[:int(sys.argv[1])]
    print('scoring',len(cand),flush=True)
    t=time.time(); results=[]
    with concurrent.futures.ThreadPoolExecutor(max_workers=20) as ex:
        futs=[ex.submit(score_one,x) for x in cand]
        for i,f in enumerate(concurrent.futures.as_completed(futs)):
            results.append(f.result())
            if (i+1)%250==0: print('done',i+1,'elapsed',round(time.time()-t,1),flush=True)
    good=[r for r in results if r[1] is not None]
    good.sort(key=lambda r:r[2] if r[2] is not None else -999, reverse=True)
    print('good',len(good),'errors',len(results)-len(good))
    for r in good[:40]:
        item,pear,spear,e=r
        print(item['dataset'], item['accession'], item.get('biological_replicates'), item.get('technical_replicates'), 'pear',round(pear,3),'spear',round(spear,3))
    json.dump([{'dataset':r[0]['dataset'],'accession':r[0]['accession'],'pear':r[1],'spear':r[2],'error':r[3]} for r in results], open('scores_fast_partial.json','w'))
