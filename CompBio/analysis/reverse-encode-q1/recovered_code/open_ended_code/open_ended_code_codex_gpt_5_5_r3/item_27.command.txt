/bin/bash -lc "python3 - <<'PY'
import gzip
rows=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
    for line in f:
        chrom,start,end,val=line.rstrip().split('\\t')
        v=float(val)
        rows.append((v,chrom,int(start),int(end)))
rows.sort(reverse=True)
for r in rows[:50]: print(r)
PY"
