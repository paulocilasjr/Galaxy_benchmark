/bin/bash -lc "python3 - <<'PY'
import json, collections, re
d=json.load(open('data/encode_dnase_grch38_experiments.json'))
files=[]
for e in d['@graph']:
    for f in e.get('files',[]):
        files.append((e,f))
print('files',len(files))
print('file keys', sorted(files[0][1].keys())[:50], sorted(files[0][1].keys())[50:])
# count file formats/output types/pipeline titles versions
cnt=collections.Counter()
vers=collections.Counter()
out=collections.Counter()
for e,f in files:
    cnt[(f.get('file_format'), f.get('output_type'), f.get('assembly'))]+=1
    out[f.get('output_type')]+=1
    for an in f.get('analyses',[]) or []:
        vers[str(an)]+=1
print('outputs top')
for k,v in out.most_common(30): print(v,k)
print('example bam')
for e,f in files:
    if f.get('file_format')=='bam':
        print(e['accession'], f.get('accession'), f.get('output_type'), f.get('assembly'), f.get('href'), f.get('analyses')[:2] if f.get('analyses') else None, f.get('biological_replicates'), f.get('technical_replicates'), f.get('read_length'), f.get('mapped_read_length'))
        print(f)
        break
PY"
