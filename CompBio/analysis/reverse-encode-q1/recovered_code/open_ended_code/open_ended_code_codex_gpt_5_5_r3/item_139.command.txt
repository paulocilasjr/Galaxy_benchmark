/bin/bash -lc "python3 - <<'PY'
import gzip,math,pyBigWig
for acc,path in [('ENCSR597URK','data/ENCFF186BOY.bigWig'),('ENCSR251INE','data/ENCFF667QSN.bigWig'),('ENCSR507XBY','data/ENCFF571GCP.bigWig')]:
 bw=pyBigWig.open(path); x=[]; y=[]; top=[]
 with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
  for line in f:
   c,s,e,v=line.rstrip().split('\\t'); v=float(v); s=int(s); e=int(e)
   st=bw.stats(c,s,e,type='mean')[0]
   st=0.0 if st is None or math.isnan(st) else float(st)
   x.append(math.log1p(st)); y.append(math.log1p(v)); top.append((v,st))
 bw.close()
 def pear(a,b):
  ma=sum(a)/len(a); mb=sum(b)/len(b); va=sum((z-ma)**2 for z in a); vb=sum((z-mb)**2 for z in b)
  return sum((a[i]-ma)*(b[i]-mb) for i in range(len(a)))/math.sqrt(va*vb)
 print(acc,'all',pear(x,y))
 for K in [40,100,1000,10000]:
  arr=sorted(top, reverse=True)[:K]
  print(acc,'top',K,pear([math.log1p(a[1]) for a in arr],[math.log1p(a[0]) for a in arr]),'mean',sum(a[1] for a in arr)/K,'max',max(a[1] for a in arr))
PY"
