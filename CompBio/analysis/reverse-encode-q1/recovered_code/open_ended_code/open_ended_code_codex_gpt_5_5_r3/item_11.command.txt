/bin/bash -lc "python3 - <<'PY'
import csv, collections
rows=list(csv.DictReader(open('data/encode_dnase_grch38_metadata.tsv'), delimiter='\\t'))
print('rows',len(rows))
print('analysis titles top')
for k,v in collections.Counter(r['File analysis title'] for r in rows).most_common(20): print(v, repr(k))
print('bam output types top')
bams=[r for r in rows if r['File format']=='bam']
for k,v in collections.Counter((r['Output type'], r['File assembly'], r['File analysis title'], r['File Status']) for r in bams).most_common(30): print(v,k)
print('alpha2 bams')
alpha=[r for r in bams if 'v3.0.0-alpha.2' in r['File analysis title']]
print(len(alpha), len(set(r['Experiment accession'] for r in alpha)))
for k,v in collections.Counter((r['Output type'], r['File assembly'], r['File Status']) for r in alpha).most_common(): print(v,k)
PY"
