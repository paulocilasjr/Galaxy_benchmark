/bin/bash -lc "cat > scripts/score_bigwigs.py <<'PY'
import csv, gzip, math, os, sys, time, statistics
from multiprocessing import Pool
import pyBigWig

META='data/encode_dnase_grch38_metadata.tsv'
TARGET='reverse.encode.q1.tsv.gz'
N=int(sys.argv[1]) if len(sys.argv)>1 else 50
WORKERS=int(sys.argv[2]) if len(sys.argv)>2 else 12
LIMIT=int(sys.argv[3]) if len(sys.argv)>3 else 0

# top bins plus some bins near quantile ranges to reduce generic open-chromatin matches
rows=[]
with gzip.open(TARGET,'rt') as f:
    for line in f:
        c,s,e,v=line.rstrip().split('\\t')
        rows.append((float(v),c,int(s),int(e)))
rows_sorted=sorted(rows, reverse=True)
bins=rows_sorted[:N]
y=[math.log1p(x[0]) for x in bins]

files=[]
with open(META) as fh:
    for r in csv.DictReader(fh, delimiter='\\t'):
        if (r['File format']=='bigWig' and r['Output type']=='read-depth normalized signal' and
            r['File assembly']=='GRCh38' and r['File Status']=='released' and
            r['File analysis title']=='ENCODE4 v3.0.0-alpha.2 GRCh38'):
            uri=r['s3_uri']
            if not uri.startswith('s3://encode-public/'): continue
            url='https://encode-public.s3.amazonaws.com/'+uri[len('s3://encode-public/'):]
            files.append((r['Experiment accession'], r['File accession'], url, r['Biological replicate(s)'], r['Technical replicate(s)'], r['Biosample term name']))
if LIMIT: files=files[:LIMIT]
print('bins',len(bins),'files',len(files),'workers',WORKERS, file=sys.stderr)

def pearson(a,b):
    n=len(a)
    ma=sum(a)/n; mb=sum(b)/n
    va=sum((x-ma)**2 for x in a); vb=sum((x-mb)**2 for x in b)
    if va==0 or vb==0: return 0.0
    return sum((a[i]-ma)*(b[i]-mb) for i in range(n))/math.sqrt(va*vb)

def score(item):
    exp, acc, url, bio, tech, bio_name=item
    t0=time.time()
    try:
        bw=pyBigWig.open(url)
        vals=[]
        for tv,c,s,e in bins:
            st=bw.stats(c,s,e,type='mean')[0]
            vals.append(0.0 if st is None or math.isnan(st) else float(st))
        bw.close()
        x=[math.log1p(v) for v in vals]
        # Score: correlation and mean high-bin signal. Both are useful; sort primarily by corr.
        corr=pearson(x,y)
        dot=sum(x[i]*y[i] for i in range(len(y)))
        mean=sum(vals)/len(vals)
        mx=max(vals) if vals else 0
        return (corr,dot,mean,mx,exp,acc,bio,tech,bio_name,round(time.time()-t0,2),'')
    except Exception as e:
        return (-9,0,0,0,exp,acc,bio,tech,bio_name,round(time.time()-t0,2),repr(e)[:120])

with Pool(WORKERS) as p:
    for res in p.imap_unordered(score, files, chunksize=1):
        print('\\t'.join(map(str,res)), flush=True)
PY
python3 scripts/score_bigwigs.py 40 16 > data/bw_scores_top40.tsv"
