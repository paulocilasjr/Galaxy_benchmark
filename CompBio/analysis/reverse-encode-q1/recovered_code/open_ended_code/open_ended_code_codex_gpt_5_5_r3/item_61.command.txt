/bin/bash -lc "python3 - <<'PY'
import csv,gzip,math,pyBigWig
# reuse scoring for selected bws with 200 bins, including high and quantile bins
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for line in f:
  c,s,e,v=line.rstrip().split('\\t')
  vals.append((float(v),c,int(s),int(e)))
rows=sorted(vals, reverse=True)
bins=rows[:200]
y=[math.log1p(v) for v,_,_,_ in bins]
def pear(a,b):
 ma=sum(a)/len(a); mb=sum(b)/len(b); va=sum((x-ma)**2 for x in a); vb=sum((x-mb)**2 for x in b)
 return sum((a[i]-ma)*(b[i]-mb) for i in range(len(a)))/math.sqrt(va*vb)
# map bw files for selected exps
sel=['ENCSR357AWH','ENCSR941MHZ','ENCSR483PGP','ENCSR566XSD','ENCSR516JCM','ENCSR000EQG','ENCSR000EQC','ENCSR367FKP']
items=[]
for r in csv.DictReader(open('data/encode_dnase_grch38_metadata.tsv'), delimiter='\\t'):
 if r['Experiment accession'] in sel and r['File format']=='bigWig' and r['Output type']=='read-depth normalized signal' and r['File analysis title']=='ENCODE4 v3.0.0-alpha.2 GRCh38' and r['File Status']=='released':
  url='https://encode-public.s3.amazonaws.com/'+r['s3_uri'][len('s3://encode-public/'):]
  items.append((r['Experiment accession'],r['File accession'],r['Biosample term name'],url))
for exp,acc,name,url in items:
 bw=pyBigWig.open(url); xs=[]
 for _,c,s,e in bins:
  st=bw.stats(c,s,e,type='mean')[0]
  xs.append(0.0 if st is None or math.isnan(st) else st)
 bw.close()
 lx=[math.log1p(x) for x in xs]
 print(pear(lx,y), sum(xs)/len(xs), max(xs), exp, acc, name)
PY"
