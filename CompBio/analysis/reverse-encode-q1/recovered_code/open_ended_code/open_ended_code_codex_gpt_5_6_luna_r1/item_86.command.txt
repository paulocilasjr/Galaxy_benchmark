/bin/bash -lc "python3 - <<'PY'
import requests, pyBigWig
u='https://www.encodeproject.org/files/ENCFF659BVQ/@@download/ENCFF659BVQ.bigWig'
r=requests.get(u,allow_redirects=False)
print(r.status_code, r.headers.get('location','')[:120],len(r.content))
f=r.headers['location']
try:
 b=pyBigWig.open(f)
 print('opened signed',b.chroms()['chr1'])
 print(b.stats('chr1',0,10000,type='sum'))
 b.close()
except Exception as e: print('err',type(e),e)
PY"
