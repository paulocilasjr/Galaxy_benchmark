def corr(a,b):
 n=len(a); ma=sum(a)/n; mb=sum(b)/n
 cov=va=vb=0.0
 for x,y in zip(a,b):
  cov+=(x-ma)*(y-mb); va+=(x-ma)**2; vb+=(y-mb)**2
