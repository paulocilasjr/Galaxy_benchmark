/bin/bash -lc "python3 - <<'PY'
import json, collections
x=json.load(open('experiments_analyses.json'))
g=x['@graph']
vers=collections.Counter()
cands=[]
for e in g:
    for a in e.get('analyses',[]):
        v=a.get('pipeline_version')
        title=a.get('title')
        if v or title: vers[(v,title)] +=1
        if v=='3.0.0-alpha.2' or (title and '3.0.0-alpha.2' in title):
            cands.append((e['accession'],a.get('accession'),v,title))
print('versions')
for k,c in vers.most_common(20): print(k,c)
print('cands',len(cands))
print(cands[:50])
PY"
