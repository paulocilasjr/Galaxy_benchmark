/bin/bash -lc "python3 - <<'PY'
import pyBigWig,time
lines=open('counts.tsv').read().splitlines()
vals=[]
for line in lines:
 c,s,e,v=line.split(); vals.append((float(v),c,int(s),int(e)))
vals.sort(reverse=True); top=vals[:50]
url='https://encode-public.s3.amazonaws.com/2020/11/06/9569425d-04bc-4802-86a2-8068b5f427ff/ENCFF450AHB.bigWig'
bw=pyBigWig.open(url)
def sum_intervals(c,s,e):
 total=0.0
 for (a,b,val) in bw.intervals(c,s,e):
  lo=max(a,s); hi=min(b,e)
  if hi>lo: total+=val*(hi-lo)
 return total
t=time.time(); res=[]
for v,c,s,e in top:
 res.append(sum_intervals(c,s,e))
print('time',time.time()-t)
print(list(zip([v for v,_,_,_ in top],res))[:10])
PY"
