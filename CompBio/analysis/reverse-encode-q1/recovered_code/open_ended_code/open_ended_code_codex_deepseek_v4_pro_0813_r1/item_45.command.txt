/bin/bash -lc "python3 - <<'PY'
import pyBigWig, time
url='https://www.encodeproject.org/files/ENCFF450AHB/@@download/ENCFF450AHB.bigWig'
t=time.time()
try:
 bw=pyBigWig.open(url)
 print('opened',time.time()-t)
 print(bw.chroms())
 for c in ['chr19','chr16','chr1']:
  print(c, bw.stats(c,0,10000,type='mean'))
except Exception as e:
 print('ERR',repr(e))
PY"
