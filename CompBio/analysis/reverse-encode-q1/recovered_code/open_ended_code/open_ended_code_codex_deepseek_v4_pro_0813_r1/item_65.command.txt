import json, pyBigWig, time, math
sel=json.load(open('selected_bins_test.json')); bins=sel['bins']; obs=sel['obs']
c=json.load(open('candidate_bw.json'))[0]
t=time.time()
for mode in ['non-exact','exact_nbins1']:
 vec=[]
 bw=pyBigWig.open(c['files'][0]['url'])
 st=time.time()
 for ch,s,e in bins:
  if mode=='non-exact':
   x=bw.stats(ch,s,e,type='mean')[0]

   x=bw.stats(ch,s,e,nBins=1,type='sum',exact=True)[0]
  vec.append(0.0 if x is None else x)
 print(mode,'bins time',time.time()-st, vec[:3])
 bw.close()
print('total',time.time()-t)
