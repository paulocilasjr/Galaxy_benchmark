import json, collections, re
files=json.load(open('files_bam.json'))['@graph']

    if acc in cand:
        groups[acc].append(f)
print('groups matched',len(groups))
cnt=collections.Counter(len(v) for v in groups.values()); print(cnt)
for acc,fs in list(groups.items())[:5]:
 print(acc, [(f['accession'],f['biological_replicates'],f['file_size']) for f in fs])
