/bin/bash -lc "python3 -u - <<'PY'
import json, pyBigWig, time, math, random
lines=open('counts.tsv').read().splitlines(); vals=[]
for line in lines:
 c,s,e,v=line.split(); vals.append((float(v),c,int(s),int(e)))
vals.sort(reverse=True)
random.seed(123); rand=random.sample(vals,200)
sel=vals[:300]+rand
seen=set(); bins=[]; obs=[]
for v,c,s,e in sel:
 if (c,s,e) in seen: continue
 seen.add((c,s,e)); bins.append((c,s,e)); obs.append(v)
print('bins',len(bins))
json.dump({'bins':bins,'obs':obs},open('selected_bins_exact.json','w'))
c=json.load(open('candidate_bw.json'))[0]
def corr(a,b):
 n=len(a); ma=sum(a)/n; mb=sum(b)/n
 cov=sum((x-ma)*(y-mb) for x,y in zip(a,b)); va=sum((x-ma)**2 for x in a); vb=sum((y-mb)**2 for y in b)
 return cov/math.sqrt(va*vb) if va and vb else 0
vec=[0.0]*len(bins)
t=time.time()
for f in c['files']:
 bw=pyBigWig.open(f['url'])
 for i,(ch,s,e) in enumerate(bins):
  x=bw.stats(ch,s,e,nBins=1,type='sum',exact=True)
  if x and x[0] is not None: vec[i]+=x[0]
 bw.close()
print('time',time.time()-t,'corr',corr(obs,vec))
PY"
