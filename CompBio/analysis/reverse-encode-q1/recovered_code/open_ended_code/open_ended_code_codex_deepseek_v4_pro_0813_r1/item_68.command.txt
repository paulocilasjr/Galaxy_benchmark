/bin/bash -lc "python3 -u - <<'PY'
import pyBigWig,time
url='https://encode-public.s3.amazonaws.com/2020/11/06/9569425d-04bc-4802-86a2-8068b5f427ff/ENCFF450AHB.bigWig'
bw=pyBigWig.open(url)
chroms=['chr1','chr2','chr3','chr4','chr5','chr6','chr7','chr8','chr9','chr10','chr11','chr12','chr13','chr14','chr15','chr16','chr17','chr18','chr19','chr20','chr21','chr22','chrX','chrY']
t=time.time(); total=0
for ch in chroms:
 L=bw.chroms(ch); n=(L+9999)//10000
 st=time.time(); arr=bw.stats(ch,0,L,nBins=n,type='mean'); total+=len(arr)
 print(ch,n,time.time()-st)
print('total',time.time()-t,'vals',total)
PY"
