/bin/bash -lc "python3 - <<'PY'
import pysam, time
url='https://www.encodeproject.org/files/ENCFF062ZPB/@@download/ENCFF062ZPB.bam'
idx=url+'.bai'
t=time.time()
try:
 af=pysam.AlignmentFile(url,'rb',index_filename=idx)
 stats=af.get_index_statistics()
 print('time',time.time()-t,'n',len(stats))
 for s in stats[:5]: print(s)
except Exception as e:
 print('ERR',repr(e))
PY"
