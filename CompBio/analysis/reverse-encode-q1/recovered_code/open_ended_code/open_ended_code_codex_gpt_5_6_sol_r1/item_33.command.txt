/bin/bash -lc "awk -F '\\t' 'BEGIN{OFS=\"\\t\"} NR==1{for(i=1;i<=NF;i++)h["'$i]=i; print "experiment","file","bio_rep","size","url"; next} $h["File analysis title"]=="ENCODE4 v3.0.0-alpha.2 GRCh38" && $h["File format"]=="bigWig" {u=$h["s3_uri"]; sub("s3://encode-public/","https://encode-public.s3.amazonaws.com/",u); print $h["Experiment accession"],$h["File accession"],$h["Biological replicate(s)"],$h["Size"],u}'"' dnase_metadata.tsv > bigwigs.tsv
wc -l bigwigs.tsv
python3 - <<'PY'
import pyBigWig,time,csv
u=next(csv.DictReader(open('bigwigs.tsv'),delimiter='\\t'))['url']
t=time.time(); b=pyBigWig.open(u); L=b.chroms()['chr1']; x=b.stats('chr1',0,L,nBins=249,type='mean'); print(len(x),sum(v is not None for v in x),time.time()-t, x[:5])
PY"
