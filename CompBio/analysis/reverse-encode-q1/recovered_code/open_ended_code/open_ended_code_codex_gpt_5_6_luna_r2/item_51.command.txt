/bin/bash -lc "python3 -u - <<'PY'
import pyBigWig,time
u='https://encode-public.s3.amazonaws.com/2020/11/12/31cec132-6555-4b4e-a906-18ba1b587b94/ENCFF389PZY.bigWig'
print('open',flush=True); b=pyBigWig.open(u); print('opened',flush=True)
L=b.chroms()['chr1']; n=L//10000; t=time.time(); x=b.stats('chr1',0,n*10000,nBins=n,exact=True); print('got',len(x),time.time()-t,sum(v or 0 for v in x),flush=True); b.close()
PY"
