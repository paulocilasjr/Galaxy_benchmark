import gzip,statistics
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  v=float(l.split()[3])
  if 0<v<=1: vals.append(v)
vals=sorted(set(vals));
# OLS with index 1..76
n=len(vals); xb=(n+1)/2; yb=sum(vals)/n
s=sum((i-xb)*(v-yb) for i,v in enumerate(vals,1))/sum((i-xb)**2 for i in range(1,n+1))
a=yb-s*xb
print(n,a,s, 'N=',1e6/s, 'resid range',min(v-(a+s*i) for i,v in enumerate(vals,1)),max(v-(a+s*i) for i,v in enumerate(vals,1)))
for i,v in enumerate(vals,1): print(i,v, v/(i), v-(a+s*i))
PY"
