/bin/bash -lc "python - <<'PY'
import csv,collections
R=list(csv.DictReader(open('encode_dnase_metadata.tsv'),delimiter='\\t'))
sel=[r for r in R if r['File assembly']=='GRCh38' and r['File analysis title']=='ENCODE4 v3.0.0-alpha.2 GRCh38' and r['File format']=='bam']
print('bam rows',len(sel),'experiments',len(set(r['Experiment accession'] for r in sel)))
print(collections.Counter(r['Output type'] for r in sel))
print(collections.Counter((r['Output type'],r['Biological replicate(s)'],r['Technical replicate(s)']) for r in sel).most_common(20))
# signal rows for same exp
ex=set(r['Experiment accession'] for r in sel if r['Output type']=='alignments')
sig=[r for r in R if r['Experiment accession'] in ex and r['File assembly']=='GRCh38' and r['File analysis title']=='ENCODE4 v3.0.0-alpha.2 GRCh38' and r['Output type']=='read-depth normalized signal']
print('alignment ex',len(ex),'signal rows',len(sig),collections.Counter(r['File format'] for r in sig))
PY"
