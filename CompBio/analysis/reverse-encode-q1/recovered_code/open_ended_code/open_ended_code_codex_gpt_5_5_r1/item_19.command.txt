/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse,collections
params={
 'type':'File','assay_title':'DNase-seq','assembly':'GRCh38','file_format':'bam','analyses.pipeline_version':'3.0.0-alpha.2','format':'json','limit':'all','frame':'object',
 'field':['accession','dataset','output_type','biological_replicates','technical_replicates','file_size','cloud_metadata','s3_uri','href','analyses','derived_from','mapped_run_type','mapped_read_length','status','quality_metrics']
}
# urllib doesn't handle list in dict straightforward
qs=[]
for k,v in params.items():
 if isinstance(v,list):
  for x in v: qs.append((k,x))
 else: qs.append((k,v))
url='https://www.encodeproject.org/search/?'+urllib.parse.urlencode(qs)
print(url[:300], file=open('encode_query_url.txt','w'))
req=urllib.request.Request(url,headers={'Accept':'application/json'})
data=json.load(urllib.request.urlopen(req,timeout=120))
print('total',data.get('total'))
open('encode_dnase_alpha2_bams.json','w').write(json.dumps(data))
c=collections.Counter(f.get('output_type') for f in data['@graph'])
print(c)
for f in data['@graph'][:3]: print(json.dumps(f,indent=2)[:1200])
PY"
