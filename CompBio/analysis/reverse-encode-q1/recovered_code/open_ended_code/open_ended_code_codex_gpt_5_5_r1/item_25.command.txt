/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, urllib.parse
# details for candidate and index files
acc='ENCFF254VNR'
for q in [f'https://www.encodeproject.org/files/{acc}/?format=json&frame=object', f'https://www.encodeproject.org/search/?type=File&file_format=bai&derived_from.accession={acc}&format=json&limit=10&frame=object']:
 data=json.load(urllib.request.urlopen(urllib.request.Request(q,headers={'Accept':'application/json'}),timeout=60))
 print('\\nURL',q)
 if '@graph' in data:
  print('total',data.get('total'))
  for f in data['@graph']:
   print(f.get('accession'),f.get('file_format'),f.get('output_type'),f.get('cloud_metadata',{}).get('url'),f.get('href'),f.get('derived_from'))
 else:
  print(data.get('accession'), data.get('dataset'), data.get('cloud_metadata',{}).get('url'), data.get('s3_uri'), data.get('href'))
  print('bio',data.get('biological_replicates'),'tech',data.get('technical_replicates'),'mapped',data.get('mapped_run_type'))
  for qm in data.get('quality_metrics',[]):
   if 'SamtoolsFlagstatsQualityMetric' in qm.get('@type',[]): print('flag total',qm.get('total'),qm.get('mapped'))
   if 'SamtoolsStatsQualityMetric' in qm.get('@type',[]): print('stats MQ0',qm.get('reads MQ0'), 'avglen', qm.get('average length'))
PY"
