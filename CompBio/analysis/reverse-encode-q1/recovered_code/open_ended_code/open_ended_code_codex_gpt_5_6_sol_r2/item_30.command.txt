/bin/bash -lc "python3 - <<'PY'
import csv,collections
# generate candidate file
flags={}
with open('flagstats.tsv') as f:
 next(f);r=csv.DictReader(f,delimiter='\\t')
 for x in r:
  if x['# reads mapped passing QC']: flags[x['Files'].strip('/').split('/')[-1]]=int(x['# reads mapped passing QC'])
rows=[]
with open('encode_files.tsv') as f:
 next(f);r=csv.DictReader(f,delimiter='\\t')
 for x in r:
  if x['Output type']=='alignments' and x['Date created']>='2020' and x['Accession'] in flags:
   x['mapped']=flags[x['Accession']];rows.append(x)
ex=collections.defaultdict(list)
for x in rows:ex[x['Dataset'].strip('/').split('/')[-1]].append(x)
bwfrom=collections.defaultdict(list)
with open('encode_bigwigs.tsv') as f:
 next(f);r=csv.DictReader(f,delimiter='\\t')
 for x in r:
  for z in x['Derived from'].split(','):
   bwfrom[z.strip('/').split('/')[-1]].append((x['Accession'],x['Date created'],x['Dataset']))
out=[]
for e,xs in ex.items():
 sm=sum(x['mapped'] for x in xs)
 if 76200000<=sm<=90000000:
  for x in xs:
   matches=[z for z in bwfrom[x['Accession']] if z[2].strip('/').split('/')[-1]==e and z[1][:10]==x['Date created'][:10]]
   out.append((e,sm,x['Accession'],x['mapped'],matches))
for z in sorted(out,key=lambda q:q[1]):print(*z,sep='\\t')
PY"
