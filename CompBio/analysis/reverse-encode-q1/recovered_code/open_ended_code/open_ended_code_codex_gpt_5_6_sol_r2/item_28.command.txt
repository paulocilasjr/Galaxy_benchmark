/bin/bash -lc "python3 -u - <<'PY'
import urllib.request,pyBigWig,gzip,numpy as np
u='https://www.encodeproject.org/files/ENCFF907ITJ/@@download/ENCFF907ITJ.bigWig';r=urllib.request.urlopen(u);u=r.geturl();r.close();b=pyBigWig.open(u)
t={}
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
 for l in f:
  c,a,z,v=l.split();
  if c in ('chr1','chr7','chr19','chrX'): t.setdefault(c,[]).append(float(v))
for c,x in t.items():
 L=b.chroms(c); n=(L+9999)//10000
 y=b.stats(c,0,L,nBins=n,type='mean',exact=False)
 y=np.nan_to_num(np.array(y,dtype=float));x=np.array(x)
 print(c,len(x),len(y),np.corrcoef(x,y)[0,1],np.corrcoef(np.log1p(x),np.log1p(y))[0,1],x.sum(),y.sum(),flush=True)
PY"
