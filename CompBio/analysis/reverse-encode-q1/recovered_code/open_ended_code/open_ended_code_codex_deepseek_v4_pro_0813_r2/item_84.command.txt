/bin/bash -lc "python3 - <<'PY'
import pyBigWig, csv, math, numpy as np, time
bw=pyBigWig.open('ENCFF389PZY.bigWig')
print('chroms',bw.chroms())
track=[]
with open('track.tsv') as f:
 for line in f:
  p=line.split(); chrom=p[0]; start=int(p[1]); end=int(p[2]); v=float(p[3]); track.append((chrom,start,end,v))
# compute for all track bins (308k) with bw.stats, may take time
bw_vals=[]
for chrom,start,end,v in track:
 try:
  s=bw.stats(chrom,start,end,nBins=1,type='mean')[0]
 except Exception as e:
  s=None
 bw_vals.append(s if s is not None else 0.0)
t=np.array([v for *_,v in track])
b=np.array(bw_vals)
mask=np.isfinite(b)
print('nan',np.isnan(b).sum(), 'nonzero',np.count_nonzero(b), 'track nonzero',np.count_nonzero(t))
for arr in [b,t]:
 print('sum',arr.sum(),'max',arr.max(),'min',arr.min())
# correlation overall and nonzero
for m in [mask]:
 print('corr all',np.corrcoef(t[m],b[m])[0,1])
 nz=b[m]>0
 print('corr nz',np.corrcoef(t[m][nz],b[m][nz])[0,1])
 # compare normalized sums maybe
 print('nz counts',nz.sum())
PY"
