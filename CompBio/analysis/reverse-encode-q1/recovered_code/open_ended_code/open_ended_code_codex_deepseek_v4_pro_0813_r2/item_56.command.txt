 e=exp[ds]; e['biosample']=bio
 e['files'].append((acc,run,length,size,mapped,seq))
 if mapped is not None: e['total_mapped']+=mapped
 if seq is not None: e['total_seq']+=seq
print('experiments',len(exp))
targets=[76275859,152551718]

 rows.sort()
 print('\\nTARGET',t)
