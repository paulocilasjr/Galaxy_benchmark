/bin/bash -lc "python3 - <<'PY'
import math
vals=[]
with open('track.tsv') as f:
 for line in f:
  v=float(line.split()[3])
  if v>0: vals.append(v)
u=min(vals)
print('u',u)
# try residues for k nearest
bad=0; maxerr=0; S=0
for v in vals[:10000]:
 k=v/u
 r=k-round(k)
 if abs(r)>0.15: bad+=1
 maxerr=max(maxerr, min(abs(k-round(k)),1-abs(k-round(k))))
print('bad first10k',bad,'maxerr',maxerr)
# Actually find best u from common differences? Let's cluster values multiples of 0.013 maybe
# Estimate S by checking integer quantization for candidate S
# For each v, k=round(v*S/1e6); error = v - k*1e6/S rounded? if original rounded, v itself is rounded to 3 decimals
for S in [76e6,77e6,78e6,80e6]:
 err=0
 for v in vals:
  k=round(v*S/1e6)
  pred=round(k*1e6/S,3)
  err+=abs(v-pred)
 print('S',S,'mean err',err/len(vals))
PY"
