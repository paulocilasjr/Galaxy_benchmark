/bin/bash -lc "python3 - <<'PY'
from fractions import Fraction
import math
vals=[]
with open('track.tsv') as f:
 for line in f:
  v=float(line.split()[3])
  if v>0: vals.append(v)
uvals=sorted(set(round(v,3) for v in vals))
# Try k0 (min count) = 1..5 and assume unique values are consecutive k = k0 + i-1? maybe no missing? We'll check intervals
for k0 in range(1,6):
 lo=Fraction(0,1); hi=Fraction(10**18,1)
 ok=True
 for i,v in enumerate(uvals[:50], start=0):
  k=k0+i
  l=Fraction(round((v+0.0005)*1000),1000)
  u=Fraction(round((v-0.0005)*1000),1000)
  if u<=0: ok=False; break
  L=Fraction(k*10**6,1)/l
  U=Fraction(k*10**6,1)/u
  lo=max(lo,L); hi=min(hi,U)
  if lo>hi:
   ok=False; break
 print('k0',k0,'ok',ok,'lo',float(lo),'hi',float(hi),'ints',math.ceil(lo),math.floor(hi))
PY"
