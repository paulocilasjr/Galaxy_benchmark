/bin/bash -lc "python3 - <<'PY'
from fractions import Fraction
vals=[]
with open('track.tsv') as f:
 for line in f:
  v=float(line.split()[3])
  if v>0: vals.append(v)
uvals=sorted(set(round(v,3) for v in vals))
print('n unique',len(uvals))
lo=0; hi=10**18
for i,v in enumerate(uvals[:100], start=1):
 # assume k=2*i
 l=Fraction(int(round((v+0.0005)*1000)),1000)
 u=Fraction(int(round((v-0.0005)*1000)),1000)
 # l>u, S in [2i*1e6/l, 2i*1e6/u]
 L=Fraction(2*i*10**6,1)/l
 U=Fraction(2*i*10**6,1)/u
 if L>lo: lo=L
 if U<hi: hi=U
print('intersection first 100 even assumption:', float(lo), float(hi), lo,hi)
# Candidate integer S range
import math
print('int range', math.ceil(lo), math.floor(hi))
PY"
