/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
url='https://www.encodeproject.org/search/?type=File&searchTerm=v3.0.0-alpha.2&limit=all&format=json'
with urllib.request.urlopen(urllib.request.Request(url,headers={'accept':'application/json'}),timeout=120) as r:data=json.load(r)
print('total',data.get('total'), 'len', len(data.get('@graph',[])))
from collections import Counter
print(Counter((f.get('assay_title'), f.get('file_format'), f.get('output_type'), f.get('assembly')) for f in data['@graph']).most_common(20))
for f in data['@graph'][:20]: print(f['accession'], f.get('dataset'), f.get('assay_title'), f.get('assembly'), f.get('file_format'), f.get('output_type'), f.get('analysis_step_version',{}).get('aliases'))
PY"
