/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://www.encodeproject.org/search/?type=File&assay_title=DNase-seq&assembly=GRCh38&file_format=bam&output_type=alignments&limit=all&format=json'
req=urllib.request.Request(url, headers={'accept':'application/json'})
with urllib.request.urlopen(req, timeout=120) as r:
    data=json.load(r)
with open('encode_dnase_grch38_bam_alignments.json','w') as f: json.dump(data,f)
print(len(data['@graph']))
# summarize analysis step version aliases and counts
from collections import Counter,defaultdict
c=Counter()
examples={}
for f in data['@graph']:
    aliases=tuple(f.get('analysis_step_version',{}).get('aliases',[]))
    c[aliases]+=1
    examples.setdefault(aliases,f['accession'])
for aliases,n in c.most_common():
    print(n, examples[aliases], aliases[:3])
PY"
