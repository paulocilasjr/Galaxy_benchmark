import subprocess,re,time,sys,gzip
bins=[]
for line in open('diagnostic_bins.bed'):
    c,s,e,v=line.split(); bins.append((c,int(s),int(e),float(v)))

def ref_end(pos0,cig):
    p=pos0
    for n,op in cig_re.findall(cig):
        if op in 'MDN=X': p+=int(n)
    return p

url=f'https://www.encodeproject.org/files/{acc}/@@download/{acc}.bam'
