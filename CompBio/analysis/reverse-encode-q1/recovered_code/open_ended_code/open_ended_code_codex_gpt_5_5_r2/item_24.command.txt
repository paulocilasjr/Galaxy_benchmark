/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
exp='ENCSR000EPK'
url=f'https://www.encodeproject.org/experiments/{exp}/?format=json'
with urllib.request.urlopen(urllib.request.Request(url,headers={'accept':'application/json'}),timeout=60) as r:data=json.load(r)
print(data['accession'], data.get('biosample_summary'), data.get('assay_title'), data.get('status'))
for f in data['files']:
    if f.get('assembly')=='GRCh38' and f.get('status')=='released':
        print(f['accession'], f.get('file_format'), f.get('output_type'), f.get('file_size'), f.get('biological_replicates'), f.get('analysis_step_version',{}).get('aliases'), f.get('href'))
PY"
