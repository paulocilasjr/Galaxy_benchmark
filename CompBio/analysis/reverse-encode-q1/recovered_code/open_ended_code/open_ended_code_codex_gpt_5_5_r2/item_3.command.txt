import gzip, math, statistics
vals=[]
with gzip.open('reverse.encode.q1.tsv.gz','rt') as f:
    for line in f:
        vals.append(float(line.split()[3]))
non=[v for v in vals if v>0]
print('bins',len(vals),'sum',sum(vals),'nonzero',len(non),'minnon',min(non),'max',max(non),'mean_non',statistics.mean(non))
print('smallest unique', sorted(set(non))[:50])
# Try estimate integer N from multiples: v ~= round(k*1e6/N,3)
# min nonzero likely 1 count. N~1e6/min
print('N from min',1e6/min(non))
# Count frequencies of small unique values
from collections import Counter
c=Counter(non)
for v in sorted(c)[:30]: print(v,c[v])
PY"
