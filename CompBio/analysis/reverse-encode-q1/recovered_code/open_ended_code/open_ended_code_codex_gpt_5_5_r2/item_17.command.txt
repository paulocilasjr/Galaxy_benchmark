import gzip
ms=[]
for line in gzip.open('reverse.encode.q1.tsv.gz','rt'):
    m=round(float(line.split()[3])*1000)
    if m>0: ms.append(m)

def T(N): return sum((m*N + 500_000_000)//1_000_000_000 for m in ms) # half up approx
# Find fixed points in 60-100M coarse
roots=[]
prevN=60_000_000; prev=T(prevN)-prevN
print('start',prev)
for N in range(60_000_000,100_000_001,10000):
    d=T(N)-N
    if d==0 or (d>0)"'!=(prev>0): roots.append((prevN,N,prev,d))
    prevN=N; prev=d
print('"'roots intervals', roots[:20], 'count', len(roots))
for a,b,_,_ in roots:
    sol=[]
    for N in range(a,b+1):
        if T(N)==N: sol.append(N)
    print(a,b,len(sol),sol[:10],sol[-10:] if sol else None)
PY"
