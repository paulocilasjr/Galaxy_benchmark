/bin/bash -lc "python3 - <<'PY'
import json
D=json.load(open('encode_dnase_grch38_bam_alignments.json'))['@graph']
# Count qm embedded types
from collections import Counter
c=Counter()
examples=[]
for f in D:
    for qm in f.get('quality_metrics',[]):
        if isinstance(qm,dict):
            c[tuple(qm.get('@type',[]))]+=1
            if len(examples)<5: examples.append(qm)
        else: c[(qm,)]+=1
print(c.most_common(20))
for f in D:
    if f.get('quality_metrics'):
        print(f['accession'], f['dataset'], f.get('file_size'), f.get('quality_metrics'))
        break
PY"
