import json,collections
p=json.load(open('encode_alpha_bams.json'))['@graph']
def m(x):
 vals=[]
 for q in x.get('quality_metrics',[]):
  if 'reads mapped' in q:vals.append(q['reads mapped'])
  if 'mapped' in q and isinstance(q['mapped'],int):vals.append(q['mapped'])
 return max(vals)
by=collections.defaultdict(list)
for x in p:by[x['dataset']].append(x)
for hi in [82,85,90,95,100,110,120,130,150,200]:
 groups=[(sum(m(x) for x in xs),d,xs) for d,xs in by.items() if 76e6<=sum(m(x) for x in xs)<=hi*1e6]
 print(hi,len(groups))
print('nearest 80-100')
for s,d,xs in sorted([(sum(m(x) for x in xs),d,xs) for d,xs in by.items() if 80e6<=sum(m(x) for x in xs)<=100e6])[:100]:
 print(f'{s/1e6:.3f}',d,[(x['accession'],m(x)) for x in xs])
PY"
