import json,collections
p=json.load(open('encode_alpha_bams.json'));g=p['@graph']
for f in ['status','biological_replicates','technical_replicates','mapped_run_type','dataset','date_created','file_size']:
 c=collections.Counter(str(x.get(f)) for x in g)
 print(f,len(c),c.most_common(30))

by=collections.defaultdict(list)
for x in g:by[x['dataset']].append(x)
print('datasets',len(by),'counts dist',collections.Counter(len(v) for v in by.values()))
print('largest grouped')
for d,v in sorted(by.items(),key=lambda z:-len(z[1]))[:10]:print(d,[(x['accession'],x['biological_replicates'],x['technical_replicates'],x['file_size']) for x in v])
PY"
