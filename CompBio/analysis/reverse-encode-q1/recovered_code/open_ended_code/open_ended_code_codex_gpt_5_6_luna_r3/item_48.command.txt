import gzip,json,pyBigWig,numpy as np
N=76275859; target={}
for l in gzip.open('reverse.encode.q1.tsv.gz','rt'):
 c,s,e,v=l.split();
 if c=='chr1':target[int(s)]=round(float(v)*N/1e6)

v=bw.stats('chr1',0,L,nBins=24896,type='sum')

print('bw stats first',b[:20]);print('target',a[:20]);
for typ in ['sum','mean','max']:
 z=np.nan_to_num(np.array(bw.stats('chr1',0,L,nBins=24896,type=typ),dtype=float));
 for f in [z,z/np.sum(z)*np.sum(a) if z.sum() else z]:
  print(typ,'sum',z.sum(),'corr',np.corrcoef(a,f)[0,1],'nz',np.count_nonzero(z),'ratio median',np.median(f[a>0]/a[a>0]))
bw.close()
