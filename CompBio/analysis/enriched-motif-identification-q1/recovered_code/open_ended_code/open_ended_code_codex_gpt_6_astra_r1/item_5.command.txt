/bin/bash -lc "cat > analysis/scan_motifs.py <<'PY'
from pathlib import Path
from Bio import motifs
import numpy as np
from collections import Counter
s=''.join(Path('dna_with_secret_motif.fasta').read_text().splitlines()[1:]).upper()
name_map={x.upper():x for x in ['FOXA1','GATA3','MAX','STAT1','OCT4','P53','NANOG','HNF4A','TBP','FOS','MYC','JUN','FOXO1','CTCF','TCF7','SOX2','E2F1','CREB1','GATA1','KLF3']}
name_map.update({'POU5F1':'OCT4','TP53':'P53'})
rows=[]
with open('analysis/jaspar2024_vertebrates.txt') as f:
 for m in motifs.parse(f,'jaspar'):
  if m.name.upper() not in name_map: continue
  m.pseudocounts=0.5
  p=m.pssm
  scores=np.maximum(p.calculate(s),p.reverse_complement().calculate(s))
  threshold=p.min+0.8*(p.max-p.min)
  rows.append((name_map[m.name.upper()],m.matrix_id,str(m.consensus),len(m),float(scores.max()),p.max,int((scores>=threshold).sum()),int((scores>=10).sum())))
rows.sort(key=lambda r:r[4],reverse=True)
out='name\\tmatrix\\tconsensus\\twidth\\tmax_score\\tideal_score\\thits_relative_0.8\\thits_10_bits\\n'
out+='\\n'.join('\\t'.join(map(str,r)) for r in rows)+'\\n'
Path('analysis/motif_scan.tsv').write_text(out)
print(out)
print('Exact inserted sequence ACCACCAGGGGGCG:',s.count('ACCACCAGGGGGCG'))
PY
python analysis/scan_motifs.py"
