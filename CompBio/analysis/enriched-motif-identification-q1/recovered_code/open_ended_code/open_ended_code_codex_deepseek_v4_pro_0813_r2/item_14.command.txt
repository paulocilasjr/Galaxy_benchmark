/bin/bash -c "python3 - <<'PY'
from pathlib import Path
from collections import Counter
s=''.join(l.strip() for l in Path('dna_with_secret_motif.fasta').read_text().splitlines() if not l.startswith('>'))
cons={
'FOXA1':'GTAAACAT','GATA3':'AGATAG','MAX':'CACGTG','STAT1':'TTCCAGGAA','OCT4':'ATGCAAA','P53':'ACATGCCCGGGCATGTC','NANOG':'GCAATCA','HNF4A':'CAAAGTCCA','TBP':'TATAAAA','FOS':'TGACTCA','MYC':'CACGTG','JUN':'ATGACTCAT','FOXO1':'GTAAACA','CTCF':'CCACCAGGGGGCG','TCF7':'CTTTGAT','SOX2':'ACAATGG','E2F1':'TTTGGCGC','CREB1':'TGACGTCA','GATA1':'AGATAG','KLF3':'GACCACGCCCA'}
comp=str.maketrans('ACGT','TGCA')
def rc(x): return x.translate(comp)[::-1]
print('motif', 'fwd', 'rev', 'max')
for k,v in cons.items():
    f=s.count(v); r=s.count(rc(v));
    print(f'{k:7} {f:5} {r:5} {max(f,r)}')
print('repeat exact variants:')
for m in ['CCACCAGGGGGCG','GCCACCAGGGGGCGC','TGGCCACCAGGGGGCGCTA','CGCCCCCTGGTGG']:
    print(m,s.count(m))
PY"
