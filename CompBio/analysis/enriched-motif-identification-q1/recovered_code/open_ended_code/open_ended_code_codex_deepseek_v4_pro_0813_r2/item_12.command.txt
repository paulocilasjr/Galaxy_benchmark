/bin/bash -c "python3 - <<'PY'
import urllib.request,json
allr=[]
url='https://jaspar.elixir.no/api/v1/matrix/?collection=CORE&tax_group=vertebrates&page_size=1000&format=json'
while url:
    js=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})).read())
    allr+=js['results']
    url=js['next']
print(len(allr))
terms=['STAT','POU5','OCT','P53','TP53','NANOG','TBP','TCF7','SOX2','KLF3']
for term in terms:
    matches=[r for r in allr if term.lower() in r['name'].lower() or term.lower() in r['matrix_id'].lower() or term.lower() in r['base_id'].lower()]
    print('\\nTERM',term,len(matches))
    for r in matches[:200]: print(r['matrix_id'],r['name'])
PY"
