s=''.join(l.strip() for l in Path('dna_with_secret_motif.fasta').read_text().splitlines() if not l.startswith('>'))

print(Counter(s))

for k in range(4,16):
    c=Counter(s[i:i+k] for i in range(len(s)-k+1))
    top=c.most_common(3)
    print(k, top)
