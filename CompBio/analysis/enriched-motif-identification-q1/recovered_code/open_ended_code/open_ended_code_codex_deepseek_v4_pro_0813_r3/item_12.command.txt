/bin/bash -lc "cat > fetch_motifs.py <<'PY'
import requests, json, time
from pathlib import Path

names_bases = {
 'FOXA1':'MA0148', 'GATA3':'MA0037', 'MAX':'MA0058', 'STAT1':'MA0137',
 'OCT4':'MA1115', 'P53':'MA0106', 'NANOG':'MA2339', 'HNF4A':'MA0114',
 'TBP':'MA0108', 'FOS':'MA1951', 'MYC':'MA0147', 'JUN':'MA0488',
 'FOXO1':'MA0480', 'CTCF':'MA1929', 'TCF7':'MA0769', 'SOX2':'MA0143',
 'E2F1':'MA0024', 'CREB1':'MA0018', 'GATA1':'MA0035', 'KLF3':'MA1516',
}

def get_json(url):
    for i in range(3):
        try:
            r=requests.get(url,timeout=30)
            if r.status_code==200:
                return r.json()
            print('status',r.status_code,url)
        except Exception as e:
            print('err',e)
        time.sleep(0.5)
    return None

allm=[]
for name,base in names_bases.items():
    d=get_json(f'https://jaspar.elixir.no/api/v1/matrix/{base}/versions/')
    if not d: continue
    versions=d.get('results',[])
    for v in versions:
        mid=v.get('matrix_id')
        md=get_json(f'https://jaspar.elixir.no/api/v1/matrix/{mid}/')
        if md:
            allm.append(md)
    print(name, len(versions), 'collected', len(allm))
Path('motifs.json').write_text(json.dumps(allm))
print('total',len(allm))
PY
python3 fetch_motifs.py"
