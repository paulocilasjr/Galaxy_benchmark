import requests, json, re
from Bio.Seq import Seq

def cons_from_pfm(pfm):
    IUPAC={'AC':'M','AG':'R','AT':'W','CG':'S','CT':'Y','GT':'K','ACG':'V','ACT':'H','AGT':'D','CGT':'B','ACGT':'N'}
    out=[]
    for col in zip(*(pfm[a] for a in 'ACGT')):
        best=max(col); letters=[a for a,n in zip('ACGT',col) if n==best]
        out.append(letters[0] if len(letters)==1 else IUPAC.get(''.join(sorted(letters)),'N'))
    return ''.join(out)
