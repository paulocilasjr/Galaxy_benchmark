/bin/bash -lc "python3 - <<'PY'
import json, math
from pathlib import Path
seq=''.join(l.strip() for l in Path('dna_with_secret_motif.fasta').read_text().splitlines() if not l.startswith('>'))
mats=json.loads(Path('matrices.json').read_text())
bases=['A','C','G','T']
bg=0.25

def pssm(pfm):
    n=len(pfm['A']); total=sum(pfm[b][0] for b in bases)
    rows=[]
    for i in range(n):
        total_i=sum(pfm[b][i] for b in bases)
        rows.append([math.log2((pfm[b][i]+0.1)/(total_i+0.4)/bg) for b in bases])
    return rows
for mid,j in mats.items():
    p=pssm(j['pfm']); n=len(p)
    scores=[]
    for i in range(len(seq)-n+1):
        s=sum(p[k][bases.index(seq[i+k])] for k in range(n))
        scores.append(s)
    mx=max(scores)
    # relative max normalized by min and max
    mn=min(scores)
    rel=[(s-mn)/(mx-mn) for s in scores]
    # count >= 0.8 rel
    cnt=sum(r>=0.8 for r in rel)
    if cnt>0 or mid in ['MA0139.2','MA1929.2','MA1930.2']:
        print(f'{mid:10} {j.get(\"name\",\"\"):18} len={n:2} max={mx:6.2f} cnt80={cnt}')
PY"
