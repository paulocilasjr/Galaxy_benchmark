/bin/bash -lc "python3 - <<'PY'
import requests, time
names='TCF7 SOX2 E2F1 CREB1 GATA1 KLF3'.split()
for n in names:
    try:
        r=requests.get(f'https://jaspar.elixir.no/api/v1/matrix/?search={n}&collection=CORE&tax_group=vertebrates&version=latest', timeout=(5,15))
        j=r.json()
        print(n, j['count'], [(x['matrix_id'],x['name']) for x in j['results']])
    except Exception as e:
        print(n, 'ERR', e)
    time.sleep(.2)
PY"
