/bin/bash -lc "python3 - <<'PY'
import requests, json, time
names='FOXA1 GATA3 MAX STAT1 OCT4 P53 NANOG HNF4A TBP FOS MYC JUN FOXO1 CTCF TCF7 SOX2 E2F1 CREB1 GATA1 KLF3'.split()
base='https://jaspar.elixir.no/api/v1/matrix/?search={}&collection=CORE&tax_group=vertebrates&version=latest'
for n in names:
    r=requests.get(base.format(n), timeout=30)
    j=r.json()
    print(n, j['count'], [(x['matrix_id'],x['name']) for x in j['results']])
    time.sleep(.2)
PY"
