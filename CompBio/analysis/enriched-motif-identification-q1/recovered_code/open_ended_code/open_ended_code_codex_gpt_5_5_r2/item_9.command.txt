/bin/bash -lc "cat > /dev/null <<'EOF'
EOF
python3 - <<'PY'
import urllib.request, json, time, math
names='FOXA1 GATA3 MAX STAT1 OCT4 P53 NANOG HNF4A TBP FOS MYC JUN FOXO1 CTCF TCF7 SOX2 E2F1 CREB1 GATA1 KLF3'.split()
for name in names:
    url=f'https://jaspar.elixir.no/api/v1/matrix/?search={name}&collection=CORE&tax_group=vertebrates&page_size=100'
    with urllib.request.urlopen(url, timeout=30) as r:
        data=json.load(r)
    exact=[x for x in data['results'] if x['name'].upper()==name.upper()]
    print(name, 'count', len(exact), [(x['matrix_id'],x['name']) for x in exact])
PY"
