/bin/bash -lc "archive_url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE242nnn/GSE242423/suppl/GSE242423_RAW.tar'
d2_barcodes_start=244328960
d2_barcodes_size=7391073
d2_barcodes_end="'$((d2_barcodes_start+d2_barcodes_size-1))
d2_barcodes_blocks=$(((d2_barcodes_size+511)/512))
d2_matrix_start=$((d2_barcodes_start+512+d2_barcodes_blocks*512))
d2_matrix_size=125124004
d2_matrix_end=$((d2_matrix_start+d2_matrix_size-1))
printf '"'%s\\n' \"barcode range: "'$d2_barcodes_start-$d2_barcodes_end" "matrix range: $d2_matrix_start-$d2_matrix_end"
curl -L --fail --silent --show-error --range "$d2_barcodes_start-$d2_barcodes_end" -o GSM7763420_D2.barcodes.tsv.gz "$archive_url"
curl -L --fail --silent --show-error --range "$d2_matrix_start-$d2_matrix_end" -o GSM7763420_D2.matrix.mtx.gz "$archive_url"
ls -l GSM7763420_D2.barcodes.tsv.gz GSM7763420_D2.matrix.mtx.gz
printf '"'barcode gzip: '; gzip -t GSM7763420_D2.barcodes.tsv.gz && echo OK
printf 'matrix gzip: '; gzip -t GSM7763420_D2.matrix.mtx.gz && echo OK"
