/bin/bash -lc 'PYTHONPATH="$PWD/packages" python - <<'"'PY'
import json
import RNA
from pathlib import Path
record = json.loads(Path('rnacentral.json').read_text())
sequence = record['sequence']
fc = RNA.fold_compound(sequence)
structure, energy = fc.mfe()
assert len(sequence) == len(structure) == record['length']
assert abs(fc.eval_structure(structure) - energy) < 0.001
result = f'{structure},{energy:.2f} kcal/mol'
assert '\\n' not in result
Path('answer.txt').write_text(result + '\\n')
Path('sequence.fasta').write_text('>URS0000668495\\n' + sequence + '\\n')
print(RNA.__version__)
print(result)
PY"
