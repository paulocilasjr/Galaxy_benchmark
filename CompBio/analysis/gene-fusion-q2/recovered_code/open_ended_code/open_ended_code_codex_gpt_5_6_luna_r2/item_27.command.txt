/bin/bash -lc "python3 - <<'PY'
import re
def g(r):
 x=r.split('|'); return x[5] if len(x)>5 else r
def qr(c):
 t=re.findall(r'(\\d+)([MIDNSHP=X])',c); p=0
 for n,o in t:
  if o in 'HS':p+=int(n)
  else:break
 a=sum(int(n) for n,o in t if o in 'MI=X'); return p,p+a
seen=set()
with open('pc_align.sam') as f:
 for line in f:
  if line[0]=='@':continue
  a=line.rstrip().split('\\t'); flag=int(a[1]); rn=a[2]; gg=g(rn)
  sa=next((x[5:] for x in a[11:] if x.startswith('SA:Z:')),None)
  if not sa:continue
  for ent in sa.strip(';').split(';'):
   b=ent.split(',')
   if len(b)<6:continue
   sg=g(b[0])
   if {gg,sg}=={'ZIC2','SP1'}:
    qn=a[0]
    if qn not in seen:
     seen.add(qn);print('\\n',qn,'primary',flag,gg,a[3],a[4],a[5], 'range',qr(a[5]),'SA',sg,b[1:])
PY"
