/bin/bash -lc "python3 - <<'PY'
import collections
cand={'MYC','SP1','CATSPER4','ZIC2'}
def g(r):
 x=r.split('|');return x[5] if len(x)>5 else r
by=collections.defaultdict(list)
with open('asm_to_pc.paf') as f:
 for l in f:
  a=l.rstrip().split('\\t')
  if len(a)<12:continue
  tags={x.split(':',2)[0]:x for x in a[12:]}
  by[a[0]].append((g(a[5]),int(a[2]),int(a[3]),int(a[7]),int(a[8]),int(a[10]),int(a[11]),a[4],a[5].split('|')[0],tags.get('tp','')))
rows=[]
for q,als in by.items():
 strong=[x for x in als if x[5]>=30]
 gs={x[0] for x in strong}
 if gs&cand:
  # summarize each gene by max query span/block and intervals
  summ={}
  for x in strong:
   gene=x[0]; old=summ.get(gene)
   if old is None or x[5]>old[0]:summ[gene]=[x[5],x[1],x[2],x[3],x[4],x[6],x[7],x[8],x[9]]
  rows.append((q,max(x[5] for x in strong),summ))
print('candidate contigs',len(rows))
for q,m,s in sorted(rows,key=lambda z:-z[1]):
 print(q,'max',m,';',' | '.join(f'{k}:q{v[1]}-{v[2]} t{v[3]}-{v[4]} b{v[0]} mq{v[6]} {v[7]}' for k,v in sorted(s.items())))
PY"
