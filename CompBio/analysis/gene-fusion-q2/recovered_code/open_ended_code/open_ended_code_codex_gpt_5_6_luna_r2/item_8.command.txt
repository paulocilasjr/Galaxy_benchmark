/bin/bash -lc "python3 - <<'PY'
import gzip, collections, statistics
fn='gene.fusion.q2.fq.gz'
counts=collections.Counter(); lengths=[]; gc=[]
with gzip.open(fn,'rt') as f:
    while True:
        h=f.readline()
        if not h: break
        s=f.readline().strip(); f.readline(); f.readline()
        counts[s]+=1; lengths.append(len(s)); gc.append((s.count('G')+s.count('C'))/len(s))
print('reads',sum(counts.values()),'unique',len(counts),'duplicate_reads',sum(n-1 for n in counts.values() if n>1))
print('lengths',collections.Counter(lengths))
print('gc_mean',sum(gc)/len(gc),'gc_median',statistics.median(gc),'gc_minmax',(min(gc),max(gc)))
print('top exact sequences:')
for s,n in counts.most_common(15): print(n,s)
PY"
