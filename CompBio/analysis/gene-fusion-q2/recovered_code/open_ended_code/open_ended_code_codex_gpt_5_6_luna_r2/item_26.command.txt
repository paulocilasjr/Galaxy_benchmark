/bin/bash -lc "python3 - <<'PY'
import re, collections
fn='pc_align.sam'
def gene(r):
    x=r.split('|'); return x[5] if len(x)>5 else r
def qrange(cigar):
    toks=re.findall(r'(\\d+)([MIDNSHP=X])',cigar)
    prefix=0
    for n,o in toks:
        n=int(n)
        if o in 'HS': prefix+=n
        else: break
    aligned=sum(int(n) for n,o in toks if o in 'MI=X')
    return prefix,prefix+aligned
pairs=collections.Counter(); reads=collections.defaultdict(list)
with open(fn) as f:
  for line in f:
    if line[0]=='@': continue
    a=line.rstrip('\\n').split('\\t'); qn=a[0]; flag=int(a[1]); rn=a[2]; mapq=int(a[4]); cigar=a[5]
    if rn=='*': continue
    g=gene(rn); qs,qe=qrange(cigar); strand='-' if flag&16 else '+'
    sa=next((x[5:] for x in a[11:] if x.startswith('SA:Z:')),None)
    if not sa: continue
    for ent in sa.strip(';').split(';'):
      if not ent: continue
      b=ent.split(',')
      if len(b)<6: continue
      srn,ss,sc,smq,snm=b[0],b[2],b[3],int(b[4]),int(b[5]); sg=gene(srn)
      if sg==g: continue
      sqs,sqe=qrange(sc)
      al=[(qs,qe,g),(sqs,sqe,sg)]; al.sort(key=lambda z:z[0], reverse=(strand=='-'))
      ordered=(al[0][2],al[1][2]); pairs[ordered]+=1
      reads[qn].append((ordered,(qs,qe),(sqs,sqe),mapq,smq,flag,sc,cigar))
print('distinct split reads',len(reads),'pair observations',sum(pairs.values()),'distinct pairs',len(pairs))
for pair,n in pairs.most_common(100):
    lens=collections.Counter(); maps=collections.Counter(); strs=collections.Counter(); qset=set()
    for q,vals in reads.items():
      for o,qr1,qr2,mq,smq,fl,sc,pc in vals:
       if o==pair:
        qset.add(q); lens[min(qr1[1]-qr1[0],qr2[1]-qr2[0])]+=1; maps[(mq,smq)]+=1; strs['-' if fl&16 else '+']+=1
    print(n, '-'.join(pair), 'reads',len(qset),'minseg',dict(lens),'strand',dict(strs),'mapq',maps.most_common(4))
PY"
