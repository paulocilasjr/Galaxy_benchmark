/bin/bash -lc "samtools faidx cand_chrs.fa && python3 - <<'PY'
import subprocess
regs=[('12',53406654,53406854),('13',99984844,99985044),('12',53409463,53409663),('1',26202386,26202586)]
seqs=[]
for chrom,s,e in regs:
    header='12' if chrom=='12' else ('1' if chrom=='1' else chrom)
    # header exact likely '12' for chr12, '1', '13'
    res=subprocess.run(['samtools','faidx','cand_chrs.fa',f'{header}:{s}-{e}'],capture_output=True,text=True)
    seq=''.join(res.stdout.splitlines()[1:])
    seqs.append(seq)
    print('\\n',chrom,s,e,res.stdout.splitlines()[0])
    print(seq)
# compare SP1/ZIC2 breakpoint sequences and SP1/CATSPER4
import difflib
for a,b,label in [(seqs[0],seqs[1],'SP1-ZIC2'),(seqs[2],seqs[3],'SP1-CATSPER4')]:
    sm=difflib.SequenceMatcher(None,a,b)
    print(label,'identity',sm.ratio())
PY"
