/bin/bash -lc "python3 - <<'PY'
import subprocess, re
# get read seqs for examples from BAM
reads={}
proc=subprocess.Popen(['samtools','view','reads_cdna.bam'],stdout=subprocess.PIPE,text=True)
for line in proc.stdout:
    p=line.split('\\t')
    if p[0] in ['READ_1307','READ_13910','READ_16346','READ_4728','READ_143722','READ_16346']:
        reads[p[0]]=p[9]
proc.wait()
# extract transcripts from fasta index by IDs list
ids=set()
with open('hs_cdna.fa.fai') as f:
    pass
# Use samtools faidx for specific IDs
ids=['ENST00000477213.1','ENST00000854917.2','ENST00000327443.9','ENST00000520751.1','ENST00000456354.7','ENST00001068767.1','ENST00000376335.8']
# make ref subset
with open('cand_refs.fa','w') as out:
    for i in ids:
        res=subprocess.run(['samtools','faidx','hs_cdna.fa',i],capture_output=True,text=True)
        out.write(res.stdout)
# query
with open('cand_reads.fa','w') as out:
    for n,s in reads.items():
        out.write(f'>{n}\\n{s}\\n')
print('read lengths', {k:len(v) for k,v in reads.items()})
PY
minimap2 -ax sr -t 2 cand_refs.fa cand_reads.fa 2>/dev/null | samtools view -S -F 0x900 - | cut -f1-6,12- | head -100"
