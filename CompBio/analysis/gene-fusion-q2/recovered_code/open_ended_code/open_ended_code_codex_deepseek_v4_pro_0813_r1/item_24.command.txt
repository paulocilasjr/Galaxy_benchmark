/bin/bash -lc "python3 - <<'PY'
import re, subprocess
trans_gene={}
with open('hs_cdna.fa') as f:
    for line in f:
        if line.startswith('>'):
            h=line[1:].strip(); tid=h.split()[0]
            for part in h.split():
                if part.startswith('gene_symbol:'): trans_gene[tid]=part.split(':',1)[1];break

def query_interval(cigar):
    # returns (qstart,qend) 0-based exclusive, plus ref consume
    m=re.findall(r'(\\d+)([MIDNSHP=X])',cigar); q=0
    qs=None; qe=None
    for ln,op in m:
        ln=int(ln)
        if op in 'M=X': qe=q+ln; q+=ln
        elif op in 'IS': q+=ln
        elif op in 'H': pass
        elif op in 'DNP': pass
    # leading/trailing clips included? use CIGAR first/last
    # determine first aligned op after leading S/H
    aligned_ops=[]
    q=0
    for i,(ln,op) in enumerate(m):
        ln=int(ln)
        if op in 'SH':
            q+=ln
        elif op in 'M=X':
            if qs is None: qs=q
            q+=ln; qe=q
        elif op=='I':
            q+=ln
        elif op in 'DNP': pass
    return (qs,qe)

proc=subprocess.Popen(['samtools','view','reads_cdna.bam'],stdout=subprocess.PIPE,text=True)
rows=[]
for line in proc.stdout:
    p=line.rstrip().split('\\t'); flag=int(p[1]); rname=p[2]
    tags={}
    for f in p[11:]:
        if ':' in f:
            k,t,v=f.split(':',2);tags[k]=v
    sa=tags.get('SA')
    if not sa: continue
    g1=trans_gene.get(rname,'')
    q1=query_interval(p[5]); seq=p[9]
    for entry in sa.rstrip(';').split(';'):
        if not entry: continue
        parts=entry.split(',');
        if len(parts)<6: continue
        rt=parts[0]; g2=trans_gene.get(rt,'')
        pair=(g1,g2)
        if pair in [('SP1','MYC'),('MYC','SP1'),('CATSPER4','SP1'),('SP1','CATSPER4'),('ZIC2','SP1'),('SP1','ZIC2')]:
            q2=query_interval(parts[3])
            ov=max(0,min(q1[1] if q1[1] else 0,q2[1] if q2[1] else 0)-max(q1[0] if q1[0] is not None else 0,q2[0] if q2[0] is not None else 0))
            rows.append((pair,p[0],p[2],p[3],p[5],parts[0],parts[1],parts[3],q1,q2,ov,seq))
proc.wait()
for pair in [('SP1','MYC'),('CATSPER4','SP1'),('ZIC2','SP1')]:
    print('\\n===',pair)
    sub=[r for r in rows if r[0]==pair]
    print('count',len(sub))
    for r in sub[:15]:
        print(r[1],r[2],r[3],r[4],'|',r[5],r[6],r[7],'q',r[8],r[9],'ov',r[10])
        print(' ',r[11])
PY"
