/bin/bash -lc "python3 - <<'PY'
import subprocess, gzip
# selected full reads
names={'READ_1307','READ_4535','READ_13910','READ_16346','READ_18554','READ_21691','READ_143722','READ_18710','READ_4728'}
reads={}
with gzip.open('gene.fusion.q2.fq.gz','rt') as f:
    while True:
        h=f.readline().strip()
        if not h:break
        s=f.readline().strip();f.readline();f.readline()
        n=h[1:] if h.startswith('@') else h
        if n in names: reads[n]=s
with open('sel_reads.fa','w') as out:
    for n,s in reads.items():out.write(f'>{n}\\n{s}\\n')
# refs ids
refs=['ENST00000376335.8','ENST00000477213.1','ENST00000854917.2','ENST00000327443.9','ENST00001068767.1','ENST00000426431.2','ENST00000520751.1','ENST00000524013.3','ENST00000456354.7','ENST00000518899.5']
for rid in refs:
    with open('one_ref.fa','w') as out:
        res=subprocess.run(['samtools','faidx','hs_cdna.fa',rid],capture_output=True,text=True)
        out.write(res.stdout)
    print('\\nREF',rid)
    p=subprocess.run(['minimap2','-ax','sr','--eqx','one_ref.fa','sel_reads.fa'],capture_output=True,text=True)
    for line in p.stdout.splitlines():
        if line.startswith('@'):continue
        f=line.split('\\t')
        if len(f)>=6:
            print(f[0],'flag',f[1],'pos',f[3],'mapq',f[4],'cigar',f[5])
PY"
