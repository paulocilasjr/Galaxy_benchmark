# quick print transcriptome directed pair counts for MYC/CATSPER4 from saved? recompute using earlier logic maybe simplified
import subprocess,re,collections,gzip
trans_gene={}
with open('hs_cdna.fa') as f:
 for line in f:
  if line.startswith('>'):
   h=line[1:].strip();tid=h.split()[0]
   for part in h.split():
    if part.startswith('gene_symbol:'):trans_gene[tid]=part.split(':',1)[1];break

pairs=collections.Counter();seen=set()
for line in subprocess.Popen(['samtools','view','-F','0x900','reads_cdna.bam'],stdout=subprocess.PIPE,text=True).stdout:
 p=line.rstrip().split('\\t');g1=trans_gene.get(p[2],'');tags={}
 for f in p[11:]:
  if ':' in f:
   k,t,v=f.split(':',2);tags[k]=v
 sa=tags.get('SA')
 if not sa:continue
 for entry in sa.rstrip(';').split(';'):
  if not entry:continue
  parts=entry.split(',')
  if len(parts)<6:continue
  g2=trans_gene.get(parts[0],'')
  if not g1 or not g2 or g1==g2:continue
  key=(p[0],tuple(sorted((g1,g2))))
  if key in seen:continue
  seen.add(key);pairs[tuple(sorted((g1,g2)))]+=1
for g in ['MYC','CATSPER4']:
 print('\\nEdges for',g)
 for k,v in sorted(pairs.items(),key=lambda kv:-kv[1]):
