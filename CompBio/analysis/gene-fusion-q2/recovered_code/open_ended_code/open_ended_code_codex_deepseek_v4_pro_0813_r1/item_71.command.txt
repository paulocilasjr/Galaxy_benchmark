import gzip
pos=127736577;chrom='8'
with gzip.open('hs.gtf.gz','rt') as f:
 for line in f:
  if line.startswith('#'):continue
  p=line.rstrip().split('\\t')
  if len(p)<9 or p[2]"'!='"'gene':continue
  if p[0]==chrom and int(p[3])<=pos<=int(p[4]):
   attrs={}
   for kv in [x.strip() for x in p[8].split(';') if x.strip()]:
    if ' ' in kv:
     k,v=kv.split(' ',1);attrs[k]=v.strip('\"')
    else:attrs[kv]=''
   print(p[0],p[3],p[4],p[6],attrs.get('gene_name'),attrs.get('gene_id'),attrs.get('gene_biotype'))
PY"
