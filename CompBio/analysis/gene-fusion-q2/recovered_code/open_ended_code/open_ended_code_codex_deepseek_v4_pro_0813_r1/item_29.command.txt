/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict, Counter
import subprocess, re
trans_gene={}
with open('hs_cdna.fa') as f:
    for line in f:
        if line.startswith('>'):
            h=line[1:].strip(); tid=h.split()[0]
            for part in h.split():
                if part.startswith('gene_symbol:'): trans_gene[tid]=part.split(':',1)[1];break

def qint(cigar):
    # query interval in stored SEQ coords (hard clipped excluded already), returns (start,end) aligned M/I? We need M only
    m=re.findall(r'(\\d+)([MIDNSHP=X])',cigar)
    q=0; qs=None; qe=None
    for ln,op in m:
        ln=int(ln)
        if op in 'SH':
            q+=ln
        elif op in 'M=X':
            if qs is None: qs=q
            q+=ln; qe=q
        elif op=='I':
            q+=ln
        elif op in 'DNP': pass
    return (qs,qe)

pair_reads=defaultdict(set)
proc=subprocess.Popen(['samtools','view','reads_cdna.bam'],stdout=subprocess.PIPE,text=True)
for line in proc.stdout:
    p=line.rstrip().split('\\t'); flag=int(p[1]); rname=p[2]
    tags={}
    for f in p[11:]:
        if ':' in f:
            k,t,v=f.split(':',2); tags[k]=v
    sa=tags.get('SA')
    if not sa: continue
    g1=trans_gene.get(rname,'')
    q1=qint(p[5])
    if not q1 or q1[1] is None: continue
    # require aligned len >=20
    if q1[1]-q1[0] < 20: continue
    for entry in sa.rstrip(';').split(';'):
        if not entry: continue
        parts=entry.split(',')
        if len(parts)<6: continue
        rt=parts[0]; g2=trans_gene.get(rt,'')
        if not g1 or not g2 or g1==g2: continue
        q2=qint(parts[3])
        if not q2 or q2[1] is None: continue
        if q2[1]-q2[0] < 20: continue
        ov=max(0,min(q1[1],q2[1])-max(q1[0],q2[0]))
        if ov<=8:
            pair_reads[(g1,g2)].add(p[0])
proc.wait()
print('pairs',len(pair_reads))
# rank by support
for (g1,g2),reads in sorted(pair_reads.items(), key=lambda kv:-len(kv[1]))[:100]:
    print(f'{len(reads):4d} {g1} -> {g2}')
PY"
