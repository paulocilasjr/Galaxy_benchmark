import gzip, re
from collections import defaultdict
# parse exons per transcript id; keep gene_name, transcript_name, start,end,strand, exon_number
exons=defaultdict(list)
trans_info={}
with gzip.open('gencode.v47.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        chrom,src,typ,start,end,score,strand,phase,attrs=p
        if typ"'!='"'exon': continue
        d=dict(re.findall(r'(\\S+)\\s+\"(["'^"]*)"'"',attrs))
        tid=d.get('transcript_id',''); gid=d.get('gene_id',''); gname=d.get('gene_name',''); tname=d.get('transcript_name',''); en=d.get('exon_number','')
        trans_info[tid]=(gname,tname)
        exons[tid].append((int(start),int(end),strand,en))
# transcripts of interest
for tid in ['ENST00000520751.1','ENST00000377970.6','ENST00000259523.10','ENST00000621592.8','ENST00000641036.1']:
    gname,tname=trans_info.get(tid,('?','?'))
    ex=sorted(exons.get(tid,[]))
    print('MYC?',tid,gname,tname,'exons',len(ex))
    for e in ex[:20]: print(' ',e)
for tid in ['ENST00000426431.2','ENST00000327443.9']:
    gname,tname=trans_info.get(tid,('?','?'))
    ex=sorted(exons.get(tid,[]))
    print('\\nSP1',tid,gname,tname,'exons',len(ex))
    # cumulative transcript coords (genomic order -> transcript order depends strand)
    # for + strand, exon order ascending; for - strand, descending
    strand=ex[0][2] if ex else '+'
    ordered=ex if strand=='+' else list(reversed(ex))
    cum=0
    for i,e in enumerate(ordered,1):
        L=e[1]-e[0]+1
        print('exon',i,e,'transcript_start0',cum,'transcript_end0',cum+L)
        cum+=L
    print('transcript length approx',cum)
for tid in ['ENST00000376335.8','ENST00000477213.1','ENST00000468291.1','ENST00000490085.5']:
    gname,tname=trans_info.get(tid,('?','?'))
    ex=sorted(exons.get(tid,[]))
    print('\\nZIC2',tid,gname,tname,'exons',len(ex))
    strand=ex[0][2] if ex else '+'
    ordered=ex if strand=='+' else list(reversed(ex))
    cum=0
    for i,e in enumerate(ordered,1):
        L=e[1]-e[0]+1

    print('len',cum)
for tid in ['ENST00000456354.7','ENST00000518899.5']:
    gname,tname=trans_info.get(tid,('?','?'))

    print('\\nCATSPER4',tid,gname,tname,'exons',len(ex))
