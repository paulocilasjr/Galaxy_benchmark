/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
from Bio.Seq import Seq
s=str(SeqIO.read('NODE_63_rc.fa','fasta').seq)
for frame in range(3):
    seq=Seq(s[frame:])
    prot=seq.translate(to_stop=False)
    # longest ORF in this frame
    import re
    orfs=[(m.start(),m.end()) for m in re.finditer(r'M["'^'"\\*]*\\*',str(prot))]
    longest=max(orfs,key=lambda x:x[1]-x[0],default=None)
    print('frame',frame,'len aa',len(prot),'longest ORF', (longest[1]-longest[0]) if longest else None, prot[:60])
PY"
