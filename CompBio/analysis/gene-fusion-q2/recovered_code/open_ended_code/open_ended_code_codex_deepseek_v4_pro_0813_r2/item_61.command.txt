/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
s=str(SeqIO.read('NODE_63_rc.fa','fasta').seq)
read='CTGCTTAGACGCTGGATTTTTTTCGGGTAGTGGAAAACTAGGTGATCATGGAGCTCAGGTTGGTCTCCATGGGGCTGGTGGTGATGGAATACATGATGACACAGCAG'
rc=str(__import__('Bio').Seq.Seq(read).reverse_complement())
print('read',read)
print('rc  ',rc)
# find approximate with one mismatch maybe using local alignment? use difflib
import difflib
for target,label in [(s,'rc_contig')]:
    for seq,sl in [(read,'read'),(rc,'read_rc')]:
        sm=difflib.SequenceMatcher(None,seq,target,autojunk=False)
        blocks=sm.get_matching_blocks()
        print(label,sl,'best match',max(blocks,key=lambda b:b.size))
        print('pos in target of best',target.find(seq[:30]))
PY"
