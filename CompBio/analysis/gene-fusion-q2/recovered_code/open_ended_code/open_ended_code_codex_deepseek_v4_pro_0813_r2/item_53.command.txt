/bin/bash -lc "bwa index NODE_63_rc.fa && bwa mem -t 8 NODE_63_rc.fa gene.fusion.q2.fq.gz > node63_rc_reads.sam; samtools view -b -F 0x904 node63_rc_reads.sam > node63_rc_reads.bam; samtools sort -o node63_rc_reads.sorted.bam node63_rc_reads.bam; samtools index node63_rc_reads.sorted.bam; python3 - <<'PY'
import pysam
bam=pysam.AlignmentFile('node63_rc_reads.sorted.bam','rb')
L=1396
dep=[0]*L
for aln in bam.fetch():
    if aln.is_unmapped: continue
    for s,e in aln.get_blocks(): 
        for i in range(max(0,s),min(L,e)): dep[i]+=1
# segments (0-based start,end) based on rc alignments
segs=[('MYC',0,393),('SP1_A',388,564),('ZIC2',559,727),('SP1_C',721,927),('CATSPER4',923,1396)]
for name,s,e in segs:
    print(name,s,e,'avg depth',sum(dep[s:e])/(e-s),'min',min(dep[s:e]),'max',max(dep[s:e]))
for i in range(0,L,50):
    print(i,sum(dep[i:i+50])/50)
PY"
