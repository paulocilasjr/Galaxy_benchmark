/bin/bash -lc "samtools view -F 0x904 aln_transcripts.sam | python3 -c '
import sys,re
from collections import Counter
c=Counter()
for line in sys.stdin:
    if line.startswith(\"@\"): continue
    parts=line.rstrip(\"\\n\").split(\"\\t\")
    rna=parts[2]
    if rna==\"*\": continue
    # header fields |: transcript id, gene id, otthumg, otthumt, tscript name, gene name, len, type
    f=rna.split(\"|\")
    gene=f[5] if len(f)>5 else rna
    c[gene]+=1
for g,n in c.most_common(100): print(n,g)
'"
