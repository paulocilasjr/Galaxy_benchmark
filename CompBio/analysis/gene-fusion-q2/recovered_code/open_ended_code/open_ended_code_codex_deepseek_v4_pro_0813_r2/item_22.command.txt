/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
for f in ['spades_out/transcripts.fasta','spades_out/hard_filtered_transcripts.fasta','spades_out/soft_filtered_transcripts.fasta']:
    print('FILE', f)
    seqs=[]
    for rec in SeqIO.parse(f,'fasta'):
        seqs.append((len(rec.seq), rec.id, rec.description))
    seqs.sort(reverse=True)
    print('count',len(seqs),'total',sum(x[0] for x in seqs),'N50-ish')
    for L,i,d in seqs[:30]:
        print(L,i)
PY"
