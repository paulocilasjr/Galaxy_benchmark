/bin/bash -lc "python3 - <<'PY'
import pysam
bam=pysam.AlignmentFile('node63_rc_reads.sorted.bam','rb')
boundaries=[('MYC-SP1',388),('SP1-ZIC2',559),('ZIC2-SP1',720),('SP1-CATSPER4',923)]
for name,pos in boundaries:
    left=right=span=0
    for aln in bam.fetch():
        if aln.is_unmapped or aln.is_secondary or aln.is_supplementary: continue
        blocks=aln.get_blocks()
        if not blocks: continue
        s=blocks[0][0]; e=blocks[-1][1]
        if s < pos-50 and e > pos+50:
            if s < pos and e > pos:
                span+=1
        # count any alignment with coverage across pos
        if any(s<=pos<e for s,e in blocks): 
            if s < pos: left+=1
            if e > pos: right+=1
    print(name,'reads covering boundary',span,'left side',left,'right side',right)
PY"
