/bin/bash -lc "python3 - <<'PY'
import sys,re
from collections import defaultdict,Counter

def gene_of(rna):
    if rna=='*': return '*'
    f=rna.split('|'); return f[5] if len(f)>5 else rna
recs=defaultdict(list)
for line in open('contigs_to_tx.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    if len(p)<11: continue
    recs[p[0]].append(p)
# get lengths/cov from name
rows=[]
for q,rs in recs.items():
    m=re.match(r'NODE_\\d+_length_(\\d+)_cov_([0-9.]+)',q)
    L=int(m.group(1)) if m else 0; cov=float(m.group(2)) if m else 0
    genes=Counter(); mapped=0
    for p in rs:
        if int(p[1])&4: continue
        g=gene_of(p[2]); genes[g]+=1; mapped+=1
    distinct=sum(1 for g,n in genes.items() if g"'!='"'*')
    if distinct>1:
        rows.append((cov,L,q,genes))
for cov,L,q,genes in sorted(rows,reverse=True)[:100]:
    # skip pseudogenes/lncRNA? print all
    print(f'{cov:.2f}\\t{L}\\t{dict(genes.most_common(10))}\\t{q}')
PY"
