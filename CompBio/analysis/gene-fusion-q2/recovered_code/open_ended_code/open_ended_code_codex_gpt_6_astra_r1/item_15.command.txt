/bin/bash -lc "python3 - <<'PY'
import gzip
names={'MYC','SP1','ZIC2','CATSPER4','LINC01766'}
with gzip.open('reference/gencode.v47.transcripts.fa.gz','rt') as f,open('reference/candidates.fa','w') as o:
 keep=False
 for l in f:
  if l.startswith('>'):keep=l.split('|')[5] in names
  if keep:o.write(l)
with gzip.open('reference/gencode.v47.annotation.gtf.gz','rt') as f,open('reference/candidates.gtf','w') as o:
 for l in f:
  if any('gene_name \"'+n+'\"' in l for n in names):o.write(l)
PY"
