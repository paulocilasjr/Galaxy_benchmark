from collections import defaultdict, Counter
hits=defaultdict(list)
for line in open('node83_reads_blast.tsv'):
    f=line.rstrip().split('\\t')
    qid=f[0]; gene=f[1].split('|')[5]; length=int(f[3]); qs=int(f[6]); qe=int(f[7])
    hits[qid].append((gene,length,qs,qe))
pairs=Counter()
examples={}
for q,hs in hits.items():
    # find all non-overlapping-ish pairs of distinct genes with min len 15 and no more than 5 gap/overlap
    arr=sorted(hs,key=lambda x:x[2])
    best_for_q=[]
    for i in range(len(arr)):
        g1,l1,qs1,qe1=arr[i]
        for j in range(i+1,len(arr)):
            g2,l2,qs2,qe2=arr[j]
            if g1==g2: continue
            if qs2 <= qe1 and qs2 >= qs1: # overlap within first
                # allow if one extends past
                pass
            if qe1 - qs2 <= 5 and qs2 - qe1 <= 5 and l1>=12 and l2>=12:
                # But this may pair overlapping HSPs from same region. Use complementary coverage: qe1 < qe2
                if qe1 < qe2:
                    pairs[(g1,g2)]+=1
                    examples.setdefault((g1,g2),q)
                    break
        if any(k for k in examples if examples[k]==q): pass
# better manually count by mapping NODE_83 positions? let's print
print(pairs.most_common())
for pair,q in examples.items():
    print(pair,q)
PY"
