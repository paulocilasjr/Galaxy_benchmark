/bin/bash -lc "python3 - <<'PY'
import pysam
bam=pysam.AlignmentFile('NODE_83_length_1396_cov_36.266219.bam','rb')
# Reads with ref alignment start between 480 and 600, so they span SP1->ZIC2 boundary ~565
names=[]
for al in bam.fetch('NODE_83_length_1396_cov_36.266219',450,620):
    if al.is_unmapped or al.is_secondary or al.is_supplementary: continue
    if 480 <= al.reference_start <= 580:
        names.append(al.query_name)
print('count',len(names))
# parse blast hits for these names
import collections
hits=collections.defaultdict(list)
for line in open('node83_reads_blast.tsv'):
    f=line.rstrip().split('\\t')
    q=f[0]
    if q in set(names):
        gene=f[1].split('|')[5]; length=int(f[3]); qs=int(f[6]); qe=int(f[7])
        hits[q].append((gene,length,qs,qe))
for n in names[:20]:
    hs=sorted(hits[n], key=lambda x:x[2])
    print(n, hs[:8])
PY"
