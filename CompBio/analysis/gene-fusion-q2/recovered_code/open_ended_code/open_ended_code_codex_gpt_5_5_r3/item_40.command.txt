/bin/bash -lc "cat > work/contig_multigene.py <<'PY'
from collections import defaultdict
import re
paf='work/contigs_vs_tx.paf'
by=defaultdict(list)
for line in open(paf):
    f=line.rstrip('\\n').split('\\t')
    q=f[0]; qlen=int(f[1]); qs=int(f[2]); qe=int(f[3]); strand=f[4]; t=f[5]
    gene=t.split('|')[5] if '|' in t else t
    m=int(f[9]); alen=int(f[10]); mapq=int(f[11])
    if alen<80 or m/alen<0.96: continue
    by[q].append((qs,qe,strand,gene,t,m,alen,mapq))

for q,als in by.items():
    genes=set(a[3] for a in als)
    if len(genes)<2: continue
    # Reduce per gene to non-nested distinct best intervals
    cand=[]
    for gene in genes:
        g=[a for a in als if a[3]==gene]
        # select intervals not contained in a longer interval from same gene, top by length
        g=sorted(g, key=lambda a: (-(a[1]-a[0]), -a[5]))[:5]
        for a in g: cand.append(a)
    # find intervals from different genes that tile contig with low overlap maybe
    cand=sorted(cand, key=lambda a:(a[0],a[1],a[3]))
    # flag if best intervals of different genes have non-overlap and total length meaningful
    # greedy: sort by score, remove overlaps >50% of shorter
    selected=[]
    for a in sorted(cand, key=lambda a: (-(a[1]-a[0]), a[0])):
        if all(min(a[1],b[1])-max(a[0],b[0]) <= 0.5*min(a[1]-a[0], b[1]-b[0]) or a[3]==b[3] for b in selected):
            selected.append(a)
    selected=sorted(selected, key=lambda a:a[0])
    if len(set(a[3] for a in selected))<2: continue
    span=sum(a[1]-a[0] for a in selected)
    if span<200: continue
    print('\\n', q, 'len', q.split('_length_')[1].split('_')[0], 'genes', [a[3] for a in selected], 'span', span)
    for a in selected[:20]:
        print(' ', a[0],a[1],a[2],a[3], 'len',a[1]-a[0], 'id',round(a[5]/a[6],3), a[4].split('|')[4])
PY
python3 work/contig_multigene.py > results/multigene_contigs.txt
head -200 results/multigene_contigs.txt"
