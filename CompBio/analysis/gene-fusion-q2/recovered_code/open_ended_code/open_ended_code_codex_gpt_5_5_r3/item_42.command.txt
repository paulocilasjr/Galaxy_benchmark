/bin/bash -lc "python3 - <<'PY'
seq=''
name=''
with open('work/rnaspades2/transcripts.fasta') as f:
    keep=False
    for line in f:
        if line.startswith('>'):
            keep=line.startswith('>NODE_63_')
            if keep: name=line.strip()
        elif keep:
            seq+=line.strip()
print(name, len(seq))
with open('work/node63.fa','w') as out:
    out.write(name+'\\n')
    for i in range(0,len(seq),60): out.write(seq[i:i+60]+'\\n')
PY
if [ "'! -f refs/gencode.v46.transcripts.fa ]; then gzip -dc refs/gencode.v46.transcripts.fa.gz > refs/gencode.v46.transcripts.fa; fi
makeblastdb -in refs/gencode.v46.transcripts.fa -dbtype nucl -out refs/gencode.v46.transcripts.blastdb
blastn -task blastn -word_size 7 -dust no -evalue 1e-5 -max_target_seqs 200 -max_hsps 20 -query work/node63.fa -db refs/gencode.v46.transcripts.blastdb -outfmt '"'6 qseqid qstart qend sseqid sstart send pident length mismatch gapopen bitscore evalue' > work/node63_blast.tsv
wc -l work/node63_blast.tsv
head -100 work/node63_blast.tsv"
