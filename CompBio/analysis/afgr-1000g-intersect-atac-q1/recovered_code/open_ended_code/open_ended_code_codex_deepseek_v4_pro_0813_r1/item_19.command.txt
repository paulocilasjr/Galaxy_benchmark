/bin/bash -lc "python3 - <<'PY'
import json,re,hashlib
# 1000G IDs
with open('3202_samples.txt') as f:
    lines=f.read().splitlines()
ids=set(line.split()[1] for line in lines[1:] if line.strip())
# AFGR ATAC experiment files
with open('afgr_atac_fields.json') as f:d=json.load(f)
rows=[]
for exp in d['@graph']:
    summary=exp.get('biosample_summary','')
    m=re.match(r'Homo sapiens (\\S+)', summary)
    sample=m.group(1) if m else ''
    for f in exp.get('files',[]):
        if f.get('file_format')=='bam' and f.get('output_type')=='alignments' and f.get('assembly')=='GRCh38':
            md5=f.get('md5sum')
            if md5:
                rows.append((sample, md5))
matched=sorted(set(md5 for sample,md5 in rows if sample in ids))
print('matched md5 count',len(matched))
out='filtered_bam_md5s.txt'
with open(out,'w') as f:
    for md5 in matched:
        f.write(md5+'\\n')
print('file bytes', open(out,'rb').read())
print('md5sum file', hashlib.md5(open(out,'rb').read()).hexdigest())
print('count',len(matched))
# check duplicates source
from collections import Counter
c=Counter(md5 for sample,md5 in rows if sample in ids)
print('dup md5s', [k for k,v in c.items() if v>1])
print('samples matched',len(set(s for s,m in rows if s in ids)))
PY
md5sum filtered_bam_md5s.txt
wc -l filtered_bam_md5s.txt"
