/bin/bash -lc "python3 - <<'PY'
import json,re
# load 1000g
with open('3202_samples.txt') as f:
    lines=f.read().splitlines()
header=lines[0]
ids=set(line.split()[1] for line in lines[1:] if line.strip())
print(len(ids), header)
# load afgr atac fields
with open('afgr_atac_fields.json') as f:d=json.load(f)
g=d['@graph']
rows=[]
for exp in g:
    summary=exp.get('biosample_summary','')
    m=re.match(r'Homo sapiens (\\S+)', summary)
    sample=m.group(1) if m else ''
    for f in exp.get('files',[]):
        if f.get('file_format')=='bam' and f.get('output_type')=='alignments':
            rows.append((sample, f['accession'], f.get('md5sum'), exp['accession']))
print('rows',len(rows),'unique sample',len(set(r[0] for r in rows)))
for r in rows:
    print(r[0], r[1], r[2], r[3], 'in1kg' if r[0] in ids else 'NOT')
# match
matched=[r for r in rows if r[0] in ids]
print('matched count',len(matched))
print('unique matched',len(set(r[0] for r in matched)))
print('not matched sample prefixes', sorted(set(r[0][:2] for r in rows if r[0] not in ids)))
PY"
