/bin/bash -lc "python3 - <<'PY'
import json
with open('afgr_atac_fields.json') as f:d=json.load(f)
g=d['@graph']
rows=[]
for exp in g:
    for f in exp.get('files',[]):
        if f.get('file_format')=='bam' and f.get('output_type')=='alignments':
            rows.append((exp['accession'], exp.get('biosample_summary'), f['accession'], f.get('md5sum'), f.get('donors'), f.get('assembly'), f.get('status'), f.get('title')))
print('filtered bams', len(rows))
for r in rows[:20]: print(r)
print('unique donors', len(set(tuple(r[4]) for r in rows)))
PY"
