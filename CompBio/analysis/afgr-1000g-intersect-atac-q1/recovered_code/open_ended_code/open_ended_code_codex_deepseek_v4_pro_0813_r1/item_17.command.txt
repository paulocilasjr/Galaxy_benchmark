/bin/bash -lc "python3 - <<'PY'
import json, urllib.request, time
with open('afgr_atac_fields.json') as f:d=json.load(f)
g=d['@graph']
donors={}
for exp in g:
    for f in exp.get('files',[]):
        if f.get('file_format')=='bam' and f.get('output_type')=='alignments':
            path=f['donors'][0]
            donors[path]=exp.get('biosample_summary')
print('unique',len(donors))
for i,(path,summary) in enumerate(donors.items()):
    url='https://www.encodeproject.org'+path+'?format=json'
    for attempt in range(3):
        try:
            with urllib.request.urlopen(url,timeout=30) as r:
                dj=json.load(r)
            aliases=dj.get('aliases',[])
            break
        except Exception as e:
            time.sleep(2)
            aliases=[]
    print(summary, path.split('/')[-1], aliases)
PY"
