/bin/bash -lc "curl -s 'https://www.encodeproject.org/search/?searchTerm=AFGR&type=Experiment&format=json&limit=all' -o afgr_experiments.json; python3 - <<'PY'
import json
with open('afgr_experiments.json') as f: d=json.load(f)
g=d['@graph']
print(len(g))
from collections import Counter
print(Counter(x.get('assay_title') for x in g))
for x in g:
    if x.get('assay_title')=='ATAC-seq':
        print(x['accession'], x.get('biosample_summary'), x.get('status'))
PY"
