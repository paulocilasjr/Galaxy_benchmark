/bin/bash -lc "python3 - <<'PY'
import json
rows=json.load(open('afgr_atac_bams.json'))
samples=set()
with open('1000g_3202_samples.tsv') as f:
    next(f)
    for line in f:
        parts=line.split()
        if len(parts)>=2:
            samples.add(parts[1])
print('1000g count', len(samples))
overlap=[r for r in rows if r['sample'] in samples]
print('overlap', len(overlap))
non=[r['sample'] for r in rows if r['sample'] not in samples]
print('non', len(non))
print('non sorted:', sorted(non))
print('overlap:')
for r in sorted(overlap, key=lambda x:x['sample']):
    print(r['sample'], r['md5sum'])
PY"
