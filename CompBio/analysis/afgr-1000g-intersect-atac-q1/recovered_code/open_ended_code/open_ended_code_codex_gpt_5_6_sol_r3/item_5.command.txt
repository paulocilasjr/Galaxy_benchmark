/bin/bash -lc "curl -L --fail --silent --show-error 'https://www.ebi.ac.uk/europepmc/webservices/rest/PMC10659267/fullTextXML' -o paper.xml
wc -c paper.xml
python3 - <<'PY'
import xml.etree.ElementTree as ET
root=ET.parse('paper.xml').getroot()
text=' '.join(root.itertext())
for phrase in ['Data availability','ENCODE','accession','Supplementary Table']:
 print('\\n',phrase)
 i=text.find(phrase)
 print(text[i:i+3000])
PY"
