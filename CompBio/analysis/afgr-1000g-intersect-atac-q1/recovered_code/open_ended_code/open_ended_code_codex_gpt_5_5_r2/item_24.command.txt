/bin/bash -lc "curl -L -s 'https://www.encodeproject.org/search/?type=File&searchTerm=AFGR&format=json&limit=all' > encode_afgr_files.json
jq '.total' encode_afgr_files.json
jq -r '.\"@graph\"[] | [.accession,.assay_term_name,.output_type,.file_format,.assembly,.dataset,.biological_replicates|tostring,.md5sum] | @tsv' encode_afgr_files.json | head -20
jq -r '.\"@graph\"[] | .dataset' encode_afgr_files.json | sort -u | head -20"
