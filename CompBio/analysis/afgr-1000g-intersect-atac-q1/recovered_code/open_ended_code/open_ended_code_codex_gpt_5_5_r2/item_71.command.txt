/bin/bash -lc "python3 - <<'PY'
import json,re
samples=set(x.strip() for x in open('igsr_30x_names.txt') if x.strip())
d=json.load(open('encode_mont_atac_exp_fields.json'))
rows=[]; non=[]; bad=[]
for exp in d['@graph']:
    disp=exp['biosample_summary'].split()[-1]
    candidates=[disp]
    if re.fullmatch(r'GM\\d+', disp): candidates.append('NA'+disp[2:])
    overlap=[c for c in candidates if c in samples]
    bams=[f for f in exp['files'] if f.get('file_format')=='bam' and f.get('output_type')=='alignments' and f.get('assembly')=='GRCh38' and f.get('status')=='released']
    if len(bams)"'!=1: bad.append((exp['"'accession'],disp,bams))
    if overlap:
        rows.append((overlap[0], disp, exp['accession'], bams[0]['accession'], bams[0]['md5sum']))
    else:
        non.append((disp, exp['accession']))
print('experiments', len(d['@graph']))
print('overlap', len(rows), 'nonoverlap', len(non), 'bad', len(bad))
print('non first', non[:30])
print('duplicate ids', len(rows), len(set(r[0] for r in rows)))
with open('afgr_1000g_overlap_bams.tsv','w') as out:
    out.write('kgp_id\\tdisplay_sample\\texperiment\\tbam_accession\\tmd5sum\\n')
    for r in sorted(rows): out.write('\\t'.join(r)+'\\n')
with open('afgr_1000g_filtered_bam_md5sums.txt','w') as out:
    for md5 in sorted(r[4] for r in rows): out.write(md5+'\\n')
PY
wc -l afgr_1000g_filtered_bam_md5sums.txt
md5sum afgr_1000g_filtered_bam_md5sums.txt
head -5 afgr_1000g_filtered_bam_md5sums.txt
tail -5 afgr_1000g_filtered_bam_md5sums.txt"
