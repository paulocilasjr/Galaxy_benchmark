import csv,re,json
# parse S3 2504
names=set()
for i,l in enumerate(open('dragen_igsr_30x.tsv')):
    if i==0: continue
    if l.strip(): names.add(l.split('\\t')[0])

    disp=exp['biosample_summary'].split()[-1]
    cand=[disp]

    if any(c in names for c in cand): rows.append(disp)
print(len(rows), rows[:10])
PY"
