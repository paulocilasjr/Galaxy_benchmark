/bin/bash -lc "curl -L -s --max-time 20 'https://1000genomes-dragen.s3.amazonaws.com/metadata/igsr-1000-genomes-30x-on-grch38.tsv' > dragen_igsr_30x.tsv || true
head -3 dragen_igsr_30x.tsv
wc -l dragen_igsr_30x.tsv
python3 - <<'PY'
import os
p='dragen_igsr_30x.tsv'
if os.path.getsize(p)>0:
    # detect first column sample names
    lines=open(p).read().splitlines()
    print('size', os.path.getsize(p), 'lines', len(lines), 'header', lines[0] if lines else '')
    # compare names if tab header available
    hdr=lines[0].split('\\t') if lines else []
    if 'Sample name' in hdr:
        idx=hdr.index('Sample name')
    elif 'Sample' in hdr:
        idx=hdr.index('Sample')
    else:
        idx=0
    names=set(l.split('\\t')[idx] for l in lines[1:] if l)
    ours=set(x.strip() for x in open('igsr_30x_names.txt') if x.strip())
    print(len(names), len(ours), len(names"'^ours))
PY'
