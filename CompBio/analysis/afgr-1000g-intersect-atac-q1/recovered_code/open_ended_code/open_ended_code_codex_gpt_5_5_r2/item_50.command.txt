/bin/bash -lc "curl -L -s -H 'Content-Type: application/json' -X POST 'https://www.internationalgenome.org/api/beta/data-collection/_search' -d '{\"size\":100,\"query\":{\"multi_match\":{\"query\":\"30x GRCh38\",\"fields\":[\"title.std\",\"shortTitle.std\",\"sequence.std\",\"alignment.std\",\"variants.std\"]}}}' > dc_search.json
jq -r '.hits.total, .hits.hits[] | [._id,._source.title,._source.shortTitle,(._source.samples.count|tostring)] | @tsv' dc_search.json"
