/bin/bash -lc "python - <<'PY'
import requests, json
base='https://lungcancer.chenlulab.com/api'
for params in [
    {'cell':'Basal','disease':'LUSC','coord':'umap','group':'cluster'},
    {'cell':'Basal','disease':'LUSC','coord':'umap','group':'stage'},
]:
    r=requests.get(base+'/coord', params=params, timeout=60)
    print('URL', r.url, 'status', r.status_code, 'len', len(r.content))
    data=r.json()
    print(data.keys())
    print('cell', data.get('cell'), 'disease', data.get('disease'), 'groups', data.get('group')[:20] if isinstance(data.get('group'), list) else data.get('group'))
    print('data keys', list(data.get('data',{}).keys())[:20])
    total=sum(len(v) for v in data.get('data',{}).values())
    print('total points', total)
    for k,v in list(data.get('data',{}).items())[:5]:
        print(' group',k,'n',len(v),'first',v[0] if v else None)
PY"
