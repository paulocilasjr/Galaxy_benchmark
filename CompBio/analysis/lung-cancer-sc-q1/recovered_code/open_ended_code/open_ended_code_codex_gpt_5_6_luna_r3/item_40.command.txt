/bin/bash -lc "python -u - <<'PY'
import urllib.request,json
base='http://lungcancer.chenlulab.com/api/'
for ep in ['cells','stats']:
 d=json.load(urllib.request.urlopen(base+ep))
 if ep=='cells':
  print('cells',d)
  print('total',sum(v for g in d.values() for c in v.values() for v in c.values()))
 else:
  print('stats keys',len(d.get('Value',{})),list(d.keys()), 'cells total',sum(sum(v.values()) for v in d['Value'].values()))
  print('meta? first',next(iter(d['Value'].items())))
PY"
