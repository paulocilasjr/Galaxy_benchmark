/bin/bash -lc "python -u - <<'PY'
import urllib.request
print('start',flush=True)
r=urllib.request.urlopen('https://lungcancer.chenlulab.com/api/download?name=meta.csv',timeout=30)
print('headers',r.headers,flush=True)
t=0
for i in range(10):
 b=r.read(1024*1024)
 print('chunk',i,len(b),flush=True)
 if not b:break
 t+=len(b)
print('total partial',t,flush=True)
PY"
