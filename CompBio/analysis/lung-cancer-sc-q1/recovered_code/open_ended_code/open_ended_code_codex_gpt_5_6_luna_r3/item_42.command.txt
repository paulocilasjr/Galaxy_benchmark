/bin/bash -lc "python -u - <<'PY'
import urllib.request
for name in ['meta.csv','raw_data.csv.gz','bulk_rna_seq.xlsx']:
 u='http://lungcancer.chenlulab.com/api/download/?name='+name
 try:
  r=urllib.request.urlopen(u,timeout=60); print(name,r.status,r.headers.get('Content-Length'),r.headers.get('Content-Type'),r.geturl())
  b=r.read(1000); print(b[:500])
 except Exception as e: print(name,e)
PY"
