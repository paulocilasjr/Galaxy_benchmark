/bin/bash -c "python - <<'PY'
import h5py, numpy as np, re
f1='lung.cancer.sc.h5ad'; f2='remote_adata.h5ad'
with h5py.File(f1,'r') as h:
    obs1=np.array([x.decode() for x in h['obs/_index'][:]])
with h5py.File(f2,'r') as h:
    obs2=np.array([x.decode() for x in h['obs/_index'][:]])
def barcode(x):
    # remove patient prefix? get last token after _, strip suffix
    parts=x.rsplit('_',1)
    seq=parts[-1]
    # strip -1/-0 or anything after last dash
    if '-' in seq:
        seq=seq.rsplit('-',1)[0]
    return seq
bc1=np.array([barcode(x) for x in obs1])
bc2=np.array([barcode(x) for x in obs2])
print(len(bc1), len(bc2), 'unique', len(set(bc1)), len(set(bc2)))
inter=set(bc1)&set(bc2)
print('intersection exact barcode sequences', len(inter))
# Check with prefixes? intersection sequences among remote datasets maybe if same 10x run but suffix differs.
for ds in ['Kim_Lee_2020','Lambrechts_Thienpont_2018_6149v1','Lambrechts_Thienpont_2018_6149v2','Maynard_Bivona_2020','Wu_Zhou_2021','Maier_Merad_2020','He_Fan_2021','Zilionis_Klein_2019','Laughney_Massague_2020']:
    with h5py.File(f2,'r') as h:
        cats=np.array([x.decode() for x in h['obs/dataset']['categories'][:]])
        codes=h['obs/dataset']['codes'][:]
        dsarr=cats[codes]
        mask=dsarr==ds
        obs_sub=obs2[mask]
        bc_sub=np.array([barcode(x) for x in obs_sub])
        inter=set(bc1)&set(bc_sub)
        print(ds, 'n', mask.sum(), 'inter', len(inter))
PY"
