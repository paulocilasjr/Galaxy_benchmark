/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
s=pd.read_parquet('scores.parquet')
# exclude zeros by filtering isin candidates? print all with mean EPCAM or LUSC high
cand=['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']
g=s.groupby('sampleID', sort=True).agg(n=('sampleID','size'), PTPRC=('PTPRC','mean'), EPCAM=('EPCAM','mean'),
 LUSC=('LUSC_basal','mean'), LUAD=('LUAD_AT2','mean'), Basal=('Basal','mean'), AT2=('AT2','mean'),
 frac_epi_gt0=('EPCAM',lambda x:(x>0.5).mean()),
 frac_lusc=('LUSC_basal',lambda x:(x>1).mean()), frac_luad=('LUAD_AT2',lambda x:(x>1).mean()),
 frac_dc=('DC',lambda x:(x>0.5).mean()), DC=('DC','mean'), KRT5=('KRT5','mean'), SOX2=('SOX2','mean'))
print(g.to_string())
print('\\nCandidates:')
print(g.loc[cand].to_string())
# sort by lusc fraction descending
print(g.sort_values('frac_lusc',ascending=False).head(30).to_string())
print('\\nall with frac_lusc>0.05')
print(g[g.frac_lusc>0.05].to_string())
PY"
