/bin/bash -c "python - <<'PY'
import h5py
f='lung.cancer.sc.h5ad'
with h5py.File(f,'r') as h:
    def show_attrs(obj):
        if obj.attrs:
            print('ATTRS', obj.name, dict(obj.attrs))
    h.visititems(lambda name,obj: show_attrs(obj))
    # first genes
    print('first 20 symbols', [x.decode() for x in h['var/symbol'][:20]])
    print('last 20 symbols', [x.decode() for x in h['var/symbol'][-20:]])
PY"
