/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
s=pd.read_parquet('scores.parquet')
cand=['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']
sub=s[s.sampleID.isin(cand)]
print(sub.groupby('sampleID').size())
cols=['LUSC_basal','LUAD_AT2','Basal','AT2','KRT5','KRT14','TP63','SOX2','NKX2-1','NAPSA','SFTPB','SFTPC','EPCAM','PTPRC','DC','cDC1','cDC2','pDC','mregDC','FCER1A','CLEC9A','CD1C','LILRA4','LAMP3','CCR7']
for sample, df in sub.groupby('sampleID', sort=True):
    print('\\n',sample, 'n',len(df))
    print(df[cols].mean().round(3).to_string())
    # fraction > thresholds
    print('frac PTPRC>0.5', (df.PTPRC>0.5).mean().round(3), 'frac EPCAM>0.5', (df.EPCAM>0.5).mean().round(3),
          'frac LUSC_basal>1', (df.LUSC_basal>1).mean().round(3), 'frac LUAD_AT2>1', (df.LUAD_AT2>1).mean().round(3))
    # candidate DC threshold
    for th in [0.3,0.5,0.7,1.0]:
        print(' frac DC >',th,(df.DC>th).mean().round(4), end='; ')
    print()
PY"
