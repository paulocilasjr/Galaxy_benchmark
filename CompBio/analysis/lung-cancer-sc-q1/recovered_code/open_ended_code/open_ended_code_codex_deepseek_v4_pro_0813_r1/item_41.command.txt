/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
s=pd.read_parquet('scores.parquet')
labels=pd.read_csv('cell_kmeans_labels.csv',index_col=0)
# ensure same order
s=s.join(labels[['kmeans']])
# map clusters to broad types based on summary
broad_map={0:'Epithelial',1:'T',2:'B',3:'DC',4:'Endothelial',5:'LUSC_basal',6:'T',7:'T',8:'Myeloid',9:'Fibroblast',10:'Myeloid',11:'NK',12:'T',13:'Epithelial',14:'pDC',15:'LUSC_basal',16:'T',17:'T',18:'T',19:'Myeloid',20:'T',21:'Epithelial',22:'Myeloid',23:'LUSC_basal',24:'Epithelial',25:'Plasma',26:'Myeloid',27:'T',28:'Mast',29:'DC'}
s['broad']=s['kmeans'].map(broad_map)
# candidate composition
cand=['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']
for p in cand:
    sub=s[s.sampleID==p]
    print('\\n',p,'n',len(sub))
    print(sub['broad'].value_counts().to_string())
    print(sub['kmeans'].value_counts().head(10).to_string())
# per-sample broad counts matrix
ct=pd.crosstab(s.sampleID,s.broad)
ct.to_csv('sample_broad_counts.csv')
# print candidates and some LUSC samples
print('\\nCandidate broad fractions:')
print(ct.loc[cand].div(ct.loc[cand].sum(axis=1),axis=0).round(3).to_string())
# Print all samples with LUSC_basal fraction
lusc_frac=(s.broad=='LUSC_basal').groupby(s.sampleID).mean()
print('\\nTop LUSC_basal samples:')
print(lusc_frac.sort_values(ascending=False).head(30).to_string())
# DC fractions by sample (cluster 3 + 29 maybe + 14 pDC)
s['is_dc']=s['broad'].isin(['DC','pDC'])
dc_frac=s.groupby('sampleID')['is_dc'].mean()
print('\\nDC fractions candidates:')
print(dc_frac.loc[cand].to_string())
print('\\nTop DC samples all:')
print(dc_frac.sort_values(ascending=False).head(20).to_string())
PY"
