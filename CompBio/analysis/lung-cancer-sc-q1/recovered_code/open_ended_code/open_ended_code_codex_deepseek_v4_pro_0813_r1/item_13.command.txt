/bin/bash -c "python - <<'PY'
import requests
url='https://huggingface.co/scvi-tools/non-small-cell-lung-cancer/resolve/main/adata.h5ad'
with requests.get(url, stream=True, timeout=30) as r:
    r.raise_for_status()
    total=int(r.headers.get('content-length',0))
    print('total',total)
    n=0
    with open('remote_adata.h5ad','wb') as f:
        for chunk in r.iter_content(chunk_size=1<<20):
            f.write(chunk); n+=len(chunk)
            if n % (50<<20) < (1<<20): print('downloaded MB', n//(1<<20))
    print('done',n)
PY"
