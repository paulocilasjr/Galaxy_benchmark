/bin/bash -c "python - <<'PY'
import h5py, numpy as np, scipy.sparse as sp, time
f='lung.cancer.sc.h5ad'
with h5py.File(f,'r') as h:
    data=h['X/data'][:]; indices=h['X/indices'][:]; indptr=h['X/indptr'][:]; shape=tuple(h['X'].attrs['shape'])
    symbols=np.array([x.decode() for x in h['var/symbol'][:]])
    sample_codes=h['obs/sampleID/codes'][:]
    sample_cats=np.array([x.decode() for x in h['obs/sampleID/categories'][:]])
X=sp.csr_matrix((data,indices,indptr),shape=shape)
print('X',X.shape,'nnz',X.nnz)
counts=np.asarray(X.sum(axis=1)).ravel()
print('total counts per cell: min/median/max', np.min(counts), np.median(counts), np.max(counts), 'mean', counts.mean())
genes=(X>0).sum(axis=1).A1
print('genes per cell min/median/max', genes.min(), np.median(genes), genes.max())
markers=['PTPRC','EPCAM','KRT5','KRT14','TP63','SOX2','NKX2-1','NAPSA','SFTPB','SFTPC','KRT7','CD3D','CD3E','CD4','CD8A','MS4A1','CD79A','CD14','FCGR3A','LYZ','CSF1R','CD68','CD1C','CLEC9A','LILRA4','FCER1A','KIT','TPSAB1','CPA3','COL1A1','DCN','PECAM1','VWF','CLDN5','MKI67','MUC1','AGER','SCGB1A1','FOXJ1']
for m in markers:
    idx=np.where(symbols==m)[0]
    if len(idx):
        expr=X[:,idx[0]].toarray().ravel()
        print(m, 'mean', expr.mean(), 'frac>0', (expr>0).mean(), 'max', expr.max())
    else:
        print(m, 'missing')
PY"
