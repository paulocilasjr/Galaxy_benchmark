/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
s=pd.read_parquet('scores.parquet')
cand=['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040']
sub=s[s.sampleID.isin(cand)]
cols=['LUSC_basal','LUAD_AT2','Basal','AT2','KRT5','KRT14','TP63','SOX2','NKX2-1','NAPSA','SFTPB','SFTPC','EPCAM','PTPRC','DC','cDC1','cDC2','pDC','mregDC','FCER1A','CLEC9A','CD1C','LILRA4']
for sample, df in sorted(sub.groupby('sampleID')):
    print('\\n',sample, 'n',len(df))
    print(df[cols].mean().round(3).to_string())
    print('frac PTPRC>0.5', round((df.PTPRC>0.5).mean(),3), 'frac EPCAM>0.5', round((df.EPCAM>0.5).mean(),3),
          'frac LUSC_basal>1', round((df.LUSC_basal>1).mean(),3), 'frac LUAD_AT2>1', round((df.LUAD_AT2>1).mean(),3))
    for th in [0.2,0.3,0.5,0.7,1.0]:
        print(' frac DC >',th,round((df.DC>th).mean(),4), end='; ')
    print()
PY"
