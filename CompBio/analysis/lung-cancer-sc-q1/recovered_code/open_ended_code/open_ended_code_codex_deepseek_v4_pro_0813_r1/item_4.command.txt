/bin/bash -c "python - <<'PY'
import h5py, numpy as np
f='lung.cancer.sc.h5ad'
with h5py.File(f,'r') as h:
    cats=h['obs/sampleID/categories'][:]
    print(len(cats), cats.tolist())
    # counts per sample using codes
    codes=h['obs/sampleID/codes'][:]
    uniq, counts=np.unique(codes, return_counts=True)
    for c in uniq:
        print(cats[c].decode(), counts[c])
PY"
