/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
s=pd.read_parquet('scores.parquet'); labels=pd.read_csv('cell_kmeans_labels.csv',index_col=0)
s=s.join(labels[['kmeans']])
broad_map={0:'Epithelial',1:'T',2:'B',3:'DC',4:'Endothelial',5:'LUSC_basal',6:'T',7:'T',8:'Myeloid',9:'Fibroblast',10:'Myeloid',11:'NK',12:'T',13:'Epithelial',14:'pDC',15:'LUSC_basal',16:'T',17:'T',18:'T',19:'Myeloid',20:'T',21:'Epithelial',22:'Myeloid',23:'LUSC_basal',24:'Epithelial',25:'Plasma',26:'Myeloid',27:'T',28:'Mast',29:'DC'}
s['broad']=s['kmeans'].map(broad_map)
g=s.groupby('sampleID').agg(n=('sampleID','size'), lusc_frac=('broad',lambda x:(x=='LUSC_basal').mean()), epi_frac=('broad',lambda x:(x=='Epithelial').mean()), lusc_mean=('LUSC_basal','mean'), luad_mean=('LUAD_AT2','mean'), krt5=('KRT5','mean'), nkx=('NKX2-1','mean'), dc_frac=('broad',lambda x:(x.isin(['DC','pDC'])).mean()))
# sort by lusc_frac and print all with lusc_frac>0.02 or epi_frac>0.2
print(g[g.lusc_frac>0.02].to_string())
print('\\nPotential LUAD (epi high):')
print(g[g.epi_frac>0.2].to_string())
# Save condition guesses
PY"
