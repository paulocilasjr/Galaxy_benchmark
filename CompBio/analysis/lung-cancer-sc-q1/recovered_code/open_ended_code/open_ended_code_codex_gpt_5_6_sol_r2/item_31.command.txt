/bin/bash -lc "python - <<'PY'
import requests
bases=['https://www.ncbi.nlm.nih.gov/pmc/articles/PMC8758688/bin/','https://pmc.ncbi.nlm.nih.gov/articles/instance/8758688/bin/','https://www.ebi.ac.uk/europepmc/articles/PMC8758688/bin/']
files=['41392_2021_824_MOESM2_ESM.xlsx','41392_2021_824_MOESM1_ESM.docx','41392_2021_824_Fig2_HTML.jpg']
for fn in files:
 for b in bases:
  r=requests.get(b+fn,timeout=60)
  print(b+fn,r.status_code,len(r.content),r.headers.get('content-type'))
  if r.status_code==200 and len(r.content)>10000:
   open(fn,'wb').write(r.content); break
PY
ls -lh 41392*"
