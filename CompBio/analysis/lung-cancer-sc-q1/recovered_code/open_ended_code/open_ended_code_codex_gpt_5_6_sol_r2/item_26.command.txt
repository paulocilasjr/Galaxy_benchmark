/bin/bash -lc "python - <<'PY'
import requests
u='https://lungcancer.chenlulab.com/api/download?name=meta.csv'

r.raise_for_status()
open('source_meta.csv','wb').write(r.content)
print(len(r.content))
PY
ls -lh source_meta.csv
python - <<'PY'
import pandas as pd
m=pd.read_csv('source_meta.csv')
print(m.shape)
print(m.columns.tolist())
for c in m.columns:
 print('\\n',c,m[c].nunique(dropna=False),m[c].value_counts(dropna=False).head(30).to_dict())
PY"
