f=h5py.File('lung.cancer.sc.h5ad','r')

f.visititems(show)

for k in f['obs/sampleID'].keys():

    print(k, d.shape, d.dtype, d[:20])

idx=f['obs/_index'][:50]
print([x.decode() if isinstance(x,bytes) else str(x) for x in idx])
print('\\nlast 20 obs index')
idx=f['obs/_index'][-20:]
print([x.decode() if isinstance(x,bytes) else str(x) for x in idx])
print('\\nvar firsts')
for k in f['var'].keys():
    obj=f['var'][k]
    if isinstance(obj,h5py.Dataset):
        print(k, obj.shape, obj.dtype, obj[:10])
    else:
        print(k, 'group', list(obj.keys()), dict(obj.attrs))
