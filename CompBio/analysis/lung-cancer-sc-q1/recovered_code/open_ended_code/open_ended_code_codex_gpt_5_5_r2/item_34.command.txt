/bin/bash -lc "python - <<'PY'
import urllib.request
url='https://lungcancer.chenlulab.com/api/download/?name=meta.csv'
out='portal_meta.csv'
with urllib.request.urlopen(url, timeout=120) as r, open(out,'wb') as f:
    total=0
    while True:
        b=r.read(1024*1024)
        if not b: break
        f.write(b); total+=len(b)
print(out,total)
print(open(out,'rb').read(500))
PY
ls -lh portal_meta.csv"
