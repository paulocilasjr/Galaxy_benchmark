/bin/bash -lc "python -u - <<'PY'
import csv
from collections import OrderedDict
ss=OrderedDict()
with open('original_meta.csv',newline='') as f:
 for r in csv.DictReader(f):
  if r['Tissue']=='Tumor' and r['Disease'] in ('LUAD','LUSC'):
   ss.setdefault(r['SampleID'],[r['PatientID'],r['Disease'],0]); ss[r['SampleID']][2]+=1
print(len(ss),sum(v[2] for v in ss.values()))
for i,(k,v) in enumerate(ss.items(),1):print(f'Patient_{i:03d}',k,*v)
PY"
