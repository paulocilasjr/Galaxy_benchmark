/bin/bash -lc "python -u - <<'PY'
import h5py, numpy as np
f=h5py.File('lung.cancer.sc.h5ad','r')
obs=f['obs']['sampleID']
cats=[x.decode() if isinstance(x,bytes) else str(x) for x in obs['categories'][:]]
codes=obs['codes'][:]
print('categories',cats)
u,n=np.unique(codes,return_counts=True)
print('\\n'.join(f'{cats[i]}\\t{c}' for i,c in zip(u,n)))
x=f['X']
for k in x: print('X',k,x[k].shape,x[k].dtype)
for k in f['var']:
 d=f['var'][k]; print('var',k,d.shape,d.dtype)
sy=f['var']['symbol'][:20]
print('symbols', [s.decode() if isinstance(s,bytes) else str(s) for s in sy])
PY"
