/bin/bash -lc "python -u - <<'PY'
import h5py,fsspec
u='https://datasets.cellxgene.cziscience.com/33165751-ae33-4a65-94ee-52d9bc38f97e.h5ad'
with fsspec.open(u,block_size=2**20,cache_type='blockcache') as h:
 with h5py.File(h,'r') as f:
  for c in ['sample','disease','dataset','_index']:
   g=f['obs'][c];print(c,type(g),g.shape if isinstance(g,h5py.Dataset) else list(g),flush=True)
   if isinstance(g,h5py.Group):print(g['categories'][:],flush=True)
   else:print(g[:5],flush=True)
PY"
