/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, scipy.sparse as sp, pandas as pd
p='lung.cancer.sc.h5ad'
with h5py.File(p,'r') as f:
    syms=np.array([x.decode() if isinstance(x,bytes) else x for x in f['var/symbol'][:]])
    cats=np.array([x.decode() if isinstance(x,bytes) else x for x in f['obs/sampleID/categories'][:]])
    codes=f['obs/sampleID/codes'][:]
    sample=cats[codes]
    data=f['X/data'][:]; indices=f['X/indices'][:]; indptr=f['X/indptr'][:]; shape=tuple(f['X'].attrs['shape'])
X=sp.csr_matrix((data, indices, indptr), shape=shape)
lib=np.asarray(X.sum(axis=1)).ravel().astype(float)
g2i={g:i for i,g in enumerate(syms)}
panels={
 'immune':['PTPRC','CD3D','CD3E','CD2','NKG7','MS4A1','CD79A','LYZ','HLA-DRA','CD74'],
 't':['CD3D','CD3E','CD2','TRAC'],
 'nk':['NKG7','GNLY','PRF1','KLRD1'],
 'b':['MS4A1','CD79A','CD79B','BANK1'],
 'myeloid':['LYZ','S100A8','S100A9','FCGR3A','CD68','CSF1R','LST1'],
 'dc':['FCER1A','CLEC10A','CD1C','CD1E','CLEC9A','XCR1','LILRA4','IL3RA','LAMP3','CCR7','FLT3','ITGAX'],
 'epithelial':['EPCAM','KRT8','KRT18','KRT19','KRT7','MUC1'],
 'basal':['KRT5','KRT6A','KRT6B','KRT14','KRT17','TP63','DSG3','SERPINB5','S100A2'],
 'adeno':['SFTPA1','SFTPA2','SFTPB','SFTPC','NAPSA','NKX2-1','SLC34A2','MUC1','AGER','SCGB3A1'],
 'endo':['PECAM1','VWF','CLDN5','ENG','KDR'],
 'fibro':['COL1A1','COL1A2','DCN','LUM','COL3A1','PDGFRA'],
 'mast':['TPSAB1','TPSB2','CPA3','KIT'],
}
allgenes=sorted({g for gs in panels.values() for g in gs if g in g2i})
cols=[g2i[g] for g in allgenes]
M=X[:,cols].toarray().astype(float)
N=np.log1p(M/lib[:,None]*1e4)
gpos={g:i for i,g in enumerate(allgenes)}
def score(gs):
    ids=[gpos[g] for g in gs if g in gpos]
    return N[:,ids].mean(axis=1) if ids else np.zeros(N.shape[0])
S={k:score(v) for k,v in panels.items()}
comp_names=['immune','epithelial','endo','fibro','mast']
comp_mat=np.vstack([S[k] for k in comp_names]).T
comp_idx=comp_mat.argmax(axis=1)
comp=np.array(comp_names, dtype=object)[comp_idx]
comp[comp_mat.max(axis=1)<0.15]='unknown'
imm_names=['t','nk','b','myeloid','dc','mast']
imm_mat=np.vstack([S[k] for k in imm_names]).T
imm_sub=np.array(imm_names, dtype=object)[imm_mat.argmax(axis=1)]
rows=[]
for sid in cats:
    m=sample==sid
    epi=m & (comp=='epithelial')
    rows.append({
        'sample':sid,'n':int(m.sum()),'immune_prop':float((m & (comp=='immune')).sum()/m.sum()),
        'epi_n':int(epi.sum()),'basal_mean_epi':float(S['basal'][epi].mean()) if epi.sum() else np.nan,
        'adeno_mean_epi':float(S['adeno'][epi].mean()) if epi.sum() else np.nan,
        'basal_cells_epi_prop':float((epi & (S['basal']>S['adeno']) & (S['basal']>0.3)).sum()/epi.sum()) if epi.sum() else np.nan,
        'adeno_cells_epi_prop':float((epi & (S['adeno']>=S['basal']) & (S['adeno']>0.3)).sum()/epi.sum()) if epi.sum() else np.nan,
        'dc_prop_all':float((m & (comp=='immune') & (imm_sub=='dc') & (S['dc']>0.2)).sum()/m.sum()),
        'dc_n':int((m & (comp=='immune') & (imm_sub=='dc') & (S['dc']>0.2)).sum()),
    })
df=pd.DataFrame(rows)
print('query samples')
print(df[df['sample'].isin(['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040'])].to_string(index=False))
print('\\ntop basal samples')
print(df.sort_values('basal_mean_epi', ascending=False).head(25).to_string(index=False))
print('\\ntop adeno samples')
print(df.sort_values('adeno_mean_epi', ascending=False).head(25).to_string(index=False))
print('\\ndc top query only sorted')
print(df[df['sample'].isin(['Patient_005','Patient_006','Patient_007','Patient_018','Patient_040'])].sort_values('dc_prop_all',ascending=False)[['sample','n','epi_n','basal_mean_epi','adeno_mean_epi','dc_n','dc_prop_all']].to_string(index=False))
lusc=(df['epi_n']>=20) & (df['basal_mean_epi']>df['adeno_mean_epi'])
print('\\nLUSC inferred', df.loc[lusc,'sample'].tolist(), 'count', int(lusc.sum()))
lusc_samples=set(df.loc[lusc,'sample'])
basal_malignant=(np.isin(sample, list(lusc_samples)) & (comp=='epithelial') & (S['basal']>S['adeno']) & (S['basal']>0.3))
nonimmune=(comp"'!='"'immune') & (comp"'!='"'unknown')
print('nonimmune', int(nonimmune.sum()), 'basal_lusc', int(basal_malignant.sum()), 'prop', float(basal_malignant.sum()/nonimmune.sum()))
print('rounded nearest 20 pct', round((basal_malignant.sum()/nonimmune.sum())*100/20)*20)
nonimm2=(comp"'!='"'immune')
print('nonimmune incl unknown', int(nonimm2.sum()), 'prop', float(basal_malignant.sum()/nonimm2.sum()), 'round', round((basal_malignant.sum()/nonimm2.sum())*100/20)*20)
PY"
