/bin/bash -lc "python - <<'PY'
import h5py
with h5py.File('lung.cancer.sc.h5ad','r') as f:
    def show_attrs(name,obj):
        if obj.attrs:
            print('\\nATTR', name or '/')
            for k,v in obj.attrs.items():
                print(' ', k, repr(v)[:500])
    f.visititems(show_attrs)
    print('\\nTOP attrs')
    for k,v in f.attrs.items(): print(k,repr(v)[:500])
PY"
