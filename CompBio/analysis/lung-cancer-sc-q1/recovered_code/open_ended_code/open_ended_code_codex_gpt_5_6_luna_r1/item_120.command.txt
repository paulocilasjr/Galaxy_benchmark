/bin/bash -lc "python - <<'PY'
import h5py,numpy as np,collections
p='lung.cancer.sc.h5ad'
with h5py.File(p,'r') as f:
 g=f['obs']['sampleID']; cats=[x.decode() if isinstance(x,bytes) else str(x) for x in g['categories'][:]]; scode=g['codes'][:]
 indptr=f['X']['indptr'][:]; data=f['X']['data']
 ncell=len(scode)
 nnz=np.diff(indptr).astype(np.int32)
 umi=np.empty(ncell,dtype=np.int64)
 # chunk loop for data sums
 for st in range(0,ncell,20000):
  en=min(ncell,st+20000); vals=[]
  for i in range(st,en): vals.append(data[indptr[i]:indptr[i+1]].sum())
  umi[st:en]=vals
 for s in range(len(cats)):
  ix=np.where(scode==s)[0]
  print(cats[s],len(ix), 'genes_med',float(np.median(nnz[ix])),'umi_med',float(np.median(umi[ix])),'genes_mean',round(nnz[ix].mean(),1),'umi_mean',round(umi[ix].mean(),1))
PY"
