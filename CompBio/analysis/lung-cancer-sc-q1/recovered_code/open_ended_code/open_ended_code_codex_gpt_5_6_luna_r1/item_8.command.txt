/bin/bash -lc "python -u - <<'PY'
import h5py
p='lung.cancer.sc.h5ad'
with h5py.File(p,'r') as f:
    def walk(name,obj):
        if isinstance(obj,h5py.Dataset):
            print(name, obj.shape, obj.dtype, obj.compression)
        else:
            print(name, '<group>')
    f.visititems(walk)
PY"
