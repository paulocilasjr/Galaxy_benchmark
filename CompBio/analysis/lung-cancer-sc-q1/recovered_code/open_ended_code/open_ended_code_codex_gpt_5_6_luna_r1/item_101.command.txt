import requests
for u in ['https://doi.org/10.57760/sciencedb.02028','https://www.scidb.cn/en/detail?dataSetId=','https://www.scidb.cn']:
 try:
  r=requests.get(u,allow_redirects=True,timeout=30)
  print(u,r.status_code,r.url,len(r.content),r.headers.get('content-type'))
 except Exception as e: print(e)
PY"
