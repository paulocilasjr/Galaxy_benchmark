import anndata as ad, numpy as np, scipy.sparse as sp
# load full atlas raw counts 6000
p='luca_full/data/20_build_atlas/annotate_datasets/35_final_atlas/full_atlas_hvg_integrated_scvi_integrated_scanvi.h5ad'
ref=ad.read_h5ad(p, backed='r')
# our var gene names
our=ad.read_h5ad('lung.cancer.sc.h5ad', backed='r')
our_genes=our.var['symbol'].astype(str).to_numpy()
ref_genes=ref.var.index.to_numpy()
# intersection positions
ref_pos={g:i for i,g in enumerate(ref_genes)}
our_pos=[ref_pos[g] for g in our_genes if g in ref_pos]
our_idx=np.array([i for i,g in enumerate(our_genes) if g in ref_pos])
ref_idx=np.array([ref_pos[g] for g in our_genes if g in ref_pos])
print('intersection',len(our_pos), 'our genes',len(our_genes), 'ref genes',len(ref_genes))
# load small sample our rows for these genes, raw? our X
start=0; end=5
Xq=our.X[start:end, our_idx]
if sp.issparse(Xq): Xq=Xq.toarray()
print('query shape',Xq.shape)
# brute nearest over ref chunks with euclidean on normalized? raw counts may large.
# normalize each row to unit norm (cosine equivalent if both normalized)
def norm_rows(X):
    X=X.toarray() if sp.issparse(X) else X
    n=np.linalg.norm(X,axis=1,keepdims=True); n[n==0]=1
    return X/n
Xq=norm_rows(Xq)
best_dist=np.full(len(Xq),np.inf); best_idx=np.full(len(Xq),-1,dtype=int)
chunk=20000
nref=ref.n_obs
for s in range(0,nref,chunk):

    # brute dot for all pairs
    sim=Xq @ Xr.T  # (q, chunk) cosine similarity
    dist=1-sim
    # track min
    minidx=np.argmin(dist,axis=1)+s
    minval=np.min(dist,axis=1)
    upd=minval<best_dist
    best_dist[upd]=minval[upd]; best_idx[upd]=minidx[upd]
    if s%50000==0: print(s)
