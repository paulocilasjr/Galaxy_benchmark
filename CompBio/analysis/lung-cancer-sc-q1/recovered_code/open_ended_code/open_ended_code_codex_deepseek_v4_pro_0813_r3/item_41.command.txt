/bin/bash -c "python - <<'PY'
import anndata as ad, pandas as pd, numpy as np
# load our obs
ours=ad.read_h5ad('lung.cancer.sc.h5ad', backed='r').obs
# derive barcode after last underscore? index format Patient_XXX_AAAC...-1
our_idx=pd.Series(ours.index.values)
print('our index head')
print(our_idx.head(10).to_string())
# suffix after first underscore? barcode is after Patient_xxx_
barcode=our_idx.str.extract(r'"'^Patient_'"\\d+_(.+)"'$'"')[0]
print('barcodes head')
print(barcode.head())
# load luca obs
luca=ad.read_h5ad('luca_scanvi_adata.h5ad', backed='r').obs
# LUCA index has barcode at end after first underscore? example 001C_AAAC...-0
luca_barcode=pd.Series(luca.index.values).str.replace(r'"'^[''^_]+_'"','',regex=True)
print('luca barcode head')
print(luca_barcode.head())
# test matches for first barcodes
for bc in barcode.head(20):
    hits=luca_barcode==bc
    print(bc, hits.sum(), luca.index[hits.values].tolist()[:3])
# overall unique matches
common=np.intersect1d(barcode.values, luca_barcode.values)
print('common unique',len(common),'of ours',barcode.nunique())
PY"
