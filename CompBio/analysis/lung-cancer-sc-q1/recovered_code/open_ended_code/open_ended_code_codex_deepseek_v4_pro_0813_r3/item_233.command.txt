/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
cs=pd.read_pickle('cell_scores.pkl')
sub=pd.read_csv('sample_subtype.csv').set_index('sample')
cs['subtype']=cs.sampleID.map(sub['subtype'])
immune_cols=['Tcell','NK','Bcell','myeloid','Mast','cDC1','cDC2','pDC','DC_mature','neutrophil'] # maybe DC_general too
nonimm_cols=['epithelial','endothelial','fibroblast','smooth_muscle']
# definitions
defs={}
defs['immune_CD45==0']=cs.immune_CD45==0
defs['immune_CD45<0.1']=cs.immune_CD45<0.1
defs['immune_CD45<=0']=cs.immune_CD45<=0
defs['max_immune_specific==0']=cs[immune_cols].max(axis=1)==0
defs['max_immune_all==0']=cs[immune_cols+['immune_CD45']].max(axis=1)==0
defs['max_nonimm>0']=cs[nonimm_cols].max(axis=1)>0
defs['max_nonimm>max_immune']=cs[nonimm_cols].max(axis=1)>=cs[immune_cols+['immune_CD45']].max(axis=1)
# LUSC_basal definitions
luscb0=cs.LUSC_basal>0
luscb005=cs.LUSC_basal>0.05
luscb01=cs.LUSC_basal>0.1
luscbmax=(cs.LUSC_basal>cs.LUAD)
# numerator only in LUSC subtype and nonimmune
for name,den in defs.items():
    for lname,lnum in [('>0',luscb0),('>0.05',luscb005),('>0.1',luscb01),('>LUAD',luscbmax)]:
        num=lnum & cs.subtype.eq('LUSC') & den
        if den.sum()>0:
            pct=num.sum()/den.sum()*100
            # nearest multiple of 20
            near=round(pct/20)*20
            print(f'{name:28s} {lname:8s} den={den.sum():6d} num={num.sum():6d} pct={pct:6.2f} near20={near}')
    print()
PY"
