/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np, pandas as pd
a=ad.read_h5ad('lung.cancer.sc.h5ad', backed='r')
varsym=a.var['symbol'].astype(str).str.upper().to_numpy()
# selected markers
markers={
'LUAD':['NKX2-1','NAPSA','KRT7','MUC1','TMPRSS2','SFTPA1','SFTPB','SFTPC','SFTPD','AGR2','CLDN5'],
'LUSC':['KRT5','KRT6A','KRT6B','KRT14','TP63','SOX2','DSG3','SERPINB3','SERPINB4','PKP1','DSC3']
}
for key,gs in markers.items():
    present=[]
    for g in gs:
        matches=np.where(varsym==g.upper())[0]
        if len(matches):
            present.append((g,int(matches[0])))
        else: print('missing',g)
    print(key,present)
PY"
