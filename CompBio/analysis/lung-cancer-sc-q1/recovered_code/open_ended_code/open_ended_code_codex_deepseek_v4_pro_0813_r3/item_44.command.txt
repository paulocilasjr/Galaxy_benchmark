/bin/bash -c "python - <<'PY'
import anndata as ad, pandas as pd
a=ad.read_h5ad('luca_scanvi_adata.h5ad', backed='r')
obs=a.obs
for col in ['cell_type','cell_type_coarse','cell_type_major','cell_type_tumor','cell_type_predicted','condition']:
    print('\\n',col)
    print(obs[col].value_counts(dropna=False).to_string())
print('\\ncell types with \"basal\"')
print(obs['cell_type'].astype(str).str.contains('basal',case=False).value_counts())
for x in obs['cell_type'].cat.categories:
    if 'basal' in x.lower() or 'tumor' in x.lower(): print(x)
print('\\nmalignant basal rows sample')
mb=obs['cell_type'].astype(str).str.lower().str.contains('malignant basal',case=False)
print(mb.sum())
print(obs.loc[mb,['cell_type','condition','sample','patient','study']].head(20).to_string())
# cross condition/cell type major tumor cells
print('\\nTumor cells by condition')
print(pd.crosstab(obs['cell_type_major'],obs['condition']).to_string())
PY"
