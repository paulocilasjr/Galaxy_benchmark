import anndata as ad, numpy as np, scipy.sparse as sp, pandas as pd
our=ad.read_h5ad('lung.cancer.sc.h5ad',backed='r')
ref=ad.read_h5ad('luca_full/data/20_build_atlas/annotate_datasets/35_final_atlas/full_atlas_hvg_integrated_scvi_integrated_scanvi.h5ad',backed='r')
def barcode(x):
    y=x.rsplit('_',1)[-1]
    if '-' in y: y=y.rsplit('-',1)[0]
    return y
# create mapping barcode -> ref positions (maybe many duplicates). Use pandas
ref_idx=np.arange(ref.n_obs)
ref_bc=np.array([barcode(x) for x in ref.obs.index])
# For first 10 our barcodes, find matches
for oi in range(10):
    ob=our.obs.index[oi]
    obc=barcode(ob)
    matches=np.where(ref_bc==obc)[0]
    # compute total counts for each candidate
    if len(matches):
        vals=[]
        for mi in matches[:10]:
            total=ref.layers['raw_counts'][mi].sum()
            vals.append((ref.obs.index[mi], int(total), ref.obs['cell_type_tumor'][mi], ref.obs['sample'][mi]))
        print(ob, 'our total', our.X[oi].sum(), 'matches', len(matches), vals[:5])
    else:
        print(ob,'no match')
