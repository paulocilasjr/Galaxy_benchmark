/bin/bash -lc "python - <<'PY'
import requests
url='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2'
r=requests.get(url,timeout=30,headers={'User-Agent':'Mozilla/5.0'})
print(r.status_code, len(r.text))
# find supplementary links
import re
for m in re.finditer(r'href=\"(["'^"]*(?:supp|Supp|media|full)[''^"]*)"'"', r.text):
    print(m.group(1))
# save text for searching
open('biorxiv_page.html','w').write(r.text)
print('saved')
PY"
