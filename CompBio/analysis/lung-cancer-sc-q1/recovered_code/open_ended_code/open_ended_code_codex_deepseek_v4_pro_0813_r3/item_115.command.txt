/bin/bash -c "python - <<'PY'
import pandas as pd, numpy as np
res=pd.read_csv('centroid_pred_celltype_tumor.csv')
immune_prefixes=('T cell','NK cell','B cell','Plasma cell','Macrophage','Monocyte','Mast cell','Neutrophils','cDC','pDC','DC mature','myeloid dividing')

nonimmune=~immune_mask
num=(res.pred.isin(['Tumor cells LUSC','Tumor cells LUSC mitotic'])).sum()
den=nonimmune.sum()
print('immune',immune_mask.sum(),'nonimmune',den,'total',len(res))
print('LUSC malignant',num,'pct nonimmune',100*num/den)
# by sample subtype maybe all LUSC patient samples
subtype=pd.read_csv('sample_subtype.csv')
res=res.merge(subtype.rename(columns={'sample':'sampleID'}),on='sampleID',how='left')
print('subtype counts in res',res.subtype.value_counts(dropna=False).to_string())
# numerator only LUSC patients should be same as pred categories
num2=((res.pred.isin(['Tumor cells LUSC','Tumor cells LUSC mitotic']))&(res.subtype=='LUSC')).sum()
print('num2 LUSC patients',num2)
# Non-immune denominator by centroid labels all samples
print('Den by category')
print(res.loc[nonimmune,'pred'].value_counts().to_string())
# DC target proportions
for s in ['Patient_018','Patient_040']:
 sub=res[res.sampleID==s]
 dc=sub.pred.isin(['cDC1','cDC2','pDC','DC mature']).sum()
 print(s,'DC',dc,'n',len(sub),'prop',dc/len(sub))
PY"
