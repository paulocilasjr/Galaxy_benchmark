/bin/bash -lc "pip install -q pymupdf && python - <<'PY'
import fitz
doc=fitz.open('paper.pdf')
print('pages',len(doc))
terms=['lung','basal','Patient_006','Patient_018','lung.cancer','single-cell']
for i,page in enumerate(doc):
 text=page.get_text()
 if any(t.lower() in text.lower() for t in terms):
  print('\\n---PAGE',i+1,'---')
  low=text.lower()
  for t in terms:
   idx=low.find(t.lower())
   if idx>=0:
    print('term',t,'snippet:',text[max(0,idx-400):idx+1200].replace('\\n',' '))
PY"
