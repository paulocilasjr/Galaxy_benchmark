/bin/bash -lc "python - <<'PY'
import h5py
with h5py.File('covid.patients.q1.h5ad','r') as f:
    def rec(g,p=''):
        for k,v in g.items():
            path=p+'/'+k if p else k
            if isinstance(v,h5py.Dataset):
                shape=v.shape
                dt=v.dtype
                if shape==() or (len(shape)>0 and shape[0]<20):
                    try: val=v[()]
                    except: val='ERR'
                    print(path,shape,dt,val if str(val)[:200] else val)
                else:
                    print(path,shape,dt)
            else:
                print(path,'GROUP attrs',dict(v.attrs))
                if path in ['uns','obsm','layers','varm','obsp','varp'] or path.startswith('uns'):
                    rec(v,path)
    rec(f['uns'],'uns')
    print('layers keys', list(f['layers'].keys()))
    print('obsm keys', list(f['obsm'].keys()))
PY"
