/bin/bash -lc "python - <<'PY'
import h5py
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
    def visit(name,obj):
        if name.count('/')<=2:
            if hasattr(obj,'shape'):
                print(name, type(obj).__name__, obj.shape, obj.dtype)
            else:
                print(name, type(obj).__name__, list(obj.attrs.items())[:5])
    f.visititems(visit)
PY"
