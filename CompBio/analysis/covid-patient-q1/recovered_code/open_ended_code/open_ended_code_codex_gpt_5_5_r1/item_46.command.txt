/bin/bash -lc "python - <<'PY'
import urllib.request,json
url='https://api.github.com/repos/Genentech/pascient/contents/data?ref=main'
try:
    arr=json.load(urllib.request.urlopen(url, timeout=20))
    for x in arr: print(x['type'], x['name'], x.get('download_url'))
except Exception as e: print(type(e).__name__, e)
PY"
