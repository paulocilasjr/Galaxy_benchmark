import urllib.request, json
for cid in ['8f126edf-5405-4731-8374-b5ce11f53e82','0a839c4b-10d0-4d64-9272-684c49a2c8ba','a72afd53-ab92-4511-88da-252fb0e26b9a']:

        data=json.load(urllib.request.urlopen(url, timeout=20))
        print('\\nCID',cid)
        print(data.get('name'))
        print('datasets',len(data.get('datasets',[])))
        for ds in data.get('datasets',[])[:10]:
            print(ds.get('dataset_id'), ds.get('title'), ds.get('cell_count'), ds.get('primary_cell_count'))
            print('assets', [(a.get('filetype'), a.get('filesize')) for a in ds.get('assets',[])])
    except Exception as e:
        print('ERR',cid,type(e).__name__,e)
PY"
