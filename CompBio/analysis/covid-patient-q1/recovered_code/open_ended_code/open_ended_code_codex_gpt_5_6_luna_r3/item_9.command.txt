/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
 def decode(a):
  return [x.decode() if isinstance(x,bytes) else str(x) for x in a]
 donors=decode(f['obs/donor_id/categories'][:]); codes=f['obs/donor_id/codes'][:]
 counts=np.bincount(codes,minlength=len(donors))
 print('donors categories:',donors)
 print('counts:',dict(zip(donors,counts)))
 genes=decode(f['var/feature_name/categories'][:]); gi=f['var/feature_name/codes'][:]
 feat=[genes[i] for i in gi]
 print('first feat',feat[:50]);print('last feat',feat[-20:]);print('unique',len(set(feat)),len(feat))
 for k in ['feature_type','feature_biotype','feature_reference']:
  cats=decode(f['var/'+k+'/categories'][:]); c=f['var/'+k+'/codes'][:]
  print(k,dict(zip(cats,np.bincount(c,minlength=len(cats)))))
PY"
