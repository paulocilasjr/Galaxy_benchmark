/bin/bash -lc "OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 MKL_NUM_THREADS=1 python - <<'PY'
import numpy as np, re
from sklearn.decomposition import PCA
z=np.load('donor_aggregates.npz',allow_pickle=True); d=z['donor_names'].astype(str); g=z['gene_names'].astype(str); s=z['sums'];
order=np.argsort([int(re.search(r'\\d+',x).group()) for x in d])
cpm=s/s.sum(1)[:,None]*1e6
x=np.log1p(cpm); var=x.var(0)
for name,mask in [('all',np.ones(len(g),bool)),('no_mt_rp',~np.array([q.startswith(('MT-','RPS','RPL')) or q=='MALAT1' for q in g]))]:
 print('\\n===',name,'===',flush=True)
 inds=np.where(mask)[0][np.argsort(var[mask])[-60:][::-1]]
 print('top variable:', ', '.join(f'{g[i]}({var[i]:.2f})' for i in inds),flush=True)
 p=PCA(n_components=5).fit(x[:,mask]); scores=p.transform(x[:,mask]); load=p.components_; gm=g[mask]
 print('explained',np.round(p.explained_variance_ratio_,3),flush=True)
 for k in range(5):
  top=np.argsort(load[k])[-12:][::-1]; bot=np.argsort(load[k])[:12]
  print('PC',k+1,'scores',dict(zip(d[order],np.round(scores[order,k],2))),flush=True)
  print(' +',','.join(gm[top]),' -',','.join(gm[bot]),flush=True)
PY"
