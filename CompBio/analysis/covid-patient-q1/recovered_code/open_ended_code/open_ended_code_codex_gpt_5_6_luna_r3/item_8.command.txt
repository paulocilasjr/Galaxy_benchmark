/bin/bash -lc "python - <<'PY'
import h5py
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
 def walk(g, prefix='', depth=2):
  for k,v in g.items():
   path=prefix+k
   if isinstance(v,h5py.Dataset): print(path,'DATA',v.shape,v.dtype,'attrs',dict(v.attrs))
   else:
    print(path,'GROUP','keys',list(v.keys()),'attrs',dict(v.attrs))
    if depth: walk(v,path+'/',depth-1)
 walk(f['obs'],'obs/',3)
 print('--- var')
 walk(f['var'],'var/',2)
 print('X attrs',dict(f['X'].attrs))
PY"
