/bin/bash -lc "python - <<'PY'
import numpy as np, pandas as pd, re
z=np.load('donor_aggregates.npz',allow_pickle=True)
d=z['donor_names'].astype(str); s=z['sums']; n=z['cell_counts']; tc=z['total_counts'];
order=np.argsort([int(re.search(r'(\\d+)',x).group(1)) for x in d])
print('donors',len(d),'genes',len(z['gene_names']),'sum shape',s.shape)
print(pd.DataFrame({'donor':d,'ncell':n,'counts':tc,'mean_counts':tc/n}).iloc[order].to_string(index=False))
print('panel available')
panel=z['panel'].astype(str); ps=z['panel_sum']; pdct=z['panel_detected']
for j,p in enumerate(panel):
 if ps[:,j].sum()>0:
  print('\\n',p)
  print('mean expr',dict(zip(d[order], np.round(ps[order,j]/n[order],3))))
  print('pct cells',dict(zip(d[order],np.round(100*pdct[order,j]/n[order],1))))
PY"
