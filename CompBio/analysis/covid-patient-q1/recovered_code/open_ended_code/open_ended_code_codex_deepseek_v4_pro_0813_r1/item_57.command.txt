/bin/bash -lc "cat > match_seurat_meta.py <<'PY'
import pandas as pd, numpy as np, sys, os
meta=pd.read_csv('seurat_meta.csv', index_col=0)
codes=np.load('donor_codes.npy')
cats=np.load('donor_categories.npy', allow_pickle=True)
print('meta shape', meta.shape, 'codes', codes.shape)
if len(meta)"'!=len(codes):
 print('"'LENGTH MISMATCH', len(meta), len(codes))
# Align by position only if same length
n=min(len(meta), len(codes))
meta=meta.iloc[:n]; codes=codes[:n]
print('columns', meta.columns.tolist())
for col in meta.columns:
    vals=meta[col].astype(str)
    mapping=[]
    bad=0
    for c in range(len(cats)):
        sub=vals[codes==c]
        if len(sub)==0:
            mapping.append((cats[c], '<empty>', np.nan))
            continue
        top=sub.mode()
        mode=top.iloc[0] if len(top)>0 else ''
        frac=(sub==mode).mean()
        if frac<0.98: bad+=1
        mapping.append((cats[c], mode, round(frac,3)))
    # show if few categories / unique
    uniq=len(set(m[1] for m in mapping))
    print(f'\\nCOLUMN {col}: unique_mode_values={uniq}, bad_mappings={bad}')
    if uniq>=30 or col in ['Class','Subject','Donor','sample','Timepoint','Batch','orig.ident']:
        for m in mapping:
            print(' ', m[0], '->', m[1], m[2])
PY"
