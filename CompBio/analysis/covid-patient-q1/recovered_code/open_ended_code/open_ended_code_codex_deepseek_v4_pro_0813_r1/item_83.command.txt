/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
m=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False)
our=np.load('donor_codes.npy')
sl=m['sample_label']
codes_sorted,_=pd.factorize(sl, sort=True) # NaN -> -1
# candidate masks
for maskname,mask in {
 'all_nonnull': sl.notna(),
 'COVID_specific_ungated': (m['Class']=='COVID')&(m['specific_gating']=='ungated'),
 'COVID_specific_ungated_QC': (m['Class']=='COVID')&(m['specific_gating']=='ungated')&(m['celltypeQC'].astype(str)=='True'),
 'COVID_QC': (m['Class']=='COVID')&(m['celltypeQC'].astype(str)=='True'),
}.items():
    seq=codes_sorted[mask.values]; seq=seq[seq>=0]
    print(maskname, len(seq), 'target',len(our))
    if len(seq)==len(our):
        print(' exact equal fraction', np.mean(seq==our))
    # greedy subsequence assuming order preserved, but with variable deletions, test whether our is subsequence of seq
    j=0; skipped=0
    for i,v in enumerate(seq):
        if j<len(our) and v==our[j]:
            j+=1
        else:
            skipped+=1
        if j==len(our): break
    print(' greedy matches',j,'skipped before',skipped,'remaining',len(seq)-(skipped+j))
    # Also check equality ignoring missing 812 cells by computing lcs? no
PY"
