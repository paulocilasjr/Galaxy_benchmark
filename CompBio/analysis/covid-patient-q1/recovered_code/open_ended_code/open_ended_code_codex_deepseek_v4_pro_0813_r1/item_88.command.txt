/bin/bash -lc "cat > celltypes_donors.py <<'PY'
import anndata as ad, numpy as np, pandas as pd, scipy.sparse as sp, time
adata=ad.read_h5ad('covid.patients.q1.h5ad')
names=np.array(adata.var['feature_name'].astype(str))
X=adata.X.tocsr()
# library sizes
lib=np.asarray(X.sum(axis=1)).ravel()
# normalize to cpm/100 and log1p without densifying all? We'll compute scores by summing selected gene log values.
# Instead compute mean raw counts per cell per type then normalize by lib? Better compute log1p counts for marker genes; normalize each cell library by CPM.
# We'll compute sum of counts per marker / lib * 1e4, then log1p.
types={
 'Monocyte':['LYZ','S100A8','S100A9','CD14','FCGR3A','MS4A7','CSF1R','LILRB2'],
 'T':['CD3D','CD3E','CD3G','IL7R','TRAC','CD2'],
 'NK':['NKG7','GNLY','KLRD1','NCAM1','KLRB1','PRF1'],
 'B':['MS4A1','CD19','CD79A','CD79B','IGHG1','JCHAIN'],
 'pDC':['LILRA4','IL3RA','GZMB','TCF4'],
 'Platelet':['PPBP','PF4','GP9','ITGA2B'],
 'Erythroid':['HBB','HBA1','HBA2','ALAS2'],
}
idx={t:[np.where(names==g)[0][0] for g in gs if g in names] for t,gs in types.items()}
# sum raw counts per cell for marker genes
rawsum=np.zeros((X.shape[0],len(types)),dtype=np.float32)
for k,(t,ids) in enumerate(idx.items()):
    if ids:
        sub=X[:,ids]
        rawsum[:,k]=np.asarray(sub.sum(axis=1)).ravel()
# score = log1p( sum counts / lib * 10000 )
scores=np.log1p(rawsum/lib[:,None]*10000)
labels=np.argmax(scores,axis=1)
# threshold low? assign none if max < log1p(0.5)? optional
maxscore=scores.max(axis=1)
labels[maxscore < np.log1p(1.0)] = -1
type_names=list(types.keys())
donors=adata.obs['donor_id'].astype(str).values
cats=np.array(sorted(set(donors), key=lambda x:int(x[5:])))
freq=pd.DataFrame(index=cats,columns=type_names+['Unknown'],dtype=float)
for d in cats:
    mask=donors==d
    counts=np.bincount(labels[mask], minlength=len(type_names)+1) # includes -1? minlength problematic
    # better
    labs=labels[mask]
    vals=np.array([(labs==k).sum() for k in range(len(type_names))]+[(labs==-1).sum()])
    freq.loc[d]=vals/len(labs)
print(freq.round(3).to_string())
freq.to_csv('celltype_freq.csv')
# cluster donors on freq
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
Xf=StandardScaler().fit_transform(freq.values)
pca=PCA(n_components=5).fit_transform(Xf)
print('freq PCA var?')
for n in [2,3,5]:
 km=KMeans(n_clusters=2,n_init=20,random_state=0).fit(pca[:,:n])
 print('kmeans',n,[sorted(cats[km.labels_==l].tolist()) for l in range(2)])
print('freq PCA coords')
for d,p in zip(cats,pca): print(d, ' '.join(f'{v:.2f}' for v in p))
PY
python celltypes_donors.py"
