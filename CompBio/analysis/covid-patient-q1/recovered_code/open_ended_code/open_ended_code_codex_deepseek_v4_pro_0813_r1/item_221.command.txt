/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.linear_model import LogisticRegressionCV
from sklearn.model_selection import cross_val_predict, StratifiedKFold
from sklearn.metrics import roc_auc_score
# original coarse freq with class
orig=pd.read_csv('orig_coarse_freq.csv')
Xo=orig[['T','NK','B','pDC','Myeloid','Other']].values
yo=(orig.Class=='HC').astype(int)
work=pd.read_csv('celltype_freq.csv',index_col=0)
work['Myeloid']=work['Monocyte']; work['Other']=work[['Platelet','Erythroid','Unknown']].sum(axis=1)
Xw=work[['T','NK','B','pDC','Myeloid','Other']].values
for model,name in [(LogisticRegressionCV(Cs=10,cv=5,class_weight='balanced',max_iter=10000,random_state=0),'LR'),(RandomForestClassifier(n_estimators=1000,random_state=0,class_weight='balanced'),'RF')]:
 model.fit(Xo,yo)
 proba=model.predict_proba(Xw)[:,1]
 print('\\n',name,'orig cv auc',roc_auc_score(yo,cross_val_predict(model,Xo,yo,cv=5,method='predict_proba')[:,1]))
 res=pd.Series(proba,index=work.index).sort_values(ascending=False)
 print(res.round(3).to_string())
 for n in [13,14,15]: print('top',n,sorted([int(x[5:]) for x in res.index[:n]]))
PY"
