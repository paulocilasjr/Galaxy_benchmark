/bin/bash -lc "cat > sex_infer.py <<'PY'
import anndata as ad, numpy as np, pandas as pd
A=ad.read_h5ad('covid.patients.q1.h5ad',backed='r')
names=np.array(A.var['feature_name'].astype(str))
D=np.load('donor_sum.npz'); sums=D['sum']; cats=D['cats']
# log1p cpm per donor
lib=sums.sum(axis=0,keepdims=True); log=np.log1p(sums/lib*1e6)
# genes sets by symbol upper
sets={
'XIST':['XIST'],
'Y1':['RPS4Y1','DDX3Y','UTY','KDM5D','ZFY','EIF1AY','NLGN4Y','USP9Y'],
'Y2':['RPS4Y1','DDX3Y','ZFY'],
'TSIX':['TSIX'],
}
def get(g):
    for i,s in enumerate(names):
        if s.upper()==g.upper(): return i
    return None
for k,gs in sets.items():
    idx=[get(g) for g in gs]
    idx=[i for i in idx if i is not None]
    vals=log[idx].mean(axis=0) if idx else np.zeros(len(cats))
    df=pd.DataFrame({'donor':cats,k:vals})
    print(k, 'genes',len(idx), 'missing',[g for g in gs if get(g) is None])
    print(df.sort_values(k,ascending=False).to_string(index=False))
# print selected values
print('\\nselected')
idx_x=get('XIST'); idy=get('RPS4Y1')
for c in cats:
    j=np.where(cats==c)[0][0]
    print(c, 'XIST',log[idx_x,j].round(3),'RPS4Y1',log[idy,j].round(3))
PY
python3 sex_infer.py"
