/bin/bash -lc "python3 - <<'PY'
import requests,json
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi'
q='\"33 COVID-19 patients\" AND \"PBMC\"'
r=requests.get(base,params={'db':'gds','term':q,'retmax':100,'retmode':'json'},timeout=30)
ids=r.json()['esearchresult']['idlist']
print('count',len(ids))
# get summaries
r2=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=30)
for id in ids:
 it=r2.json()['result'].get(id)
 if it:
  print(id,it.get('accession'),it.get('title'))
PY"
