/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis, QuadraticDiscriminantAnalysis
from sklearn.naive_bayes import GaussianNB
from sklearn.preprocessing import StandardScaler
orig=pd.read_csv('orig_signature_scores.csv',index_col=0); work=pd.read_csv('working_signature_scores2.csv',index_col=0)
fs=['up','down','diff','ISG','Mono','T','NK','B','pDC','Platelet','Erythroid','Neutro','myel_lymph_ratio','isg_lymph_ratio']
X=orig[fs].values; y=orig['label'].values; W=work[fs].values
# standardize using orig for models requiring; LDA no scale necessary but use
sc=StandardScaler().fit(X); Xz=sc.transform(X); Wz=sc.transform(W)
for name,m in [('LDA',LinearDiscriminantAnalysis()),('QDA',QuadraticDiscriminantAnalysis(reg_param=0.5)),('NB',GaussianNB())]:
 m.fit(Xz,y); p=m.predict_proba(Wz)[:,1]; df=pd.DataFrame({'d':work.index.str[5:].astype(int),'p':p}).sort_values('p',ascending=False)
 print('\\n',name); print(df.to_string(index=False)); print('top14',','.join(map(str,df.head(14).d.tolist())))
PY"
