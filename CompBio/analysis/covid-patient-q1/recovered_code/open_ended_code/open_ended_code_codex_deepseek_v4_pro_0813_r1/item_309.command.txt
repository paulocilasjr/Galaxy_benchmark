/bin/bash -lc "python3 - <<'PY'
import pandas as pd
meta=pd.read_csv('seurat_meta.csv',low_memory=False)
vc=meta.groupby('Donor').size()
classes=meta.drop_duplicates('Donor').set_index('Donor')['Class']
for th in [0,1000,1500,1800,2000,2500,3000,3500,4000,4500,5000,5500,6000,7000,8000,9000,10000]:
 keep=vc[vc>=th].index
 n=len(keep); hc=(classes.loc[keep]=='HC').sum(); cov=n-hc
 print(th,n,'HC',hc,'COVID',cov)
 if n<=40:
  print('  removed', [x for x in vc.index if x not in keep])
PY"
