/bin/bash -lc "python3 - <<'PY'
import requests,json
base='https://genentech-compbiobench-leaderboard-v1.hf.space/gradio_api/call/refresh'
r=requests.post(base,json={'data':[]},timeout=120); eid=r.json()['event_id']
r=requests.get(base+'/'+eid,timeout=120)
text=r.text
# extract data lines
lines=[l for l in text.splitlines() if l.startswith('data: ')]
print('lines',len(lines))
for i,l in enumerate(lines):
 payload=l[6:]
 print('line',i,'len',len(payload))
 try:
  obj=json.loads(payload)
  print('type',type(obj),'len',len(obj) if hasattr(obj,'__len__') else '')
  if isinstance(obj,list):
   for j,o in enumerate(obj):
    if isinstance(o,dict):
     print(' obj',j,'keys',list(o.keys()),'headers',o.get('headers'),'data rows',len(o.get('data',[])))
    else: print(' obj',j,type(o),str(o)[:100])
 except Exception as e:
  print('parse err',e,payload[:200])
PY"
