/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
odf=pd.read_csv('orig_signature_scores.csv',index_col=0); wdf=pd.read_csv('working_signature_scores2.csv',index_col=0)
features=['up','down','diff','ISG','Mono','T','NK','B','pDC','Platelet','Erythroid','Neutro','myel_lymph_ratio','isg_lymph_ratio']
Xo=odf[features].values; yo=odf.label.values
scaler=StandardScaler().fit(Xo)
lr=LogisticRegression(C=10,max_iter=10000).fit(scaler.transform(Xo),yo)
for d in ['donor1','donor22','donor16','donor6']:
 x=scaler.transform(wdf.loc[[d],features].values)[0]
 z=np.dot(lr.coef_[0],x)+lr.intercept_[0]
 print(d,'z',z,'p',1/(1+np.exp(-z)))
 print(' coef*features:')
 for f,c,xi in zip(features,lr.coef_[0],x): print('  ',f,'c',round(c,2),'x',round(xi,2),'contrib',round(c*xi,2))
PY"
