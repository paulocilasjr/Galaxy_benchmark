/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
m=pd.read_csv('seurat_meta.csv', index_col=0, low_memory=False)
print(m['mergedcelltype'].value_counts())
print('\\nWCTmergedcelltype')
print(m['WCTmergedcelltype'].value_counts())
# per donor mergedcelltype frequencies
freq=m.groupby(['Donor','Class'])['mergedcelltype'].value_counts(normalize=True).unstack(fill_value=0)
print(freq.round(3).to_string())
freq.to_csv('orig_mergedcelltype_donor_freq.csv')
# WCT
freq2=m.groupby(['Donor','Class'])['WCTmergedcelltype'].value_counts(normalize=True).unstack(fill_value=0)
print('\\nWCT freq')
print(freq2.round(3).to_string())
freq2.to_csv('orig_wctcelltype_donor_freq.csv')
PY"
