/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np, scipy.sparse as sp, pandas as pd, time
p='covid.patients.q1.h5ad'
adata=ad.read_h5ad(p)
donors=pd.Series(adata.obs['donor_id'].astype(str).values)
cats=np.array(sorted(donors.unique(), key=lambda x:int(x[5:])))
num=np.array([int(x[5:])-1 for x in donors])
D=sp.csr_matrix((np.ones(len(num), dtype=np.float32), (np.arange(len(num)), num)), shape=(len(num), len(cats)))
X=adata.X.tocsr()
sum_gd=(X.T @ D).toarray()
Xbin=(X>0).astype(np.float32)
nnz_gd=(Xbin.T @ D).toarray()
cell_counts=np.asarray(D.sum(axis=0)).ravel().astype(int)
np.savez_compressed('donor_sum.npz', sum=sum_gd, nnz=nnz_gd, cats=cats, cell_counts=cell_counts)
print('saved', cell_counts)
PY
python cluster_donors.py"
