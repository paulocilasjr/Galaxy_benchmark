/bin/bash -lc "python3 - <<'PY'
import requests,json
q='\"33 COVID-19 patients\" AND \"PBMC\"'
r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':q,'retmax':100,'retmode':'json'},timeout=30)
ids=r.json()['esearchresult']['idlist']
r2=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',params={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=30)
for id in ids:
 it=r2.json()['result'].get(id)
 if not it: continue
 title=it.get('title','') or ''
 summary=it.get('summary','') or ''
 if any(k in title.lower() for k in ['single-cell','scrna','single cell']) or 'single-cell' in summary.lower():
  print(id,it.get('accession'),title[:120], '|', summary[:200].replace('\\n',' '))
PY"
