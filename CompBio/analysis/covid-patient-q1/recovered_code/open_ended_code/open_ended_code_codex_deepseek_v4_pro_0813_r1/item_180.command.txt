/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
orig=pd.read_csv('group_sums_Donor.csv',index_col=0)
O=orig.values.astype(float); lib=O.sum(axis=0,keepdims=True); log=np.log1p(O/lib*1e6)
mu=log.mean(axis=1,keepdims=True); sd=log.std(axis=1,keepdims=True); sd[sd==0]=1
Z=(log-mu)/sd
names=np.array([x.upper() for x in orig.index])
cols=np.array(orig.columns)
def sc(g):
 a=np.where(names==g)[0]; return Z[a[0]] if len(a) else np.zeros(len(cols))
X=sc('XIST'); Y=sum(sc(g) for g in ['RPS4Y1','DDX3Y','UTY','KDM5D','ZFY','USP9Y','EIF1AY'])
score=X-Y
meta=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False).drop_duplicates('Donor').set_index('Donor')
for c,x,y,s in zip(cols,X,Y,score):
 print(c,'gender',meta.loc[c,'Gender'] if c in meta.index else '?', 'X',round(x,1),'Ysum',round(y,1),'score',round(s,1), 'infer','F' if s>5 else 'M' if s<-5 else '?')
PY"
