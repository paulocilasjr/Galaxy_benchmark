import numpy as np,pandas as pd,anndata as ad
from sklearn.metrics.pairwise import cosine_similarity
from scipy.optimize import linear_sum_assignment
A=ad.read_h5ad('covid.patients.q1.h5ad',backed='r'); wnames=np.array([str(x) if x is not None else '' for x in A.var['feature_name']]); wl=np.char.lower(wnames.astype('U'))
orig=pd.read_csv('group_sums_Donor.csv',index_col=0); onames=np.array([str(x) for x in orig.index]); ocols=np.array(orig.columns.astype(str)); O=orig.values.astype(float); ol=np.char.lower(onames.astype('U')); common,wi,oi=np.intersect1d(wl,ol,return_indices=True); D=np.load('donor_sum.npz'); W=D['sum'][wi,:].astype(float); Or=O[oi,:].astype(float)
def logcpm(X): return np.log1p(X/X.sum(axis=0,keepdims=True)*1e6)
W=logcpm(W); Or=logcpm(Or); Wz=(W-W.mean(1,keepdims=True))/(W.std(1,keepdims=True)+1e-8); Oz=(Or-Or.mean(1,keepdims=True))/(Or.std(1,keepdims=True)+1e-8)
meta=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False).drop_duplicates('Donor').set_index('Donor'); labels=meta.loc[ocols,'Class'].values
hc_idx=np.where(labels=='HC')[0]; covid_idx=np.where(labels=='COVID')[0]
cats=D['cats']; donor_nums=np.array([int(d[5:]) for d in cats])
for metric,M in [('cos',cosine_similarity(Wz.T,Oz.T)),('pear',np.corrcoef(Wz.T,Oz.T)[:Wz.shape[1],Wz.shape[1]:])]:
 print('\\nMETRIC',metric)
 for name,set14 in [('B',[1,5,6,10,11,12,16,17,21,23,24,27,29,32]),('B22',[1,5,6,10,11,12,17,21,22,23,24,27,29,32]),('knn19',[1,5,6,10,11,12,17,19,21,23,24,27,29,32]),('cell14',[1,5,10,11,12,16,17,21,22,24,25,27,29,32])]:
  widx=[np.where(donor_nums==d)[0][0] for d in set14]
  # similarity to original HC submatrix
  sim_hc=M[np.ix_(widx,hc_idx)]
  r,c=linear_sum_assignment(-sim_hc)
  print(name,'totalHC',round(sim_hc[r,c].sum(),3),'assignments',[(set14[rr],ocols[hc_idx[cc]],round(sim_hc[rr,cc],2)) for rr,cc in zip(r,c)])
PY"
