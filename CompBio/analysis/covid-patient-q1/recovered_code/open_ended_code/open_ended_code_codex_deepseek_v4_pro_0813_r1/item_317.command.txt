import pandas as pd,numpy as np,anndata as ad
from sklearn.metrics.pairwise import cosine_similarity
A=ad.read_h5ad('covid.patients.q1.h5ad',backed='r'); wnames=np.array([str(x) if x is not None else '' for x in A.var['feature_name']]); wl=np.char.lower(wnames.astype('U'))
for f in ['group_sums_sample_name.csv','group_sums_sample_id.csv','group_sums_sample.csv']:

 def logcpm(X): return np.log1p(X/X.sum(axis=0,keepdims=True)*1e6)

  return sub['Class'].mode().iloc[0] if len(sub) else 'NA'
 labels=np.array([cls_for(c) for c in ocols])
 cos=cosine_similarity(Wz.T,Oz.T)

 rows=[]
 for j,d in enumerate(cats):

 print('\\nFILE',f)
