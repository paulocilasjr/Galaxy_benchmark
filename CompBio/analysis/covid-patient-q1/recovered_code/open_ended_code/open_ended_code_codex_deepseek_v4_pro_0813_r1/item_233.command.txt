/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
freq=pd.read_csv('celltype_freq.csv',index_col=0)
sig=pd.read_csv('working_signature_scores2.csv',index_col=0)
freq['lymph']=freq[['T','NK','B']].sum(axis=1)
freq['myel']=freq['Monocyte']
# score using normalized ranks? simple standardized
for col in ['lymph','myel','ISG','Mono','Neutro','T','NK','B']:
 if col in sig: sig[col]=sig[col]
sig['lymph_freq']=freq['lymph']; sig['myel_freq']=freq['myel']
sig['ratio']=(sig['T']+sig['NK']+sig['B'])/3 - (sig['ISG']+sig['Mono']+sig['Neutro'])/3
sig['ratio_freq']=sig['lymph_freq']-sig['myel_freq']
print('ratio')
print(sig.sort_values('ratio',ascending=False)[['ratio','ISG','Mono','T','NK','B']].round(3).to_string())
for n in [13,14,15]:
 print(n,sorted([int(x[5:]) for x in sig.sort_values('ratio',ascending=False).index[:n]]))
print('\\nfreq ratio')
print(sig.sort_values('ratio_freq',ascending=False)[['ratio_freq','lymph_freq','myel_freq','ISG']].round(3).to_string())
for n in [13,14,15]:
 print(n,sorted([int(x[5:]) for x in sig.sort_values('ratio_freq',ascending=False).index[:n]]))
PY"
