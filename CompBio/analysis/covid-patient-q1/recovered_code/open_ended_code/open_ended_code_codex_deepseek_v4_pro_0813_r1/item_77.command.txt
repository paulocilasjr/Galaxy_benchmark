/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, anndata as ad
adata=ad.read_h5ad('covid.patients.q1.h5ad')
our_names=np.array(adata.var['feature_name'].astype(str))
D=np.load('donor_sum.npz'); our_sums=D['sum']; our_cats=D['cats']; our_cc=D['cell_counts']
our_libs=our_sums.sum(axis=0, keepdims=True)
our_cpm=our_sums/our_libs*1e6
our_log=np.log1p(our_cpm)

def match_original(path):
    orig=pd.read_csv(path,index_col=0)
    orig.index=orig.index.astype(str)
    common=[g for g in our_names if g in orig.index]
    if len(common)<1000:
        print('too few',path,len(common)); return
    seen=set(); idx=[]
    for g in common:
        if g not in seen:
            seen.add(g)
            idx.append(np.where(our_names==g)[0][0])
    idx=np.array(idx); genes=[our_names[i] for i in idx]
    A=our_log[idx,:]
    Braw=orig.loc[genes]
    Bcols=Braw.columns
    B=np.log1p(Braw.values / Braw.values.sum(axis=0, keepdims=True)*1e6)
    C=np.corrcoef(A.T, B.T)[:len(our_cats), len(our_cats):]
    best=C.argmax(axis=1)
    print('\\nPATH',path,'common genes',len(idx))
    for i,c in enumerate(our_cats):
        tops=np.argsort(C[i])[::-1][:5]
        print(c, 'best', Bcols[best[i]], round(C[i,best[i]],3), 'tops', [(Bcols[t], round(C[i,t],3)) for t in tops])
    return C,Bcols
for path in ['group_sums_Donor.csv','group_sums_sample_id.csv','group_sums_sample_name.csv','group_sums_Subject.csv','group_sums_sample.csv']:
    match_original(path)
PY"
