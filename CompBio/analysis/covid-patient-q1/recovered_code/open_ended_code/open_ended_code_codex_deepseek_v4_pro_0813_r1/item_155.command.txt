/bin/bash -lc "cat > cluster_sigs.py <<'PY'
import pandas as pd, numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.mixture import GaussianMixture
from scipy.cluster.hierarchy import linkage, fcluster
from scipy.spatial.distance import pdist

# Load features
sig=pd.read_csv('working_signature_scores2.csv',index_col=0)
freq=pd.read_csv('celltype_freq.csv',index_col=0)
# combine selected
feat_sig=['up','down','diff','ISG','Mono','T','NK','B','pDC','Platelet','Erythroid','Neutro','myel_lymph_ratio','isg_lymph_ratio']
X1=sig[feat_sig].values
# freq features (maybe standardized separately)
freq2=freq.copy(); freq2['Myeloid']=freq2['Monocyte']; freq2['Other']=freq2[['Platelet','Erythroid','Unknown']].sum(axis=1)
X2=freq2[['T','NK','B','pDC','Myeloid','Other']].values
# concat standardized each
from sklearn.preprocessing import StandardScaler
X=np.concatenate([StandardScaler().fit_transform(X1), StandardScaler().fit_transform(X2)],axis=1)
donors=sig.index.tolist()

def show(labels,name):
 print('\\n===',name,'===')
 for lab in sorted(set(labels)):
  mem=sorted([int(d[5:]) for d,l in zip(donors,labels) if l==lab])
  print(lab,len(mem),mem)

for n_clusters in [2,3,4]:
 km=KMeans(n_clusters=n_clusters,n_init=100,random_state=0).fit(X)
 show(km.labels_,f'KMeans {n_clusters}')
 # agg ward
 agg=AgglomerativeClustering(n_clusters=n_clusters,linkage='ward').fit(X)
 show(agg.labels_,f'Agg {n_clusters}')
# GMM
for comp in [2,3,4]:
 gmm=GaussianMixture(n_components=comp,random_state=0,covariance_type='full',n_init=20).fit(X)
 show(gmm.predict(X),f'GMM {comp}')
# PCA 2D coords
pca=PCA(n_components=2,random_state=0).fit(X)
P=pca.transform(X)
print('\\nPCA var',pca.explained_variance_ratio_)
for d,(x,y) in zip(donors,P): print(d,round(x,3),round(y,3))
# linkage and print clusters at distance thresholds
Z=linkage(X,method='ward')
for t in [3,4,5,6,7,8]:
 labels=fcluster(Z,t,criterion='distance')
 show(labels,f'ward dist {t}')
PY
python3 cluster_sigs.py"
