/bin/bash -lc "Rscript - <<'RS'
x <- readRDS('repo/A4/input/healthy_vs_covid_COVID-Healthy_module_score_gsva_filtered_samples_genes.rds')
x"'$group <- ifelse(grepl('"'_HC_', x"'$sample), '"'HC', ifelse(grepl('_T0_', x"'$sample), '"'COVID_T0', 'OTHER'))
y <- subset(x, group %in% c('HC','COVID_T0'))
paths <- c('GO_RESPONSE_TO_TYPE_I_INTERFERON','HALLMARK_INTERFERON_ALPHA_RESPONSE','HALLMARK_INTERFERON_GAMMA_RESPONSE','reactome_Interferon Signaling','reactome_Interferon alpha/beta signaling','btm_M75_antiviral IFN signature','GO_CELLULAR_RESPONSE_TO_INTERFERON_GAMMA','GO_INTERFERON_GAMMA_MEDIATED_SIGNALING_PATHWAY','reactome_ISG15 antiviral mechanism','reactome_Antiviral mechanism by IFN-stimulated genes','GO_RESPONSE_TO_INTERFERON_GAMMA','GO_RESPONSE_TO_INTERFERON_BETA')
for(p in paths){
 sub <- y[y"'$pathway==p,]
 if(nrow(sub)==0) next
 agg <- aggregate(module.score ~ celltype + group, data=sub, FUN=mean)
 wide <- reshape(agg, idvar='"'celltype', timevar='group', direction='wide')
 wide"'$diff <- wide$module.score.COVID_T0 - wide$module.score.HC
 cat('"'\\n',p,'overall diff',mean(wide"'$diff),'"'\\n')
 print(wide[order(-abs(wide"'$diff)),], row.names=FALSE)
 le <- unique(unlist(strsplit(sub$leadingEdge, '"' ')))
 cat('LE genes', paste(le, collapse=' '), '\\n')
}
RS"
