/bin/bash -lc "python3 - <<'PY'
# use saved? rerun quickly compute top class and margins from matrices
import numpy as np,pandas as pd,anndata as ad
from sklearn.metrics.pairwise import cosine_similarity
A=ad.read_h5ad('covid.patients.q1.h5ad',backed='r'); wnames=np.array([str(x) if x is not None else '' for x in A.var['feature_name']]); orig=pd.read_csv('group_sums_Donor.csv',index_col=0); onames=np.array([str(x) for x in orig.index]); ocols=np.array(orig.columns.astype(str)); O=orig.values.astype(float); wl=np.char.lower(wnames.astype('U')); ol=np.char.lower(onames.astype('U')); common,wi,oi=np.intersect1d(wl,ol,return_indices=True); D=np.load('donor_sum.npz'); W=D['sum'][wi,:].astype(float); Or=O[oi,:].astype(float)
def logcpm(X): return np.log1p(X/X.sum(axis=0,keepdims=True)*1e6)
W=logcpm(W); Or=logcpm(Or); Wz=(W-W.mean(1,keepdims=True))/(W.std(1,keepdims=True)+1e-8); Oz=(Or-Or.mean(1,keepdims=True))/(Or.std(1,keepdims=True)+1e-8)
meta=pd.read_csv('seurat_meta.csv',index_col=0,low_memory=False).drop_duplicates('Donor').set_index('Donor'); labels=meta.loc[ocols,'Class'].values; cats=D['cats']
cos=cosine_similarity(Wz.T,Oz.T)
for j,d in enumerate(cats):
 order=np.argsort(-cos[j])
 top=order[0]; second=order[1]
 print(f'{d}: top {ocols[top]} {labels[top]} {cos[j,top]:.3f} second {ocols[second]} {labels[second]} {cos[j,second]:.3f} topHC_rank=',np.where(labels[order]==\"HC\")[0][0] if any(labels[order]==\"HC\") else '')
PY"
