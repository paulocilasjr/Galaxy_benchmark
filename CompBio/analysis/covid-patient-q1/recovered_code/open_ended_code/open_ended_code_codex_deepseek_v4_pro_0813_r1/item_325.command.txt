/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
s=pd.read_csv('working_signature_scores2.csv',index_col=0)
p=pd.read_csv('classifier_probs.csv').set_index('donor')
cell=pd.read_csv('cell_covid_scores.csv').set_index('donor')
# Build ranks (healthy lower for diff scores)
donors=[f'donor{i}' for i in range(1,34)]
df=pd.DataFrame(index=donors)
df['diff_rank']=s['diff'].rank()
df['up_rank']=s['up'].rank()
df['ISG_rank']=s['ISG'].rank()
df['Mono_rank']=s['Mono'].rank()
df['lymph_rank']=(-(s['T']+s['NK']+s['B'])).rank()
df['cell_diff_rank']=cell['diff'].rank()
df['p_hc_rank']=p['p_covid'].rank(ascending=False)
df['myel_ratio_rank']=s['myel_lymph_ratio'].rank()
df['isg_lymph_rank']=s['isg_lymph_ratio'].rank()
# composite mean rank with chosen weights (all lower healthy)
weights={'diff_rank':3,'up_rank':1,'ISG_rank':2,'Mono_rank':1,'lymph_rank':1,'cell_diff_rank':2,'p_hc_rank':2,'isg_lymph_rank':1}
score=sum(df[k]*w for k,w in weights.items())/sum(weights.values())
df['score']=score
res=df.sort_values('score')
print(res.to_string())
print('top14',','.join(map(str,res.head(14).index.str[5:].tolist())))
PY"
