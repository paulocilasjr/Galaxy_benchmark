/bin/bash -lc "cat > cell_scores.py <<'PY'
import anndata as ad, numpy as np, pandas as pd, scipy.sparse as sp
A=ad.read_h5ad('covid.patients.q1.h5ad',backed='r')
names=np.array(A.var['feature_name'].astype(str))
obs=A.obs['donor_id'].astype(str).values
donors=sorted(set(obs))
def read_genes(path):
 return [l.strip().upper() for l in open(path) if l.strip()]
up=read_genes('covid_up_genes.txt'); dn=read_genes('covid_dn_genes.txt')
nameup=np.array([s.upper() for s in names])
idx_up=np.array([np.where(nameup==g)[0][0] for g in up if np.any(nameup==g)])
idx_dn=np.array([np.where(nameup==g)[0][0] for g in dn if np.any(nameup==g)])
print('up found',len(idx_up),'dn found',len(idx_dn))
N=A.shape[0]
# compute donor means using streaming for selected genes? full matrix slices
# store accumulators
acc=np.zeros((len(donors),4)) # sum score, n, mean up, mean dn? 
# We'll compute log1p normalized by cell total then score avg up - avg dn
up_acc=np.zeros(len(donors)); dn_acc=np.zeros(len(donors)); n=np.zeros(len(donors))
di={d:i for i,d in enumerate(donors)}
# Need map obs to index chunks
chunk=10000
for start in range(0,N,chunk):
    X=A.X[start:start+chunk]
    if sp.issparse(X):
        Xc=X.tocsr()
        tot=np.asarray(Xc.sum(axis=1)).ravel()
        Xsub_up=Xc[:,idx_up].astype(np.float32)
        Xsub_dn=Xc[:,idx_dn].astype(np.float32)
        # log1p cpm
        denom=np.maximum(tot[:,None],1)
        up_vals=np.log1p(Xsub_up/denom*1e6).mean(axis=1)
        dn_vals=np.log1p(Xsub_dn/denom*1e6).mean(axis=1)
    else:
        tot=X.sum(axis=1)
        up_vals=np.log1p(X[:,idx_up]/tot[:,None]*1e6).mean(axis=1)
        dn_vals=np.log1p(X[:,idx_dn]/tot[:,None]*1e6).mean(axis=1)
    ob=np.array([di[d] for d in obs[start:start+chunk]])
    # use bincount
    for k in range(len(ob)):
        up_acc[ob[k]] += up_vals[k]
        dn_acc[ob[k]] += dn_vals[k]
        n[ob[k]] += 1
    if start%50000==0: print(start)
res=pd.DataFrame({'donor':donors,'n':n,'mean_up':up_acc/n,'mean_dn':dn_acc/n,'diff':up_acc/n-dn_acc/n})
res['donor_num']=res.donor.str[5:].astype(int)
res=res.sort_values('diff')
print(res.to_string(index=False))
# save
res.to_csv('cell_covid_scores.csv',index=False)
PY
python3 cell_scores.py"
