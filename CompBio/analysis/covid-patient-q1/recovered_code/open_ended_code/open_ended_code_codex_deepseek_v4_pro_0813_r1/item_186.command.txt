/bin/bash -lc "python3 - <<'PY'
import json,requests,time
j=json.load(open('[REDACTED LOCAL PATH]'))
cands=[]
for d in j:
 name=d.get('name','') or ''
 if any(k in name.lower() for k in ['covid','pbmc','blood','immune']):
  cands.append(d)
print('cands',len(cands))
for d in cands[:200]:
 did=d['id']
 try:
  r=requests.get(f'https://api.cellxgene.cziscience.com/dp/v1/datasets/{did}',timeout=20)
  if r.status_code==200:
   det=r.json()
   # donor_count field?
   dc=det.get('donor_count') or det.get('donors') or ''
   cc=det.get('cell_count')
   print(did,d.get('name'),'cc',cc,'donor_count',dc,'keys',[k for k in det.keys() if 'donor' in k.lower() or 'cell' in k.lower() or 'name' in k.lower()])
 except Exception as e:
  pass
PY"
