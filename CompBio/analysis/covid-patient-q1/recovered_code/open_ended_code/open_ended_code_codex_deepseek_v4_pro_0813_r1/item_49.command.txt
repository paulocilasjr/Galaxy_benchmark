/bin/bash -lc "python - <<'PY'
import numpy as np, anndata as ad
D=np.load('donor_sum.npz'); sums=D['sum']; cats=D['cats']; cc=D['cell_counts']
adata=ad.read_h5ad('covid.patients.q1.h5ad')
names=list(adata.var['feature_name'].astype(str))
libs=sums.sum(axis=0, keepdims=True); cpm=sums/libs*1e6; log=np.log1p(cpm)
# score of ISG normalized across donors
isgs=['ISG15','IFI27','IFI44L','IFIT1','IFIT2','IFIT3','IFIT5','MX1','MX2','OAS1','OAS2','OAS3','OASL','IFI6','IFIH1','IFI44','IFI35','IFITM1','IFITM2','IFITM3','ISG20']
idx=[names.index(g) for g in isgs if g in names]
# zscore per gene across donors then mean
z=(log[idx,:]-log[idx,:].mean(axis=1,keepdims=True))/log[idx,:].std(axis=1,keepdims=True)
score=z.mean(axis=0)
print('ranked ISG score')
for c,s in sorted(zip(cats,score),key=lambda x:x[1], reverse=True):
 print(c, round(s,3), 'IFI27cpm', round(cpm[names.index('IFI27'),list(cats).index(c)],1), 'IFITM3cpm', round(cpm[names.index('IFITM3'),list(cats).index(c)],1))
print('\\nIFI27 ranked')
for c in sorted(cats, key=lambda x:cpm[names.index('IFI27'),list(cats).index(x)], reverse=True):
 print(c, round(cpm[names.index('IFI27'),list(cats).index(c)],1), 'score', round(score[list(cats).index(c)],2))
PY"
