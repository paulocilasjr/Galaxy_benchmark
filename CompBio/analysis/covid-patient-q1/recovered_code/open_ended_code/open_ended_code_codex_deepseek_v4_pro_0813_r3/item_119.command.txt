/bin/bash -c "python - <<'PY'
import h5py, numpy as np
p='GSE180578_annotated.h5ad'
with h5py.File(p,'r') as f:
    X=f['X'][:]
    varidx=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/_index'][:]]
    varfeat=[x.decode() if isinstance(x,bytes) else str(x) for x in f['var/features'][:]]
    pid=[x.decode() if isinstance(x,bytes) else str(x) for x in f['obs/patient_id'][:]]
    disease=[x.decode() if isinstance(x,bytes) else str(x) for x in f['obs/disease'][:]]
print('X shape',X.shape,X.dtype)
print('varidx first',varidx[:20])
print('varfeat first',varfeat[:20])
print('unique patients',len(set(pid)))
# save arrays for matching
np.save('ann_X.npy',X)
np.save('ann_var.npy',np.array(varidx,dtype=object))
np.save('ann_pid.npy',np.array(pid,dtype=object))
np.save('ann_disease.npy',np.array(disease,dtype=object))
PY"
