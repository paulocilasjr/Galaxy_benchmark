/bin/bash -c "python - <<'PY'
import requests, json
r=requests.get('https://api.cellxgene.cziscience.com/curation/v1/datasets',timeout=120)
data=r.json()
for d in data:
    donors=d.get('donor_id') or []
    n=len(set(donors))
    if n==33:
        cc=d.get('cell_count')
        # check disease fields or title
        disease=d.get('disease') or []
        tissue=d.get('tissue') or []
        # print if disease contains covid/normal or tissue blood
        print(d.get('dataset_id'), 'cells',cc,'donors',n,'title',(d.get('title') or '')[:100], 'tissue',tissue[:3], 'disease',disease[:3])
PY"
