    return x.decode() if isinstance(x,bytes) else str(x)
df=pd.DataFrame({k:[dec(v) for v in vals] for k,vals in obs.items()})
print(df.head())
print(df.groupby(['patient_id','disease','sample_type']).size().reset_index(name='n'))
print('unique patients',df.patient_id.nunique())
# map disease per patient
for pid,g in df.groupby('patient_id'):
    print(pid, g.disease.iloc[0], g.sample_type.iloc[0], len(g))
