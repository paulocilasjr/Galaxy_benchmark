/bin/bash -c "python - <<'PY'
import requests
data=requests.get('https://api.cellxgene.cziscience.com/curation/v1/datasets',timeout=120).json()
for d in data:
    cc=d.get('cell_count',0)
    title=d.get('title','') or ''
    if 250000 <= cc <= 320000 and ('covid' in title.lower() or 'PBMC' in title or 'blood' in title.lower()):
        donors=d.get('donor_id') or []
        print(d.get('dataset_id'),cc,len(set(donors)),title)
PY"
