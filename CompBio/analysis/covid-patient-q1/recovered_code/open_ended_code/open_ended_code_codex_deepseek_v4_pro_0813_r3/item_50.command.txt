python - <<'PY'
import re
text=open('GSE157344_gsm.txt').read()
samples=text.split('"'^SAMPLE = '"')[1:]
print('n',len(samples))
blood=[]; bal=[]; other=[]
for s in samples:
    lines=s.strip().splitlines()
    gsm=lines[0].strip()
    title=''; src=''; char=[]
    for l in lines:
        if l.startswith('"'!Sample_title'"'): title=l.split('= ',1)[1].strip()
        if l.startswith('"'!Sample_source_name_ch1'"'): src=l.split('= ',1)[1].strip()
        if l.startswith('"'!Sample_characteristics_ch1'"'): char.append(l.split('= ',1)[1].strip())
    if 'blood' in title.lower() or 'blood' in src.lower() or any('blood' in c.lower() for c in char):
        blood.append((gsm,title,src,char))
    elif 'broncho' in title.lower() or 'bal' in title.lower() or any('broncho' in c.lower() for c in char):
        bal.append((gsm,title,src,char))
    else: other.append((gsm,title,src,char))
print('blood',len(blood),'bal',len(bal),'other',len(other))
for x in blood: print('BLOOD',x)
PY"
