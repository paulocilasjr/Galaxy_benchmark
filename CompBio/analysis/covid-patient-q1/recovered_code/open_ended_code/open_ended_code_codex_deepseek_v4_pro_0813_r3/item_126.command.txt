/bin/bash -c "python - <<'PY'
exec(open('classify_with_annotated.py').read().split(\"# load annotated raw X\")[0])
PY
# simpler modify and run via sed? Let's use python heredoc rewritten
python - <<'PY'
import numpy as np, h5py, scipy.sparse as sp
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import LeaveOneGroupOut
from sklearn.metrics import accuracy_score
with h5py.File('GSE180578_annotated.h5ad','r') as f:
    data=f['raw/X/data'][:]; indices=f['raw/X/indices'][:]; indptr=f['raw/X/indptr'][:]
    ann_var=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in f['raw/var/_index'][:]])
    pid=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in f['obs/patient_id'][:]])
    disease=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in f['obs/disease'][:]])
ann_raw=sp.csr_matrix((data,indices,indptr),shape=(len(pid),len(ann_var)))
PB=np.load('pb_dense.npy'); donors=np.load('donor_names.npy',allow_pickle=True)
genes=np.array([g.decode() if isinstance(g,bytes) else str(g) for g in np.load('gene_names.npy',allow_pickle=True)])
common,ann_idx,our_idx=np.intersect1d(np.array([g.upper() for g in ann_var]), np.array([g.upper() for g in genes]),return_indices=True)
patients=np.unique(pid)
ann_counts=np.zeros((len(patients),len(common)))
ann_dis=[]; ann_pat=[]
for i,p in enumerate(patients):
    m=pid==p; ann_counts[i]=np.asarray(ann_raw[m][:,ann_idx].sum(axis=0)).ravel(); ann_dis.append(disease[m][0]); ann_pat.append(p)
def logcpm(M):
    lib=M.sum(axis=1,keepdims=True); lib[lib==0]=1
    return np.log2(M/lib*1e6+1)
ann_log=logcpm(ann_counts)
our_log=logcpm(PB[our_idx,:].T)
mask=np.isin(np.array(ann_dis),['healthy','covid'])
X=ann_log[mask]; y=(np.array(ann_dis)[mask]=='covid').astype(int); groups=np.array(ann_pat)[mask]
# filter low var and all same across train/test
gmask=(X.var(axis=0)>0.01) & (our_log.var(axis=0)>0)
X=X[:,gmask]; our=our_log[:,gmask]
print('train',X.shape,'healthy',sum(y==0),'covid',sum(y==1))
scaler=StandardScaler().fit(X); Xs=scaler.transform(X); ours=scaler.transform(our)
clf=LogisticRegression(max_iter=3000,C=1.0).fit(Xs,y)
proba=clf.predict_proba(ours)[:,1]
for j,d in enumerate(donors):
    print(f'{d} prob_covid {proba[j]:.3f} pred', 'COVID' if proba[j]>=0.5 else 'HEALTHY')
print('healthy IDs', [int(d[5:]) for j,d in enumerate(donors) if proba[j]<0.5])
# LOPO accuracy
preds=[]; tests=[]
for tr,te in LeaveOneGroupOut().split(Xs,y,groups):
    c=LogisticRegression(max_iter=3000,C=1.0).fit(Xs[tr],y[tr]); preds.append(c.predict(Xs[te])); tests.append(te)
preds=np.concatenate(preds); tests=np.concatenate(tests)
print('LOPO acc',accuracy_score(y[tests],preds))
PY"
