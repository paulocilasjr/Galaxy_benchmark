/bin/bash -c "python - <<'PY'
import requests
u='https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE171555&targ=self&form=text&view=brief'
for i in range(3):
    try:
        r=requests.get(u,timeout=60)
        print(r.status_code,len(r.text))
        open('GSE171555_brief.txt','w').write(r.text)
        print(r.text[:2500])
        break
    except Exception as e:
        print('retry',i,e)
PY"
