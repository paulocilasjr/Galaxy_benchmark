/bin/bash -c "cat > de.py <<'PY'
import numpy as np, pandas as pd
from scipy import stats
PB=np.load('pb_dense.npy')
donors=np.load('donor_names.npy', allow_pickle=True)
genes=np.load('gene_names.npy', allow_pickle=True)
lib=PB.sum(axis=0, keepdims=True)
log=np.log2(PB/lib*1e6+1)
A=[1,5,11,12,17,21,24,27,29,32]
B=[i for i in range(1,34) if i not in A]
idxA=[np.where(donors==f'donor{i}')[0][0] for i in A]
idxB=[np.where(donors==f'donor{i}')[0][0] for i in B]
print(idxA, idxB)
for group in [A,B]:
    arr=log[:,[np.where(donors==f'donor{i}')[0][0] for i in group]]
    print('group', group, 'mean log expr top', np.mean(arr,axis=1).max())
# t test log values, mean diff group A - B
mA=log[:,idxA].mean(axis=1)
mB=log[:,idxB].mean(axis=1)
diff=mA-mB
# compute p
_,p=stats.ttest_ind(log[:,idxA], log[:,idxB], axis=1, equal_var=False)
# rank by abs diff with p<0.05
order=np.argsort(diff)
print('TOP UP in A (positive diff)')
for i in order[::-1][:50]:
    print(genes[i], f'{diff[i]:.3f}', f'mA {mA[i]:.2f} mB {mB[i]:.2f}', f'p {p[i]:.2g}')
print('TOP UP in B (negative diff)')
for i in order[:50]:
    print(genes[i], f'{diff[i]:.3f}', f'mA {mA[i]:.2f} mB {mB[i]:.2f}', f'p {p[i]:.2g}')
PY
python de.py"
