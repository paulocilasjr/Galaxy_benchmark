/bin/bash -lc "python - <<'PY'
import pandas as pd, anndata as ad
src=pd.read_csv('source_donor_counts.csv')
adata=ad.read_h5ad('covid.patients.q1.h5ad', backed='r')
local=adata.obs['donor_id'].astype(str).value_counts().rename_axis('local').reset_index(name='n')
local['id']=local['local'].str.replace('donor','').astype(int)
local=local.sort_values('id')
for _,r in local.iterrows():
    src['diff']=(src['n']-r.n).abs()
    near=src.sort_values('diff').head(5)
    print(r['id'], r['n'], '; '.join(f\"{a}:{b}:{c}:d{d}\" for a,b,c,d in zip(near.donor_id, near.disease, near.n, near['diff'])))
PY"
