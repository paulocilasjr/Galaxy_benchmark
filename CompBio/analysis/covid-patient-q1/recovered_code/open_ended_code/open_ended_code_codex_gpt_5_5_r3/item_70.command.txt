import cellxgene_census, anndata as ad
import numpy as np, pandas as pd, time

# focused genes: disease, lineage, and top PC markers from local analysis
panel = ['IFI27','IFI6','ISG15','MX1','OAS1','OAS2','OAS3','IFIT1','IFIT2','IFIT3','RSAD2','SIGLEC1','LY6E','STAT1','IRF7','XAF1','USP18','IFI44L','IFI44','RNF213','SP100','ISG20','ADAR','SP110',

'CD3D','CD3E','TRAC','IL7R','CCR7','LTB','TCF7','LEF1','S1PR1','SELL','MAL','IL32','ETS1',
'MS4A1','CD79A','CD79B','BANK1','CD22','TCL1A','IGKC','IGLC2','IGLC3',
'MZB1','JCHAIN','XBP1','PRDM1','TNFRSF17','DERL3','SSR4','IGHG1','IGHA1',

t=time.time()
with cellxgene_census.open_soma(census_version='latest') as census:
    src=cellxgene_census.get_anndata(census,'Homo sapiens',obs_value_filter=\"dataset_id == 'c7775e88-49bf-4ba2-a03b-93f00447c958'\",var_value_filter=filt,obs_column_names=['donor_id','disease'],var_column_names=['feature_id','feature_name'])

keys=sorted(obs['key'].unique())
key_to_i={k:i for i,k in enumerate(keys)}
codes=np.array([key_to_i[k] for k in obs['key']], dtype=np.int32)

cell_counts=np.bincount(codes, minlength=len(keys))
lib=S.sum(axis=1)

src_names=src.var['feature_name'].astype(str).values

print('saved source summaries', len(keys), len(src_names))
print(pd.DataFrame({'key':keys,'cells':cell_counts}).to_string(index=False))
