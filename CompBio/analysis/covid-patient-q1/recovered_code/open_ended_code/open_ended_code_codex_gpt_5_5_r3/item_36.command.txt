/bin/bash -lc "python - <<'PY'
import numpy as np, pandas as pd
from sklearn.metrics import silhouette_score
from sklearn.cluster import KMeans

d=np.load('donor_pseudobulk.npz', allow_pickle=True)
sums=d['sums']; names=d['names'].astype(str); donors=d['donors'].astype(str); lib=d['lib']; detected=d['detected']; cell_counts=d['cell_counts']
ids=np.array([int(x.replace('donor','')) for x in donors]); idx={g:i for i,g in enumerate(names)}
cpm=sums/lib[:,None]*1e6; logcpm=np.log1p(cpm)
ref=pd.read_csv('pascient_repo/data/ref_genes_covid.csv', header=None)[1].astype(str).tolist()
ref_idx=[idx[g] for g in ref if g in idx]
print('ref genes present', len(ref_idx), 'of', len(ref))
score=logcpm[:,ref_idx].mean(axis=1)
pct=(detected[:,ref_idx]/cell_counts[:,None]).mean(axis=1)
df=pd.DataFrame({'id':ids,'score':score,'pct':pct}).sort_values('score')
print(df.to_string(index=False))
# also score IFN subset from ref
for geneset_name, genes in [('ref_isg',[g for g in ref if g in ['LY6E','RNF213','SP100','ISG15','ISG20','XAF1','IFI6','SP110','ADAR','MX1','IFI44L','OAS1','IFIT1','STAT1']]),('ref_healthy_t',[g for g in ref if g in ['SELL','LTB','MAL','IL32','ETS1','TCF7','LEF1','CCR7','IL7R']])]:
    gi=[idx[g] for g in genes if g in idx]
    s=logcpm[:,gi].mean(axis=1)
    print('\\n',geneset_name, genes, 'present', len(gi))
    print(pd.DataFrame({'id':ids,geneset_name:s}).sort_values(geneset_name).to_string(index=False))
PY"
