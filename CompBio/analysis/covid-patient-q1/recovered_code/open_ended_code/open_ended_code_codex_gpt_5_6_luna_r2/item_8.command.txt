/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
 cats=f['obs/donor_id/categories'].asstr()[:]
 codes=f['obs/donor_id/codes'][:]
 print('donors',len(cats), list(enumerate(cats)))
 counts=np.bincount(codes,minlength=len(cats))
 print('counts')
 for i,(c,n) in enumerate(zip(cats,counts)): print(i,repr(c),int(n))
 genes=f['var/feature_name/categories'].asstr()[:]
 gcodes=f['var/feature_name/codes'][:]
 varidx=f['var/_index'].asstr()[:]
 print('var index first/last',varidx[:10],varidx[-10:])
 print('feature names first/last',genes[gcodes[:10]],genes[gcodes[-10:]])
 print('unique names',len(set(genes[gcodes])))
PY"
