/bin/bash -lc "python - <<'PY'
import requests, json, re
base='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main?recursive=true&expand=false'
items=[]
next_url=base
while next_url:
    r=requests.get(next_url,timeout=60)
    print('fetch',next_url[:120],r.status_code,len(r.text))
    data=r.json()
    items.extend(data)
    link=r.headers.get('Link','')
    m=re.search(r'<(["'^>]+)>;'"\\s*rel=\"next\"',link)
    next_url=m.group(1) if m else None
    if len(items)>50000: break
print('total items',len(items))
matches=[x for x in items if 'covid' in x.get('path','').lower()]
print('matches',len(matches))
for x in matches[:300]: print(x['path'],x['type'],x.get('size'))
# save list
import pickle
pickle.dump(items,open('traces_items.pkl','wb'))
PY"
