/bin/bash -lc "python - <<'PY'
import requests, pandas as pd, io
base='https://huggingface.co/datasets/amanutej/trustworthy-biology-agents-traces/resolve/main/'
for f in ['index.parquet','traces/index.parquet','traces/tasks/index.parquet','traces/derived/index.parquet']:
 r=requests.get(base+f,timeout=30)
 print(f,r.status_code,r.headers.get('content-length'))
 if r.ok:
  try:
   df=pd.read_parquet(io.BytesIO(r.content))
   print(df.head(), df.columns)
  except Exception as e: print(e)
PY"
