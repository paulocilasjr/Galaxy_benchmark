if ids:
 res=requests.post('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',data={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=60).json()['result']
 for uid in ids:
  it=res[uid]; print(it.get('accession'),len(it.get('samples',[])),it.get('title'))
PY"
