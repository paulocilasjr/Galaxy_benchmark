base='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main'

while next_url:
 r=requests.get(next_url,timeout=30)
 data=r.json()

 items.extend(data)

 print('got',len(data), 'last',data[-1]['path'] if data else '')
 break
matches=[x for x in items if 'covid' in x.get('path','').lower()]

for x in matches[:200]: print(x['path'],x['type'],x.get('size'))
PY"
