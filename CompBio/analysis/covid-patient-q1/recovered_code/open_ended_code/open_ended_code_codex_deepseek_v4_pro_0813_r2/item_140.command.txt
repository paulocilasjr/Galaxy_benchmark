/bin/bash -lc "python - <<'PY'
import requests,re
url='https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE161918&targ=gsm&form=text&view=brief'
text=requests.get(url,timeout=30).text
parts=text.split('"'^SAMPLE = '"')[1:]
print('samples',len(parts))
# summarize unique sample title patterns and characteristics
for part in parts[:20]:
 s=part.split('\\n',1)[0].strip()
 title=re.search(r'"'!Sample_title = ([''^'"\\n]+)',part)
 chars=re.findall(r'"'!Sample_characteristics_ch1 = ([''^'"\\n]+)',part)
 print(s,title.group(1) if title else '', chars[:8])
PY"
