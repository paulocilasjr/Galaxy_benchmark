/bin/bash -lc "python - <<'PY'
import requests
url='https://zenodo.org/api/records/19443186'
r=requests.get(url,timeout=30)
print(r.status_code)
data=r.json()
print(data.get('metadata',{}).get('title'))
print(data.get('metadata',{}).get('description','')[:2000])
files=data.get('files',[])
for f in files: print(f.get('key'), f.get('size'))
PY"
