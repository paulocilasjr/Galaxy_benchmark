/bin/bash -lc "python - <<'PY'
import requests,re
for gse in ['GSE263817','GSE218131']:
 url=f'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc={gse}&targ=gsm&form=text&view=brief'
 text=requests.get(url,timeout=30).text
 parts=text.split('"'^SAMPLE = '"')[1:]
 print('\\n',gse,'samples',len(parts))
 for part in parts:
  s=part.split('\\n',1)[0].strip()
  title=re.search(r'"'!Sample_title = ([''^'"\\n]+)',part)
  chars=re.findall(r'"'!Sample_characteristics_ch1 = ([''^'"\\n]+)',part)
  supp=re.findall(r'"'!Sample_supplementary_file_'"\\d+ = (.*)',part)
  print(s, title.group(1) if title else '', chars)
  if supp: print(' supp', supp[:2])
PY"
