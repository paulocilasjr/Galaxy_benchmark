/bin/bash -lc "python - <<'PY'
import requests,re
url='https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE163668&targ=gsm&form=text&view=brief'
text=requests.get(url,timeout=30).text
samples=re.findall(r'\\"'^SAMPLE = (GSM'"\\d+)',text)
print('count',len(samples))
for s in samples:
    m=re.search(r'\\"'^SAMPLE = '"'+s+r'\\n(.*?)(?=\\"'^SAMPLE|'"\\Z)', text, re.S)
    block=m.group(1)
    title_m=re.search(r'"'!Sample_title = (.*)'"',block)
    title=title_m.group(1) if title_m else ''
    chars=re.findall(r'"'!Sample_characteristics_ch1 = (.*)'"',block)
    supp=re.findall(r'"'!Sample_supplementary_file_'"\\d+ = (.*)',block)
    if 'GEX' in title or 'RNA' in title or any('GEX' in x for x in supp) or 'matrix' in title.lower():
      print(s,title,chars)
PY"
