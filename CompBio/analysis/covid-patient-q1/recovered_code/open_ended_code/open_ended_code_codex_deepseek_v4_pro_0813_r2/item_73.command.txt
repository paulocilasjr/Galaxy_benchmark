/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, pandas as pd, time
url='https://cellxgene-census-public-us-west-2.s3.us-west-2.amazonaws.com/cell-census/2025-11-08/h5ads/c7775e88-49bf-4ba2-a03b-93f00447c958.h5ad'
f=h5py.File(url,'r',driver='ros3',aws_region=b'us-west-2')
donor_codes=f['obs/donor_id/codes'][:]
donor_cats=f['obs/donor_id/categories'][:].astype(str)
disease_codes=f['obs/disease/codes'][:]
disease_cats=f['obs/disease/categories'][:].astype(str)
status_codes=f['obs/Status/codes'][:]
status_cats=f['obs/Status/categories'][:].astype(str)
counts=pd.Series(donor_codes).value_counts().sort_index()
df=pd.DataFrame({'n':counts.values}, index=donor_cats)
# disease per donor mode
for i,d in enumerate(donor_cats):
    df.loc[d,'disease']=disease_cats[disease_codes[donor_codes==i].min()]
    df.loc[d,'Status']=status_cats[status_codes[donor_codes==i].min()]
print('num donors',len(df))
print(df.to_string())
# compare our counts sequence sorted numeric

print('our sorted numeric counts', our)
print('Stephenson counts sorted?')
print(sorted(df['n'].tolist()))
PY"
