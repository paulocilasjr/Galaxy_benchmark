/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
D=np.load('cell_scores.npz', allow_pickle=True)
cats=D['cats']; df={}
for k in D.keys():
 if k"'!='"'cats': df[k]=D[k]
df=pd.DataFrame(df,index=cats)
for k in ['type_CD14_Mono','type_CD16_Mono','type_T','type_NK','type_B','type_Plasma','type_cDC','type_pDC','type_Platelet','type_RBC']:
 df[k+'_prop']=df[k]/df['total_cells']

df['ISG_all']=df['ISG_all_sum']/df['total_cells']
df['lymph']=df[['type_T_prop','type_NK_prop','type_B_prop','type_Plasma_prop']].sum(axis=1)
df['mono']=df[['type_CD14_Mono_prop','type_CD16_Mono_prop']].sum(axis=1)
df['NK_T_ratio']=df['type_NK_prop']/(df['type_T_prop']+1e-6)
for col in ['lymph','mono','type_T_prop','type_NK_prop','type_B_prop','type_Plasma_prop','type_CD16_Mono_prop','ISG_mono']:
 print('\\n',col,'low to high')
 print(df[col].sort_values().round(3).to_string())
print('\\nsummary')
print(df[['ISG_mono','ISG_all','lymph','mono','type_T_prop','type_NK_prop','type_B_prop','type_Plasma_prop','type_CD16_Mono_prop']].sort_values('ISG_mono').round(3).to_string())
PY"
