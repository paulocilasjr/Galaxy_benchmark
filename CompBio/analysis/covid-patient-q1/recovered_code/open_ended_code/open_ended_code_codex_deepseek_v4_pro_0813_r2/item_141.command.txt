/bin/bash -lc "python - <<'PY'
import numpy as np, pandas as pd
from sklearn.mixture import GaussianMixture
D=np.load('cell_scores.npz',allow_pickle=True); cats=D['cats']
df={k:D[k] for k in D.keys() if k"'!='"'cats'}
df=pd.DataFrame(df,index=cats)
df['ISG_mono']=df['ISG_mono_sum']/df['cnt_mono']
df['ISG_all']=df['ISG_all_sum']/df['total_cells']
# fit GMM on ISG_all, ISG_mono, ISG_T, ISG_B
X=df[['ISG_all','ISG_mono','ISG_T','ISG_B']].values
for cov in ['full','diag','tied','spherical']:
 for seed in range(3):
  g=GaussianMixture(n_components=2,covariance_type=cov,n_init=10,random_state=seed).fit(X)
  lab=g.predict(X)
  groups=[sorted(int(d[5:]) for d,l in zip(cats,lab) if l==k) for k in [0,1]]
  print(cov,seed,groups, 'means',g.means_[:,0].round(2))
  break
PY"
