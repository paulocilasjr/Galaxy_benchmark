/bin/bash -lc "python - <<'PY'
import urllib.request,json
for path in ['traces','traces/capability']:
 for qs in ['?recursive=true&limit=100000','?recursive=false&limit=100000']:
  u='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/'+path+qs
  try:
   x=json.load(urllib.request.urlopen(u)); print(path,qs,'n',len(x),'next?',x[-1] if x and 'cursor' in x[-1] else '')
   ps=[y.get('path','') for y in x]
   print('comp',sum('compbio' in p.lower() for p in ps),'covid',sum('covid' in p.lower() for p in ps),'samplelast',ps[-3:])
  except Exception as e: print(e)
PY"
