import numpy as np
from scipy.stats import ttest_ind, mannwhitneyu
z=np.load('donor_aggregate.npz',allow_pickle=True); S=z['sums']; d=z['donor_names'].astype(str); g=z['gene_names'].astype(str); u=z['donor_umis']

 print('\\n',typ)

 for i in ixs:

  if n>=50: break
# means cell marker genes groups

 ix=np.where(g==q)[0]
 if len(ix):

PY"
