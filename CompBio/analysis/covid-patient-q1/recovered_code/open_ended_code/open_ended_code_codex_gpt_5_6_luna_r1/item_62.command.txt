/bin/bash -lc "python - <<'PY'
import h5py
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
 def rec(g,prefix='',d=0):
  for k,v in g.items():
   path=prefix+'/'+k
   print(path, type(v).__name__, getattr(v,'shape',''), getattr(v,'dtype',''), dict(v.attrs))
   if isinstance(v,h5py.Group) and d<2: rec(v,path,d+1)
 rec(f)
PY"
