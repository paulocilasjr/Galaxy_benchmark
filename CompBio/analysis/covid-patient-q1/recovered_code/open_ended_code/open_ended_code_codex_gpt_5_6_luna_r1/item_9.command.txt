import h5py
p='covid.patients.q1.h5ad'
with h5py.File(p,'r') as f:
 for gname in ['obs/donor_id','var/feature_types','var/feature_name','var/feature_length','var/feature_biotype','var/feature_reference','var/feature_type']:
  o=f[gname]
  print('\\n',gname, type(o), list(o.keys()) if isinstance(o,h5py.Group) else o.shape, o.dtype if isinstance(o,h5py.Dataset) else '')
  if isinstance(o,h5py.Group):
   for k in o.keys():
    d=o[k]; print(' ',k,d.shape,d.dtype,d[:20] if d.size<100 else d[:10])
  else:
   print(o[:20])
 print('obs index first',f['obs/_index'][:10])
 print('var index first',f['var/_index'][:20])
PY"
