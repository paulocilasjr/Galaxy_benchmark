/bin/bash -lc "python -u - <<'PY'
import csv,gzip,io,tarfile
with tarfile.open('GSE201088_RAW.tar','r') as tf:
 for m in tf:
  if 'GSM6050543' in m.name and 'RSEC_MolsPerCell' in m.name:
   rd=csv.reader(io.TextIOWrapper(gzip.GzipFile(fileobj=tf.extractfile(m)),encoding='utf8',errors='replace'))
   h=None;n=0
   for row in rd:
    if not row or row[0].startswith('#'): continue
    if h is None: h=row; continue
    n+=1
   print(n,len(h))
PY"
