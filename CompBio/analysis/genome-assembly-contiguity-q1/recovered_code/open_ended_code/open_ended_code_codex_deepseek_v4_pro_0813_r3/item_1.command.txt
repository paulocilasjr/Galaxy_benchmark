/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
path=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
lengths=[]
cur=0
with path.open('r') as f:
    for line in f:
        if line.startswith('>'):
            if cur:
                lengths.append(cur)
                cur=0
        else:
            cur += len(line.strip())
    if cur:
        lengths.append(cur)
lengths.sort(reverse=True)
total=sum(lengths)
cum=0
n50=None
for l in lengths:
    cum+=l
    if cum*2 >= total:
        n50=l
        break
print('num_seqs',len(lengths))
print('total',total)
print('max',lengths[0])
print('min',lengths[-1])
print('n50',n50)
print('answer')
print(n50)
PY"
