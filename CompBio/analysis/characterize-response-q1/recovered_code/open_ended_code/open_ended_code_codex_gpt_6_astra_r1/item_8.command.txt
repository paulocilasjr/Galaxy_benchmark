/bin/bash -lc "PYTHONPATH=.packages python - <<'PY'
import requests,bs4,concurrent.futures
urls=['https://www.google.com/search?q=LIF+monocytes+Ifitm1&udm=14','https://www.bing.com/search?q=%22LIF%22+%22monocytes%22+%22Ifitm1%22','https://html.duckduckgo.com/html/?q=%22LIF%22+%22monocytes%22+%22Ifitm1%22']
def f(u):
 r=requests.get(u,timeout=30); h=bs4.BeautifulSoup(r.text,'html.parser')
 for s in h(['script','style']):s.decompose()
 return u,r.status_code,h.get_text(' ',strip=True)
for x in concurrent.futures.ThreadPoolExecutor().map(f,urls):print(x)
PY"
