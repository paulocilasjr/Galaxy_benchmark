/bin/bash -lc "PYTHONPATH=.packages python - <<'PY'
import openpyxl,collections
q=open('characterize.response.q1.txt').read().split();qs=set(q)
w=openpyxl.load_workbook('cytokine_signatures.xlsx',read_only=True,data_only=True)
res=[]
for s in w:
 groups=collections.defaultdict(list)
 for r in list(s.values)[1:]:groups[(r[3],r[5]>0)].append((r[4],r[5]))
 for (cy,sign),rows in groups.items():
  genes={r[0] for r in rows}; n=len(qs&genes)
  res.append((n/len(qs|genes),n,s.title,cy,sign,len(genes)))
  if cy=='LIF' and s.title=='Monocyte':print('LIF mono',rows)
print('BEST',*sorted(res,reverse=True)[:20],sep='\\n')
PY"
