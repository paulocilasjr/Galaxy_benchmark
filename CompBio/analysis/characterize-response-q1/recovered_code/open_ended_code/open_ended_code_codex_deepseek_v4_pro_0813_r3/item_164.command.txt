/bin/bash -lc "cat > cellmarkers.py <<'PY'
import glob, os
from scipy.stats import spearmanr
q=[l.strip().upper() for l in open('characterize.response.q1.txt') if l.strip()]; qset=set(q); qmap={g:i for i,g in enumerate(q)}
files=[f for f in glob.glob('all_libs/*.gmt') if any(x in os.path.basename(f) for x in ['CellMarker','Panglao','ARCHS4_Tissues','Descartes','Tabula_Sapiens','MAGNET','TISSUES_Experimental'])]
keys=['monocyte','dendritic','neutrophil','cd8','cd4','b cell','nk cell','natural killer','macrophage','myeloid']
rows=[]
for f in files:
 lib=os.path.basename(f)[:-4]
 for line in open(f,errors='ignore'):
  p=line.rstrip('\\n').split('\\t')
  if len(p)<3: continue
  term=p[0]; txt=term+' '+p[1]
  if not any(k in txt.lower() for k in keys): continue
  genes=[g.upper() for g in p[1:] if g.strip()]
  s=set(genes); ov=qset&s
  if not ov: continue
  seen=set(); og=[]
  for g in genes:
   if g in qset and g not in seen: seen.add(g); og.append(g)
  rho=None
  if len(og)>=3: rho,_=spearmanr([qmap[g] for g in og],list(range(len(og))))
  jac=len(ov)/(len(s)+112-len(ov))
  rows.append((len(ov),jac,rho,lib,term,len(genes)))
rows.sort(key=lambda r:(-r[0],-r[1]))
for x,jac,rho,lib,term,ntot in rows[:200]:
 print(f'{x:2d}/{ntot:<5d} jac={jac:.3f} rho={rho if rho is None else round(rho,2)} {lib:40s} | {term}')
PY
python3 cellmarkers.py > cellmarkers.txt
sed -n '1,220p' cellmarkers.txt"
