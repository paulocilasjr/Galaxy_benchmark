import re, json
r=open('sample-select.js').read()

 hits=[]

  tool=''

  if tm: tool=tm.group(1)
  if kw.lower() in (title+' '+tool).lower():
   hits.append((title,tool))
 if hits:
  print('\\n###',kw,len(hits))
  for title,tool in hits[:20]: print(title,'::',tool.replace('\\n',' ')[:300])
