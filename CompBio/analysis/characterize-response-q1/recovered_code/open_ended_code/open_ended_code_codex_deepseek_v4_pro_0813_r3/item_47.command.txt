/bin/bash -c "python3 - <<'PY'
import glob
q=set(g.upper() for g in open('characterize.response.q1.txt').read().split())
print('q',len(q))
for path in glob.glob('libs/*.gmt'):
 n=0; intercnt=0; sample=None
 for line in open(path):
  p=line.rstrip('\\n').split('\\t')
  if len(p)<3: continue
  genes=[]
  for x in p[2:]:
   if x: genes.append(x.split(',')[0].upper())
  n+=1
  if q & set(genes):
   intercnt+=1
   if not sample: sample=p[0]
 print(path, 'terms',n,'with overlap',intercnt,'sample',sample)
PY"
