/bin/bash -c "python3 - <<'PY'
import requests, gzip, io
url='https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL3nnn/GPL339/annot/GPL339.annot.gz'
r=requests.get(url,timeout=60)
print(r.status_code, len(r.content), r.url)
if r.status_code==200:
    open('GPL339.annot.gz','wb').write(r.content)
    import gzip
    with gzip.open('GPL339.annot.gz','rt') as f, open('GPL339.annot.txt','w') as out:
        out.write(f.read())
    print('saved')
PY
head -20 GPL339.annot.txt"
