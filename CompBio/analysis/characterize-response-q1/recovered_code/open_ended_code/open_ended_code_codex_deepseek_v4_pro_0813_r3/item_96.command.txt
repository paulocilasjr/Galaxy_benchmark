/bin/bash -c "python3 - <<'PY'
import requests,gzip,os
for gse in ['GSE17721','GSE14769','GSE2706','GSE3982','GSE22103','GSE63739']:
 url=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{gse[:5]}nnn/{gse}/matrix/{gse}_series_matrix.txt.gz'
 # path actually GSE17721 -> GSE17nnn etc; use actual directories? pattern for GSE2706 is GSE2nnn, GSE3982 GSE3nnn; GSE22103 GSE22nnn; GSE63739 GSE63nnn
 r=requests.get(url,timeout=120)
 print(gse,r.status_code,len(r.content))
 if r.status_code==200:
  gz=f'{gse}.txt.gz'; open(gz,'wb').write(r.content)
  with gzip.open(gz,'rt') as f, open(f'{gse}.txt','w') as out: out.write(f.read())
PY
ls -lh GSE*.txt"
