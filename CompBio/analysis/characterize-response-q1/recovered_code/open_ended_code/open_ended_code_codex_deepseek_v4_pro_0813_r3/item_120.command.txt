cat > analyze_libs.py <<'PY'
import os, glob, math
q=[l.strip().upper() for l in open('characterize.response.q1.txt') if l.strip()]
qs=set(q)
rows=[]
files=glob.glob('all_libs/*.gmt')
for f in files:
    lib=os.path.basename(f)[:-4]
    with open(f, errors='ignore') as fh:
      for line in fh:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<3: continue
        term=p[0]
        genes=[g.upper() for g in p[1:] if g.strip()]
        # The second field is desc but can be long URL? It won't match q mostly; genes include desc too, but no harm? could match query if description contains? no spaces not.
        s=set(genes)
        ov=qs & s
        if not ov: continue
        pos=[q.index(g) for g in ov if g in q]
        rows.append((len(ov), sum(pos)/len(pos), lib, term, sorted(pos), len(genes)))
rows.sort(key=lambda r:(-r[0], r[1]))
print('top 150 by overlap')
for x,mean,lib,term,pos,ntot in rows[:150]:
    print(f'{x:2d}/{ntot:<5d} meanpos={mean:6.1f} {lib:45s} | {term}')
print('\\nTOP mean pos overlap>=8')
f=sorted([r for r in rows if r[0]>=8], key=lambda r:r[1])
for x,mean,lib,term,pos,ntot in f[:150]:
    print(f'{x:2d}/{ntot:<5d} meanpos={mean:6.1f} {lib:45s} | {term}')
PY
python3 analyze_libs.py > libs_analysis.txt
wc -l libs_analysis.txt
sed -n '1,320p' libs_analysis.txt"
