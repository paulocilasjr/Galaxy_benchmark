/bin/bash -lc "cat > [REDACTED LOCAL PATH] <<'PY'
import mygene, json, sys
mg=mygene.MyGeneInfo()
genes=[l.strip() for l in open('characterize.response.q1.txt') if l.strip()]
print('n genes',len(genes))
out=mg.querymany(genes, scopes='symbol', species='mouse', fields='symbol,ensembl.gene,taxid,ortholog', verbose=False)
# querymany returns list
mp={}
for d in out:
    q=d.get('query'); sym=d.get('symbol'); orth=d.get('ortholog')
    hits=[]
    if isinstance(orth,dict):
        hits=[orth]
    elif isinstance(orth,list):
        hits=orth
    hum=None
    for o in hits:
        if o.get('taxid')==9606:
            hum=o.get('symbol') or o.get('name')
            break
    mp[q]=hum
    print(q, sym, hum)
json.dump(mp, open('mouse_to_human.json','w'))
print('mapped',sum(v is not None for v in mp.values()),'/',len(mp))
