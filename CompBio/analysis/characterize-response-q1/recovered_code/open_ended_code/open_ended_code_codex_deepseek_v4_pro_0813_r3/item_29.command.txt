/bin/bash -c "python3 - <<'PY'
import requests
urls=['https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL3nnn/GPL339/soft/GPL339.soft.gz','https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL339/soft/GPL339.soft.gz','https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL339/annot/GPL339.annot.gz','https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL339/soft/GPL339_family.soft.gz']
for u in urls:
 r=requests.get(u,timeout=30); print(u,r.status_code,len(r.content),r.headers.get('content-type'))
 if r.status_code==200:
  open('GPL339_soft.gz','wb').write(r.content); break
PY"
