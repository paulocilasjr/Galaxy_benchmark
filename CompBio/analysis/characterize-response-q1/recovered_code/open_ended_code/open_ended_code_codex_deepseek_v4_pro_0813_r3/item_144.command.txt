/bin/bash -lc "python3 - <<'PY'
import os, glob, math
from scipy.stats import spearmanr
q=[l.strip().upper() for l in open('characterize.response.q1.txt') if l.strip()]
qset=set(q); qmap={g:i for i,g in enumerate(q)}
# list of condition keyword sets
keywords=['LPS','lipopolysaccharide','exhaust','LIF','IL-39','IL39','COVID','sepsis','MED12','lupus','SLE','BMS-34554','aldesleukin','IL-2','aging','rheumatoid','arthritis']
terms=[]
for f in glob.glob('all_libs/*.gmt'):
 lib=os.path.basename(f)[:-4]
 for line in open(f,errors='ignore'):
  p=line.rstrip('\\n').split('\\t')
  if len(p)<3: continue
  term=p[0]; desc=p[1]
  text=term+' '+desc
  genes=[g.upper() for g in p[1:] if g.strip()]
  s=set(genes); ov=qset&s
  if not ov: continue
  # rank corr on term order
  seen=set(); og=[]
  for g in genes:
   if g in qset and g not in seen: seen.add(g); og.append(g)
  rho=None
  if len(og)>=3:
   rho,_=spearmanr([qmap[g] for g in og],list(range(len(og))))
  jac=len(ov)/(len(s)+112-len(ov))
  terms.append((len(ov),jac,rho,lib,term,desc,og))
# print per keyword groups, top by overlap/jac
for kw in keywords:
 print('\\n### KEYWORD',kw)
 hits=[t for t in terms if kw.lower() in (t[3]+' '+t[4]+' '+t[5]).lower()]
 # sort by overlap then jac
 hits.sort(key=lambda t:(-t[0],-t[1]))
 for x,jac,rho,lib,term,desc,og in hits[:15]:
  print(f'{x:2d}/{jac:.3f} rho={rho if rho is None else round(rho,2)} {lib:50s} | {term}')
PY"
