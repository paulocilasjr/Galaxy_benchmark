/bin/bash -c "python3 - <<'PY'
import mygene, re, os, math
mg=mygene.MyGeneInfo()
genes=open('characterize.response.q1.txt').read().split()
res=mg.querymany(genes, scopes='symbol', species='mouse', fields='agr.orthologs,symbol', returnall=True, verbose=False)
human=[]
for out in res['out']:
    hs=set()
    for o in out.get('agr',{}).get('orthologs',[]):
        if o.get('taxid')==9606 and o.get('symbol'):
            hs.add(o['symbol'].upper())
    human.append((out['query'], hs))
print('mapped human',sum(bool(h) for _,h in human),'of',len(genes))
# print unmapped
for g,h in human:
 if not h: print('no human',g)
# parse c7
q=set()
for g,h in human: q |= h
terms=[]
for line in open('c7.immunesigdb.v2024.1.Hs.symbols.gmt'):
 p=line.rstrip('\\n').split('\\t')
 if len(p)<3: continue
 name=p[0]; gs={x.upper() for x in p[2:] if x}
 inter=q & gs
 if inter:
  terms.append((len(inter), name, inter, len(gs)))
terms.sort(key=lambda x:-x[0])
print('top C7 terms:')
for t in terms[:100]:
 print(t[0], t[1], 'size',t[3], 'genes',','.join(t[2][:20]))
# filter option keywords
conds={'LPS':r'LPS|LIPOPOLYSACCHARIDE','exhaustion':r'EXHAUST','LIF':r'LIF|LEUKEMIA_INHIBITORY','IL-39':r'IL39|INTERLEUKIN_39','COVID':r'COVID|SARS_COV|CORONAVIRUS','sepsis':r'SEPSIS|SEPTIC','Med12':r'MED12','SLE':r'LUPUS|SLE|SYSTEMIC_LUPUS','BMS':r'BMS|34554','aldesleukin':r'ALDESLEUKIN|IL2|INTERLEUKIN_2','aging':r'AG(?:E|ING)|AGING','RA':r'RHEUMATOID|ARTHRITIS'}
cells={'monocytes':r'MONOCYTE|MONO','CD8':r'CD8|CYTOTOXIC_T','CD4':r'CD4|TCELL|T_CELL|TH1|TH2','B':r'B_CELL|BCELL|B_LYMPH|FOLLICULAR_B','DC':r'DENDRITIC|DC_|PDC','NK':r'NKCELL|NK_CELL|NATURAL_KILLER','Neutrophils':r'NEUTROPHIL|GRANULOCYTE'}
print('\\nfiltered options:')
for t in terms:
 n,name,inter,size=t
 ch=[c for c,pat in conds.items() if re.search(pat,name)]
 cl=[c for c,pat in cells.items() if re.search(pat,name)]
 if ch and cl:
  print(n,name,'size',size,'cond',ch,'cell',cl, 'genes',','.join(inter[:25]))
PY"
