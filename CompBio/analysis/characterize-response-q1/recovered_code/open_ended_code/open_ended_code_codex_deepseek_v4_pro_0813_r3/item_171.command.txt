/bin/bash -lc "python3 - <<'PY'
import geo_de, pickle
q=[l.strip().upper() for l in open('characterize.response.q1.txt') if l.strip()]; qset=set(q); qmap={g:i+1 for i,g in enumerate(q)}
cache=pickle.load(open('probe_symbol_cache.pkl','rb')); pmap={k:v.upper() for k,v in cache.items() if isinstance(v,str)}
m,ids,titles,mat=geo_de.parse_geo('GSE63739.txt')
A=[i for i,t in enumerate(titles) if t=='BM PMN_WT_LPS 3h']; B=[i for i,t in enumerate(titles) if t=='BM PMN_WT_LPS 0h']
res=geo_de.de_contrast(mat,A,B); agg=geo_de.aggregate_gene(res,pmap,qset); df=agg[agg.gene.isin(qset)].sort_values('logFC',ascending=False)
df['qpos']=[qmap[g] for g in df.gene]
print(df[['gene','qpos','logFC','t']].to_string(index=False))
PY"
