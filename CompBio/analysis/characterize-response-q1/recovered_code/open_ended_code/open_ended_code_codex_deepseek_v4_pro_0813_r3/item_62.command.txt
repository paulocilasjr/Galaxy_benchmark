/bin/bash -c "python3 - <<'PY'
import requests,gzip
for plat in ['GPL96','GPL97']:
 url=f'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/matrix/GSE22886-{plat}_series_matrix.txt.gz'
 r=requests.get(url,timeout=120)
 print(plat,r.status_code,len(r.content))
 if r.status_code==200:
  fn=f'GSE22886-{plat}_series_matrix.txt.gz'; open(fn,'wb').write(r.content)
  with gzip.open(fn,'rt') as f:
   text=f.read()
  open(f'GSE22886-{plat}_series_matrix.txt','w').write(text)
  print('saved')
PY
for f in GSE22886-*.txt; do echo ==="'$f; grep '"'"'^!Sample_title'"' "'$f | cut -c1-1000; done'
