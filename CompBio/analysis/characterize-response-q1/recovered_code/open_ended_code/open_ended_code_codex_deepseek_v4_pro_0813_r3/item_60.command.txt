/bin/bash -c "python3 - <<'PY'
import requests
for u in [
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/matrix/',
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/',
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/soft/GSE22886_family.soft.gz',
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/miniml/GSE22886_family.xml.tgz',
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE22nnn/GSE22886/suppl/']:
 r=requests.get(u,timeout=30)
 print(u,r.status_code,len(r.content),r.headers.get('content-type'))
 if r.status_code==200 and 'text/html' in r.headers.get('content-type',''):
  print(r.text[:500])
PY"
