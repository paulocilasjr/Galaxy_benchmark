python3 - <<'PY'
import json
mp=json.load(open('mouse_to_human3.json'))
q=list(dict.fromkeys(mp.values()))
qs=set(mp.values())
print('n q human',len(qs),len(q))
terms=[]
for line in open('c7.immunesigdb.v2024.1.Hs.symbols.gmt'):
    parts=line.rstrip('\\n').split('\\t')
    if len(parts)<3: continue
    name, desc=parts[0], parts[1]
    genes=[g for g in parts[2:] if g]
    terms.append((name,desc,genes))
print('terms',len(terms))
# filter terms with any overlap or keywords
keywords=['LPS','DC','MONO','NEUTRO','CD8','EXHAUST','IL2','ALDESLEUK','SLE','LUPUS','AGE','RHEUM','SEPSIS','BMS','LIF','IL39','COVID','MACROPHAGE','MYELOID','DENDRITIC']
for name,desc,genes in terms:
    s=set(genes)
    ov=qs & s
    if ov:
        keywords_hit=[k for k in keywords if k.lower() in (name+' '+desc).lower()]
        if keywords_hit:
            print(f'{len(ov):2d} {name} | {desc[:100]}')
PY"
