/bin/bash -c "python3 - <<'PY'
import requests, gzip, io
for url in ['https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL3nnn/GPL339/soft/GPL339_family.soft.gz','https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL3nnn/GPL339/soft/GPL339.annot.gz','https://ftp.ncbi.nlm.nih.gov/geo/platforms/GPL3nnn/GPL339/GPL339_full.soft.gz']:
 r=requests.get(url,timeout=60); print(url,r.status_code,len(r.content),r.url)
 if r.status_code==200: break
if r.status_code==200:
 open('GPL339_soft.gz','wb').write(r.content)
 import gzip
 txt=gzip.decompress(r.content).decode('utf8','ignore')
 open('GPL339_soft.txt','w').write(txt)
 print('saved', txt[:500])
PY"
