/bin/bash -lc "cat > search_libs_keywords.py <<'PY'
import os, glob
q=[l.strip().upper() for l in open('characterize.response.q1.txt') if l.strip()]
qs=set(q)
files=glob.glob('all_libs/*.gmt')
keywords=['LPS','lipopolysaccharide','exhaust','LIF','IL39','IL-39','COVID','sepsis','MED12','lupus','SLE','BMS-34554','aldesleukin','aging','rheumatoid','arthritis','neutrophil','dendritic','monocyte','macrophage','CD8','CD4','B cell','NK cell']
# collect matching terms with overlap
rows=[]
for f in files:
 lib=os.path.basename(f)[:-4]
 with open(f, errors='ignore') as fh:
  for line in fh:
   p=line.rstrip('\\n').split('\\t')
   if len(p)<3: continue
   term=p[0]
   # full text = term + p[1]
   txt=(term+' '+p[1]).lower()
   if not any(k.lower() in txt for k in keywords): continue
   genes=[g.upper() for g in p[1:] if g.strip()]
   s=set(genes)
   ov=qs & s
   if not ov: continue
   pos=[q.index(g) for g in ov if g in q]
   rows.append((len(ov), sum(pos)/len(pos), lib, term, p[1], sorted(pos), len(genes)))
# print all sorted
rows.sort(key=lambda r:(-r[0], r[1]))
for x,mean,lib,term,desc,pos,ntot in rows[:500]:
 print(f'{x:2d}/{ntot:<5d} meanpos={mean:6.1f} {lib:50s} | {term} | {desc[:60]}')
PY
python3 search_libs_keywords.py > search_libs_keywords.txt
wc -l search_libs_keywords.txt
sed -n '1,500p' search_libs_keywords.txt"
