/bin/bash -lc "python3 - <<'PY'
import requests
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE7nnn/GSE7219/matrix/GSE7219_series_matrix.txt.gz'
r=requests.get(url)
print(r.status_code, len(r.content), r.url)
if r.status_code==200:
 open('GSE7219_series_matrix.txt.gz','wb').write(r.content)
 import gzip, shutil
 with gzip.open('GSE7219_series_matrix.txt.gz','rb') as f, open('GSE7219_series_matrix.txt','wb') as o: shutil.copyfileobj(f,o)
 print('done')
else:
 print(r.text[:200])
PY
ls -lh GSE7219* 2>/dev/null"
