/bin/bash -lc "python - <<'PY'
import io, urllib.request, zipfile, xml.etree.ElementTree as ET, re
url='https://media.springernature.com/original/springer-static/esm/art%3A10.1038%2Fs41586-023-06816-9/MediaObjects/41586_2023_6816_MOESM5_ESM.xlsx'
z=zipfile.ZipFile(io.BytesIO(urllib.request.urlopen(url).read()))
NS='http://schemas.openxmlformats.org/spreadsheetml/2006/main'; ns={'m':NS}
root=ET.fromstring(z.read('xl/sharedStrings.xml'))
ss=[]
for si in root.findall('m:si',ns):
    ss.append(''.join(t.text or '' for t in si.findall('.//m:t',ns)))
print('shared strings',len(ss),ss[:20])
root=ET.fromstring(z.read('xl/worksheets/sheet3.xml'))
def val(c):
    v=c.find('m:v',ns); x='' if v is None else v.text
    if c.attrib.get('t')=='s' and x"'!='"'': return ss[int(x)]
    return x
rows=[]
for row in root.findall('.//m:row',ns):
    d={c.attrib['r']:val(c) for c in row.findall('m:c',ns)}
    vals=[]
    # by columns sorted numerical
    for ref,x in d.items():
      col=re.match('[A-Z]+',ref).group(); n=0
      for ch in col:n=n*26+ord(ch)-64
      vals.append((n,ref,x))
    rows.append((int(row.attrib['r']),vals))
print('rows',len(rows))
for rn, vals in rows[:12]: print(rn,[(ref,x) for n,ref,x in vals])
print('LIF matches')
for rn, vals in rows:
    if any('LIF' in x.upper() for n,ref,x in vals): print(rn,[(ref,x) for n,ref,x in vals])
PY"
