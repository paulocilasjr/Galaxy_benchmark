import pandas as pd
p='immune_dictionary_table3.xlsx'
target=[x.strip() for x in open('characterize.response.q1.txt') if x.strip()]

print(g.sort_values('Avg_log2FC',ascending=False)[['Gene','Avg_log2FC']].to_string(index=False))
print('missing',sorted(set(target)-set(g.Gene)))
print('extra target-like',sorted(set(g.Gene)-set(target)))
PY"
