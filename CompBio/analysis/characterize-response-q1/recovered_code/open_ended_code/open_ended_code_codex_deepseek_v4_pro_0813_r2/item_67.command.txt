/bin/bash -c "python3 - <<'PY'
import requests, os
# likely GSE2002 matrix
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE2nnn/GSE2002/matrix/GSE2002_series_matrix.txt.gz'
r=requests.get(url,timeout=60); print(r.status_code,len(r.content),r.url)
if r.ok:
 open('GSE2002_series_matrix.txt.gz','wb').write(r.content)
 os.system('gzip -dc GSE2002_series_matrix.txt.gz | head -30')
PY"
