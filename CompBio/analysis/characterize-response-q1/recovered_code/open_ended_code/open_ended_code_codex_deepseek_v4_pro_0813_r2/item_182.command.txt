/bin/bash -lc "python3 - <<'PY'
import requests, re, xml.etree.ElementTree as ET
# search GDS for Med12 immune terms
queries=['Med12 knockout immune','Med12 KO macrophage','Med12 dendritic','Med12 monocyte','MED12 CD8 T cell']
for q in queries:
 u='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gds&term='+requests.utils.quote(q)+'&retmax=30'
 r=requests.get(u)
 root=ET.fromstring(r.text)
 ids=[e.text for e in root.findall('.//Id')]
 print('\\n',q,'count',root.findtext('.//Count'),'ids',ids)
 if ids:
  s=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=gds&id='+','.join(ids[:20]))
  sr=ET.fromstring(s.text)
  for d in sr.findall('.//DocSum'):
   title=d.find(\".//Item[@Name='title']\").text if d.find(\".//Item[@Name='title']\") is not None else ''
   print(' ',d.find('Id').text,title[:150])
PY"
