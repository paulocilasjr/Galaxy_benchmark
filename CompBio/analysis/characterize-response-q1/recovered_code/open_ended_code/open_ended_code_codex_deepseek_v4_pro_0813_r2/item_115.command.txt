/bin/bash -lc "python3 - <<'PY'
import pandas as pd, scipy.stats as st
if False:
 pass
# load
xl=pd.ExcelFile('GSE78209_DE_analysis.xls')
df=xl.parse('DatasetEV8', header=[0,1], index_col=0)
sym=df[('Symbol','Unnamed: 1_level_1')].astype(str).str.strip()
for col in [('LPS 3 h','LFC'),('LPS 6 h','LFC')]:
    s=pd.Series(pd.to_numeric(df[col],errors='coerce'),index=sym)
    # sort by descending lfc, filter genes
    s=s.dropna()
    # unique symbol keep first? average
    s=s.groupby(s.index.str.upper()).mean()
    input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
    mapped=input.copy()
    ranks={g:i+1 for i,g in enumerate(input)}
    got=s[s.index.isin(set(input))].copy()
    # rank in dataset among all genes desc
    ranked=s.sort_values(ascending=False)
    rank={g:i+1 for i,g in enumerate(ranked.index)}
    print('\\n',col,'num input mapped',len(got),'median lfc',got.median())
    print('top 30 input lfc ranks:')
    for g in input[:30]:
        if g in s.index:
            print(g, 'lfc',round(s[g],3),'dataset rank',rank.get(g),'input rank',ranks[g])
        else:
            print(g,'not mapped')
    # spearman between input rank and dataset lfc for mapped
    a=pd.Series([ranks[g] for g in input if g in s.index])
    b=pd.Series([s[g] for g in input if g in s.index])
    print('spearman input rank vs lfc', st.spearmanr(a,b))
    # top 112 dataset genes (positive side) overlap
    top=ranked.head(112).index.tolist()
    print('overlap with top112 dataset',len(set(top)&set(input)), 'input among top112', [g for g in input if g in set(top)][:50])
PY"
