import pandas as pd, numpy as np
expr=pd.read_csv('GSE17721_expr.csv')
expr['probe']=expr.iloc[:,0].str.replace('\"','')
map=pd.read_csv('GSE17721_probe_map.csv')
# map columns; drop ID_REF
expr=expr.merge(map.rename(columns={'PROBEID':'probe'}),on='probe',how='left')
# group by SYMBOL uppercase mean of numeric cols
cols=[c for c in expr.columns if c not in ['ID_REF','probe','SYMBOL']]
expr['SYMBOL']=expr['SYMBOL'].str.upper()
out=expr.groupby('SYMBOL')[cols].mean().reset_index()
out.to_csv('GSE17721_symbol_expr.csv',index=False)
print(out.shape)
PY
