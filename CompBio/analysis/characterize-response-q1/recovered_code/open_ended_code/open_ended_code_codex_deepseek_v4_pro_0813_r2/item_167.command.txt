/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.stats as st
path='GSE233796_Med12_BMDCs.csv.gz'
df=pd.read_csv(path)
# aggregate gene name
ko=[f'DCs_Med12KO_{i}' for i in (1,2,3)]; nt=[f'DCs_NT_{i}' for i in (1,2,3)]
df['lfc']=np.log2((df[ko].mean(axis=1)+1)/(df[nt].mean(axis=1)+1))
s=df.groupby('Gene_name')['lfc'].mean()
s.index=s.index.str.upper()
input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
ri={g:i+1 for i,g in enumerate(input)}
got=s[s.index.isin(set(input))]
print('mapped',len(got))
a=[ri[g] for g in input if g in s.index]; b=[s[g] for g in input if g in s.index]
print('spearman input rank vs lfc',st.spearmanr(a,b))
ranked=s.sort_values(ascending=False); top=ranked.head(112).index.tolist()
print('overlap top112',len(set(top)&set(input)), [g for g in input if g in set(top)])
print('top input lfc:',[(g,round(s.get(g,float('nan')),2)) for g in input[:30] if g in s.index])
# top dataset genes
print('top dataset genes:',ranked.head(30).index.tolist())
PY"
