/bin/bash -lc "python3 - <<'PY'
import pandas as pd, scipy.stats as st
xl=pd.ExcelFile('GSE78209_DE_analysis.xls')
df=xl.parse('DatasetEV8', header=[0,1], index_col=0)
sym=df[('Symbol','Unnamed: 1_level_1')].astype(str).str.strip().values
for col in [('LPS 3 h','LFC'),('LPS 6 h','LFC')]:
    s=pd.Series(pd.to_numeric(df[col],errors='coerce').values,index=sym).dropna()
    s=s.groupby([x.upper() for x in s.index]).mean()
    input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
    ranks={g:i+1 for i,g in enumerate(input)}
    got=s[s.index.isin(set(input))]
    print('\\n',col,'mapped',len(got),'median',got.median())
    ranked=s.sort_values(ascending=False)
    rank={g:i+1 for i,g in enumerate(ranked.index)}
    for g in input[:30]:
        if g in s.index:
            print(g, round(s[g],3),'dataset rank',rank.get(g),'input rank',ranks[g])
        else: print(g,'not')
    a=[ranks[g] for g in input if g in s.index]; b=[s[g] for g in input if g in s.index]
    print('spearman',st.spearmanr(a,b))
    top=ranked.head(112).index.tolist()
    print('overlap top112',len(set(top)&set(input)), [g for g in input if g in set(top)])
    # bottom 112 (down)
    bot=ranked.tail(112).index.tolist()
    print('overlap bottom112',len(set(bot)&set(input)), [g for g in input if g in set(bot)])
PY"
