/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, math, scipy.stats as st
df=pd.read_csv('GSE49449_cpm.tsv.gz',sep='\\t')
print(df.columns)
# collapse symbol upper
for compname, groups in [
 ('KO_vs_WT_unstim', ['PEC Ptp1bko -IL10','PEC Ptp1bwt -IL10']),
 ('IL10_WT', ['PEC Ptp1bwt +IL10','PEC Ptp1bwt -IL10']),
 ('IL10_KO', ['PEC Ptp1bko +IL10','PEC Ptp1bko -IL10'])]:
    a=df[groups[0]]; b=df[groups[1]]
    lfc=np.log2((a+0.5)/(b+0.5))
    s=pd.Series(lfc.values,index=df['name'].str.upper()).groupby(level=0).mean()
    inp=[x.strip().upper() for x in open('characterize.response.q1.txt')]
    got=s[s.index.isin(set(inp))]
    print('\\n',compname, 'mapped',len(got),'median lfc',got.median())
    ranked=s.sort_values(ascending=False); rank={g:i+1 for i,g in enumerate(ranked.index)}
    print('input top 20 lfc/ranks:')
    for g in inp[:20]:
      print(g, round(s.get(g,float('nan')),3), rank.get(g,''), sep=' ')
    a=[i+1 for i,g in enumerate(inp) if g in s.index]; b=[s[g] for g in inp if g in s.index]
    print('spearman input rank vs lfc',st.spearmanr(a,b))
    top=ranked.head(112).index.tolist(); bot=ranked.tail(112).index.tolist()
    print('overlap top112',len(set(top)&set(inp)), [g for g in inp if g in set(top)])
    print('overlap bottom112',len(set(bot)&set(inp)), [g for g in inp if g in set(bot)])
PY"
