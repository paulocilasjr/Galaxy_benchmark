/bin/bash -c "python3 - <<'PY'
import requests, xml.etree.ElementTree as ET, re, time
ids=['GSE1871','GSE4479','GSE36700','GSE59184','GSE411','GDS1030','GDS4551','GSE3184','GSE1623','GSE10500']
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils'
for gid in ids:
    r=requests.get(base+'/esearch.fcgi',params={'db':'gds','term':gid,'retmode':'json'},timeout=20)
    uid=r.json().get('esearchresult',{}).get('idlist',[])
    print('\\n###',gid,'ids',uid)
    if not uid: continue
    r2=requests.get(base+'/esummary.fcgi',params={'db':'gds','id':','.join(uid[:5]),'retmode':'json'},timeout=20)
    try:
      res=r2.json()['result']
      for k in uid[:5]:
        it=res[k]
        print(k, '|', it.get('title'), '|', it.get('summary'), '|', it.get('gdsType'))
    except Exception as e: print(r2.text[:500])
    time.sleep(0.4)
PY"
