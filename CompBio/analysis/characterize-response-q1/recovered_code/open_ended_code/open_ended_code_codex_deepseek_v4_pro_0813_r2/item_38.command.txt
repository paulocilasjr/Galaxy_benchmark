/bin/bash -c "python3 - <<'PY'
import requests, gzip, io, os
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE233nnn/GSE233796/suppl/GSE233796_Med12_BMDCs.csv.gz'
r=requests.get(url,timeout=120)
print(r.status_code, len(r.content), r.url)
if r.status_code==200:
 open('GSE233796_Med12_BMDCs.csv.gz','wb').write(r.content)
 os.system('gzip -dc GSE233796_Med12_BMDCs.csv.gz | head -30')
PY"
