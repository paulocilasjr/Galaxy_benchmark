/bin/bash -c "python3 - <<'PY'
import requests, json
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils'
queries=[
 'aging[Title] AND monocyte[Title] AND Mus musculus[Organism] AND expression profiling[Filter]',
 'aged[Title] AND monocyte[Title] AND Mus musculus[Organism]',
 'aging[Title] AND dendritic cell[Title] AND Mus musculus[Organism]',
 'aging[Title] AND CD4 T cell[Title] AND Mus musculus[Organism]',
 'aging[Title] AND CD8 T cell[Title] AND Mus musculus[Organism]',
 'aging[Title] AND B cell[Title] AND Mus musculus[Organism]',
 'aging[Title] AND neutrophil[Title] AND Mus musculus[Organism]',
]
for q in queries:
 r=requests.get(base+'/esearch.fcgi',params={'db':'gds','term':q,'retmax':20,'retmode':'json'},timeout=20)
 js=r.json().get('esearchresult',{})
 print('\\nQ',q,'count',js.get('count'))
 ids=js.get('idlist',[])[:10]
 if ids:
  r2=requests.get(base+'/esummary.fcgi',params={'db':'gds','id':','.join(ids),'retmode':'json'},timeout=20)
  res=r2.json().get('result',{})
  for i in ids:
   it=res.get(i,{})
   print(i,'|',it.get('accession'),'|',it.get('title'),'|',(it.get('summary') or '')[:150])
PY"
