/bin/bash -c "python3 - <<'PY'
import requests, json, time
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils'
queries=[
 'LPS[Title] AND monocyte[Title] AND Mus musculus[Organism] AND expression profiling by high throughput sequencing[DataSet Type]',
 'lipopolysaccharide[Title] AND monocyte[Title] AND Mus musculus[Organism]',
 'LPS[Title] AND macrophage[Title] AND Mus musculus[Organism] AND expression profiling by high throughput sequencing[DataSet Type]',
 'lipopolysaccharide[Title] AND dendritic cell[Title] AND Mus musculus[Organism]',
 'lipopolysaccharide[Title] AND CD4[Title] AND Mus musculus[Organism]',
 'LPS[Title] AND B cell[Title] AND Mus musculus[Organism]',
]
for q in queries:
 r=requests.get(base+'/esearch.fcgi',params={'db':'gds','term':q,'retmax':50,'retmode':'json'},timeout=20)
 js=r.json().get('esearchresult',{})
 print('\\n',q,'count',js.get('count'))
 ids=js.get('idlist',[])[:20]
 if ids:
  # summarize titles
  r2=requests.get(base+'/esummary.fcgi',params={'db':'gds','id':','.join(ids[:20]),'retmode':'json'},timeout=20)
  res=r2.json().get('result',{})
  for i in ids:
   it=res.get(i,{})
   print(i, '|', it.get('accession'), '|', it.get('title'), '|', it.get('summary')[:120] if it.get('summary') else '')
  time.sleep(0.4)
 time.sleep(0.4)
PY"
