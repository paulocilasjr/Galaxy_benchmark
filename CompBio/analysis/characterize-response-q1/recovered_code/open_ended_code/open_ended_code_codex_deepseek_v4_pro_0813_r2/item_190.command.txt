/bin/bash -lc "python3 - <<'PY'
import requests, json
base='http://amp.pharm.mssm.edu/CREEDS/download'
for typ,fn in [('single_gene_perturbations-v1.0.json','creeds_gene.json'),('disease_signatures-v1.0.json','creeds_disease.json'),('drug_signatures-v1.0.json','creeds_drug.json')]:
 try:
  r=requests.get(f'{base}/{typ}',timeout=60)
  print(typ,r.status_code,len(r.content))
  if r.status_code==200:
   open(fn,'w').write(r.text)
 except Exception as e: print(e)
PY
ls -lh creeds*.json"
