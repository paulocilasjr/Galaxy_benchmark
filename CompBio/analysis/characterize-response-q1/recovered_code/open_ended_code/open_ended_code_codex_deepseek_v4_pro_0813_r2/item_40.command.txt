/bin/bash -c "python3 - <<'PY'
import requests, json, time, math
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
N=len(genes); gs=set(genes); rank={g:i+1 for i,g in enumerate(genes)}
libs=[
'Disease_Perturbations_from_GEO_up','Disease_Perturbations_from_GEO_down',
'Drug_Perturbations_from_GEO_2014','Drug_Perturbations_from_GEO_up','Drug_Perturbations_from_GEO_down',
'Ligand_Perturbations_from_GEO_up','Ligand_Perturbations_from_GEO_down',
'Gene_Perturbations_from_GEO_up','Gene_Perturbations_from_GEO_down',
'Aging_Perturbations_from_GEO_up','Aging_Perturbations_from_GEO_down',
'Microbe_Perturbations_from_GEO_up','Microbe_Perturbations_from_GEO_down',
'Kinase_Perturbations_from_GEO_up','Kinase_Perturbations_from_GEO_down',
'Virus_Perturbations_from_GEO_up','Virus_Perturbations_from_GEO_down',
'RNA-Seq_Disease_Gene_and_Drug_Signatures_from_GEO',
'MCF7_Perturbations_from_GEO_up','MCF7_Perturbations_from_GEO_down',
'CellMarker_Augmented_2021','CellMarker_2024','PanglaoDB_Augmented_2021','Azimuth_Cell_Types_2021','Descartes_Cell_Types_and_Tissue_2021','ARCHS4_Tissues'
]
allrows=[]
for lib in libs:
    try:
        r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
        if r.status_code"'!=200:
            print('"'bad',lib,r.status_code); continue
        terms=r.json()[lib]['terms']
    except Exception as e:
        print('err',lib,e); continue
    for term,vals in terms.items():
        vals=[v.upper() for v in vals]
        s=set(vals)
        inter=gs & s
        if len(inter) < 8: continue
        union=len(gs|s)
        jac=len(inter)/union
        # coverage: fraction of our genes present in term
        coverage=len(inter)/N
        # rank score: lower avg term position of our genes = better
        present=[g for g in genes if g in vals]
        avg=sum(vals.index(g)+1 for g in present)/len(present)
        # top-position score
        top=0
        for j,v in enumerate(vals[:N]):
            if v in rank: top += 1/(abs(j+1-rank[v])+1)
        # enrichment style simple hypergeom not needed
        allrows.append((len(inter),jac,coverage,top,-avg,lib,term,present))
allrows.sort(key=lambda x:(x[0], x[2], x[3]), reverse=True)
print('TOP OVERLAP COUNT')
for row in allrows[:60]:
    print(f'{row[0]} jac={row[1]:.3f} cov={row[2]:.2f} pos={row[3]:.2f} avgterm={-row[4]:.0f} {row[5]} | {row[6][:100]}')
print('\\nTOP COVERAGE')
allrows2=sorted(allrows,key=lambda x:(x[2],x[0],x[3]),reverse=True)
for row in allrows2[:60]:
    print(f'{row[0]} jac={row[1]:.3f} cov={row[2]:.2f} pos={row[3]:.2f} avgterm={-row[4]:.0f} {row[5]} | {row[6][:100]}')
PY"
