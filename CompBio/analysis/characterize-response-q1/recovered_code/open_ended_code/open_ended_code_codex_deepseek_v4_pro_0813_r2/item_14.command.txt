/bin/bash -c "python3 - <<'PY'
import requests, json, time, math
genes=open('characterize.response.q1.txt').read().strip().split('\\n')
gs=set(genes)
libs=['Disease_Perturbations_from_GEO_up','Disease_Perturbations_from_GEO_down','Ligand_Perturbations_from_GEO_up','Ligand_Perturbations_from_GEO_down','Gene_Perturbations_from_GEO_up','Gene_Perturbations_from_GEO_down','CellMarker_Augmented_2021','PanglaoDB_Augmented_2021','ARCHS4_Tissues']
for lib in libs:
    r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
    print('fetched',lib,r.status_code,len(r.text))
    if r.status_code"'!=200: continue
    data=r.json()[lib]['"'terms']
    rows=[]
    for term,vals in data.items():
        s=set(vals)
        inter=gs & s
        if len(inter)>=15:
            # Jaccard
            union=len(gs|s)
            jac=len(inter)/union
            # rank sum by gene positions? weighted top
            score=sum(1/(i+1) for i,g in enumerate(genes) if g in s)
            rows.append((len(inter),jac,score,term))
    rows.sort(reverse=True)
    print('Top for',lib)
    for row in rows[:30]:
        print(row)
    time.sleep(0.5)
PY"
