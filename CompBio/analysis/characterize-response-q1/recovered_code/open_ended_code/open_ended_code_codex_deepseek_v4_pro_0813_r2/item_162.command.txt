import gzip,csv,json
with gzip.open('GSE2002_series_matrix.txt.gz','rt') as f: lines=f.read().splitlines()
b=next(i for i,l in enumerate(lines) if l.startswith('"'!series_matrix_table_begin'"'))+1
e=next(i for i,l in enumerate(lines) if l.startswith('"'!series_matrix_table_end'"'))

data=[l.strip('\\n').split('\\t') for l in lines[b+1:e] if l.strip()]
with open('GSE2002_expr.csv','w',newline='') as f: w=csv.writer(f); w.writerow(h); w.writerows(data)
meta={}
for l in lines:
 if l.startswith('"'!Sample_title'"'):
  titles=[x.strip('\"') for x in l.split('\\t')[1:]]
 if l.startswith('"'!Sample_geo_accession'"'):
  gsms=[x.strip('\"') for x in l.split('\\t')[1:]]
for g,t in zip(gsms,titles): meta[g]=t
json.dump(meta,open('GSE2002_meta.json','w'))
print(h[:5],len(data),list(meta.items())[:5])
PY
