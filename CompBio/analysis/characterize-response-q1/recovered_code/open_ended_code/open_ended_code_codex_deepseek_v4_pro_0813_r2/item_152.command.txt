/bin/bash -lc "Rscript - <<'RS'
library(data.table)
library(mouse4302.db)
expr <- fread('GSE17721_expr.csv')
probes <- gsub('\"','',expr[[1]])
map <- select(mouse4302.db, keys=probes, columns='SYMBOL', keytype='PROBEID')
setDT(map)
expr[, probe := probes]
# numeric expression columns
numcols <- setdiff(names(expr), c('ID_REF','probe'))
for(j in numcols) set(expr, j=j, value=as.numeric(expr[[j]]))
m <- merge(expr, map, by='probe', allow.cartesian=TRUE)
# keep rows with non-NA SYMBOL
m <- m["'!is.na(SYMBOL)]
# average by uppercase symbol
m[, SYMBOL := toupper(SYMBOL)]
out <- m[, lapply(.SD, mean), by=SYMBOL, .SDcols=numcols]
fwrite(out, '"'GSE17721_symbol_expr.csv')
cat('symbols',nrow(out),'\\n')
RS
head -1 GSE17721_symbol_expr.csv | cut -c1-200"
