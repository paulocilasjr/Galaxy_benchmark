/bin/bash -c "python3 - <<'PY'
import requests, json, time
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
gs=set(genes)
libs=['Disease_Perturbations_from_GEO_up','Disease_Perturbations_from_GEO_down','Ligand_Perturbations_from_GEO_up','Ligand_Perturbations_from_GEO_down','Gene_Perturbations_from_GEO_up','Gene_Perturbations_from_GEO_down','CellMarker_Augmented_2021','PanglaoDB_Augmented_2021','ARCHS4_Tissues']
for lib in libs:
    r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
    if r.status_code"'!=200: print('"'bad',lib,r.status_code); continue
    data=r.json()[lib]['terms']
    rows=[]
    for term,vals in data.items():
        s=set(v.upper() for v in vals)
        inter=gs & s
        if len(inter)>=5:
            union=len(gs|s)
            jac=len(inter)/union
            # weighted top enrichment: sum 1/(rank+1) for genes present
            score=sum(1/(i+1) for i,g in enumerate(genes) if g in s)
            # also overlap count and genes
            overlap_genes=[g for g in genes if g in s]
            rows.append((len(inter),jac,score,term,overlap_genes))
    rows.sort(key=lambda x:(x[2]), reverse=True)
    print('\\n###',lib,'rows',len(rows))
    for row in rows[:20]:
        print(row[:4], 'genes=', ','.join(row[4][:20]))
PY"
