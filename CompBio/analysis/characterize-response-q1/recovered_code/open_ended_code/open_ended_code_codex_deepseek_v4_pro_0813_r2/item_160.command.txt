/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, json, scipy.stats as st
meta=json.load(open('GSE14769_meta.json'))
df=pd.read_csv('GSE14769_symbol_expr.csv').set_index('SYMBOL')
input=[x.strip().upper() for x in open('characterize.response.q1.txt')]
ri={g:i+1 for i,g in enumerate(input)}
# group samples by time; titles contain LPS1-0 etc. Extract time token after '-' or _LPS_xxx_min
def time_of(title):
    t=title.upper()
    if t.endswith('MIN') or '_MIN' in t:
        import re
        m=re.search(r'LPS[_A-Z-]*(\\d+)_?MIN',t)
        if m:return m.group(1)
    if '-' in t:
        # LPS1-120 etc
        return t.split('-')[1]
    return None
groups={}
for c in df.columns:
    t=meta.get(c,'')
    tp=time_of(t)
    groups.setdefault(tp,[]).append(c)
print(groups)
for tp in sorted(groups, key=lambda x:int(x) if x else -1):
    if tp is None or tp=='0': continue
    cols=groups[tp]
    zero=groups.get('0',[])
    if not zero: continue
    lfc=np.log2((df[cols].mean(axis=1)+1)/(df[zero].mean(axis=1)+1))
    s=lfc
    got=s[s.index.isin(set(input))]
    a=[ri[g] for g in input if g in s.index]; b=[s[g] for g in input if g in s.index]
    rho=st.spearmanr(a,b)[0]
    ranked=s.sort_values(ascending=False); top=ranked.head(112).index.tolist()
    print(f'time {tp} min: n lps {len(cols)} n0 {len(zero)}, mapped {len(got)}, spearman {rho:.3f}, overlap top112 {len(set(top)&set(input))}')
    if tp in ['120','240','360']:
      print('  lfc top input:', [(g, round(s.get(g,float('nan')),2)) for g in input[:20] if g in s.index])
PY"
