/bin/bash -c "Rscript - <<'RS'
suppressMessages(library(msigdbr))
genes <- toupper(readLines('characterize.response.q1.txt'))
for (org in c('Mus musculus','Homo sapiens')) {
  for (col in c('C7')) {
    m <- msigdbr(species=org, collection=col)
    cat('\\n###',org,col,nrow(m),'sets\\n')
    # aggregate by gs_name
    sets <- split(m"'$gene_symbol, m$gs_name)
    inter <- sapply(sets, function(s) length(intersect(genes,toupper(s))))
    # coverage fraction of our genes
    cover <- inter/length(genes)
    # weighted top rank
    rank <- seq_along(genes)
    weighted <- sapply(sets, function(s) { s<-toupper(s); sum(1/rank[genes %in% s])})
    idx <- order(inter, cover, weighted, decreasing=TRUE)
    top <- head(idx, 40)
    for(i in top){
      if(inter[i]>=5) cat(sprintf('"'%3d cov %.2f weighted %.2f | %s\\n', inter[i], cover[i], weighted[i], names(sets)[i]))
    }
  }
}
RS"
