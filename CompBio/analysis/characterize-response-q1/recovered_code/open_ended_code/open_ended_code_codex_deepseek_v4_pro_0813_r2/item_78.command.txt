/bin/bash -c "python3 - <<'PY'
import requests, math
from math import comb, log
genes=[g.upper() for g in open('characterize.response.q1.txt').read().strip().split('\\n')]
N=20000 # approx
q=len(genes)
def hyper_score(s, inter):
    # use fisher log p upper tail via comb approximations or scipy
    import scipy.stats as st
    return st.hypergeom.sf(inter-1, N, len(s), q)
libs=['CellMarker_Augmented_2021','CellMarker_2024','PanglaoDB_Augmented_2021','Azimuth_Cell_Types_2021','Descartes_Cell_Types_and_Tissue_2021','ARCHS4_Tissues']
for lib in libs:
 r=requests.get('https://maayanlab.cloud/Enrichr/geneSetLibrary',params={'libraryName':lib},timeout=120)
 terms=r.json()[lib]['terms']; rows=[]
 for term,vals in terms.items():
  s=set(v.upper() for v in vals)
  inter=genes and s; inter=len(set(genes)&s)
  if inter>=5:
   p=hyper_score(s,inter)
   score=sum(1/(i+1) for i,g in enumerate(genes) if g in s)
   rows.append((p,score,len(s),inter,term))
 rows.sort(key=lambda x:(x[0],-x[1]))
 print('\\n',lib)
 for p,score,size,inter,term in rows[:15]: print(f'{p:.1e} score={score:.2f} size={size} inter={inter} {term}')
PY"
