/bin/bash -lc "python3 - <<'PY'
import GEOparse, pandas as pd, numpy as np
query=[g.upper() for g in open('characterize.response.q1.txt').read().split()]
for f,comp in [('geo/GSE108289_family.soft.gz',('immature','mature')),('geo/GSE69076_family.soft.gz',('ctrl','lps6'))]:
 gse=GEOparse.get_GEO(filepath=f, silent=True)
 gpl=list(gse.gpls.values())[0]; sym=gpl.table.set_index('ID')['GENE_SYMBOL'].to_dict()
 mats={n:gsm.table.set_index('ID_REF')['VALUE'] for n,gsm in gse.gsms.items()}
 mat=pd.DataFrame(mats)
 if 'GSE108289' in f:
  a=[c for c in mat.columns if 'immature' in gse.gsms[c].metadata['title'][0]]
  b=[c for c in mat.columns if 'maturation' in gse.gsms[c].metadata['title'][0]]
 else:
  a=[c for c in mat.columns if 'Unstimulated' in gse.gsms[c].metadata['title'][0]]
  b=[c for c in mat.columns if 'LPS 6hr' in gse.gsms[c].metadata['title'][0]]
 A=mat[a].mean(axis=1); B=mat[b].mean(axis=1); lfc=B-A
 for direction,name in [(lfc,'up B vs A'),(-lfc,'down B vs A')]:
  arr=direction.abs().sort_values(ascending=False)
  res=[]; seen=set()
  for pid in arr.index:
   s=sym.get(pid)
   if not isinstance(s,str) or s.lower()=='nan' or not s: continue
   su=s.upper()
   if su in seen: continue
   seen.add(su); res.append(s)
  top=[x.upper() for x in res[:112]]
  ov=sum(1 for g in query if g in top)
  print('\\n',f,name,'overlap',ov,'common first', [x for x in res[:112] if x.upper() in query])
PY"
