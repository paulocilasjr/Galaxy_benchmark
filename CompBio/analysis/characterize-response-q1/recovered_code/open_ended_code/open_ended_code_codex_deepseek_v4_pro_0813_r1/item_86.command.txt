import GEOparse,numpy as np,pandas as pd,re
from collections import defaultdict
QFILE='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-df9db11b2debf840fc5650c6/workspace/characterize.response.q1.txt'
DEST='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-df9db11b2debf840fc5650c6/workspace/geo'
g=GEOparse.get_GEO(geo='GSE138018',destdir=DEST,silent=True)
mats=[]
for name,gsm in g.gsms.items():
 t=gsm.table.copy();vc='VALUE' if 'VALUE' in t.columns else [c for c in t.columns if c not in ['ID_REF']][0]
 t=t[['ID_REF',vc]].rename(columns={vc:'value'});t['sample']=name;mats.append(t)
df=pd.concat(mats);expr=df.pivot(index='ID_REF',columns='sample',values='value').astype(float)
sym=None
for gpl in g.gpls.values():

 for name in expr.columns:
  title=g.gsms[name].metadata.get('title');
  if isinstance(title,list):title=title[0]
  if re.search(condpat,title,re.I):cond.append(name)

 ranked=sorted(best.items(),key=lambda kv:-kv[1][0]);top=[x for x,v in ranked[:112]]

 rank={x:i+1 for i,(x,v) in enumerate(ranked)}
 print([(x.capitalize(),rank.get(x.upper())) for x in qlist if x.upper() in rank][:40])
comp(r'pDC \\+ LIF\\.',r'pDC \\+ medium\\.','LIF vs medium')

PY"
