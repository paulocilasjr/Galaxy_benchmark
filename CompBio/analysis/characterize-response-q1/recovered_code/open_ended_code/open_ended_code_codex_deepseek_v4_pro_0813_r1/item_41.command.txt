/bin/bash -lc "python3 - <<'PY'
import requests, urllib.parse, re, time
queries=[
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" \"Clec4n\"',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" LPS',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" dendritic',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" monocytes',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" BMS-34554',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" sepsis',
 '\"Ifitm1\" \"Tspo\" \"Lgmn\" aging',
]
for q in queries:
    url='https://html.duckduckgo.com/html/?q='+urllib.parse.quote(q)
    r=requests.get(url,headers={'User-Agent':'Mozilla/5.0'},timeout=30)
    text=re.sub('<["'^>]+>'"',' ',r.text); text=re.sub(r'\\s+',' ',text)
    print('\\nQUERY',q)
    # print around results
    parts=text.split('result__a')
    for p in parts[1:6]:
        m=re.search(r'href=\"(["'^"]+)"[''^>]*>(.*?)</a>'"',p)
        if m:
            import html
            print(html.unescape(m.group(2))[:150], m.group(1)[:100])
    if 'No results' in text: print('No results')
    time.sleep(.5)
PY"
