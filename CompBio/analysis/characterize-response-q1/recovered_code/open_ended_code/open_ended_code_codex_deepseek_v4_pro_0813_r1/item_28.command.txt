/bin/bash -lc "cd geo && python3 - <<'PY'
import gzip
fn='GSE4479_family.soft.gz'
lines=[]
with gzip.open(fn,'rt',encoding='latin1') as f:
    lines=f.readlines()
# find platform table
start=next(i for i,l in enumerate(lines) if l.startswith('"'!platform_table_begin'"'))
end=next(i for i,l in enumerate(lines) if l.startswith('"'!platform_table_end'"'))
print(start,end)
header=lines[start+1].rstrip('\\n').split('\\t')
print(header[:30])
for i in range(start+2,start+5):
    print(lines[i].rstrip().split('\\t')[:15])
# sample tables starts
indices=[i for i,l in enumerate(lines) if l.startswith('"'!sample_table_begin'"')]
print('sample tables',indices[:5])
print(lines[indices[0]+1].rstrip().split('\\t'))
print(lines[indices[0]+2].rstrip().split('\\t')[:6])
PY"
