/bin/bash -lc "python3 - <<'PY'
import requests,time
queries=[
'lipopolysaccharide[Title] AND monocyte[Title] AND Mus musculus[Organism]',
'LPS[Title] AND monocyte[Title] AND Mus musculus[Organism]',
'lipopolysaccharide[Title] AND macrophage[Title] AND Mus musculus[Organism]',
'lipopolysaccharide[Title] AND dendritic cell[Title] AND Mus musculus[Organism]',
]
for q in queries:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':q,'retmax':100,'retmode':'json'},timeout=60)
 j=r.json()['esearchresult'];print('\\n',q,j['count']);ids=j.get('idlist',[]);print(ids[:30])
 # summaries first 10
 if ids:
  ss=requests.post('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',data={'db':'gds','id':','.join(ids[:20]),'retmode':'json'},timeout=60).json()['result']
  for uid in ids[:20]:
   v=ss.get(uid,{}); print(' ',v.get('accession'),v.get('title'))
 time.sleep(.2)
PY"
