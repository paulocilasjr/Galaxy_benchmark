/bin/bash -lc "python3 - <<'PY'
import GEOparse,pandas as pd,re
qset=set(g.upper() for g in open('characterize.response.q1.txt').read().split())
f='geo/GSE138018_family.soft.gz'
gse=GEOparse.get_GEO(filepath=f,silent=True)
gpl=list(gse.gpls.values())[0]
sym={}
for pid,v in gpl.table.set_index('ID')['gene_assignment'].items():
 if isinstance(v,str):
  part=v.split('//')[1].strip() if len(v.split('//'))>1 else ''
  if part: sym[pid]=part
mats={n:gsm.table.set_index('ID_REF')['VALUE'] for n,gsm in gse.gsms.items()}
mat=pd.DataFrame(mats)
titles={n:gsm.metadata.get('title',[''])[0] for n,gsm in gse.gsms.items()}
# groups
medium=[n for n,t in titles.items() if '+ medium' in t]
lif=[n for n,t in titles.items() if '+ LIF' in t]
cpg=[n for n,t in titles.items() if '+ CpG-B' in t]
cpglif=[n for n,t in titles.items() if '+ CpG-B + LIF' in t]
print('groups',len(medium),len(lif),len(cpg),len(cpglif))
def comp(a,b):
 A=mat[a].mean(axis=1);B=mat[b].mean(axis=1);lfc=B-A
 rows=[];seen=set()
 for pid in lfc.index:
  s=sym.get(pid)
  if not isinstance(s,str) or not s: continue
  su=s.upper()
  if su in seen: continue
  seen.add(su);rows.append((s,lfc[pid]))
 df=pd.DataFrame(rows,columns=['g','x']).sort_values('x',ascending=False)
 up=df.head(112).g.str.upper().tolist(); dn=df.tail(112).g.str.upper().tolist()
 print('up',len(set(up)&qset),'down',len(set(dn)&qset))
 print([x for x in up if x in qset])
comp(medium,lif);comp(cpg,cpglif)
PY"
