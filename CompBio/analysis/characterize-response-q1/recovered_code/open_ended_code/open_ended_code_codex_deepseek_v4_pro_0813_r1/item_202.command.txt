/bin/bash -lc "python3 - <<'PY'
import requests, time, json, collections, sys
genes=['Clec4n','Cd180','Ms4a6d','Mcemp1','C1qc','Lgmn','Mafb','Tspo','Cybb','Ncf1','Scimp','Wfdc17','Gpr35','Cyp7b1','Ifitm1','Ifitm2','Clec4e','Ms4a4c']
acc_sets={}
for gene in genes:
 ids=[]; retstart=0
 while True:
  r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'geoprofiles','term':f'{gene}[Gene Symbol]','retstart':retstart,'retmax':10000,'retmode':'json'},timeout=60)
  j=r.json().get('esearchresult',{}); batch=j.get('idlist',[]); ids.extend(batch)
  if len(batch)<10000 or retstart+len(batch)>=int(j.get('count','0')): break
  retstart += len(batch)
  time.sleep(.1)
 acc=set()
 # esummary in batches of 500
 for i in range(0,len(ids),500):
  batch=ids[i:i+500]
  r=requests.post('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',data={'db':'geoprofiles','id':','.join(batch),'retmode':'json'},timeout=60)
  jj=r.json().get('result',{})
  for uid in batch:
   val=jj.get(uid,{})
   # summary may have GDS or GSE fields
   for key in ['GDS','GSE','GSM']:
    if key in val: acc.add(val[key])
  time.sleep(.1)
 acc_sets[gene]=acc
 print(gene,len(ids),len(acc))
 common=set.intersection(*acc_sets.values()) if acc_sets else set()
 print('common accessions',len(common),sorted(common)[:100])
 # also intersection of all subsets decreasing
PY"
