/bin/bash -lc "python3 - <<'PY'
import GEOparse, glob, os
for f in sorted(glob.glob('geo/*_family.soft.gz')):
 try:
  gse=GEOparse.get_GEO(filepath=f, silent=True)
  for gn,gpl in gse.gpls.items():
   cols=list(gpl.table.columns)
   symcols=[c for c in cols if 'SYMBOL' in c.upper() or c.upper()=='GENE' or c.upper().startswith('GENE_')]
   print(os.path.basename(f),gn,symcols[:10], 'ID col?', 'ID' in cols)
 except Exception as e:
  print('ERR',f,e)
PY"
