/bin/bash -lc "python3 - <<'PY'
# print top lists for GSE17721 LPS 1h/6h/24h no dedup
import GEOparse,pandas as pd,re,collections
q=[g.upper() for g in open('characterize.response.q1.txt').read().split()]
qc=collections.Counter(q)
f='geo/GSE17721_family.soft.gz';g=GEOparse.get_GEO(filepath=f,silent=True);gpl=list(g.gpls.values())[0]
cols=gpl.table.columns; sc=[c for c in cols if 'Symbol' in c or 'SYMBOL' in c]; sym=gpl.table.set_index('ID')[sc[0]].to_dict()
mats={n:gsm.table.set_index('ID_REF')['VALUE'] for n,gsm in g.gsms.items()};mat=pd.DataFrame(mats);titles={n:gsm.metadata.get('title',[''])[0] for n,gsm in g.gsms.items()}
ctrl=[n for n,t in titles.items() if '0_hours' in t]
for tp in ['1_hours','6_hours','12_hours','24_hours']:
 tr=[n for n,t in titles.items() if tp in t]
 lfc=mat[tr].mean(axis=1)-mat[ctrl].mean(axis=1)
 rows=[]
 for pid,x in lfc.items():

  if not isinstance(s,str):continue
  s=re.split(r'///|\\|',s)[0].strip()
  if not s or s.lower()=='nan':continue
  rows.append((s.upper(),x))
 df=pd.DataFrame(rows,columns=['g','x']).sort_values('x',ascending=False)
 print('\\n',tp,'UP first 120:', ' '.join(df.head(120).g.tolist()))
 print(tp,'DOWN last 120:', ' '.join(df.tail(120).g.tolist()))
