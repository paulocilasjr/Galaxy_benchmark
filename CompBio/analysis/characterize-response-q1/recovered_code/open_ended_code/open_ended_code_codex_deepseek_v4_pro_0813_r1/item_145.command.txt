/bin/bash -lc "python3 - <<'PY'
import glob, os
q=[g.upper() for g in open('characterize.response.q1.txt').read().split()]
for path in glob.glob('enrichr_libs/*.txt'):
 lib=os.path.basename(path)[:-4]
 for line in open(path):
  p=line.rstrip('\\n').split('\\t')
  term=p[0]; genes=[g.upper() for g in p[1:] if g.strip()]
  # dedupe preserve order
  seen=set(); gl=[]
  for g in genes:
   if g not in seen:
    seen.add(g); gl.append(g)
  # compute longest prefix matching q
  n=0
  while n<min(len(q),len(gl)) and q[n]==gl[n]:
   n+=1
  if n>=20:
   print(lib,term,'prefix',n,'term_len',len(gl))
PY"
