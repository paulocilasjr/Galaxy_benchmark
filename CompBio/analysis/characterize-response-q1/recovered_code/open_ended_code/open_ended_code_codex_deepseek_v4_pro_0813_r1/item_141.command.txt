/bin/bash -lc "python3 - <<'PY'
import os, glob, re
query=[g.upper() for g in open('characterize.response.q1.txt').read().split()]
qset=set(query)
allrows=[]
for path in glob.glob('enrichr_libs/*.txt'):
 lib=os.path.basename(path)[:-4]
 for line in open(path):
  line=line.rstrip('\\n')
  if not line.strip(): continue
  parts=line.split('\\t')
  term=parts[0]
  genes=[p.upper() for p in parts[1:] if p.strip()]
  # dedupe preserve order
  seen=set(); gl=[]
  for g in genes:
   if g not in seen:
    seen.add(g); gl.append(g)
  gset=set(gl)
  ov=len(gset & qset)
  jac=ov/len(gset|qset) if len(gset|qset) else 0
  allrows.append((jac,ov,len(gset),lib,term,gl))
for row in sorted(allrows, reverse=True, key=lambda x:(x[0],x[1]))[:200]:
 print(f'{row[0]:.3f}\\t{row[1]}\\t{row[2]}\\t{row[3]}\\t{row[4]}')
PY"
