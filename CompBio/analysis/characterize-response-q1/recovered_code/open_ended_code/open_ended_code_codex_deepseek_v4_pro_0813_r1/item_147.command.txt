/bin/bash -lc "python3 - <<'PY'
import glob, os, heapq
q=[g.upper() for g in open('characterize.response.q1.txt').read().split()]
qset=set(q)
heap=[]
for path in glob.glob('enrichr_libs/*.txt'):
 lib=os.path.basename(path)[:-4]
 # process lines
 for line in open(path):
  p=line.rstrip('\\n').split('\\t')
  term=p[0]
  # build set and dedup maybe
  genes=[]
  seen=set()
  for g in p[1:]:
   gu=g.upper()
   if gu not in seen:
    seen.add(gu); genes.append(gu)
  gset=set(genes)
  ov=len(gset & qset)
  if ov==0: continue
  recall=ov/len(qset)
  jac=ov/len(gset|qset)
  # maybe weighted score recall + jac/2
  score=recall
  item=(score,ov,jac,len(gset),lib,term,genes)
  if len(heap)<200:
   heapq.heappush(heap,item)
  else:
   heapq.heappushpop(heap,item)
for item in sorted(heap,reverse=True,key=lambda x:(x[0],x[1],x[2])):
 print(f'{item[0]:.3f}\\tov={item[1]}\\tj={item[2]:.3f}\\tn={item[3]}\\t{item[4]}\\t{item[5]}')
PY"
