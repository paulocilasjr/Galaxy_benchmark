/bin/bash -lc "python3 - <<'PY'
import requests,re
for gse in ['GSE134869','GSE120807','GSE120808','GSE62746','GSE66178','GSE66315','GSE96628','GSE56929','GSE55068','GSE204955','GSE204957','GSE181134','GSE163975']:
 nnn=gse[:-3]+'nnn'
 url=f'https://ftp.ncbi.nlm.nih.gov/geo/series/{nnn}/{gse}/matrix/'
 try:
  r=requests.get(url,timeout=30)
  print('\\n',gse,r.status_code,len(r.text))
  files=re.findall(r'href=\"(["'^"]+)"'"',r.text)
  print([f for f in files if 'matrix' in f or '.gz' in f][:20])
 except Exception as e: print(gse,e)
PY"
