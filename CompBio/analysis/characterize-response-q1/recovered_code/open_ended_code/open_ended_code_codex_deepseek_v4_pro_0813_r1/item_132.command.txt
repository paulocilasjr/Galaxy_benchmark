/bin/bash -lc "python3 - <<'PY'
import requests, xml.etree.ElementTree as ET
queries=[
'LPS[Title] AND dendritic[Title] AND Mus musculus[Organism]',
'lipopolysaccharide[Title] AND dendritic cells[Title] AND Mus musculus[Organism]',
'LPS[Title] AND monocyte[Title] AND Mus musculus[Organism]',
'lipopolysaccharide[Title] AND macrophages[Title] AND Mus musculus[Organism]',
'IL-39[Title] AND Mus musculus[Organism]',
'Med12 knockout[All Fields] AND Mus musculus[Organism]',
'BMS-345541[All Fields] AND Mus musculus[Organism]',
'aldesleukin[Title] AND Mus musculus[Organism]',
]
for q in queries:
    r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gds','term':q,'retmax':100,'retmode':'json'},timeout=60)
    print('\\nQUERY',q,'count',r.json()['esearchresult']['count'])
    ids=r.json()['esearchresult'].get('idlist',[])
    print(ids[:50])
    if ids:
        s=requests.post('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi',data={'db':'gds','id':','.join(ids[:50]),'retmode':'json'},timeout=60).json()
        for uid,val in s['result'].items():
            if uid=='uids': continue
            print(uid, val.get('accession'), val.get('title'))
PY"
