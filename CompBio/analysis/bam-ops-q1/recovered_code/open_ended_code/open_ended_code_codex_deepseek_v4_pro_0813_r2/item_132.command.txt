/bin/bash -c "python - <<'PY'
import pysam
# For each QNAME, count primary proper-pair records and q30 records, all records
from collections import defaultdict
for mode in ['primary','all']:
    d=defaultdict(lambda:[0,0])
    for r in pysam.AlignmentFile('mt.sorted.bam','rb'):
        if not r.is_proper_pair: continue
        if mode=='primary' and (r.is_secondary or r.is_supplementary): continue
        d[r.query_name][0]+=1
        if r.mapping_quality>=30: d[r.query_name][1]+=1
    at_least_one=sum(1 for a,b in d.values() if b>0)
    both=sum(1 for a,b in d.values() if b>=2)
    print(mode,'unique templates with >=1 q30 proper alignment',at_least_one,'with both',both)
PY"
