/bin/bash -c "python - <<'PY'
import pysam
for scope in ['all','primary']:
    counts={}
    for r in pysam.AlignmentFile('mt.sorted.bam','rb'):
        if r.is_proper_pair and r.mapping_quality>=30:
            if scope=='primary' and (r.is_secondary or r.is_supplementary): continue
            key=('r1' if r.is_read1 else 'r2' if r.is_read2 else 'other')
            counts[key]=counts.get(key,0)+1
    print(scope, counts)
PY"
