/bin/bash -c "python - <<'PY'
import requests, json
# use urllib to POST and then get event
import urllib.request, time, json
base='https://genentech-compbiobench-leaderboard-v1.hf.space'
req=urllib.request.Request(base+'/gradio_api/call/refresh', data=json.dumps({'data':[]}).encode(), headers={'Content-Type':'application/json'})
with urllib.request.urlopen(req, timeout=60) as r:
    info=json.loads(r.read())
eid=info['event_id']
url=base+'/gradio_api/call/refresh/'+eid
with urllib.request.urlopen(url, timeout=60) as r:
    text=r.read().decode()
# Extract all data: lines
for line in text.splitlines():
    if line.startswith('data:'):
        data=line[5:].strip()
        if data:
            try:
                obj=json.loads(data)
                print('type', type(obj), 'len', len(obj) if hasattr(obj,'__len__') else '')
                if isinstance(obj,list):
                    for i,x in enumerate(obj):
                        print('item',i, 'type',type(x), 'keys?', x.keys() if isinstance(x,dict) else '')
                        if isinstance(x,dict):
                            print('headers', x.get('headers'))
                            print('data len', len(x.get('data',[])))
                            # save
                            json.dump(obj, open('[REDACTED LOCAL PATH]','w'))
                else:
                    print(str(obj)[:1000])
            except Exception as e:
                print('parse error',e, data[:1000])
PY"
