/bin/bash -lc "curl -sS --get 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi' --data 'db=nuccore' --data 'id=NC_000022.11' --data 'seq_start=27079126' --data 'seq_stop=27080622' --data 'rettype=fasta' --data 'retmode=text' -o hg38.fa

minimap2 -c --eqx hg38.fa hg38_clone.fa 2>/dev/null"
