/bin/bash -lc "cat > work/phase.py <<'PY'
import pysam,collections,itertools,json
positions=[2145,2205,2505,2665,2915,3114,3118,3365]
alleles=[['A','C'],['C','T'],['A','T'],['D','G'],['C','G'],['A','D'],['D','G'],['A','G']]
b=pysam.AlignmentFile('work/reads.bam')
byread=collections.defaultdict(lambda:collections.defaultdict(list))
for r in b:
 calls={}
 for q,p in r.get_aligned_pairs():
  if p is not None and p+1 in positions:
   i=positions.index(p+1);a=r.query_sequence[q] if q is not None else 'D'
   if a in alleles[i]:byread[r.query_name][i].append(a)
reads={}
for name,cs in byread.items():
 reads[name]={i:a[0] for i,a in cs.items() if len(set(a))==1}
edges=collections.defaultdict(collections.Counter)
for calls in reads.values():
 for i,j in itertools.combinations(sorted(calls),2):edges[i,j][calls[i]+calls[j]]+=1
for (i,j),c in sorted(edges.items()):print(positions[i],positions[j],dict(c))
scored=[]
for bits in itertools.product([0,1],repeat=len(positions)-1):
 bits=(0,)+bits
 h1=[aa[v] for aa,v in zip(alleles,bits)];h2=[aa[1-v] for aa,v in zip(alleles,bits)]
 cost=0;good=0;bad=[]
 for name,calls in reads.items():
  m=min(sum(a"'!=h1[i] for i,a in calls.items()),sum(a!=h2[i] for i,a in calls.items()))
  cost+=m
  if m:bad.append([name,calls,m])
  else:good+=1
 scored.append((cost,h1,h2,good,bad))
scored.sort()
for cost,h1,h2,good,bad in scored[:5]:print('"'PHASE',cost,good,'-'.join(h1)+'|'+'-'.join(h2))
json.dump({'positions':positions,'alleles':alleles,'phases':scored[:2]},open('work/phase-evidence.json','w'))
PY
python3 work/phase.py"
