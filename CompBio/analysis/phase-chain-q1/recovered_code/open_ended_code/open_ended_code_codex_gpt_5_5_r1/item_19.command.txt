import urllib.request, urllib.parse, re
seq='AGGCCTCCTGACACCTGAGACATGGTCTGAGAGGCAGGCTGCCGTGGGTACCCACATGTACCACAGGTACTCACCTGGGCTTGCCAGCCT'
p=urllib.parse.urlencode({'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','MEGABLAST':'on','QUERY':seq,'FORMAT_TYPE':'XML'})
data=urllib.request.urlopen('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=p.encode(),timeout=60).read().decode('latin1','replace')
open('ncbi_put.html','w').write(data)
print('len',len(data))
for pat in ['RID = ([A-Z0-9-]+)','RID = (["'^'"\\n<]+)','name=\"RID\" value=\"(["'^"]+)"'"','RID\" value=\"(["'^"]+)'"']:
    m=re.search(pat,data)
    print(pat, m.group(1).strip() if m else None)
for pat in ['RTOE = (\\d+)','RTOE = (["'^'"\\n<]+)']:
    m=re.search(pat,data); print(pat, m.group(1).strip() if m else None)
print(data[-2000:])
PY"
