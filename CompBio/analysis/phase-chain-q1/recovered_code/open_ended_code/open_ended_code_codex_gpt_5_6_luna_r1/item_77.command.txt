/bin/bash -lc 'pwd; for h in 1 2; do echo H${h}; samtools view reads_hap${h}.bam | awk '"'{print "'$6}'"' | sort | uniq -c | sort -k2,2 | tail -10; done
python3 - <<'PY'
import pysam,collections
b=pysam.AlignmentFile('reads_vs_contigs.bam'); cs=collections.Counter()
for a in b.fetch(until_eof=True):
 if a.is_secondary or a.is_supplementary: continue
 for op,n in a.cigartuples or []:
  if op in (1,2): cs[(op,n)]+=1
print(cs)
PY"
