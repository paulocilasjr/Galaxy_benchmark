/bin/bash -lc "python - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('reads.sorted.bam')
# genomic 1-based sites/events; true candidate provisional
sites=[27079306,27079366,27079666,27079826,27080076,27080275,27080279,27080526]
labels=['s306','s366','s666','e826','s076','e275','e279','s526']
# event alleles inferred from observed ref / deletion; retain all bases

def obs(rec,p):
 # return base if aligned at p, D if deletion starts at p, . unknown
 target=p-1
 rpos=rec.reference_start; qpos=0
 q=rec.query_sequence
 for op,n in rec.cigartuples:
  if op in (0,7,8):
   if rpos<=target<rpos+n:
    return q[qpos+target-rpos]
   rpos+=n;qpos+=n
  elif op==1: qpos+=n
  elif op==2:
   if rpos<=target<rpos+n:
    return 'D'
   rpos+=n
  elif op==3: rpos+=n
  elif op==4: qpos+=n
 return '.'

pairs=collections.defaultdict(list)
for rec in bam.fetch('chr22',27079000,27081000):
 if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
 pairs[rec.query_name].append(rec)
# print pair observations with >=2 calls
states=collections.Counter(); full=[]
for name,recs in pairs.items():
 a=[]
 for p,l in zip(sites,labels):
  vals=[obs(r,p) for r in recs]; vals=[v for v in vals if v"'!='"'.']
  # require consistent; combine unique
  v='/' .join(sorted(set(vals))) if vals else '.'
  a.append(v)
 key=' '.join(f'{l}={v}' for l,v in zip(labels,a) if v"'!='"'.')
 if sum(v"'!='"'.' for v in a)>=2:
  states[tuple(a)]+=1; full.append((name,a,recs))
print('pairs',len(pairs),'with >=2 site calls',len(full),'unique states',len(states))
for st,n in states.most_common(): print(n, ' '.join(f'{l}:{v}' for l,v in zip(labels,st)))
print('\\nindividual observation counts')
for p,l in zip(sites,labels):
 c=collections.Counter()
 for name,recs in pairs.items():
  vs=[obs(r,p) for r in recs]; vs=[v for v in vs if v"'!='"'.']
  if vs: c['/'.join(sorted(set(vs)))]+=1
 print(l,c)
PY"
