/bin/bash -lc "python - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('reads.sorted.bam')
sites=[27079306,27079366,27079666,27079826,27080076,27080275,27080279,27080526]
labels=['s306','s366','s666','e826','s076','e275','e279','s526']
haps=[['C','T','A','G','G','D','G','A'],['A','C','T','D','C','A','D','G']]
def obs(rec,p):
 target=p-1; rpos=rec.reference_start; qpos=0; q=rec.query_sequence
 for op,n in rec.cigartuples:
  if op in (0,7,8):
   if rpos<=target<rpos+n:return q[qpos+target-rpos]
   rpos+=n;qpos+=n
  elif op==1:qpos+=n
  elif op==2:
   if rpos<=target<rpos+n:return 'D'
   rpos+=n
  elif op==3:rpos+=n
  elif op==4:qpos+=n
 return '.'
pairs=collections.defaultdict(list)
for rec in bam.fetch('chr22',27079000,27081000):
 if not (rec.is_secondary or rec.is_supplementary):pairs[rec.query_name].append(rec)
assign={}; conflicts=[]
for name,recs in pairs.items():
 scores=[]
 for h in haps:
  sc=0; n=0
  for p,exp in zip(sites,h):
   for rec in recs:
    x=obs(rec,p)
    if x"'!='"'.':
     n+=1; sc += (x==exp)
  scores.append((sc,n))
 if scores[0][0]>scores[1][0] and scores[0][0]>=1:assign[name]=0
 elif scores[1][0]>scores[0][0] and scores[1][0]>=1:assign[name]=1
 elif scores[0][0]==scores[1][0] and scores[0][0]>=1: conflicts.append((name,scores))
# counts by assigned hap
for hi in (0,1):
 print('\\nHAP',hi+1,'pairs',sum(v==hi for v in assign.values()))
 for p,l in zip(sites,labels):
  c=collections.Counter()
  for name,recs in pairs.items():
   if assign.get(name)==hi:
    for rec in recs:
     x=obs(rec,p)
     if x"'!='"'.':c[x]+=1
  print(l,c)
print('assigned',len(assign),'conflicts',len(conflicts),conflicts[:10])
# Show pair assignment states with all calls / mismatch
for name in list(pairs)[:0]:pass
PY"
