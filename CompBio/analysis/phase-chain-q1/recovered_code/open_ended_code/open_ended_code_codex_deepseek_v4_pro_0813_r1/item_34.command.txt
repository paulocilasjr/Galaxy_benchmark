/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('phase.bam','rb')
start=27079000; end=27080700
ins=collections.Counter()
for read in bam.fetch('chr22', start, end):
    if read.is_unmapped or read.is_secondary or read.is_supplementary: continue
    rp=read.reference_start; qp=0
    for op,l in read.cigartuples:
        if op==0: rp+=l; qp+=l
        elif op==1:
            seq=read.query_sequence[qp:qp+l]
            ins[(rp,l,seq)]+=1
            qp+=l
        elif op==2: rp+=l
        elif op==3: rp+=l
        elif op==4: qp+=l
        elif op==5: pass
        elif op==6: pass
        elif op in (7,8): qp+=l; rp+=l
print('total distinct insertions', len(ins), 'total calls', sum(ins.values()))
for k,v in ins.most_common(20):
    print(k,v)
PY"
