/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('phase.bam','rb')
start=27079000; end=27080700
counts=collections.defaultdict(collections.Counter)
for pileupcolumn in bam.pileup('chr22', start, end, truncate=True, min_base_quality=0, stepper='nofilter', max_depth=1000000):
    pos=pileupcolumn.reference_pos+1
    for pr in pileupcolumn.pileups:
        if pr.is_del or pr.is_refskip:
            continue
        q=pr.alignment.query_sequence[pr.query_position]
        counts[pos][q]+=1
for pos in range(start,end+1):
    c=counts.get(pos)
    if not c: continue
    tot=sum(c.values())
    if tot==0: continue
    # identify nonmajor
    top=c.most_common()
    if len(c)>1 or (top[0][1]<tot):
        print(pos, tot, dict(c.most_common()), 'nonref?')
PY"
