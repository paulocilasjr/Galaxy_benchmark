/bin/bash -lc "python3 - <<'PY'
import pysam
bam=pysam.AlignmentFile('phase.bam','rb')
fasta=pysam.FastaFile('hg38.fa')
start=27079790; end=27079860
ref=fasta.fetch('chr22', start-1, end)
print('REF', start, ref)
for read in bam.fetch('chr22', start, end):
    if read.is_unmapped: continue
    # compute aligned pairs positions in [start,end]
    pairs=read.get_aligned_pairs(matches_only=False)
    seg=[]
    for q,r in pairs:
        if r is not None and start <= r+1 <= end:
            if q is not None:
                seg.append(f'{r+1}:{read.query_sequence[q]}')
            else:
                seg.append(f'{r+1}:D')
        # insertions between ref positions maybe
    # print compact around 27079820-27079840
    print(read.query_name, read.reference_start, read.cigarstring, ' '.join(seg[:100]))
PY"
