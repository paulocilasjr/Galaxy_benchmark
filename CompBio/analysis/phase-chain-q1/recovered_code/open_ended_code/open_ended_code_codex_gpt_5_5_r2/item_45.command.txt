/bin/bash -lc "curl -s 'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr22;start=27079125;end=27080624' > ucsc_seq.json
python3 - <<'PY'
import json, textwrap
j=json.load(open('ucsc_seq.json'))
seq=j['dna'].upper()
print(len(seq), seq[:50], seq[-50:])
with open('region.fa','w') as f:
    f.write('>chr22:27079126-27080624\\n')
    for i in range(0,len(seq),60): f.write(seq[i:i+60]+'\\n')
PY
head -3 region.fa"
