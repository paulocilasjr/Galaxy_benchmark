/bin/bash -lc "python3 - <<'PY'
import pysam, collections, json
bam=pysam.AlignmentFile('local.sorted.bam')
ins=collections.Counter(); dels=collections.Counter(); cig=collections.Counter(); soft=0
for r in bam.fetch('local'):
    if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
    q=0; rp=r.reference_start+1
    for op,l in r.cigartuples:
        cig[(op,l)]+=1
        if op in (0,7,8): q+=l; rp+=l
        elif op==1:
            ins[(rp-1, r.query_sequence[q:q+l])] += 1; q+=l
        elif op==2:
            dels[(rp,l)] += 1; rp+=l
        elif op==4: soft+=1; q+=l
        elif op==3: rp+=l
print('insertions >=5')
for k,n in ins.most_common():
    if n>=5: print(k,n)
print('deletions >=5')
for k,n in dels.most_common():
    if n>=5: print(k,n)
print('softclips',soft)
PY
printf '%s\\n' 'A-C-T-D-C-A-D-G|C-T-A-G-G-D-G-A' > answer.txt
wc -l answer.txt; awk '{print length, "'$0}'"' answer.txt"
