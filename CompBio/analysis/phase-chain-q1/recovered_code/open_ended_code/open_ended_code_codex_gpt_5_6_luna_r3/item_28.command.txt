from Bio.Seq import Seq

out=urllib.request.urlopen(url,timeout=30).read().decode();ref=''.join(x.strip() for x in out.splitlines() if not x.startswith('>'))
def reads(p):
 with gzip.open(p,'rt') as f:
  while True:
   h=f.readline().strip()
   if not h:return
   s=f.readline().strip();f.readline();f.readline();yield h,s
def parse(c): return [(op,int(n)) for n,op in re.findall(r'(\\d+)([=XID])',c)]
def aln(s):
 a=[]
 for ori,q in [('+',s),('-',str(Seq(s).reverse_complement()))]:

 return min(a,key=lambda x:(x[0],x[3][0]))

for fn in ['phase.chain.q1.R1.fq.gz','phase.chain.q1.R2.fq.gz']:

  for op,n in parse(c):

print('base candidates alt>=20% and cov>=40')
for i,c in enumerate(base):
 if cov[i]>=40 and len(c)>1 and c.most_common(2)[1][1]/cov[i]>=.15:
  print(60001+i,ref[i],cov[i],c.most_common())
print('ins candidates >=10')
for (t,s),n in ins.most_common():

print('del candidates >=10')
for (t,l),n in dele.most_common():
 if n>=10:print(60001+t,'deleted',ref[t:t+l],n)
