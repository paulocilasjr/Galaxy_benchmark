/bin/bash -lc "python3 -u - <<'PY'
import urllib.parse,urllib.request,re
seq='AGGCCTCCTGACACCTGAGACATGGTCTGAGAGGCAGGCTGCCGTGGGTACCCACATGTACCACAGGTACTCACCTGGGCTTGCCAGCCTGACTCAGAGTCCGTTCCCCTCCATGCACAGTGCTGCCTCCCCCTCCCTGGCGGAGCAATC'
data=urllib.parse.urlencode({'CMD':'Put','PROGRAM':'blastn','DATABASE':'refseq_genomic','QUERY':seq,'MEGABLAST':'on','HITLIST_SIZE':'20','FORMAT_TYPE':'XML','EXPECT':'1e-10','FILTER':'L'}).encode()
r=urllib.request.urlopen(urllib.request.Request('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=data,headers={'User-Agent':'Mozilla/5.0'}),timeout=60)
out=r.read().decode('utf8','replace')
print('response',len(out),out[:100])
print('errors',re.findall(r'<Message>(.*?)</Message>|<Error>(.*?)</Error>',out,re.S)[:5])
print('rid',re.findall(r'<input name=\"RID\"["'^>]*value="([''^"]+)"'"',out)[:5])
print('qblast',re.findall(r'QBlastInfoBegin.*?QBlastInfoEnd',out,re.S)[:2])
PY"
