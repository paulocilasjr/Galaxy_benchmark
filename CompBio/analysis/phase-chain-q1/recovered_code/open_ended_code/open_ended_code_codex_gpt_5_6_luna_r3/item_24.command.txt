/bin/bash -lc "python3 - <<'PY'
import urllib.parse,urllib.request,re,time,xml.etree.ElementTree as ET
seq='AGGCCTCCTGACACCTGAGACATGGTCTGAGAGGCAGGCTGCCGTGGGTACCCACATGTACCACAGGTACTCACCTGGGCTTGCCAGCCTGACTCAGAGTCCGTTCCCCTCCATGCACAGTGCTGCCTCCCCCTCCCTGGCGGAGCAATC'
for db in ['refseq_genomic','nt']:
 data=urllib.parse.urlencode({'CMD':'Put','PROGRAM':'blastn','DATABASE':db,'QUERY':seq,'MEGABLAST':'on','HITLIST_SIZE':'20','FORMAT_TYPE':'XML','EXPECT':'1e-10','FILTER':'L','ENTREZ_QUERY':'Homo sapiens[organism]' if db=='refseq_genomic' else ''}).encode()
 out=urllib.request.urlopen(urllib.request.Request('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=data,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read().decode('utf8','replace')
 m=re.search(r'<input name=\"RID\"["'^>]*value="([''^"]+)"'"',out)
 rid=m.group(1) if m else None
 print('db',db,'rid',rid)
 if not rid: continue
 for i in range(20):
  time.sleep(3)
  url='https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML'})
  out2=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read()
  if b'<Hit>' in out2:
   root=ET.fromstring(out2)
   for hit in root.findall('.//Hit')[:20]:
    hsp=hit.find('./Hit_hsps/Hsp')
    print(hit.findtext('Hit_num'),hit.findtext('Hit_id'),hit.findtext('Hit_def'),hsp.findtext('Hsp_identity'),hsp.findtext('Hsp_hit-from'),hsp.findtext('Hsp_hit-to'))
   break
PY"
