/bin/bash -lc "python3 - <<'PY'
import pysam
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
from collections import Counter
c=Counter()
for r in b.fetch('chr22',27079100,27080630):
    if r.is_unmapped: continue
    pos=r.reference_start
    for op,l in r.cigartuples:
        if op==2:
            c[(pos,l)]+=1
            # print first few
        if op in (0,2,3,7,8): pos += l
for k,v in c.most_common(10): print(k,v)
PY"
