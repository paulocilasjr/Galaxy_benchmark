/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
c=Counter(); ex=[]
for r in b.fetch('chr22',27079100,27080630):
    if r.is_unmapped: continue
    pos=r.reference_start
    for op,l in r.cigartuples:
        if op==1:
            c[(pos,l)]+=1
            if len(ex)<10: ex.append((r.qname,pos,l,r.seq))
        if op in (0,2,3,7,8): pos += l
print('insertions')
for k,v in c.most_common(20): print(k,v)
print('examples',ex[:3])
PY"
