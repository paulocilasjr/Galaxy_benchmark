/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
start=27079126; end=27080624; L=end-start+1
b=pysam.AlignmentFile('mapped.sorted.bam','rb')
pc=[Counter() for _ in range(L)]
for rec in b.fetch('chr22',start-1,end+1):
 if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
 seq=rec.query_sequence
 for qpos,rpos in rec.get_aligned_pairs(matches_only=False):
  if rpos is None: continue
  idx=rpos-(start-1)
  if 0<=idx<L: pc[idx]['-' if qpos is None else seq[qpos]]+=1
for i,c in enumerate(pc):
 total=sum(c.values())
 if total<=10:
  print(start+i,total,c.most_common())
PY"
