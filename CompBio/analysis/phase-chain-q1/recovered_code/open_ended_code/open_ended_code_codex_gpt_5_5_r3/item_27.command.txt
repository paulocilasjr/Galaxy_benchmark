/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('aln.bam','rb'); chrom='AL008638.1'
for p in range(64728,64740):
    cnt=collections.Counter(); dels=0
    for col in bam.pileup(chrom,p-1,p,truncate=True,min_base_quality=0,ignore_overlaps=False,stepper='all'):
      for pr in col.pileups:
        if pr.is_del: dels+=1
        elif not pr.is_refskip: cnt[pr.alignment.query_sequence[pr.query_position]]+=1
    print(p, cnt, 'D',dels)
PY"
