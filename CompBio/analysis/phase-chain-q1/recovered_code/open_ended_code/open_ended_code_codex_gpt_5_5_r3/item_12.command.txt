/bin/bash -lc "python3 - <<'PY'
import gzip, urllib.parse, urllib.request, time, re
seq='AGGCCTCCTGACACCTGAGACATGGTCTGAGAGGCAGGCTGCCGTGGGTACCCACATGTACCACAGGTACTCACCTGGGCTTGCCAGCCTGACTCAGAGTCCGTTCCCCTCCATGCACAGTGCTGCCTCCCCCTCCCTGGCGGAGCAATC'
params=urllib.parse.urlencode({'CMD':'Put','PROGRAM':'blastn','DATABASE':'nt','QUERY':seq,'MEGABLAST':'on','FORMAT_TYPE':'Text','HITLIST_SIZE':10})
req=urllib.request.Request('https://blast.ncbi.nlm.nih.gov/Blast.cgi', data=params.encode(), headers={'User-Agent':'Mozilla/5.0'})
try:
    txt=urllib.request.urlopen(req,timeout=30).read().decode('utf-8','ignore')
    print(txt[:2000])
    rid=re.search(r'RID = (\\S+)',txt)
    rtoe=re.search(r'RTOE = (\\d+)',txt)
    print('RID', rid.group(1) if rid else None, 'RTOE', rtoe.group(1) if rtoe else None)
except Exception as e:
    print('ERR',repr(e))
PY"
