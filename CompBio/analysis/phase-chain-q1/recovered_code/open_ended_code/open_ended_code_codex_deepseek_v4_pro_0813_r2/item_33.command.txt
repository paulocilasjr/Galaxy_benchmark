/bin/bash -lc "python3 - <<'PY'
import re
rows=[]
with open('variants.vcf') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<10: continue
        info=p[7]; fmt=p[8]; sample=p[9]
        # parse AD from sample field
        d=dict(zip(fmt.split(':'), sample.split(':')))
        ad=d.get('AD','')
        if ',' in ad or 'INDEL' in info:
            rows.append((p[1],p[3],p[4],info,ad,d.get('DP')))
print(len(rows))
for r in rows:
    print(r)
PY"
