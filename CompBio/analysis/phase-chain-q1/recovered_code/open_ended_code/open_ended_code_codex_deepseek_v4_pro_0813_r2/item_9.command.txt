/bin/bash -lc "python3 - <<'PY'
import urllib.parse, subprocess, urllib.request, re
seq=subprocess.check_output(\"gzip -cd phase.chain.q1.R1.fq.gz | awk 'NR==2{print; exit}'\", shell=True, text=True).strip()
params={'CMD':'Put','PROGRAM':'blastn','DATABASE':'refseq_genomic','QUERY':seq,'FORMAT_TYPE':'JSON2','HITLIST_SIZE':'5','EXPECT':'1e-10','WORD_SIZE':'11','FILTER':'m'}
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi'
data=urllib.parse.urlencode(params).encode()
req=urllib.request.Request(url,data=data,headers={'User-Agent':'Mozilla/5.0'})
text=urllib.request.urlopen(req,timeout=120).read().decode()
# print QBlastInfo or error
for pat in [r'QBlastInfo(.*)', r'RID = (["'^<'"\\s]+)', r'<textarea["'^>]*>(.*?)</textarea>'"', r'error["'^<]*'"']:
    m=re.findall(pat,text,re.S|re.I)
    print(pat,m[:3])
print(text[-2000:])
PY"
