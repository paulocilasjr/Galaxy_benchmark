import urllib.parse, urllib.request, time, re, json
rid='83PKJDYB014'
url='https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi'
for i in range(30):
    params={'CMD':'Get','RID':rid,'FORMAT_TYPE':'JSON2'}
    req=urllib.request.Request(url+'?'+urllib.parse.urlencode(params),headers={'User-Agent':'Mozilla/5.0'})
    try:
        text=urllib.request.urlopen(req,timeout=30).read().decode()
    except Exception as e:
        print('err',e); break
    if 'Status=READY' in text or 'Status=UNKNOWN' in text or 'QBlastInfo' in text and 'Status=WAITING' not in text and 'Status=FAILED' not in text and 'Status=UNKNOWN' not in text:
        print(text[:5000]); break
    if 'Status=FAILED' in text or 'Status=UNKNOWN' in text:
        print(text[:3000]); break
    time.sleep(5)
else:
    print('timeout', text[:1000])
PY"
