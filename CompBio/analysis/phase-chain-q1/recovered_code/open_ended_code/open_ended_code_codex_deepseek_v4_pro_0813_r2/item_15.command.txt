/bin/bash -lc "python3 - <<'PY'
import gzip
from itertools import islice

def read_fastq(fn):
    with gzip.open(fn,'rt') as f:
        while True:
            name=f.readline().strip()
            if not name: break
            seq=f.readline().strip(); plus=f.readline().strip(); qual=f.readline().strip()
            yield name,seq,qual

def rc(s):
    comp=str.maketrans('ACGT','TGCA')
    return s.translate(comp)[::-1]
r1=list(islice(read_fastq('phase.chain.q1.R1.fq.gz'),10))
r2=list(islice(read_fastq('phase.chain.q1.R2.fq.gz'),10))
for (n1,s1,q1),(n2,s2,q2) in zip(r1,r2):
    s2rc=rc(s2)
    # max overlap exact
    best=None
    for k in range(min(len(s1),len(s2rc)),9,-1):
        if s1[-k:]==s2rc[:k]:
            best=k; break
    print(n1, len(s1), 'overlap R1 tail vs R2rc head', best, 'insert estimated', len(s1)+len(s2)-best)
    print(' R1',s1[:60])
    print(' R2rc',s2rc[:60])
PY"
