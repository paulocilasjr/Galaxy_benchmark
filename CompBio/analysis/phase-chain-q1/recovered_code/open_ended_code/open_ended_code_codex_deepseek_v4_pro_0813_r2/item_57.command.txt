/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('aln_bwa.sorted.bam','rb')
ins=collections.Counter()
for r in bam.fetch():
    if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
    refpos=r.reference_start
    for op,length in r.cigartuples:
        if op==1:
            ins[(refpos,length)]+=1
        if op in (0,2,3): refpos+=length
        elif op==4: pass
bam.close()
print('insertions with count>=5')
for (pos,length),n in sorted(ins.items()):
    if n>=5:
        print(pos, 'length',length,'count',n, 'genomic1',pos+27078000+1 if pos is not None else None)
PY"
