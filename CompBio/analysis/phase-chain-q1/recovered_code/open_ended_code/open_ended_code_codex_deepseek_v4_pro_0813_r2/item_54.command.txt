/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('aln_bwa.sorted.bam','rb')
counts=collections.defaultdict(collections.Counter)
delcount=collections.Counter(); cover=collections.Counter()
for r in bam.fetch():
    if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
    refpos=r.reference_start; qpos=0
    for op,length in r.cigartuples:
        if op==0:
            for i in range(length):
                b=r.query_sequence[qpos+i]; q=r.query_qualities[qpos+i] if r.query_qualities else 20
                if q>=10:
                    counts[refpos+i][b]+=1; cover[refpos+i]+=1
            refpos+=length; qpos+=length
        elif op==1: qpos+=length
        elif op==2:
            if length==1: delcount[refpos]+=1
            refpos+=length
        elif op==3: refpos+=length
        elif op==4: qpos+=length
bam.close()
# genomic 1-based = local_start + p0; local_start=27078000
local_start=27078000
print('coverage min/max positions with cov>0')
covered=[p for p,n in cover.items() if n>0]
print(min(covered),max(covered), local_start+min(covered), local_start+max(covered))
# print all positions with two alleles >= max(1,0.3*total) and total>=3, excluding known
known={27079306,27079366,27079666,27079826,27080076,27080275,27080279,27080526}
print('candidate variants (local0, genomic1, cov, counts, del)')
for p in sorted(counts):
    c=counts[p]; tot=sum(c.values())
    if tot<3: continue
    sortedc=c.most_common()
    if len(sortedc)>=2:
        if sortedc[1][1]>=max(1,0.3*tot):
            gen=local_start+p
            if gen not in known:
                print(p,gen,tot,c.most_common(5),'del',delcount[p])
print('deletions')
for p,n in sorted(delcount.items()):
    if n>=3:
        print('del local',p,'gen',local_start+p,n,'bases',counts[p].most_common(5))
PY"
