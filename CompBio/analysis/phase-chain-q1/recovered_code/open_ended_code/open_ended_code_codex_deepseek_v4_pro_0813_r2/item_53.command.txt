/bin/bash -lc "python3 - <<'PY'
import pysam, collections
loci=[27079306,27079366,27079666,27079826,27080076,27080275,27080279,27080526]
X=['C','T','A','G','G','D','G','A']
Y=['A','C','T','D','C','A','D','G']
local_start=27078000
bam=pysam.AlignmentFile('aln_bwa.sorted.bam','rb')
def allele_at(r,pos):
    lp=pos-local_start
    for qp,rp in r.get_aligned_pairs(matches_only=False):
        if rp==lp:
            return 'D' if qp is None else r.query_sequence[qp]
    return None
reads={}
for r in bam.fetch():
    if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
    reads[r.query_name,'R1' if r.is_read1 else 'R2']=[allele_at(r,p) for p in loci]
bam.close()
pair_calls={}
for name in set(k[0] for k in reads if k[1]=='R1'):
    arr=[None]*len(loci)
    for mate in ['R1','R2']:
        for i,c in enumerate(reads.get((name,mate),[])):
            if c is None: continue
            if arr[i] is None: arr[i]=c
            elif arr[i]"'!=c: arr[i]=None
    pair_calls[name]=arr
scores=[]
for name,arr in pair_calls.items():
    sx=sy=0; n=0
    details=[]
    for i,a in enumerate(arr):
        if a is None: continue
        n+=1
        if a==X[i]: sx+=1
        if a==Y[i]: sy+=1
    if n==0: continue
    # classify best
    best=max(sx,sy)
    scores.append((name,best/n,sx,sy,n,arr))
# print some inconsistent pairs
bad=[s for s in scores if s[1]<1]
print('"'pairs',len(scores),'perfect',sum(s[1]==1 for s in scores),'bad',len(bad))
for s in bad[:20]:
    name,f,sx,sy,n,arr=s
    print(name,f,sx,sy,n,arr)
PY"
