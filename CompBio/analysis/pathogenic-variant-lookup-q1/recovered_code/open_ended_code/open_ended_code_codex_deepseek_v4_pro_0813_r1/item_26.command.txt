/bin/bash -lc "python3 - <<'PY'
import urllib.request, json

def get(url):
    req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
    return json.load(urllib.request.urlopen(req))
# cDNA (spliced transcript sequence)
for typ in ['cdna','cds']:
    d=get(f'https://rest.ensembl.org/sequence/id/ENST00000297261?type={typ};multiple_sequences=1')
    s=d[0]['seq'] if isinstance(d,list) else d['seq']
    print(typ, len(s), s[:120], s[-120:])
# genomic sequence exon A/B/C using region endpoints, plus strand
import urllib.parse
for name,start,end in [('A',155811823,155812463),('B',155806296,155806557),('C',155799980,155803726)]:
    url=f'https://rest.ensembl.org/sequence/region/human/{start}..{end}:1?coord_system_version=GRCh38'
    req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
    out=urllib.request.urlopen(req).read().decode()
    print(name, start,end, out[:200])
PY"
