rows=[]
for fname in ['clinvar_summary.json','clinvar_0001_summary.json']:
 data=json.load(open(fname))['result']
 for uid in data['uids']:

  if 'missense variant' not in r['molecular_consequence_list']:continue
  v=r['variation_set'][0];seq,pos,ref,alt=v['canonical_spdi'].split(':');pos=int(pos)
  assert 155811822<=pos<155812122 and seq=='NC_000007.14'
  score=next(x['value'] for x in json.load(open('alphaMissense_'+alt+'.json'))['alphaMissense_'+alt] if x['start']<=pos<x['end'])
  rows.append({'variant':f'chr7:{pos+1}','protein':r['protein_change'],'ref':ref,'alt':alt,'score':score,'omim':[x['db_id'] for x in v['variation_xrefs'] if x['db_source']=='OMIM']})
rows.sort(key=lambda r:r['score'],reverse=True)
open('ranked_variants.json','w').write(json.dumps(rows,indent=2)+'\\n')
answer=rows[0]['variant'];assert re.fullmatch(r'chr7:[1-9][0-9]*',answer)
