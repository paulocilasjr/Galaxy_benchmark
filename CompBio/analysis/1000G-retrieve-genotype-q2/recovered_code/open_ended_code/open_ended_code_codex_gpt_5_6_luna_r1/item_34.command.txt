/bin/bash -lc 'split -l 20 candidate_regions.tsv batch_regions_
for f in batch_regions_*; do
  bcftools query -R "$f" -f '"'%CHROM\\t%POS\\t%REF\\t%ALT[\\t%GT]\\n' 'https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.vcf.gz' > \""'$f.out" 2> "$f.err" &
done
wait
printf '"'%s\\n' BATCHES_DONE
for f in batch_regions_*.out; do
  cat \""'$f"
done > queried_vcf_parallel.tsv
wc -l queried_vcf_parallel.tsv
du -h queried_vcf_parallel.tsv'
