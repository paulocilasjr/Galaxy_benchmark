/bin/bash -lc 'PYTHONPATH="$PWD/deps" python3 - <<'"'PY'
import pysam
base='https://ftp-trace.ncbi.nlm.nih.gov/1000genomes/ftp/1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/'
ann=base+'20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.annotated.vcf.gz'
raw=base+'20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.vcf.gz'
cand={(x[0],int(x[1])) for x in map(lambda s:s.rstrip().split('\\t'),open('candidate_sites.tsv'))}
ivs=[(x[0],int(x[1]),int(x[2])) for x in map(lambda s:s.rstrip().split('\\t'),open('top1000_chr1_intervals.bed'))]
out=set()
with open('chr1_common_top1000_encode_1000G.tsv') as f:
 next(f)
 for line in f:
  x=line.split('\\t',4); out.add((x[0],int(x[1]),x[2],x[3]))
v=pysam.VariantFile(ann); found=set()
for c,s,e in ivs:
 for r in v.fetch(c,s,e):
  if (r.chrom,r.pos) in cand and len(r.ref)==1 and r.ref in 'ACGT' and r.alts and len(r.alts)==1 and len(r.alts[0])==1 and r.alts[0] in 'ACGT': found.add((r.chrom,r.pos,r.ref,r.alts[0]))
diff=sorted(found-out,key=lambda z:z[1])
print('annotated minus raw output',len(diff),diff)
v=pysam.VariantFile(raw)
for z in diff:
 rows=list(v.fetch(z[0],z[1]-1,z[1]))
 print('\\nTARGET',z)
 for r in rows: print(r.chrom,r.pos,r.ref,r.alts,r.id,r.filter.keys(),list(r.format.keys()))
PY"
