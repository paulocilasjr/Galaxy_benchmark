/bin/bash -lc "python - <<'PY'
from pathlib import Path
p=Path('genotypes_chr1_top1000.tsv')
lines=p.read_text().splitlines()
header=lines[0].split('\\t')
rows=[x.split('\\t') for x in lines[1:]]
print('lines',len(lines),'header_fields',len(header),'rows',len(rows))
print('header_prefix',header[:4])
print('sample_count',len(header)-4,'samples_sorted',header[4:]==sorted(header[4:]),'unique_samples',len(set(header[4:])))
coords=[(r[0],int(r[1])) for r in rows]
print('coordinate_sorted',coords==sorted(coords),'unique_coords',len(set(coords)))
print('metadata_strict',all(r[0]=='chr1' and len(r[2])==1 and r[2] in 'ACGT' and len(r[3])==1 and r[3] in 'ACGT' for r in rows))
allowed={'0/0','0/1','1/1','./.'}
print('allowed_gt',all(g in allowed for r in rows for g in r[4:]))
print('first_row',rows[0][:10])
print('last_row',rows[-1][:10])
PY
md5sum genotypes_chr1_top1000.tsv && md5 genotypes_chr1_top1000.tsv"
