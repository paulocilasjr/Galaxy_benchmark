/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
p=Path('genotypes.tsv')
with p.open() as f:
    header=f.readline().rstrip('\\n').split('\\t')
    rows=[line.rstrip('\\n').split('\\t') for line in f]
print('lines_including_header',1+len(rows),'columns',len(header),'samples',len(header)-4)
print('header_prefix',header[:4],'samples_sorted_unique',header[4:]==sorted(header[4:]) and len(set(header[4:]))==len(header[4:]))
print('rows_sorted_unique',all(int(a[1])<int(b[1]) for a,b in zip(rows,rows[1:])))
print('chroms',sorted(set(r[0] for r in rows)),'alleles_ok',all(len(r[2])==len(r[3])==1 and r[2] in 'ACGT' and r[3] in 'ACGT' for r in rows),'gts_ok',all(g in {'0/0','0/1','1/1','./.'} for r in rows for g in r[4:]))
print('first',rows[0][:8]); print('last',rows[-1][:8])
PY
md5sum genotypes.tsv | awk '{print \"md5\","'$1}'"'
wc -l genotypes.tsv"
