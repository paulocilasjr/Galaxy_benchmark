/bin/bash -lc "mkdir -p region_parts query_parts
split -l 20 common_snps.regions.bed region_parts/part_
rawbase='https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.vcf.gz'
for part in region_parts/part_*; do
 name="'${part##*/}
 (bcftools view -R "$part" -S sorted_samples.txt -Ou "$rawbase" 2>"query_parts/$name.err" | bcftools query -f '"'%CHROM\\t%POS\\t%REF\\t%ALT[\\t%GT]\\n' -o \"query_parts/"'$name.tsv") &
done
wait
for f in query_parts/*.tsv; do wc -l "$f"; done'
