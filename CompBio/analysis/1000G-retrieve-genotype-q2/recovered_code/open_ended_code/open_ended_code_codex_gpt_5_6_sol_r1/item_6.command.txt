/bin/bash -lc 'gzip -dc ENCFF717TCQ.bed.gz | head -3
gzip -dc ENCFF717TCQ.bed.gz | wc -l
gzip -dc ENCFF717TCQ.bed.gz | sort -t $'"'\\t' -k9,9nr | head -1000 > top1000.bed
awk -F '\\t' '"'$1=="chr1"'"' top1000.bed | sort -k2,2n > top1000.chr1.bed
wc -l top1000.bed top1000.chr1.bed
head -5 top1000.chr1.bed
curl -L --fail --silent --show-error 'https://api.genome.ucsc.edu/getData/track?genome=hg38;track=snp151Common;chrom=chr1;start=1000000;end=1100000' -o ucsc_test.json
python3 - <<'PY'
import json
j=json.load(open('ucsc_test.json'))
print(j.keys())
print(len(j.get('snp151Common',[])))
print(j.get('snp151Common',[None])[0])
PY"
