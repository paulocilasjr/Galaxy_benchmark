/bin/bash -lc "python3 - <<'PY'
seen=set()
with open('data/selected_raw.vcf') as f:
    for line in f:
        if line.startswith('#'): continue
        p=int(line.split('\\t')[1])
        seen.add(p)
pos=[int(x.split()[1]) for x in open('data/selected_snp_positions.txt')]
missing=[p for p in pos if p not in seen]
print('missing',missing)
# count duplicates
from collections import Counter
c=Counter(p for line in open('data/selected_raw.vcf') if not line.startswith('#') for p in [int(line.split('\\t')[1])])
dups=[(p,n) for p,n in c.items() if n>1]
print('dups',dups)
PY"
