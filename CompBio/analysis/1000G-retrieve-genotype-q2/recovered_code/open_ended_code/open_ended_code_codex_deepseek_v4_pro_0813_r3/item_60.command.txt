/bin/bash -lc "python3 - <<'PY'
sel=set(int(x.split()[1]) for x in open('data/selected_snp_positions.txt'))
valid=set()
for line in open('data/selected_raw.vcf'):
    if line.startswith('#'): continue
    f=line.rstrip('\\n').split('\\t')
    chrom,pos,ref,alt=f[0],int(f[1]),f[3],f[4]
    if chrom=='chr1' and pos in sel and len(ref)==1 and len(alt)==1 and ref in 'ACGT' and alt in 'ACGT':
        valid.add(pos)
out=set(int(x.split('\\t')[1]) for x in open('genotypes.tsv').read().splitlines()[1:])
print('selected',len(sel),'valid selected VCF',len(valid),'output',len(out))
print('missing valid not output',sorted(valid-out))
print('extra output',sorted(out-valid))
PY"
