/bin/bash -lc "tail -1 data/selected_raw.vcf | cut -f1,2,4,5; grep -vc '"'^#'"' data/selected_raw.vcf; python3 - <<'PY'
seen=set()
with open('data/selected_raw.vcf') as f:
    for line in f:
        if line.startswith('#'): continue
        seen.add(int(line.split('\\t')[1]))
pos=[int(x.split()[1]) for x in open('data/selected_snp_positions.txt')]
rem=[p for p in pos if p not in seen]
print('remaining candidate positions',len(rem))
print(rem)
PY"
