/bin/bash -lc "python3 - <<'PY'
from collections import Counter
import sys
cnt=Counter()
bad=[]
ncols=None
with open('genotypes.tsv') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    print('header cols',len(header),'first4',header[:4],'last',header[-3:])
    for ln,line in enumerate(f,2):
        fields=line.rstrip('\\n').split('\\t')
        if ncols is None: ncols=len(fields)
        if len(fields)"'!=ncols:
            bad.append((ln,len(fields))); break
        # check CHROM POS REF ALT
        if fields[0]!='"'chr1' or fields[2] not in 'ACGT' or fields[3] not in 'ACGT':
            bad.append(('fixed',ln,fields[:4]))
        for val in fields[4:]:
            cnt[val]+=1
            if val not in ('0/0','0/1','1/1','./.'):
                bad.append((ln,val))
print('ncols',ncols,'data lines',ln,'counts',cnt,'bad count',len(bad))
if bad[:10]: print(bad[:10])
# ensure sorted POS
pos=[int(x.split('\\t')[1]) for x in open('genotypes.tsv').read().splitlines()[1:]]
print('sorted',pos==sorted(pos),'unique',len(set(pos)),len(pos))
PY"
