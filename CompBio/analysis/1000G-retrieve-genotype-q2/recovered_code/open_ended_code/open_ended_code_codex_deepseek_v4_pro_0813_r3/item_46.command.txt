/bin/bash -lc "tail -1 data/selected_raw.vcf | cut -f1,2,4,5; python3 - <<'PY'
# compare selected positions and positions already in file to estimate remaining
seen=set()
with open('data/selected_raw.vcf') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')[1]
        seen.add(int(p))
pos=[int(x.split()[1]) for x in open('data/selected_snp_positions.txt')]
rem=[p for p in pos if p not in seen]
print('seen',len(seen),'remaining',len(rem),'maxseen',max(seen) if seen else None,'minrem',min(rem) if rem else None,'maxrem',max(rem) if rem else None)
PY"
