/bin/bash -lc "VCF='https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.vcf.gz'
awk 'BEGIN{OFS=\"\\t\"}{print "'$1,$3,$4}'"' tmp/snp151Common.top1000.chr1.filtered.tsv > tmp/snp_targets.bed
bcftools query -l \""'$VCF" | LC_ALL=C sort > tmp/samples.lexicographic.txt
bcftools view -R tmp/snp_targets.bed "$VCF" | python3 scripts/make_tsv.py tmp/samples.lexicographic.txt out/answer.tsv
wc -l out/answer.tsv
md5sum out/answer.tsv
awk -F'"'\\t' 'NR==1{print NF, "'$1,$2,$3,$4,$5,$NF; next} {bad=0; for(i=5;i<=NF;i++) if($i!="0/0"&&$i!="0/1"&&$i!="1/1"&&$i!="./.") bad++; if(bad){print "bad",NR,bad; exit}} END{if(!bad) print "genotypes_ok"}'"' out/answer.tsv"
