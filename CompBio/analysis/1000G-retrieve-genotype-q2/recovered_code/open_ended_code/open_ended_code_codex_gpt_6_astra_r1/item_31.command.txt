/bin/bash -lc "python3 - <<'PY'
import zlib
p='data/selected_raw.vcf.gz'
b=open(p,'rb').read()
text=b''
while b:
 d=zlib.decompressobj(31)
 try:
  text+=d.decompress(b)
 except zlib.error:
  break
 if not d.eof:break
 b=d.unused_data
rows=[r.split('\\t',3)[:3] for r in text.decode(errors='replace').splitlines() if not r.startswith('#')]
print('retrieval progress:',len(rows),'records; last:',rows[-1] if rows else None)
PY"
