/bin/bash -lc '# Top 1000 genome-wide peaks by qValue, chr1 subset
(gzip -cd data/ENCFF717TCQ.bed.gz | sort -t $'"'\\t' -k9,9gr | head -1000 || true) \\
  | awk -v OFS='\\t' '"'$1=="chr1" {print $1,$2,$3}'"' \\
  | sort -k1,1 -k2,2n -k3,3n > work/top1000_peaks.chr1.bed

# chr1 common SNPs with exactly two distinct observed A/C/G/T alleles.
gzip -cd data/snp151Common.txt.gz \\
  | awk -v OFS='\\t' '"'$2=="chr1" && $10 ~ /''^[ACGT]'"\\/[ACGT]"'$/ {split($10,a,"/"); if (a[1] != a[2]) print $2,$3,$3+1,$5,$10}'"' \\
  | sort -k1,1 -k2,2n -k3,3n > work/snp151Common.chr1.biallelic_acgt.bed

python3 - <<'PY'
from pathlib import Path
peaks=[]
with open('work/top1000_peaks.chr1.bed') as f:
    for line in f:
        c,s,e=line.rstrip('\\n').split('\\t')[:3]
        peaks.append((int(s),int(e)))
peaks.sort()
# merge peaks for simpler containment/overlap
merged=[]
for s,e in peaks:
    if not merged or s>merged[-1][1]:
        merged.append([s,e])
    elif e>merged[-1][1]:
        merged[-1][1]=e
out=[]
i=0
with open('work/snp151Common.chr1.biallelic_acgt.bed') as f:
    for line in f:
        c,s,e,name,obs=line.rstrip('\\n').split('\\t')[:5]
        s=int(s); e=int(e)
        while i < len(merged) and merged[i][1] <= s:
            i += 1
        if i < len(merged) and merged[i][0] < e and s < merged[i][1]:
            out.append((c,s,e,name,obs))
# unique by coordinate; if UCSC has duplicate rs at same base, keep coordinate once.
seen=set()
with open('work/target_snps.bed','w') as bed, open('work/target_snps.regions','w') as reg:
    for c,s,e,name,obs in out:
        key=(c,s,e)
        if key in seen:
            continue
        seen.add(key)
        bed.write(f'{c}\\t{s}\\t{e}\\n')
        reg.write(f'{c}:{s+1}-{s+1}\\n')
print('chr1_top1000_peaks', len(peaks), 'merged', len(merged))
print('chr1_biallelic_common_snps', sum(1 for _ in open('work/snp151Common.chr1.biallelic_acgt.bed')))
print('target_snps', len(seen))
PY
wc -l work/top1000_peaks.chr1.bed work/snp151Common.chr1.biallelic_acgt.bed work/target_snps.bed
head -5 work/target_snps.regions; tail -5 work/target_snps.regions"
