/bin/bash -lc "gzip -cd data/snp151Common.txt.gz \\
  | awk -v OFS='\\t' '"'$2=="chr1" && $4==$3+1 && $10 ~ /''^[ACGT]'"\\/[ACGT]"'$/ {split($10,a,"/"); if (a[1] != a[2]) print $2,$3,$4,$5,$10}'"' \\
  | sort -k1,1 -k2,2n -k3,3n > work/snp151Common.chr1.biallelic_acgt.len1.bed
python3 - <<'PY'
peaks=[]
for line in open('work/top1000_peaks.chr1.bed'):
    c,s,e=line.rstrip('\\n').split('\\t')[:3]
    peaks.append((int(s),int(e)))
peaks.sort()
merged=[]
for s,e in peaks:
    if not merged or s>merged[-1][1]: merged.append([s,e])
    elif e>merged[-1][1]: merged[-1][1]=e
seen=set(); i=0
with open('work/target_snps.len1.bed','w') as bed:
    for line in open('work/snp151Common.chr1.biallelic_acgt.len1.bed'):
        c,s,e,name,obs=line.rstrip('\\n').split('\\t')[:5]
        s=int(s); e=int(e)
        while i < len(merged) and merged[i][1] <= s: i += 1
        if i < len(merged) and merged[i][0] < e and s < merged[i][1] and (c,s,e) not in seen:
            seen.add((c,s,e)); bed.write(f'{c}\\t{s}\\t{e}\\n')
print(len(seen))
PY
wc -l work/snp151Common.chr1.biallelic_acgt.len1.bed work/target_snps.len1.bed
comm -3 <(cut -f1-3 work/target_snps.bed | sort) <(cut -f1-3 work/target_snps.len1.bed | sort) | head -20"
