/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
p=Path('out/genotypes.raw20201028.tsv')
samples=[x.strip() for x in open('work/raw_samples.lex.txt')]
valid={'0/0','0/1','1/1','./.'}
last=0
n=0
bad=[]
with p.open() as f:
    header=f.readline().rstrip('\\n').split('\\t')
    print('header_cols',len(header),'expected',4+len(samples),'header_samples_match',header[4:]==samples)
    for ln,line in enumerate(f, start=2):
        fields=line.rstrip('\\n').split('\\t')
        n+=1
        if len(fields)"'!=len(header): bad.append((ln,'"'width',len(fields)))
        chrom,pos,ref,alt=fields[:4]
        pos=int(pos)
        if chrom"'!='"'chr1' or pos<last: bad.append((ln,'sort',chrom,pos,last))
        last=pos
        if len(ref)"'!=1 or ref not in '"'ACGT' or len(alt)"'!=1 or alt not in '"'ACGT': bad.append((ln,'allele',ref,alt))
        wrong=set(fields[4:])-valid
        if wrong: bad.append((ln,'gt',sorted(wrong)[:5]))
print('variant_rows',n)
print('bad_count',len(bad))
if bad[:10]: print(bad[:10])
PY
wc -l out/genotypes.raw20201028.tsv
md5sum out/genotypes.raw20201028.tsv"
