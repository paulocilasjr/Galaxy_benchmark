/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
# Max q per unique interval, top 1000 unique intervals genome-wide.
best={}
with __import__('gzip').open('data/ENCFF717TCQ.bed.gz','rt') as f:
    for line in f:
        if not line.strip() or line.startswith(('#','track','browser')): continue
        fields=line.rstrip('\\n').split('\\t')
        key=tuple(fields[:3])
        q=float(fields[8])
        if key not in best or q>best[key]: best[key]=q
rows=sorted(best.items(), key=lambda kv: kv[1], reverse=True)[:1000]
chr1=[(int(k[1]),int(k[2]),q) for k,q in rows if k[0]=='chr1']
print('unique_total',len(best),'top1000_unique_chr1',len(chr1))
# intersect with len1 SNPs
peaks=sorted((s,e) for s,e,q in chr1)
merged=[]
for s,e in peaks:
    if not merged or s>merged[-1][1]: merged.append([s,e])
    elif e>merged[-1][1]: merged[-1][1]=e
seen=set(); i=0
for line in open('work/snp151Common.chr1.biallelic_acgt.len1.bed'):
    c,s,e,*_=line.rstrip('\\n').split('\\t')
    s=int(s); e=int(e)
    while i<len(merged) and merged[i][1]<=s: i+=1
    if i<len(merged) and merged[i][0]<e and s<merged[i][1]: seen.add((c,s,e))
print('unique_alt_targets',len(seen),'merged_peaks',len(merged))
PY"
