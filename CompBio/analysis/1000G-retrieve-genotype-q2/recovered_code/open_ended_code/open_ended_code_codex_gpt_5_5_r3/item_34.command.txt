/bin/bash -lc "set -euo pipefail
# Top 1000 genome-wide peaks by narrowPeak qValue (column 9), then retain chr1 intervals.
gzip -cd data/ENCFF717TCQ.bed.gz \\
  | sort -t "'$'"'\\t' -k9,9gr \\
  | head -1000 \\
  | awk -v OFS='\\t' '"'$1=="chr1" {print $1,$2,$3}'"' \\
  | sort -k1,1 -k2,2n -k3,3n > work/top1000_peaks.chr1.bed

# UCSC snp151Common: chr1, observed exactly B1/B2 where B1,B2 are distinct A/C/G/T.
gzip -cd data/snp151Common.txt.gz \\
  | awk -v OFS='\\t' '"'$2=="chr1" && $10 ~ /''^[ACGT]'"\\/ [ACGT]"'$/ {print}'"' > /dev/null"
