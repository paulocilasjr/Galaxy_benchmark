/bin/bash -lc "command -v aws || true
curl -L --fail --silent 'https://1000genomes.s3.amazonaws.com/?list-type=2&prefix=1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/&delimiter=/' -o s3_listing.xml

import xml.etree.ElementTree as ET
root=ET.parse('s3_listing.xml').getroot()
ns={'s':'http://s3.amazonaws.com/doc/2006-03-01/'}
cs=root.findall('s:Contents',ns)
print('count',len(cs), 'truncated',root.findtext('s:IsTruncated',namespaces=ns))
for c in cs[:100]:
 print(c.findtext('s:Key',namespaces=ns),c.findtext('s:Size',namespaces=ns))
PY"
