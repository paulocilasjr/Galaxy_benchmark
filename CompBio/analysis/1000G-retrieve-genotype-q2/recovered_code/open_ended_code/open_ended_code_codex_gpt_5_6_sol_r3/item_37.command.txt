/bin/bash -lc "printf 'records=' && bcftools index -n raw_candidates.vcf.gz
printf 'samples=' && bcftools query -l raw_candidates.vcf.gz | wc -l
bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\n' raw_candidates.vcf.gz | awk 'length("'$3)==1 && $3~/''^[ACGT]$/ && length($4)==1 && $4~/''^[ACGT]$/'"' | wc -l
bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\n' raw_candidates.vcf.gz | head -10"
