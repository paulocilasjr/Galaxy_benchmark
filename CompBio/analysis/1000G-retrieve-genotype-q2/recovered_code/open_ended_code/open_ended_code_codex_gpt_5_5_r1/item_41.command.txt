/bin/bash -lc 'set -euo pipefail
base=https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20201028_3202_raw_GT_with_annot/20201028_CCDG_14151_B01_GRM_WGS_2020-08-05_chr1.recalibrated_variants.vcf.gz
if [ -e data/vcf_target_gt_lex.tsv ]; then mv data/vcf_target_gt_lex.tsv data/vcf_target_gt_lex.position_partial.tsv; fi
bcftools view -S data/1000g_samples_lex.txt -R data/top1000_peaks_chr1.bed -T data/query_regions.tsv -Ou "$base" '"\\
  | bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT[\\t%GT]\\n' > data/vcf_target_gt_lex.tsv
wc -l data/vcf_target_gt_lex.tsv
awk -F'\\t' '{print NF; exit}' data/vcf_target_gt_lex.tsv
head -n 1 data/vcf_target_gt_lex.tsv | cut -f1-8"
