/bin/bash -lc "set -euo pipefail
python3 scripts/write_final_matrix.py data/1000g_samples_lex.txt data/candidate_ucsc_snps_chr1_toppeaks.bed data/vcf_target_gt_lex.tsv out/1000g_chr1_encode_ucsc_common_snps.tsv
wc -l out/1000g_chr1_encode_ucsc_common_snps.tsv
awk -F'\\t' 'NR==1{print NF; print "'$1,$2,$3,$4,$5,$NF; exit}'"' out/1000g_chr1_encode_ucsc_common_snps.tsv
awk -F'\\t' 'NR>1{for(i=5;i<=NF;i++) if("'$i!="0/0" && $i!="0/1" && $i!="1/1" && $i!="./."){print "bad",NR,i,$i; exit 1}} END{print "gt_ok"}'"' out/1000g_chr1_encode_ucsc_common_snps.tsv
awk -F'\\t' 'NR>1{if(prev && "'$2+0<prev){print "sort_bad",NR; exit 1} prev=$2+0} END{print "sort_ok"}'"' out/1000g_chr1_encode_ucsc_common_snps.tsv
md5sum out/1000g_chr1_encode_ucsc_common_snps.tsv"
