/bin/bash -lc "set -euo pipefail
curl -L https://hgdownload.soe.ucsc.edu/goldenPath/hg38/database/snp151Common.txt.gz \\
  | gzip -cd \\
  | awk -F'\\t' 'BEGIN{OFS=\"\\t\"} "'$2=="chr1" {n=split($10,a,"/"); if(n==2 && length(a[1])==1 && length(a[2])==1 && a[1]!=a[2] && a[1]~/''^[ACGT]$/ && a[2]~/''^[ACGT]$/) print $2,$3,$4,$5,$10}'"' \\
  | sort -k1,1 -k2,2n > data/snp151Common_chr1_biallelic_acgt.bed
bedtools intersect -u -a data/snp151Common_chr1_biallelic_acgt.bed -b data/top1000_peaks_chr1.bed \\
  | awk 'BEGIN{OFS=\"\\t\"}{print "'$1,$2,$3,$4,$5,$2+1}'"' \\
  | sort -k1,1 -k2,2n -u > data/candidate_ucsc_snps_chr1_toppeaks.bed
wc -l data/snp151Common_chr1_biallelic_acgt.bed data/candidate_ucsc_snps_chr1_toppeaks.bed
head -n 5 data/candidate_ucsc_snps_chr1_toppeaks.bed
tail -n 5 data/candidate_ucsc_snps_chr1_toppeaks.bed"
