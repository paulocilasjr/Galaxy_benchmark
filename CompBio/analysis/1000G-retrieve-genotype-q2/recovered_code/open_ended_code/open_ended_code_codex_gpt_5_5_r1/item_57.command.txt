/bin/bash -lc "set -euo pipefail
python3 - <<'PY'
from collections import Counter,defaultdict
BASES=set('ACGT')
# target positions
target=set()
for line in open('data/candidate_ucsc_snps_chr1_toppeaks.bed'):
    f=line.rstrip('\\n').split('\\t')
    target.add((f[0], int(f[5])))
reasons=Counter(); valid=[]; bypos=defaultdict(list)
for line in open('data/vcf_target_gt_lex.tsv'):
    f=line.rstrip('\\n').split('\\t')
    chrom,pos,ref,alt=f[:4]; pos=int(pos)
    if (chrom,pos) not in target:
        reasons['not_target']+=1; continue
    if ref not in BASES or alt not in BASES:
        reasons[f'invalid_ref_alt:{ref}>{alt}']+=1; continue
    valid.append((chrom,pos,ref,alt)); bypos[(chrom,pos)].append((ref,alt))
print('vcf_rows', sum(1 for _ in open('data/vcf_target_gt_lex.tsv')))
print('valid_rows', len(valid), 'unique_records', len(set(valid)), 'unique_positions', len(bypos))
print('reasons', reasons)
print('positions_with_multiple_valid', sum(1 for v in bypos.values() if len(v)>1))
for k,v in bypos.items():
    if len(v)>1:
        print(k,v)
PY
wc -l out/1000g_chr1_encode_ucsc_common_snps.tsv
md5sum out/1000g_chr1_encode_ucsc_common_snps.tsv"
