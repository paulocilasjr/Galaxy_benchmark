/bin/bash -lc "python3 - <<'PY'
import pymysql
# load intervals
intervals=[]
with open('data/top1000_peaks.bed') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if p[0]=='chr1': intervals.append((int(p[1]),int(p[2])))
conn=pymysql.connect(host='genome-mysql.soe.ucsc.edu', user='genome', password='', db='hg38', read_timeout=300, connect_timeout=30)
cur=conn.cursor()
conds=' OR '.join(['(chromStart < %d AND chromEnd > %d)'%(e,s) for s,e in intervals])
# broad: observed regex only, intersect intervals, no class/locType
sql=\"SELECT chromStart,chromEnd,name,observed,class,locType FROM snp151Common WHERE chrom='chr1' AND observed REGEXP '"'^[ACGT]/[ACGT]$'"' AND (\"+conds+\") ORDER BY chromStart\"
cur.execute(sql)
rows=cur.fetchall()
print('broad candidate rows', len(rows))
exact=[r for r in rows if r[4]=='single' and r[5]=='exact']
nonexact=[r for r in rows if not (r[4]=='single' and r[5]=='exact')]
print('exact',len(exact),'nonexact',len(nonexact))
for r in nonexact: print(r)
cur.close(); conn.close()
PY"
