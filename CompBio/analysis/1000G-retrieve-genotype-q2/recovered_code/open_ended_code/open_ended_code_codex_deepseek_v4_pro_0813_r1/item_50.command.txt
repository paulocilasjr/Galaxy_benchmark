/bin/bash -lc "python3 - <<'PY'
import csv
cand=set()
with open('data/candidates.tsv') as f:
    r=csv.DictReader(f,delimiter='\\t')
    for row in r: cand.add(int(row['POS']))
print('cand', len(cand))
records={}
with open('data/candidate_raw.vcf') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        pos=int(p[1]); ref=p[3]; alt=p[4]
        records[pos]=(ref,alt,p[2])
print('records in raw fetched', len(records))
missing=cand-records.keys()
print('candidate positions absent from fetched raw', len(missing), sorted(missing)[:50])
nonacgt=[]
for pos in cand & records.keys():
    ref,alt,vid=records[pos]
    if len(ref)"'!=1 or ref not in '"'ACGT' or len(alt)"'!=1 or alt not in '"'ACGT':
        nonacgt.append((pos,ref,alt,vid))
print('candidate pos present but not biallelic ACGT', len(nonacgt))
for x in nonacgt[:50]: print(x)
PY"
