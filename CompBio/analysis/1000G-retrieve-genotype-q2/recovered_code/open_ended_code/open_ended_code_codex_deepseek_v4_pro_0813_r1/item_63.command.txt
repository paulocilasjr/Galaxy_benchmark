/bin/bash -lc "python3 - <<'PY'
import sys
# get original sample order from candidate_raw.vcf
samples=[]
with open('data/candidate_raw.vcf') as f:
    for line in f:
        if line.startswith('#CHROM'):
            p=line.rstrip('\\n').split('\\t')
            samples=p[9:]
            break
out_samples=sorted(samples)
def norm(gt):
    s=gt.replace('|','/')
    parts=s.split('/')
    if len(parts)"'!=2: return '"'./.'
    if all(x in '01' for x in parts):
        t=tuple(sorted(parts, key=lambda x:int(x)))
        return {'00':'0/0','01':'0/1','11':'1/1'}.get(t[0]+t[1], './.')
    return './.'
# candidate positions
cand=set()
with open('data/candidates.tsv') as f:
    f.readline()
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        cand.add(int(p[2]))
rows=[]
with open('data/query_rows.tsv') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)"'!=4+len(samples):
            print('"'bad cols', p[0],p[1],len(p),file=sys.stderr); continue
        chrom,pos,ref,alt=p[0],int(p[1]),p[2],p[3]
        if chrom"'!='"'chr1' or pos not in cand: continue
        if len(ref)"'!=1 or ref not in '"'ACGT': continue
        if len(alt)"'!=1 or alt not in '"'ACGT': continue
        # map original order to normalized, then sorted
        orig=p[4:]
        d=dict(zip(samples,map(norm,orig)))
        row=[chrom,str(pos),ref,alt]+[d[s] for s in out_samples]
        rows.append((pos,row))
rows.sort(key=lambda x:x[0])
with open('data/verify.tsv','w') as f:
    f.write('\\t'.join(['CHROM','POS','REF','ALT']+out_samples)+'\\n')
    for _,row in rows: f.write('\\t'.join(row)+'\\n')
print('verify rows',len(rows))
PY
diff -q data/output.tsv data/verify.tsv; echo diff_exit="'$?; wc -l data/verify.tsv; md5sum data/verify.tsv'
