"'$/="'"\\n>\";
my "'$q="CACACACAGGAGAT";
my $rc=reverse $q; $rc =~ tr/ACGT/TGCA/;
while (<>) {
  s/''^>//;
  my ($h,@lines)=split /'"\\n/;
  my ("'$id)=split /'"\\s+/, "'$h;
  my $s=uc(join("",@lines));
  for my $pair (["+",$q],["-",$rc]) {
    my ($strand,$needle)=@$pair;
    my $off=0;
    while (($off=index($s,$needle,$off)) >= 0) {
      print "$id'"\\t"'$off'"\\t\",("'$off+length($needle)),"'"\\t"'$strand'"\\t\",substr("'$s,$off,length($needle)),"'"\\n\";
      "'$off++;
    }
  }
}
'"' GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna"
