from pathlib import Path
p=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
want='NT_033779.5'; name=None; pos=0; linebuf=[]
import sys
with p.open() as f:
    for raw in f:
        line=raw.rstrip('\\n')
        if line.startswith('>'):
            if name==want:
                break
            name=line[1:].split()[0]
            pos=0; linebuf=[]
        elif name==want:
            s=line.strip()
            for ch in s:
                linebuf.append(ch); pos+=1
                if len(linebuf)>14: linebuf.pop(0)
                if len(linebuf)==14 and ''.join(linebuf)=='CACACACAGGAGAT':
                    print('hit at 0based', pos-14)
                    sys.exit()
print('none')
