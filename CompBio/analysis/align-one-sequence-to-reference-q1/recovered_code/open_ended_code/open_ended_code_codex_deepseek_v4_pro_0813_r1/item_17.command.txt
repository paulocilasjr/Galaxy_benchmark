/bin/bash -c "python3 - <<'PY'
from Bio import SeqIO
from pathlib import Path
q='CACACACAGGAGAT'
comp=str.maketrans('ACGT','TGCA')
rc=q.translate(comp)[::-1]
for rec in SeqIO.parse('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna','fasta'):
    s=str(rec.seq).upper()
    hits=[]
    for qq,label in [(q,'+'),(rc,'rc')]:
        start=0
        while True:
            i=s.find(qq,start)
            if i<0:break
            hits.append((i,i+len(q),label))
            start=i+1
    if hits:
        print(rec.id, len(s), hits[:10], 'count', len(hits))
PY"
