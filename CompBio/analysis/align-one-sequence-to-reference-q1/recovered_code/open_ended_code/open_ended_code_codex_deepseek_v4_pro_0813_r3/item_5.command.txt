/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
s='CACACACAGGAGAT'
rev=str.maketrans('ACGTNacgtn','TGCNAtgcna')
rc=s.translate(rev)[::-1]
print('rc',rc, flush=True)
path=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
data=path.read_text()
chunks=data.split('>')
found=[]
for chunk in chunks[1:]:
    lines=chunk.splitlines()
    header=lines[0].split()[0]
    seq=''.join(lines[1:]).upper()
    for strand,pat in [('+',s),('-',rc)]:
        start=0
        while True:
            i=seq.find(pat,start)
            if i<0: break
            found.append((header,i,i+len(pat),strand,len(seq)))
            start=i+1
print('found',len(found), flush=True)
for f in found:
    print(f)
PY"
