/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
p=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
seq='CACACACAGGAGAT'
rc=str.maketrans('ACGT','TGCA')
rev=seq.translate(rc)[::-1]
print('len',len(seq))
name=None
buf=[]
for line in p.open():
    line=line.strip()
    if not line: continue
    if line.startswith('>'):
        if name:
            s=''.join(buf)
            f=s.find(seq)
            r=s.find(rev)
            print(name, 'forward', f, 'reverse', r, 'len', len(s))
        name=line[1:].split()[0]
        buf=[]
    else:
        buf.append(line.upper())
if name:
    s=''.join(buf)
    f=s.find(seq)
    r=s.find(rev)
    print(name, 'forward', f, 'reverse', r, 'len', len(s))
PY"
