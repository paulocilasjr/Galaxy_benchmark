/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
fn=Path('GCF_000001215.4_Release_6_plus_ISO1_MT_genomic.fna')
q='CACACACAGGAGAT'
seqs={}
name=None
parts=[]
for line in fn.open():
    line=line.strip()
    if not line: continue
    if line.startswith('>'):
        if name is not None: seqs[name]=''.join(parts).upper()
        name=line[1:].split()[0]
        parts=[]
    else:
        parts.append(line)
if name is not None: seqs[name]=''.join(parts).upper()
comp=str.maketrans('ACGTNacgtn','TGCANtgcan')
rc=q.translate(comp)[::-1].upper()
print('contigs',len(seqs),'query len',len(q),'rc',rc)
for orient,s in [('+',q),('-',rc)]:
    count=0
    for name,seq in seqs.items():
        start=0
        while True:
            i=seq.find(s,start)
            if i==-1: break
            print(name,i,i+len(s),orient)
            count+=1
            start=i+1
    print('count',orient,count)
PY"
