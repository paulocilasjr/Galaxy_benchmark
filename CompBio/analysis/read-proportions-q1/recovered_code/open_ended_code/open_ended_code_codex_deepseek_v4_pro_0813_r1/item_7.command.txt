/bin/bash -lc "python3 - <<'PY'
import gzip
from pathlib import Path
from collections import defaultdict
seqs={}
cur=None
for line in Path('read.proportions.q1.fa').read_text().splitlines():
    if line.startswith('>'): cur=line[1:].strip(); seqs[cur]=[]
    else: seqs[cur].append(line.strip())
for c in seqs: seqs[c]=''.join(seqs[c])
comp=str.maketrans('ACGT','TGCA')
K=36
# sets per read? create dict kmer -> list (c,strand)
d=defaultdict(list)
for c,s in seqs.items():
    for i in range(len(s)-K+1):
        d[s[i:i+K]].append((c,'F'))
    rc=s.translate(comp)[::-1]
    for i in range(len(rc)-K+1):
        d[rc[i:i+K]].append((c,'R'))
cnt=defaultdict(int); both=0
n=0
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h:break
        seq=f.readline().strip(); f.readline(); f.readline(); n+=1
        hits=d.get(seq,[])
        strands=set(x[1] for x in hits)
        if len(strands)==1: cnt[next(iter(strands))]+=1
        elif len(strands)==2: both+=1
        else: cnt['none']+=1
print(n, dict(cnt), 'bothstrands exact',both)
PY"
