/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict,Counter
import gzip,math,numpy as np
from scipy.optimize import minimize
fa={}
for l in open('read.proportions.q1.fa'):
 l=l.strip()
 if l.startswith('>'):n=l[1:];fa[n]=''
 else:fa[n]+=l
keys=list(fa);M=2965;comp=str.maketrans('ACGTN','TGCAN');rc=lambda s:s.translate(comp)[::-1]
ix=defaultdict(set)
for ci,c in enumerate(keys):
 for p in range(M): ix[fa[c][p:p+36]].add(ci)
# predictions orientation-forward exact match sets
pred=[]
for c in keys:pred.append(Counter(tuple(sorted(ix.get(fa[c][p:p+36],set()))) for p in range(M)))
# observed fwd and rev separately
ob={'f':Counter(),'r':Counter()}; N={'f':0,'r':0}
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
 while f.readline():
  q=f.readline().strip();f.readline();f.readline()
  a=tuple(sorted(ix.get(q,set())));b=tuple(sorted(ix.get(rc(q),set())))
  if a:ob['f'][a]+=1;N['f']+=1
  elif b:ob['r'][b]+=1;N['r']+=1
classes=sorted(set(ob['f'])|set(ob['r']))
A=np.array([[pred[c][S]/M for c in range(4)] for S in classes])
for ori in 'fr':
 y=np.array([ob[ori][S]/N[ori] for S in classes])
 fun=lambda x:-sum(y[j]*math.log(max(1e-300,float(A[j]@x))) for j in range(len(classes)))
 res=minimize(fun,[.25]*4,bounds=[(0,1)]*4,constraints={'type':'eq','fun':lambda x:x.sum()-1},method='SLSQP',options={'ftol':1e-14,'maxiter':1000})
 print(ori,N[ori],res.x,res.fun)
 print('obs vs expected at mle')
 for S,row,yy in zip(classes,A,y): print(''.join(str(i+1) for i in S),ob[ori][S],round(yy,5),round(row@res.x,5))
PY"
