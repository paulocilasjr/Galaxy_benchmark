/bin/bash -lc "python3 - <<'PY'
import gzip
fa={}
name=None
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'):
        name=line[1:].split()[0]; fa[name]=''
    else: fa[name]+=line
names=list(fa)
rc=str.maketrans('ACGTNacgtn','TGCANtgcan')
def revcomp(s): return s.translate(rc)[::-1]
counts={n:0 for n in names}; unique=both=none=0; examples=[]
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h: break
        seq=f.readline().strip(); f.readline(); f.readline()
        hits=[]
        for n in names:
            x=fa[n].find(seq); y=fa[n].find(revcomp(seq))
            if x>=0 or y>=0: hits.append(n)
        if len(hits)==1: counts[hits[0]]+=1; unique+=1
        elif len(hits)==0:
            none+=1
            if len(examples)<5: examples.append(seq)
        else: both+=1
print('names',names,'exact unique',unique,'ambiguous',both,'none',none)
print(counts)
print('examples none',examples)
PY"
