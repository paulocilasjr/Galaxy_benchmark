from pathlib import Path
import gzip, collections, math, time
trans=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(trans)[::-1]
# parse fasta
seqs=[]; name=None; chunks=[]
for line in Path('read.proportions.q1.fa').read_text().splitlines():
    if line.startswith('>'):
        if name: seqs.append((name,''.join(chunks)))
        name=line[1:].split()[0]; chunks=[]
    else: chunks.append(line.strip())
if name: seqs.append((name,''.join(chunks)))
L=36
# use 6 segments of 6? Any <=5 mism has one exact segment, includes true high errors, more candidates but small genome
seg_starts=[0,6,12,18,24,30]; seg_len=6
windows=[]
index=[collections.defaultdict(list) for _ in seg_starts]
for ci,(n,s) in enumerate(seqs):
    for ss in (s, rc(s)):
        for pos in range(len(ss)-L+1):
            w=ss[pos:pos+L]
            wid=len(windows)
            windows.append((ci,w))
            for j,st in enumerate(seg_starts):
                index[j][w[st:st+seg_len]].append(wid)
reads=[]
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h: break
        r=f.readline().strip(); f.readline(); f.readline(); reads.append(r)
features=[]
maxm=12
bestglobal=collections.Counter()
for r in reads:
    cands=set()
    for j,st in enumerate(seg_starts): cands.update(index[j].get(r[st:st+seg_len],()))
    nby=[[0]*(maxm+1) for _ in range(4)]
    best=99
    for wid in cands:
        ci,w=windows[wid]
        m=0
        # early cutoff maxm
        for a,b in zip(r,w):
            if a"'!=b:
                m+=1
                if m>maxm: break
        if m<=maxm:
            nby[ci][m]+=1
            if m<best: best=m
    bestglobal[best]+=1
    features.append(nby)
print('"'bestglobal', bestglobal)

def em_fixed(e, max_use=maxm):
    a=e/(3*(1-e))
    powa=[a**m for m in range(max_use+1)]
    pi=[0.25]*4
    for it in range(200):
        sums=[0.0]*4; ll=0.0
        for nby in features:
            lik=[]
            for ci in range(4):
                lik.append(sum(nby[ci][m]*powa[m] for m in range(max_use+1)))
            denom=sum(pi[i]*lik[i] for i in range(4))
            if denom==0:
                # uninformative rare; assign prior
                for i in range(4): sums[i]+=pi[i]
            else:
                ll+=math.log(denom)
                for i in range(4): sums[i]+=pi[i]*lik[i]/denom
        new=[x/len(features) for x in sums]
        if max(abs(new[i]-pi[i]) for i in range(4))<1e-12: break
        pi=new
    return pi,ll,it
for e in [0.003,0.005,0.0075,0.008,0.009,0.01,0.012,0.015,0.02,0.03]:
    pi,ll,it=em_fixed(e, max_use=8)
    print('e',e,'pi', ['%.5f'%x for x in pi], 'pct', ['%.2f'%(100*x) for x in pi], 'll', round(ll,2),'it',it)
# joint rough: grid optimize e by ll after EM
best=None
for k in range(1,101):
    e=k/10000 # .0001-.0100
    pi,ll,it=em_fixed(e, max_use=8)
    if best is None or ll>best[0]: best=(ll,e,pi)
print('grid best <=.01',best[1], ['%.5f'%x for x in best[2]], best[0])
for k in range(101,301):
    e=k/10000
    pi,ll,it=em_fixed(e, max_use=8)
    if ll>best[0]: best=(ll,e,pi)
print('grid best <=.03',best[1], ['%.5f'%x for x in best[2]], best[0])
