/bin/bash -lc "python -u - <<'PY'
import subprocess
from collections import Counter
seqs={}
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'): h=line[1:].split()[0]; seqs[h]=''
    else: seqs[h]+=line.upper()
chroms=list(seqs); patterns=Counter(); skipped=Counter()
p=subprocess.Popen(['bwa','samse','-n','20','read.proportions.q1.fa','read.proportions.q1.sai','read.proportions.q1.fq.gz'],stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True)
for line in p.stdout:
    if line.startswith('@'): continue
    f=line.rstrip().split('\\t'); flag=int(f[1]); cigar=f[5]
    if flag&4: skipped['unmapped']+=1; continue
    if cigar"'!='"'36M': skipped[cigar]+=1; continue
    pos=int(f[3])-1; read=f[9].upper()
    d=tuple(sum(a"'!=b for a,b in zip(read,seqs[c][pos:pos+36])) for c in chroms)
    patterns[d]+=1
p.wait(); n=sum(patterns.values())
print('"'used',n,'skipped',dict(skipped),flush=True)
for pat,num in patterns.most_common(20): print(pat,num,flush=True)
for e in [.002,.005,.008,.01,.02,.03,.05]:
    pi=[.25]*4; ratio=(e/3)/(1-e)
    for it in range(1000):
        sums=[0.0]*4
        for d,count in patterns.items():
            ws=[pi[k]*(ratio**d[k]) for k in range(4)]; z=sum(ws)
            for k in range(4): sums[k]+=count*ws[k]/z
        new=[x/n for x in sums]
        if max(abs(new[k]-pi[k]) for k in range(4))<1e-13: pi=new; break
        pi=new
    print('e',e,'pi',*[round(100*x,4) for x in pi],'iters',it,flush=True)
PY"
