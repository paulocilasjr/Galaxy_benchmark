from Bio import SeqIO
from collections import defaultdict,Counter
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
names=list(seqs); comp=str.maketrans('ACGT','TGCA'); L=36;k=5
rev={n:s.translate(comp)[::-1] for n,s in seqs.items()}
index={name:{'f':defaultdict(list),'r':defaultdict(list)} for name in names}

    for i in range(len(s)-L+1): index[name]['f'][s[i:i+k]].append(i)

    for i in range(len(rs)-L+1): index[name]['r'][rs[i:i+k]].append(i)
def min_ds(read):
    ds=[]
    for name in names:
        best=L+1
        for strand in ['f','r']:
            idx=index[name][strand]; sstr=seqs[name] if strand=='f' else rev[name]
            aligns=set()
            for start in range(0,L-k+1,k):

                if d<best: best=d
                if best==0: break

        ds.append(best)
    return ds

            read=t[a:a+L]
            ds=min_ds(read)
            md=min(ds); ns=[i for i,d in enumerate(ds) if d==md]

print('total',totals)
for c in range(4):
