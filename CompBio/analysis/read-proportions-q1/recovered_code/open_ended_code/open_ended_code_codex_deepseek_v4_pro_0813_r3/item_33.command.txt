/bin/bash -lc "python3 - <<'PY'
import pickle, math
with open('edit_dn.pkl','rb') as f: data=pickle.load(f)
all_d=data['d']; all_n=data['n']; N=len(all_d)
def run(weight):
    # lik = n * weight"'^d
    pi=[.25]*4; prev=-1e100
    for it in range(1000):
        new=[0.0]*4; ll=0.0
        for i in range(N):
            vals=[pi[j]*all_n[i][j]*(weight**all_d[i][j]) for j in range(4)]
            den=sum(vals); ll+=math.log(den)
            for j in range(4): new[j]+=vals[j]/den
        pi=[x/N for x in new]
        if abs(ll-prev)<1e-12: break
        prev=ll
    return pi,ll,it
for w in [0.001,0.003,0.005,0.008,0.01,0.015,0.02,0.03,0.05,0.08,0.1]:
    pi,ll,it=run(w)
    print(w,pi,[100*x for x in pi],ll,it)
PY'
