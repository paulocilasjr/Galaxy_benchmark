from Bio import SeqIO
from collections import Counter
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
names=list(seqs); comp=str.maketrans('ACGT','TGCA'); L=36
rev={n:s.translate(comp)[::-1] for n,s in seqs.items()}
# precompute all windows? function min d using brute for each generated read to all refs
def min_ds(read):
    ds=[]
    for n in names:
        b=99
        for t in (seqs[n],rev[n]):
            for a in range(len(t)-L+1):
                d=sum(t[a+j]"'!=read[j] for j in range(L))
                if d<b: b=d
                if b==0: break
            if b==0: break
        ds.append(b)
    return ds
probs=Counter(); totals=Counter()
for c,n in enumerate(names):
    for strand,t in [('"'f',seqs[n]),('r',rev[n])]:
        for a in range(len(t)-L+1):
            read=t[a:a+L]
            ds=min_ds(read)
            md=min(ds)
            ns=[i for i,d in enumerate(ds) if d==md]
            totals[c]+=1
            if len(ns)==1:
                probs[(c,ns[0])]+=1
print('total windows per chr',totals)
print('informative counts matrix source->assigned')
for c in range(4):
    print(c,[probs[(c,j)] for j in range(4)], 'informative total',sum(probs[(c,j)] for j in range(4)))
# observed unique counts from data
obs=[6282,995,1351,402]

# predict obs counts for pi = 50,20,20,10 and = EM
for pi in ([0.5,0.2,0.2,0.1],[0.504071,0.198444,0.196970,0.100515],[0.6957,0.1102,0.1496,0.0445]):
