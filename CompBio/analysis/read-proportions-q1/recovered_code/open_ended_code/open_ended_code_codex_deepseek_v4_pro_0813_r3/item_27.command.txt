/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
import gzip, edlib, time, pickle
from collections import Counter
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
names=list(seqs); comp=str.maketrans('ACGT','TGCA'); L=36
rev={n:s.translate(comp)[::-1] for n,s in seqs.items()}
reads=[]
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    for i in range(40000):
        h=f.readline(); seq=f.readline().strip().upper(); f.readline(); f.readline(); reads.append(seq)
# Test first 100, then all maybe
start=time.time(); edit_mins=[]
for i,r in enumerate(reads):
    ds=[]
    for n in names:
        b=99
        for t in (seqs[n], rev[n]):
            res=edlib.align(r,t,mode='HW',task='distance')
            d=res['editDistance']
            if d<b: b=d
        ds.append(b)
    edit_mins.append(ds)
    if i==0 or (i+1)%5000==0: print('done',i+1,'elapsed',time.time()-start)
print('total',time.time()-start)
with open('edit_mins.pkl','wb') as f: pickle.dump(edit_mins,f)
from collections import Counter
hist=Counter(min(ds) for ds in edit_mins)
print('edit min hist',hist)
unique=Counter()
for ds in edit_mins:
    md=min(ds); ns=[names[j] for j,d in enumerate(ds) if d==md]
    if len(ns)==1: unique[ns[0]]+=1
print('unique',unique)
PY"
