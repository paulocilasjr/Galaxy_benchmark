/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
from collections import defaultdict, Counter
seqs={r.id:str(r.seq).upper() for r in SeqIO.parse('read.proportions.q1.fa','fasta')}
comp=str.maketrans('ACGT','TGCA')
# forward index per chr count sets
for name,s in seqs.items():
    kmers=set(s[i:i+36] for i in range(len(s)-35))
    print(name,'seqlen',len(s),'unique forward',len(kmers))
allf=defaultdict(list)
for name,s in seqs.items():
    for k in set(s[i:i+36] for i in range(len(s)-35)):
        allf[k].append(name)
print('forward multi',sum(1 for v in allf.values() if len(set(v))>1))
# check rev
print('sample duplicated')
c=0
for k,v in allf.items():
    if len(set(v))>1:
        print(k,v); c+=1
        if c>=5: break
# rev index
allr=defaultdict(list)
for name,s in seqs.items():
    rs=s.translate(comp)[::-1]
    for k in set(rs[i:i+36] for i in range(len(rs)-35)):
        allr[k].append(name)
print('rev multi',sum(1 for v in allr.values() if len(set(v))>1))
# combined duplicate that affect reads maybe same sequence in f/r
PY"
