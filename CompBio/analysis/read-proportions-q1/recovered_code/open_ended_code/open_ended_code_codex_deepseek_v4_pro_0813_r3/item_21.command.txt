        idx=index[name][strand]; sstr=seqs[name] if strand=='f' else seqs[name].translate(comp)[::-1]
        aligns=set()

            d=sum(sstr[a+j]"'!=read[j] for j in range(L))

            if d<=maxd: cnt[d]+=1
