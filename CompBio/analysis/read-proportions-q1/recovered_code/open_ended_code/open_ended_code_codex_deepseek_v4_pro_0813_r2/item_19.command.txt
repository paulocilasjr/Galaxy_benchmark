/bin/bash -lc "python3 - <<'PY'
import numpy as np
cnt=np.load('cnt.npy')
N=cnt.shape[0]; K=37
dvals=np.arange(K)
pi=np.ones(4)/4; eps=0.01
for it in range(200):
    P=np.array([(1-eps)**(K-1-d)*(eps/3)**d for d in range(K)])
    L=cnt.astype(float)@P # N,4
    ll=np.log(L)+np.log(pi)
    mx=ll.max(axis=1,keepdims=True); w=np.exp(ll-mx); gamma=w/w.sum(axis=1,keepdims=True)
    newpi=gamma.mean(axis=0)
    # expected stats with component posterior
    # For each r,c, expected matches = sum_d cnt*(36-d)*P_d / L; mismatches similarly
    matches=((cnt.astype(float)*(K-1-dvals)[None,None,:])@P)/L  # N,4
    mism=((cnt.astype(float)*dvals[None,None,:])@P)/L
    A=(gamma*matches).sum()
    B=(gamma*mism).sum()
    neweps=B/(A+B)
    diff=max(np.max(abs(newpi-pi)),abs(neweps-eps))
    if it<20 or it%10==0:
        print(it,'pi',pi*100,'eps',eps,'newpi',newpi*100,'neweps',neweps,'diff',diff)
    pi,eps=newpi,neweps
    if diff<1e-10: break
print('FINAL',pi*100,'eps',eps,'round',[int(round(x*100/5)*5) for x in pi])
PY"
