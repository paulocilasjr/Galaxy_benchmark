/bin/bash -lc "python3 - <<'PY'
import numpy as np
cnt=np.load('cnt.npy')
mins=np.full((cnt.shape[0],4),99,dtype=int)
for d in range(37):
    m=cnt[:,:,d]>0
    mins[m]=np.minimum(mins[m],d)
best=mins.min(axis=1,keepdims=True)
assigned=(mins==best).astype(float)
assigned/=assigned.sum(axis=1,keepdims=True)
print('hard frac %', assigned.mean(axis=0)*100)
print('best min dist counts',dict(zip(*np.unique(best, return_counts=True))))
PY"
