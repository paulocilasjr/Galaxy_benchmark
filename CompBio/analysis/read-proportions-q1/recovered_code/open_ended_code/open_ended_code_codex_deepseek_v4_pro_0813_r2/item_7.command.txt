from collections import defaultdict
seqs={};name=None
for line in open('read.proportions.q1.fa'):
 line=line.strip()
 if line.startswith('>'): name=line[1:];seqs[name]=''
 else: seqs[name]+=line.upper()
idx=defaultdict(set)
for c,s in seqs.items():
 for i in range(len(s)-36+1): idx[s[i:i+36]].add(c)
dist=defaultdict(int)
for k,cs in idx.items():
 if len(cs)>1: dist[tuple(sorted(cs))]+=1
print('total unique kmers',len(idx))
print('multi',sum(v for k,v in dist.items()),dist)
PY"
