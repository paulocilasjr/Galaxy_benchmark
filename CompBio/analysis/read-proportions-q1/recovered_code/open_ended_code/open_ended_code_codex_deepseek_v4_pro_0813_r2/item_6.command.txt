/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
seqs={}; name=None
for line in open('read.proportions.q1.fa'):
    line=line.strip()
    if line.startswith('>'): name=line[1:]; seqs[name]=''
    else: seqs[name]+=line.upper()
comp=str.maketrans('ACGT','TGCA')
def rc(x): return x.translate(comp)[::-1]
idx=defaultdict(set)  # kmer -> set chroms
for c,s in seqs.items():
    for i in range(len(s)-36+1):
        idx[s[i:i+36]].add(c)
import gzip
counts={c:0 for c in seqs}; amb=0;un=0
with gzip.open('read.proportions.q1.fq.gz','rt') as f:
    while True:
        h=f.readline().strip()
        if not h: break
        s=f.readline().strip().upper(); f.readline(); f.readline()
        hits=idx.get(s, set()) | idx.get(rc(s), set())
        if len(hits)==1: counts[next(iter(hits))]+=1
        elif len(hits)>1: amb+=1
        else: un+=1
print(counts,'amb',amb,'un',un,'sum',sum(counts.values()))
PY"
