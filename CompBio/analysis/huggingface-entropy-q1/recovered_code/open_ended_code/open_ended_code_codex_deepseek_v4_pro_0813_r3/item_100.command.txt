/bin/bash -lc "cd /tmp && python - <<'PY'
import torch,json
W='[REPO_ROOT]/compbiobench/runs/.agent_runtime/exec-6e845b76af504ee4e726ec8c/workspace'
seq=json.load(open(f'{W}/seq_7792299_7793299.json'))['dna'].upper()
logits=torch.load(f'{W}/caduceus/logits_full_no_sep.pt', map_location='cpu')
L=len(seq)
ml=logits[torch.arange(L),torch.arange(L)]
def runs(ent,thr=0.5):
    best=[];cur=[]
    for i,e in enumerate(ent):
        if e<thr: cur.append(i)
        else:
            if len(cur)>len(best): best=cur
            cur=[]
    if len(cur)>len(best): best=cur
    return best
# ACGT renorm
ids=[7,8,9,10]
p=torch.softmax(ml[:,ids].float(),-1); e=-(p*torch.log2(p+1e-12)).sum(-1)
r=runs(e); print('ACGT',len(r),r[0] if r else None, seq[r[0]:r[0]+len(r)] if r else '')
# all 16 softmax
pall=torch.softmax(ml.float(),-1); eall=-(pall*torch.log2(pall+1e-12)).sum(-1)
r=runs(eall); print('all16',len(r),r[0] if r else None, seq[r[0]:r[0]+len(r)] if r else '', eall.min().item())
# ACGTN renorm
ids5=[7,8,9,10,11]
p5=torch.softmax(ml[:,ids5].float(),-1); e5=-(p5*torch.log2(p5+1e-12)).sum(-1)
r=runs(e5); print('ACGTN',len(r),r[0] if r else None, seq[r[0]:r[0]+len(r)] if r else '')
# no renorm over ACGT (softmax over all, sum acgt entropy conditional? maybe)
PY"
