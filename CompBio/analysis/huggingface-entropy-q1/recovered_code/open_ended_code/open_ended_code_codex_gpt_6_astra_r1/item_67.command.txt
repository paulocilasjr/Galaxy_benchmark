import json,re
from pathlib import Path
r=json.load(open('data/analysis.json'))
for name in ['sep','no_sep']:
 assert r[name]['completed']==1000, (name,r[name]['completed'])
 for vocab in ['all','acgt']:
  runs=r[name][vocab]['longest_runs']
  assert runs[0]['length']>runs[1]['length']
  assert runs[0]['substring']==r['sep']['all']['longest_runs'][0]['substring']
answer=r['sep']['all']['longest_runs'][0]['substring']
assert re.fullmatch('[ACGT]+',answer)
Path('data/answer.txt').write_text(answer+'\\n')
assert Path('data/answer.txt').read_text().splitlines()==[answer]
print('Final answer verified:',answer)
PY"
