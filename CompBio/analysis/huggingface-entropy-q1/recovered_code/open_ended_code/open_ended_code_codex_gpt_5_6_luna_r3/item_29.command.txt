/bin/bash -lc "python -u - <<'PY'
import torch,time
from transformers import AutoTokenizer,AutoModelForMaskedLM
seq=open('data/sequence.txt').read().strip()[:1000]+'A'*24
tok=AutoTokenizer.from_pretrained('model_files',trust_remote_code=True,local_files_only=True)
ids=tok(seq,return_tensors='pt',add_special_tokens=False).input_ids
print(ids.shape,flush=True)
model=AutoModelForMaskedLM.from_pretrained('model_files',trust_remote_code=True,local_files_only=True).eval().to('mps')
with torch.no_grad():
 t=time.time(); y=model(ids.to('mps')).logits; torch.mps.synchronize(); print(y.shape,time.time()-t,y[0,0,7:11].cpu(),flush=True)
PY"
