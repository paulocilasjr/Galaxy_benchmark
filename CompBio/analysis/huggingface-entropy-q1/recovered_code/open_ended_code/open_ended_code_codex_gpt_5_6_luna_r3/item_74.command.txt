/bin/bash -lc "python - <<'PY'
import json, numpy as np
z=np.load('data/entropy_results.npz')
for k in ['base_logits','vocab_logits','base_entropy','vocab_entropy']:
 print(k,z[k].shape, np.isfinite(z[k]).all())
seq=json.load(open('data/hg38_chr12_7792299_7793299.json'))['dna'].upper()

def runs(h):
 ok=h<0.5
 ans=[]; i=0
 while i<len(ok):
  if not ok[i]: i+=1; continue
  j=i
  while j<len(ok) and ok[j]: j+=1
  ans.append((j-i,i,j,seq[i:j],float(h[i:j].min()),float(h[i:j].max())))
  i=j
 return sorted(ans,reverse=True)
for name in ['base_entropy','vocab_entropy']:
 print('\\n',name)
 rr=runs(z[name])
 for x in rr[:20]: print(x)
print('\\nbase entropy min/max/mean',z['base_entropy'].min(),z['base_entropy'].max(),z['base_entropy'].mean())
# print sequence and entropy around each longest
rr=runs(z['base_entropy']); n=rr[0][0]; i=rr[0][1];
print('winner',n,i,i+n,seq[i:i+n])
print('entropies',np.array2string(z['base_entropy'][i:i+n],precision=6))
PY"
