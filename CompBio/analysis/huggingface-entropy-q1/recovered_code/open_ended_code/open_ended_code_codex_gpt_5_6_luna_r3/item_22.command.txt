/bin/bash -lc "python - <<'PY'
import time, torch
from transformers import AutoTokenizer, AutoModelForMaskedLM
seq='AAAATTGGTGAATCTCCATTCTGACCTATTAACACCCAGTGTGGGAGCTTTTGAGCATCAGTTTAAAGCAAAAGGTGGTAGCGCC'*12
seq=seq[:1000]
tok=AutoTokenizer.from_pretrained('model_files',trust_remote_code=True,local_files_only=True)
ids=tok(seq,return_tensors='pt',add_special_tokens=False).input_ids
for name,dev in [('cpu','cpu'),('mps','mps')]:
 print('device',name)
 model=AutoModelForMaskedLM.from_pretrained('model_files',trust_remote_code=True,local_files_only=True).eval().to(dev)
 x=ids.to(dev)
 with torch.no_grad():
  t=time.time(); y=model(x).logits; 
  if dev=='mps': torch.mps.synchronize()
  print(y.shape,'elapsed',time.time()-t,'sample',y[0,0,7:11].detach().cpu())
PY"
