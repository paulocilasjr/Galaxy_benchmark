/bin/bash -lc "python - <<'PY'
import json,hashlib,math
from pathlib import Path
src=json.loads(Path('artifacts/ucsc_sequence.json').read_text())
seq=src['dna'].upper()
r=json.loads(Path('results/entropy_0_1000.json').read_text())
assert len(seq)==1000 and len(r)==1000
assert [x['position'] for x in r]==list(range(1000))
old=json.loads(Path('results/entropy_0_20.json').read_text())
for a,b in zip(old,r[:20]):
    assert a==b
for key in ('base_entropy','full_entropy'):
    runs=[]; start=None
    for i,x in enumerate(r+[{}]):
        low=i<1000 and x[key] < 0.5
        if low and start is None: start=i
        if not low and start is not None:
            runs.append((i-start,start,i,seq[start:i]))
            start=None
    runs.sort(reverse=True)
    print(key,'run_count',len(runs),'top5')
    for run in runs[:5]: print(run)
    length,start,end,answer=runs[0]
    print('boundary',start-1,r[start-1][key],start,r[start][key],end-1,r[end-1][key],end,r[end][key])
    print('answer_sha256',hashlib.sha256(answer.encode()).hexdigest())
print('max_full_minus_base',max(abs(x['full_entropy']-x['base_entropy']) for x in r))
print('sequence_slice_194_221',seq[194:221],len(seq[194:221]))
print('genomic_interval',src['chrom'],src['start']+194,src['start']+221)
PY"
