import json,numpy as np
x=json.load(open('entropy_results.json')); seq=x['sequence']
for key in ['entropy_base4','entropy_vocab16']:

  j=i

  runs.append((j-i,i,j-1))
  i=j
 runs.sort(reverse=True)
 print('\\n'+key+' min '+str(e.min())+' max '+str(e.max())+' n<.5 '+str(ok.sum()),flush=True)
 for n,a,b in runs[:20]:

 print('five lowest',sorted(enumerate(e),key=lambda z:z[1])[:10],flush=True)
PY"
