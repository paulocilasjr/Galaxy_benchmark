/bin/bash -lc "python - <<'PY'
import time,torch
from pure_caduceus import load_model
m=load_model().eval().to('mps'); ids=torch.tensor([[7,8,9,10]*250]).to('mps')
t=time.time()
with torch.no_grad(): y=m(ids)
torch.mps.synchronize(); print('elapsed',time.time()-t,'shape',tuple(y.shape),'finite',bool(torch.isfinite(y).all()),'range',y.min().item(),y.max().item(),flush=True)
PY"
