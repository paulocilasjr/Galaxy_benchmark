/bin/bash -lc ". .venv/bin/activate && python3 - <<'PY'
import json, statistics
r=json.load(open('outputs/caduceus_entropy_results.json'))
seq=r['sequence']
for key,longkey in [('base_entropy','base_longest'),('vocab_entropy','vocab_longest')]:
    vals=r[key]
    run=r[longkey]
    print(longkey, run)
    print('min', min(vals), 'mean', statistics.mean(vals), 'count<0.5', sum(v<0.5 for v in vals))
    s,e=run['start0'],run['end0']
    print('boundary', vals[max(0,s-2):min(len(vals),e+2)])
    print('substring_check', seq[s:e])
PY"
