/bin/bash -lc "PYTHONPATH=./packages:. python - <<'PY'
import torch, json, urllib.request
seq=json.loads(urllib.request.urlopen('https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=7792299;end=7793299').read().decode())['dna'].upper()
for name,path in [('base','entropies.pt'),('full','full_vocab_entropies.pt')]:
 e=torch.load(path)
 flags=(e<0.5).tolist()
 runs=[]; st=None
 for i,f in enumerate(flags+[False]):
  if f and st is None: st=i
  if (not f) and st is not None:
   runs.append((st,i-st,seq[st:i])); st=None
 print(name, sorted(runs,key=lambda x:x[1], reverse=True)[:10])
 print('minmax', float(e.min()), float(e.max()))
PY"
