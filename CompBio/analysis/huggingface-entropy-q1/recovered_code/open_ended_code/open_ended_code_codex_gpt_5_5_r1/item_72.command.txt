/bin/bash -lc ". venv/bin/activate && python - <<'PY'
import json
res=json.load(open('data/entropy_results.json'))
seq=res['sequence']
ent=res['base_entropy_acgt']
runs=[]
s=None
for i,x in enumerate(ent+[1.0]):
    if x<0.5 and s is None: s=i
    if x>=0.5 and s is not None:
        runs.append((s,i,i-s,seq[s:i],min(ent[s:i]),max(ent[s:i])))
        s=None
print('num_runs', len(runs))
for r in sorted(runs, key=lambda x:x[2], reverse=True)[:10]: print(r)
print('flanks', seq[185:230])
PY"
