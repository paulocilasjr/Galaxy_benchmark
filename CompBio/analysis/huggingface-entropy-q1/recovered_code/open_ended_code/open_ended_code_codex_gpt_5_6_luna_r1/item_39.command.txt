/bin/bash -lc "PYTHONPATH='deps/mamba_ssm-2.2.2:.' python - <<'PY'
import json, torch
import torch.nn.functional as F
from model.configuration_caduceus import CaduceusConfig
from model.modeling_caduceus import CaduceusForMaskedLM
s=json.load(open('interval_ucsc.json'))['dna'].upper(); ids={'A':7,'C':8,'G':9,'T':10}
x=torch.tensor([[ids[c] for c in s]],dtype=torch.long); x[0,0]=3
config=CaduceusConfig.from_pretrained('model'); config.fused_add_norm=False
m=CaduceusForMaskedLM.from_pretrained('model',config=config,local_files_only=True).eval()
with torch.no_grad(): logits=m(x,return_dict=True).logits[0,0]
print('len',len(s),'logits',logits.tolist())
for inds,name in [([7,8,9,10],'bases'),(list(range(16)),'all'),([7,8,9,10,11],'basesN')]:
 p=F.softmax(logits[inds],dim=-1); h=-(p*torch.log2(p)).sum()
 print(name,'p',p.tolist(),'H',float(h),'arg',inds[int(p.argmax())])
PY"
