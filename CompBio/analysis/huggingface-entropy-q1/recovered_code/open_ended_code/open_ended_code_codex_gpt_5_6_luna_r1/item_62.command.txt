import torch.nn.functional as F
from model.configuration_caduceus import CaduceusConfig
from model.modeling_caduceus import CaduceusForMaskedLM

 bh=-(bp*torch.log2(bp)).sum(); rh=-(rp*torch.log2(rp)).sum()
 print(pos,'max_logit_diff',float(d.max()),'mean',float(d.mean()),'Hfast',float(bh),'Href',float(rh),'pred', 'ACGT'[int(bp.argmax())])
