import os, sys, time
sys.path.insert(0, os.getcwd())
import torch
from transformers import AutoTokenizer, AutoModelForMaskedLM
seq=open('sequence.txt').read().strip()
tok=AutoTokenizer.from_pretrained('model_files', trust_remote_code=True)
model=AutoModelForMaskedLM.from_pretrained('model_files', trust_remote_code=True); model.eval().to('mps')
base_ids=tok(''.join(seq), add_special_tokens=False).input_ids
for B in [32,64]:
    try:
        masked=torch.tensor([base_ids for _ in range(B)], device='mps')
        for b in range(B): masked[b,b]=tok.mask_token_id
        t0=time.time()
        with torch.no_grad():
            out=model(masked)
        dt=time.time()-t0
        print('B',B,'time',dt,'per seq',dt/B, flush=True)
        del out, masked
    except Exception as e:
        print('ERR B',B,repr(e), flush=True)
        break
