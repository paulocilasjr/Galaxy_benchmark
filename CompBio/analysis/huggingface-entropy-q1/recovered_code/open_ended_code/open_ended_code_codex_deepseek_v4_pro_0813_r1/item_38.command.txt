/bin/bash -lc "python - <<'PY'
import requests, pathlib
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=7792299;end=7793299'
d=requests.get(url,timeout=30).json()
seq=d['dna']
print('length',len(seq))
pathlib.Path('sequence.fa').write_text('>chr12:7792299-7793299\\n'+seq+'\\n')
pathlib.Path('sequence.txt').write_text(seq+'\\n')
print(seq[:100]); print(seq[-100:])
print('N count', seq.count('N'), 'lower count', sum(c.islower() for c in seq))
PY"
