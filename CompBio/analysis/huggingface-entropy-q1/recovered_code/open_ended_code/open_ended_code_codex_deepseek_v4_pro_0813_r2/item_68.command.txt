import sys, torch, time, pathlib
sys.path.insert(0,'.')
from safetensors.torch import load_file
from cadmodel.configuration_caduceus import CaduceusConfig
from cadmodel.modeling_caduceus import CaduceusForMaskedLM

B=3; batch=[]
for i in range(B): x=ids[0].clone(); x[i]=3; batch.append(x)
inp=torch.stack(batch)
# CPU
model.to('cpu'); inp_cpu=inp
with torch.no_grad(): out_cpu=model(input_ids=inp_cpu)
lc=out_cpu.logits[:, :, :].cpu()
# MPS
model.to('mps'); inp_m=inp.to('mps')
with torch.no_grad(): out_m=model(input_ids=inp_m)
lm=out_m.logits.cpu()
for i in range(B):
 diff=(lc[i,i]-lm[i,i]).abs().max().item(); print(i,'masked pos max diff',diff,'cpu top',lc[i,i].topk(5).values.tolist(),'mps top',lm[i,i].topk(5).values.tolist())
PY"
