import requests, tarfile, io
url='https://files.pythonhosted.org/packages/source/m/mamba-ssm/mamba_ssm-2.2.2.tar.gz'

for n in tf.getnames():
 if n.endswith('mamba_simple.py') or n.endswith('block.py'):
  print('---',n)
  data=tf.extractfile(n).read().decode()
  print(data[:16000])
  print('... len', len(data))
  open(n.split('/')[-1],'w').write(data)
PY"
