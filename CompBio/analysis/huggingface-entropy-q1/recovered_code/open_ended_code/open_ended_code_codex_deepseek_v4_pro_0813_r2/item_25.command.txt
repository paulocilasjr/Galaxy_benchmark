/bin/bash -lc "python - <<'PY'
import requests
url='https://huggingface.co/kuleshov-group/caduceus-ph_seqlen-131k_d_model-256_n_layer-16/resolve/main/model.safetensors?download=true'
with requests.get(url, stream=True, timeout=120) as r:
 print(r.status_code, r.headers.get('content-length'))
 r.raise_for_status()
 with open('model.safetensors','wb') as f:
  for chunk in r.iter_content(1<<20): f.write(chunk)
print('size', __import__('os').path.getsize('model.safetensors'))
PY"
