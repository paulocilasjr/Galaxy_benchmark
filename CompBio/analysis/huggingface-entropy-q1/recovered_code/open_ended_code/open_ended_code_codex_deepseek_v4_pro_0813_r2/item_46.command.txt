/bin/bash -lc "python - <<'PY'
import sys, torch, time, pathlib
sys.path.insert(0,'.')
from safetensors.torch import load_file
from cadmodel.configuration_caduceus import CaduceusConfig
from cadmodel.modeling_caduceus import CaduceusForMaskedLM
seq=pathlib.Path('sequence.fa').read_text().split('\\n')[1].strip()
mapping={'A':7,'C':8,'G':9,'T':10}
ids=torch.tensor([[mapping[c] for c in seq]],dtype=torch.long)
model=CaduceusForMaskedLM(CaduceusConfig.from_pretrained('./cadmodel'))
model.load_state_dict(load_file('model.safetensors'), strict=False); model.tie_weights(); model.eval(); model.to('mps'); ids=ids.to('mps')
for B in [25,50,100]:
 ids_b=ids.repeat(B,1)
 print('batch',B, flush=True)
 try:
  t=time.time()
  with torch.no_grad(): out=model(input_ids=ids_b)
  print(' time',time.time()-t, flush=True)
  del out, ids_b
 except Exception as e:
  print(' error',type(e),e)
  break
PY"
