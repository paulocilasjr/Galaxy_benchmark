from transformers import AutoModelForMaskedLM,AutoTokenizer

for name,mm,hh in [('fwd',layer.mixer.mamba_fwd,h),('rev',layer.mixer.mamba_rev,h.flip(1))]:
 xz=torch.nn.functional.linear(hh,mm.in_proj.weight).transpose(1,2); xx=xz.chunk(2,dim=1)[0]; xx=torch.nn.functional.silu(mm.conv1d(xx)[...,:hh.shape[1]])
 q=torch.nn.functional.linear(xx.transpose(1,2),mm.x_proj.weight); dt=q[...,:mm.dt_rank]; dt=torch.nn.functional.softplus(torch.nn.functional.linear(dt,mm.dt_proj.weight,mm.dt_proj.bias))
 print(name, float(dt.min().cpu()),float(dt.max().cpu()),float(dt.mean().cpu()))
