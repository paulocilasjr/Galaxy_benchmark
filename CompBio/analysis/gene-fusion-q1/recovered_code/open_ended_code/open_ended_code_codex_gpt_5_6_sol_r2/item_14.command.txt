/bin/bash -lc "awk '/"'^>/{id=substr($1,2); gene=""; for(i=2;i<=NF;i++) if($i ~ /''^gene_symbol:/){gene=$i; sub(/''^gene_symbol:/,"",gene)} if(gene!="") print id"'"\\t\"gene}' Homo_sapiens.GRCh38.cdna.all.fa > tx2gene.tsv
samtools view -f 64 -F 2308 cdna_align.bam | awk 'BEGIN{FS=OFS=\"\\t\"} NR==FNR{g["'$1]=$2;next} ($3 in g){print $1,g[$3],$3,$5,$2,$4,$6}'"' tx2gene.tsv - | LC_ALL=C sort -k1,1 > r1_primary.tsv
samtools view -f 128 -F 2308 cdna_align.bam | awk 'BEGIN{FS=OFS=\"\\t\"} NR==FNR{g["'$1]=$2;next} ($3 in g){print $1,g[$3],$3,$5,$2,$4,$6}'"' tx2gene.tsv - | LC_ALL=C sort -k1,1 > r2_primary.tsv
join -t "'$'"'\\t' -1 1 -2 1 r1_primary.tsv r2_primary.tsv > paired_primary.tsv
awk -F'\\t' '"'$2!=$8 {a=$2;b=$8;if(a>b){t=a;a=b;b=t} print a"'"\\t\"b}' paired_primary.tsv | sort | uniq -c | sort -nr | head -30
wc -l tx2gene.tsv r1_primary.tsv r2_primary.tsv paired_primary.tsv"
