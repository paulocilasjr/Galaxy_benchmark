/bin/bash -c "python3 - <<'PY'
import pysam, json
sym=json.load(open('sym.json'))
def g(ref): return sym.get(ref, ref)
infile=pysam.AlignmentFile('aln.sam','r')
for rec in infile:
    if rec.query_name in ('READ_42026','READ_55182','READ_89056'):
        rg=g(rec.reference_name)
        print('\\n', rec.query_name, 'flag', rec.flag, 'ref', rec.reference_name, rg, 'pos', rec.reference_start, 'mapq', rec.mapping_quality, 'cigar', rec.cigarstring, 'rnext', rec.next_reference_name, 'pnext', rec.next_reference_start, 'tlen', rec.template_length)
        print('seq', rec.query_sequence)
        print('SA', rec.get_tag('SA') if rec.has_tag('SA') else '')
PY"
