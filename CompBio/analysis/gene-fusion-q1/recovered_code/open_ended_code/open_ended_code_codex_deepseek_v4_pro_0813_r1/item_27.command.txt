/bin/bash -c "python3 - <<'PY'
import pysam, json, collections, re
sym=json.load(open('sym.json'))
def g(ref):
    return sym.get(ref, ref)
infile=pysam.AlignmentFile('aln.sam','r')
sa_pairs=collections.Counter(); mate_pairs=collections.Counter(); examples_sa={}; examples_mate={}
for rec in infile:
    if rec.is_unmapped: continue
    if rec.is_secondary or rec.is_supplementary: continue
    rname=rec.reference_name
    rg=g(rname)
    # SA
    if rec.has_tag('SA'):
        for entry in rec.get_tag('SA').split(';'):
            if not entry: continue
            parts=entry.split(',')
            if len(parts)<6: continue
            ref=parts[0]
            rg2=g(ref)
            if rg2 "'!= rg:
                pair=tuple(sorted([rg,rg2]))
                sa_pairs[pair]+=1
                examples_sa.setdefault(pair,(rec.query_name, rname, rg, ref, rg2, rec.get_tag('"'SA')))
    # mate
    nref=rec.next_reference_name
    if nref and nref "'!= '"'=' and nref "'!= rname:
        ng=g(nref)
        if ng != rg:
            pair=tuple(sorted([rg,ng]))
            mate_pairs[pair]+=1
            examples_mate.setdefault(pair,(rec.query_name, rname, rg, nref, ng, rec.flag, rec.tlen))
print('"'TOP SA pairs:')
for p,c in sa_pairs.most_common(30):
    print(p,c, examples_sa[p])
print('\\nTOP mate pairs:')
for p,c in mate_pairs.most_common(30):
    print(p,c, examples_mate[p])
PY"
