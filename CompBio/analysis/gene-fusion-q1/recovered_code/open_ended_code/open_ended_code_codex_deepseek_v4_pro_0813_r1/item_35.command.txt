/bin/bash -c "python3 - <<'PY'
import gzip, re, json
meta={}
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            sym=None; biotype=None; chrom=None
            m=re.search(r'gene_symbol:(["'^'"\\s]+)', line)
            if m: sym=m.group(1)
            m=re.search(r'gene_biotype:(["'^'"\\s]+)', line)
            if m: biotype=m.group(1)
            # fields chromosome:GRCh38:... or scaffold...
            m=re.search(r'chromosome:GRCh38:(["'^:]+):'"', line)
            if m: chrom=m.group(1)
            elif 'scaffold:GRCh38:' in line:
                chrom='scaffold'
            meta[tid]=(sym,biotype,chrom)
json.dump(meta, open('meta.json','w'))
print(len(meta))
print(list(meta.items())[:3])
PY"
