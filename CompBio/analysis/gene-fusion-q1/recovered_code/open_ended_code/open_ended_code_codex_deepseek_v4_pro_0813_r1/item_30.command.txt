/bin/bash -c "python3 - <<'PY'
import pysam, json, collections, re
sym=json.load(open('sym.json'))
def g(ref): return sym.get(ref, ref)
infile=pysam.AlignmentFile('aln.sam','r')
sa=collections.Counter(); mate=collections.Counter()
for rec in infile:
    if rec.is_unmapped or rec.is_secondary or rec.is_supplementary: continue
    rg=g(rec.reference_name)
    if rec.has_tag('SA'):
        seen=set()
        for entry in rec.get_tag('SA').split(';'):
            if not entry: continue
            p=entry.split(',')
            rg2=g(p[0])
            if rg2"'!=rg:
                pair=tuple(sorted([rg,rg2])); seen.add(pair)
        for p in seen: sa[p]+=1
    nr=rec.next_reference_name
    if nr and nr!='"'=' and nr"'!=rec.reference_name:
        ng=g(nr)
        if ng!=rg:
            pair=tuple(sorted([rg,ng])); mate[pair]+=1
print('"'Pairs with SA and mate:')
for p,c in sa.most_common(100):
    mc=mate.get(p,0)
    if c>=2 or mc>=10:
        print(f'{p[0]}-{p[1]}\\tSA={c}\\tmate={mc}')
print('\\nTop mate only:')
for p,c in mate.most_common(50):
    print(f'{p[0]}-{p[1]}\\tmate={c}\\tSA={sa.get(p,0)}')
PY"
