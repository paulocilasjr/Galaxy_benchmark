/bin/bash -lc "python3 - <<'PY'
import re,subprocess,collections
mp={}
with open('MANE.GRCh38.v1.4.refseq_rna.fna') as f:
 for l in f:
  if l.startswith('>'):
   acc=l[1:].split()[0]
   m=re.search(r'\\((["'^()]*)'"\\), transcript',l)
   mp[acc]=m.group(1) if m else acc
cnt=collections.Counter(); examples={}
p=subprocess.Popen(['samtools','view','mane.name.bam'],stdout=subprocess.PIPE,text=True)
group=[]; last=None
def process(g):
 a=[]
 for l in g:
  z=l.rstrip().split('\\t'); flag=int(z[1]);
  if flag&0x4 or flag&0x100 or flag&0x800: continue
  a.append((1 if flag&0x40 else 2,z[2],int(z[4]),z[5],z[9]))
 if len(a)==2 and a[0][1]"'!=a[1][1]:
  pair=(mp.get(a[0][1],a[0][1]),mp.get(a[1][1],a[1][1]))
  # unordered genes for initial evidence
  pair=tuple(sorted(pair))
  cnt[pair]+=1
  if pair not in examples: examples[pair]=a
for l in p.stdout:
 q=l.split('"'\\t',1)[0]
 if last is not None and q"'!=last: process(group); group=[]
 group.append(l);last=q
process(group)
for k,n in cnt.most_common(30): print(n,k,examples[k])
PY'
