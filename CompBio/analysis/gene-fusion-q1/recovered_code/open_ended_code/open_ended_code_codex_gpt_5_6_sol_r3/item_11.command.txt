import re,subprocess,collections
mp={}
with open('MANE.GRCh38.v1.4.refseq_rna.fna') as f:

  if l.startswith('>'):
   acc=l[1:].split()[0]; m=re.search(r'\\((["'^()]*)'"\\),',l); mp[acc]=m.group(1) if m else acc
p=subprocess.Popen(['samtools','view','mane.name.bam'],stdout=subprocess.PIPE,text=True)
d=collections.defaultdict(list)
for l in p.stdout:
 z=l.rstrip().split('\\t'); flag=int(z[1])
 if flag&0x4 or flag&0x100: continue
 d[(z[0],1 if flag&0x40 else 2)].append((flag,mp[z[2]],z[2],int(z[3]),int(z[4]),z[5],z[9]))
c=collections.Counter(); ex={}
for key,a in d.items():
 genes=tuple(sorted(set(x[1] for x in a)))
 if len(genes)>1:
  c[genes]+=1;ex.setdefault(genes,(key,a))
for genes,n in c.most_common(30):
 print('\\n',n,genes,ex[genes])
