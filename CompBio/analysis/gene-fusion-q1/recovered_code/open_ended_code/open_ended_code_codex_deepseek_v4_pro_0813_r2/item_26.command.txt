/bin/bash -lc "python3 - <<'PY'
import pysam, re, gzip
sym={}
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            m=re.search(r'gene_symbol:(\\S+)',line)
            if m: sym[tid]=m.group(1)
sam=pysam.AlignmentFile('raw_transcriptome.sam','r')
for a in sam.fetch(until_eof=True):
    if a.query_name=='READ_133' and not a.is_secondary:
        refs=[a.reference_name]
        if a.has_tag('SA'):
            for item in a.get_tag('SA').split(';'):
                if item: refs.append(item.split(',')[0])
        genes=[sym.get(r,r) for r in refs]
        print(a.query_name,'flag',a.flag,'ref',a.reference_name,'gene',sym.get(a.reference_name),'start',a.reference_start,'cigar',a.cigarstring,'strand','-' if a.is_reverse else '+','seq',a.query_sequence)
        print('all refs/genes', list(zip(refs,genes)))
        if a.has_tag('SA'): print('SA',a.get_tag('SA'))
PY"
