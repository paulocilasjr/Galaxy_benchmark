/bin/bash -lc "python3 - <<'PY'
import pysam, re, gzip, collections
sym={}
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as fh:
    for line in fh:
        if line.startswith('>'):
            tid=line[1:].split()[0]
            m=re.search(r'gene_symbol:(\\S+)',line)
            if m: sym[tid]=m.group(1)
sam=pysam.AlignmentFile('raw_transcriptome.sam','r')
count=0
for a in sam.fetch(until_eof=True):
    if a.is_unmapped: continue
    refs=[a.reference_name]
    genes={sym.get(a.reference_name, a.reference_name)}
    if a.has_tag('SA'):
        for item in a.get_tag('SA').split(';'):
            if not item: continue
            parts=item.split(',')
            refs.append(parts[0])
            genes.add(sym.get(parts[0],parts[0]))
    if 'GAPDH' in genes and 'SOX2' in genes:
        count+=1
        print('QNAME',a.query_name,'flag',a.flag,'primary_ref',a.reference_name,'primary_start',a.reference_start,'cigar',a.cigarstring,'seq',a.query_sequence[:40])
        if a.has_tag('SA'): print(' SA',a.get_tag('SA'))
        print(' refs genes', genes)
        if count>=15: break
print('shown',count)
PY"
