/bin/bash -lc "cat > analyze.py <<'PY'
import gzip, numpy as np, time, os
sizes={l.split()[0]:int(l.split()[1]) for l in open('hg38.chrom.sizes') if l.split()[0] in ['chr'+str(i) for i in range(1,23)]+['chrX','chrY']}
for name in ['signal','control']:
 out=name+'.bins.npz'
 if os.path.exists(out): continue
 t=time.time(); arr={c:np.zeros((s+199)//200,dtype=np.uint32) for c,s in sizes.items()}; n=0; ignored=0; lengths={}
 with gzip.open('histone.chip.q1.'+name+'.tagalign.gz','rt') as f:
  for line in f:
   v=line.split('\\t'); c=v[0]
   if c not in arr: ignored+=1; continue
   a=int(v[1]); b=int(v[2]); pos=(a+b)//2
   arr[c][pos//200]+=1; n+=1
   if n%5000000==0: print(name,n,round(time.time()-t,1),flush=True)
 np.savez_compressed(out,**arr)
 print(name,'total',n,'ignored',ignored,'seconds',time.time()-t,flush=True)
PY
python -u analyze.py"
