/bin/bash -lc "python3 - <<'PY'
import gzip
with gzip.open('rmsk.txt.gz','rt') as f, open('rmsk_canonical.bed','w') as out:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<13: continue
        chrom=p[5]; start=int(p[6]); end=int(p[7]); strand=p[9]; name=p[10]; cls=p[11]; fam=p[12]
        if not chrom.startswith('chr') or '_' in chrom: continue
        out.write(f'{chrom}\\t{start}\\t{end}\\t{name}\\t0\\t{strand}\\t{cls}\\t{fam}\\n')
PY
wc -l rmsk_canonical.bed
head -2 rmsk_canonical.bed
bedtools sort -g hg38.chrom.sizes -i rmsk_canonical.bed > rmsk_sorted.bed
head -2 rmsk_sorted.bed
wc -l rmsk_sorted.bed"
