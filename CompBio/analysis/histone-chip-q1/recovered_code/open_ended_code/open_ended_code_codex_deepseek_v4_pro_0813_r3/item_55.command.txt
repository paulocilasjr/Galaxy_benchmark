/bin/bash -lc "python3 - <<'PY'
import requests, json
url='https://www.encodeproject.org/search/'
for params in [
 {'type':'File','file_format':'bed','file_format_type':'tagAlign','limit':'5','format':'json'},
 {'type':'File','file_format_type':'tagAlign','limit':'5','format':'json'},
 {'type':'File','output_type':'alignments','file_format':'bed','limit':'5','format':'json'},
 {'type':'File','file_format':'tagAlign','limit':'5','format':'json'},
]:
 r=requests.get(url,params=params,headers={'Accept':'application/json'})
 print(params, r.status_code, len(r.text))
 try:
  j=r.json(); print('total',j.get('total'),'graph',len(j.get('@graph',[])))
  for x in j.get('@graph',[])[:2]: print(x.get('accession'), x.get('file_format'), x.get('file_format_type'), x.get('title'))
 except Exception as e: print(e)
PY"
