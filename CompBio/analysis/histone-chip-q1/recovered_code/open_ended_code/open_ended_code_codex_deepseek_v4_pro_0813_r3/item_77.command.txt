/bin/bash -lc "python3 - <<'PY'
import gzip
# hg19 refGene coding gene bodies and promoters
rows=[]
with gzip.open('hg19.refGene.txt.gz','rt') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  chrom=p[2]; strand=p[3]; txStart=int(p[4]); txEnd=int(p[5]); cdsStart=int(p[6]); cdsEnd=int(p[7]); name=p[12]
  if cdsStart>=cdsEnd: continue
  if '_' in chrom or not chrom.startswith('chr'): continue
  tss=txStart if strand=='+' else txEnd
  if txEnd-txStart<100: continue
  rows.append((chrom,txStart,txEnd,name,strand,tss))
with open('hg19_gene_bodies.bed','w') as f:
 for r in sorted(rows,key=lambda x:(x[0],x[1])):
  f.write('\\t'.join(map(str,[r[0],r[1],r[2],r[3],'.',r[4]]))+'\\n')
with open('hg19_promoters_2kb.bed','w') as f:
 ints=[]
 for r in rows:
  s=max(0,r[5]-2000); e=r[5]+2000
  ints.append((r[0],s,e,r[3],'.',r[4]))
 for r in sorted(ints,key=lambda x:(x[0],x[1])):
  f.write('\\t'.join(map(str,r))+'\\n')
print('hg19 coding transcripts',len(rows))
# cpg bed
with gzip.open('hg19.cpgIslandExt.txt.gz','rt') as f, open('hg19_cpgislands.bed','w') as out:
 for line in f:
  p=line.rstrip('\\n').split('\\t'); chrom=p[1]; start=int(p[2]); end=int(p[3])
  if '_' in chrom: continue
  out.write(f'{chrom}\\t{start}\\t{end}\\n')
PY
wc -l hg19_gene_bodies.bed hg19_promoters_2kb.bed hg19_cpgislands.bed
for f in cand_H3K27me3.tagalign.gz cand_H3K9me3.tagalign.gz cand_H3K4me3.tagalign.gz; do echo -n \""'$f "; gzip -cd "$f" | awk '"'"'$1 ~ /''^chr([0-9]+|X|Y|M)$/ {n++} END{print n}'"'; done"
