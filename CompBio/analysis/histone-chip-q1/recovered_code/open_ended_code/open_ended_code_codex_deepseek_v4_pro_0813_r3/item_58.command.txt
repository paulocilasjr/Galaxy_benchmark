/bin/bash -lc "python3 - <<'PY'
import requests, json, gzip
url='https://www.encodeproject.org/search/'
params={'type':'File','file_format':'tagAlign','limit':'all','format':'json',
        'field':['accession','file_size','target.label','dataset','assembly','href','read_length','biological_replicates','technical_replicates','biosample_ontology.term_name','simple_biosample_summary']}
r=requests.get(url,params=params,headers={'Accept':'application/json'},timeout=120)
print('status',r.status_code,'len',len(r.text))
j=r.json(); print('total',j.get('total'))
rows=[]
for x in j.get('@graph',[]):
    targets=[t.get('label') for t in x.get('target',[])] if isinstance(x.get('target'),list) else [x.get('target',{}).get('label') if isinstance(x.get('target'),dict) else x.get('target')]
    rows.append((x.get('accession'),x.get('file_size'),targets,x.get('dataset'),x.get('assembly'),x.get('href'),x.get('read_length'),x.get('simple_biosample_summary')))
with open('enc_tagalign_meta.tsv','w') as f:
    f.write('accession\\tfile_size\\ttargets\\tdataset\\tassembly\\thref\\tread_length\\tbiosample\\n')
    for row in rows:
        f.write('\\t'.join(['' if v is None else str(v) for v in row])+'\\n')
print('saved',len(rows))
# find near our sizes
for target_size,label in [(321498264,'signal'),(199777948,'control')]:
    print('\\nNear',label,target_size)
    matches=sorted(rows,key=lambda r: abs((r[1] or 0)-target_size))[:20]
    for r in matches:
        print(r[0], r[1], r[2], r[3], r[4], r[6], r[7])
PY"
