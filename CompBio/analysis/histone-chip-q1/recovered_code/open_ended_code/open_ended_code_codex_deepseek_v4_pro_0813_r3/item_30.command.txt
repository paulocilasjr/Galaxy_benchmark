/bin/bash -lc "python3 - <<'PY'
canon={}
with open('hg38.chrom.sizes') as f:
 for line in f:
  c,s=line.split()
  if c.startswith('chr') and '_' not in c:
   canon[c]=int(s)
# region lengths sum by awk from bed files
import subprocess
lengths={}
for name in ['gene_bodies.bed','promoters_2kb.bed','tss_500.bed']:
 s=subprocess.check_output(f\"awk '{{s+="'$3-$2}} END{{print s}}'"' {name}\",shell=True,text=True).strip()
 lengths[name]=int(s)
print('canon genome',sum(canon.values()),'lengths',lengths)
# counts
counts={
 'gene_signal':16816791,'gene_control':12847429,
 'prom_signal':1139917,'prom_control':1256107,
}
total_s=44795356; total_c=27088288
genome_len=sum(canon.values())
for region in ['gene','prom']:
 ls=lengths['gene_bodies.bed' if region=='gene' else 'promoters_2kb.bed']
 cs=counts[f'{region}_signal']; cc=counts[f'{region}_control']
 print(region,'len',ls,'signal density rpm/Mb',cs/(total_s/1e6)/(ls/1e6),'control',cc/(total_c/1e6)/(ls/1e6),'fold', (cs/(ls/1e6))/(cc/(ls/1e6))* (total_c/total_s))
# background density
print('genome bg signal rpm/Mb', total_s/(total_s/1e6)/(genome_len/1e6))
print('genome bg control', total_c/(total_c/1e6)/(genome_len/1e6))
# intergenic approximate
ig_len=genome_len-lengths['gene_bodies.bed']
igs=total_s-counts['gene_signal']; igc=total_c-counts['gene_control']
print('intergenic len',ig_len,'signal density',igs/(total_s/1e6)/(ig_len/1e6),'control',igc/(total_c/1e6)/(ig_len/1e6),'fold signal/control norm', (igs/(ig_len/1e6))/(igc/(ig_len/1e6))*(total_c/total_s))
PY"
