/bin/bash -lc "python3 - <<'PY'
import gzip
with gzip.open('cpgIslandExt.txt.gz','rt') as f, open('cpgislands.bed','w') as out:
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  chrom=p[1]; start=int(p[2]); end=int(p[3])
  if '_' in chrom: continue
  out.write(f'{chrom}\\t{start}\\t{end}\\n')
PY
bedtools sort -g canonical.chrom.sizes -i cpgislands.bed > cpgislands_sorted.bed
wc -l cpgislands_sorted.bed
bedtools coverage -sorted -g canonical.chrom.sizes -counts -a cpgislands_sorted.bed -b signal_canonical.tagalign.gz > cpg_signal_counts.bed && echo signal_done
bedtools coverage -sorted -g canonical.chrom.sizes -counts -a cpgislands_sorted.bed -b control_canonical.tagalign.gz > cpg_control_counts.bed && echo control_done"
