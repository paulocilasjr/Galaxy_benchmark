/bin/bash -lc "python3 - <<'PY'
import collections, math
scale=1.6392283037914148
# parse avg to name->mean0
def parse(path):
    d={}
    with open(path) as f:
        for line in f:
            p=line.rstrip().split('\\t')
            d[p[0]]=float(p[4])
    return d
sig=parse('bins/rmsk.signal.avg.tab'); ctrl=parse('bins/rmsk.control.avg.tab')
cls_s=collections.defaultdict(float); cls_c=collections.defaultdict(float); cls_n=collections.defaultdict(int)
with open('bins/rmsk.meta.tsv') as f:
    for line in f:
        name,cls=line.rstrip().split('\\t')
        if name in sig and name in ctrl:
            cls_s[cls]+=sig[name]; cls_c[cls]+=ctrl[name]; cls_n[cls]+=1
print('class\\tn\\tmean_sig\\tmean_ctrl\\tctrl_norm\\tlog2_enr')
rows=[]
for cls in sorted(cls_n, key=lambda x:-cls_n[x]):
    n=cls_n[cls]; ms=cls_s[cls]/n; mc=cls_c[cls]/n; cn=mc*scale; r=math.log2((ms+0.1)/(cn+0.1))
    rows.append((cls,n,ms,mc,cn,r))
    print(f'{cls}\\t{n}\\t{ms:.4f}\\t{mc:.4f}\\t{cn:.4f}\\t{r:.3f}')
# cpg
sig=parse('bins/cpg.signal.avg.tab'); ctrl=parse('bins/cpg.control.avg.tab')
ms=sum(sig.values())/len(sig); mc=sum(ctrl.values())/len(ctrl); cn=mc*scale
print('CpG_islands\\t',len(sig),f'{ms:.4f}\\t{mc:.4f}\\t{cn:.4f}\\t{math.log2((ms+0.1)/(cn+0.1)):.3f}')
PY"
