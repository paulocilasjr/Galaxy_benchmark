/bin/bash -lc "python3 - <<'PY'
import pyBigWig, math
sig=pyBigWig.open('bigwig/signal.bw'); ctrl=pyBigWig.open('bigwig/control.bw')
ssum=sig.header()['sumData']; csum=ctrl.header()['sumData']; scale=ssum/csum
chroms=[c for c in sig.chroms() if c.startswith('chr') and '_' not in c and c not in ('chrM','chrY')]
# sort natural
chroms=sorted(chroms, key=lambda x:int(x[3:]) if x[3:].isdigit() else 99)
print('chrom\\tlen\\tmean_sig\\tmean_ctrl\\tctrl_norm\\tlog2')
for c in chroms:
 length=sig.chroms(c)
 s=sig.stats(c,0,length,type='mean')[0] or 0
 ct=ctrl.stats(c,0,length,type='mean')[0] or 0
 cn=ct*scale
 print(c, length, f'{s:.4f}', f'{ct:.4f}', f'{cn:.4f}', f'{math.log2((s+0.1)/(cn+0.1)):.3f}')
PY"
