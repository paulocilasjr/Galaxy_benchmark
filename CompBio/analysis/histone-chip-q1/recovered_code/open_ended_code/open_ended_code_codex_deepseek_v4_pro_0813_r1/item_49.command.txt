/bin/bash -lc "python3 - <<'PY'
import gzip
canon=set(['chr'+str(i) for i in range(1,23)]+['chrX','chrY'])
# rmsk
with gzip.open('ref/rmsk.txt.gz','rt') as f, open('bins/rmsk.bed','w') as o, open('bins/rmsk.meta.tsv','w') as m:
    for i,line in enumerate(f):
        p=line.rstrip('\\n').split('\\t')
        if len(p)<12: continue
        chrom=p[5]; start=int(p[6]); end=int(p[7]); cls=p[11]
        if chrom not in canon: continue
        name=f'r{i}'
        o.write(f'{chrom}\\t{start}\\t{end}\\t{name}\\n')
        m.write(f'{name}\\t{cls}\\n')
print('rmsk done')
# cpg
with gzip.open('ref/cpgIslandExt.txt.gz','rt') as f, open('bins/cpg.bed','w') as o:
    for i,line in enumerate(f):
        p=line.rstrip('\\n').split('\\t')
        if len(p)<4: continue
        chrom=p[1]; start=int(p[2]); end=int(p[3])
        if chrom not in canon: continue
        o.write(f'{chrom}\\t{start}\\t{end}\\tc{i}\\n')
print('cpg done')
PY
wc -l bins/rmsk.bed bins/rmsk.meta.tsv bins/cpg.bed; head -3 bins/rmsk.bed; head -3 bins/cpg.bed"
