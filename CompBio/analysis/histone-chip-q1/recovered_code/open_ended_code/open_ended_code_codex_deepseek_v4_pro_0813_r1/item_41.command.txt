/bin/bash -lc "python3 - <<'PY'
genes='ref/genes.pc.longest.tsv'
out='bins/bins.uniq.bed'
idx=0
with open(genes) as f, open(out,'w') as o:
    next(f)
    for line in f:
        p=line.rstrip().split('\\t')
        gene,chrom,strand=p[0],p[1],p[2]; start=int(p[3]); end=int(p[4])
        if strand=='+':
            tss=start; tes=end
        else:
            tss=end; tes=start
        for b in range(100):
            s=tss-5000+b*100; e=s+100
            o.write(f'{chrom}\\t{max(0,s)}\\t{e}\\tg{idx}_T{b}\\n')
        for b in range(100):
            s=tes-5000+b*100; e=s+100
            o.write(f'{chrom}\\t{max(0,s)}\\t{e}\\tg{idx}_E{b}\\n')
        length=end-start
        for b in range(50):
            s=start + int(length*b/50); e=start + int(length*(b+1)/50)
            if e<=s: e=s+1
            o.write(f'{chrom}\\t{s}\\t{e}\\tg{idx}_B{b}\\n')
        idx+=1
print('genes',idx)
PY
head -3 bins/bins.uniq.bed && wc -l bins/bins.uniq.bed"
