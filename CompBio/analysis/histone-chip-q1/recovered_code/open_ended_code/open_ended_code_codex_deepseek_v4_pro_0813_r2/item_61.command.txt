/bin/bash -lc "python3 - <<'PY'
import gzip, collections, os
order={}
with open('refs/hg38.chrom.sizes') as f:
    for i,line in enumerate(f): order[line.split()[0]]=i
# use C byte sort
lexchroms=sorted(order.keys(), key=lambda s:s.encode())
lexorder={c:i for i,c in enumerate(lexchroms)}
classes=collections.defaultdict(list)
with gzip.open('refs/rmsk.txt.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<13: continue
        chrom=p[5]; start=int(p[6]); end=int(p[7]); strand=p[9]; cls=p[11]
        if chrom not in order: continue
        classes[cls].append((lexorder[chrom],start,end,f'{chrom}\\t{start}\\t{end}\\t{cls}\\t0\\t{strand}'))
os.makedirs('refs/repeats',exist_ok=True)
for cls,rows in classes.items():
    rows.sort(key=lambda x:(x[0],x[1],x[2]))
    with open(f'refs/repeats/{cls}.bed','w') as f:
        for _,_,_,line in rows: f.write(line+'\\n')
print({k:len(v) for k,v in sorted(classes.items(), key=lambda kv:-len(kv[1]))})
PY
ls refs/repeats | head -20; wc -l refs/repeats/* | tail -1"
