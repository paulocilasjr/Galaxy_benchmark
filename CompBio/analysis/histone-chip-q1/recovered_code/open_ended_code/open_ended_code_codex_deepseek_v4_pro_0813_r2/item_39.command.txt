/bin/bash -lc "python3 - <<'PY'
import gzip
for fn in ['histone.chip.q1.signal.tagalign.gz']:
    seen=[]
    seen_set=set()
    with gzip.open(fn,'rt') as f:
        for line in f:
            c=line.split('\\t')[0]
            if c not in seen_set:
                seen_set.add(c); seen.append(c)
                if len(seen)>=80: break
    for i,c in enumerate(seen): print(i,c)
PY"
