/bin/bash -lc "python3 - <<'PY'
import gzip
for fn in ['histone.chip.q1.signal.tagalign.gz','histone.chip.q1.control.tagalign.gz']:
    last=None; bad=0; total=0; firstbad=[]
    with gzip.open(fn,'rt') as f:
        for line in f:
            p=line.split('\\t'); key=(p[0], int(p[1]))
            if last is not None and key<last:
                bad+=1
                if len(firstbad)<5: firstbad.append((last,key))
            last=key; total+=1
    print(fn,'total',total,'bad',bad,'examples',firstbad)
PY"
