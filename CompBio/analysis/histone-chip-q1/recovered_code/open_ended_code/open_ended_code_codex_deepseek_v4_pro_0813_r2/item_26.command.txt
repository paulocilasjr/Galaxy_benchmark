/bin/bash -lc "python3 - <<'PY'
import gzip, csv, os
# load canonical transcripts
canon=set()
with gzip.open('refs/knownCanonical.txt.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)>=5:
            canon.add(p[4].split('.')[0])  # transcript base w/o version
print('canonical transcripts', len(canon))
# parse refGene, filter canonical
genes={}
with gzip.open('refs/refGene.txt.gz','rt') as f:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<13: continue
        name=p[1]; chrom=p[2]; strand=p[3]; txStart=int(p[4]); txEnd=int(p[5]); gene=p[12]
        if name.split('.')[0] in canon:
            genes.setdefault(gene,[]).append((chrom,strand,txStart,txEnd,name))
print('canonical genes in refGene', len(genes))
# one longest transcript per gene to avoid duplicates
rows=[]
for gene,ts in genes.items():
    t=max(ts,key=lambda x:x[3]-x[2])
    rows.append((t[0],t[2],t[3],t[1],gene))
# sort by chrom/start (chrom lexical matches ucsc order? use custom sort)
order={}
with open('refs/hg38.chrom.sizes') as f:
    for i,line in enumerate(f):
        c=line.split()[0]; order[c]=i
rows.sort(key=lambda x:(order.get(x[0],999), x[1]))
with open('refs/canonical_genes.bed','w') as f:
    for chrom,start,end,strand,gene in rows:
        f.write(f'{chrom}\\t{start}\\t{end}\\t{gene}\\t0\\t{strand}\\n')
# TSS windows
with open('refs/tss_2kb.bed','w') as f:
    for chrom,start,end,strand,gene in rows:
        tss=start if strand=='+' else end
        f.write(f'{chrom}\\t{max(0,tss-2000)}\\t{tss+2000}\\t{gene}\\t0\\t{strand}\\n')
# gene bodies minus 2kb TSS? create full gene body and also 5kb after TSS maybe
with open('refs/gene_bodies.bed','w') as f:
    for chrom,start,end,strand,gene in rows:
        f.write(f'{chrom}\\t{start}\\t{end}\\t{gene}\\t0\\t{strand}\\n')
# exons from canonical longest
with open('refs/canonical_exons.bed','w') as f:
    with gzip.open('refs/refGene.txt.gz','rt') as fin:
        for line in fin:
            p=line.rstrip('\\n').split('\\t')
            if len(p)<11: continue
            name=p[1]
            if name.split('.')[0] not in canon: continue
            chrom=p[2]; strand=p[3]; starts=list(map(int,p[9].rstrip(',').split(','))); ends=list(map(int,p[10].rstrip(',').split(',')))
            for s,e in zip(starts,ends):
                f.write(f'{chrom}\\t{s}\\t{e}\\t{name}\\t0\\t{strand}\\n')
print('rows', len(rows))
PY
wc -l refs/canonical_genes.bed refs/tss_2kb.bed refs/gene_bodies.bed refs/canonical_exons.bed"
