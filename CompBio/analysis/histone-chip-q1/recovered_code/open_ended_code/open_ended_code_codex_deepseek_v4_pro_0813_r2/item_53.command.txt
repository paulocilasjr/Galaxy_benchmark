/bin/bash -lc "python3 - <<'PY'
# union of tagalign order and hg38 sizes; sort lexicographically C locale
chroms=set()
for fn in ['refs/tagalign_order.txt','refs/hg38.chrom.sizes']:
    with open(fn) as f:
        for line in f:
            c=line.split()[0]
            if c: chroms.add(c)
# sort using C locale by bytes via python
import functools
def cmp(a,b):
    return (a>b)-(a<b)
lst=sorted(chroms, key=lambda s: s.encode('utf-8'))
with open('refs/lex.genome','w') as f:
    for c in lst:
        f.write(f'{c}\\t1\\n')
print('n',len(lst))
print('\\n'.join(lst[:40]))
PY
cd refs && for f in canonical_genes.bed tss_2kb.bed gene_bodies.bed canonical_exons.bed ccre_CTCF_only.bed ccre_DNase_H3K4me3.bed ccre_PLS.bed ccre_pELS.bed ccre_dELS.bed cpg_islands.bed blacklist.bed; do bedtools sort -g lex.genome -i \""'$f" > tmp_sorted.bed && mv tmp_sorted.bed "$f"; done
head -5 refs/tss_2kb.bed'
