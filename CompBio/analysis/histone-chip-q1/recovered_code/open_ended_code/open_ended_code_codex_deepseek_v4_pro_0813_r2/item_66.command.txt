/bin/bash -lc "python3 - <<'PY'
import gzip, collections
files=['histone.chip.q1.signal.lexsort.tagalign.gz','histone.chip.q1.control.lexsort.tagalign.gz']
counts=[]
for fn in files:
    c=collections.Counter()
    with gzip.open(fn,'rt') as f:
        for line in f:
            c[line.split('\\t')[0]]+=1
    counts.append(c)
s,c=counts
main=['chr1','chr2','chr3','chr4','chr5','chr6','chr7','chr8','chr9','chr10','chr11','chr12','chr13','chr14','chr15','chr16','chr17','chr18','chr19','chr20','chr21','chr22','chrX','chrY']
st=sum(s.values()); ct=sum(c.values())
print('totals',st,ct)
for ch in main:
    print(f'{ch:5s} sig={s[ch]/st*1e6:9.2f}/M ctrl={c[ch]/ct*1e6:9.2f}/M normfold={(s[ch]/st)/(c[ch]/ct):6.2f} sigcount={s[ch]}')
PY"
