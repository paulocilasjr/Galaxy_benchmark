/bin/bash -lc "python3 - <<'PY'
rows=[]
with open('refs/tss_2kb.bed') as f:
    for i,line in enumerate(f):
        if i>=20000: break
        p=line.rstrip('\\n').split('\\t')
        chrom=p[0]; strand=p[5]
        tss=int(p[1])+2000
        for j in range(-30,31):
            s=tss+j*100; e=s+100
            if e<=0: continue
            if s<0: s=0
            rows.append((chrom,s,e,f'{p[3]}_{j}'))
order={}
with open('refs/lex.genome') as f:
    for k,line in enumerate(f): order[line.split()[0]]=k
rows.sort(key=lambda x:(order.get(x[0],999),x[1]))
with open('refs/tss_bins_20k.bed','w') as f:
    for chrom,s,e,name in rows:
        f.write(f'{chrom}\\t{s}\\t{e}\\t{name}\\n')
print(len(rows))
PY
for sample in signal control; do
  echo START "'$sample
  bedtools map -sorted -g refs/lex.genome -a refs/tss_bins_20k.bed -b histone.chip.q1.${sample}.lexsort.tagalign.gz -c 1 -o count > counts/${sample}_tss_bins_20k.txt
  echo DONE $sample $(wc -l < counts/${sample}_tss_bins_20k.txt)
done'
