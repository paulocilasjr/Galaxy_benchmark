/bin/bash -lc 'for f in promoters_2kb.merged.bed gene_bodies_no_prom.merged.bed genes_full.merged.bed; do bedtools sort -g hg38.main.chrom.sizes -i "$f" > "${f%.bed}.gsort.bed"; done
bedtools complement -i genes_full.merged.gsort.bed -g hg38.main.chrom.sizes > intergenic.bed
wc -l intergenic.bed && head -5 intergenic.bed
for kind in signal control; do bedtools coverage -counts -a intergenic.bed -b <(gzip -cd histone.chip.q1.${kind}.tagalign.gz) > intergenic.${kind}.counts.bed; done
python3 - <<'"'PY'
N={'signal':44898157,'control':27109157}
files={
 'intergenic':'intergenic.{kind}.counts.bed',
 'genes_full':'genes_full.merged.bed.{kind}.counts.bed',
 'promoters_2kb':'promoters_2kb.merged.bed.{kind}.counts.bed',
 'gene_bodies_no_prom':'gene_bodies_no_prom.merged.bed.{kind}.counts.bed',
}
for label,pat in files.items():
 print('\\n'+label)
 vals={}
 for kind in ['signal','control']:
  bp=cnt=0
  with open(pat.format(kind=kind)) as f:
   for line in f:
    a=line.split('\\t'); bp+=int(a[2])-int(a[1]); cnt+=int(a[3])
  dens=cnt/N[kind]*1e6/bp
  vals[kind]=dens
  print(kind, 'Mb', round(bp/1e6,2), 'reads', cnt, 'rpm_per_bp', dens)
 print('norm fold S/C', vals['signal']/vals['control'])
PY"
