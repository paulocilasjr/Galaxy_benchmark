/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict
# parse refGene into promoter/gene intervals for main chromosomes, one representative per gene per strand/TSS
main={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}
chrom_sizes={}
with open('hg38.chrom.sizes') as f:
 for line in f:
  c,s=line.split()[:2]
  if c in main: chrom_sizes[c]=int(s)
# keep all transcript TSS, merge later
prom=[]; bodies=[]; fullgenes=[]; tss=[]
with gzip.open('refGene.txt.gz','rt') as f:
 for line in f:
  a=line.rstrip('\\n').split('\\t')
  if len(a)<13: continue
  chrom,strand=a[2],a[3]
  if chrom not in chrom_sizes: continue
  txStart,txEnd=int(a[4]),int(a[5])
  if txEnd-txStart<1000: continue
  pos=txStart if strand=='+' else txEnd
  tss.append((chrom,max(0,pos-1),min(chrom_sizes[chrom],pos),strand,a[12]))
  prom.append((chrom,max(0,pos-2000),min(chrom_sizes[chrom],pos+2000)))
  fullgenes.append((chrom,txStart,txEnd))
  if strand=='+':
   s=max(txStart,pos+2000); e=txEnd
  else:
   s=txStart; e=min(txEnd,pos-2000)
  if e>s: bodies.append((chrom,s,e))
for name, rows in [('tss_1bp.bed',tss),('promoters_2kb.bed',prom),('gene_bodies_no_prom.bed',bodies),('genes_full.bed',fullgenes)]:
 with open(name,'w') as out:
  for r in sorted(rows): out.write('\\t'.join(map(str,r[:3]))+'\\n')
PY
for f in promoters_2kb.bed gene_bodies_no_prom.bed genes_full.bed; do sort -k1,1 -k2,2n \""'$f" | bedtools merge -i - > "${f%.bed}.merged.bed"; done
wc -l *.merged.bed'
