/bin/bash -lc "python3 - <<'PY'
import gzip, collections, math, statistics, os, sys
for fn in ['histone.chip.q1.signal.tagalign.gz','histone.chip.q1.control.tagalign.gz']:
    chrom=collections.Counter(); strand=collections.Counter(); lengths=collections.Counter();
    # bins 1kb and 10kb counts; only primary chromosomes
    b1=collections.Counter(); b10=collections.Counter(); b100=collections.Counter()
    starts=[]; ends=[]
    n=0; malformed=0
    with gzip.open(fn,'rt') as f:
      for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<6: malformed+=1; continue
        c,s,e=p[0],int(p[1]),int(p[2]); st=p[5]
        n+=1; chrom[c]+=1; strand[st]+=1; lengths[e-s]+=1
        if c in {'chrX','chrY','chrM'} or c.startswith('chr') and c[3:].isdigit():
          b1[(c,s//1000)]+=1; b10[(c,s//10000)]+=1; b100[(c,s//100000)]+=1
    vals=sorted(lengths.items())
    print('\\n',fn,'n',n,'malformed',malformed,'chroms',len(chrom),'strand',strand)
    print('lengths',vals[:20], '... nuniq',len(vals))
    print('chrom counts')
    for c,k in chrom.most_common(): print(c,k,round(k/n,4))
    for label,b in [('1kb',b1),('10kb',b10),('100kb',b100)]:
      counts=list(b.values());
      print(label,'occupied',len(counts),'meanocc',sum(counts)/len(counts),'max',max(counts),'q50',statistics.median(counts),'q90',statistics.quantiles(counts,n=10)[8],'q99',statistics.quantiles(counts,n=100)[98])
PY"
