/bin/bash -lc "awk -F '\\t' '
function primary(c){return (c ~ /"'^chr([0-9]+|X|Y)$/)}
FNR==1{file++}
primary($1){k=$1 ":" int($2/10000); u[k]=1; if(file==1){s[k]++; ns++} else {c[k]++; nc++}}
END{for(k in u) print k, s[k]+0, c[k]+0, ns, nc}'"' <(gzip -cd histone.chip.q1.signal.tagalign.gz) <(gzip -cd histone.chip.q1.control.tagalign.gz) | python3 - <<'PY'
import sys, math, collections, statistics
rows=[]
for line in sys.stdin:
 p=line.split(); rows.append((p[0],int(p[1]),int(p[2])))
S=sum(x[1] for x in rows); C=sum(x[2] for x in rows)
print('bins',len(rows),'S',S,'C',C,'norm',S/C)
for name,j,total in [('signal',1,S),('control',2,C)]:
 vals=sorted((x[j] for x in rows),reverse=True)
 print(name,'nonzero',sum(v>0 for v in vals),'q50',vals[len(vals)//2], 'q90',vals[int(len(vals)*.1)], 'q99',vals[int(len(vals)*.01)],'max',vals[0])
 for f in [.001,.005,.01,.02,.05,.1,.2]:
  k=max(1,math.ceil(len(vals)*f)); print(f,'topbinreads',sum(vals[:k]),'frac',round(sum(vals[:k])/total,4),end='; ')
 print()
# normalized ratio quantiles for bins with any c and count/fold
folds=[]
for k,s,c in rows:
 # ratio of normalized density, pseudocount perhaps 0.5 to avoid sparse
 if s>0 and c>0: folds.append((s/S)/(c/C))
print('ratio both',len(folds),'q', [round(x,3) for x in [statistics.quantiles(folds,n=100)[i] for i in [0,9,24,49,74,89,94,97,98]]], 'max',max(folds))
for threshold in [1.2,1.5,2,3,4,8,16]:
 sr=sum(s for _,s,c in rows if (s/S)/((c/C) if c else 1e-12)>=threshold)
 n=sum(1 for _,s,c in rows if (s/S)/((c/C) if c else 1e-12)>=threshold)
 print('fold>=',threshold,'bins',n,'signalfrac',round(sr/S,4))
# chrom normalized ratios
by=collections.defaultdict(lambda:[0,0])
for k,s,c in rows: by[k.split(':')[0]][0]+=s; by[k.split(':')[0]][1]+=c
print('chrom ratios')
for k in sorted(by): print(k,by[k][0],by[k][1],round((by[k][0]/S)/(by[k][1]/C),3))
PY"
