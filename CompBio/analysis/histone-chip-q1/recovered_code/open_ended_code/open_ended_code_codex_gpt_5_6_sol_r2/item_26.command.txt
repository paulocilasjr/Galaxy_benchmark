/bin/bash -lc "(gzip -cd histone.chip.q1.signal.tagalign.gz | awk -v b=100000 '{k="'$1 FS int($2/b); if(k!=p){if(NR>1) print n,p; p=k;n=1}else n++} END{print n,p}'"' > signal.100kb.counts) &
(gzip -cd histone.chip.q1.control.tagalign.gz | awk -v b=100000 '{k="'$1 FS int($2/b); if(k!=p){if(NR>1) print n,p; p=k;n=1}else n++} END{print n,p}'"' > control.100kb.counts) &
wait
wc -l signal.100kb.counts control.100kb.counts"
