/bin/bash -lc "python - <<'PY'
import gzip, json, urllib.request
variants=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        chrom,pos,_,ref,*rest=line.split('\\t')
        pos=int(pos)
        if pos <= 200000:
            variants.append((pos,ref.upper()))
print('early variants',len(variants), variants[:3], variants[-1] if variants else None)
url='https://api.genome.ucsc.edu/getData/sequence?genome=hs1;chrom=chr20;start=0;end=200000'
with urllib.request.urlopen(url, timeout=60) as r:
    data=json.load(r)
seq=data['dna'].upper()
match=0; examples=[]
for pos,ref in variants:
    b=seq[pos-1]
    if b==ref:
        match+=1
    elif len(examples)<10:
        examples.append((pos, ref, b))
print('hs1 chr20 first200kb match', match, 'of', len(variants), 'rate', match/len(variants) if variants else None, 'examples', examples)
PY"
