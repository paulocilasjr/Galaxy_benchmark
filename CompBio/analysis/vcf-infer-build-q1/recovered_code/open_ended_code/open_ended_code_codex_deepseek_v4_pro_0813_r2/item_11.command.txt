/bin/bash -c "python3 - <<'PY'
import urllib.request, json, time
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom=chr20;start=0;end=10000000'
t=time.time()
try:
    data=json.load(urllib.request.urlopen(url, timeout=120))
    print(data.get('error'), len(data.get('dna','')), time.time()-t)
except Exception as e: print('ERR',e)
PY"
