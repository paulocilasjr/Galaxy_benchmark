/bin/bash -lc "perl -e '
use strict; use warnings;
my "'$ref = shift;
my $rfh;
open($rfh, "-|", "gzip", "-cd", "$ref.chr20.fa.gz") or die $!;
my $seq = "";
while (my $line = <$rfh>) { chomp $line; next if $line =~ /''^>/; $seq .= uc($line); }
close $rfh;
my ($total,$match,$mismatch) = (0,0,0);
my $vfh;
open($vfh, "-|", "gzip", "-cd", "vcf.infer.build.q1.vcf.gz") or die $!;
while (my $line = <$vfh>) {
  next if $line =~ /''^#/;
  chomp $line;
  my @f = split /'"\\t/, "'$line;
  my ($pos,$want) = ($f[1], uc($f[3]));
  my $got = substr($seq, $pos-1, length($want));
  $total++;
  if ($got eq $want) { $match++; }
  else { $mismatch++; print "mismatch $f[0]:$pos want=$want got=$got'"\\n\" if "'$mismatch <= 8; }
}
close $vfh;
printf "build=%s length=%d total=%d match=%d mismatch=%d rate=%.6f'"\\n\", "'$ref,length($seq),$total,$match,$mismatch,$match/$total;
'"' 'hg19'"
