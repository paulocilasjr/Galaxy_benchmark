/bin/bash -lc "python3 - <<'PY'
import gzip
from pathlib import Path

vcf='vcf.infer.build.q1.vcf.gz'
refs={'hg18':'refs/hg18.chr20.fa.gz','hg19':'refs/hg19.chr20.fa.gz','hg38':'refs/hg38.chr20.fa.gz'}
records=[]
with gzip.open(vcf,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        parts=line.split('\\t')
        records.append((int(parts[1]), parts[3].upper()))
print('records', len(records), 'minmax', records[0][0], records[-1][0])
for name,path in refs.items():
    seq=[]
    with gzip.open(path,'rt') as f:
        for line in f:
            if line.startswith('>'): continue
            seq.append(line.strip().upper())
    s=''.join(seq)
    total=0; match=0; n=0; beyond=0; examples=[]
    for pos,ref in records:
        if pos>len(s):
            beyond+=1; continue
        base=s[pos-1:pos-1+len(ref)].upper()
        if 'N' in base:
            n+=1
        total+=1
        if base==ref:
            match+=1
        elif len(examples)<8:
            examples.append((pos,ref,base))
    print(name, 'len', len(s), 'compared', total, 'match', match, 'mismatch', total-match, 'N', n, 'beyond', beyond, 'pct', round(match/total*100,4) if total else None)
    print('examples', examples)
PY"
