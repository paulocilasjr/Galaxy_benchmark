/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
pos=61795
for genome in ['hg18','hg19','hg38']:
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome={genome};chrom=chr20;start={pos-1};end={pos}'
    try:
        data=json.load(urllib.request.urlopen(url, timeout=20))
        print(genome, data.get('dna'))
    except Exception as e:
        print(genome, 'ERR', e)
PY"
