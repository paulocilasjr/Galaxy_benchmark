/bin/bash -lc "python3 - <<'PY'
import gzip
# load hs1 sequence
seq=[]
with open('refs/hs1.chr20.fa') as f:
    f.readline()
    for line in f:
        seq.append(line.strip())
hs=''.join(seq).upper()
print('len hs1', len(hs))
pos_ref=[]
with gzip.open('vcf.infer.build.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')
        pos_ref.append((int(p[1]), p[3].upper()))
matches=0; mism=0; ex=[]
for p,r in pos_ref:
    b=hs[p-1] if p-1 < len(hs) else 'N'
    if b==r:
        matches+=1
    else:
        mism+=1
        if len(ex)<20: ex.append((p,r,b))
print('hs1 matches', matches, 'mismatches', mism)
print(ex)
PY"
