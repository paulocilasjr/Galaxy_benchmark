/bin/bash -lc 'for f in refs/hg18.chr20.fa.gz refs/hg19.chr20.fa.gz refs/hg38.chr20.fa.gz; do gzip -dk "$f"; done
for f in refs/*.chr20.fa; do samtools faidx "$f"; done
for build in hg18 hg19 hg38; do total=$(bcftools view -H vcf.infer.build.q1.vcf.gz | wc -l | tr -d '"' '); mismatches="'$(bcftools norm -f "refs/${build}.chr20.fa" -c w vcf.infer.build.q1.vcf.gz -Ou 1>/dev/null 2>"refs/${build}.mismatch.log"; grep -c '"'REF_MISMATCH' \"refs/"'${build}.mismatch.log" || true); printf '"'%s total=%s mismatches=%s matches=%s\\n' \""'$build" "$total" "$mismatches" "$((total-mismatches))"; head -n 3 "refs/${build}.mismatch.log"; done'
