import gzip

def get(name,lo,hi,file):
 found=False;pos=0;out=[]
 with gzip.open(file,'rt') as f:
  for line in f:
   if line.startswith('>'):
    found=line[1:].split()[0]==name;pos=0;continue
   if found:
    s=line.strip();a=max(0,lo-pos-1);b=min(len(s),hi-pos)
    if a<b:out.append(s[a:b])
    pos+=len(s)
    if pos>=hi:break
 return ''.join(out)
r=get('CAUYPG010105159.1',30208,30848,'semjoh_genomic.fna.gz')
print('ref len',len(r),r)
comp=str.maketrans('ACGTNacgtn','TGCANtgcan')
cdna=r.translate(comp)[::-1]
print('cdna len',len(cdna),cdna)
print('ATGs',[i+1 for i in range(len(cdna)-2) if cdna[i:i+3]=='ATG'])
PY"
