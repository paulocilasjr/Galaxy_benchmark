/bin/bash -lc "python - <<'PY'
import json,glob,re,os
for f in glob.glob('*json'):
 if f in ['human_ens.json','species.json','hom.json']: continue
 try: x=json.load(open(f))
 except: continue
 tr=x.get('id'); ex=x.get('Exon',[]); strand=x.get('strand'); trans=x.get('Translation',{})
 raw=open(f.replace('.json','.cdna.fa')).read() if os.path.exists(f.replace('.json','.cdna.fa')) else ''
 seq=''.join(raw.splitlines())
 # sequence endpoint returned raw DNA, but if header strip
 if seq.startswith('>'): seq=''.join(seq.split('\\n')[1:])
 ex0=ex[0]
 elen=abs(ex0['end']-ex0['start'])+1
 # start codon genomic coordinate is translation.end on minus / start plus
 codg=trans['end'] if strand==-1 else trans['start']
 cstart=(ex0['end']-codg+1) if strand==-1 else (codg-ex0['start']+1)
 seg=seq[cstart-1:elen]
 print('\\n',f,tr,x.get('display_name'),x.get('species'),'strand',strand,'cdna_len',len(seq),'exons',[(e['start'],e['end']) for e in ex],'trans',trans['length'],'codinggen',codg,'cstart',cstart,'firstlen',elen,'seglen',len(seg))
 print('seg',seg)
 print('seq at start',seq[cstart-1:cstart+29])
PY"
