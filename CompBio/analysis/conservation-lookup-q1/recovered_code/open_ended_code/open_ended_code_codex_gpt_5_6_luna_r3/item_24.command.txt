/bin/bash -lc "curl -L --fail --silent --show-error -H 'Content-Type: application/json' 'https://rest.ensembl.org/homology/symbol/homo_sapiens/SHH?format=condensed;sequence=cdna' -o hom.json && python - <<'PY'
import json
x=json.load(open('hom.json'))
print(type(x),x.keys())
for h in x.get('data',[]):
 print(h.get('id'),h.get('species'),len(h.get('homologies',[])))
 for z in h.get('homologies',[]):
  t=z.get('target',{}); print(' ',z.get('type'),t.get('species'),t.get('id'),t.get('align_seq', '')[:30], len(t.get('align_seq','')))
PY"
