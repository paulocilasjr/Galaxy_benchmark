/bin/bash -lc "python3 - <<'PY'
import json
out=json.load(open('ensembl_segments.json'))
h=out['homo_sapiens']['seq']
# species from Ensembl; use best 300-bp window within extracted first coding exon segment to represent the human-corresponding region
def lev(a,b):
 p=list(range(len(b)+1))
 for i,x in enumerate(a,1):
  c=[i]+[0]*len(b)
  for j,y in enumerate(b,1): c[j]=min(p[j]+1,c[j-1]+1,p[j-1]+(x"'!=y))
  p=c
 return p[-1]
for sp in ['"'pan_troglodytes','macaca_mulatta','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']:
 s=out[sp]['seq']
 best=None
 for i in range(0,len(s)-len(h)+1):
  w=s[i:i+len(h)]
  d=lev(h,w)
  if best is None or d<best[0]: best=(d,i,w[:30],w[-30:])
 print(sp,'full_len',len(s),'best_dist',best[0],'offset',best[1],best[2],best[3],'full_dist',lev(h,s))
for fa,name in [('indri_region.fa','indri_indri'),('sem_region.fa','semnopithecus_johnii')]:
 seq=''.join(l.strip() for l in open(fa) if not l.startswith('>')).upper()
 print(name,'len',len(seq),'dist',lev(h,seq),seq[:30],seq[-30:])
PY"
