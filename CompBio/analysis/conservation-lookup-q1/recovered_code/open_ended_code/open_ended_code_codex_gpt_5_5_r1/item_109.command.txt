/bin/bash -lc "curl -s 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=CAUYPG010105159.1&seq_start=30208&seq_stop=30507&strand=2&rettype=fasta&retmode=text' > sem_region.fa && python3 - <<'PY'
import json
h=json.load(open('ensembl_segments.json'))['homo_sapiens']['seq']
seq=''.join(l.strip() for l in open('sem_region.fa') if not l.startswith('>')).upper()
print(len(seq),seq[:30],seq[-30:])
def lev(a,b):
 p=list(range(len(b)+1))
 for i,x in enumerate(a,1):
  c=[i]+[0]*len(b)
  for j,y in enumerate(b,1): c[j]=min(p[j]+1,c[j-1]+1,p[j-1]+(x"'!=y))
  p=c
 return p[-1]
print(lev(h,seq))
PY'
