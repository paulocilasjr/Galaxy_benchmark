/bin/bash -lc "sleep 10; python3 - <<'PY'
import urllib.parse, urllib.request, time
runs={'indri_prefix':'6NRJDPR8014','sem_prefix':'6NRJMH19014','sem_org':'6NRJTBMJ014'}
for name,rid in runs.items():
 print('fetch',name,rid)
 for i in range(10):
  url='https://blast.ncbi.nlm.nih.gov/Blast.cgi?'+urllib.parse.urlencode({'CMD':'Get','RID':rid,'FORMAT_TYPE':'XML'})
  text=urllib.request.urlopen(url,timeout=120).read().decode()
  if 'Status=WAITING' in text:
   print(' waiting')
   time.sleep(10); continue
  open(name+'.xml','w').write(text)
  print(' saved',len(text),text[:70].replace('\\n',' '))
  break
PY"
