/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, json, time
species=['Indri indri','Semnopithecus johnii','Pan troglodytes','Macaca mulatta','Pan paniscus','Ailuropoda melanoleuca','Delphinapterus leucas']
for sp in species:
 q=urllib.parse.quote(f'{sp}[Organism] AND latest[filter]')
 url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=assembly&term={q}&retmode=json&retmax=10'
 j=json.load(urllib.request.urlopen(url))
 ids=j['esearchresult']['idlist']
 print('\\n',sp,j['esearchresult']['count'],ids)
 if ids:
  s=json.load(urllib.request.urlopen('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=assembly&id='+','.join(ids)+'&retmode=json'))
  for i in ids[:5]:
   d=s['result'][i]
   print(i,d.get('assemblyaccession'),d.get('assemblyname'),d.get('assemblystatus'),d.get('refseq_category'),d.get('ftppath_refseq') or d.get('ftppath_genbank'))
 time.sleep(.35)
PY"
