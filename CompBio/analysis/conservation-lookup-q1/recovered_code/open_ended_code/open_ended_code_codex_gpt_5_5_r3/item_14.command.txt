import urllib.request,json
for url in [
'https://rest.ensembl.org/homology/id/human/ENSG00000164690?target_species=pan_troglodytes;type=orthologues;content-type=application/json',
'https://rest.ensembl.org/homology/id/ENSG00000164690?target_species=pan_troglodytes;type=orthologues;content-type=application/json',
'https://rest.ensembl.org/homology/id/ENSG00000164690?target_species=pan_troglodytes&content-type=application/json',
]:
 print('\\n',url)
 try:
  data=json.load(urllib.request.urlopen(url,timeout=30))
  print(data.keys(), len(data.get('data',[])))
  print(data['data'][0]['homologies'][0]['type'], data['data'][0]['homologies'][0]['target']['id'])
 except Exception as e:
  print(type(e), e)
