/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
sp=json.load(urllib.request.urlopen('https://rest.ensembl.org/info/species?content-type=application/json',timeout=30))['species']
for target in ['Pan troglodytes','Macaca mulatta','Pan paniscus','Indri indri','Semnopithecus johnii','Ailuropoda melanoleuca','Delphinapterus leucas']:
    print('\\nTARGET',target)
    key=target.lower().replace(' ','_')
    hits=[s for s in sp if key in (s.get('name','')+' '+s.get('display_name','')+' '+str(s.get('aliases',''))).lower()]
    for s in hits[:10]:
        print(s['name'], s.get('display_name'), s.get('assembly'), s.get('division'), s.get('aliases'))
PY"
