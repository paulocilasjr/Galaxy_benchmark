/bin/bash -c "python3 - <<'PY'
import requests,json,time
url='https://rest.ensembl.org/info/species'
for i in range(5):
    r=requests.get(url,params={'division':'EnsemblVertebrates'},headers={'Content-Type':'application/json'},timeout=60)
    print('try',i,r.status_code,len(r.text))
    if r.status_code==200:
      data=r.json()
      sp=data.get('species',[])
      print('count',len(sp))
      names=[s['name'] for s in sp]
      for n in names:
        if 'indri' in n or 'semn' in n or 'johnii' in n or 'trach' in n:
          print('match',n)
      # print some around indri
      for n in names:
        if 'indri' in n:
          print([s for s in sp if s['name']==n])
      break
    time.sleep(5)
PY"
