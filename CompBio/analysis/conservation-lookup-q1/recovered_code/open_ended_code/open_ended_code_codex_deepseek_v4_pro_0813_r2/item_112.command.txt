/bin/bash -c "python3 - <<'PY'
import requests,re
id='3037588952'
r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi',params={'db':'nuccore','id':id,'rettype':'gbwithparts','retmode':'text'},timeout=30)
text=r.text
print(text[:3000])
# parse CDS locations and sequence maybe
# find first CDS
print('CDS lines', re.findall(r'CDS\\s+.*',text)[:5])
# extract sequence
seq=''.join(re.findall(r'"'^'"\\s+\\d+\\s+([acgtn]+)',text,re.M)).upper()
print('seq len',len(seq),seq[:100])
PY"
