/bin/bash -c "python3 - <<'PY'
import requests,time
terms=[
'Indri indri SHH sonic hedgehog',
'Semnopithecus johnii SHH sonic hedgehog',
'Trachypithecus johnii SHH',
'Indri indri sonic hedgehog mRNA',
'Semnopithecus johnii sonic hedgehog mRNA',
'Trachypithecus johnii sonic hedgehog mRNA',
]
for term in terms:
    for db in ['nucleotide','protein','gene','assembly']:
      r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':db,'term':term,'retmode':'json','retmax':10},timeout=30)
      ids=r.json().get('esearchresult',{}).get('idlist')
      if ids:
        print(db, term, ids)
      time.sleep(0.25)
PY"
