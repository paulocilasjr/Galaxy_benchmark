/bin/bash -c "python3 - <<'PY'
import requests,time,re
rid='81NYKU4A014'
for i in range(30):
 r=requests.get('https://blast.ncbi.nlm.nih.gov/blast/Blast.cgi',params={'CMD':'Get','FORMAT_TYPE':'JSON2_S','RID':rid},timeout=60)
 if 'Status=WAITING' in r.text:
  print(i,'WAITING'); time.sleep(10)
 else:
  print(i,'not waiting, len',len(r.text), r.text[:200].replace('\\n',' '))
  open('indri_blast_result.txt','w').write(r.text)
  break
PY"
