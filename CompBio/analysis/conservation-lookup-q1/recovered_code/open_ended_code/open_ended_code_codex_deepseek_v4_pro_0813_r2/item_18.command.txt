/bin/bash -c "python3 - <<'PY'
import requests,json
for org in ['Indri indri','Semnopithecus johnii','Trachypithecus johnii']:

 j=r.json().get('esearchresult',{})
 print(org,j.get('count'),j.get('idlist'))
PY"
