/bin/bash -c "python3 - <<'PY'
import requests,time
sp='trachypithecus_francoisi'

 r=requests.get(f'https://rest.ensembl.org/lookup/symbol/{sp}/SHH',params={'expand':'1'},headers={'Content-Type':'application/json'},timeout=60)
 print(i,r.status_code,len(r.text),r.text[:300])
 if r.status_code==200: break
 time.sleep(3)
PY"
