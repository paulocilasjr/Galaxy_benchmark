import json,glob,os

def parse(p):
 data=json.load(open(p))
 for rep in data.get('BlastOutput2',[]):
  rep=rep.get('report',rep)
  print('report',rep.get('program'),'query_len',rep.get('results',{}).get('search',{}).get('query_len'))
  hits=rep.get('results',{}).get('hits',[])
  print('hits',len(hits))
  for h in hits[:10]:
   d=(h.get('description') or [{}])[0]
   print(' hit',h.get('num'),d.get('id'),d.get('title'),'accession',d.get('accession'),'len',h.get('len'))
   for hs in h.get('hsps',[])[:3]:
    print('  hsp q',hs.get('qseq'),'h',hs.get('hseq'),'qcov',hs.get('qcov'),'identity',hs.get('identity'),'gaps',hs.get('gaps'),'align_len',hs.get('align_len'),'qfrom',hs.get('query_from'),'qto',hs.get('query_to'),'hfrom',hs.get('hit_from'),'hto',hs.get('hit_to'),'strand',hs.get('hit_strand'))
for p in ['blast_nt_indri.json','blast_wgs_indri.json']:
 print('\\n###',p, os.path.getsize(p))
 if os.path.getsize(p)>100:
  parse(p)
PY"
