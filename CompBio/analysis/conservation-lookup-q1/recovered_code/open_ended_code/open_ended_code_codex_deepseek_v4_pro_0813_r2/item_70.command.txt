def get_json(url,params=None):
 for i in range(15):
  try:
   r=requests.get(url,params=params,headers={'Content-Type':'application/json'},timeout=60)
   if r.status_code==200:return r.json()
   print('json',r.status_code,'try',i,url)
  except Exception as e: print('err',e,'try',i)
  time.sleep(2+2*i)
 return None
def get_cdna(tid):
 for i in range(15):
  try:
   r=requests.get(f'https://rest.ensembl.org/sequence/id/{tid}',params={'type':'cdna'},headers={'Accept':'text/x-fasta'},timeout=60)
   if r.status_code==200:return ''.join(r.text.splitlines()[1:])
   print('seq',r.status_code,'try',i,tid)
  except Exception as e: print('err seq',e,'try',i)
  time.sleep(2+2*i)
 return None
def lev(a,b):
 if len(a)<len(b):a,b=b,a
 prev=list(range(len(b)+1))
 for i,ca in enumerate(a,1):
  cur=[i]
  for j,cb in enumerate(b,1):
   cur.append(min(prev[j]+1,cur[-1]+1,prev[j-1]+(ca"'!=cb)))
  prev=cur
 return prev[-1]

 ts.sort(key=key)
