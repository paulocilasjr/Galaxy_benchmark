/bin/bash -c "python3 - <<'PY'
import requests
for org in ['Microcebus murinus','Propithecus coquereli','Prolemur simus','Lemur catta','Eulemur fulvus','Daubentonia madagascariensis']:
 r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':'gene','term':f'{org}[orgn] AND SHH[gene]','retmode':'json','retmax':10},timeout=30)
 j=r.json().get('esearchresult',{})
 print(org,j.get('count'),j.get('idlist'))
PY"
