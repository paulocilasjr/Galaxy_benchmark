/bin/bash -c "python3 - <<'PY'
import requests,time,json
for org in ['Semnopithecus johnii','Trachypithecus johnii']:
    for db in ['nucleotide','protein','gene','assembly','sra','biosample']:
      r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi',params={'db':db,'term':f'{org}[orgn]','retmode':'json','retmax':20},timeout=30)
      j=r.json().get('esearchresult',{})
      print(db, org, j.get('count'), j.get('idlist'))
      time.sleep(0.25)
PY"
