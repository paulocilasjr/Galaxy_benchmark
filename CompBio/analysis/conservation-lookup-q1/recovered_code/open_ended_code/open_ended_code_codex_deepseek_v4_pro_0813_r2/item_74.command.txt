import gzip
from Bio.Seq import Seq

def get_seq(fn,target,start,end,strand):
 with gzip.open(fn,'rt') as f:
  name=None; parts=[]
  for line in f:
   if line.startswith('>'):
    if name==target: break
    name=line[1:].split()[0]; parts=[]
   elif name==target:
    parts.append(line.strip())
  s=''.join(parts)
 sub=s[start-1:end]
 if strand=='-': sub=str(Seq(sub).reverse_complement())
 return sub
# Indri plus: human segment starts at 17146, upstream 9 bases positions 17137-17145
for strand,start,end in [('+',17137,17145)]:
 print('indri upstream 17137-17145:',get_seq('indri.fna.gz','CAUYTU010176261.1',start,end,strand))
 print('indri combined upstream+seg:',get_seq('indri.fna.gz','CAUYTU010176261.1',start,17445,strand)[:60])
# Sem minus: cDNA 5' is high. Human start high coordinate 30507. upstream cDNA is positions 30516-30508 (higher). Extract reverse complement.
print('sem upstream high coords 30508-30516 rc:',get_seq('semjoh.fna.gz','CAUYPG010105159.1',30508,30516,'-'))
print('sem combined upstream+seg rc:',get_seq('semjoh.fna.gz','CAUYPG010105159.1',30208,30516,'-')[:60])
PY"
