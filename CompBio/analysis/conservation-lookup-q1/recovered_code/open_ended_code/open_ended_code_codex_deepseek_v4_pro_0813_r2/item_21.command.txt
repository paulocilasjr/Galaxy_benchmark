/bin/bash -c "python3 - <<'PY'
import requests
for base in ['https://rapid.ensembl.org','https://www.ensembl.org']:
    try:
      r=requests.get(base+'/info/species',headers={'Content-Type':'application/json'},timeout=30)
      print(base,r.status_code,len(r.text),r.text[:200])
    except Exception as e: print(e)
PY"
