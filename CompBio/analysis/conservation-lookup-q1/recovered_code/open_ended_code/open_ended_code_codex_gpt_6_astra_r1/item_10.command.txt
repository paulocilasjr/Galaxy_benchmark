/bin/bash -lc "cat > distances.py <<'PY'
import json

def lev(a,b):
 prev=list(range(len(b)+1))
 for i,c in enumerate(a,1):
  row=[i]
  for j,d in enumerate(b,1): row.append(min(row[-1]+1,prev[j]+1,prev[j-1]+(c"'!=d)))
  prev=row
 return prev[-1]

human=json.load(open('"'human_cds.json'))['seq'][:300]
seqs={'homo_sapiens':human}
for s in ['pan_troglodytes','macaca_mulatta','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']:
 x=json.load(open(s+'_gene.json'))
 t=max([t for t in x['Transcript'] if 'Translation' in t],key=lambda t:(t.get('is_canonical',0),t['Translation']['length']))
 ex=t['Exon'][0]; tr=t['Translation']
 n=tr['end']-ex['start']+1 if t['strand']==-1 else ex['end']-tr['start']+1
 seq=json.load(open(s+'_cds.json'))['seq'][:n]; seqs[s]=seq
 print(s,n,lev(human,seq),seq)
json.dump(seqs,open('segments.json','w'),indent=2)
PY
python distances.py"
