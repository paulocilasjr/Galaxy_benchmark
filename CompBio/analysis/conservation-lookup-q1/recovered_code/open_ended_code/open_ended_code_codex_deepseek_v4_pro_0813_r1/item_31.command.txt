/bin/bash -lc "datasets summary gene gene-id 743371 472586 716553 100995872 100481788 111176269 --report product --as-json-lines > species_products.jsonl
wc -l species_products.jsonl
python3 - <<'PY'
import json
for line in open('species_products.jsonl'):
    j=json.loads(line)
    print('\\nGENE',j.get('gene_id'),j.get('symbol'),j.get('taxname'), 'orientation?', j.get('orientation'))
    for t in j.get('transcripts',[]):
        cds=t.get('cds',{})
        rng=cds.get('range',[{}])
        exons=[]
        gls=t.get('genomic_locations',[])
        if gls:
            exons=gls[0].get('exons',[])
        print(' ',t.get('accession_version'), t.get('name'), t.get('type'), 'len',t.get('length'), 'cds',rng, 'exons',[(e.get('begin'),e.get('end'),e.get('order')) for e in exons], 'ens',t.get('ensembl_transcript'), 'select',t.get('select_category'))
PY"
