/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
for sp in ['Indri indri','Semnopithecus johnii','Trachypithecus johnii']:
 u='https://rest.uniprot.org/uniprotkb/search?'+urllib.parse.urlencode({'query':f'gene:SHH AND organism_name:\"{sp}\"','format':'json','size':'10'})
 req=urllib.request.Request(u,headers={'User-Agent':'Python'})
 try:
  j=json.loads(urllib.request.urlopen(req).read().decode())
  print(sp, 'count', len(j.get('results',[])))
  for r in j.get('results',[])[:5]:
   print(r.get('primaryAccession'),r.get('sequence',{}).get('length'),r.get('sequence',{}).get('value','')[:80])
 except Exception as e: print(e)
PY"
