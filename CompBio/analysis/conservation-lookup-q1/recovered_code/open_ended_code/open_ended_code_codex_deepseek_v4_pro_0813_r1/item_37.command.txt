/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, time
pairs={'pan_troglodytes':'ENSPTRG00000019912','macaca_mulatta':'ENSMMUG00000062128','pan_paniscus':'ENSPPAG00000033504','ailuropoda_melanoleuca':'ENSAMEG00000028933','delphinapterus_leucas':'ENSDLEG00000014841'}
for sp,g in pairs.items():
 u=f'https://rest.ensembl.org/lookup/id/{g}?content-type=application/json;expand=1'
 d=json.loads(urllib.request.urlopen(u).read().decode())
 print('\\n',sp,g,'canonical',d.get('canonical_transcript'), 'biotype',d.get('biotype'))
 # expand may include Transcript list
 trs=d.get('Transcript',[])
 print('transcripts count',len(trs))
 for t in trs:
  print(t.get('id'),t.get('biotype'),t.get('is_canonical'),'len',t.get('length'),'Translation',t.get('Translation'))
 # fetch canonical transcript details separately
 c=d.get('canonical_transcript')
 if c:
  u2=f'https://rest.ensembl.org/lookup/id/{c}?content-type=application/json;expand=1'
  td=json.loads(urllib.request.urlopen(u2).read().decode())
  print('canonical exons', [(e['start'],e['end']) for e in td.get('Exon',[])], 'translation',td.get('Translation'))
 time.sleep(0.2)
PY"
