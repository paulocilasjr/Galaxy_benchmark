def readfa(path): return ''.join(open(path).read().splitlines()[1:])
def lev(a,b):
    if a==b:return 0
    n,m=len(a),len(b)
    if n<m: a,b=b,a;n,m=len(a),len(b)
    prev=list(range(m+1))
    for i,ca in enumerate(a,1):
        cur=[i]+[0]*m
        for j,cb in enumerate(b,1):
            cur[j]=min(prev[j]+1,cur[j-1]+1,prev[j-1]+(ca"'!=cb))
        prev=cur
    return prev[m]
