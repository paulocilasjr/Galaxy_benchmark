/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
for db in ['protein','nucleotide']:
 for sp in ['Indri indri','Semnopithecus johnii','Trachypithecus johnii']:
  u='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':db,'term':f'SHH[All Fields] AND {sp}[Organism]','retmode':'json'})
  j=json.loads(urllib.request.urlopen(u).read().decode())
  print(db,sp,j['esearchresult'].get('count'),j['esearchresult'].get('idlist')[:5])
PY"
