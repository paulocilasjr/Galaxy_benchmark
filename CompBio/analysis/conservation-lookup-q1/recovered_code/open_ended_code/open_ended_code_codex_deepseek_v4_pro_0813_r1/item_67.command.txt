/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
for sp in ['Pan troglodytes','Macaca mulatta','Pan paniscus']:
 u='https://rest.uniprot.org/uniprotkb/search?'+urllib.parse.urlencode({'query':f'gene:SHH AND organism_name:\"{sp}\"','format':'json','size':'5'})
 req=urllib.request.Request(u,headers={'User-Agent':'Python'})
 data=urllib.request.urlopen(req).read().decode()
 j=json.loads(data)
 print('\\n',sp,j.get('results',[])[0].get('primaryAccession') if j.get('results') else None)
 for r in j.get('results',[])[:3]:
  print(r.get('primaryAccession'), r.get('proteinDescription',{}).get('recommendedName',{}).get('fullName',{}).get('value'), r.get('sequence',{}).get('length'), r.get('sequence',{}).get('value','')[:50])
PY"
