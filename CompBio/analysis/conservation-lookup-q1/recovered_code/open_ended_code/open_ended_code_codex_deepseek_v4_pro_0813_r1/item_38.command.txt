/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, time, sys
pairs={'pan_troglodytes':'ENSPTRG00000019912','macaca_mulatta':'ENSMMUG00000062128','pan_paniscus':'ENSPPAG00000033504','ailuropoda_melanoleuca':'ENSAMEG00000028933','delphinapterus_leucas':'ENSDLEG00000014841'}
for sp,g in pairs.items():
 try:
  d=json.loads(urllib.request.urlopen(f'https://rest.ensembl.org/lookup/id/{g}?content-type=application/json;expand=1').read().decode())
  trs=d.get('Transcript',[])
  c=d.get('canonical_transcript')
  print('\\n',sp,g,'canonical',c,'transcripts:')
  for t in trs:
   print(' ',t.get('id'),'len',t.get('length'),'biotype',t.get('biotype'),'canonical?',t.get('is_canonical'))
  if c:
   u2=f'https://rest.ensembl.org/lookup/id/{c}?content-type=application/json;expand=1'
   td=json.loads(urllib.request.urlopen(u2).read().decode())
   print(' canonical',c,'strand',td.get('strand'),'exons',[(e['start'],e['end'],e.get('strand')) for e in td.get('Exon',[])],'translation',td.get('Translation'))
 except Exception as e:
  print('ERROR',sp,e)
 time.sleep(0.2)
PY"
