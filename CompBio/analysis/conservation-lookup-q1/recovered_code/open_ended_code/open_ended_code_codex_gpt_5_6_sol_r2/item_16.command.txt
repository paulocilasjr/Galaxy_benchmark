/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,time
species=['pan_troglodytes','macaca_mulatta','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']
for sp in species:
 url=f'https://rest.ensembl.org/lookup/symbol/{sp}/SHH?expand=1;content-type=application/json'
 try:
  data=urllib.request.urlopen(url,timeout=60).read(); open(f'{sp}_lookup.json','wb').write(data)
  x=json.loads(data); print(sp,x.get('id'),x.get('canonical_transcript'),x.get('display_name'),len(x.get('Transcript',[])))
  for t in x.get('Transcript',[]):
   if t.get('biotype')=='protein_coding': print(' ',t['id'],t.get('is_canonical'),t.get('length'),len(t.get('Exon',[])),t.get('Translation',{}).get('length'))
 except Exception as e: print(sp,'ERROR',e)
 time.sleep(.3)
PY"
