/bin/bash -lc "curl -sS 'https://rest.ensembl.org/lookup/symbol/homo_sapiens/SHH?expand=1;content-type=application/json' -o shh_gene.json
python3 - <<'PY'
import json
x=json.load(open('shh_gene.json'))
print(x['id'],x.get('canonical_transcript'),len(x.get('Transcript',[])))
for t in x['Transcript']:
 print(t['id'],t.get('is_canonical'),t.get('biotype'),t.get('start'),t.get('end'),len(t.get('Exon',[])), t.get('Translation',{}).get('id'))
PY"
