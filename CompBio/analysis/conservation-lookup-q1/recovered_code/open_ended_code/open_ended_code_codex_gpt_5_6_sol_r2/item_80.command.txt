/bin/bash -lc "curl --max-time 60 -sS 'https://rest.ensembl.org/sequence/id/ENSAMET00000046595?type=cdna;content-type=text/plain' -o panda_ensembl_cdna.fa &
curl --max-time 60 -sS 'https://rest.ensembl.org/overlap/id/ENSAMET00000046595?feature=exon;content-type=application/json' -o panda_exons.json &
wait
python3 - <<'PY'
import json
s=''.join(open('panda_ensembl_cdna.fa').read().split()); ex=[e for e in json.load(open('panda_exons.json')) if e['Parent']=='ENSAMET00000046595']
print(len(s),[(e['rank'],e['start'],e['end']) for e in ex]); print(s.find('ATGCTG'),s[:100])
PY"
