/bin/bash -lc 'for s in pan_troglodytes macaca_mulatta pan_paniscus indri_indri semnopithecus_johnii ailuropoda_melanoleuca delphinapterus_leucas; do curl -L --fail --silent --show-error "https://rest.ensembl.org/lookup/symbol/${s}/SHH?expand=1" -H '"'Content-Type:application/json' -o \""'${s}_lookup.json" || echo "FAILED ${s}"; done
ls -l *_lookup.json
python3 - <<'"'PY'
import json,glob
for f in glob.glob('*_lookup.json'):
 try:
  x=json.load(open(f)); print(f,x.get('species'),x.get('id'),x.get('display_name'),x.get('canonical_transcript'),len(x.get('Transcript',[])))
 except Exception as e: print(f,e)
PY"
