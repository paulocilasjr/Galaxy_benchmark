/bin/bash -lc "curl -L --fail --silent --show-error 'https://rest.ensembl.org/info/species?content-type=application/json' -o ensembl_species.json && python3 - <<'PY'
import json
x=json.load(open('ensembl_species.json'))
for s in x['species']:
 n=s['name']
 if any(q in n for q in ['indri','semnopithecus','johnii','troglodytes','pan_paniscus','macaca','ailuropoda','delphinapterus']): print(s)
PY"
