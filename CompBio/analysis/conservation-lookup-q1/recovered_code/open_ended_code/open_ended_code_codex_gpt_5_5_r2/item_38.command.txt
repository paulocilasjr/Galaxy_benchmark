/bin/bash -lc "python3 - <<'PY'
import subprocess,json,urllib.parse,requests
species=['Pan troglodytes','Macaca mulatta','Pan paniscus','Indri indri','Semnopithecus johnii','Ailuropoda melanoleuca','Delphinapterus leucas']
base='https://api.ncbi.nlm.nih.gov/datasets/v2alpha/gene/symbol/SHH/taxon/'
for sp in species:
    url=base+urllib.parse.quote(sp)
    r=requests.get(url,timeout=20)
    print('\\n',sp, r.status_code)
    try:j=r.json()
    except Exception as e: print(r.text[:200]); continue
    print('total',j.get('total_count'), 'keys',j.keys())
    for rep in j.get('reports',[])[:5]:
        g=rep.get('gene',{})
        print(g.get('gene_id'), g.get('symbol'), g.get('taxname'), g.get('common_name'), g.get('type'), 'transcripts',g.get('transcript_count'),'proteins',g.get('protein_count'))
        if 'annotations' in g:
            print(' ann',[(a.get('assembly_accession'),a.get('assembly_name'),a.get('annotation_name')) for a in g['annotations'][:2]])
PY"
