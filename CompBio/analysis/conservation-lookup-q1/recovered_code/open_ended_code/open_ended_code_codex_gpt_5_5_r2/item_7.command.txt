/bin/bash -lc "python3 - <<'PY'
import requests, json
base='https://rest.ensembl.org'
h={'Content-Type':'application/json'}
gene='ENSG00000164690'
r=requests.get(f'{base}/lookup/id/{gene}',headers=h,params={'expand':1,'mane':1})
print(r.status_code)
data=r.json()
print(data.keys())
print(data.get('canonical_transcript'))
for t in data.get('Transcript',[]):
    print(t.get('id'), 'biotype',t.get('biotype'), 'canonical',t.get('is_canonical'), 'translation', t.get('Translation',{}).get('id') if t.get('Translation') else None, 'len',t.get('length'), 'start/end', t.get('start'), t.get('end'))
    if t.get('MANE') or t.get('mane_select') or t.get('MANE_Select'):
        print(' MANE?', {k:v for k,v in t.items() if 'MANE' in k or 'mane' in k})
PY"
