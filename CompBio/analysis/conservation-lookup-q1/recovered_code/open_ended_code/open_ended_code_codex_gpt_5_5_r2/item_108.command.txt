/bin/bash -lc "python3 - <<'PY'
from extract_ncbi_shh import get_segment, WORK, levenshtein
human=get_segment(WORK/'ncbi_human_run1/pkg/ncbi_dataset/data')[6]
# Ensembl values from cross-check for mapped region; NCBI extracted sequence for panda same as Ensembl, beluga skip9 same.
seqs={}
for sp,path in [('Pan troglodytes','ncbi_species/Pan_troglodytes/pkg/ncbi_dataset/data'),('Macaca mulatta','ncbi_species/Macaca_mulatta/pkg/ncbi_dataset/data'),('Pan paniscus','ncbi_species/Pan_paniscus/pkg/ncbi_dataset/data'),('Ailuropoda melanoleuca','ncbi_species/Ailuropoda_melanoleuca/pkg/ncbi_dataset/data'),('Delphinapterus leucas','ncbi_species/Delphinapterus_leucas/pkg/ncbi_dataset/data')]:
    seg=get_segment(WORK/path)[6]
    # mapped to human start: skip the upstream species-specific 9 nt when present.
    if len(seg)==309:
        seg=seg[9:]
    seqs[sp]=seg
for sp,fn in [('Indri indri','indri_pgdp_blast.tsv'),('Semnopithecus johnii','semjoh_pgdp_blast.tsv')]:
    seqs[sp]=open(fn).readline().rstrip('\\n').split('\\t')[13].replace('-','')
order=['Pan troglodytes','Macaca mulatta','Pan paniscus','Indri indri','Semnopithecus johnii','Ailuropoda melanoleuca','Delphinapterus leucas']
for sp in order:
    print(sp, len(seqs[sp]), seqs[sp][:15], seqs[sp][-15:], levenshtein(human,seqs[sp]))
print(','.join(str(levenshtein(human,seqs[sp])) for sp in order))
PY"
