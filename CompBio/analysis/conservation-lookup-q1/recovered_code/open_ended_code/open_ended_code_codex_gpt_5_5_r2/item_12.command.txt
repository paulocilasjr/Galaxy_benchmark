/bin/bash -lc "curl -s -H 'Content-Type: application/json' 'https://rest.ensembl.org/info/species?division=EnsemblVertebrates' > species.json && python3 - <<'PY'
import json
s=json.load(open('species.json'))['species']
for q in ['troglodytes','mulatta','paniscus','indri','semnopithecus','ailuropoda','delphinapterus']:
    print('\\n',q)
    for sp in s:
        if q in sp['name'] or q in sp.get('display_name','').lower() or q in sp.get('common_name','').lower():
            print(sp['name'], sp.get('display_name'), sp.get('common_name'), sp.get('assembly'))
PY"
