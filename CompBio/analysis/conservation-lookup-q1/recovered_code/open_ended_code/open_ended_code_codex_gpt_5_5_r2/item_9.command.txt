/bin/bash -lc "python3 - <<'PY'
import requests,json
base='https://rest.ensembl.org'; h={'Content-Type':'application/json'}
tid='ENST00000297261'
d=requests.get(f'{base}/lookup/id/{tid}',headers=h,params={'expand':1,'mane':1}).json()
print('keys',d.keys())
print('strand',d['strand'],'len',d.get('length'),'start end',d['start'],d['end'])
print('Translation',json.dumps(d.get('Translation'),indent=2)[:1000])
for i,e in enumerate(sorted(d['Exon'], key=lambda e:e['start'], reverse=d['strand']==-1),1):
    print(i,e['id'],e['start'],e['end'],e.get('rank'), 'len', e['end']-e['start']+1)
seq=requests.get(f'{base}/sequence/id/{tid}',headers={'Content-Type':'text/plain'},params={'type':'cdna'}).text.strip()
print('cdna len',len(seq),seq[:100],seq[-100:])
PY"
