import json,re

def fasta(path):return ''.join(x.strip() for x in open(path) if not x.startswith('>')).upper()
def lev(a,b):
 if len(a)<len(b): a,b=b,a
 prev=list(range(len(b)+1))
 for i,x in enumerate(a,1):
  cur=[i]
  for j,y in enumerate(b,1): cur.append(min(cur[-1]+1,prev[j]+1,prev[j-1]+(x"'!=y)))
  prev=cur
 return prev[-1]
# coordinate-based cDNA start and end computed from lookup json
infos={
 '"'human':('ensembl_human_gene.json','ENST00000297261'),
 'chimp':('pan_troglodytes_gene.json','ENSPTRT00000109799'),
 'macaque':('macaca_mulatta_gene.json','ENSMMUT00000090541'),
 'bonobo':('pan_paniscus_gene.json','ENSPPAT00000043352'),
 'bear':('ailuropoda_melanoleuca_gene.json','ENSAMET00000046595'),
 'beluga':('delphinapterus_leucas_gene.json','ENSDLET00000022460'),
}
seqs={}
for name,(file,tid) in infos.items():
 x=json.load(open(file)); t=next(t for t in x['Transcript'] if t['id']==tid)
 ex=t['Exon']; e=ex[0]; strand=t['strand']; tr=t['Translation']
 if strand==1:
  cds_start=tr['start']; cpos=cds_start-e['start']+1
 else:
  cds_start=tr['end']; cpos=e['end']-cds_start+1
 exon_end=sum(abs(z['end']-z['start'])+1 for z in ex[:1])
 s=fasta(name+'_ens_cdna.fa')
 seg=s[cpos-1:exon_end]
 seqs[name]=seg
 print(name,'txlen',len(s),'strand',strand,'cdsstartgen',cds_start,'cpos',cpos,'exonlen',exon_end,'seglen',len(seg),'start',seg[:15],'end',seg[-15:])
print('human segment',seqs['human'])
for n in ['chimp','macaque','bonobo','bear','beluga']:
 print(n,lev(seqs['human'],seqs[n]))
PY
python3 compute_ensembl.py"
