/bin/bash -lc "python3 - <<'PY'
import gzip
def rec(f,key):
 out=[];ok=False
 for l in gzip.open(f,'rt'):
  if l.startswith('>'):
   if ok:break
   ok=key in l
  elif ok:out.append(l.strip())
 return ''.join(out).upper()
def read(f):return ''.join(l.strip() for l in open(f) if not l.startswith('>')).upper()
def rc(s):return s.translate(str.maketrans('ACGTN','TGCAN'))[::-1]
for name,fa,parts,strand,gz,key in [
 ('indri','indri_shh_contig.fa',[(17137,17445),(22533,22794),(25423,26225)],'+','indri_tib.codingseq.gz','g6088'),
 ('semno','semno_shh_contig.fa',[(30208,30516),(24696,24957),(21296,22122)],'-','semno_tib.codingseq.gz','g5701')]:
 s=read(fa); seq=''.join((s[a-1:b] if strand=='+' else rc(s[a-1:b])) for a,b in parts)
 tib=rec(gz,key)
 print(name,len(seq),len(tib),seq==tib,seq[:30],tib[:30],seq[-30:],tib[-30:])
PY"
