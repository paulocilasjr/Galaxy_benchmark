/bin/bash -lc "curl -L --fail --silent --show-error 'https://api.ncbi.nlm.nih.gov/datasets/v2/gene/id/6469/orthologs' -o ncbi_shh_orthologs.json && python3 - <<'PY'
import json,collections
x=json.load(open('ncbi_shh_orthologs.json'))
print(x.keys(),len(x.get('reports',[])))
for r in x.get('reports',[]):
 g=r.get('gene',{});
 if g.get('taxname') in ['Pan troglodytes','Macaca mulatta','Pan paniscus','Indri indri','Trachypithecus johnii','Semnopithecus johnii','Ailuropoda melanoleuca','Delphinapterus leucas']:
  print(g.get('taxname'),g.get('gene_id'),g.get('symbol'),g.get('ensembl_gene_ids'),g.get('transcript_count'),g.get('protein_count'),g.get('annotations'),g.keys())
print(collections.Counter(r.get('gene',{}).get('taxname') for r in x.get('reports',[])).most_common(20))
PY"
