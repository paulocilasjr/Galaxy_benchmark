/bin/bash -lc "python3 - <<'PY'
import subprocess, os
# extract local subject regions and write fasta
for label,ref,name,start in [('indri','indri_data/ncbi_dataset/data/GCA_004363605.1/GCA_004363605.1_IndInd_v1_BIUU_genomic.fna','RJWJ010014441.1',16166),
                              ('sem','sem_data/ncbi_dataset/data/GCA_963574185.1/GCA_963574185.1_PGDP_SemJoh_genomic.fna','CAUYPG010105159.1',30208)]:
    s=max(1,start-2000); e=start+299+2000
    out=subprocess.check_output(['samtools','faidx',ref,f'{name}:{s}-{e}'],text=True)
    lines=out.splitlines(); seq=''.join(lines[1:])
    with open(f'{label}_local.fa','w') as f:f.write(f'>{name}_{s}_{e}\\n{seq}\\n')
    print(label,name,s,e,len(seq))
PY
for label in indri sem; do blastn -query human_segment.fa -subject "'${label}_local.fa -task blastn -dust no -word_size 7 -evalue 1e-20 -outfmt '"'6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore qseq sseq' -out "'${label}_blast.tsv; echo ${label} exit:$? lines:$(wc -l < ${label}_blast.tsv); cat ${label}_blast.tsv; done'
