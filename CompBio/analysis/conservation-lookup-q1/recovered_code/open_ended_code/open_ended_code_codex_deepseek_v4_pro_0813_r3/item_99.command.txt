/bin/bash -lc "python3 - <<'PY'
import subprocess, json, time, os
# selected accession and cds begin, first exon length from product report
items=[
('Pan troglodytes','XM_001147185.7',333,641),
('Macaca mulatta','XM_001106515.5',333,641),
('Pan paniscus','XM_034965340.4',1542,1850),
('Ailuropoda melanoleuca','XM_034650041.1',310,609), # need first exon length compute later
('Delphinapterus leucas','XM_022576607.2',136,444),
]
# compute first exon lengths from product json
prod=json.load(open('ncbi_products.json'))
for name,acc,b,e in items:
    # find actual exons
    for rep in prod['reports']:
      p=rep['product']
      if p['taxname']==name:
       for tr in p['transcripts']:
        if tr['accession_version']==acc:
         gl=tr['genomic_locations'][0]
         ex1=[e for e in gl['exons'] if e['order']==1][0]
         e_len=ex1['end']-ex1['begin']+1
         print(name,acc,'ex1 len',e_len,'cds begin',tr['cds']['range'][0]['begin'])
         items[items.index((name,acc,b,e))]=(name,acc,b,e_len)
    out=f'ncbi_{acc}.fa'
    if not os.path.exists(out):
      url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id={acc}&rettype=fasta&retmode=text'
      subprocess.run(['curl','-s',url,'-o',out])
      # rate limit sleep? we may get errors
    print(open(out).read()[:100])
PY"
