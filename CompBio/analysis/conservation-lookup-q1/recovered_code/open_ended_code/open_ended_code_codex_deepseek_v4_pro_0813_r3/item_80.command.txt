/bin/bash -lc "curl -s \"https://rest.ensembl.org/info/species?division=EnsemblVertebrates&collection=quick\" -H \"Content-Type: application/json\" -o species_quick.json; python3 - <<'PY'
import json
j=json.load(open('species_quick.json'))
print(j.keys(), len(j.get('species',[])))
for target in ['pan_troglodytes','macaca_mulatta','pan_paniscus','indri_indri','semnopithecus_johnii','ailuropoda_melanoleuca','delphinapterus_leucas']:
    hits=[s for s in j['species'] if s.get('name')==target]
    print(target, [(h.get('name'),h.get('display_name'),h.get('assembly')) for h in hits])
PY"
