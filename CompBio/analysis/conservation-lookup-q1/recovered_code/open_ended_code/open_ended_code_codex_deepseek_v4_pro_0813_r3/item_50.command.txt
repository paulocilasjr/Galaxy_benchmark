/bin/bash -lc "python3 - <<'PY'
import subprocess, json, time, os
species_map={
'Pan troglodytes':'pan_troglodytes',
'Macaca mulatta':'macaca_mulatta',
'Pan paniscus':'pan_paniscus',
'Indri indri':'indri_indri',
'Semnopithecus johnii':'semnopithecus_johnii',
'Ailuropoda melanoleuca':'ailuropoda_melanoleuca',
'Delphinapterus leucas':'delphinapterus_leucas'
}
os.makedirs('hom',exist_ok=True)
for common,ens in species_map.items():
    out=f'hom/{ens}.json'
    url=f'https://rest.ensembl.org/homology/id/human/ENSG00000164690?target_species={ens};type=orthologues'
    for attempt in range(5):
        code=subprocess.run(['curl','-s','-o',out,'-w','%{http_code}',url,'-H','Content-Type: application/json'],capture_output=True,text=True).stdout
        if code=='200':
            try:
                j=json.load(open(out))
                data=j['data'][0]['homologies']
                print(common, 'OK', len(data))
                for h in data:
                    print(' ',h['type'], h.get('target',{}).get('id'), h.get('target',{}).get('protein_id'), h.get('target',{}).get('taxon_id'))
                break
            except Exception as e:
                print(common,'parse error',e,'code',code,open(out).read()[:200])
                break
        else:
            time.sleep(1)
    else:
        print(common,'FAIL',code,open(out).read()[:200])
PY"
