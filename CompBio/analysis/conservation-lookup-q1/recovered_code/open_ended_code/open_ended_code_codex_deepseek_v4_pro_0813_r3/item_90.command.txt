import subprocess, json, time, os, sys
from urllib.parse import quote

def curl_json(url, out, attempts=12):
    for i in range(attempts):
        code=subprocess.run(['curl','-s','-o',out,'-w','%{http_code}',url,'-H','Content-Type: application/json'],capture_output=True,text=True).stdout.strip()
        if code=='200':
            try:
                j=json.load(open(out)); return j
            except Exception as e:
                print('parse',url,e,file=sys.stderr)
                time.sleep(1)
                continue
        # if 404/400 no retry?
        if code in ('400','404'):
            print('bad code',code,url,file=sys.stderr); return None
        time.sleep(1.5)
    print('fail',url,code,open(out).read()[:100] if os.path.exists(out) else '',file=sys.stderr)
    return None

def fetch_seq(tid, typ):
    url=f'https://rest.ensembl.org/sequence/id/{tid}?type={typ}'
    out=f'seq_{tid}_{typ}.json'
    if os.path.exists(out) and os.path.getsize(out)>100:
        try: return json.load(open(out))['seq']
        except: pass
    for i in range(12):
        code=subprocess.run(['curl','-s','-o',out,'-w','%{http_code}',url,'-H','Content-Type: application/json'],capture_output=True,text=True).stdout.strip()
        if code=='200':
            try:
                j=json.load(open(out)); return j['seq']
            except Exception as e:
                print('parse seq',tid,typ,e,file=sys.stderr); time.sleep(1); continue
        time.sleep(1.5)
    raise RuntimeError(f'could not fetch seq {tid} {typ}')

def get_transcript(tid, species):
    url=f'https://rest.ensembl.org/lookup/id/{tid}?expand=1'
    out=f'trans_{tid}.json'
    j=curl_json(url,out)
    if j is None: return None
    # lookup id returns transcript object directly
    return j

def segment_for(tid, species):
    t=get_transcript(tid,species)
    if t is None: return None,None
    strand=t.get('strand')
    exons=t.get('Exon',[])
    if not exons: return None,None
    exons_sorted=sorted(exons, key=lambda e:e['start'], reverse=(strand==-1))
    first=exons_sorted[0]
    trans=t.get('Translation')
    if not trans: return None,None
    cds_start_genomic=trans['start'] if strand==1 else trans['end']
    if strand==1:
        offset=cds_start_genomic-first['start']
    else:
        offset=first['end']-cds_start_genomic
    if offset<0 or offset>=first['end']-first['start']+1:
        print('CDS start not in first exon?',tid,strand,first,trans,file=sys.stderr)
        return None,None
    cdna=fetch_seq(tid,'cdna')
    first_len=first['end']-first['start']+1
    seg=cdna[offset:first_len]
    # verify seg starts ATG? CDS maybe starts ATG
    cds=fetch_seq(tid,'cds')
    # seg should be prefix of cds
    print(tid, species, 'strand',strand,'first',first['id'],first['start'],first['end'],'offset',offset,'first_len',first_len,'seg_len',len(seg),'seg_prefix_cds',cds.startswith(seg),'seg_start',seg[:3])
    return seg, cdna

species_info=[
('9606','homo_sapiens','ENST00000297261','Homo sapiens'),
('9598','pan_troglodytes','ENSPTRT00000109799','Pan troglodytes'),

]
results={}
for taxid,ens,tid,common in species_info:

    if seg:
        results[common]=seg
        open(f'segment_{common.replace(\" \",\"_\")}.txt','w').write(seg)
    else:
        print('no segment',common)
# save human
print('\\nRESULTS')
human=results.get('Homo sapiens')
for k,v in results.items():
    if k"'!='"'Homo sapiens' and human:

        def lev(a,b):
            m,n=len(a),len(b)
            prev=list(range(n+1))
            for i in range(1,m+1):
                cur=[i]+[0]*n
                for j in range(1,n+1):
                    cur[j]=min(prev[j]+1,cur[j-1]+1,prev[j-1]+(a[i-1]"'!=b[j-1]))
                prev=cur
            return prev[n]
        print(k, len(v), lev(human,v))
