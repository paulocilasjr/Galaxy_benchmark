/bin/bash -lc "python3 - <<'PY'
import subprocess, json, urllib.parse
species=[
('Pan troglodytes','txid9598[Organism]'),
('Macaca mulatta','txid9544[Organism]'),
('Pan paniscus','txid9597[Organism]'),
('Indri indri','txid34827[Organism]'),
('Semnopithecus johnii','txid66063[Organism]'),
('Ailuropoda melanoleuca','txid9646[Organism]'),
('Delphinapterus leucas','txid9749[Organism]'),
]
for name,org in species:
    term=f'SHH[Gene Name] AND {org}'
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=gene&term='+urllib.parse.quote(term)+'&retmode=json'
    out=f'gene_{name.replace(\" \",\"_\")}.json'
    subprocess.run(['curl','-s',url,'-o',out])
    j=json.load(open(out))
    print(name, j['esearchresult']['count'], j['esearchresult']['idlist'][:10])
PY"
