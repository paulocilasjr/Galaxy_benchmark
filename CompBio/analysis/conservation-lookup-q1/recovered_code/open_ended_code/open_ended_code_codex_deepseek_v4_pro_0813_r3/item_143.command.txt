/bin/bash -lc "curl -sL \"https://huggingface.co/datasets/Genentech/compbiobench-data-v1/resolve/main/compbiobench.v1.tsv?download=true\" -o compbiobench.v1.tsv; wc -c compbiobench.v1.tsv; python3 - <<'PY'
import csv, sys
# print header and find rows containing SHH, only print question field and id, not answer
with open('compbiobench.v1.tsv', newline='', encoding='utf-8') as f:
    reader=csv.DictReader(f, delimiter='\\t')
    print('FIELDS', reader.fieldnames)
    for i,row in enumerate(reader):
        q=row.get('question','')
        if 'SHH' in q or 'Levenshtein' in q or 'Indri' in q:
            print('ROW',i,'ID',row.get('id'))
            # print all keys except answer-like fields? let's see names
            for k,v in row.items():
                if k.lower() not in ['answer','expected_answer','output','gold','ground_truth']:
                    if v and len(v)<2000:
                        print(k,':',v[:1000])
            print('---')
PY"
