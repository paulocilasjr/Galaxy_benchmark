/bin/bash -lc "python3 - <<'PY'
import subprocess, json, time, os
prod=json.load(open('ncbi_products.json'))
names=['Pan troglodytes','Macaca mulatta','Pan paniscus','Ailuropoda melanoleuca','Delphinapterus leucas']
accs={'Pan troglodytes':'XM_001147185.7','Macaca mulatta':'XM_001106515.5','Pan paniscus':'XM_034965340.4','Ailuropoda melanoleuca':'XM_034650041.1','Delphinapterus leucas':'XM_022576607.2'}
for name in names:
    acc=accs[name]
    for rep in prod['reports']:
      p=rep['product']
      if p['taxname']==name:
       for tr in p['transcripts']:
        if tr['accession_version']==acc:
         gl=tr['genomic_locations'][0]
         ex1=[e for e in gl['exons'] if e.get('order')==1][0]
         e_len=int(ex1['end'])-int(ex1['begin'])+1
         cds_begin=int(tr['cds']['range'][0]['begin'])
         print(name,acc,'ex1 len',e_len,'cds begin',cds_begin,'seg len',e_len-cds_begin+1)
         out=f'ncbi_{acc}.fa'
         if not os.path.exists(out) or os.path.getsize(out)<10:
           url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id={acc}&rettype=fasta&retmode=text'
           for attempt in range(8):
             subprocess.run(['curl','-s',url,'-o',out])
             txt=open(out,errors='ignore').read()
             if 'API rate limit exceeded' not in txt and txt.startswith('>'):
               break
             time.sleep(1)
         txt=open(out).read().replace('\\n','')
         header_end=txt.find(' ',txt.find(' '))
         seq=txt[txt.index('\\n')+1:] if '\\n' in txt else ''
         seq=''.join(open(out).read().splitlines()[1:])
         seg=seq[cds_begin-1:e_len]
         open(f'segment_ncbi_{name.replace(\" \",\"_\")}.txt','w').write(seg)
         print(' seq len',len(seq),'seg',seg[:30],seg[-20:])
PY"
