/bin/bash -lc "curl -s \"https://rest.ensembl.org/info/species?division=EnsemblVertebrates\" -H \"Content-Type: application/json\" -o species_vert.json; python3 - <<'PY'
import json
j=json.load(open('species_vert.json'))
print(type(j), j.keys() if isinstance(j,dict) else len(j))
species=j.get('species', j if isinstance(j,list) else [])
print(len(species))
for target in ['pan_troglodytes','macaca_mulatta','pan_paniscus','indri_indri','semnopithecus_johnii','ailuropoda_melanoleuca','delphinapterus_leucas']:
    hits=[s for s in species if s.get('name')==target]
    print(target, [(h.get('name'),h.get('display_name'),h.get('assembly')) for h in hits])
PY"
