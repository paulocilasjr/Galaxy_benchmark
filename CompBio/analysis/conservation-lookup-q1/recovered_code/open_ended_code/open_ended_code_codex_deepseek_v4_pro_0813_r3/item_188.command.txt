/bin/bash -lc "python3 - <<'PY'
import requests,json
url='https://www.ensembl.org/data/graphql/core'
for name in ['Pan paniscus','Pan troglodytes','Macaca mulatta','Ailuropoda melanoleuca','Delphinapterus leucas']:
 q='''query { genomes(by_keyword: { scientific_name: \"%s\" }) { genome_id assembly_accession scientific_name release_number } }''' % name
 r=requests.post(url,json={'query':q},headers={'Content-Type':'application/json'}).json()
 print(name, json.dumps(r,indent=2)[:1500])
PY"
