/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('orthologs.json'))
taxes={}
for r in j['reports']:
    key=(r.get('taxname'),r.get('tax_id'))
    taxes.setdefault(key,0); taxes[key]+=1
for sub in ['Pan','Macaca','Ailuropoda','Delphinapterus','Indri','Semnopithecus','Monodelphis']:
    print('\\n--',sub)
    for (name,tid),cnt in taxes.items():
        if name and sub.lower() in name.lower(): print(name,tid,cnt)
PY"
