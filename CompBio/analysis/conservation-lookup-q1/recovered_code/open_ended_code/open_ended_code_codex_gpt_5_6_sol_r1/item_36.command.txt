/bin/bash -lc "python3 - <<'PY'
import requests,re
seq=''.join(x.strip() for x in open('human_target.fa') if not x.startswith('>'))
for name,tax,asm in [('indri',34827,'GCA_004363605.1'),('johnii',66063,'GCA_963574185.1')]:
 p={'CMD':'Put','PROGRAM':'blastn','DATABASE':f'genomic/{tax}/{asm}','QUERY':seq,'MEGABLAST':'on','HITLIST_SIZE':10,'BLAST_SPEC':'GDH_'+asm,'PAGE_TYPE':'BlastSearch','PROG_DEF':'blastn'}
 r=requests.post('https://blast.ncbi.nlm.nih.gov/Blast.cgi',data=p,timeout=60)
 rid=re.search(r'RID = ([A-Z0-9-]+)',r.text); rtoe=re.search(r'RTOE = (\\d+)',r.text)
 print(name,rid.group(1) if rid else None,rtoe.group(1) if rtoe else None)
 open('submit2_'+name+'.html','w').write(r.text)
PY"
