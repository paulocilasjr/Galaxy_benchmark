/bin/bash -lc "pwd && curl -s 'https://rest.ensembl.org/info/species?content-type=application/json' -o ensembl_species.json && python3 - <<'PY'
import json
x=json.load(open('ensembl_species.json'))
wanted=['homo_sapiens','pan_troglodytes','macaca_mulatta','pan_paniscus','indri_indri','semnopithecus_johnii','ailuropoda_melanoleuca','delphinapterus_leucas']
for w in wanted:
 s=[z for z in x['species'] if z['name']==w]
 print(w, s)
PY"
