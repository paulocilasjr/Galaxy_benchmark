/bin/bash -lc "python3 -u - <<'PY'
import requests,time
species=['pan_troglodytes','macaca_mulatta','pan_paniscus','ailuropoda_melanoleuca','delphinapterus_leucas']
for s in species:
 try:
  u=f'https://rest.ensembl.org/xrefs/symbol/{s}/SHH'
  x=requests.get(u,headers={'Content-Type':'application/json'},timeout=45)
  print(s,x.status_code,x.text[:1000],flush=True)
 except Exception as e: print(s,e,flush=True)
 time.sleep(1)
PY"
