/bin/bash -lc "curl -s 'https://ftp.ensembl.org/pub/rapid-release/species_metadata.json' -o rapid_species_metadata.json && python3 - <<'PY'

x=json.load(open('rapid_species_metadata.json'))
for s in x:
 if s.get('scientific_name') in ('Indri indri','Semnopithecus johnii') or 'indri' in str(s).lower() or 'johnii' in str(s).lower(): print(s)
PY"
