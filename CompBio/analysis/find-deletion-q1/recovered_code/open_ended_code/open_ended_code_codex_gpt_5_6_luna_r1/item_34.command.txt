/bin/bash -lc "printf 'Insert-size distribution for chr22 mapped pairs (absolute TLEN, one record per pair):\\n'; samtools view -F 0x904 -f 1 reads.sorted.bam chr22 | awk '"'$9!=0 && and($2,64){n[$9<0?- $9:$9]++} END{for(x in n)print x,n[x]}'"' | sort -n | head -n 30
printf '\\nLargest absolute TLEN values:\\n'; samtools view -F 0x904 -f 1 reads.sorted.bam chr22 | awk '"'$9!=0{a=$9<0?- $9:$9; if(a>1000)print $1,$2,$3,$4,$5,$6,$7,$8,$9}'"' | sort -k9,9nr | head -n 50
printf '\\nSoft-clipped or supplementary-like CIGAR records on chr22:\\n'; samtools view -F 4 reads.sorted.bam chr22 | awk '"'$6 !~ /''^150M$/{print $1,$2,$3,$4,$5,$6,$7,$8,$9}'"' | head -n 100
printf '\\nRead pairs with one end near upstream boundary and mate near downstream boundary:\\n'; samtools view -F 0x904 reads.sorted.bam chr22 | awk '("'$4<12910000 && $4>12850000 && $8>15000000) || ($4>15100000 && $4<15200000 && $8<13000000){print $1,$2,$3,$4,$5,$6,$7,$8,$9}'"' | head -n 100"
