/bin/bash -lc "python3 - <<'PY'
import csv
rows=[]
with open('chr22.covN10k.txt') as f:
    for line in f:
        parts=line.strip().split()
        if len(parts)<4: continue
        start=int(parts[0]); end=int(parts[1]); cov=float(parts[2]); n=int(parts[3])
        rows.append((start,end,cov,n))
# find zero runs with median N < 8000
runs=[]; i=0
while i<len(rows):
    if rows[i][2]==0:
        j=i
        while j<len(rows) and rows[j][2]==0: j+=1
        length=rows[j-1][1]-rows[i][0]
        total_n=sum(r[3] for r in rows[i:j])
        med_n=sorted(r[3] for r in rows[i:j])[len(rows[i:j])//2]
        if length>=20000:
            runs.append((rows[i][0], rows[j-1][1], length, j-i, total_n, med_n))
        i=j
    else: i+=1
for r in runs:
    print('run',r[0],r[1],'len',r[2],'windows',r[3],'Nsum',r[4],'medN',r[5])
PY"
