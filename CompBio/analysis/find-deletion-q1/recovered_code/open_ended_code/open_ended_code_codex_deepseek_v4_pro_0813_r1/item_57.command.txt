/bin/bash -c "python3 - <<'PY'
from collections import Counter
BIN=10000
bins=Counter()
with open('chr22_alns.tsv') as f:
    next(f)
    for line in f:
        p=line.rstrip().split('\\t')
        pos=int(p[2])
        bins[pos//BIN]+=1
maxkey=max(bins)
arr=[bins.get(i,0) for i in range(maxkey+1)]
print('bins nonzero',len(bins),'total',sum(arr),'maxkey',maxkey)
runs=[]; inz=False; start=0
for i,c in enumerate(arr):
    if c==0 and not inz: start=i; inz=True
    elif c>0 and inz: runs.append((start,i-1,i-start)); inz=False
if inz: runs.append((start,len(arr)-1,len(arr)-start))
for s,e,n in runs:
    if n>=5:
        print(f'zero {s*BIN:>10}-{(e+1)*BIN-1:>10} len {n*BIN} nbins {n}')
        lo=max(0,s-5); hi=min(len(arr)-1,e+5)
        for i in range(lo,hi+1): print(f'  {i*BIN:>10}: {arr[i]}')
PY"
