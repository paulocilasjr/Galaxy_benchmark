/bin/bash -c "python3 - <<'PY'
from collections import Counter, defaultdict
chrom=None
bins=defaultdict(int)
reading=False
with open('counts.tsv') as f:
    for line in f:
        if line.startswith('BINS'):
            reading=True; continue
        if not reading: continue
        c,b,count=line.rstrip().split('\\t')
        bins[(c,int(b))]=int(count)
print('chr22 bins', sum(1 for (c,b) in bins if c=='chr22'))
arr=[]
for b in range(0, max([x[1] for x in bins if x[0]=='chr22'], default=0)+100000, 100000):
    cnt=bins.get(('chr22',b),0)
    arr.append((b,cnt))
print('chr22 total count', sum(c for b,c in arr))
print('len',len(arr),'maxbin',arr[-1][0] if arr else 0)
# print all bins around low counts
low=[(b,c) for b,c in arr if c<20]
print('bins c<20:',len(low))
for b,c in low:
    print(b,c)
PY"
