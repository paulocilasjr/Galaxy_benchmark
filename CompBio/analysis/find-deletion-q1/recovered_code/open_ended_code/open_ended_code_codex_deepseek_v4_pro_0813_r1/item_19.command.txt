/bin/bash -c 'set -e
mkdir -p hg38_parts
base=896044570
end=983659423
chunks=12
chunk=$(((end-base+1)/chunks))
for i in $(seq 0 $((chunks-1))); do
  s=$((base+i*chunk))
  e=$((s+chunk-1))
  if [ $i -eq $((chunks-1)) ]; then e=$end; fi
  curl -L --fail --silent --show-error -r ${s}-${e} -o hg38_parts/part.${i} https://hgdownload2.soe.ucsc.edu/goldenPath/hg38/bigZips/hg38.fa.gz &
done
wait
for i in $(seq 0 $((chunks-1))); do cat hg38_parts/part.${i}; done >> hg38.fa.gz
ls -l hg38.fa.gz'
