/bin/bash -c "python3 - <<'PY'
from collections import defaultdict
bins=defaultdict(int)
reading=False
with open('counts.tsv') as f:
    for line in f:
        if line.startswith('BINS'):
            reading=True; continue
        if not reading: continue
        c,b,count=line.rstrip().split('\\t')
        if c=='chr22':
            bins[int(b)]=int(count)
maxb=50800000
arr=[bins.get(b,0) for b in range(0,maxb+1,100000)]
# print all bins with >0 or zero runs summary
runs=[]
in_zero=False; start=0
for i,c in enumerate(arr):
    if c==0 and not in_zero:
        start=i; in_zero=True
    elif c>0 and in_zero:
        runs.append((start,i-1,i-start))
        in_zero=False
if in_zero: runs.append((start,len(arr)-1,len(arr)-start))
print('Zero runs (start_bin,end_bin,nbins):')
for s,e,n in runs:
    print(f'{s*100000:>10}-{min(e*100000+99999,maxb):>10} n={n}')
print('Nonzero bins count',sum(c>0 for c in arr), 'max',max(arr),'mean',sum(arr)/len(arr))
# print counts around runs
for s,e,n in runs:
    lo=max(0,s-3); hi=min(len(arr)-1,e+3)
    print('\\nAround run',s*100000, 'n',n)
    for i in range(lo,hi+1):
        print(f'{i*100000:>10}: {arr[i]}')
PY"
