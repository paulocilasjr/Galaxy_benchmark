/bin/bash -c "python3 - <<'PY'
import gzip
for f in ['find.deletion.r1.fq.gz','find.deletion.r2.fq.gz']:
    n=0; lengths=[]
    with gzip.open(f,'rt') as fh:
        for i,line in enumerate(fh):
            if i%4==1:
                lengths.append(len(line.strip())); n+=1
                if n>=10000: break
    print(f, 'n',n,'minmax',min(lengths),max(lengths),'mean',sum(lengths)/n)
PY"
