/bin/bash -lc "python3 - <<'PY'
import json,glob,os
import numpy as np
q=np.load('q_coarse_counts.npy')
s=json.load(open('q_profile_samples.json'))
tot=q.sum(0)
X=np.log1p(q/tot*1e6)
# center per vector
for f in glob.glob('bw_coarse_*.npy'):
 tr=f[len('bw_coarse_'):-4]
 y=np.log1p(np.load(f))
 # restrict variable regions? all
 cor=np.corrcoef(np.column_stack([X,y]).T)[:-1,-1]
 order=np.argsort(cor)[::-1]
 print(tr, ' '.join(f'{s[i]}={cor[i]:.3f}' for i in order[:5]))
PY"
