x <- readRDS('proj_Axolotl.rds')
cat('attrs names:',paste(names(attributes(x)),collapse=','),'\\n')

print(head(cc[,c('Sample','cell_type','subanno','lineage')]))
for(n in c('Sample','cell_type','subanno','lineage','predictedGroup_Un')) if(n %in% names(cc)){cat('TABLE',n,'\\n');print(table(cc[[n]]))}

RS"
