/bin/bash -lc "python3 - <<'PY'
import gzip,collections,numpy as np,re
# summarize per matrix chrom and sample normalized by total
sums=collections.defaultdict(lambda:np.zeros(15,dtype=np.int64)); n=collections.Counter()
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
 names=f.readline().rstrip().split('\\t')[3:]
 for l in f:
  x=l.rstrip().split('\\t'); c=x[0]; sums[c]+=np.array(x[3:],dtype=np.int64);n[c]+=1
lib=sum(sums.values())
for c in sorted(sums, key=lambda x:(int(re.match(r'chr(\\d+)',x).group(1)),x)):
 z=sums[c]/lib*1e6
 print(c,n[c],' '.join(f'{v:.0f}' for v in z))
PY"
