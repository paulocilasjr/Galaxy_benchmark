/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,re
url='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true&expand=true'
all=[]
for page in range(10):
 req=urllib.request.Request(url)
 with urllib.request.urlopen(req) as r:
  data=json.load(r); link=r.headers.get('Link','')
 all.extend(data)
 m=re.search(r'<(["'^>]+)>;'"\\s*rel=\"next\"',link)
 print(page,len(data),bool(m))
 if not m: break
 url=m.group(1)
json.dump(all,open('hf_tree_all.json','w'))
print('total',len(all))
for z in all:
 p=z.get('path','')
 if 'sample.swap' in p.lower() or 'answer' in p.lower() or 'chrom.sizes' in p.lower(): print(p,z.get('size'))
PY"
