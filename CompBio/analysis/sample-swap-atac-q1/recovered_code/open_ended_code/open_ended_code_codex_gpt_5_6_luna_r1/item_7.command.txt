/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, time
p='sample.swap.atac.q1.tsv.gz'
t0=time.time()
with gzip.open(p,'rt') as f:
    names=f.readline().rstrip().split('\\t')[3:]
    a=np.loadtxt(f,delimiter='\\t',usecols=range(3,18),dtype=np.float32)
print('loaded',a.shape,'sec',round(time.time()-t0,1),flush=True)
for label,x in [('raw',a),('log1p',np.log1p(a))]:
    c=np.corrcoef(x,rowvar=False)
    print(label,flush=True)
    iu=np.triu_indices(len(names),1)
    pairs=sorted(((float(c[i,j]),names[i],names[j]) for i,j in zip(*iu)), reverse=True)
    print(pairs[:20],flush=True)
PY"
