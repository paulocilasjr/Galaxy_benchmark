/bin/bash -lc 'mkdir -p gtf_parts
for spec in 0:9999999 10000000:19999999 20000000:29999999 30000000:39999999 40000000:49999999 50000000:51302806
do
  start=${spec%%:*}
  end=${spec##*:}
  curl -L --fail --silent --show-error -r "$start-$end" -o "gtf_parts/$start-$end" '"'https://www.axolotl-omics.org/dl/AmexT_v47-AmexG_v6.0-DD.gtf.gz'
done
cat gtf_parts/* > AmexT_v47-AmexG_v6.0-DD.clean.gtf.gz
ls -lh AmexT_v47-AmexG_v6.0-DD.clean.gtf.gz
gzip -t AmexT_v47-AmexG_v6.0-DD.clean.gtf.gz
echo gzip_status="'$?'
