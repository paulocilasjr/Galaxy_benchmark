/bin/bash -lc "python3 - <<'PY'
import csv, numpy as np
D={}; s=[]
for r in csv.reader(open('promoter_scores.tsv'),delimiter='\\t'):
 if not s:s=r[1:];continue
 D[r[0]]=np.array(list(map(float,r[1:])))
for g in ['GKN1','MUC5AC','TFF1','TFF2','TFF3.2','ATP4A','ATP4B','GIF','PGC','GAST','MUC2','FABP1','FABP2','SLC15A1','ALPI','CDX2','SATB2','CLDN18','CHIA','MUC5B','CCL19','TFF3','KRT19','KRT7','KRT8','KRT18','SOX2']:
 if g in D:
  v=D[g]/D[g].sum(); o=np.argsort(v)[::-1][:7]
  print(f'{g:10s}', ' '.join(f'{s[i]}={v[i]:.3f}' for i in o))
 else: print(g,'NA')
PY"
