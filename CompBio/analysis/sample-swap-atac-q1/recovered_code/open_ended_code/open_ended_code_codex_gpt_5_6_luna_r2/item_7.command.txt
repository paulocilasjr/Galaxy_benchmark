/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np, time
p='sample.swap.atac.q1.tsv.gz'
t0=time.time()
head=pd.read_csv(p,sep='\\t',nrows=3)
cols=list(head.columns[3:])
print('cols', cols)
print('start')
tot=np.zeros(len(cols),dtype=np.int64)
row_nonzero=np.zeros(len(cols),dtype=np.int64)
row_sum=np.zeros(len(cols),dtype=np.float64)
row_sumsq=np.zeros(len(cols),dtype=np.float64)
# get counts, totals only
n=0
for ch in pd.read_csv(p,sep='\\t',usecols=range(3,18),chunksize=200000):
    x=ch.to_numpy(dtype=np.int64,copy=False)
    tot += x.sum(0)
    row_nonzero += (x>0).sum(0)
    row_sum += x.sum(0)
    row_sumsq += (x*x).sum(0)
    n += len(x)
print('n',n,'sec',time.time()-t0)
for c,a,b,d in zip(cols,tot,row_nonzero,row_sumsq): print(c, 'total',int(a),'nonzero',int(b),'mean',a/n,'sd',np.sqrt(d/n-(a/n)**2))
PY"
