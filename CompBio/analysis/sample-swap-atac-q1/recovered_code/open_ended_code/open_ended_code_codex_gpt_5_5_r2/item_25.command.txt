/bin/bash -lc "python3 - <<'PY'
import urllib.request, re
for acc in ['GSM8426940','GSM8426943','GSM8426949','GSM8426963','GSM8426967','GSM8426973']:
    url=f'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc={acc}'
    html=urllib.request.urlopen(url, timeout=30).read().decode('utf-8','ignore')
    print('ACC',acc)
    for m in re.findall(r'https?://["'^"<> ]+'"', html):
        if 'geo/download' in m or 'ftp.ncbi' in m:
            print(m[:300])
    # also href around Supplementary
    for line in html.splitlines():
        if 'download' in line.lower() and ('.bed.gz' in line or 'format=file' in line):
            print(line.strip()[:500])
PY"
