/bin/bash -lc "python3 - <<'PY'
import gzip
import numpy as np
cols=None; n=0
sums=None; nz=None
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    cols=header[3:]
    sums=np.zeros(len(cols), dtype=float); nz=np.zeros(len(cols), dtype=int)
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        vals=np.array(list(map(float,parts[3:])))
        sums += vals; nz += vals>0; n += 1
print(n, cols)
for c,s,z in zip(cols,sums,nz):
    print(c, int(s), z)
PY"
