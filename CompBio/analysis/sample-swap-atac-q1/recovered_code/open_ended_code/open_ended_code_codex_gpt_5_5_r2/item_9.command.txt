/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
header=None; X=[]
# stream compute sums for log1p normalized counts maybe load float32 2.4M x15=145MB OK
rows=[]
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
    header=f.readline().strip().split('\\t')
    cols=header[3:]
    for i,line in enumerate(f):
        vals=np.fromstring('\\t'.join(line.rstrip('\\n').split('\\t')[3:]), sep='\\t', dtype=np.float32)
        rows.append(vals)
        # if i>100000: break
X=np.vstack(rows)
print('shape', X.shape)
lib=X.sum(axis=0)
Y=np.log1p(X/lib*1e6)
C=np.corrcoef(Y.T)
print('Pearson logCPM top pairs')
pairs=[]
for i in range(len(cols)):
 for j in range(i+1,len(cols)):
  pairs.append((C[i,j],cols[i],cols[j]))
for r,a,b in sorted(pairs, reverse=True)[:30]: print(f'{r:.4f} {a} {b}')
print('bottom')
for r,a,b in sorted(pairs)[:20]: print(f'{r:.4f} {a} {b}')
# Euclidean/correlation dendro maybe pca projections
# highly variable bins
var=Y.var(axis=1)
idx=np.argsort(var)[-50000:]
C2=np.corrcoef(Y[idx].T)
print('top HV pairs')
pairs=[]
for i in range(len(cols)):
 for j in range(i+1,len(cols)):
  pairs.append((C2[i,j],cols[i],cols[j]))
for r,a,b in sorted(pairs, reverse=True)[:30]: print(f'{r:.4f} {a} {b}')
print('bottom HV')
for r,a,b in sorted(pairs)[:20]: print(f'{r:.4f} {a} {b}')
PY"
