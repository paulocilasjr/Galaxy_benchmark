/bin/bash -lc "python3 - <<'PY'
import gzip, subprocess, urllib.parse
from collections import defaultdict
limit=10000000
cols=None; local_bins=[]; local_counts={}
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
    cols=f.readline().rstrip('\\n').split('\\t')[3:]
    want={'Cloaca','Stomach'}
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if p[0]"'!='"'chr1p_1': break
        st=int(p[1]); en=int(p[2])
        if st>=limit: break
        b=(st,en); local_bins.append(b)
        vals=list(map(int,p[3:]))
        for c,v in zip(cols,vals):
            if c in want: local_counts.setdefault(c,{})[b]=v

def stream(acc,fname):
    url='https://www.ncbi.nlm.nih.gov/geo/download/?acc=%s&format=file&file=%s'%(acc,urllib.parse.quote(fname))
    proc=subprocess.Popen(['bash','-lc',f\"curl -L --fail --silent {url"'!r} | gzip -dc"],stdout=subprocess.PIPE,text=True)
    counts=defaultdict(int)
    for line in proc.stdout:
        p=line.rstrip('"'\\n').split('\\t')
        if len(p)<3: continue
        chrom=p[0]; st=int(p[1]); en=int(p[2])
        if chrom"'!='"'chr1p_1' or st>=limit: break
        s=(st//10000)*10000; e=((en-1)//10000)*10000
        for x in range(s,min(e,limit-1)+1,10000): counts[(x,x+10000)]+=1
    proc.kill()
    return counts
for name,acc,fname in [
    ('source_Cloaca','GSM8426946','GSM8426946_Axolotl_Cloaca_Fragments.bed.gz'),
    ('source_Stomach','GSM8426973','GSM8426973_Axolotl_Stomach_Fragments.bed.gz')]:
    sc=stream(acc,fname)
    print(name, 'bins', len(local_bins))
    for col in ['Cloaca','Stomach']:
        mism=sum(sc.get(b,0)"'!=local_counts[col][b] for b in local_bins)
        absdiff=sum(abs(sc.get(b,0)-local_counts[col][b]) for b in local_bins)
        print('"'  vs',col,'mism',mism,'abs',absdiff)
PY"
