/bin/bash -lc "python3 - <<'PY'
import numpy as np
D=np.load('tmp/all_promoter_activity.npz',allow_pickle=True)
syms=list(D['syms']); idx={s:i for i,s in enumerate(syms)}
samples=list(D['samples']); libs=D['libs']; act=D['sym_act']
x=np.log1p(act/libs*1e6)
std=x.std(axis=1); z=np.zeros_like(x); mask=std>0; z[mask]=(x[mask]-x[mask].mean(axis=1,keepdims=True))/std[mask,None]
sets={
'Stomach': ['GKN1','GKN2','MUC5AC','CLDN18','VSIG1','TFF3','TFF3.2','PGC','SOX2','BARX1','MUC6','GIF'],
'IntestineColonCloaca':['MUC2','SATB2','CDX2','FABP2','APOA4','VIL1','SLC15A1','MGAM','PRAP1','CDHR2','SLC5A1','ENPP7','TMPRSS15','MUC3A','KRT13','TP63','PCK1'],
'GallBladderBiliary':['KRT19','EPCAM','SOX17','CFTR','GGT1','MUC1','KRT7','KRT8','KRT18','SLC4A2','MUC5B'],
'BladderUro':['UPK1A','UPK1B','UPK2','UPK3A','UPK3B','KRT13','TP63','KRT5','KRT20','FOXA1'],
'Liver':['ALB','FGA','FGB','FGG','APOA1','APOB','TTR','TF-A','TF','FETUB','HPX','AGT','ARG1'],
'Pancreas':['PRSS1','PRSS2','PRSS3','CTRB1','CTRB2','CELA1','CELA2A','CPA1','CPB1','CTRL','AMY2A','INS','PTF1A','PNLIP'],
}
for name,genes in sets.items():
    present=[g for g in genes if g in idx]
    mat=np.vstack([z[idx[g]] for g in present])
    score=mat.mean(axis=0)
    print('\\nSET',name,'present',present)
    for s,val in sorted(zip(samples,score), key=lambda x:-x[1]):
        print(f'{s}:{val:.2f}',end=' ')
    print('\\nGene ranks/top for key genes:')
    for g in present:
        vals=z[idx[g]]
        top=sorted(zip(samples,vals,act[idx[g]]/libs*1e6), key=lambda x:-x[1])[:5]
        print(g, ' '.join(f'{s}:{v:.2f}/{c:.1f}' for s,v,c in top))
PY"
