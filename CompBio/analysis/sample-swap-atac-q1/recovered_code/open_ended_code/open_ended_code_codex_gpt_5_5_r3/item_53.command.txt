/bin/bash -lc "python3 - <<'PY'
import urllib.request,re,time
samples={
'Bladder':'GSM8426940','Brain':'GSM8426943','Cloaca':'GSM8426946','GallBladder':'GSM8426949','Gill':'GSM8426952','Heart':'GSM8426953','Intestine':'GSM8426954','Kidney':'GSM8426957','Limb':'GSM8426960','Liver':'GSM8426963','Lung':'GSM8426966','Pancreas':'GSM8426967','Spleen':'GSM8426970','Stomach':'GSM8426973','Prostate':'GSM9059487'}
for tissue,gsm in samples.items():
    url=f'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc={gsm}'
    try:
        txt=urllib.request.urlopen(url,timeout=20).read().decode('utf-8','replace')
        # find Supplementary file line
        m=re.search(r'(GSM\\d+_["'^<]*?Fragments'"\\.bed\\.gz)\\s*([0-9.]+)\\s*([GMK]b)',txt)
        title=re.search(r'Title.*?UUATAC-seq datasets of (["'^<'"\\n]+)',txt)
        print(tissue,gsm, title.group(1).strip() if title else '?', m.groups() if m else 'NOFILE')
    except Exception as e: print('ERR',tissue,e)
PY"
