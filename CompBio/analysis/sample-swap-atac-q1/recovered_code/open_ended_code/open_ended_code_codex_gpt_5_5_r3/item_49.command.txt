/bin/bash -lc "python3 - <<'PY'
import json, glob, re, os
terms=['GKN1','GKN2','CLDN18','VSIG1','TFF3','MUC2','SATB2','PCK1','FABP2','VIL1','APOA4','CFTR','SOX2','BARX1','MUC5AC','PGC']
for fn in sorted(glob.glob('tmp/DCA_top_markers_WT_*.json')):
    tissue=fn.split('DCA_top_markers_')[1].split('.json')[0]
    data=json.load(open(fn))['data']
    hits=[]
    for row in data:
        gene=row['gene'].upper()
        for t in terms:
            if re.search(r'("'^|[-_.])'"'+re.escape(t)+r'("'$|[-_.])'"',gene) or gene.endswith('-'+t):
                hits.append((t,row['avg_diff'],row['pct_1'],row['pct_2'],row['alias'],row['gene']))
    if hits:
        print('\\n',tissue)
        for h in hits[:20]: print(h)
PY"
