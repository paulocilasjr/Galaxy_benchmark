/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
url='https://bis.zju.edu.cn/ACA/call/mca_stats.php'
try:
    req=urllib.request.Request(url, data=b'', headers={'User-Agent':'Mozilla/5.0'})
    data=urllib.request.urlopen(req,timeout=20).read()
    print(len(data), data[:1000])
    open('tmp/mca_stats.json','wb').write(data)
except Exception as e: print('ERR',e)
PY
python3 - <<'PY'
from pathlib import Path
p=Path('tmp/mca_stats.json')
if p.exists():
    txt=p.read_text(errors='replace')
    for term in ['Cloaca','Gall','Limb','Bladder','tissue']:
        print(term, txt.find(term))
    print(txt[:2000])
PY"
