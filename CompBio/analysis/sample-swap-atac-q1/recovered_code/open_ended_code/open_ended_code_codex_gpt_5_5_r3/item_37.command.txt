import urllib.request,re
url='https://bis.zju.edu.cn/ACA/dpline.html?tissue=WT_Stomach'
text=urllib.request.urlopen(url).read().decode('utf-8','replace')
open('tmp/WT_Stomach.html','w').write(text)
for pat in ['dpline','json','csv','marker','ajax','tissue']:
    print('\\nPAT',pat)
    for m in re.finditer(pat,text,re.I):
        print(m.start(), text[max(0,m.start()-120):m.start()+200].replace('\\n',' ')[:350])
        break
print('len',len(text))
print('\\n'.join(re.findall(r'<script["'^>]+src="([''^"]+)"'"',text)))
print('\\nlinks')
print('\\n'.join(re.findall(r'href=\"(["'^"]+)"'"',text)[:50]))
PY
