# Manually use parsed rows above excluding missing prostate for swap depth diagnostics
import numpy as np
rows=[('Bladder',7361379926,701506327),('Brain',12048619808,1168148790),('Cloaca',4079467003,484000505),('GallBladder',4980611273,565994637),('Gill',11566811496,1094125058),('Heart',9197722750,868549497),('Intestine',6322122007,699794434),('Kidney',7755800716,745602917),('Limb',11646072827,1147926115),('Liver',3644606637,398317869),('Lung',9346073087,877686879),('Pancreas',12518564809,1398702406),('Spleen',5192026012,497829833),('Stomach',4355106294,455786546)]
