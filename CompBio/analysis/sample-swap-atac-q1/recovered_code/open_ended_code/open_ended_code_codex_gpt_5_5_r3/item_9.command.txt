/bin/bash -lc "python3 - <<'PY'
import gzip, csv, math, statistics
from collections import defaultdict
path='sample.swap.atac.q1.tsv.gz'
with gzip.open(path,'rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    samples=header[3:]
    sums=[0]*len(samples); nnz=[0]*len(samples); n=0
    chrom_sums=defaultdict(lambda:[0]*len(samples))
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        vals=list(map(int,parts[3:]))
        n+=1
        c=parts[0]
        cs=chrom_sums[c]
        for i,v in enumerate(vals):
            sums[i]+=v
            cs[i]+=v
            if v: nnz[i]+=1
print('n',n,'samples',samples)
for s,total,nonz in zip(samples,sums,nnz):
    print(s,total,nonz,total/n)
print('chroms',len(chrom_sums))
PY"
