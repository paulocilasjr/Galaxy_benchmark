/bin/bash -lc "gzip -cd sample.swap.atac.q1.tsv.gz | wc -l
python3 - <<'PY'
import pandas as pd, numpy as np
p='sample.swap.atac.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t')
X=df.iloc[:,3:]
print('shape',df.shape)
print('totals')
print(X.sum().to_string())
print('nonzero_fraction')
print((X>0).mean().to_string())
print('pearson log1p')
print(np.log1p(X).corr().round(3).to_string())
print('spearman raw')
print(X.corr(method='spearman').round(3).to_string())
PY"
