/bin/bash -lc "python3 - <<'PY'
import numpy as np
d=np.load('matrix_data.npz',allow_pickle=True); X=d['X']; names=list(d['names'])
ct=np.load('celltrack_data.npz',allow_pickle=True); A=ct['celltrack']; celltypes=list(ct['celltypes']); samples=list(ct['samples']); C=ct['C']; F=ct['F']
# Pearson between each X column and each celltype column
Xc=X-X.mean(axis=0,keepdims=True)
Ac=A-A.mean(axis=0,keepdims=True)
denX=np.sqrt((Xc**2).sum(axis=0)); denA=np.sqrt((Ac**2).sum(axis=0))
P=(Xc.T@Ac)/np.outer(denX,denA)
print('P shape',P.shape)
# compare rows P to ref C/F by cosine and correlation
for ref,label in [(C,'count'),(F,'frag')]:
    # maybe P has positive/negative; use cosine
    cos=(P@ref.T)/(np.linalg.norm(P,axis=1)[:,None]*np.linalg.norm(ref,axis=1)[None,:]+1e-12)
    print('\\nPearson celltype cosine',label)
    for i,s in enumerate(names):
        order=np.argsort(cos[i])[::-1]
        print(f'{s:12s} best {samples[order[0]]:12s} {cos[i,order[0]]:.3f} 2nd {samples[order[1]]:12s} {cos[i,order[1]]:.3f} diag {cos[i,i]:.3f}')
    np.save(f'pearson_cos_{label}.npy',cos)
PY"
