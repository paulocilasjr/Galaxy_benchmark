/bin/bash -lc "R --vanilla --quiet <<'RS'
library(GenomicRanges)
x <- readRDS('proj_Axolotl.rds')
ga <- attr(x,'geneAnnotation')
genes <- ga"'$genes
tss <- ga$TSS
# associate TSS to genes by order
df <- data.frame(gene=genes$symbol, chrom=as.character(seqnames(tss)), pos=start(tss), strand=as.character(strand(tss)), stringsAsFactors=FALSE)
write.table(df, '"'gene_tss.tsv', sep='\\t', row.names=FALSE, col.names=TRUE, quote=FALSE)
cat('wrote',nrow(df),'\\n')
RS
head gene_tss.tsv"
