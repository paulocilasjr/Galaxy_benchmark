/bin/bash -lc "python3 - <<'PY'
import collections, csv
cell=collections.Counter(); sample_cell=collections.defaultdict(collections.Counter)
with open('cell_meta.tsv.gz') as f:
 r=csv.DictReader(f,delimiter='\\t')
 for row in r:
  s=row['sample']; c=row['cell_type']
  cell[c]+=1; sample_cell[s][c]+=1
print('cell types',len(cell))
for c,v in cell.most_common(): print(v,c)
print('samples',len(sample_cell))
PY"
