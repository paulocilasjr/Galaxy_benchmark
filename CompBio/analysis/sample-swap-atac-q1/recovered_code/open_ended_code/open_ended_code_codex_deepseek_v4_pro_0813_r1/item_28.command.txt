/bin/bash -lc "R --vanilla --quiet <<'RS'
x <- readRDS('proj_Axolotl.rds')
cc <- attr(x,'cellColData')
df <- as.data.frame(cc[,c('Sample','nFrags','nMonoFrags','nDiFrags','ReadsInPeaks','ReadsInTSS')])
# Sample is Rle maybe as.factor
for(s in names(attr(x,'sampleColData'))){
 sub <- df[df"'$Sample==s,]
 cat(s, '"'cells', nrow(sub), 'sumNFrags', sum(sub"'$nFrags), '"'sumMono', sum(sub"'$nMonoFrags), '"'sumDi', sum(sub"'$nDiFrags), '"'sumPeaks', sum(sub"'$ReadsInPeaks), '"'sumTSS', sum(sub"'$ReadsInTSS), '"'\\n')
}
RS"
