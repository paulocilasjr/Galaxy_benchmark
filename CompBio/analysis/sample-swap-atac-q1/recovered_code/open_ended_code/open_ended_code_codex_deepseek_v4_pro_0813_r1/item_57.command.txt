/bin/bash -lc "python3 - <<'PY'
import numpy as np
D=np.load('deconv_coef.npz',allow_pickle=True)
coef=D['coef']; coef_norm=D['coef_norm']; names=list(D['names']); samples=list(D['samples']); C=D['C']; F=D['F']; celltypes=list(D['celltypes'])
# align samples order same as names likely
print('names',names)
print('sample ref order',samples)
# compute cosine and L1 between coef_norm (rows samples) and each reference comp (rows tissues)
for ref,label in [(C,'count'),(F,'frag')]:
 cos=(coef_norm @ ref.T)/(np.linalg.norm(coef_norm,axis=1)[:,None]*np.linalg.norm(ref,axis=1)[None,:]+1e-12)
 print('\\nCOSINE',label)
 print('input\\t'+'\\t'.join(samples))
 for i,s in enumerate(names):
  print(s+'\\t'+'\\t'.join(f'{cos[i,j]:.3f}' for j in range(len(samples))))
 print('Argmax per input')
 for i,s in enumerate(names):
  order=np.argsort(cos[i])[::-1]
  print(f'{s:12s} best {samples[order[0]]:12s} {cos[i,order[0]]:.3f} 2nd {samples[order[1]]:12s} {cos[i,order[1]]:.3f}')
 np.save(f'deconv_cos_{label}.npy',cos)
PY"
