/bin/bash -lc "cat > chromstats.awk <<'AWK'
NR==1{for(i=4;i<=NF;i++) name[i]="'$i}
NR>1{
 chrom=$1; n[chrom]++;
 for(i=4;i<=NF;i++){
   key=chrom SUBSEP i;
   sum[key]+=$i;
   if($i==0) zero[key]++;
 }
}
END{
 for(c in n){
   printf "%s n=%d", c, n[c];
   for(i=4;i<=NF;i++){
     key=c SUBSEP i;
     printf " %s_zero=%d_avg=%.1f", name[i], zero[key]+0, sum[key]/n[c];
   }
   printf "'"\\n\";
 }
}
AWK
gzip -cd sample.swap.atac.q1.tsv.gz | awk -f chromstats.awk | sort -k1,1V"
