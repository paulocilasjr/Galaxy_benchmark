/bin/bash -lc "python3 - <<'PY'
import numpy as np
D=np.load('promoter_matrix.npz',allow_pickle=True); G=D['G']; genes=list(D['genes']); names=list(D['names'])
Md=np.load('matrix_data.npz',allow_pickle=True); totals=Md['X'].sum(axis=0)
from collections import defaultdict
idx=defaultdict(list)
for i,g in enumerate(genes): idx[g].append(i)
pats=['CPA1','GCG','SST','PDX1','INS','AMY2A']
for p in pats:
 inds=[i for i,g in enumerate(genes) if p in g]
 if not inds: continue
 print('\\n',p, [genes[i] for i in inds[:10]])
 for i in inds[:3]:
  vals=G[i]/totals*1e6
  order=np.argsort(vals)[::-1]
  print(' row',genes[i], 'top', ', '.join(f'{names[j]}:{vals[j]:.1f}' for j in order[:5]))
PY"
