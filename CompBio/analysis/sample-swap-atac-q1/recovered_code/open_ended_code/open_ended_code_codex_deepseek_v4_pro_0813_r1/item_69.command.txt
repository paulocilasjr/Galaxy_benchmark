/bin/bash -lc "python3 - <<'PY'
import requests, tarfile, os, io
url='https://www.ncbi.nlm.nih.gov/geo/download/?acc=GSE182746&format=file'
print('downloading',url)
r=requests.get(url,stream=True,timeout=120)
r.raise_for_status()
total=int(r.headers.get('content-length',0)); print('total',total)
with open('GSE182746_RAW.tar','wb') as f:
 done=0
 for chunk in r.iter_content(chunk_size=1<<20):
  f.write(chunk); done+=len(chunk)
  if done%(50<<20)==0: print(done)
print('done',done,os.path.getsize('GSE182746_RAW.tar'))
PY"
