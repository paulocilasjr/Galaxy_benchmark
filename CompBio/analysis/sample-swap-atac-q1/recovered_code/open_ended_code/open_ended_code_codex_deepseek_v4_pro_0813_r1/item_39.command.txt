/bin/bash -lc "cat > enrichment.py <<'PY'
import numpy as np
d=np.load('matrix_data.npz',allow_pickle=True)
X=d['X']; ref=d['ref']; names=list(d['names'])
# normalize columns to total 1
Xn=X/X.sum(axis=0,keepdims=True)
n=Xn.shape[0]
for binary in [True, False]:
    W=(ref>0).astype(float) if binary else ref.copy()
    Wsum=W.sum(axis=0)
    base=Xn.sum(axis=0)/n  # same for all columns ~1/n
    # enrichment per sample, tissue
    En=(Xn.T@W)/(Wsum[None,:])
    # divide by global mean per sample? use fold over overall avg
    fold=En/base[:,None]
    print('\\nBinary' if binary else '\\nWeighted')
    print('sample\\t'+'\\t'.join(names))
    for i,s in enumerate(names):
        print(s+'\\t'+'\\t'.join(f'{fold[i,j]:.4f}' for j in range(len(names))))
    print('\\nArgmax per sample')
    for i,s in enumerate(names):
        order=np.argsort(fold[i])[::-1]
        print(f'{s:12s} best {names[order[0]]:12s} {fold[i,order[0]]:.3f} second {names[order[1]]:12s} {fold[i,order[1]]:.3f}')
    # save binary fold
    if binary: np.save('enrich_binary.npy',fold)
PY
python3 enrichment.py"
