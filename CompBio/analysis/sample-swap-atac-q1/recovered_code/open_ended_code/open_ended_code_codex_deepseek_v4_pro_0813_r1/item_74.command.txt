import gzip, glob, os, re, numpy as np, collections
# Parse RNA featurecount files for neotenic only
files=glob.glob('rna/GSM*neotenic*featurecounts.txt.gz')
tissue_files=collections.defaultdict(list)
for f in files:
    base=os.path.basename(f)
    m=re.match(r'GSM\\d+_B\\d+_neotenic_(.+)_featurecounts\\.txt\\.gz',base)
    if m:
        tissue=m.group(1).replace('_',' ')
        tissue_files[tissue].append(f)
# tissue canonical mapping
canon={'Cloacal chamber':'Cloaca','Fore limb':'Limb','Hind limb':'Limb'}
tissues=[]
for t in sorted(tissue_files):
    c=canon.get(t,t)
    if c not in tissues: tissues.append(c)
print('RNA tissues',tissues)
# Read all counts, sum batches per tissue and duplicate for limb? combine fore/hind average?
# Store counts per file keyed by canonical; combine sum for all files in canonical group
counts=collections.defaultdict(dict); genes_order=None
for t,files in tissue_files.items():

    for f in files:
        with gzip.open(f,'rt') as fh:
            for line in fh:
                if line.startswith('#'): continue
                parts=line.rstrip('\\n').split('\\t')
                if len(parts)<7: continue
                if parts[0]=='Geneid': continue
                geneid=parts[0]
                sym=geneid.split('_')[-1].upper()

                counts[c][sym]=counts[c].get(sym,0)+count

allgenes=sorted(set(g for c in counts.values() for g in c))
