/bin/bash -lc "python3 - <<'PY'
import numpy as np, re
RNA=np.load('rna_matrix.npz',allow_pickle=True); Z=RNA['Z']; rna_genes=list(RNA['genes']); rna_tissues=list(RNA['tissues'])
PM=np.load('promoter_matrix.npz',allow_pickle=True); G=PM['G']; genes=list(PM['genes']); names=list(PM['names'])
Md=np.load('matrix_data.npz',allow_pickle=True); totals=Md['X'].sum(axis=0)
# maps
from collections import defaultdict
p_idx=defaultdict(list)
for i,g in enumerate(genes):
 base=g.split('[')[0].upper(); p_idx[base].append(i)
 for part in g.split('|'):
  s=part.split('[')[0].strip().upper(); p_idx[s].append(i)
# top genes for Cloaca and Stomach from RNA
for tissue in ['Cloaca','Stomach']:
 ti=rna_tissues.index(tissue); order=np.argsort(Z[ti])[::-1]
 print('\\n',tissue,'top matched promoter genes')
 for gi in order[:20]:
  sym=rna_genes[gi]
  inds=p_idx.get(sym,[])
  if inds:
   vals=G[inds[0]]/totals*1e6
   order2=np.argsort(vals)[::-1]
   print(f'{sym:20s}', ', '.join(f'{names[j]}:{vals[j]:.1f}' for j in order2[:8]))
PY"
