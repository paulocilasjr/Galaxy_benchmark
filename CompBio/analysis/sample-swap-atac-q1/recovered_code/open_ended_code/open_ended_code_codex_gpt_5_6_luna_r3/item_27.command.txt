/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
f='sample.swap.atac.q1.tsv.gz'
df=pd.read_csv(f, sep='\\t')
x=df.iloc[:,3:].to_numpy(dtype=np.int64)
names=df.columns[3:].tolist()
print('quantiles')
for j,n in enumerate(names):
 a=x[:,j]
 print(n, 'sum',a.sum(),'mean',round(a.mean(),2),'sd',round(a.std(),2),'q',np.quantile(a,[0,.25,.5,.75,.9,.95,.99,.999]).astype(int).tolist(),'zeros',round(np.mean(a==0),3),'max',a.max())
# normalized log with CPM
z=np.log1p(x/x.sum(axis=0)*1e8)
c=np.corrcoef(z.T)
for i,n in enumerate(names):
 ix=np.argsort(-c[i]); print(n, ' '.join(f'{names[k]}:{c[i,k]:.3f}' for k in ix[1:5]))
from sklearn.decomposition import PCA
pca=PCA(n_components=5).fit(z[::10])
p=pca.transform(z[::10])
print('PCA explained',pca.explained_variance_ratio_)
for n,row in zip(names,p): print(n, ' '.join(f'{v:.2f}' for v in row))
PY"
