/bin/bash -lc "cat > scan_promoter.py <<'PY'
import gzip, csv, pickle, time
import numpy as np
with open('keymap.pkl','rb') as f: keymap=pickle.load(f)
with open('gene_info.pkl','rb') as f: info=pickle.load(f)
G=len(info['gene_names'])
names=info['gene_names']
acc=np.zeros((G,15),dtype=np.float64)
n=0
t0=time.time()
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t')
    header=next(r)
    samples=header[3:]
    for row in r:
        key=(row[0],int(row[1]))
        inds=keymap.get(key)
        if inds:
            vals=np.array([int(x) for x in row[3:]],dtype=np.float64)
            acc[inds]+=vals
        n+=1
        if n%1000000==0:
            print('rows',n,'hits',int((acc.sum(axis=1)>0).sum()),'elapsed',time.time()-t0, flush=True)
print('done rows',n,'nonzero genes',int((acc.sum(axis=1)>0).sum()))
np.save('promoter_acc.npy',acc)
with open('samples.txt','w') as f:
    f.write('\\n'.join(samples))
# save info
import json
with open('gene_names.json','w') as f: json.dump(names,f)
print('saved')
PY
python3 scan_promoter.py"
