/bin/bash -lc "python3 - <<'PY'
import pandas as pd,pickle,json,numpy as np,gzip
from scipy.stats import zscore
cell=pd.read_csv('axolotl_cellColData.csv',index_col=0)
samples=['Bladder','Brain','Cloaca','GallBladder','Gill','Heart','Intestine','Kidney','Limb','Liver','Lung','Pancreas','Prostate','Spleen','Stomach']
lineages=sorted(cell['subanno'].unique())
counts=cell.groupby(['Sample','subanno']).size().unstack(fill_value=0).reindex(index=samples,columns=lineages,fill_value=0)
props=counts.div(counts.sum(axis=1),axis=0).values
d=pickle.load(open('cluster_exp_dict.pkl','rb')); gi=json.load(open('selected_gene_indices.json')); names=list(gi.keys())
acc=np.load('promoter_acc.npy')
tot=np.zeros(15)
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
 f.readline()
 for line in f:
  p=line.split('\\t'); tot+=np.array([int(x) for x in p[3:]],float)
P=np.zeros((len(names),15))
for i,n in enumerate(names): P[i]=acc[gi[n]].sum(axis=0)
Pnorm=np.log2(P/tot*1e6+1)
E=np.zeros_like(Pnorm)
for i,n in enumerate(names):
 arr=np.array([d[n].get(L,0) for L in lineages]); E[i]=props.dot(arr)
# specificity by max z within row
for tissue in ['Stomach','Cloaca','Intestine','Liver']:
 ti=samples.index(tissue)
 order=[]
 for i,n in enumerate(names):
  # score = E zscore across tissues at tissue
  z=(E[i]-E[i].mean())/(E[i].std()+1e-9)
  order.append((z[ti],i,n))
 order.sort(reverse=True)
 print('\\n=== Expected high',tissue)
 for z,i,n in order[:20]:
  print(f'{n:40s} E_t={E[i,ti]:.3f} z={z:.2f} oldCloaca={Pnorm[i,samples.index(\"Cloaca\")]:.3f} oldStomach={Pnorm[i,samples.index(\"Stomach\")]:.3f}')
PY"
