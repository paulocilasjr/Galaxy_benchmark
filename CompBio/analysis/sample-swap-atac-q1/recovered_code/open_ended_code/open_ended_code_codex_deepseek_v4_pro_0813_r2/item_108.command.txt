/bin/bash -lc "python3 - <<'PY'
import re,urllib.parse,json,numpy as np
html=open('cluster_exp_listing.html').read(); hrefs=re.findall(r'href=\"(["'^"]+'"\\.json)\"',html)
listing=[urllib.parse.unquote(h)[:-5] for h in hrefs]
listing_set=set(listing)
gn=json.load(open('gene_names.json')); acc=np.load('promoter_acc.npy')
# aggregate unique nonzero gene names
from collections import defaultdict
idx_by_name=defaultdict(list)
for i,n in enumerate(gn):
    if acc[i].sum()>0:
        idx_by_name[n].append(i)
sel=[n for n in idx_by_name if n in listing_set]
print('selected genes',len(sel))
with open('selected_genes.txt','w') as f:
    for n in sel: f.write(n+'\\n')
# save mapping selected to indices
with open('selected_gene_indices.json','w') as f:
    json.dump({n:idx_by_name[n] for n in sel},f)
PY
wc -l selected_genes.txt"
