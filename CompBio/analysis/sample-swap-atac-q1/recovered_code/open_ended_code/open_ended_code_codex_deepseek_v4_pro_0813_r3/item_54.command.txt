/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
A=pd.read_csv('atac_gene_promoter_counts.tsv',sep='\\t')
G=A.groupby('symbol').sum(numeric_only=True)
tot=G.sum(axis=0)
for markers in [
 ['INS','GCG','AMY2A','CPA1','PDX1','PTF1A','CEL'],
 ['ACPP','KLK3','NKX3-1','MSMB','HOXB13'],
 ['MUC5B','CFTR','FGF19','SLC4A4','AQP1','KRT19'],
 ['HOXD13','PRRX1','SHH','FGF8','TBX5','HAND2'],
 ['CD3D','PTPRC','MS4A1','CD19','CD79A','NKG7'],
]:
 print('\\n',markers)
 sub=G[G.index.isin(markers)]
 cpm=sub.div(tot,axis=1)*1e6
 # show columns sorted by sum maybe
 print(cpm.T.assign(sum=cpm.sum(axis=0)).sort_values('sum',ascending=False).round(1))
PY"
