/bin/bash -lc "python3 - <<'PY'
import gzip, csv, sys, collections, statistics
fn='sample.swap.atac.q1.tsv.gz'
cols=None
rows=0
totals=None
with gzip.open(fn,'rt') as f:
    r=csv.reader(f,delimiter='\\t')
    header=next(r)
    cols=header[3:]
    totals=[0]*len(cols)
    n=0
    max_vals=[0]*len(cols)
    for row in r:
        n+=1
        for i,v in enumerate(row[3:]):
            x=int(v); totals[i]+=x
            if x>max_vals[i]: max_vals[i]=x
print('header',header)
print('cols',cols)
print('rows',n)
print('totals')
for c,t,m in zip(cols,totals,max_vals):
    print(c,t,m)
PY"
