/bin/bash -lc "python3 - <<'PY'
import gzip, csv, collections, re
sizes={}
for line in open('sample.swap.atac.q1.chrom.sizes'):
    c,s=line.split(); sizes[c]=int(s)
cols=None; sums=collections.defaultdict(lambda:[0]*15); rows=collections.defaultdict(int)
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
    r=csv.reader(f,delimiter='\\t')
    header=next(r); cols=header[3:]
    for row in r:
        c=row[0]; rows[c]+=1
        arr=sums[c]
        vals=list(map(int,row[3:]))
        for i,v in enumerate(vals): arr[i]+=v
def key(c):
    m=re.match(r'chr(\\d+)([pq])',c)
    return int(m.group(1)), m.group(2), c
for c in sorted(sizes, key=key):
    arr=sums[c]
    print(c, rows[c], sizes[c], arr)
PY"
