/bin/bash -lc "python3 - <<'PY'
import gzip,re,collections
markers=['ACPP','AQP1','AQP2','ATP4A','CD19','CD3D','FGF19','GAST','KLK3','KRT20','KRT7','MS4A1','NKG7','NPPA','SCGB1A1','SCGB3A2','SLC17A7','UPK2','ATP6V1E1','EPCAM','SLC12A2','CA4','NHE3','FOXI1','CD79A','TFF2','GKN1']
found=collections.defaultdict(set)
with gzip.open('AmexT_v47-AmexG_v6.0-DD.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        attrs=p[8]
        m=re.search(r'gene_name \"(["'^"]+)"'"',attrs)
        if not m: continue
        gn=m.group(1)
        for part in gn.split('|'):
            name=part.strip()
            base=re.sub(r'\\s*\\[(?:hs|nr|mm|dr|xt|zf|gg|xl|sc)\\]"'$'"','',name).strip()
            if not base: continue
            base2=base.split()[0]
            if base2 in markers:
                found[base2].add(gn)
for mk in markers:
    print(mk, len(found[mk]))
    for x in list(found[mk])[:8]: print('  ',x)
PY"
