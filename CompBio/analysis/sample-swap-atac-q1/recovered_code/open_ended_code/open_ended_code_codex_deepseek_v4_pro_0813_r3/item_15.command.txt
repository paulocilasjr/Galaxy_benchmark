/bin/bash -lc "awk '{s+="'$2; n++} END{print n,s,s/1e9}'"' sample.swap.atac.q1.chrom.sizes; python3 - <<'PY'
# compare windows coverage per chrom
import gzip,csv,collections
rows=collections.Counter(); bases=collections.Counter()
with gzip.open('sample.swap.atac.q1.tsv.gz','rt') as f:
 r=csv.reader(f,delimiter='\\t'); next(r)
 for x in r:
  c=x[0]; rows[c]+=1; bases[c]+=int(x[2])-int(x[1])
for c in sorted(rows):
 print(c, rows[c], bases[c])
PY"
