/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np,itertools
atac=pd.read_csv('atac_gene_promoter_counts.tsv',sep='\\t'); rna=pd.read_csv('bulk_rna_counts_clean.tsv',sep='\\t')
A=atac.groupby('symbol').sum(numeric_only=True); R=rna.set_index('symbol'); common=A.index.intersection(R.index)
Ac=A.loc[common].div(A.loc[common].sum(axis=0),axis=1)*1e6; Rc=R.loc[common].div(R.loc[common].sum(axis=0),axis=1)*1e6
mask=((Ac>0).sum(axis=1)>0)&((Rc>0).sum(axis=1)>0); Ac=Ac[mask]; Rc=Rc[mask]
Al=np.log1p(Ac); Rl=np.log1p(Rc)
def z(X):
 mu=X.mean(axis=1); sd=X.std(axis=1).replace(0,1); return X.sub(mu,axis=0).div(sd,axis=0)
Az=z(Al); Rz=z(Rl)
P=pd.DataFrame(np.corrcoef(Az.T,Rz.T)[:len(A.columns),len(A.columns):],index=A.columns,columns=R.columns)
# reference map for only tissues present in RNA
ref={'Bladder':'Bladder_featurecounts','Brain':'Brain_featurecounts','Cloaca':'Cloacal_chamber_featurecounts','Gill':'Gill_featurecounts','Heart':'Heart_featurecounts','Intestine':'Intestine_featurecounts','Kidney':'Kidney_featurecounts','Limb':'Fore_limb_featurecounts','Liver':'Liver_featurecounts','Lung':'Lung_featurecounts','Spleen':'Spleen_featurecounts','Stomach':'Stomach_featurecounts'}
known=[c for c in A.columns if c in ref]
# For each pair, calculate total diagonal corr if swap pair
base=sum(P.loc[c,ref[c]] for c in known)
print('base diag',base)
results=[]
for a,b in itertools.combinations(known,2):
 score=base - P.loc[a,ref[a]] - P.loc[b,ref[b]] + P.loc[a,ref[b]] + P.loc[b,ref[a]]
 results.append((score,a,b,P.loc[a,ref[b]],P.loc[b,ref[a]]))
results.sort(reverse=True)
for r in results[:20]: print(round(r[0],3),r[1],r[2],'off1',round(r[3],3),'off2',round(r[4],3))
PY"
