/bin/bash -lc "python3 - <<'PY'
# quick inspect overlap between atac symbols and bulk raw symbols from current bulk file
import csv
atac=set()
with open('atac_gene_promoter_counts.tsv') as f:
 r=csv.reader(f,delimiter='\\t'); next(r)
 for row in r:
  atac.add(row[1])
bulk=set()
with open('bulk_rna_counts.tsv') as f:
 r=csv.reader(f,delimiter='\\t'); next(r)
 for row in r:
  bulk.add(row[0])
print(len(atac),len(bulk),len(atac&bulk))
# list overlap examples
for s in list(atac&bulk)[:50]: print(s)
PY"
