/bin/bash -lc "python3 - <<'PY'
import gzip, re
markers=['NEUROD1','SOX2','GFAP','MBP','SLC17A7','TNNT2','MYH6','NPPA','NKX2-5','NKX2-5','ALB','APOA1','FGA','HNF4A','TTR','CDX2','VIL1','FABP2','MUC2','SLC12A1','AQP2','UMOD','PAX8','NPHS2','SFTPC','NKX2-1','SCGB1A1','INS','GCG','AMY2A','CPA1','PDX1','KLK3','ACPP','NKX3-1','UPK1B','UPK2','UPK3A','KRT20','MUC5B','CFTR','FGF19','ATP6V1E1','HOXD13','PRRX1','SHH','FGF8','CD3D','PTPRC','MS4A1','CD19','ATP4A','ATP4B','PGC','GAST','MUC5AC','MYL7','MYL2','NKX2-5','EPCAM','KRT8','KRT18']
# exact names as before pipe from gene_name
with gzip.open('AmexT_v47-AmexG_v6.0-DD.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9 or p[2]"'!='"'gene': continue
        attrs=p[8]
        m=re.search(r'gene_name \"(["'^"]+)"'"',attrs)
        if not m: continue
        gn=m.group(1)
        # split by |
        for part in gn.split('|'):
            name=part.strip()
            base=re.sub(r'\\s*\\[(?:hs|nr|mm|dr|xt)\\]"'$'"','',name).strip()
            if base in markers:
                print(p[0],p[3],p[4],p[6],base,'||',gn)
PY"
