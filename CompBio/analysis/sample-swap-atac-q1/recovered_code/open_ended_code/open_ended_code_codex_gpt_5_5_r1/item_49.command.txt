/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, re, glob, bisect, collections
# samples/libs
path='sample.swap.atac.q1.tsv.gz'
cols=pd.read_csv(path,sep='\\t',nrows=0).columns.tolist(); samples=cols[3:]
lib=np.array([701506327,1168148790,484000505,565994637,1094125058,868549497,699794434,745602917,1147926115,398317869,877686879,1398702406,973845979,497829833,455786546],dtype=np.float64)
# chunk offsets
sizes=[]
with open('sample.swap.atac.q1.chrom.sizes') as f:
    for line in f:
        c,s=line.split(); sizes.append((c,int(s)))
offsets={}; chrom_chunks=collections.defaultdict(list)
for c,s in sizes:
    base=c.rsplit('_',1)[0]
    chrom_chunks[base].append((c,s))
for base,lst in chrom_chunks.items():
    off=0
    # assumes listed order
    for c,s in lst:
        offsets[c]=off; off+=s
# reverse map original pos to chunk
orig_chunks={base:[] for base in chrom_chunks}
for base,lst in chrom_chunks.items():
    off=0
    for c,s in lst:
        orig_chunks[base].append((off,off+s,c)); off+=s

def to_chunk(chrom,start,end):
    out=[]
    for a,b,c in orig_chunks.get(chrom,[]):
        st=max(start,a); en=min(end,b)
        if st<en:
            out.append((c, st-a, en-a))
    return out
markers = {
'Brain':['Rbfox3','Elavl3','Elavl4','Syn1','Snap25','Map2','Tubb3','Nefl','Gad1','Slc17a7','Mbp','Sox10'],
'Heart':['Tnnt2','Tnni3','Myh6','Myh7','Actc1','Nkx2-5','Nppa','Nppb','Ryr2','Myl2','Myl3'],
'Liver':['Alb','Apoa1','Apob','Ttr','Fga','Fgb','Fgg','Hnf4a','Serpina1'],
'Pancreas':['Ins','Gcg','Sst','Pdx1','Ptf1a','Prss1','Prss2','Cpa1','Cela1','Amy2a','Krt19'],
'Kidney':['Slc12a1','Slc12a3','Aqp1','Aqp2','Pax2','Pax8','Umod','Nphs1','Nphs2','Kcnj1'],
'Spleen':['Cd3e','Cd79a','Lck','Ptprc','Spi1','Nkg7','Hbb','Hba1','Lyz'],
'Lung':['Sftpa1','Sftpb','Sftpc','Sftpd','Nkx2-1','Scgb1a1','Foxj1','Ager','Muc5b'],
'Gill':['Krt5','Krt14','Tp63','Foxi1','Atp1a1','Cftr','Aqp3','Cldn10'],
'Limb':['Hoxa13','Hoxd13','Msx1','Msx2','Pax7','Myod1','Myog','Sox9','Col2a1','Prrx1'],
'Intestine':['Vil1','Cdx2','Muc2','Si','Alpi','Fabp2','Krt20','Lgr5'],
'Stomach':['Atp4a','Atp4b','Gif','Muc5ac','Muc6','Gkn1','Pgc','Tff1'],
'Bladder':['Upk1a','Upk1b','Upk2','Upk3a','Krt7','Krt8','Krt13','Foxa1'],
'Prostate':['Nkx3-1','Klk3','Klk2','Acpp','Ar','Msmb','Hoxb13'],
'GallBladder':['Krt7','Krt19','Sox17','Hnf1b','Epcam','Muc1','Muc5b','Cftr'],
'Cloaca':['Krt13','Krt14','Tp63','Foxa1','Cdx2','Muc2','Krt8']
}
key_to_gene={m.upper().replace('-','').replace('.',''):m for ms in markers.values() for m in ms}
gene_orgs=collections.defaultdict(set)
for org,ms in markers.items():
    for m in ms: gene_orgs[m].add(org)
# gather unique TSS per marker gene (all genes_putative primarily plus other transcript tracks)
records=[]
for fn in glob.glob('*.primary.tsv'):
    track=fn.replace('.primary.tsv','')
    df=pd.read_csv(fn,sep='\\t')
    if not {'name','txStart','txEnd'}.issubset(df.columns): continue
    for _,r in df.iterrows():
        for p in str(r['name']).split('|'):
            nm=re.sub(r'\\s*\\[["'^'"\\]]+\\]','',p).strip()
            if nm.startswith('AMEX') or not nm: continue
            key=nm.upper().replace('-','').replace('.','')
            if key in key_to_gene:
                gene=key_to_gene[key]
                tss=int(r['txStart']) if r['strand']=='+' else int(r['txEnd'])
                records.append((gene, str(r['chrom']), tss, track, str(r['strand']), int(r['txStart']), int(r['txEnd'])))
# collapse exact duplicate gene/chrom/tss
seen=set(); rec=[]
for x in records:
    k=x[:3]
    if k not in seen:
        seen.add(k); rec.append(x)
print('unique marker TSS',len(rec))
# evaluate several windows
for win in [10000,25000,50000,100000,250000]:
    intervals=collections.defaultdict(list)
    gene_ids={g:i for i,g in enumerate(sorted(gene_orgs))}
    for gene,chrom,tss,*rest in rec:
        for c,st,en in to_chunk(chrom, max(0,tss-win), tss+win):
            intervals[c].append((st,en,gene_ids[gene]))
    # sort intervals by start; create starts for bisect
    starts={}
    for c,lst in intervals.items():
        lst.sort(); starts[c]=[x[0] for x in lst]
    sums=np.zeros((len(gene_ids),len(samples)), dtype=np.float64)
    nbins=np.zeros(len(gene_ids), dtype=int)
    # stream rows
    for chunk in pd.read_csv(path, sep='\\t', chunksize=200000):
        arr=chunk[samples].to_numpy(dtype=np.float64)
        ch=chunk['chrom'].to_numpy(); st=chunk['start'].to_numpy(); en=chunk['end'].to_numpy()
        for i,(c,a,b) in enumerate(zip(ch,st,en)):
            if c not in intervals: continue
            lst=intervals[c]; ss=starts[c]
            # intervals with start < row_end, scan back a bit up to those possibly overlapping
            j=bisect.bisect_left(ss, b)
            for (x,y,gid) in lst[:j]:  # inefficient but intervals small per chrom
                if y>a and x<b:
                    sums[gid]+=arr[i]
                    nbins[gid]+=1
    # per gene normalized logCPM from summed counts per matching bins; row zscore
    log=np.log2(sums/lib*1e6 + 1)
    z=(log-log.mean(axis=1,keepdims=True))/(log.std(axis=1,keepdims=True)+1e-9)
    gene_names=sorted(gene_ids, key=lambda g:gene_ids[g])
    print('\\nWINDOW',win,'genes with bins',(nbins>0).sum())
    # marker set scores average z per organ marker genes
    score={}
    for org,ms in markers.items():
        idx=[gene_ids[m] for m in ms if m in gene_ids and nbins[gene_ids[m]]>0]
        if idx:
            score[org]=z[idx].mean(axis=0)
    print('marker set predicted top samples:')
    for org in markers:
        if org not in score: continue
        vals=score[org]
        order=np.argsort(-vals)
        print(f'{org:12s}', ' '.join(f'{samples[j]}:{vals[j]:.2f}' for j in order[:5]))
    # for each sample top marker sets
    print('sample top marker sets:')
    orgs=list(score)
    mat=np.vstack([score[o] for o in orgs])
    for j,s in enumerate(samples):
        order=np.argsort(-mat[:,j])
        print(f'{s:12s}', ' '.join(f'{orgs[i]}:{mat[i,j]:.2f}' for i in order[:5]))
PY"
