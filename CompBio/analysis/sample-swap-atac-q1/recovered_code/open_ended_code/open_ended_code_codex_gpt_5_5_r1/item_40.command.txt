/bin/bash -lc "python3 - <<'PY'
import pandas as pd, re, glob
markers = {
'Brain':['RBFOX3','NEUN','ELAVL3','ELAVL4','SYN1','SNAP25','MAP2','TUBB3','NEFL','GAD1','SLC17A7','MBP','SOX10'],
'Heart':['TNNT2','TNNI3','MYH6','MYH7','ACTC1','NKX2-5','NPPA','NPPB','RYR2','MYL2','MYL3'],
'Liver':['ALB','AFP','APOA1','APOB','TTR','CYP3A4','FGA','FGB','FGG','HNF4A','SERPINA1'],
'Pancreas':['INS','GCG','SST','PDX1','PTF1A','PRSS1','PRSS2','CPA1','CELA1','AMY2A','KRT19'],
'Kidney':['SLC12A1','SLC12A3','AQP1','AQP2','PAX2','PAX8','UMOD','NPHS1','NPHS2','KCNJ1'],
'Spleen':['CD3D','CD3E','CD79A','MS4A1','LCK','PTPRC','SPI1','NKG7','HBB','HBA1','HBA2','LYZ'],
'Lung':['SFTPA1','SFTPB','SFTPC','SFTPD','NKX2-1','SCGB1A1','FOXJ1','AGER','MUC5B'],
'Gill':['KRT5','KRT14','TP63','FOXI1','ATP1A1','CFTR','AQP3','CLDN10'],
'Limb':['HOXA13','HOXD13','MSX1','MSX2','PAX7','MYOD1','MYOG','SOX9','COL2A1','PRRX1'],
'Intestine':['VIL1','CDX2','MUC2','SI','ALPI','FABP2','KRT20','LGR5'],
'Stomach':['ATP4A','ATP4B','GIF','MUC5AC','MUC6','GKN1','PGC','TFF1'],
'Bladder':['UPK1A','UPK1B','UPK2','UPK3A','KRT7','KRT8','KRT13','FOXA1'],
'Prostate':['NKX3-1','KLK3','KLK2','ACPP','AR','MSMB','HOXB13'],
'GallBladder':['KRT7','KRT19','SOX17','HNF1B','EPCAM','MUC1','MUC5B','CFTR'],
'Cloaca':['KRT13','KRT14','TP63','FOXA1','CDX2','MUC2','KRT8']
}
allm={m.upper() for ms in markers.values() for m in ms}
rows=[]
for fn in glob.glob('*.primary.tsv'):
    track=fn.replace('.primary.tsv','')
    df=pd.read_csv(fn,sep='\\t',comment='#')
    if 'name' not in df.columns: continue
    for _,r in df.iterrows():
        name=str(r['name'])
        parts=name.split('|')
        names=[]
        for p in parts:
            # strip ID and tags
            p=re.sub(r'\\s*\\[["'^'"\\]]+\\]','',p).strip()
            if p.startswith('AMEX'): continue
            if p: names.append(p)
        for nm in names:
            base=nm.upper().replace('-','').replace('.','')
            # also exact uppercase with punctuation
            if nm.upper() in allm or base in {x.replace('-','').replace('.','') for x in allm}:
                tss=int(r['txStart']) if r['strand']=='+' else int(r['txEnd'])
                rows.append((track,nm,r['name'],r['chrom'],r['strand'],int(r['txStart']),int(r['txEnd']),tss))
# group found by marker org
print('found rows',len(rows))
found_names={}
for rec in rows:
    for org,ms in markers.items():
        for m in ms:
            if rec[1].upper().replace('-','').replace('.','')==m.upper().replace('-','').replace('.',''):
                found_names.setdefault(org,set()).add(m)
print('found markers by organ')
for org in markers:
    print(org, sorted(found_names.get(org,set())))
print('\\nfirst 100 records')
for rec in rows[:100]: print('\\t'.join(map(str,rec)))
PY"
