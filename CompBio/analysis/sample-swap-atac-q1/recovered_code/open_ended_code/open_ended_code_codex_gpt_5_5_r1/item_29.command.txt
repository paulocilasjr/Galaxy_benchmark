/bin/bash -lc 'for track in annotation_patch genes_putative genes_fragmented genes_predicted genes_non_annotated genes_other; do
  echo $track
  wget -qO ${track}.gtf "https://genome.axolotl-omics.org/cgi-bin/hgTables?db=ambMex60DD&hgta_group=genes&hgta_track=${track}&hgta_table=${track}&hgta_regionType=genome&hgta_outputType=gff&hgta_doTopSubmit=1"
  wc -l ${track}.gtf
  head -n 1 ${track}.gtf
 done'
