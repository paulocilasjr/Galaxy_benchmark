/bin/bash -lc "wget -qO hgTables.html 'https://genome.axolotl-omics.org/cgi-bin/hgTables?db=ambMex60DD'
python3 - <<'PY'
import re
s=open('hgTables.html',errors='ignore').read()
# print option values containing Gene/gene/Annotation etc around hgta_track/table selects
for pat in ['hgta_track','hgta_table','Annotation','gene']:
    print('---',pat, s.find(pat))
# list select option values/text briefly
opts=re.findall(r'<OPTION["'^>]*VALUE="?([''^" >]+)"?[''^>]*>([''^<]+)'"', s, re.I)
for v,t in opts[:200]:
    if any(k in (v+' '+t).lower() for k in ['gene','trans','patch','amext','refseq','known','augustus']):
        print(v, '=>', t.strip())
print('nopts',len(opts))
PY"
