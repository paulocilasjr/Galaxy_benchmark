/bin/bash -c "python3 - <<'PY'
import json
p='[REDACTED LOCAL PATH]'
data=json.load(open(p))
for rec in data:
    print('id', rec.get('id'), 'input', rec.get('input'))
    print('allele_string', rec.get('allele_string'))
    print('most_severe', rec.get('most_severe_consequence'))
    print('assembly', rec.get('assembly_name'), 'seq_region', rec.get('seq_region_name'), 'start', rec.get('start'), 'end', rec.get('end'))
    print('transcripts', len(rec.get('transcript_consequences',[])))
    for tc in rec.get('transcript_consequences',[]):
        print(tc.get('transcript_id'), tc.get('gene_symbol'), tc.get('consequence_terms'), tc.get('impact'), tc.get('variant_allele'), tc.get('amino_acids'), tc.get('codons'), tc.get('protein_start'), tc.get('protein_end'), tc.get('hgvsp'), tc.get('hgvsc'))
PY"
