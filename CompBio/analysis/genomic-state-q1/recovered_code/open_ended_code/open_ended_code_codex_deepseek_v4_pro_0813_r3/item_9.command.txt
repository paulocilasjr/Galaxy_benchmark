/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for url in [
 'https://api.genome.ucsc.edu/list/tracks?genome=hg38;chrom=chr11;start=124738681;end=124738772',
 'https://api.genome.ucsc.edu/list/hg38',
 'https://api.genome.ucsc.edu/list/hubs?genome=hg38',
]:
 try:
  req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
  with urllib.request.urlopen(req,timeout=30) as r: data=json.loads(r.read())
  print(url, '\\n', str(data)[:1000])
 except Exception as e: print('ERR',url,e)
PY"
