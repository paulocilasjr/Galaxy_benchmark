/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
base='https://api.genome.ucsc.edu/getData/track'
for track,start,end in [('knownGene',124600000,124850000),('ncbiRefSeq',124600000,124850000)]:
    q=urllib.parse.urlencode({'genome':'hg38','track':track,'chrom':'chr11','start':start,'end':end})
    u=base+'?'+q
    req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
    with urllib.request.urlopen(req,timeout=30) as r: txt=r.read().decode()
    data=json.loads(txt)
    items=data.get(track,[])
    print('===',track,'items',len(items))
    for it in items[:50]:
        print(it.get('name2'), it.get('name'), it.get('chromStart'), it.get('chromEnd'), it.get('strand'), it.get('geneName'), it.get('geneType'))
PY"
