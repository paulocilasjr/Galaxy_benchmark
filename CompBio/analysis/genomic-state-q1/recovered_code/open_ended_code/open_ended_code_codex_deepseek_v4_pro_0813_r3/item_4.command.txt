/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
for track in ['knownGene','ncbiRefSeq']:
 u='https://api.genome.ucsc.edu/getData/track?'+urllib.parse.urlencode({'genome':'hg38','track':track,'chrom':'chr11','start':124680000,'end':124780000})
 req=urllib.request.Request(u,headers={'User-Agent':'Mozilla/5.0'})
 with urllib.request.urlopen(req,timeout=30) as r: data=json.loads(r.read())
 print('===',track)
 for it in data.get(track,[]):
  if track=='knownGene':
   print(it['chromStart'],it['chromEnd'],it['name2'],it['geneType'],it['strand'],it.get('name'))
  else:
   print(it['chromStart'],it['chromEnd'],it.get('name2'),it.get('name'),it.get('strand'))
PY"
