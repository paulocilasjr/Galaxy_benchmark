import json, urllib.request
base='https://api.genome.ucsc.edu/getData/track?genome=hg38;chrom=chr11;start=124738500;end=124739000;track='
tracks=['cCREregistry','coreCcres','encodeCcreCombined','geneHancer','refSeqFuncElems','vistaEnhancersBb','wgEncodeReg4Atac','wgEncodeReg4Dnase','wgEncodeReg4MarkH3k27ac','wgEncodeReg4MarkH3k4me1','wgEncodeReg4MarkH3k4me3','wgEncodeReg4Txn','wgEncodeRegDnaseClustered','wgEncodeRegDnaseWig','wgEncodeRegMarkH3k27ac','wgEncodeRegMarkH3k4me1','wgEncodeRegMarkH3k4me3','wgEncodeRegTxn','robustPeaks','encRegTfbsClustered','jaspar']
for tr in tracks:
 u=base+tr
 try:
  d=json.load(urllib.request.urlopen(u,timeout=60))
  key=tr
  print('\\n###',tr, d.get('trackType'),d.get('dataTime'))
  for k,v in d.items():
   if k in ('downloadTime','downloadTimeStamp','dataTime','dataTimeStamp','genome','trackType','track','chrom','chromSize','start','end','itemsReturned','bigDataUrl'): continue
   print(k, json.dumps(v)[:5000])
 except Exception as e: print('\\n###',tr,type(e).__name__,e)
PY"
