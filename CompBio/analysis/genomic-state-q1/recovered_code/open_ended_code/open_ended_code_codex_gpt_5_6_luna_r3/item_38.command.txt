/bin/bash -lc "python - <<'PY'
import urllib.request,gzip,io
base='https://personal.broadinstitute.org/cboix/epimap/ChromHMM/observed_aux_18_hg38/CALLS/'
ids=['BSS00478','BSS00483','BSS00484','BSS00554','BSS00556','BSS01158','BSS01159','BSS01164','BSS01168','BSS01169','BSS01170','BSS01519','BSS00369','BSS00371','BSS00729','BSS00125','BSS00127']
for sid in ids:
 u=base+sid+'_18_CALLS_segments.bed.gz'
 try:
  raw=urllib.request.urlopen(u,timeout=120).read()
  hits=[]
  with gzip.GzipFile(fileobj=io.BytesIO(raw)) as f:
   for line in f:
    p=line.decode().rstrip().split('\\t')
    if p[0]=='chr11' and int(p[1])<124738772 and int(p[2])>124738680:
     hits.append(p)
  print('\\n',sid,'bytes',len(raw))
  for p in hits: print('\\t'.join(p))
 except Exception as e: print('\\n',sid,type(e).__name__,e)
PY"
