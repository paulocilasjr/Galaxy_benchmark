/bin/bash -lc "python - <<'PY'
import urllib.request,json
for u in ['https://zenodo.org/api/records/19443186','https://zenodo.org/api/records/19443185']:
 print('\\n',u)
 try:
  d=json.load(urllib.request.urlopen(u,timeout=60))
  print('title',d.get('metadata',{}).get('title'))
  for f in d.get('files',[]): print(f.get('key'),f.get('size'),f.get('links',{}).get('self'))
 except Exception as e: print(e)
PY"
