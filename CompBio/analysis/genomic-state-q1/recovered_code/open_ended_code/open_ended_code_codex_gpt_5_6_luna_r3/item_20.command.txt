/bin/bash -lc "python - <<'PY'
import json, urllib.request
x=json.load(urllib.request.urlopen('https://api.genome.ucsc.edu/list/tracks?genome=hg38',timeout=90))['hg38']
def walk(obj,path=''):
 if isinstance(obj,dict):
  s=' '.join(str(v) for v in obj.values() if isinstance(v,(str,int,float)))
  if any(t in s.lower() for t in ['frontal','liver','embryonic stem','esc','polycomb','ubiquit','poised','enhancer']):
   print(path, s[:1000].replace('\\n',' '))
  for k,v in obj.items(): walk(v,path+'/'+k)
 elif isinstance(obj,list):
  for i,v in enumerate(obj): walk(v,path+'/'+str(i))
walk(x)
PY"
