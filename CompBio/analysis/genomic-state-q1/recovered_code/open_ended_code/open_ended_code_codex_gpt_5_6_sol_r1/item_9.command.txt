/bin/bash -lc "curl -fsSL 'https://api.genome.ucsc.edu/getData/track?genome=hg38;track=ncbiRefSeq;chrom=chr11;start=124700000;end=124780000' -o ucsc_refseq.json
jq -r '.ncbiRefSeq[] | [.chromStart,.chromEnd,.name,.name2,.strand,.txStart,.txEnd] | @tsv' ucsc_refseq.json | head -30
curl -fsSL 'https://api.genome.ucsc.edu/getData/track?genome=hg38;track=encodeCcreCombined;chrom=chr11;start=124738000;end=124739500' -o ucsc_ccre.json
jq -r '.. | objects | select(has(\"chromStart\")) | [.chromStart,.chromEnd,.name,.ucscLabel,.description] | @tsv' ucsc_ccre.json | head -30"
