/bin/bash -lc "python - <<'PY'
import json, urllib.request
for track in ['knownGene','ncbiRefSeqCurated','gencodeV49','wgEncodeGencodeCompV44','ensGene','refGene']:
 u=f'https://api.genome.ucsc.edu/getData/track?genome=hg38;track={track};chrom=chr11;start=124730000;end=124750000'
 try:
  with urllib.request.urlopen(u) as f: d=json.load(f)
  print('\\nTRACK',track, 'keys',d.keys())
  vals=d.get(track,[])
  print(json.dumps(vals[:20],indent=2)[:8000])
 except Exception as e: print(track,e)
PY"
