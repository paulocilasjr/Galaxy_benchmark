/bin/bash -lc "python - <<'PY'
import json,urllib.request
for track in ['cCREregistry','encodeCcreCombined','coreCcres','vistaEnhancersBb','mprabase']:
 u=f'https://api.genome.ucsc.edu/getData/track?genome=hg38;track={track};chrom=chr11;start=124737000;end=124741500'
 try:
  with urllib.request.urlopen(u) as f:d=json.load(f)
  print('\\n',track)
  print(json.dumps(d.get(track,[]),indent=2)[:20000])
 except Exception as e: print(e)
PY"
