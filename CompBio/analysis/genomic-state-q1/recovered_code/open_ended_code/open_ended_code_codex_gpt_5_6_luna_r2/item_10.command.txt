/bin/bash -lc "python - <<'PY'
import json, urllib.request
with urllib.request.urlopen('https://api.genome.ucsc.edu/list/tracks?genome=hg38') as f:d=json.load(f)['hg38']
for k,v in d.items():
 s=(k+' '+str(v.get('shortLabel',''))+' '+str(v.get('longLabel',''))+' '+str(v.get('group',''))).lower()
 if any(x in s for x in ['roadmap','chromatin','epigenom','state']):
  print(k, '|',v.get('shortLabel'), '|',v.get('longLabel'), '|',v.get('type'), '| parent',v.get('parent'))
PY"
