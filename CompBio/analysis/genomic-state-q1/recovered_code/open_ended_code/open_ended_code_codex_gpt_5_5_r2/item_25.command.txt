/bin/bash -lc "python3 - <<'PY'
import gzip
start=124738680; end=124738772; chrom='chr11'
with gzip.open('data/hg38ToHg19.over.chain.gz','rt') as f:
    in_chain=False
    for line in f:
        if line.startswith('chain'):
            p=line.split()
            # chain score tName tSize tStrand tStart tEnd qName qSize qStrand qStart qEnd id
            tName=p[2]; tStart=int(p[5]); tEnd=int(p[6]); qName=p[7]; qStrand=p[9]
            if tName==chrom and not (tEnd<=start or tStart>=end):
                header=p
                tpos=tStart; qpos=int(p[10])
                in_chain=True
            else:
                in_chain=False
            continue
        if not in_chain or not line.strip():
            continue
        nums=list(map(int,line.split()))
        size=nums[0]
        b_t_start=tpos; b_t_end=tpos+size
        b_q_start=qpos; b_q_end=qpos+size
        if b_t_end>start and b_t_start<end:
            ov_start=max(start,b_t_start); ov_end=min(end,b_t_end)
            q_ov_start=b_q_start+(ov_start-b_t_start)
            q_ov_end=b_q_start+(ov_end-b_t_start)
            print('header', header)
            print('block', b_t_start,b_t_end,'->',b_q_start,b_q_end,'overlap',ov_start,ov_end,'maps',qName,q_ov_start,q_ov_end,'qStrand',qStrand)
        if len(nums)==3:
            tpos += size + nums[1]
            qpos += size + nums[2]
        else:
            tpos += size; qpos += size
PY"
