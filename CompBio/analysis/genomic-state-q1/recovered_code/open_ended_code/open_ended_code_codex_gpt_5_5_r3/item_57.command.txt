/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('ucsc_tracks_hg19.json'))['hg19']
h=j['wgEncodeBroadHmm']

def walk(prefix,obj):
    if isinstance(obj,dict):
        if 'type' in obj and 'bed' in str(obj.get('type')).lower():
            name=prefix[-1]
            s=' '.join([name,str(obj.get('shortLabel','')),str(obj.get('longLabel',''))]).lower()
            if any(x in s for x in ['liver','hepg2','h1','h9','esc','brain','neuro','nh-a','frontal']):
                print(name,'|',obj.get('shortLabel'),'|',obj.get('longLabel'),'|',obj.get('type'))
        for k,v in obj.items(): walk(prefix+[k],v)
walk(['wgEncodeBroadHmm'],h)
PY"
