/bin/bash -lc "python3 - <<'PY'
import json, urllib.request
tracks={
 'brain_avg':'wgEncodeReg4MarkH3k27acBrain',
 'liver_avg':'wgEncodeReg4MarkH3k27acLiver',
 'H1':'ENCFF919FBG',
 'H9':'ENCFF988WEQ',
 'frontal84':'ENCFF646KVN',
 'liver_sample':'ENCFF764VSN',
}
for label,track in tracks.items():
    url=f'https://api.genome.ucsc.edu/getData/track?genome=hg38;track={track};chrom=chr11;start=124738680;end=124738772'
    try:
        data=json.load(urllib.request.urlopen(url, timeout=20))
        arr=data.get(track,[])
        vals=[]
        for r in arr:
            vals += [r['value']] * (r['end']-r['start'])
        print(label, track, 'n=',len(vals),'mean=',round(sum(vals)/len(vals),3) if vals else None,'minmax=',(round(min(vals),3),round(max(vals),3)) if vals else None)
    except Exception as e:
        print(label, track, 'ERR', e)
PY"
