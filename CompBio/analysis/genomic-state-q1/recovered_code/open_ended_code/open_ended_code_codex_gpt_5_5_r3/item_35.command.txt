import json
j=json.load(open('ucsc_tracks_hg38.json'))['hg38']

        s=(name+' '+str(meta.get('shortLabel',''))+' '+str(meta.get('longLabel',''))).lower()
        if any(x in s for x in ['liver','frontal','middle_frontal','brain','h1','h9','esc']):
            print(name, '|', meta.get('shortLabel'), '|', meta.get('longLabel'), '| parent', meta.get('parent'), '| groups', 'organ='+str(meta.get('organ','')), 'data='+str(meta.get('dataType','')))
PY"
