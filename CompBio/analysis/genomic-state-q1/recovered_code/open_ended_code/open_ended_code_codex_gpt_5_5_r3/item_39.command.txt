/bin/bash -lc "python3 - <<'PY'
import json
j=json.load(open('ucsc_tracks_hg38.json'))['hg38']
cc=j['coreCcres']

def walk(prefix,obj):
    if isinstance(obj,dict):
        if 'type' in obj and ('bigBed' in str(obj.get('type')) or 'bigWig' in str(obj.get('type'))):
            name=prefix[-1] if prefix else None
            s=' '.join([name or '',str(obj.get('shortLabel','')),str(obj.get('longLabel','')),str(obj.get('organ','')),str(obj.get('simpleBiosample','')),str(obj.get('dataType',''))]).lower()
            if any(x in s for x in ['liver','frontal','middle','brain','h1','h9','esc']):
                print(name, '|', obj.get('shortLabel'), '|', obj.get('longLabel'), '| type', obj.get('type'), '| organ', obj.get('organ'), '| biosample', obj.get('simpleBiosample'), '| data', obj.get('dataType'))
        for k,v in obj.items(): walk(prefix+[k],v)
walk(['coreCcres'],cc)
PY"
