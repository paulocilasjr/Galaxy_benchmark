/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
tumor=C['Tumor_Core'].values>0; tcoords=coords[tumor]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
for measure in ['count','presence']:
 print('\\n',measure,'Spearman with negative distance (closer)')
 for ct in cts[:-1]:
  x=C[ct].values if measure=='count' else (C[ct].values>0).astype(int)
  r,p=scipy.stats.spearmanr(x,-D)
  print(f'{ct:20s} r={r:.3f} p={p:.3g}')
# partial correlation with distance after controlling total cells? not necessary
PY"
