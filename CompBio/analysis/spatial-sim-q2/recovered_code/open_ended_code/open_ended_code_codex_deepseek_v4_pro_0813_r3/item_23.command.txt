/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
sc=pd.read_csv('data/spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('data/spatial_q_sc_metadata.csv', index_col=0)
cts=sorted(meta['cell_type'].unique())
means=pd.DataFrame({ct: sc.values[meta['cell_type']==ct].mean(axis=0) for ct in cts}, index=sc.columns)
X=sc.values
Xcpm=X/X.sum(axis=1,keepdims=True)*1e4
# use all gene z-scores from SC? or spatial? compute z across cells
Xz=(Xcpm-Xcpm.mean(axis=0))/(Xcpm.std(axis=0)+1e-9)
scores=np.zeros((X.shape[0],len(cts)))
for i,ct in enumerate(cts):
 spec=means[ct]/(means.sum(axis=1)+1e-9)
 genes=spec[spec>0.5].index.tolist()
 scores[:,i]=Xz[:, [sc.columns.get_loc(g) for g in genes]].mean(axis=1)
pred=np.array(cts)[scores.argmax(axis=1)]
print('accuracy', (pred==meta['cell_type'].values).mean())
print(pd.crosstab(meta['cell_type'],pred))
PY"
