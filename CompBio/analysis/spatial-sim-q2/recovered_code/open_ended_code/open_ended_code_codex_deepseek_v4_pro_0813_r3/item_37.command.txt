/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.io, scipy.optimize, time, itertools, math
sc=pd.read_csv('data/spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('data/spatial_q_sc_metadata.csv', index_col=0)
cts=sorted(meta['cell_type'].unique()); m=len(cts)
S=np.column_stack([sc.values[meta['cell_type']==ct].mean(axis=0) for ct in cts])
Y=scipy.io.mmread('data/matrix.mtx.gz').toarray()
# comps up to 6
comps_by_k={}
for k in range(1,7):
 comps=[]
 def rec(start, rem, cur):
  if rem==0:
   counts=np.zeros(m,dtype=int)
   for idx in cur: counts[idx]+=1
   comps.append(counts); return
  for i in range(start,m):
   cur.append(i); rec(i, rem-1, cur); cur.pop()
 rec(0,k,[])
 comps_by_k[k]=comps
print({k:len(v) for k,v in comps_by_k.items()})
pred=[]
start=time.time()
for yi in Y:
 best=None; bestll=-1e18; bestk=None
 for k,comps in comps_by_k.items():
  # skip k too far from nnls sum? could but no
  for comp in comps:
   lam=S@comp
   ll=np.sum(yi*np.log(lam+1e-12)-lam)
   if ll>bestll: bestll=ll; best=comp; bestk=k
 pred.append((bestk,best))
 if len(pred)%100==0: print(len(pred),'time',time.time()-start)
pred_counts=np.array([x[1] for x in pred])
pred_k=np.array([x[0] for x in pred])
print('k counts',pd.Series(pred_k).value_counts())
df=pd.DataFrame(pred_counts,columns=cts)
df.to_csv('map_compositions.csv',index=False)
print('presence counts')
print((df>0).sum())
print('sum counts',df.sum(axis=1).value_counts())
# compare to nnls presence
P=pd.read_csv('nnls_weights.csv'); P.columns=cts
print('presence agreement per type')
for ct in cts:
 print(ct, ((P[ct]>0.2)==(df[ct]>0)).mean())
print('exact total weight corr', np.corrcoef(pred_k,P.sum(axis=1))[0,1])
print(df.head(20))
PY"
