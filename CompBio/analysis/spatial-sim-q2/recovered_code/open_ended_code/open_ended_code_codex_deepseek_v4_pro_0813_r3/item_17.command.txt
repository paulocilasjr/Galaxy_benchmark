/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.stats
sc=pd.read_csv('data/spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('data/spatial_q_sc_metadata.csv', index_col=0)
cts=sorted(meta['cell_type'].unique())
means=pd.DataFrame({ct: sc.values[meta['cell_type']==ct].mean(axis=0) for ct in cts}, index=sc.columns)
# top genes by max/min or specificity: mean ct / sum
means_norm=means.div(means.sum(axis=1),axis=0)
# top specificity per type
for ct in cts:
 scores=means_norm[ct].sort_values(ascending=False)
 print('\\n',ct,'top10',[(g,round(s,3),round(means.loc[g,ct],2)) for g,s in scores.head(10).items()])
# log2fc top vs others using mean+1
for ct in cts:
 others=[c for c in cts if c"'!=ct]
 fc=np.log2((means[ct]+1)/(means[others].mean(axis=1)+1))
 top=fc.sort_values(ascending=False).head(10)
 print(ct,'"'topFC',[(g,round(v,2)) for g,v in top.items()])
PY"
