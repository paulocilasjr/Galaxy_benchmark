/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.sparse, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
P=pd.read_csv('nnls_weights.csv')
P.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
presence=(P>0.2).values
types=P.columns.tolist()
tumor=presence[:,types.index('Tumor_Core')]
print('tumor spots',tumor.sum())
tcoords=coords[tumor]
# distance matrix from all spots to nearest tumor
D=scipy.spatial.distance.cdist(coords,tcoords,metric='euclidean').min(axis=1)
for ct in types:
 if ct=='Tumor_Core': continue
 sel=presence[:,types.index(ct)]
 observed=D[sel].mean()
 obs_med=np.median(D[sel])
 # all spots baseline (including tumor? maybe non-tumor spots)
 base=D[~tumor].mean() if (~tumor).any() else np.nan
 # all spots baseline excluding tumor-containing? distance to tumor from non-tumor spots
 base_all=D[~tumor].mean()
 # random permutation among all spots of sel count
 rng=np.random.default_rng(42)
 n=sel.sum(); perm_means=[]
 for _ in range(10000):
  idx=rng.choice(900,size=n,replace=False)
  perm_means.append(D[idx].mean())
 p=(np.array(perm_means)<=observed).mean()
 print(f'{ct:20s} n={n:3d} obs_mean={observed:.3f} obs_med={obs_med:.3f} baseline_nontumor={base_all:.3f} p_less={p:.5f} z={(observed-np.mean(perm_means))/np.std(perm_means):.2f}')
PY"
