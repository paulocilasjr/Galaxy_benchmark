/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
T=C[C['Tumor_Core']>0]
print(T.shape)
print('tumor count distribution',T['Tumor_Core'].value_counts())
print('total cells per tumor spot',T.sum(axis=1).value_counts())
print('other types presence among tumor spots')
for ct in cts[:-1]: print(ct,(T[ct]>0).sum(),(T[ct]>0).mean())
print('composition combinations')
# unique composition counts in tumor spots
combos=T.apply(lambda r: tuple(int(r[c]) for c in cts),axis=1).value_counts()
print(combos)
PY"
