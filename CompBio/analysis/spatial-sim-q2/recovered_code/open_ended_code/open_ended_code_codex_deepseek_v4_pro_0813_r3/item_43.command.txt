import pandas as pd, numpy as np, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
# use marker scores dominant
bar=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0]

for method,labels in [('marker',labels)]:
 tumor=labels=='Tumor_Core'; tcoords=coords[tumor]
 D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)

 for ct in cts[:-1]:
  sel=(labels==ct)&~tumor
  obs=D[sel].mean(); base=D[~tumor].mean()
  u,p=scipy.stats.mannwhitneyu(D[sel],D[(~tumor)&(labels"'!=ct)],alternative='"'less')
  print(f'{ct:20s} n={sel.sum():3d} mean={obs:.3f} base={base:.3f} p={p:.4g}')
# NNLS max labels from exact counts ties choose max then first? Use max count
C=pd.read_csv('map_compositions.csv'); C.columns=cts

 labels2.append(cand)

print('\\nNNLS max labels counts',labels2.value_counts())
tumor=labels2=='Tumor_Core'; tcoords=coords[tumor]

print('tumor n',tumor.sum())

 print(f'{ct:20s} n={sel.sum():3d} mean={obs:.3f} base={base:.3f} p={p:.4g}')
