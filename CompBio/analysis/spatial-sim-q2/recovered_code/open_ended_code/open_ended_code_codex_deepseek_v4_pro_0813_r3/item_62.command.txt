/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
C=pd.read_csv('map_compositions.csv'); C.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
tumor_count=C['Tumor_Core'].values
tumor_pres=(tumor_count>0).astype(float)
dist=scipy.spatial.distance.cdist(coords,coords)
for radius in [1.5,2.5,3.5,5.1]:
 adj=(dist<=radius)&(dist>0)
 neigh_tumor=adj@tumor_pres
 neigh_degree=adj.sum(axis=1)
 neigh_rate=neigh_tumor/np.where(neigh_degree>0,neigh_degree,1)
 print('\\nradius',radius)
 for ct in C.columns[:-1]:
  x=C[ct].values
  r,p=__import__('scipy.stats').stats.spearmanr(x,neigh_rate)
  print(f'{ct:20s} spearman with neighbor tumor rate r={r:.3f} p={p:.3g}')
PY"
