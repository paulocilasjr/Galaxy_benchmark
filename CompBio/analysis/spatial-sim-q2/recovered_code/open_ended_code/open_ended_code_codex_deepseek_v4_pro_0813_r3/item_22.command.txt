/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
pos=pd.read_csv('data/tissue_positions.csv',index_col=0)
bar=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0].tolist()
scores=pd.read_csv('marker_scores.csv',index_col=0); scores.index=bar
props=pd.read_csv('nnls_props.csv'); props.index=bar
for df,name in [(scores,'score'),(props,'prop')]:
 d=df.copy(); d['row']=pos['array_row'].values; d['col']=pos['array_col'].values
 for ct in ['Tumor_Core','Macrophage','Endothelial','Fibroblast_Stroma','T_Cell','B_Cell']:
  grid=d.pivot(index='row',columns='col',values=ct)
  if name=='score':
   thr=np.quantile(d[ct],0.8)
  else:
   thr=0.2
  chars=np.where(grid.values>thr,'#','.')
  print('\\n',ct,name,'thr',round(thr,3))
  for i in range(grid.shape[0]):
   print(''.join(chars[i]))
PY"
