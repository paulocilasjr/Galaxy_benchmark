/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
tumor=C['Tumor_Core'].values>0; tcoords=coords[tumor]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
print('tumor spots',tumor.sum())
for ct in cts[:-1]:
 counts=C[ct].values
 # binary presence excluding tumor
 sel=(counts>0)&~tumor
 n=sel.sum(); wmean=np.average(D[sel],weights=counts[sel]); mean=D[sel].mean()
 # all occurrences in non-tumor spots? use counts repeated
 # compare to expected by permutation of counts across non-tumor spots preserving spatial positions? do simple random assignments of spots
 print(f'{ct:20s} binary_n={n:3d} mean={mean:.3f} weighted={wmean:.3f} total_cells={counts.sum()}')
 # co-occurrence with tumor
 print('   overlap spots',(counts>0 & tumor).sum(),'cells in tumor spots',counts[tumor].sum())
PY"
