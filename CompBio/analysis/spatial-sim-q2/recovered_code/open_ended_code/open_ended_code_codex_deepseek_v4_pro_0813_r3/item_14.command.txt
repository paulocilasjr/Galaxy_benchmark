/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.io, scipy.optimize
sc=pd.read_csv('data/spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('data/spatial_q_sc_metadata.csv', index_col=0)
cts=sorted(meta['cell_type'].unique())
S=np.column_stack([sc.values[meta['cell_type']==ct].mean(axis=0) for ct in cts])
Y=scipy.io.mmread('data/matrix.mtx.gz').toarray()
P=np.zeros((len(cts),Y.shape[1]))
for j in range(Y.shape[1]):
 p,_=scipy.optimize.nnls(S,Y[:,j]); P[:,j]=p
wsum=P.sum(axis=0)
print(pd.Series(wsum).describe())
print('corr with spot sums', np.corrcoef(wsum,Y.sum(axis=0))[0,1])
PY"
