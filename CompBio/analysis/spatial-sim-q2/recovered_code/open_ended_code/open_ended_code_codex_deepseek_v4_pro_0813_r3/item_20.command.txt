/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
# barcodes order matches matrix rows? pos rows sorted? Check barcodes file order vs pos order
bar=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0].tolist()
print('bar first',bar[:5],'pos first',pos.index[:5].tolist())
print('same order?',bar==pos.index.tolist())
scores=pd.read_csv('marker_scores.csv',index_col=0)
props=pd.read_csv('nnls_props.csv')
props.index=bar
# merge positions
for df,name in [(scores,'score'),(props,'prop')]:
 d=df.copy(); d['row']=pos['array_row']; d['col']=pos['array_col']
 # pivot for each type
 for ct in ['Tumor_Core','Macrophage','Endothelial','Fibroblast_Stroma','T_Cell','B_Cell']:
  grid=d.pivot(index='row',columns='col',values=ct)
  # print threshold map as chars
  if name=='score':
   # signs? use top 10%?
   thr=np.quantile(d[ct],0.8)
  else:
   thr=0.2
  chars=np.where(grid.values>thr,'#','.')
  print('\\n',ct,name,'thr',thr)
  for i in range(grid.shape[0]):
   print(''.join(chars[i]))
PY"
