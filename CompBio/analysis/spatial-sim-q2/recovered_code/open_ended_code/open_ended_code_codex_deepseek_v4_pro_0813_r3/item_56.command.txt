/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
C=pd.read_csv('map_compositions.csv'); C.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
tumor=(C['Tumor_Core']>0).values; tcoords=coords[tumor]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
fib=(C['Fibroblast_Stroma']>0).values
near=(D<=1.5)&~tumor
fib_near=coords[fib&near]
# for each fib near spot count adjacent tumor spots
tumor_adj_counts=[]
for p in fib_near:
 d=np.abs(tcoords-p); d=np.minimum(d,30-d); # not torus for edge? use euclid <=1.5 direct
 euc=np.sqrt(((tcoords-p)**2).sum(axis=1))
 tumor_adj_counts.append((euc<=1.5).sum())
print('fib near unique',len(fib_near),'adj tumor sum',sum(tumor_adj_counts),'dist',sorted(tumor_adj_counts)[-10:])
# total fibro incidence to tumor spots from all fibro spots (not only non-tumor)
fib_all=coords[fib]
cnts=[]
for p in fib_all:
 euc=np.sqrt(((tcoords-p)**2).sum(axis=1)); cnts.append((euc<=1.5).sum())
print('all fib adj tumor sum',sum(cnts),'fib total',len(cnts),'cnts dist',pd.Series(cnts).value_counts())
# direct matrix observed
dist=scipy.spatial.distance.cdist(coords,coords); adj=(dist<=1.5)&(dist>0)
print('obs matrix',(fib[:,None]&tumor[None,:]&adj).sum())
PY"
