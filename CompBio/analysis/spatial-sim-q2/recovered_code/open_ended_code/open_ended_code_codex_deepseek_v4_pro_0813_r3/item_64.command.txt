/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial, scipy.stats
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
bar=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0]
P=pd.read_csv('nnls_props.csv'); P.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']; P.index=bar
# raw weights threshold variants
W=pd.read_csv('nnls_weights.csv'); W.columns=P.columns; W.index=bar
for thresh in [0.01,0.05,0.1,0.2,0.3]:
 tumor=(W['Tumor_Core']>thresh).values
 if tumor.sum()<2: continue
 tcoords=coords[tumor]; D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
 print('\\nthr',thresh,'tumor',tumor.sum())
 for ct in P.columns[:-1]:
  sel=(W[ct]>0.2).values & ~tumor
  obs=D[sel].mean(); base=D[~tumor].mean()
  u,p=scipy.stats.mannwhitneyu(D[sel],D[(~tumor)&(~sel)],alternative='less')
  print(f'{ct:20s} mean={obs:.3f} base={base:.3f} p={p:.4g}')
PY"
