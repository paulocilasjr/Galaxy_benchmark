/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, scipy.spatial
pos=pd.read_csv('data/tissue_positions.csv', index_col=0)
coords=pos[['array_row','array_col']].values.astype(float)
cts=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
C=pd.read_csv('map_compositions.csv'); C.columns=cts
tumor=C['Tumor_Core'].values>0; tcoords=coords[tumor]
D=scipy.spatial.distance.cdist(coords,tcoords).min(axis=1)
# bins of D, rate presence among nonT spots
bins=[0,1.001,1.501,2.001,2.501,3.501,5.001,8,15,30]
labels=[f'{bins[i]:.1f}-{bins[i+1]:.1f}' for i in range(len(bins)-1)]
print('bin counts',pd.cut(D[~tumor],bins=bins).value_counts().sort_index())
for ct in cts[:-1]:
 p=(C[ct]>0).values
 rates=[]
 for i in range(len(bins)-1):
  mask=(D>bins[i])&(D<=bins[i+1])&~tumor
  rates.append(p[mask].mean() if mask.sum() else np.nan)
 print(f'{ct:20s}', ' '.join(f'{r:.2f}' if not np.isnan(r) else ' NA ' for r in rates))
PY"
