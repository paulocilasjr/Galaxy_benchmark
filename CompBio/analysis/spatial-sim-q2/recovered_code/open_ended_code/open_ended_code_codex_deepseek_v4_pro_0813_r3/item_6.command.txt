import scipy.io, scipy.sparse, gzip, numpy as np, pandas as pd
mat=scipy.io.mmread('data/matrix.mtx.gz')
print(type(mat), mat.shape, mat.nnz)
if scipy.sparse.issparse(mat): mat=mat.tocsr()
# spot sums
sums=np.asarray(mat.sum(axis=1)).ravel()
print('spot sums min max mean median', sums.min(), sums.max(), sums.mean(), np.median(sums))
# gene sums
gs=np.asarray(mat.sum(axis=0)).ravel()
print('gene sums min max mean median', gs.min(), gs.max(), gs.mean(), np.median(gs))
# positions grid
pos=pd.read_csv('data/tissue_positions.csv')
print(pos[['array_row','array_col']].agg(['min','max']))
print(pos[['pxl_row_in_fullres','pxl_col_in_fullres']].agg(['min','max']))
PY"
