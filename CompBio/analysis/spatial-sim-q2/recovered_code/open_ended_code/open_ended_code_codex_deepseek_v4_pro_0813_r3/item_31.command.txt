/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
P=pd.read_csv('nnls_weights.csv'); P.columns=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
S=pd.read_csv('marker_scores.csv',index_col=0); S.index=pd.read_csv('data/barcodes.tsv.gz',header=None,compression='gzip')[0]
# marker tumor threshold top quantiles
for q in [0.8,0.9,0.93,0.95]:
 thr=np.quantile(S['Tumor_Core'],q); mask=S['Tumor_Core']>thr
 print('marker q',q,'n',mask.sum(),'overlap with nnls tumor', (mask & (P['Tumor_Core']>0.2)).sum())
# nnls tumor score distribution
print(pd.DataFrame({'marker_tumor':S['Tumor_Core'],'nnls_tumor':P['Tumor_Core']>0.2}).groupby('nnls_tumor')['marker_tumor'].describe())
# correlation tumor marker with NNLS weight
print('corr',np.corrcoef(S['Tumor_Core'],P['Tumor_Core'])[0,1])
# top marker tumor map
PY"
