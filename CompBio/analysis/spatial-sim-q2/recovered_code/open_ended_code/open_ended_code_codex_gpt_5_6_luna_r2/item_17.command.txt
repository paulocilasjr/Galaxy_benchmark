/bin/bash -lc "python3 -u - <<'PY'
import numpy as np,csv
from scipy.io import mmread
from scipy.optimize import nnls
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001))
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels));
S=np.array([sc[labels==t].mean(0) for t in types])
sp=mmread('data/matrix.mtx.gz').toarray()
X=S.T
coef=[]; res=[]
for j in range(sp.shape[1]):
 c,r=nnls(X,sp[:,j]); coef.append(c);res.append(r)
coef=np.array(coef); res=np.array(res)
for i,t in enumerate(types):
 print(t,'coef min median max sum',*(round(float(x),3) for x in [coef[:,i].min(),np.median(coef[:,i]),coef[:,i].max(),coef[:,i].sum()]))
props=coef/(coef.sum(1,keepdims=True)+1e-9)
pa=np.argmax(props,axis=1)
print('nnls dominant assignments',[(types[i],int((pa==i).sum())) for i in range(len(types))])
for i,t in enumerate(types):
 ii=np.argsort(props[:,i])[::-1][:10]
 print(t,'top',[(int(j),round(float(coef[j,i]),3),round(float(props[j,i]),3),round(float(sp[:,j].sum()),0)) for j in ii])
PY"
