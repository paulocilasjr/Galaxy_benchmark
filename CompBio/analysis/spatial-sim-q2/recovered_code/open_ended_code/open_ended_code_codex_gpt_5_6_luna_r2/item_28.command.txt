/bin/bash -lc "python3 -u - <<'PY'
import numpy as np,csv
from scipy.io import mmread
from scipy.optimize import nnls
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001))
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels)); S=np.array([sc[labels==t].mean(0) for t in types]); sp=mmread('data/matrix.mtx.gz').toarray(); blocks=[(0,50),(50,100),(100,150),(150,200),(200,250),(250,300)]; X=np.array([[S[i,a:b].sum() for i in range(6)] for a,b in blocks]);Y=np.array([sp[a:b].sum(0) for a,b in blocks]).T; C=np.array([nnls(X,Y[j])[0] for j in range(900)]); cr=np.rint(C).astype(int)
with open('data/tissue_positions.csv') as f: pos=list(csv.DictReader(f)); xy=np.array([[int(p['array_row']),int(p['array_col'])] for p in pos],float)
for i,t in enumerate(types):
 w=C[:,i] # counts continuous
 print(t,'n>0.5',sum(w>.5),'sum',w.sum(),'centroid',np.average(xy,axis=0,weights=w).round(2),'weighted sd',np.sqrt(np.average((xy-np.average(xy,axis=0,weights=w))**2,axis=0,weights=w)).round(2),'corr row,col',np.corrcoef(w,xy[:,0])[0,1].round(3),np.corrcoef(w,xy[:,1])[0,1].round(3))
# count number types / cell counts distribution
print('total cells per spot',[(n,int((cr.sum(1)==n).sum())) for n in sorted(set(cr.sum(1)))])
print('all type exact counts',[(t,{n:int((cr[:,i]==n).sum()) for n in range(0,5)}) for i,t in enumerate(types)])
PY"
