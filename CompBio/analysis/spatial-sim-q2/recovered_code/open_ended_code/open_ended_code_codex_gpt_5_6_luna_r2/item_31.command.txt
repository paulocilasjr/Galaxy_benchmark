/bin/bash -lc "python3 -u - <<'PY'
import numpy as np,csv
from scipy.io import mmread
from scipy.optimize import nnls
from scipy.stats import fisher_exact
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001));
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels)); S=np.array([sc[labels==t].mean(0) for t in types]); sp=mmread('data/matrix.mtx.gz').toarray(); blocks=[(0,50),(50,100),(100,150),(150,200),(200,250),(250,300)]; X=np.array([[S[i,a:b].sum() for i in range(6)] for a,b in blocks]);Y=np.array([sp[a:b].sum(0) for a,b in blocks]).T; C=np.array([nnls(X,Y[j])[0] for j in range(900)]);
with open('data/tissue_positions.csv') as f: pos=list(csv.DictReader(f)); xy=np.array([[int(p['array_row']),int(p['array_col'])] for p in pos],float); D=np.sqrt(((xy[:,None]-xy[None,:])**2).sum(2)); tum=C[:,types.index('Tumor_Core')]>0.5; d=D[:,tum].min(1)
print('tumor',tum.sum())
for R in [0,1,1.5,2,2.5,3,4,5,6,8,10]:
 near=d<=R
 print('\\nR',R,'near spots',near.sum())
 for t in ['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell']:
  i=types.index(t); pr=C[:,i]>.5
  a=(pr&near).sum(); b=((~pr)&near).sum(); c=(pr&~near).sum(); e=((~pr)&~near).sum();
  odds,p=fisher_exact([[a,b],[c,e]],alternative='greater')
  print(t,'near frac',round(a/near.sum(),3),'far frac',round(c/(~near).sum(),3),'OR',round(odds,2),'p',f'{p:.2g}',end='; ')
 print()
PY"
