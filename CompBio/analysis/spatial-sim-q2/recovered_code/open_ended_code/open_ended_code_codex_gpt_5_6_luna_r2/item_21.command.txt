/bin/bash -lc "python3 -u - <<'PY'
import numpy as np,csv
from scipy.io import mmread
from scipy.optimize import nnls
from scipy.stats import mannwhitneyu,ttest_ind,permutation_test
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001))
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels)); S=np.array([sc[labels==t].mean(0) for t in types]); sp=mmread('data/matrix.mtx.gz').toarray(); coef=np.array([nnls(S.T,sp[:,j])[0] for j in range(sp.shape[1])]);
with open('data/tissue_positions.csv') as f: pos=list(csv.DictReader(f))
xy=np.array([[int(p['array_row']),int(p['array_col'])] for p in pos],float)
# distance from each spot to tumor-positive spots
tum=coef[:,types.index('Tumor_Core')]>0.8
D=np.sqrt(((xy[:,None,:]-xy[None,:,:])**2).sum(2)); d=D[:,tum].min(1)
print('tumor n',tum.sum(),'coords',xy[tum].astype(int).tolist())
print('distance global',[(x,int((d==x).sum())) for x in sorted(set(d))[:10]])
for th in [.05,.1,.2,.5,.8]:
 print('\\nthreshold',th)
 for i,t in enumerate(types):
  if t=='Tumor_Core': continue
  present=coef[:,i]>th
  vals=d[present]
  others=d[~present]
  mw=mannwhitneyu(vals,others,alternative='less')
  # mean/permutation? 
  print(t,'n',present.sum(),'mean',round(vals.mean(),3),'median',np.median(vals),'q25/75',np.quantile(vals,[.25,.75]).round(2),'not mean',round(others.mean(),3),'MW p',mw.pvalue,'U',mw.statistic)
# spot type dominant only, distance
print('\\nDominant type among nnls, tumor spots dominant only')
dom=np.argmax(coef,axis=1)
tumd=dom==types.index('Tumor_Core'); dd=D[:,tumd].min(1)
for i,t in enumerate(types):
 if t=='Tumor_Core': continue
 vals=dd[dom==i]; oth=dd[dom"'!=i]
 print(t,'"'n',len(vals),'mean',vals.mean() if len(vals) else 0,'med',np.median(vals) if len(vals) else 0,'p',mannwhitneyu(vals,oth,alternative='less').pvalue)
PY"
