/bin/bash -lc "python3 -u - <<'PY'
import numpy as np,csv
from scipy.io import mmread
from scipy.optimize import nnls
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001));
with open('data/spatial_q_sc_metadata.csv') as f: rows=list(csv.reader(f))[1:]
labels=np.array([r[1] for r in rows]); types=sorted(set(labels)); S=np.array([sc[labels==t].mean(0) for t in types]); sp=mmread('data/matrix.mtx.gz').toarray(); C=np.array([nnls(S.T,sp[:,j])[0] for j in range(900)]); 
with open('data/tissue_positions.csv') as f: pos=list(csv.DictReader(f)); xy=np.array([[int(p['array_row']),int(p['array_col'])] for p in pos],float)
D=np.sqrt(((xy[:,None]-xy[None,:])**2).sum(2)); ti=types.index('Tumor_Core'); tum=C[:,ti]>.8
for i,t in enumerate(types):
 pr=C[:,i]>.8; overlap=pr&tum
 dd=D[:,tum].min(1)
 dnon=[]
 for j in np.where(pr)[0]:
  inds=np.where(tum & (np.arange(900)"'!=j))[0]
  dnon.append(D[j,inds].min() if len(inds) else np.nan)
 dnon=np.array(dnon)
 print(t,'"'n',pr.sum(),'overlap tumor',overlap.sum(),'mean d',round(dd[pr].mean(),3),'mean nonself',round(dnon.mean(),3),'med',np.median(dd[pr]),'fraction d<=1',round(np.mean(dd[pr]<=1),3),'frac overlap',round(np.mean(overlap[pr]),3))
PY"
