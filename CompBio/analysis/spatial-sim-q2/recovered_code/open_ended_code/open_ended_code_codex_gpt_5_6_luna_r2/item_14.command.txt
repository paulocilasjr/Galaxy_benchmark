import numpy as np,csv
from scipy.io import mmread
sc=np.loadtxt('data/spatial_q_sc_counts.csv',delimiter=',',skiprows=1,usecols=range(1,1001))

 other=nmeans[np.arange(len(labels))"'!=i].mean(0)
 score=np.log2((nmeans[i]+.01)/(other+.01))

 print('"'\\n',t,'mean lib',sc[types==t].sum(1).mean(),'top:',[(f'Gene_{j}',round(float(nmeans[i,j]),1),round(float(score[j]),2)) for j in idx])
print('\\nmeans first 80')
for j in range(80):
 print(j, [round(x,2) for x in nmeans[:,j]])
