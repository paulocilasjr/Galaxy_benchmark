/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
from scipy.optimize import nnls
sc=pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
ctypes=meta['cell_type'].astype('category').cat.categories.tolist()
sig=np.array([sc.values[meta['cell_type']==t].mean(axis=0) for t in ctypes])
X=mmread('matrix.mtx.gz').T.toarray()
W=np.array([nnls(sig.T,x)[0] for x in X])
w=W[:,ctypes.index('Tumor_Core')]
print(pd.Series(w).describe())
print('value counts rounded')
print(pd.Series(np.rint(w)).value_counts())
print('nonzero fractional counts: w>0.1 not rounded')
print(pd.Series(w[w>0.01]).value_counts(bins=20).head())
# possible tumor spots with w 0.5-1.5
print('w between .4 and 1.5', np.where((w>0.4)&(w<1.5))[0][:20])
PY"
