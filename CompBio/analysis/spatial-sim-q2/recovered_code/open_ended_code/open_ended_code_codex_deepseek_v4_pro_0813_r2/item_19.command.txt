/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.optimize import nnls
from sklearn.metrics import accuracy_score, confusion_matrix
sc=pd.read_csv('spatial_q_sc_counts.csv', index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)
ctypes=meta['cell_type'].astype('category').cat.categories.tolist()
sig=np.array([sc.values[meta['cell_type']==t].mean(axis=0) for t in ctypes])
# leave-one? classify each sc cell by nearest signature correlation / NNLS? 
pred=[]
for i in range(len(sc)):
    x=sc.values[i]
    # NNLS weights with signatures excluding? just all
    w=nnls(sig.T,x)[0]
    pred.append(ctypes[np.argmax(w)])
acc=accuracy_score(meta['cell_type'], pred)
print('single-cell NNLS max weight accuracy', acc)
print(confusion_matrix(meta['cell_type'], pred, labels=ctypes))
# correlations
corr=sc.values @ sig.T
pred2=np.argmax(corr,axis=1)
print('corr accuracy', accuracy_score(meta['cell_type'], ctypes[pred2]))
PY"
