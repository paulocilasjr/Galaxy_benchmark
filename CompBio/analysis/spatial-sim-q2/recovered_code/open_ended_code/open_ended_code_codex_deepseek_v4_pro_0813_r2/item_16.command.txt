import pandas as pd, numpy as np
from scipy.io import mmread
from scipy.stats import zscore

meta=pd.read_csv('spatial_q_sc_metadata.csv', index_col=0)

means=sc.groupby(meta['cell_type']).mean()
# top genes per type by mean relative specificity

    specificity=(means.loc[t] - means.drop(index=t).mean(axis=0))/ (means.loc[t]+1)
    top=specificity.sort_values(ascending=False).head(5)
    print('\\n',t, top.index.tolist(), [round(v,2) for v in top.values])
