/bin/bash -lc "python3 - <<'PY'
import pandas as pd,numpy as np
from scipy.io import mmread
from scipy.spatial.distance import cdist
from scipy.stats import spearmanr
comp=pd.read_csv('deconv_composition.csv')
coords=comp[['array_row','array_col']].values; tm=comp['Tumor_Core']>0; D=cdist(coords,coords[tm]); nearest=D.min(1)
genes=pd.read_csv('features.tsv.gz',sep='\\t',header=None)[1].tolist(); X=mmread('matrix.mtx.gz').T.toarray()
# top marker gene lists from prior block
markers={'Tumor_Core':list(range(0,50)),'Fibroblast_Stroma':list(range(50,100)),'T_Cell':list(range(100,150)),'B_Cell':list(range(150,200)),'Macrophage':list(range(200,250)),'Endothelial':list(range(250,300))}
for t, idxs in markers.items():
    corrs=[]
    for g in idxs:
        r,_=spearmanr(nearest, X[:,g]); corrs.append((r,genes[g]))
    corrs_sorted=sorted(corrs)
    print('\\n',t,'num neg',sum(c<0 for c,_ in corrs),'mean rho',np.mean([c for c,_ in corrs]),'top neg',corrs_sorted[:5],'top pos',corrs_sorted[-3:])
PY"
