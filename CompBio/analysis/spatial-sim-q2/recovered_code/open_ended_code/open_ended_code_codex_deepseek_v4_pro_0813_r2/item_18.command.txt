/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
from scipy.stats import fisher_exact
comp=pd.read_csv('deconv_composition.csv')
ctypes=['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell','Tumor_Core']
coords=comp[['array_row','array_col']].values
tm=comp['Tumor_Core']>0; D=cdist(coords,coords[tm]); nearest=D.min(1)
for cutoff in [1,2,3,4,5,6,7,8]:
    near=nearest<=cutoff
    print('\\ncutoff',cutoff,'near spots',near.sum(),'far',(~near).sum())
    for t in ctypes:
        if t=='Tumor_Core': continue
        a=((comp[t]>0)&near).sum(); b=((comp[t]>0)&(~near)).sum(); c=((comp[t]==0)&near).sum(); d=((comp[t]==0)&(~near)).sum()
        # presence ratio near vs far
        pnear=a/near.sum(); pfar=b/(~near).sum()
        odds,p=fisher_exact([[a,b],[c,d]])
        # mean counts
        mn=comp.loc[near,t].mean(); mf=comp.loc[~near,t].mean()
        print(f'{t:20s} presence near={pnear:.3f} far={pfar:.3f} OR={odds:.3f} p={p:.2e} mean near={mn:.3f} far={mf:.3f}')
PY"
