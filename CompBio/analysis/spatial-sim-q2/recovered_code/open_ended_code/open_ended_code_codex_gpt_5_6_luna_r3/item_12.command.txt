/bin/bash -lc "python - <<'PY'
import pandas as pd,numpy as np,gzip,scipy.io as sio
S=sio.mmread(gzip.open('matrix.mtx.gz','rt')).tocsr()
# avoid all? 
tot=np.asarray(S.sum(axis=0)).ravel()
print('unique totals',len(np.unique(tot)))
print(np.quantile(tot,[0,.01,.05,.1,.2,.3,.4,.5,.6,.7,.8,.9,.95,.99,1]))
for k in range(1,7): print(k,((tot>=(k-0.45)*1242)&(tot<(k+0.45)*1242)).sum())
# display sorted totals
print(np.sort(tot)[:50]);print(np.sort(tot)[-50:])
# spot total grid by row col rough
pos=pd.read_csv('tissue_positions.csv')
grid=np.full((30,30),0.)
for i,p in pos.iterrows():grid[int(p.array_row),int(p.array_col)]=tot[i]/1242
for row in grid: print(' '.join(f'{x:.1f}' for x in row))
PY"
