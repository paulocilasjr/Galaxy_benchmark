/bin/bash -lc "python -u - <<'PY'
import pandas as pd,numpy as np
from scipy.io import mmread
from scipy.optimize import nnls
sc=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
meta=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0).iloc[:,0]
X=mmread('matrix.mtx.gz').toarray().T # spots x genes
cts=sorted(meta.unique()); A=np.column_stack([sc.loc[meta.eq(c)].mean().values for c in cts])
co=np.array([nnls(A,x)[0] for x in X])
np.set_printoptions(precision=3,suppress=True)
print('types',cts)
print('coeff quantiles')
for j,c in enumerate(cts): print(c,np.quantile(co[:,j],[0,.05,.25,.5,.75,.95,1]))
print('sum coef q',np.quantile(co.sum(1),[0,.05,.25,.5,.75,.95,1]))
print('distance each coeff to integer',np.quantile(np.abs(co-np.round(co)),[0,.25,.5,.75,.9,.95,.99,1]))
print('rounded sum distribution',dict(zip(*np.unique(np.rint(co).astype(int).sum(1),return_counts=True))))
print('raw lib sorted gaps/classes')
libs=X.sum(1)
for n in range(1,6):
 m=np.rint(co.sum(1)).astype(int)==n
 print(n,m.sum(),libs[m].min(),np.quantile(libs[m],[.1,.5,.9]),libs[m].max())
print('round residual max per spot q',np.quantile(np.max(abs(co-np.rint(co)),axis=1),[0,.25,.5,.75,.9,.95,.99,1]))
print('presence spot counts',dict(zip(cts,(co>.4).sum(0))))
PY"
