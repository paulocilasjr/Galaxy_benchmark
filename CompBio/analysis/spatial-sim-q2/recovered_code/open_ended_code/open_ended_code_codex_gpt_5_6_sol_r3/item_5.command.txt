/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
m=pd.read_csv('spatial_q_sc_metadata.csv',index_col=0)
print(m.cell_type.value_counts())
x=pd.read_csv('spatial_q_sc_counts.csv',index_col=0)
print(x.shape)
means=x.groupby(m.cell_type).mean()
print('\\nTop genes by type / specificity:')
for t in means.index:
    other=means.drop(t).mean()
    score=(means.loc[t]+.1)/(other+.1)
    print(t, list(score.nlargest(12).items()))
PY"
