/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
R=pd.read_csv('deconv_raw.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[R.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
tidx=np.where(R['Tumor_Core'].values>0.85)[0]
d=cdist(coords,coords[tidx]).min(axis=1)
C=pd.DataFrame(index=R.index)
for ct in R.columns:
    C[ct]=np.where(R[ct].values>0.5,np.round(R[ct].values),0).astype(int)
for radius in [100,150,200,250,300]:
    near=d<=radius
    print('\\nradius',radius,'near spots',near.sum())
    for ct in [c for c in C.columns if c"'!='"'Tumor_Core']:
        n_near=C.loc[near,ct].sum(); n_far=C.loc[~near,ct].sum()
        cells_near=C[ct].sum()
        total_cells=C[[c for c in C.columns if c"'!='"'Tumor_Core']].values.sum()
        total_near=C.loc[near,[c for c in C.columns if c"'!='"'Tumor_Core']].values.sum()
        # enrichment = (n_near / cells_near) / (total_near/total_cells) maybe
        enrich=(n_near/cells_near)/(total_near/total_cells) if total_near else np.nan
        print(f'{ct}: near={n_near} far={n_far} total={cells_near} near_all_non={total_near} total_non={total_cells} enrich={enrich:.2f}')
PY"
