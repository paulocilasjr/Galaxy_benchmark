/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
sp=mmread('matrix.mtx.gz').toarray() # genes x spots
blocks=[('Tumor_Core',0,50),('Fibroblast_Stroma',50,100),('T_Cell',100,150),('B_Cell',150,200),('Macrophage',200,250),('Endothelial',250,300)]
S=pd.DataFrame(index=pd.read_csv('barcodes.tsv.gz',sep='\\t',header=None)[0],columns=[b[0] for b in blocks])
for name,start,end in blocks:
    S[name]=sp[start:end,:].sum(axis=0)
print(S.head())
print(S.describe())
# normalized each spot by total marker sum
Ssum=S.sum(axis=1)
print('total marker range', Ssum.describe())
P=S.div(Ssum,axis=0)
print(P.describe())
# dominant
print(P.idxmax(axis=1).value_counts())
P.to_csv('marker_proportions.csv')
PY"
