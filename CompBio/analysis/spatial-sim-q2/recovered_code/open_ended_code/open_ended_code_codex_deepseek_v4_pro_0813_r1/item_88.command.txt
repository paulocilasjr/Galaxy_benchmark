/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,re
repo='amanutej/trustworthy-biology-agents-traces'
url='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/traces/compbiobench/compbiobench?recursive=true&expand=true&limit=100'
found=[];seen=0
while url:
 req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
 resp=urllib.request.urlopen(req,timeout=30)
 data=json.load(resp)
 for f in data:
  p=f.get('path','')
  if '/spatial-sim-q2/' in p or p.endswith('/spatial-sim-q2'):
   found.append((f['type'],p,f.get('size')))
 seen+=len(data)
 link=resp.headers.get('Link'); url=None
 if link:
  for part in link.split(','):
   if 'rel=\"next\"' in part:
    m=re.search(r'<(["'^>]+)>'"',part)
    if m: url=m.group(1)
 if seen>5000: break
print('seen',seen,'found',len(found))
for x in found: print(x)
PY"
