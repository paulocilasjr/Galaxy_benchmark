/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for lim in [100,200,500,1000]:
 url=f'https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true&expand=true&limit={lim}'
 try:
  data=json.load(urllib.request.urlopen(url))
  print('lim',lim,'entries',len(data))
  if any('spatial.sim' in f['path'] for f in data):
   print('found')
   for f in data:
    if 'spatial' in f['path'].lower() or 'tar.gz' in f['path'].lower(): print(f['path'],f['type'])
   break
 except Exception as e: print('ERR',lim,e)
PY"
