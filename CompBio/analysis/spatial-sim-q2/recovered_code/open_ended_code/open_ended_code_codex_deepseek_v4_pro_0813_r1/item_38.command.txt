/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
C=pd.read_csv('mle_counts.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[C.index,['pxl_row_in_fullres','pxl_col_in_fullres']].values
tidx=np.where(C['Tumor_Core'].values>0)[0]
print('tumor spots',len(tidx))
d=cdist(coords,coords[tidx]).min(axis=1)
types=[c for c in C.columns if c"'!='"'Tumor_Core']
# cell-level average distance
all_counts=np.zeros(len(C),dtype=int)
for ct in types: all_counts += C[ct].values
instance_d=np.repeat(d,all_counts)
for ct in types:
    counts=C[ct].values
    n=counts.sum(); mean=np.sum(counts*d)/n
    # random sample n cells among all non-tumor instances
    rng=np.random.default_rng(42)
    sims=np.array([instance_d[rng.choice(len(instance_d),size=n,replace=False)].mean() for _ in range(5000)])
    p=(np.sum(sims<=mean)+1)/(len(sims)+1); z=(mean-sims.mean())/sims.std()
    print(ct,'n',n,'mean',mean,'all',instance_d.mean(),'z',z,'p',p)
PY"
