/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
for query in ['spatial-sim-q2','spatial.sim.tar.gz','\"Fibroblast_Stroma\" \"Macrophage\" spatial-sim']:
    url='https://huggingface.co/api/datasets?search='+urllib.parse.quote(query)
    try:
        data=json.load(urllib.request.urlopen(url,timeout=30))
        print('QUERY',query,'results',len(data))
        for d in data[:10]: print(d['id'])
    except Exception as e: print('ERR',e)
PY"
