/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,urllib.parse
# list dataset files
for ds in ['amanutej/trustworthy-biology-agents-traces']:
 url=f'https://huggingface.co/api/datasets/{ds}/tree/main?recursive=true&expand=true&limit=100'
 try:
  data=json.load(urllib.request.urlopen(url,timeout=30))
  print(ds,'len',len(data))
  for f in data:
   p=f['path']
   if 'spatial' in p.lower(): print(f['type'],p,f.get('size'))
 except Exception as e: print('ERR',e)
PY"
