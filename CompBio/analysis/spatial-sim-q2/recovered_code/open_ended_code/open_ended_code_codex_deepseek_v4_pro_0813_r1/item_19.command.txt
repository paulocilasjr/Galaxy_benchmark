/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
R=pd.read_csv('deconv_raw.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv')
pos.index=pos['barcode']
# assign grid coords
for ct in ['Tumor_Core']+[c for c in R.columns if c"'!='"'Tumor_Core']:
    arr=np.full((30,30),'.',dtype='<U1')
    for bc in R.index:
        i=int(pos.loc[bc,'array_row']); j=int(pos.loc[bc,'array_col'])
        if ct=='Tumor_Core':
            val=R.loc[bc,ct]>0.85
        else:
            val=R.loc[bc,ct]>0.5
        if val: arr[i,j]='X' if ct=='Tumor_Core' else 'o'
    print('\\n',ct)
    for i in range(30): print(''.join(arr[i,j] for j in range(30)))
PY"
