/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for repo in ['Genentech/compbiobench-evaluator-v1']:
 for api in [f'https://huggingface.co/api/spaces/{repo}/tree/main?recursive=true&expand=true', f'https://huggingface.co/api/spaces/{repo}/tree/main']:
  try:
   req=urllib.request.Request(api,headers={'User-Agent':'Mozilla/5.0'})
   data=json.load(urllib.request.urlopen(req,timeout=30))
   print(api,'len',len(data) if isinstance(data,list) else data)
   if isinstance(data,list):
    for f in data:
     print(f.get('type'),f.get('path'),f.get('size'))
  except Exception as e: print('ERR',api,e)
PY"
