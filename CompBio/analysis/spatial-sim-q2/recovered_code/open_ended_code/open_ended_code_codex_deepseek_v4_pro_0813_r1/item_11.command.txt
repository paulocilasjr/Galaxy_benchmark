/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
P=pd.read_csv('deconv_proportions.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv',index_col='barcode')
pos=pos.loc[P.index]
for ct in P.columns:
    print('\\n',ct,'nonzeros',np.sum(P[ct]>0), 'mean among >0',P.loc[P[ct]>0,ct].mean(), 'quantiles',P[ct].quantile([0,.5,.75,.9,.95,.99]).round(3).to_dict())
print('\\nTumor_Core positive count distribution')
print(P['Tumor_Core'][P['Tumor_Core']>0].describe())
# for each ct correlation of proportion with row/col
for ct in P.columns:
    print(ct,'corr row/col', np.corrcoef(P[ct],pos['pxl_row_in_fullres'])[0,1].round(3), np.corrcoef(P[ct],pos['pxl_col_in_fullres'])[0,1].round(3))
PY"
