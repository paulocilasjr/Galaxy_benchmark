/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.spatial.distance import cdist
from scipy.stats import fisher_exact, mannwhitneyu
C=pd.read_csv('mle_counts.csv',index_col=0)
pos=pd.read_csv('tissue_positions.csv').set_index('barcode')
coords=pos.loc[C.index,['array_row','array_col']].values.astype(float)
dom=C.idxmax(axis=1).values
target=dom=='Tumor_Core'
d=cdist(coords,coords[target]).min(axis=1)
print('target',target.sum())
for ct in ['B_Cell','Endothelial','Fibroblast_Stroma','Macrophage','T_Cell']:
 m=dom==ct
 u,p=mannwhitneyu(d[m],d[~m],alternative='less')
 print(ct,'n',m.sum(),'mean',d[m].mean(),'median',np.median(d[m]),'MWp',p)
 for R in [1,2,3,4,5]:
  near=d<=R
  table=[[(m&near).sum(),(m&~near).sum()],[((~m)&near).sum(),((~m)&~near).sum()]]
  odds,fp=fisher_exact(table, alternative='greater')
  print(' R',R,'odds',round(odds,3),'p',fp)
PY"
