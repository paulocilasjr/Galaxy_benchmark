/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
repo='amanutej/trustworthy-biology-agents-traces'
base=f'https://huggingface.co/api/datasets/{repo}/tree/main?recursive=true&expand=true&limit=100'
url=base
count=0
while url:
    req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
    data=json.load(urllib.request.urlopen(req,timeout=30))
    if isinstance(data,dict): print('dict?',data); break
    print('batch len',len(data))
    for f in data:
        p=f.get('path','')
        if 'spatial' in p.lower(): print(f['type'],p,f.get('size'))
    count+=len(data)
    # next cursor
    url=None
    # HuggingFace API uses Link header? not easy. try inspect? no.
    # Use sibling? break after 100 for now
    break
print('count',count)
PY"
