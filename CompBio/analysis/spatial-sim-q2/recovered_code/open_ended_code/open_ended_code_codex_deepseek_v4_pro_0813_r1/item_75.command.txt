/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, urllib.parse
repo='amanutej/trustworthy-biology-agents-traces'
url=f'https://huggingface.co/api/datasets/{repo}/tree/main?recursive=true&expand=true&limit=100'
seen=0
while url:
    req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
    resp=urllib.request.urlopen(req,timeout=30)
    data=json.load(resp)
    print('batch',len(data), 'url?', url[-80:])
    for f in data:
        p=f.get('path','')
        if 'spatial' in p.lower():
            print('FOUND',f['type'],p,f.get('size'))
    seen+=len(data)
    link=resp.headers.get('Link')
    next_url=None
    if link:
        # parse rel next
        import re
        parts=link.split(',')
        for part in parts:
            if 'rel=\"next\"' in part:
                m=re.search(r'<(["'^>]+)>'"',part)
                if m: next_url=m.group(1)
    url=next_url
    if seen>1000: break
print('seen',seen)
PY"
