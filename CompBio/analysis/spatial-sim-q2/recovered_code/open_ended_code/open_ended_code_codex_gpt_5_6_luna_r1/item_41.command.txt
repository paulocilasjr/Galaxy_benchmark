/bin/bash -lc "python - <<'PY'
import json,urllib.request
for doi in ['19443186','19443185']:
 try:
  d=json.load(urllib.request.urlopen('https://zenodo.org/api/records/'+doi,timeout=60))
  print(doi,d.get('title'),'files',len(d.get('files',[])))
  for f in d.get('files',[]): print(' ',f.get('key'),f.get('size'))
 except Exception as e: print('ERR',doi,repr(e))
PY"
