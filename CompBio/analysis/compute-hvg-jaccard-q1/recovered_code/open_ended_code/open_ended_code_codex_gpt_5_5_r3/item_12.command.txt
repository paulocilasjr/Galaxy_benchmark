/bin/bash -lc ". .venv/bin/activate && python - <<'PY'
import scanpy as sc
for path in ['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']:
    adata = sc.read_h5ad(path, backed='r')
    print(path)
    print('shape', adata.shape)
    print('var_names first5', list(map(str, adata.var_names[:5])))
    print('var columns', list(adata.var.columns))
    if 'gene_symbols' in adata.var.columns:
        s = adata.var['gene_symbols']
        print('gene_symbols missing', int(s.isna().sum()), 'empty', int((s.astype(str) == '').sum()))
        print('gene_symbols first5', list(map(str, s.iloc[:5])))
    print('var index unique', adata.var_names.is_unique)
PY"
