/bin/bash -lc ". .venv/bin/activate && python - <<'PY'
import math
import warnings
import scanpy as sc

warnings.filterwarnings('ignore')
paths = ['pbmc3k.h5ad', '10x_pbmc68k_reduced.h5ad']
adatas = [sc.read_h5ad(p) for p in paths]

filtered = []
for a in adatas:
    if 'gene_symbols' in a.var.columns:
        mask = a.var['gene_symbols'].notna() & (a.var['gene_symbols'].astype(str) "'!= '"'')
        a = a[:, mask].copy()
    filtered.append(a)

common_set = set(map(str, filtered[0].var_names))
for a in filtered[1:]:
    common_set &= set(map(str, a.var_names))
common = [g for g in map(str, filtered[0].var_names) if g in common_set]

hvg_sets = []
truthy = ['True', '1', 'TRUE']
for p, a in zip(paths, filtered):
    a = a[:, common].copy()
    sc.pp.highly_variable_genes(a, flavor='seurat')
    hv = set(map(str, a.var_names[a.var['highly_variable'].astype(str).isin(truthy)]))
    hvg_sets.append(hv)
    print(p, 'shape_common', a.shape, 'hvg_count', len(hv))

union = hvg_sets[0] | hvg_sets[1]
if not common or not union:
    result = float('nan')
else:
    result = len(hvg_sets[0] & hvg_sets[1]) / len(union)
print('common_count', len(common))
print('intersection', len(hvg_sets[0] & hvg_sets[1]), 'union', len(union))
print('result_raw', result)
print('answer', 'NaN' if math.isnan(result) else f'{result:.2f}')
PY"
