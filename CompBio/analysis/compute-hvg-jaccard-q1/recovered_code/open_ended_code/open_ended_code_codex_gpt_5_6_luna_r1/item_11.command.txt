/bin/bash -lc "python -u - <<'PY'
import warnings
warnings.filterwarnings('ignore')
import scanpy as sc
import anndata
import numpy as np

print('scanpy',sc.__version__)
a=anndata.read_h5ad('pbmc3k.h5ad')
b=anndata.read_h5ad('10x_pbmc68k_reduced.h5ad')
# Exact string-valued var_names; retain common genes in a's order.
a_names=[str(x) for x in a.var_names]
b_names=set(str(x) for x in b.var_names)
common=[g for g in a_names if g in b_names]
print('shapes',a.shape,b.shape,'common',len(common))
a=a[:,common].copy()
b=b[:,common].copy()
print('subset',a.shape,b.shape,'var cols',list(a.var),list(b.var))
# No gene_symbols columns exist in these files; all exact var_names are nonmissing gene identifiers.
sc.pp.highly_variable_genes(a, flavor='seurat')
sc.pp.highly_variable_genes(b, flavor='seurat')
accepted={'True','1','TRUE'}
ha={str(v) in accepted for v in a.var['highly_variable']}
hb={str(v) in accepted for v in b.var['highly_variable']}
# above set is only for values, but calculate names explicitly
ha_names={str(g) for g,v in zip(a.var_names,a.var['highly_variable']) if str(v) in accepted}
hb_names={str(g) for g,v in zip(b.var_names,b.var['highly_variable']) if str(v) in accepted}
inter=ha_names & hb_names
union=ha_names | hb_names
print('a hvgs',len(ha_names),'b hvgs',len(hb_names),'intersection',len(inter),'union',len(union),'jaccard',len(inter)/len(union) if union else np.nan)
print('a true counts',a.var['highly_variable'].value_counts(dropna=False).to_dict())
print('b true counts',b.var['highly_variable'].value_counts(dropna=False).to_dict())
print('a hvg names',sorted(ha_names))
print('b hvg names',sorted(hb_names))
PY"
