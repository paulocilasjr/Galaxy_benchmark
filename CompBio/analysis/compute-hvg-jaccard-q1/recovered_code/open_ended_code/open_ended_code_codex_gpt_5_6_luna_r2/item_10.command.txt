/bin/bash -lc "python -u - <<'PY'
import warnings
warnings.filterwarnings('ignore')
import scanpy as sc
paths=['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']
ads=[sc.read_h5ad(p) for p in paths]
common=ads[0].var_names.intersection(ads[1].var_names)
sets=[]
for a in ads:
 a=a[:,common].copy()
 sc.pp.highly_variable_genes(a, flavor='seurat')
 vals=a.var['highly_variable'].astype(str)
 keep=vals.isin(list(dict.fromkeys(['True','True','1','TRUE'])))
 s=set(a.var_names[keep])
 sets.append(s)
 print(len(s), sorted(s))
i,j=sets
print('intersection',len(i&j), sorted(i&j))
print('union',len(i|j), sorted(i|j))
print('jaccard',len(i&j)/len(i|j) if i|j else float('nan'))
PY"
