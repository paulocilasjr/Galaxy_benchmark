import scanpy as sc
print('scanpy', sc.__version__)
for f in ['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']:
    a=sc.read_h5ad(f)
    print(f, 'shape', a.shape, 'var_names dtype', a.var_names.dtype, 'unique', a.var_names.is_unique)
    print('var columns', list(a.var.columns))
    print('var index first/last', list(a.var_names[:5]), list(a.var_names[-5:]))
    if 'gene_symbols' in a.var:
        print('gene_symbols missing', int(a.var['gene_symbols'].isna().sum()), 'unique', a.var['gene_symbols'].nunique(dropna=True), 'dtype', a.var['gene_symbols'].dtype)
a=sc.read_h5ad('pbmc3k.h5ad'); b=sc.read_h5ad('10x_pbmc68k_reduced.h5ad')
common=set(a.var_names).intersection(b.var_names)
print('common', len(common), 'a',a.n_vars,'b',b.n_vars)
print('common missing symbols a,b', int(a[:,list(common)].var['gene_symbols'].isna().sum()), int(b[:,list(common)].var['gene_symbols'].isna().sum()))
PY"
