/bin/bash -lc "python - <<'PY'
import scanpy as sc
import numpy as np

f1='pbmc3k.h5ad'; f2='10x_pbmc68k_reduced.h5ad'
a1=sc.read_h5ad(f1)
a2=sc.read_h5ad(f2)
print('initial', a1.shape, a2.shape)
# filter missing gene_symbols if present
for a in [a1,a2]:
    if 'gene_symbols' in a.var.columns:
        print('filtering missing', a.shape, a.var['gene_symbols'].isna().sum())
        a = a[:, ~a.var['gene_symbols'].isna()]
# common
common=np.intersect1d(np.asarray(a1.var_names, dtype=str), np.asarray(a2.var_names, dtype=str))
print('common len', len(common), common[:10])
a1=a1[:, common]
a2=a2[:, common]
print('sub shapes', a1.shape, a2.shape)
for name,a in [('pbmc',a1),('68k',a2)]:
    print('calling hvg', name)
    sc.pp.highly_variable_genes(a, flavor='seurat')
    v=a.var['highly_variable'].astype(str)
    allowed=set(['True','1','TRUE'])
    hv=a.var_names[v.isin(allowed)]
    print(name, 'n HVG', len(hv), 'of', a.n_vars)
    print(hv[:30].tolist())
    print('value counts', v.value_counts())
print('jaccard exact X', len(np.intersect1d(a1.var_names[a1.var['highly_variable'].astype(str).isin(['True','1','TRUE'])], a2.var_names[a2.var['highly_variable'].astype(str).isin(['True','1','TRUE'])]))/len(np.union1d(a1.var_names[a1.var['highly_variable'].astype(str).isin(['True','1','TRUE'])], a2.var_names[a2.var['highly_variable'].astype(str).isin(['True','1','TRUE'])])))
PY"
