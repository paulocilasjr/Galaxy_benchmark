/bin/bash -lc "python - <<'PY'
import scanpy as sc, numpy as np, warnings
warnings.filterwarnings('ignore')
f1='pbmc3k.h5ad'; f2='10x_pbmc68k_reduced.h5ad'
a1=sc.read_h5ad(f1)
a2=sc.read_h5ad(f2)
# raw to adata
r1=a1.raw.to_adata()
r2=a2.raw.to_adata()
print('raw shapes',r1.shape,r2.shape)
common=np.intersect1d(np.asarray(r1.var_names,dtype=str), np.asarray(r2.var_names,dtype=str))
print('common',len(common))
r1=r1[:,common].copy(); r2=r2[:,common].copy()
for name,a in [('pbmc_raw',r1),('68k_raw',r2)]:
 sc.pp.highly_variable_genes(a, flavor='seurat')
 v=a.var['highly_variable'].astype(str)
 hv=a.var_names[v.isin(['True','1','TRUE'])]
 print(name,'n',len(hv))
sets=[set(a.var_names[a.var['highly_variable'].astype(str).isin(['True','1','TRUE'])]) for a in [r1,r2]]
print('jaccard raw',len(sets[0]&sets[1])/len(sets[0]|sets[1]), len(sets[0]&sets[1]),len(sets[0]),len(sets[1]),len(sets[0]|sets[1]))
PY"
