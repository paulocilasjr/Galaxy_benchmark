import scanpy as sc, numpy as np, warnings
warnings.filterwarnings('ignore')
a1=sc.read_h5ad('pbmc3k.h5ad'); a2=sc.read_h5ad('10x_pbmc68k_reduced.h5ad')
# use current var_names intersection

for name,a in [('b1',b1),('b2',b2)]:
 sc.pp.highly_variable_genes(a, flavor='seurat')
 hv=set(a.var_names[a.var['highly_variable'].astype(str).isin(['True','1','TRUE'])])
 print(name,len(hv))
sets=[set(a.var_names[a.var['highly_variable'].astype(str).isin(['True','1','TRUE'])]) for a in [b1,b2]]
print(len(sets[0]&sets[1]), len(sets[0]|sets[1]), len(sets[0]&sets[1])/len(sets[0]|sets[1]))
