/bin/bash -lc "python -u - <<'PY'
import scanpy as sc, anndata, numpy as np, pandas as pd
files=['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']
a=[anndata.read_h5ad(f) for f in files]
common=a[0].var_names.intersection(a[1].var_names)
allowed=list(dict.fromkeys(['True','True','1','TRUE']))
sets=[]
for fn,x in zip(files,a):
 b=x[:,common].copy()
 sc.pp.highly_variable_genes(b, flavor='seurat')
 mask=b.var['highly_variable'].astype(str).isin(allowed)
 s=set(b.var_names[mask])
 sets.append(s)
 print(fn, 'n_vars',b.n_vars,'hvg_true_values',b.var.loc[mask,'highly_variable'].astype(str).value_counts().to_dict(),'n',len(s))
 print('all value strings',b.var['highly_variable'].astype(str).value_counts().to_dict())
print('intersection',len(sets[0]&sets[1]))
print('union',len(sets[0]|sets[1]))
print('jaccard',len(sets[0]&sets[1])/len(sets[0]|sets[1]))
print('overlap',sorted(sets[0]&sets[1]))
print('only1',sorted(sets[0]-sets[1]))
print('only2',sorted(sets[1]-sets[0]))
PY"
