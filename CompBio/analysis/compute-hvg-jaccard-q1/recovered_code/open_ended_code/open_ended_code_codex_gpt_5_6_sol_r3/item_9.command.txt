/bin/bash -lc "PYTHONPATH=.scanpy_deps NUMBA_DISABLE_JIT=1 python - <<'PY'
import scanpy as sc, numpy as np
from scipy import sparse
for f in ['pbmc3k.h5ad','10x_pbmc68k_reduced.h5ad']:
 a=sc.read_h5ad(f)
 print('\\nFILE',f,'shape',a.shape,'X type',type(a.X),'dtype',a.X.dtype)
 print('var columns',list(a.var.columns), 'obs columns', list(a.obs.columns))
 print('var_names first',a.var_names[:15].tolist())
 print('var_names last',a.var_names[-5:].tolist())
 print('unique',a.var_names.is_unique,'missing index strings',sum(x is None for x in a.var_names))
 for c in a.var.columns:
  print('VARCOL',c,'dtype',a.var[c].dtype,'na',int(a.var[c].isna().sum()),'first',a.var[c].head().tolist())
 x=a.X
 if sparse.issparse(x): vals=x.data
 else: vals=np.asarray(x).ravel()
 print('X finite',np.isfinite(vals).all(),'min',np.nanmin(vals),'max',np.nanmax(vals),'mean',np.nanmean(vals),'first',vals[:10])
 print('layers',list(a.layers.keys()),'raw',None if a.raw is None else a.raw.shape)
PY"
