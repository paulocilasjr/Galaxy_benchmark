/bin/bash -lc "set -o pipefail
for url in \\
  'https://api.ncbi.nlm.nih.gov/variation/v0/beta/refsnp/200533370' \\
  'https://api.ncbi.nlm.nih.gov/variation/v0/beta/refsnp/241311' \\
  'https://rest.ensembl.org/variation/human/rs200533370?content-type=application/json' \\
  'https://rest.ensembl.org/variation/human/rs241311?content-type=application/json'; do
  echo \"URL "'$url"
  curl -L --fail --silent --show-error "$url" | head -c 2000
  echo
 done'
