/bin/bash -lc "python3 - <<'PY'
import requests, json
term='NM_015041.3(CLUAP1):c.817C>T'
for db in ['clinvar']:
    r=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi', params={'db':db,'term':term,'retmode':'json'})
    print(db,r.status_code,r.text)
    if r.ok:
        ids=r.json()['esearchresult'].get('idlist',[])
        print('ids',ids)
        if ids:
            es=requests.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi', params={'db':db,'id':','.join(ids),'retmode':'json'})
            print(es.text[:8000])
PY"
