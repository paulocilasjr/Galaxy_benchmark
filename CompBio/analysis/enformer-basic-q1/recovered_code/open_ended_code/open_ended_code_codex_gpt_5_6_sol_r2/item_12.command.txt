/bin/bash -lc "python - <<'PY'
C=1536
# helpers
conv=lambda i,o,k:i*o*k+o
bn=lambda d:2*d
pool=lambda d:d*d
stem=conv(4,C//2,15)+bn(C//2)+conv(C//2,C//2,1)+pool(C//2)
print('stem',stem)
fs=[768,896,1024,1152,1280,1536]
p=C//2
tower=0
for d in fs:
 x=bn(p)+conv(p,d,5)+bn(d)+conv(d,d,1)+pool(d)
 print('tower',p,d,x)
 tower+=x;p=d
print('tower total',tower)
H=8;K=64;V=C//H;R=C//H
ln=2*C
mha=2*C*(K*H)+C*(V*H)+(V*H)*C+C + R*(K*H)+2*H*K
mlp=2*C + C*(2*C)+(2*C)+(2*C)*C+C
per=ln+mha+mlp # careful ln mha + MLP includes its LN
print('mha',mha,'mlp with ln',mlp,'per',per,'trans',11*per)
final=bn(C)+conv(C,2*C,1)
heads=(2*C+1)*(5313+1643)
print('final',final,'heads',heads)
print('trunk',stem+tower+11*per+final)
print('total',stem+tower+11*per+final+heads)
PY"
