import struct
path='enformer_saved/variables/variables.index'; data=open(path,'rb').read()

def read_varint(buf,pos):
    shift=0; result=0
    while True:
        b=buf[pos]; pos+=1
        result |= (b & 0x7f) << shift
        if not (b & 0x80): return result,pos
        shift+=7
pos=0; prev=b''; entries=[]
while pos < restart_base:
    shared,pos=read_varint(content,pos)
    nonshared,pos=read_varint(content,pos)
    vlen,pos=read_varint(content,pos)
    key=prev[:shared]+content[pos:pos+nonshared]; pos+=nonshared
    val=content[pos:pos+vlen]; pos+=vlen
    prev=key
    entries.append((key,val))
print('entries',len(entries),'endpos',pos)
for i,(k,v) in enumerate(entries[:20]):
    print(i, len(k), len(v), k[:120], v.hex()[:100])
print('last')
for i,(k,v) in enumerate(entries[-10:], start=len(entries)-10):
