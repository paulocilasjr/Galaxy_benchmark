/bin/bash -c "cd enformer-pytorch && python - <<'PY'
from enformer_pytorch.modeling_enformer import Enformer
from enformer_pytorch.config_enformer import EnformerConfig

config = EnformerConfig.from_hparams(
    dim = 1536,
    depth = 11,
    heads = 8,
    output_heads = dict(human = 5313, mouse = 1643),
    target_length = 896,
    use_tf_gamma = False,
)
model = Enformer(config)
total = sum(p.numel() for p in model.parameters() if p.requires_grad)
print(total)
for name,p in model.named_parameters():
    if p.requires_grad:
        pass
# print module groups
groups = {
 'stem': sum(p.numel() for n,p in model.stem.named_parameters()),
 'conv_tower': sum(p.numel() for n,p in model.conv_tower.named_parameters()),
 'transformer': sum(p.numel() for n,p in model.transformer.named_parameters()),
 'crop_final': sum(p.numel() for n,p in model.crop_final.named_parameters()),
 'final_pointwise': sum(p.numel() for n,p in model.final_pointwise.named_parameters()),
 'heads': sum(p.numel() for n,p in model._heads.named_parameters()),
}
print(groups)
# individual heads
for head,m in model._heads.items():
    print(head, sum(p.numel() for p in m.parameters()))
PY"
