/bin/bash -lc "python3 - <<'PY'
C=1536
filters=[768,896,1024,1152,1280,1536]
def lin(inp,out,bias=True): return (inp+(1 if bias else 0))*out
def conv(inp,out,w,bias=True): return (w*inp+(1 if bias else 0))*out
stem=conv(4,C//2,15)+2*(C//2)+conv(C//2,C//2,1)+(C//2)**2
tower=0; inp=C//2
for F in filters:
    tower += 2*inp+conv(inp,F,5)+2*F+conv(F,F,1)+F*F
    inp=F
mha=conv(C,8*64,1,bias=False)*2+conv(C,C,1,bias=False)+lin(C,C)+192*512+2*8*64
block=mha+2*C+lin(C,2*C)+lin(2*C,C)
trans=11*block
final=2*C+conv(C,2*C,1)
heads=lin(2*C,5313)+lin(2*C,1643)
total=stem+tower+trans+final+heads
print('stem',stem)
print('tower',tower)
print('mha',mha,'block',block,'trans',trans)
print('final',final)
print('heads human',lin(2*C,5313),'mouse',lin(2*C,1643),'total',heads)
print('TOTAL',total)
print(f'{total:,}')
PY"
