/bin/bash -lc "curl -L --fail -H 'Content-Type: text/plain' 'https://rest.ensembl.org/sequence/region/human/12:120196699..120201111:1?coord_system_version=GRCh38' -o downloads/ensembl_sequence.txt
curl -L --fail 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_000012.12&seq_start=120196699&seq_stop=120201111&strand=1&rettype=fasta&retmode=text' -o downloads/ncbi_sequence.fa
python3 - <<'PY'
import json, hashlib
from pathlib import Path
u=json.loads(Path('downloads/ucsc_sequence.json').read_text())['dna'].upper()
e=''.join(Path('downloads/ensembl_sequence.txt').read_text().split()).upper()
n=''.join(x.strip() for x in Path('downloads/ncbi_sequence.fa').read_text().splitlines() if not x.startswith('>')).upper()
for label,s in [('UCSC',u),('Ensembl',e),('NCBI',n)]:
 print(label,len(s),hashlib.sha256(s.encode()).hexdigest())
print('all_equal',u==e==n)
PY"
