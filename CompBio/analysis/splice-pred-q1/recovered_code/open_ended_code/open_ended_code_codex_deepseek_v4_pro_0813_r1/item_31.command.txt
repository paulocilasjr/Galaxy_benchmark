/bin/bash -lc "python - <<'PY'
import torch, time
from Bio.Seq import Seq
import openspliceai.predict.predict as P
P.setup_device = lambda: torch.device('cpu')

def run(seq, tag):
    t=time.time(); y=P.predict(seq, 'repo/models/openspliceai-mane/10000nt',10000); print(tag,time.time()-t,y.shape); torch.save(y, tag+'.pt')
    acc=(y[:,1]>=0.9).nonzero().flatten().tolist()
    don=(y[:,2]>=0.9).nonzero().flatten().tolist()
    # coordinate conventions
    acceptors=[i-1 for i in acc]
    donors=don[:]
    print(tag, 'acc score indices', acc, 'don indices', don)
    return y, acceptors, donors
seq_plus=open('locus.fa').read().split('\\n',1)[1].replace('\\n','')
seq_minus=str(Seq(seq_plus).reverse_complement())
yplus, aplus, dplus=run(seq_plus,'plus')
yminus, aminus, dminus=run(seq_minus,'minus')
for tag,a,d in [('plus',aplus,dplus),('minus',aminus,dminus)]:
    print(tag, 'acceptors(last intron coords)',a)
    print(tag, 'donors(first intron coords)',d)
PY"
