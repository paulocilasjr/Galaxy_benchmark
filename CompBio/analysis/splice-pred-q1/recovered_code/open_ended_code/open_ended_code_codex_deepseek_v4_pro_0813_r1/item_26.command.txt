/bin/bash -lc "python - <<'PY'
import urllib.request, json
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start=120196698;end=120201111'
with urllib.request.urlopen(url, timeout=60) as r:
    data=json.load(r)
print(data.keys())
print(data['dna'][:100], len(data['dna']))
open('locus.fa','w').write('>locus\\n'+data['dna']+'\\n')
PY
wc -c locus.fa; head -c 100 locus.fa"
