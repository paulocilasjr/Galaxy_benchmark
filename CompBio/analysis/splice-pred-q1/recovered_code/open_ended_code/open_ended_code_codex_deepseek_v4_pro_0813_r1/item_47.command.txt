# parse BED intervals; convert to gene strand coordinates from strand '-'
def parse(fn):
    res=[]

        c,s,e,name,score,strand=line.split()
        s=int(s); e=int(e)
        # for minus: chrom_start=end_1b-seq_end, chrom_end=end_1b-seq_start; so gene p_start = end_1b - chrom_end? Let's derive BED interval [chrom_start,chrom_end)
        # sequence interval seq [seq_start,seq_end), p = seq_start for start, seq_end for end.
        # chrom_start = end_1b - seq_end => seq_end=end_1b-chrom_start.
        # chrom_end = end_1b - seq_start => seq_start=end_1b-chrom_end.
        seq_start=S - e
        seq_end=S - s
        res.append((seq_start,seq_end,float(score),name))
    return res
