/bin/bash -lc ".venv/bin/python - <<'PY'
import numpy as np
p=np.load('results/ensemble_predictions.npz')['ensemble']
A=list(np.flatnonzero(p[1]>=.9)); D=list(np.flatnonzero(p[2]>=.9)); n=p.shape[1]
internal=[]
for a in A:
    d=next((x for x in D if x>a),None)
    if d is not None: internal.append((a,d))
exons=[(0,D[0]),*internal,(A[-1],n)]
print('acceptors',A)
print('donors',D)
print('internal',internal)
print('exons',exons)
print('count',len(exons))
PY"
