/bin/bash -c 'PYTHONNOUSERSITE=1 PYTHONPATH="$PWD/py_pkgs:$PWD/openspliceai_src" python3 -X faulthandler - <<'"'PY'
import torch, urllib.request, json, urllib.parse
from openspliceai.predict.predict import create_datapoints, load_pytorch_models
from openspliceai.predict import utils
params={'genome':'hg38','chrom':'chr12','start':120196698,'end':120201111}
url='https://api.genome.ucsc.edu/getData/sequence?'+urllib.parse.urlencode(params)
seq=json.load(urllib.request.urlopen(url))['dna'].upper()
seq=seq.translate(str.maketrans('ACGTN','TGCAN'))[::-1]
consts=utils.initialize_constants(10000)
print('seq', len(seq))
x=create_datapoints(seq, consts['SL'], consts['CL_max'])
print('x np', x.shape, x.dtype)
x=torch.tensor(x,dtype=torch.float32).permute(0,2,1)
print('x tensor', x.shape)
models,_=load_pytorch_models('models/OSAI-MANE/10000nt',torch.device('cpu'),consts['SL'],10000)
print('models loaded')
for i, model in enumerate(models):
  print('model',i,'start')
  with torch.no_grad():
    y=model(x)
  print('model',i,'y',y.shape, y[:,1,:].max().item(), y[:,2,:].max().item())
print('done')
PY"
