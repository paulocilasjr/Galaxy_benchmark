/bin/bash -lc "python - <<'PY'
import urllib.request, json
# fetch correct forward plus from UCSC 0-based half-open [120196698,120201111)
start=120196698; end=120201111
url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr12;start={start};end={end}'
data=json.load(urllib.request.urlopen(url, timeout=60))
seq=data['dna'].upper()
print('len',len(seq))
comp=str.maketrans('ACGTNacgtn','TGCANtgcan')
rc=seq.translate(comp)[::-1]
open('locus_plus.fa','w').write(f'>chr12:120196699-120201111(+)\\n{seq}\\n')
open('locus_minus.fa','w').write(f'>chr12:120196699-120201111(-)\\n{rc}\\n')
print('plus first',seq[:60])
print('minus first',rc[:60])
PY
cat > run_predict_cpu.py <<'PY'
import os, time
import torch
import numpy as np
from Bio import SeqIO
from openspliceai.predict import predict as predmod
from openspliceai.constants import SL
seq=str(next(SeqIO.parse('locus_minus.fa','fasta')).seq)
X=predmod.create_datapoints(seq, SL, 10000)
X=torch.tensor(X,dtype=torch.float32).permute(0,2,1)
device=torch.device('cpu')
models, params=predmod.load_pytorch_models(os.path.abspath('repo/models/openspliceai-mane/10000nt'), device, SL, 10000)
with torch.no_grad():
    y=torch.mean(torch.stack([m(X).detach().cpu() for m in models]),axis=0)
y=y.permute(0,2,1).contiguous().view(-1, y.shape[1])[:len(seq),:]
np.save('predictions_avg_cpu.npy', y.numpy())
print('shape', y.shape)
print('acc>=0.9', np.where(y[:,1].numpy()>=0.9)[0].tolist())
print('don>=0.9', np.where(y[:,2].numpy()>=0.9)[0].tolist())
PY
time python run_predict_cpu.py"
