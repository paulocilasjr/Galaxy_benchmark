/bin/bash -lc "python - <<'PY'
import numpy as np
y=np.load('predictions_avg_cpu.npy')
seq_len=len(y)
acc=[int(i) for i in np.where(y[:,1]>=0.9)[0]]
don=[int(i) for i in np.where(y[:,2]>=0.9)[0]]
print('len',seq_len,'acc',acc,'don',don)
# sort combined sites with type
sites=sorted([(p,'D') for p in don]+[(p,'A') for p in acc])
print(sites)
# Terminal first exon: boundary 0 to first donor (exclusive maybe 0..donor)
# Terminal last: last acceptor to boundary end
# internal exons: each acceptor paired with next donor
first_don=sites[0] if sites[0][1]=='D' else None
last_acc=sites[-1] if sites[-1][1]=='A' else None
internal=0
for idx,(p,typ) in enumerate(sites):
    if typ=='A':
        # next site
        if idx+1 < len(sites) and sites[idx+1][1]=='D':
            internal+=1
            print('internal', idx, p, sites[idx+1][0], 'exon maybe start', p+1, 'end', sites[idx+1][0])
print('first terminal',0,'to',first_don,'len',first_don[0])
print('last terminal',last_acc[0],'to',seq_len,'len',seq_len-last_acc[0])
print('internal count',internal,'total exons',2+internal)
# Alternative using donor intervals:
# Exons = intervals between [transcript start, first donor], [acceptor+1, next donor], ... [last acceptor+1, transcript end]
# count = number donors (if each donor ends exon) = 7? number exons = #donors + maybe? if first and last.
print('donor count',len(don),'acceptor count',len(acc),'total if #donors+1',len(don)+1)
PY"
