/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://api.genome.ucsc.edu/getData/track?genome=hg38;track=refGene;chrom=chr12;start=120196699;end=120201111'
try:
 data=json.load(urllib.request.urlopen(url))
 for rec in data.get('refGene',[]):
  print(rec.get('name'), rec.get('strand'), rec.get('exonStarts'), rec.get('exonEnds'))
except Exception as e:
 print('err',e)
PY"
