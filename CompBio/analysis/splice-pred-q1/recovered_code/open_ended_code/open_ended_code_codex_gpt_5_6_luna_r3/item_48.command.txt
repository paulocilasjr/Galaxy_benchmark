/bin/bash -lc "curl -L --fail --silent --show-error 'https://api.genome.ucsc.edu/getData/track?genome=hg38;track=ncbiRefSeq;chrom=chr12;start=120196698;end=120201111' -o downloads/ucsc_ncbiRefSeq_interval.json && python3 - <<'PY'
import json
x=json.load(open('downloads/ucsc_ncbiRefSeq_interval.json'))
for k,v in x.items():
    if k=='ncbiRefSeq':
      print('records',len(v))
      for rec in v:
        print({z:rec.get(z) for z in ['name','chrom','txStart','txEnd','strand','exonCount','exonStarts','exonEnds','geneName','name2']})
PY"
