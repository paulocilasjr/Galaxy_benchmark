/bin/bash -lc "python3 - <<'PY'
import json
from pathlib import Path
import numpy as np
j=json.loads(Path('results/site_calls.json').read_text())
z=np.load('results/averaged_scores.npz'); s=z['scores'];
# known exon boundaries from UCSC, genomic 1-based inclusive
starts=[120196699,120197322,120198554,120198854,120199118,120199310,120200730,120201083]
ends=[120196934,120197462,120198739,120199000,120199205,120199485,120200891,120201111]
# map any genomic base to transcript 0-based: 120201111-g
for name, coords in [('starts',starts),('ends',ends)]:
  print(name)
  for g in coords:
    p=120201111-g
    # max scores in +/- 5 positions with channel and locations
    ar=[]
    for q in range(max(0,p-5),min(len(s),p+6)):
      ar.append((q+1, float(s[q,1]), float(s[q,2])))
    ma=max(ar,key=lambda x:max(x[1],x[2]))
    print(g,'transcript0',p,'max',ma,'at site',[(x[0],round(x[1],3),round(x[2],3)) for x in ar if x[1]>=.5 or x[2]>=.5])
PY"
