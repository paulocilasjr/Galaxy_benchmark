/bin/bash -lc "python3 - <<'PY'
import pysam, collections, time
p='pooled.infer.donors.q1.bam'
counts=collections.Counter(); qcounts=collections.Counter(); chrom=collections.Counter(); tags=collections.Counter(); n=0; t=time.time()
with pysam.AlignmentFile(p,'rb') as f:
 for a in f.fetch(until_eof=True):
  n+=1
  if a.has_tag('CB'):
   cb=a.get_tag('CB'); counts[cb]+=1
   if a.has_tag('GX'): tags['GX']+=1
   if a.has_tag('GN'): tags['GN']+=1
   if a.has_tag('UB'): tags['UB']+=1
  else: tags['noCB']+=1
  if a.is_duplicate: qcounts['dup']+=1
  if a.mapping_quality>=30: qcounts['mapq30']+=1
  chrom[a.reference_name]+=1
print('reads',n,'elapsed',round(time.time()-t,1),flush=True)
print('barcodes',len(counts),'total tagged',sum(counts.values()),'tagstats',tags,flush=True)
print('depth quantiles', [sorted(counts.values())[int((len(counts)-1)*x)] for x in (0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1)],flush=True)
print('top',counts.most_common(20),flush=True)
print('qstats',qcounts,flush=True)
print('chrom top',chrom.most_common(30),flush=True)
PY"
