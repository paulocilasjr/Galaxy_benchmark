/bin/bash -lc "cat > validate_donors.py <<'PY'
import os
os.environ['MPLCONFIGDIR']=os.path.abspath('tmp/matplotlib')
import numpy as np,json,glob
from scipy import sparse
from sklearn.metrics import adjusted_rand_score
from vireoSNP import Vireo
ad=sparse.load_npz('AD.npz').tocsc();dp=sparse.load_npz('DP.npz').tocsc()
sites=[l.split() for l in open('sites.tsv')]
chr=np.array([int(a[0]) for a in sites]);pos=np.array([int(a[1]) for a in sites])
base=np.load('models/k4.s0.npz')['ID'].argmax(1)
for subset in ['nonHLA','odd','even']:
 keep=~((chr==6)&(pos>25000000)&(pos<35000000))
 if subset=='odd':keep&=(chr%2==1)
 if subset=='even':keep&=(chr%2==0)
 aa=ad[keep];dd=dp[keep]
 for k in ([3,4,5,6] if subset=='nonHLA' else [4]):
  for seed in range(3):
   np.random.seed(100+seed)
   m=Vireo(n_cell=aa.shape[1],n_var=aa.shape[0],n_donor=k)
   m.fit(aa,dd,max_iter=250,delay_fit_theta=5,verbose=False)
   labels=m.ID_prob.argmax(1);conf=m.ID_prob.max(1)>.95
   print(json.dumps({'subset':subset,'sites':aa.shape[0],'k':k,'seed':seed,'elbo':float(m.ELBO_[-1]),'sizes':np.bincount(labels,minlength=k).tolist(),'confident':int(conf.sum()),'ARI_vs_full':adjusted_rand_score(base[conf],labels[conf])}),flush=True)
PY
PYTHONPATH=\""'$PWD/pydeps" OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 VECLIB_MAXIMUM_THREADS=1 python -u validate_donors.py > validate.log 2>&1'
