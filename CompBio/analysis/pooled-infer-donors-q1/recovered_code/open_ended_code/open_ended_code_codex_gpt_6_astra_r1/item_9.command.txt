/bin/bash -lc "python -u - <<'PY'
import pysam,collections,time
print('pysam imported',flush=True)
b=pysam.AlignmentFile('pooled.infer.donors.q1.bam');c=collections.Counter(); n=0;t=time.time()
for a in b:
 if a.has_tag('CB'):c[a.get_tag('CB')]+=1
 n+=1
 if n%5000000==0: print(n,len(c),round(time.time()-t),flush=True)
with open('barcodes.counts.tsv','w') as f:
 for k,v in c.most_common():f.write(f'{k}\\t{v}\\n')
print('DONE',n,len(c),c.most_common(10),flush=True)
PY"
