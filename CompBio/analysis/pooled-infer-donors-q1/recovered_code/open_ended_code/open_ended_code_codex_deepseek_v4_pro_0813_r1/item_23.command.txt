/bin/bash -lc "python3 - <<'PY'
import pysam
fn='pooled.infer.donors.q1.bam'
inb=pysam.AlignmentFile(fn,'rb',check_sq=False)
outs={}
for bc in ['ACAACTTG','CTCCAACA','TGTGTGAC','GAGTGCGT']:
    outs[bc]=pysam.AlignmentFile(f'chr1_{bc}.bam','wb',template=inb)
count=0
for r in inb:
    if r.reference_name=='1':
        bc=r.get_tag('BC')
        outs[bc].write(r)
        count+=1
print('chr1 reads',count)
for o in outs.values(): o.close()
PY
ls -lh chr1_*.bam"
