/bin/bash -lc "python3 - <<'PY'
import pysam, collections
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb',check_sq=False)
c=collections.Counter()
for r in b:
    c[(r.get_tag('BC'),r.get_tag('CB'))]+=1
print('pairs',len(c),'minmax',min(c.values()),max(c.values()))
print('value counts',collections.Counter(c.values()).most_common(10))
print('unique CB per BC', collections.Counter(bc for bc,cb in c))
PY"
