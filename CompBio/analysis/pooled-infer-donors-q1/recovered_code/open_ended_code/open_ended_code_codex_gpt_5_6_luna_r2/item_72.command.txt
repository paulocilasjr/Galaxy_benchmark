/bin/bash -lc "python - <<'PY'
import numpy as np
z=np.load('allele_counts.npz')
for k in z.files:
 a=z[k]; print(k,a.shape,a.dtype)
keys=z['keys']; dp=z['dp']; ad=z['ad']; sites=z['sites']; cb=z['barcodes']
rows=keys>>16; cells=keys & 65535
print('n cells',len(cb),'n sites',len(sites),'pairs',len(keys),'reads',int(dp.sum()),'alt',int(ad.sum()))
print('cell sites quantiles',np.quantile(np.bincount(cells,minlength=len(cb)),[0,.01,.1,.5,.9,.99,1]))
print('cell reads quantiles',np.quantile(np.bincount(cells,weights=dp,minlength=len(cb)),[0,.01,.1,.5,.9,.99,1]))
site_pairs=np.bincount(rows,minlength=len(sites)); site_dp=np.bincount(rows,weights=dp,minlength=len(sites)); site_ad=np.bincount(rows,weights=ad,minlength=len(sites))
print('site cells quantiles',np.quantile(site_pairs,[0,.1,.5,.9,.99,1]))
print('site reads quantiles',np.quantile(site_dp,[0,.1,.5,.9,.99,1]))
f=site_ad/np.maximum(site_dp,1)
print('allele freq quantiles',np.quantile(f[site_dp>0],[0,.1,.25,.5,.75,.9,.99,1]))
for thresh in [20,50,100,200,500]:
 sel=(site_pairs>=thresh)
 print('sites pairs>=',thresh,sel.sum())
PY"
