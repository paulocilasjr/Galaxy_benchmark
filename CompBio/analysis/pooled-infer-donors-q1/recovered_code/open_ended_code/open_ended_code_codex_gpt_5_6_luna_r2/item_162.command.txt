import numpy as np
from scipy.io import mmread
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
AD=mmread('cellSNP_common_20/cellSNP.tag.AD.mtx').tocsr().astype(float);DP=mmread('cellSNP_common_20/cellSNP.tag.DP.mtx').tocsr().astype(float); f=np.asarray(AD.sum(1)).ravel()/np.maximum(np.asarray(DP.sum(1)).ravel(),1)
X=np.zeros((DP.shape[1],DP.shape[0]),dtype='float32')
for j in range(DP.shape[0]):
 d=DP.getrow(j).toarray().ravel();a=AD.getrow(j).toarray().ravel();X[:,j]=(a-d*f[j])/np.sqrt(d+1)
p=PCA(n_components=20,random_state=1).fit_transform(X);print('shape',X.shape,'var',PCA(n_components=10,random_state=1).fit(X).explained_variance_ratio_[:5])
for k in range(2,11):
 lab=KMeans(n_clusters=k,n_init=30,random_state=1).fit_predict(p[:,:10]);print(k,round(silhouette_score(p[:,:10],lab),4),np.bincount(lab).tolist())
PY"
