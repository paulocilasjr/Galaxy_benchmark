import gzip,numpy as np
from scipy.optimize import linear_sum_assignment
def gt(path):
 d={}
 with gzip.open(path,'rt') as fh:
  for l in fh:
   if l.startswith('#'):continue
   f=l.rstrip().split('\\t');g=[]
   for x in f[9:]:
    y=x.split(':')[0];g.append(0 if y in ('0/0','0|0') else 1 if y in ('1/1','1|1') else .5)
   d[(f[0],int(f[1]))]=g
 return d
for n in [4,5,6]:
 base=gt(f'vireo_common_20_N{n}/GT_donors.vireo.vcf.gz'); bkeys=list(base); B=np.array([base[k] for k in bkeys]).T
 print('N',n)
 for t in [30,50,100]:
  q=gt(f'vireo_common_{t}_N{n}/GT_donors.vireo.vcf.gz'); common=[k for k in bkeys if k in q]; A=np.array([base[k] for k in common]).T;C=np.array([q[k] for k in common]).T
  cost=np.mean(abs(A[:,None,:]-C[None,:,:]),2);ri,ci=linear_sum_assignment(cost);print(t,'sites',len(common),'mean',round(cost[ri,ci].mean(),3),'max',round(cost[ri,ci].max(),3))
PY"
