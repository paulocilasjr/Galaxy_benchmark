/bin/bash -lc "python - <<'PY'
import numpy as np
z=np.load('allele_counts.npz'); keys=z['keys'];dp=z['dp'];ad=z['ad'];sites=z['sites']; rows=keys>>16
sp=np.bincount(rows,minlength=len(sites)); sd=np.bincount(rows,weights=dp,minlength=len(sites)); sa=np.bincount(rows,weights=ad,minlength=len(sites)); f=sa/sd
ix=np.argsort(-sd)
for i in ix[:40]: print(i,sites[i].tolist(),int(sp[i]),int(sd[i]),int(sa[i]),round(f[i],3))
PY"
