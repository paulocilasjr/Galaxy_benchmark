/bin/bash -lc "python - <<'PY'
import re,numpy as np,pandas as pd
z=np.load('allele_counts.npz'); keys=z['keys']; dp=z['dp'].astype(int); ad=z['ad'].astype(int); rows=keys>>16; cells=keys&65535; cb=z['barcodes'].astype(str)
holdset=set(np.load('cv_site_sets.npz')['hold'].tolist()); hm=np.array([int(r) in holdset for r in rows]); hr=rows[hm]; hc=cells[hm]; hdp=dp[hm]; had=ad[hm]
print('heldout',len(hr),hdp.sum(),len(np.unique(hr)))
for n in range(2,9):
 df=pd.read_csv(f'cv_train_N{n}/donor_ids.tsv',sep='\\t'); c2i={x:i for i,x in enumerate(cb)}
 lab=np.full(len(cb),-1,dtype=int); prob=np.zeros(len(cb))
 for _,row in df.iterrows():
  i=c2i[row['cell']]; m=re.fullmatch(r'donor(\\d+)',str(row['donor_id']))
  if m: lab[i]=int(m.group(1));prob[i]=float(row['prob_max'])
 for clean in (False,True):
  keep=(lab[hc]>=0)&((prob[hc]>=.9) if clean else True)
  rr=hr[keep]; cc=hc[keep]; dd=hdp[keep]; aa=had[keep]
  geno={}
  for g in range(n):
   q=(lab[cc]==g)
   if not q.any(): continue
   code=(rr[q].astype(np.int64)<<32)|g
   uk,inv=np.unique(code,return_inverse=True)
   rs=(uk>>32).astype(int)
   ads=np.bincount(inv,weights=aa[q]); dps=np.bincount(inv,weights=dd[q])
   for site,a,d in zip(rs,ads,dps):
    scores=[]
    for state in (0.,.5,1.):
     pp=.01+.98*state
     scores.append((a*np.log(pp)+(d-a)*np.log1p(-pp),pp))
    geno[(int(site),g)]=max(scores)[1]
  ll=0.; total=0
  for site,c,d,a in zip(rr,cc,dd,aa):
   pp=geno[(int(site),int(lab[c]))]
   ll += a*np.log(pp)+(d-a)*np.log1p(-pp); total+=d
  print(f'N{n} clean={clean} cells={len(np.unique(cc))} reads={total} sites={len(geno)} ll={ll:.1f} per={ll/total:.6f}')
PY"
