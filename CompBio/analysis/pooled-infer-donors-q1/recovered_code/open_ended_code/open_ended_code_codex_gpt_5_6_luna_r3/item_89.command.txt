/bin/bash -lc "python - <<'PY'
import numpy as np
A=np.load('cell_site_counts.npz')['counts']; R=A[:,:,0].T.astype(float); Alt=A[:,:,1].T.astype(float); p=np.array([.03,.5,.97]);
for K in [2,3,4,5,6]:
 z=np.load(f'em_all_K{K}.npz'); L=z['labels']; ix=z['site_indices']; r=R[:,ix]; a=Alt[:,ix];
 out=[]
 for fold in range(5):
  rng=np.random.default_rng(700+fold); tr=np.zeros(len(L),bool)
  for k in range(K):
   ii=np.where(L==k)[0]; rng.shuffle(ii); tr[ii[:len(ii)//2]]=1
  te=~tr; g=np.zeros((K,len(ix)),int)
  for k in range(K):
   rr=r[tr & (L==k)].sum(0); aa=a[tr & (L==k)].sum(0)
   g[k]=np.argmax(rr[:,None]*np.log1p(-p)[None,:]+aa[:,None]*np.log(p)[None,:],1)
  ll=np.zeros((len(L),K))
  for k in range(K):
   pp=p[g[k]]; ll[:,k]=r@np.log1p(-pp)+a@np.log(pp)
  fixed=ll[te,np.array(L[te])].sum(); mx=ll[te].max(1).sum(); n=(r[te]+a[te]).sum()
  out.append((fixed,mx,n))
 print('K',K,'fixed',round(sum(x[0] for x in out),1),'max',round(sum(x[1] for x in out),1),'perread',round(sum(x[0] for x in out)/sum(x[2] for x in out),4), 'folds', [round(x[0],0) for x in out])
PY"
