/bin/bash -lc "python - <<'PY'
import pysam, collections, time
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
c=collections.Counter(); n=0; t=time.time()
for r in b.fetch(until_eof=True):
    cb=r.get_tag('CB', True)
    if cb is not None:
        c[cb]+=1; n+=1
    if n and n%2000000==0: print('n',n,'distinct',len(c),'elapsed',time.time()-t, flush=True)
print('done reads with CB',n,'distinct',len(c),'elapsed',time.time()-t)
print('top counts')
for cb,ct in c.most_common(30): print(ct,cb)
PY"
