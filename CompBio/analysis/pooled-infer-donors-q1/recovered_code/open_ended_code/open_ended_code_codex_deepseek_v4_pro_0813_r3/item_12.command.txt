/bin/bash -lc "python - <<'PY'
import pysam, collections
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
bc=collections.Counter(); cb_bc=collections.defaultdict(set); n=0
for r in b.fetch(until_eof=True):
    v=r.get_tag('BC', True)
    if v is not None:
        bc[v]+=1
        cb=r.get_tag('CB', True)
        cb_bc[cb].add(v)
    n+=1
print('reads',n,'distinct BC',len(bc))
for v,c in bc.most_common(): print(c,v)
print('num CB with multiple BC', sum(len(s)>1 for s in cb_bc.values()))
print('num CB with 1 BC', sum(len(s)==1 for s in cb_bc.values()))
# distribution
multi=collections.Counter(len(s) for s in cb_bc.values())
print('CB per numBC', multi)
PY"
