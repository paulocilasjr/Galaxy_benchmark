/bin/bash -lc "python - <<'PY'
import pysam, collections
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
bc_run=collections.defaultdict(collections.Counter)
for r in b.fetch(until_eof=True):
    prefix=r.query_name.split(':')[0]
    bc_run[r.get_tag('BC')][prefix]+=1
for bc,d in bc_run.items(): print(bc, dict(d))
PY"
