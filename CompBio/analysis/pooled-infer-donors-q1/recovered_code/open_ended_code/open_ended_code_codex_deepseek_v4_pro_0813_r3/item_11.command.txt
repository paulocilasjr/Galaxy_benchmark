/bin/bash -lc "python - <<'PY'
import pysam, collections
fn='pooled.infer.donors.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
tags=collections.Counter(); vals=collections.defaultdict(set); n=0
for r in b.fetch(until_eof=True):
    for t,v in r.get_tags():
        tags[t]+=1
        if n<200000: vals[t].add((v,type(v).__name__))
    n+=1
    if n>=2000000: break
print('reads',n)
for t,c in tags.most_common(): print(t,c, 'sample', list(vals[t])[:10])
PY"
