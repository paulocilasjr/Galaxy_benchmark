/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json, time
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
def get(url):
    with urllib.request.urlopen(url,timeout=30) as r:
        return r.read().decode()
def esummary(db, ids):
    url=base+'esummary.fcgi?db='+db+'&id='+','.join(map(str,ids))+'&retmode=json'
    return json.loads(get(url))
for id in [90527,6652,84419,58472,9153]:
    try:
        d=esummary('gene',[id])
        r=d['result'][str(id)]
        print(id, r.get('name'), r.get('description'), r.get('otheraliases'), 'chr', r.get('locationhist') or r.get('maplocation'))
    except Exception as e: print(id,e)
    time.sleep(.3)
PY"
