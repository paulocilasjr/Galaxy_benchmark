/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/'
def j(url):
    return json.load(urllib.request.urlopen(url))
# exact gene symbol search
for term in ['TERB2[Gene Name] AND human[Organism]','TERB2[Gene Name]','BEND5[Gene Name] AND human[Organism]','BEND5[Gene Name]']:
    d=j(base+'esearch.fcgi?db=gene&retmode=json&term='+urllib.parse.quote(term))
    print(term, d['esearchresult']['idlist'])
    ids=d['esearchresult']['idlist'][:20]
    if ids:
        s=j(base+'esummary.fcgi?db=gene&id='+','.join(ids)+'&retmode=json')
        for i in ids:
            r=s['result'].get(i,{})
            print(' ',i,r.get('name'),r.get('description'))
PY"
