/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for sym in ['TERB2','SLC28A2','DUOXA1','COXFA4L3','SQOR','BEND5','SHF','SORD']:
    u=f'https://rest.ensembl.org/lookup/symbol/homo_sapiens/{sym}?content-type=application/json'
    try:
        d=json.load(urllib.request.urlopen(u,timeout=20))
        print(sym,d.get('id'),d.get('display_name'),d.get('biotype'))
    except Exception as e:
        print(sym,e)
PY"
