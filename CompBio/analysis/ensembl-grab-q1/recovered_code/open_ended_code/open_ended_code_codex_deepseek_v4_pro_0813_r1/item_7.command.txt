/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
# Ensembl GRCh38 overlap chr15:45167214-45187966
url='https://rest.ensembl.org/overlap/region/human/15:45167214-45187966?feature=gene;content-type=application/json'
req=urllib.request.Request(url, headers={'Content-Type':'application/json'})
data=json.load(urllib.request.urlopen(req,timeout=30))
for g in data:
    print(g.get('id'), g.get('external_name'), g.get('biotype'), g.get('seq_region_name'), g.get('start'), g.get('end'))
PY"
