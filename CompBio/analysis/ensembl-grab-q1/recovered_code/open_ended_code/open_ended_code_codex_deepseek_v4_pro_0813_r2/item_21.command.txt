import urllib.request, urllib.parse, json
queries=['BEN domain containing 5 transcript variant 6','BEN domain containing 5 variant 6','BEND5 transcript variant 6','BEND5']

   print(h.get('_id'),h.get('symbol'),h.get('name'),h.get('ensembl'),h.get('refseq',{}).get('rna'))
 except Exception as e: print('ERR',q,e)
