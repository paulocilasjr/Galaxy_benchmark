/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
queries=[
('NM_001276266.2','refseq.rna:NM_001276266.2'),
('NP_922946.1','refseq.protein:NP_922946.1'),
('ENST00000267814','ensembl.transcript:ENST00000267814'),
('MIM617658','mim:617658'),
]
for label,q in queries:
 u='https://mygene.info/v3/query?q='+urllib.parse.quote(q)+'&fields=symbol,ensembl.gene,name,refseq'
 try:
  j=json.loads(urllib.request.urlopen(u,timeout=30).read().decode())
  print('\\n',label,'total',j['total'])
  for h in j.get('hits',[])[:5]:
   print(h.get('_id'),h.get('symbol'),h.get('name'),h.get('ensembl'),h.get('refseq'))
 except Exception as e: print(label,'ERR',e)
PY"
