/bin/bash -lc "python - <<'PY'
import json, urllib.request, time
symbols=['DUOXA1','TERB2','SORD','SHF','SLC28A2','COXFA4L3','SQOR','BEND5']
for s in symbols:
 u=f'https://rest.ensembl.org/xrefs/symbol/homo_sapiens/{s}?content-type=application/json'
 try:
  data=json.load(urllib.request.urlopen(u,timeout=30))
  print(s, [(x.get('id'),x.get('type')) for x in data])
 except Exception as e: print(s,type(e).__name__,e)
 time.sleep(.5)
PY"
