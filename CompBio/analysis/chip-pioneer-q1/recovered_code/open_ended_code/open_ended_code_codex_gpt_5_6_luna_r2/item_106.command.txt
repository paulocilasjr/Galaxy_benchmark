import os,math
regs=['old_r1','old_r2','old_union','old_both']
D={}
for l in open('fast_old_counts.tsv'):
 p=l.rstrip().split('\\t'); b=os.path.basename(p[0]); tf=b.split('.')[0]; typ='c' if 'control' in b else 'r'; D.setdefault((tf,typ),[]).append((int(p[1]),int(p[2]),{k:tuple(map(int,v.split('/'))) for k,v in zip(regs,p[3:])}))
for tf in [f'TF{i}' for i in range(1,7)]:
 rn=sum(x[0] for x in D[tf,'r']);cn=sum(x[0] for x in D[tf,'c'])
 print(tf,end=' ')
 for reg in regs:
  ra=sum(x[2][reg][0] for x in D[tf,'r']);ca=sum(x[2][reg][0] for x in D[tf,'c']);fa=ra/rn;fc=ca/cn
  print(f'{reg}:A{fa:.4f}/{fc:.4f},O{math.log2((1-fa)/(1-fc)):.3f}',end=' ')
 print()
PY"
