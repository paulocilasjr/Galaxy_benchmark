/bin/bash -lc "cat > analysis/summarize_counts.py <<'PY'
import csv,os
p='analysis/fragment_peak_counts.tsv'
rows={}
for r in csv.DictReader(open(p),delimiter='\\t'):
 name=os.path.basename(r['file']).replace('.bed','');rows[name]={k:int(r[k]) for k in ['total','open','closed','both']}
print('TF\\treplicate\\tnet_open_fraction\\tnet_closed_fraction\\tclosed_share_of_net_signal\\tclosed_fold_over_input')
for i in range(1,7):
 tf=f'TF{i}'
 for rep in [1,2]:
  name=f'{tf}.rep{rep}';control=f'{tf}.control{rep}'
  if control not in rows:control=f'{tf}.control'
  if name not in rows or control not in rows:continue
  t=rows[name];c=rows[control]
  op=t['open']/t['total']-c['open']/c['total'];cl=t['closed']/t['total']-c['closed']/c['total']
  fold=(t['closed']/t['total'])/(c['closed']/c['total'])
  print(tf,rep,f'{op:.6f}',f'{cl:.6f}',f'{cl/(op+cl):.4f}',f'{fold:.3f}',sep='\\t')
PY
python3 analysis/summarize_counts.py; cat analysis/summarize_dnase_signal_partial.log"
