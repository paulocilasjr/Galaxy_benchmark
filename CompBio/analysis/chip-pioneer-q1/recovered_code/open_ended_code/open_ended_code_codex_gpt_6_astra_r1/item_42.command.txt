/bin/bash -lc "cat > analysis/check_reproducibility.py <<'PY'
import os,time,subprocess,csv
names=['TF4','TF4.rep1','TF4.rep2']
while not all(os.path.exists(f'analysis/peaks/{n}_peaks.narrowPeak') and 'Done"'!'"' in open(f'analysis/peaks/{n}.log').read() for n in names):time.sleep(10)
subprocess.run(['python3','analysis/summarize_peaks.py'],check=True)
a='analysis/peaks/TF4_peaks.narrowPeak'
for r in [1,2]:
 out=f'analysis/TF4.supported_by_{r}.narrowPeak'
 with open(out,'w') as f:subprocess.run(['bedtools','intersect','-u','-a',a,'-b',f'analysis/peaks/TF4.rep{r}_peaks.narrowPeak'],stdout=f,check=True)
 a=out
names={line.split()[3] for line in open(a)}
rows=[r for r in csv.DictReader(open('analysis/TF4.annotated.tsv'),delimiter='\\t') if r['name'] in names]
for n in [0,1000,5000,10000]:
 x=sorted(rows,key=lambda r:float(r['p']),reverse=True)
 if n:x=x[:n]
 print('TF4_reproducible',n,len(x),'closed_fraction',sum(r['hotspots2018_open']=='False' for r in x)/len(x),flush=True)
PY
python3 analysis/check_reproducibility.py > analysis/reproducibility.log 2>&1"
