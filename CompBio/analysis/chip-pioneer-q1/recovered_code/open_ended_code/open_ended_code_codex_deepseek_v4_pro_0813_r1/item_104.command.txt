/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os, concurrent.futures, time
bw=os.path.abspath('dnase/ENCFF723WAJ.bigWig')
files=sorted(glob.glob('chip.pioneer.q1/*.bed.gz'))
out='fragment_dnase_means.tsv'

def run(f):
    cmd1=f\"gzip -cd {f} | awk 'BEGIN{{OFS=\\\"\\\\t\\\"}}{{mid=int(("'$2+$3)/2); print $1, mid, mid+1, NR, 0}}'"'\"
    p1=subprocess.Popen(cmd1, shell=True, stdout=subprocess.PIPE)
    p2=subprocess.Popen(['bigWigAverageOverBed', bw, '/dev/stdin', '/dev/stdout'], stdin=p1.stdout, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
    p1.stdout.close()
    count=0; ssum=0.0; firstrow=None
    for line in p2.stdout:
        parts=line.rstrip(b'\\n').split(b'\\t')
        if len(parts)>=6:
            try:
                v=float(parts[5]); ssum+=v; count+=1
                if firstrow is None: firstrow=parts
            except: pass
    p2.wait()
    return os.path.basename(f), count, ssum, ssum/count if count else None, firstrow
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
    futs=[ex.submit(run,f) for f in files]
    results={}
    for fut in concurrent.futures.as_completed(futs):
        name,count,ssum,mean,first=fut.result(); results[name]=(count,mean)
        print(name,'n',count,'mean',round(mean,5) if mean is not None else None,flush=True)
with open(out,'w') as o:
    o.write('file\\tcount\\tmean_dnase\\n')
    for f in files:
        name=os.path.basename(f); c,m=results[name]
        o.write(f'{name}\\t{c}\\t{m}\\n')
print('done')
PY"
