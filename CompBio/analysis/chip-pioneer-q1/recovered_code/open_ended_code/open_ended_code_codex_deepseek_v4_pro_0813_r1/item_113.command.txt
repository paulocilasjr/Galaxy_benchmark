/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os, pyBigWig, statistics
peaks='dnase/dnase_peaks.merged.bed'
bw=pyBigWig.open('dnase/ENCFF723WAJ.bigWig')
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    f1=f'macs_peaks_q001/{tf}_rep1_peaks.narrowPeak'; f2=f'macs_peaks_q001/{tf}_rep2_peaks.narrowPeak'
    p=subprocess.Popen(['bedtools','intersect','-a',f1,'-b',f2,'-wa','-u'],stdout=subprocess.PIPE)
    lines=p.stdout.readlines(); p.wait()
    vals=[]
    for line in lines:
        c=line.decode().rstrip().split('\\t'); chrom=c[0]; start=int(c[1]); end=int(c[2])
        try: s=bw.stats(chrom,start,end,type='mean',exact=True)[0]
        except: continue
        if s is not None: vals.append(s)
    print(f'{tf:4s} shared={len(lines):6d} dnase_mean={statistics.mean(vals):.4f} median={statistics.median(vals):.4f}')
bw.close()
PY"
