/bin/bash -lc "python3 - <<'PY'
import glob, os, subprocess
peaks='dnase/dnase_peaks.merged.bed'
for f in sorted(glob.glob('macs_peaks_q001/*_union.narrowPeak')):
    # sort by score col8 descending, take top 5000/10000/20000/50000
    name=os.path.basename(f).replace('_union.narrowPeak','')
    lines=open(f).read().splitlines()
    parsed=[]
    for line in lines:
        p=line.split('\\t')
        if len(p)>=10:
            parsed.append((float(p[6]), line))
    parsed.sort(reverse=True)
    for k in [5000,10000,20000,50000]:
        if len(parsed)<k: continue
        # write temp
        tmp=f'[REDACTED LOCAL PATH]'
        with open(tmp,'w') as o:
            for score,line in parsed[:k]: o.write(line+'\\n')
        p=subprocess.Popen(['bedtools','intersect','-a',tmp,'-b',peaks,'-wa','-u'],stdout=subprocess.PIPE)
        open_count=sum(1 for _ in p.stdout); p.wait()
        print(f'{name:8s} top{k:5d} open_frac={open_count/k:.3f} resistant={1-open_count/k:.3f}')
    print()
PY"
