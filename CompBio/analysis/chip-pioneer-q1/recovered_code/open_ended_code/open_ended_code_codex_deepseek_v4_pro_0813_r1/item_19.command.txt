/bin/bash -lc "python3 - <<'PY'
totals={... as before}
open_counts={}
for line in open('open_counts.tsv'):
 p=line.rstrip().split('\\t')
 if p[0]=='file': continue
 open_counts[p[0]]=int(p[2])
import math
# define controls per tf list
controls={'TF1':['TF1.control1.bed.gz','TF1.control2.bed.gz'], 'TF2':['TF2.control1.bed.gz','TF2.control2.bed.gz'], 'TF3':['TF3.control1.bed.gz','TF3.control2.bed.gz'], 'TF4':['TF4.control1.bed.gz','TF4.control2.bed.gz'], 'TF5':['TF5.control.bed.gz'], 'TF6':['TF6.control.bed.gz']}
# pool controls by sum counts
for tf, cs in controls.items():
    T=sum(totals[c] for c in cs); O=sum(open_counts[c] for c in cs); C=T-O
    for rep in ['rep1','rep2']:
        name=f'{tf}.{rep}.bed.gz'; t=totals[name]; o=open_counts[name]; c=t-o
        ORo=(o/t)/(O/T); ORc=(c/t)/(C/T)
        CO=(c/C)/(o/O) # normalized closed/open enrichment ratio (ChIP/input closed vs open)
        print(f'{name:20s} OR_open={ORo:.3f} OR_closed={ORc:.3f} closed/open_enrich={CO:.3f}')
    print('control', tf, 'open_frac',O/T)
PY"
