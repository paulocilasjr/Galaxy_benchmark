/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os
peaks='dnase/dnase_peaks.merged.bed'
files=sorted(glob.glob('macs_peaks_q001/*_peaks.narrowPeak'))
for f in files:
    total=sum(1 for _ in open(f))
    p1=subprocess.Popen(['bedtools','intersect','-a',f,'-b',peaks,'-wa','-u'], stdout=subprocess.PIPE)
    open_count=sum(1 for _ in p1.stdout)
    p1.wait()
    name=os.path.basename(f).replace('_peaks.narrowPeak','')
    print(f'{name:12s} peaks={total:7d} open={open_count:7d} open_frac={open_count/total:.3f} resistant_frac={1-open_count/total:.3f}')
PY"
