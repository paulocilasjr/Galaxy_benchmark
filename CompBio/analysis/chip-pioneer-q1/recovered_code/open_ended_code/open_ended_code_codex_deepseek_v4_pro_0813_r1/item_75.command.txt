/bin/bash -lc "python3 - <<'PY'
import csv, math, itertools
rows=list(csv.DictReader(open('sra_runinfo.csv')))
spot={r['SampleName']:int(r['spots']) for r in rows}
constructs={
'FOXA1wt': ['GSM6806959','GSM6806961','GSM6806960','GSM6806962'],
'FOXA1dbd': ['GSM6806963','GSM6806965','GSM6806964','GSM6806966'],
'FOXA1nhaa': ['GSM6806967','GSM6806969','GSM6806968','GSM6806970'],
'FOXA1rraa': ['GSM6806971','GSM6806973','GSM6806972','GSM6806974'],
'HNF4A': ['GSM6806975','GSM6806977','GSM6806976','GSM6806978'],
'H2B_t6': ['GSM6806982','GSM6806983','GSM6806984',None],
}
vec={
'TF1':[30262780,34212511,37039899,45007567],
'TF2':[9520602,18620491,26894612,26693947],
'TF3':[15298277,17189525,23274203,13216643],
'TF4':[26225809,19481343,28277779,23754574],
'TF5':[16135021,14755327,16452140,None],
'TF6':[8156838,5302061,16833124,None],
}
def ratio_vec(a,b):
    out=[]
    for x,y in zip(a,b):
        if x is None or y is None: continue
        out.append(x/y)
    return out
def cost(a,b):
    r=ratio_vec(a,b)
    if not r:return 1e9
    mean=sum(r)/len(r); cv=math.sqrt(sum((x-mean)**2 for x in r)/len(r))/mean if mean else 0
    # penalize mean >1 or too small
    pen=0
    if mean>1: pen+=(mean-1)*10
    if mean<0.1: pen+=(0.1-mean)*10
    return cv+pen*0.01
# only consider construct with one ctrl for TF5/TF6? But global maybe assign those with one ctrl first.
best=[]
for perm in itertools.permutations(constructs.keys(), len(vec)):
    total=0
    for tf, con in zip(sorted(vec), perm):
        total += cost(vec[tf], [spot[x] if x else None for x in constructs[con]])
    best.append((total, dict(zip(sorted(vec),perm))))
best.sort()
for total,mp in best[:20]:
    print(round(total,3), mp)
PY"
