/bin/bash -lc "python3 - <<'PY'
import pyBigWig, gzip, glob, os, statistics
bw=pyBigWig.open('dnase/ENCFF723WAJ.bigWig')
files=sorted(glob.glob('chip.pioneer.q1/*.bed.gz'))
# sample up to 5M fragments per file for speed
for f in files:
    vals=[]; n=0; maxn=5000000
    with gzip.open(f,'rt') as fh:
        for line in fh:
            p=line.rstrip().split('\\t'); chrom=p[0]; start=int(p[1]); end=int(p[2]); mid=(start+end)//2
            try:
                s=bw.stats(chrom,mid,mid+1,type='mean',exact=True)[0]
            except Exception:
                continue
            if s is not None: vals.append(s)
            n+=1
            if n>=maxn: break
    if vals:
        print(f'{os.path.basename(f):20s} n={len(vals)} dnase_mean={statistics.mean(vals):.4f} median={statistics.median(vals):.4f}')
bw.close()
PY"
