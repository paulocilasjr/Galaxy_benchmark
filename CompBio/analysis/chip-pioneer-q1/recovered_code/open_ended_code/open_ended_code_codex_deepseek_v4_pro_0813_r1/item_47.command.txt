/bin/bash -lc "python3 - <<'PY'
import subprocess, os, concurrent.futures, time
outdir='macs_peaks_q001'
os.makedirs(outdir, exist_ok=True)
pairs=[]
for tf in ['TF1','TF2','TF3','TF4']:
    pairs.append((f'{tf}.rep1.bed.gz', f'{tf}.control1.bed.gz', f'{tf}_rep1'))
    pairs.append((f'{tf}.rep2.bed.gz', f'{tf}.control2.bed.gz', f'{tf}_rep2'))
for tf in ['TF5','TF6']:
    pairs.append((f'{tf}.rep1.bed.gz', f'{tf}.control.bed.gz', f'{tf}_rep1'))
    pairs.append((f'{tf}.rep2.bed.gz', f'{tf}.control.bed.gz', f'{tf}_rep2'))
base='chip.pioneer.q1'

def run(item):
    t,c,n=item
    tpath=os.path.join(base,t); cpath=os.path.join(base,c)
    cmd=['macs2','callpeak','-t',tpath,'-c',cpath,'-f','BED','-g','hs','--outdir',outdir,'-n',n,'-q','0.001','--nomodel','--extsize','200','--shift','-100','--keep-dup','all','--call-summits']
    start=time.time()
    with open(os.path.join(outdir,n+'.log'),'w') as log:
        r=subprocess.run(cmd, stdout=log, stderr=subprocess.STDOUT)
    return n,r.returncode,time.time()-start
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
    futs=[ex.submit(run,p) for p in pairs]
    for fut in concurrent.futures.as_completed(futs):
        n,rc,elapsed=fut.result(); print(n,rc,round(elapsed,1),flush=True)
PY"
