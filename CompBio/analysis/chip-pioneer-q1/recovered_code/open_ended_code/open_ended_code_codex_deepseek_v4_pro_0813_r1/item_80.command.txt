/bin/bash -lc "python3 - <<'PY'
import subprocess, glob, os
black='hg38_blacklist.bed.gz'
for peaksfile in ['dnase/dnase_peaks.merged.bed','dnase/dnase_hotspots.merged.bed']:
    print('USING', peaksfile)
    for f in sorted(glob.glob('macs_peaks_q001/*_union.narrowPeak')):
        # filter blacklist via intersect -v
        p=subprocess.Popen(['bedtools','intersect','-a',f,'-b',black,'-v'], stdout=subprocess.PIPE)
        filtered=[]
        for line in p.stdout: filtered.append(line)
        p.wait()
        total=len(filtered)
        if total==0: continue
        # write temp? use subprocess second with input bytes from filtered
        inp=b''.join(filtered)
        p=subprocess.Popen(['bedtools','intersect','-a','-','-b',peaksfile,'-wa','-u'], stdin=subprocess.PIPE, stdout=subprocess.PIPE)
        out,_=p.communicate(inp)
        open_count=out.count(b'\\n')
        name=os.path.basename(f).replace('_union.narrowPeak','')
        print(f'{name:8s} filtered_peaks={total:6d} open={open_count:6d} open_frac={open_count/total:.3f} resistant={1-open_count/total:.3f}')
PY"
