/bin/bash -lc "python3 - <<'PY'
import subprocess, time
bw='dnase/ENCFF723WAJ.bigWig'
f='chip.pioneer.q1/TF1.rep1.bed.gz'
start=time.time()
# generator command: gzip -cd f | awk add line number unique
cmd1=f\"gzip -cd {f} | awk 'BEGIN{{OFS=\\\"\\\\t\\\"}}{{mid=int(("'$2+$3)/2); print $1, mid, mid+1, NR, 0}}'"'\"
p1=subprocess.Popen(cmd1, shell=True, stdout=subprocess.PIPE)
p2=subprocess.Popen(['bigWigAverageOverBed', bw, '/dev/stdin', '/dev/stdout'], stdin=p1.stdout, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
p1.stdout.close()
count=0; ssum=0.0; first=None; last=None
for line in p2.stdout:
    parts=line.rstrip(b'\\n').split(b'\\t')
    if len(parts)>=6:
        try:
            v=float(parts[5]); ssum+=v; count+=1
            if first is None: first=parts
            last=parts
        except: pass
p2.wait()
print('count',count,'sum',ssum,'mean',ssum/count if count else None,'elapsed',time.time()-start)
print('first',first)
print('last',last)
PY"
