/bin/bash -lc "python3 - <<'PY'
import subprocess, os, glob
peaks='dnase/dnase_peaks.merged.bed'
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    files=glob.glob(f'macs_peaks_q001/{tf}_rep*_peaks.narrowPeak')
    union=f'macs_peaks_q001/{tf}_union.narrowPeak'
    cmd=f\"cat {' '.join(files)} | cut -f1-3 | sort -k1,1 -k2,2n | bedtools merge -i - > {union}\"
    subprocess.run(cmd,shell=True,check=True)
    total=sum(1 for _ in open(union))
    p=subprocess.Popen(['bedtools','intersect','-a',union,'-b',peaks,'-wa','-u'], stdout=subprocess.PIPE)
    open_count=sum(1 for _ in p.stdout); p.wait()
    print(f'{tf:4s} union={total:6d} open={open_count:6d} open_frac={open_count/total:.3f} resistant={1-open_count/total:.3f}')
PY"
