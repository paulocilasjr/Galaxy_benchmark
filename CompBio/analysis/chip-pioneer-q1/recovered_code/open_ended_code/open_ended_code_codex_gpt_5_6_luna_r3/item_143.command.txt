/bin/bash -lc "python - <<'PY'
import pyBigWig,numpy as np
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
 bws=[pyBigWig.open(x) for x in ['data/ENCFF107IJY.bigWig','data/ENCFF708XPM.bigWig']]
 vals=[]
 with open(f'peaks_sub/{tf}/{tf}_peaks.narrowPeak') as f:
  for l in f:
   a=l.split('\\t'); c,s,e=a[0],int(a[1]),int(a[2]); x=[bw.stats(c,s,e,nBins=1,type='mean',exact=True)[0] or 0 for bw in bws]; vals.append(sum(x)/2)
 for bw in bws: bw.close()
 vals=np.array(vals)
 print(tf,len(vals),'closed',np.mean(vals<=.05),'accessible',np.mean(vals>.05),'quant',np.round(np.quantile(vals,[0,.25,.5,.75,.9,.95]),4),'mean',round(vals.mean(),4))
PY"
