/bin/bash -lc "python - <<'PY'
# correlations on 5kb bins; use per-bin normalized IP/input log2 ratios and DNase signal
from collections import defaultdict
import math
S={'TF1':(['TF1.rep1','TF1.rep2'],['TF1.control1','TF1.control2']),'TF2':(['TF2.rep1','TF2.rep2'],['TF2.control1','TF2.control2']),'TF3':(['TF3.rep1','TF3.rep2'],['TF3.control1','TF3.control2']),'TF4':(['TF4.rep1','TF4.rep2'],['TF4.control1','TF4.control2']),'TF5':(['TF5.rep1','TF5.rep2'],['TF5.control']),'TF6':(['TF6.rep1','TF6.rep2'],['TF6.control'])}
D={}
with open('data/dnase_5kb.tsv') as f:
 next(f)
 for l in f:
  c,s,e,x,y=l.rstrip().split('\\t');D[(c,int(s))]=(float(x)+float(y))/2
def load(xs):
 d=defaultdict(int);n=0
 for x in xs:
  for l in open('data/'+x+'.5kb.tsv'):
   c,s,e,v=l.rstrip().split('\\t');d[(c,int(s))]+=int(v);n+=int(v)
 return d,n
for tf,(cs,bs) in S.items():
 c,cn=load(cs);b,bn=load(bs); xs=[];ys=[];ws=[]
 for k,d in D.items():
  # pseudocount 0.1 fragment/bin
  xs.append(math.log2((c[k]+.1)/cn/((b[k]+.1)/bn)))
  ys.append(math.log1p(d)); ws.append(c[k]+b[k])
 def corr(x,y,w=None):
  if w is None:w=[1]*len(x)
  sw=sum(w); mx=sum(a*b for a,b in zip(x,w))/sw;my=sum(a*b for a,b in zip(y,w))/sw
  return sum(ww*(a-mx)*(b-my) for a,b,ww in zip(x,y,w))/(sum(ww*(a-mx)**2 for a,ww in zip(x,w))*sum(ww*(b-my)**2 for b,ww in zip(y,w)))**.5
 print(tf,'unweighted',corr(xs,ys),'weighted',corr(xs,ys,ws))
PY"
