/bin/bash -lc "python - <<'PY'
import pyBigWig,numpy as np
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
 bws=[pyBigWig.open(x) for x in ['data/ENCFF107IJY.bigWig','data/ENCFF708XPM.bigWig']]; vals=[]
 p=f'peaks_q/{tf}/{tf}_peaks.narrowPeak'
 with open(p) as f:
  for l in f:
   a=l.split('\\t');c,s,e=a[0],int(a[1]),int(a[2]); vals.append(sum((bw.stats(c,s,e,nBins=1,type='mean',exact=True)[0] or 0) for bw in bws)/2)
 for bw in bws:bw.close()
 vals=np.array(vals)
 print(tf,len(vals),'closed',np.mean(vals<=.05),'mean',vals.mean() if len(vals) else None,'q',np.round(np.quantile(vals,[.25,.5,.75]),3) if len(vals) else None)
PY"
