/bin/bash -lc "python - <<'PY'
import pyBigWig,glob,os,numpy as np
bws=[pyBigWig.open(x) for x in ['data/ENCFF107IJY.bigWig','data/ENCFF708XPM.bigWig']]
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
 p=f'peaks_sub/{tf}/{tf}_peaks.narrowPeak'; vals=[]
 with open(p) as f:
  for l in f:
   a=l.split('\\t'); c,s,e=a[0],int(a[1]),int(a[2]); x=[]
   for bw in bws: x.append(bw.stats(c,s,e,nBins=1,type='mean',exact=True)[0] or 0)
   vals.append(sum(x)/2)
 vals=np.array(vals)
 print(tf,len(vals),'closed',np.mean(vals<=.05),'accessible',np.mean(vals>.05),'quant',np.quantile(vals,[0,.25,.5,.75,.9,.95]),'mean',vals.mean(),'sum',vals.sum())
 for bw in bws: bw.close()
PY"
