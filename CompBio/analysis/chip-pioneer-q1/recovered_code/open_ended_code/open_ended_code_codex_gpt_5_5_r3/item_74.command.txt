/bin/bash -lc "gzip -cd chip.pioneer.q1/TF6.rep2.bed.gz | awk 'last"'!=$1{print $1; last=$1}'"' > work/fragment_chrom_order.txt
python3 - <<'PY'
sizes={}
for line in open('work/hg38.chrom.sizes'):
    c,s=line.split()[:2]; sizes[c]=s
seen=set(); out=[]
for line in open('work/fragment_chrom_order.txt'):
    c=line.strip()
    if c in sizes and c not in seen:
        out.append((c,sizes[c])); seen.add(c)
for c,s in out:
    print(c,s,sep='\\t')
PY > work/hg38.fragment_order.chrom.sizes
head -40 work/fragment_chrom_order.txt
wc -l work/fragment_chrom_order.txt work/hg38.fragment_order.chrom.sizes
bedtools coverage -sorted -g work/hg38.fragment_order.chrom.sizes -counts -a work/hg38.500bp.natural.windows.bed -b chip.pioneer.q1/TF6.rep2.bed.gz > work/test_TF6rep2.counts.bed
awk '{s+="'$4} END{print NR,s}'"' work/test_TF6rep2.counts.bed"
