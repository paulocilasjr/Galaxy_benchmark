/bin/bash -lc "(
  gzip -cd encode/hotspots/ENCFF519XYN.bed.gz
  gzip -cd encode/hotspots/ENCFF855TSQ.bed.gz
  gzip -cd encode/hotspots/ENCFF964XQP.bed.gz
) | awk 'BEGIN{OFS=\"\\t\"} "'$1 ~ /''^chr([0-9]+|X|Y)$/ {print $1,$2,$3}'"' | sort -k1,1 -k2,2n | bedtools merge -i - > work/BJ_DNase_hotspot_union.bed
wc -l work/BJ_DNase_hotspot_union.bed
awk '{s += "'$3-$2} END{print s}'"' work/BJ_DNase_hotspot_union.bed
mkdir -p work/pooled
for tf in TF1 TF2 TF3 TF4 TF5 TF6; do
  gzip -cd chip.pioneer.q1/"'${tf}.rep*.bed.gz | awk '"'BEGIN{OFS=\"\\t\"} "'$1 ~ /''^chr([0-9]+|X|Y)$/ {print $1,$2,$3}'"' | sort -k1,1 -k2,2n > work/pooled/"'${tf}.treat.bed
  gzip -cd chip.pioneer.q1/${tf}.control*.bed.gz | awk '"'BEGIN{OFS=\"\\t\"} "'$1 ~ /''^chr([0-9]+|X|Y)$/ {print $1,$2,$3}'"' | sort -k1,1 -k2,2n > work/pooled/"'${tf}.control.bed
  printf '"'%s\\t' "'$tf; wc -l work/pooled/${tf}.treat.bed work/pooled/${tf}.control.bed | tail -1
done'
