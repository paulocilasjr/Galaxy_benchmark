/bin/bash -lc "curl -L 'https://hgdownload.soe.ucsc.edu/goldenPath/hg38/bigZips/hg38.chrom.sizes' -o work/hg38.chrom.sizes
awk '"'$1 ~ /''^chr([0-9]+|X|Y)$/ {print}'"' work/hg38.chrom.sizes > work/hg38.primary.chrom.sizes
head work/hg38.primary.chrom.sizes && wc -l work/hg38.primary.chrom.sizes
bedtools makewindows -g work/hg38.primary.chrom.sizes -w 500 > work/hg38.500bp.windows.bed
wc -l work/hg38.500bp.windows.bed"
