/bin/bash -lc "python3 - <<'PY'
import pyBigWig, numpy as np
bw=pyBigWig.open('dnase_rep1_ENCFF768KOM.bigWig')
sizes=bw.chroms()
def peak_vals(peakfile):
    out=[]
    with open(peakfile) as f:
        for line in f:
            p=line.split('\\t'); chrom=p[0]; start=int(p[1]); end=int(p[2])
            if chrom not in sizes: continue
            start=max(0,start); end=min(end,sizes[chrom])
            if start>=end: continue
            v=bw.stats(chrom,start,end,type='mean',exact=True)[0]
            if v is not None: out.append(v)
    return np.array(out)
for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
    a=peak_vals(f'processed/union_peaks_q001/{tf}.bed')
    for t in [0.02,0.03,0.05,0.1]:
        acc=(a>t).mean(); res=1-acc
        print(f'{tf} n={len(a)} thr={t}: acc={acc:.3f} res={res:.3f}', end=' | ')
    print()
bw.close()
PY"
