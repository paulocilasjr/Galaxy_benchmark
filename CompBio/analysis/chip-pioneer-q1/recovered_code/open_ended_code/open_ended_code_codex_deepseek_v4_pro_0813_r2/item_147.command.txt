/bin/bash -lc "python3 - <<'PY'
import subprocess
sets={
 'narrow_union':'processed/dnase_access_union.bed',
 'hotspot_union_150':'processed/dnase_hotspot_union.bed',
 'hotspot3_broad':'processed/dnase_hotspot3_broad.bed',
}
for name,dnase in sets.items():
    print('\\n',name)
    for fac in ['FOXA1wt','SOX2wt','HNF4A','FOXA1nhaa','FOXA1rraa','FOXA1dbd','SOX2dbd']:
        geo=f'geo_peaks_hg38/{fac}.bed'
        n=sum(1 for _ in open(geo))
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {geo} -b {dnase} | wc -l']).decode().strip())
        print(f'{fac}: acc={a/n:.3f} res={(n-a)/n:.3f} n={n}')
PY"
