/bin/bash -lc "python3 - <<'PY'
import gzip, subprocess, os
sets={
 'old_rep1_ENCFF001WEF':'dnase_hg19_hotspot_old_rep1_ENCFF001WEF.bed.gz',
 'old_rep2_ENCFF001WEG':'dnase_hg19_hotspot_old_rep2_ENCFF001WEG.bed.gz',
 'new_rep1_ENCFF371GVV':'dnase_hg19_hotspot_rep1_ENCFF371GVV.bed.gz',
 'new_rep2_ENCFF615DUM':'dnase_hg19_hotspot_rep2_ENCFF615DUM.bed.gz',
}
for name,fn in sets.items():
    raw=f'processed/tmp_hg19_{name}.bed'
    with gzip.open(fn,'rt') as f, open(raw,'w') as o:
        for line in f:
            if line.startswith('#'): continue
            p=line.split('\\t'); o.write(f'{p[0]}\\t{p[1]}\\t{p[2]}\\n')
    merged=f'processed/tmp_hg19_{name}_merged.bed'
    subprocess.run(f'bedtools sort -i {raw} | bedtools merge -i - > {merged}',shell=True,check=True)
    for fac in ['FOXA1wt','HNF4A','FOXA1nhaa','FOXA1rraa','FOXA1dbd']:
        geo=f'geo_peaks/{fac}.bed.gz'
        # create decompressed bed3
        rawgeo=f'processed/tmp_geo_{fac}.bed'
        if not os.path.exists(rawgeo):
            with gzip.open(geo,'rt') as f, open(rawgeo,'w') as o:
                for line in f:
                    if line.startswith('#'): continue
                    p=line.split('\\t'); o.write(f'{p[0]}\\t{p[1]}\\t{p[2]}\\n')
        n=sum(1 for _ in open(rawgeo))
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {rawgeo} -b {merged} | wc -l']).decode().strip())
        print(f'{name}\\t{fac}\\t{n}\\t{a}\\t{a/n:.3f}')
    print()
PY"
