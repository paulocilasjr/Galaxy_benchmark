/bin/bash -lc "python3 - <<'PY'
import subprocess, os, gzip
sets={
 'old_rep2_ENCFF815LIR':'dnase_peaks.bed.gz',
 'narrow_rep1_ENCFF812QOW':'dnase_rep1_ENCFF812QOW.bed.gz',
 'narrow_rep2_ENCFF407NXV':'dnase_rep2_ENCFF407NXV.bed.gz',
 'hotspot_rep1_ENCFF876DOW':'dnase_hotspot_rep1_ENCFF876DOW.bed.gz',
 'hotspot_rep2_ENCFF764MOY':'dnase_hotspot_rep2_ENCFF764MOY.bed.gz',
 'hotspot3_rep2_ENCFF034CWP':'dnase_rep2_hotspot_ENCFF034CWP.bed.gz',
}
# prepare sorted merged or raw cut first3
for name,fn in sets.items():
    raw=f'processed/tmp_{name}.bed'
    if name=='hotspot3_rep2_ENCFF034CWP':
        # file has header #chr, columns are already bed3 with header
        with gzip.open(fn,'rt') as f, open(raw,'w') as o:
            for line in f:
                if line.startswith('#'): continue
                p=line.split('\\t'); o.write(f'{p[0]}\\t{p[1]}\\t{p[2]}\\n')
    else:
        with gzip.open(fn,'rt') as f, open(raw,'w') as o:
            for line in f:
                if line.startswith('#'): continue
                p=line.split('\\t'); o.write(f'{p[0]}\\t{p[1]}\\t{p[2]}\\n')
    merged=f'processed/tmp_{name}_merged.bed'
    subprocess.run(f'bedtools sort -i {raw} | bedtools merge -i - > {merged}',shell=True,check=True)
    for fac in ['FOXA1wt','HNF4A','FOXA1nhaa','FOXA1rraa','FOXA1dbd']:
        geo=f'geo_peaks_hg38/{fac}.bed'
        n=int(subprocess.check_output(['wc','-l','<',geo]).decode().strip())
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {geo} -b {merged} | wc -l']).decode().strip())
        print(f'{name}\\t{fac}\\t{n}\\t{a}\\t{a/n:.3f}')
    print()
PY"
