/bin/bash -lc "python3 - <<'PY'
import pyBigWig, numpy as np
bw=pyBigWig.open('dnase_rep1_ENCFF768KOM.bigWig')
sizes=bw.chroms()
def vals_for(fac):
    out=[]
    with open(f'geo_peaks_hg38/{fac}.bed') as f:
        for line in f:
            p=line.split('\\t'); chrom=p[0]; start=int(p[1]); end=int(p[2])
            if chrom not in sizes: continue
            start=max(0,start); end=min(end,sizes[chrom])
            if start>=end: continue
            v=bw.stats(chrom,start,end,type='mean',exact=True)[0]
            if v is not None: out.append(v)
    return np.array(out)
factors=['FOXA1wt','SOX2wt','HNF4A','FOXA1dbd','FOXA1nhaa','FOXA1rraa','SOX2dbd']
arrays={f:vals_for(f) for f in factors}
for t in [0.01,0.02,0.03,0.05,0.1,0.2,0.5]:
    print('threshold',t)
    for f in factors:
        a=arrays[f]; frac=(a>t).mean()
        print(f'{f}: acc={frac:.3f} res={1-frac:.3f}', end='; ')
    print()
bw.close()
PY"
