/bin/bash -lc 'mkdir -p processed/motif_regions processed/motif_fa
for tf in TF1 TF2 TF3 TF4 TF5 TF6; do
  sort -k5,5nr processed/macs/${tf}_rep1_summits.bed | head -10000 | bedtools slop -i - -g processed/hg38.chrom.sizes -b 100 > processed/motif_regions/${tf}_top10k.bed
  bedtools getfasta -fi processed/hg38.fa.gz -bed processed/motif_regions/${tf}_top10k.bed -fo processed/motif_fa/${tf}_top10k.fa
  echo $tf $(grep -c '"'"'^>'"' processed/motif_fa/"'${tf}_top10k.fa)
done'
