import pyBigWig, numpy as np, glob, os
bw=pyBigWig.open('dnase_rep1_ENCFF768KOM.bigWig')
sizes=bw.chroms()
def vals(peakfile):
    out=[]
    with open(peakfile) as f:
        for line in f:
            if line.startswith('track'): continue
            p=line.split('\\t'); chrom=p[0]; start=int(p[1]); end=int(p[2])
            if chrom not in sizes: continue
            start=max(0,start); end=min(end,sizes[chrom])
            if start>=end: continue
            v=bw.stats(chrom,start,end,type='mean',exact=True)[0]
            if v is not None: out.append(v)
    return np.array(out)
for f in sorted(glob.glob('processed/macs/*_peaks.narrowPeak')):
    # filter q0.001 by col9 >=3
    arr=[]
    with open(f) as fh:
        for line in fh:
            if line.startswith('track'): continue
            p=line.rstrip('\\n').split('\\t')
            if float(p[8])<3: continue
            chrom=p[0]; start=int(p[1]); end=int(p[2])
            if chrom not in sizes: continue
            start=max(0,start); end=min(end,sizes[chrom])
            if start>=end: continue
            v=bw.stats(chrom,start,end,type='mean',exact=True)[0]
            if v is not None: arr.append(v)
    a=np.array(arr)
    for t in [0.05]:
        acc=(a>t).mean(); print(f'{os.path.basename(f)} n={len(a)} res={1-acc:.3f} acc={acc:.3f}')
bw.close()
PY"
