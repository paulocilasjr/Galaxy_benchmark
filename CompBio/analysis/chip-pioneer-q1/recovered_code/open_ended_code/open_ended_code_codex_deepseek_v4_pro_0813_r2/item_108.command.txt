/bin/bash -lc "python3 - <<'PY'
import pandas as pd
D=pd.read_csv('processed/geo_overlap_matrix.tsv',sep='\\t',header=None,names=['sample','factor','covered','total','sum','mean','frac'])
totals=pd.read_csv('processed/frag_access_counts.tsv',sep='\\t',header=None,names=['sample','acc','inacc','total']).set_index('sample')['total'].to_dict()
D['rpm_per_peak']=D['mean']/D['sample'].map(totals)*1e6
print('totals')
for k,v in totals.items(): print(k,v)
piv=D.pivot_table(index='sample',columns='factor',values='rpm_per_peak')
print('\\nRPM per peak')
print(piv.round(3).to_string())
print('\\nCoverage fraction')
piv2=D.pivot_table(index='sample',columns='factor',values='frac')
print(piv2.round(3).to_string())
# rank factor by rpm per sample
print('\\nBest factor per sample by rpm')
for s,row in piv.iterrows():
    print(s, row.idxmax(), row.max().round(3))
PY"
