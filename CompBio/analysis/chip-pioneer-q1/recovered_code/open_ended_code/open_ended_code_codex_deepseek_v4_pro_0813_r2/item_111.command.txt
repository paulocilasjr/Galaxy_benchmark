/bin/bash -lc "python3 - <<'PY'
import os, subprocess
samples=[]
for f in sorted(os.listdir('processed/macs')):
    if f.endswith('_peaks.narrowPeak'):
        samples.append(f[:-len('_peaks.narrowPeak')])
factors=['FOXA1wt','FOXA1dbd','FOXA1nhaa','FOXA1rraa','HNF4A']
print('\\t'.join(['sample']+factors+['n']))
for s in samples:
    peak='processed/macs/'+s+'_peaks.narrowPeak'
    n=int(subprocess.check_output(['wc','-l','<',peak]).decode().strip())
    vals=[]
    for fac in factors:
        geo=f'geo_peaks_hg38/{fac}.bed'
        inter=subprocess.check_output(['bedtools','intersect','-u','-a',peak,'-b',geo], text=True)
        # count lines (bedtools may include track header? narrowPeak no track? macs narrowPeak has header track)
        count=sum(1 for line in inter.splitlines() if line.strip() and not line.startswith('track'))
        vals.append(f'{count/n:.3f}')
    print('\\t'.join([s]+vals+[str(n)]))
PY"
