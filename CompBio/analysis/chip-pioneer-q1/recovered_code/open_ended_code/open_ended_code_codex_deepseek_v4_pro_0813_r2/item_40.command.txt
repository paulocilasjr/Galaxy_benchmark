/bin/bash -lc "python3 - <<'PY'
from pypdf import PdfReader
r=PdfReader('paper.pdf')
with open('paper.txt','w') as f:
    for i,p in enumerate(r.pages):
        f.write(f'\\n===PAGE {i+1}===\\n')
        f.write(p.extract_text() or '')
print('done',len(r.pages))
PY
wc -l paper.txt
grep -n -i -E 'pioneer|HNF4A|FOXA1|SOX2|DNase|48' paper.txt | head -100"
