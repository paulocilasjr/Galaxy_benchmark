/bin/bash -lc "python3 - <<'PY'
from pyliftover import LiftOver
import gzip, os
lo=LiftOver('processed/hg19ToHg38.over.chain.gz')
for fn in ['SOX2wt.bed.gz','SOX2dbd.bed.gz']:
    factor=fn[:-7]
    out=f'geo_peaks_hg38/{factor}.bed'
    n=m=0
    with gzip.open('geo_peaks/'+fn,'rt') as f, open(out,'w') as o:
        for line in f:
            if line.startswith('#'): continue
            p=line.rstrip('\\n').split('\\t')
            if len(p)<3: continue
            chrom=p[0]; start=int(p[1]); end=int(p[2]); width=end-start; mid=(start+end)//2
            try: res=lo.convert_coordinate(chrom,mid,'+')
            except: res=[]
            if res:
                newchrom,newpos,_,_=res[0]
                o.write(f'{newchrom}\\t{max(0,newpos-width//2)}\\t{max(0,newpos-width//2)+width}\\n'); m+=1
            n+=1
    print(factor,n,m)
PY
for f in geo_peaks_hg38/SOX2*.bed; do echo "'$f $(wc -l < $f); done'
