/bin/bash -lc "python3 - <<'PY'
import subprocess
for fac in ['FOXA1wt','FOXA1nhaa','FOXA1rraa','FOXA1dbd','HNF4A','SOX2wt','SOX2dbd']:
    geo=f'geo_peaks_hg38/{fac}.bed'
    n=sum(1 for _ in open(geo))
    vals=[]
    for tf in ['TF1','TF2','TF3','TF4','TF5','TF6']:
        peaks=f'processed/union_peaks_q001/{tf}.bed'
        a=int(subprocess.check_output(['sh','-c',f'bedtools intersect -u -a {geo} -b {peaks} | wc -l']).decode().strip())
        vals.append(f'{tf}:{a/n:.3f}')
    print(fac,'\\t'.join(vals))
PY"
