/bin/bash -lc "curl -L --fail --silent --show-error 'https://www.encodeproject.org/files/ENCFF855TSQ/@@download/ENCFF855TSQ.bed.gz' -o dnase/rep1.hotspots.bed.gz & curl -L --fail --silent --show-error 'https://www.encodeproject.org/files/ENCFF519XYN/@@download/ENCFF519XYN.bed.gz' -o dnase/rep2.hotspots.bed.gz & wait
(gzip -cd dnase/rep1.narrowPeak.bed.gz; gzip -cd dnase/rep2.narrowPeak.bed.gz) | LC_ALL=C sort -k1,1 -k2,2n | bedtools merge -i - > dnase/union_narrow.bed
(gzip -cd dnase/rep1.hotspots.bed.gz; gzip -cd dnase/rep2.hotspots.bed.gz) | LC_ALL=C sort -k1,1 -k2,2n | bedtools merge -i - > dnase/union_hotspots.bed
wc -l dnase/union_narrow.bed dnase/union_hotspots.bed
for tf in TF5 TF6; do
 awk 'BEGIN{OFS=\"\\t\"}{s="'$2+$10; print $1,s,s+1}'"' macs_pooled/"'$tf/${tf}_peaks.narrowPeak > macs_pooled/$tf/${tf}_summit1.bed
 total=$(wc -l < macs_pooled/$tf/${tf}_summit1.bed)
 n_open=$(bedtools intersect -u -a macs_pooled/$tf/${tf}_summit1.bed -b dnase/union_narrow.bed | wc -l)
 h_open=$(bedtools intersect -u -a macs_pooled/$tf/${tf}_summit1.bed -b dnase/union_hotspots.bed | wc -l)
 echo "$tf total=$total narrow_open=$n_open hotspot_open=$h_open narrow_closed=$((total-n_open)) hotspot_closed=$((total-h_open))"
done'
