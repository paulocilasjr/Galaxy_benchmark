import csv
for tf in ('TF5','TF6'):
 a=[]
 for r in csv.reader(open(f'macs_pooled/{tf}/{tf}_annot.tsv'),delimiter='\\t'):
  a.append({'fe':float(r[6]),'lp':float(r[7]),'lq':float(r[8]),'open':int(r[11])>0})
 print('\\n',tf,len(a))
 for key in ('fe','lp','lq'):
  for th in ([3,5,10,20] if key=='fe' else [3,5,10,20,50,100]):
   z=[x for x in a if x[key]>=th]
   if z: print(key,th,len(z),sum(not x['open'] for x in z),round(sum(not x['open'] for x in z)/len(z),3))
 for n in (1000,5000,10000,20000):
  z=sorted(a,key=lambda x:x['lp'],reverse=True)[:n]
  print('top',n,'closed',sum(not x['open'] for x in z),round(sum(not x['open'] for x in z)/len(z),3))
