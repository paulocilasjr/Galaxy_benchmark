/bin/bash -lc 'mkdir -p macs_pooled/TF2 macs_pooled/TF5
macs3 callpeak -t chip.pioneer.q1/TF2.rep1.bed.gz chip.pioneer.q1/TF2.rep2.bed.gz -c chip.pioneer.q1/TF2.control1.bed.gz chip.pioneer.q1/TF2.control2.bed.gz -f BEDPE -g hs --keep-dup all --call-summits -q 0.01 -n TF2 --outdir macs_pooled/TF2 > macs_pooled/TF2/run.log 2>&1 &
macs3 callpeak -t chip.pioneer.q1/TF5.rep1.bed.gz chip.pioneer.q1/TF5.rep2.bed.gz -c chip.pioneer.q1/TF5.control.bed.gz -f BEDPE -g hs --keep-dup all --call-summits -q 0.01 -n TF5 --outdir macs_pooled/TF5 > macs_pooled/TF5/run.log 2>&1 &
wait
wc -l macs_pooled/TF2/TF2_peaks.narrowPeak macs_pooled/TF5/TF5_peaks.narrowPeak
tail -n 4 macs_pooled/TF2/run.log
tail -n 4 macs_pooled/TF5/run.log'
