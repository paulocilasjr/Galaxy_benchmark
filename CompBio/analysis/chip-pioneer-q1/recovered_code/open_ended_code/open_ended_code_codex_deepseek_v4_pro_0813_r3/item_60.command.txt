/bin/bash -lc "python3 - <<'PY'
import glob, os, subprocess
regions={'hotspots':'dnase/hotspots_union.bed','peaks':'dnase/peaks_union.bed'}
files=sorted(glob.glob('macs/TF*_peaks.narrowPeak'))
for regname,reg in regions.items():
    print('\\nREGION',regname,'main top')
    for ntop in [1000,3000,5000,10000,20000]:
        print('top',ntop,end=' ')
        for f in files:
            tf=os.path.basename(f).split('_')[0]
            peaks=[]
            with open(f) as fh:
                for line in fh:
                    p=line.rstrip().split('\\t')
                    if len(p)<9: continue
                    if not (p[0].startswith('chr') and p[0] in ['chr'+str(i) for i in range(1,23)]+['chrX','chrY','chrM']): continue
                    q=float(p[8]) if p[8] not in ('.','-1') else 0
                    peaks.append((q,p))
            peaks.sort(key=lambda x:x[0], reverse=True)
            top=peaks[:min(ntop,len(peaks))]
            with open('[REDACTED LOCAL PATH]','w') as out:
                for q,p in top: out.write('\\t'.join(p[:3])+'\\n')
            total=len(top)
            acc=int(subprocess.check_output(f\"bedtools intersect -u -a [REDACTED LOCAL PATH] -b {reg} | wc -l\",shell=True,text=True).strip())
            print(f'{tf}:{ (total-acc)/total:.3f}', end=' ')
        print()
PY"
