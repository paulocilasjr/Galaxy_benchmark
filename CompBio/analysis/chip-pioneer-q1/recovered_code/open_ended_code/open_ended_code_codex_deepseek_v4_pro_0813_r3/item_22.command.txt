/bin/bash -lc "python3 - <<'PY'
import re, math, collections
rows=[]
for line in open('overlap_counts.tsv'):
    f,r,n,a=line.strip().split('\\t')
    n=int(n); a=int(a)
    m=re.search(r'TF(\\d+)\\.(rep\\d|control\\d?)', f)
    tf=int(m.group(1)); typ=m.group(2)
    reg='hot' if 'hotspots' in r else 'peak'
    rows.append((tf,typ,reg,n,a))
# print per file metrics
print('per file fraction accessible (hot/peak), closed enrichment vs pooled? no')
for tf,typ,reg,n,a in rows:
    if reg=='hot':
        print(f'TF{tf} {typ:8s} total={n:>10} acc={a:>10} fa={a/n:.6f}')
# aggregate by TF kind for each region
for regname, reg in [('hotspots','hot'),('peaks','peak')]:
    print('\\nREGION', regname)
    data=collections.defaultdict(lambda: {'chip_n':0,'chip_a':0,'ctrl_n':0,'ctrl_a':0})
    for tf,typ,key,n,a in rows:
        if key"'!=reg: continue
        d=data[tf]
        if typ.startswith('"'rep'):
            d['chip_n']+=n; d['chip_a']+=a
        else:
            d['ctrl_n']+=n; d['ctrl_a']+=a
    print('TF chip_total chip_acc chip_fa ctrl_total ctrl_acc ctrl_fa chip_fc ctrl_fc log2fc_enrich closed_diff')
    vals=[]
    for tf in sorted(data):
        d=data[tf]
        cfa=d['chip_a']/d['chip_n']; tfa=d['ctrl_a']/d['ctrl_n']
        cfc=1-cfa; tfc=1-tfa
        l2=math.log2(cfc/tfc)
        diff=cfc-tfc
        vals.append((l2,tf,cfc,tfc,diff,cfa,tfa))
        print(tf, d['chip_n'], d['chip_a'], round(cfa,6), d['ctrl_n'], d['ctrl_a'], round(tfa,6), round(cfc,6), round(tfc,6), round(l2,6), round(diff,6))
    print('ranking by closed enrichment')
    for l2,tf,cfc,tfc,diff,cfa,tfa in sorted(vals, reverse=True):
        print(f'TF{tf}: log2={l2:.6f} closed_chip={cfc:.6f} closed_ctrl={tfc:.6f} diff={diff:.6f} open_fa_chip={cfa:.6f} open_fa_ctrl={tfa:.6f}')
PY"
