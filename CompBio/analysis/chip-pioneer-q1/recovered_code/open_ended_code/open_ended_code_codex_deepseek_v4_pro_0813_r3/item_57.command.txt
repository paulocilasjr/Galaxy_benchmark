/bin/bash -lc "python3 - <<'PY'
import glob, os, subprocess
# build individual sorted hotspot main files maybe already unsorted? create union from each? use gzip and bedtools sort? let's create quick by sorting cat with genome order
for rep in ['rep1','rep2']:
    # create temp sorted main hotspots
    subprocess.run(f\"(gzip -dc dnase/{rep}_hotspots.bed.gz) | awk 'BEGIN{{FS=OFS=\\\"\\\\t\\\"}} "'$1 ~ /''^chr([1-9]|1[0-9]|2[0-2]|X|Y|M)$/ {{print $1,$2,$3}}'"' | bedtools sort -i - -g dnase/genome.order | bedtools merge -i - > dnase/{rep}_hotspots_union.bed\", shell=True, check=True)
    reg=f'dnase/{rep}_hotspots_union.bed'
    print('\\n',rep)
    for f in sorted(glob.glob('macs/TF*_peaks.narrowPeak')):
        tf=os.path.basename(f).split('_')[0]
        total=int(subprocess.check_output(f\"wc -l < '{f}'\",shell=True,text=True).strip())
        acc=int(subprocess.check_output(f\"bedtools intersect -u -a '{f}' -b '{reg}' | wc -l\",shell=True,text=True).strip())
        print(tf, round((total-acc)/total,3), end=' | ')
    print()
PY"
