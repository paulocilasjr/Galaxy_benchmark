/bin/bash -lc "set -e
for x in a b; do :; done
gzip -cd ENCFF169SUG.bed.gz | awk 'BEGIN{OFS=\"\\t\"} "'$1 !~ /''^#/ {print $1,$2,$3}'"' | sort -k1,1 -k2,2n > dnase19.a.bed
gzip -cd ENCFF474ARN.bed.gz | awk 'BEGIN{OFS=\"\\t\"} "'$1 !~ /''^#/ {print $1,$2,$3}'"' | sort -k1,1 -k2,2n > dnase19.b.bed

for f in GSE299017_*peaks_hg19.bed.gz; do
 b="'$(basename "$f" .bed.gz)
 n=$(gzip -cd "$f" | awk '"'"'$1 !~ /''^#/{n++} END{print n}'"')
 for s in union intersection; do
  o="'$(gzip -cd "$f" | awk '"'"'$1 !~ /''^#/{print $1,$2,$3}'"' OFS='\\t' | bedtools intersect -a stdin -b \"dnase19."'$s.bed" -u | wc -l | tr -d '"' ')
  printf '%s\\t%s\\t%s\\t%s\\n' \""'$b" "$n" "$o" "$((n-o))"
 done
done > peak_overlap19.tsv
wc -l dnase19.*; column -t peak_overlap19.tsv'
