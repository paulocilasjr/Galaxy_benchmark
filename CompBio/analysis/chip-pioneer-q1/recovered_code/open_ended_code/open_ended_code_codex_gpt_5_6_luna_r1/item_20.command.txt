/bin/bash -lc 'set -e
: > counts.current001.union.tsv
for f in chip.pioneer.q1/*.bed.gz; do
 base=$(basename "$f" .bed.gz)
 gzip -cd "$f" | bedtools intersect -a stdin -b current001.union.bed -c | awk -v x="$base" '"'BEGIN{t=o=0} {t++; if("'$NF>0)o++} END{printf "%s'"\\t%d\\t%d\\t%d\\n\",x,t,o,t-o}' >> counts.current001.union.tsv
done
sort counts.current001.union.tsv"
