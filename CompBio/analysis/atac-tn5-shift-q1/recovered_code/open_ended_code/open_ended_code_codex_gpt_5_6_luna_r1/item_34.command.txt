/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip, numpy as np, time
f=TwoBitFile('hg38.2bit'); offs=range(-12,13)
cs=np.zeros((25,4),int); ce=np.zeros((25,4),int); ce_rev=np.zeros((25,4),int)
ix=np.full(256,-1,dtype=int); ix[ord('A')]=0; ix[ord('C')]=1; ix[ord('G')]=2; ix[ord('T')]=3
comp=np.array([3, -1, 1, 2],dtype=int) # index complement A,C,G,T
n=0;t=time.time()
with gzip.open('unknown.shift.frag.bed.gz','rt') as h:
 for line in h:
  c,s,e,*_=line.rstrip().split('\\t'); s=int(s); e=int(e); seq=f[c]
  a=seq[s-12:s+13].upper().encode()
  b=seq[e-12:e+13].upper().encode()
  # oriented end: reverse complement of window e-12:e+13, relative e at index 12? approximate
  cr=b.translate(bytes.maketrans(b'ACGT',b'TGCA'))[::-1]
  for j,x in enumerate(a):
   z=ix[x]
   if z>=0: cs[j,z]+=1
  for j,x in enumerate(b):
   z=ix[x]
   if z>=0: ce[j,z]+=1
  for j,x in enumerate(cr):
   z=ix[x]
   if z>=0: ce_rev[j,z]+=1
  n+=1
print('n',n,'sec',time.time()-t)
print('offset  start A C G T  endf A C G T  endrev A C G T')
for k,a,b,c in zip(offs,cs,ce,ce_rev):
 print(f'{k:3d}',*(f'{x/x.sum():.4f}' for x in a),*(f'{x/x.sum():.4f}' for x in b),*(f'{x/x.sum():.4f}' for x in c))
PY"
