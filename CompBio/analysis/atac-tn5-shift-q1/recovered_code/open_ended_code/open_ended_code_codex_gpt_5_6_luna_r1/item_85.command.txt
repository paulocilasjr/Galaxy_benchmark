/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip,numpy as np,collections
f=TwoBitFile('hg38.2bit'); trans=str.maketrans('ACGT','TGCA');ix={b:i for i,b in enumerate('ACGT')}
def run(kind, weighted=False):
 S=np.zeros((41,4));n=0
 with gzip.open('unknown.shift.frag.bed.gz','rt') as h:
  for line in h:
   c,s,e,*rest=line.rstrip().split('\\t')
   if c=='chrM':continue
   s=int(s);e=int(e); w=int(rest[-1]) if weighted else 1;seq=f[c]
   a=seq[s-20:s+21] if kind=='start' else seq[e-21:e+20].translate(trans)[::-1]
   for j,x in enumerate(a.upper()):
    if x in ix:S[j,ix[x]]+=w
   n+=w
 S/=S.sum(1,keepdims=True)
 arr=[]
 for c in range(-6,7):
  l=np.array([S[c-d+20] for d in range(1,9)]);r=np.array([S[c+d+20][[3,2,1,0]] for d in range(1,9)])
  arr.append((np.mean((l-r)**2),c,np.corrcoef(l.ravel(),r.ravel())[0,1]))
 print(kind,'weighted',weighted,'n',n,sorted(arr)[:4])
for w in [False,True]:
 run('start',w);run('end',w)
PY"
