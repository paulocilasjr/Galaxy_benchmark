print('center error corr')
for c in range(-8,9):
 left=[];right=[]
 for d in range(1,9):
  i=c-d+20;j=c+d+20;left.append(S[i]);right.append(S[j][comp])
 print(c,np.mean((np.array(left)-np.array(right))**2),np.corrcoef(np.array(left).ravel(),np.array(right).ravel())[0,1])
