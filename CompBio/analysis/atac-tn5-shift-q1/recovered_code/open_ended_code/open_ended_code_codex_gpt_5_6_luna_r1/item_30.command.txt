/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip, numpy as np
f=TwoBitFile('hg38.2bit'); offs=range(-12,13)
cs=np.zeros((25,4),int); ce=np.zeros((25,4),int); ce_rev=np.zeros((25,4),int)
ix={b:i for i,b in enumerate('ACGT')}; comp={'A':'T','C':'G','G':'C','T':'A'}
with gzip.open('unknown.shift.frag.bed.gz','rt') as h:
 for line in h:
  c,s,e,*_=line.rstrip().split('\\t'); s=int(s); e=int(e); seq=f[c]
  for j,k in enumerate(offs):
   b=seq[s+k].upper() if s+k>=0 else 'N'
   if b in ix: cs[j,ix[b]]+=1
   b=seq[e+k].upper() if e+k>=0 else 'N'
   if b in ix: ce[j,ix[b]]+=1
   b=comp.get(seq[e-1-k].upper(),'N') if e-1-k>=0 else 'N'
   if b in ix: ce_rev[j,ix[b]]+=1
print('offset start(A C G T) endf(A C G T) endrev(A C G T) ATs ATf ATr')
for k,a,b,c in zip(offs,cs,ce,ce_rev):
 print(f'{k:3d}',*(f'{x/x.sum():.4f}' for x in a),*(f'{x/x.sum():.4f}' for x in b),*(f'{x/x.sum():.4f}' for x in c), f'{(a[0]+a[3])/a.sum():.4f}',f'{(b[0]+b[3])/b.sum():.4f}',f'{(c[0]+c[3])/c.sum():.4f}')
PY"
