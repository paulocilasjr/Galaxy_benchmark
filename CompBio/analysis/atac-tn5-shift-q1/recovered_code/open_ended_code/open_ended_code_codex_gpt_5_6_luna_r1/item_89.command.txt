/bin/bash -lc "PYTHONPATH=pydeps python3 - <<'PY'
from twobitreader import TwoBitFile
import gzip,numpy as np
f=TwoBitFile('hg38.2bit'); trans=str.maketrans('ACGT','TGCA');ix={b:i for i,b in enumerate('ACGT')}
def center(which):
 S=np.zeros((31,4));n=0
 with gzip.open('unknown.shift.frag.bed.gz','rt') as h:
  for line in h:
   c,s,e,*_=line.rstrip().split('\\t');s=int(s);e=int(e)
   # Convert effective unknown shifts (2,-1) to canonical (+4,-5): +2 and -4
   if which=='start': q=s+2; a=f[c][q-15:q+16]
   else: q=e-4; a=f[c][q-15:q+16].translate(trans)[::-1]
   for j,x in enumerate(a.upper()):
    if x in ix:S[j,ix[x]]+=1
   n+=1
 S/=S.sum(1,keepdims=True);best=[]
 for c in range(-5,6):
  l=np.array([S[c-d+15] for d in range(1,7)]);r=np.array([S[c+d+15][[3,2,1,0]] for d in range(1,7)])
  best.append((np.mean((l-r)**2),c,np.corrcoef(l.ravel(),r.ravel())[0,1]))
 print(which,sorted(best)[:3])
center('start');center('end')
PY"
