import json,urllib.request
u='https://api.genome.ucsc.edu/getData/track?genome=hg38;track=chromInfo'

print(type(d),d.keys())
for k,v in d.items():
 if isinstance(v,list):
  print(k,len(v),[x for x in v if x.get('chrom') in ('chrM','chr1','chrUn_KI270304v1')])
PY"
