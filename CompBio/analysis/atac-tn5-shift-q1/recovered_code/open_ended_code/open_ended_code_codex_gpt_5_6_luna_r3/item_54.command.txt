   if side=='"'R':seq=''.join(comp.get(x,'n') for x in seq[::-1])

    i=24+p;b=seq[i] if i<len(seq) else 'n'

  P={p:{b:C[p][b]/sum(C[p].values()) for b in 'acgt'} for p in C}

    q=2*cen-p

   vals.append((sum(ds)/len(ds),cen))
