with gzip.open('unknown.shift.frag.bed.gz','rt') as f, gzip.open('standardized_chr1.frag.bed.gz','wt') as out:

        p=line.rstrip('\\n').split('\\t')
        if p[0]"'!='"'chr1': continue
        start=int(p[1])-2
        end=int(p[2])+1
        # start = plus-2 (actual +6 to standard +4); end = minus+1 (actual -6 to standard -5)
        if start<0 or start>=end: continue
        out.write(f\"{p[0]}\\t{start}\\t{end}\\t{p[3]}\\t{p[4]}\\n\")
PY
