/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, pysam, collections, time
# chr1 fragments
s=[];e=[]
with gzip.open('unknown.shift.frag.bed.gz','rt') as f:
 for l in f:
  a=l.split('\\t')
  if a[0]=='chr1':s.append(int(a[1]));e.append(int(a[2]))
s=np.array(s);e=np.array(e)
f=pysam.FastaFile('chr1.fa.bgz')
comp=str.maketrans('ACGTN','TGCAN')
def contexts(pos,side):
 # side left: forward; right: reverse complement, coordinate as BED coordinate
 out=[]
 for x in pos:
  # positions x-10...x+10, 0-based bases
  z=f.fetch('chr1',max(0,x-12),x+13).upper()
  # pad if needed; use positions relative to x; fetched starts could differ only at chr start
  if side=='R': z=z.translate(comp)[::-1]
  # center coordinate x corresponds index 12 in forward and reverse context if fetch exact; for near start pad
  if len(z)==25: out.append(z)
  else: out.append('N'*25)
 return out
for side,x in [('L',s),('R',e)]:
 t0=time.time();
 C=contexts(x,side); print(side,'contexts',len(C),'secs',time.time()-t0)
 for off in range(-10,11):
  c=collections.Counter(z[off+12] for z in C)
  n=sum(c[b] for b in 'ACGT')
  # info content over base freq, KL against genome uniform; GC
  p=np.array([c[b]/n for b in 'ACGT']) if n else np.zeros(4)
  kl=sum(q*np.log2(q/.25) for q in p if q)
  print(f'{off:+d}', ''.join(f'{b}:{c[b]/n:.3f} ' for b in 'ACGT'),f'GC {p[1]+p[2]:.3f} KL {kl:.4f}')
PY"
