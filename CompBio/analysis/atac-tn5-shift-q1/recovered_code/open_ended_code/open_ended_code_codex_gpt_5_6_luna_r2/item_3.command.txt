/bin/bash -lc "set -o pipefail
printf 'lines: '; gzip -cd unknown.shift.frag.bed.gz | wc -l
printf 'chromosomes:\\n'; gzip -cd unknown.shift.frag.bed.gz | cut -f1 | sort | uniq -c | head -n 30
printf 'field count: '; gzip -cd unknown.shift.frag.bed.gz | awk -F '\\t' '{print NF; if(NR==20) exit}' | sort -nu | tr '\\n' ' '; echo
printf 'length stats:\\n'; gzip -cd unknown.shift.frag.bed.gz | awk -F '\\t' '{n++; l="'$3-$2; s+=l; ss+=l*l; if(l<0) neg++; if(l<100) a++; if(l<200) b++; if(l<500) c++;} END{print n,s/n,neg,a,b,c}'"'
printf 'min/max endpoints:\\n'; gzip -cd unknown.shift.frag.bed.gz | awk -F '\\t' 'BEGIN{mi=1e18; ma=-1} {if("'$2<mi)mi=$2;if($3>ma)ma=$3} END{print mi,ma}'"'
printf 'strand/count col5 stats:\\n'; gzip -cd unknown.shift.frag.bed.gz | awk -F '\\t' '{s+="'$5; if($5>1)d++} END{print "total",s,"duplicates",d}'"'
"
