/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse
for term in ['GCF_000001405.39[Assembly Accession]','GRCh38.p13[Assembly Name]','GCA_000001305.2[Assembly Accession]']:
    q=urllib.parse.quote(term)
    url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=assembly&term={q}&retmode=json'
    data=urllib.request.urlopen(url,timeout=30).read().decode()
    print(term)
    print(data)
PY"
