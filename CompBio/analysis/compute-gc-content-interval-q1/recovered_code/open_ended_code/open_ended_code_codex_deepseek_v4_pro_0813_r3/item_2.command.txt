/bin/bash -c "python3 - <<'PY'
import urllib.request, json, hashlib
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr1;start=999999;end=1000100'
obj=json.load(urllib.request.urlopen(url))
seq=obj['dna']
up=seq.upper()
gc=sum(1 for c in up if c in 'GC')
n=up.count('N')
print(len(seq), len(up), gc, round(gc/len(up)*100,2), n, hashlib.md5(up.encode()).hexdigest())
print(up)
PY"
