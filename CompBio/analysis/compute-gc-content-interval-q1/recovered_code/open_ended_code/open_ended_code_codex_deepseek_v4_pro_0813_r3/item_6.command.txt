/bin/bash -c "python3 - <<'PY'
import urllib.request, gzip, hashlib, time
url='https://ftp.ncbi.nlm.nih.gov/genomes/all/GCF/000/001/405/GCF_000001405.39_GRCh38.p13/GCF_000001405.39_GRCh38.p13_assembly_structure/Primary_Assembly/assembled_chromosomes/FASTA/chr1.fna.gz'
start=1_000_000
end=1_000_100
response=urllib.request.urlopen(url, timeout=120)
g=gzip.GzipFile(fileobj=response)
collected=[]; pos=0; started=False; first=True
header=None
t=time.time()
for raw in g:
    if first:
        header=raw.decode().strip(); first=False
        print('header', header)
        continue
    line=raw.decode().strip()
    if not line: continue
    # one sequence per file, no extra headers expected
    ls=len(line)
    if pos < end:
        # overlap from current line
        take_start=max(0, start-pos)
        take_end=min(ls, end-pos)
        if take_end>take_start:
            collected.append(line[take_start:take_end])
    pos += ls
    if pos >= end:
        break
seq=''.join(collected)
print('len seq', len(seq), 'pos', pos, 'time', round(time.time()-t,2))
print(seq)
up=seq.upper()
gc=sum(1 for c in up if c in 'GC')
n=up.count('N')
print('metrics', len(up), gc, round(gc/len(up)*100,2), n, hashlib.md5(up.encode()).hexdigest())
PY"
