/bin/bash -c "python3 - <<'PY'
import urllib.request
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr1;start=999999;end=1000100'
print(url)
try:
    with urllib.request.urlopen(url, timeout=30) as r:
        data=r.read().decode()
    print(data[:1000])
except Exception as e:
    print('ERR',e)
PY"
