/bin/bash -lc "python3 - <<'PY'
from urllib.request import urlopen
from collections import Counter
ref=''.join(urlopen('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text').read().decode().splitlines()[1:]).upper()
tr=str.maketrans('ACGTN','TGCAN')
def rc(s):return s.translate(tr)[::-1]
def align(s):
 hits=[]
 for strand,q in [(1,s),(-1,rc(s))]:
  p=ref.find(q)
  if p>=0:hits.append((0,p,strand))
 if hits:return min(hits)
 best=(51,0,0)
 for strand,q in [(1,s),(-1,rc(s))]:
  for i in range(len(ref)-len(q)+1):
   d=sum(a"'!=b for a,b in zip(q,ref[i:i+len(q)]))
   if d<best[0]:best=(d,i,strand)
 return best
for n in ['"'exo','endo','mix']:
 with open(f'exogenous.mix.reads.q1.{n}.fq') as f:z=[x.rstrip() for x in f]
 A=[align(s) for s in z[1::4]]
 pos=[x[1] for x in A]
 outside=sum(not (597<=p<=1987) for p in pos)
 low=sum(p<597 for p in pos); high=sum(p>1987 for p in pos)
 N=len(pos); endofrac=(outside/N)/((597+(2886-1987))/2887)
 print(n,'N',N,'outside CDS-support',outside,'low',low,'high',high,'implied endo frac',endofrac,'exo',1-endofrac)
 print('CDS pos quantiles',*[sorted(pos)[int((N-1)*q)] for q in [0,.01,.05,.25,.5,.75,.95,.99,1]])
 print('orientation',Counter(x[2] for x in A),'mismatches',Counter(x[0] for x in A))
PY"
