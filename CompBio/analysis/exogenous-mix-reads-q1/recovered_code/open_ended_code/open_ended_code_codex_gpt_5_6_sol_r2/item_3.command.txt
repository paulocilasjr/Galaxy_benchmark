from collections import Counter
fs=['exo','endo','mix']
data={}

  lines=[x.rstrip() for x in f]

 data[n]=s
 print(n,len(s),'unique',len(set(s)),'lengths',Counter(map(len,s)))
for a in fs:
 for b in fs:
  if a<b:
   print(a,b,'exact shared unique',len(set(data[a])&set(data[b])))
