/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict

def readseqs(path):
    out=[]
    with open(path) as f:
        for i,line in enumerate(f):
            if i%4==1: out.append(line.strip().upper())
    return out
seqs={k:readseqs(f'exogenous.mix.reads.q1.{k}.fq') for k in ['exo','endo','mix']}
for k in [8,10,12,15,20,25,31]:
    km={}
    for cls in ['exo','endo']:
        c=Counter()
        for s in seqs[cls]:
            for i in range(len(s)-k+1): c[s[i:i+k]]+=1
        km[cls]=c
    exo_only=set(km['exo'])-set(km['endo'])
    endo_only=set(km['endo'])-set(km['exo'])
    both=set(km['exo'])&set(km['endo'])
    print('\\nk',k,'exo km',len(km['exo']),'endo',len(km['endo']),'exo_only',len(exo_only),'endo_only',len(endo_only),'both',len(both))
    class_counts=Counter(); read_score=[]
    for s in seqs['mix']:
        eo=sum(1 for i in range(len(s)-k+1) if s[i:i+k] in exo_only)
        no=sum(1 for i in range(len(s)-k+1) if s[i:i+k] in endo_only)
        if eo>0 and no==0: cl='exo'
        elif no>0 and eo==0: cl='endo'
        elif eo>no: cl='exo_mixed'
        elif no>eo: cl='endo_mixed'
        elif eo==no==0: cl='unknown'
        else: cl='tie'
        class_counts[cl]+=1
        read_score.append((eo,no,s))
    print(class_counts, 'exo-ish', class_counts['exo']+class_counts['exo_mixed'], 'endo-ish', class_counts['endo']+class_counts['endo_mixed'])
PY"
