/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def readseqs(path):
    with open(path) as f:
        return [line.strip().upper() for i,line in enumerate(f) if i%4==1]
seqs={k:readseqs(f'exogenous.mix.reads.q1.{k}.fq') for k in ['exo','endo','mix']}
for k in [8,10,12,15,20,25,31,35,40,45,50]:
    km={}
    for cls in ['exo','endo']:
        S=set()
        for s in seqs[cls]:
            for i in range(len(s)-k+1): S.add(s[i:i+k])
        km[cls]=S
    exo_only=km['exo']-km['endo']; endo_only=km['endo']-km['exo']
    def cls_counts(reads):
        c=Counter()
        for s in reads:
            eo=any(s[i:i+k] in exo_only for i in range(len(s)-k+1))
            no=any(s[i:i+k] in endo_only for i in range(len(s)-k+1))
            if eo and not no: c['exo']+=1
            elif no and not eo: c['endo']+=1
            elif eo and no: c['both']+=1
            else: c['unknown']+=1
        return c
    ce=cls_counts(seqs['exo']); cn=cls_counts(seqs['endo']); cm=cls_counts(seqs['mix'])
    pe=ce['exo']/len(seqs['exo']); pn=cn['endo']/len(seqs['endo'])
    # solve from exo conf (assuming no false) and endo conf separately: alpha_exo=m_exo/(N*pe), alpha_endo=1-m_endo/(N*pn)
    ax=cm['exo']/len(seqs['mix'])/pe if pe else None
    an=1-cm['endo']/len(seqs['mix'])/pn if pn else None
    # combine by least squares categories exo/endo/unknown with p vectors
    print('k',k,'train exo',ce,'endo',cn,'mix',cm,'pe/pn',round(pe,3),round(pn,3),'alpha_x/n',None if ax is None else round(ax,3),None if an is None else round(an,3))
PY"
