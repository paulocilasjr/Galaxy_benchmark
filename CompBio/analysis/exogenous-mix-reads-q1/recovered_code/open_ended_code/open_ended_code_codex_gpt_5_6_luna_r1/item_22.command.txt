/bin/bash -lc "python -u - <<'PY'
from urllib.request import urlopen
from pathlib import Path
from collections import Counter
ref=''.join(urlopen('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text',timeout=30).read().decode().splitlines()[1:]).upper()
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
def reads(p): return Path(p).read_text().splitlines()[1::4]

def best(s):
 best=(999,None,None)
 for ori,t in [('+',s),('-',rc(s))]:
  for p in range(len(ref)-len(t)+1):
   # early cutoff
   d=sum(a"'!=b for a,b in zip(t,ref[p:p+len(t)]))
   if d<best[0]: best=(d,p,ori)
 return best
for name in ['"'exo','endo','mix']:
 a=reads(f'exogenous.mix.reads.q1.{name}.fq'); no=[]
 for s in a:
  if ref.find(s)<0 and ref.find(rc(s))<0: no.append(s)
 print('\\n',name,'nohits',len(no))
 ds=[]
 for s in no:
  b=best(s); ds.append(b[0]); print(b,s) if name=='mix' else None
 print('dist counts',Counter(ds))
PY"
