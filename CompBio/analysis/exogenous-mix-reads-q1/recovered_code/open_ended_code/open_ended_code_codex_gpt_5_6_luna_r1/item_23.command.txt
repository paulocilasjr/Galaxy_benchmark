/bin/bash -lc "python -u - <<'PY'
from urllib.request import urlopen
from pathlib import Path
from collections import Counter
ref=''.join(urlopen('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text',timeout=30).read().decode().splitlines()[1:]).upper()
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
def reads(p): return Path(p).read_text().splitlines()[1::4]
def best(s):
 f=ref.find(s); r=ref.find(rc(s))
 if f>=0:return f,0,'+'
 if r>=0:return r,0,'-'
 bestd=99; bestp=None; besto=None
 for ori,t in [('+',s),('-',rc(s))]:
  for p in range(len(ref)-49):
   d=0
   for a,b in zip(t,ref[p:p+50]):
    d += a"'!=b
   if d<bestd:bestd,bestp,besto=d,p,ori
 return bestp,bestd,besto
CDS0,CDS1=597,2037
for name in ['"'exo','endo','mix']:
 ps=[]; ds=[]
 for s in reads(f'exogenous.mix.reads.q1.{name}.fq'):
  p,d,o=best(s);ps.append(p);ds.append(d)
 out=[p for p in ps if p<CDS0 or p+50>CDS1]
 whole=[p for p in ps if p+50<=CDS0 or p>=CDS1]
 print(name,'n',len(ps),'dist',Counter(ds),'range',min(ps),max(ps),'out/intersect',len(out),'whole utr',len(whole),'boundary',len(out)-len(whole),'p',len(out)/len(ps))
 print('bins200',' '.join(f'{a}:{sum(a<=p<a+200 for p in ps)}' for a in range(0,len(ref),200)))
q=sum(p<CDS0 or p+50>CDS1 for p in [best(s)[0] for s in reads('exogenous.mix.reads.q1.endo.fq')])/395
obs=sum(p<CDS0 or p+50>CDS1 for p in [best(s)[0] for s in reads('exogenous.mix.reads.q1.mix.fq')])/1965
print('f ratio',1-obs/q,'q',q,'obs',obs)
PY"
