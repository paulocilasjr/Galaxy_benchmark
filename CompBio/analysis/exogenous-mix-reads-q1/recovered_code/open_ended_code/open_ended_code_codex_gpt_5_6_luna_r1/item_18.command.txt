/bin/bash -lc "python -u - <<'PY'
from urllib.request import urlopen
from pathlib import Path
from collections import Counter
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.6&rettype=fasta&retmode=text'
ref=''.join(urlopen(url,timeout=30).read().decode().splitlines()[1:]).upper()
comp=str.maketrans('ACGTN','TGCAN')
def rc(s): return s.translate(comp)[::-1]
def reads(p): return Path(p).read_text().splitlines()[1::4]
def hits(a):
 out=[]
 for s in a:
  f=ref.find(s); r=ref.find(rc(s))
  out.append(f if f>=0 else r if r>=0 else None)
 return out
for name in ['exo','endo','mix']:
 hs=hits(reads(f'exogenous.mix.reads.q1.{name}.fq'))
 print('\\n',name,'n',len(hs),'hits',sum(x is not None for x in hs),'no',sum(x is None for x in hs))
 bins=[(0,199),(200,399),(400,599),(600,799),(800,999),(1000,1199),(1200,1399),(1400,1599),(1600,1799),(1800,1999),(2000,2199),(2200,2399),(2400,2599),(2600,2799),(2800,2999)]
 print('bins')
 print(' '.join(f'{a}-{b}:{sum(a<=x<=b for x in hs if x is not None)}' for a,b in bins))
 print('top bin 50')
 print(' '.join(f'{a}:{sum(a<=x<a+50 for x in hs if x is not None)}' for a in range(0,len(ref),50)))
PY"
