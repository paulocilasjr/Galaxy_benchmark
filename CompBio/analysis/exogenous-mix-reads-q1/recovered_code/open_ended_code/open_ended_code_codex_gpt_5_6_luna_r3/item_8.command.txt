/bin/bash -lc "python - <<'PY'
from collections import Counter

def recs(path):
    with open(path) as f:
        lines=[x.rstrip('\\n') for x in f]
    return Counter(zip(lines[1::4], lines[3::4]))
R={x:recs(f'exogenous.mix.reads.q1.{x}.fq') for x in ('exo','endo','mix')}
for x in R:
    print(x, 'total',sum(R[x].values()),'unique',len(R[x]))
for a,b in [('exo','endo'),('exo','mix'),('endo','mix')]:
    inter=set(R[a])&set(R[b])
    print(a,b,'unique intersection',len(inter),'multiset overlap',sum(min(R[a][k],R[b][k]) for k in inter))
    print('records in b matching a (with multiplicity)',sum(R[b][k] for k in R[a]))
print('mix source exact assignment:')
print('endo',sum(R['mix'][k] for k in R['endo']))
print('exo',sum(R['mix'][k] for k in R['exo']))
print('both',sum(R['mix'][k] for k in set(R['endo'])&set(R['exo'])))
print('unmatched',sum(v for k,v in R['mix'].items() if k not in R['exo'] and k not in R['endo']))
# frequency relationships
for name in ('exo','endo'):
    vals=[R['mix'][k] for k in R[name]]
    print(name,'mix count for records',Counter(vals), 'sum',sum(vals))
PY"
