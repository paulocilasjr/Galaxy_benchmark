/bin/bash -lc "python - <<'PY'
# manually reuse exact BLAST coords by calling via shell; print categories for various exo-support bounds
import subprocess,collections
P={}
for x in ('exo','endo','mix'):
 q=f\"awk 'NR%4==1{{print \\\">\\\"substr("'$0,2)}} NR%4==2{{print}}'"' exogenous.mix.reads.q1.{x}.fq\"
 cmd=f\"blastn -task blastn-short -query <({q}) -subject <(curl -L --fail --silent 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.4&rettype=fasta&retmode=text') -outfmt '6 qseqid sstart send length pident mismatch gaps' -max_target_seqs 1 -max_hsps 1 -evalue 100\"
 out=subprocess.run(cmd,shell=True,executable='/bin/bash',capture_output=True,text=True,check=True).stdout.splitlines()
 P[x]=[(min(int(f.split('\\t')[1]),int(f.split('\\t')[2])),max(int(f.split('\\t')[1]),int(f.split('\\t')[2]))) for f in out]
for lo,hi in [(595,2034),(601,2033),(602,2033),(602,1983),(601,1983),(600,2034)]:
 print('support',lo,hi)
 for x in P:
  out=sum(a<lo or b>hi for a,b in P[x])
  print(x,out, '/',len(P[x]),round(out/len(P[x]),3),end='; ')
 print()
 e=sum(a<lo or b>hi for a,b in P['endo'])/len(P['endo'])
 m=sum(a<lo or b>hi for a,b in P['mix'])/len(P['mix'])
 print('p_est',round(1-m/e,3))
PY"
