/bin/bash -lc "python - <<'PY'
from urllib.request import urlopen
from collections import Counter,defaultdict
import re
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.4&rettype=fasta&retmode=text'
raw=urlopen(url,timeout=30).read().decode()
ref=''.join(raw.splitlines()[1:]).upper()
print('reference length',len(ref),raw.splitlines()[0])

def reads(path):
    with open(path) as f:
        ls=[x.strip() for x in f]
    return [ls[i] for i in range(1,len(ls),4)]
for name in ('exo','endo','mix'):
    rs=reads(f'exogenous.mix.reads.q1.{name}.fq')
    exact=0; rev=0; no=0; poss=0; positions=[]
    for s in rs:
        p=ref.find(s)
        rc=s.translate(str.maketrans('ACGT','TGCA'))[::-1]
        q=ref.find(rc)
        if p>=0: exact+=1; positions.append(p)
        elif q>=0: exact+=1; rev+=1; positions.append(q)
        else: no+=1
    print(name,'n',len(rs),'exact',exact,'rev',rev,'not exact',no,'unique positions',len(set(positions)),'pos range', (min(positions),max(positions)) if positions else None)
PY"
