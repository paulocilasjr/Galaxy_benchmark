/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict
def reads(p):
    return [l.strip() for i,l in enumerate(open(p)) if i%4==1]
mix=reads('exogenous.mix.reads.q1.mix.fq'); exo=reads('exogenous.mix.reads.q1.exo.fq'); endo=reads('exogenous.mix.reads.q1.endo.fq')
Es=set(exo); En=set(endo)
cnt=Counter(mix)
cat=defaultdict(int)
for s,n in cnt.items():
    e=s in Es; d=s in En
    if e and d: c='both'
    elif e: c='exo_only'
    elif d: c='endo_only'
    else: c='neither'
    cat[c]+=n
print(cat, sum(cat.values()))
for s in mix:
    if s not in Es and s not in En:
        print('neither eg',s); break
# counts weighted by source support: classify if in exo or endo
PY"
