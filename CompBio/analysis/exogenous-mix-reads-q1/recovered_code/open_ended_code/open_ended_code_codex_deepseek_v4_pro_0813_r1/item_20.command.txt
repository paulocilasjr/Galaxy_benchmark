/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
samples=['mix','exo','endo']
for n in samples:
    b=pysam.AlignmentFile(f'exogenous.mix.reads.q1.{n}.bam'); starts=np.array([r.reference_start for r in b if not r.is_unmapped]); b.close()
    inside=((starts>=594)&(starts<=1984)).sum(); outside=len(starts)-inside
    print(n,'total',len(starts),'inside CDS',inside,'outside',outside,'outside frac',outside/len(starts))
PY"
