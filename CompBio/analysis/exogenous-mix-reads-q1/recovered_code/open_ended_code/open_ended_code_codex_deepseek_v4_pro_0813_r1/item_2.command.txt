/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
from collections import Counter
def reads(p):
    seqs=[]
    with open(p) as fh:
        for i,l in enumerate(fh):
            if i%4==1: seqs.append(l.strip())
    return seqs
names=['mix','exo','endo']
data={n:reads('exogenous.mix.reads.q1.'+n+'.fq') for n in names}
for n,s in data.items():
    print(n,len(s),len(set(s)),'unique',Counter(s).most_common(10)[:5])
sets={n:set(s) for n,s in data.items()}
for a in names:
 for b in names:
  print(a,b,'intersection',len(sets[a]&sets[b]),'jaccard seq set',len(sets[a]&sets[b])/len(sets[a]|sets[b]))
PY"
