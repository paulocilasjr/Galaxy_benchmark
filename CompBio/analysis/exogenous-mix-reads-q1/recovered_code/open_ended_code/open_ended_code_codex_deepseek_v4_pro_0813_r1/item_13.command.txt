/bin/bash -lc "python3 - <<'PY'
import pysam, collections, numpy as np
for n in ['exo','endo','mix']:
    f=f'exogenous.mix.reads.q1.{n}.bam'
    b=pysam.AlignmentFile(f)
    mism=[]
    nm=[]
    for r in b:
        if r.is_unmapped: continue
        md=r.get_tag('MD') if r.has_tag('MD') else None
        nmv=r.get_tag('NM') if r.has_tag('NM') else None
        nm.append(nmv)
        if nmv: mism.append((nmv,md,r.query_sequence,r.reference_start))
    print(n,'NM dist',collections.Counter(nm))
    print('mismatched first',mism[:5])
    b.close()
PY"
