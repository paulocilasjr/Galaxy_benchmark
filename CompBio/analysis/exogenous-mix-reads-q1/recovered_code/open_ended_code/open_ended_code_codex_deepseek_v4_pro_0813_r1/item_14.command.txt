/bin/bash -lc "python3 - <<'PY'
import pysam, collections
from collections import Counter,defaultdict
# count per-position bases for each sample from reads with NM maybe
samples=['exo','endo','mix']
ref=pysam.FastaFile('klf4_mrna.fa').fetch('NM_004235.4').upper()
L=len(ref)
for n in samples:
    b=pysam.AlignmentFile(f'exogenous.mix.reads.q1.{n}.bam')
    counts=[Counter() for _ in range(L)]
    depths=[0]*L
    for r in b:
        if r.is_unmapped: continue
        for qpos,refpos in r.get_aligned_pairs(matches_only=True):
            base=r.query_sequence[qpos]
            counts[refpos][base]+=1
            depths[refpos]+=1
    b.close()
    print('\\n',n,'positions with nonref majority or >?')
    # positions depth >=3 and nonref freq >0.2
    out=[]
    for pos in range(L):
        c=counts[pos]
        if not c: continue
        refbase=ref[pos]
        nonref=sum(v for k,v in c.items() if k"'!=refbase)
        if depths[pos]>=3 and nonref/depths[pos]>0.2:
            out.append((pos+1,refbase,depths[pos],dict(c),nonref/depths[pos]))
    print('"'count',len(out))
    for x in out[:80]: print(x)
PY"
