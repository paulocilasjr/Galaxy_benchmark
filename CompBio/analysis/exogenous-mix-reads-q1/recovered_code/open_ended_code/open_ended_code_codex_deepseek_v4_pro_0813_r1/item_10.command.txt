/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np, collections
files=['exogenous.mix.reads.q1.mix.bam','exogenous.mix.reads.q1.exo.bam','exogenous.mix.reads.q1.endo.bam']
datas={}
for f in files:
    b=pysam.AlignmentFile(f); starts=[r.reference_start for r in b if not r.is_unmapped]; b.close(); datas[f]=np.array(starts)
    print(f, len(datas[f]))
# print hist per 100 for each
for f in files:
    s=datas[f]
    hist,bins=np.histogram(s,bins=np.arange(0,3001,100))
    print('\\n',f)
    for c,(lo,hi) in zip(hist,bins[:-1]):
        if c: print(f'{lo:4d}-{hi:4d}: {c}')
PY"
