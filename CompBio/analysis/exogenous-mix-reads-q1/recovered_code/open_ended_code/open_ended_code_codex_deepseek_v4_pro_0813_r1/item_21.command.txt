/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
from scipy.stats import chisquare
for n in ['exo','endo','mix']:
    b=pysam.AlignmentFile(f'exogenous.mix.reads.q1.{n}.bam'); s=np.array([r.reference_start for r in b if not r.is_unmapped]); b.close()
    L=2949
    # 50mer starts 0..2899
    edges=np.arange(0,2901,100); hist=np.histogram(s,bins=edges)[0]
    # expected by positions length per bin (last partial)
    lens=np.array([min(100,2900-lo) for lo in edges[:-1]])
    exp=hist.sum()*lens/2900
    # combine small? print chisq
    mask=exp>0
    chi,p=chisquare(hist[mask],f_exp=exp[mask])
    print(n,'chisq',chi,'p',p,'hist',hist.tolist())
PY"
