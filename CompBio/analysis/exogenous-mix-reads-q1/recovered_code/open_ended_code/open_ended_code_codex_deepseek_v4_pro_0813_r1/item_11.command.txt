/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
files=['exogenous.mix.reads.q1.mix.bam','exogenous.mix.reads.q1.exo.bam','exogenous.mix.reads.q1.endo.bam']
for f in files:
    b=pysam.AlignmentFile(f); s=np.array([r.reference_start for r in b if not r.is_unmapped]); b.close()
    hist,edges=np.histogram(s,bins=np.arange(0,3001,100))
    print('\\n'+f)
    for c,lo in zip(hist,edges[:-1]):
        if c: print(f'{lo:4d}-{lo+99:4d}: {c}')
PY"
