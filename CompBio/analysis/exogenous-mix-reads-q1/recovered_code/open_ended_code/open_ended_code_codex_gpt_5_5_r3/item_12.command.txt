/bin/bash -lc "python3 - <<'PY'
from collections import Counter,defaultdict

def fq(path):
    out=[]
    with open(path) as f:
        rec=[]
        for i,line in enumerate(f):
            rec.append(line.strip())
            if len(rec)==4:
                out.append(tuple(rec)); rec=[]
    return out
D={n:fq(f'exogenous.mix.reads.q1.{n}.fq') for n in ['mix','exo','endo']}
for a in D:
    for b in D:
        if a<b:
            seq_a=Counter(r[1] for r in D[a]); seq_b=Counter(r[1] for r in D[b])
            qual_a=Counter(r[3] for r in D[a]); qual_b=Counter(r[3] for r in D[b])
            bothseq=sum((seq_a & seq_b).values()); bothqual=sum((qual_a & qual_b).values())
            print(a,b,'seq multiset overlap',bothseq,'qual multiset overlap',bothqual,'unique qual overlap',len(set(qual_a)&set(qual_b)))
# How often qual i same across files same index
for a,b in [('exo','endo'),('mix','exo'),('mix','endo')]:
    n=min(len(D[a]),len(D[b]));
    print(a,b,'same qual by index',sum(D[a][i][3]==D[b][i][3] for i in range(n)),'/',n,'same seq by index',sum(D[a][i][1]==D[b][i][1] for i in range(n)))
PY"
