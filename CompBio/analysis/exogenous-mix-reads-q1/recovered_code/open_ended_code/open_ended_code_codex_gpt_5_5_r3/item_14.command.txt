/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def pairs(path):
    c=Counter(); seq=Counter(); qual=Counter()
    with open(path) as f:
        while True:
            h=f.readline()
            if not h: break
            s=f.readline().strip(); p=f.readline(); q=f.readline().strip()
            c[(s,q)]+=1; seq[s]+=1; qual[q]+=1
    return c,seq,qual
D={n:pairs(f'exogenous.mix.reads.q1.{n}.fq') for n in ['mix','exo','endo']}
for src in ['endo','exo']:
    c_src,seq_src,qual_src=D[src]
    c_mix,seq_mix,qual_mix=D['mix']
    print(src,'pair overlap',sum((c_src&c_mix).values()),'of',sum(c_src.values()))
    print('seq missing count',sum(max(0,seq_src[k]-seq_mix.get(k,0)) for k in seq_src))
    print('pair missing count',sum(max(0,c_src[k]-c_mix.get(k,0)) for k in c_src))
    print('qual missing count',sum(max(0,qual_src[k]-qual_mix.get(k,0)) for k in qual_src))
# Counts if all endo pairs embedded in mix; locate mix pairs from source exactly
mix_pairs=D['mix'][0]
exo_pairs=D['exo'][0]
endo_pairs=D['endo'][0]
assign_endo=sum(min(mix_pairs[p], endo_pairs.get(p,0)) for p in mix_pairs)
assign_exo=sum(min(mix_pairs[p], exo_pairs.get(p,0)) for p in mix_pairs)
print('mix overlaps pairs with endo/exo',assign_endo, assign_exo, 'mix total',sum(mix_pairs.values()))
print('source intersection pairs',sum((exo_pairs&endo_pairs).values()), 'unique',len(set(exo_pairs)&set(endo_pairs)))
PY"
