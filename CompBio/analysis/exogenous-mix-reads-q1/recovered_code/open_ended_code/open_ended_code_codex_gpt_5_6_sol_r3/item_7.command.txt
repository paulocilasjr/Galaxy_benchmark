/bin/bash -lc "python3 - <<'PY'
from collections import Counter
import math, random

def reads(fn):
 with open(fn) as f:return [x.strip() for i,x in enumerate(f) if i%4==1]
E=reads('exogenous.mix.reads.q1.exo.fq'); N=reads('exogenous.mix.reads.q1.endo.fq'); M=reads('exogenous.mix.reads.q1.mix.fq')
comp=str.maketrans('ACGTN','TGCAN')
def feat(R,k,canon=False):
 c=Counter()
 for s in R:
  for i in range(len(s)-k+1):
   z=s[i:i+k]
   if canon:
    rc=z.translate(comp)[::-1]; z=min(z,rc)
   c[z]+=1
 tot=sum(c.values())
 return {x:v/tot for x,v in c.items()}
def fit(k,canon=False):
 e,n,m=feat(E,k,canon),feat(N,k,canon),feat(M,k,canon)
 keys=set(e)|set(n)|set(m)
 a=sum((m.get(x,0)-n.get(x,0))*(e.get(x,0)-n.get(x,0)) for x in keys)
 b=sum((e.get(x,0)-n.get(x,0))**2 for x in keys)
 return a/b, math.sqrt(sum((m.get(x,0)-(a/b*e.get(x,0)+(1-a/b)*n.get(x,0)))**2 for x in keys))
for canon in (False,True):
 print('canonical',canon)
 for k in [1,2,3,4,5,6,7,8,10,12,15,20,25,31,40,50]:
  print(k,*(round(x,4) for x in fit(k,canon)))
PY"
