/bin/bash -lc 'for src in exo endo mix; do samtools sort -o "${src}.sorted.bam" "${src}.bam"; done
samtools depth -aa exo.sorted.bam endo.sorted.bam mix.sorted.bam | awk '"'BEGIN{OFS=\"\\t\"} {for(i=3;i<=5;i++) s[i]+="'$i; if(($3+$4+$5)>0 && !on){st=$2;on=1} if(($3+$4+$5)==0&&on){print st,$2-1,s[3],s[4],s[5]; delete s;on=0}} END{if(on) print st,$2,s[3],s[4],s[5]}'"'
samtools view exo.sorted.bam | awk '{print "'$4}'"' | awk '{b=int(("'$1-1)/100)*100+1;c[b]++} END{for(b in c) print b,c[b]}'"' | sort -n
samtools view endo.sorted.bam | awk '{print "'$4}'"' | awk '{b=int(("'$1-1)/100)*100+1;c[b]++} END{for(b in c) print b,c[b]}'"' | sort -n
samtools view mix.sorted.bam | awk '{print "'$4}'"' | awk '{b=int(("'$1-1)/100)*100+1;c[b]++} END{for(b in c) print b,c[b]}'"' | sort -n"
