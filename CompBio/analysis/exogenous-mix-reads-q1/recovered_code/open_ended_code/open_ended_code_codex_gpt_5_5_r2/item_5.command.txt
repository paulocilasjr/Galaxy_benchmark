from collections import Counter
files=['exogenous.mix.reads.q1.mix.fq','exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.endo.fq']
def readseqs(fn):
    seqs=[]
    quals=[]
    with open(fn) as f:
        while True:
            h=f.readline().strip()
            if not h: break
            s=f.readline().strip(); p=f.readline(); q=f.readline().strip()
            seqs.append(s); quals.append(q)
    return seqs, quals
for fn in files:
    seqs,quals=readseqs(fn)
    c=Counter(seqs)
    print(fn, len(seqs), 'unique seqs', len(c), 'dup reads', sum(v-1 for v in c.values() if v>1), 'readlen', Counter(map(len,seqs)))

mix=Counter(readseqs(files[0])[0]); exo=Counter(readseqs(files[1])[0]); endo=Counter(readseqs(files[2])[0])
print('unique overlaps: mix-exo', len(set(mix)&set(exo)), 'mix-endo', len(set(mix)&set(endo)), 'exo-endo', len(set(exo)&set(endo)), 'all', len(set(mix)&set(exo)&set(endo)))
for name,A in [('exo',exo),('endo',endo)]:
    print('mix reads matching',name, sum(v for s,v in mix.items() if s in A), 'unique mix seqs', sum(1 for s in mix if s in A))
print('mix reads class exact: exo_only', sum(v for s,v in mix.items() if s in exo and s not in endo), 'endo_only', sum(v for s,v in mix.items() if s in endo and s not in exo), 'both', sum(v for s,v in mix.items() if s in exo and s in endo), 'neither', sum(v for s,v in mix.items() if s not in exo and s not in endo))
PY"
