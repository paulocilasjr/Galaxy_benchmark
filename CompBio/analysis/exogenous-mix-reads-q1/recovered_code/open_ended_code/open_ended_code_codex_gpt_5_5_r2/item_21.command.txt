import re, glob
from collections import Counter,defaultdict
for sam in sorted(glob.glob('refs/*.sam')):
    total=0; mapped=0; pos=Counter(); flags=Counter(); mapq=Counter()
    with open(sam) as f:
        for line in f:
            if line.startswith('@'): continue
            total+=1
            parts=line.split('\\t')
            flag=int(parts[1]); flags[flag]+=1
            if not (flag & 4):
                mapped+=1; pos[int(parts[3])]+=1; mapq[int(parts[4])]+=1
    print(sam,total,mapped,'flags',flags.most_common(5),'mapq',mapq.most_common(5),'pos range', (min(pos) if pos else None, max(pos) if pos else None), 'unique starts', len(pos))
PY"
