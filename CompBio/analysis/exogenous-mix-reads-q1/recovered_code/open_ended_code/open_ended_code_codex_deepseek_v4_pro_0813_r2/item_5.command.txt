# basic count exact matches in transcript for each sample
python3 - <<'PY'
ref=open('KLF4_NM_004235.5.fa').read().split('\\n',1)[1].replace('\\n','')
from collections import Counter
def reads(fn):
    out=[]
    with open(fn) as f:
        for h in f:
            s=f.readline().strip(); f.readline(); f.readline(); out.append(s)
    return out
for fn in ['exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q1.mix.fq']:
    seqs=reads(fn)
    exact=sum(1 for s in seqs if s in ref)
    rc_exact=sum(1 for s in seqs if s.translate(str.maketrans('ACGT','TGCA'))[::-1] in ref)
    print(fn, len(seqs), 'exact fwd',exact,'rc',rc_exact, 'either',exact+rc_exact)
PY"
