/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
def getpos(fn):
 a=[]
 with pysam.AlignmentFile(fn,'r') as f:
  for r in f:
   if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
   a.append(r.reference_start)
 return np.array(a)
regions={'5UTR':(0,548),'CDS':(548,1938),'3UTR':(1938,2854)}
for fn in ['exogenous.mix.reads.q1.exo.sam','exogenous.mix.reads.q1.endo.sam','exogenous.mix.reads.q1.mix.sam']:
 p=getpos(fn)
 print(fn)
 for name,(a,b) in regions.items():
  print(name, ((p>=a)&(p<b)).sum(),'expected lengths',b-a)
 print('total',len(p))
PY"
