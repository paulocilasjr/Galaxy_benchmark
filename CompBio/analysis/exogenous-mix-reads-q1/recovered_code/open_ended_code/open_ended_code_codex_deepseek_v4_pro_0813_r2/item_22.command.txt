/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
from scipy import stats
def getpos(fn):
 a=[]
 with pysam.AlignmentFile(fn,'r') as f:
  for r in f:
   if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
   a.append(r.reference_start)
 return np.array(a)
for k in ['endo','exo']:
 p=getpos(f'exogenous.mix.reads.q1.{k}.sam')
 print(k,'KS uniform',stats.kstest(p, 'uniform', args=(0,2854)).statistic, stats.kstest(p,'uniform',args=(0,2854)).pvalue)
 # only endo outside vs uniform outside distribution piece? not needed
PY"
