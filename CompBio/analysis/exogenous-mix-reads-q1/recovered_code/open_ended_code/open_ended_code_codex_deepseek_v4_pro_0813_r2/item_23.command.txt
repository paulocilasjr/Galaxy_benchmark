/bin/bash -lc "python3 - <<'PY'
import pysam,numpy as np
from scipy import stats
def getpos(fn):
 a=[]
 with pysam.AlignmentFile(fn,'r') as f:
  for r in f:
   if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
   a.append(r.reference_start)
 return np.array(a)
p=getpos('exogenous.mix.reads.q1.exo.sam')
print(stats.kstest(p,'uniform',args=(548,1938)))
PY"
