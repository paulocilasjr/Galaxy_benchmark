/bin/bash -lc "python3 - <<'PY'
from collections import Counter
from pathlib import Path
import numpy as np
seqs={}
for n in ['mix','endo','exo']:
    ls=Path(f'exogenous.mix.reads.q1.{n}.fq').read_text().splitlines()
    seqs[n]=ls[1::4]
mapb=np.full(256, 4, dtype=np.int8)
for i,b in enumerate(b'ACGT'): mapb[b]=i
arr={n:mapb[np.frombuffer(''.join(v).encode(),dtype=np.uint8)].reshape(len(v),50) for n,v in seqs.items()}
def mindist(a,b,bs=200):
    vals=[]
    for x in a:
        vals.append(np.count_nonzero(x[None,:]"'!=b,axis=1).min())
    return np.asarray(vals)
de=mindist(arr['"'mix'],arr['exo'])
dn=mindist(arr['mix'],arr['endo'])
print('sizes', {n:(len(v),len(set(v))) for n,v in seqs.items()})
print('de',Counter(map(int,de)))
print('dn',Counter(map(int,dn)))
print('class',Counter(np.where(de<dn,'exo',np.where(dn<de,'endo','tie'))))
print('pairs',Counter(zip(map(int,de),map(int,dn))))
print('margins',Counter(map(int,de-dn)))
# exact seq membership
exo_set=set(seqs['exo']); endo_set=set(seqs['endo'])
print('exact categories',Counter((s in exo_set,s in endo_set) for s in seqs['mix']))
PY"
