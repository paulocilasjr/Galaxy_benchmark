/bin/bash -lc "python3 - <<'PY'
from collections import Counter
from pathlib import Path
seqs={}
for n in ['mix','endo','exo']:
    ls=Path(f'exogenous.mix.reads.q1.{n}.fq').read_text().splitlines()
    seqs[n]=ls[1::4]
print({n:(len(v),len(set(v))) for n,v in seqs.items()}, flush=True)
for a,b in [('mix','exo'),('mix','endo'),('exo','endo')]:
    out=[]
    for s in seqs[a]:
        best=50
        for t in seqs[b]:
            d=sum(x"'!=y for x,y in zip(s,t))
            if d<best: best=d
        out.append(best)
    print(a,b,Counter(out),flush=True)
PY'
