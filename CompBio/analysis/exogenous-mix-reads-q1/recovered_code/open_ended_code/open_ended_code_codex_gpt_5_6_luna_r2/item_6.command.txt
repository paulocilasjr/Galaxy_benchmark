/bin/bash -lc "python3 - <<'PY'
from collections import Counter, defaultdict
from pathlib import Path

names=['mix','endo','exo']
seqs={}
for n in names:
    lines=Path(f'exogenous.mix.reads.q1.{n}.fq').read_text().splitlines()
    seqs[n]=lines[1::4]

def hd(a,b):
    return sum(x"'!=y for x,y in zip(a,b))
for n in names:
    print(n, len(seqs[n]), len(set(seqs[n])))
print('"'pairwise min Hamming distributions')
for n in ['mix','exo','endo']:
  for m in ['exo','endo']:
    if n==m: continue
    d=[]
    for s in seqs[n]:
      d.append(min(hd(s,t) for t in seqs[m]))
    print(n,'to',m,Counter(d))

print('\\nclassification of mix by min Hamming to reference reads')
counts=Counter(); details=Counter(); margins=[]
for s in seqs['mix']:
  de=min(hd(s,t) for t in seqs['exo'])
  dn=min(hd(s,t) for t in seqs['endo'])
  if de<dn: lab='exo'
  elif dn<de: lab='endo'
  else: lab='tie'
  counts[lab]+=1; details[(lab,de,dn)]+=1; margins.append(de-dn)
print(counts)
print('distance pairs:')
for k,v in sorted(details.items(), key=lambda kv:(kv[0][1],kv[0][2],kv[0][0])):
 print(k,v)
print('margin distribution',Counter(margins))

# exact membership category counts by reads
print('\\nexact membership categories')
cat=Counter()
for s in seqs['mix']:
  e=s in set(seqs['exo']); n=s in set(seqs['endo'])
  cat[(e,n)]+=1
print(cat)
# unique sequences category and read count
ucat=Counter()
for s,c in Counter(seqs['mix']).items(): ucat[(s in set(seqs['exo']),s in set(seqs['endo']))]+=1
print('unique',ucat)
PY"
