/bin/bash -lc "python3 - <<'PY'
import subprocess,numpy as np
cmd=\"curl -L --fail --silent 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.5&rettype=fasta&retmode=text' | minimap2 -a -x sr - {fq} 2>/dev/null\"
for criterion in ['mapped','exact']:
 vals={}
 for n in ['exo','endo','mix']:
  out=subprocess.check_output(cmd.format(fq=f'exogenous.mix.reads.q1.{n}.fq'),shell=True,text=True)
  arr=[]
  for l in out.splitlines():
   if l.startswith('@'): continue
   x=l.split('\\t'); flag=int(x[1]);
   if flag&4: continue
   if criterion=='exact' and not any(t=='NM:i:0' for t in x[11:]): continue
   arr.append(int(x[3]))
  vals[n]=np.array(arr)
 eout=sum(~((vals['endo']>=549)&(vals['endo']<=1988))); etot=len(vals['endo'])
 mout=sum(~((vals['mix']>=549)&(vals['mix']<=1988))); mtot=len(vals['mix'])
 print(criterion,'counts', {n:(len(a),sum((a>=549)&(a<=1988)),sum(~((a>=549)&(a<=1988)))) for n,a in vals.items()},'f',1-(mout/mtot)/(eout/etot))
PY"
