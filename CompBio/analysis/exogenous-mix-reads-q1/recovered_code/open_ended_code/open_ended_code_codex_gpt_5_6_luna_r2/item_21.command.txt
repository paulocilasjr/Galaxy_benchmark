import subprocess, numpy as np
from collections import Counter
cmd=\"curl -L --fail --silent 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NM_004235.5&rettype=fasta&retmode=text' | minimap2 -a -x sr - {fq} 2>/dev/null\"

  if not flag&4: p.append(int(x[3]))
 ps[n]=np.array(p)
for a,b in [(549,1988),(549,1939),(556,1938),(550,1988),(501,2000)]:
 print('\\nrange',a,b)

 pe=sum((ps['exo']>=a)&(ps['exo']<=b))/len(ps['exo'])
 pd=sum((ps['endo']>=a)&(ps['endo']<=b))/len(ps['endo'])
 pm=sum((ps['mix']>=a)&(ps['mix']<=b))/len(ps['mix'])
 print('f in-bin formula', (pm-pd)/(pe-pd), 'f outside', 1-(1-pm)/(1-pd))
