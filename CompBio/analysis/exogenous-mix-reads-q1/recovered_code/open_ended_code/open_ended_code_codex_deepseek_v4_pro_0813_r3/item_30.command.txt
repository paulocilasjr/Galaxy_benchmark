/bin/bash -lc "python3 - <<'PY'
import subprocess, math

def positions(bam):
 out=subprocess.check_output(f'samtools view {bam}',shell=True,text=True).splitlines()
 return [int(l.split('\\t')[3])-1 for l in out]
en=positions('exogenous.mix.reads.q1.endo.mrna.bam')
mx=positions('exogenous.mix.reads.q1.mix.mrna.bam')
L=2903; lo,hi=555,1937; Le=hi-lo+1
# profile likelihood p, q optimized for each p (grid .001)
def negll(p):
 best=None
 qs=[i/1000 for i in range(1001)]
 for q in qs:
  ll=0; ok=True
  for pos in en:
   pr=(q/Le if lo<=pos<=hi else 0)+(1-q)/L
   if pr<=0: ok=False; break
   ll += math.log(pr)
  if not ok: continue
  for pos in mx:
   pr=(p/Le if lo<=pos<=hi else 0)+(1-p)/L
   if pr<=0: ok=False; break
   ll += math.log(pr)
  if not ok: continue
  if best is None or ll>best[0]: best=(ll,q)
 return best
maxll=None
for pi in range(0,1001):
 p=pi/1000
 ll,q=negll(p)
 if maxll is None or ll>maxll[0]: maxll=(ll,p,q)
print('MLE',maxll)
# print profile around p
for pi in range(0,701,50):
 p=pi/1000
 ll,q=negll(p)
 print('p',p,'q',q,'2*diff',2*(maxll[0]-ll))
PY"
