/bin/bash -lc "python3 - <<'PY'
from collections import Counter
import math, numpy as np

def reads(fn):
 lines=open(fn).read().splitlines(); return [lines[i+1] for i in range(0,len(lines),4)]
exo=reads('exogenous.mix.reads.q1.exo.fq'); endo=reads('exogenous.mix.reads.q1.endo.fq'); mix=reads('exogenous.mix.reads.q1.mix.fq')
def kcounts(seqs,k):
 c=Counter()
 for s in seqs:
  for i in range(len(s)-k+1): c[s[i:i+k]]+=1
 return c
for k in range(1,8):
 ce=kcounts(exo,k); cn=kcounts(endo,k); cm=kcounts(mix,k)
 keys=sorted(set(ce)|set(cn))
 fe=np.array([ce.get(x,0) for x in keys],float); fn=np.array([cn.get(x,0) for x in keys],float); fm=np.array([cm.get(x,0) for x in keys],float)
 fe/=fe.sum(); fn/=fn.sum(); fm/=fm.sum()
 # grid p 0..1 minimize L2
 best=(None,None)
 for j in range(1001):
  p=j/1000
  d=np.sum((p*fe+(1-p)*fn-fm)**2)
  if best[0] is None or d<best[1]: best=(p,d)
 # KL? p grid with KL(fm || mixture)
 bestkl=(None,None)
 for j in range(1001):
  p=j/1000
  v=p*fe+(1-p)*fn
  kl=sum(fm[i]*math.log((fm[i]+1e-12)/(v[i]+1e-12)) for i in range(len(fm)))
  if bestkl[0] is None or kl<bestkl[1]: bestkl=(p,kl)
 print('k',k,'L2 p',best[0],'KL p',bestkl[0],'GC fit', ( (sum(x.count(\"G\")+x.count(\"C\") for x in mix)/50/len(mix)) - (sum(x.count(\"G\")+x.count(\"C\") for x in endo)/50/len(endo)) ) / ((sum(x.count(\"G\")+x.count(\"C\") for x in exo)/50/len(exo)) - (sum(x.count(\"G\")+x.count(\"C\") for x in endo)/50/len(endo))))
PY"
