/bin/bash -lc "python3 - <<'PY'
import subprocess
from collections import Counter
s=open('klf4_mrna.fa').read().split('\\n',1)[1].replace('\\n','')
for name in ['exo','endo','mix']:
 out=subprocess.check_output(f'samtools view exogenous.mix.reads.q1.{name}.mrna.bam',shell=True,text=True).splitlines()
 gc_obs=[]
 gc_ref=[]
 for l in out:
  parts=l.split('\\t'); pos=int(parts[3])-1; read=parts[9]
  gc_obs.append((read.count('G')+read.count('C'))/len(read))
  # reference at aligned pos if full length, but if reverse, read is revcomp of ref; GC same
  ref=s[pos:pos+len(read)]
  gc_ref.append((ref.count('G')+ref.count('C'))/len(ref))
 print(name,'mean observed GC',sum(gc_obs)/len(gc_obs),'mean ref GC at positions',sum(gc_ref)/len(gc_ref),'n',len(gc_obs))
PY"
