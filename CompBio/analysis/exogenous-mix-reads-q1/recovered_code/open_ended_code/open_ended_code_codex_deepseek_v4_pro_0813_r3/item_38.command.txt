/bin/bash -lc "python3 - <<'PY'
import urllib.request
for uid,name in [('1675154260','klf4_v1.fa'),('1519314269','klf4_v2.fa')]:

 data=urllib.request.urlopen(url,timeout=20).read().decode()
 open(name,'w').write(data)
 seq=''.join(data.split('\\n',1)[1].split('\\n')) if '\\n' in data else ''
 print(name,'data len',len(data),'seq len',len(seq),'header',data.split('\\n')[0])
 print('seq first',seq[:100])
PY"
