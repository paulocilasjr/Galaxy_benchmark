/bin/bash -lc "python3 - <<'PY'
# find reads not mapping to mRNA from each file using bam read names
import subprocess

def reads(fn):
 lines=open(fn).read().splitlines(); return [lines[i] for i in range(0,len(lines),4)],[lines[i+1] for i in range(0,len(lines),4)]
for fn,b in [('exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.exo.mrna.bam'),('exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q1.endo.mrna.bam'),('exogenous.mix.reads.q1.mix.fq','exogenous.mix.reads.q1.mix.mrna.bam')]:
 names,seqs=reads(fn); mapped=set(subprocess.check_output(f'samtools view {b} | cut -f1',shell=True,text=True).split())
 miss=[(n,s) for n,s in zip(names,seqs) if n not in mapped]
 print(fn,'missing',len(miss))
 for n,s in miss[:20]: print(n,s)
PY"
