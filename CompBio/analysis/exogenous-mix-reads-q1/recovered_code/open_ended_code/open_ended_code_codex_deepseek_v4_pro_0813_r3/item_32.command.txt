/bin/bash -lc "python3 - <<'PY'
import subprocess

def reads(fn):
 lines=open(fn).read().splitlines(); return [(lines[i],lines[i+1]) for i in range(0,len(lines),4)]
for fn,bam in [('exogenous.mix.reads.q1.endo.fq','exogenous.mix.reads.q1.endo.mrna.bam'),('exogenous.mix.reads.q1.mix.fq','exogenous.mix.reads.q1.mix.mrna.bam'),('exogenous.mix.reads.q1.exo.fq','exogenous.mix.reads.q1.exo.mrna.bam')]:
 mapped=set(subprocess.check_output(f'samtools view {bam} | cut -f1',shell=True,text=True).split())
 miss=[(n,s) for n,s in reads(fn) if n not in mapped]
 print(fn,'total',len(reads(fn)),'unmapped',len(miss))
 for n,s in miss[:20]: print(n,s)
 print()
PY"
