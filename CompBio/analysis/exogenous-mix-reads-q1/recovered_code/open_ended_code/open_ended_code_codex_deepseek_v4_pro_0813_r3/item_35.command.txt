/bin/bash -lc "python3 - <<'PY'
import subprocess
for name in ['exo','endo','mix']:
 def names(b): return set(subprocess.check_output(f'samtools view {b} | cut -f1',shell=True,text=True).split())
 m=names(f'exogenous.mix.reads.q1.{name}.mrna.bam'); g=names(f'exogenous.mix.reads.q1.{name}.genomic.bam')
 print(name,'mRNA',len(m),'genomic',len(g),'both',len(m&g),'only_m',len(m-g),'only_g',len(g-m))
PY"
