/bin/bash -lc "python3 - <<'PY'
import subprocess
from collections import Counter
for name in ['exo','endo','mix']:
 out=subprocess.check_output(f'samtools view exogenous.mix.reads.q1.{name}.mrna.bam',shell=True,text=True).splitlines()
 c=Counter(int(l.split('\\t')[3])-1 for l in out)
 print(name)
 # print coverage counts in 10bp bins from 500 to 2000
 for a in range(500,2000,50):
  cnt=sum(v for p,v in c.items() if a<=p<a+50)
  print(f'{a}-{a+49}: {cnt}',end='  ')
 print('\\n')
PY"
