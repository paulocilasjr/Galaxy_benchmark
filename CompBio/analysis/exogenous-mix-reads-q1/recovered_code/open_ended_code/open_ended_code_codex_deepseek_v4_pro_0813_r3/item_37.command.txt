/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, re
# esearch nuccore KLF4 human mRNA
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?db=nuccore&term=KLF4%5BGene%5D+AND+Homo+sapiens%5BOrganism%5D+AND+biomol_mrna%5BProperties%5D&retmax=50&retmode=json'
data=json.load(urllib.request.urlopen(url,timeout=20))
ids=data['esearchresult']['idlist']
print('count',data['esearchresult']['count'],'ids',ids)
# efetch summaries
if ids:
 url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=nuccore&id='+','.join(ids[:30])+'&retmode=json'
 summ=json.load(urllib.request.urlopen(url,timeout=20))
 for uid in ids[:30]:
  r=summ['result'].get(uid)
  if r: print(uid, r.get('title'), r.get('slen'))
PY"
