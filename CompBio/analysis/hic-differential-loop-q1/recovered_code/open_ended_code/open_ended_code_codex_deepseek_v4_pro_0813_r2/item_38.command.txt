/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
import re, html
s=Path('pmc.xml').read_text()
# strip tags rough
for m in re.finditer(r'<fig["'^>]*>(.*?)</fig>'"',s,re.S):
 txt=re.sub(r'<["'^>]+>'"',' ',m.group(1))
 txt=html.unescape(re.sub(r'\\s+',' ',txt))
 print(txt[:1500])
 print('---')
PY"
