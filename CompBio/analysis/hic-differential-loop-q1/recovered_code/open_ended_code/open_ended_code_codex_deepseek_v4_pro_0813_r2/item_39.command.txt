from pathlib import Path
import re, html
s=Path('pmc.xml').read_text()

for i,m in enumerate(figs,1):
 txt=re.sub(r'<["'^>]+>'"',' ',m)
 txt=html.unescape(re.sub(r'\\s+',' ',txt))
 if i>=5:
  print('FIGURE',i, txt[:3000]); print('---')
PY"
