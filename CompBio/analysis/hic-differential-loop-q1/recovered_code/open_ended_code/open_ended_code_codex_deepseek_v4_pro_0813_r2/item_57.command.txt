import gzip

def load(fn):
 d=[]
 for raw in gzip.open(fn,'rt'):
  if raw.startswith('chrom1'): continue
  p=raw.rstrip('\\n').split('\\t')
  if len(p)<11: continue
  c1,s1,e1,c2,s2,e2=p[:6]
  if c1"'!='"'chr12' or c2"'!='"'chr12': continue
  s1,e1,s2,e2=map(int,[s1,e1,s2,e2])
  if s1>s2 or (s1==s2 and e1>e2): s1,e1,s2,e2=s2,e2,s1,e1
  # fields 8 c_size, 9 count, 12 donut.value, 16 donut.qval
  try:
   count=int(p[9]); q=float(p[16]); donut=float(p[12])
  except: count=0; q=1; donut=0
  d.append((s1,e1,s2,e2,count,q,donut,p))
 return d
H1=load('H1_dots.bedpe.gz'); HFF=load('HFF_dots.bedpe.gz')
print(len(H1),len(HFF))
for name,data in [('H1',H1),('HFF',HFF)]:
 print('\\n###',name)
 for x in data:
  s1,e1,s2,e2,count,q,donut,p=x
  # both anchors? region around NANOG 6.5-8.5
  if (6500000 <= s1 <= 8500000) or (6500000 <= s2 <= 8500000):
   print(f'{s1}\\t{e1}\\t{s2}\\t{e2}\\tcount={count}\\tdonut={donut:.1f}\\tq={q:.2g}')
