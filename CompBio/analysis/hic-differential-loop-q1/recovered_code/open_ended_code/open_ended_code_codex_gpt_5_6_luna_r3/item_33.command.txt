/bin/bash -lc "python -u - <<'PY'
import fsspec,h5py,cooler,numpy as np
url='https://4dn-open-data-public.s3.amazonaws.com/fourfront-webprod/wfoutput/27bbfb40-0224-427c-be41-398b43d70c48/4DNFI9FVHJZQ.mcool'
with fsspec.open(url,'rb',block_size=8*1024*1024,cache_type='readahead') as f:
 with h5py.File(f,'r') as h:
  c=cooler.Cooler(h['resolutions/1000'])
  raw=c.matrix(balance=False,sparse=False).fetch('chr12:7620000-7820000').astype(np.int32)
  np.save('HFF_1kb_raw.npy',raw)
  print(raw.shape,raw.sum(),flush=True)
PY
ls -lh HFF_1kb_raw.npy"
