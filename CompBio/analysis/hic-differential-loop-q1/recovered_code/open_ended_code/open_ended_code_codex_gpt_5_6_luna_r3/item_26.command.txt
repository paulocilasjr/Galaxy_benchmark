/bin/bash -lc "python - <<'PY'
import fsspec,h5py,cooler,numpy as np,time
url='https://4dn-open-data-public.s3.amazonaws.com/fourfront-webprod/wfoutput/27bbfb40-0224-427c-be41-398b43d70c48/4DNFI9FVHJZQ.mcool'
with fsspec.open(url,'rb',block_size=8*1024*1024,cache_type='readahead') as f:
 with h5py.File(f,'r') as h:
  c=cooler.Cooler(h['resolutions/10000'])
  region='chr12:7620000-7820000'
  bins=c.bins().fetch(region)[:]
  raw=c.matrix(balance=False,sparse=False).fetch(region).astype(float)
  bal=c.matrix(balance=True,sparse=False).fetch(region)
  print('info',c.info)
  print('bins',bins[['start','end']].to_records(index=False).tolist())
  np.set_printoptions(precision=3,suppress=True,linewidth=240)
  print('raw');print(raw.astype(int))
  print('bal');print(bal)
  for matlabel,mat in [('raw',raw),('bal',bal)]:
   exp=np.array([np.nanmean(np.diag(mat,k)) for k in range(mat.shape[0])])
   d=np.abs(np.arange(mat.shape[0])[:,None]-np.arange(mat.shape[0])[None,:])
   oe=mat/exp[d]
   print(matlabel,'OE');print(oe)
PY"
