import fsspec,h5py,cooler,numpy as np,time
url='https://4dn-open-data-public.s3.amazonaws.com/fourfront-webprod/wfoutput/d13aa1ea-053c-4113-94fe-a1e7ab1dbbab/4DNFI9GMP2J8.mcool'

  c=cooler.Cooler(h['resolutions/10000'])
  print(c.info, time.time()-t)
  print(c.chromsizes['chr12'])
  a=c.matrix(balance=False,sparse=False).fetch('chr12:7629950-7809597')
  print(a.shape,a[:3,:3],np.nanmax(a),a.sum())
PY"
