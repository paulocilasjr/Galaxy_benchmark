/bin/bash -lc "python - <<'PY'
import numpy as np
h1=np.load('H1_1kb_raw.npy').astype(float);h2=np.load('HFF_1kb_raw.npy').astype(float)
w1=np.load('H1_1kb_weights.npy');w2=np.load('HFF_1kb_weights.npy')
# fix nan to 0 for irrelevant first bin
b1=h1*w1[:,None]*w1[None,:]; b2=h2*w2[:,None]*w2[None,:]
for bs in [1,2,5,10]:
 n=200//bs; coords=7620000+np.arange(n)*1000*bs
 # sum raw, balanced sums (not same as aggregate balance but okay)
 a1=h1[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3));a2=h2[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3))
 q1=b1[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3));q2=b2[:n*bs,:n*bs].reshape(n,bs,n,bs).sum((1,3))
 print('\\nBS',bs)
 for label,x,y,pseudo in [('raw',a1,a2,.5),('bal',q1,q2,1e-6)]:
  # total-scale normalize first
  x=x/x.sum();y=y/y.sum()
  vals=[]
  for i in range(n):
   for j in range(i+4,n):
    # local 5x5 neighborhood excluding center, same-ish diagonal positions maybe square ring
    i0=max(0,i-2);i1=min(n,i+3);j0=max(0,j-2);j1=min(n,j+3)
    z=np.r_[x[i0:i1,j0:j1].ravel(), y[i0:i1,j0:j1].ravel()]
    # score center against mean of surrounding 5x5 excluding exact center for each map
    nbx=x[i0:i1,j0:j1].copy();nby=y[i0:i1,j0:j1].copy();nbx[i-i0,j-j0]=np.nan;nby[i-i0,j-j0]=np.nan
    sx=np.log2((x[i,j]+pseudo)/(np.nanmedian(nbx)+pseudo)); sy=np.log2((y[i,j]+pseudo)/(np.nanmedian(nby)+pseudo))
    vals.append((sx-sy,sx,sy,i,j,a1[i,j],a2[i,j],coords[i],coords[j]))
  print(label,'top diff')
  for z in sorted(vals,reverse=True)[:20]:print('%+.2f (%.2f,%.2f) %d/%d %d;%d'%z)
PY"
