/bin/bash -lc "python - <<'PY'
import csv,json
lo,hi=7629950,7809597
loops={}
for sample,accession in [('H1','4DNES21D8SP8'),('HFF','4DNESWST3UBH')]:
 meta=json.load(open(accession+'.json'))
 for group in meta['other_processed_files']:
  for f in group['files']:
   if f.get('file_type')=='dot calls':print(sample,f['accession'],'assembly:',f.get('genome_assembly'))
 rows=list(csv.DictReader(open(sample+'.bedpe'),delimiter='\\t'))
 loops[sample]=[r for r in rows if r['chrom1']==r['chrom2']=='chr12' and lo<=int(r['start1'])<=hi and lo<=int(r['start2'])<=hi]
 print(sample,'regional loops:',[(r['start1'],r['end1'],r['start2'],r['end2']) for r in loops[sample]])
for r in loops['H1']:
 if not any(abs(int(r['start1'])-int(s['start1']))<=20000 and abs(int(r['start2'])-int(s['start2']))<=20000 for s in loops['HFF']):
  anchors=[(int(r['start1'])+int(r['end1']))/2,(int(r['start2'])+int(r['end2']))/2]
  result=';'.join(str(round(v/20000)*20000) for v in anchors)
  print('H1-specific loop rounded:',result)
  open('answer.txt','w').write(result+'\\n')
PY"
