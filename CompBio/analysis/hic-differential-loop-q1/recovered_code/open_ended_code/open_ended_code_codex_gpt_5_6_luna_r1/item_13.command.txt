/bin/bash -lc "curl -L --fail --silent --show-error 'https://data.4dnucleome.org/experiment-set-replicates/4DNESWST3UBH/?format=json' -o hff.json && curl -L --fail --silent --show-error 'https://data.4dnucleome.org/experiment-set-replicates/4DNES21D8SP8/?format=json' -o h1.json && python3 - <<'PY'
import json
for fn in ['hff.json','h1.json']:
 d=json.load(open(fn))
 print(fn, d.keys())
 # recursively print keys containing file etc
 def walk(x,path=''):
  if isinstance(x,dict):
   for k,v in x.items():
    if any(s in k.lower() for s in ('file','matrix','pair','url','download')):
     print(path+'/'+k, type(v).__name__, (str(v)[:300] if not isinstance(v,(dict,list)) else ('len='+str(len(v)))) )
    walk(v,path+'/'+k)
  elif isinstance(x,list):
   for i,v in enumerate(x[:100]): walk(v,path+f'/{i}')
 walk(d)
PY"
