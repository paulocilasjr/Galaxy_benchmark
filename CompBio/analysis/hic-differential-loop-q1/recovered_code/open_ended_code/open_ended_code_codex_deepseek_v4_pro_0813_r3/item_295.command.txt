/bin/bash -lc "python3 - <<'PY'
import fitz
doc=fitz.open('supp2_real.pdf')
for pno in [7]: # 0-index page8
 page=doc[pno]
 for dpi in [300,600]:
  pix=page.get_pixmap(dpi=dpi)
  f=f'supp2_p08_dpi{dpi}.png'
  pix.save(f)
  print(f,pix.width,pix.height)
PY
python3 - <<'PY'
from rapidocr_onnxruntime import RapidOCR
ocr=RapidOCR()
for f in ['supp2_p08_dpi300.png','supp2_p08_dpi600.png']:
 print('\\n====',f)
 result,_=ocr(f)
 if result:
  for box,text,conf in result:
   print(text, round(float(conf),2))
PY"
