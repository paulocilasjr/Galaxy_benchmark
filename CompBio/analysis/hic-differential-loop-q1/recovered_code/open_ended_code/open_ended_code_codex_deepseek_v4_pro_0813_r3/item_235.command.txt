import numpy as np
from scipy.ndimage import maximum_filter

def load(f):
 d=np.load(f); return int(d['start']),int(d['res']),d['H1'],d['HFF']

def local_max(mat, mask, size=3):
 # local maximum within square excluding self? use maximum_filter
 from scipy.ndimage import maximum_filter
 lm=maximum_filter(np.where(mask,mat,np.nan),size=size,mode='constant',cval=np.nan)
 return mask & (mat==lm) & (mat>0)
