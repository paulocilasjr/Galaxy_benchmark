/bin/bash -lc "python3 - <<'PY'
from PIL import Image,ImageOps
img=Image.open('supp2_p08_dpi600.png').convert('L')
# Crop bottom axis strip right half
W,H=img.size
im=img.crop((int(W*0.45),int(H*0.94),W,int(H*0.99)))
im=im.resize((im.width//2,im.height//2))
im=ImageOps.autocontrast(im)
# binarize inverse: text black -> ascii #
th=120
# downsample further
im=im.resize((im.width//2,im.height//2))
chars=' .:-=+*#%@'
pix=im.load()
for y in range(im.height):
 line=''
 for x in range(im.width):
  v=pix[x,y]
  line+=chars[min(9,max(0,(255-v)*9//255))]
 print(line)
PY"
