import urllib.request, re
url='https://data.4dnucleome.org/publications/a716e6b4-9cfa-4f8d-a2c7-cabf21d42b95/'
req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
data=urllib.request.urlopen(req,timeout=60).read().decode('utf-8','ignore')
print(len(data))
for m in re.finditer(r'href=\"(["'^"]+)"[''^>]*>([''^<]+)</a>'"',data):
    href=m.group(1); text=m.group(2).strip()
    if 'experiment-set' in href or 'experiment' in href or '4DNES' in text or '4DNES' in href:
        print(text, href)
print('experiment sets IDs', re.findall(r'4DNES[A-Z0-9]{8}', data)[:100])
PY"
