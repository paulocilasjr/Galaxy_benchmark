/bin/bash -lc "python3 - <<'PY'
import requests, hashlib, re
s=requests.Session(); s.headers['User-Agent']='Mozilla/5.0'
u='https://pmc.ncbi.nlm.nih.gov/articles/PMC7222625/pdf/nihms-1574796.pdf'
r=s.get(u,timeout=30); challenge=re.search(r'const POW_CHALLENGE = \"(["'^"]+)"'"',r.text).group(1)
diff=int(re.search(r'const POW_DIFFICULTY = \"(["'^"]+)"'"',r.text).group(1)); name=re.search(r'const POW_COOKIE_NAME = \"(["'^"]+)"'"',r.text).group(1)
nonce=0; target='0'*diff
while not hashlib.sha256((challenge+str(nonce)).encode()).hexdigest().startswith(target): nonce+=1
print('nonce',nonce)
s.cookies.set(name,challenge+','+str(nonce),path='/')
r=s.get(u,timeout=60)
print(r.status_code,r.headers.get('content-type'),len(r.content),r.content[:5])
open('article_real.pdf','wb').write(r.content)
PY"
