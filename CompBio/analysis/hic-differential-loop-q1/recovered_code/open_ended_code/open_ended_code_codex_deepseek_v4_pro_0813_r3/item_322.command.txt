/bin/bash -lc "python3 - <<'PY'
import requests
for u in ['https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.full.pdf','https://www.biorxiv.org/content/10.64898/2026.04.06.716850v1.full.pdf','https://www.biorxiv.org/content/biorxiv/early/2026/05/02/2026.04.06.716850.full.pdf']:
 r=requests.get(u,timeout=60)
 print(u,r.status_code,r.headers.get('content-type'),len(r.content),r.content[:4])
 if r.content[:4]==b'%PDF': open('compbio_paper.pdf','wb').write(r.content); break
PY"
