/bin/bash -lc "python3 - <<'PY'
import requests,json
for uuid,name in [('1e8876a1-6b9b-4119-8ea2-3bd9fe9c3e0f','H1_supp'),('64316922-a767-4564-813d-f8486997a68e','HFF_supp')]:
 u=f'https://data.4dnucleome.org/higlass-view-configs/{uuid}/?format=json'
 r=requests.get(u,timeout=30)
 print(uuid,r.status_code,len(r.text))
 open(f'{name}.json','w').write(r.text)
 j=r.json()
 vc=j.get('viewconfig',{})
 print('views',len(vc.get('views',[])))
 for v in vc.get('views',[]):
  print(v.get('uid'), v.get('initialXDomain'), v.get('initialYDomain'))
  # track types and tileset uids
  for pos,tracks in v.get('tracks',{}).items():
   for t in tracks:
    if isinstance(t,dict):
     print(' ',pos,t.get('type'),t.get('uid'),t.get('tilesetUid'),t.get('options',{}).get('name'))
     if 'contents' in t:
      for c in t['contents']:
       print('   content',c.get('tilesetUid'),c.get('type'))
PY"
