import urllib.request,json
url='https://data.4dnucleome.org/search/?format=json&q=NANOG&limit=50'

 lab=item.get('lab',{}).get('display_title') if isinstance(item.get('lab'),dict) else ''
 desc=item.get('description') or ''
 print(item.get('accession'), item.get('display_title'), item.get('file_type'), lab, desc[:200])
PY"
