/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://huggingface.co/api/datasets/Genentech/compbiobench-data-v1/tree/main?recursive=true'

for x in data: print(x.get('path'), x.get('size'), x.get('type'))
PY"
