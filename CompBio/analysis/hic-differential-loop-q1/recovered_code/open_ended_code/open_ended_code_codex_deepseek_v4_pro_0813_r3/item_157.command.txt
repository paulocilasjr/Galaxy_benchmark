/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
# Gradio API call info to see named endpoints
url='https://genentech-compbiobench-leaderboard-v1.hf.space/gradio_api/info'
d=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=30).read())
print(json.dumps(d,indent=2)[:5000])
# call refresh
for endpoint in ['/refresh','/submit_predictions']:
 if endpoint in d.get('named_endpoints',{}):
  callurl='https://genentech-compbiobench-leaderboard-v1.hf.space/gradio_api/call'+endpoint
  print('\\ncall',callurl)
  try:
   req=urllib.request.Request(callurl,data=b'{}',headers={'User-Agent':'Mozilla/5.0','Content-Type':'application/json'})
   resp=urllib.request.urlopen(req,timeout=60).read().decode()
   print(resp[:1000])
  except Exception as e: print('ERR',repr(e))
PY"
