/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
base='https://data.4dnucleome.org/search/?type=File&format=json'
# search for lab/dekker or publication? Let's try file_type=differential loops
for q in ['file_type=differential+loops','file_type=dot+calls&lab.title=Job+Dekker%2C+UMMS','file_type=dot+calls&publications.display_title=Ultrastructural+Details+of+Mammalian+Chromosome+Architecture']:
 url=base+'&'+q
 print('URL',url)
 try:
  d=json.loads(urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=120).read())
  print('graph',len(d.get('@graph',[])))
  for item in d.get('@graph',[]):
   print(item.get('accession'), item.get('display_title'), item.get('file_type'), item.get('open_data_url'))
 except Exception as e: print('ERR',e)
PY"
