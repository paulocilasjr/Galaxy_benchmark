from PIL import Image, ImageOps, ImageFilter
img=Image.open('supp2_p08_dpi600.png').convert('L')
W,H=img.size
# bottom right panel roughly right half, lower 20%? Let's crop several strips
boxes={
 'br_axis':(int(W*0.45), int(H*0.90), W, int(H*0.985)),
 'br_axis2':(int(W*0.45), int(H*0.92), W, H),
 'br_lower':(int(W*0.45), int(H*0.75), W, H),
 'br_all':(int(W*0.45), int(H*0.5), W, H),
}
for n,b in boxes.items():
 im=img.crop(b)
 # upscale 2x, autocontrast, binarize
 im=im.resize((im.width*2,im.height*2),Image.LANCZOS)
 im=ImageOps.autocontrast(im)
 im=im.point(lambda x:255 if x>140 else 0)
 im.save(f'axis_{n}.png')
 print(n,b,im.size)
PY
