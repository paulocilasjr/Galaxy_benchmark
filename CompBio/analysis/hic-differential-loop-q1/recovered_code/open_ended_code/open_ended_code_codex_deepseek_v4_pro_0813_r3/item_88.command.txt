/bin/bash -lc "python3 - <<'PY'
import urllib.request,re,json
url='https://data.4dnucleome.org/publications/a716e6b4-9cfa-4f8d-a2c7-cabf21d42b95/'
data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})).read().decode('utf-8','ignore')
uuids=re.findall(r'/higlass-view-configs/([0-9a-f-]+)/',data)
print(len(uuids), sorted(set(uuids)))
for u in sorted(set(uuids)):
    api='https://data.4dnucleome.org/higlass-view-configs/'+u+'/?format=json'
    try:
        d=json.loads(urllib.request.urlopen(urllib.request.Request(api,headers={'User-Agent':'Mozilla/5.0'}),timeout=60).read())
        print('\\n',u, d.get('display_title'), d.get('name'))
        # print tracks with tilesetUid and file accession?
        cfg=d.get('viewconfig',{})
        for view in cfg.get('views',[]):
            for t in view.get('tracks',{}).get('top',[]):
                print(' track',t.get('type'), t.get('tilesetUid'), t.get('fileUrl'), t.get('name'))
    except Exception as e: print('ERR',u,e)
PY"
