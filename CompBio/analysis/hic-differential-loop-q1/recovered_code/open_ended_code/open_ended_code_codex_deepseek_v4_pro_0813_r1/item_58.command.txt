/bin/bash -lc "python - <<'PY'
import gzip,csv,math
loops={}
for name,fn in [('H1','h1_loops.bedpe.gz'),('HFF','hff_loops.bedpe.gz')]:
 rows=[]
 with gzip.open(fn,'rt') as f:
  for d in csv.DictReader(f,delimiter='\\t'):
   if d['chrom1']=='chr12' and d['chrom2']=='chr12':
    s1=int(d['start1']);e1=int(d['end1']);s2=int(d['start2']);e2=int(d['end2'])
    # loop intersects broad region 7.0-8.2M
    if not (e1 < 7000000 or s1 > 8200000) and not (e2 < 7000000 or s2 > 8200000):
     rows.append((s1,e1,s2,e2,float(d['la_exp.donut.qval']),int(d['count'])))
 loops[name]=rows
# sort and print
sub=(7629950,7809597)
for name,rows in loops.items():
 print('\\n',name)
 for r in sorted(rows):
  s1,e1,s2,e2,q,c=r
  # overlap length with sub interval
  ov=lambda a,b: max(0,min(b,sub[1])-max(a,sub[0]))
  o1=ov(s1,e1);o2=ov(s2,e2)
  inside = (s1>=sub[0] and e1<=sub[1] and s2>=sub[0] and e2<=sub[1])
  partial = o1>0 and o2>0
  print(f'{s1}-{e1} / {s2}-{e2} q={q:.2e} cnt={c} inside={inside} partial={partial} overlap=({o1},{o2})')
PY"
