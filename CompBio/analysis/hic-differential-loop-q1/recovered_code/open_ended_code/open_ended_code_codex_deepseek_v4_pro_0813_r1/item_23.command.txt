/bin/bash -lc "python - <<'PY'
import requests
url='https://www.ncbi.nlm.nih.gov/research/bionlp/RESTful/pmcoa.cgi/BioC_json/PMC7222625/unicode'
r=requests.get(url,timeout=30); data=r.json()
text=[]
for doc in data[0]['documents']:
 for p in doc['passages']:
  text.append(p['text'])
full='\\n'.join(text)
open('pmc_bioc_text.txt','w').write(full)
for i,line in enumerate(text):
 if 'NANOG' in line.upper() or 'chr12' in line.lower() or 'differential' in line.lower():
  print('---',i, line[:500])
PY"
