/bin/bash -lc "python - <<'PY'
import fsspec,h5py
url='https://4dn-open-data-public.s3.amazonaws.com/fourfront-webprod/wfoutput/d13aa1ea-053c-4113-94fe-a1e7ab1dbbab/4DNFI9GMP2J8.mcool'
fs=fsspec.filesystem('http')
with fs.open(url,block_size=1<<20) as f:
 h=h5py.File(f,'r')
 g=h['resolutions']['5000']
 print('group keys',list(g.keys()))
 for k,v in g.items():
  print(k, type(v), getattr(v,'shape',None), getattr(v,'dtype',None), dict(v.attrs) if hasattr(v,'attrs') else '')
 print('bins keys', list(g['bins'].keys()))
 print('bins chrom',g['bins']['chrom'][:10],g['bins']['chrom'].dtype)
 print('bins start',g['bins']['start'][:10])
 print('bins end',g['bins']['end'][:10])
 print('pixels keys', list(g['pixels'].keys()))
 for k,v in g['pixels'].items(): print('pix',k,v.shape,v.dtype,dict(v.attrs))
 print('indexes keys',list(g['indexes'].keys()))
 for k,v in g['indexes'].items(): print('idx',k,v.shape,v.dtype,dict(v.attrs))
 print('chroms',g['chroms']['name'][:],g['chroms']['length'][:])
PY"
