/bin/bash -lc "python - <<'PY'
import requests
urls={
'pmc_manuscript.pdf':'https://pmc.ncbi.nlm.nih.gov/articles/instance/7222625/pdf/nihms-1574796.pdf',
'pmc_supp2.pdf':'https://pmc.ncbi.nlm.nih.gov/articles/instance/7222625/bin/NIHMS1574796-supplement-2.pdf',
}
for fn,u in urls.items():
 r=requests.get(u,headers={'User-Agent':'Mozilla/5.0'},timeout=120)
 print(fn,r.status_code,len(r.content),r.headers.get('content-type'))
 open(fn,'wb').write(r.content)
PY"
