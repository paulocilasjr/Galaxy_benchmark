/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
p='single_cell_dynamics_question.csv'
d=pd.read_csv(p)
e=d[['enh_x','enh_y','enh_z']].to_numpy(); q=d[['prom_x','prom_y','prom_z']].to_numpy()
d['distance']=np.sqrt(np.sum((e-q)**2,axis=1))
d['contact']=(d.distance<=260).astype(int)
print('shape',d.shape)
print('dist quantiles',d.distance.quantile([0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]).to_dict())
print('contact proportion',d.contact.mean(),d.contact.sum())
print('tx proportion',d.transcription.mean())
print('contingency\\n',pd.crosstab(d.contact,d.transcription,normalize='index'))
print('counts\\n',pd.crosstab(d.contact,d.transcription))
print('mean distance tx',d.groupby('transcription').distance.agg(['count','mean','median','std']))
print('mean distance contact',d.groupby('contact').distance.agg(['count','mean','median','std']))
# association stats
print('chi2',stats.chi2_contingency(pd.crosstab(d.contact,d.transcription)))
print('pearson contact tx',np.corrcoef(d.contact,d.transcription)[0,1])
# per cell rates
x=d.groupby('cell_id').agg(contact=('contact','mean'), tx=('transcription','mean'), n=('contact','size'))
print('per-cell corr',x.contact.corr(x.tx), 'means',x[['contact','tx']].mean().to_dict())
print('per-cell quantiles',x.quantile([.1,.5,.9]).to_string())
PY"
