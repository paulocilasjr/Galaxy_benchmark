/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd, numpy as np
D=pd.read_csv('single_cell_dynamics_question.csv')
D['dist']=np.sqrt((D.enh_x-D.prom_x)**2+(D.enh_y-D.prom_y)**2+(D.enh_z-D.prom_z)**2)
C=(D.dist.to_numpy().reshape(-1,250)<=260).astype(int)
T=D.transcription.to_numpy().reshape(-1,250).astype(int)
# all transitions previous -> current; rows prev c,t, columns current c/t
for name,A in [('C',C),('T',T)]:
 print(name,'transition conditional on own previous')
 for prev in [0,1]:
  now=A[:,1:][A[:,:-1]==prev]
  print(prev,'n',len(now),'P1',now.mean())
print('joint transition states prev (C,T) -> current')
for cp in [0,1]:
 for tp in [0,1]:
  mask=(C[:,:-1]==cp)&(T[:,:-1]==tp)
  print('prev',cp,tp,'n',mask.sum(),'P(Cnow=1)',C[:,1:][mask].mean(),'P(Tnow=1)',T[:,1:][mask].mean(),'joint',np.mean(C[:,1:][mask]*T[:,1:][mask]))
print('Cnow probability by Tprev controlling Cprev')
for cp in [0,1]:
 for tp in [0,1]:
  mask=(C[:,:-1]==cp)&(T[:,:-1]==tp)
  vals=C[:,1:][mask]
  print('Cprev',cp,'Tprev',tp,'n',len(vals),'P',vals.mean())
print('Tnow probability by Cprev controlling Tprev')
for tp in [0,1]:
 for cp in [0,1]:
  mask=(T[:,:-1]==tp)&(C[:,:-1]==cp)
  vals=T[:,1:][mask]
  print('Tprev',tp,'Cprev',cp,'n',len(vals),'P',vals.mean())
# Compare to transitions from states etc
PY"
