/bin/bash -lc "python - <<'PY'
import csv,math
from collections import defaultdict
A=defaultdict(list)
with open('single_cell_dynamics_question.csv') as f:
 for x in csv.DictReader(f):
  ex=[float(x[k]) for k in ('enh_x','enh_y','enh_z')]; px=[float(x[k]) for k in ('prom_x','prom_y','prom_z')]
  d=math.sqrt(sum((a-b)**2 for a,b in zip(ex,px)))
  A[int(x['cell_id'])].append((int(x['time']),d,int(d<=260),int(x['transcription'])))
for cell in range(10):
 print('\\ncell',cell)
 print(' '.join(f'{t}:{d:.0f}/{c}/{y}' for t,d,c,y in A[cell][:50]))
PY"
