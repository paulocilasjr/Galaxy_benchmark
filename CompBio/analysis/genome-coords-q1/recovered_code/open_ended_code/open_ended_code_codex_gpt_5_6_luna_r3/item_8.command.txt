/bin/bash -lc "python - <<'PY'
import csv,math
from collections import defaultdict,Counter
A=defaultdict(list)
with open('single_cell_dynamics_question.csv') as f:
 for x in csv.DictReader(f):
  ex=[float(x[k]) for k in ('enh_x','enh_y','enh_z')]; px=[float(x[k]) for k in ('prom_x','prom_y','prom_z')]
  d=math.sqrt(sum((a-b)**2 for a,b in zip(ex,px)))
  A[int(x['cell_id'])].append((int(x['time']),int(d<=260),int(x['transcription']),d))
for vals in A.values(): vals.sort()
for lag in range(0,8):
 n=Counter()
 for vals in A.values():
  for i in range(lag,len(vals)):
   n[(vals[i-lag][1],vals[i][2])]+=1
 print('\\nlag',lag,n)
 for c in [0,1]:
  tot=sum(n[c,y] for y in [0,1])
  print('Cprev',c,'n',tot,'Y1',n[c,1]/tot)
# Separate event onset / offset maybe
for lag in range(1,5):
 n=Counter()
 for vals in A.values():
  for i in range(lag,len(vals)):
   c0,y0=vals[i-lag][1:3]; c1,y1=vals[i][1:3]
   n[(c0,y0,c1,y1)]+=1
 print('lag',lag)
 for k,v in sorted(n.items()): print(k,v)
PY"
