import csv, math
import numpy as np
T=250; cells=[]
with open('single_cell_dynamics_question.csv') as f:
 r=csv.DictReader(f); cur=[]; last=None
 for row in r:
  cid=int(row['cell_id'])
  if last is not None and cid"'!=last: cells.append(cur); cur=[]
  last=cid
  ex=float(row['"'enh_x']);ey=float(row['enh_y']);ez=float(row['enh_z']);px=float(row['prom_x']);py=float(row['prom_y']);pz=float(row['prom_z'])
  d=math.sqrt((ex-px)**2+(ey-py)**2+(ez-pz)**2)
  cur.append((int(d<=260.0), int(row['transcription'])))
 cells.append(cur)
# per-cell delta C_t given T_{t-1}=1 vs 0
dC=[]; dT=[]
for c in cells:
 n=[[0,0],[0,0]]; # prev T 0/1 -> curr C 0/1

 if p1 is not None and p0 is not None and n[1].sum()>0 and n[0].sum()>0:
  dC.append(p1-p0)
 # T_t given C_{t-1}
 m=[[0,0],[0,0]]
 for t in range(1,T):
  prevC=c[t-1][0]; curT=c[t][1]

 q1=m[1][1]/m[1].sum() if m[1].sum() else None
 q0=m[0][1]/m[0].sum() if m[0].sum() else None
 if q1 is not None and q0 is not None and m[1].sum()>0 and m[0].sum()>0:
  dT.append(q1-q0)
print('per-cell T->C delta: n',len(dC),'mean',np.mean(dC),'median',np.median(dC),'positive',sum(x>0 for x in dC),'negative',sum(x<0 for x in dC),'zero',sum(x==0 for x in dC))

from scipy.stats import wilcoxon, ttest_1samp, binomtest
