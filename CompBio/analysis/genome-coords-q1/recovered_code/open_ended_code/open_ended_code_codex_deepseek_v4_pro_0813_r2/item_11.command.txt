import csv, math
import numpy as np
from scipy.optimize import minimize
T=250; cells=[]
with open('single_cell_dynamics_question.csv') as f:
 r=csv.DictReader(f); cur=[]; last=None
 for row in r:
  cid=int(row['cell_id'])
  if last is not None and cid"'!=last: cells.append(cur); cur=[]
  last=cid
  ex=float(row['"'enh_x']);ey=float(row['enh_y']);ez=float(row['enh_z']);px=float(row['prom_x']);py=float(row['prom_y']);pz=float(row['prom_z'])
  d=math.sqrt((ex-px)**2+(ey-py)**2+(ez-pz)**2)
  cur.append((int(d<=260.0), int(row['transcription'])))
 cells.append(cur)

def make(outcome, pred_lags):
    # outcome var 0/1 and predictors are lags for C/T; each predictor tuple (var, lag>0)
    Y=[]; X=[]
    for c in cells:
        for t in range(2,T): # allow lag2
            # check all lags
            if all(t-lag>=0 for _,lag in pred_lags):
                Y.append(c[t][outcome])
                X.append([c[t-lag][var] for var,lag in pred_lags])
    return np.array(X,float),np.array(Y,float)

def fit(X,Y):
    X1=np.column_stack([np.ones(len(X)),X])
    def nll(beta):
        z=X1@beta
        p=1/(1+np.exp(-np.clip(z,-30,30)))
        return -np.sum(Y*np.log(p+1e-15)+(1-Y)*np.log(1-p+1e-15))
    res=minimize(nll, np.zeros(X1.shape[1]), method='BFGS')
    beta=res.x
    # Hessian
    z=X1@beta; p=1/(1+np.exp(-np.clip(z,-30,30)))
    W=p*(1-p)
    H=X1.T@(W[:,None]*X1)
    se=np.sqrt(np.diag(np.linalg.inv(H+np.eye(H.shape[0])*1e-8)))
    zs=beta/se
    from scipy.stats import norm
    pvals=2*(1-norm.cdf(np.abs(zs)))
    return beta,se,pvals,res.fun,len(Y)

# Models for T_t with C_t-1 and T_t-1, and C_t-2, T_t-2
for outcome,name,specs in [
 (1,'T_t',[('C',1),('T',1),('C',2),('T',2)]),
 (0,'C_t',[('T',1),('C',1),('T',2),('C',2)]),
]:
    pred=[]
    for var,lag in specs:
        pred.append((var,lag))
    X,Y=make(outcome,pred)
    print('\\nModel',name,'n',len(Y),'base',Y.mean())
    beta,se,p,z=fit(X,Y)
    labels=['intercept']+[f'{var} lag{lag}' for var,lag in pred]

        print(f'{lab:12s} coef {b:+.4f} se {s:.4f} z {b/s:+.2f} p {pv:.4g} OR {math.exp(b):.3f}')
