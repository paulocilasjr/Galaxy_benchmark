 on=(X[:,1:]==1)&(X[:,:-1]==0)
 print('\\nsource onsets',srcname,'n',on.sum())
 for off in range(-5,11):
  # onset at t=1..249, indexed on col t-1
  ts=np.where(on)[0]*250+np.where(on)[1]+1
  # convert row/col direct
  rows, cols=np.where(on); cols=cols+1
  valid=(cols+off>=0)&(cols+off<250)
  vals=Y[rows[valid], cols[valid]+off]
  print(f'offset {off:+d}: n={len(vals)} rate={vals.mean():.6f}')
# Compare target rate by source state at each offset, perhaps target conditional state not onset
for srcname,X,Y in [('C',C,T),('T',T,C)]:
 print('\\nsource state',srcname)
 for off in range(-3,6):
  if off>=0:
   x=X[:,:250-off].ravel(); y=Y[:,off:].ravel()
  else:
   x=X[:,-off:].ravel(); y=Y[:,:250+off].ravel()

for srcname,X,Y in [('C',C,T),('T',T,C)]:

 for off in range(0,6):

  yy=tar_on[rows[valid],t[valid]+off-1] # tar_on index is target time -1
  print('target onset at offset',off,'rate',yy.mean(),'n',len(yy))
