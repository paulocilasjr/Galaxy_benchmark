/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1)).reshape(600,250)
C=(D<=260).astype(int); T=df['transcription'].values.reshape(600,250)
states=C[:,:249]*2+T[:,:249]  # C high bit, T low bit
nexts=C[:,1:]*2+T[:,1:]
# transition counts/probs
for s in range(4):
    mask=states==s
    n=mask.sum()
    print('state C=%d T=%d n=%d'%(s//2,s%2,n),'next probs',[nexts[mask].tolist().count(k)/n if n else np.nan for k in range(4)])
PY"
