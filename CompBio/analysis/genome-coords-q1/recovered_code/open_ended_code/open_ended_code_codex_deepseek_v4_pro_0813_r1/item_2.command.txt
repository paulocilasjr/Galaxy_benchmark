/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1))
C=(D<=260).astype(np.int8); T=df['transcription'].values.astype(np.int8)
# reshape assumes sorted cell,time; verify sorted
print('is sorted?', df.groupby('cell_id')['time'].is_monotonic_increasing.all())
C=C.reshape(600,250); T=T.reshape(600,250)
# overall lag probabilities
print('\\nP(T_t=1 | C_{t-lag}=1) and P(T_t=1 | C=0), diff, odds')
for lag in range(-10,11):
    if lag<=0:
        # contact leads transcription by -lag? define C at t-lag; t=lag..249 if lag>=0, if negative t=0..249+lag
        x=C[:, max(0,lag):250+min(0,lag)] # C_{t-lag}: if lag=0 all; if lag=-3, indices? let's simply arrays aligned below.
    # easier arrays:
    if lag>=0:
        c=C[:, :250-lag]
        y=T[:, lag:]
        x=c
    else:
        c=C[:, -lag:]
        y=T[:, :250+lag]
        x=c
    mask=x==1
    p1=y[mask].mean() if mask.sum() else np.nan
    p0=y[~mask].mean()
    print(lag, 'contact lead lag',lag, 'P trans after/at',p1,p0,'diff',p1-p0)
print('\\nP(C_t=1 | T_{t-lag}=1)')
for lag in range(-10,11):
    if lag>=0:
        tt=T[:, :250-lag]
        c=C[:, lag:]
    else:
        tt=T[:, -lag:]
        c=C[:, :250+lag]
    mask=tt==1
    p1=c[mask].mean() if mask.sum() else np.nan
    p0=c[~mask].mean()
    print(lag, 'trans lead lag',lag, 'P contact after',p1,p0,'diff',p1-p0)
PY"
