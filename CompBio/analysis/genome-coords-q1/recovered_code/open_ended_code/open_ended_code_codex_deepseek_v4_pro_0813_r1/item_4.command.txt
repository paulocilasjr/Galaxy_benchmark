/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import chi2
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1))
C=(D<=260).astype(int); T=df['transcription'].values.astype(int)
C=C.reshape(600,250); T=T.reshape(600,250)
# Logistic regression via statsmodels if available; else scipy optimize
try:
 import statsmodels.api as sm
 print('statsmodels yes')
except Exception as e:
 print('no statsmodels', e); raise
# create lag1 arrays, flatten all t 1..249
T_prev=T[:,:249].ravel(); C_prev=C[:,:249].ravel(); T_cur=T[:,1:].ravel(); C_cur=C[:,1:].ravel()
# T outcome
X=sm.add_constant(np.c_[T_prev,C_prev])
res=sm.Logit(T_cur,X).fit(disp=0)
print('Predict T_t from T_prev,C_prev')
print(res.summary2().tables[1])
print('odds', np.exp(res.params))
# C outcome
X=sm.add_constant(np.c_[C_prev,T_prev])
res2=sm.Logit(C_cur,X).fit(disp=0)
print('Predict C_t from C_prev,T_prev')
print(res2.summary2().tables[1])
print('odds', np.exp(res2.params))
# include simultaneous C_t in T model; and T_t in C model
X=sm.add_constant(np.c_[T_prev,C_prev,C_cur])
res3=sm.Logit(T_cur,X).fit(disp=0)
print('Predict T_t from T_prev,C_prev,C_cur')
print(res3.summary2().tables[1]); print(np.exp(res3.params))
X=sm.add_constant(np.c_[C_prev,T_prev,T_cur])
res4=sm.Logit(C_cur,X).fit(disp=0)
print('Predict C_t from C_prev,T_prev,T_cur')
print(res4.summary2().tables[1]); print(np.exp(res4.params))
PY"
