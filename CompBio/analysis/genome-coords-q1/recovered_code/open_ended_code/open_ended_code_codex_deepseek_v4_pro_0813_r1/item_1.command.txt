/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
print(df.shape, df.columns.tolist())
print(df.dtypes)
print(df['cell_id'].nunique(), df['time'].min(), df['time'].max(), df['time'].nunique())
print(df['transcription'].describe())
# distance
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1))
df['dist']=D
df['contact']=(D<=260).astype(int)
print(df[['dist','contact']].describe())
print('contact counts',df['contact'].value_counts())
print('transcription counts',df['transcription'].value_counts())
print('joint',pd.crosstab(df['contact'],df['transcription']))
print('within contact trans rate',df.loc[df.contact==1,'transcription'].mean(),'outside',df.loc[df.contact==0,'transcription'].mean())
print('prob contact when trans',df.loc[df.transcription==1,'contact'].mean(),'when no',df.loc[df.transcription==0,'contact'].mean())
PY"
