/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D=np.sqrt(((df[['enh_x','enh_y','enh_z']].values-df[['prom_x','prom_y','prom_z']].values)**2).sum(1)).reshape(600,250)
T=df['transcription'].values.reshape(600,250)
# mean distance around T=1 times (lag relative to transcription)
print('Mean distance around T_t=1, lag -10..10 (lag negative before, positive after)')
for lag in range(-10,11):
    if lag>=0:
        # D at t+lag aligned with T_t
        d=D[:,lag:250] if lag>0 else D
        t=T[:,:250-lag] if lag>0 else T
    else:
        l=-lag
        d=D[:,:250-l]
        t=T[:,l:]
    mask=t==1
    print(lag, np.mean(d[mask]) if mask.sum() else np.nan, 'n',mask.sum())
# compare distance distribution T=0 vs T=1 at same time, and lag +1
print('\\nsame time mean dist T0 vs T1',D[T==0].mean(),D[T==1].mean())
# mean distance at t+1 given T_t
mask=T[:,:249]==1; print('D next after T_t=1',D[:,1:][mask].mean(),'D next after T_t=0',D[:,1:][~mask].mean())
# mean distance at t-1 before T_t
mask=T[:,1:]==1; print('D prev before T_t=1',D[:,:249][mask].mean(),'prev before T=0',D[:,:249][~mask].mean())
PY"
