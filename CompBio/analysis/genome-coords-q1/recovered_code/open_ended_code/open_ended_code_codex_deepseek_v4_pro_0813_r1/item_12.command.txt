/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
E=df[['enh_x','enh_y','enh_z']].values.reshape(600,250,3)
P=df[['prom_x','prom_y','prom_z']].values.reshape(600,250,3)
D=np.sqrt(((E-P)**2).sum(2))
T=df['transcription'].values.reshape(600,250)
C=(D<=260).astype(int)
# displacement magnitude per step enh/prom
dE=np.sqrt(((E[:,1:,:]-E[:,:-1,:])**2).sum(2))
dP=np.sqrt(((P[:,1:,:]-P[:,:-1,:])**2).sum(2))
dD=D[:,1:]-D[:,:-1]
# distance change from t to t+1 given T_t=1 vs 0
for name, state in [('T_t', T[:,:-1]),('C_t', C[:,:-1])]:
    print('\\nconditional on',name)
    for v in [0,1]:
        mask=state==v
        print(v,'n',mask.sum(),'mean dD (D_t+1-D_t)',dD[mask].mean(),'mean dE',dE[mask].mean(),'mean dP',dP[mask].mean())
# vector displacement toward promoter for enhancer: component of enhancer displacement along vector from enhancer to promoter (positive if moves toward promoter)
vecP_minus_E=P-E # from E to P
unit=vecP_minus_E/np.maximum(D[...,None],1e-6)
dispE=E[:,1:,:]-E[:,:-1,:]
projE_toward=(dispE*unit[:,:-1]).sum(2)
dispP=P[:,1:,:]-P[:,:-1,:]
unit_from_E_to_P=unit[:,:-1]
projP_toward=-(dispP*unit_from_E_to_P).sum(2) # promoter moving toward enhancer along E->P: displacement P opposite vector => negative dot if toward E? E->P vector, P moves toward E has dot negative, so toward= -dot.
for name,state in [('T_t', T[:,:-1]),('C_t', C[:,:-1])]:
    print('\\nprojections cond',name)
    for v in [0,1]:
        mask=state==v
        print(v,'mean enh toward prom',projE_toward[mask].mean(),'mean prom toward enh',projP_toward[mask].mean(),'mean distance change',dD[mask].mean())
PY"
