/bin/bash -lc "python -S -u - <<'PY'
import csv, math
from collections import Counter,defaultdict
cells=defaultdict(list); counts=Counter(); n=0; sd=0
with open('single_cell_dynamics_question.csv') as f:
 for r in csv.DictReader(f):
  d=math.sqrt(sum((float(r['enh_'+a])-float(r['prom_'+a]))**2 for a in 'xyz'))
  c=int(d<=260); t=int(r['transcription']); counts[c,t]+=1
  cells[int(r['cell_id'])].append((int(r['time']),c,t)); n+=1; sd+=d
print('rows',n,'cells',len(cells),'mean distance',sd/n,'contact/transcription counts',dict(counts))
for lag in [-10,-5,-2,-1,0,1,2,5,10]:
 tab=Counter()
 for v in cells.values():
  v.sort()
  for i in range(max(0,-lag),min(len(v),len(v)-lag)):
   tab[v[i][1],v[i+lag][2]]+=1
 print('lag',lag,'pTgivenC',tab[1,1]/(tab[1,0]+tab[1,1]),'pTgivenNoC',tab[0,1]/(tab[0,0]+tab[0,1]))
PY"
