cells=collections.defaultdict(list)
with open('single_cell_dynamics_question.csv',newline='') as f:
 for r in csv.DictReader(f):
  d=((float(r['enh_x'])-float(r['prom_x']))**2+(float(r['enh_y'])-float(r['prom_y']))**2+(float(r['enh_z'])-float(r['prom_z']))**2)**.5
  cells[int(r['cell_id'])].append((int(r['time']), int(d<=260), int(r['transcription']), d))
# transition counts indexed C_t,T_t,out
for outcome in ['Tnext','Cnext']:
 tab=collections.Counter()
 for v in cells.values():
  v=sorted(v)
  for a,b in zip(v,v[1:]):
   c,t=a[1],a[2]
   y=b[2] if outcome=='Tnext' else b[1]
   tab[c,t,y]+=1
 print(outcome)
 for c in [0,1]:
  for t in [0,1]:
   n0,n1=tab[c,t,0],tab[c,t,1]

for outcome in ['Tnext','Cnext']:
 effects=[]
 for v in cells.values():
  v=sorted(v); vals={}
  # for Tnext compare c current within each current t. For Cnext compare t current within each current c
  for hold in [0,1]:
   group=[[],[]]
   for a,b in zip(v,v[1:]):

    pred=c if outcome=='Tnext' else t
    h=t if outcome=='Tnext' else c

   if all(group): effects.append((outcome,hold,sum(group[1])/len(group[1])-sum(group[0])/len(group[0]),len(group[0]),len(group[1])))
 print(outcome,'mean cell-stratum diff',sum(e[2] for e in effects)/len(effects),'n strata',len(effects))
# event transition pair counts C_t,T_t -> Cnext,Tnext
pair=collections.Counter()
for v in cells.values():
 v=sorted(v)
 for a,b in zip(v,v[1:]): pair[(a[1],a[2],b[1],b[2])]+=1
print('joint transitions')
for k in sorted(pair): print(k,pair[k])
