/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D2=((df[['enh_x','enh_y','enh_z']].values - df[['prom_x','prom_y','prom_z']].values)**2).sum(axis=1)
df['dist']=np.sqrt(D2); df['contact']=(df.dist<=260).astype(int)
D=np.stack([s.values for _,s in df.groupby('cell_id')['dist']])
T=np.stack([s.values for _,s in df.groupby('cell_id')['transcription']])
C=np.stack([s.values for _,s in df.groupby('cell_id')['contact']])
print('mean dist by T current:', D[T==1].mean(), D[T==0].mean())
# continuous distance vs future/past transcription lag
for k in range(-5,6):
    if k<0:
        kk=-k; pred=T[:,kk:]; target=D[:,:-kk]
    elif k>0:
        pred=T[:,:-k]; target=D[:,k:]
    else:
        pred=T; target=D
    print('lag',k,'T=1 mean future/current D:',target[pred==1].mean(),'T=0:',target[pred==0].mean(), 'diff', target[pred==1].mean()-target[pred==0].mean())
print('\\nContact future/past by T:')
for k in range(-5,6):
    if k<0:
        kk=-k; pred=T[:,kk:]; target=C[:,:-kk]
    elif k>0:
        pred=T[:,:-k]; target=C[:,k:]
    else:
        pred=T; target=C
    print('lag',k,'P(C|T=1):',target[pred==1].mean(),'P(C|T=0):',target[pred==0].mean())
print('\\nT future/past by C:')
for k in range(-5,6):
    if k<0:
        kk=-k; pred=C[:,kk:]; target=T[:,:-kk]
    elif k>0:
        pred=C[:,:-k]; target=T[:,k:]
    else:
        pred=C; target=T
    print('lag',k,'P(T|C=1):',target[pred==1].mean(),'P(T|C=0):',target[pred==0].mean())
PY"
