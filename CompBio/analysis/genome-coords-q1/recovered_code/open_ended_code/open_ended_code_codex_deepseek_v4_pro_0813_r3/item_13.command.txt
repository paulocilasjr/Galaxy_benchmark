/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np, statsmodels.api as sm
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
D2=((df[['enh_x','enh_y','enh_z']].values - df[['prom_x','prom_y','prom_z']].values)**2).sum(axis=1)
df['dist']=np.sqrt(D2); df['contact']=(df.dist<=260).astype(int)
T=np.stack([s.values for _,s in df.groupby('cell_id')['transcription']])
C=np.stack([s.values for _,s in df.groupby('cell_id')['contact']])
# Predict T_t using T_{t-1}, C_{t-1}, T_{t-2}, C_{t-2}, maybe T_{t-3}, C_{t-3}
def make_data(target, lags=3):
    Y=target[:,lags:].ravel()
    cols=[]
    names=[]
    for lag in range(1,lags+1):
        cols.append(T[:,lags-lag:-lag].ravel() if lag<lags else T[:,:-lags].ravel() if lags==lag else None)
    # Simpler: use loop
    arrays=[]; names=[]
    for lag in range(1,lags+1):
        arrays.append(T[:,lags-lag:T.shape[1]-lag].ravel()); names.append(f'T_lag{lag}')
        arrays.append(C[:,lags-lag:C.shape[1]-lag].ravel()); names.append(f'C_lag{lag}')
    X=np.column_stack(arrays)
    X=sm.add_constant(X)
    return Y,X,names
for target,label in [(T,'T'),(C,'C')]:
    Y,X,names=make_data(target,3)
    model=sm.Logit(Y,X).fit(disp=0)
    print('\\nPredicting',label,'t from lags 1-3')
    print(pd.DataFrame({'var':['const']+names,'coef':model.params,'p':model.pvalues}).to_string(index=False))
    print('Pseudo R2',model.prsquared)
PY"
