/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
p='single_cell_dynamics_question.csv'
df=pd.read_csv(p)
# distance
d2=((df[['enh_x','enh_y','enh_z']].values - df[['prom_x','prom_y','prom_z']].values)**2).sum(axis=1)
df['dist']=np.sqrt(d2)
df['contact']=(df.dist<=260).astype(int)
print('contact rate',df.contact.mean())
print('dist quantiles',df.dist.quantile([0,.01,.05,.1,.25,.5,.75,.9,.99,1]).to_string())
print('contact corr contemporaneous with trans', np.corrcoef(df.contact, df.transcription)[0,1])
print(pd.crosstab(df.contact,df.transcription,normalize='all'))
print(pd.crosstab(df.contact,df.transcription,normalize='index'))
PY"
