/bin/bash -lc "python - <<'PY'
import csv, math, numpy as np
p='single_cell_dynamics_question.csv'
a=np.empty((600,250),float); y=np.empty((600,250),int)
with open(p,newline='') as f:
    rd=csv.DictReader(f)
    for row in rd:
        i=int(row['cell_id']); t=int(row['time'])
        dx=float(row['enh_x'])-float(row['prom_x']); dy=float(row['enh_y'])-float(row['prom_y']); dz=float(row['enh_z'])-float(row['prom_z'])
        a[i,t]=math.sqrt(dx*dx+dy*dy+dz*dz); y[i,t]=int(row['transcription'])
c=(a<=260).astype(int)
print('distance mean sd quant', a.mean(),a.std(),np.quantile(a,[0,.01,.05,.1,.25,.5,.75,.9,.95,.99,1]))
print('contact rate',c.mean(),'tx rate',y.mean())
for cv in [0,1]:
 print('Ptx|contact',cv,y[c==cv].mean(), 'n', (c==cv).sum())
for yv in [0,1]:
 print('Pcontact|tx',yv,c[y==yv].mean())
# lag associations P(y_t=1 | c_t-k); P(c_t=1 | y_t-k)
for k in range(-10,11):
    # corr C(t) with Y(t+k) if k positive contact leads tx
    if k>=0: x=c[:,:250-k or None].ravel(); z=y[:,k:].ravel()
    else: kk=-k; x=c[:,kk:].ravel(); z=y[:,:250-kk].ravel()
    corr=np.corrcoef(x,z)[0,1]
    diff=z[x==1].mean()-z[x==0].mean()
    print(f'lag {k:+d} corr {corr:.5f} diffY {diff:.5f}')
# transition predictive effects conditional on previous same outcome
print('contact leading tx conditional Yprev')
for yp in [0,1]:
 for cp in [0,1]:
  mask=(y[:,:-1]==yp)&(c[:,:-1]==cp)
  print(yp,cp,y[:,1:][mask].mean(),mask.sum())
print('tx leading contact conditional Cprev')
for cp in [0,1]:
 for yp in [0,1]:
  mask=(c[:,:-1]==cp)&(y[:,:-1]==yp)
  print(cp,yp,c[:,1:][mask].mean(),mask.sum())
PY"
