/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
from scipy.stats import spearmanr
from sklearn.decomposition import PCA
from sklearn.preprocessing import normalize
fn='sample.swap.atac.q2.tsv.gz'
with gzip.open(fn,'rt') as f:
    hdr=f.readline().rstrip().split('\\t')
    names=hdr[3:]
    rows=[]
    coords=[]
    for line in f:
        p=line.rstrip().split('\\t')
        coords.append(tuple(p[:3])); rows.append([int(x) for x in p[3:]])
X=np.array(rows,dtype=np.float64)
print(X.shape, names)
lib=X.sum(0); print('libs',dict(zip(names,lib.astype(int))))
# normalize counts per library to 1m, log1p
Y=np.log1p(X/lib*1e6)
print('Pearson log normalized')
C=np.corrcoef(Y.T)
for i,n in enumerate(names):
    print(n, ' '.join(f'{names[j]}:{C[i,j]:.4f}' for j in np.argsort(-C[i]) if j"'!=i))
print('"'Spearman raw?')
S=spearmanr(X).statistic
for i,n in enumerate(names):
    print(n, ' '.join(f'{names[j]}:{S[i,j]:.4f}' for j in np.argsort(-S[i]) if j"'!=i))
# PCA weighted bins?
p=PCA().fit(Y)
print('"'PCA var',p.explained_variance_ratio_[:5])
Z=p.transform(Y)
for i,n in enumerate(names):print(n, *(f'{v:.1f}' for v in Z[i,:4]))
# top z-normalized / markers
print('per-column max z bins')
for j,n in enumerate(names):
    z=(Y[:,j]-Y[:,j].mean())/Y[:,j].std()
    ix=np.argsort(-z)[:15]
    print('\\n',n)
    for i in ix[:10]: print(coords[i], int(X[i,j]), round(z[i],2))
PY"
