/bin/bash -lc "python - <<'PY'
import json, subprocess
markers='RHO PDE6A PDE6B GNAT1 NRL NR2E3 SAG GNB1 ROM1 ARR3 GNAT2 PDE6C OPN1SW OPN1MW OPN1LW GNB3 RXRG GLUL RLBP1 SLC1A3 AQP4 VIM SOX9 KCNJ10 CLDN11 GFAP ALDH1L1 S100B AGT GJA1 C1QA C1QB C1QC AIF1 P2RY12 TMEM119 CSF1R LST1 TYROBP RBPMS SNCG THY1 POU4F1 POU4F2 NEFL GAP43 ISL1 DDIT4 LHX1 LHX2 ONECUT1 ONECUT2 CALB1 PROX1 VSX1 GAD1 GAD2 SLC32A1 PAX6 TFAP2A TFAP2B SLC6A9 PRKCA CABP5'.split()
cmd=['curl','-L','--max-time','60','-sS','-X','POST','-H','Content-Type: application/json','-H','Accept: application/json','https://rest.ensembl.org/lookup/symbol/homo_sapiens','--data',json.dumps({'symbols':markers})]
d=json.loads(subprocess.check_output(cmd,text=True))
for g in markers:
 x=d.get(g,{})
 print(f'{g}\\t{x.get(\"seq_region_name\")}\\t{x.get(\"start\")}\\t{x.get(\"end\")}')
PY"
