/bin/bash -lc "python3 - <<'PY'
import requests, json, sys, time
genes=['RHO','NRL','PDE6B','PDE6C','ARR3','OPN1SW','OPN1MW','RLBP1','GLUL','SOX9','APOE','P2RY12','CSF1R','CX3CR1','TYROBP','GFAP','SLC1A3','AQP4','POU4F1','RBPMS','SLC17A6','THY1','ONECUT1','ONECUT2','LHX1','CALB1','GAD1','GAD2','SLC6A9','TFAP2A','TFAP2B','PCP4','CPLX3','NEFL','GRM6','VSX2','OTX2','PRDM1','LHX2','ISL1','NR2E3','CRX']
for g in genes:
    url=f'https://rest.ensembl.org/lookup/symbol/homo_sapiens/{g}?expand=1'
    try:
        r=requests.get(url,headers={'Content-Type':'application/json'},timeout=20)
        d=r.json()
        print(g, d.get('seq_region_name'), d.get('start'), d.get('end'), d.get('strand'))
    except Exception as e:
        print('ERR',g,e)
    time.sleep(0.1)
PY"
