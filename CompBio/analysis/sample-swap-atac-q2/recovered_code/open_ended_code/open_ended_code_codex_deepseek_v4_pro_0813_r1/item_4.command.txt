/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
fn='sample.swap.atac.q2.tsv.gz'
cols=None
X=[]
with gzip.open(fn,'rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    cols=header[3:]
    for line in f:
        p=line.rstrip().split('\\t')
        X.append([int(x) for x in p[3:]])
X=np.array(X,dtype=np.float64)
print(X.shape)
# log? Pearson and Spearman on sparse counts
from scipy.stats import spearmanr, pearsonr
print('Pearson')
for i,a in enumerate(cols):
    for j in range(i):
        c=np.corrcoef(X[:,i],X[:,j])[0,1]
        print(a,cols[j],c)
print('\\nSpearman maybe subsample/log?')
Xl=np.log1p(X)
for i,a in enumerate(cols):
    for j in range(i):
        c=np.corrcoef(Xl[:,i],Xl[:,j])[0,1]
        print(a,cols[j],c)
PY"
