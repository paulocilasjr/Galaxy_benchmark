/bin/bash -lc "python3 - <<'PY'
import requests,time
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE196nnn/GSE196235/suppl/GSE196235_RAW.tar'
# Rod data offset from index
import json
files=json.load(open('tar_index.json'))
rod=[x for x in files if 'Rod_frags' in x[0] and 'Rodbipolar' not in x[0]][0]
dataoff=rod[1]
print(rod)
start=time.time()
r=requests.get(url,headers={'Range':f'bytes={dataoff}-{dataoff+10000000}'},timeout=60)
dt=time.time()-start
print(r.status_code,len(r.content),dt,len(r.content)/dt/1e6,'MB/s')
PY"
