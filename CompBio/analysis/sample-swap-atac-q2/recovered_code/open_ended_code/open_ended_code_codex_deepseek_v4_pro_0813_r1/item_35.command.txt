/bin/bash -lc "python3 - <<'PY'
import numpy as np, gzip
cols=['AIIamacrine','Astrocyte','Cone','GABAamacrine','Horizontal','Microglia','Mullerglia','Retinalganglioncell','Rod']
X=np.zeros((295420,len(cols)),dtype=np.float64)
with gzip.open('sample.swap.atac.q2.tsv.gz','rt') as f:
    f.readline()
    for i,line in enumerate(f):
        p=line.rstrip().split('\\t')
        X[i]=[int(x) for x in p[3:]]
print('X loaded')
for refname in ['AIIamacrine','Astrocyte','Cone','GABAamacrine','Horizontal','Microglia','Mullerglia','Retinalganglioncell']:
    ref=np.load(f'ref_{refname}.npy')
    print('\\nREF',refname)
    scores=[]
    for j,col in enumerate(cols):
        # Pearson raw
        c=np.corrcoef(ref,X[:,j])[0,1]
        # log pearson
        c2=np.corrcoef(np.log1p(ref),np.log1p(X[:,j]))[0,1]
        scores.append((col,c,c2))
    for col,c,c2 in sorted(scores,key=lambda x:-x[1]):
        print(f'{col:25s} raw {c:.6f} log {c2:.6f}')
PY"
