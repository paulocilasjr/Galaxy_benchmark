/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np
path='sample.swap.atac.q2.tsv.gz'
cols=None; auto=np.zeros(0); x=np.zeros(0)
with gzip.open(path,'rt') as f:
    r=csv.reader(f,delimiter='\\t'); header=next(r); cols=header[3:]
    for row in r:
        if len(row)<12: continue
        vals=np.array([float(x) for x in row[3:]],dtype=float)
        if row[0]=='chrX':
            if not len(x): x=vals.copy()
            else: x+=vals
        elif row[0].startswith('chr') and row[0] not in ['chrX','chrY']:
            if not len(auto): auto=vals.copy()
            else: auto+=vals
print('cols',cols)
print('autosome totals', {c:auto[i] for i,c in enumerate(cols)})
print('chrX totals', {c:x[i] for i,c in enumerate(cols)})
print('X/auto ratio', {c:round(x[i]/auto[i],4) for i,c in enumerate(cols)})
PY"
