/bin/bash -lc "python3 - <<'PY'
import requests, struct, json
base='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE196nnn/GSE196235/suppl/GSE196235_RAW.tar'
def get_range(url, start, end):
    headers={'Range': f'bytes={start}-{end}'}
    r=requests.get(url, headers=headers, timeout=30)
    if r.status_code not in (200,206):
        print('bad',r.status_code,r.text[:200])
        return None
    return r.content
entries=[]
offset=0
for idx in range(100):
    b=get_range(base, offset, offset+511)
    if not b or len(b)<512: break
    if b[:512]==b'\\0'*512:
        # maybe zero block; check next
        b2=get_range(base, offset+512, offset+1023)
        if b2 and b2[:512]==b'\\0'*512:
            break
        offset+=512
        continue
    name=b[0:100].split(b'\\0')[0].decode(errors='replace')
    size_field=b[124:136].split(b'\\0')[0].strip()
    size=int(size_field or b'0',8)
    typeflag=b[156:157]
    entries.append((offset,name,size,typeflag))
    print(idx, offset, name, size, typeflag)
    data_start=offset+512
    data_padded=(size+511)//512*512
    offset=data_start+data_padded
print('total entries',len(entries))
json.dump(entries,open('tar_entries.json','w'))
PY"
