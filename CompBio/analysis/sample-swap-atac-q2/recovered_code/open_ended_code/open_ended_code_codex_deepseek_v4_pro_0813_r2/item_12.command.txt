/bin/bash -lc "python3 - <<'PY'
import gzip,csv,numpy as np, requests, time, itertools, json, pickle, os
path='sample.swap.atac.q2.tsv.gz'
# cache tss? 
markers = {
'AIIamacrine': ['GJD2','GJD3','PRKCA','TRPM1','GRM6','ISL1','TFAP2A','TFAP2B'],
'Astrocyte': ['GFAP','S100B','AQP4','ALDH1L1','SLC1A3','GJA1','SOX9','VIM'],
'Cone': ['PDE6H','GNAT2','OPN1SW','OPN1MW','PDE6C','ARR3','THRB','RXRG'],
'GABAamacrine': ['GAD1','GAD2','SLC6A1','SLC32A1','TFAP2A','TFAP2B','ONECUT2','MEIS2'],
'Horizontal': ['ONECUT1','ONECUT2','LHX1','CALB1','GABRA1','GABRG2','SEMA6A'],
'Microglia': ['P2RY12','TMEM119','CX3CR1','CSF1R','AIF1','C1QA','C1QB','TYROBP'],
'Mullerglia': ['RLBP1','GLUL','SOX9','APOE','CLU','SLC1A3','AQP4','DKK3'],
'Retinalganglioncell': ['RBPMS','POU4F2','POU4F1','ISL1','SNCG','THY1','NEFL','SLC17A6'],
'Rod': ['RHO','NRL','NR2E3','PDE6B','GNAT1','CNGA1','PDE6A','ROM1','CNGB1','SAG'],
}
allgenes=sorted(set(sum(markers.values(),[])))
gene_tss={}
for gene in allgenes:
    try:
        r=requests.get(f'https://rest.ensembl.org/lookup/symbol/homo_sapiens/{gene}?content-type=application/json',timeout=20)
        if r.status_code==200:
            j=r.json(); gene_tss[gene]=('chr'+j['seq_region_name'], int(j['start']), int(j['end']), int(j['strand']))
    except Exception: pass
    time.sleep(0.03)
# load matrix cpm
cols=None; data={}
with gzip.open(path,'rt') as f:
    r=csv.reader(f,delimiter='\\t'); header=next(r); cols=header[3:]
    for row in r:
        if len(row)<12: continue
        data[(row[0],int(row[1]))]=np.array([float(x) for x in row[3:]],dtype=float)
tot=sum(data.values())
# cpm
for k,v in data.items(): data[k]=v/tot*1e6

def promoter_vals(chrom,tss_start,tss_end,strand):
    tss=tss_start if strand==1 else tss_end
    lo=tss-2000; hi=tss+2000
    b=(lo//10000)*10000; vals=[]
    while b<=hi:
        if (chrom,b) in data: vals.append(data[(chrom,b)])
        b+=10000
    return np.sum(vals,axis=0) if vals else np.zeros(len(cols))
# precompute per marker set matrix M: ncelltypes x ncolumns = mean cpm across genes
M=np.zeros((len(cols),len(cols)))
valid_counts=[]
for ct_idx,ct in enumerate(cols):
    genes=markers[ct]
    scores=np.zeros(len(cols)); ng=0
    for gene in genes:
        if gene not in gene_tss: continue
        v=promoter_vals(*gene_tss[gene])
        if v.sum()==0: continue
        # normalize each gene across celltypes by max then avg? use raw cpm mean (could dominate)
        scores += v/v.max(); ng+=1
    M[ct_idx]=scores/max(ng,1)
    valid_counts.append(ng)
print('valid counts',dict(zip(cols,valid_counts)))
print('baseline diag (self marker score):')
for i,c in enumerate(cols): print(c, round(M[i,i],3))

# For each candidate swap, compute objective = sum_i M[i, perm(i)] where perm(i) is column assigned to label i under swap.
# Also compute gain
base=np.trace(M)
print('base objective', base)
results=[]
for i,j in itertools.combinations(range(len(cols)),2):
    perm=list(range(len(cols))); perm[i],perm[j]=perm[j],perm[i]
    obj=sum(M[k,perm[k]] for k in range(len(cols)))
    gain=obj-base
    results.append((gain,i,j,obj))
for gain,i,j,obj in sorted(results, reverse=True)[:20]:
    print(f'swap {cols[i]},{cols[j]}: gain {gain:.3f} obj {obj:.3f}')
print('\\nworst')
for gain,i,j,obj in sorted(results)[:10]:
    print(f'swap {cols[i]},{cols[j]}: gain {gain:.3f} obj {obj:.3f}')
PY"
