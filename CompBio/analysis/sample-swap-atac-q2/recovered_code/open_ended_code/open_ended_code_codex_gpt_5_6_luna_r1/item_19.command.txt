/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for g in ['GJD2','SLC6A9','GFAP','C1QA','RLBP1','ARR3','GAD1','ONECUT1','NEFL','RHO','PDE6A']:
 u='https://rest.ensembl.org/lookup/symbol/homo_sapiens/'+g+'?expand=1'
 req=urllib.request.Request(u,headers={'Content-Type':'application/json','User-Agent':'codex'})
 try:
  x=json.load(urllib.request.urlopen(req,timeout=20))
  print(g,x.get('seq_region_name'),x.get('start'),x.get('end'),x.get('strand'),x.get('assembly_name'),x.get('display_name'))
 except Exception as e: print(g,e)
PY"
