/bin/bash -lc "python3 - <<'PY'
import gzip,re,pandas as pd
markers=['RLBP1','GLUL','SOX9','VIM','AQP4','CRYAB','CLU','SLC1A3','SLC1A2','DIO2','APOE','HES1','HES5','ID3','FABP7','GFAP','ALDH1L1','S100B','GJA1','PAX2','NFIA','NFIB','PITX2','HEY2','CRX','OTX2','RHO','GNAT1','OPN1LW','OPN1SW','ARR3','RBPMS','SNCG','THY1','POU4F2','CX3CR1','C1QA','TYROBP','AIF1','CSF1R']
need=set(markers); coords={}; pat=re.compile(r'gene_name \"(["'^"]+)"'"')
with gzip.open('gencode.v44.annotation.gtf.gz','rt') as f:
 for line in f:
  if line.startswith('#'): continue
  fs=line.rstrip('\\n').split('\\t')
  if len(fs)>=9 and fs[2]=='gene':
   m=pat.search(fs[8])
   if m and m.group(1) in need: coords[m.group(1)]=(fs[0],int(fs[3])-1,int(fs[4]))
df=pd.read_csv('sample.swap.atac.q2.tsv.gz',sep='\\t'); cts=list(df.columns[3:]); tot=df[cts].sum()
for g in markers:
 if g not in coords: print(g,'missing'); continue
 chrom,start,end=coords[g]; lo=max(0,start-50000); hi=end+50000
 mask=df.chrom.eq(chrom)&(df.end>lo)&(df.start<hi)
 vals=(df.loc[mask,cts].sum()/tot*1e6).sort_values(ascending=False)
 print(g, 'top', ', '.join(f'{k}:{v:.1f}' for k,v in vals.head(5).items()))
PY"
