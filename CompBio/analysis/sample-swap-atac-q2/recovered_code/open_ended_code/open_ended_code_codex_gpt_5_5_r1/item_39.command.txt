/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
rna=pd.read_csv('multiome.match.atac.rna.q1.rna.tsv.gz', sep='\\t')
cols=[str(i) for i in range(8)]
markers = {
'Rod':['RHO','NRL','GNAT1','PDE6A','PDE6B','SAG','CNGA1','RCVRN'],
'Cone':['OPN1LW','OPN1MW','OPN1SW','ARR3','GNAT2','PDE6C','CNGB3','GNGT2'],
'Mullerglia':['RLBP1','GLUL','SOX9','VIM','AQP4','CRYAB','CLU','SLC1A3'],
'Astrocyte':['GFAP','ALDH1L1','S100B','AQP4','SOX9','GJA1','SLC1A3'],
'Microglia':['CX3CR1','P2RY12','AIF1','TYROBP','C1QA','C1QB','CSF1R','SPI1','PTPRC'],
'Retinalganglioncell':['RBPMS','SNCG','THY1','POU4F1','POU4F2','NEFL','NEFM'],
'Horizontal':['LHX1','ONECUT1','ONECUT2','PROX1','CALB1','LIM2'],
'GABAamacrine':['GAD1','GAD2','SLC6A1','SLC32A1','DLX1','DLX2','TFAP2A','TFAP2B'],
'AIIamacrine':['GRIK1','SLC1A6','PCP2','CACNA2D3','MEIS2','TFAP2A','TFAP2B']
}
# log tpm z by gene across columns; average marker raw and z
rna_idx=rna.set_index('gene')
print('Marker raw top:')
for group,gs in markers.items():
    present=[g for g in gs if g in rna_idx.index]
    if not present: continue
    vals=rna_idx.loc[present,cols].astype(float).mean(axis=0)
    print(group, 'present', present, 'top3', ', '.join(f'{k}:{v:.2f}' for k,v in vals.sort_values(ascending=False).head(3).items()))
print('\\nPer-column top marker groups by z-sum')
scores=pd.DataFrame(index=markers.keys(), columns=cols, dtype=float)
log=np.log2(rna_idx[cols].astype(float)+1)
z=(log-log.mean(axis=1).values[:,None])/(log.std(axis=1,ddof=0).replace(0,np.nan).values[:,None])
z=z.fillna(0)
for group,gs in markers.items():
    present=[g for g in gs if g in z.index]
    scores.loc[group]=z.loc[present].mean(axis=0)
print(scores.round(2).to_string())
for c in cols:
    print(c, scores[c].sort_values(ascending=False).head(4).to_dict())
PY"
