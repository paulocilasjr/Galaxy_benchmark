/bin/bash -lc "python3 - <<'PY'
import gzip
import numpy as np
names=[]
rows=[]
with gzip.open('sample.swap.atac.q2.tsv.gz','rt') as f:
    names=f.readline().rstrip().split('\\t')
    for line in f:
        a=line.rstrip().split('\\t')
        rows.append((a[0],int(a[1]),int(a[2]),*[int(x) for x in a[3:]]))
print(names, len(rows))
a=np.array([r[3:] for r in rows], dtype=np.float64)
# log normalize / correlation across bins
for transform in ['raw','log','binary']:
    if transform=='raw': x=a
    elif transform=='log': x=np.log1p(a)
    else: x=(a>0).astype(float)
    # cosine, corr
    c=np.corrcoef(x.T)
    print('\\n',transform)
    print(' '.join(f'{n:22s}' for n in names[3:]))
    for n,row in zip(names[3:],c): print(f'{n:22s}', ' '.join(f'{v:.3f}' for v in row))
    # nearest each excluding self
    print('nearest')
    for i,n in enumerate(names[3:]):
        j=np.argsort(c[i])[::-1][1]
        print(n,names[3+j],c[i,j])
PY"
