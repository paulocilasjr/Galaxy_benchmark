/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
fn='sample.swap.atac.q2.tsv.gz'
df=pd.read_csv(fn,sep='\\t')
print(df.shape)
cols=list(df.columns[3:])
print(cols)
lib=df[cols].sum()
print('library sizes')
print(lib.sort_values())
# filter variable/accessed bins
mat=df[cols].astype(float)
print('nonzero bins per col')
print((mat>0).sum().sort_values())
# normalize CPM log1p
cpm=mat.div(lib, axis=1)*1e6
log=np.log1p(cpm)
cor=log.corr(method='pearson')
print('pearson logCPM')
print(cor.round(3))
print('spearman raw')
print(mat.corr(method='spearman').round(3))
PY"
