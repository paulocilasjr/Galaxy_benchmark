/bin/bash -lc "python3 - <<'PY'
import requests,pandas as pd,numpy as np
fn='sample.swap.atac.q2.tsv.gz'
df=pd.read_csv(fn,sep='\\t',compression='gzip')
cols=list(df.columns[3:]); libs=df[cols].sum()
syms=['SLC6A9','GLRA2','PROX1','TFAP2A','TFAP2B','GAD1','GAD2','SLC6A1','SLC32A1','PAX6','CALB2','CHAT','TH','GRIK1','PCP2','GNB3','LHX1','ONECUT1','ONECUT2','RBPMS','RHO','ARR3','GFAP','RLBP1','PTPRC']
r=requests.post('https://mygene.info/v3/query',data={'q':','.join(syms),'scopes':'symbol,alias','fields':'symbol,genomic_pos_hg38,genomic_pos','species':'human'},timeout=60)
coord={}
for it in r.json():
 gp=it.get('genomic_pos_hg38') or it.get('genomic_pos')
 if not gp: continue
 if isinstance(gp,list):
  gps=[g for g in gp if str(g.get('chr')) in list(map(str,range(1,23)))+['X','Y']]
  gp=gps[0] if gps else gp[0]
 ch='chr'+str(gp['chr']).replace('chr','')
 if ch=='chrMT': ch='chrM'
 coord[it['query']]=(it.get('symbol'),ch,int(gp['start']),int(gp['end']),int(gp.get('strand',0) or 0))
for mode,pad in [('prom10',10000),('gene10',10000),('gene50',50000)]:
 print('\\n',mode)
 rows=[]
 for s in syms:
  if s not in coord: continue
  off,ch,st,en,strand=coord[s]
  if mode.startswith('prom'):
   tss=st if strand>=0 else en; wst=tss-pad; wen=tss+pad
  else: wst=st-pad; wen=en+pad
  sub=df[(df.chrom==ch)&(df.end>wst)&(df.start<wen)]
  if sub.empty: continue
  score=np.log1p(sub[cols].sum()/libs*1e6)
  top=score.sort_values(ascending=False).head(5)
  rows.append([s]+[f'{k}:{v:.2f}' for k,v in top.items()])
 for row in rows: print('\\t'.join(row))
PY"
