/bin/bash -lc 'set -e
for cell in AIIamacrine Microglia Retinalganglioncell Horizontal GABAamacrine Cone Mullerglia; do
  gz=GSM5866073_${cell}_frags.tsv.gz
  npy=${cell}_ref_counts.npy
  if [ -s "$npy" ]; then echo "$cell already done"; continue; fi
  echo "downloading $cell"
  curl -sS -L -o "$gz" "https://www.ncbi.nlm.nih.gov/geo/download/?acc=GSM5866073&format=file&file=${gz}"
  echo "processing $cell"
  python3 build_reference_counts.py "$gz" "$npy"
done
ls -lh *_ref_counts.npy'
