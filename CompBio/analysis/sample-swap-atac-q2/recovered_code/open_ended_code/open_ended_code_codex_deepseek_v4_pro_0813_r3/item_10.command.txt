/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
# load only counts
df = pd.read_csv('data.tsv', sep='\\t', usecols=lambda c: c not in ['chrom','start','end'])
# normalize CPM
cpm = df.div(df.sum(axis=0), axis=1)*1e6
print(df.sum())
print('\\nPearson correlation of CPM:')
print(cpm.corr().round(3).to_string())
print('\\nSpearman (sampled bins):')
samp = cpm.sample(n=min(50000,len(cpm)), random_state=0)
print(samp.corr(method='spearman').round(3).to_string())
PY"
