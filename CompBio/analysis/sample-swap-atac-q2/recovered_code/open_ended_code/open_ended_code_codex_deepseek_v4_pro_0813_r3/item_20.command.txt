/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
df=pd.read_csv('data.tsv',sep='\\t'); cols=list(df.columns[3:])
gt=pd.read_csv('gene_tss.tsv',sep='\\t'); bins=df[['chrom','start','end']].copy()
genes=['SLC6A9','GLRA1','GLRA2','GLRB','SLC32A1','GAD1','GAD2','SLC6A1','GJD2','GJA1','TFAP2A','TFAP2B','EBF3','PAX6','CALB2','SLC17A8','CHAT','SLC18A3','ONECUT1','ONECUT2','LHX1','CALB1','SLC12A7','DAB1','NEUROD1','NEUROD2','GRM7','RBPMS','POU4F1','SLC17A6']
res={}
for g in genes:
    r=gt[gt.gene==g]
    if r.empty: continue
    r=r.iloc[0]; s=r.tss-2000;e=r.tss+2000
    mask=(bins.chrom==r.chrom)&(bins.start<e)&(bins.end>s)
    res[g]=df.loc[mask,cols].sum(axis=0)
raw=pd.DataFrame(res,index=cols)
cpm=raw.div(df[cols].sum(axis=0),axis=0)*1e6
print(cpm.round(2).to_string())
PY"
