/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, gzip
# load matrix
df = pd.read_csv('data.tsv', sep='\\t')
cols = list(df.columns[3:])
gt = pd.read_csv('gene_tss.tsv', sep='\\t')
# protein coding only, unique gene names
gt = gt[gt.gene_type=='protein_coding'].drop_duplicates(subset='gene')
print('protein_coding', len(gt))
bins = df[['chrom','start','end']].copy()
# Map each gene to bin whose overlap with TSS +/- 2000 and choose max overlap or first
# Build interval index? simple loop over genes may be okay 20k; use dictionary by chrom bins arrays.
gene_bins = []
# Add bin midpoint from matrix index
bins_reset = bins.reset_index(drop=True)
for chrom, group in gt.groupby('chrom'):
    bchr = bins_reset[bins_reset.chrom==chrom].reset_index(drop=True)
    if bchr.empty: continue
    starts=bchr.start.values; ends=bchr.end.values
    for _,r in group.iterrows():
        s=r.tss-2000; e=r.tss+2000
        # find bins overlapping; choose max overlap
        ov = np.minimum(ends,e) - np.maximum(starts,s)
        idx = int(np.argmax(ov)) if len(ov) else -1
        if idx>=0 and ov[idx]>0:
            gene_bins.append((r.gene, bchr.index[idx]))
gdf = pd.DataFrame(gene_bins, columns=['gene','bin_idx'])
print('mapped genes', len(gdf))
# aggregate if multiple genes per bin? keep separate fine.
X = df.iloc[gdf.bin_idx.values][cols].values.astype(float)
sums = df[cols].sum(axis=0).values
cpm = X / sums * 1e6
gdf = gdf.copy()
for j,c in enumerate(cols): gdf[c] = cpm[:,j]
# specificity score: z-score across columns per gene, then top positive
z = gdf[cols].sub(gdf[cols].mean(axis=1), axis=0).div(gdf[cols].std(axis=1).replace(0,np.nan), axis=0)
gdf_z = gdf[['gene']].copy(); gdf_z[cols]=z
# top genes for each column by zscore (require raw CPM maybe > 5 and max specificity)
print('\\nTop 30 genes by specificity z-score for each column (promoter +/-2kb):')
for c in cols:
    sub=gdf_z.nlargest(30,c)[['gene',c]].copy()
    # include cpm value and rank
    sub['cpm']=gdf.loc[sub.index,c].values
    print('\\n---',c,'---')
    print(sub.to_string(index=False))
PY"
