/bin/bash -c "cat > predict_borzoi.py <<'PY'
import json, time, math
import numpy as np
import torch
from borzoi_pytorch import Borzoi

torch.set_num_threads(12)
# constants
chrom='chr1'
out_start=70157360
out_end=70353968
seq_start=out_start-163840
seq_end=seq_start+524288
print('seq region', chrom, seq_start, seq_end, 'out', out_start, out_end, 'len', seq_end-seq_start, flush=True)

# fetch sequence from UCSC
import urllib.request, urllib.parse
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=%s;start=%d;end=%d' % (chrom, seq_start, seq_end)
print('fetching sequence', flush=True)
t=time.time()
with urllib.request.urlopen(url, timeout=120) as r:
    data=json.loads(r.read().decode())
seq=data['dna']
print('fetched', len(seq), 'elapsed', time.time()-t, flush=True)
assert len(seq)==seq_end-seq_start
# one-hot Nx4xL
mapping={'A':0,'C':1,'G':2,'T':3}
x=np.zeros((1,4,len(seq)), dtype=np.float32)
for i,b in enumerate(seq):
    if b in mapping:
        x[0,mapping[b],i]=1.0
print('onehot done', x.shape, x.sum(), flush=True)

# load model (already cached likely)
print('loading model', flush=True)
t=time.time()
model=Borzoi.from_pretrained('johahi/borzoi-replicate-0')
model.eval()
print('loaded', time.time()-t, flush=True)

x=torch.from_numpy(x)
with torch.no_grad():
    print('forward start', flush=True)
    t=time.time()
    y=model(x, is_human=True)
    print('forward done', time.time()-t, 'shape', y.shape, flush=True)

np.save('borzoi_pred_rep0.npy', y.cpu().numpy().astype(np.float32))
print('saved', flush=True)
PY
python3 predict_borzoi.py"
