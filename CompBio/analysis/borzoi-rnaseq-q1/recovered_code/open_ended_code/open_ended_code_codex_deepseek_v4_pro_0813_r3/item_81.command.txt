/bin/bash -c "cat > predict_borzoi_upper.py <<'PY'
import time
import numpy as np
import torch
from borzoi_pytorch import Borzoi

torch.set_num_threads(1)
chrom='chr1'; out_start=70157360; out_end=70353968; seq_start=out_start-163840; seq_end=seq_start+524288
seq=open('std_region_seq.txt').read().upper()
print('seq len', len(seq), 'region', seq_start, seq_end, 'nonN', sum(c in 'ACGT' for c in seq), flush=True)
assert len(seq)==524288
mapping={'A':0,'C':1,'G':2,'T':3}
x=np.zeros((1,4,len(seq)), dtype=np.float32)
for i,b in enumerate(seq):
    if b in mapping:
        x[0,mapping[b],i]=1.0
print('onehot sum', x.sum(), flush=True)
print('loading model', flush=True)
model=Borzoi.from_pretrained('johahi/borzoi-replicate-0')
model.eval()
with torch.no_grad():
    print('forward start', flush=True)
    t=time.time()
    y=model(torch.from_numpy(x), is_human=True)
    print('forward done', time.time()-t, 'shape', y.shape, flush=True)
np.save('borzoi_pred_rep0_upper.npy', y.cpu().numpy().astype(np.float32))
print('saved', flush=True)
PY
OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 python3 predict_borzoi_upper.py"
