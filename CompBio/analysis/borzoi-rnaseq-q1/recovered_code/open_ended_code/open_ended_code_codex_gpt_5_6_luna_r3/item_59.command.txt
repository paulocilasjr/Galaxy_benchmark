/bin/bash -lc "python - <<'PY'
import numpy as np
f=np.load('encff281bwx_rep0_forward_raw.npy').astype(np.float64)
r=np.load('encff281bwx_rep0_rc_minus_raw.npy').astype(np.float64)[::-1]
avg=(f+r)/2
x=avg/0.3
x=np.where(x>384,(x-384)**2+384,x)
c=x**(4/3)
print('direct raw sum',f.sum(),'rc raw sum',r.sum(),'avg raw sum',avg.sum(),'avg counts',c.sum(),'rounded',round(c.sum()/10)*10)
# average counts instead
fc=f/0.3;fc=np.where(fc>384,(fc-384)**2+384,fc);fc=fc**(4/3)
rc=r/0.3;rc=np.where(rc>384,(rc-384)**2+384,rc);rc=rc**(4/3)
print('avg counts sum',(fc+rc).sum()/2,'round',round(((fc+rc).sum()/2)/10)*10)
print('correlation',np.corrcoef(f,r)[0,1], 'diff raw', np.abs(f-r).sum())
PY"
