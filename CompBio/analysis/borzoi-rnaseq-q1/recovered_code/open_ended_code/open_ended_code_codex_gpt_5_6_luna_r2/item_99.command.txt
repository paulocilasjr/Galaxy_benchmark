import gzip,collections
cnt=collections.Counter(); examples=[]
with gzip.open('borzoi_repo/data/sequences_human.bed.gz','rt') as f:

  if len(a)>=3:
   cnt[(int(a[1])%32,int(a[2])%32)]+=1
   if len(examples)<10: examples.append((a[0],int(a[1]),int(a[2]),int(a[1])%32,int(a[2])%32))
print(examples,cnt)
PY"
