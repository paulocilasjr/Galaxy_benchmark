/bin/bash -lc "python3 - <<'PY'
import requests, json
chrom='chr1'; start=69993520; end=70517808
url='https://api.genome.ucsc.edu/getData/sequence'
r=requests.get(url, params={'genome':'hg38','chrom':chrom,'start':start,'end':end}, timeout=60)
print(r.url)
print(r.status_code, len(r.text))
r.raise_for_status()
d=r.json()
seq=d.get('dna','')
print(len(seq), seq[:20], seq[-20:])
assert len(seq)==end-start
open('data/chr1_69993520_70517808.fa','w').write(f'>{chrom}:{start}-{end}\\n')
for i in range(0,len(seq),80): open('data/chr1_69993520_70517808.fa','a').write(seq[i:i+80].upper()+'\\n')
PY"
