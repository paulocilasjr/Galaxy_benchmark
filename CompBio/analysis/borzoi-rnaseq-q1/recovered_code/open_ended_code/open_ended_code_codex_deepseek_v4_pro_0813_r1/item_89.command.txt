/bin/bash -lc "python3 - <<'PY'
import pysam, numpy as np
fa=pysam.FastaFile('hg38/chr1.fa')
start=69993520; end=70517808
seq=fa.fetch('chr1',start,end).upper()
print(len(seq), seq[:20], seq[-20:])
# one-hot A,C,G,T
arr=np.zeros((len(seq),4), dtype=np.float32)
for i,b in enumerate(seq):
    if b=='A': arr[i,0]=1
    elif b=='C': arr[i,1]=1
    elif b=='G': arr[i,2]=1
    elif b=='T': arr[i,3]=1
    else:
        # assign N as zeros (will not sum to 1); okay
        pass
np.save('seq_onehot.npy', arr)
print('saved', arr.shape, arr.sum())
PY"
