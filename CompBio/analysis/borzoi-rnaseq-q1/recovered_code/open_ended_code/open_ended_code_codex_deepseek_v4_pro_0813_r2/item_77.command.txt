/bin/bash -lc "python3 - <<'PY'
import h5py, requests, os
url='https://huggingface.co/Genentech/borzoi-model/resolve/main/human_state_dict_rep0.h5'
path='[REDACTED LOCAL PATH]'
if not os.path.exists(path):
 print('downloading')
 with requests.get(url, stream=True, timeout=300) as r:
  r.raise_for_status()
  with open(path,'wb') as f:
   for chunk in r.iter_content(1024*1024): f.write(chunk)
 print('size', os.path.getsize(path))
try:
 f=h5py.File(path,'r')
 print('keys', list(f.keys())[:10], len(list(f.keys())))
 print('first key', list(f.keys())[0])
 for k in list(f.keys())[:3]:
  print(k, f[k].shape if hasattr(f[k],'shape') else type(f[k]))
except Exception as e:
 print('h5 error', e)
PY"
