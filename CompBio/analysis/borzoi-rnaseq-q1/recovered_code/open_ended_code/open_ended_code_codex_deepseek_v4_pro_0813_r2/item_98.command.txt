/bin/bash -lc "curl -L -s 'https://www.encodeproject.org/files/ENCFF281BWX/@@download/ENCFF281BWX.bigWig' -o ENCFF281BWX.bigWig && ls -lh ENCFF281BWX.bigWig && python3 - <<'PY'
import pyBigWig
bw=pyBigWig.open('ENCFF281BWX.bigWig')
print(bw.chroms('chr1'))
start=70157360; end=70353968
vals=bw.values('chr1', start, end)
import numpy as np
arr=np.array(vals,dtype=float)
print('n',len(arr),'sum',np.nansum(arr),'mean',np.nanmean(arr),'max',np.nanmax(arr),'nonnan',np.sum(~np.isnan(arr)))
# sum 32bp bins exactly maybe use stats with bins
bins=bw.stats('chr1',start,end,nBins=6144,type='sum')
print('stats n',len(bins),'sum',np.nansum(bins),'max',np.nanmax(bins),'mean',np.nanmean(bins))
PY"
