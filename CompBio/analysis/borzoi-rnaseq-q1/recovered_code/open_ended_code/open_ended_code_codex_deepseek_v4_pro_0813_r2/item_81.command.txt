/bin/bash -lc "python3 - <<'PY'
import numpy as np
z=np.load('track_6068_mps.npy').astype(np.float64)
scale=np.float64(0.3); cs=np.float64(384)
r=z/scale
r=np.where(r>cs, cs+(r-cs)**2, r)
y=r**(np.float64(4)/3)
print('sum',y.sum())
print('round10', round(float(y.sum())/10)*10)
# compare untransform_preds1 default (scale kept)
print('scaled sum', (y*scale).sum(), round(float((y*scale).sum())/10)*10)
# no scale paper
z2=z.copy(); y2=np.where(z2<=cs, z2**(4/3), (cs+(z2-cs)**2)**(4/3))
print('no scale', y2.sum(), round(float(y2.sum())/10)*10)
PY"
