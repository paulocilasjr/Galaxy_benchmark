/bin/bash -lc "python3 - <<'PY'
import urllib.request
for url in [
'https://huggingface.co/datasets/Genentech/compbiobench-results-v1',
'https://huggingface.co/datasets/Genentech/compbiobench-results-v1/tree/main',
'https://huggingface.co/datasets/Genentech/compbiobench-results-v1/resolve/main/results.tsv',
'https://huggingface.co/datasets/Genentech/compbiobench-results-v1/resolve/main/all_results.tsv',
'https://huggingface.co/datasets/Genentech/compbiobench-submissions-v1/resolve/main/submissions.tsv']:
    print('URL',url)
    try:
        req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
        data=urllib.request.urlopen(req, timeout=20).read().decode()
        print(data[:1000])
        print('len',len(data))
    except Exception as e:
        print('ERR',e)
PY"
