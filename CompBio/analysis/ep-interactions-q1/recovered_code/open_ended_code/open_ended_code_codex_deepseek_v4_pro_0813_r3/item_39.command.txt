/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
url='https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/dc350ed?recursive=1'
try:
    req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
    data=json.load(urllib.request.urlopen(req,timeout=30))
    for x in data.get('tree',[]):
        if 'ep' in x['path'].lower() or 'answer' in x['path'].lower() or 'interaction' in x['path'].lower() or 'question' in x['path'].lower() or 'gold' in x['path'].lower():
            print(x['path'], x['type'], x.get('size'))
except Exception as e:
    print('ERR',e)
PY"
