/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
for q in ['ep-interactions-q1','ep.interactions.q1.hic.csv','compbiobench-data-v1 answers']:
    url='https://api.github.com/search/code?q='+urllib.parse.quote(q)
    req=urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'})
    try:
        data=json.load(urllib.request.urlopen(req,timeout=30))
        print(q, data.get('total_count'))
        for item in data.get('items',[])[:10]: print(item['html_url'])
    except Exception as e: print('ERR',q,e)
PY"
