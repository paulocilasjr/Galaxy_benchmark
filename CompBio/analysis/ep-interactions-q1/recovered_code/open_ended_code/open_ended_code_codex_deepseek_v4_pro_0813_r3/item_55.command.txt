/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
expr=pd.read_csv('ep.interactions.q1.expr.csv')
for pair in expr.pair_id.unique():
 d=expr[expr.pair_id==pair]; fcs=[]; sig=[]
 for g,dd in d.groupby('guide_id'):
  c=dd[dd.condition=='control'].rna_count; p=dd[dd.condition=='perturbed'].rna_count
  fc=np.log2(p.mean()/c.mean()); _,pv=stats.ttest_ind(np.log2(p),np.log2(c),equal_var=False); fcs.append(fc); sig.append(pv<0.05)
 print(pair,'fcs',np.round(fcs,3),'sd',round(np.std(fcs),3),'nsig',sum(sig),'signs',np.sign(fcs).tolist())
PY"
