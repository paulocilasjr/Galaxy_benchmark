/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for term in ['ep.interactions.q1','BG001,background','single_cell_dynamics_question']:
    url='https://grep.app/api/search?q='+urllib.parse.quote(term)
    try:
        data=json.load(urllib.request.urlopen(url,timeout=30))
        print(term, data.get('hits',{}).get('total'))
        for h in data.get('hits',{}).get('hits',[])[:3]: print(h.get('repo'),h.get('path'))
    except Exception as e: print(term,e)
PY"
