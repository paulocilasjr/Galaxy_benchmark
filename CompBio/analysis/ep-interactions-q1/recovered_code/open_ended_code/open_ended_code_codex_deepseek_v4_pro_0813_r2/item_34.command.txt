/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
hic=pd.read_csv('ep.interactions.q1.hic.csv'); expr=pd.read_csv('ep.interactions.q1.expr.csv')
cand=hic[hic.set=='candidate'].set_index('pair_id')
rows=[]
for p,g in expr.groupby('pair_id'):
    c=g[g.condition=='control'].rna_count; t=g[g.condition=='perturbed'].rna_count
    logfc=np.log2(t.mean()/c.mean()); pv=stats.ttest_ind(c,t,equal_var=False).pvalue
    rows.append((p,cand.loc[p,'mean_count'] if 'mean_count' in cand else cand.loc[p,['count_rep1','count_rep2','count_rep3']].mean(),logfc,abs(logfc),pv))
df=pd.DataFrame(rows,columns=['pair','hic','logFC','abslogFC','p'])
print(df)
print('corr hic vs logFC', df.hic.corr(df.logFC))
print('corr hic vs abslogFC', df.hic.corr(df.abslogFC))
print('corr hic vs -log10p', df.hic.corr(-np.log10(df.p)))
PY"
