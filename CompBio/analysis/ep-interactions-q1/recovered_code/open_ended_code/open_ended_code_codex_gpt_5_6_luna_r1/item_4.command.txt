/bin/bash -lc "python - <<'PY'
import csv, statistics, collections, math
with open('ep.interactions.q1.hic.csv',newline='') as f: h=list(csv.DictReader(f))
print('HIC candidates:')
for r in h:
    if r['set']=='candidate':
        x=[int(r[f'count_rep{i}']) for i in range(1,4)]
        print(r['pair_id'], 'dist',r['distance_bp'], 'counts',x,'mean',sum(x)/3)

with open('ep.interactions.q1.expr.csv',newline='') as f: e=list(csv.DictReader(f))
print('\\nExpression:')
for p in sorted(set(r['pair_id'] for r in e), key=lambda x:int(x[2:])):
    byg={}
    for r in e:
        if r['pair_id']==p:
            byg.setdefault(r['guide_id'],{}).setdefault(r['condition'],[]).append(int(r['rna_count']))
    print('\\n',p)
    allc=[]; allp=[]
    for g,d in byg.items():
        c=d['control']; q=d['perturbed']
        cm=sum(c)/len(c); pm=sum(q)/len(q)
        # log2 fold perturbed/control, percent decrease
        l=math.log2(pm/cm)
        print(g, 'C',c,'P',q,'means',round(cm,2),round(pm,2),'P/C',round(pm/cm,3),'log2',round(l,3),'delta',round(pm-cm,2))
        allc += c; allp += q
    cm=sum(allc)/len(allc);pm=sum(allp)/len(allp)
    print('aggregate means',cm,pm,'P/C',pm/cm,'log2',math.log2(pm/cm),'delta',pm-cm)
PY"
