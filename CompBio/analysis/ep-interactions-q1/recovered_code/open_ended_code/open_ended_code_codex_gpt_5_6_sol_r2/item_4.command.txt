/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
from scipy import stats
import statsmodels.api as sm
h=pd.read_csv('ep.interactions.q1.hic.csv')
h['mean']=h.filter(like='count').mean(axis=1)
b=h[h.set=='background']; c=h[h.set=='candidate'].copy()
# log-log OLS background mean + 0.5
X=sm.add_constant(np.log(b.distance_bp)); m=sm.OLS(np.log(b['mean']+.5),X).fit()
c['expected_log']=m.predict(sm.add_constant(np.log(c.distance_bp)))
c['resid']=np.log(c['mean']+.5)-c.expected_log
# empirical local percent: backgrounds within +-200kb / distance fold 1.5
for _,r in c.iterrows():
 local=b[(b.distance_bp>=r.distance_bp/1.5)&(b.distance_bp<=r.distance_bp*1.5)]
 print(r.pair_id,'dist',r.distance_bp,'mean',r['mean'],'exp',round(np.exp(r.expected_log)-.5,2),'resid',round(r.resid,2),'localn',len(local),'percentile',round(stats.percentileofscore(local['mean'],r['mean']),1),'local median',local['mean'].median())
print(m.summary())
PY"
