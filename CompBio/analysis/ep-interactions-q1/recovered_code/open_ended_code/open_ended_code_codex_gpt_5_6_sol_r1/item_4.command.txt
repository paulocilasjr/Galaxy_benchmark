/bin/bash -lc "python3 - <<'PY'
import csv, collections, math
p=collections.defaultdict(lambda: collections.defaultdict(lambda: collections.defaultdict(list)))
with open('ep.interactions.q1.expr.csv') as f:
 for x in csv.DictReader(f): p[x['pair_id']][x['guide_id']][x['condition']].append(int(x['rna_count']))
print('Expression guide-level pert/control ratios:')
for pair in sorted(p):
 vals=[]
 for g in sorted(p[pair]):
  c=sum(p[pair][g]['control'])/len(p[pair][g]['control']); t=sum(p[pair][g]['perturbed'])/len(p[pair][g]['perturbed']); vals.append(t/c)
 print(pair, ' '.join(f'{x:.3f}' for x in vals), 'median',f'{sorted(vals)[len(vals)//2-1:len(vals)//2+1][0] if False else 0}')
print('\\nHiC candidate mean and nearby-background empirical percentile:')
rows=[]
with open('ep.interactions.q1.hic.csv') as f:
 for x in csv.DictReader(f):
  rows.append((x['pair_id'],x['set'],int(x['distance_bp']),sum(int(x[k]) for k in ('count_rep1','count_rep2','count_rep3'))/3))
bg=[r for r in rows if r[1]=='background']
for pid,_,d,m in [r for r in rows if r[1]=='candidate']:
 near=[x[3] for x in bg if abs(math.log(x[2]/d))<.25]
 pct=sum(v<=m for v in near)/len(near)
 print(pid,d,f'{m:.2f}',len(near),f'pct={pct:.3f}',f'near_med={sorted(near)[len(near)//2]:.2f}')
PY"
