/bin/bash -lc "python3 - <<'PY'
import csv, statistics, math
from collections import defaultdict
try:
 import scipy.stats as stats
except Exception as e:
 print('no scipy',e); stats=None
rows=list(csv.DictReader(open('ep.interactions.q1.expr.csv')))
by=defaultdict(lambda:{'control':[],'perturbed':[]})
for r in rows:
 by[r['pair_id']][r['condition']].append(int(r['rna_count']))
for ep in sorted(by):
 c=by[ep]['control']; p=by[ep]['perturbed']
 # Welch t p<c?
 if stats:
  tt=stats.ttest_ind(p,c,equal_var=False)
  mw=stats.mannwhitneyu(p,c,alternative='two-sided')
  print(ep,'meanC',statistics.mean(c),'meanP',statistics.mean(p),'ratio',statistics.mean(p)/statistics.mean(c),'diff',statistics.mean(p)-statistics.mean(c),'t',tt.statistic,'p',tt.pvalue,'mw',mw.pvalue)
PY"
