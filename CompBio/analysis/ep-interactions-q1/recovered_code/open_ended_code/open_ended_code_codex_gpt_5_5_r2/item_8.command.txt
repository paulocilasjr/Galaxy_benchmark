/bin/bash -lc "python3 -S - <<'PY'
import csv, math, statistics as st
hic=[]
with open('ep.interactions.q1.hic.csv') as f:
 for r in csv.DictReader(f):
  counts=[int(r['count_rep1']),int(r['count_rep2']),int(r['count_rep3'])]
  r['distance_bp']=int(r['distance_bp']); r['counts']=counts; r['mean_count']=sum(counts)/3
  hic.append(r)
print('Candidates HIC')
for r in hic:
 if r['set']=='candidate': print(r['pair_id'], r['distance_bp'], r['counts'], round(r['mean_count'],3))
expr=[]
with open('ep.interactions.q1.expr.csv') as f:
 for r in csv.DictReader(f):
  r['rna_count']=int(r['rna_count'])
  expr.append(r)
print('\\nExpr summary')
pids=sorted({r['pair_id'] for r in expr})
for pid in pids:
 c=[r['rna_count'] for r in expr if r['pair_id']==pid and r['condition']=='control']
 p=[r['rna_count'] for r in expr if r['pair_id']==pid and r['condition']=='perturbed']
 print(pid, 'ctrl',round(sum(c)/len(c),1),'pert',round(sum(p)/len(p),1),'log2fc',round(math.log2((sum(p)/len(p)+1)/(sum(c)/len(c)+1)),3),'diff',round(sum(p)/len(p)-sum(c)/len(c),1))
print('\\nPer guide')
for pid in pids:
 print(pid)
 guides=sorted({r['guide_id'] for r in expr if r['pair_id']==pid})
 for guide in guides:
  c=[r['rna_count'] for r in expr if r['pair_id']==pid and r['guide_id']==guide and r['condition']=='control']
  p=[r['rna_count'] for r in expr if r['pair_id']==pid and r['guide_id']==guide and r['condition']=='perturbed']
  print(' ',guide,'log2fc',round(math.log2((sum(p)/len(p)+1)/(sum(c)/len(c)+1)),3),'c',round(sum(c)/len(c),1),'p',round(sum(p)/len(p),1))
PY"
