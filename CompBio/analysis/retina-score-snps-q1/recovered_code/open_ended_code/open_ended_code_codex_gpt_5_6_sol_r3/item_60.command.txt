/bin/bash -lc "python3 - <<'PY'
import pandas as pd
from sklearn.metrics import average_precision_score,auc,precision_recall_curve
x=pd.read_csv('rod_scores.tsv',sep='\\t')
p,r,_=precision_recall_curve(x.label,x.abs_Rod_fold_avg_lfc)
print('AP',average_precision_score(x.label,x.abs_Rod_fold_avg_lfc))
print('AUC',auc(r,p))
print('floor AP',int(average_precision_score(x.label,x.abs_Rod_fold_avg_lfc)*100)/100)
print('round AP',round(average_precision_score(x.label,x.abs_Rod_fold_avg_lfc),2))
PY"
