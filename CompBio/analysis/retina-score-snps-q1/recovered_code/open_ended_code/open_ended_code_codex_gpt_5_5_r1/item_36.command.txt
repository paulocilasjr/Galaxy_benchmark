/bin/bash -lc "python3 - <<'PY'
import csv,json
from pathlib import Path
INP=2114; CENTER=INP//2-1
rows=list(csv.DictReader(open('retina.score.snps.q1.tsv'), delimiter='\\t'))
for off in range(-3,4):
    mr=ma=both=0
    for row in rows:
        chrom=row['chr']; pos=int(row['start']); start=pos-INP//2; end=pos+INP//2
        seq=json.load(open(Path('cache/ucsc_hg19')/f'{chrom}_{start}_{end}.json'))['dna'].upper()
        b=seq[CENTER+off]
        if b==row['ref'].upper(): mr+=1
        if b==row['alt'].upper(): ma+=1
        if b in (row['ref'].upper(), row['alt'].upper()): both+=1
    print(off,'refmatch',mr,'altmatch',ma,'either',both)
PY"
