/bin/bash -lc "python - <<'PY'
import csv, urllib.request, json, time
rows=list(csv.DictReader(open('retina.score.snps.q1.tsv'), delimiter='\\t'))[:50]
for delta in [-2,-1,0,1,2]:
    m=0
    vals=[]
    for r in rows:
        pos=int(r['start'])+delta
        url=f\"https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom={r['chr']};start={pos};end={pos+1}\"
        with urllib.request.urlopen(url, timeout=20) as resp:
            seq=json.load(resp).get('dna','').upper()
        vals.append(seq)
        if seq==r['ref']: m+=1
    print(delta, m, vals[:10], [r['ref'] for r in rows[:10]])
PY"
