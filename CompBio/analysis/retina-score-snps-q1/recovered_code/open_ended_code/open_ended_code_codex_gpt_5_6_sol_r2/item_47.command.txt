/bin/bash -lc "tf_arm_env/bin/python - <<'PY'
import math
import numpy as np
import pandas as pd
from sklearn.metrics import average_precision_score
x=pd.read_csv('retina.score.snps.q1.rod_scored.tsv',sep='\\t')
y=x.label.to_numpy(); s=np.abs(x.Rod_fold_avg_lfc.to_numpy())
order=np.argsort(-s,kind='stable'); yo=y[order]
manual=(np.cumsum(yo)/(np.arange(len(y))+1)*yo).sum()/yo.sum()
lib=average_precision_score(y,s)
print(len(x),y.sum(),np.isfinite(s).all(),manual,lib,manual-lib,math.floor(lib*100)/100)
vals,counts=np.unique(s,return_counts=True)
print('ties',sum(counts>1),'max multiplicity',counts.max(),'zero',sum(s==0))
print('mean stored-vs-foldmean maxdiff',np.max(np.abs(x.Rod_fold_avg_lfc.to_numpy()-x[[f'Rod_fold{i}_lfc' for i in range(5)]].mean(axis=1).to_numpy())))
PY"
