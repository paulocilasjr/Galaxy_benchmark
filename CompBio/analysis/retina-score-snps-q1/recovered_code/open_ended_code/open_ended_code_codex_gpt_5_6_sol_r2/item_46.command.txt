/bin/bash -lc "tf_arm_env/bin/python - <<'PY'
import math
import numpy as np
import pandas as pd
from sklearn.metrics import average_precision_score
x=pd.read_csv('retina.score.snps.q1.rod_scored.tsv',sep='\\t')
y=x.label.to_numpy()
s=np.abs(x.Rod_fold_avg_lfc.to_numpy())
order=np.argsort(-s,kind='stable')
yo=y[order]
manual=(np.cumsum(yo)/(np.arange(len(y))+1)*yo).sum()/yo.sum()
lib=average_precision_score(y,s)
assert len(x)==500 and y.sum()==250 and np.isfinite(s).all()
assert abs(manual-lib)<1e-15
assert math.floor(lib*100)/100==0.67
print(f'rows={len(x)} positives={y.sum()} sklearn={lib:.15f} manual={manual:.15f} floor={math.floor(lib*100)/100:.2f}')
print('fold columns', [c for c in x if c.startswith('Rod_fold')])
PY"
