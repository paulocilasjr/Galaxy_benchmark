/bin/bash -lc "python -u - <<'PY'
import pandas as pd, json, urllib.request
x=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
c=[0,0,0]; examples=[]
for j,r in x.iterrows():
 p=int(r.start)
 url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom={r.chr};start={p-1};end={p+1}'
 try: d=json.load(urllib.request.urlopen(url,timeout=20)); dna=d['dna'].upper()
 except Exception as e: print('ERR',j,e); break
 m0=dna[1]==r.ref.upper(); m1=dna[0]==r.ref.upper()
 c[0 if m0 else 1 if m1 else 2]+=1
 if len(examples)<12 and not m0: examples.append((j,r.chr,p,r.ref,dna,m1))
print('ref match p',c[0],'p-1 only',c[1],'neither',c[2]); print(examples)
PY"
