/bin/bash -lc "mkdir -p zenodo/models/fold0 zenodo/models/fold1 zenodo/models/fold2 zenodo/models/fold3 zenodo/models/fold4 && python - <<'PY'
from pathlib import Path
import zlib, binascii
for fold in ['fold0','fold1','fold2','fold3','fold4']:
 src=Path('zenodo')/(fold+'_Rod.deflate.full')
 dst=Path('zenodo/models')/fold/'Rod.h5'
 data=src.read_bytes()
 raw=zlib.decompress(data,-15)
 dst.write_bytes(raw)
 print(fold,'compressed',len(data),'raw',len(raw),'crc32',format(binascii.crc32(raw)&0xffffffff,'08x'))
PY
python - <<'PY'
try:
 import h5py
 print('h5py',h5py.__version__)
 for f in ['zenodo/models/fold0/Rod.h5']:
  with h5py.File(f,'r') as h:
   print('keys',list(h.keys()))
except Exception as e: print(type(e).__name__,e)
PY"
