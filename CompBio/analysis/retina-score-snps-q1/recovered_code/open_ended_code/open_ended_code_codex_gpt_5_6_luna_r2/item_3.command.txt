/bin/bash -lc "mkdir -p zenodo && curl -L --fail --silent --show-error https://zenodo.org/api/records/6330053 -o zenodo/record_6330053.json && curl -L --fail --silent --show-error https://zenodo.org/api/records/6796067 -o zenodo/record_6796067.json && echo '--- 6330053 files ---' && python - <<'PY'

for fn in ['zenodo/record_6330053.json','zenodo/record_6796067.json']:

PY"
