/bin/bash -lc "set -o pipefail
curl -L --fail --silent --show-error https://zenodo.org/api/records/6330053 -o zenodo_6330053.json
curl -L --fail --silent --show-error https://zenodo.org/api/records/6796067 -o zenodo_6796067.json
python - <<'PY'
import json
for fn in ['zenodo_6330053.json','zenodo_6796067.json']:
    print('\\n---',fn,'---')
    d=json.load(open(fn))
    print('title:',d.get('metadata',{}).get('title'))
    for f in d.get('files',[]):
        print(f.get('key'), f.get('size'), f.get('checksum'), f.get('links',{}).get('self'), f.get('links',{}).get('content'))
PY"
