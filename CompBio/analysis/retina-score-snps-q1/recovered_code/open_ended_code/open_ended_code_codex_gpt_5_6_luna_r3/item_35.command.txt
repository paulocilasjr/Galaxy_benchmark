/bin/bash -lc "python -u - <<'PY'
import pandas as pd, urllib.request, urllib.parse, json
print('imports',flush=True)
x=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
for i,r in x.head(3).iterrows():
 print('row',i,flush=True)
 for genome in ['hg19','hg38']:
  for mode,st in [('0based',int(r.start)),('1based',int(r.start)-1)]:
   url='https://api.genome.ucsc.edu/getData/sequence?'+urllib.parse.urlencode({'genome':genome,'chrom':r.chr,'start':st,'end':st+1})
   try:
    z=json.load(urllib.request.urlopen(url,timeout=30))['dna']
    print(genome,mode,z,flush=True)
   except Exception as e: print('ERR',e,flush=True)
PY"
