/bin/bash -lc "python - <<'PY'
import struct
b=open('model_zip_tail.bin','rb').read()
tail_start=4172833344-len(b)
cd_off_abs=4172816903
cd_size=16419
p=cd_off_abs-tail_start
end=p+cd_size
print('tail start',tail_start,'cd relative',p,'end',end,'len',len(b))
entries=[]
while p<end:
    if b[p:p+4]"'!=b'"'PK\\x01\\x02':
        print('bad signature at',p,b[p:p+16]);break
    # central dir fixed 46: signature + 42
    vals=struct.unpack_from('<4s6H3L5H2L',b,p)
    (_,ver_made,ver_need,flag,comp,mtime,mdate,crc,csize,usize,nlen,xlen,clen,disk,intattr,extattr,loff)=vals
    name=b[p+46:p+46+nlen].decode('utf-8',errors='replace')
    entries.append((name,usize,csize,comp,loff,flag))
    p += 46+nlen+xlen+clen
print('parsed',len(entries),'p-end',p-end)
for e in entries:
    print('\\t'.join(map(str,e)))
PY"
