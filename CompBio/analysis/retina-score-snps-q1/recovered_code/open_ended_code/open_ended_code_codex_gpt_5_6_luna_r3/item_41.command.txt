/bin/bash -lc "curl -L --fail --silent --show-error --retry 3 https://hgdownload.soe.ucsc.edu/goldenPath/hg19/bigZips/hg19.fa.gz -o hg19.fa.gz
wc -c hg19.fa.gz
gzip -t hg19.fa.gz
gzip -dc hg19.fa.gz > hg19.fa
ls -lh hg19.fa.gz hg19.fa
python - <<'PY'
import pyfaidx
f=pyfaidx.Fasta('hg19.fa')
print('keys',len(f),list(f.keys())[:5],list(f.keys())[-5:])
print('chr1',len(f['chr1']))
PY"
