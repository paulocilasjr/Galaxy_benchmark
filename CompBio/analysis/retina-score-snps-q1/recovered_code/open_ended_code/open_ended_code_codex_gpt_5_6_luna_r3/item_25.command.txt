/bin/bash -lc "python - <<'PY'
import zlib,glob
for fn in sorted(glob.glob('*_Rod.deflate')):
 b=open(fn,'rb').read()
 try:
  d=zlib.decompressobj(-15)
  out=d.decompress(b)
  print(fn,'compressed',len(b),'out',len(out),'eof',d.eof,'unused',len(d.unused_data),'unconsumed',len(d.unconsumed_tail))
 except Exception as e: print(fn,'ERROR',type(e).__name__,e)
PY"
