/bin/bash -c "python - <<'PY'
from pyliftover import LiftOver
import pandas as pd
lo=LiftOver('ref/hg19ToHg38.over.chain.gz')
df=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
from collections import Counter
strands=[]
multi=[]
for i,r in df.iterrows():
    res=lo.convert_coordinate(r['chr'],r['start']-1)
    if not res:
        strands.append('none'); continue
    ss=[x[2] for x in res]
    strands.append(ss[0])
    if len(res)>1: multi.append((i,res))
print(Counter(strands))
print('multi',len(multi),multi[:5])
PY"
