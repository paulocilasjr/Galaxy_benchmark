/bin/bash -c "python - <<'PY'
import gzip
qpos=51540904
path='ref/hg19ToHg38.over.chain.gz'
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        if p[7]=='chr10' and int(p[10])<=qpos<=int(p[11]):
            header=p; blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            break
tStart=int(header[5]); qStart=int(header[10])
t=tStart; q=qStart
for b in blocks:
    parts=b.split()
    if len(parts)==3: size,dt,dq=map(int,parts)
    else: size=int(parts[0]); dt=dq=0
    if q <= qpos < q+size:
        off=qpos-q
        print('found',b,'t',t,'q',q,'size',size,'off',off,'mapped t0',t+off,'t1',t+off+1)
        break
    t += size+dt; q += size+dq
PY"
