/bin/bash -c "python - <<'PY'
import torch,h5py,numpy as np,time
f=h5py.File('work/Rod_fold0.h5','r')
def ds(p): return torch.from_numpy(np.array(f[p]))
w1=ds('model_weights/conv1d/conv1d/kernel:0').permute(2,1,0); b1=ds('model_weights/conv1d/conv1d/bias:0')
ws=[]
for i in range(1,9): ws.append((ds(f'model_weights/conv1d_{i}/conv1d_{i}/kernel:0').permute(2,1,0),ds(f'model_weights/conv1d_{i}/conv1d_{i}/bias:0')))
wct=ds('model_weights/logcounts/logcounts/kernel:0').reshape(1,512,1); bct=ds('model_weights/logcounts/logcounts/bias:0').reshape(1)
f.close()
@torch.no_grad()
def forward(x):
    out=torch.relu(torch.nn.functional.conv1d(x,w1,b1))
    for i,(w,b) in enumerate(ws,1):
        y=torch.relu(torch.nn.functional.conv1d(out,w,b,dilation=2**i))
        xlen=out.shape[2]; ylen=y.shape[2]; crop=(xlen-ylen)//2
        out=y+out[:,:,crop:crop+ylen]
    avg=out.mean(dim=2)
    return torch.nn.functional.linear(avg,wct.reshape(1,512),bct)
x=torch.zeros(1,4,2114); x[:,:,0]=1
t=time.time(); y=forward(x); print(y, 'time',time.time()-t)
x64=torch.zeros(64,4,2114); x64[:,:,0]=1
t=time.time(); y=forward(x64); print(y[:3], 'batch64 time',time.time()-t)
PY"
