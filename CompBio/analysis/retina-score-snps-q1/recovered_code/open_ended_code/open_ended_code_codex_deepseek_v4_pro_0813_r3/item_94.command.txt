/bin/bash -c "python - <<'PY'
import pandas as pd, urllib.request,json,concurrent.futures
from pyliftover import LiftOver
lo=LiftOver('ref/hg19ToHg38.over.chain.gz')
df=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
def rc(s): return s.upper().translate(str.maketrans('ACGT','TGCA'))[::-1]
# compute lifted target candidates for each; for strand '-' use p0-1
rows=[]
for i,r in df.iterrows():
    res=lo.convert_coordinate(r['chr'], r['start'])
    if not res:
        rows.append((i,'none',None,'+',r['ref'],r['alt'])); continue
    c,p0,strand,score=res[0]
    if strand=='-':
        p=p0-1
        ref=rc(r['ref']); alt=rc(r['alt'])
    else:
        p=p0
        ref=r['ref'].upper(); alt=r['alt'].upper()
    rows.append((i,strand,p,strand,ref,alt))
# verify base at p (0-based) for mapped
def check(row):
    i,kind,p,strand,ref,alt=row
    if p is None: return i,kind,p,None,ref
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={df.iloc[i]['chr']};start={p};end={p+1}'
    try: b=json.load(urllib.request.urlopen(url))['dna'].upper()
    except Exception as e: b='ERR'
    return i,kind,p,b,ref
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:
    results=list(ex.map(check,rows))
mism=[]
for i,kind,p,b,ref in results:
    if b is None: continue
    if b=='ERR' or (ref"'!='"'N' and b"'!=ref):
        mism.append((i,kind,p,b,ref,df.iloc[i].to_dict()))
print('"'num mism',len(mism))
for x in mism[:20]: print(x)
# count strands
from collections import Counter
print(Counter(k for _,k,_,_,_ in rows))
PY"
