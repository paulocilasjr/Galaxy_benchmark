/bin/bash -c "python - <<'PY'
import urllib.request, os, zlib, struct, concurrent.futures, time
url='https://zenodo.org/api/records/6330053/files/retina_models.zip/content'
rods={0:(3978450820,63984526,'work/Rod_fold0.h5'),1:(3146890825,64978361,'work/Rod_fold1.h5'),2:(639001594,64329784,'work/Rod_fold2.h5'),3:(2314589870,64246455,'work/Rod_fold3.h5'),4:(1473529213,64976922,'work/Rod_fold4.h5')}
def download(item):
    f,(start,csize,path)=item
    req=urllib.request.Request(url, headers={'Range':f'bytes={start}-{start+csize-1}'})
    t=time.time()
    data=urllib.request.urlopen(req).read()
    print('downloaded',f,len(data),'in',time.time()-t)
    # decompress deflate
    # HDF5 file itself starts with signature, zlib decompressobj raw
    dobj=zlib.decompressobj(-15)
    out=dobj.decompress(data)+dobj.flush()
    print('decompressed',f,len(out), 'expected 76662176')
    open(path,'wb').write(out)
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as ex:
    list(ex.map(download, rods.items()))
print('done')
for p in ['work/Rod_fold0.h5','work/Rod_fold1.h5','work/Rod_fold2.h5','work/Rod_fold3.h5','work/Rod_fold4.h5']:
    print(p, os.path.getsize(p))
PY"
