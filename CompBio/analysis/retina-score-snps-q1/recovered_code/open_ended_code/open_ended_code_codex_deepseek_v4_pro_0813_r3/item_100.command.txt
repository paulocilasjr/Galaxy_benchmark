/bin/bash -c "python - <<'PY'
import gzip
pos=110222992
path='ref/hg19ToHg38.over.chain.gz'
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        if p[2]=='chr2' and int(p[5])<=pos<=int(p[6]):
            header=p; blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            break
print('header',header)
s=int(header[5]); t=int(header[10])
prev=None
for j,b in enumerate(blocks):
    parts=b.split()
    if len(parts)==3: size,sgap,tgap=map(int,parts)
    else: size=int(parts[0]); sgap=tgap=0
    if s<=pos<s+size:
        off=pos-s
        print('contains block',j,b,'s',s,'t',t,'size',size,'off',off,'tp',t+off)
        break
    if s>pos:
        print('first after',j,b,'s',s,'t',t,'prev',prev)
        break
    prev=(j,b,s,t,size)
    s += size+sgap; t += size+tgap
else:
    print('ended after blocks s,t',s,t)
PY"
