/bin/bash -c "python - <<'PY'
import pandas as pd
from pyliftover import LiftOver
lo=LiftOver('ref/hg19ToHg38.over.chain.gz')
df=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
print(df.shape, df.columns)
# Convert 0-based positions? hg19 coords in file are 1-based.
rows=[]
for i,r in df.iterrows():
    res=lo.convert_coordinate(r['chr'], r['start']-1) # pyliftover uses 0-based start? likely.
    if res:
        chrom,pos,strand,_=res[0]
        rows.append((r['chr'],r['start'],chrom,pos+1,r['ref'],r['alt'],r['label']))
    else:
        rows.append((r['chr'],r['start'],None,None,r['ref'],r['alt'],r['label']))
print('unmapped',sum(1 for x in rows if x[2] is None))
print(rows[:10])
# save
out=pd.DataFrame(rows,columns=['chr19','pos19','chr38','pos38','ref','alt','label'])
out.to_csv('work/snps_lifted.tsv',sep='\\t',index=False)
print(out.head())
PY"
