/bin/bash -c "python - <<'PY'
from pyfaidx import Fasta
# create small fasta in memory via temp file
import tempfile, os
fd,path=tempfile.mkstemp(suffix='.fa'); os.close(fd)
open(path,'w').write('>chr1\\nACGTACGTAC\\n')
f=Fasta(path)
r=f['chr1']
for args in [(0,5),(1,5),(0,1),(1,1),(1,2)]:
    try: print(args, r[args[0]:args[1]])
    except Exception as e: print(args,e)
print('len',len(r))
PY"
