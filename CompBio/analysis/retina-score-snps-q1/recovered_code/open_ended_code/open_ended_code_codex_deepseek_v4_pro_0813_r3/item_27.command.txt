/bin/bash -c "python - <<'PY'
import urllib.request, json, pandas as pd
out=pd.read_csv('work/snps_lifted.tsv',sep='\\t')
for idx in [0,1,2,3,4]:
    r=out.iloc[idx]
    for build,chrom,pos in [('hg19',r['chr19'],r['pos19']),('hg38',r['chr38'],int(r['pos38']))]:
        if chrom is None or pd.isna(pos): continue
        url=f'https://api.genome.ucsc.edu/getData/sequence?genome={build};chrom={chrom};start={pos-2};end={pos+1}'
        try:
            d=json.load(urllib.request.urlopen(url))['dna']
        except Exception as e:
            d='ERR'
        print(idx,build,chrom,pos,'dna',d,'ref',r['ref'])
PY"
