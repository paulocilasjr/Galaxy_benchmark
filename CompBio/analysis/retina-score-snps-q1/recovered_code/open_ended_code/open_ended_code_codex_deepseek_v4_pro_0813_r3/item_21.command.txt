import urllib.request, json, time
sites=[('chr1',20014826),('chr1',20015797),('chr1',21097399)]

    for chrom,pos in sites[:1]:

        print(build,url)
        try:
            data=json.load(urllib.request.urlopen(url))

        except Exception as e:
            print('ERR',e)
