/bin/bash -c "python - <<'PY'
import urllib.request, struct, os
url='https://zenodo.org/api/records/6330053/files/retina_models.zip/content'
size=4172833344
# download last 2MB
tail=2*1024*1024
req=urllib.request.Request(url, headers={'Range':f'bytes={size-tail}-{size-1}'})
data=urllib.request.urlopen(req).read()
open('work/zip_tail.bin','wb').write(data)
print(len(data))
# find EOCD signature
idx=data.rfind(b'PK\\x05\\x06')
print('eocd idx',idx)
eocd=data[idx:idx+22]
print(eocd.hex())
fields=struct.unpack('<4s4H2LH', eocd)
print(fields)
cd_size, cd_off = fields[5], fields[6]
print('cd_size',cd_size,'cd_off',cd_off,'tail_start',size-tail)
PY"
