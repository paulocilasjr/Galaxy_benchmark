/bin/bash -c "python - <<'PY'
import gzip
# parse matching chain and print block around t=46054916
qpos=51540904; target=46054916
path='ref/hg19ToHg38.over.chain.gz'
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        if p[7]=='chr10' and int(p[10])<=qpos<=int(p[11]):
            header=p; blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            break
tStart=int(header[5]); qStart=int(header[10])
t=tStart; q=qStart
for j,b in enumerate(blocks):
    parts=b.split()
    if len(parts)==3: size,dt,dq=map(int,parts)
    else: size=int(parts[0]); dt=dq=0
    if t <= target < t+size:
        off=target-t
        print('found target in block',j,b,'t',t,'q',q,'size',size,'off',off,'qpos mapped',q+off)
        break
    t += size+dt; q += size+dq
PY"
