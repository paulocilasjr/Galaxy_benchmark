/bin/bash -c "python - <<'PY'
import torch, h5py, numpy as np, time
# test loading weights
p='work/Rod_fold0.h5'
f=h5py.File(p,'r')
# helper get dataset
def ds(path): return torch.from_numpy(np.array(f[path]))
w_conv1 = ds('model_weights/conv1d/conv1d/kernel:0').permute(2,1,0).contiguous() # [512,4,21]
b_conv1 = ds('model_weights/conv1d/conv1d/bias:0')
conv_layers=[]
for i in range(1,9):
    w=ds(f'model_weights/conv1d_{i}/conv1d_{i}/kernel:0').permute(2,1,0).contiguous()
    b=ds(f'model_weights/conv1d_{i}/conv1d_{i}/bias:0')
    conv_layers.append((w,b))
w_ct=ds('model_weights/logcounts/logcounts/kernel:0').reshape(1,512,1)
b_ct=ds('model_weights/logcounts/logcounts/bias:0').reshape(1)
print(w_conv1.shape,b_conv1.shape,w_ct.shape,b_ct.shape)
f.close()

@torch.no_grad()
def forward(x):
    # x [B,4,L]
    out=torch.relu(torch.nn.functional.conv1d(x,w_conv1,b_conv1))
    for i,(w,b) in enumerate(conv_layers,1):
        y=torch.nn.functional.conv1d(out,w,b,dilation=2**i)
        xlen=out.shape[2]; ylen=y.shape[2]; crop=(xlen-ylen)//2
        out=y + out[:,:,crop:crop+ylen]
    avg=out.mean(dim=2) # [B,512]
    logcts=torch.nn.functional.linear(avg,w_ct.reshape(1,512),b_ct) # [B,1]
    return logcts, out
x=torch.randn(2,4,2114)
t=time.time(); y,_=forward(x); print(y, 'time',time.time()-t)
PY"
