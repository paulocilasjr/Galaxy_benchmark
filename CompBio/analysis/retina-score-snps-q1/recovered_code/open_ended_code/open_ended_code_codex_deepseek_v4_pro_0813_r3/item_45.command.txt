/bin/bash -c "python - <<'PY'
import pandas as pd, urllib.request, json
out=pd.read_csv('work/snps_lifted.tsv',sep='\\t')
for idx in [52,53,288]:
    r=out.iloc[idx]
    print('\\n',idx,r.to_dict())
    for build,chrom,pos in [('hg19',r['chr19'],int(r['pos19'])),('hg38',r['chr38'],int(r['pos38']) if pd.notna(r['chr38']) else None)]:
        if pos is None: continue
        # fetch ±10
        url=f'https://api.genome.ucsc.edu/getData/sequence?genome={build};chrom={chrom};start={pos-10};end={pos+10}'
        d=json.load(urllib.request.urlopen(url))['dna']
        print(build,chrom,pos, d)
PY"
