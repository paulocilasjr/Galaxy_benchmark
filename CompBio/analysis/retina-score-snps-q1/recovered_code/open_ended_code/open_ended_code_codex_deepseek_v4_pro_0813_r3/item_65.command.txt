/bin/bash -c "python - <<'PY'
import gzip
# input hg19 0-based
tpos=51540904
path='ref/hg19ToHg38.over.chain.gz'
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        # target hg19
        tName=p[2]; tStart=int(p[5]); tEnd=int(p[6])
        qName=p[7]; qStart=int(p[10]); qEnd=int(p[11])
        if tName=='chr10' and tStart<=tpos<=tEnd:
            header=p; blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            break
print('header',header)
t=tStart; q=qStart
for j,b in enumerate(blocks):
    parts=b.split()
    if len(parts)==3: size,dt,dq=map(int,parts)
    else: size=int(parts[0]); dt=dq=0
    if t <= tpos < t+size:
        off=tpos-t
        # qStrand +, map same orientation
        print('found',j,b,'t',t,'q',q,'size',size,'off',off,'q0',q+off,'q1',q+off+1)
        break
    t += size+dt; q += size+dq
PY"
