/bin/bash -c "python - <<'PY'
import numpy as np, pandas as pd
from sklearn.metrics import average_precision_score, precision_recall_curve
out=pd.read_pickle('work/snps_final_hg38_with_seq.pkl')
lfcs=[]
for f in range(5):
    l=np.load(f'work/lfc_fold{f}.npy'); lfcs.append(l)
lfc=np.mean(lfcs,axis=0)
out['Rod_fold_avg_lfc']=lfc
out['abs_lfc']=np.abs(lfc)
ap=average_precision_score(out['label'], out['abs_lfc'])
print('AUPRC',ap)
print('floor2',np.floor(ap*100)/100)
print('format',f'{np.floor(ap*100)/100:.2f}')
print(out[['idx','chr','pos1','ref','alt','strand','label','Rod_fold_avg_lfc','abs_lfc']].head(20))
print(out['Rod_fold_avg_lfc'].describe())
out.to_csv('work/rod_scores.tsv',sep='\\t',index=False)
np.save('work/rod_lfc.npy',lfc)
# PR summary
prec,rec,thr=precision_recall_curve(out['label'], out['abs_lfc'])
print('pr len',len(prec))
PY"
