/bin/bash -c "python - <<'PY'
import gzip
pos=110222992 # 0-based source (file start)
path='ref/hg19ToHg38.over.chain.gz'
found=[]
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        source_name=p[2]; source_start=int(p[5]); source_end=int(p[6])
        if source_name=='chr2' and source_start<=pos<=source_end:
            score=int(p[1]); target_name=p[7]; target_strand=p[9]; target_start=int(p[10]); target_end=int(p[11]); target_size=int(p[8])
            found.append((score,source_start,source_end,target_name,target_strand,target_start,target_end,target_size,line.strip()))
            print('chain found',score,target_strand,target_start,target_end,line.strip())
            # parse blocks to see if pos in block
            blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            s=source_start; t=target_start
            for b in blocks:
                parts=b.split()
                if len(parts)==3: size,sgap,tgap=map(int,parts)
                else: size=int(parts[0]); sgap=tgap=0
                if s<=pos<s+size:
                    off=pos-s
                    print(' block',b,'s',s,'t',t,'size',size,'off',off)
                    # compute correct target for strand
                    if target_strand=='+':
                        tp=t+off
                    else:
                        # correct mapping? within block reverse: tp = t + (size-1-off)? but target coords maybe plus? Need output
                        tp = t + (size-1-off)
                    print('  -> target plus?',tp,'tp1',tp+1)
                    break
                s += size+sgap; t += size+tgap
print('num chains',len(found))
PY"
