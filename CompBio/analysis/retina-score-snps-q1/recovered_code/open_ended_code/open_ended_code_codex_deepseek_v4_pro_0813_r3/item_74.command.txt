/bin/bash -c "python - <<'PY'
import gzip
path='ref/hg19ToHg38.over.chain.gz'
score=22059922; pos=51540904
with gzip.open(path,'rt') as f:
    for line in f:
        if not line.startswith('chain'): continue
        p=line.split()
        if int(p[1])==score:
            header=p; blocks=[]
            for line2 in f:
                if line2.startswith('chain'): break
                if line2.strip(): blocks.append(line2.strip())
            break
print('header',header)
source_start=int(header[5]); target_start=int(header[10]); target_size=int(header[8]); target_strand=header[9]
s=source_start; t=target_start
for j,b in enumerate(blocks):
    parts=b.split()
    if len(parts)==3: size,sgap,tgap=map(int,parts)
    else: size=int(parts[0]); sgap=tgap=0
    if s <= pos < s+size:
        off=pos-s
        tp=t+off
        if target_strand=='-': tp=target_size-1-tp
        print('found block',j,b,'s',s,'t',t,'size',size,'off',off,'tp0',tp,'tp1',tp+1)
        break
    s += size+sgap; t += size+tgap
PY"
