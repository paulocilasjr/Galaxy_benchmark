/bin/bash -c "python - <<'PY'
import pandas as pd, urllib.request,json
# fetch direct bases and compare file ref with base at start, start+1 (0-based coordinates)
# If file start is 0-based, base at API start (0-based) matches.
# If file start is 1-based, base at API start-1 (0-based) matches.
df=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t').head(10)
for i,r in df.iterrows():
    bases=[]
    for off in [-1,0,1]:
        s=r['start']+off
        url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom={r.chr};start={s};end={s+1}'
        b=json.load(urllib.request.urlopen(url))['dna'].upper()
        bases.append(b)
    print(i,r['start'],r['ref'],bases,'match offset -1',bases[0]==r['ref'],'0',bases[1]==r['ref'])
PY"
