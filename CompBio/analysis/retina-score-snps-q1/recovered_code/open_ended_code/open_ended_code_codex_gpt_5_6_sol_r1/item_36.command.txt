/bin/bash -lc "python - <<'PY'
import math
import numpy as np
import pandas as pd
from sklearn.metrics import average_precision_score, auc, precision_recall_curve, roc_auc_score
x=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
folds=np.stack([np.load(f'rod_fold{i}_lfc.npy') for i in range(5)],axis=1)
avg=folds.mean(axis=1)
pred=np.abs(avg)
ap=average_precision_score(x.label,pred)
p,r,t=precision_recall_curve(x.label,pred)
print('shape',folds.shape,'finite',np.isfinite(folds).all())
print('fold APs',[average_precision_score(x.label,np.abs(folds[:,i])) for i in range(5)])
print('average_precision',repr(ap))
print('trapezoidal_pr_auc',auc(r,p))
print('roc_auc',roc_auc_score(x.label,pred))
print('floor2',f'{math.floor(ap*100)/100:.2f}')
print('round2',f'{ap:.2f}')
print('mean_lfc stats',avg.min(),avg.max(),avg.mean(),np.median(pred))
for label in [0,1]:
 s=pred[x.label.to_numpy()==label]
 print('label',label,'mean abs',s.mean(),'median abs',np.median(s),'max',s.max())
# Confirm duplicate rows receive identical scores and dump the evidence table.
out=x.copy(); out['Rod_fold_avg_lfc']=avg; out['abs_Rod_fold_avg_lfc']=pred
out.to_csv('retina.score.snps.q1.rod.scored.tsv',sep='\\t',index=False)
print(out[out.duplicated(['chr','start','ref','alt'],keep=False)].sort_values(['chr','start']).to_string(index=False))
PY"
