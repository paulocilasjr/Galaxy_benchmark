/bin/bash -lc "python - <<'PY'
import pandas as pd, requests
INP_LEN=2114; BUFFER=500
snps=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
for _,r in snps.head(5).iterrows():
    s=int(r.start)-INP_LEN//2; e=int(r.start)+INP_LEN//2+BUFFER
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom={r.chr};start={s};end={e}'
    j=requests.get(url,timeout=30).json()
    seq=j['dna'].upper()
    print(r.chr,r.start,r.ref, len(seq), seq[INP_LEN//2-1:INP_LEN//2-1+len(r.ref)])
PY"
