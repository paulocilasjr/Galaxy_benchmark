/bin/bash -lc "python - <<'PY'
import pandas as pd, requests
snps=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t').head(5)
for _,r in snps.iterrows():
    p=int(r.start)
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg19;chrom={r.chr};start={p-5};end={p+5}'
    seq=requests.get(url,timeout=30).json()['dna'].upper()
    print(r.chr,p,r.ref,seq, [(i-5,b) for i,b in enumerate(seq) if b==r.ref])
PY"
