/bin/bash -lc "python - <<'PY'
import h5py, json
with h5py.File('retina_models/fold0/Rod.h5','r') as f:
    cfg=json.loads(f.attrs['model_config'])
    for i,layer in enumerate(cfg['config']['layers']):
        c=layer['config']
        print('\\n',i,layer['name'],layer['class_name'])
        keys=['batch_input_shape','padding','strides','dilation_rate','filters','kernel_size','activation','cropping','pool_size','axis','units','use_bias']
        for k in keys:
            if k in c: print(' ',k, c[k])
        print(' inbound', layer.get('inbound_nodes'))
    print('\\nweights')
    def walk(g,p=''):
        for k,v in g.items():
            if isinstance(v,h5py.Dataset): print(p+k, v.shape)
            else: walk(v,p+k+'/')
    walk(f['model_weights'])
PY"
