/bin/bash -lc "python - <<'PY'
from pathlib import Path
import struct
b=Path('model-tail.bin').read_bytes()
p=b.rfind(b'PK\\x05\\x06')
print('eocd',p, struct.unpack_from('<4s6H2LH',b,p))
q=b.find(b'PK\\x01\\x02')
while q>=0 and q < p:
    vals=struct.unpack_from('<4s6H3L5H2L',b,q)
    sig,vermade,verneed,flag,comp,mt,md,crc,csize,usize,nlen,elen,clen,disk,iattr,eattr,loff=vals
    name=b[q+46:q+46+nlen].decode('utf8','replace')
    if 'Rod' in name or name.startswith('retina_models'):
        print(q, name, 'comp',comp,'csize',csize,'usize',usize,'loff',loff,'flag',flag,'extra',elen)
    q=b.find(b'PK\\x01\\x02',q+1)
PY"
