/bin/bash -lc "python - <<'PY'
import numpy as np
x=np.load('rod_scores_hg19.npz'); y=x['labels'].astype(np.int64); s=np.abs(x['avg_lfc'])
order=np.argsort(-s,kind='mergesort'); ys=y[order]; ss=s[order]
tp=np.cumsum(ys); idx=np.flatnonzero(np.r_[ss[1:] "'!= ss[:-1], True])
prec=tp[idx]/(idx+1)
rec=tp[idx]/tp[-1]
prev=np.r_[0.,rec[:-1]]
ap=np.sum((rec-prev)*prec)
print('"'positives',int(y.sum()),'negatives',int((1-y).sum()))
print('AP',repr(float(ap)))
print('floor',format(np.floor(ap*100)/100,'.2f'))
print('range',s.min(),s.max(),'unique',len(np.unique(s)))
PY"
