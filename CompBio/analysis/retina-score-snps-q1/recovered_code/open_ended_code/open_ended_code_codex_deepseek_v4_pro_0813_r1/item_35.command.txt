/bin/bash -lc "python - <<'PY'
import pandas as pd, requests, time
s=pd.read_csv('retina.score.snps.q1.tsv',sep='\\t')
url='https://api.genome.ucsc.edu/getData/sequence'
for i,row in s.head(20).iterrows():
    out=[]
    for g in ['hg19','hg38']:
        p=int(row['start'])
        try:
            r=requests.get(url,params={'genome':g,'chrom':row['chr'],'start':p-1,'end':p},timeout=20)
            dna=r.json().get('dna','ERR').upper()
        except Exception as e: dna='ERR'
        out.append(dna)
    print(i,row['chr'],row['start'],row['ref'],'hg19',out[0],'hg38',out[1], 'match', out[0]==row['ref'],out[1]==row['ref'])
PY"
