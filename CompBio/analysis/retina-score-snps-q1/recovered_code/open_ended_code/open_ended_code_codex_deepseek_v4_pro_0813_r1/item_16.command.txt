import struct
b=open('central_dir.bin','rb').read(); pos=0
while pos+46 <= len(b):
    vals=struct.unpack('<4s6H3L5H2L',b[pos:pos+46])
    (_,vmake,vneed,flags,method,mtime,mdate,crc,csize,usize,fnlen,extralen,commentlen,disk,intattr,extattr,lho)=vals
    name=b[pos+46:pos+46+fnlen].decode('utf8','replace')
    if 'Rod.h5' in name and not name.startswith('__MACOSX'):
        print(name,'method',method,'flags',flags,'csize',csize,'usize',usize,'lho',lho,'extralen',extralen,'fnlen',fnlen)
    pos += 46+fnlen+extralen+commentlen
PY"
