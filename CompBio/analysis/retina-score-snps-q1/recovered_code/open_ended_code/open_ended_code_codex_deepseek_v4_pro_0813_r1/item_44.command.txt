/bin/bash -lc "python - <<'PY'
import requests
url='https://api.genome.ucsc.edu/getData/sequence'
# SNP file first from repo

for label,start,end in [('B-1 1-based',B-1,B),('B 0based?',B,B+1),('B+1',B+1,B+2)]:
 r=requests.get(url,params={'genome':'hg38','chrom':chrom,'start':start,'end':end},timeout=20)
 d=r.json().get('dna','ERR').upper()
 print(label,start,end,d)
# fetch 2114+500 window and index1056
s=B-1057; e=B+1057+500
r=requests.get(url,params={'genome':'hg38','chrom':chrom,'start':s,'end':e},timeout=30)
dna=r.json()['dna'].upper()
print('len',len(dna),'idx1056',dna[1056],'idx1057',dna[1057])
print('file ref',ref)
PY"
