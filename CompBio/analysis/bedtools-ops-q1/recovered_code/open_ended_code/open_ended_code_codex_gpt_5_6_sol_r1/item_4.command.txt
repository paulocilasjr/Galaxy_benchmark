/bin/bash -lc "gzip -cd ENCFF333TAT.bed.gz | awk 'BEGIN{OFS=\"\\t\"} {summit="'$2+$10; start=summit-250; end=summit+250; if(start<0) start=0; print $1,start,end}'"' | LC_ALL=C sort -k1,1 -k2,2n -k3,3n > centered500.bed
awk 'BEGIN{OFS=\"\\t\"} NR==1{chrom="'$1; start=$2; end=$3; next} $1==chrom && $2<=end{if($3>end) end=$3; next} {print chrom,start,end; chrom=$1; start=$2; end=$3} END{if(NR) print chrom,start,end}'"' centered500.bed > merged500.bed
awk 'BEGIN{FS=OFS=\"\\t\"} NR==FNR{n["'$1]++; bs[$1,n[$1]]=$2; be[$1,n[$1]]=$3; next} {hit=0; for(i=1;i<=n[$1];i++){if(bs[$1,i] >= $3) break; if(be[$1,i] > $2){hit=1; break}} if(!hit) print}'"' <(gzip -cd ENCFF356LFX.bed.gz | LC_ALL=C sort -k1,1 -k2,2n -k3,3n) merged500.bed > final.bed
awk '{bases += "'$3-$2} END{print "centered_records",NR,"raw_bases",bases}'"' centered500.bed
awk '{bases += "'$3-$2} END{print "merged_records",NR,"merged_bases",bases}'"' merged500.bed
awk '{bases += "'$3-$2} END{print "final_records",NR,"final_bases",bases}'"' final.bed"
