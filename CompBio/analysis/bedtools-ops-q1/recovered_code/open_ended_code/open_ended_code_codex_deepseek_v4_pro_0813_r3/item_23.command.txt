/bin/bash -c "set -e
gzip -dc ENCFF333TAT.bed.gz | awk -F '\\t' 'BEGIN{OFS=\"\\t\"} {s="'$2+$10; print $1,s-251,s+249}'"' | LC_ALL=C sort -k1,1 -k2,2n > peaks_shift.bed
bedtools merge -i peaks_shift.bed > merged_shift.bed
bedtools intersect -a merged_shift.bed -b ENCFF356LFX.bed.gz -v > final_shift.bed
awk '{sum+="'$3-$2} END{print "intersect_v",sum}'"' final_shift.bed
awk '{sum+="'$3-$2} END{print "merged",sum}'"' merged_shift.bed
bedtools subtract -a merged_shift.bed -b ENCFF356LFX.bed.gz | awk '{sum+="'$3-$2} END{print "subtract",sum}'"'"
