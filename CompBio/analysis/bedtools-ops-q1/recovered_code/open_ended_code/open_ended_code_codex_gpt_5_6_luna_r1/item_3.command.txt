/bin/bash -lc "set -e

gzip -cd ENCFF333TAT.bed.gz |
awk 'BEGIN{FS=OFS=\"\\t\"} {summit="'$2+$10; print $1, summit-250, summit+250}'"' > centered.bed
bedtools sort -i centered.bed > centered.sorted.bed
bedtools merge -i centered.sorted.bed > merged.bed
gzip -cd ENCFF356LFX.bed.gz | bedtools sort -i - > blacklist.sorted.bed
bedtools intersect -v -a merged.bed -b blacklist.sorted.bed > kept.intersect-v.bed
bedtools subtract -a merged.bed -b blacklist.sorted.bed > kept.subtract.bed
printf '%s\\n' '--- counts and base totals ---'
for f in centered.bed centered.sorted.bed merged.bed kept.intersect-v.bed kept.subtract.bed; do
  awk 'BEGIN{n=0; b=0} {n++; b += "'$3-$2} END{print FILENAME, "intervals=" n, "bases=" b}'"' \""'$f"
done
printf '"'%s\\n' '--- merged intervals overlapping blacklist ---'
bedtools intersect -wa -u -a merged.bed -b blacklist.sorted.bed | wc -l
printf '%s\\n' '--- first/last final inverse ---'
sed -n '1,3p' kept.intersect-v.bed
tail -n 3 kept.intersect-v.bed"
