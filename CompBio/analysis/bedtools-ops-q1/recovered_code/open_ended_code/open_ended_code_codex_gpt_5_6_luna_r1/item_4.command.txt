/bin/bash -lc "printf '%s\\n' '--- blacklist overlap detail ---'
bedtools intersect -wa -wb -a merged.bed -b blacklist.sorted.bed
printf '%s\\n' '--- overlap interval lengths / neighboring context ---'
awk 'BEGIN{FS=OFS=\"\\t\"} NR==1 {c="'$1;s=$2;e=$3; next} {if($1==c && $2<=e){print "merged block",c,s,e,e-s; c=""; exit}}'"' merged.bed || true
printf '%s\\n' '--- file integrity ---'
awk 'BEGIN{FS=\"\\t\"} NF"'!=3 || $2<0 || $3<$2 || $1=="" {bad++} END{print "merged bad",bad+0}'"' merged.bed
awk 'BEGIN{FS=\"\\t\"} NF"'!=3 || $2<0 || $3<$2 || $1=="" {bad++} END{print "final bad",bad+0}'"' kept.intersect-v.bed"
