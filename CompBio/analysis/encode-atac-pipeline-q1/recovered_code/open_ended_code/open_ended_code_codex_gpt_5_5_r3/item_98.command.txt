/bin/bash -lc "./envs/atac/bin/python3 - <<'PY'
import os, subprocess
os.environ['PATH'] = os.getcwd() + '/envs/atac/bin:' + os.environ['PATH']
for args in [['/bin/bash','-c','echo "'$PATH; which cutadapt'"'], ['/bin/bash','-o','pipefail','-c','echo "'$PATH; which cutadapt'"']]:
    p=subprocess.Popen(args, stdout=subprocess.PIPE, stderr=subprocess.PIPE, universal_newlines=True)
    out,err=p.communicate()
    print(args, 'rc', p.returncode)
    print(out, err)
PY"
