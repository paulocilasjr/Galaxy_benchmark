/bin/bash -lc "python - <<'PY'
import gzip,numpy as np,pandas as pd
from scipy.io import mmread
from scipy import sparse
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f: g=pd.read_csv(f,sep='\\t').gene_symbols.astype(str).to_numpy()
Xs=[]
for z in ['1','2']:
 with gzip.open(f'subtype.inflammation.q1.{z}.mtx.gz','rb') as f: Xs.append(mmread(f).tocsr())
X=sparse.hstack(Xs,format='csr'); n1=Xs[0].shape[1]
lib=np.asarray(X.sum(0)).ravel(); scale=1e4/np.maximum(lib,1); Xn=X.multiply(scale).tocsr(); Xn.data=np.log1p(Xn.data)
# gene mean norm and detection per eye
for b,(a,c) in enumerate([(0,n1),(n1,X.shape[1])],1):
 m=np.asarray(Xn[:,a:c].mean(1)).ravel(); d=np.asarray((X[:,a:c]>0).mean(1)).ravel()
 if b==1: m1,d1=m,d
 else: m2,d2=m,d
# ranking difference log ratio mean normalized, require prevalence
lf=np.log2((m1+1e-5)/(m2+1e-5)); dd=d1-d2
for title,score in [('log2mean',lf),('detdiff',dd)]:
 print('\\nTOP',title,'eye1')
 for i in np.argsort(score)[-60:][::-1]:
  if max(d1[i],d2[i])>.002: print(g[i],round(score[i],2),round(m1[i],3),round(m2[i],3),round(d1[i],3),round(d2[i],3))
 print('\\nTOP',title,'eye2')
 for i in np.argsort(score)[:60]:
  if max(d1[i],d2[i])>.002: print(g[i],round(score[i],2),round(m1[i],3),round(m2[i],3),round(d1[i],3),round(d2[i],3))
PY"
