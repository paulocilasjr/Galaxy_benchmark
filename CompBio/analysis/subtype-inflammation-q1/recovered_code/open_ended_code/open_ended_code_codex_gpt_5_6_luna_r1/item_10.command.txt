import gzip,numpy as np,pandas as pd
from scipy.io import mmread
from scipy import sparse
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f:g=pd.read_csv(f,sep='\\t').gene_symbols.astype(str).to_numpy()

for gene in ['CD69','CCL5','IL18RAP','CSF3R','OLR1','P2RY12','RHO','SLC1A3','GFAP','AQP4','RLBP1','SAG','GNAT1','NRL','C1QA','TYROBP','LYZ']:
 i=mp.get(gene); z=X[i,:].toarray().ravel() if i is not None else np.array([])
 print(gene, 'n',int((z>0).sum()),'sum',int(z.sum()),'indices first',np.where(z>0)[0][:10])
for label,ii in [('CD69+', X[mp['CD69'],:]>0),('CD69-',X[mp['CD69'],:]==0),('CCL5+',X[mp['CCL5'],:]>0),('CSF3R+',X[mp['CSF3R'],:]>0)]:
 ii=np.asarray(ii).ravel().astype(bool)
 print('\\nSUBSET',label,'n',ii.sum(),'lib',np.percentile(lib[ii],[0,25,50,75,100]))
 m=np.asarray(Xn[:,ii].mean(1)).ravel(); d=np.asarray((X[:,ii]>0).mean(1)).ravel();
 # top by mean normalized
 for ix in np.argsort(m)[-100:][::-1]:
  if g[ix].upper() in ['MALAT1']: continue
  print(g[ix],round(m[ix],2),round(d[ix],2),end='; ')
 print()
