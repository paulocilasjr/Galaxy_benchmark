/bin/bash -lc "python - <<'PY'
import gzip,numpy as np,pandas as pd,time
from scipy.io import mmread
from sklearn.decomposition import TruncatedSVD
from sklearn.cluster import KMeans
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f:g=pd.read_csv(f,sep='\\t').gene_symbols.astype(str).to_numpy()
with gzip.open('subtype.inflammation.q1.1.mtx.gz','rb') as f:X1=mmread(f).tocsr()
with gzip.open('subtype.inflammation.q1.2.mtx.gz','rb') as f:X2=mmread(f).tocsr()
from scipy import sparse
X=sparse.hstack([X1,X2],format='csr'); n1=X1.shape[1]; batch=np.r_[np.zeros(n1,int),np.ones(X2.shape[1],int)]
lib=np.asarray(X.sum(0)).ravel(); Xn=X.multiply(1e4/np.maximum(lib,1)).tocsr(); Xn.data=np.log1p(Xn.data); Xc=Xn.T.tocsr()
mean=np.asarray(Xc.mean(0)).ravel(); mean2=np.asarray(Xc.multiply(Xc).mean(0)).ravel(); var=mean2-mean*mean
# restrict genes present in at least 20 cells and ignore mitochondrial/ribo maybe
present=np.asarray((X>0).sum(1)).ravel(); mask=(present>=20)&~np.char.startswith(np.char.upper(g),'MT-')
idx=np.where(mask)[0][np.argsort(var[mask])[::-1][:3000]]
print('hvg',len(idx),'matrix',Xc[:,idx].shape)
t=time.time(); svd=TruncatedSVD(n_components=50,n_iter=5,random_state=1); Z=svd.fit_transform(Xc[:,idx]); print('svd',time.time()-t,svd.explained_variance_ratio_.sum())
t=time.time(); km=KMeans(n_clusters=50,n_init=5,random_state=1,algorithm='lloyd').fit(Z); print('kmeans',time.time()-t)
clusters=km.labels_; cd=np.asarray(X.getrow(mp) if False else X[0,:])
# make markers
mp={x.upper():i for i,x in enumerate(g)}
markers={
'astro':['GFAP','AQP4','ALDH1L1','SLC1A3','GJA1','SOX9','CLDN11','ATP1B2','AGT'],
'micro':['P2RY12','TMEM119','CX3CR1','C1QA','C1QB','C1QC','AIF1','TYROBP','CSF1R','LST1','CTSS','SPI1'],
'RPE':['RPE65','BEST1','RLBP1','PMEL','TYR','MITF','LRAT','MERTK','RDH5'],
'cone':['ARR3','GNB3','PDE6H','OPN1SW','OPN1MW','OPN1LW','GNAT2','RXRG','GUCA1C','GRK7'],
'rod':['RHO','GNAT1','PDE6A','PDE6B','SAG','NRL','NR2E3','ROM1','GNGT1','CNGB1'],
'horizontal':['LHX1','LHX2','ONECUT1','PROX1','CALB1','CACNA1A','GAD1','GAD2','VSX1'],
'bipolar':['VSX2','VSX1','PRKCA','GRM6','TRPM1','CABP5','PCP4','OTX2','GPR149','LHX3'],
'RGC':['RBPMS','SNCG','ISL1','POU4F1','POU4F2','THY1','NEFL','SLC17A6','GAP43','CPLX3'],
'T':['CD3D','CD3E','TRBC1','TRBC2','CD2','CD247','IL7R','LTB','CCL5'],
'macro':['LYZ','FCER1G','TYROBP','CTSD','CD68','LILRB1','AIF1','CTSS','CD14','MS4A7','LGALS3','MARCO','ADGRE1','OLR1'],
'Schwann':['S100B','SOX10','PLP1','MPZ','NCAM1','ERBB3','MAG','PMP22','GFAP'],
'pericyte':['RGS5','CSPG4','MCAM','PDGFRB','ACTA2','COL4A1','DES','NOTCH3','ABCC9','KCNJ8'],
'B':['CD79A','MS4A1','CD37','CD74','HLA-DRA','CD79B','CD22','CD19','CD83'],
'fibro':['COL1A1','COL1A2','DCN','LUM','COL3A1','COL6A1','PDGFRA','C7','FBLN1','COL6A2'],
'Muller':['SLC1A3','RLBP1','GLUL','VIM','SOX9','RAX','KCNJ10','AQP4','FABP7','SLC1A2','DIO2'],
'endothelial':['EMCN','KDR','ESAM','PECAM1','VWF','ENG','CLDN5','RAMP2','PLVAP','RGCC']}
ct=list(markers)
def inds(gs):return [mp[x] for x in gs if x in mp]
sc=np.array([np.asarray(Xc[:,inds(gs)].mean(1)).ravel() for gs in markers.values()]).T
# cluster summary
for c in range(50):
 ii=clusters==c
 if ii.sum()<10:continue
 m=sc[ii].mean(0); top=np.argsort(m)[::-1][:3]
 cd69=np.asarray(X[mp['CD69'],ii]).ravel()>0
 ccl5=np.asarray(X[mp['CCL5'],ii]).ravel()>0
 print(c,'n',ii.sum(),'e1/e2',int((ii& (batch==0)).sum()),int((ii&(batch==1)).sum()),'cd69',cd69.sum(),'ccl5',ccl5.sum(),'markers',[(ct[j],round(float(m[j]),2)) for j in top])
# print clusters sorted by CD69 count
print('CD69 clusters')
for c in np.argsort([((clusters==c)&(batch==0)&(np.asarray(X[mp['CD69'],:].toarray()).ravel()>0)).sum() for c in range(50)])[::-1][:10]:
 ii=(clusters==c); 
 print(c,'n',ii.sum(),'e1/e2',[(ii&(batch==b)).sum() for b in [0,1]],'cd69',((ii)&(batch==0)&(np.asarray(X[mp['CD69'],:].toarray()).ravel()>0)).sum())
 # cluster top genes normalized mean versus all
 cm=np.asarray(Xc[ii].mean(0)).ravel(); bg=np.asarray(Xc.mean(0)).ravel(); score=cm-bg
 print('; '.join(g[i]+'='+str(round(float(score[i]),2)) for i in np.argsort(score)[-30:][::-1]))
PY"
