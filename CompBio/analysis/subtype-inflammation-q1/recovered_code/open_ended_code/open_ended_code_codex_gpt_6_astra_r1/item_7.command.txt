/bin/bash -lc "cat > analysis/cluster.py <<'PY'
import sys, os
sys.path.insert(0,os.path.abspath('packages'))
os.environ['NUMBA_CACHE_DIR']=os.path.abspath('.cache/numba')
os.environ['MPLCONFIGDIR']=os.path.abspath('.cache/matplotlib')
import scanpy as sc
import numpy as np,pandas as pd
from scipy import sparse
sc.settings.n_jobs=6
G=pd.read_csv('subtype.inflammation.q1.genes.txt.gz',sep='\\t')
x=sparse.vstack([sparse.load_npz(f'analysis/eye{i}.npz') for i in [1,2]],format='csr')
a=sc.AnnData(x)
a.var_names=G.gene_symbols.values
a.var_names_make_unique()
a.obs['eye']=pd.Categorical(['1']*8646+['2']*9355)
a.obs['counts']=np.asarray(x.sum(axis=1)).ravel()
a.layers['counts']=x.copy()
sc.pp.normalize_total(a,target_sum=1e4)
sc.pp.log1p(a)
sc.pp.highly_variable_genes(a,n_top_genes=3000)
print('HVG done',flush=True)
sc.tl.pca(a,n_comps=50)
sc.pp.neighbors(a,n_neighbors=20,n_pcs=35)
sc.tl.leiden(a,resolution=0.6,key_added='cluster',flavor='igraph',n_iterations=2)
print('clustering done',flush=True)
sc.tl.rank_genes_groups(a,'cluster',method='wilcoxon',pts=True)
r=sc.get.rank_genes_groups_df(a,group=None)
r.to_csv('analysis/cluster_markers.csv',index=False)
a.write('analysis/data.h5ad')
print(pd.crosstab(a.obs.cluster,a.obs.eye).to_string())
for c in a.obs.cluster.cat.categories:
 print(c,', '.join(r[r.group==c].head(35).names))
PY"
