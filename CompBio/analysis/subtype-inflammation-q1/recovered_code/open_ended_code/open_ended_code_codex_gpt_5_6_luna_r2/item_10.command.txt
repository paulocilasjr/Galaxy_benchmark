/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
from scipy.io import mmread
# load gene symbols
g=[]
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as h:
 next(h)
 for line in h: g.append(line.rstrip().split('\\t')[1])
Xs=[]
for f in ['subtype.inflammation.q1.1.mtx.gz','subtype.inflammation.q1.2.mtx.gz']:
 x=mmread(f).tocsr().astype(np.float64)
 lib=np.asarray(x.sum(axis=0)).ravel()
 # means CPM per eye, pseudocount transform at average? use normalized mean
 Xs.append(x.multiply(1e4/lib).mean(axis=1).A.ravel())
a,b=Xs
# log ratio of means; filter enough
for title,rat in [('eye1 higher',np.log2((a+0.01)/(b+0.01))),('eye2 higher',np.log2((b+0.01)/(a+0.01)))]:
 print('\\n'+title)
 ix=np.argsort(rat)[::-1]
 for i in ix[:100]:
  if a[i]+b[i] < .05: continue
  print(f'{g[i]:30s} log2={rat[i]:6.2f} mean1={a[i]:7.2f} mean2={b[i]:7.2f}')
  if sum(1 for j in ix[:list(ix).index(i)+1] if a[j]+b[j]>=.05)>=30: break
PY"
