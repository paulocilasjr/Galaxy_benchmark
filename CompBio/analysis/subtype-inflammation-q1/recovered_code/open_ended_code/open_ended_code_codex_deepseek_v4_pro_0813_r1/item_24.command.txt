/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('combined.h5ad')
# ensure X log1p
genes=['RLBP1','GLUL','SLC1A3','CLU','DKK3','VIM','APOE','GFAP','AQP4','S100B','P2RY12','CSF1R','CX3CR1','AIF1','LYZ','CD14','MRC1','CD68','FCGR3A','PDGFRB','RGS5','ACTA2','NOTCH3','DCN','COL1A1','PECAM1','VWF','CLDN5','RPE65','BEST1','SERPINF1','RHO','ARR3','VSX2','GRIK1','RBPMS','THY1','CD74','HLA-DRA','C1QA','C1QB','C1QC','CXCL8','IL1B','NFKBIA','CXCL10','CCL2','STAT1','IRF1','ISG15','IFITM3','SOD2','FOS','JUN','B2M']
present=[g for g in genes if g in adata.var_names]
for cl in ['10','11']:
 sub=adata[adata.obs['leiden']==cl]
 print('\\nCluster',cl,'n',sub.n_obs,'sample',sub.obs['sample'].value_counts().to_dict())
 vals={g:float(np.asarray(sub[:,g].X.mean())) for g in present}
 for g,v in sorted(vals.items(), key=lambda kv:kv[1], reverse=True)[:30]:
  print(f'{g}: {v:.3f}', end='  ')
 print()
PY"
