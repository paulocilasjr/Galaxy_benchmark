/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
a=sc.read_h5ad('combined.h5ad')
# if assignments file exists load; else recompute rough? use saved
obs=pd.read_csv('celltype_assignments.csv',index_col=0)
a.obs['pred_type']=obs['pred_type']
print('astrocyte cluster counts')
print(a.obs[a.obs.pred_type=='astrocyte'].groupby('leiden').size())
print('muller cluster counts')
print(a.obs[a.obs.pred_type=='muller glia cell'].groupby('leiden').size())
# mean markers in predicted astrocyte vs muller
for typ in ['astrocyte','muller glia cell']:
 sub=a[a.obs.pred_type==typ]
 print(typ,sub.n_obs, 'sample',sub.obs['sample'].value_counts().to_dict())
 for g in ['GFAP','AQP4','S100B','RLBP1','GLUL','SLC1A3','CLU','DKK3','APOE','CD14','IL1B','CXCL10']:
  x=np.asarray(sub[:,g].X.toarray()).ravel()
  print(' ',g,round(x.mean(),3),(x>0).mean()*100)
PY"
