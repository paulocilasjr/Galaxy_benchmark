import scanpy as sc, pandas as pd, numpy as np
adata=sc.read_h5ad('combined.h5ad')

# get arrays

for cond in ['CD14','IL1B','CXCL10']:
 x=arr[cond]
 pos=x>0
 neg=~pos
 print('\\nCondition',cond,'pos',pos.sum(),'neg',neg.sum())
 for g in ['RLBP1','GLUL','SLC1A3','GFAP','AQP4','P2RY12','CSF1R','AIF1','LYZ']:
  print(g, 'pos mean',arr[g][pos].mean(), 'neg mean',arr[g][neg].mean())
PY"
