/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np
from scipy.stats import ranksums, mannwhitneyu
adata=sc.read_h5ad('combined.h5ad')
# Use cells predicted as muller from saved assignments if exists, or recompute quickly from clusters 10 and 11
muller=adata[adata.obs['leiden'].isin(['10','11'])].copy()
print('muller cluster cells', muller.n_obs, muller.obs['sample'].value_counts().to_dict())
genes=['RLBP1','GLUL','SLC1A3','CLU','DKK3','APOE','GFAP','AQP4','C1QA','C1QB','C1QC','HLA-DRA','HLA-DRB1','HLA-DPA1','CD74','CXCL8','IL1B','NFKBIA','CXCL10','CCL2','STAT1','IRF1','ISG15','IFITM3','SOD2','FOS','JUN','B2M','VIM','PDGFRB','P2RY12','CD14']
present=[g for g in genes if g in adata.var_names]
eye1=muller[muller.obs['sample']=='eye1']
eye2=muller[muller.obs['sample']=='eye2']
for g in present:
    x=np.asarray(eye1[:,g].X.toarray()).ravel() if hasattr(eye1[:,g].X,'toarray') else np.asarray(eye1[:,g].X).ravel()
    y=np.asarray(eye2[:,g].X.toarray()).ravel() if hasattr(eye2[:,g].X,'toarray') else np.asarray(eye2[:,g].X).ravel()
    try:
        stat,p=mannwhitneyu(x,y,alternative='two-sided')
    except Exception as e:
        p=np.nan
    print(f'{g}: eye1={x.mean():.3f}, eye2={y.mean():.3f}, diff={x.mean()-y.mean():.3f}, pct1={(x>0).mean()*100:.1f}%, pct2={(y>0).mean()*100:.1f}%, p={p:.2g}')
PY"
