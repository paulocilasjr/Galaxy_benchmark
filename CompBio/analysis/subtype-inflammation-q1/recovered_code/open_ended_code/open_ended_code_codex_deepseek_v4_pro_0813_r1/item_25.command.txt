/bin/bash -c "cat > plot_markers.py <<'PY'
import scanpy as sc, matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
adata=sc.read_h5ad('combined.h5ad')
markers=['RLBP1','SLC1A3','GLUL','CLU','DKK3','APOE','GFAP','AQP4','RPE65','BEST1','PDGFRB','P2RY12','CD14','C1QA','C1QB','CXCL10','IL1B','CCL2','NFKBIA','HLA-DRA','CD74','VSX2','GRIK1','ARR3','RHO','RBPMS','THY1','DCN','PECAM1']
markers=[g for g in markers if g in adata.var_names]
# Dotplot by leiden
sc.pl.dotplot(adata, markers, groupby='leiden', standard_scale='var', save='_dotplot.png')
sc.pl.umap(adata, color=['leiden','sample','RLBP1','SLC1A3','CXCL10','IL1B','P2RY12','CD14'], save='_umap.png')
print('done')
PY
python3 plot_markers.py"
