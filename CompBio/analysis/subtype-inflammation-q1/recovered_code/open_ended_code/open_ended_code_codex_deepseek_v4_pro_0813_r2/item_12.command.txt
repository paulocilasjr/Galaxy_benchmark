/bin/bash -c "cat > markers.py <<'PY'
import scanpy as sc, pandas as pd, numpy as np
adata=sc.read('retina_combined.h5ad')
sc.tl.rank_genes_groups(adata, 'leiden', method='wilcoxon', use_raw=True, layer='counts')
result=adata.uns['rank_genes_groups']
groups=result['names'].dtype.names
for g in groups:
    genes=list(result['names'][g][:25])
    scores=list(result['scores'][g][:25])
    logfc=list(result['logfoldchanges'][g][:25])
    print('CLUSTER',g)
    for x,y,z in zip(genes,scores,logfc):
        print(f' {x}\\t{y:.1f}\\t{z:.2f}')
    print()
PY
python3 markers.py > markers_out.txt
head -200 markers_out.txt"
