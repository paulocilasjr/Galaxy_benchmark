/bin/bash -c "python3 - <<'PY'
import gzip, scipy.io, scipy.sparse as sp, numpy as np, pandas as pd
genes=pd.read_csv('subtype.inflammation.q1.genes.txt.gz',sep='\\t')
Xs=[]
for f in ['subtype.inflammation.q1.1.mtx.gz','subtype.inflammation.q1.2.mtx.gz']:
 with gzip.open(f,'rb') as fh: X=scipy.io.mmread(fh).tocsr()
 Xs.append(X)
A=sp.hstack(Xs,format='csr')
counts=np.asarray(A.sum(axis=0)).ravel()
ncell=np.asarray((A>0).sum(axis=0)).ravel()
print('mean genes/cell', ncell.mean(), np.percentile(ncell,[1,10,50,90,99]))
print('mean umi/cell', counts.mean(), np.percentile(counts,[1,10,50,90,99]))
PY"
