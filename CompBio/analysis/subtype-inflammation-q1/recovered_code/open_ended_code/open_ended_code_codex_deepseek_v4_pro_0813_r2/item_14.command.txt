/bin/bash -c "python3 - <<'PY'
import warnings; warnings.filterwarnings('ignore')
import scanpy as sc
adata=sc.read('retina_combined.h5ad')
sc.tl.rank_genes_groups(adata, 'leiden', method='wilcoxon', use_raw=True)
result=adata.uns['rank_genes_groups']
groups=result['names'].dtype.names
with open('markers_clean.txt','w') as f:
 for g in groups:
  f.write('CLUSTER '+g+'\\n')
  for x,y,z in zip(result['names'][g][:20], result['scores'][g][:20], result['logfoldchanges'][g][:20]):
   f.write(f'{x}\\t{y:.1f}\\t{z:.2f}\\n')
print('done')
PY
sed -n '/CLUSTER/p' markers_clean.txt | head -30
# print top 8 markers each cluster compact
python3 - <<'PY'
lines=open('markers_clean.txt').read().splitlines()
cluster=None; buf=[]
for line in lines:
 if line.startswith('CLUSTER'):
  if cluster is not None:
   print(cluster, ', '.join([x.split('\\t')[0] for x in buf[:12]]))
  cluster=line; buf=[]
 else: buf.append(line)
if cluster is not None: print(cluster, ', '.join([x.split('\\t')[0] for x in buf[:12]]))
PY"
