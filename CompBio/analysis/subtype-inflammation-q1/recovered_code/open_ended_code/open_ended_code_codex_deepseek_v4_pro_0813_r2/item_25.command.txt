/bin/bash -c "python3 - <<'PY'
import scanpy as sc, pandas as pd, numpy as np, warnings; warnings.filterwarnings('ignore')
adata=sc.read('retina_combined.h5ad')
X=adata.raw.X; varn=list(adata.raw.var.index); obs=adata.obs
infl=['IL1B','IL6','TNF','CXCL8','CCL2','CCL3','CCL4','CCL5','CXCL10','CXCL9','CXCL11','IL18','IL10','TGFB1','NFKB1','NFKBIA','STAT1','STAT2','IRF1','IRF7','ISG15','IFIT1','IFIT2','IFIT3','MX1','OAS1','OAS2','OASL','RSAD2','GBP1','GBP2','HLA-DRA','HLA-DRB1','CD74','B2M','C3','CFB','SERPING1','PTGS2','NOS2']
present=[g for g in infl if g in varn]
inds=[varn.index(g) for g in present]
V=np.asarray(X[:,inds].mean(axis=1)).ravel()
obs['infl_score']=V
# Also log sum? mean
print('top infl score cells:')
idx=np.argsort(V)[::-1][:100]
sub=obs.iloc[idx][['batch','leiden']].copy(); sub['score']=V[idx]
print(sub.groupby(['batch','leiden']).size().head(30))
print('\\ncluster mean infl score by batch')
print(obs.groupby(['batch','leiden']).infl_score.mean().unstack().T.round(3).to_string())
print('\\ncluster difference eye1-eye2')
d=obs.groupby(['batch','leiden']).infl_score.mean().unstack().T
d['diff']=d['eye1']-d['eye2']
d['ratio']=d['eye1']/(d['eye2']+1e-4)
print(d.sort_values('diff',ascending=False).head(15).round(3).to_string())
# top genes difference per cluster maybe later
PY"
