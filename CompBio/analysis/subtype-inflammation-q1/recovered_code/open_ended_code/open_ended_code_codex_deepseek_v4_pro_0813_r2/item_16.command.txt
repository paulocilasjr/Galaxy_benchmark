/bin/bash -c "cat > annotate_scores.py <<'PY'
import scanpy as sc, pandas as pd, numpy as np, warnings; warnings.filterwarnings('ignore')
adata=sc.read('retina_combined.h5ad')
Xraw=adata.raw.X
var=adata.raw.var.index.tolist()
def expr(genes):
    inds=[var.index(g) for g in genes if g in var]
    return np.asarray(Xraw[:,inds].mean(axis=1)).ravel()
sets={
'rod':['RHO','NRL','NR2E3','PDE6A','PDE6B','CNGB1','SAG','GNGT1','GNAT1','RCVRN','ABCA4','PRPH2'],
'cone':['ARR3','OPN1SW','OPN1MW','OPN1LW','PDE6H','GNAT2','CNGB3','RXRG','GUCA1A','GUCA1B'],
'bipolar':['VSX2','OTX2','CABP5','GRM6','TRPM1','GRIK1','ISL1','PRKCA','NYX','GNG13','GPR179'],
'rgc':['POU4F1','POU4F2','RBPMS','THY1','SLC17A6','NEFL','NEFM','SNCG','GAP43','ISL1'],
'amacrine':['GAD1','GAD2','TFAP2A','TFAP2B','SLC6A9','CALB2','CCK','ST18'],
'horizontal':['ONECUT1','ONECUT2','LHX1','CALB1','TNR','SEPT4'],
'muller':['RLBP1','GLUL','AQP4','SLC1A3','VIM','SOX9','CLU','DKK3','HES1','S100B'],
'astrocyte':['GFAP','ALDH1L1','S100B','AQP4','PAX2','VIM'],
'microglia':['P2RY12','TMEM119','CX3CR1','CSF1R','C1QA','C1QB','C1QC','TYROBP','AIF1','CD68','ITGAM','SPI1','P2RY13','TBXAS1'],
'macrophage':['CD163','MRC1','F13A1','LYZ','S100A8','S100A9','CD14','FCGR3A','ITGAX','MSR1'],
'tcell':['CD3D','CD3E','CD3G','TRAC','TRBC1','TRBC2','CD2','IL7R','LCK','GZMA','NKG7','CCL5'],
'bcell':['CD79A','CD79B','MS4A1','CD19','BANK1','MZB1','IGHM','JCHAIN','CD74'],
'endothelial':['PECAM1','VWF','CDH5','CLDN5','KDR','FLT1','ENG','EMCN','PLVAP'],
'pericyte':['PDGFRB','RGS5','NOTCH3','CSPG4','ACTA2','TAGLN','COX4I2'],
'fibroblast':['DCN','LUM','COL1A1','COL1A2','COL3A1','COL6A3','FBLN1','PDGFRA','IGFBP5'],
'schwann':['MPZ','PLP1','SOX10','NGFR','CDH19','PMP22'],
'rpe':['RPE65','BEST1','TYRP1','PMEL','MLANA','DCT','RGR','LRAT','SERPINF1','SLC16A8'],
}
scores=pd.DataFrame(index=adata.obs.index)
for k,gs in sets.items():
    e=expr(gs)
    scores[k]=e
scores['leiden']=adata.obs['leiden'].values
means=scores.groupby('leiden').mean()
# scaled z scores per cluster for easier ranking
z=means.copy()
for c in z.columns:
    z[c]=(z[c]-z[c].mean())/z[c].std()
# print top sets per cluster
for cl in means.index:
    top=means.loc[cl].sort_values(ascending=False)[:5]
    print(cl, '; '.join(f'{k}={v:.2f}' for k,v in top.items()))
    # print raw key maybe immune if positive
print('\\nScaled scores per cluster')
print(z.round(2).to_string())
z.to_csv('cluster_type_scores.csv')
means.to_csv('cluster_type_means.csv')
# save per cell scores
scores.to_csv('cell_type_scores.csv', index=False)
PY
python3 annotate_scores.py"
