/bin/bash -lc "OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 python -u - <<'PY'
import pandas as pd, numpy as np
from scipy.io import mmread
g=pd.read_csv('subtype.inflammation.q1.genes.txt.gz',sep='\\t').gene_symbols.astype(str).tolist(); gi={x:i for i,x in enumerate(g)}
x=mmread('subtype.inflammation.q1.1.mtx.gz').tocsr().T.tocsr()
for target in ['CCL5','CXCL10','FOXP2','WIF1']:
 m=x[:,gi[target]].toarray().ravel()>0
 print('\\n',target,m.sum())
 for z in ['GLUL','RLBP1','SLC1A3','CLU','APOE','VIM','AQP4','GFAP','ALDH1L1','SLC1A2','SOX9','PAX2','WIF1','CHRDL1','TF','FOXP2','GABRG3','EGF','IL1R1','RGR','RPE65','BEST1','KRT18','P2RY12','C1QA','AIF1','PECAM1','RGS5']:
  v=x[:,gi[z]].toarray().ravel()
  print(z, 'pct+',round((v[m]>0).mean(),3),'avg+',round(v[m].mean(),2),'pct-',round((v[~m]>0).mean(),3),'enrich',round((v[m].mean()+.001)/(v[~m].mean()+.001),1))
PY"
