/bin/bash -lc "python - <<'PY'
import gzip,numpy as np
from scipy.io import mmread

def load(p):
 with gzip.open(p,'rt') as f:return mmread(f).tocsr()
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f:
 next(f); genes=[l.rstrip('\\n').split('\\t')[1] for l in f]
pos={g:i for i,g in enumerate(genes)}; x=load('subtype.inflammation.q1.1.mtx.gz')
infl=['IL1B','CCL2','CCL7','CXCL9','CXCL10','CXCL11','IL6','CSF3','S100A8','S100A9','NFKBIA','CCL3','CCL4','CCL20','CD69','TNF','TNFRSF9','HLA-DRA','HLA-DRB1','CD74','MARCO']
score=np.asarray(x[[pos[g] for g in infl],:].sum(0)).ravel()
# Print top cells with > 20 raw inflammation count, all relevant marker expression and top genes excluding housekeeping
markers=['RHO','GNAT1','PDE6A','PDE6B','SAG','NRL','NR2E3','RCVRN','ROM1','GRK1','CNGB1','EYS','TF','ARR3','OPN1SW','OPN1MW','OPN1LW','GNAT2','PDE6C','RXRG','THRB','GFAP','AQP4','SLC1A3','S100B','ALDOC','VIM','CLU','GLUL','P2RY12','TMEM119','CX3CR1','C1QA','C1QB','C1QC','AIF1','TYROBP','LST1','CSF1R','TREM2','CTSS','FCER1G','CST3','APOE','LYZ','CD68','MS4A7','ITGAM','MRC1','MARCO','FCGR3A','RBPMS','THY1','POU4F1','POU4F2','SNCG','NEFL','VSX2','GRM6','TRPM1','CABP5','PCP2','PRKCA','GNB3','RPE65','BEST1','MITF','PMEL','MERTK','KDR','ESM1','PECAM1','VWF','CLDN5','EMCN','KCNJ8','CSPG4','PDGFRB','RGS5','COL1A1','COL1A2','DCN','LUM','MPZ','SOX10','PLP1','MAL','MBP','CD3D','CD3E','TRBC1','TRBC2','LCK','MS4A1','CD79A','CD74','HLA-DRA']
for j in np.flatnonzero(score>=15)[:50]:
 vals=x[:,j].toarray().ravel(); top=np.argsort(vals)[-40:][::-1]
 print('\\nCELL',j,'score',score[j],'lib',vals.sum())
 print('relevant',[(g,int(vals[pos[g]])) for g in markers if vals[pos[g]]])
 print('top',[(genes[i],int(vals[i])) for i in top if vals[i]][:30])
PY"
