/bin/bash -lc "python - <<'PY'
import gzip,numpy as np
from scipy.io import mmread

def load(p):
 with gzip.open(p,'rt') as f:return mmread(f).tocsr()
with gzip.open('subtype.inflammation.q1.genes.txt.gz','rt') as f:
 next(f); genes=[l.rstrip('\\n').split('\\t')[1] for l in f]
pos={g:i for i,g in enumerate(genes)}
a=load('subtype.inflammation.q1.1.mtx.gz'); b=load('subtype.inflammation.q1.2.mtx.gz')
for label,x in [('eye1',a),('eye2',b)]:
 lib=np.asarray(x.sum(0)).ravel()
 print('\\n',label)
 for g in ['IL1B','CCL2','CCL7','CXCL9','CXCL10','CXCL11','IL6','CSF3','S100A8','S100A9','NFKBIA','CD69','HLA-DRA','CD74','AIF1','TYROBP','P2RY12','TMEM119','C1QA','C1QB','C1QC','APOE','TREM2','LST1','FCER1G','LYZ','CST3','CD68','MARCO','MRC1','MS4A7','ITGAM','ITGAX','FCGR3A']:
  i=pos[g]; vals=x.getrow(i).toarray().ravel(); ids=np.flatnonzero(vals)
  print(f'{g:8s} nz {len(ids):4d} max {vals.max():5.0f} total {vals.sum():7.0f} top',[(int(j),int(vals[j]),int(lib[j])) for j in ids[np.argsort(vals[ids])[-5:][::-1]]])
 # cells any inflammation gene
 infl=['IL1B','CCL2','CCL7','CXCL9','CXCL10','CXCL11','IL6','CSF3','S100A8','S100A9','NFKBIA','CCL3','CCL4','CCL20','CD69','TNF','TNFRSF9','HLA-DRA','HLA-DRB1','CD74']
 ids=[pos[g] for g in infl]
 score=np.asarray(x[ids,:].sum(axis=0)).ravel()
 print('inflamm score nonzero',np.count_nonzero(score),'top')
 for j in np.argsort(score)[-30:][::-1]:
  if score[j]==0: break
  nz=x[:,j].toarray().ravel(); top=np.argsort(nz)[-12:][::-1]
  print(int(j), 'score',int(score[j]),'lib',int(lib[j]), 'genes',[(genes[i],int(nz[i])) for i in top if nz[i]>0])
PY"
