/bin/bash -lc "python3 - <<'PY'
import math, csv
# load sample genotypes keyed by pos ref alt
q={}
for line in open('query_variants.tsv'):
    chrom,pos,ref,alt,*gts=line.strip().split('\\t')
    q[(chrom,pos,ref,alt)]=gts
pops=['AFR','EUR','EAS','SAS','AMR']
logs=[[0.0]*len(pops) for _ in range(4)]
counts=[0]*4
n=0
for line in open('refs/1000g_chr1_overlap_af.tsv'):
    p=line.strip().split('\\t')
    if len(p)<9: continue
    chrom,pos,ref,alt=p[:4]
    vals=p[4:9]
    key=(chrom,pos,ref,alt)
    if key not in q: continue
    freqs=[]
    ok=True
    for v in vals:
        try: freqs.append(float(v))
        except: ok=False
    if not ok: continue
    gts=q[key]
    n+=1
    for i,gt in enumerate(gts):
        alleles=gt.replace('|','/').split('/')
        if '.' in alleles: continue
        g=sum(1 for a in alleles if a"'!='"'0')
        if g==0: continue
        counts[i]+=1
        for j,pop in enumerate(pops):
            f=max(min(freqs[j],1-1e-6),1e-6)
            # conditional on non-ref call at site
            den=1-(1-f)**2
            prob=(2*f*(1-f) if g==1 else f*f)/den
            logs[i][j]+=math.log(max(prob,1e-300))
print('matched',n,'counts',counts)
for i in range(4):
    ranked=sorted(zip(pops,logs[i]), key=lambda x:x[1], reverse=True)
    print('Sample',i+1, ranked, 'delta', ranked[0][1]-ranked[1][1])
PY"
