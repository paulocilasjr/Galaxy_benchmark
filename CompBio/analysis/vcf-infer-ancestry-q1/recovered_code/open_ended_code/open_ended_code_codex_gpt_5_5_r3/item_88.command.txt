/bin/bash -lc "python3 - <<'PY'
import math
sel={}
for line in open('selected_lifted.tsv'):
    chrom,pos,ref,alt,hg38pos,*gs=line.strip().split('\\t')
    sel[(chrom,pos,ref,alt)]=[int(x) for x in gs]
pops=['AFR','EUR','EAS','SAS','AMR']
logs=[[0.0]*len(pops) for _ in range(4)]; counts=[0]*4; used=0; bad=0
for line in open('refs/1000g_selected_af.tsv'):
    p=line.strip().split('\\t')
    if len(p)<9: continue
    chrom,pos,ref,alts=p[:4]
    altlist=alts.split(',')
    # find if any selected alt matches
    found=None
    for alt in altlist:
        key=(chrom,pos,ref,alt)
        if key in sel:
            found=key; break
    if not found:
        bad+=1; continue
    idx=altlist.index(found[3])
    vals=[]; ok=True
    for v in p[4:9]:
        try:
            # if multi AF comma, take matching allele index
            vals.append(float(v.split(',')[idx]))
        except Exception:
            ok=False
    if not ok: continue
    used+=1
    ugs=sel[found]
    for i,u in enumerate(ugs):
        counts[i]+=1
        for j,f in enumerate(vals):
            f=max(min(f,1-1e-6),1e-6)
            den=1-(1-f)**2
            prob=(2*f*(1-f) if u==1 else f*f)/den
            logs[i][j]+=math.log(max(prob,1e-300))
print('used',used,'bad',bad,'counts',counts)
for i in range(4):
    ranked=sorted(zip(pops,logs[i]), key=lambda x:x[1], reverse=True)
    print('Sample',i+1,ranked,'delta',ranked[0][1]-ranked[1][1])
PY"
