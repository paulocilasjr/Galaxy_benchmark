/bin/bash -lc "python3 - <<'PY'
import gzip
from pyliftover import LiftOver
lo=LiftOver('refs/hg19ToHg38.over.chain.gz')
records=[]; last=-10**9
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line[0]=='#': continue
        p=line.rstrip().split('\\t')
        chrom,pos,ref,alt=p[0],int(p[1]),p[3],p[4]
        if chrom"'!='"'1' or len(ref)"'!=1 or len(alt)!=1 or '"',' in alt: continue
        g=[]
        for field in p[9:]:
            gt=field.split(':',1)[0].replace('|','/')
            if '.' in gt: break
            c=sum(1 for a in gt.split('/') if a"'!='"'0')
            g.append(c)
        if len(g)"'!=4 or 0 in g or len(set(g))==1: continue
        if pos-last < 5000: continue
        records.append((chrom,pos,ref,alt,g)); last=pos
        if len(records)>=1500: break
with open('"'selected1500_hg19.tsv','w') as out, open('selected1500_hg19.bed','w') as bed:
    for chrom,pos,ref,alt,g in records:
        out.write('\\t'.join(map(str,[chrom,pos,ref,alt]+g))+'\\n')
        bed.write(f'{chrom}\\t{pos-1}\\t{pos}\\n')
print(len(records))
PY
bcftools query -R selected1500_hg19.bed -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%INFO/AFR_AF\\t%INFO/EUR_AF\\t%INFO/EAS_AF\\t%INFO/SAS_AF\\t%INFO/AMR_AF\\n' https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz > refs/1000g_selected1500_af.tsv
wc -l refs/1000g_selected1500_af.tsv"
