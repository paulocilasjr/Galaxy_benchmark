/bin/bash -lc 'out=refs/gnomad_selected_af_batched.tsv
: > "$out"
dir=af_chunks_$(date +%s)
mkdir "$dir"
split -l 25 selected_hg38.bed "$dir/chunk_"
fmt='"'%CHROM\\t%POS\\t%REF\\t%ALT\\t%INFO/gnomad_AF_afr\\t%INFO/gnomad_AF_asj\\t%INFO/gnomad_AF_nfe\\t%INFO/gnomad_AF_eas\\t%INFO/gnomad_AF_mid\\t%INFO/gnomad_AF_sas\\t%INFO/AC_papuan\\t%INFO/AN_papuan\\t%INFO/AC_melanesian\\t%INFO/AN_melanesian\\n'
for f in \""'$dir"/chunk_*; do
  ok=0
  for t in 1 2 3; do
    tmp="$f.out"
    if bcftools query -R "$f" -f "$fmt" https://storage.googleapis.com/gcp-public-data--gnomad/release/3.1.2/vcf/genomes/gnomad.genomes.v3.1.2.hgdp_tgp.chr1.vcf.bgz > "$tmp"; then
      cat "$tmp" >> "$out"
      ok=1
      break
    fi
    sleep 1
  done
  if [ "$ok" -ne 1 ]; then echo "FAILED $f" >&2; fi
done
wc -l "$out"'
