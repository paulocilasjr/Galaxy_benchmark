/bin/bash -lc "python3 - <<'PY'
import gzip
n=0
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f, open('biallelic_sites.bed','w') as bed, open('query_variants.tsv','w') as q:
    for line in f:
        if line[0]=='#': continue
        p=line.rstrip().split('\\t')
        chrom,pos,ref,alt=p[0],int(p[1]),p[3],p[4]
        if len(ref)==1 and len(alt)==1 and ',' not in alt:
            bed.write(f'{chrom}\\t{pos-1}\\t{pos}\\n')
            gts=[x.split(':',1)[0] for x in p[9:]]
            q.write('\\t'.join([chrom,str(pos),ref,alt]+gts)+'\\n')
            n+=1
print(n)
PY
bcftools query -R biallelic_sites.bed -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%INFO/AFR_AF\\t%INFO/EUR_AF\\t%INFO/EAS_AF\\t%INFO/SAS_AF\\t%INFO/AMR_AF\\n' https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz > refs/1000g_chr1_overlap_af.tsv
wc -l refs/1000g_chr1_overlap_af.tsv
head refs/1000g_chr1_overlap_af.tsv"
