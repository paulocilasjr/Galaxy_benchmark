/bin/bash -lc "cat > score_hgdp.py <<'PY'
import numpy as np
z=np.load('matched.npz');pos=z['pos'];g=z['g'];q=z['q'];r=z['refg'];meta=z['meta']
print('Data',r.shape,'genotypes',np.unique(r))
for col in [5,2]:
 groups=sorted(set(meta[:,col]));ll=[]
 for pop in groups:
  a=r[:,meta[:,col]==pop];n=(a>=0).sum(axis=1)*2;k=np.maximum(a,0).sum(axis=1)
  p=(k+0.5)/(n+1)
  ph=np.clip(2*(1-p)/(2-p),0.005,0.995)
  ll.append(np.where(g==1,np.log(ph[:,None]),np.log1p(-ph[:,None])))
 ll=np.array(ll)
 for thin in [10000,50000]:
  keep=[];last=-1e9
  for i,p in enumerate(pos):
   if p-last>=thin and min(q[i])>=30 and p>1000000 and (p<120000000 or p>145000000):keep.append(i);last=p
  scores=ll[:,keep,:].sum(axis=1)
  print('group',col,'spacing',thin,'markers',len(keep))
  for s in range(4):
   ranks=np.argsort(-scores[:,s]);best=scores[ranks[0],s]
   print('Sample',s+1,[(groups[x],round(scores[x,s]-best,1)) for x in ranks[:12]])
PY
PYTHONPATH=./packages python3 score_hgdp.py"
