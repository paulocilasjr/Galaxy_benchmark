/bin/bash -lc "python3 - <<'PY'
# cluster file
import pandas as pd
info=pd.read_csv('hgdp_1kgp.info.txt',sep='\\t')
# read merged.fam order
fam=pd.read_csv('merged.fam',sep='\\s+',header=None,names=['FID','IID','PAT','MAT','SEX','PHENO'])
clusters=[]
for _,r in fam.iterrows():
    if r['IID'].startswith('Sample'):
        clusters.append((r['FID'],r['IID'],'MYID'))
    else:
        # info lookup
        match=info[info['IID']==r['IID']]
        if len(match)==1:
            clusters.append((r['FID'],r['IID'],'HGDP_1kGP'))
        else:
            clusters.append((r['FID'],r['IID'],'UNKNOWN'))
with open('merged.cluster','w') as f:
    for c in clusters: f.write('\\t'.join(map(str,c))+'\\n')
print('rows',len(clusters), 'query',sum(1 for c in clusters if c[1].startswith('Sample')))
# create pop/region mapping for centroids later
PY
./plink --bfile merged --pca --within merged.cluster --pca-cluster-names HGDP_1kGP --out merged_pca --allow-no-sex --allow-extra-chr 2>&1 | tail -n 80"
