/bin/bash -lc "python3 - <<'PY'
import gzip
ours=set()
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.split('\\t')
        # store pos only
        ours.add((p[0],int(p[1])))
ref=[]
for line in open('HGDP_1kGP.exon_UTRS.bim'):
    p=line.rstrip('\\n').split('\\t')
    ref.append((p[0],int(p[3]),p[4],p[5]))
print('ours',len(ours),'ref chr1',sum(1 for x in ref if x[0]=='1'))
rset={(x[0],x[1]) for x in ref if x[0]=='1'}
print('overlap pos',len(ours & rset))
print(list(ours & rset)[:20])
PY"
