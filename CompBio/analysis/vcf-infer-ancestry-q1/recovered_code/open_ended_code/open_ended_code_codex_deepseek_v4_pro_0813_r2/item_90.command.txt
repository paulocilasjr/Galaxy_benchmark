/bin/bash -lc "python3 - <<'PY'
import numpy as np, pandas as pd
ids=pd.read_csv('merged_dist.mdist.id',sep='\\t',header=None,names=['FID','IID'])['IID'].tolist()
D=np.loadtxt('merged_dist.mdist')
info=pd.read_csv('hgdp_1kgp.info.txt',sep='\\t')
sample_indices=[ids.index(s) for s in ['Sample1','Sample2','Sample3','Sample4']]
print('sample indices',sample_indices)
for si,label in zip(sample_indices,['Sample1','Sample2','Sample3','Sample4']):
    row=D[si]
    order=np.argsort(row)
    print('\\nQuery',label)
    shown=0
    for idx in order:
        iid=ids[idx]
        if iid.startswith('Sample'): continue
        pop=''
        m=info[info['IID']==iid]
        if len(m): pop=f\"{m.iloc[0]['POP']}/{m.iloc[0]['myREG']}\"
        print(f'{iid:12s} dist={row[idx]:.6f} IBS={1-row[idx]:.6f} {pop}')
        shown+=1
        if shown>=20: break
PY"
