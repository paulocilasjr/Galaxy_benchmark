/bin/bash -lc "python3 - <<'PY'
import gzip, collections
samples=None
pair_same=collections.Counter(); pair_shared=collections.Counter(); pair_ibd=collections.Counter(); pair_shared_alleles=collections.Counter()
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f:
    for line in f:
        if line.startswith('##'): continue
        if line.startswith('#CHROM'):
            h=line.strip().split('\\t'); samples=h[9:]; continue
        p=line.strip().split('\\t')
        g=[]
        for x in p[9:]:
            gt=x.split(':')[0]
            if '.' in gt: g.append(None)
            else:
                a=gt.replace('|','/').split('/')
                # diploid; multiallelic sum works for dosage but for allele matching maybe not
                g.append(tuple(int(z) for z in a[:2]))
        for i in range(4):
            for j in range(i+1,4):
                if g[i] is None or g[j] is None: continue
                pair_shared[(i,j)]+=1
                if g[i]==g[j]: pair_same[(i,j)]+=1
                # IBS0/1/2: count alleles matching ignoring order? use multiset
                ci=collections.Counter(g[i]); cj=collections.Counter(g[j]); shared=0
                for k,v in ci.items(): shared += min(v,cj.get(k,0))
                pair_shared_alleles[(i,j)] += shared
print('pairwise same fraction')
for k in sorted(pair_same):
    i,j=k; den=pair_shared[k]
    print(samples[i],samples[j], pair_same[k], den, pair_same[k]/den, 'mean shared alleles', pair_shared_alleles[k]/(2*den))
PY"
