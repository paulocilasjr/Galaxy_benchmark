/bin/bash -lc "python3 - <<'PY'
import requests, json
# search repo tree recursively for info.txt or metadata
url='https://api.github.com/repos/JianingYao/scRNA-seq-genetic-ancestry/git/trees/main?recursive=1'
r=requests.get(url).json()
for item in r.get('tree',[]):
    p=item['path']
    if 'info' in p.lower() or 'pop' in p.lower() or 'region' in p.lower() or 'meta' in p.lower() or '1kg' in p.lower() or 'hgdp' in p.lower():
        print(p, item.get('size'))
PY"
