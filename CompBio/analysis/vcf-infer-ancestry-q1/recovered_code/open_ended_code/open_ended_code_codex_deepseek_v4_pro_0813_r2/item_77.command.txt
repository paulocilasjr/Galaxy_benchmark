/bin/bash -lc "python3 - <<'PY'
ref=[]
for line in open('HGDP_1kGP.exon_UTRS.bim'):
    p=line.rstrip('\\n').split('\\t'); ref.append((p[0],int(p[3]),frozenset([p[4],p[5]]),p[1]))
qmap={}
for line in open('query_simple.bim'):
    p=line.rstrip('\\n').split('\\t'); qmap[(p[0],int(p[3]),frozenset([p[4],p[5]]))]=p[1]
ren=[]; common=[]
for chrom,pos,alleles,rid in ref:
    qid=qmap.get((chrom,pos,alleles))
    if qid:
        ren.append((qid,rid)); common.append(rid)
print(len(ren))
open('query_simple.rename','w').write('\\n'.join(f'{q}\\t{r}' for q,r in ren)+'\\n')
open('common.ids','w').write('\\n'.join(common)+'\\n')
PY
head query_simple.rename; wc -l query_simple.rename common.ids"
