/bin/bash -lc "python3 - <<'PY'
import pandas as pd
panel=pd.read_csv('1kg.panel', sep='\\t')
samples=open('1kg.samples.txt').read().splitlines()
import numpy as np
sum_ibs=np.load('1kg_match_ibs.npy'); count=np.load('1kg_match_count.npy'); exact=np.load('1kg_match_exact.npy')
valid_count=count//4
print('count range',valid_count.min(),valid_count.max())
for qi in range(4):
    avg=sum_ibs[qi]/(2*valid_count)
    ex=exact[qi]/valid_count
    order=np.argsort(-avg, kind='stable')[:30]
    print('\\nSample',qi+1,'top avg IBS')
    for j in order:
        pop=panel.set_index('sample').loc[samples[j]]
        print(f'{j:4d} {samples[j]:8s} pop={pop[\"pop\"]:3s} super={pop[\"super_pop\"]:4s} avgIBS={avg[j]:.6f} exact={ex[j]:.6f}')
PY"
