/bin/bash -c "python3 - <<'PY'
import requests,re
for pop in ['PapuanHighlands','Bedouin','Yoruba','French']:
 u=f'https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/HGDP/data/{pop}/'
 r=requests.get(u,timeout=30)
 print('\\n',pop,r.status_code)
 for a in re.findall(r'href=\"(["'^"]+)"'"',r.text):
  if not a.startswith('?'): print(a)
PY"
