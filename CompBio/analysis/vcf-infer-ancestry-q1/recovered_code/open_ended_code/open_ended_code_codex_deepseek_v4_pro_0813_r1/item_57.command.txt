/bin/bash -c "python3 - <<'PY'
import gzip, pickle, numpy as np, math
G=np.load('samples.G.npy')
with open('sites.pkl','rb') as f: sites=pickle.load(f)
refaf={}
with open('1000g_overlap_af.tsv') as f:
 for line in f:
  parts=line.rstrip().split('\\t')
  chrom,pos,ref,alt,afr,eur,eas,sas,amr=parts
  refaf[int(pos)]=(ref,alt,[afr,eur,eas,sas,amr])
labels=['AFR','EUR','EAS','SAS','AMR']
LL=np.zeros((4,5)); counts=np.zeros((4,5),dtype=int)
used=0; refskip=0; multiskip=0; altskip=0
for i,(c,pos,ref,alts) in enumerate(sites):
 if pos not in refaf: continue
 rref,ralt,afstr=refaf[pos]
 if ref "'!= rref:
  refskip+=1; continue
 if len(alts)>1:
  multiskip+=1; continue
 ralts=ralt.split('"',')
 if alts[0]"'!=ralts[0]:
  altskip+=1; continue
 vals=[]
 for s in afstr:
  v=s.split('"',')[0]
  try: v=float(v)
  except: v=0.0
  v=min(max(v,1e-6),1-1e-6)
  vals.append(v)
 used+=1
 for j in range(4):
  g=int(G[i,j])
  if g==3: continue
  for k,p in enumerate(vals):
   if g==0: lp=2*math.log(1-p)
   elif g==1: lp=math.log(2)+math.log(p)+math.log(1-p)
   else: lp=2*math.log(p)
   LL[j,k]+=lp; counts[j,k]+=1
print('used',used,'refskip',refskip,'multiskip',multiskip,'altskip',altskip)
print('counts per sample:')
for j in range(4): print('S',j+1,counts[j])
print('LL matrix:')
print(LL)
print('best labels:')
for j in range(4):
 best=int(np.argmax(LL[j])); print('Sample',j+1,labels[best], 'LLs', [round(x,1) for x in LL[j]])
PY"
