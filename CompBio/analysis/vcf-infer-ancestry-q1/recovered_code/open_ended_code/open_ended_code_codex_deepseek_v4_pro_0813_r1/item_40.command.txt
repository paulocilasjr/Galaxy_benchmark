/bin/bash -c "# generate target positions file one per line for bcftools -T (chr pos)
python3 - <<'PY'
import gzip
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f, open('first1000.targets.txt','w') as out:
 n=0
 for line in f:
  if line.startswith('#'): continue
  c,p=line.split('\\t')[:2]
  out.write(f'{c}\\t{p}\\n')
  n+=1
  if n>=1000: break
PY
/usr/bin/time -p bcftools query -T first1000.targets.txt -f '%POS\\n' https://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/ALL.chr1.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz > [REDACTED LOCAL PATH]
wc -l [REDACTED LOCAL PATH] head [REDACTED LOCAL PATH]"
