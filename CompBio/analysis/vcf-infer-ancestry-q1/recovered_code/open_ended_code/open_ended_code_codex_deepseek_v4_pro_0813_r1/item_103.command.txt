/bin/bash -c "python3 - <<'PY'
import pickle,numpy as np
G=np.load('samples.G.npy')
with open('sites.pkl','rb') as f:sites=pickle.load(f)
refaf={}
for line in open('1000g_overlap_af.tsv'):
 p=line.rstrip().split('\\t'); refaf[int(p[1])]=(p[2],p[3],p[4:9])
labels=['AFR','EUR','EAS','SAS','AMR']
# collect arrays for biallelic matching
arrs=[[] for _ in range(4)]
exp=[[] for _ in range(5)]
for i,(c,pos,ref,alts) in enumerate(sites):
 if pos not in refaf: continue
 rref,ralt,afstr=refaf[pos]
 if ref"'!=rref or len(alts)>1 or alts[0]!=ralt.split('"',')[0]: continue
 vals=[]
 for s in afstr:
  v=float(s.split(',')[0]); v=min(max(v,1e-6),1-1e-6); vals.append(v)
 for j in range(4):
  g=int(G[i,j]);
  if g==3: continue
  arrs[j].append(g)
 for k,v in enumerate(vals): exp[k].append(2*v/(2-v))
for j in range(4):
 a=np.array(arrs[j],float); cors=[]
 for k in range(5):
  e=np.array(exp[k]); cors.append(np.corrcoef(a,e)[0,1])
 print('S',j+1,['%.3f'%c for c in cors],'best',labels[int(np.argmax(cors))])
PY"
