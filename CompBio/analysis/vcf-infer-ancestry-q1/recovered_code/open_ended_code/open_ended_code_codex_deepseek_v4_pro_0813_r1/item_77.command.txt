/bin/bash -c "python3 - <<'PY'
import requests
url='https://gnomad.broadinstitute.org/api'
queries={
'v2_genome_pop':'query("'$variantId:String!,$dataset:DatasetId!){variant(variantId:$variantId,dataset:$dataset){variant_id genome{ac an populations{id ac an homozygote_count}}}}'"',
'v2_exome_pop':'query("'$variantId:String!,$dataset:DatasetId!){variant(variantId:$variantId,dataset:$dataset){variant_id exome{ac an populations{id ac an homozygote_count}}}}'"',
'v4_genome_pop':'query("'$variantId:String!,$dataset:DatasetId!){variant(variantId:$variantId,dataset:$dataset){variant_id genome{ac an populations{id ac an homozygote_count}}}}'"',
}
for name,q in queries.items():
 r=requests.post(url,json={'query':q,'variables':{'variantId':'1-13116-T-G','dataset':'gnomad_r2_1' if name.startswith('v2') else 'gnomad_r4'}},timeout=30)
 print(name,r.status_code,r.text[:800])
PY"
