/bin/bash -c "python3 - <<'PY'
import pickle, numpy as np, math
G=np.load('samples.G.npy')
with open('sites.pkl','rb') as f: sites=pickle.load(f)
refaf={}
for line in open('1000g_overlap_af.tsv'):
 p=line.rstrip().split('\\t'); refaf[int(p[1])]=(p[2],p[3],p[4:9])
labels=['AFR','EUR','EAS','SAS','AMR']
LL=np.zeros((4,5))
used=0
for i,(c,pos,ref,alts) in enumerate(sites):
 if pos not in refaf: continue
 rref,ralt,afstr=refaf[pos]
 if ref"'!=rref or len(alts)>1: continue
 if alts[0]!=ralt.split('"',')[0]: continue
 vals=[]
 for s in afstr:
  v=float(s.split(',')[0])
  v=min(max(v,1e-6),1-1e-6); vals.append(v)
 used+=1
 for j in range(4):
  g=int(G[i,j])
  if g==3: continue
  for k,p in enumerate(vals):
   # conditional on non-ref
   if g==1: lp=math.log(2*(1-p)/(2-p))
   elif g==2: lp=math.log(p/(2-p))
   else: lp=-1e9
   LL[j,k]+=lp
print('used',used)
print(LL)
for j in range(4): print('S',j+1,labels[int(np.argmax(LL[j]))], [round(x,1) for x in LL[j]])
PY"
