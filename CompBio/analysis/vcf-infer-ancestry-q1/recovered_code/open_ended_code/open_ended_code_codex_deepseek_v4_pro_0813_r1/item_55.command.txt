/bin/bash -c "python3 - <<'PY'
import gzip
with gzip.open('vcf.infer.ancestry.q1.vcf.gz','rt') as f, open('oursites.tsv','w') as out:
 for line in f:
  if line.startswith('#'): continue
  c,p=line.split('\\t')[:2]
  out.write(f'{c}\\t{p}\\n')
PY
/usr/bin/time -p bcftools query -T oursites.tsv -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%INFO/AFR_AF\\t%INFO/EUR_AF\\t%INFO/EAS_AF\\t%INFO/SAS_AF\\t%INFO/AMR_AF\\n' 1000G.chr1.v5a.vcf.gz > 1000g_overlap_af.tsv
wc -l 1000g_overlap_af.tsv; head 1000g_overlap_af.tsv"
