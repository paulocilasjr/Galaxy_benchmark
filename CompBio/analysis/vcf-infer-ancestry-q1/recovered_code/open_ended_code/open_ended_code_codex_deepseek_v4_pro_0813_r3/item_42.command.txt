/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
cols=['FID','IID']+[f'PC{i}' for i in range(1,11)]
vec=pd.read_csv('pca_all.eigenvec', sep=r'\\s+', header=None, names=cols)
ps=pd.read_csv('phase3_corrected.psam', sep='\\t', comment='#', header=None, names=['IID','PAT','MAT','SEX','SuperPop','Population'])
mp=dict(zip(ps['IID'], ps['SuperPop']))
vec['pop']=vec.IID.map(mp)
print(vec[vec.IID.str.startswith('Sample')].to_string())
pc=[f'PC{i}' for i in range(1,7)]
for p in pc:
 print('\\n',p)
 print(vec.groupby('pop')[p].agg(['mean','std','min','max']).sort_values('mean').to_string())
samples=vec[vec.IID.str.startswith('Sample')][pc].values
g=vec.groupby('pop')[pc].mean()
pops=g.index
print('\\nsample correlations over PC1-6')
for j,sid in enumerate(['Sample1','Sample2','Sample3','Sample4']):
    cors=[np.corrcoef(samples[j], g.loc[p].values)[0,1] for p in pops]
    print(sid, list(zip(pops,np.round(cors,3))))
print('\\nEuclidean distances to means PC1-6')
for j,sid in enumerate(['Sample1','Sample2','Sample3','Sample4']):
    ds=[np.linalg.norm(samples[j]-g.loc[p].values) for p in pops]
    print(sid, list(zip(pops,np.round(ds,4))))
PY"
