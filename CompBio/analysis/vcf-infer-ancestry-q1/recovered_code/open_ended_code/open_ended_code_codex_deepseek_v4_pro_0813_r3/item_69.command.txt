/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
cols=['FID','IID']+[f'PC{i}' for i in range(1,11)]
vec=pd.read_csv('pca_hgdp.eigenvec', sep=r'\\s+', header=None, names=cols)
ps=pd.read_csv('hgdp.psam', sep='\\t', comment='#')
# columns from hgdp.psam: #IID SEX population latitude longitude region
mp=dict(zip(ps['#IID'], ps['region']))
pop=dict(zip(ps['#IID'], ps['population']))
vec['region']=vec.IID.map(mp)
vec['pop']=vec.IID.map(pop)
print(vec[vec.IID.str.startswith('Sample')].to_string())
for p in ['PC1','PC2','PC3','PC4','PC5','PC6']:
 print('\\n',p)
 print(vec.groupby('region')[p].agg(['mean','std','min','max']).sort_values('mean').to_string())
pc=['PC1','PC2','PC3','PC4','PC5','PC6']
samples=vec[vec.IID.str.startswith('Sample')][pc].values
g=vec.groupby('region')[pc].mean()
regs=g.index
print('\\nsample correlations PC1-6')
for j,sid in enumerate(['Sample1','Sample2','Sample3','Sample4']):
    cors=[np.corrcoef(samples[j], g.loc[p].values)[0,1] for p in regs]
    print(sid, list(zip(regs,np.round(cors,3))))
print('\\nEuclidean distances')
for j,sid in enumerate(['Sample1','Sample2','Sample3','Sample4']):
    ds=[np.linalg.norm(samples[j]-g.loc[p].values) for p in regs]
    print(sid, list(zip(regs,np.round(ds,4))))
PY"
