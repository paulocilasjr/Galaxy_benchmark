/bin/bash -lc "python3 - <<'PY'
# Build q1 set by pos + REF + ALT set; output ref IDs where matching.
import sys
qset={}
with open('q1id.pvar') as f:
    for line in f:
        if line.startswith('#'):
            continue
        c=line.rstrip('\\n').split('\\t')
        if len(c)<5: continue
        chrom,pos,vid,ref,alt=c[:5]
        alts=set(alt.split(','))
        # store each alt
        qset.setdefault((chrom,pos,ref),set()).update(alts)
print('qset keys',len(qset))
out=open('ref_extract_ids.txt','w')
n=0
# process ref pvar, maybe big; first pass header skip
with open('chr1_phase3.pvar') as f:
    for line in f:
        if line.startswith('#'):
            continue
        c=line.rstrip('\\n').split('\\t')
        if len(c)<5: continue
        chrom,pos,rid,ref,alt=c[:5]
        if (chrom,pos,ref) in qset:
            alts=set(alt.split(','))
            # output if any q alt matches any ref alt, or ref multi
            if alts & qset[(chrom,pos,ref)]:
                out.write(rid+'\\n')
                n+=1
out.close()
print('ref matching IDs',n)
PY
wc -l ref_extract_ids.txt; head ref_extract_ids.txt"
