/bin/bash -lc "python3 - <<'PY'
# create q1_to_rsid.txt mapping current q1id IDs to 1KG rsIDs where pos/ref/alt matches
qdict={}
with open('q1id.pvar') as f:
    for line in f:
        if line.startswith('#'): continue
        c=line.rstrip('\\n').split('\\t')
        if len(c)<5: continue
        chrom,pos,vid,ref,alt=c[:5]
        qdict.setdefault((chrom,pos,ref,alt),[]).append(vid)
out=open('q1_to_rsid.txt','w')
seen=set()
with open('chr1_phase3.pvar') as f:
    for line in f:
        if line.startswith('#'): continue
        c=line.rstrip('\\n').split('\\t')
        if len(c)<5: continue
        chrom,pos,rsid,ref,alt=c[:5]
        key=(chrom,pos,ref,alt)
        if key in qdict:
            for vid in qdict[key]:
                if vid not in seen:
                    out.write(f'{vid}\\t{rsid}\\n')
                    seen.add(vid)
out.close()
print('mapped q1 variants',len(seen))
PY
head q1_to_rsid.txt; wc -l q1_to_rsid.txt"
