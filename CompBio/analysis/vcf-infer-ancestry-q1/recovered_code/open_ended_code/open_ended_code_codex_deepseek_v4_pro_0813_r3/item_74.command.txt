/bin/bash -lc "python3 - <<'PY'
import numpy as np
for name in ['dist_1kg','dist_hgdp']:
    with open(name+'.mdist.id') as f: ids=[line.strip() for line in f]
    rows=[]
    with open(name+'.mdist') as f:
        for line in f:
            rows.append([float(x) for x in line.split()])
    print(name,'ids',len(ids),'rows',len(rows))
    for sid in ['Sample1','Sample2','Sample3','Sample4']:
        if sid not in ids:
            print(sid,'not found'); continue
        i=ids.index(sid)
        # row i has vals 0..i
        vals=rows[i]
        # for j>i need column j row j element i
        dist=[]
        for j,v in enumerate(vals): dist.append((ids[j],v))
        for j in range(i+1,len(ids)):
            dist.append((ids[j], rows[j][i]))
        dist=np.array([d[1] for d in dist]); idx=np.argsort(dist)
        print('\\n',sid, 'top 10')
        for k in idx[:10]:
            print(ids[k], round(dist[k],6))
PY"
