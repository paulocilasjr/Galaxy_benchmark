/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
vec=pd.read_csv('pca_all.eigenvec', sep=r'\\s+', header=None, names=['FID','IID']+[f'PC{i}' for i in range(1,11)])
ps=pd.read_csv('phase3_corrected.psam', sep='\\t', comment='#')
print(ps.columns.tolist())
mp=dict(zip(ps['IID'], ps['SuperPop']))
vec['pop']=vec.IID.map(mp)
print(vec[vec.IID.str.startswith('Sample')].to_string())
for i in range(1,7):
 print('\\nPC',i)
 print(vec.groupby('pop')[f'PC{i}'].agg(['mean','std','min','max']).sort_values('mean').to_string())
samples=vec[vec.IID.str.startswith('Sample')].iloc[:,2:12].values
g=vec.groupby('pop').mean().iloc[:,2:12]
pops=g.index
print('\\nsample correlations')
for j,sid in enumerate(['Sample1','Sample2','Sample3','Sample4']):
    cors=[np.corrcoef(samples[j], g.loc[p].values)[0,1] for p in pops]
    print(sid, list(zip(pops,np.round(cors,3))))
PY"
