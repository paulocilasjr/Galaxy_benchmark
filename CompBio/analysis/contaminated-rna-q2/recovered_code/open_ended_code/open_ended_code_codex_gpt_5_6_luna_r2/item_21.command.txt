/bin/bash -lc "curl -sS --max-time 30 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data 'CMD=Put' --data 'PROGRAM=blastn' --data 'DATABASE=nt' --data 'MEGABLAST=on' --data 'FORMAT_TYPE=XML' --data-urlencode 'QUERY@known.fa' > blast_put.html
grep -E 'RID|RTOE|ERROR|error|Message' blast_put.html | tail -n 20"
