/bin/bash -lc "gzip -cd contaminated.rna.q2.fq.gz | awk 'NR%4==1{h=substr("'$0,2); n++} NR%4==2{s=$0} NR%4==0 && n%1000==1{print ">" h "'"\\n\" s}' > sample_even.fa
printf 'queries: '; grep -c '"'^>'"' sample_even.fa
curl -sS --max-time 60 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data 'CMD=Put' --data 'PROGRAM=blastn' --data 'DATABASE=nt' --data 'MEGABLAST=on' --data 'FORMAT_TYPE=XML' --data-urlencode 'QUERY@sample_even.fa' > sample_even_put.html
grep -E 'RID =|RTOE =' sample_even_put.html | tail -n 5"
