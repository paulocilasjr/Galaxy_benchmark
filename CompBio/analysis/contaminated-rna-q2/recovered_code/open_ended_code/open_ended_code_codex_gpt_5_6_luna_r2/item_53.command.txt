samtools fastq -f 4 -0 candidate_nonhuman.fq unmapped_rrna.bam 2> candidate_fastq.log

printf 'human-rRNA mapped: '; samtools view -c -F 4 unmapped_rrna.bam
printf 'candidate GC distribution:\\n'; awk 'NR%4==2{gc=gsub(/[GC]/,\"&\"); print gc}' candidate_nonhuman.fq | sort -n | uniq -c | awk '{printf \"%2d%% %6d\\n\","'$2*100/75,$1}'"' | head -n 100"
