/bin/bash -lc "for item in 'human NC_012920' 'chimp NC_001643' 'bonobo NC_001644' 'gorilla NC_001645' 'macaque NC_005943' 'mouse NC_005089'; do set -- "'$item; printf '"'> %s\\n' \""'$1"; curl -L --fail --max-time 30 -sS "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=$2&rettype=fasta&retmode=text" | sed '"'1d'; done > primate_mt.fa
sed -n '1,12p' primate_mt.fa
awk '/"'^>/{if(seq){print name,length(seq)};name=$0;seq="";next}{seq=seq $0}END{print name,length(seq)}'"' primate_mt.fa"
