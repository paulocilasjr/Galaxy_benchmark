/bin/bash -lc "awk 'NR%4==1{h=substr("'$0,2)} NR%4==2{s=$0} NR%4==0{n++; if(n%100==1) print ">"h "'"\\n\" s}' candidate_nonhuman.fq > candidate_sample.fa
printf 'queries: '; grep -c '"'^>'"' candidate_sample.fa
curl -sS --max-time 60 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data 'CMD=Put' --data 'PROGRAM=blastn' --data 'DATABASE=nt' --data 'MEGABLAST=on' --data 'FORMAT_TYPE=XML' --data-urlencode 'QUERY@candidate_sample.fa' > candidate_sample_put.html
grep -E 'RID =|RTOE =' candidate_sample_put.html | tail -n 5"
