/bin/bash -lc "awk 'NR%4==1{print \">\" substr("'$0,2)} NR%4==2{print}'"' candidate_nonhuman.fq > candidate_nonhuman.fa
blastn -query candidate_nonhuman.fa -subject primate_mt.fa -task blastn-short -dust no -evalue 1e-5 -max_target_seqs 20 -max_hsps 1 -outfmt '6 qseqid sseqid pident length qstart qend sstart send evalue bitscore' > candidate_mt.blast.tsv
printf 'alignments: '; wc -l < candidate_mt.blast.tsv
printf 'sample alignments:\\n'; head -n 30 candidate_mt.blast.tsv
printf 'target counts: '; cut -f2 candidate_mt.blast.tsv | sort | uniq -c | sort -nr"
