/bin/bash -lc "perl -pi -e 's/"'^> />/'"' primate_mt.fa
grep -n '"'^>'"' primate_mt.fa
minimap2 -ax sr -N 20 primate_mt.fa candidate_nonhuman.fq 2> candidate_mt.log | samtools view -b -o candidate_mt.bam -
printf 'mapped alignments: '; samtools view -c -F 4 candidate_mt.bam
printf 'primary target counts:\\n'; samtools view -F 256 -F 4 candidate_mt.bam | cut -f3 | sort | uniq -c | sort -nr
printf 'all target counts:\\n'; samtools view -F 4 candidate_mt.bam | cut -f3 | sort | uniq -c | sort -nr
cat candidate_mt.log"
