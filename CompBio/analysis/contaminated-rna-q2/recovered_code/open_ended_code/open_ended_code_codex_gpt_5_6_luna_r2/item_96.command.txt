/bin/bash -lc "awk '"'$2>=140 && $2>$3 && $2>$4 {print $1}'"' primate_scores.tsv > strict_chimp.ids
awk 'NR==FNR{keep["'$1]=1; next} /''^>/{h=substr($0,2); want=(h in keep)} want{print}'"' strict_chimp.ids candidate_nonhuman.fa > strict_chimp.fa
printf 'strict chimp queries: '; grep -c '"'^>'"' strict_chimp.fa
curl -sS --max-time 60 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' --data 'CMD=Put' --data 'PROGRAM=blastn' --data 'DATABASE=nt' --data 'MEGABLAST=on' --data 'FORMAT_TYPE=XML' --data 'HITLIST_SIZE=10' --data-urlencode 'QUERY@strict_chimp.fa' > strict_chimp_put.html
grep -E 'RID =|RTOE =' strict_chimp_put.html | tail -n 5"
