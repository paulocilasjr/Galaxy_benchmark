/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
best=defaultdict(lambda: {'HUMAN':(-1,None),'MOUSE':(-1,None),'RAT':(-1,None)})
for line in open('hmr111_all.paf'):
    f=line.rstrip().split('\\t')
    if len(f)<12: continue
    q=f[0]; sp=f[5].split('|',1)[0]
    if sp not in best[q]: continue
    matches=int(f[9]); block=int(f[10])
    if (matches,block)>best[q][sp][0]:
        best[q][sp]=(matches,f[5])
rat=[]
for q,d in best.items():
    vals={s:d[s][0] for s in d}
    ordered=sorted(vals.items(), key=lambda kv: kv[1], reverse=True)
    if ordered[0][0]=='RAT' and ordered[0][1]-ordered[1][1]>=5:
        rat.append((q, ordered[0][1]-ordered[1][1], d['RAT'][1], d['HUMAN'][0], d['MOUSE'][0]))
print(len(rat))
for x in rat[:10]: print(x)
PY"
