/bin/bash -lc "python3 - <<'PY'
# create id list of unclassified from kraken_unmapped_output
ids=set()
for line in open('kraken_unmapped_output.txt'):
    p=line.rstrip().split('\\t')
    if p[0]=='U':
        ids.add(p[1])
# extract from unmapped_human.fq
with open('unmapped_human.fq') as fin, open('unmapped_human_krakenU.fq','w') as out:
    while True:
        h=fin.readline().rstrip('\\n')
        if not h: break
        s=fin.readline().rstrip('\\n'); p=fin.readline().rstrip('\\n'); q=fin.readline().rstrip('\\n')
        rid=h[1:].split()[0]
        if rid in ids:
            out.write(f'{h}\\n{s}\\n{p}\\n{q}\\n')
print(len(ids))
PY
wc -l unmapped_human_krakenU.fq; python3 - <<'PY'
print(open('unmapped_human_krakenU.fq').read().count('@') // 4)
PY"
