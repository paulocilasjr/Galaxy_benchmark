/bin/bash -lc "python3 - <<'PY'
from collections import Counter
cnt=Counter(); human=0;mouse=0;unm=0;amb=0
strong_human=0;strong_mouse=0
# also count primary read flags 0,16,4; supplementary flags include 2048
for line in open('combined_bwa.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    flag=int(p[1]); ref=p[2]
    if flag & 0x800: # supplementary
        continue
    if ref=='*':
        unm+=1; cnt['unmapped']+=1; continue
    if ref.startswith('human_'): human+=1
    elif ref.startswith('mouse|'): mouse+=1
    else: amb+=1
    cnt[ref.split('|')[0]]+=1
    as_=None
    for tag in p[11:]:
        if tag.startswith('AS:i:'): as_=int(tag[5:])
    if as_ is not None and as_>=120:
        if ref.startswith('human_'): strong_human+=1
        elif ref.startswith('mouse|'): strong_mouse+=1
print('primary human',human,'mouse',mouse,'unmapped',unm,'amb',amb)
print('strong AS>=120 human',strong_human,'mouse',strong_mouse)
print(cnt.most_common(10))
# count mouse primary types
typecnt=Counter(); ex=[]
for line in open('combined_bwa.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    flag=int(p[1]); ref=p[2]
    if flag & 0x800 or ref=='*': continue
    if ref.startswith('mouse|'):
        rid=ref.split('|',1)[1]
        typ='NM' if rid.startswith('NM_') else 'XM' if rid.startswith('XM_') else 'NR' if rid.startswith('NR_') else 'XR' if rid.startswith('XR_') else 'other'
        typecnt[typ]+=1
        if len(ex)<10: ex.append((p[0],rid,flag))
print('mouse types',typecnt)
for e in ex: print(e)
PY"
