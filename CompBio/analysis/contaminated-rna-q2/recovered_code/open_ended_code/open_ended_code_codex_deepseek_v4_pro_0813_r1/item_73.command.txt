/bin/bash -lc "python3 - <<'PY'
# build mapping of read to human cDNA mapped status from SAM (flag not 4)
human_cdna_mapped=set()
for line in open('human_mapped.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    if int(p[1]) & 4 == 0:
        human_cdna_mapped.add(p[0])
# mouse primary reads from all_combined.sam (minimap2), type NM/XM, AS>=120
mouse_mrna=[]
for line in open('all_combined.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    flag=int(p[1]); ref=p[2]
    if flag & 0x100 or ref=='*': continue
    if ref.startswith('mouse|'):
        rid=ref.split('|',1)[1]
        if (rid.startswith('NM_') or rid.startswith('XM_')):
            as_=None
            for tag in p[11:]:
                if tag.startswith('AS:i:'): as_=int(tag[5:])
            if as_ and as_>=100:
                mouse_mrna.append((p[0],rid,as_))
print('mouse mRNA primary >=100',len(mouse_mrna))
print('of those also mapped to human cDNA:',sum(1 for x in mouse_mrna if x[0] in human_cdna_mapped))
for x in mouse_mrna[:20]: print(x)
PY"
