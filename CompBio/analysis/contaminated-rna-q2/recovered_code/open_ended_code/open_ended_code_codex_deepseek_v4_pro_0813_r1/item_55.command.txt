/bin/bash -lc "python3 - <<'PY'
# ids mapped to human ncRNA (primary or supplementary) from SAM flags? Use samtools view -F 4 unique qnames already.
import subprocess
# generate mapped id set via samtools view
p=subprocess.run(['/opt/miniconda3/envs/benchmark_codex_553074deb34f45ddb3da/bin/samtools','view','-F','4','human_ncrna_unmappedU.sam'],capture_output=True,text=True)
mapped=set(line.split('\\t')[0] for line in p.stdout.splitlines() if line)
# extract reads not in mapped from unmapped_human_krakenU.fq
with open('unmapped_human_krakenU.fq') as fin, open('remaining_nonhuman.fq','w') as out:
    while True:
        h=fin.readline().rstrip('\\n')
        if not h: break
        s=fin.readline().rstrip('\\n'); plus=fin.readline().rstrip('\\n'); q=fin.readline().rstrip('\\n')
        rid=h[1:].split()[0]
        if rid not in mapped:
            out.write(f'{h}\\n{s}\\n{plus}\\n{q}\\n')
print('mapped',len(mapped))
PY
wc -l remaining_nonhuman.fq"
