/bin/bash -lc "python3 - <<'PY'
import gzip
# map read id -> taxid from kraken
rid2tax={}
for line in open('kraken_output.txt'):
    p=line.rstrip().split('\\t')
    if p[0]=='C':
        rid2tax[p[1]]=p[2]
# parse fasta
from Bio import SeqIO
recs=list(SeqIO.parse('nonhuman_top.fa','fasta'))
for tax in ['10376','29382','1396','492670','83986','1279']:
    chosen=[r for r in recs if rid2tax.get(r.id)==tax][:5]
    SeqIO.write(chosen, f'sample_{tax}.fa','fasta')
PY
ls -lh sample_*.fa"
