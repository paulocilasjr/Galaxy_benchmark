/bin/bash -lc "python3 - <<'PY'
from collections import Counter
cnt=Counter(); strong=[]
for line in open('ebv_all.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t'); flag=int(p[1])
    if flag & 0x100: continue
    if p[2]=='*': cnt['unmapped']+=1
    else:
        cnt['mapped']+=1
        nm=None; as_=None
        for tag in p[11:]:
            if tag.startswith('NM:i:'): nm=int(tag[5:])
            if tag.startswith('AS:i:'): as_=int(tag[5:])
        if nm is not None and nm<=2: strong.append((p[0],as_,nm))
print(cnt,'strong',len(strong))
for e in strong[:30]: print(e)
PY"
