/bin/bash -lc "python3 - <<'PY'
from collections import Counter

def parse(sam):
    d={}
    for line in open(sam):
        if line.startswith('@'): continue
        p=line.rstrip().split('\\t')
        flag=int(p[1]); rid=p[0]
        if flag & 0x100: continue
        if p[2]=='*': d[rid]=None; continue
        as_=None; nm=None
        for tag in p[11:]:
            if tag.startswith('AS:i:'): as_=int(tag[5:])
            if tag.startswith('NM:i:'): nm=int(tag[5:])
        if rid not in d or (as_ or 0)>(d[rid][0] or 0): d[rid]=(as_,nm,p[2])
    return d
h=parse('human_all.sam'); m=parse('mouse_all.sam'); y=parse('yeast_all.sam')
cnt=Counter(); examples=[]
for rid in set(h)|set(m)|set(y):
    hv=h.get(rid); mv=m.get(rid); yv=y.get(rid)
    hs=hv[0] if hv else -1; ms=mv[0] if mv else -1; ys=yv[0] if yv else -1
    best=max(hs,ms,ys)
    if ys<0 and best<0: cat='neither'
    elif ys>hs and ys>ms: cat='yeast_best'
    elif ys==best and ys>=0 and (hs==ys or ms==ys): cat='tie_with_yeast'
    elif hs>=ms and hs>ys: cat='human_best'
    elif ms>hs and ms>ys: cat='mouse_best'
    else: cat='other'
    cnt[cat]+=1
    if cat=='yeast_best' and len(examples)<20:
        examples.append((rid,hs,ms,ys,hv[2] if hv else None,mv[2] if mv else None,yv[2] if yv else None,hv[1] if hv else None,mv[1] if mv else None,yv[1] if yv else None))
print(cnt)
print('yeast_best',cnt['yeast_best'])
for e in examples: print(e)
PY"
