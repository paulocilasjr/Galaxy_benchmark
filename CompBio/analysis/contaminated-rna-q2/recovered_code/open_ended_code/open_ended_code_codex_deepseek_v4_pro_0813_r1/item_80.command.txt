/bin/bash -lc "python3 - <<'PY'
from Bio import SeqIO
# write human_all
with open('human_all.fa','w') as out:
    for fn in ['human_cdna.fa','human_ncrna.fa']:
        for line in open(fn):
            out.write(line)
    for rec in SeqIO.parse('rrna_refs.fa','fasta'):
        if rec.id in ('NR_003286.4','NR_003287.4'):
            out.write(f'>{rec.id} human_rRNA\\n{rec.seq}\\n')
# mouse_all
with open('mouse_all.fa','w') as out:
    for line in open('mouse_rna.fa'):
        out.write(line)
    for rec in SeqIO.parse('rrna_refs.fa','fasta'):
        if rec.id in ('NR_003278.3','NR_003279.1'):
            out.write(f'>{rec.id} mouse_rRNA\\n{rec.seq}\\n')
PY
ls -lh human_all.fa mouse_all.fa"
