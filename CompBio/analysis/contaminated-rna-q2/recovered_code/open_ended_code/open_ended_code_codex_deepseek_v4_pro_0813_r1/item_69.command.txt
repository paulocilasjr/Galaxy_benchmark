/bin/bash -lc "python3 - <<'PY'
# prefix headers and concatenate
import gzip
files=['human_cdna.fa','human_ncrna.fa','mouse_rna.fa']
prefixes=['human_cdna|','human_ncrna|','mouse|']
with open('combined_human_mouse.fa','w') as out:
    for fn,pre in zip(files,prefixes):
        with open(fn) as f:
            for line in f:
                if line.startswith('>'):
                    out.write('>'+pre+line[1:])
                else:
                    out.write(line)
PY
