/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict
best={}
all_hits=defaultdict(list)
for line in open('remaining_rrna_blast.tsv'):
    p=line.rstrip().split('\\t')
    q,s,pid,length,mis,gap,qs,qe,ss,se,e,bits,qcov=p
    bits=float(bits); e=float(e)
    all_hits[q].append((s,bits,e,pid,length))
    key=(bits,-e)
    if q not in best or key>best[q][0]:
        best[q]=(key,s,bits,e,pid,length)
# classify human/mouse based on best
h=0;m=0;amb=0
h_ids=[];m_ids=[]
for q,(key,s,b,e,pid,l) in best.items():
    if s in ('NR_003286.4','NR_003287.4'): h+=1; h_ids.append(q)
    elif s in ('NR_003278.3','NR_003279.1'): m+=1; m_ids.append(q)
    else: amb+=1
print('queries with hits',len(best),'human best',h,'mouse best',m,'amb',amb)
# count top scores with ties: For each query, compare max human bits and max mouse bits
strong_mouse=0; strong_human=0; tie=0
for q, hits in all_hits.items():
    hb=max([b for s,b,e,pid,l in hits if s in ('NR_003286.4','NR_003287.4')], default=-1)
    mb=max([b for s,b,e,pid,l in hits if s in ('NR_003278.3','NR_003279.1')], default=-1)
    if hb<0 and mb>0: strong_mouse+=1
    elif mb<0 and hb>0: strong_human+=1
    elif hb>mb: strong_human+=1
    elif mb>hb: strong_mouse+=1
    else: tie+=1
print('strong human',strong_human,'strong mouse',strong_mouse,'tie/no',tie)
# among top hits by bits
from collections import Counter
print(Counter(best[q][1] for q in best))
# print some mouse-best examples with bits/pident
mouse_best=[(q,best[q]) for q in best if best[q][1].startswith('NR_00327')][:10]
for q,val in mouse_best: print('MOUSE_BEST',q,val)
PY"
