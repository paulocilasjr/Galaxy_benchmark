/bin/bash -lc "python3 - <<'PY'
import gzip
# Extract 34 staph reads from kraken output taxid 29382
ids=[]
for line in open('kraken_output.txt'):
    p=line.rstrip().split('\\t')
    if p[0]=='C' and p[2]=='29382': ids.append(p[1])
with gzip.open('contaminated.rna.q2.fq.gz','rt') as f, open('staph_kraken_reads.fa','w') as out:
    while True:
        h=f.readline().rstrip('\\n')
        if not h: break
        s=f.readline().rstrip('\\n'); plus=f.readline().rstrip('\\n'); q=f.readline().rstrip('\\n')
        rid=h[1:].split()[0]
        if rid in ids:
            out.write(f'>{rid}\\n{s}\\n')
print(len(ids))
PY
makeblastdb -in staph_cohnii.fna -dbtype nucl -out staph_db >/dev/null
blastn -task blastn -db staph_db -query staph_kraken_reads.fa -outfmt '6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore qcovs' -evalue 1e-5 -max_target_seqs 2 -out staph_kraken_blast.tsv
wc -l staph_kraken_blast.tsv; head -20 staph_kraken_blast.tsv"
