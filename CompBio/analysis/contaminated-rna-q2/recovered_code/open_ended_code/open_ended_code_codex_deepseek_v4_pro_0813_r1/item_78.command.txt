/bin/bash -lc "python3 - <<'PY'
from collections import Counter
cnt=Counter(); strong=Counter(); mouse_types=Counter()
for line in open('combined_bwa.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t')
    flag=int(p[1]); ref=p[2]; mapq=int(p[4])
    if flag & 0x800: continue
    if ref=='*': cnt['unmapped']+=1; continue
    prefix=ref.split('|')[0]
    cnt[prefix]+=1
    nm=None
    for tag in p[11:]:
        if tag.startswith('NM:i:'): nm=int(tag[5:])
    if nm is not None and nm<=2:
        strong[prefix]+=1
        if prefix=='mouse':
            rid=ref.split('|',1)[1]
            typ='NM' if rid.startswith('NM_') else 'XM' if rid.startswith('XM_') else 'NR' if rid.startswith('NR_') else 'XR' if rid.startswith('XR_') else 'other'
            mouse_types[typ]+=1
print('all primary',cnt)
print('NM<=2 strong',strong)
print('mouse NM<=2 types',mouse_types)
PY"
