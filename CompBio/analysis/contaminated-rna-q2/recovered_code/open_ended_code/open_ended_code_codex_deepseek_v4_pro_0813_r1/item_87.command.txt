/bin/bash -lc "python3 - <<'PY'
from collections import Counter
cnt=Counter(); strong=0
for line in open('yeast_all.sam'):
    if line.startswith('@'): continue
    p=line.rstrip().split('\\t'); flag=int(p[1])
    if flag & 0x100: continue
    if p[2]=='*': cnt['unmapped']+=1
    else:
        cnt['mapped']+=1
        nm=None
        for tag in p[11:]:
            if tag.startswith('NM:i:'): nm=int(tag[5:])
        if nm is not None and nm<=2: strong+=1
print(cnt, 'strong', strong)
PY"
