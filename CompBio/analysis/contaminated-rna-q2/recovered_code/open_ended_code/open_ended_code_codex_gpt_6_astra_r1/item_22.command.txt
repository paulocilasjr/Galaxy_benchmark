/bin/bash -lc "python3 - <<'PY'
import xml.etree.ElementTree as E,collections
r=E.parse('sample.blast.xml')
for it in r.findall('.//Iteration'):
 hits=it.findall('./Iteration_hits/Hit')
 if not hits:continue
 h=hits[0];hs=h.find('./Hit_hsps/Hsp')
 print(it.findtext('Iteration_query-def'),hs.findtext('Hsp_identity'),hs.findtext('Hsp_align-len'),h.findtext('Hit_def')[:250])
un=[];counts=collections.Counter()
with open('human.sam') as f:
 for l in f:
  if l.startswith('@'):continue
  p=l.split('\\t'); flag=int(p[1]);counts['unmapped' if flag&4 else 'mapped']+=1
  if flag&4:un.append((p[0],p[9]));
print(counts)
with open('unmapped.fa','w') as f:
 for h,s in un:f.write(f'>{h}\\n{s}\\n')
c=collections.Counter(s for h,s in un)
with open('unmapped.top.fa','w') as f:
 for i,(s,n) in enumerate(c.most_common(100)):f.write(f'>unmappedtop_{i}_count{n}\\n{s}\\n')
print(c.most_common(10))
PY"
