/bin/bash -lc "python3 - <<'PY'
import gzip,re,pysam,collections
# ids for human-unmapped and pig high-confidence
h={}
with pysam.AlignmentFile('results/human.map.bam') as b:
 for r in b:
  if r.is_secondary:h[r.query_name]=r.is_unmapped
p=[]
with pysam.AlignmentFile('results/pig.map.bam') as b:
 for r in b:
  if r.is_secondary:continue
  if h.get(r.query_name,False) and not r.is_unmapped and r.mapping_quality>=30 and r.has_tag('NM') and r.get_tag('NM')<=2 and r.query_alignment_length>=70:
   p.append((r.query_name,r.reference_name,r.mapping_quality,r.get_tag('NM'),r.query_alignment_length))
ids={x[1] for x in p}
heads={}
with gzip.open('refs/Sus_scrofa.Sscrofa11.1.cdna.all.fa.gz','rt') as f:
 for line in f:
  if line.startswith('>'):
   tok=line[1:].split()[0]
   if tok in ids: heads[tok]=line.strip()[1:]
for ref,n in collections.Counter(x[1] for x in p).most_common():
 print(n,ref,heads.get(ref,''))
PY"
