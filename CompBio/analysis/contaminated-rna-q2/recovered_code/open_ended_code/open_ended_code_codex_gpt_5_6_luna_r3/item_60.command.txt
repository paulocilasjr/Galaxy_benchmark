/bin/bash -lc "python3 - <<'PY'
import pysam, collections, math
files={'human':'results/human.map.bam','pig':'results/pig.map.bam'}
data={}
for org,f in files.items():
 d={}
 with pysam.AlignmentFile(f) as b:
  for r in b:
   if r.is_secondary: continue
   d[r.query_name]=(r.is_unmapped,r.mapping_quality,r.get_tag('AS') if r.has_tag('AS') else -999,r.get_tag('NM') if r.has_tag('NM') else -999,r.query_alignment_length,r.reference_name,r.cigarstring)
 data[org]=d
# bins for pig on human-unmapped
c=collections.Counter(); vals=[]
for q,h in data['human'].items():
 p=data['pig'][q]
 if h[0] and not p[0]:
  ident=(p[4]-p[3])/p[4] if p[4] else 0
  c[(p[1],p[2],p[3],p[4])]+=1
  vals.append((ident,p[1],p[2],p[3],p[4],q,p[5],p[6]))
print('human unmapped',sum(v[0] for v in data['human'].values()),'pig mapped',len(vals))
for cutoff in [0.99,0.98,0.95,0.90,0.80]: print('identity >=',cutoff,sum(x[0]>=cutoff for x in vals))
print('top by identity/length')
for x in sorted(vals,reverse=True)[:50]: print(x)
print('counts by nm/aln/mq')
print(c.most_common(30))
PY"
