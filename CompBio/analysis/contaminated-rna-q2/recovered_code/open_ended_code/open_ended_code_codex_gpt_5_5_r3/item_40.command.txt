/bin/bash -lc 'mkdir -p refs/candidates
for sp in mus_musculus rattus_norvegicus drosophila_melanogaster caenorhabditis_elegans saccharomyces_cerevisiae danio_rerio arabidopsis_thaliana; do
  url=$(curl -fsSL "https://ftp.ensembl.org/pub/current_fasta/${sp}/cdna/" | awk -F'"'\"' '/cdna\\.all\\.fa\\.gz/{print "'$2; exit}'"')
  echo \""'$sp $url"
  if [ -n "$url" ]; then curl -L --retry 3 -o "refs/candidates/${sp}.cdna.fa.gz" "https://ftp.ensembl.org/pub/current_fasta/${sp}/cdna/${url}"; fi
done
ls -lh refs/candidates'
