/bin/bash -lc "curl -L --retry 3 -o refs/genomes/rattus_norvegicus.chr11.fa.gz https://ftp.ensembl.org/pub/current_fasta/rattus_norvegicus/dna/Rattus_norvegicus.GRCr8.dna.primary_assembly.11.fa.gz
ls -lh refs/genomes/rattus_norvegicus.chr11.fa.gz
minimap2 -t 4 -x sr refs/genomes/rattus_norvegicus.chr11.fa.gz nonhuman_enriched.fa > rattus_chr11.paf
awk '{all["'$1]=1; if($10>=60 && $11>=60 && $10/$11>=0.95) hi[$1]=1; if($11>=70 && $10/$11>=0.99) vhi[$1]=1} END{print "rat_chr11 all", length(all), "hi95", length(hi), "hi99", length(vhi), "alns", NR}'"' rattus_chr11.paf
awk '"'$11>=70 && $10/$11>=0.99{print $1,$6,$8,$9,$10,$11,$5}'"' rattus_chr11.paf | head -10"
