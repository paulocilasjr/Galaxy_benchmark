/bin/bash -lc 'for f in refs/candidates2/*.cdna.fa.gz; do
  sp=$(basename "$f" .cdna.fa.gz)
  minimap2 -t 4 -x sr "$f" nonhuman_enriched.fa > "${sp}.paf" 2> "${sp}.minimap2.log"
  awk -v sp="$sp" '"'{all["'$1]=1; if($10>=60 && $11>=60 && $10/$11>=0.95) hi[$1]=1; if($10>=70 && $11>=70 && $10/$11>=0.98) vhi[$1]=1} END{print sp, "all", length(all), "hi95", length(hi), "hi98", length(vhi), "alns", NR}'"' \""'${sp}.paf"
done'
