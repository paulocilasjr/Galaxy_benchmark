/bin/bash -lc 'mkdir -p refs/candidates2
for sp in mus_musculus rattus_norvegicus drosophila_melanogaster caenorhabditis_elegans saccharomyces_cerevisiae danio_rerio; do
  url=$(curl -fsSL "https://ftp.ensembl.org/pub/current_fasta/${sp}/cdna/" | sed -n '"'s/.*href=\"\\(["'^"]*cdna'"\\.all\\.fa\\.gz\\)\".*/\\1/p' | head -1)
  echo \""'$sp $url"
  if [ -n "$url" ]; then curl -L --retry 3 -o "refs/candidates2/${sp}.cdna.fa.gz" "https://ftp.ensembl.org/pub/current_fasta/${sp}/cdna/${url}"; fi
done
ls -lh refs/candidates2'
