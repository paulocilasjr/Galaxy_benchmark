/bin/bash -lc "curl -sS --fail --max-time 60 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_007605.1&rettype=fasta&retmode=text' -o ebv.fa
curl -sS --fail --max-time 60 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_045512.2&rettype=fasta&retmode=text' -o sars.fa
curl -sS --fail --max-time 60 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=LT963440.1&rettype=fasta&retmode=text' -o staph_cohnii.fa
curl -sS --fail --max-time 60 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=LT963441.1&rettype=fasta&retmode=text' -o staph_haemolyticus.fa
for spec in ebv sars staph_cohnii staph_haemolyticus; do minimap2 -d \""'${spec}.mmi" "${spec}.fa" >/dev/null 2>"${spec}.idx.log"; minimap2 -ax sr -t 8 --secondary=no "${spec}.mmi" unmapped2.fq 2>"${spec}.map.log" | samtools view -b -o "${spec}.bam" -; printf '"'%-20s ' \""'$spec"; samtools view -c -F 4 "${spec}.bam"; done'
