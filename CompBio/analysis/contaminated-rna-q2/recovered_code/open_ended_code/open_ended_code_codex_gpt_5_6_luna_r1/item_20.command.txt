/bin/bash -lc 'rid=$(grep -o '"'RID = [A-Z0-9]*' blast_put.txt | tail -1 | awk '{print "'$3}'"')
echo RID="'$rid
for i in 1 2 3 4 5 6; do
  curl -sS --max-time 60 "https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&RID=${rid}&FORMAT_TYPE=Tabular&FORMAT_OBJECT=Alignment" -o blast_get.txt
  echo -n "poll $i "; grep -E '"'Status=|ThereAreHits|Status' blast_get.txt | head -4 | tr '\\n' ' '; echo
  grep -v '"'^#'"' blast_get.txt | head -3
  if grep -q 'Status=READY' blast_get.txt; then break; fi
  sleep 10
done
wc -c blast_get.txt"
