/bin/bash -lc "cat blast_results/*.result 2>/dev/null | awk '/"'^# Query:/{q=$3;next} /''^[''^#[:space:]]/{if(!(q in seen)){print q"'"\\t\""'$2"'"\\t\""'$3"'"\\t\""'$4"'"\\t\""'$11"'"\\t\""'$12;seen[q]=1}}'"' > unique_best_sofar.tsv
accs="'$(cut -f2 unique_best_sofar.tsv | sort -u | paste -sd, -)
curl -sS --max-time 60 "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=nuccore&id=${accs}" -o unique_esummary_sofar.xml
awk -F'"'[<>]' '/Item Name=\"Caption\"/{c="'$3} /Item Name="Title"/{t=$3} /Item Name="TaxId"/{tax=$3} /Item Name="AccessionVersion"/{print c"'"\\t\"tax\"\\t\"t; c=t=tax=\"\"}' unique_esummary_sofar.xml | sort -k2,2 -k1,1 > unique_species_sofar.tsv
printf 'best query rows: '; wc -l < unique_best_sofar.tsv; printf 'accession metadata: '; grep -c '"'^'"' unique_species_sofar.tsv; column -t -s "'$'"'\\t' unique_species_sofar.tsv | sed -n '1,160p'"
