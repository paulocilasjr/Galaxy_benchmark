/bin/bash -lc "# Calculate mean quality of all unmapped2 reads and mapped reference groups using portable awk's ord approximation.
awk 'NR%4==1{h="'$0;sub(/''^@/,"",h)} NR%4==2{seq=$0} NR%4==4{q=$0;sum=0;for(i=1;i<=length(q);i++){c=substr(q,i,1);v=index("!'"\\\"#"'$%&'"\\x27()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\\\]"'^_`abcdefghijklmnopqrstuvwxyz{|}~",c)-1;sum+=v}; print h,sum/length(q)}'"' unmapped2.fq > unmapped2_quality.tsv
for spec in ebv sars staph_cohnii pig_map; do echo -n \""'$spec "; awk -v s="$spec" '"'"'$2==s{print $1}'"' map_labels.tsv | sort -u | while read id; do awk -v id=\""'$id" '"'"'$1==id{print $2}'"' unmapped2_quality.tsv; done | awk '{n++;sum+="'$1}END{print n,sum/n}'"'; done
printf 'all '; awk '{s+="'$2;n++}END{print n,s/n}'"' unmapped2_quality.tsv"
