/bin/bash -lc "samtools fastq -f 4 -0 unmapped2.fq -n genome_map.bam 2>/dev/null
gzip -c unmapped2.fq > unmapped2.fq.gz
printf 'records: '; awk 'END{print NR/4}' unmapped2.fq
printf 'unique sequences: '; awk 'NR%4==2{print}' unmapped2.fq | sort -u | wc -l
printf 'top sequences:\\n'; awk 'NR%4==2{print}' unmapped2.fq | sort | uniq -c | sort -nr | head -40
printf 'MAPQ/NM for mapped genome reads:\\n'; samtools view genome_map.bam | awk '{for(i=12;i<=NF;i++)if("'$i~/''^NM:/)nm=$i; print $5,nm}'"' | sort | uniq -c | sort -nr | head -20"
