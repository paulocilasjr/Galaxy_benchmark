/bin/bash -lc 'cut -f2 unique_best_sofar.tsv | sort -u > best_accessions.txt
split -l 50 -a 2 best_accessions.txt best_acc_
: > accession_metadata.tsv
for f in best_acc_*; do ids=$(paste -sd, "$f"); curl -sS --max-time 60 -X POST '"'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi' --data-urlencode db=nuccore --data-urlencode \"id="'${ids}" -o tmp_esum.xml; awk -F'"'[<>]' '/Item Name=\"Caption\"/{c="'$3} /Item Name="Title"/{t=$3} /Item Name="TaxId"/{tax=$3} /Item Name="AccessionVersion"/{print c"'"\\t\"tax\"\\t\"t; c=t=tax=\"\"}' tmp_esum.xml >> accession_metadata.tsv; done
sort -u accession_metadata.tsv > accession_metadata_unique.tsv
printf 'metadata rows: '; wc -l < accession_metadata_unique.tsv
column -t -s "'$'"'\\t' accession_metadata_unique.tsv | sed -n '1,220p'"
