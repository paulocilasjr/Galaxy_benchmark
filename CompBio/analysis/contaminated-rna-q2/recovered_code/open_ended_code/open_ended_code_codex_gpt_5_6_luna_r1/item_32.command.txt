/bin/bash -lc "awk 'NR%4==1{h="'$0} NR%4==2{if((++n%5)==0){sub(/''^@/,">",h); print h; print}}'"' unmapped.fq > unmapped_every5.fa
wc -l unmapped_every5.fa
curl -sS --max-time 120 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' \\
  --data-urlencode 'CMD=Put' \\
  --data-urlencode 'PROGRAM=blastn' \\
  --data-urlencode 'DATABASE=nt' \\
  --data-urlencode 'MEGABLAST=off' \\
  --data-urlencode 'EXPECT=1e-10' \\
  --data-urlencode 'HITLIST_SIZE=10' \\
  --data-urlencode 'QUERY@unmapped_every5.fa' \\
  -o unmapped_put.txt
rid="'$(grep -o '"'RID = [A-Z0-9]*' unmapped_put.txt | tail -1 | awk '{print "'$3}'"')
echo RID="'$rid; grep -E '"'NUM_QUERIES|QUERY_LENGTH|RTOE|DATABASE' unmapped_put.txt | tail -10"
