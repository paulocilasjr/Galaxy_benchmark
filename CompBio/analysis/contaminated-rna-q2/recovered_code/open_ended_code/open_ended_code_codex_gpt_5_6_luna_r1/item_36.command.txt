/bin/bash -lc "awk 'NR%4==1{h="'$0} NR%4==2 && $0=="GTCGGCATGTATTAGCTCTAGAATTACCACAGTTATCCAAGTAGGAGAGGAGCGAGCGACCAAAGGAACCATAAC"{print ">"substr(h,2); print; exit}'"' unmapped.fq > top.fa
cat top.fa
curl -sS --max-time 120 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' \\
  --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' \\
  --data-urlencode 'MEGABLAST=off' --data-urlencode 'EXPECT=1e-10' --data-urlencode 'HITLIST_SIZE=20' \\
  --data-urlencode 'QUERY@top.fa' -o top_put.txt
rid="'$(grep -o '"'RID = [A-Z0-9]*' top_put.txt | tail -1 | awk '{print "'$3}'"'); echo RID="'$rid
for i in 1 2 3 4; do sleep 10; curl -sS --max-time 60 "https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&RID=${rid}&FORMAT_OBJECT=SearchInfo&FORMAT_TYPE=Text" -o top_status.txt; s=$(grep -o '"'Status=[A-Z]*' top_status.txt | head -1); echo "'$s; [ "$s" = '"'Status=READY' ] && break; done
curl -sS --max-time 60 \"https://blast.ncbi.nlm.nih.gov/Blast.cgi?RESULTS_FILE=on&RID="'${rid}&FORMAT_TYPE=Text&FORMAT_OBJECT=Alignment&DESCRIPTIONS=20&ALIGNMENT_VIEW=Tabular&CMD=Get" -o top_result.txt
sed -n '"'1,35p' top_result.txt"
