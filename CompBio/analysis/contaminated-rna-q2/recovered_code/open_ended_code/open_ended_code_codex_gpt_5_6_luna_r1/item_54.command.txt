/bin/bash -lc 'accs=$(cut -f2 bacterial_best.tsv | sort -u | paste -sd, -)
curl -sS --max-time 60 "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db=nuccore&id=${accs}" -o bacterial_esummary.xml
printf '"'records: '; grep -c '<DocSum>' bacterial_esummary.xml
awk -F'[<>]' '/Item Name=\"Caption\"/{c="'$3} /Item Name="Title"/{t=$3} /Item Name="TaxId"/{tax=$3} /Item Name="AccessionVersion"/{print c"'"\\t\"tax\"\\t\"t; c=t=tax=\"\"}' bacterial_esummary.xml > bacterial_species.tsv
column -t -s "'$'"'\\t' bacterial_species.tsv | sed -n '1,120p'"
