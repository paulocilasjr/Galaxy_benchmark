/bin/bash -lc "awk 'NR%4==1{h="'$0} NR%4==2{if(!seen[$0]++){sub(/''^@/,">",h); print h; print; if(++n>=100) exit}}'"' unmapped2.fq > bacterial_candidates.fa
wc -l bacterial_candidates.fa
curl -sS --max-time 120 -X POST 'https://blast.ncbi.nlm.nih.gov/Blast.cgi' \\
  --data-urlencode 'CMD=Put' --data-urlencode 'PROGRAM=blastn' --data-urlencode 'DATABASE=nt' \\
  --data-urlencode 'MEGABLAST=off' --data-urlencode 'EXPECT=1e-10' --data-urlencode 'HITLIST_SIZE=10' \\
  --data-urlencode 'QUERY@bacterial_candidates.fa' -o bacterial_put.txt
rid="'$(grep -o '"'RID = [A-Z0-9]*' bacterial_put.txt | tail -1 | awk '{print "'$3}'"'); echo RID="'$rid
for i in 1 2 3; do sleep 10; curl -sS --max-time 60 "https://blast.ncbi.nlm.nih.gov/Blast.cgi?CMD=Get&RID=${rid}&FORMAT_OBJECT=SearchInfo&FORMAT_TYPE=Text" -o bacterial_status.txt; s=$(grep -o '"'Status=[A-Z]*' bacterial_status.txt | head -1); echo "'$s; [ "$s" = '"'Status=READY' ] && break; done
curl -sS --max-time 60 \"https://blast.ncbi.nlm.nih.gov/Blast.cgi?RESULTS_FILE=on&RID="'${rid}&FORMAT_TYPE=Text&FORMAT_OBJECT=Alignment&DESCRIPTIONS=10&ALIGNMENT_VIEW=Tabular&CMD=Get" -o bacterial_result.txt
wc -c bacterial_result.txt; grep -v '"'"'^#'"' bacterial_result.txt | head -30"
