/bin/bash -lc "BLASTDB=refs/blastdb blastx -db landmark -query results/unmapped_human_genome.fa -max_target_seqs 10 -evalue 1e-5 -outfmt '6 qseqid sacc staxids pident length evalue bitscore stitle' -num_threads 4 -out results/unmapped_humangenome_landmark_blastx.tsv

awk '{tax["'$2]++; title[$2]=$5" "$6" "$7" "$8" "$9} END{for(t in tax) print tax[t],t,title[t]}'"' results/unmapped_humangenome_landmark_blastx_best.tsv | sort -nr | head -n 60"
