/bin/bash -lc "curl -L --fail --retry 2 -o human_genome.fna.gz https://ftp.ncbi.nlm.nih.gov/genomes/refseq/vertebrate_mammalian/Homo_sapiens/all_assembly_versions/GCF_000001405.40_GRCh38.p14/GCF_000001405.40_GRCh38.p14_genomic.fna.gz
ls -lh human_genome.fna.gz
minimap2 -x sr --secondary=no -t 8 human_genome.fna.gz contaminated.rna.q2.fq.gz > human_genome.paf
awk '{print "'$1}'"' human_genome.paf | sort -u > human_genome.ids
wc -l human_genome.ids"
