from Bio import SeqIO
s=next(SeqIO.parse('query_unitig2.fa','fasta')).seq

  for frame in range(3):
    p=seq[frame:].translate(table=2)
    parts=str(p).split('*')
    longest=max(parts,key=len)
    print(strand,frame,len(longest),longest[:100])
PY"
