/bin/bash -lc "curl -L --fail --retry 2 -o pig_refseq_rna.fna.gz https://ftp.ncbi.nlm.nih.gov/genomes/refseq/vertebrate_mammalian/Sus_scrofa/all_assembly_versions/GCF_000003025.6_Sscrofa11.1/GCF_000003025.6_Sscrofa11.1_rna.fna.gz
ls -lh pig_refseq_rna.fna.gz
minimap2 -c -x sr --secondary=no -t 8 pig_refseq_rna.fna.gz contaminated.rna.q2.fq.gz > pig_rna_c.paf
awk '{print "'$1}'"' pig_rna_c.paf | sort -u | wc -l
python - <<'PY'
from collections import Counter

def load(fn):
 d={}
 for l in open(fn):
  a=l.split(); tags={x[:2]:x[5:] for x in a[12:] if len(x)>5}
  key=(int(a[9]),-int(a[10]),-float(tags.get('de','1')))
  if key>d.get(a[0],((-1,),None))[0]: d[a[0]]=(key,a)
 return d
H=load('human_genome_c.paf'); P=load('pig_rna_c.paf')
c=Counter()
for q,p in P.items():
 if q not in H:c['pig_only']+=1
 elif p[0]>H[q][0]:c['pig_win']+=1
 elif H[q][0]>p[0]:c['human_win']+=1
 else:c['tie']+=1
print(c)
PY"
