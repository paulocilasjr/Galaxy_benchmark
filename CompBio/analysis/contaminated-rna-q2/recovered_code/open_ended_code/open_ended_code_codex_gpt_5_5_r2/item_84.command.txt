/bin/bash -lc "python3 - <<'PY'
import urllib.request
for acc in ['NC_007605.1','NC_009334.1','AJ507799.2']:
    url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id={acc}&rettype=fasta&retmode=text'
    data=urllib.request.urlopen(url,timeout=30).read().decode()
    print(acc, data.split('\\n',1)[0], len(data))
    open(acc+'.fa','w').write(data)
PY
cat NC_007605.1.fa NC_009334.1.fa AJ507799.2.fa > ebv_refs.fa
minimap2 -t 2 -a -x sr ebv_refs.fa unmapped_to_human_genome.fq | samtools view -@ 2 -bS - > ebv_on_human_unmapped.bam
samtools flagstat ebv_on_human_unmapped.bam
samtools view ebv_on_human_unmapped.bam | awk '"'$3!="*"{print $1,$3,$5,$6}'"' | head -50"
