/bin/bash -lc "python3 - <<'PY'
# write reads without obvious adapter/polyA/vector restriction motifs
bad_motifs=['AGATCGGAAGAGCACACGTCTGAACTCCAGTCAC','GCGGCCGC']
with open('unmapped_to_human_genome.fq') as f, open('clean_unmapped.fa','w') as out:
    rec=[]; kept=0; total=0
    for line in f:
        rec.append(line.rstrip('\\n'))
        if len(rec)==4:
            total+=1; h,s,_,q=rec
            bad=any(m in s for m in bad_motifs) or s.count('A')>=55 or s.count('T')>=55
            if not bad:
                kept+=1; out.write('>'+h[1:]+'\\n'+s+'\\n')
            rec=[]
print(total, kept)
PY
head -10 clean_unmapped.fa"
