/bin/bash -lc "python3 - <<'PY'
import math
from pathlib import Path

def load(p):
    d={}
    for line in Path(p).read_text().splitlines()[1:]:
        if not line: continue
        g,c=line.split('\\t')
        d[g]=int(c)
    return d
A=load('results/sampleA.gene_counts.tsv')
B=load('results/sampleB.gene_counts.tsv')
allg=set(A)|set(B)
rows=[]
for g in allg:
    a=A.get(g,0); b=B.get(g,0)
    diff=b-a
    fc=(b+1)/(a+1)
    rows.append((abs(diff), abs(math.log2(fc)), g,a,b,diff,math.log2(fc)))
for r in sorted(rows, reverse=True)[:30]:
    _,_,g,a,b,diff,l2=r
    print(f'{g}\\tA={a}\\tB={b}\\tdiff={diff}\\tlog2FC={l2:.3f}')
PY"
