/bin/bash -c "Rscript - <<'RS'
suppressMessages(library(DESeq2))
x <- read.delim('gene_counts.tsv', stringsAsFactors=FALSE)
rownames(x) <- x"'$gene_id
cts <- as.matrix(x[,c('"'countA','countB')])
colnames(cts) <- c('A','B')
coldata <- data.frame(condition=factor(c('A','B')), row.names=colnames(cts))
dds <- DESeqDataSetFromMatrix(countData=cts, colData=coldata, design=~condition)
dds <- DESeq(dds, quiet=TRUE)
res <- results(dds, contrast=c('condition','B','A'), independentFiltering=FALSE, cooksCutoff=FALSE)
res <- as.data.frame(res)
res"'$gene_id <- rownames(res)
res <- merge(res, x[,c('"'gene_id','gene_symbol')], by='gene_id')
res <- res[order(res"'$pvalue, -abs(res$log2FoldChange)),]
print(head(res, 30))
RS'
