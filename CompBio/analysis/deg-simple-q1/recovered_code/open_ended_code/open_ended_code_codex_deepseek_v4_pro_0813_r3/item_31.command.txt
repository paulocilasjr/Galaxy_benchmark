/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('gene_counts.tsv',sep='\\t')
x['max']=x[['countA','countB']].max(axis=1)
x['log2FC']=np.log2((x['countB']+0.5)/(x['countA']+0.5))
y=x[(x['max']>=5)&(x['log2FC'].abs()>=1.5)].sort_values('log2FC',key=lambda s:s.abs(),ascending=False)
print(y[['gene_symbol','gene_id','countA','countB','log2FC']].to_string(index=False))
print('n=',len(y))
PY"
