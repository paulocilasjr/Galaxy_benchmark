/bin/bash -c "python3 - <<'PY'
import gzip, re, json
path='Homo_sapiens.GRCh38.cdna.all.fa.gz'
out=[]
with gzip.open(path,'rt') as f:
    for line in f:
        if line.startswith('>'):
            parts=line[1:].strip().split()
            tid=parts[0]
            gene=ensg=symbol=None
            for p in parts[1:]:
                if p.startswith('gene:'): ensg=p[5:]
                if p.startswith('gene_symbol:'): symbol=p[12:]
            out.append((tid,ensg,symbol))
print(len(out), out[:3], out[-3:])
with open('transcript_gene.tsv','w') as f:
    f.write('transcript_id\\tgene_id\\tgene_symbol\\n')
    for tid,ensg,symbol in out:
        f.write(f'{tid}\\t{ensg or \"\"}\\t{symbol or \"\"}\\n')
PY
wc -l transcript_gene.tsv"
