/bin/bash -c "python3 - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('gene_counts.tsv',sep='\\t')
# filter at least 100 in max
x=x[x[['countA','countB']].max(axis=1)>=100].copy()
x['log2FC']=np.log2((x['countB']+1)/(x['countA']+1))
x['diff']=x['countB']-x['countA']
print(x.sort_values('log2FC',key=lambda s:abs(s),ascending=False).head(40).to_string(index=False))
PY"
