/bin/bash -lc "python3 - <<'PY'
from collections import Counter
from scipy.stats import fisher_exact, binomtest

def counts(path):
 c=Counter(); n=0
 with open(path) as f:
  for line in f:
   x=line.rstrip('\\n').split('\\t'); n+=1; c[x[5].split('|')[5]]+=1
 return c,n
A,na=counts('sampleA.paf'); B,nb=counts('sampleB.paf')
# report normalized difference and Fisher p for genes total >= 10
rows=[]
for g in set(A)|set(B):
 a,b=A[g],B[g]
 if a+b<10: continue
 # ratio expected using mapped totals, difference counts relative to expected B
 d=a-(na/nb)*b
 # exact 2x2
 _,p=fisher_exact([[a,na-a],[b,nb-b]], alternative='two-sided')
 l2=((a+.5)/na)/((b+.5)/nb)
 rows.append((p,-abs(d),d,l2,a,b,g))
print('TOP by fisher p')
for p,_,d,l2,a,b,g in sorted(rows)[:100]: print(f'p={p:.3g} d={d:.1f} l2={l2:.2f} A={a} B={b} {g}')
print('\\nTOP by abs normalized difference')
for p,_,d,l2,a,b,g in sorted(rows,key=lambda x:abs(x[2]),reverse=True)[:100]: print(f'p={p:.3g} d={d:.1f} l2={l2:.2f} A={a} B={b} {g}')
PY"
