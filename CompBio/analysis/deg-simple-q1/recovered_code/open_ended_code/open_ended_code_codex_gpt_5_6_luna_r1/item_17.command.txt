/bin/bash -lc "python3 - <<'PY'
from collections import Counter
from math import log2

def counts(path):
 c=Counter(); n=0
 with open(path) as f:
  for line in f:
   x=line.rstrip('\\n').split('\\t'); n+=1; c[x[5].split('|')[5]]+=1
 return c,n
A,na=counts('sampleA.paf'); B,nb=counts('sampleB.paf')
rows=[]
for g in set(A)|set(B):
 a,b=A[g],B[g]
 if a+b<10: continue
 d=a-(na/nb)*b
 l2=log2(((a+.5)/na)/((b+.5)/nb))
 # approximate chi-square 1 df for 2 proportions, statistic
 tot=a+b; expa=tot*na/(na+nb); expb=tot*nb/(na+nb)
 # 2x2 chi-square
 chi=(a-expa)**2/expa+(b-expb)**2/expb+((na-a)-(na+nb-tot)*na/(na+nb))**2/((na+nb-tot)*na/(na+nb))+((nb-b)-(na+nb-tot)*nb/(na+nb))**2/((na+nb-tot)*nb/(na+nb)) if tot<na+nb else 0
 # p approx erfc(sqrt(chi/2))
 from math import erfc,sqrt
 p=erfc(sqrt(chi/2))
 rows.append((p,abs(d),d,l2,a,b,g))
print('mapped',na,nb,'genes',len(set(A)|set(B)),flush=True)
print('TOP by chi p',flush=True)
for p,_,d,l2,a,b,g in sorted(rows)[:100]: print(f'p={p:.3g} d={d:.1f} l2={l2:.2f} A={a} B={b} {g}',flush=True)
print('\\nTOP by abs normalized difference',flush=True)
for p,_,d,l2,a,b,g in sorted(rows,key=lambda x:x[1],reverse=True)[:100]: print(f'p={p:.3g} d={d:.1f} l2={l2:.2f} A={a} B={b} {g}',flush=True)
PY"
