/bin/bash -lc "python3 - <<'PY'
from collections import Counter
from pathlib import Path

def counts(path, mapq_min=None):
    c=Counter(); n=0; mapped=0; mq=Counter()
    with open(path) as f:
        for line in f:
            if not line.strip(): continue
            x=line.rstrip('\\n').split('\\t')
            mapped+=1
            mq[int(x[11])] += 1
            if mapq_min is not None and int(x[11]) < mapq_min: continue
            gene=x[5].split('|')[5] if '|' in x[5] else x[5]
            c[gene]+=1
    return c,mapped,mq
A,na,mqa=counts('sampleA.paf'); B,nb,mqb=counts('sampleB.paf')
print('mapped',na,nb,'genes',len(set(A)|set(B)))
for cutoff in [0,10,30,100]:
    rows=[]
    for g in set(A)|set(B):
        a,b=A[g],B[g]
        if a+b<cutoff: continue
        # normalize to mapped reads, pseudocount 0.5
        l2=((a+.5)/(na+1))/((b+.5)/(nb+1))
        rows.append((abs(l2),l2,a,b,g))
    print('\\ncutoff',cutoff,'top abs')
    for row in sorted(rows,reverse=True)[:40]: print('%8.3f %8.3f %7d %7d %s'%row)
print('\\nMAPQ A',mqa.most_common(10)); print('MAPQ B',mqb.most_common(10))
PY"
