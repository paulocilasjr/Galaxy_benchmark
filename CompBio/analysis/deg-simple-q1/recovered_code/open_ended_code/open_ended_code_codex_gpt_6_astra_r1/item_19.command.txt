/bin/bash -lc "cat > scripts/compare_symbols.py <<'PY'
import csv,math,collections
counts={};high={}
for s in ['A','B']:
 counts[s]=collections.Counter();high[s]=collections.Counter()
 for r in csv.DictReader(open(f'results/{s}.counts.tsv'),delimiter='\\t'):
  counts[s][r['symbol']]+=int(r['count']);high[s][r['symbol']]+=int(r['high_mapq'])
f=sum(counts['B'].values())/sum(counts['A'].values())
rows=[]
for g in counts['A'].keys()|counts['B'].keys():
 a,b=counts['A'][g],counts['B'][g]
 fc=math.log2((b+.5)/(a+.5)/f);z=(b-a*f)/math.sqrt((a+b)*f)
 rows.append((g,a,b,fc,z,high['A'][g],high['B'][g]))
rows.sort(key=lambda r:abs(r[4]),reverse=True)
with open('results/symbol_comparison.tsv','w') as out:
 w=csv.writer(out,delimiter='\\t');w.writerow(['symbol','A','B','log2FC','z','high_A','high_B']);w.writerows(rows)
print('TOP DIFFERENCES')
for r in rows[:30]:print(r)
print('TOP FOLD CHANGE WITH >=100 READS')
for r in sorted([r for r in rows if r[1]+r[2]>=100],key=lambda r:abs(r[3]),reverse=True)[:20]:print(r)
PY
python scripts/compare_symbols.py"
