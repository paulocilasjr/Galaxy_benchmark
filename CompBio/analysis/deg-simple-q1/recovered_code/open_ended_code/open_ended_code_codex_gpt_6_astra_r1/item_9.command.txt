/bin/bash -lc "cat > scripts/compare_counts.py <<'PY'
import pandas as pd,numpy as np
x=pd.read_csv('results/A.counts.tsv',sep='\\t').merge(pd.read_csv('results/B.counts.tsv',sep='\\t'),on=['gene','symbol'],how='outer',suffixes=('_A','_B')).fillna(0)
f=x['count_B'].sum()/x['count_A'].sum()
x['log2FC_B_A']=np.log2((x.count_B+.5)/(x.count_A+.5)/f)
x['z']=(x.count_B-x.count_A*f)/np.sqrt((x.count_A+x.count_B)*f)
x['abs_z']=x.z.abs()
x=x.sort_values('abs_z',ascending=False)
x.to_csv('results/comparison.tsv',sep='\\t',index=False)
print('Totals:',x.count_A.sum(),x.count_B.sum(),'normalization:',f)
print(x.head(35).to_string(index=False))
print('\\nLargest fold changes with at least 100 reads:')
y=x[x.count_A+x.count_B>=100]
print(y.loc[y.log2FC_B_A.abs().sort_values(ascending=False).index].head(25).to_string(index=False))
PY"
