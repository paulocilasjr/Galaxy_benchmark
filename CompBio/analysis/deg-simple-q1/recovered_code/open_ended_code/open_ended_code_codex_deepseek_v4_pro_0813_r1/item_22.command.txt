/bin/bash -lc "python - <<'PY'
import pandas as pd, gzip
# parse headers for KLF4 and perhaps KLF genes
wanted=[]
with gzip.open('Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as f:
    for line in f:
        if line.startswith('>'):
            parts=line.rstrip().split()
            tid=parts[0][1:]
            gene=''
            for p in parts:
                if p.startswith('gene_symbol:'):
                    gene=p.split(':',1)[1]
            if 'KLF4' in gene or gene in ['KLF1','KLF2','KLF3','KLF4','KLF5','KLF6']:
                wanted.append((tid,gene))
print('transcripts wanted', wanted)
a=pd.read_csv('outA/abundance.tsv',sep='\\t').set_index('target_id')
b=pd.read_csv('outB/abundance.tsv',sep='\\t').set_index('target_id')
for tid,g in wanted:
    print(g, tid, 'A', a.loc[tid,'est_counts'] if tid in a.index else None, 'B', b.loc[tid,'est_counts'] if tid in b.index else None)
PY"
