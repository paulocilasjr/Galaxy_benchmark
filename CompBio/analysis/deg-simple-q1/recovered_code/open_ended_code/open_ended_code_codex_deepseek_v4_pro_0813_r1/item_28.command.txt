/bin/bash -lc "python - <<'PY'
import pandas as pd
from scipy.stats import binomtest
x=pd.read_csv('gene_counts_abundance.tsv',sep='\\t')
x=x[['gene','count_A','count_B']]
# library sizes from kallisto counts total
N_A=x['count_A'].sum(); N_B=x['count_B'].sum()
print('lib',N_A,N_B)
rows=[]
for _,r in x.iterrows():
    k=r['count_B']; n=r['count_A']+r['count_B']
    if n<1: continue
    p=N_B/(N_A+N_B)
    # two-sided? use exact binomial test
    bt=binomtest(int(round(k)), int(round(n)), p=p, alternative='two-sided')
    fc=(r['count_B']+0.5)/(r['count_A']+0.5)
    rows.append((r['gene'], r['count_A'], r['count_B'], fc, bt.pvalue))
res=pd.DataFrame(rows,columns=['gene','A','B','FC','p'])
res=res.sort_values('p')
print(res.head(40).to_string(index=False))
PY"
