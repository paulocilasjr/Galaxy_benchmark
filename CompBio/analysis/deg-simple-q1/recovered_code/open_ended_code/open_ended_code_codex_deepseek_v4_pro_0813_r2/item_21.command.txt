/bin/bash -lc "python3 - <<'PY'
import pysam, collections, math
for minmq in [1,10,20]:
    print('minmq', minmq)
    counts={}
    for sample in ['A','B']:
        gc=collections.Counter(); total=0; uniq=0
        with pysam.AlignmentFile(f'sample{sample}.bam','rb') as bam:
            for read in bam.fetch(until_eof=True):
                if read.is_unmapped: continue
                total+=1
                if read.mapping_quality < minmq: continue
                uniq+=1
                name=read.reference_name
                parts=name.split('|')
                sym=parts[5] if len(parts)>5 else parts[0]
                gc[sym]+=1
        counts[sample]=gc
        print(sample, 'unique', uniq, 'of', total, 'KLF4', gc.get('KLF4',0))
    genes=set(counts['A'])|set(counts['B'])
    rows=sorted(((abs(counts['A'].get(g,0)-counts['B'].get(g,0)), counts['A'].get(g,0), counts['B'].get(g,0), g) for g in genes), reverse=True)[:15]
    print(rows)
PY"
