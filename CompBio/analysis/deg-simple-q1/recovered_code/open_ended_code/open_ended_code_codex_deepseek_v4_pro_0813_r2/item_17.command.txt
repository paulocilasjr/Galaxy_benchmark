/bin/bash -lc "python3 - <<'PY'
import pysam, collections
for sample in ['A','B']:
    counts=collections.Counter()
    with pysam.AlignmentFile(f'sample{sample}.bam','rb') as bam:
        for read in bam.fetch(until_eof=True):
            if read.is_unmapped: continue
            counts[read.reference_name]+=1
    print(sample, 'aligned records', sum(counts.values()), 'unique transcripts', len(counts))
    top=counts.most_common(10)
    print(top)
    with open(f'sample{sample}.transcript_counts.tsv','w') as f:
        f.write('transcript\\tcount\\n')
        for t,c in counts.items():
            f.write(f'{t}\\t{c}\\n')
PY"
