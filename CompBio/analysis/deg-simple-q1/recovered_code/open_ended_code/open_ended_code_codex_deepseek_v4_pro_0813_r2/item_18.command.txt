/bin/bash -lc "python3 - <<'PY'
import collections, re, math
counts={}
for sample in ['A','B']:
    gene_counts=collections.Counter()
    total=0
    with open(f'sample{sample}.transcript_counts.tsv') as f:
        next(f)
        for line in f:
            line=line.rstrip('\\n')
            t,c=line.split('\\t')
            c=int(c)
            total+=c
            parts=t.split('|')
            symbol=parts[5] if len(parts)>5 else parts[0]
            gene_counts[symbol]+=c
    counts[sample]=gene_counts
    print(sample, 'total', total, 'genes', len(gene_counts))
# merge
genes=sorted(set(counts['A'])|set(counts['B']))
rows=[]
for g in genes:
    a=counts['A'].get(g,0); b=counts['B'].get(g,0)
    # pseudocount
    fc=(b+1)/(a+1)
    log2fc=math.log2(fc)
    # diff
    rows.append((abs(a-b), log2fc, g, a,b))
print('top absolute diff')
for r in sorted(rows, reverse=True)[:30]: print(r)
print('top log2FC (b vs a)')
for r in sorted(rows, key=lambda x: abs(x[1]), reverse=True)[:30]: print(r)
PY"
