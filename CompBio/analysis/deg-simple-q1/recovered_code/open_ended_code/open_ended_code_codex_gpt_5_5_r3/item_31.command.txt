/bin/bash -lc "python3 - <<'PY'
import math
from pathlib import Path

def load(fn):
    d={}
    meta={}
    for line in Path(fn).read_text().splitlines():
        g,c=line.split('\\t')
        c=int(c)
        if g.startswith('__'):
            meta[g]=c
        else:
            d[g]=c
    return d, meta
A, Am = load('sampleA.gene_counts.tsv')
B, Bm = load('sampleB.gene_counts.tsv')
TA=Am['__PRIMARY_MAPPED__']; TB=Bm['__PRIMARY_MAPPED__']
rows=[]
for g in set(A)|set(B):
    a=A.get(g,0); b=B.get(g,0)
    an=a/TA*1e6; bn=b/TB*1e6
    lfc=math.log2((an+1)/(bn+1))
    rows.append((abs(lfc), lfc, g, a, b, an, bn, a-b))
print('meta', Am, Bm)
print('top_abs_lfc count>=20 either')
for r in sorted([r for r in rows if max(r[3],r[4])>=20], reverse=True)[:30]:
    print(f'{r[2]}\\tA={r[3]}\\tB={r[4]}\\tApm={r[5]:.1f}\\tBpm={r[6]:.1f}\\tlog2={r[1]:.2f}\\tdiff={r[7]}')
print('top_abs_diff')
for r in sorted(rows, key=lambda x: abs(x[7]), reverse=True)[:30]:
    print(f'{r[2]}\\tA={r[3]}\\tB={r[4]}\\tApm={r[5]:.1f}\\tBpm={r[6]:.1f}\\tlog2={r[1]:.2f}\\tdiff={r[7]}')
PY"
