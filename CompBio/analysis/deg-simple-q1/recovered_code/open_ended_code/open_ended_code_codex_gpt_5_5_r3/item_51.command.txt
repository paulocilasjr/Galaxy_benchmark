/bin/bash -lc "python3 - <<'PY'
import pathlib, math

def load(fn):
    d={}; meta={}
    for line in pathlib.Path(fn).read_text().splitlines():
        g,c=line.split('\\t'); c=int(c)
        if g.startswith('__'): meta[g]=c
        else: d[g]=c
    return d, meta
A,Am=load('sampleA.any_gene_counts.tsv'); B,Bm=load('sampleB.any_gene_counts.tsv')
TA=Am['__READS__']; TB=Bm['__READS__']
rows=[]
for g in set(A)|set(B):
    a=A.get(g,0); b=B.get(g,0)
    if max(a,b)>=20:
        an=a/TA*1e6; bn=b/TB*1e6
        lfc=math.log2((an+1)/(bn+1))
        rows.append((abs(lfc), g, a, b, lfc, a-b))
print('read totals', TA, TB)
for r in sorted(rows, reverse=True)[:20]:
    print(f'{r[1]}\\tA={r[2]}\\tB={r[3]}\\tlog2={r[4]:.2f}\\tdiff={r[5]}')
PY"
