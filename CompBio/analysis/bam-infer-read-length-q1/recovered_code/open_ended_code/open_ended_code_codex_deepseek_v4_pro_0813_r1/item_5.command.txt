/bin/bash -lc "python3 - <<'PY'
import pysam
b=pysam.AlignmentFile('mt.sorted.bam','rb')
paired=0; single=0; lengths={}
for r in b:
    if r.flag & 1:
        paired+=1
    else:
        single+=1
    lengths[r.query_length]=lengths.get(r.query_length,0)+1
print('paired',paired,'single',single)
print('lengths',lengths)
print('mapped', sum(1 for _ in pysam.AlignmentFile('mt.sorted.bam','rb') if not _.is_unmapped))
PY"
