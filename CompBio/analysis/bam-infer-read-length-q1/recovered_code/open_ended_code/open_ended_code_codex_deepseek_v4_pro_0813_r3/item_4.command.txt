/bin/bash -lc "python3 - <<'PY'
import pysam
p='mt.sorted.bam'
counts={}; lens=[]; paired_count=0; flags=set()
with pysam.AlignmentFile(p,'rb') as f:
    for r in f:
        if r.is_secondary or r.is_supplementary: continue
        counts[len(r.query_sequence)] = counts.get(len(r.query_sequence),0)+1
        lens.append(len(r.query_sequence))
        if r.is_paired: paired_count+=1
        flags.add(r.flag)
print('primary reads', len(lens))
print('paired flag count', paired_count)
print('length distribution', counts)
print('unique flags', flags)
# read1/read2 counts and read lengths
r1=[];r2=[]
with pysam.AlignmentFile(p,'rb') as f:
    for r in f:
        if r.is_secondary or r.is_supplementary: continue
        if r.is_read1: r1.append(len(r.query_sequence))
        elif r.is_read2: r2.append(len(r.query_sequence))
print('read1 count',len(r1),'read2 count',len(r2),'r1 len dist', {x:r1.count(x) for x in set(r1)}, 'r2 len dist', {x:r2.count(x) for x in set(r2)})
PY"
