/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    header=f.readline().strip();
    lines=[]
    for line in f:
        if line.startswith('%'): continue
        lines.append(line.strip())
        if len(lines)==1: break
    dims=tuple(map(int,lines[0].split()))
    vals=np.fromiter((float(x) for line in f for x in line.split()),dtype=np.float64)
print(header, dims, vals.size, vals.min(), vals.max(), np.count_nonzero(vals), 'zero frac', np.mean(vals==0))
X=vals.reshape(dims, order='F')
lib=X.sum(0); det=(X>0).sum(0)
print('lib',np.percentile(lib,[0,1,5,10,25,50,75,90,95,99,100]))
print('det',np.percentile(det,[0,1,5,10,25,50,75,90,95,99,100]))
print('gene means quant',np.percentile(X.mean(1),[0,25,50,75,90,99,100]))
print('cell first 10 totals',lib[:10], 'det',det[:10])
PY"
