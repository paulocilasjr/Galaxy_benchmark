/bin/bash -lc "python - <<'PY'
import gzip,numpy as np
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
with gzip.open('cell.proportions.q1.mtx.gz','rt') as f:
 for line in f:
  if line.startswith('%'): continue
  g,n=map(int,line.split()); break
 v=np.fromiter((float(x) for line in f for x in line.split()),np.float32); X=v.reshape((g,n),order='F')
Y=np.log1p(X/X.sum(0)*1e4).T; sel=np.argsort(Y.var(0))[-300:]; p=PCA(20,random_state=1).fit_transform(Y[:,sel]); lab=KMeans(6,n_init=50,random_state=1).fit_predict(p)
order=np.argsort(np.bincount(lab))[::-1]; mp={old:i for i,old in enumerate(order)}; lab=np.array([mp[x] for x in lab])
# run lengths
st=0
for i in range(1,n+1):
 if i==n or lab[i]"'!=lab[st]: print(st,i-1,'"'label',lab[st],'len',i-st); st=i
# counts by broad blocks 0-? 
print('first labels',lab[:100]); print('last labels',lab[-100:])
PY"
