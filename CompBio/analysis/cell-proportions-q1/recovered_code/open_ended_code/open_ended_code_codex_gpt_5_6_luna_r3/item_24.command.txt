/bin/bash -lc "python -u - <<'PY'
import gzip,numpy as np
from sklearn.decomposition import PCA
from scipy.stats import spearmanr
with gzip.open('cell.proportions.q1.mtx.gz','rt') as f:
 for line in f:
  if not line.startswith('%'): break
 dims=tuple(map(int,line.split())); vals=np.fromiter((float(x) for x in f),dtype=np.float32)
X=vals.reshape(dims,order='F').T
lib=X.sum(1); Y=np.log1p(X/lib[:,None]*1e4)
blocks=[(0,2500),(2500,4250),(4250,5000),(5000,5100),(5100,5150),(5150,5350)]
M=np.array([Y[a:b].mean(0) for a,b in blocks]); R=np.corrcoef(M)
print('block sizes',list(map(lambda x:x[1]-x[0],blocks)))
print('profile correlations:\\n',np.round(R,3))
print('profile Euclidean distances')
print(np.round(np.sqrt(((M[:,None,:]-M[None,:,:])**2).sum(2)),1))
for i,(a,b) in enumerate(blocks):
 z=M[i]-M.mean(0)
 tops=np.argsort(z)[::-1][:20]; lows=np.argsort(z)[:10]
 print('\\nBLOCK',i,'n',b-a,'lib',X[a:b].sum(1).mean(),'det',(X[a:b]>0).sum(1).mean())
 print('top expr genes',np.argsort(M[i])[::-1][:20].tolist())
 print('top marker z',tops.tolist(),np.round(z[tops],2).tolist())
 print('low marker',lows.tolist(),np.round(z[lows],2).tolist())
# means per group selected top 100 variable genes, output compact matrix for distinguishing genes
v=Y.var(0); vars=np.argsort(v)[-100:]
print('\\nTop 100 var profile means range',np.round(M[:,vars].mean(1),2))
print('block overall means',np.round(M.mean(1),3))
PY"
