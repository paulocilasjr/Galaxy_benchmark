/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, time
path='cell.proportions.q1.mtx.gz'
# Read matrix market array
with gzip.open(path,'rt') as f:
    line=f.readline().strip(); print(line)
    # comments
    while True:
        pos=f.tell(); l=f.readline()
        if not l.startswith('%'):
            dims=l.strip(); break
    nrow,ncol=map(int,dims.split())
    print('dims', nrow,ncol)
    data=np.fromiter((float(x) for x in f), dtype=np.float64, count=nrow*ncol)
print('read', data.shape, 'minmax', data.min(), data.max(), 'mean', data.mean())
X=data.reshape((nrow,ncol), order='F')  # MatrixMarket array column-major
# check alternately order? Values list is column-major by spec.
print('X', X.shape)
lib=X.sum(axis=0); detected=(X>0).sum(axis=0)
print('lib quantiles', np.quantile(lib,[0,0.01,0.05,0.1,0.25,0.5,0.75,0.9,0.95,0.99,1]))
print('det quantiles', np.quantile(detected,[0,0.01,0.05,0.1,0.25,0.5,0.75,0.9,0.95,0.99,1]))
print('cell mean q', np.quantile(X.mean(axis=0),[0,0.01,0.05,0.1,0.5,0.9,0.95,0.99,1]))
print('gene mean q', np.quantile(X.mean(axis=1),[0,0.01,0.05,0.1,0.5,0.9,0.95,0.99,1]))
PY"
