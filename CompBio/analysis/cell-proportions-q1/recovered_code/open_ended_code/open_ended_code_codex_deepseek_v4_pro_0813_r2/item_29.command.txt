/bin/bash -c "python3 - <<'PY'
import numpy as np, gzip
from scipy.io import mmread
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f: M=mmread(f)
X=np.asarray(M.T,dtype=float)
lib=X.sum(1); Xn=np.log1p(X/lib[:,None]*10000)
A=Xn[:,0:100].mean(1); B=Xn[:,100:200].mean(1); C=Xn[:,200:300].mean(1)
S=np.stack([A,B,C],axis=1)
print('score quantiles by module')
for name,arr in [('A',A),('B',B),('C',C)]:
 print(name, [round(float(np.quantile(arr,q)),3) for q in [0,.01,.05,.1,.5,.9,.95,.99,1]])
# classify by thresholds: high >3? Counts
for th in [2.5,3.0,3.2,3.5]:
 high=S>th
 n=np.sum(high,axis=1)
 counts=np.bincount(n,minlength=4)
 print('th',th,'nhigh counts',counts)
 # count combinations among n>=1
 from collections import Counter
 print(Counter([tuple(np.where(h)[0]) for h in high if h.sum()>=1]).most_common(20))
PY"
