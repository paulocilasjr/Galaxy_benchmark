/bin/bash -c "python3 - <<'PY'
import scanpy as sc, numpy as np, pandas as pd
adata=sc.read('adata_unfiltered.h5ad')
cl=adata.obs['leiden_0.5'].astype(int).values
feat=adata.obs.nFeature.values
cnt=adata.obs.nCount.values
for cond_name, mask in [
 ('feat<=929', feat<=929),
 ('feat between 878 and 929', (feat>=878)&(feat<=929)),
 ('feat between 878 and 920', (feat>=878)&(feat<=920)),
 ('feat between 878 and 930', (feat>=878)&(feat<=930)),
 ('cnt low? >9250?', cnt>9250), # no
 ('feat>=878 & cnt?', (feat>=878)&(feat<=929)&(cnt>9250)),
 ('feat<=929 & cnt>=9300', (feat<=929)&(cnt>=9300)),
]:
 tab=pd.Series(cl[mask]).value_counts().sort_index()
 print(cond_name,'kept total',tab.sum(),'tab',tab.to_dict(),'props',[round(v/tab.sum()*100,1) for v in tab])
# MAD thresholds
for name,arr in [('feat',feat),('cnt',cnt)]:
 med=np.median(arr); mad=np.median(np.abs(arr-med)); lo=med-3*mad; hi=med+3*mad
 print(name,'med',med,'mad',mad,'lo',lo,'hi',hi,'low drop',(arr<lo).sum(),'high drop',(arr>hi).sum())
 mask=(arr>=lo)&(arr<=hi)
 tab=pd.Series(cl[mask]).value_counts().sort_index(); print('within',tab.to_dict(), 'total',tab.sum(),'props',[round(v/tab.sum()*100,1) for v in tab])
PY"
