/bin/bash -c "python3 - <<'PY'
# Check marker gene indices ranges in top 50 for each cluster
import scanpy as sc, numpy as np, gzip
from scipy.io import mmread
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f: M=mmread(f)
X=np.asarray(M.T,dtype=float)
adata=sc.read('adata_unfiltered.h5ad'); cl=adata.obs['leiden_0.5'].astype(int).values
lib=X.sum(1); Xn=np.log1p(X/lib[:,None]*10000)
for c in [0,1,2]:
 m=Xn[cl==c].mean(0)
 top=np.argsort(m)[::-1][:100]
 print('cluster',c,'top gene ranges min max',min(top),max(top),'count under?')
 # counts by 100 block
 hist, _ = np.histogram(top,bins=np.arange(0,1001,50))
 print('hist50',hist)
 print('top20',top[:20])
PY"
