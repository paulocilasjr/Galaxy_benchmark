/bin/bash -c "python3 - <<'PY'
import scanpy as sc, numpy as np, pandas as pd
adata=sc.read('adata_unfiltered.h5ad')
cl=adata.obs['leiden_0.5'].astype(int).values
feat=adata.obs.nFeature.values
# threshold examine
for t in range(910,971,5):
 mask=feat>t
 print('t',t,'drop',mask.sum(),'drop by cluster',pd.Series(cl[mask]).value_counts().to_dict())
# exact boundaries
for c in sorted(np.unique(cl)):
 vals=feat[cl==c]
 print('cl',c,'n',len(vals),'min max',vals.min(),vals.max(), 'quant',np.quantile(vals,[0,.01,.05,.5,.95,.99,1]))
PY"
