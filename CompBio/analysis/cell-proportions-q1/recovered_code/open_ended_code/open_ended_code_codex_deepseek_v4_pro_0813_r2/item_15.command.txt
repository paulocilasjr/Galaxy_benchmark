 print('\\nCluster',c,'size',(cl==c).sum())
 print(' top mean lognorm genes:')
 print('  '+', '.join([f'g{i}:{m[i]:.2f}' for i in top]))
# Pairwise correlations of cluster means
means=np.array([Xnorm[cl==c].mean(0) for c in sorted(np.unique(cl))])
