/bin/bash -c "python3 - <<'PY'
import scanpy as sc, numpy as np, pandas as pd
adata=sc.read('adata_unfiltered.h5ad')
cl=adata.obs['leiden_0.5'].astype(int).values
feat=adata.obs.nFeature.values
for t in [929,930,920]:
 mask=feat<=t
 tab=pd.Series(cl[mask]).value_counts().sort_index()
 print('threshold <=',t,'kept',tab.to_dict(),'total',tab.sum(),'props',[f'{v/tab.sum()*100:.1f}' for v in tab], 'rounded',[int(round(v/tab.sum()*100/5)*5) for v in tab])
# after dropping only clusters 3-5
keep=~np.isin(cl,[3,4,5])
tab=pd.Series(cl[keep]).value_counts().sort_index(); print('drop doublet clusters kept',tab.to_dict(),'total',tab.sum(),'props',[f'{v/tab.sum()*100:.1f}' for v in tab])
PY"
