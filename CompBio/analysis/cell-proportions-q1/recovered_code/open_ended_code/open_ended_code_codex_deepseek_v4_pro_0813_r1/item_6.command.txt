/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrows,ncols=map(int,line.split()); break
    arr=np.loadtxt(f)
X=arr.reshape((nrows,ncols), order='F')
tot=X.sum(0); ng=(X>0).sum(0)
# print histogram bins
for name, a in [('tot',tot),('ng',ng)]:
    hist,bins=np.histogram(a,bins=30)
    print(name,'bins')
    for h,b0,b1 in zip(hist,bins[:-1],bins[1:]):
        print(f'{b0:.0f}-{b1:.0f}: {h}')
PY"
