/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
from sklearn.decomposition import PCA
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrows,ncols=map(int,line.split()); break
    arr=np.loadtxt(f)
X=arr.reshape((nrows,ncols), order='F')
lib=X.sum(0,keepdims=True); Y=np.log1p(X/lib*1e4)
pca=PCA(n_components=5).fit(Y.T); Z=pca.transform(Y.T)
# print quantiles / summary of PCs; correlation with total/ng
for i in range(5):
    corr_t=np.corrcoef(Z[:,i],X.sum(0))[0,1]
    corr_g=np.corrcoef(Z[:,i],(X>0).sum(0))[0,1]
    print('PC',i+1,'corr tot/ng',corr_t,corr_g,'range',Z[:,i].min(),Z[:,i].max())
# save z for plotting maybe
np.save('[REDACTED LOCAL PATH]',Z)
PY"
