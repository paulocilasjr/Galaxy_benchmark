/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, anndata as ad, scanpy as sc
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrows,ncols=map(int,line.split()); break
    arr=np.loadtxt(f)
X=arr.reshape((nrows,ncols), order='F')
adata=ad.AnnData(X.T) # cells x genes
adata.var_names=[f'gene{i}' for i in range(X.shape[0])]
adata.obs_names=[f'cell{i}' for i in range(X.shape[1])]
sc.pp.calculate_qc_metrics(adata, percent_top=None, inplace=True)
print(adata)
print(adata.obs[['total_counts','n_genes_by_counts']].describe(percentiles=[.01,.05,.1,.25,.5,.75,.9,.95,.99]).to_string())
# identify cells outside typical: mad-based thresholds standard used by scanpy (is_outlier)
import scipy.stats as st
for metric in ['total_counts','n_genes_by_counts']:
    vals=adata.obs[metric]
    med=np.median(vals); mad=st.median_abs_deviation(vals)
    print(metric,'median',med,'mad',mad,'threshold 5 mad', med-5*mad, med+5*mad, 'outliers low', (vals<med-5*mad).sum(),'high',(vals>med+5*mad).sum(),'3mad', (vals<med-3*mad).sum(),(vals>med+3*mad).sum())
PY"
