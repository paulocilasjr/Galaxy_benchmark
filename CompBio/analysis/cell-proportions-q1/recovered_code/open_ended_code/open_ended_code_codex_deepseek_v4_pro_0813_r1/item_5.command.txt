/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
from scipy import sparse
from sklearn.decomposition import PCA, TruncatedSVD
from sklearn.preprocessing import normalize, StandardScaler
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrows,ncols=map(int,line.split()); break
    arr=np.loadtxt(f)
X=arr.reshape((nrows,ncols), order='F')
print('X shape',X.shape)
# CPM + log1p
lib=X.sum(axis=0,keepdims=True)
Y=np.log1p(X/lib*1e4)
# PCA
pca=PCA(n_components=50, random_state=0).fit(Y.T)
print('pca var',pca.explained_variance_ratio_[:20].round(4), 'cum',np.cumsum(pca.explained_variance_ratio_)[:20].round(3))
Z=pca.transform(Y.T)
# Kmeans k=2..10 on top pcs
for k in range(2,11):
    km=KMeans(n_clusters=k, n_init=10, random_state=0).fit(Z[:,:20])
    sil=silhouette_score(Z[:,:20], km.labels_, sample_size=2000, random_state=0)
    print('k',k,'sil',sil,'counts',np.bincount(km.labels_))
PY"
