/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, anndata as ad, scanpy as sc
import warnings; warnings.filterwarnings('ignore')
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        nrows,ncols=map(int,line.split()); break
    arr=np.loadtxt(f)
X=arr.reshape((nrows,ncols), order='F')
adata=ad.AnnData(X.T)
adata.var_names=[f'gene{i}' for i in range(X.shape[0])]
adata.obs_names=[f'cell{i}' for i in range(X.shape[1])]
sc.pp.calculate_qc_metrics(adata, inplace=True)
sc.pp.normalize_total(adata, target_sum=1e4)
sc.pp.log1p(adata)
sc.pp.highly_variable_genes(adata, n_top_genes=1000)
sc.pp.scale(adata, max_value=10)
sc.tl.pca(adata, svd_solver='arpack', n_comps=50)
sc.pp.neighbors(adata, n_neighbors=15, n_pcs=50)
sc.tl.leiden(adata, resolution=0.2, random_state=0, key_added='leiden')
cl=adata.obs['leiden'].astype(int)
for c in sorted(cl.unique()):
    m=(cl==c)
    print('cluster',c,'n',m.sum(), 'total mean med min max', adata.obs.loc[m,'total_counts'].mean().round(1), adata.obs.loc[m,'total_counts'].median(), adata.obs.loc[m,'total_counts'].min(), adata.obs.loc[m,'total_counts'].max(), 'genes mean med', adata.obs.loc[m,'n_genes_by_counts'].mean().round(1), adata.obs.loc[m,'n_genes_by_counts'].median(), 'pct cells high total > median+3mad', (adata.obs.loc[m,'total_counts']>9775+3*170).sum(),'genes >899+3*7', (adata.obs.loc[m,'n_genes_by_counts']>920).sum())
PY"
