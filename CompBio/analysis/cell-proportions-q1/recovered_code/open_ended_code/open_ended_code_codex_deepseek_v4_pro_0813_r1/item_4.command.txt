/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        hdr=line.strip().split(); break
    arr=np.loadtxt(f)
print('hdr',hdr,'flat',arr.shape)
nrows,ncols=map(int,hdr)
X=arr.reshape((nrows,ncols), order='F')
print('X',X.shape, X.dtype, X.min(), X.max(), X.mean(), (X==0).sum())
tot=X.sum(axis=0)
ng=(X>0).sum(axis=0)
for q in [0,1,5,10,25,50,75,90,95,99,100]:
 print('tot pct',q,np.percentile(tot,q))
for q in [0,1,5,10,25,50,75,90,95,99,100]:
 print('ng pct',q,np.percentile(ng,q))
idx=np.argsort(tot)
print('low totals', list(zip(idx[:30].tolist(), tot[idx[:30]].tolist(), ng[idx[:30]].tolist())))
print('high totals', list(zip(idx[-20:].tolist(), tot[idx[-20:]].tolist(), ng[idx[-20:]].tolist())))
PY"
