/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np
p='cell.proportions.q1.mtx.gz'
vals=[]
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%'): continue
        break
    hdr=next(f).strip().split()
    print('header',hdr)
    arr=np.loadtxt(f)
print(arr.shape, arr.dtype, arr.min(), arr.max(), arr.mean())
# reshape column major (MatrixMarket array stores column-major)
nrows,ncols=map(int,hdr)
X=arr.reshape((nrows,ncols), order='F')
print(X.shape, X.min(), X.max(), X.mean(), np.sum(X==0))
# per cell totals
tot=X.sum(axis=0)
ng=(X>0).sum(axis=0)
print('totals min max median mean',tot.min(),np.percentile(tot,[1,5,25,50,75,95,99]),tot.max())
print('genes min max median mean',ng.min(),np.percentile(ng,[1,5,25,50,75,95,99]),ng.max())
# print top/bottom
idx=np.argsort(tot)
print('low totals', list(zip(idx[:20], tot[idx[:20]], ng[idx[:20]])))
print('high totals', list(zip(idx[-20:], tot[idx[-20:]], ng[idx[-20:]])))
PY"
