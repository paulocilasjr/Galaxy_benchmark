/bin/bash -lc "python - <<'PY'
import numpy as np, scipy.io as sio, gzip
from scipy.io import mmread
X = mmread('cell.proportions.q1.mtx.gz')
print(type(X), X.shape, X.dtype)
X = np.asarray(X, dtype=np.float32) # genes x cells
print('min max mean', X.min(), X.max(), X.mean())
tot = X.sum(axis=0)
det = (X>0).sum(axis=0)
print('tot quantiles', np.quantile(tot,[0,0.01,0.05,0.1,0.25,0.5,0.75,0.9,0.95,0.99,1]))
print('det quantiles', np.quantile(det,[0,0.01,0.05,0.1,0.25,0.5,0.75,0.9,0.95,0.99,1]))
print('zero fraction', np.mean(X==0), 'negative', np.mean(X<0))
# gene stats
print('gene mean q', np.quantile(X.mean(axis=1), [0,0.1,0.5,0.9,1]))
print('gene var q', np.quantile(X.var(axis=1), [0,0.1,0.5,0.9,1]))
PY"
