/bin/bash -lc "python - <<'PY'
import gzip, numpy as np
p='cell.proportions.q1.mtx.gz'
with gzip.open(p,'rt') as f:
    for line in f:
        if line.startswith('%%'): continue
        if line.startswith('%'): continue
        dims=tuple(map(int,line.split())); break
    print('dims',dims)
    vals=np.fromiter((float(x) for line in f for x in [line]),dtype=np.float64,count=dims[0]*dims[1])
print('nvals',vals.size,'min/max/mean',vals.min(),vals.max(),vals.mean(),'zeros',np.sum(vals==0),'zero fraction',np.mean(vals==0))
X=vals.reshape(dims,order='F')
cs=X.sum(axis=0); gn=(X>0).sum(axis=0); gs=X.sum(axis=1); gc=(X>0).sum(axis=1)
for name,a in [('cell_sum',cs),('cell_nnz',gn),('gene_sum',gs),('gene_nnz',gc)]:
 print(name,'min',np.min(a),'p1',np.percentile(a,1),'p5',np.percentile(a,5),'p25',np.percentile(a,25),'median',np.median(a),'p75',np.percentile(a,75),'p95',np.percentile(a,95),'p99',np.percentile(a,99),'max',np.max(a))
print('first sums',cs[:10]); print('first genes',gs[:20]); print('cell sums quant unique rounded', np.unique(np.round(cs,3),return_counts=True)[0][:20])
PY"
