/bin/bash -lc "python - <<'PY'
import gzip,numpy as np,time
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score,calinski_harabasz_score
with gzip.open('cell.proportions.q1.mtx.gz','rt') as f:
 for line in f:
  if line.startswith('%'): continue
  nr,nc=map(int,line.split());break
 vals=np.fromiter((float(line) for line in f),dtype=np.float32,count=nr*nc)
 X=vals.reshape((nr,nc),order='F').T
# library size normalize and log
X=X/X.sum(1,keepdims=True)*1e4
X=np.log1p(X)
# select variable genes top 500 by variance
var=X.var(0)
idx=np.argsort(var)[-500:]
Xs=X[:,idx]
# PCA
pca=PCA(n_components=30,random_state=0)
Z=pca.fit_transform(Xs)
print('PCA variance',np.round(pca.explained_variance_ratio_[:15],4), 'cum30',pca.explained_variance_ratio_.sum())
for k in range(2,11):
 t=time.time(); km=KMeans(n_clusters=k,n_init=10,random_state=1,algorithm='lloyd').fit(Z[:,:20])
 sil=silhouette_score(Z[:,:20],km.labels_,sample_size=3000,random_state=1)
 ch=calinski_harabasz_score(Z[:,:20],km.labels_)
 counts=np.bincount(km.labels_)
 print('k',k,'sil',round(sil,4),'CH',round(ch,1),'counts',sorted(counts,reverse=True),'inertia',round(km.inertia_,1),'time',round(time.time()-t,1))
PY"
