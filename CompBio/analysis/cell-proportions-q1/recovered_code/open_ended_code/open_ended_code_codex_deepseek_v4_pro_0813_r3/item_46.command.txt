/bin/bash -lc "python - <<'PY'
import numpy as np
from sklearn.cluster import KMeans
from sklearn.mixture import GaussianMixture
from sklearn.metrics import silhouette_score, calinski_harabasz_score, davies_bouldin_score
Z=np.load('pca_standardized.npy')
g=GaussianMixture(n_components=6,covariance_type='full',random_state=42,n_init=50,max_iter=500).fit(Z[:,:2])
lab=g.predict(Z[:,:2])
counts=np.bincount(lab)
small=np.argsort(counts)[:3]
keep=~np.isin(lab,small)
Zk=Z[keep]
for n in [2,3,5,10,20,50]:
 data=Zk[:,:n]
 print('\\nnpc',n)
 for k in range(2,7):
  km=KMeans(n_clusters=k,n_init=50,random_state=0).fit(data)
  sil=silhouette_score(data,km.labels_,sample_size=2000,random_state=0)
  ch=calinski_harabasz_score(data,km.labels_)
  db=davies_bouldin_score(data,km.labels_)
  print(k, 'sil %.3f ch %.0f db %.2f'%(sil,ch,db),end='; ')
 print()
PY"
