/bin/bash -lc "python - <<'PY'
import scipy.io,gzip,numpy as np,time
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f:A=scipy.io.mmread(f)
A=np.asarray(A).astype(np.float64)
# normalize
lib=A.sum(0)
X=np.log1p(A/lib*1e4)
# optionally standardize genes? scRNA uses scale max? Let's first just center/scale
Xc=X-X.mean(1,keepdims=True)
Xs=Xc/X.std(1,keepdims=True)
print('nan?',np.isnan(Xs).sum(), np.isfinite(Xs).all())
t=time.time()
pca=PCA(n_components=50, random_state=0)
Z=pca.fit_transform(Xs.T) # samples x pc
print('pca time',time.time()-t,'explained first 20',np.round(pca.explained_variance_ratio_[:20],4), 'cum50',pca.explained_variance_ratio_.sum())
np.save('pca_standardized.npy',Z)
print('saved')
PY"
