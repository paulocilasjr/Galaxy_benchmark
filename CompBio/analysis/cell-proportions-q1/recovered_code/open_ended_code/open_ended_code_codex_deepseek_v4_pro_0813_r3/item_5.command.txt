/bin/bash -lc "python - <<'PY'
import scipy.io, gzip, numpy as np, pandas as pd
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f:
    A=scipy.io.mmread(f)
A=np.asarray(A)
print(A.shape)
# QC per cell
libsize=A.sum(0)
ngenes=(A>0).sum(0)
print('libsize min/max/median', libsize.min(),libsize.max(),np.median(libsize))
print('ngenes min/max/median', ngenes.min(),ngenes.max(),np.median(ngenes))
print('libsize percentiles', np.percentile(libsize,[0,1,5,10,25,50,75,90,95,99,100]))
print('ngenes percentiles', np.percentile(ngenes,[0,1,5,10,25,50,75,90,95,99,100]))
# potential outlier low
for thr in [1000,1500,2000,3000,4000,5000]:
    print('lib<thr',thr,(libsize<thr).sum())
for thr in [100,200,300,400,500]:
    print('ngenes<thr',thr,(ngenes<thr).sum())
PY"
