/bin/bash -lc "python - <<'PY'
import numpy as np
from scipy.stats import gaussian_kde
Z=np.load('pca_standardized.npy')
x=Z[:,0]; y=Z[:,1]
kde=gaussian_kde(np.vstack([x,y]), bw_method='scott')
# grid
xx,yy=np.meshgrid(np.linspace(x.min(),x.max(),200), np.linspace(y.min(),y.max(),200))
d=kde(np.vstack([xx.ravel(),yy.ravel()])).reshape(xx.shape)
# local maxima find
from scipy.ndimage import maximum_filter
loc=(d==maximum_filter(d,size=15)) & (d>np.percentile(d,95))
peaks=np.argwhere(loc)
vals=d[loc]
print('n local max',len(vals))
for (i,j),v in sorted(zip(peaks,vals),key=lambda t:-t[1])[:20]:
    print('peak x,y,val', xx[i,j], yy[i,j], v)
# contour density percentile counts
for q in [90,95,99]:
 print(q,(d>np.percentile(d,q)).sum())
PY"
