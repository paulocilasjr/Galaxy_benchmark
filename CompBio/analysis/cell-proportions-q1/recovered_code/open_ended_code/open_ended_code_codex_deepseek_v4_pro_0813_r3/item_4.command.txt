/bin/bash -lc "python - <<'PY'
import scipy.io, gzip, time, numpy as np
p='cell.proportions.q1.mtx.gz'
t=time.time()
with gzip.open(p,'rb') as f:
    X=scipy.io.mmread(f)
print('time',time.time()-t, 'type', type(X), 'shape', X.shape, 'nnz', X.nnz if hasattr(X,'nnz') else '')
print(X.dtype)
# inspect
if hasattr(X,'toarray'):
    A=X.toarray()
else: A=X
print(A.min(),A.max(),A.mean())
print('rows sum stats', A.sum(1).min(), A.sum(1).max())
print('cols sum stats', A.sum(0).min(), A.sum(0).max())
print(A[:5,:5])
PY"
