/bin/bash -lc "python - <<'PY'
import numpy as np
from scipy.ndimage import maximum_filter
Z=np.load('pca_standardized.npy')
x=Z[:,0]; y=Z[:,1]
for nb in [80,120,160]:
 H,xe,ye=np.histogram2d(x,y,bins=nb)
 # local maxima with size
 loc=(H==maximum_filter(H,size=5))&(H>0)
 peaks=np.argwhere(loc)
 print('\\nnb',nb,'peaks',len(peaks))
 items=[]
 for i,j in peaks:
  xx=(xe[i]+xe[i+1])/2; yy=(ye[j]+ye[j+1])/2
  items.append((H[i,j],xx,yy))
 for v,xx,yy in sorted(items,key=lambda t:-t[0])[:20]:
  print(' count',v,'x',round(xx,2),'y',round(yy,2))
PY"
