/bin/bash -lc "python - <<'PY'
import numpy as np
Z=np.load('pca_standardized.npy')
x=Z[:,0]; y=Z[:,1]
# use 80x50 bins, print quantized ascii with log counts
xb=np.linspace(x.min(),x.max(),81); yb=np.linspace(y.min(),y.max(),41)
H,_,_=np.histogram2d(x,y,bins=[xb,yb])
# trim zero rows/cols
for i in range(H.shape[0]-1,-1,-1):
    row=H[i]
    mx=row.max()
    if mx==0: continue
    line=''.join(' .:-=+*#%@'[min(int(np.log1p(v)/np.log1p(mx)*9),9)] if v>0 else ' ' for v in row)
    print(line.rstrip())
PY"
