/bin/bash -lc "python - <<'PY'
import numpy as np
Z=np.load('pca_standardized.npy')
x=Z[:,0]; y=Z[:,1]
# grid counts 50x30 within range
xmin,xmax=x.min(),x.max();ymin,ymax=y.min(),y.max()
xb=np.linspace(xmin,xmax,61); yb=np.linspace(ymin,ymax,31)
H,_,_=np.histogram2d(x,y,bins=[xb,yb])
chars=' .:-=+*#%@'
# print row max scaling
for i in range(H.shape[0]-1,-1,-1):
    row=H[i]
    mx=row.max()
    line=''.join(chars[min(int(v/mx*9),9)] if mx>0 else ' ' for v in row)
    print(line)
PY"
