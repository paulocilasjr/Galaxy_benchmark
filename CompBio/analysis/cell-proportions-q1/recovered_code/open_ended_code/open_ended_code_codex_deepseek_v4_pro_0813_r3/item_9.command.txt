/bin/bash -lc "python - <<'PY'
import numpy as np, time
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score, calinski_harabasz_score, davies_bouldin_score
Z=np.load('pca_standardized.npy')
# Use PC1-2 only? scRNA cluster often all PCs. Let's test KMeans on first N PCs varying.
for n in [2,5,10,20,50]:
    data=Z[:,:n]
    print('\\nnpc',n)
    for k in range(2,11):
        km=KMeans(n_clusters=k, n_init=20, random_state=42).fit(data)
        sil=silhouette_score(data,km.labels_,sample_size=2000,random_state=0)
        ch=calinski_harabasz_score(data,km.labels_)
        print(k, 'sil %.3f ch %.0f'%(sil,ch), end='; ')
    print()
PY"
