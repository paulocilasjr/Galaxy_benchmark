/bin/bash -lc "python - <<'PY'
import numpy as np
from sklearn.mixture import GaussianMixture
Z=np.load('pca_standardized.npy')
for n in [2,3,4,5,10]:
    data=Z[:,:n]
    g=GaussianMixture(n_components=6,covariance_type='full',random_state=42,n_init=50,max_iter=500).fit(data)
    lab=g.predict(data)
    counts=np.bincount(lab,minlength=6)
    print('\\nnpc',n)
    print('counts',counts)
    print('percents',np.round(counts/len(lab)*100,2))
    print('means\\n',np.round(g.means_,2)[:,:min(n,4)])
PY"
