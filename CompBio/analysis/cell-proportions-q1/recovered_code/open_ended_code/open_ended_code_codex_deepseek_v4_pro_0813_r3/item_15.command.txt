/bin/bash -lc "python - <<'PY'
import scipy.io,gzip,numpy as np
from sklearn.cluster import KMeans
Z=np.load('pca_standardized.npy')
with gzip.open('cell.proportions.q1.mtx.gz','rb') as f:A=scipy.io.mmread(f)
A=np.asarray(A)
lib=A.sum(0); genes=(A>0).sum(0); maxexpr=A.max(0)
# use KMeans k=6 PC2
labels=KMeans(n_clusters=6,n_init=100,random_state=42).fit_predict(Z[:,:2])
for c in range(6):
 idx=np.where(labels==c)[0]
 print('cluster',c,'n',len(idx),'lib median/min/max',np.median(lib[idx]),lib[idx].min(),lib[idx].max(),'genes med/min/max',np.median(genes[idx]),genes[idx].min(),genes[idx].max(),'maxexpr med',np.median(maxexpr[idx]))
# also percentage zeros
zero=(A==0).mean(0)
for c in range(6):
 idx=np.where(labels==c)[0]
 print('c',c,'zero med',np.median(zero[idx]),'zero range',zero[idx].min(),zero[idx].max())
PY"
