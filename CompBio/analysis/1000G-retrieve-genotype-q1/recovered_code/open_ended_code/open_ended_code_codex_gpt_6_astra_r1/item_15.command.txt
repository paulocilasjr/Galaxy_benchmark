from pathlib import Path
samples = ['HG01248','NA19204','NA18614','NA21123','NA19171','HG02303','HG01274','HG03134','NA18639','HG01383']
rows = Path('genotypes.tsv').read_text().splitlines()
assert len(rows) == 1
fields = rows[0].split('\\t')
assert fields[:4] == ['chr10','110918899','G','T']
genotypes = []
for sample, field in zip(samples, fields[4:], strict=True):
    name, gt = field.split('=')
    assert name == sample
    genotypes.append('/'.join(sorted(gt.replace('|','/').split('/'))))
answer = ','.join(genotypes)
assert len(genotypes) == 10 and all(gt in {'0/0','0/1','1/1'} for gt in genotypes)
Path('answer.txt').write_text(answer + '\\n')
print(answer)
PY"
