/bin/bash -lc "python3 - <<'PY'
import subprocess
samples='HG01248,NA19204,NA18614,NA21123,NA19171,HG02303,HG01274,HG03134,NA18639,HG01383'
url='http://ftp.1000genomes.ebi.ac.uk/vol1/ftp/data_collections/1000G_2504_high_coverage/working/20220422_3202_phased_SNV_INDEL_SV/1kGP_high_coverage_Illumina.chr10.filtered.SNV_INDEL_SV_phased_panel.vcf.gz'
out=subprocess.check_output(['bcftools','query','-s',samples,'-f','[\\t%GT]','-r','chr10:110918899-110918899',url], stderr=subprocess.DEVNULL).decode().strip().lstrip('\\t')
gts=[]
for gt in out.split('\\t'):
    a,b=gt.split('|')
    if int(a)>int(b): a,b=b,a
    gts.append(f'{a}/{b}')
print(','.join(gts))
PY"
