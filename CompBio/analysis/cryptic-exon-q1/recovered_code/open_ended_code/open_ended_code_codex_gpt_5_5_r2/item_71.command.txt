/bin/bash -lc "python - <<'PY' > results/top_loci_junction_counts.tsv
import re,collections
counts=collections.Counter()
for line in open('work/top_loci.splice.sam'):
    if line.startswith('@'): continue
    f=line.rstrip('\\n').split('\\t')
    if len(f)<6 or f[2]=='*' or 'N' not in f[5]: continue
    r=f[2]; gene=r.split('|')[0]
    pos=int(f[3])-1 # 0-based offset in fetched locus
    for n,op in re.findall(r'(\\d+)([MIDNSHP=X])', f[5]):
        n=int(n)
        if op in 'M=XDN':
            if op=='N':
                counts[(gene,r,pos,pos+n)] += 1
            pos += n
for (gene,r,s,e),c in counts.most_common():
    print(c,gene,r,s,e,sep='\\t')
PY
head -80 results/top_loci_junction_counts.tsv"
