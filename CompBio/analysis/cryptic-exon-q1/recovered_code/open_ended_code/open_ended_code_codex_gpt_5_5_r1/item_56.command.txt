/bin/bash -lc "python3 - <<'PY'
import gzip,re
symbols={'GNG10','DNAJC25-GNG10','DNAJC25'}
for line in gzip.open('refs/gencode.v47.annotation.gtf.gz','rt'):
    if '\\texon\\t' not in line: continue
    if any(f'gene_name \"{s}\"' in line for s in symbols):
        f=line.rstrip().split('\\t')
        attrs=dict(re.findall(r'(\\S+) \"(["'^"]+)"'"', f[8]))
        print(attrs.get('gene_name'), attrs.get('transcript_id'), f[3], f[4], f[6], attrs.get('exon_number'))
PY"
