/bin/bash -lc "python3 -u - <<'PY'
import gzip,re
attr_re=re.compile(r'(gene_name|gene_id|gene_biotype) \"(["'^"]+)"'"')
info={}; n=0
with gzip.open('Homo_sapiens.GRCh38.116.chr.gtf.gz','rt') as fh:
 for line in fh:
  if line.startswith('#'):continue
  f=line.rstrip('\\n').split('\\t')
  if len(f)<9 or f[2]"'!='"'gene':continue
  a=dict(attr_re.findall(f[8])); info[a.get('gene_id','')]=(f[0],f[6],int(f[3]),int(f[4]))
  n+=1
print('genes',n,len(info),flush=True)
PY"
