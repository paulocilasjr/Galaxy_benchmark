/bin/bash -lc "python3 - <<'PY'
import pysam
bam=pysam.AlignmentFile('aligned.bam','rb')
both=0; only1=0; only2=0
for r in bam.fetch('9',111660000,111667500):
    if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
    js=[]
    ref=r.reference_start
    for op,l in r.cigartuples:
        if op==0: ref+=l
        elif op==1: pass
        elif op==2: ref+=l
        elif op==3:
            js.append((ref+1,ref+l)); ref+=l
    nov1=(111661716,111664536); nov2=(111664590,111666814)
    if nov1 in js and nov2 in js: both+=1
    elif nov1 in js: only1+=1
    elif nov2 in js: only2+=1
print('both',both,'only1',only1,'only2',only2)
PY"
