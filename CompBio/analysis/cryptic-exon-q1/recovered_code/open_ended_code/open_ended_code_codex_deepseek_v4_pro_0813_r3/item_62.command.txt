/bin/bash -lc "samtools view aligned.bam 9:111660000-111667500 > gng10_region.sam; wc -l gng10_region.sam; python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('aligned.bam','rb')
reads=collections.defaultdict(list)
jcount=collections.Counter()
for r in bam.fetch('9',111660000,111667500):
    if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
    ref=r.reference_start
    js=[]
    for op,l in r.cigartuples:
        if op==0: ref+=l
        elif op==1: pass
        elif op==2: ref+=l
        elif op==3:
            js.append((ref+1,ref+l))
            jcount[(ref+1,ref+l)]+=1
            ref+=l
    if js: reads[r.query_name].append((r.reference_name,ref if False else r.reference_start,js, r.get_tag('XS') if r.has_tag('XS') else '.'))
print('junctions in GNG10 region:')
for j,c in sorted(jcount.items(), key=lambda x:x[1], reverse=True):
    print(j,c)
print('\\nReads spanning novel junctions and/or both:')
nov1=(111661716,111664536); nov2=(111664590,111666814)
for name, aligns in reads.items():
    alljs=[j for a in aligns for j in a[2]]
    if nov1 in alljs or nov2 in alljs:
        print(name, aligns)
PY"
