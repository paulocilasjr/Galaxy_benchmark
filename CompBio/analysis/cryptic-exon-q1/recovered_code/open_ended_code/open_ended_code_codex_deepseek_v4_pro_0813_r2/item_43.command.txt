/bin/bash -lc "python3 - <<'PY'
import collections
rows=[]
for line in open('novel_junctions_pc.tsv'):
    if line.startswith('gene_id'): continue
    p=line.rstrip().split('\\t')
    if len(p)<7: continue
    gid,gname,chrom,strand,start,end,cnt=p
    rows.append((gid,gname,chrom,strand,int(start),int(end),int(cnt)))
g=collections.defaultdict(list)
for r in rows: g[(r[0],r[1])].append(r)
print('total genes',len(g))
# summary top by total count
for (gid,name),lst in sorted(g.items(), key=lambda kv:-sum(x[6] for x in kv[1]))[:30]:
    print(name, gid, 'njunc',len(lst), 'totcount',sum(x[6] for x in lst), 'coords', [(x[4],x[5],x[6]) for x in lst])
print('\\nEXACTLY TWO NOVEL JUNCTIONS')
for (gid,name),lst in sorted(g.items(), key=lambda kv:-sum(x[6] for x in kv[1])):
    if len(lst)==2:
        print(name,gid,'tot',sum(x[6] for x in lst),[(x[4],x[5],x[6]) for x in lst])
PY"
