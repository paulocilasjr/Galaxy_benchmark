/bin/bash -lc "python3 - <<'PY'
import re,collections
exons=collections.defaultdict(lambda:collections.defaultdict(list))
for line in open('gencode.v47.annotation.gtf'):
    if line.startswith('#'):continue
    f=line.rstrip('\\n').split('\\t')
    if len(f)<9 or f[2]"'!='"'exon':continue
    attrs=dict(re.findall(r'(\\S+?) \"(["'^"]*)"'"',f[8]))
    if attrs.get('gene_name')=='GLS':
        exons['GLS'][attrs.get('transcript_id','')].append((int(f[3]),int(f[4])))
for tx,es in exons['GLS'].items():
    es=sorted(es)
    print('TX',tx, 'exons',es[:20])
    ints=[(e1+1,s2-1) for (s1,e1),(s2,e2) in zip(es,es[1:]) if s2>e1+1]
    print(' ints near', [(a,b) for a,b in ints if a<=190921200 and b>=190910000][:10])
PY"
