/bin/bash -lc "python3 - <<'PY'
import re,collections
exons=collections.defaultdict(lambda:collections.defaultdict(list))
for line in open('gencode.v47.annotation.gtf'):
    if line.startswith('#'):continue
    f=line.rstrip('\\n').split('\\t')
    if len(f)<9 or f[2]"'!='"'exon':continue
    attrs=dict(re.findall(r'(\\S+?) \"(["'^"]*)"'"',f[8]))
    exons[attrs.get('gene_name','')][attrs.get('transcript_id','')].append((int(f[3]),int(f[4])))
for tx,es in exons['ACTB'].items():
    es=sorted(es); ints=[(e1+1,s2-1) for (s1,e1),(s2,e2) in zip(es,es[1:]) if s2>e1+1]
    print(tx, ints)
PY"
