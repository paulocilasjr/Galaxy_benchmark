/bin/bash -lc "python3 - <<'PY'
import urllib.request,json,time
url='https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chr6;start=73489308;end=73525587'
t=time.time()
try:
    with urllib.request.urlopen(url, timeout=60) as r:
        d=json.load(r)
    print(len(d.get('dna','')), d.get('dna','')[:20], time.time()-t)
except Exception as e: print('ERR',e)
PY"
