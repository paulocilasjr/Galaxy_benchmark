/bin/bash -lc "python3 - <<'PY'
import gzip,re,collections,pickle
known=set(); exons=collections.defaultdict(list)
with gzip.open('gencode.v38.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'):continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9 or p[2]"'!='"'exon':continue
        attrs=dict(re.findall(r'(\\S+?) \"(["'^"]*)"'"',p[8])); tx=attrs.get('transcript_id','')
        exons[tx].append((p[0],int(p[3]),int(p[4])))
for tx,es in exons.items():
    es=sorted(es)
    for (chrom,s1,e1),(chrom2,s2,e2) in zip(es,es[1:]):
        if chrom==chrom2 and s2>e1+1:
            known.add((chrom,e1+1,s2-1))
pickle.dump(known,open('known_v38.pkl','wb'))
print('known v38 introns',len(known))
PY"
