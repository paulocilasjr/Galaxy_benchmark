/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
cands={
'PHB2':('chr12',[(6965713,6966414),(6967249,6967670)]),
'EEF1A1':('chr6',[(73517719,73518159),(73518444,73519409)]),
'RPS4X':('chrX',[(72273967,72275052),(72275175,72275543)]),
'HLA-C':('chr6',[(31268901,31269087),(31269184,31269337)]),
'CD74':('chr5',[(150402021,150402562),(150402626,150405084)]),
}
def get(chrom,start,end):
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={start-1};end={end}'
    with urllib.request.urlopen(url,timeout=20) as r:d=json.load(r)
    return d['dna']
for gene,(chrom,ints) in cands.items():
    print('\\n',gene,chrom)
    for gs,ge in ints:
        left=get(chrom,gs-5,gs+4); right=get(chrom,ge-3,ge+6)
        print('intron',gs,ge,'lower',[(gs-4+i,c) for i,c in enumerate(left)],'upper',[(ge-2+i,c) for i,c in enumerate(right)])
PY"
