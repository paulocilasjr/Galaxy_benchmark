/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
cands={
'EEF1A1':[(73517719,73518159),(73518444,73519409)],
'RPS4X':[(72273967,72275052),(72275175,72275543)],
'HLA-C':[(31268901,31269087),(31269184,31269337)],
'CD74':[(150402021,150402562),(150402626,150405084)],
}
chroms={'EEF1A1':'chr6','RPS4X':'chrX','HLA-C':'chr6','CD74':'chr5'}
def get(chrom,start,end):
    url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom={chrom};start={start-1};end={end}'
    with urllib.request.urlopen(url,timeout=20) as r:d=json.load(r)
    return d['dna']
for gene,ints in cands.items():
    print('\\n',gene)
    for gs,ge in ints:
        left=get(chroms[gene],gs-5,gs+4)
        right=get(chroms[gene],ge-3,ge+6)
        # UCSC start gs-5 0-based -> 1-based gs-4..gs+4? Let's print raw with positions via enumerate
        print('intron',gs,ge)
        print(' left', ''.join(f'{i}:{c}' for i,c in enumerate(left,gs-4)))
        print(' right', ''.join(f'{i}:{c}' for i,c in enumerate(right,ge-2)))
PY"
