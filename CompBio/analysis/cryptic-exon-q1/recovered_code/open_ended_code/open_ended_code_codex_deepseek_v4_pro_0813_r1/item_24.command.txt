/bin/bash -lc "python3 - <<'PY'
import subprocess, collections
name2={}
for line in open('gencode.v47.transcripts.fa'):
    if line.startswith('>'):
        h=line[1:].strip()
        parts=h.split('|')
        name2[h]=parts[5] if len(parts)>5 else '?'
counts=collections.Counter()
for line in subprocess.Popen(['samtools','view','-F','256','trans_aln.sorted.bam'],stdout=subprocess.PIPE,text=True).stdout:
    fields=line.split('\\t')
    if len(fields)<3: continue
    ref=fields[2]
    if ref=='*': continue
    counts[name2.get(ref,'?')]+=1
for gene,c in counts.most_common(50):
    print(f'{gene}\\t{c}')
print('total genes',len(counts))
PY"
