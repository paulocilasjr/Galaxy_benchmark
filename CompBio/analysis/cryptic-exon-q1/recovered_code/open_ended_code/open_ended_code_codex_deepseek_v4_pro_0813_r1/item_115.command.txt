/bin/bash -lc "python3 - <<'PY'
import collections,gzip
# Build set of unmapped read IDs
ids=set()
import subprocess
for line in subprocess.Popen(['samtools','view','-f','4','trans_aln.sorted.bam'],stdout=subprocess.PIPE,text=True).stdout:
    ids.add(line.split('\\t')[0])
print('unmapped',len(ids))
seqs=collections.Counter(); total=0
with gzip.open('cryptic.exon.q1.fq.gz','rt') as f:
    while True:
        h=f.readline()
        if not h:break
        s=f.readline().strip(); f.readline(); f.readline()
        total+=1
        if h[1:].split()[0] in ids:
            seqs[s]+=1
print('seqs',len(seqs),'total counted',sum(seqs.values()))
for s,c in seqs.most_common(30):
    print(c,s)
PY"
