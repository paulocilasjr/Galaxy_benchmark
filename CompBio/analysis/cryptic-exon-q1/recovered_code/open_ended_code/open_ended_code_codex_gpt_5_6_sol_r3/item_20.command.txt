/bin/bash -lc "samtools view -F 2308 cryptic_tmp/genome.splice.bam | perl -lane '"'$chr=$F[2]; $p=$F[3]; $c=$F[5]; while($c=~/('"\\d+)([MIDNSHP=X])/g){("'$n,$op)=($1,$2); if($op eq "N"){print join("'"\\t\","'$chr,$p-1,$p+$n,$n,$F[0],$F[4],$F[5])} $p+=$n if $op=~/[MDN=X]/}'"' > cryptic_tmp/junction_reads.tsv
cut -f1-3 cryptic_tmp/junction_reads.tsv | sort | uniq -c | sort -nr | head -100"
