/bin/bash -lc "gzip -cd cryptic.exon.q1.fq.gz | awk 'NR%4==2{print}' | sort | uniq -c | sort -nr | head -20 | awk '{print \">top\"NR\"_count\""'$1"'"\\n\""'$2}'"' > top20.fa
blastn -remote -db nt -query top20.fa -out top20.blast.tsv -outfmt '6 qseqid saccver pident length mismatch gapopen qstart qend sstart send evalue bitscore stitle' -max_target_seqs 5 -task blastn"
