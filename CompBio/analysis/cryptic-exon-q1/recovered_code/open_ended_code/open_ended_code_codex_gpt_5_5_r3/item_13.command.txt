/bin/bash -lc "python3 - <<'PY'
import gzip, collections
k=31
c=collections.Counter()
with gzip.open('cryptic.exon.q1.fq.gz','rt') as f:
    for i,line in enumerate(f):
        if i%4==1:
            s=line.strip()
            for j in range(len(s)-k+1):
                c[s[j:j+k]]+=1
for mer,n in c.most_common(30):
    print(n, mer)
PY"
