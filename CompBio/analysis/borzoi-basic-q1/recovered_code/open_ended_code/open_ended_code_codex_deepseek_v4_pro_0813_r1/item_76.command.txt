/bin/bash -c "python3 - <<'PY'
import h5py, numpy as np
for fn in ['models/model0_best.h5','models/model1_best.h5']:
    print('\\n',fn)
    with h5py.File(fn,'r') as f:
        for name in f['model_weights'].keys():
            # count datasets
            cnt=0; shape=None; trainable=0
            for k,o in f['model_weights'][name].items():
                if isinstance(o,h5py.Dataset):
                    cnt+=np.prod(o.shape,dtype=np.int64)
                    if not k.startswith('moving_'): trainable+=np.prod(o.shape,dtype=np.int64)
                    if k=='kernel:0': shape=o.shape
            if trainable>1000000 or 'dense_1' in name or 'dense_2' in name:
                print(name, 'trainable',trainable, 'kernel_shape',shape)
PY"
