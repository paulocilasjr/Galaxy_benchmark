/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, json
for p in ['models/model0_best.h5','models/model1_best.h5']:
 with h5py.File(p,'r') as f:
  cfg=json.loads(f.attrs['model_config'])
  finals=[l for l in cfg['config']['layers'] if l['class_name']=='Dense']
  prefix='/'.join(['model_weights', finals[-1]['name'], finals[-1]['name']])
  head=0
  for n in f['model_weights']:
   if n.startswith(prefix+'/'):
    ds=f['model_weights'][n]
    head+=int(np.prod(ds.shape))
  full=0; moving=0
  def visit(name,obj):
   global full,moving
   if isinstance(obj,h5py.Dataset):
    n=int(np.prod(obj.shape)); full+=n
    if name.split('/')[-1].split(':')[0] in ('moving_mean','moving_variance'): moving+=n
  f['model_weights'].visititems(visit)
  print('\\n',p)
  print('head name', finals[-1]['name'], 'head elements', head)
  print('full all',full,'full trainable excl moving',full-moving)
  print('nonhead all = full-head', full-head)
  print('nonhead trainable = full-moving-head', full-moving-head)
PY"
