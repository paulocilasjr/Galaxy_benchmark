/bin/bash -c "python3 - <<'PY'
import h5py, numpy as np
for fn in ['models/model0_best.h5','models/model1_best.h5']:
    print('\\n', fn)
    with h5py.File(fn,'r') as f:
        counts=[]
        non=[]
        for n in f['model_weights'].keys():
            arrs = {}
            for k in f['model_weights'][n].keys():
                d=f['model_weights'][n][k]
                # dataset names often d/kernel:0 etc
                for kk in d.keys():
                    ds=d[kk]
                    arrs[kk]=np.prod(ds.shape,dtype=np.int64)
            total=sum(arrs.values())
            # heuristic moving stats only batch norm has moving_mean/moving_variance
            moving=sum(v for k,v in arrs.items() if k.startswith('moving_'))
            counts.append((n,total,total-moving,moving,arrs))
            if moving: non.append((n,moving))
        print('layers', len(counts),'total',sum(c[1] for c in counts),'trainable',sum(c[2] for c in counts),'moving',sum(c[3] for c in counts))
        for c in non[:20]: print('non-trainable layer',c)
        # Print head-like layers / last layers
        for c in counts[-10:]: print(c[0], c[1], c[2], c[4])
PY"
