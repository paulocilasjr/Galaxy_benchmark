/bin/bash -c "python3 - <<'PY'
import h5py,numpy as np
f0=h5py.File('models/model0_best.h5','r')['model_weights']
f1=h5py.File('models/model1_best.h5','r')['model_weights']
# compare all common top-level names excluding head names
heads={'dense_20','dense_21'}
common=[n for n in f0.keys() if n in f1.keys() and n not in heads]
mismatch=[]
for n in common:
    def arrays(g):
        out=[]
        def rec(o):
            for k,x in o.items():
                if isinstance(x,h5py.Dataset): out.append((k,x[()]))
                else: rec(x)
        rec(g); return out
    a=arrays(f0[n]); b=arrays(f1[n])
    if len(a)"'!=len(b): mismatch.append((n,'"'len',len(a),len(b))); continue
    for (ka,va),(kb,vb) in zip(a,b):
        if va.shape"'!=vb.shape or not np.allclose(va,vb,equal_nan=True):
            mismatch.append((n,ka,va.shape,vb.shape))
            break
print('"'common trunk layers compared',len(common),'mismatches',len(mismatch))
print(mismatch[:10])
PY"
