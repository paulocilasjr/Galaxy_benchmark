/bin/bash -c "python3 - <<'PY'
import h5py,numpy as np
for fn in ['models/model0_best.h5','models/model1_best.h5']:
 print('\\n',fn)
 with h5py.File(fn,'r') as f:
  rows=[]
  for name,obj in f['model_weights'].items():
   t=[0]; shapes=[]
   def rec(g):
    for k,o in g.items():
     if isinstance(o,h5py.Dataset):
      t[0]+=np.prod(o.shape,dtype=np.int64); shapes.append((k,o.shape))
     else: rec(o)
   rec(obj)
   rows.append((t[0],name,shapes))
  for t,name,shapes in sorted(rows, reverse=True)[:20]:
   print(t,name,shapes[:4])
PY"
