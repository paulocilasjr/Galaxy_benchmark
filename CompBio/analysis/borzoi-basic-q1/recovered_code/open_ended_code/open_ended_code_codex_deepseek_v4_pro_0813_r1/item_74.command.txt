import h5py
with h5py.File('models/model0_best.h5','r') as f:
    def show(g,indent=''):
        if hasattr(g,'shape') and isinstance(g.shape,tuple):
            print(indent, 'DATASET', g.name, g.shape); return
        for k in g.keys():
            print(indent,k)
            show(g[k], indent+'  ')
    show(f['model_weights'])
PY"
