/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np
path='model0_best.h5'
total=0
train=0
nontrain=0
train_names=[]
nontrain_names=[]
with h5py.File(path,'r') as f:
    for name, obj in f['model_weights'].items():
        # recursively collect datasets
        pass

def walk(group, prefix=''):
    global total
    datasets=[]
    for key in group:
        obj=group[key]
        if isinstance(obj, h5py.Group):
            datasets += walk(obj, prefix+'/'+key)
        else:
            datasets.append((prefix+'/'+key, obj))
    return datasets

with h5py.File(path,'r') as f:
    datasets=walk(f['model_weights'])
    for name,ds in datasets:
        n=int(np.prod(ds.shape, dtype=np.int64))
        total += n
        # trainable except BatchNorm moving stats
        if '/moving_mean:0' in name or '/moving_variance:0' in name:
            nontrain += n
            nontrain_names.append(name)
        else:
            train += n
            train_names.append(name)
print('num datasets', len(datasets))
print('total elements', total)
print('nontrain moving', nontrain)
print('train (all except moving)', train)
print('nontrain names', nontrain_names[:5], '...', len(nontrain_names))
# also print by top categories maybe
from collections import Counter
cats=Counter()
for n,d in datasets:
    parts=n.split('/')
    cats[parts[1] if len(parts)>1 else n] += int(np.prod(d.shape,dtype=np.int64))
for k,v in cats.most_common(20): print(k,v)
PY"
