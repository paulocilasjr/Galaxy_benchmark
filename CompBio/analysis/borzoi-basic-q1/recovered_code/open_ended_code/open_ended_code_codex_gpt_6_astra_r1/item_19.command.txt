/bin/bash -lc "python3 - <<'PY'
import json

def read_weights(filename):
    return {w['name']:w for w in json.load(open(filename+'.weights.json')) if not any(x in w['name'] for x in ['moving_mean','moving_variance'])}
h=read_weights('model0_best.h5')
m=read_weights('model1_best.h5')
shared=h.keys() & m.keys()
assert all(h[k]['shape']==m[k]['shape'] for k in shared)
shared_count=sum(h[k]['count'] for k in shared)
human_only={k:h[k] for k in h.keys()-m.keys()}
mouse_only={k:m[k] for k in m.keys()-h.keys()}
print('Shared trunk:',shared_count)
print('Human head:',human_only)
print('Mouse head:',mouse_only)
assert sum(w['count'] for w in human_only.values()) == 1921*7611
assert sum(w['count'] for w in mouse_only.values()) == 1921*2608
count=sum(w['count'] for w in h.values())+sum(w['count'] for w in m.values())-shared_count
assert count == 190902667
answer=str(count)
assert answer.isdigit() and '\\n' not in answer
open('answer.txt','w').write(answer+'\\n')
print(answer)
PY"
