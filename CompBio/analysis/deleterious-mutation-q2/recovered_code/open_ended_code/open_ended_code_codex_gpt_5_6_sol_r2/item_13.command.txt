/bin/bash -lc "bcftools view -H -r chr9:127661125-127661125 calls.vcf.gz; samtools mpileup -f hg38.chr9.fa -r chr9:127661125-127661125 -q 20 -Q 20 -d 100000 reads.hg38.bam; python - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('reads.hg38.bam')
pos=127661125
for mq,bq in [(0,0),(20,0),(20,20),(30,30)]:
 c=collections.Counter(); strand=collections.Counter(); bqs=collections.defaultdict(list); mqs=collections.defaultdict(list)
 for col in bam.pileup('chr9',pos-1,pos,truncate=True,min_mapping_quality=mq,min_base_quality=bq,max_depth=100000,stepper='all'):
  for p in col.pileups:
   if p.is_del or p.is_refskip: continue
   a=p.alignment
   base=a.query_sequence[p.query_position].upper()
   c[base]+=1; strand[(base,'R' if a.is_reverse else 'F')]+=1
   bqs[base].append(a.query_qualities[p.query_position]); mqs[base].append(a.mapping_quality)
 print('filters',mq,bq,'counts',dict(c),'strand',dict(strand))
 print('quality', {x:(min(bqs[x]),round(sum(bqs[x])/len(bqs[x]),1),min(mqs[x]),round(sum(mqs[x])/len(mqs[x]),1)) for x in c})
PY"
