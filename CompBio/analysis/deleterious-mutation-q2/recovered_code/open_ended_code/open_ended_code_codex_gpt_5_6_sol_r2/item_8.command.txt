/bin/bash -lc "bcftools mpileup -Ou -f hg38.chr9.fa -q 20 -Q 20 -d 100000 -a FORMAT/DP,FORMAT/AD,FORMAT/ADF,FORMAT/ADR reads.hg38.bam | bcftools call -m -v -Oz -o calls.vcf.gz && bcftools index calls.vcf.gz && bcftools stats calls.vcf.gz | sed -n '1,40p' && bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL[\\t%DP\\t%AD\\t%ADF\\t%ADR]\\n' calls.vcf.gz | awk -F'\\t' 'length("'$3)==1 && length($4)==1 && $6>=8 {split($7,a,","); if(a[2]>=2 && a[2]/$6>=0.03 && a[2]/$6<=0.45) print}'"' | sort -k5,5nr | head -100"
