/bin/bash -lc "bcftools mpileup -f chr9.fa -q 20 -Q 20 -a FORMAT/DP,FORMAT/AD -Ou reads.chr9.bam | bcftools call -mv -Oz -o calls.vcf.gz
bcftools index -f calls.vcf.gz
bcftools stats calls.vcf.gz | sed -n '1,80p'
bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t[%DP\\t%AD\\t%AF]\\n' calls.vcf.gz | head -30"
