/bin/bash -lc "set -o pipefail
bcftools mpileup -f chr9.fa -q 20 -Q 20 -d 100000 -a FORMAT/AD,FORMAT/DP -Ou q2.bam | bcftools call -m -v -Ou | bcftools norm -f chr9.fa -Oz -o q2.calls.vcf.gz
bcftools index q2.calls.vcf.gz
bcftools stats q2.calls.vcf.gz | sed -n '/"'^SN/p'"'
bcftools query -f '%CHROM\\t%POS\\t%REF\\t%ALT\\t%QUAL\\t[%DP\\t%AD]\\n' q2.calls.vcf.gz | awk -F'\\t' 'BEGIN{OFS=\"\\t\"} {split("'$7,a,","); if(length($3)==1 && length($4)==1 && a[2]>=2) print $0, a[2]/(a[1]+a[2])}'"' | sort -k8,8nr | head -30"
