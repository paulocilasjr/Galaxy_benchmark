/bin/bash -lc "python3 - <<'PY'
import numpy as np,itertools
samples=[line.split()[1] for line in open('identify.related.q1.tfam') if line.strip()]
gr=np.load('relatedness_grm.npy')
for lo,hi in [(0.3530,0.3542),(0.34,0.36),(0.30,0.40)]:
    pairs=[]
    for i,j in itertools.combinations(range(len(samples)),2):
        r=gr[i,j]
        if lo <= r < hi:
            pairs.append((r,samples[i],samples[j]))
    print('range',lo,hi,'count',len(pairs))
    for r,a,b in sorted(pairs, reverse=True)[:50]:
        print(f'{r:.9f}\\t{a}\\t{b}')
    print()
for cutoff in [0.354,2*(2**-2.5),0.3535,0.35,0.4]:
    flagged=set(); npairs=0
    for i,j in itertools.combinations(range(len(samples)),2):
        if gr[i,j] >= cutoff:
            npairs+=1; flagged.add(samples[i]); flagged.add(samples[j])
    print('cutoff',cutoff,'pairs',npairs,'nind',len(flagged),','.join(sorted(flagged)))
PY"
