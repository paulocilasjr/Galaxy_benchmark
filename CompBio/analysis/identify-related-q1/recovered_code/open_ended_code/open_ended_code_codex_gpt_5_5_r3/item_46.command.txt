/bin/bash -lc "python3 - <<'PY'
from pathlib import Path
path=Path('plink_q1_genome.genome')
with path.open() as f:
    header=f.readline().split()
    rows=[]
    for line in f:
        vals=line.split()
        d=dict(zip(header, vals))
        rows.append(d)
rows.sort(key=lambda d: float(d['PI_HAT']), reverse=True)
print('nrows',len(rows))
for d in rows[:80]:
    print(d['IID1'],d['IID2'],d['RT'],d['Z0'],d['Z1'],d['Z2'],d['PI_HAT'],d['DST'],d['PPC'])
for cutoff in [0.5,0.45,0.4,0.375,0.354,0.35,0.25,0.177]:
    flag=set(); npairs=0
    for d in rows:
        if float(d['PI_HAT']) >= cutoff:
            npairs+=1; flag.add(d['IID1']); flag.add(d['IID2'])
    print('cutoff',cutoff,'pairs',npairs,'nind',len(flag),','.join(sorted(flag)))
PY"
