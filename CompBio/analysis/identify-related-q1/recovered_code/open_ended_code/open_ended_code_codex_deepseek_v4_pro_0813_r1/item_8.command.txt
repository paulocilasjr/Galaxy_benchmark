/bin/bash -lc "python3 - <<'PY'
import gzip, sys, collections
f=gzip.open('identify.related.q1.tped.gz','rt')
alleles=collections.Counter(); ncols=None; missing=0; line_count=0
for line in f:
    parts=line.split()
    if ncols is None:
        ncols=len(parts); print('ncols',ncols)
    else:
        assert len(parts)==ncols, (line_count,len(parts),ncols)
    for a in parts[4:]:
        if a=='0': missing+=1
        else: alleles[a]+=1
    line_count+=1
    if line_count>=100000: break
print('first100k lines',line_count,'missing',missing,'alleles',alleles.most_common(20))
PY"
