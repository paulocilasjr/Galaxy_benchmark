/bin/bash -lc "python3 - <<'PY'
import csv, collections
rows=[]
with open('q1_genome.genome') as f:
    r=csv.DictReader(f, delimiter='\\t')
    for d in r:
        d['PI_HAT']=float(d['PI_HAT']); d['Z0']=float(d['Z0']); d['Z1']=float(d['Z1']); d['Z2']=float(d['Z2'])
        rows.append(d)
# sort pihat
rows.sort(key=lambda d:d['PI_HAT'], reverse=True)
print('top 40 pihat:')
for d in rows[:40]:
    print(d['IID1'], d['IID2'], 'PI_HAT', round(d['PI_HAT'],4), 'Z0',round(d['Z0'],4),'Z1',round(d['Z1'],4),'Z2',round(d['Z2'],4),'DST',d['DST'])
# distribution thresholds
from collections import Counter
print('counts PI_HAT ranges')
for lo,hi in [(0.5,2),(0.45,0.5),(0.4,0.45),(0.35,0.4),(0.3,0.35),(0.25,0.3),(0.2,0.25),(0.1,0.2),(0,0.1)]:
    c=sum(1 for d in rows if lo<=d['PI_HAT']<hi)
    print(lo,hi,c)
print('pairs with pihat>0.4:', sum(1 for d in rows if d['PI_HAT']>0.4))
print('samples involved >0.4:', len(set([d['IID1'] for d in rows if d['PI_HAT']>0.4]+[d['IID2'] for d in rows if d['PI_HAT']>0.4])))
PY"
