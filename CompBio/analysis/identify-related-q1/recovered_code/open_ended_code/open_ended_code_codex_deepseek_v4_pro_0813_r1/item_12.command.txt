/bin/bash -lc "python3 - <<'PY'
import gzip, collections
f=gzip.open('identify.related.q1.tped.gz','rt')
dist=collections.Counter(); multi=0; mono=0
for i,line in enumerate(f,1):
    p=line.split()
    vals=p[4:]
    s=set(vals)
    dist[len(s)]+=1
    if len(s)>2: multi+=1
    if len(s)==1: mono+=1
    if i==1:
        print('line1 unique',s)
print(dist, 'multi',multi,'mono',mono)
PY"
