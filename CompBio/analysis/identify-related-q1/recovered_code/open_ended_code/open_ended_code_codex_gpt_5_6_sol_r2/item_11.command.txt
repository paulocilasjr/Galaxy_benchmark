/bin/bash -lc "python3 - <<'PY'
import numpy as np
n=205
m=500000
bpm=(n+3)//4
ibs0=np.zeros((n,n),dtype=np.int64)
hh=np.zeros((n,n),dtype=np.int64)
hets=np.zeros(n,dtype=np.int64)
lut=np.empty((256,4),dtype=np.uint8)
for b in range(256):
    lut[b]=[(b>>(2*k))&3 for k in range(4)]
with open('identify.related.q1.bed','rb') as f:
    assert f.read(3)==bytes([0x6c,0x1b,0x01])
    for start in range(0,m,10000):
        k=min(10000,m-start)
        raw=np.frombuffer(f.read(k*bpm),dtype=np.uint8).reshape(k,bpm)
        c=lut[raw].reshape(k,bpm*4)[:,:n]
        h=(c==2).astype(np.int64)
        a=(c==0).astype(np.int64)
        z=(c==3).astype(np.int64)
        hets += h.sum(axis=0)
        hh += h.T@h
        ibs0 += a.T@z + z.T@a
kin=(hh-2*ibs0)/(hets[:,None]+hets[None,:])
names=[line.split()[1] for line in open('identify.related.q1.fam')]
pairs=sorted(((kin[i,j],ibs0[i,j],hh[i,j],names[i],names[j]) for i in range(n) for j in range(i+1,n)),reverse=True)
for row in pairs[:25]: print(*row)
PY"
