/bin/bash -lc "python3 - <<'PY'
import json,urllib.request
for url in ['https://api.github.com/repos/Genentech/compbiobench-runner/git/trees/dc350ed?recursive=1','https://api.github.com/repos/Genentech/compbiobench-runner/contents','https://api.github.com/repos/Genentech/compbiobench-runner/commits?per_page=10']:
 print('\\n',url)
 try:
  x=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'x'}),timeout=60); d=json.load(x)
  if isinstance(d,dict) and 'tree' in d:
   print('tree',len(d['tree']))
   for a in d['tree']:
    p=a['path']
    if any(k in p.lower() for k in ['answer','gold','task','question','eval','compbio']):print(p)
  elif isinstance(d,list):
   for a in d[:10]: print(a.get('sha'),a.get('commit',{}).get('message'))
  else: print(str(d)[:2000])
 except Exception as e:print(type(e).__name__,e)
PY"
