/bin/bash -lc "python3 - <<'PY'
import json,urllib.request,urllib.parse
repo='amanutej/trustworthy-biology-agents-traces';base='https://huggingface.co/api/datasets/'+repo+'/tree/main'
all=[]
for page in range(20):
 u=base+'?recursive=true&expand=false&limit=1000&cursor='+urllib.parse.quote(str(page))
 try:d=json.load(urllib.request.urlopen(u,timeout=120))
 except Exception as e:print('err',page,e);break
 print('page',page,'n',len(d));all.extend(d)
 if len(d)<1000:break
print('total',len(all))
for x in all:
 p=x.get('path','')
 if any(k in p.lower() for k in ['compbio','gwas','reverse','source']):print(p,x.get('size'))
print('first')
for x in all[:20]:print(x.get('path'))
print('last')
for x in all[-20:]:print(x.get('path'))
PY"
