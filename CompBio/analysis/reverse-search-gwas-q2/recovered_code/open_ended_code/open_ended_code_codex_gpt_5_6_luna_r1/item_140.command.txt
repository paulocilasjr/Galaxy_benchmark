/bin/bash -lc "python3 -u - <<'PY'
import urllib.request,json
for rs in ['rs1150779','rs6909732','rs2483211','rs7742369','rs4308051','rs8094261','rs7235010','rs4800455','rs7244739','rs2871960','rs6918981','rs224333']:
 try:
  u='https://grch37.rest.ensembl.org/variation/human/'+rs+'?content-type=application/json'
  d=json.load(urllib.request.urlopen(u,timeout=30));
  print('\\n',rs,'MAF',d.get('MAF'),'syn',d.get('synonyms'),flush=True)
  for p in d.get('populations',[]):
   q=p.get('population','')
   if any(k in q.lower() for k in ['1000genomes','korean','gnomad','eas','eur','afr','sas','amr']): print(q,p.get('allele'),p.get('frequency'),flush=True)
 except Exception as e:print('\\n',rs,'ERR',repr(e),flush=True)
PY"
