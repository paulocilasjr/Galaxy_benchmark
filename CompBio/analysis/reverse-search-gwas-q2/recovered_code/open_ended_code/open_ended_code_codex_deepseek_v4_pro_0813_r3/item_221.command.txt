/bin/bash -c "python3 - <<'PY'
import gzip, statistics
p='reverse.search.gwas.q2.gz'
ns=[]; f05=[]
with gzip.open(p,'rt') as f:
 next(f)
 for line in f:
  a=line.rstrip().split('\\t')
  if len(a)<10: continue
  try: af=float(a[6]); se=float(a[8])
  except: continue
  if not (0<af<1 and se>0): continue
  N=1/(2*af*(1-af)*se*se)
  if af>.05 and af<.95 and se<.02: ns.append(N)
  if .4<af<.6: f05.append(N)
print('common count',len(ns),'medianN',statistics.median(ns),'mean',statistics.mean(ns))
print('quantiles',[round(statistics.quantiles(ns,n=10)[i]) for i in [0,2,5,7,9]])
print('f05',len(f05),'median',statistics.median(f05))
for af,se in [(.703,.00669),(.258,.00697),(.631,.00629),(.364,.00625),(.660,.00637)]:
 print(af,se,1/(2*af*(1-af)*se*se))
PY"
