/bin/bash -c "python3 - <<'PY'
import requests, json, time, collections
snps=['rs1150779','rs6909732','rs2483211','rs7742369','rs4308051','rs2871960','rs224333','rs2011603','rs3791679','rs9435734','rs798485','rs1812175']
assocs=[]
for snp in snps:
 url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{snp}/associations?size=1000'
 try:
  d=requests.get(url,timeout=30).json()
  arr=d.get('_embedded',{}).get('associations',[])
  print(snp,len(arr))
  for a in arr:
   study=a['_links']['study']['href']
   assocs.append((snp,a.get('pvalue'),a.get('betaNum'),a.get('standardError'),a.get('riskFrequency'),study))
 except Exception as e: print('err',snp,e)
 time.sleep(.1)
# unique studies
studies={}
for snp,p,b,se,rf,href in assocs:
 studies[href]=studies.get(href,0)+1
print('unique studies',len(studies))
# fetch study info, print relevant N/height
count=0
for href,cnt in studies.items():
 try:
  s=requests.get(href,timeout=30).json()
  trait=s.get('diseaseTrait',{}).get('trait','')
  n=s.get('initialSampleSize','')
  pmid=s.get('publicationInfo',{}).get('pubmedId')
  acc=s.get('accessionId')
  if 'height' in trait.lower() or 'body' in trait.lower() or 'spleen' in trait.lower() or 'anthrop' in trait.lower():
   print('STUDY',acc,trait,n,pmid,'cnt',cnt)
 except Exception as e: pass
print('done')
PY"
