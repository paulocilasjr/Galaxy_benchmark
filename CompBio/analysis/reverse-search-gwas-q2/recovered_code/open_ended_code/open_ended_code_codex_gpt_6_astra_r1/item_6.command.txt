/bin/bash -lc "cat > lookup.py <<'PY'
import requests,concurrent.futures,json
snps=['rs4308051','rs2871960','rs224333','rs2011603','rs3791679']
def work(rs):
 j=requests.get(f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?projection=associationBySnp').json();json.dump(j,open(rs+'.json','w'))
 out=[]
 for x in j.get('_embedded',{}).get('associations',[]):
  s=requests.get(x['_links']['study']['href']).json()
  out.append((rs,{k:x.get(k) for k in ['pvalue','betaNum','betaUnit','riskFrequency']},s.get('publicationInfo'),s.get('diseaseTrait'),s.get('accessionId')))
 return out
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as ex:
 for res in ex.map(work,snps):
  for x in res:print(x,flush=True)
PY
python lookup.py"
