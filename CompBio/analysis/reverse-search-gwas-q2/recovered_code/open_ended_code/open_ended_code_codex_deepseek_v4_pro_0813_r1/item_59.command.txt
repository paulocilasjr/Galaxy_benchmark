/bin/bash -lc "python3 - <<'PY'
import requests, json, math
base='https://www.ebi.ac.uk/gwas/rest/api/studies/GCST90095033/associations?projection=associationByStudy&size=100'
assocs=[]; url=base
while url:
    r=requests.get(url,headers={'Accept':'application/json'},timeout=30)
    d=r.json(); emb=d.get('_embedded',{}).get('associations',[]); assocs+=emb
    url=d.get('_links',{}).get('next',{}).get('href')
print('n',len(assocs))
for a in assocs:
    # risk allele name includes rs
    rs=a.get('loci',[{}])[0].get('strongestRiskAlleles',[{}])[0].get('riskAlleleName','').split('-')[0]
    print(rs, a.get('pvalue'), a.get('betaNum'), a.get('betaDirection'), a.get('standardError'), a.get('riskFrequency'))
PY"
