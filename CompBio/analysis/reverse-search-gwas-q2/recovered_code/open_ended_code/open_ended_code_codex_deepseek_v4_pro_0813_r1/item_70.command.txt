/bin/bash -lc "python3 - <<'PY'
import requests, json, re
base='https://www.ebi.ac.uk/gwas/rest/api/studies/search/findByDiseaseTrait?diseaseTrait=height&size=100'
url=base; items=[]; pages=0
while url and pages<20:
    r=requests.get(url,headers={'Accept':'application/json'},timeout=60)
    d=r.json(); emb=d.get('_embedded',{}).get('studies',[]); items+=emb
    url=d.get('_links',{}).get('next',{}).get('href')
    pages+=1
print('total',len(items))
for s in items:
    anc=s.get('ancestries',[])
    init=sum(a.get('numberOfIndividuals',0) for a in anc if a.get('type')=='initial')
    pub=s.get('publicationInfo',{}).get('pubmedId')
    trait=s.get('diseaseTrait',{}).get('trait')
    title=s.get('publicationInfo',{}).get('title')
    if init and 30000<=init<=100000:
        print(s.get('accessionId'),pub,init,s.get('snpCount'),trait,title[:80])
PY"
