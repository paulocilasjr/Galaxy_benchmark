/bin/bash -lc "python3 - <<'PY'
import requests,re
html=requests.get('http://ftp.ebi.ac.uk/pub/databases/gwas/summary_statistics/',timeout=30).text
folders=re.findall(r'href=\"(GCST["'^"/]+/)'"',html)
# find those containing 90095033 by checking likely 90095001-90100000? Use folder list
for f in folders:
    # parse ranges
    m=re.match(r'GCST(\\d+)-GCST(\\d+)/',f)
    if m:
        lo,hi=map(int,m.groups())
        if lo<=90095033<=hi:
            print(f)
# print all around 9xxxx
for f in folders:
    if '900' in f or '901' in f:
        print(f)
PY"
