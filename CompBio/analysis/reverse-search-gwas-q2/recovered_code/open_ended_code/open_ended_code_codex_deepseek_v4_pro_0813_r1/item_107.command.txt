import json
j=json.load(open('gwasinfo.json')); ds=j['datasets_compressed']; cod=j['majority_fields_and_coding']

    arr=val[0]; meta=val[1] if len(val)>1 else {}
    nsnp=meta.get('nsnp'); sample=arr[10]
    if nsnp and 600000 <= nsnp <= 650000:
        res.append((id,arr[0],sample,nsnp,meta.get('pmid'),meta.get('note'),meta.get('doi'),cod['population'][arr[4]] if arr[4] is not None else None))
print(len(res))
for r in res: print(r)
