/bin/bash -lc "python3 - <<'PY'
import requests
for rs in ['rs16895971','rs2780226','rs3816804','rs4239437','rs3791675','rs76803230']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations?size=100'
    r=requests.get(url,headers={'Accept':'application/json'},timeout=30)
    assocs=r.json().get('_embedded',{}).get('associations',[]) if r.status_code==200 else []
    print('\\n',rs,len(assocs))
    for a in assocs[:30]:
        print(a.get('pvalue'),a.get('betaNum'),a.get('betaDirection'),a['_links']['study']['href'].split('/')[-2])
PY"
