/bin/bash -lc "python3 - <<'PY'
import requests
for rs in ['rs1150779','rs4308051','rs2871960']:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations'
    r=requests.get(url, headers={'Accept':'application/json'}, timeout=30)
    print('\\n',rs, r.status_code)
    if r.status_code==200:
        data=r.json()
        assocs=data.get('_embedded',{}).get('associations',[])
        print('n',len(assocs))
        for a in assocs[:20]:
            print(a.get('pValue'), a.get('orPerCopyNum'), a.get('riskFrequency'), a.get('traitName'), a.get('study',{}).get('publicationInfo',{}).get('pubmedId'), a.get('study',{}).get('accessionId'))
            # maybe beta
    else: print(r.text[:500])
PY"
