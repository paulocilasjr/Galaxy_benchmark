/bin/bash -lc "python3 - <<'PY'
import requests, json
rslist=['rs1150779','rs6909732','rs2483211','rs7742369','rs4308051','rs8094261','rs7235010','rs4800455','rs2871960','rs9846396']
for rs in rslist:
    url=f'https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{rs}/associations'
    r=requests.get(url,headers={'Accept':'application/json'},timeout=30)
    if r.status_code"'!=200:
        print(rs,r.status_code); continue
    assocs=r.json().get('"'_embedded',{}).get('associations',[])
    print('\\n###',rs, 'n',len(assocs))
    for a in assocs[:10]:
        p=a.get('pvalue'); beta=a.get('betaNum'); dir=a.get('betaDirection')
        # fetch study link
        study_url=a['_links']['study']['href']
        sr=requests.get(study_url,headers={'Accept':'application/json'},timeout=30)
        st=sr.json() if sr.ok else {}
        pub=st.get('publicationInfo',{}).get('pubmedId')
        title=st.get('publicationInfo',{}).get('title')
        trait=st.get('diseaseTrait',{}).get('trait')
        init=st.get('initialSampleSize'); repl=st.get('replicateSampleSize')
        print(p,beta,dir,'|',pub,'|',trait,'|',title,'|',init,'|',repl)
PY"
