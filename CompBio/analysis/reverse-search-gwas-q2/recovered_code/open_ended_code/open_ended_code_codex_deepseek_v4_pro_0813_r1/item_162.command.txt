/bin/bash -lc "python3 - <<'PY'
import gzip, csv
# load original partial (has header)
orig={}
with open('GCST90245993_buildGRCh37.tsv') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        chrom,pos,ea,oa,beta,se,eaf,pval,rs,n=p[:10]
        orig[(chrom,pos,rs)] = p
# our file all chr1 rows
ours=[]
with gzip.open('reverse.search.gwas.q2.gz','rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        if p[2]=='1':
            ours.append(p)
print('ours chr1',len(ours))
matches=0; total=0
for p in ours:
    chrom=p[2]; pos=p[3]; rs=p[1]
    key=(chrom,pos,rs)
    if key in orig:
        total+=1
        op=orig[key]
        # columns orig: chrom,pos,effect,other,beta,se,eaf,pval,rs,n
        # ours: SNPID,RSID,CHR,POS,EA,OA,EAF,BETA,SE,P
        if (op[2]==p[4] and op[3]==p[5] and abs(float(op[4])-float(p[7]))<1e-12 and abs(float(op[5])-float(p[8]))<1e-12 and abs(float(op[6])-float(p[6]))<1e-12 and abs(float(op[7])-float(p[9]))<1e-12):
            matches+=1
        else:
            if total<=5: print('mismatch',p,op)
    else:
        if total<=5: print('missing orig',p)
print('matched exact',matches,'of total in orig',total)
PY"
