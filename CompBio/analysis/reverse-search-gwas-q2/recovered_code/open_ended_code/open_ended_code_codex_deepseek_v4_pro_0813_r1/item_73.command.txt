/bin/bash -lc "python3 - <<'PY'
import requests, json
for rs in ['rs1150779','rs7742369','rs4308051','rs2871960','rs224333','rs1812175','rs3791679','rs798485']:
    u=f'https://grch37.rest.ensembl.org/variation/human/{rs}?pops=1'
    r=requests.get(u,headers={'Content-Type':'application/json'},timeout=30)
    if not r.ok:
        print(rs,r.status_code,r.text[:100]); continue
    d=r.json()
    pops=d.get('populations',[])
    freqs=[]
    for p in pops:
        if p.get('population') in ['1000GENOMES:phase_3:EUR','1000GENOMES:phase_3:EAS','1000GENOMES:phase_3:AMR','1000GENOMES:phase_3:AFR']:
            freqs.append((p['population'].split(':')[-1],p.get('allele'),p.get('frequency')))
    print(rs,d.get('mappings',[{}])[0].get('allele_string'),freqs)
PY"
