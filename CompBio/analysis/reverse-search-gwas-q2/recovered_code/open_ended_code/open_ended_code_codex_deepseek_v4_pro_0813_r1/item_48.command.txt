/bin/bash -lc "python3 - <<'PY'
import requests
for base in ['https://europepmc.org/articles/PMC9281111/bin/','https://www.ebi.ac.uk/europepmc/webservices/rest/PMC9281111/supplementaryFiles','https://europepmc.org/articles/PMC9281111?pdf=render']:
    url=base+'12916_2022_2450_MOESM2_ESM.docx' if 'bin' in base else base
    r=requests.get(url,timeout=60)
    print(url,r.status_code,len(r.content),r.headers.get('content-type'),r.url)
    if r.status_code==200 and r.headers.get('content-type','').startswith('application') or b'PK' in r.content[:4]:
        open('MOESM2_test','wb').write(r.content); print('saved'); break
    print(r.text[:200])
PY"
