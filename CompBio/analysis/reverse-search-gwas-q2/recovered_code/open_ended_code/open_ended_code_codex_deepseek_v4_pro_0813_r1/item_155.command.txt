/bin/bash -lc "python3 - <<'PY'
import requests
for rs in ['rs1150779','rs7742369','rs4308051','rs2871960','rs224333','rs1812175','rs3791679','rs798485']:
    d=requests.get(f'https://grch37.rest.ensembl.org/variation/human/{rs}?pops=1',headers={'Content-Type':'application/json'},timeout=30).json()
    pops={p['population'].split(':')[-1]:{p.get('allele'):p.get('frequency')} for p in d.get('populations',[]) if p['population'].startswith('1000GENOMES')}
    print(rs, {k:v for k,v in pops.items() if k in ['SAS','AMR','EUR','EAS','AFR']})
PY"
