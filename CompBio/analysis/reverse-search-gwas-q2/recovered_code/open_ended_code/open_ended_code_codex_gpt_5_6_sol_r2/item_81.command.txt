/bin/bash -lc "gzip -cd reverse.search.gwas.q2.gz | awk 'NR>1&&"'$10!=""{print $10,$2}'"' | sort -g -k1,1 | head -25 | awk '{print "'$2}'"' > top25_snps.txt
xargs -P 4 -I{} sh -c 'curl -L --silent \"https://www.ebi.ac.uk/gwas/rest/api/singleNucleotidePolymorphisms/{}/associations?projection=associationBySnp\" -o \"topassoc_{}.json\"' < top25_snps.txt
ls topassoc_*.json | wc -l"
