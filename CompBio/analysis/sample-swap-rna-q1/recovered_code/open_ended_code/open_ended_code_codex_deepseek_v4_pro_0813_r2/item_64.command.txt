/bin/bash -lc "python3 - <<'PY'
import gzip
# count lines and columns quickly
for p in ['ref/sc_counts.tsv.gz','ref/metacell.tsv.gz']:
    with gzip.open(p,'rt') as f:
        header=f.readline().rstrip('\\n').split('\\t')
        print(p, 'cols', len(header), 'first', header[:5], 'last', header[-5:])
        for _ in range(5):
            line=f.readline()
            if not line: break
            parts=line.rstrip('\\n').split('\\t')
            print('rowlen',len(parts), parts[0], parts[1:5])
PY"
