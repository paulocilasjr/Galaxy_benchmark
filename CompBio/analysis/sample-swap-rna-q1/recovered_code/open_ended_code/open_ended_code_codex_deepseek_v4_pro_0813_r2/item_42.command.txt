/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, json
p='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t',index_col=0)
cols=df.columns
markers={}
for col in cols:
    a=df[col].values
    others=df.drop(columns=col).values
    rate_a=a/df[col].sum(); rate_b=others/others.sum()
    lfc=np.log2((rate_a+1e-6)/(rate_b+1e-6))
    idx=np.argsort(-lfc)[:40]
    markers[col]=[(df.index[i], float(lfc[i]), float(a[i])) for i in idx]
    print(col, len(markers[col]))
with open('top_markers.json','w') as f: json.dump(markers,f,indent=2)
PY
head -30 top_markers.json"
