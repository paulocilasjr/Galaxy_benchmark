/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
p='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t',index_col=0)
# cpm-like; compute log2((mean in group+1e-5)/(mean other+1e-5))
cols=df.columns
# scale each column? sums approx equal; use raw
for col in cols:
    a=df[col].values
    b=(df.drop(columns=col).sum(axis=1)).values
    # total per group mean with same total? compare rates: a/sumA, b/sumB
    rate_a=a/df[col].sum(); rate_b=b/df.drop(columns=col).values.sum()
    lfc=np.log2((rate_a+1e-6)/(rate_b+1e-6))
    idx=np.argsort(-lfc)[:25]
    print('\\n###',col)
    for i in idx:
        print(df.index[i], 'lfc=%.2f'%lfc[i], 'val=%.1f'%a[i], 'other=%.3f'%(b[i]/ (len(cols)-1)))
PY"
