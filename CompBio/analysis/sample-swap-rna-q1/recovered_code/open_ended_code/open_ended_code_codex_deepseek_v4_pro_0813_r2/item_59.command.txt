/bin/bash -lc "python3 - <<'PY'
import json
markers=json.load(open('top_markers.json'))
mp=json.load(open('gbff_id_gene.json'))
for col, arr in markers.items():
    print('\\n##', col)
    for gid,lfc,val in arr[:25]:
        m=mp.get(gid,{})
        print(f'{gid:15s} lfc={lfc:5.2f} val={val:8.1f} gene={m.get(\"gene\")} locus={m.get(\"locus\")}')
PY"
