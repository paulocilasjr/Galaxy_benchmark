/bin/bash -lc "python3 - <<'PY'
import pandas as pd
ann=pd.read_csv('GSE255742_Amphioxus_21hpf_sc_annotations.tsv.gz',sep='\\t')
print(ann.shape)
print(ann.columns.tolist())
print('cell_type unique',ann.cell_type.nunique())
print(sorted(ann.cell_type.unique()))
print('broad unique',ann.broad_cell_type.nunique())
print(sorted(ann.broad_cell_type.unique()))
print('\\nmetacell per broad')
print(ann.groupby('broad_cell_type')['metacell'].nunique().sort_index().to_string())
print('\\ncell per broad')
print(ann.groupby('broad_cell_type')['cell'].count().sort_index().to_string())
print('\\nmetacell labels')
print(ann.drop_duplicates('metacell').sort_values('metacell')[['metacell','cell_type','broad_cell_type']].to_string(index=False))
PY"
