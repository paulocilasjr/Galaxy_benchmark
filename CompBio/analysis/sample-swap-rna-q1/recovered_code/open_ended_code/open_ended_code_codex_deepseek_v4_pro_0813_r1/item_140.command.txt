/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
g=pd.read_csv('gene_pseudobulk.tsv',sep='\\t',index_col=0)
cols=g.columns.tolist()
# log fold changes for all pairs, focus epidermis vs tailbud_neural and epidermal_neural vs tailbud_neural etc
for a,b in [('epidermis','tailbud_neural'),('epidermal_neural','tailbud_neural'),('neural','tailbud_neural'),('epidermal_neural','neural'),('putative_neuro_endo','epidermal_neural'),('somite','muscular_somites'),('tailbud','mesodermal_notochord')]:
    x=np.log1p(g[a]); y=np.log1p(g[b])
    diff=x-y
    up=diff.sort_values(ascending=False)
    down=diff.sort_values(ascending=True)
    print('\\n###',a,'vs',b)
    print('UP in',a)
    for gene,val in up.head(20).items(): print(f'{gene:15s} {g.loc[gene,a]:9.1f} {g.loc[gene,b]:9.1f} diff={val:.2f}')
    print('UP in',b)
    for gene,val in down.head(20).items(): print(f'{gene:15s} {g.loc[gene,a]:9.1f} {g.loc[gene,b]:9.1f} diff={val:.2f}')
PY"
