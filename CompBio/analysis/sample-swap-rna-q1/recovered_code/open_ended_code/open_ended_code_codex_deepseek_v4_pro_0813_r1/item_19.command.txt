/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse
for db in ['gene','protein','sra']:
    ids=['BLAG01000002','BLAG01000003','BLAG01000020','BLAG90000847']
    url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi?db={db}&retmode=json&id='+','.join(ids)
    try:
        with urllib.request.urlopen(url, timeout=20) as r:
            data=r.read()
        print(db, data.decode()[:500])
    except Exception as e: print(db,'ERR',e)
PY"
