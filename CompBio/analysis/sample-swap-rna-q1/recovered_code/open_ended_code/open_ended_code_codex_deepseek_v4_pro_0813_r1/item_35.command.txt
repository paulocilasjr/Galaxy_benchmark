/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
p='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t',index_col=0)
X=np.log1p(df)
# scale genes
Z=StandardScaler().fit_transform(X.T) # rows samples columns genes
pca=PCA(n_components=5).fit(Z)
print('explained',pca.explained_variance_ratio_)
coords=pca.transform(Z)
for i,c in enumerate(df.columns):
    print(c, np.round(coords[i],4))
print('\\nloadings top genes PC1')
loadings=pca.components_[0]
idx=np.argsort(-np.abs(loadings))[:20]
for j in idx:
    print(df.index[j], loadings[j])
PY"
