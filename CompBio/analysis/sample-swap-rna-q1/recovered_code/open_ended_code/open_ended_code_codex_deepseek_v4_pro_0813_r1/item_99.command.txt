/bin/bash -lc "python3 - <<'PY'
import pandas as pd, re
g=pd.read_csv('gene_pseudobulk.tsv',sep='\\t',index_col=0)
cols=g.columns.tolist()
# normalized per gene (max=1) for easier pattern
norm=g.div(g.max(axis=1),axis=0)
# list markers
markers=['KRT13','KRT75','KRT8','KRT18','KRT19','CDH1','TP63','SOX2','SOX3','SOX14','SOX19','SOXB1','ELAVL1','ELAVL2','ELAVL3','ELAVL4','FOXA1','FOXA2','FOXA3','HHEX','SOX17','GATA4','GATA5','GATA6','NKX2-1','NKX2.1','TPO','FOXE1','FOXE3','PAX8','BRA2','BRACHYURY','T','NOTO','FOXJ1','SHH','MYOD1','MYF5','MYOG','MEF2C','MYH1','MYH2','MYH3','MYH4','MYH6','MYH7','MYL1','TTN','ACTN2','TPM1','TNNI1','CSRP1','TBX6','TBX16','WNT1','WNT3','WNT5A','FGF8','FGF17','CDX1','CDX2','MSGN1','EVX1','EVX2','NEUROD1','NEUROG1','ASCL1','SYP','NEFM','TUBB2A','TUBB2B']
for m in markers:
    # exact or regex
    hits=[x for x in g.index if x.upper()==m.upper()]
    if not hits:
        pat=re.compile(re.escape(m), re.I)
        hits=[x for x in g.index if pat.search(x)]
    for x in hits[:5]:
        print(m, '->', x, [round(v,1) for v in g.loc[x].tolist()], 'norm', [round(v,2) for v in norm.loc[x].tolist()])
PY"
