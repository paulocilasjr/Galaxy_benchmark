/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, time
terms=['BLAG01000002[All Fields]','BLAG01000003[All Fields]','BLAG01000933[All Fields]','BLAG90000847[All Fields]']
for db in ['protein','nucleotide']:
  for term in terms:
    url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':db,'term':term,'retmode':'json','retmax':3})
    try:
      with urllib.request.urlopen(url, timeout=30) as r: data=r.read().decode()
      print(db,term,data[:300])
    except Exception as e: print(db,term,'ERR',e)
    time.sleep(.3)
PY"
