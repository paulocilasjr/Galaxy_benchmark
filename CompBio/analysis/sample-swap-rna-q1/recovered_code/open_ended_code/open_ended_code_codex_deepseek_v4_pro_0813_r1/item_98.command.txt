/bin/bash -lc "python3 - <<'PY'
import gzip, json, pandas as pd, numpy as np
# load mapping
mapping=json.load(open('blag_mrna_mapping.json'))
# load expression
ids=[]; data=[]
with gzip.open('sample.swap.rna.q1.tsv.gz','rt') as f:
    header=f.readline().rstrip('\\n').split('\\t')
    cols=header[1:]
    for line in f:
        parts=line.rstrip('\\n').split('\\t')
        ids.append(parts[0]); data.append(list(map(float,parts[1:])))
df=pd.DataFrame(data,index=ids,columns=cols)
print(df.shape)
# gene symbol mapping, missing use id
sym=[]
for i in ids:
    m=mapping.get(i)
    sym.append(m['gene'] if m and m['gene'] else i)
df['gene']=sym
gdf=df.groupby('gene').sum()
print('gene matrix',gdf.shape)
gdf.to_csv('gene_pseudobulk.tsv',sep='\\t')
print('top total genes')
print(gdf.sum(axis=1).sort_values(ascending=False).head(50))
PY"
