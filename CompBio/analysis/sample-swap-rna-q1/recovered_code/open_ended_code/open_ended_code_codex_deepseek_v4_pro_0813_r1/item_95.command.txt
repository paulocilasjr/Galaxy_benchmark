import zipfile
z=zipfile.ZipFile('[REDACTED LOCAL PATH]')
with z.open('ncbi_dataset/data/GCA_927797965.1/genomic.gff') as f:
    count=0
    for line in f:
        l=line.decode('utf8','replace').rstrip('\\n')
        if l.startswith('#') or l=='':
            continue
        parts=l.split('\\t')
        if len(parts)<9: continue
        if parts[2]=='gene':
            print(l[:1000])
            count+=1
            if count>=10: break
PY"
