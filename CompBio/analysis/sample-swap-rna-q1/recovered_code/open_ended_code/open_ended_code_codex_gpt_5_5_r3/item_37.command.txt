expr.columns=[int(float(c)) for c in expr.columns]
expr=expr.astype(float)
# Collapse fine metacell annotation to task labels
map_ct={
 'endoderm':'endoderm',
 'endostyle':'endostyle_csg',
 'epidermal neural 1':'epidermal_neural',
 'epidermal neural 2':'epidermal_neural',
 'anterior epidermis':'epidermis',
 'posterior epidermis':'epidermis',
 'mesodermal notochord':'mesodermal_notochord',
 'muscular somites':'muscular_somites',
 'neural':'neural',

 'tailbud neural':'tailbud_neural',

transforms.append(('q_FCmean_vs_FC', q_fc_mean, ref.loc[common]))
for name,qmat,rmat in transforms:
    # select variable genes maybe
    for subset in ['all','var5000','marker_ref_fc1.5','marker_query_top']:
        genes=common
        if subset=='var5000':

            genes=v.head(5000).index
