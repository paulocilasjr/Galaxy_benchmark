/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np
x=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
for typ,y in [('logpearson',np.log1p(x).corr()),('rawpearson',x.corr()),('spearman',x.corr(method='spearman'))]:
 print('\\n',typ)
 for c in y:
  z=y[c].drop(c).sort_values(ascending=False)
  print(f'{c:22s}', ', '.join(f'{i}:{v:.3f}' for i,v in z.head(4).items()))
PY"
