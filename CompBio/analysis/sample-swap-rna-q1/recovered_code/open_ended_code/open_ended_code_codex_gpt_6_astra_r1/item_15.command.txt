/bin/bash -lc "python -u - <<'PY'
import requests,concurrent.futures
urls=['https://www.ebi.ac.uk/europepmc/webservices/rest/PMC11136973/supplementaryFiles','https://ftp.ncbi.nlm.nih.gov/geo/series/GSE255nnn/GSE255742/suppl/','https://sebelab.crg.eu','http://sebelab.crg.eu/amphioxus-neurula-21hpf/','https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE255742&targ=self&form=text&view=full']
def run(u):
 try:
  r=requests.get(u,timeout=20);print(u,r.status_code,r.headers.get('Content-Type'),len(r.content));
  if 'zip' in r.headers.get('Content-Type',''):open('supplements.zip','wb').write(r.content)
  else:print(r.text[:12000])
 except Exception as e:print(u,str(e))
with concurrent.futures.ThreadPoolExecutor() as e:list(e.map(run,urls))
PY"
