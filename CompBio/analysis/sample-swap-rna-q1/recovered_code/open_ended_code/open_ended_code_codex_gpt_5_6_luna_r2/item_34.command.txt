/bin/bash -lc "python3 -u - <<'PY'
import pandas as pd,numpy as np
D=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
R=pd.read_csv('umi_pseudobulk_exact.tsv.gz',sep='\\t',index_col=0).reindex(D.index)
# expected group definitions based on released annotation categories
G={
'endoderm':['endoderm'],
'endostyle_csg':['endostyle_csg'],
'epidermal_neural':['epidermal_neural1','epidermal_neural2'],
'epidermis':['epidermis_1','epidermis_2'],
'mesodermal_notochord':['mesodermal_notochord'],
'muscular_somites':['muscular_somites'],
'neural':['neural'],
'neural_like':['neural_like1','neural_like2'],
'putative_neuro_endo':['putative_neuro_endo'],
'somite':['somite'],
'tailbud':['tailbud'],
'tailbud_neural':['tailbud_neural'],
}
E={}
for k,v in G.items():
 x=R[v].sum(axis=1)
 E[k]=x/x.sum()*1e6
E=pd.DataFrame(E)
print('MAE and max abs error to corresponding expected groups')
for c in D.columns:
 print(c, 'mae',np.mean(abs(D[c]-E[c])).round(4),'max',np.max(abs(D[c]-E[c])).round(4), 'corr',D[c].corr(E[c]).round(5))
print('\\nfull correlation')
C=pd.DataFrame([[D[a].corr(E[b]) for b in E] for a in D],index=D.columns,columns=E.columns)
print(C.round(5).to_string())
print('\\nBest mappings')
for a in D:
 print(a, C.loc[a].nlargest(4).round(5).to_dict())
PY"
