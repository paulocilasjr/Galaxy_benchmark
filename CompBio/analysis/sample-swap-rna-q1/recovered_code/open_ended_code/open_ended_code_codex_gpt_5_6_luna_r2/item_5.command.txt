/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
p='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(p,sep='\\t',index_col=0)
print(df.shape)
print(df.columns.tolist())
print('means')
print(df.mean().round(2).to_string())
print('nonzero')
print((df>0).sum().to_string())
# top genes by specificity: log2((x+1)/(mean others+1)) and raw x
for c in df.columns:
    x=df[c]
    rest=df.drop(columns=c).mean(axis=1)
    score=np.log2((x+1)/(rest+1))
    print('\\n##',c)
    print(pd.DataFrame({'gene':df.index,'x':x,'rest':rest,'score':score}).sort_values('score',ascending=False).head(25).to_string(index=False,formatters={'x':'{:.1f}'.format,'rest':'{:.1f}'.format,'score':'{:.2f}'.format}))
PY"
