import gzip, pandas as pd, numpy as np
f='sample.swap.rna.q1.tsv.gz'
df=pd.read_csv(f,sep='\\t',index_col=0)
print('shape',df.shape)
print('columns',list(df.columns))
print('dtypes',df.dtypes.unique())
print('nan',df.isna().sum().sum())
print('\\nsummary')
print(df.describe(percentiles=[.5,.9,.99]).T.to_string())
print('\\ncor raw')
print(df.corr().round(3).to_string())
# log correlation
lg=np.log1p(df)
print('\\ncor log')
print(lg.corr().round(3).to_string())
for c in df:
    x=df[c]
    print('\\n###',c,'sum',x.sum(),'nonzero',(x>0).sum(),'top')
    print(x.sort_values(ascending=False).head(15).to_string())
PY"
