/bin/bash -lc "python3 - <<'PY'
import gzip, numpy as np, pandas as pd, time
D=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
genes=set(D.index); nG=len(D); target=sorted(D.index); pos={g:i for i,g in enumerate(target)}
a=pd.read_csv('GSE255742_Amphioxus_21hpf_sc_annotations.tsv.gz',sep='\\t')
labmap=dict(zip(a.cell,a.cell_type))
with gzip.open('GSE255742_Amphioxus_21hpf_sc_UMI_counts.tsv.gz','rb') as f:
    header=f.readline().decode().rstrip('\\n').split('\\t')
labels=np.array([labmap[x] for x in header]); types=sorted(set(labels)); codes=np.array([types.index(x) for x in labels]); k=len(types)
# sums for target genes and all genes per cell type
S=np.zeros((nG,k),float); totals=np.zeros(k,float); found=0;t0=time.time()
with gzip.open('GSE255742_Amphioxus_21hpf_sc_UMI_counts.tsv.gz','rb') as f:
    f.readline()
    for n,line in enumerate(f,1):
        p=line.split(b'\\t',1); g=p[0].decode()
        v=np.fromstring(p[1].decode(),sep='\\t')
        totals += np.bincount(codes,weights=v,minlength=k)
        j=pos.get(g)
        if j is not None:
            S[j] += np.bincount(codes,weights=v,minlength=k); found+=1
        if n%5000==0: print(n,round(time.time()-t0,1),flush=True)
print('rows',n,'found',found,'types',types,'sec',round(time.time()-t0,1),flush=True)
A=pd.DataFrame((S/totals[None,:]*1e6),index=target,columns=types).loc[D.index]
print('D sums',D.sum().round(1).to_dict(),flush=True)
print('A sums',A.sum().round(1).to_dict(),flush=True)
for d in D:
 out=[]
 for t in A:
  out.append((t,np.corrcoef(D[d],A[t])[0,1],np.corrcoef(np.log1p(D[d]),np.log1p(A[t]))[0,1],np.abs(D[d]-A[t]).mean()))
 print(d,sorted(out,key=lambda z:z[1],reverse=True)[:5],flush=True)
PY"
