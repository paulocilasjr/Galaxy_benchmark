/bin/bash -lc "python3 - <<'PY'
import urllib.request
for u in [
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE255nnn/GSE255742/matrix/',
'https://ftp.ncbi.nlm.nih.gov/geo/series/GSE255nnn/GSE255742/suppl/',
'https://www.ncbi.nlm.nih.gov/geo/query/acc.cgi?acc=GSE255742&targ=self&form=xml&view=quick'
]:
 print('\\nURL',u)
 try:
  b=urllib.request.urlopen(u,timeout=30).read()
  print(b[:5000].decode(errors='replace'))
 except Exception as e: print(type(e).__name__,e)
PY"
