/bin/bash -lc "python3 - <<'PY'
import gzip, pandas as pd, numpy as np
D=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
# ranks by specificity: value / mean other columns; log fold and perhaps max
for c in D:
    x=D[c]
    rest=D.drop(columns=c).mean(axis=1)
    score=np.log2((x+0.1)/(rest+0.1))
    print('\\n###',c)
    for g in score.nlargest(30).index:
        print(g, f'{D.loc[g,c]:.2f}', f'{rest[g]:.2f}', f'LFC={score[g]:.2f}')
PY"
