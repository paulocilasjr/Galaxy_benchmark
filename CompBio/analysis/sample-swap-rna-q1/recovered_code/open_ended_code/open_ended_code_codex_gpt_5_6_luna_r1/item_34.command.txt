/bin/bash -lc "python - <<'PY'
import pandas as pd, numpy as np, time
ann=pd.read_csv('GSE255742_annotations.tsv.gz',sep='\\t')
merge={'epidermis_1':'epidermis','epidermis_2':'epidermis','neural_like1':'neural_like','neural_like2':'neural_like','epidermal_neural1':'epidermal_neural','epidermal_neural2':'epidermal_neural'}
ann['grp']=ann.cell_type.replace(merge)
wanted=['endoderm','endostyle_csg','epidermal_neural','epidermis','mesodermal_notochord','muscular_somites','neural','neural_like','putative_neuro_endo','somite','tailbud','tailbud_neural']
ctmap=dict(zip(ann.cell,ann.grp))
cells=pd.read_csv('GSE255742_counts.tsv.gz',sep='\\t',nrows=0,index_col=0).columns.tolist()
groups=np.array([ctmap.get(c,'') for c in cells])
print('cells',len(cells),'desired',[(g,int((groups==g).sum())) for g in wanted])
out=[]; genes=[]; t=time.time(); nrows=0
for k,ch in enumerate(pd.read_csv('GSE255742_counts.tsv.gz',sep='\\t',index_col=0,chunksize=500)):
    a=ch.to_numpy(dtype=np.float64,copy=False)
    out.append(np.column_stack([a[:,groups==g].sum(axis=1) for g in wanted]))
    genes.extend(ch.index.tolist()); nrows+=len(ch)
    if (k+1)%10==0: print('chunk',k+1,'rows',nrows,'sec',round(time.time()-t,1),flush=True)
agg=np.vstack(out); genes=np.array(genes)
print('agg shape',agg.shape,'time',round(time.time()-t,1))
sums=agg.sum(axis=0); cpm=agg/sums*1e6
x=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
lookup={g:i for i,g in enumerate(genes)}
common=[g for g in x.index if g in lookup]
ix=np.array([lookup[g] for g in common]); a=x.loc[common,wanted].to_numpy(); b=cpm[ix]
print('common',len(common),'source sums',sums.astype(int).tolist())
from scipy.optimize import linear_sum_assignment
for transname,aa,bb in [('rawCPM',a,b),('log',np.log1p(a),np.log1p(b)),('rank',a.argsort(0).argsort(0),b.argsort(0).argsort(0))]:
 cor=np.corrcoef(aa.T,bb.T)[:12,12:]
 print('\\n',transname,'diagonal',np.diag(cor).round(3).tolist())
 print(pd.DataFrame(cor,index=wanted,columns=wanted).round(3).to_string())
 best=cor.argmax(axis=1)
 print('best',[(wanted[i],wanted[j],round(cor[i,j],3)) for i,j in enumerate(best)])
 ri,cj=linear_sum_assignment(-cor)
 print('assign',[(wanted[i],wanted[j],round(cor[i,j],3)) for i,j in zip(ri,cj)],'sum',round(cor[ri,cj].sum(),3))
PY"
