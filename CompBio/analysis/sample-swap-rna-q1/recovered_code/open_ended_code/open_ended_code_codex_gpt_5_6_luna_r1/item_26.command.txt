/bin/bash -lc "python - <<'PY'
import pandas as pd,numpy as np
m=pd.read_csv('GSE255742_metacell.tsv.gz',sep='\\t',index_col=0)
print('m',m.shape,'sum range',m.sum().min(),m.sum().max(),'median',m.median().median(),'max',m.max().max(),'nonzero fraction',(m>1).mean().mean())
print(m.iloc[:10,:5].to_string())
# source cell annotations composition for each metacell
ann=pd.read_csv('GSE255742_annotations.tsv.gz',sep='\\t')
ct=ann.pivot_table(index='metacell',columns='cell_type',values='cell',aggfunc='count',fill_value=0)
print('composition pure fraction', (ct.max(axis=1)/ct.sum(axis=1)).describe().to_string())
print('metacells',ct.shape)
print(ct.assign(dom=ct.idxmax(1), n=ct.sum(1)).groupby('dom').agg(m=('n','count'),cells=('n','sum')).to_string())
print(ct.head())
PY"
