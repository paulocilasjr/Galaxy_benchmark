/bin/bash -lc "python - <<'PY'
import numpy as np, pandas as pd
from scipy.stats import pearsonr, spearmanr
z=np.load('reference_pseudobulk.npz')
r=pd.DataFrame(z['counts'],index=z['genes'],columns=z['labels'])
r=1e6*r/r.sum()
x=pd.read_csv('sample.swap.rna.q1.tsv.gz',sep='\\t',index_col=0)
r=r.loc[x.index]
# Pearson log1p and maximum absolute difference under best mapping.
corr=pd.DataFrame(index=x.columns,columns=r.columns,dtype=float)
for a in x:
 for b in r:
  corr.loc[a,b]=pearsonr(np.log1p(x[a]),np.log1p(r[b]))[0]
print(corr.round(4).to_string())
print('\\nBest match and absolute errors:')
for a in x:
 b=corr.loc[a].idxmax()
 d=np.abs(x[a]-r[b])
 print(a,'=>',b,'corr',corr.loc[a,b],'maxerr',d.max(),'meanerr',d.mean())
PY"
