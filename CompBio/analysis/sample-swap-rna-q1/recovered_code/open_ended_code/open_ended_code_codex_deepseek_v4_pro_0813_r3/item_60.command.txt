import gzip,csv,json,re,numpy as np
id2name=json.load(open('id2name.json'))
with gzip.open('sample.swap.rna.q1.tsv.gz','rt') as f:

for p in patterns:
    rr=re.compile('"'^'"'+p,re.I)
    m=[]
    for i,g in enumerate(ids):
        n=id2name.get(g,g)
        if rr.search(n): m.append((n,g,A[i]))
    if m:
        print('\\n##',p)
        for n,g,v in m[:15]: print(f'{n:12s} {g:15s} ' + ' '.join(f'{x:7.1f}' for x in v))
