/bin/bash -lc "python3 - <<'PY'
import urllib.request, json, zipfile, io
acc='GCA_927797965.1'
# report
for endpoint in ['dataset_report','annotation']:
    url=f'https://api.ncbi.nlm.nih.gov/datasets/v2/genome/accession/{acc}/{endpoint}'
    try:
        r=urllib.request.urlopen(url,timeout=60)
        data=r.read()
        print(endpoint, r.status, r.headers.get('content-type'), len(data), data[:1000])
        if endpoint=='annotation':
            open('braLan3_ann.zip','wb').write(data)
            try:
                z=zipfile.ZipFile(io.BytesIO(data))
                for n in z.namelist(): print(n, z.getinfo(n).file_size)
            except Exception as e: print('zip err',e)
    except Exception as e: print(endpoint,e)
PY"
