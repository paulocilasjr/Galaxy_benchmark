/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
base='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi'
for db in ['gds','sra','bioproject']:
  for term in ['amphioxus single cell RNA-seq', '\"epidermal_neural\"', '\"tailbud_neural\"', '\"endostyle_csg\"']:
    url=base+'?'+urllib.parse.urlencode({'db':db,'term':term,'retmode':'json','retmax':20})
    try:
      data=json.loads(urllib.request.urlopen(url,timeout=30).read())
      print(db,term,'count',data['esearchresult']['count'],'ids',data['esearchresult'].get('idlist',[]))
    except Exception as e: print(e)
PY"
