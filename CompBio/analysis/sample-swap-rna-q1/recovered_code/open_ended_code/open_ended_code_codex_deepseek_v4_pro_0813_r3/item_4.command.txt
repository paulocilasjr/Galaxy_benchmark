/bin/bash -lc "python3 - <<'PY'
import gzip, csv, numpy as np
f='sample.swap.rna.q1.tsv.gz'
with gzip.open(f,'rt') as fh:
    r=csv.reader(fh,delimiter='\\t')
    header=next(r)
    rows=[]
    genes=[]
    for row in r:
        genes.append(row[0]); rows.append([float(x) for x in row[1:]])
A=np.array(rows)
print(header)
print(A.shape, len(genes))
print('col sums', A.sum(axis=0))
print('col nonzeros', (A>0).sum(axis=0))
PY"
