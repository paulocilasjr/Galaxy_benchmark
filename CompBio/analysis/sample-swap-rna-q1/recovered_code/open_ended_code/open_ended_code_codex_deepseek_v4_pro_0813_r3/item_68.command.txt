import urllib.parse, urllib.request, json, re
queries=['\"endoderm\" \"endostyle_csg\" \"epidermal_neural\" amphioxus','\"tailbud_neural\" \"putative_neuro_endo\"','\"endostyle_csg\" scRNA']
for q in queries:

        data=urllib.request.urlopen(req,timeout=30).read().decode()
        print('\\nQUERY',q)

            print(re.sub('<.*?>','',m.group(2)), m.group(1)[:120])
        for m in re.finditer(r'<a class=\"result__snippet\"["'^>]*>(.*?)</a>'"',data,re.S):
            print(' SNIP', re.sub('<.*?>','',m.group(1))[:300])
    except Exception as e: print(e)
PY"
