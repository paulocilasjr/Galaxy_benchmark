/bin/bash -lc "python3 - <<'PY'
from remotezip import RemoteZip
from pathlib import Path
import shutil
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
out=Path('saluki_paper/datasets/deeplearning/train_gru')
out.mkdir(parents=True, exist_ok=True)
with RemoteZip(url) as z:
    targets=[i.filename for i in z.infolist() if i.filename.startswith('datasets/deeplearning/train_gru/') and (i.filename.endswith('model0_best.h5') or i.filename.endswith('params.json'))]
    print('extracting', len(targets))
    for idx,name in enumerate(targets,1):
        rel=Path(name).relative_to('datasets/deeplearning/train_gru')
        dest=out/rel
        dest.parent.mkdir(parents=True, exist_ok=True)
        with z.open(name) as src, open(dest,'wb') as dst:
            shutil.copyfileobj(src,dst)
        if idx % 10 == 0: print(idx)
print('done')
PY"
