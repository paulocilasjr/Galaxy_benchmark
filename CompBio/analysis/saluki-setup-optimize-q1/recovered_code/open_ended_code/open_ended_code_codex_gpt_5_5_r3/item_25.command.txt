/bin/bash -lc "python3 - <<'PY'
from remotezip import RemoteZip
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
with RemoteZip(url) as z:
    infos=[i for i in z.infolist() if i.filename.startswith('datasets/deeplearning/train_gru/') and i.filename.endswith(('model0_best.h5','params.json'))]
    total=sum(i.file_size for i in infos if i.filename.endswith('model0_best.h5'))
    for i in infos[:20]: print(i.filename, i.file_size, i.compress_size)
    print('h5 total GB', total/1e9)
PY"
