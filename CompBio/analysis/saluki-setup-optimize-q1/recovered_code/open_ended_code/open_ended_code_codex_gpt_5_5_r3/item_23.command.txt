/bin/bash -lc "python3 - <<'PY'
from remotezip import RemoteZip
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
with RemoteZip(url) as z:
    names=z.namelist()
    matches=[n for n in names if 'deeplearning/train_gru' in n and (n.endswith('params.json') or n.endswith('model0_best.h5'))]
    print('total', len(names), 'matches', len(matches))
    for n in matches[:200]: print(n)
PY"
