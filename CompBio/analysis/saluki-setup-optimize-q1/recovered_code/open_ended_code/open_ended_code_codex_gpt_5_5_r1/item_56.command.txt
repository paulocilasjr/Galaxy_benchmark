/bin/bash -lc "python3 - <<'PY'
import h5py
p='datasets/deeplearning/train_gru/f0_c0/train/model0_best.h5'
with h5py.File(p,'r') as h:
    for name in h['model_weights']:
        g=h['model_weights'][name]
        if hasattr(g,'visititems'):
            def show(n,o,prefix=name):
                if hasattr(o,'shape'):
                    print(prefix+'/'+n, o.shape)
            g.visititems(show)
PY"
