/bin/bash -lc "PYTHONPATH=./pydeps python3 - <<'PY'
from remotezip import RemoteZip
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
with RemoteZip(url) as z:
    infos={i.filename:i for i in z.infolist()}
    model_names=[n for n in infos if n.startswith('datasets/deeplearning/train_gru/') and n.endswith('/train/model0_best.h5')]
    print('models', len(model_names))
    print('total_compressed_MB', round(sum(infos[n].compress_size for n in model_names)/1e6,2))
    print('total_uncompressed_MB', round(sum(infos[n].file_size for n in model_names)/1e6,2))
    for n in sorted(model_names)[:5]: print(n, infos[n].file_size, infos[n].compress_size)
    p='datasets/deeplearning/train_gru/params.json'
    print('base params', infos[p].file_size)
PY"
