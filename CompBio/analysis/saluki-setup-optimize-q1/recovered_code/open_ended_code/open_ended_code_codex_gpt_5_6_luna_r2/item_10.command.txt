/bin/bash -lc "curl -L --fail --silent --show-error https://zenodo.org/api/records/6326409 -o zenodo_record.json

import json
p=json.load(open('zenodo_record.json'))
print(p.get('title'))
for f in p.get('files',[]):
    print(f.get('key'), f.get('size'), f.get('links',{}).get('self'))
PY"
