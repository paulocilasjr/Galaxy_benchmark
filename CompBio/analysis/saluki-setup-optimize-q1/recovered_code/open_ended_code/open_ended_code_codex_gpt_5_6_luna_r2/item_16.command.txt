/bin/bash -lc "python - <<'PY'
import struct
b=open('zip_central.bin','rb').read(); i=0; n=0; hits=[]
while i<len(b):
    if b[i:i+4]"'!=b'"'PK\\x01\\x02':
        print('bad signature at',i,b[i:i+16]); break
    vals=struct.unpack_from('<4s6H3I5H2I',b,i)
    fnl,exl,coml=vals[10:13]
    name=b[i+46:i+46+fnl].decode('utf-8','replace')
    if any(s in name.lower() for s in ['model','saluki','gru','params','train_gru']): hits.append((name,vals[8],vals[9],vals[13],vals[14],vals[15],i,exl))
    i+=46+fnl+exl+coml; n+=1
print('entries',n,'end',i,'len',len(b),'hits',len(hits))
for x in hits[:1000]: print(x[:6])
PY"
