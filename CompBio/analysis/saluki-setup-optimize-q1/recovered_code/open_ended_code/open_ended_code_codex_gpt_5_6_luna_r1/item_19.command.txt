b=open('zip_central.bin','rb').read(); i=0; rows=[]

 # sig ver made need flag method time date crc csize usize name extra comment disk int ext offset

 _,vm,vn,flag,method,tm,dt,crc,cs,us,nl,xl,cl,disk,internal,external,off=vals

 rows.append((name,cs,us,off,method,flag))
 i+=46+nl+xl+cl
print('parsed',len(rows),'end',i,len(b))
for row in rows:
 name,cs,us,off,m,f=row
 if any(x in name.lower() for x in ['train_gru','model','saluki','params','fold']):
