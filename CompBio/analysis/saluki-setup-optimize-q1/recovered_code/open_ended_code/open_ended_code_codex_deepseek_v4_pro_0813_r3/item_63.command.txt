/bin/bash -lc "mkdir -p models_h5
python3 - <<'PY'
import zip_utils, os, re, concurrent.futures
recs=zip_utils.RECS
names=[]
for f in recs:
    m=re.search(r'"'^datasets/deeplearning/train_gru/(f'"\\d+_c\\d+)/train/model0_best\\.h5"'$'"', f)
    if m:
        names.append((f, m.group(1)))
print('count',len(names))
def work(item):
    f,label=item
    dest=f'models_h5/{label}_model0_best.h5'
    if os.path.exists(dest) and os.path.getsize(dest)>1000000:
        return label,'exists',os.path.getsize(dest)
    n=zip_utils.save(f,dest)
    return label,'done',n
with concurrent.futures.ThreadPoolExecutor(max_workers=12) as ex:
    for res in ex.map(work,names):
        if res[1]"'!='"'exists': print(res)
PY
ls models_h5 | head; ls models_h5 | wc -l; du -sh models_h5"
