/bin/bash -lc "cat > run_evo_correct2.py <<'PY'
import os,glob,time
import numpy as np, tensorflow as tf
tf.get_logger().setLevel('ERROR')
from basenji import rnann
SEQ0='GCCGCCACCATGGTGAGCAAGGGCGAGTAGTGTACATAATAAGGACT'; L=12288; CDS_START=9; CDS_END=27  # stop excluded
P={'model':{'activation':'relu','rnn_type':'gru','seq_length':L,'augment_shift':3,'num_targets':1,'heads':2,'filters':64,'kernel_size':5,'dropout':0.3,'l2_scale':0.001,'ln_epsilon':0.007,'num_layers':6,'bn_momentum':0.90}}
def encode_batch(seqs):
 X=np.zeros((len(seqs),L,6),np.float32)
 for si,seq in enumerate(seqs):
  for i,nt in enumerate(seq): X[si,i,'ACGT'.index(nt)]=1
  for i in range(CDS_START,CDS_END,3): X[si,i,4]=1
  X[si,len(seq)-1,5]=1
 return X
def run(files,label):
 model=rnann.RnaNN(P['model']); seq=SEQ0; utr=list(range(30,len(seq)))
 print('===',label,'models',len(files),flush=True)
 for rnd in range(1,4):
  cands=[]
  for pos in utr:
   for nt in 'ACGT':
    if nt"'!=seq[pos]:
     s=list(seq); s[pos]=nt; cands.append('"''.join(s))
  seqs=[seq]+cands; X=encode_batch(seqs); tot=np.zeros((len(seqs),1),np.float64)
  for mf in files:
   model.restore(mf,0); preds=model.predict(X,verbose=0,batch_size=len(X)); tot+=preds.astype(np.float64)
  avg=tot/len(files); bi=int(np.argmax(avg[1:]))+1; best=seqs[bi]
  print('round',rnd,'ref',float(avg[0,0]),'best',float(avg[bi,0]),'utr',best[30:],'mut',[(i,seq[i],best[i]) for i in range(len(seq)) if seq[i]"'!=best[i]],flush=True)
  seq=best
 print('"'FINAL',label,seq,flush=True)
 return seq
allfiles=sorted(glob.glob('models_h5/*_model0_best.h5'))
seq=run(allfiles,'all50')
open('final_answer.txt','w').write(seq+'\\n')
files_c0=[f for f in allfiles if os.path.basename(f).split('_model')[0].endswith('_c0')]
run(files_c0,'c0only')
PY
TF_CPP_MIN_LOG_LEVEL=3 PYTHONPATH=basenji_repo ./tfenv_arm/bin/python run_evo_correct2.py"
