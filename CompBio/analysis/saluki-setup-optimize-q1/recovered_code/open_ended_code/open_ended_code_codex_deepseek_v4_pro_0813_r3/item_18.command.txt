/bin/bash -lc "python3 - <<'PY'
import struct
b=open('tail.bin','rb').read()
# find EOCD sig 0x06054b50
pos=b.rfind(b'PK\\x05\\x06')
print('eocd pos',pos)
e=b[pos:pos+22]
print(e.hex())
fields=struct.unpack('<4s4H2LH', e)
print(fields)
sig,disk,cddisk,total_this,total,cd_size,cd_off,comment_len=fields
print('cd_size',cd_size,'cd_off',cd_off,'total entries',total)
PY"
