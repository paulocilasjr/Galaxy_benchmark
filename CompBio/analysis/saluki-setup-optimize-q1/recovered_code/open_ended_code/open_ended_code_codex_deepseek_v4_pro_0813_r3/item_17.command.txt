/bin/bash -lc "python3 - <<'PY'
import requests
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
headers={'Range':'bytes=-100000'}
r=requests.get(url,headers=headers,allow_redirects=True,stream=True,timeout=60)
print(r.status_code)
print(r.headers.get('Content-Range'), r.headers.get('Content-Length'))
data=r.content
print(len(data), data[:20], data[-20:])
open('tail.bin','wb').write(data)
PY"
