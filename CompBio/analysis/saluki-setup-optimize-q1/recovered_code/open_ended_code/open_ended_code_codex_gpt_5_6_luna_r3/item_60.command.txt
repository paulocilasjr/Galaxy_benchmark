/bin/bash -lc "PYTHONPATH=. python - <<'PY'
import gzip
import pandas as pd, numpy as np
from saluki_torch import SalukiTorch, one_hot_batch
construct=pd.read_table('saluki_paper/Fig6_S7/cxcl2_Erle_testSet/BTV_construct.txt',index_col=0,header=None).values
with gzip.open('validation_data/cxcl2_Erle_testSet/fastUTR_predictions/f0_c0.txt.gz','rt') as f:
 for line in f:
  if line.startswith('TGGCACCAGAG'):
   seq,expected=line.rstrip().split('\\t'); expected=float(expected); break
myseq=(construct[0]+construct[1]+construct[2]+[seq]+construct[3])[0]
model=SalukiTorch('saluki_models/train_gru/f0_c0/train/model0_best.h5')
batch=one_hot_batch([myseq],coding_start=len(construct[0][0]),coding_end=len(construct[0][0])+len(construct[1][0]))
pred=float(model(batch).cpu().numpy()[0,0])
print('mylen',len(myseq),'expected',repr(expected),'pred',repr(pred),'diff',pred-expected)
PY"
