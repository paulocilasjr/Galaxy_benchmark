/bin/bash -lc "python - <<'PY'
from remotezip import RemoteZip
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
with RemoteZip(url) as z:
    infos=z.infolist()
    for zi in infos:
        name=zi.filename
        if name.startswith('datasets/deeplearning/train_gru/') and (name.endswith('model0_best.h5') or name.endswith('model1_best.h5') or name.endswith('params.json')):
            print(f'{name}\\t{zi.file_size}\\t{zi.header_offset}')
PY"
