/bin/bash -lc "mkdir -p models/train_gru && python - <<'PY'
from remotezip import RemoteZip
import os, time
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
outbase='models/train_gru'
with RemoteZip(url) as z:
    infos=z.infolist()
    targets=[]
    for zi in infos:
        name=zi.filename
        if name.startswith('datasets/deeplearning/train_gru/') and name.endswith('/train/model0_best.h5'):
            parts=name.split('/')
            fold=parts[-3]  # fX_cY? name: datasets/deeplearning/train_gru/f0_c0/train/model0_best.h5 -> parts[-3]=f0_c0
            out=f'{outbase}/{fold}_model0_best.h5'
            targets.append((name,out))
    print('targets',len(targets))
    t=time.time()
    for i,(name,out) in enumerate(targets):
        data=z.read(name)
        open(out,'wb').write(data)
        print(i, name.split('/')[-3], len(data), f'{time.time()-t:.1f}s', flush=True)
    print('done',time.time()-t)
PY
ls -lh models/train_gru | head && ls models/train_gru | wc -l"
