/bin/bash -lc "python3 - <<'PY'
import glob,json,numpy as np
from saluki_numpy_optimize import SalukiNumpy, candidate_sequences, average_predict
seq='GCCGCCACCATGGTGAGCAAGGGCGAGTAGTGTACATAATAAGGACT'
utr_start=seq.index('TAG')+3
params=json.load(open('saluki_models/datasets/deeplearning/train_gru/params.json'))
files=sorted(glob.glob('saluki_models/datasets/deeplearning/train_gru/f*_c0/train/model0_best.h5'))
models=[SalukiNumpy(p, params['model']['ln_epsilon']) for p in files]
cur=seq
for r in range(3):
    cands=list(candidate_sequences(cur, utr_start))
    preds=average_predict(models, [c[2] for c in cands], utr_start)
    bi=int(np.argmax(preds)); pos,nt,cur=cands[bi]
    print('round',r+1,pos,nt,preds[bi],cur)
print(cur)
PY"
