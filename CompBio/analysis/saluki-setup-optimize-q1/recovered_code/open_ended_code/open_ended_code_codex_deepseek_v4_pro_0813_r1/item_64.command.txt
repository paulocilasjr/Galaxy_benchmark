/bin/bash -lc "python - <<'PY'
import json, urllib.request, struct, zlib, os, time
entries={e['name']:e for e in json.load(open('central_entries.json'))}
name='datasets/deeplearning/train_gru/f0_c0/ism/scores.h5'
e=entries[name]; dest='models/train_gru/f0_c0/ism_scores.h5'
start=e['local_off']; end=start+e['comp_size']+65536-1
print('fetching',e)
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
req=urllib.request.Request(url, headers={'Range':f'bytes={start}-{end}'})
with urllib.request.urlopen(req, timeout=120) as r:
    data=r.read()
rec=struct.unpack('<IHHHHHIIIHH', data[:30])
name_len,extra_len=rec[9],rec[10]; method=rec[3]
comp=data[30+name_len+extra_len:30+name_len+extra_len+e['comp_size']]
out=zlib.decompress(comp,-15) if method==8 else comp
open(dest,'wb').write(out)
print(len(out), e['uncomp_size'])
PY
ls -lh models/train_gru/f0_c0/ism_scores.h5"
