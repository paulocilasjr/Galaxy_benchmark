/bin/bash -lc "python - <<'PY'
import json, urllib.request, struct, zlib, os, concurrent.futures
entries={e['name']:e for e in json.load(open('central_entries.json'))}
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
items=[]
for fi in range(10):
    for ci in range(1,5):
        name=f'datasets/deeplearning/train_gru/f{fi}_c{ci}/train/model0_best.h5'
        dest=f'models/train_gru/f{fi}_c{ci}/train/model0_best.h5'
        if not os.path.exists(dest):
            items.append((name,dest))

def fetch(item):
    name,dest=item; e=entries[name]
    os.makedirs(os.path.dirname(dest),exist_ok=True)
    start=e['local_off']; end=start+e['comp_size']+65536-1
    req=urllib.request.Request(url,headers={'Range':f'bytes={start}-{end}'})
    with urllib.request.urlopen(req,timeout=120) as r: data=r.read()
    rec=struct.unpack('<IHHHHHIIIHH',data[:30])
    if rec[0]"'!=0x04034b50: raise Exception('"'bad local header '+name)
    nl,el,method=rec[9],rec[10],rec[3]
    comp=data[30+nl+el:30+nl+el+e['comp_size']]
    out=zlib.decompress(comp,-15) if method==8 else comp
    assert len(out)==e['uncomp_size'], (name,len(out),e['uncomp_size'])
    open(dest,'wb').write(out)
    return dest
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
    for res in ex.map(fetch,items):
        print('done',res)
print('downloaded',len(items))
PY"
