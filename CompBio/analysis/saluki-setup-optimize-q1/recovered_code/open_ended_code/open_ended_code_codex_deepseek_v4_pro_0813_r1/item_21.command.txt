/bin/bash -lc "python - <<'PY'
import struct
fn='tail.zip'
data=open(fn,'rb').read()
# find EOCD signature 0x06054b50 scanning from end
idx=data.rfind(b'PK\\x05\\x06')
print('eocd idx', idx, 'len', len(data))
if idx>=0:
    e=data[idx:idx+22]
    fields=struct.unpack('<4s4H2LH', e)
    print(fields)
    cd_size, cd_off, n=fields[5],fields[6],fields[4]
    print(cd_size, cd_off, n)
PY"
