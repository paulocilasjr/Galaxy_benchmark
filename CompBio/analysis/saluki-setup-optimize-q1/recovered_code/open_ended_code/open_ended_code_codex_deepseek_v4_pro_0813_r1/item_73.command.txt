/bin/bash -lc "python - <<'PY'
import numpy as np, torch, h5py
torch.set_num_threads(1)
import saluki_torch
with h5py.File('models/train_gru/f0_c0/ism_scores.h5','r') as f:
 seqs=f['seqs'][:1]; coding=f['coding'][:1]; splice=f['splice'][:1]
X=np.concatenate([seqs.astype(np.float32),coding[...,None].astype(np.float32),splice[...,None].astype(np.float32)],axis=2)
print('X ready',X.shape,X.dtype)
X=torch.from_numpy(X).permute(0,2,1).contiguous()
print('X tensor',X.shape, X.is_contiguous())
model=saluki_torch.Saluki(); saluki_torch.load_weights(model,'models/train_gru/f0_c0/train/model0_best.h5'); model.eval()
print('model ready')
with torch.no_grad():
 x=model.conv_initial(X); print('conv init',x.shape)
 for i in range(6):
  x=model.lns[i](x); x=torch.relu(x); x=model.convs[i](x); print('conv',i,x.shape); x=model.pools[i](x); print('pool',i,x.shape)
 x=model.lns[6](x); x=torch.relu(x); x=x.transpose(1,2); print('before flip',x.shape); x=x.flip(dims=[1]); print('flipped',x.shape)
 out,h=model.gru(x); print('gru out',out.shape,h.shape)
 h=h.squeeze(0); h=model.bn1(h.unsqueeze(2)).squeeze(2); h=torch.relu(h); h=model.dense1(h); h=model.bn2(h.unsqueeze(2)).squeeze(2); h=torch.relu(h); y=model.dense2(h)
 print('y',y)
PY"
