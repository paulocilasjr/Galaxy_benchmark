/bin/bash -lc "python - <<'PY'
import numpy as np, h5py, torch
torch.set_num_threads(1)
from saluki_torch import Saluki, load_weights
with h5py.File('models/train_gru/f0_c0/ism_scores.h5','r') as f:
    seqs=f['seqs'][:5]
    coding=f['coding'][:5]
    splice=f['splice'][:5]
X=np.concatenate([seqs.astype(np.float32), coding[...,None].astype(np.float32), splice[...,None].astype(np.float32)],axis=2)
X=torch.from_numpy(X).permute(0,2,1).contiguous()
model=Saluki(); load_weights(model,'models/train_gru/f0_c0/train/model0_best.h5'); model.eval()
with torch.no_grad(): pred=model(X).numpy().ravel()
with h5py.File('models/train_gru/f0_c0/test0_preds.h5','r') as f:
    ref=f['preds'][:5].ravel()
print('pred',pred)
print('ref ',ref)
print('maxdiff',np.max(np.abs(pred-ref)))
PY"
