import struct
data=open('tail.zip','rb').read()
for sig in [b'PK\\x06\\x07', b'PK\\x06\\x06']:
    print(sig, data.rfind(sig))
idx=data.rfind(b'PK\\x06\\x07')
print(data[idx:idx+60])
idx2=data.rfind(b'PK\\x06\\x06')
if idx2>=0:
    rec=data[idx2:idx2+80]
    print(rec)
    # zip64 EOCD record fields: sig 4, size 8, version made 2, need 2, disk 4, cd disk 4, entries disk 8, entries total 8, cd size 8, cd offset 8
    vals=struct.unpack('<4sQHHIIQQQQ', rec[:56])
    print(vals)
PY"
