/bin/bash -lc "python - <<'PY'
import urllib.request, struct, zlib, json, pathlib
entries={e['name']:e for e in json.load(open('central_entries.json'))}
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
for name in ['datasets/deeplearning/train_gru/f0_c0/data0/genes.tsv','datasets/deeplearning/train_gru/f0_c0/data1/genes.tsv']:
    e=entries[name]
    off=e['local_off']; csz=e['comp_size']; usz=e['uncomp_size']
    # fetch local header + compressed data + maybe descriptor
    total=csz+512
    req=urllib.request.Request(url, headers={'Range':f'bytes={off}-{off+total-1}'})
    r=urllib.request.urlopen(req, timeout=120)
    data=r.read()
    print(name, r.status, len(data), 'expected', total)
    vals=struct.unpack('<IHHHHHIIIHH', data[:30])
    nlen=vals[9]; xlen=vals[10]; method=vals[2]
    data_start=30+nlen+xlen
    comp=data[data_start:data_start+csz]
    if len(comp)<csz:
        raise RuntimeError('short')
    raw=zlib.decompress(comp,-15)
    print('uncomp', len(raw), usz)
    out=pathlib.Path('genes_'+name.split('/')[-1].replace('genes.tsv',name.split('/')[-2]+'.tsv'))
    # actually simple names
    outname=name.split('/')[-2]+'_genes.tsv'
    pathlib.Path(outname).write_bytes(raw)
    print('saved',outname, raw[:200])
PY
ls -lh *_genes.tsv"
