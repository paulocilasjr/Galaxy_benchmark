import json, urllib.request, struct, zlib, os, concurrent.futures, time
entries={e['name']:e for e in json.load(open('central_entries.json'))}
url='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
want=[]
def add(n,d):
    want.append((n,d))
add('datasets/deeplearning/train_gru/params.json','models/train_gru/params.json')
add('datasets/deeplearning/train_gru/f0_c0/data0/statistics.json','models/train_gru/f0_c0/data0/statistics.json')

    add(f'datasets/deeplearning/train_gru/f{fi}_c0/train/model0_best.h5', f'models/train_gru/f{fi}_c0/train/model0_best.h5')

def fetch(item):
    name,dest=item; e=entries[name]
    os.makedirs(os.path.dirname(dest),exist_ok=True)
    # fetch local header with generous 65536 extra; compressed files up to ~1.8MB
    start=e['local_off']; end=start+e['comp_size']+65536-1
    req=urllib.request.Request(url, headers={'Range':f'bytes={start}-{end}'})
    with urllib.request.urlopen(req, timeout=120) as r:
        data=r.read()
    rec=struct.unpack('<IHHHHHIIIHH', data[:30])
    if rec[0] "'!= 0x04034b50: raise Exception('"'bad local header '+name)
    name_len, extra_len = rec[9], rec[10]
    method=rec[3]
    comp_start=30+name_len+extra_len
    comp=data[comp_start:comp_start+e['comp_size']]
    if method==8:
        out=zlib.decompress(comp,-15)
    elif method==0:
        out=comp
    else:
        raise Exception('method '+str(method))
    if len(out)"'!=e['"'uncomp_size']:
        print('size mismatch',name,len(out),e['uncomp_size'])
    open(dest,'wb').write(out)
    return name,len(out)

with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
    for res in ex.map(fetch,want):
        print('done',res)
print('all downloaded')
