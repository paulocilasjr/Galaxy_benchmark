/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, struct, sys
sys.path.insert(0,'proto'); from example_pb2 import Example
with h5py.File('models/train_gru/f0_c0/ism_scores.h5','r') as f: s=f['seqs'][0]
idx=np.where(s.sum(axis=-1)"'!=0)[0]; start=idx[0]; end=idx[-1]+1
scores_dna='"''.join('ACGT'[np.argmax(s[i])] for i in range(start,end))
fn='models/train_gru/f0_c0/test-0.tfr.dec'; data=open(fn,'rb').read()
length=struct.unpack('<Q',data[0:8])[0]; payload=data[12:12+length]; ex=Example(); ex.ParseFromString(payload); raw=ex.features.feature['sequence'].bytes_list.value[0]
tfr_dna=''.join('ACGT'[b] for b in raw)
print('scores len',len(scores_dna),'tfr len',len(tfr_dna))
print('scores',scores_dna[:60])
print('tfr   ',tfr_dna[:60])
# find match offset
for off in [-1,0,1]:
 print('off',off,'match',scores_dna==tfr_dna[off:off+len(scores_dna)] if len(tfr_dna)>=len(scores_dna)+off else None)
# compare suffix
print('scores tail',scores_dna[-30:]); print('tfr tail',tfr_dna[-30:])
PY"
