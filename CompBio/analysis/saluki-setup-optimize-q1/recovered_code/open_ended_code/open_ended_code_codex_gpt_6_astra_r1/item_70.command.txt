/bin/bash -lc "cat > fetch_last.py <<'PY'
from pathlib import Path
import requests,zipfile,struct,zlib,time
z=zipfile.ZipFile('archive_sparse.zip')
i=z.getinfo('datasets/deeplearning/train_gru/f8_c1/train/model0_best.h5')
u='https://zenodo.org/api/records/6326409/files/datasets.zip/content'
r=requests.get(u,headers={'Range':f'bytes={i.header_offset}-{i.header_offset+i.compress_size+4095}'},timeout=120)
print(r.status_code,r.headers,len(r.content),flush=True)
r.raise_for_status();assert r.status_code==206
h=struct.unpack_from('<4s5H3L2H',r.content);start=30+h[-2]+h[-1];data=zlib.decompress(r.content[start:start+i.compress_size],-15)
assert len(data)==i.file_size and zlib.crc32(data)&0xffffffff==i.CRC
p=Path('models/f8_c1/train/model0_best.h5');temp=p.with_suffix('.download');temp.write_bytes(data);temp.replace(p)
PY
python -u fetch_last.py > fetch_last.log 2>&1"
