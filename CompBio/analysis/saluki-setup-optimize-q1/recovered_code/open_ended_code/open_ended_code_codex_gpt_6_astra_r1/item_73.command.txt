/bin/bash -lc "cat results/history.json && cat validation.json && python - <<'PY'
import csv,json
from pathlib import Path
import numpy as np
initial='GCCGCCACCATGGTGAGCAAGGGCGAGTAGTGTACATAATAAGGACT'
prev=initial
history=json.load(open('results/history.json'))
for r in range(1,4):
 rows=list(csv.DictReader(open(f'results/round{r}_candidates.csv')))
 preds=np.load(f'results/round{r}_fold_predictions.npy')
 assert preds.shape==(50,52) and np.isfinite(preds).all()
 assert np.std(preds[:,0])>0
 expected={prev}|{prev[:i]+b+prev[i+1:] for i in range(30,len(prev)) for b in 'ACGT' if b"'!=prev[i]}
 assert set(row['"'sequence'] for row in rows)==expected
 means=preds.mean(axis=0,dtype=np.float64)
 assert np.allclose(means,[float(row['mean_human_prediction']) for row in rows],rtol=0,atol=0)
 winner=rows[int(means.argmax())]['sequence']
 assert winner==history[r-1]['sequence']
 assert winner[:30]==initial[:30]
 assert sum(a"'!=b for a,b in zip(prev,winner))==1
 print('"'round',r,'best',winner,'gain',means.max()-means[0],'margin',np.sort(means)[-1]-np.sort(means)[-2])
 prev=winner
final=Path('results/final_sequence.txt').read_text()
assert final==prev+'\\n' and len(final.splitlines())==1
assert len(prev)==47 and set(prev)<=set('ACGT')
print('Changed positions:',[(i+1,a,b) for i,(a,b) in enumerate(zip(initial,prev)) if a"'!=b])
print('"'Verified final:',repr(final))
PY"
