/bin/bash -c "python3 - <<'PY'
import gzip, collections
tx=set(); ttype={}; exons=[]
with gzip.open('gencode.vM31.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        if 'gene_name \"Lepr\"' in p[8]:
            attr=dict(item.strip().split(' ',1) for item in p[8].split(';') if ' ' in item.strip())
            if p[2]=='transcript':
                tx.add(attr.get('transcript_id',''))
                ttype[attr.get('transcript_id','')]=attr.get('transcript_type','')
            elif p[2]=='exon':
                exons.append((int(p[3]),int(p[4]),attr.get('transcript_id','')))
print('transcripts',len(tx))
for k in sorted(tx):
    print(k, ttype[k])
print('exon records',len(exons))
PY"
