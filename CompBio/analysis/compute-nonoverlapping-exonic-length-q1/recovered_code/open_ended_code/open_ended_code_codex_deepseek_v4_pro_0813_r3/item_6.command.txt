/bin/bash -c "python3 - <<'PY'
import gzip
intervals=[]
with gzip.open('gencode.vM31.annotation.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        if p[2]=='exon' and 'gene_name \"Lepr\"' in p[8]:
            intervals.append((int(p[3]), int(p[4])))
intervals.sort()
union=[]
for a,b in intervals:
    if not union or a>union[-1][1]:
        union.append([a,b])
    elif b>union[-1][1]:
        union[-1][1]=b
print(len(intervals), sum(b-a+1 for a,b in union), union[:5], union[-5:])
PY"
