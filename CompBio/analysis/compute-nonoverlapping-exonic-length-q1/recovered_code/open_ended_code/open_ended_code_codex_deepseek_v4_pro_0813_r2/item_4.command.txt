/bin/bash -lc "python3 - <<'PY'
import gzip, re
path='gencode.vM31.annotation.gtf.gz'
pos=set(); rows=0
with gzip.open(path,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip().split('\\t')
        if len(p)<9 or p[2]"'!='"'exon': continue
        attrs=p[8]
        if re.search(r'gene_name \"Lepr\"', attrs):
            s,e=map(int,p[3:5])
            pos.update(range(s,e+1)); rows+=1
print(rows, len(pos))
PY"
