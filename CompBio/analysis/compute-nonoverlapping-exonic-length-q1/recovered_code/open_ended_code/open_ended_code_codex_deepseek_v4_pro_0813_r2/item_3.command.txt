/bin/bash -lc "python3 - <<'PY'
import gzip, re, sys
path='gencode.vM31.annotation.gtf.gz'
intervals=[]
transcripts=set()
exon_ids=set()
gene_ids=set()
with gzip.open(path,'rt') as f:
    for line in f:
        if line.startswith('#'):
            continue
        parts=line.rstrip('\\n').split('\\t')
        if len(parts)<9: continue
        chrom, source, ftype, start, end, score, strand, frame, attrs = parts
        if ftype "'!= '"'exon':
            continue
        # parse attributes
        d={}
        for m in re.finditer(r'(\\S+?)\\s+\"(["'^"]*)"'"', attrs):
            d[m.group(1)] = m.group(2)
        if d.get('gene_name') == 'Lepr':
            intervals.append((int(start), int(end)))
            transcripts.add(d.get('transcript_id'))
            exon_ids.add(d.get('exon_id'))
            gene_ids.add(d.get('gene_id'))
print('gene_ids', gene_ids)
print('transcripts', len(transcripts), sorted(transcripts)[:5], sorted(transcripts)[-5:])
print('exon rows', len(intervals), 'unique exon ids', len(exon_ids))
intervals.sort()
merged=[]
for s,e in intervals:
    if not merged or s > merged[-1][1]:
        merged.append([s,e])
    else:
        if e > merged[-1][1]:
            merged[-1][1]=e
total=sum(e-s+1 for s,e in merged)
print('merged intervals', len(merged))
print('merged', merged)
print('total', total)
PY"
