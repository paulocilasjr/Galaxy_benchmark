/bin/bash -lc "bash make_bins.sh 10000 && bash make_bins.sh 1000000 && python3 - <<'PY'
import os, glob
import numpy as np
for dirname in ['bins10000','bins100000','bins1000000']:
    files=sorted(glob.glob(dirname+'/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0]))
    keys=set(); vals=[]
    for f in files:
        d={}
        with open(f) as fh:
            for line in fh:
                k,v=line.rstrip('\\n').split('\\t')
                d[k]=int(v); keys.add(k)
        vals.append(d)
    keys=sorted(keys)
    X=np.array([[d.get(k,0) for k in keys] for d in vals], dtype=float)
    print('\\nDIR', dirname, 'bins', X.shape[1])
    transforms = {
        'raw': X,
        'log1p': np.log1p(X),
        'sqrt': np.sqrt(X),
        'binary': (X>0).astype(float),
    }
    # also cap each sample at its 99.5 percentile nonzero to reduce focal pileups
    cap=X.copy()
    for r in range(cap.shape[0]):
        nz=cap[r][cap[r]>0]
        q=np.percentile(nz,99.5)
        cap[r]=np.minimum(cap[r], q)
    transforms['cap995_log1p']=np.log1p(cap)
    for name,Y in transforms.items():
        C=np.corrcoef(Y)
        avgs=[]
        for i,row in enumerate(C): avgs.append((row.sum()-1)/(len(row)-1))
        order=np.argsort(avgs)+1
        print(name, 'avg_order', ' '.join(f'{i}:{avgs[i-1]:.4f}' for i in order))
PY"
