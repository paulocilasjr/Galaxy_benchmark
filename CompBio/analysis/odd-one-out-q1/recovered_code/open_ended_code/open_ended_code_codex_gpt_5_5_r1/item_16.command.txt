/bin/bash -lc "python3 - <<'PY'
import os, glob, math
import numpy as np
files=sorted(glob.glob('bins100k/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0]))
keys=set(); vals=[]
for f in files:
    d={}
    for line in open(f):
        k,v=line.rstrip('\\n').split('\\t')
        d[k]=int(v); keys.add(k)
    vals.append(d)
keys=sorted(keys)
X=np.array([[d.get(k,0) for k in keys] for d in vals], dtype=float)
for transform in ['raw','log1p','binary']:
    if transform=='log1p': Y=np.log1p(X)
    elif transform=='binary': Y=(X>0).astype(float)
    else: Y=X
    C=np.corrcoef(Y)
    print('\\n',transform)
    print('avg corr to others')
    for idx,row in enumerate(C,1):
        avg=(row.sum()-1)/(len(row)-1)
        med=np.median(np.delete(row,idx-1))
        print(idx, f'avg={avg:.4f}', f'med={med:.4f}', 'minmax', f'{np.delete(row,idx-1).min():.4f}', f'{np.delete(row,idx-1).max():.4f}')
    print('matrix')
    print('    '+' '.join(f'{i:>6}' for i in range(1,11)))
    for i,row in enumerate(C,1): print(f'{i:>2} '+' '.join(f'{x:6.3f}' for x in row))
PY"
