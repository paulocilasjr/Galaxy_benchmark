/bin/bash -lc "python3 - <<'PY'
import gzip, os, glob, bisect, re
from collections import defaultdict
classes=['PLS','pELS','dELS','DNase-H3K4me3','CTCF-only']
tmp={c:defaultdict(list) for c in classes}
with gzip.open('refs/ENCFF924IMH.bed.gz','rt') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  chrom,s,e=p[0],int(p[1]),int(p[2]); label=p[9]
  base=label.split(',')[0]
  if base in tmp and re.match(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y)$'"', chrom):
   tmp[base][chrom].append((s,e))
# merge per class
ints={}
for c,d in tmp.items():
 ints[c]={}
 for chrom,arr in d.items():
  arr=sorted(arr); merged=[]
  for s,e in arr:
   if not merged or s>merged[-1][1]: merged.append([s,e])
   elif e>merged[-1][1]: merged[-1][1]=e
  ints[c][chrom]=([x[0] for x in merged],[x[1] for x in merged])
def ov(c,chrom,s,e):
 if chrom not in ints[c]: return False
 starts,ends=ints[c][chrom]
 i=bisect.bisect_right(ends,s)
 return i<len(starts) and starts[i]<e
primary=re.compile(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y|M):'"')
print('sample\\t'+'\\t'.join(classes)+'\\tany_cCRE')
for bf in sorted(glob.glob('bins10000/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0])):
 sample=os.path.basename(bf).split('.')[0]
 totals={c:0 for c in classes}; anytot=0; total=0
 for line in open(bf):
  key,v=line.rstrip('\\n').split('\\t'); v=int(v); total+=v
  if not primary.match(key): continue
  chrom,b=key.split(':'); s=int(b)*10000; e=s+10000
  anyhit=False
  for c in classes:
   if ov(c,chrom,s,e): totals[c]+=v; anyhit=True
  if anyhit: anytot+=v
 print(sample, *(f'{totals[c]/total:.4f}' for c in classes), f'{anytot/total:.4f}', sep='\\t')
PY"
