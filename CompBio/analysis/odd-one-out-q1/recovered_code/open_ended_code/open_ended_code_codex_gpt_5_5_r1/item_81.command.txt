/bin/bash -lc "python3 - <<'PY'
import bisect, glob, os, re
from collections import defaultdict
SETS = {
 'promoter':'refs/promoter_2kb_merged.bed',
 'gene':'refs/gene_merged.bed',
 'exon':'refs/exon_merged.bed',
 'intron':'refs/intron_no_promoter.bed',
}
def load(path):
 d=defaultdict(lambda:([],[]))
 tmp=defaultdict(lambda:[[],[]])
 for line in open(path):
  chrom,s,e=line.split()[:3]; tmp[chrom][0].append(int(s)); tmp[chrom][1].append(int(e))
 return tmp
def ov(ints, chrom,s,e):
 if chrom not in ints: return False
 starts,ends=ints[chrom]
 i=bisect.bisect_right(ends,s)
 return i < len(starts) and starts[i] < e
ints={k:load(v) for k,v in SETS.items()}
primary=re.compile(r'"'^chr([1-9]|1[0-9]|2[0-2]|X|Y|M):'"')
print('sample\\tpromoter\\tgene\\texon\\tintron')
for f in sorted(glob.glob('bins10000/*.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0])):
 sample=os.path.basename(f).split('.')[0]
 totals={k:0 for k in SETS}; total=0
 for line in open(f):
  key,c=line.rstrip('\\n').split('\\t'); c=int(c); total+=c
  if not primary.match(key): continue
  chrom,b=key.split(':'); s=int(b)*10000; e=s+10000
  for name in SETS:
   if ov(ints[name], chrom,s,e): totals[name]+=c
 print(sample, *(f'{totals[k]/total:.4f}' for k in ['promoter','gene','exon','intron']), sep='\\t')
PY"
