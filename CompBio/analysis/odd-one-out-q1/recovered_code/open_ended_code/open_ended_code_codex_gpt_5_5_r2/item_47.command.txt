/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections, time
files=sorted(glob.glob('odd.one.out.q1/*.tagAlign.gz'), key=lambda p:int(os.path.basename(p).split('.')[0]))
for path in files:
    id_=os.path.basename(path).split('.')[0]; t=time.time()
    plus=collections.Counter(); minus=collections.Counter(); n=0
    # use chr20: enough reads and avoids chr11 hotspot artifact
    with gzip.open(path,'rt') as fh:
        for line in fh:
            f=line.split('\\t')
            if f[0] "'!= '"'chr20': continue
            if f[5][0]=='+': plus[int(f[1])] += 1
            else: minus[int(f[2])-1] += 1
            n+=1
    vals=[]
    for lag in range(0,501):
        s=0
        # plus upstream of minus, typical fragment lag between 5p ends
        for p,c in plus.items():
            m=minus.get(p+lag)
            if m: s += c*m
        vals.append(s)
    # ignore zero/readlen duplicate peak? report maxima in ranges
    top=max(range(len(vals)), key=lambda i: vals[i])
    top_gt50=max(range(51,len(vals)), key=lambda i: vals[i])
    print(id_, 'chr20tags', n, 'top', top, vals[top], 'top_gt50', top_gt50, vals[top_gt50], 'v50', vals[50], 'v100', vals[100], 'v150', vals[150], 'v200', vals[200], 'sec', f'{time.time()-t:.1f}', flush=True)
PY"
