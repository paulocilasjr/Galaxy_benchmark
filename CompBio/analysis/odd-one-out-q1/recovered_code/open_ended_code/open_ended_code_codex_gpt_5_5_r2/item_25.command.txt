import glob, numpy as np, os
files=sorted(glob.glob('analysis/*.100kb.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0]))

    counts=[]
    for line in open(f): counts.append(int(line.split()[3]))

    total=x.sum(); xs=np.sort(x)[::-1]
    print(os.path.basename(f).split('.')[0], 'nonzero_bins', len(x), 'max', int(xs[0]), 'top10_frac', xs[:10].sum()/total, 'top100_frac', xs[:100].sum()/total, 'top1000_frac', xs[:1000].sum()/total, 'median_nz', np.median(x), 'p99_nz', np.quantile(x,.99))
PY
