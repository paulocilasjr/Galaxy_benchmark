/bin/bash -lc "python3 - <<'PY'
import glob, math
import numpy as np
files=sorted(glob.glob('analysis/*.100kb.tsv'), key=lambda p:int(p.split('/')[-1].split('.')[0]))
keys=[]; idx={}
rows=[]
for f in files:
    d={}
    for line in open(f):
        id_,chr_,bin_,cnt=line.split()
        key=(chr_,int(bin_))
        d[key]=int(cnt)
        if key not in idx:
            idx[key]=len(keys); keys.append(key)
    rows.append(d)
X=np.zeros((len(rows),len(keys)), dtype=float)
for i,d in enumerate(rows):
    for k,v in d.items(): X[i,idx[k]]=v
ids=[int(f.split('/')[-1].split('.')[0]) for f in files]
print('shape', X.shape, 'ids', ids)
for name,Y in [('raw',X),('log1p',np.log1p(X)),('sqrt',np.sqrt(X))]:
    C=np.corrcoef(Y)
    print('\\n',name)
    print('mean corr to others')
    for i,id_ in enumerate(ids):
        m=(C[i].sum()-1)/(len(ids)-1)
        print(id_, f'{m:.4f}', 'minpair', f'{np.min(np.delete(C[i],i)):.4f}', 'maxpair', f'{np.max(np.delete(C[i],i)):.4f}')
    print('matrix')
    print('    '+' '.join(f'{id_:>6}' for id_ in ids))
    for i,id_ in enumerate(ids):
        print(f'{id_:>3} '+' '.join(f'{C[i,j]:6.3f}' for j in range(len(ids))))
# Euclidean distances standardized per bin? zscore columns? maybe cosine of centered rows included by corr
PY"
