/bin/bash -lc "python3 - <<'PY'
import glob, os, numpy as np
files=sorted(glob.glob('analysis_dedup/*.100kb.dedup.tsv'), key=lambda p:int(os.path.basename(p).split('.')[0]))
idx={}; keys=[]; rows=[]
for f in files:
    d={}
    for line in open(f):
        id_,chr_,bin_,cnt=line.split()
        key=(chr_,int(bin_)); d[key]=int(cnt)
        if key not in idx: idx[key]=len(keys); keys.append(key)
    rows.append(d)
X=np.zeros((len(rows),len(keys)),float)
for i,d in enumerate(rows):
    for k,v in d.items(): X[i,idx[k]]=v
ids=[int(os.path.basename(f).split('.')[0]) for f in files]
print('shape', X.shape, ids)
for name,Y in [('raw',X),('log1p',np.log1p(X)),('sqrt',np.sqrt(X))]:
    C=np.corrcoef(Y)
    print('\\n',name)
    for i,id_ in enumerate(ids):
        others=np.delete(C[i],i)
        print(id_, 'mean', f'{others.mean():.4f}', 'min', f'{others.min():.4f}', 'max', f'{others.max():.4f}')
    print('matrix')
    print('   '+' '.join(f'{id_:>5}' for id_ in ids))
    for i,id_ in enumerate(ids): print(f'{id_:>2} '+' '.join(f'{C[i,j]:5.2f}' for j in range(len(ids))))
print('\\nconcentration')
for i,id_ in enumerate(ids):
    xs=np.sort(X[i][X[i]>0])[::-1]; total=xs.sum()
    print(id_, 'totaluniq', int(total), 'max', int(xs[0]), 'top10', f'{xs[:10].sum()/total:.4f}', 'top100', f'{xs[:100].sum()/total:.4f}', 'top1000', f'{xs[:1000].sum()/total:.4f}')
PY"
