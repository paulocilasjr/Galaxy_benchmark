/bin/bash -lc "python3 - <<'PY'
import json,numpy as np
from scipy.ndimage import gaussian_filter1d
D=sorted(json.load(open('endpoint_bias.json')),key=lambda x:int(x['file'].split('.')[0]))
for x in D:
 c=x['adjacent']; a=np.array([c.get(f'+-:{i}',0) for i in range(1,501)],float)
 sm=gaussian_filter1d(a,3); baseline=gaussian_filter1d(a,30)+1; ratio=sm/baseline
 inds=np.argsort(ratio[19:])[::-1][:8]+19
 print(x['file'].split('.')[0],'ratio peaks',[(i+1,round(ratio[i],2),int(a[i])) for i in inds], 'selected',[(j,int(a[j-1])) for j in [20,30,40,50,75,100,125,150,175,200,250,300,400,500]])
PY"
