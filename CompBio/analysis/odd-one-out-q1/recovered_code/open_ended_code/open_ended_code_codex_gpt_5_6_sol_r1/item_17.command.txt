/bin/bash -lc "python - <<'PY'
import gzip,numpy as np
from analyze_tagalign import OFFSETS,BIN,CHROMS
T=set()
with gzip.open('refGene.txt.gz','rt') as h:
 for l in h:
  f=l.rstrip().split('\\t'); c=f[2]
  if c not in OFFSETS: continue
  p=int(f[4]) if f[3]=='+' else int(f[5])-1
  T.add((c,p//1000))
idx=[]
for c,b in T:
 if b>=50 and b+50<(CHROMS[c]+999)//1000: idx.append(OFFSETS[c]+b)
idx=np.array(idx)
print('TSS',len(idx))
for i in range(1,11):
 x=np.load(f'features/{i}.npy').astype(float)
 prof=np.array([x[idx+d].mean() for d in range(-50,51)])
 flank=np.r_[prof[:40],prof[-40:]].mean()
 print(i,'center',np.round(prof[48:53],2),'fold',round(prof[50]/flank,2),'near5',round(prof[45:56].mean()/flank,2),'near20',round(prof[30:71].mean()/flank,2))
PY"
