import numpy as np
from pathlib import Path
from scipy.spatial.distance import pdist, squareform
X=np.stack([np.load(f'features/{i}.npy') for i in range(1,11)])
print('shape',X.shape,'sums',X.sum(1))
for i,x in enumerate(X,1):
 nz=x[x>0]
 print(i,'nonzero',len(nz),'max',x.max(),'top1k',np.sort(x)[-1000:].sum(),'sqsum',np.dot(x.astype(float),x),'q',np.quantile(nz,[.5,.9,.99,.999]))
for transform in ['raw','log','binary']:
 Y={'raw':X.astype(float),'log':np.log1p(X),'binary':(X>0).astype(float)}[transform]
 C=np.corrcoef(Y)
 print('\\n',transform,'correlation')
 print(np.array2string(C,precision=3,suppress_small=True))
 print('mean offdiag',[(i+1,(C[i].sum()-1)/9) for i in range(10)])
PY"
