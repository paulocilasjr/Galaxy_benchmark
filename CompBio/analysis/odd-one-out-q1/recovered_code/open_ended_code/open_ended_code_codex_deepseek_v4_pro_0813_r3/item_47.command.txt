/bin/bash -lc "python3 - <<'PY'
import gzip, collections, os
f='odd/odd.one.out.q1/4.tagAlign.gz'
c=collections.Counter()
with gzip.open(f,'rt') as fh:
    for line in fh:
        p=line.split('\\t'); c[p[0]]+=1
canon=sum(c[ch] for ch in [f'chr{i}' for i in range(1,23)]+['chrX','chrY'])
non=[(ch,n) for ch,n in c.items() if ch not in [f'chr{i}' for i in range(1,23)]+['chrX','chrY']]
print('total',sum(c.values()),'canon',canon,'non',len(non),'nonreads',sum(n for _,n in non))
for ch,n in sorted(non, key=lambda x:-x[1])[:30]: print(ch,n)
PY"
