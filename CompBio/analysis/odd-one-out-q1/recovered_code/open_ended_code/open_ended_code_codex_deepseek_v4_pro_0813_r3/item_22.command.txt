/bin/bash -lc "python3 - <<'PY'
# create tss_windows 100bp from -5000 to +5000 (relative to TSS), excluding tss point window maybe
with open('tss.bed') as f, open('tss_windows.bed','w') as out:
    for line in f:
        p=line.split()
        chrom=p[0]; pos=int(p[2]); strand=p[5]
        for rel in range(-5000,5000,100):
            s=pos+rel; e=s+100
            if s<0: continue
            out.write(f'{chrom}\\t{s}\\t{e}\\t{rel}\\n')
PY
wc -l tss_windows.bed; head -3 tss_windows.bed"
