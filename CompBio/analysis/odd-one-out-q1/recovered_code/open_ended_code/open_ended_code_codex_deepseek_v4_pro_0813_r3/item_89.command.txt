/bin/bash -lc "python3 - <<'PY'
import collections
path='/opt/miniconda3/envs/benchmark_codex_a75c7e44e7bb415d9720/annotation/hg38_RefSeq_exon.txt'
g=collections.defaultdict(list)
with open(path) as f:
    next(f)
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        g[p[0]].append((p[1],int(p[2]),int(p[3]),p[4]))
with open('genes.bed','w') as out:
    for gene,exons in g.items():
        chrom=exons[0][0]; strand=exons[0][3]
        start=min(e[1] for e in exons); end=max(e[2] for e in exons)
        out.write(f'{chrom}\\t{start}\\t{end}\\t{gene}\\t0\\t{strand}\\n')
PY
wc -l genes.bed"
