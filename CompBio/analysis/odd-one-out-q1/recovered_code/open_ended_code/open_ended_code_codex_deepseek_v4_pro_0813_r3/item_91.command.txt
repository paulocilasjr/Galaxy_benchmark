import glob,os,collections
files=sorted(glob.glob('genecov/*.tsv'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    name=os.path.basename(f).split('.')[0]
    tot=0; n=0; lengths=[]
    with open(f) as fh:
        for line in fh:
            p=line.split(); length=int(p[2])-int(p[1]); count=int(p[4]); tot+=count; n+=1; lengths.append(length)
    print(name,'reads_in_genes',tot,'mean_per_gene',tot/n,'frac_of_10M',tot/1e6)
PY"
