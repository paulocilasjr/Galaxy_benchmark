/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections, statistics
files=sorted(glob.glob('odd/odd.one.out.q1/*.tagAlign.gz'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    chrom=collections.Counter(); lens=collections.Counter(); strands=collections.Counter(); starts=[]
    with gzip.open(f,'rt') as fh:
        for line in fh:
            p=line.rstrip('\\n').split('\\t')
            chrom[p[0]]+=1; lens[int(p[2])-int(p[1])]+=1; strands[p[5]]+=1; starts.append(int(p[1]))
    print(os.path.basename(f), 'chroms',len(chrom), 'top',chrom.most_common(8))
    print('  lens top', lens.most_common(5), 'strands', strands)
    print('  start min/max mean median sd', min(starts),max(starts),statistics.mean(starts),statistics.median(starts),statistics.pstdev(starts))
PY"
