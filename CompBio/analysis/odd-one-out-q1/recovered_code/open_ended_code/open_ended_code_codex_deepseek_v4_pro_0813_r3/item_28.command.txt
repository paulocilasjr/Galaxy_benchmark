/bin/bash -lc "python3 - <<'PY'
import numpy as np, glob, os
files=sorted(glob.glob('cov/*.counts'), key=lambda x:int(os.path.basename(x).split('.')[0]))
for f in files:
    a=np.loadtxt(f,usecols=3).astype(np.float64)
    total=a.sum(); n=len(a)
    sorted_desc=np.sort(a)[::-1]
    csum=np.cumsum(sorted_desc)
    frac_top01=csum[int(n*0.001)-1]/total
    frac_top1=csum[int(n*0.01)-1]/total
    frac_top5=csum[int(n*0.05)-1]/total
    frac_top10=csum[int(n*0.1)-1]/total
    nonzero=(a>0).sum()/n
    entropy=-(a[a>0]/total*np.log2(a[a>0]/total)).sum()
    gini=1-2*((np.sort(a)/total).cumsum()).sum()/n + (np.sort(a)/total).sum()/n # maybe
    print(os.path.basename(f).split('.')[0], 'nonzero',round(nonzero,4), 'top0.1%',round(frac_top01,4),'top1%',round(frac_top1,4),'top5%',round(frac_top5,4),'top10%',round(frac_top10,4),'entropy',round(entropy,3))
PY"
