/bin/bash -lc "python3 - <<'PY'
import glob, os, collections, numpy as np
files=sorted(glob.glob('tsscov/*.tsv'), key=lambda x:int(os.path.basename(x).split('.')[0]))
profiles={}
names=[]
for f in files:
    name=os.path.basename(f).split('.')[0]; names.append(name)
    d=collections.defaultdict(int)
    n=collections.defaultdict(int)
    with open(f) as fh:
        for line in fh:
            p=line.split()
            rel=int(p[3]); c=int(p[4])
            d[rel]+=c; n[rel]+=1
    rels=sorted(d)
    arr=np.array([d[r]/n[r] for r in rels])
    profiles[name]=(rels,arr)
print('mean TSS coverage per 100bp by rel')
for name in names:
    rels,arr=profiles[name]
    # print few values and shape metrics
    total=arr.sum(); peak=arr.max(); peakrel=rels[int(np.argmax(arr))]
    print(name,'sum',round(total,2),'peak',round(peak,3),'peakrel',peakrel,
          'near',round(arr[rels.index(0):rels.index(0)+10].mean(),3) if 0 in rels else None,
          'up',round(arr[:rels.index(0)].mean(),3) if 0 in rels else None,
          'down',round(arr[rels.index(0)+1:].mean(),3) if 0 in rels else None)
# write matrix and plot? Save profiles
np.savez('tss_profiles.npz', names=names, rels=rels, **{f'p{n}':profiles[n][1] for n in names})
# print normalized profile around TSS first 100bp bins relative -1000 to 1000
for name in names:
    rels,arr=profiles[name]
    idx={r:i for i,r in enumerate(rels)}
    sub=[round(arr[idx[r]],3) for r in range(-1000,1000,100)]
    print(name, sub)
PY"
