/bin/bash -lc "python3 - <<'PY'
import csv
chroms=[]
with open('/opt/miniconda3/envs/benchmark_codex_a75c7e44e7bb415d9720/lib/python3.11/site-packages/bioframe/io/data/hg38.seqinfo.tsv') as f:
    r=csv.DictReader(f, delimiter='\\t')
    for row in r:
        name=row['name']; length=int(row['length'])
        if name in [f'chr{i}' for i in range(1,23)]+['chrX','chrY']:
            chroms.append((name,length))
with open('bins_10kb.bed','w') as out:
    for name,length in chroms:
        for s in range(0,length,10000):
            e=min(length,s+10000)
            out.write(f'{name}\\t{s}\\t{e}\\n')
print(len(chroms))
PY
wc -l bins_10kb.bed"
