/bin/bash -lc "python3 - <<'PY'
import csv, collections
path='/opt/miniconda3/envs/benchmark_codex_a75c7e44e7bb415d9720/annotation/hg38_RefSeq_exon.txt'
g=collections.defaultdict(list)
with open(path) as f:
    r=csv.DictReader(f, delimiter='\\t')
    for row in r:
        g[row['GeneID']].append((row['Chr'],int(row['Start']),int(row['End']),row['Strand']))
with open('tss.bed','w') as out:
    for gene,exons in g.items():
        chrom=exons[0][0]; strand=exons[0][3]
        starts=[e[1] for e in exons]; ends=[e[2] for e in exons]
        if strand=='+':
            tss=min(starts)
        else:
            tss=max(ends)
        out.write(f'{chrom}\\t{max(0,tss-1)}\\t{tss}\\t{gene}\\t0\\t{strand}\\n')
print(len(g))
PY
wc -l tss.bed && head tss.bed"
