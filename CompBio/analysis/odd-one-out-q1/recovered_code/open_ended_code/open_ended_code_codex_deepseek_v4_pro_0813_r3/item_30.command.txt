/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np, time, os, gzip
f='odd/odd.one.out.q1/1.tagAlign.gz'
t=time.time()
chunks=[]
reader=pd.read_csv(f, sep='\\t', header=None, usecols=[0,1,5], names=['chrom','start','strand'], chunksize=1000000, dtype={'chrom':'category','start':np.int32,'strand':'category'})
for ch in reader:
    ch['bin']=(ch['start']//10).astype(np.int32)
    g=ch.groupby(['chrom','strand','bin'], observed=True).size().reset_index(name='count')
    chunks.append(g)
df=pd.concat(chunks, ignore_index=True)
print('read',time.time()-t, df.shape, df.memory_usage(deep=True).sum()/1e6)
print(df.head())
print('n chroms', df.chrom.nunique())
PY"
