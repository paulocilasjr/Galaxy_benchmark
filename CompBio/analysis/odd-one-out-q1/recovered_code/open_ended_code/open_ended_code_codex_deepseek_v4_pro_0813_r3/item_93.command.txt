/bin/bash -lc "python3 - <<'PY'
# create metagene windows: 5kb upstream (100bp bins), 100 bins over gene body, 5kb downstream (100bp bins)
with open('genes.bed') as f, open('gene_windows.bed','w') as out:
    for line in f:
        p=line.rstrip('\\n').split('\\t')
        chrom=p[0]; start=int(p[1]); end=int(p[2]); strand=p[5]
        # upstream
        for i in range(50):
            s=start-5000+i*100; e=s+100
            if s<0: continue
            out.write(f'{chrom}\\t{s}\\t{e}\\tup_{i}\\n')
        length=end-start
        prev=start
        for i in range(100):
            s=start + int(round(length*i/100))
            e=start + int(round(length*(i+1)/100))
            if e>s:
                out.write(f'{chrom}\\t{s}\\t{e}\\tbody_{i}\\n')
            prev=e
        for i in range(50):
            s=end+i*100; e=s+100
            out.write(f'{chrom}\\t{s}\\t{e}\\tdown_{i}\\n')
PY
wc -l gene_windows.bed; head -3 gene_windows.bed"
