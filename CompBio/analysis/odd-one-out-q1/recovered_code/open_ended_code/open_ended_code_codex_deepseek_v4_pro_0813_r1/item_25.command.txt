/bin/bash -c "python3 - <<'PY'
import numpy as np, glob, re, os
from scipy.stats import zscore
files=glob.glob('tss_counts_*.bed'); files.sort(key=lambda x:int(re.search(r'tss_counts_(\\d+)\\.bed',x).group(1)))
names=[os.path.basename(f).replace('tss_counts_','').replace('.bed','') for f in files]
W=100; X=[]
for f in files:
 a=np.loadtxt(f,usecols=6,dtype=float).reshape(-1,W)
 prof=a.mean(axis=0)
 # normalize shape: z-score of log? raw
 X.append(prof)
X=np.array(X)
print('raw profile corr')
C=np.corrcoef(X)
print('\\t'.join(names))
for i,n in enumerate(names): print(n+'\\t'+'\\t'.join(f'{x:.3f}' for x in C[i]))
print('zscore profile corr')
Z=np.array([zscore(x) for x in X]); C=np.corrcoef(Z)
print('\\t'.join(names))
for i,n in enumerate(names): print(n+'\\t'+'\\t'.join(f'{x:.3f}' for x in C[i]))
# pairwise distances of zscore
print('pairwise euclid zscore')
D=np.linalg.norm(Z[:,None,:]-Z[None,:,:],axis=2)
for i,n in enumerate(names): print(n+'\\t'+'\\t'.join(f'{x:.1f}' for x in D[i]))
PY"
