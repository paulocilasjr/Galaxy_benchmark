/bin/bash -c "python3 - <<'PY'
import gzip
regions=[('chr11',24100000,24200000),('chr3',141200000,141300000),('chr17',61400000,61500000),('chr20',57200000,57300000),('chr17',82000000,82100000),('chr1',600000,700000)]
# build genes from refGene
with gzip.open('refGene.txt.gz','rt') as f:
 for line in f:
  p=line.rstrip().split('\\t')
  if len(p)<16: continue
  chrom=p[2]; s=int(p[4]); e=int(p[5]); name=p[12]
  for r in regions:
   if chrom==r[0] and e>=r[1] and s<=r[2]:
    print(r, name, s,e, p[3]); break
PY"
