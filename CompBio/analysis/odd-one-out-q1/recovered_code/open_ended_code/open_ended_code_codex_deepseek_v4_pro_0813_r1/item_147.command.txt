/bin/bash -c "python3 - <<'PY'
import requests
for output in ['alignments','unfiltered alignments']:
 q=f'https://www.encodeproject.org/search/?type=File&mapped_read_length=50&assembly=GRCh38&file_format=bam&output_type={requests.utils.quote(output)}&format=json&limit=5'
 r=requests.get(q,timeout=30); print(output,r.status_code,len(r.json().get('@graph',[])))
 data=r.json().get('@graph',[])
 for f in data[:5]: print(f.get('accession'),f.get('assay_title'),f.get('output_type'),f.get('mapped_read_length'))
PY"
