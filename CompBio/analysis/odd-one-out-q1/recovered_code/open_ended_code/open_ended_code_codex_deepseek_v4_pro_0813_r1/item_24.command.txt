/bin/bash -c "python3 - <<'PY'
import numpy as np, glob, re, os
files=glob.glob('tss_counts_*.bed'); files.sort(key=lambda x:int(re.search(r'tss_counts_(\\d+)\\.bed',x).group(1)))
names=[os.path.basename(f).replace('tss_counts_','').replace('.bed','') for f in files]
W=100
data=[]
for f in files:
 a=np.loadtxt(f,usecols=6,dtype=float)
 assert len(a)%W==0
 data.append(a.reshape(-1,W))
print('genes',data[0].shape)
for name,mat in zip(names,data):
 prof=mat.mean(axis=0)
 norm=prof/prof.mean()
 print(name, ' '.join(f'{v:.2f}' for v in norm[::10]))
 print('  TSS_enrich', norm[40:61].mean()/max(1e-9,(norm[0:20].mean()+norm[80:100].mean())/2))
PY"
