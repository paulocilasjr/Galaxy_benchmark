/bin/bash -c "python3 - <<'PY'
import gzip, re
# parse refGene
genes={}
with gzip.open('refGene.txt.gz','rt') as f:
 for line in f:
  p=line.rstrip('\\n').split('\\t')
  if len(p)<16: continue
  chrom=p[2]; strand=p[3]; txStart=int(p[4]); txEnd=int(p[5]); name2=p[12]
  if not chrom.startswith('chr') or '_' in chrom or chrom=='chrM': continue
  tss=txStart if strand=='+' else txEnd
  key=(name2,chrom)
  # keep longest span; tie largest tss? deterministic
  span=txEnd-txStart
  if key not in genes or span>genes[key][1] or (span==genes[key][1] and tss>genes[key][0]):
   genes[key]=(tss,span,strand)
# write windows
bins=list(range(-5000,5000,100))  # 100 bins, last end 5000
with open('refGene_tss_windows.bed','w') as out:
 for (gene,chrom),(tss,span,strand) in genes.items():
  for s in bins:
   out.write(f'{chrom}\\t{max(0,tss+s)}\\t{max(0,tss+s+100)}\\t{gene}\\t0\\t{strand}\\n')
print('genes',len(genes),'windows',len(genes)*len(bins))
PY
sort -k1,1 -k2,2n refGene_tss_windows.bed > refGene_tss_windows_lex.bed
wc -l refGene_tss_windows_lex.bed
head refGene_tss_windows_lex.bed"
