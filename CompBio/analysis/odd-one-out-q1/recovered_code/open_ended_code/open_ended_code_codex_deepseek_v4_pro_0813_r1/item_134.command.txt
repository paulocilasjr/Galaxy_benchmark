/bin/bash -c "python3 - <<'PY'
import requests
# get one DNase experiment MCF-7
url='https://www.encodeproject.org/search/?type=Experiment&assay_title=DNase-seq&replicates.library.biosample.organism.scientific_name=Homo%20sapiens&assembly=GRCh38&format=json&limit=all'
exps=requests.get(url,timeout=120).json()['@graph']
for exp in exps:
 if exp.get('biosample_ontology',{}).get('term_name')=='MCF-7' and exp.get('biosample_ontology',{}).get('classification')=='cell line':
  acc=exp['accession']; print(acc)
  r=requests.get(f'https://www.encodeproject.org/search/?type=File&dataset=/experiments/{acc}/&format=json&limit=all',timeout=60)
  files=r.json().get('@graph',[])
  for f in files:
   if f.get('file_format')=='bed':
    print(f.get('accession'),f.get('file_type'),f.get('output_type'),f.get('file_size'))
  break
PY"
