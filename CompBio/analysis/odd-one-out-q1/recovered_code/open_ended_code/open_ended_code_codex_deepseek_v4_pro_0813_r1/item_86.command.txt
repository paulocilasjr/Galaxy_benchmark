/bin/bash -c "python3 - <<'PY'
import numpy as np, glob, re, os
files=glob.glob('counts10k_*.bed'); files.sort(key=lambda x:int(re.search(r'counts10k_(\\d+)\\.bed',x).group(1)))
names=[os.path.basename(f).replace('counts10k_','').replace('.bed','') for f in files]
X=np.vstack([np.loadtxt(f,usecols=3,dtype=float) for f in files]).T
print(X.shape)
def corrmat(Y, label):
 print('\\n'+label)
 C=np.corrcoef(Y.T)
 print('\\t'.join(names))
 for i,n in enumerate(names): print(n+'\\t'+'\\t'.join(f'{x:.3f}' for x in C[i]))
# raw
corrmat(X,'pearson raw')
# log1p
corrmat(np.log1p(X),'pearson log1p')
# spearman via rankdata each col
from scipy.stats import rankdata
R=np.column_stack([rankdata(X[:,i]) for i in range(X.shape[1])])
corrmat(R,'spearman rank')
# remove bins where any file top? use median filtering per file top 0.1%
# clip each file at its 99.9th pct
Y=np.clip(X,0,np.quantile(X,0.999,axis=0))
corrmat(np.log1p(Y),'pearson log1p clipped 99.9')
# remove bins that are top 0.1% in any file
mask=np.ones(X.shape[0],dtype=bool)
for i in range(X.shape[1]):
 thr=np.quantile(X[:,i],0.999)
 mask &= (X[:,i]<=thr)
print('removed',np.sum(~mask))
corrmat(np.log1p(X[mask]),'pearson log1p removed top0.1 any')
PY"
