/bin/bash -c "python3 - <<'PY'
import pandas as pd
D=pd.read_csv('feature_counts.tsv',sep='\\t',header=None,names=['ann','file','count'])
totals={f:10000000 for f in range(1,11)}
P=D.pivot(index='file',columns='ann',values='count')
P['promoter_ratio']=P['tss']/totals
P['gene_body_ratio']=P['genes']/totals
P['exon_ratio']=P['exons']/totals
P['intron_ratio']=P['introns']/totals
P['intron_gene']=P['introns']/P['genes']
P['tss_gene']=P['tss']/P['genes']
print(P.to_string())
PY"
