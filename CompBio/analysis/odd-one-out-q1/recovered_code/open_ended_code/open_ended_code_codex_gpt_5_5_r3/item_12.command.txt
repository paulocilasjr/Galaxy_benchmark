/bin/bash -lc "python3 - <<'PY'
import gzip, glob, os, collections, math
import numpy as np
paths=sorted(glob.glob('odd.one.out.q1/*.tagAlign.gz'), key=lambda p:int(os.path.basename(p).split('.')[0]))
canon=[f'chr{i}' for i in range(1,23)]+['chrX','chrY']
canon=set(canon)
# sparse counts per 100kb canonical genomic bins
bin_size=100_000
all_keys=set(); dicts=[]; plus_minus=[]
for path in paths:
    d=collections.Counter(); pm=collections.Counter()
    with gzip.open(path,'rt') as fh:
        for line in fh:
            c,s,e,_,_,st=line.rstrip('\\n').split('\\t')[:6]
            if c in canon:
                k=(c,int(s)//bin_size)
                d[k]+=1
                pm[(k,st)]+=1
    dicts.append(d); all_keys.update(d)
keys=sorted(all_keys)
mat=np.array([[d.get(k,0) for k in keys] for d in dicts], dtype=float)
# normalize to counts per million then log1p to emphasize signal shape without letting one bin dominate entirely
mat=mat / mat.sum(axis=1, keepdims=True) * 1e6
log=np.log1p(mat)
corr=np.corrcoef(log)
print('Pearson corr log1p 100kb bins')
print('    '+' '.join(f'{i:>5}' for i in range(1,11)))
for i,row in enumerate(corr,1):
    print(f'{i:>2}  '+' '.join(f'{x:5.2f}' for x in row))
print('\\nmean corr to other 9:')
for i,row in enumerate(corr,1):
    vals=[row[j] for j in range(10) if j"'!=i-1]
    print(i, round(float(np.mean(vals)),3), '"'median', round(float(np.median(vals)),3))
# top bins per sample to see chr11 dominance
print('\\ntop bins')
for i,d in enumerate(dicts,1):
    print(i, d.most_common(5))
PY"
