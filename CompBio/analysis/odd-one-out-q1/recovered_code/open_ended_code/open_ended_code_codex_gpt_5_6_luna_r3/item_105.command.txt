/bin/bash -lc "curl -fLs 'https://www.encodeproject.org/search/?type=File&file_format=tagAlign&limit=100&format=json' -o encode_tagalign.json && python3 - <<'PY'
import json,collections
x=json.load(open('encode_tagalign.json'))['@graph']
c=collections.Counter()
for f in x: c[str(f.get('assay_term_name'))]+=1
print('returned',len(x),'total',json.load(open('encode_tagalign.json')).get('total'),'assays',c)
for f in x[:20]: print(f.get('accession'),f.get('assembly'),f.get('assay_title'),f.get('assay_term_name'),f.get('file_size'),f.get('cloud_metadata',{}).get('url'))
PY"
