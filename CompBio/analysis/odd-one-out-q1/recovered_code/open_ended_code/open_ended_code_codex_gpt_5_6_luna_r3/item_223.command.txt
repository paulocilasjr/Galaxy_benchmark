/bin/bash -lc "python3 - <<'PY'
import numpy as np
from pathlib import Path
D=[]
for j in range(1,11):
 d={}
 for line in Path(f'bins10_{j}.tsv').open():
  c,b,n=line.split(); d[(c,int(b))]=int(n)
 D.append(d)
# all bins with count > thresholds in at least one, show median/mean ranks and file8
for threshold in [100,200,500,1000]:
 keys=set().union(*(set(k for k,v in d.items() if v>=threshold) for d in D))
 A=np.array([[d.get(k,0) for k in keys] for d in D],float)
 # normalized per file, rank as z log
 Y=np.log1p(A)
 C=np.corrcoef(Y)
 print('thr',threshold,'keys',len(keys),'corr',np.round(C[:,7],3),'mean8',A[7].mean(),'other mean',np.delete(A,7,0).mean())
# common high bins in >=5 files and counts
for threshold in [100,500,1000]:
 keys=[]
 for k in set().union(*(d.keys() for d in D)):
  n=sum(d.get(k,0)>=threshold for d in D)
  if n>=5:keys.append(k)
 print('common',threshold,len(keys),'file8 median',np.median([D[7].get(k,0) for k in keys]),'others',np.median([D[j].get(k,0) for j in range(10) if j"'!=7 for k in keys]))
PY'
