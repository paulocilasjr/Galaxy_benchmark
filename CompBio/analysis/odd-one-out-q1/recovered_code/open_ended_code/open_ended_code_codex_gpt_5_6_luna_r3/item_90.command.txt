import numpy as np
from pathlib import Path
files=[f'{i}' for i in range(1,11)]
keys={}
for j,b in enumerate(files):
    p=Path(f'bins10_{b}.tsv')
    for line in p.open():
        c,k,n=line.split()
        key=c+':'+k
        if key not in keys: keys[key]=np.zeros(10,dtype=np.float64)
        keys[key][j]=int(n)
X=np.stack(list(keys.values()))
print('shape',X.shape,'totals',X.sum(0).astype(int))
# scores on all bins, with various transforms
for label,Y in [('raw',X),('log',np.log1p(X)),('sqrt',np.sqrt(X))]:
    C=np.corrcoef(Y,rowvar=False)
    print('\\n',label,'mean corr excluding self',np.round((C.sum(0)-1)/9,3),'lowest',np.argmin((C.sum(0)-1)/9)+1)
    # cosine
    Z=Y-Y.mean(0)
    # Pearson same
# top-bin overlap
for frac in [0.001,0.005,0.01,0.05,0.1]:
    n=int(X.shape[0]*frac)
    tops=[]
    for j in range(10): tops.append(set(np.argpartition(X[:,j],-n)[-n:]))
    ov=np.array([[len(tops[i]&tops[j])/len(tops[i]|tops[j]) for j in range(10)] for i in range(10)])
    print('top',frac,'mean',np.round((ov.sum(1)-1)/9,3),'lowest',np.argmin((ov.sum(1)-1)/9)+1)
# correlations after exclude chr11 and alt/unplaced
keep=np.array([not k.startswith('chr11:') and k.startswith(tuple('chr'+str(x)+':' for x in range(1,23))) or k.startswith('chrX:') or k.startswith('chrY:') for k in keys])
# above precedence messy; make proper
kk=list(keys)
keep=np.array([(k.split(':')[0] in {f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}) for k in kk])
for rem in ['no_chr11', 'no_top_0.01_each']:
    Y=X[keep] if rem=='no_chr11' else X.copy()
    if rem=='no_top_0.01_each':
        # remove union of per-file top 1% bins
        bad=set()
        n=int(X.shape[0]*.01)
        for j in range(10): bad.update(np.argpartition(X[:,j],-n)[-n:])
        Y=np.delete(X,list(bad),axis=0)
    C=np.corrcoef(np.log1p(Y),rowvar=False)
    print(rem,'mean corr',np.round((C.sum(0)-1)/9,3),'lowest',np.argmin((C.sum(0)-1)/9)+1)
PY"
