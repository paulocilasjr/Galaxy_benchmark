/bin/bash -lc "gzip -cd gencode.gtf.gz | awk -F '\\t' 'BEGIN{OFS=\"\\t\"} "'$1 ~ /''^(1[0-9]|2[0-2]|[1-9]|X|Y)$/ && $3=="transcript" {t=($7=="+")?$4-1:$5-1; s=t-2000; if(s<0)s=0; print "chr"$1,s,t+2001}'"' | LC_ALL=C sort -k1,1 -k2,2n | bedtools merge -i - > promoters_4kb.bed && gzip -cd gencode.gtf.gz | awk -F '\\t' 'BEGIN{OFS=\"\\t\"} "'$1 ~ /''^(1[0-9]|2[0-2]|[1-9]|X|Y)$/ && $3=="exon" {print "chr"$1,$4-1,$5}'"' | LC_ALL=C sort -k1,1 -k2,2n | bedtools merge -i - > exons_union.bed && wc -l promoters_4kb.bed exons_union.bed"
