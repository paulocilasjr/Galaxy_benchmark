import numpy as np
rows=[]
for l in open('cross8_5bp.tsv'):
 z=l.split(); c=z[0]; a=np.array(list(map(float,z[1:])))

 print(c,'raw peak', (np.argmax(raw[1:])+1)*5, 'raw ratio',raw[1:].max()/raw[0], 'binary peak',(np.argmax(binary[1:])+1)*5,'binary ratio',binary[1:].max()/binary[0], 'd0',raw[0],binary[0])
 rows.append((raw,binary))
for name,idx in [('raw',0),('binary',1)]:
 x=np.sum([r[idx] for r in rows],axis=0); i=np.argmax(x[1:])+1
 print('ALL',name,'peakbp',i*5,'ratio',x[i]/x[0], 'values',','.join(f'{v/x[0]:.3f}' for v in x[:41]))
PY"
