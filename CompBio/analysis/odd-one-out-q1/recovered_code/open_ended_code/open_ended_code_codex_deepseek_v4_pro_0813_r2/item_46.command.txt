/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
url='https://www.encodeproject.org/search/?type=File&file_format=tagAlign&format=json&limit=all'
with urllib.request.urlopen(url,timeout=60) as r: j=json.load(r)
print('total',j.get('total'),'graph',len(j.get('@graph',[])))
open('encode_tagalign_hg19.json','w').write(json.dumps(j))
# inspect keys of one file
f=j['@graph'][0]
print(f.keys())
print({k:f.get(k) for k in ['accession','file_format','assembly','file_size','biological_replicates','technical_replicates','dataset','lab','status','md5sum','read_count','output_type']})
PY"
