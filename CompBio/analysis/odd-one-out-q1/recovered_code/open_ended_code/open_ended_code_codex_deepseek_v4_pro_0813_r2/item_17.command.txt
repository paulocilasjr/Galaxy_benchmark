/bin/bash -lc "cat > cov_profile.sh <<'EOF'
#"'!/bin/bash
f="$1"
name=$(basename "$f")
gzip -cd "$f" | awk '"'NR==FNR{s["'$1]=1;next} ($1 in s){print}'"' hg38.chrom.sizes - | bedtools genomecov -i stdin -g hg38.chrom.sizes -bga | awk -v name=\""'$name" '"'{d="'$4; if(d>0){if(d<=5)b="1-5"; else if(d<=20)b="6-20"; else if(d<=100)b="21-100"; else b="100+"; len=$3-$2; sum[b]+=len; tot+=len}} END{printf "%s",name; for(k in sum) printf " %s=%.0f(%.4f%%)",k,sum[k],100*sum[k]/tot; print " total_covered="tot}'"' 
EOF
chmod +x cov_profile.sh
ls odd.one.out.q1/*.tagAlign.gz | xargs -n1 -P 4 ./cov_profile.sh | sort > cov_profiles.txt
cat cov_profiles.txt"
