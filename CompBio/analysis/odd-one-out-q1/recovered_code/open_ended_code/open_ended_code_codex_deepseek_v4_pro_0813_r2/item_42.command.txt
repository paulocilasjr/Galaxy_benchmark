/bin/bash -lc "python3 - <<'PY'
import urllib.parse, urllib.request, json
url='https://www.encodeproject.org/search/?type=File&file_format=tagAlign&assembly=GRCh38&format=json&limit=10'
with urllib.request.urlopen(url,timeout=30) as r: j=json.load(r)
print('total',j.get('total'))
print('filters',j.get('filters'))
print('graph len',len(j.get('@graph',[])))
for f in j.get('@graph',[])[:10]: print(f.get('accession'), f.get('title'), f.get('file_format'), f.get('assembly'), f.get('file_size'))
PY"
