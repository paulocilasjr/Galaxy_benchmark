/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
# parse bin number from 4th col (name_1), group by bin, sum counts
profiles={}
for n in range(1,11):
    df=pd.read_csv(f'tss_bins_cov_{n}.txt',sep='\\t',header=None,names=['chr','start','end','name','count'])
    df['bin']=df['name'].str.rsplit('_',n=1).str[1].astype(int)
    grp=df.groupby('bin')['count'].sum().sort_index()
    profiles[n]=grp.values/10000000*1e6  # reads per million per bin (sum over TSSs)
# print table bin profiles selected
bins=np.arange(1,81)
print('bin\\t'+'\\t'.join(str(i) for i in range(1,11)))
for b in bins:
    print(b, '\\t'.join(f'{profiles[i][b-1]:.2f}' for i in range(1,11)))
PY"
