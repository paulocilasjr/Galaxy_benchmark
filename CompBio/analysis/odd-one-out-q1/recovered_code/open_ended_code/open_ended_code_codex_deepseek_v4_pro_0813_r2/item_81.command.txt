/bin/bash -lc "python3 - <<'PY'
import pandas as pd
# Find the 1kb window overlapping chr11:24159491 in kb_cov files? windows start 24159000? We can grep around.
for n in range(1,11):
    df=pd.read_csv(f'kb_cov_{n}.txt',sep='\\t',header=None,names=['chr','start','end','count'])
    # window containing position 24159491: start floor/1000 = 24159000
    row=df[(df.chr=='chr11')&(df.start<=24159491)&(df.end>24159491)]
    print(n,row[['start','end','count']].values)
PY"
