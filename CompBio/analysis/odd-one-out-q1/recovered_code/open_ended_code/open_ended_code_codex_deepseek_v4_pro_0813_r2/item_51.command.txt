/bin/bash -lc "python3 - <<'PY'
import gzip, math, os, collections, re, json
files=[f'odd.one.out.q1/{i}.tagAlign.gz' for i in range(1,11)]
chrom_counts=[]
for f in files:
    c=collections.Counter()
    with gzip.open(f,'rt') as fh:
        for line in fh:
            parts=line.split('\\t')
            c[parts[0]]+=1
    chrom_counts.append(c)
for i,c in enumerate(chrom_counts,1):
    total=sum(c.values())
    # entropy canonical
    canon={k:v for k,v in c.items() if k.startswith('chr') and '_' not in k}
    ct=sum(canon.values())
    ent=-sum((v/ct)*math.log(v/ct) for v in canon.values() if v)
    print(i,'chrM',c.get('chrM',0), 'chrX',c.get('chrX',0), 'chrY',c.get('chrY',0), 'entropy',round(ent,4), 'topchrom', c.most_common(1)[0])
PY"
