/bin/bash -lc "python3 - <<'PY'
import urllib.request,json
for url in [
'https://www.encodeproject.org/search/?type=File&file_format=tagAlign&format=json&limit=10',
'https://www.encodeproject.org/search/?type=File&file_format=tagAlign.gz&format=json&limit=10',
'https://www.encodeproject.org/search/?type=File&output_type=alignments&file_format=tagAlign&format=json&limit=10',
'https://www.encodeproject.org/search/?type=File&file_format=bed&format=json&limit=10']:
 print('\\nURL',url)
 try:
  with urllib.request.urlopen(url,timeout=20) as r: j=json.load(r)
  print('total',j.get('total'),'graph',len(j.get('@graph',[])),'filters',j.get('filters'))
  for f in j.get('@graph',[])[:3]: print(f.get('accession'),f.get('file_format'),f.get('title'))
 except Exception as e: print('ERR',e)
PY"
