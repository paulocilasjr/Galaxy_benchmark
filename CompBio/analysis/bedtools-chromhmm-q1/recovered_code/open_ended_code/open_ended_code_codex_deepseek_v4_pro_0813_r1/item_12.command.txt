/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict
path='E055_15_coreMarks_dense.bed.gz'
intervals=defaultdict(list)
with gzip.open(path,'rt') as f:
    for line in f:
        if line.startswith('track'): continue
        c,s,e,*rest=line.rstrip().split('\\t')
        s=int(s); e=int(e)
        intervals[c].append((s,e))
for c in sorted(intervals):
    ints=sorted(intervals[c])
    total=sum(e-s for s,e in ints)
    gaps=0; prev_end=0
    for s,e in ints:
        if s>prev_end: gaps+=s-prev_end
        prev_end=max(prev_end,e)
    print(c,total,'gaps',gaps,'last',prev_end)
PY"
