/bin/bash -lc "python3 - <<'PY'
import gzip
from collections import defaultdict, Counter
cnt=Counter()
rows=[]
with gzip.open('ENCFF333TAT.bed.gz','rt') as f:
    for line in f:
        p=line.rstrip().split('\\t')
        cnt[(p[0],int(p[1]),int(p[2]))]+=1
        rows.append((p[0],int(p[1]),int(p[2]),p[3]))
print('total lines',len(rows))
print('unique exact coords',len(cnt))
print('max duplicates',cnt.most_common(5))
print('dup>1 coords',sum(1 for v in cnt.values() if v>1),'sum extra',sum(v-1 for v in cnt.values() if v>1))
# sample duplicated
for k,v in cnt.items():
    if v>1:
        print(k,v); break
PY"
