import pysam
b=pysam.AlignmentFile('variant.status.q1.bam','rb')
pos0=154398499
for r in b.fetch(until_eof=True):
 if r.reference_name=='chrX' and r.reference_start<=pos0<r.reference_end and r.is_reverse:
  for q,ref in r.get_aligned_pairs(matches_only=False):
   if ref==pos0:
    print('name',r.query_name,'flag',r.flag,'qpos',q)
    print('query_sequence',r.query_sequence[q],r.query_sequence[:15])
    print('forward_sequence',r.get_forward_sequence()[len(r.query_sequence)-1-q],r.get_forward_sequence()[:15])
    print('seq len',len(r.query_sequence))
    break
PY"
