/bin/bash -lc "python3 - <<'PY'
import pysam, collections
b=pysam.AlignmentFile('variant.status.q1.bam','rb')
counts=collections.Counter(); bases=collections.Counter(); pos0=154398499
for r in b.fetch(until_eof=True):
 counts[r.reference_name]+=1
 if r.reference_name=='chrX' and r.reference_start<=pos0<r.reference_end:
  for q,p in r.get_aligned_pairs(matches_only=False):
   if p==pos0:
    bases[r.query_sequence[q] if q is not None else 'DEL']+=1
print('records chrX chrY chrM',counts['chrX'],counts['chrY'],counts['chrM'])
print('target base counts',dict(bases))
PY"
