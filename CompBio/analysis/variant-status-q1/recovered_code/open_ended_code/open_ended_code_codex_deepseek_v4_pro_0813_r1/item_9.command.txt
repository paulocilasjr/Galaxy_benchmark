/bin/bash -c "python - <<'PY'
import pysam
bam='variant.status.q1.bam'
pos=154398500
af=pysam.AlignmentFile(bam,'rb')
counts={}
rows=[]
for r in af.fetch('chrX', pos-1, pos):
    seq=r.query_sequence  # SEQ as stored; orientation-corrected? pysam gives query_sequence as original SEQ field, but alignment sequence property is query_alignment_sequence
    pairs=r.get_aligned_pairs(matches_only=False)
    qpos = None
    for qp,rp in pairs:
        if rp==pos-1:
            qpos=qp
            break
    if qpos is None:
        # maybe deletion at position
        base=None
    else:
        base=seq[qpos]
    # orientation correction: if reverse flag, stored SEQ is reverse complemented relative to reference? query_alignment_sequence is oriented to reference
    counts[base.upper()]=counts.get(base.upper(),0)+1
    rows.append((r.query_name, r.flag, r.reference_start+1, r.cigarstring, qpos, pos, base, r.mapping_quality, r.query_qualities[qpos] if qpos is not None else None, r.query_alignment_sequence[qpos] if qpos is not None else None))
print('counts stored seq', counts)
print('total rows', len(rows))
for row in rows:
    print('\\t'.join(map(str,row)))
af.close()
PY"
