/bin/bash -lc "samtools view variant.status.q1.bam chrX:154398500-154398500 | perl -lane '
"'$t=154398500; ($qname,$flag,$rname,$pos,$mapq,$cigar,undef,undef,undef,$seq,$qual)=@F;
$ref=$pos; $read=0; $base="NA"; $bq="NA";
while ($cigar =~ /('"\\d+)([MIDNSHP=X])/g) {
  ("'$n,$op)=($1,$2);
  if ($op =~ /[M=X]/) {
    if ($t >= $ref && $t < $ref+$n) { $idx=$read+($t-$ref); $base=substr($seq,$idx,1); $bq=ord(substr($qual,$idx,1))-33; last; }
    $ref += $n; $read += $n;
  } elsif ($op =~ /[DN]/) { $ref += $n; }
  elsif ($op =~ /[IS]/) { $read += $n; }
}
$strand=($flag & 16) ? "-" : "+"; $dup=($flag & 1024) ? "dup" : "nodup"; $sec=($flag & 256) ? "sec" : "pri"; $supp=($flag & 2048) ? "supp" : "main";
print join("'"\\t\","'$qname,$flag,$pos,$cigar,$base,$strand,$dup,$sec,$supp,$mapq,$bq);
'"' | sort | tee target_read_bases.tsv | perl -lane '"'$k=join(":",@F[4..8]); $c{$k}++; END{for $k (sort keys %c){print "$k'"\\t"'$c{$k}"}}'"'"
