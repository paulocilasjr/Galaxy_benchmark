import subprocess,re
pos=154398500
ops_re=re.compile(r'(\\d+)([MIDNSHP=X])')
for min_qpos in [0,1,2,3,5,10]:
    counts={}
    out=subprocess.check_output(['samtools','view','variant.status.q1.bam','chrX:154398500-154398500'], text=True)
    for line in out.strip().splitlines():
        f=line.split('\\t'); start=int(f[3]); cigar=f[5]; seq=f[9]; qual=f[10]; ref=start; qi=0
        base=None; qpos=None
        for n,op in ops_re.findall(cigar):
            n=int(n)
            if op in 'M=X':
                if ref <= pos < ref+n:
                    qpos=qi+pos-ref; base=seq[qpos]; break
                ref+=n; qi+=n
            elif op in 'DN': ref+=n
            elif op in 'IS': qi+=n
        if base and qpos>=min_qpos:
            counts[base]=counts.get(base,0)+1
    print('min_qpos',min_qpos,counts)
PY"
