/bin/bash -lc "python3 - <<'PY'
import pysam
from collections import Counter
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
posA=154398499; posB=154398500
c=Counter(); rows=[]
for read in bam.fetch('chrX',154398300,154398600):
    pairs=read.get_aligned_pairs(matches_only=False)
    qa=qb=None
    for q,r in pairs:
        if r==posA-1: qa=q
        if r==posB-1: qb=q
    if qa is not None and qb is not None:
        ba=read.query_sequence[qa]; bb=read.query_sequence[qb]
        c[(ba,bb)]+=1
        rows.append((read.query_name,read.reference_start+1,read.cigarstring,qa,qb,ba,bb))
print('haplotypes pos499-500:',c)
for r in rows:
    print(r)
PY"
