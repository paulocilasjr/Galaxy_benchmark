/bin/bash -lc "python3 - <<'PY'
import urllib.request, json
start=154398450; end=154398580
url=f'https://api.genome.ucsc.edu/getData/sequence?genome=hg38;chrom=chrX;start={start-1};end={end}'
data=json.load(urllib.request.urlopen(url, timeout=20))
seq=data['dna']
print('positions', start, 'to', end, 'len', len(seq))
for i,b in enumerate(seq):
    print(f'{start+i}\\t{b}\\t{i}')
PY"
