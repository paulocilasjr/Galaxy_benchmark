/bin/bash -lc "python3 - <<'PY'
import pysam, collections
bam=pysam.AlignmentFile('variant.status.q1.bam','rb')
pos=154398500
for min_dist in [0,1,2,3,5,10]:
 c=collections.Counter(); detail=[]
 for read in bam.fetch('chrX', pos-500, pos+500):
  pairs=read.get_aligned_pairs(matches_only=False)
  qpos=None
  for q,r in pairs:
   if r==pos-1:
    qpos=q; break
  if qpos is None: continue
  # distance from query ends
  left=qpos
  right=read.query_length-1-qpos
  if left>=min_dist and right>=min_dist:
   b=read.query_sequence[qpos]
   c[b]+=1
   detail.append((read.query_name,read.reference_start+1,qpos,b,left,right,read.cigarstring))
 print('min_dist',min_dist,'counts',c)
 if min_dist in [2,5]:
  for d in detail[:50]: print(' ',d)
PY"
