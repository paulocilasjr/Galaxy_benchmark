/bin/bash -lc "python3 - <<'PY'
import pysam
b=pysam.AlignmentFile('variant.status.q1.bam','rb')
for pos in [154398498,154398499,154398500,154398501]:
    counts={}; qs={}
    for pc in b.pileup('chrX',pos-1,pos,truncate=True,stepper='nofilter'):
        if pc.reference_pos==pos-1:
            for pr in pc.pileups:
                if pr.is_del or pr.is_refskip: continue
                q=pr.alignment.query_sequence[pr.query_position]
                counts[q]=counts.get(q,0)+1
                qs.setdefault(q,[]).append(pr.alignment.query_qualities[pr.query_position] if pr.alignment.query_qualities else None)
    print(pos, counts, {k:(len(v),min(v),max(v)) for k,v in qs.items()})
PY"
