/bin/bash -lc "python3 - <<'PY'
import pysam
fn='variant.status.q1.bam'
b=pysam.AlignmentFile(fn,'rb')
pos=154398500
for pc in b.pileup('chrX', pos-1, pos, truncate=True, stepper='nofilter'):
    if pc.reference_pos==pos-1:
        print('n', pc.nsegments)
        for pr in pc.pileups:
            a=pr.alignment
            if pr.is_del or pr.is_refskip:
                print('skip/del', a.query_name)
                continue
            qp=pr.query_position
            base=a.query_sequence[qp]
            qual=a.query_qualities[qp] if a.query_qualities else None
            print(a.query_name, base, qual, 'mapq', a.mapping_quality, 'flag', a.flag, 'cigar', a.cigarstring, 'start', a.reference_start+1, 'qpos',qp, 'NM', a.get_tag('nM') if a.has_tag('nM') else '')
PY"
