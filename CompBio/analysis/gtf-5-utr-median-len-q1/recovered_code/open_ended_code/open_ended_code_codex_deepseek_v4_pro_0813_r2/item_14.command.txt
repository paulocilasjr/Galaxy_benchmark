/bin/bash -lc "python3 - <<'PY'
import gzip,re
from collections import defaultdict
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
attr_re=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
cds=defaultdict(list); start=defaultdict(list)
with gzip.open(fn,'rt') as f:
  for line in f:
    if line.startswith('#'): continue
    p=line.rstrip().split('\\t')
    if len(p)<9: continue
    a=dict(attr_re.findall(p[8])); tx=a.get('transcript_id','')
    if p[2]=='CDS': cds[tx].append((int(p[3]),int(p[4])))
    elif p[2]=='start_codon': start[tx].append((int(p[3]),int(p[4])))
bad=[]
for tx,starts in start.items():
  for s,e in starts:
    if not any(max(s,cs)<=min(e,ce) for cs,ce in cds.get(tx,[])):
      bad.append((tx,s,e))
print('start features',sum(map(len,start.values())),'non-overlap',len(bad))
print(bad[:5])
PY"
