/bin/bash -lc "python3 - <<'PY'
import gzip,re
from collections import defaultdict
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
attr_re=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
cds=defaultdict(list); stop=defaultdict(list)
with gzip.open(fn,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip().split('\\t')
        if len(p)<9: continue
        attrs=dict(attr_re.findall(p[8]))
        tx=attrs.get('transcript_id','')
        if p[2]=='CDS': cds[tx].append((int(p[3]),int(p[4])))
        elif p[2]=='stop_codon': stop[tx].append((int(p[3]),int(p[4])))
count_overlap=0; count_adjacent=0; double=0
for tx, stops in stop.items():
    for s,e in stops:
        for cs,ce in cds.get(tx,[]):
            if max(s,cs)<=min(e,ce):
                count_overlap+=1; double+=max(0,min(e,ce)-max(s,cs)+1)
            if ce+1==s or e+1==cs:
                count_adjacent+=1
print('stop features',sum(map(len,stop.values())),'overlap with cds',count_overlap,'overlap_bases',double,'adjacent',count_adjacent)
PY"
