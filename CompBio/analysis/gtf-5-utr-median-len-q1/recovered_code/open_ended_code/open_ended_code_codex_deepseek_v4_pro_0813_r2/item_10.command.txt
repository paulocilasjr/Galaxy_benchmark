/bin/bash -lc "python3 - <<'PY'
import gzip,re
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
STANDARD={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}
attr_re=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
sel=set(); chroms=set(); all_selected_chroms=set()
with gzip.open(fn,'rt') as f:
  for line in f:
    if line.startswith('#'): continue
    p=line.rstrip().split('\\t')
    if len(p)<9: continue
    a=dict(attr_re.findall(p[8])); tx=a.get('transcript_id','')
    if not tx: continue
    if p[2]=='transcript':
      if p[0] in STANDARD and 'MANE Select' in a.get('tag',''):
        sel.add(tx)
# now check features
with gzip.open(fn,'rt') as f:
  for line in f:
    if line.startswith('#'): continue
    p=line.rstrip().split('\\t')
    if len(p)<9: continue
    a=dict(attr_re.findall(p[8])); tx=a.get('transcript_id','')
    if tx in sel and p[2] in ('exon','CDS','stop_codon'):
      all_selected_chroms.add(p[0])
      if p[0] not in STANDARD: chroms.add(p[0])
print('selected tx',len(sel),'nonstandard feature chroms',chroms)
print('all feature chroms count',len(all_selected_chroms))
PY"
