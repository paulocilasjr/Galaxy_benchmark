/bin/bash -lc "python3 - <<'PY'
import gzip, re, math
from collections import defaultdict, Counter
import numpy as np
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std={*(f'chr{i}' for i in range(1,23)),'chrX','chrY','chrM'}
rx=re.compile(r'(\\S+) \"(["'^"]*)";'"')
def parse(s):
    d=defaultdict(list)
    for k,v in rx.findall(s): d[k].append(v)
    return d
def union_len(iv):
    if not iv: return 0
    iv=sorted(iv)
    total=0; cs,ce=iv[0]
    for s,e in iv[1:]:
        if s<=ce+1:
            if e>ce: ce=e
        else:
            total += ce-cs+1; cs,ce=s,e
    return total+ce-cs+1
sel={}; f=defaultdict(lambda: defaultdict(list)); tagsets=Counter(); chrom_sel=Counter()
with gzip.open(path,'rt') as h:
    for line in h:
        if line.startswith('#'): continue
        c=line.rstrip('\\n').split('\\t')
        if len(c)"'!=9: continue
        chrom,_,feat,start,end,_,strand,_,as_=c
        a=parse(as_); tid=a.get('"'transcript_id',[''])[0]; tags=tuple(a.get('tag',[]))
        if feat=='transcript':
            tagsets[tags]+=1
            if 'MANE Select' in tags:
                chrom_sel[chrom]+=1
                if chrom in std: sel[tid]=(chrom,strand)
        if chrom in std and tid and feat in {'exon','CDS','stop_codon'}:
            f[tid][feat].append((int(start),int(end)))
vals=[[],[],[]]
cds_overlap=[]
for tid,(chrom,strand) in sel.items():
    coding=f[tid]['CDS']+f[tid]['stop_codon']
    cds_sum=sum(e-s+1 for s,e in coding)
    cds_union=union_len(coding)
    if cds_sum"'!=cds_union: cds_overlap.append((tid,cds_sum,cds_union))
    lo=min(s for s,e in coding); hi=max(e for s,e in coding)
    before=sum(max(0,min(e,lo-1)-s+1) for s,e in f[tid]['"'exon'] if s<lo)
    after=sum(max(0,e-max(s,hi+1)+1) for s,e in f[tid]['exon'] if e>hi)
    vals[0].append(before if strand=='+' else after)
    vals[1].append(cds_union)
    vals[2].append(after if strand=='+' else before)
print('transcript tagsets top', tagsets.most_common(5))
print('select standard', len(sel), 'select chrom categories', len(chrom_sel), 'nonstandard', sum(n for ch,n in chrom_sel.items() if ch not in std))
print('cds overlaps', cds_overlap[:3], 'count', len(cds_overlap))
print('medians', [np.median(x) for x in vals])
print(f'{math.floor(np.median(vals[0]))};{math.floor(np.median(vals[1]))};{math.floor(np.median(vals[2]))}')
PY"
