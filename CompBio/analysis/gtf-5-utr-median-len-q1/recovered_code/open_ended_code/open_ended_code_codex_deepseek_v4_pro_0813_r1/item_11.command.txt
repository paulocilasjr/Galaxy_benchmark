/bin/bash -lc "python3 - <<'PY'
# minimal validation using functions imported from solve? rerun a few manually
import gzip,re
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'; attr_re=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
want='NM_001005484.2'
ex=[];cd=[];st=[];strand=None
with gzip.open(path,'rt') as f:
 for line in f:
  if line.startswith('#'):continue
  p=line.rstrip().split('\\t')
  if len(p)"'!=9:continue
  a={k:v for k,v in attr_re.findall(p[8])}
  if a.get('"'transcript_id')"'!=want:continue
  if p[2]=='"'transcript': strand=p[6]
  elif p[2]=='exon': ex.append((int(p[3]),int(p[4])))
  elif p[2]=='CDS': cd.append((int(p[3]),int(p[4])))
  elif p[2]=='stop_codon': st.append((int(p[3]),int(p[4])))
def merge(x):
 x=sorted(x); out=[list(x[0])]
 for s,e in x[1:]:
  if s<=out[-1][1]+1: out[-1][1]=max(out[-1][1],e)
  else: out.append([s,e])
 return out
def il(x): return sum(e-s+1 for s,e in x)
def inter(x,lo,hi): return sum(max(0,min(e,hi)-max(s,lo)+1) for s,e in x)
exm=merge(ex); cod=merge(cd+st)
print(strand, exm, cod)
print('exon len',il(exm),'cds',il(cod))
print('five/three', inter(exm,exm[0][0],cod[0][0]-1) if strand=='+' else inter(exm,cod[-1][1]+1,exm[-1][1]), inter(exm,cod[-1][1]+1,exm[-1][1]) if strand=='+' else inter(exm,exm[0][0],cod[0][0]-1))
PY"
