/bin/bash -lc "python3 - <<'PY'
import gzip, re, numpy as np
# quick rerun with validation by parsing all selected features from transcript line onwards; compute per tx exon/cds/stop, check total
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std=re.compile(r'"'^chr(?:[1-9]|1[0-9]|2[0-2]|X|Y)$'"')
attr_re=re.compile(r'(\\S+)\\s+\"(["'^"]*)"'"')
def pa(s): return {k:v for k,v in attr_re.findall(s)}
sel=set(); ex={}; cd={}; st={}; strand={}
with gzip.open(path,'rt') as f:
  for line in f:
    if line.startswith('#'): continue
    p=line.rstrip().split('\\t')
    if len(p)"'!=9: continue
    if not std.match(p[0]): continue
    a=pa(p[8]); tid=a.get('"'transcript_id','')
    if not tid: continue
    feat=p[2]
    if feat=='transcript':
      if a.get('tag','')=='MANE Select':
        sel.add(tid); strand[tid]=p[6]
      continue
    if tid not in sel: continue
    s,e=int(p[3]),int(p[4])
    (ex if feat=='exon' else cd if feat=='CDS' else st if feat=='stop_codon' else {}).setdefault(tid,[]).append((s,e))
# note this .setdefault weird but works? Actually dict.setdefault returns list and appends? let's not.
PY"
