/bin/bash -lc "python3 - <<'PY'
import gzip, re, math
from collections import defaultdict
import numpy as np

path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std={f'chr{i}' for i in range(1,23)}|{'chrX','chrY','chrM'}
attr_re=re.compile(r'(\\S+) \"(["'^"]*)";'"')

def attrs(s):
    d=defaultdict(list)
    for k,v in attr_re.findall(s):
        d[k].append(v)
    return d

selected={}
data=defaultdict(lambda: {'exon':[], 'cds':[], 'stop':[], 'chrom':None, 'strand':None, 'gene':None})
all_tx=plus=select=select_std=0
with gzip.open(path,'rt') as fh:
    for line in fh:
        if not line.strip() or line.startswith('#'):
            continue
        fields=line.rstrip('\\n').split('\\t')
        if len(fields)<9:
            continue
        chrom, source, feature, start, end, score, strand, frame, attr=fields
        a=attrs(attr)
        tid=a.get('transcript_id',[''])[0]
        tags=a.get('tag',[])
        if feature=='transcript':
            all_tx+=1
            if 'MANE Plus Clinical' in tags: plus+=1
            if 'MANE Select' in tags:
                select+=1
                if chrom in std:
                    select_std+=1
                    selected[tid]=(chrom,strand,a.get('gene_id',[''])[0])
        if tid:
            rec=data[tid]
            rec['chrom']=chrom; rec['strand']=strand; rec['gene']=a.get('gene_id',[''])[0]
            if feature=='exon': rec['exon'].append((int(start),int(end)))
            elif feature=='CDS': rec['cds'].append((int(start),int(end)))
            elif feature=='stop_codon': rec['stop'].append((int(start),int(end)))

five=[]; cds_lengths=[]; three=[]
missing=[]
examples=[]
for tid,(chrom,strand,gene) in selected.items():
    rec=data[tid]
    exons=rec['exon']
    coding=rec['cds']+rec['stop']
    if not exons or not coding:
        missing.append(tid); continue
    c_start=min(s for s,e in coding); c_end=max(e for s,e in coding)
    # CDS length as genomic union of CDS and stop_codon intervals to avoid any accidental overlap double count.
    merged=[]
    for s,e in sorted(coding):
        if not merged or s>merged[-1][1]+1:
            merged.append([s,e])
        else:
            if e>merged[-1][1]: merged[-1][1]=e
    cds_len=sum(e-s+1 for s,e in merged)
    def ov_len(interval, lo, hi):
        s,e=interval
        a=max(s,lo); b=min(e,hi)
        return max(0,b-a+1)
    if strand=='+':
        f=sum(ov_len(ex, -10**18, c_start-1) for ex in exons)
        t=sum(ov_len(ex, c_end+1, 10**18) for ex in exons)
    elif strand=='-':
        f=sum(ov_len(ex, c_end+1, 10**18) for ex in exons)
        t=sum(ov_len(ex, -10**18, c_start-1) for ex in exons)
    else:
        raise ValueError((tid,strand))
    five.append(f); cds_lengths.append(cds_len); three.append(t)
    if len(examples)<3:
        examples.append((tid,gene,strand,f,cds_len,t))

print('all_tx',all_tx,'select',select,'plus',plus,'select_std',select_std,'used',len(five),'missing',len(missing))
print('examples',examples)
print('median_raw',np.median(five),np.median(cds_lengths),np.median(three))
print('answer',f'{math.floor(np.median(five))};{math.floor(np.median(cds_lengths))};{math.floor(np.median(three))}')
print('zero counts',sum(x==0 for x in five),sum(x==0 for x in cds_lengths),sum(x==0 for x in three))
PY"
