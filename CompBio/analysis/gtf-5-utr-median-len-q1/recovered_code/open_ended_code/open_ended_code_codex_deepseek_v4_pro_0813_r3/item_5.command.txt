import gzip,re,collections
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'

with gzip.open(fn,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        chrom=p[0]; typ=p[2]; attrs=p[8]
        chroms.add(chrom)
        if typ"'!='"'transcript': continue
        if 'tag \"MANE Select\"' not in attrs: continue
        cnt[chrom]+=1
        if re.match(r'chr(?:[1-9]|1\\d|2[0-2]|[XY])"'$'"',chrom): cntstd[chrom]+=1
print('all contigs', sorted(chroms))
print('counts all select', cnt)
print('sum', sum(cnt.values()))
print('std sum', sum(cntstd.values()))
PY"
