/bin/bash -lc "python3 - <<'PY'
import numpy as np, gzip, re, collections
fn='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
std=set([f'chr{i}' for i in list(range(1,23))+['X','Y']])
raw=collections.defaultdict(int)
with gzip.open(fn,'rt') as f:
 for line in f:
  if line.startswith('#'): continue
  p=line.rstrip('\\n').split('\\t')
  if len(p)<9: continue
  chrom=p[0]; typ=p[2]; start=int(p[3]); end=int(p[4]); attrs=p[8]
  if chrom not in std: continue
  m=re.search(r'transcript_id \"(["'^"]+)"'"', attrs)
  if not m: continue
  tid=m.group(1)
  if 'tag \"MANE Select\"' not in attrs: continue
  if typ in ('CDS','stop_codon'):
   raw[tid]+=end-start+1
print(len(raw), sorted(raw.items())[:2])
print('median raw cds', int(np.floor(np.median(list(raw.values())))))
PY"
