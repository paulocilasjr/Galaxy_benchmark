/bin/bash -lc "python3 - <<'PY'
import gzip, re, collections, math
import numpy as np
path='MANE.GRCh38.v1.3.refseq_genomic.gtf.gz'
standard={f'chr{i}' for i in range(1,23)} | {'chrX','chrY','chrM','chrMT','MT','M'}
attr_re=re.compile(r'(\\S+) \"(["'^"]*)";'"')

def attrs(s):
    d=collections.defaultdict(list)
    for k,v in attr_re.findall(s): d[k].append(v)
    return d
sel={}
features=collections.defaultdict(lambda: collections.defaultdict(list))
with gzip.open(path,'rt') as f:
    for line in f:
        if line.startswith('#'): continue
        p=line.rstrip('\\n').split('\\t')
        if len(p)<9: continue
        chrom,src,feat,start,end,score,strand,frame,attr = p
        if chrom not in standard: continue
        a=attrs(attr)
        tid=a.get('transcript_id',[''])[0]
        if feat=='transcript' and tid and 'MANE Select' in a.get('tag',[]):
            key=(tid,chrom,strand)
            sel[key]=1
        if feat in {'exon','CDS','stop_codon'} and tid:
            features[(tid,chrom,strand)][feat].append((int(start),int(end)))

def merge(xs):
    xs=sorted(xs)
    out=[]
    for s,e in xs:
        if not out or s>out[-1][1]+1: out.append([s,e])
        else: out[-1][1]=max(out[-1][1],e)
    return [(s,e) for s,e in out]

def ilen(xs): return sum(e-s+1 for s,e in xs)

def intersect(a,b):
    s=max(a[0],b[0]); e=min(a[1],b[1])
    return (s,e) if s<=e else None

m1=[]; m2=[]; gap_count=0
for key in sorted(sel):
    tid,chrom,strand=key
    ex=features[key]['exon']; code=merge(features[key]['CDS']+features[key]['stop_codon'])
    assert ex and code, key
    # method 1, genomic side partition
    cmin=min(s for s,e in code); cmax=max(e for s,e in code)
    a5=a3=0
    for s,e in ex:
        if strand=='+':
            if s<cmin: a5 += max(0,min(e,cmin-1)-s+1)
            if e>cmax: a3 += max(0,e-max(s,cmax+1)+1)
        else:
            if e>cmax: a5 += max(0,e-max(s,cmax+1)+1)
            if s<cmin: a3 += max(0,min(e,cmin-1)-s+1)
    clen=ilen(code)
    m1.append((a5,clen,a3))
    # method 2, transcript coordinates
    ordered=sorted(ex, reverse=(strand=='-'))
    pos=1; mapped=[]; tx_len=0
    for es,ee in ordered:
        el=ee-es+1
        tx_len += el
        for cs,ce in code:
            ov=intersect((es,ee),(cs,ce))
            if not ov: continue
            os,oe=ov
            if strand=='+':
                ts=pos+(os-es); te=pos+(oe-es)
            else:
                # high genomic coordinate maps to lower transcript coordinate
                ts=pos+(ee-oe); te=pos+(ee-os)
            mapped.append((ts,te))
        pos += el
    mapped=merge(mapped)
    b5=min(s for s,e in mapped)-1
    b3=tx_len-max(e for s,e in mapped)
    bclen=ilen(mapped)
    if max(e for s,e in mapped)-min(s for s,e in mapped)+1 "'!= bclen:
        gap_count += 1
    m2.append((b5,bclen,b3))
    if m1[-1]!=m2[-1]:
        print('"'mismatch',key,m1[-1],m2[-1],ex,code,mapped); raise SystemExit(1)
print('checked',len(m1),'all match','coding gaps in tx coords',gap_count)
for i,name in enumerate(['utr5','cds','utr3']):
    vals=[x[i] for x in m2]
    print(name, np.median(vals), math.floor(np.median(vals)))
print(f\"{math.floor(np.median([x[0] for x in m2]))};{math.floor(np.median([x[1] for x in m2]))};{math.floor(np.median([x[2] for x in m2]))}\")
PY"
