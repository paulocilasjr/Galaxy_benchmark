import gzip,re,numpy as np
from collections import defaultdict
std={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}
def A(s):return dict(re.findall(r'(\\S+) \"(["'^"]*)"'"',s))

 if not iv:return 0
 q=sorted(iv);a,b=q[0];n=0
 for c,d in q[1:]:
  if c<=b+1:b=max(b,d)
  else:n+=b-a+1;a,b=c,d
 return n+b-a+1
records=[];selected=set()
with gzip.open('MANE.GRCh38.v1.3.refseq_genomic.gtf.gz','rt') as f:

 D[t]['strand']=p[6]

for t,d in D.items():

   if a<lo:fiv.append((a,min(b,lo-1)))

  else:

a=np.array(vals)
