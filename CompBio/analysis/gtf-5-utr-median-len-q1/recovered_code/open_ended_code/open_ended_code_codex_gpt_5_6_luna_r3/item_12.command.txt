/bin/bash -lc "python - <<'PY'
# Check all stop codons are included in the CDS lengths and not UTRs under the chosen interpretation.
import gzip,re
from collections import defaultdict,Counter
std={f'chr{i}' for i in range(1,23)}|{'chrX','chrY'}
def A(s):
 d=defaultdict(list)
 for k,v in re.findall(r'(\\S+) \"(["'^"]*)"'"',s): d[k].append(v)
 return d
def U(iv):
 if not iv:return 0
 q=sorted(iv); a,b=q[0]; n=0
 for c,d in q[1:]:
  if c<=b+1:b=max(b,d)
  else:n+=b-a+1;a,b=c,d
 return n+b-a+1
D=defaultdict(lambda:defaultdict(list))
with gzip.open('MANE.GRCh38.v1.3.refseq_genomic.gtf.gz','rt') as f:
 for l in f:
  if l.startswith('#'):continue
  p=l.rstrip().split('\\t');a=A(p[8])
  if p[0] in std and 'MANE Select' in a.get('tag',[]): D[a['transcript_id'][0]][p[2]].append((int(p[3]),int(p[4])))
print('transcripts',len(D),'stop length counts',Counter(U(d['stop_codon']) for d in D.values()))
print('CDS separate vs combined differences',sum(U(d['CDS'])+U(d['stop_codon'])"'!=U(d['"'CDS']+d['stop_codon']) for d in D.values()))
PY"
