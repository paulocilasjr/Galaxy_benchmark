/bin/bash -lc "python -u - <<'PY'
import anndata as ad
import numpy as np

adata = ad.read_h5ad('pbmc3k.h5ad')
labels = adata.obs['louvain'].astype(str).to_numpy()
idx = np.flatnonzero(labels == 'B cells')
gidx = adata.var_names.get_loc('S100B')
vals = adata.X[idx, gidx]
if hasattr(vals, 'toarray'):
    vals = vals.toarray().ravel()
else:
    vals = np.asarray(vals).ravel()
mean = float(vals.mean())
print('shape:', adata.shape)
print('louvain counts:', adata.obs['louvain'].value_counts().to_dict())
print('S100B column:', gidx)
print('B cells:', len(idx))
print('sum:', float(vals.sum()))
print('mean:', repr(mean))
print('formatted:', format(mean, '.2f'))
PY"
