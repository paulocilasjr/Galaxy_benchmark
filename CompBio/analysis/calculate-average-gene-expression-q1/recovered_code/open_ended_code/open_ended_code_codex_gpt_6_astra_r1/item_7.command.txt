/bin/bash -lc "python - <<'PY'
import h5py
import numpy as np
with h5py.File('pbmc3k.h5ad','r') as f:
 categories = f['uns/louvain_categories'][:]
 print('categories:', categories)
 category = np.flatnonzero(categories == b'B cells')
 gene = np.flatnonzero(f['var']['index'] == b'S100B')
 assert len(category) == len(gene) == 1
 mask = f['obs']['louvain'] == category[0]
 values = f['X'][:, int(gene[0])][mask]
 print('cells:',len(values))
 print('mean:',repr(values.mean()))
 print(f'{values.mean():.2f}')
PY"
