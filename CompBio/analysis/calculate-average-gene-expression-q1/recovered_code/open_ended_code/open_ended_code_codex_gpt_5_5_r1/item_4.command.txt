/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
with h5py.File('pbmc3k.h5ad','r') as f:
    var_index = f['var']['index'][:].astype(str)
    matches = np.where(var_index == 'S100B')[0]
    print('matches', matches)
    cats = f['uns/louvain_categories'][:].astype(str)
    print('cats', list(enumerate(cats)))
    b_code = int(np.where(cats == 'B cells')[0][0])
    obs_louvain = f['obs']['louvain'][:]
    cells = np.where(obs_louvain == b_code)[0]
    print('b_code', b_code, 'n_cells', len(cells))
    vals = f['X'][cells, matches[0]]
    print(vals.shape, vals.dtype, vals[:10])
    print('mean', np.mean(vals), 'rounded', f'{np.mean(vals):.2f}')
PY"
