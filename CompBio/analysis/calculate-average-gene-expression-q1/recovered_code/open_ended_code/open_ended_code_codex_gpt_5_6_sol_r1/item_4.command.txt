/bin/bash -lc "python - <<'PY'
import anndata as ad
import numpy as np
from scipy import sparse
adata = ad.read_h5ad('pbmc3k.h5ad')
print('shape', adata.shape)
print('louvain labels', adata.obs['louvain'].value_counts(dropna=False).to_dict())
print('S100B matches', int(np.sum(adata.var_names == 'S100B')))
mask = np.asarray(adata.obs['louvain'] == 'B cells')
gene_idx = int(np.flatnonzero(np.asarray(adata.var_names == 'S100B'))[0])
values = adata.X[mask, gene_idx]
if sparse.issparse(values):
    mean = float(values.mean())
    arr = values.toarray().ravel()
else:
    arr = np.asarray(values).ravel()
    mean = float(arr.mean())
print('n_cells', int(mask.sum()))
print('sum', format(float(arr.sum()), '.12f'))
print('mean', repr(mean))
print('rounded', f'{mean:.2f}')
PY"
