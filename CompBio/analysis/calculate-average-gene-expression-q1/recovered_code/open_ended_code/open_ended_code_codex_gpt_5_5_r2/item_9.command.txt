/bin/bash -lc "python - <<'PY'
import h5py
import numpy as np
with h5py.File('pbmc3k.h5ad','r') as f:
    var_names = np.array([x.decode() if isinstance(x, bytes) else str(x) for x in f['var']['index'][()]])
    cats = np.array([x.decode() if isinstance(x, bytes) else str(x) for x in f['uns']['louvain_categories'][()]])
    obs = f['obs'][()]
    louvain_codes = obs['louvain']
    gene_idx = int(np.where(var_names == 'S100B')[0][0])
    b_code = int(np.where(cats == 'B cells')[0][0])
    mask = louvain_codes == b_code
    vals = f['X'][mask, gene_idx]
    print('categories:', cats.tolist())
    print('S100B idx:', gene_idx)
    print('B cells code/count:', b_code, int(mask.sum()))
    print('mean:', float(vals.mean()))
    print(f'{float(vals.mean()):.2f}')
PY"
