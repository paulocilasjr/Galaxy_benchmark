/bin/bash -lc "python - <<'PY'
import h5py
import numpy as np
with h5py.File('pbmc3k.h5ad','r') as f:
    cats = [x.decode('utf-8') for x in f['uns/louvain_categories'][:]]
    print('categories:', cats)
    b_code = cats.index('B cells')
    obs = f['obs'][:]
    mask = obs['louvain'] == b_code
    print('B cells count:', int(mask.sum()), 'code', b_code)
    var_names = [x.decode('utf-8') for x in f['var'][:]['index']]
    gene_idx = var_names.index('S100B')
    print('S100B index:', gene_idx)
    vals = f['X'][:, gene_idx][mask]
    mean = float(np.mean(vals))
    print('mean raw:', repr(mean))
    print('answer:', f'{mean:.2f}')
PY"
