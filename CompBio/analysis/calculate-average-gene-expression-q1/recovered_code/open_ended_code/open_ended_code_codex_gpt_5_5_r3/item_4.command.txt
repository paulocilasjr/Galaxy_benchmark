/bin/bash -lc "python - <<'PY'
import h5py
print('h5py', h5py.__version__)
with h5py.File('pbmc3k.h5ad','r') as f:
    def show(name, obj):
        if name.count('/') <= 2:
            kind = type(obj).__name__
            shape = getattr(obj, 'shape', '')
            dtype = getattr(obj, 'dtype', '')
            print(name, kind, shape, dtype)
    f.visititems(show)
PY"
