/bin/bash -lc "python3 - <<'PY'
import gzip, re
transcript_id='ENST00000269305'
with gzip.open('Homo_sapiens.GRCh38.115.gtf.gz','rt') as f:
    for line in f:
        if line.startswith('#'):
            continue
        fields=line.rstrip('\\n').split('\\t')
        if len(fields) < 9 or fields[2] "'!= '"'transcript':
            continue
        attrs=dict(re.findall(r'(\\S+) \"(["'^"]+)";'"', fields[8]))
        if attrs.get('transcript_id') == transcript_id:
            chrom,start,end,strand=fields[0], int(fields[3]), int(fields[4]), fields[6]
            tss = start if strand == '+' else end
            up, down = 500, 100
            if strand == '+':
                p_start, p_end = tss - up, tss + down - 1
            else:
                p_start, p_end = tss - down + 1, tss + up
            seq=[]
            with gzip.open('Homo_sapiens.GRCh38.dna.chromosome.17.fa.gz','rt') as fa:
                for l in fa:
                    if not l.startswith('>'):
                        seq.append(l.strip().upper())
            s=''.join(seq)
            prom=s[p_start-1:p_end]
            gc=sum(1 for b in prom if b in 'GC')
            print(chrom, start, end, strand, tss, p_start, p_end, len(prom), gc, f'{gc/len(prom):.2f}')
            print(f'{chrom};{p_start};{p_end};{strand};{len(prom)};{gc};{gc/len(prom):.2f}')
            break
PY"
