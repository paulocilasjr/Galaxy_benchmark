/bin/bash -lc "python3 - <<'PY'
import urllib.request, urllib.parse, json
for query in ['compbiobench','compbiobench ground truth','compbiobench answers']:
    url='https://huggingface.co/api/datasets?search='+urllib.parse.quote(query)
    req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    try:
        data=json.load(urllib.request.urlopen(req, timeout=30))
        print('QUERY',query, len(data))
        for x in data[:10]: print(x['id'], x.get('private'))
    except Exception as e: print('ERR',e)
PY"
