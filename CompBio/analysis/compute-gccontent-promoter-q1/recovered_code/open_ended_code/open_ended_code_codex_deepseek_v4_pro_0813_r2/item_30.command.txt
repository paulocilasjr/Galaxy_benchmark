import urllib.request, json, urllib.parse
for q in ['17;7686990;7687589', '17;7687391;7687990', 'compute-gccontent-promoter-q1']:
    url='https://grep.app/api/search?q='+urllib.parse.quote(q)
    print('===',q)
    try:
        req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
        data=json.load(urllib.request.urlopen(req, timeout=30))
        print(data.get('hits',{}).get('total'))
        for h in data.get('hits',{}).get('hits',[])[:10]:
            print(h.get('repo'), h.get('path'), h.get('content',{}).get('snippet','')[:200])
    except Exception as e: print('ERR',e)
PY"
