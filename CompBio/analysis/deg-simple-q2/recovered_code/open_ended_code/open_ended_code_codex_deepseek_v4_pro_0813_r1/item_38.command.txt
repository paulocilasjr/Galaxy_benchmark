/bin/bash -lc "python3 - <<'PY'
import pysam, gzip, re, collections
# header mapping
hdr={}
with gzip.open('ref/Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as f:
    for line in f:
        if line.startswith('>'):
            tid=line.split()[0][1:]
            gs=re.search(r'gene_symbol:(\\S+)', line)
            gene=re.search(r'gene:(\\S+)', line)
            hdr[tid]=(gs.group(1) if gs else '', gene.group(1) if gene else '')
# set of transcript ids for candidate genes
cands={'EEF1A1','GAPDH','LCP1','CD74','HSP90AB1','IRF4','MYH9','CCL22','HSP90AA1','PABPC1','STAT1'}
tids_by_gs=collections.defaultdict(list)
for t,(gs,g) in hdr.items():
    if gs in cands:
        tids_by_gs[gs].append(t)
for s in ['sampleA','sampleB']:
    bam=pysam.AlignmentFile(f'align/{s}.cdna.bam','rb')
    print('\\nSample',s)
    for gs in cands:
        tset=set(tids_by_gs[gs])
        primary=collections.Counter(); unique=collections.Counter(); allmap=collections.Counter(); mapqhist=collections.Counter()
        for read in bam.fetch(until_eof=True):
            if read.is_unmapped: continue
            if read.reference_name in tset:
                if not read.is_secondary and not read.is_supplementary:
                    primary[read.reference_name]+=1
                    mapqhist[read.mapping_quality]+=1
                    if read.mapping_quality>=60:
                        unique[read.reference_name]+=1
        print(gs,'primary',sum(primary.values()),'unique>=60',sum(unique.values()),'transcripts',len(primary))
        for t,c in primary.most_common(12):
            print('  ',t,c,'uniq',unique[t],'mapqhist top',mapqhist.most_common(3))
PY"
