/bin/bash -lc "python3 - <<'PY'
import pysam, gzip, re, collections, math, csv
# header mapping transcript->gs,gene
hdr={}
with gzip.open('ref/Homo_sapiens.GRCh38.cdna.all.fa.gz','rt') as f:
    for line in f:
        if line.startswith('>'):
            tid=line.split()[0][1:]
            gs=re.search(r'gene_symbol:(\\S+)', line)
            gene=re.search(r'gene:(\\S+)', line)
            hdr[tid]=(gs.group(1) if gs else '', gene.group(1) if gene else '')
# count unique mapq>=60 and all primary mapped per transcript per sample
counts={}
for s in ['sampleA','sampleB']:
    bam=pysam.AlignmentFile(f'align/{s}.cdna.bam','rb')
    uni=collections.Counter(); prim=collections.Counter()
    for r in bam.fetch(until_eof=True):
        if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
        prim[r.reference_name]+=1
        if r.mapping_quality>=60:
            uni[r.reference_name]+=1
    counts[s]=(uni,prim)
# get all transcripts with unique count in either
tids=set(hdr)&(set(counts['sampleA'][0])|set(counts['sampleB'][0]))
data=[]
for t in tids:
    a=counts['sampleA'][0].get(t,0); b=counts['sampleB'][0].get(t,0)
    if a+b>0:
        data.append((t,hdr[t][0],hdr[t][1],a,b,counts['sampleA'][1].get(t,0),counts['sampleB'][1].get(t,0)))
print('n transcripts with unique reads',len(data))
# aggregate genes
genes=collections.defaultdict(list)
for row in data: genes[row[2]].append(row)
print('genes with unique reads',len(genes))
from scipy.stats import chi2
res=[]
for g,rows in genes.items():
    table=[(r[3],r[4]) for r in rows if r[3]+r[4]>0]
    if len(table)<2: continue
    totA=sum(x for x,y in table); totB=sum(y for x,y in table)
    if totA<1 or totB<1: continue
    n=totA+totB
    G=0
    for x,y in table:
        rowsum=x+y
        for obs,col in [(x,totA),(y,totB)]:
            exp=rowsum*col/n
            if obs>0 and exp>0: G += obs*math.log(obs/exp)
    G*=2; df=len(table)-1; p=chi2.sf(G,df)
    l1=sum(abs(x/totA-y/totB) for x,y in table)
    res.append((G,p,l1,totA,totB,g,rows))
res.sort(key=lambda x:-x[0])
print('\\nTop 60 by unique-read G-test (counts A/B)')
for G,p,l1,totA,totB,g,rows in res[:60]:
    gs=rows[0][1]
    print(f'G={G:.1f} p={p:.3g} L1={l1:.3f} tot={totA}/{totB} {gs} {g}')
    for r in sorted(rows, key=lambda r:-(r[3]+r[4]))[:6]:
        print('   ',r[1],r[0],r[3],'/',r[4])
PY"
