from collections import defaultdict
info={}
with open('gencode.v47.transcripts.fa') as f:
    for line in f:
        if line.startswith('>'):
            p=line[1:].rstrip().split('|'); info[p[0]]=p
for sample in ['A','B']:
    c=defaultdict(int)
    for line in open(f'sample{sample}.transcript_counts.tsv'):
        p=line.rstrip().rsplit(' ',1); c[p[0].split('|')[0]]=int(p[1])
    print('sample',sample)
    for tid,n in sorted(c.items(), key=lambda kv:-kv[1]):
        if info.get(tid,[''])[1]=='ENSG00000203970.4':
            print(tid,n,info[tid][5],info[tid][7])
PY"
