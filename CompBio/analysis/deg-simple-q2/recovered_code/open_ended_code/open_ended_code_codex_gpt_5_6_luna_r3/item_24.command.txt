/bin/bash -lc "python3 - <<'PY'
import gzip, re, csv, math
from collections import defaultdict
q={}
for s in 'AB':
  d={}
  with open(f'results/salmon_{s}/quant.sf') as f:
    for row in csv.DictReader(f, delimiter='\\t'):
      d[row['Name'].split('.')[0]]=(float(row['TPM']),float(row['NumReads']))
  q[s]=d
# gtf map
m={}
with gzip.open('refs/gencode.v47.annotation.gtf.gz','rt') as f:
 for line in f:
  if line.startswith('#'): continue
  a=line.rstrip().split('\\t')
  if a[2]"'!='"'transcript': continue
  attrs=dict(re.findall(r'(\\w+) \"(["'^"]*)"'"',a[8]))
  tid=attrs.get('transcript_id','').split('.')[0]
  if tid: m[tid]=(attrs.get('gene_id','').split('.')[0],attrs.get('gene_name',''),attrs.get('gene_type',''),attrs.get('transcript_type',''))
print('mapped',len(m),'quant',len(q['A']))
# effective count ratio with pseudocount, filter count total >= 5
rows=[]
for tid in q['A']:
  ta,ca=q['A'][tid]; tb,cb=q['B'][tid]
  if ca+cb>=2:
    # ratio normalized same total (check)
    l=math.log2((ca+.25)/(cb+.25))
    rows.append((abs(l),l,ca,cb,ta,tb,tid,*m.get(tid,('?','?','?','?'))))
for r in sorted(rows,reverse=True)[:100]: print('\\t'.join(map(str,r)))
print('totals',sum(x[1] for x in q['A'].values()),sum(x[1] for x in q['B'].values()))
PY"
