/bin/bash -lc "cat > scripts/compare_pure.py <<'PY'
import csv,math,collections
A={r['target_id']:r for r in csv.DictReader(open('results/A/abundance.tsv'),delimiter='\\t')}
B={r['target_id']:r for r in csv.DictReader(open('results/B/abundance.tsv'),delimiter='\\t')}
genes=collections.defaultdict(list)
for t,r in A.items():
 a=float(r['est_counts']);b=float(B[t]['est_counts']);name=t.split('|')[5]
 if a+b:genes[name].append((t.split('|')[0],a,b))
res=[]
for name,ts in genes.items():
 a=sum(t[1] for t in ts);b=sum(t[2] for t in ts);chi=0
 for _,ta,tb in ts:
  if ta+tb<10:continue
  ea=(ta+tb)*a/(a+b);eb=(ta+tb)*b/(a+b)
  if ea and eb:chi+=(ta-ea)**2/ea+(tb-eb)**2/eb
 tv=sum(abs(ta/max(a,1)-tb/max(b,1)) for _,ta,tb in ts)/2
 res.append((chi,name,a,b,tv))
res.sort(reverse=True)
print('GENE COMPOSITION: chi2, gene, A, B, total variation')
for row in res[:30]:print(row)
print('\\nTOP GENE TRANSCRIPTS')
for _,g,*_ in res[:10]:
 print(g)
 for t in sorted(genes[g],key=lambda t:t[1]+t[2],reverse=True)[:12]:print(t)
print('\\nLARGEST TRANSCRIPT DIFFERENCES')
diffs=sorted([(abs(a-b),g,t,a,b) for g,ts in genes.items() for t,a,b in ts],reverse=True)
for t in diffs[:30]:print(t)
with open('results/gene_ranking.tsv','w') as f:
 w=csv.writer(f,delimiter='\\t');w.writerow(['chi2','gene','A','B','TV']);w.writerows(res)
PY
python3 scripts/compare_pure.py > results/comparison.txt
cat results/comparison.txt"
