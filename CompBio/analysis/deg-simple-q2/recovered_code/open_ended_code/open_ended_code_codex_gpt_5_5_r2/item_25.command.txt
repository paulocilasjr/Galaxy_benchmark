/bin/bash -lc "python - <<'PY'
from collections import defaultdict,Counter
import math
files={'A':'results/sampleA.paf','B':'results/sampleB.paf'}
counts={s:defaultdict(Counter) for s in files}
for s,path in files.items():
    with open(path) as f:
        for line in f:
            parts=line.rstrip('\\n').split('\\t')
            hdr=parts[5]
            a=hdr.split('|')
            tid=a[0].split('.')[0]
            tname=a[4]
            gene=a[5]
            counts[s][gene][tname]+=1
# union genes
rows=[]
for g in set(counts['A'])|set(counts['B']):
    ca=counts['A'].get(g,Counter()); cb=counts['B'].get(g,Counter())
    ta=sum(ca.values()); tb=sum(cb.values())
    if ta+tb<100: continue
    iso=set(ca)|set(cb)
    if len(iso)<2: continue
    # L1 proportion difference
    l1=sum(abs(ca[i]/ta-cb[i]/tb) for i in iso)/2
    # chi-square statistic for 2xK with pseudofilter no p
    chi=0.0
    for i in iso:
        obs=[ca[i],cb[i]]; col=sum(obs); total=ta+tb
        for o,rowtot in zip(obs,[ta,tb]):
            exp=rowtot*col/total
            if exp>0: chi+=(o-exp)**2/exp
    # total expr ratio
    ratio=(ta+1)/(tb+1)
    rows.append((l1,chi,ta+tb,g,ta,tb,ratio,ca,cb))
rows.sort(reverse=True, key=lambda x:(x[0],x[1]))
for r in rows[:50]:
    l1,chi,n,g,ta,tb,ratio,ca,cb=r
    print(f\"{g}\\tL1={l1:.3f}\\tchi={chi:.1f}\\tn={n}\\tA={ta}\\tB={tb}\\tratio={ratio:.2f}\")
    for i in sorted(set(ca)|set(cb), key=lambda k: -(ca[k]+cb[k]))[:8]:
        print(f\"  {i}\\t{ca[i]}\\t{cb[i]}\\tpa={ca[i]/ta:.2f}\\tpb={cb[i]/tb:.2f}\")
PY"
