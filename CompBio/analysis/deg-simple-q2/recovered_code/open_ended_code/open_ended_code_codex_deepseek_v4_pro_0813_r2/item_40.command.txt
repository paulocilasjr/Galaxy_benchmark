/bin/bash -lc "python3 - <<'PY'
import pysam,gzip,re,collections
# quick no annotation? use known txs
counts={}
for s,b in [('A','A.bam'),('B','B.bam')]:
 c=collections.Counter(); f=pysam.AlignmentFile(b,'rb')
 for r in f.fetch(until_eof=True):
  if r.is_unmapped or r.is_secondary or r.is_supplementary: continue
  c[re.sub(r'\\.\\d+"'$'"','',r.reference_name)]+=1
 counts[s]=c
txs={'ENST00000371148','ENST00000393660'}
ca=sum(counts['A'].get(t,0) for t in txs); cb=sum(counts['B'].get(t,0) for t in txs)
print(ca,cb)
print(max(txs,key=lambda t: counts['A'].get(t,0)), max(txs,key=lambda t: counts['B'].get(t,0)))
print([(t,counts['A'].get(t,0),counts['B'].get(t,0)) for t in txs])
PY"
