/bin/bash -lc "python3 - <<'PY'
import pandas as pd, numpy as np
from scipy.stats import chi2_contingency, fisher_exact
x=pd.read_csv('txcounts.combined.tsv',sep='\\t',header=None,names=['tx','gene','symbol','A','B'])
# remove no symbol / weird empty
out=[]
for gene,g in x.groupby('gene',sort=False):
    if len(g)<2: continue
    a=g.A.to_numpy(); b=g.B.to_numpy()
    tot=a.sum()+b.sum()
    if tot<20 or a.sum()<5 or b.sum()<5: continue
    tab=np.array([a,b])
    try: chi,p,df,exp=chi2_contingency(tab,correction=False)
    except: continue
    # max abs fraction diff among isoforms, require observed total >= 5
    t=a+b; keep=t>=3
    da=a/a.sum()-b/b.sum()
    imax=np.argmax(np.abs(np.where(keep,da,0)))
    out.append((p,chi,tot,len(g),g.symbol.iloc[0],gene, g.tx.iloc[imax],a[imax],b[imax],a.sum(),b.sum(),da[imax]))
out.sort(key=lambda z:z[0])
print('genes tested',len(out))
print('p chi total n symbol gene tx A B Atot Btot diff')
for z in out[:100]: print(f'{z[0]:.3g} {z[1]:.1f} {z[2]} {z[3]} {z[4]} {z[5]} {z[6]} {z[7]} {z[8]} {z[9]} {z[10]} {z[11]:+.3f}')
PY"
