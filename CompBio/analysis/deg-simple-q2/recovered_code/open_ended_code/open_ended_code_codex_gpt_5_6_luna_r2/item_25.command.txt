import gzip
p='gencode.v44.annotation.gtf.gz'

 with gzip.open(p,'rt') as f:
  n=0
  for line in f:
   n+=1
   if n<=2: print(line.rstrip())
 print('lines',n)
except Exception as e:
 print(type(e).__name__,str(e))
PY
xxd -l 32 gencode.v44.annotation.gtf.gz; tail -c 32 gencode.v44.annotation.gtf.gz | xxd"
