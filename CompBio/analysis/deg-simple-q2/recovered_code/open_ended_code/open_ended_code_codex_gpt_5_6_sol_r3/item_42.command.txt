/bin/bash -lc 'tools/salmon-cli-x86_64-apple-darwin/salmon quant -i reference/salmon_gencode47_pc -l A -r deg.simple.q2.sampleA.fq.gz -p 4 --seqBias --gcBias -o quant/v47pc_sampleA > quant/v47pc_sampleA.log 2>&1 &
