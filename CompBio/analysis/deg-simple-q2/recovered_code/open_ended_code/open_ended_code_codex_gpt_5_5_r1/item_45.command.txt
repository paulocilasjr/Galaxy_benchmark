/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def load(path):
    d={}
    with open(path) as f:
        next(f)
        for line in f:
            tid,length,eff,c,tpm=line.rstrip('\\n').split('\\t')
            parts=tid.split('|')
            d[tid]=(parts[1],parts[5],parts[4],float(c),float(tpm))
    return d
A=load('quant/sampleA/abundance.tsv'); B=load('quant/sampleB/abundance.tsv')
by=defaultdict(list)
for tid,v in A.items(): by[(v[0],v[1])].append(tid)
for thr in [50,100,200,500,1000]:
    rows=[]
    for (gid,gname),tids in by.items():
        if len(tids)<2: continue
        ca=[A[t][3] for t in tids]; cb=[B[t][3] for t in tids]
        ta=sum(ca); tb=sum(cb)
        if ta<thr or tb<thr: continue
        pa=[x/ta for x in ca]; pb=[x/tb for x in cb]
        l1=sum(abs(x-y) for x,y in zip(pa,pb))/2
        maxdiff=max(abs(x-y) for x,y in zip(pa,pb))
        rows.append((l1,maxdiff,min(ta,tb),ta,tb,gname,gid,len(tids)))
    rows.sort(reverse=True)
    print('\\nTHR',thr,'N',len(rows))
    for r in rows[:20]: print('\\t'.join(map(str,r)))
PY"
