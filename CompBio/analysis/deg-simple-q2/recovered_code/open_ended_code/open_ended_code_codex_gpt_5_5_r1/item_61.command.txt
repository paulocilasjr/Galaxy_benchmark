/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def load(path):
    d={}
    with open(path) as f:
        next(f)
        for line in f:
            name,l,el,tpm,num=line.rstrip('\\n').split('\\t')
            parts=name.split('|')
            d[name]={'gid':parts[1],'gname':parts[5],'tx':parts[4],'cnt':float(num),'tpm':float(tpm),'len':int(float(l))}
    return d
A=load('quant/salmon_sampleA/quant.sf'); B=load('quant/salmon_sampleB/quant.sf')
by=defaultdict(list)
for tid,v in A.items(): by[(v['gid'],v['gname'])].append(tid)
for thr in [50,100,200,500,1000]:
    rows=[]
    for (gid,gname),tids in by.items():
        if len(tids)<2: continue
        ta=sum(A[t]['cnt'] for t in tids); tb=sum(B[t]['cnt'] for t in tids)
        if ta<thr or tb<thr: continue
        l1=sum(abs(A[t]['cnt']/ta-B[t]['cnt']/tb) for t in tids)/2
        maxdiff=max(abs(A[t]['cnt']/ta-B[t]['cnt']/tb) for t in tids)
        rows.append((l1,maxdiff,min(ta,tb),ta,tb,gname,gid,len(tids)))
    rows.sort(reverse=True)
    print('\\nTHR',thr,'N',len(rows))
    for r in rows[:20]: print('\\t'.join(map(str,r)))
PY"
