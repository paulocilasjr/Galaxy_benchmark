/bin/bash -lc "python3 - <<'PY'
from collections import defaultdict

def load(path):
    d={}
    with open(path) as f:
        next(f)
        for line in f:
            tid,length,eff,c,tpm=line.rstrip('\\n').split('\\t')
            parts=tid.split('|')
            d[tid]={'gid':parts[1],'gname':parts[5],'tx':parts[4],'cnt':float(c),'tpm':float(tpm),'len':int(length),'type':parts[7] if len(parts)>7 else ''}
    return d
A=load('quant/sampleA/abundance.tsv'); B=load('quant/sampleB/abundance.tsv')
by=defaultdict(list)
for tid,v in A.items(): by[v['gname']].append(tid)
for gene in ['RPS3','STAT1','RPL15','FNBP1','RPL19','TPT1','LCP1','IKZF2','HUWE1','FLNA','HNRNPC','PDIA3','HNRNPK']:
    tids=by[gene]
    ta=sum(A[t]['cnt'] for t in tids); tb=sum(B[t]['cnt'] for t in tids)
    rows=[]
    for t in tids:
        ca=A[t]['cnt']; cb=B[t]['cnt']
        if ca+cb>1:
            rows.append((abs(ca/ta-cb/tb), ca, cb, ca/ta if ta else 0, cb/tb if tb else 0, A[t]['tx'], t.split('|')[0], A[t]['len']))
    rows.sort(reverse=True)
    l1=sum(r[0] for r in rows)/2
    print('\\n',gene,'totalA',ta,'totalB',tb,'L1',l1,'n_tx_nonzero',len(rows))
    for r in rows[:8]: print('  ', '\\t'.join(map(lambda x: f'{x:.4g}' if isinstance(x,float) else str(x), r)))
PY"
