/bin/bash -lc 'mkdir -p results/sampleA results/sampleB
./tools_env/bin/salmon quant -i reference/salmon_gencode49 -l A -r deg.simple.q2.sampleA.fq.gz --fldMean 200 --fldSD 80 --validateMappings --seqBias -p 4 -o results/sampleA > results/sampleA.run.log 2>&1 &
pid_a=$!
./tools_env/bin/salmon quant -i reference/salmon_gencode49 -l A -r deg.simple.q2.sampleB.fq.gz --fldMean 200 --fldSD 80 --validateMappings --seqBias -p 4 -o results/sampleB > results/sampleB.run.log 2>&1 &
pid_b=$!
wait "$pid_a" "$pid_b"
tail -n 12 results/sampleA.run.log
tail -n 12 results/sampleB.run.log'
