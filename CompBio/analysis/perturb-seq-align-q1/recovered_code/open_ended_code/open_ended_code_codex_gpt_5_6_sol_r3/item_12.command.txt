/bin/bash -lc "python -u - <<'PY'
import anndata as ad, numpy as np, pandas as pd
from scipy import sparse
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 a=ad.read_h5ad(fn)
 print('\\n',fn)
 print(a.obs['guide'].value_counts().sort_index().to_string())
 # CPM log mean signatures
 X=a.X.tocsr().astype(float)
 sf=np.asarray(X.sum(1)).ravel(); X=X.multiply(1e4/np.maximum(sf,1)[:,None]); X.data=np.log1p(X.data)
 cats=list(a.obs.guide.cat.categories)
 means=np.vstack([np.asarray(X[a.obs.guide.values==g].mean(0)).ravel() for g in cats])
 genes=np.array(a.var_names)
 for target in ['PABPC1','NUDT21','LEO1']:
  j=np.where(genes==target)[0][0]
  vals=pd.Series(means[:,j],index=cats)
  ntmean=vals[[x.startswith('NT-') for x in vals.index]].mean()
  print('\\nTARGET',target,'NT',ntmean)
  print((vals-ntmean).sort_values().head(12).to_string())
PY"
