import anndata as ad
for f in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:

    print('n obs',len(a.obs),'n guide unique', a.obs['guide'].nunique())
    print('guide counts:\\n', a.obs['guide'].value_counts().to_string())
    if 'target_gene' in a.obs:

        q=a.obs[a.obs['target_gene'].astype(str).str.upper().isin(['PABPC1','NUDT21','LEO1'])]
        print('requested rows:\\n',q.to_string())
        print('requested map:\\n',q[['guide','target_gene']].drop_duplicates().to_string(index=False))
