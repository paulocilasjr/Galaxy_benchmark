import h5py
import numpy as np

def decode(x):
 if isinstance(x, bytes): return x.decode()
 return str(x)
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as h:
 for c in ['guide','target_gene']:
  cats=[decode(x) for x in h['obs'][c]['categories'][()]]
  codes=h['obs'][c]['codes'][()]
  print(c, len(cats), cats)
  if c=='guide': guides=cats
  else: targets=cats
 guides_for={}
 for gcode,tcode in zip(h['obs']['guide']['codes'][()],h['obs']['target_gene']['codes'][()]):
  guides_for.setdefault(guides[gcode],set()).add(targets[tcode])
 print('MAP')
 for g,ts in guides_for.items(): print(repr(g),'->',sorted(ts))
 print('TARGET REQUESTED')
 for target in ['PABPC1','NUDT21','LEO1']:
  print(target,[g for g,ts in guides_for.items() if target in ts])
with h5py.File('perturb.seq.align.q1.query.h5ad','r') as h:
