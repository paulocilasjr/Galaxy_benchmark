/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, pandas as pd

def cat_col(f, path):
    g=f[path]
    cats=np.array(g['categories']).astype(str)
    codes=np.array(g['codes'])
    return pd.Categorical.from_codes(codes, cats)
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as f:
    guides=cat_col(f,'obs/guide')
    targets=cat_col(f,'obs/target_gene')
    df=pd.DataFrame({'guide':guides.astype(str),'target_gene':targets.astype(str)})
    print('guide cats:', list(np.array(f['obs/guide/categories']).astype(str)))
    print('target cats:', list(np.array(f['obs/target_gene/categories']).astype(str)))
    print('\\nRequested:')
    for t in ['PABPC1','NUDT21','LEO1']:
        sub=df[df.target_gene==t]
        print(t, sub.groupby('guide').size().sort_values(ascending=False).to_dict())
    print('\\nAll guide-target counts:')
    print(df.groupby(['target_gene','guide']).size().to_string())
with h5py.File('perturb.seq.align.q1.query.h5ad','r') as f:
    print('\\nquery guides:', list(np.array(f['obs/guide/categories']).astype(str)))
PY"
