/bin/bash -lc "python - <<'PY'
import h5py, numpy as np

def dec(ds): return [x.decode() if isinstance(x,bytes) else str(x) for x in ds[:]]
def col(f,j):
    indptr=f['X/indptr']; indices=f['X/indices']; data=f['X/data']; n=int(f['X'].attrs['shape'][0])
    a=int(indptr[j]); b=int(indptr[j+1]); v=np.zeros(n,float); v[indices[a:b]]=data[a:b]; return v
with h5py.File('perturb.seq.align.q1.query.h5ad','r') as f:
    guides=dec(f['obs/guide/categories']); codes=f['obs/guide/codes'][:].astype(int); genes=dec(f['var/_index'])
    ntmask=np.array([guides[i].startswith('NT-') for i in codes])
    for gene in ['CDC73','CPSF3L']:
        if gene not in genes: print(gene,'missing'); continue
        vals=col(f, genes.index(gene))
        print('\\n',gene,'NTmean',vals[ntmask].mean())
        rows=[]
        for i,g in enumerate(guides):
            m=codes==i
            if not m.sum() or g.startswith('NT-'): continue
            rows.append((g, vals[m].mean(), int(m.sum())))
        for r in sorted(rows,key=lambda x:x[1])[:15]: print(r[0], round(r[1],3), r[2])
PY"
