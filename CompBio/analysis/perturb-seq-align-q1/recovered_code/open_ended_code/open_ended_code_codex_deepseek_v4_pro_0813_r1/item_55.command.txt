/bin/bash -lc "python3 - <<'PY'
import h5py,numpy as np
from collections import Counter

def load_meta(f):
 with h5py.File(f,'r') as h:
  guides=h['obs/guide/categories'][:].astype(str); codes=h['obs/guide/codes'][:]
  targets=h['obs/target_gene/categories'][:].astype(str) if 'obs/target_gene/categories' in h else None
  tcodes=h['obs/target_gene/codes'][:] if targets is not None else None
 return guides,codes,targets,tcodes
gr,cr,tr,tcr=load_meta('perturb.seq.align.q1.ref.h5ad')
gq,cq,_,_=load_meta('perturb.seq.align.q1.query.h5ad')
r={g:(cr==i).sum() for i,g in enumerate(gr) if not g.startswith('NT-')}
q={g:(cq==i).sum() for i,g in enumerate(gq) if not g.startswith('NT-')}
# total perturb cells
print('ref perturb',sum(r.values()),'query perturb',sum(q.values()),'ratio',sum(q.values())/sum(r.values()))
# sort by counts
for target in ['PABPC1','NUDT21','LEO1']:
 rg=[g for g in gr if not g.startswith('NT-') and tr[tcr[cr==gr.tolist().index(g)]][0]==target][0]
 print(target,rg,r[rg], 'scaled',round(r[rg]*sum(q.values())/sum(r.values()),1),'query near counts:')
 for qg,c in sorted(q.items(), key=lambda x:abs(x[1]-r[rg]*sum(q.values())/sum(r.values())))[:10]:
  print(' ',qg,c)
PY"
