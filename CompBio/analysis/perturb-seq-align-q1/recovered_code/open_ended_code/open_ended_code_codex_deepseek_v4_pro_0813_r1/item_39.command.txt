/bin/bash -lc "python3 - <<'PY'
import requests,re,html
u='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.full'
text=requests.get(u,headers={'User-Agent':'Mozilla/5.0'},timeout=30).text
plain=html.unescape(re.sub('<["'^>]+>'"',' ',text))
for term in ['Supplementary','supplement','Data availability','Zenodo','huggingface','github']:
 for m in re.finditer(term,plain,re.I):
  print(term, plain[m.start()-100:m.start()+300].replace('\\n',' ')[:400]); print('---'); break
PY"
