/bin/bash -lc "python3 - <<'PY'
import requests,time
url='https://ftp.ncbi.nlm.nih.gov/geo/series/GSE269nnn/GSE269596/suppl/GSE269596_CPA_K562.Rds.gz'
r=requests.get(url,stream=True,timeout=30)
print(r.status_code,r.headers.get('content-length'))
start=time.time(); chunk=0
for c in r.iter_content(1024*1024):
 chunk+=len(c); 
 if chunk>=20*1024*1024: break
print('downloaded',chunk,'in',time.time()-start,'sec')
PY"
