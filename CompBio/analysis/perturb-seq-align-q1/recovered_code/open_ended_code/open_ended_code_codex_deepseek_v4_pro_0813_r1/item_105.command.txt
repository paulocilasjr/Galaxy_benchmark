/bin/bash -lc "python3 - <<'PY'
import requests,re
u='https://www.biorxiv.org/content/10.64898/2026.04.06.716850v2.supplementary-material'
text=requests.get(u,headers={'User-Agent':'Mozilla/5.0'},timeout=30).text
for m in re.finditer(r'href=[\"\\'](["'^"'"\\']+)[\"\\']', text, re.I):
 h=m.group(1)
 if any(x in h.lower() for x in ['.xlsx','.xls','.csv','.tsv','.zip','suppl','media']):
  print(h)
# also look for file names in text
for term in ['Table S1','TableS1','xlsx']:
 print(term, text.find(term))
PY"
