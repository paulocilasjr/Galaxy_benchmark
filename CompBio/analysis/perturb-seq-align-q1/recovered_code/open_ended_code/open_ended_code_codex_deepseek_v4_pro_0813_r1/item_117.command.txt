/bin/bash -lc "cat > inspect_k562.R <<'EOF'
suppressMessages({library(Seurat); library(Matrix)})
cat('reading\\n')
obj <- readRDS('CPA_K562.Rds.gz')
cat('class', class(obj), '\\n')
cat('slots', slotNames(obj), '\\n')
cat('meta cols', paste(colnames(obj@meta.data), collapse=', '), '\\n')
cat('dims cells', nrow(obj@meta.data), '\\n')
print(head(obj@meta.data, 2))
cat('assays', names(obj@assays), '\\n')
print(slotNames(obj@assays"'$RNA))
cat('"'counts class', class(obj@assays"'$RNA@counts), '"'dim', dim(obj@assays"'$RNA@counts), '"'\\n')
cat('data class', class(obj@assays"'$RNA@data), '"'dim', dim(obj@assays"'$RNA@data), '"'\\n')
# find columns that look guide/target
for (cc in colnames(obj@meta.data)) {
 vals <- as.character(obj@meta.data[[cc]])
 if (any(grepl('"'^NT|PABPC1|NUDT21|LEO1|CDC73'"', vals))) cat('candidate',cc,'examples',paste(head(unique(vals),10),collapse=','),'\\n')
}
EOF
Rscript inspect_k562.R > inspect_k562.out 2>&1"
