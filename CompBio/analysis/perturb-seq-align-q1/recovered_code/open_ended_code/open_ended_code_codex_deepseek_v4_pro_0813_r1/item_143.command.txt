/bin/bash -lc "cat > inspect_assay5.R <<'EOF'
suppressMessages({library(Seurat); library(Matrix)})
con<-pipe('gzip -dc CPA_K562.Rds.gz | gzip -dc','rb'); obj<-readRDS(con); close(con)
rna <- obj@assays"'$RNA
for(sn in c('"'cells','features','layers')) {
 cat('\\nslot',sn,'class',class(slot(rna,sn)),'length/dim',dim(slot(rna,sn)),'len',length(slot(rna,sn)),'\\n')
 print(str(slot(rna,sn), max.level=1))
}
cat('cells head as char', paste(head(as.character(rna@cells),10),collapse=','),'\\n')
cat('features head as char', paste(head(as.character(rna@features),10),collapse=','),'\\n')
cat('meta order first/last layers maybe\\n')
EOF
Rscript inspect_assay5.R > inspect_assay5.out 2>&1"
