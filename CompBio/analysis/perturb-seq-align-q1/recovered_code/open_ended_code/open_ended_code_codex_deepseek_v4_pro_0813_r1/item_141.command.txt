/bin/bash -lc "cat > inspect_layers2.R <<'EOF'
suppressMessages({library(Seurat); library(Matrix)})
con<-pipe('gzip -dc CPA_K562.Rds.gz | gzip -dc','rb'); obj<-readRDS(con); close(con)
rna <- obj@assays"'$RNA
cat('"'layer names', names(rna@layers), '\\n')
for(nm in names(rna@layers)) {
  cat(nm, 'dim', dim(rna@layers[[nm]]), 'headcols', paste(head(colnames(rna@layers[[nm]]),3),collapse=','), 'tailcols', paste(tail(colnames(rna@layers[[nm]]),3),collapse=','), '\\n')
}
cat('meta head rownames', paste(head(rownames(obj@meta.data),10),collapse=','), '\\n')
cat('meta tail rownames', paste(tail(rownames(obj@meta.data),10),collapse=','), '\\n')
cat('layer1 rownames head', paste(head(rownames(rna@layers[[1]]),5),collapse=','), '\\n')
EOF
Rscript inspect_layers2.R > inspect_layers2.out 2>&1"
