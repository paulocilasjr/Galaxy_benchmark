/bin/bash -lc "cat > inspect_logmap.R <<'EOF'
suppressMessages({library(Seurat); library(Matrix)})
con<-pipe('gzip -dc CPA_K562.Rds.gz | gzip -dc','rb'); obj<-readRDS(con); close(con)
rna<-obj@assays"'$RNA
cm <- slot(rna@cells,'"'data'); fm <- slot(rna@features,'data')
cat('cells logmap dim',dim(cm),'class',class(cm),'\\n')
print(cm[1:5,1:4])
cat('features logmap dim',dim(fm),'\\n')
print(fm[1:5,1:4])
cat('rownames cells logmap? ', paste(head(rownames(cm),5),collapse=','), '\\n')
cat('colnames cells logmap? ', paste(head(colnames(cm),5),collapse=','), '\\n')
cat('rownames features logmap? ', paste(head(rownames(fm),5),collapse=','), '\\n')
cat('colnames features logmap? ', paste(head(colnames(fm),5),collapse=','), '\\n')
cat('layer colnames empty? yes\\n')
# try labels function from SeuratObject namespace directly
cat('try labels cells\\n')
print(SeuratObject:::labels.LogMap(rna@cells, values=rownames(obj@meta.data)[1:3]))
EOF
Rscript inspect_logmap.R > inspect_logmap.out 2>&1"
