/bin/bash -lc "python3 - <<'PY'
import h5py, numpy as np, scipy.sparse as sp, scipy.io as sio, os

def load(f):
 with h5py.File(f,'r') as h:
  X=sp.csr_matrix((h['X/data'][:],h['X/indices'][:],h['X/indptr'][:]),shape=(len(h['var/_index']),len(h['obs/_index']))).T.tocsr()
  guides=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['obs/guide/categories'][:]])
  codes=h['obs/guide/codes'][:]
  genes=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['var/_index'][:]])
  obsnames=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['obs/_index'][:]])
  targets=np.array([x.decode() if isinstance(x,bytes) else str(x) for x in h['obs/target_gene/categories'][:]]) if 'obs/target_gene/categories' in h else None
  tcodes=h['obs/target_gene/codes'][:] if targets is not None else None
 return X,guides,codes,genes,obsnames,targets,tcodes
Xr,gr,cr,gr_genes,robs,tr,tcr=load('perturb.seq.align.q1.ref.h5ad')
Xq,gq,cq,gq_genes,qobs,tq,tcq=load('perturb.seq.align.q1.query.h5ad')
common,ri,qi=np.intersect1d(gr_genes,gq_genes,return_indices=True)
os.makedirs('r_input',exist_ok=True)
# Save common genes and matrices (cells x genes)
sio.mmwrite('r_input/ref_matrix.mtx', Xr[:,ri].tocoo())
sio.mmwrite('r_input/query_matrix.mtx', Xq[:,qi].tocoo())
np.savetxt('r_input/genes.txt', common, fmt='%s')
# metadata
with open('r_input/ref_meta.tsv','w') as f:
 f.write('cell\\tguide\\ttarget_gene\\n')
 for i in range(len(robs)):
  t=tr[tcr[i]] if tr is not None else 'NA'
  f.write(f'{robs[i]}\\t{gr[cr[i]]}\\t{t}\\n')
with open('r_input/query_meta.tsv','w') as f:
 f.write('cell\\tguide\\n')
 for i in range(len(qobs)):
  f.write(f'{qobs[i]}\\t{gq[cq[i]]}\\n')
print('saved',Xr[:,ri].shape,Xq[:,qi].shape)
PY
ls -lh r_input"
