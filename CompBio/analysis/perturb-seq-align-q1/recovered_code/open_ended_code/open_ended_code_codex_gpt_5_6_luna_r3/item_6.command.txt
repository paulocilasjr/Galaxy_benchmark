import h5py

def visit(name,obj):
    if isinstance(obj,h5py.Dataset):
        print(name,'DATA',obj.shape,obj.dtype)
    else:
        print(name,'GROUP')
        for k,v in obj.attrs.items():
            print('  attr',k,repr(v)[:300])
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 print('\\n###',fn)
 with h5py.File(fn,'r') as f:
  f.visititems(visit)
PY"
