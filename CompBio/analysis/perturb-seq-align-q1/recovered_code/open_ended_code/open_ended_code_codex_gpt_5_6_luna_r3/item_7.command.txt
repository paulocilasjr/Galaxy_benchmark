/bin/bash -lc "python - <<'PY'
import h5py, numpy as np
for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
 print('\\n',fn)
 with h5py.File(fn,'r') as f:
  cats={}
  for col in f['obs'].attrs['column-order']:
   col=col.decode() if isinstance(col,bytes) else str(col)
   g=f['obs'][col]
   if 'categories' in g:
    cats[col]=[x.decode() if isinstance(x,bytes) else str(x) for x in g['categories'][...]]
    print(col, len(cats[col]), cats[col])
  if 'target_gene' in cats:
   guides=np.array(f['obs']['guide']['codes'][...])
   targets=np.array(f['obs']['target_gene']['codes'][...])
   gc=cats['guide']; tc=cats['target_gene']
   print('mapping')
   for t in tc:
    gs=sorted(set(gc[g] for g,tt in zip(guides,targets) if tc[tt]==t))
    print(t,gs)
PY"
