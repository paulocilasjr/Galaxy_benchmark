/bin/bash -lc "python - <<'PY'
# print full mapping for raw effect var1000 and all, for context
import anndata as ad, numpy as np, pandas as pd
from scipy import sparse
from scipy.optimize import linear_sum_assignment
ref=ad.read_h5ad('perturb.seq.align.q1.ref.h5ad'); q=ad.read_h5ad('perturb.seq.align.q1.query.h5ad')
targets=set(ref.obs['target_gene'].astype(str).unique())-{'NT'}
common=[g for g in ref.var_names if g in set(q.var_names) and g not in targets]
r=ref[:,common].copy(); q=q[:,common].copy()
def norm_log(a):
 X=a.X.astype(float).tocsr(); lib=np.asarray(X.sum(1)).ravel(); sf=1e4/np.maximum(lib,1); X=sparse.diags(sf).dot(X); X.data=np.log1p(X.data); return X.toarray()
R=norm_log(r); Q=norm_log(q)
rguide=r.obs.guide.astype(str).values; qguide=q.obs.guide.astype(str).values
rg=[g for g in r.obs.guide.cat.categories if not g.startswith('NT-')]
qg=[g for g in q.obs.guide.cat.categories if not g.startswith('NT-')]
rnt=R[np.array([g.startswith('NT') for g in rguide])].mean(0); qnt=Q[np.array([g.startswith('NT') for g in qguide])].mean(0)
Er=np.array([R[rguide==g].mean(0)-rnt for g in rg]); Eq=np.array([Q[qguide==g].mean(0)-qnt for g in qg])
def corr(A,B):
 A=A-A.mean(1,keepdims=True); B=B-B.mean(1,keepdims=True); A/=np.linalg.norm(A,axis=1,keepdims=True)+1e-9; B/=np.linalg.norm(B,axis=1,keepdims=True)+1e-9; return A@B.T
for k in [1000,11857]:
 v=Er.var(0)+Eq.var(0); idx=np.argsort(v)[-k:] if k<11857 else np.arange(Er.shape[1])
 C=corr(Er[:,idx],Eq[:,idx])
 ri,ci=linear_sum_assignment(-C)
 print('\\nk',k,'score',sum(C[i,j] for i,j in zip(ri,ci)))
 for i,j in sorted(zip(ri,ci)):
  print(f'{rg[i]:12s} -> {qg[j]:7s} corr {C[i,j]:.3f} (best {qg[np.argmax(C[i])]} {C[i].max():.3f})')
PY"
