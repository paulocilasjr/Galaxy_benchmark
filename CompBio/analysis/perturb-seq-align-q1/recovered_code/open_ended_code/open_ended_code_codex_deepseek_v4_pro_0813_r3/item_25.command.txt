base='https://huggingface.co/datasets/amanutej/trustworthy-biology-agents-traces/resolve/main/'
paths=[
'traces/compbiobench/compbiobench/codex_gpt-5.5_20260613_224735/questions/perturb-seq-align-q1/result.json',
'traces/compbiobench/compbiobench/codex_gpt-5.5_20260613_224735/questions/perturb-seq-align-q1/trace.md',

'traces/compbiobench/compbiobench/codex_gpt-5.5_20260613_224735/questions/perturb-seq-align-q1/raw_stdout.jsonl',
]
for p in paths:
    local='hf_'+p.replace('/','_')
    try:
        urllib.request.urlretrieve(base+p, local)
        print('saved',local, os.path.getsize(local))
        if p.endswith('.json'):
            data=json.load(open(local))
            print(json.dumps(data,indent=2)[:3000])
    except Exception as e: print('err',p,e)
PY"
