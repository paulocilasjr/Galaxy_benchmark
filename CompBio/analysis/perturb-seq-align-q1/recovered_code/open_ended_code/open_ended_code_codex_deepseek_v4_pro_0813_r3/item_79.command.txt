/bin/bash -lc "python - <<'PY'
import gzip, time, re, subprocess
fn='GSE269596_CPA_K562.Rds.gz'
offset=2709000000
count=5000000
chunk=16*1024*1024
print('skipping')
t=time.time()
with gzip.open(fn,'rb') as f1, gzip.GzipFile(fileobj=f1, mode='rb') as f2:
    skipped=0
    while skipped<offset:
        data=f2.read(min(chunk, offset-skipped))
        if not data: raise EOFError
        skipped+=len(data)
    seg=f2.read(count)
open('k562_hto_guide_segment.bin','wb').write(seg)
print('wrote',len(seg),'time',time.time()-t)
lines=[x.strip() for x in subprocess.check_output(['strings','-n','4','k562_hto_guide_segment.bin'], text=True).splitlines()]
hto=[x for x in lines if x.startswith('HTO-')]
pat=re.compile(r'"'^(?:[A-Z0-9]+-I'"\\d+|NT-I\\d+)"'$'"')
guides=[x for x in lines if pat.match(x)]
print('lines',len(lines),'hto',len(hto),'guides',len(guides))
print('first guide index', next((i for i,x in enumerate(lines) if pat.match(x)), None))
print('hto counts', __import__('collections').Counter(hto).most_common())
PY"
