/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np, pandas as pd
from scipy import sparse
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.preprocessing import StandardScaler
ref=ad.read_h5ad('perturb.seq.align.q1.ref.h5ad'); q=ad.read_h5ad('perturb.seq.align.q1.query.h5ad')
targets=set(ref.obs['target_gene'].astype(str).unique())-{'NT'}
common=[g for g in ref.var_names if g in set(q.var_names) and g not in targets]
r=ref[:,common].copy(); q=q[:,common].copy()
def norm_log(a):
 X=a.X.astype(float).tocsr(); lib=np.asarray(X.sum(1)).ravel(); sf=1e4/np.maximum(lib,1); X=sparse.diags(sf).dot(X); X.data=np.log1p(X.data); return X.toarray()
R=norm_log(r); Q=norm_log(q)
rguide=r.obs.guide.astype(str).values; qguide=q.obs.guide.astype(str).values
rnt=R[np.array([g.startswith('NT') for g in rguide])].mean(0); qnt=Q[np.array([g.startswith('NT') for g in qguide])].mean(0)
rg=[g for g in r.obs.guide.cat.categories if not g.startswith('NT-')]; qg=[g for g in q.obs.guide.cat.categories if not g.startswith('NT-')]
Er=np.array([R[rguide==g].mean(0)-rnt for g in rg]); Eq=np.array([Q[qguide==g].mean(0)-qnt for g in qg])
# restrict to variable genes? maybe all
v=Er.var(0)+Eq.var(0); idx=np.argsort(v)[-2000:]
Xr=Er[:,idx]; Xq=Eq[:,idx]
# standardize each gene across all ref guides for LDA
scaler=StandardScaler().fit(Xr)
Xr_s=scaler.transform(Xr); Xq_s=scaler.transform(Xq)
classes=['LEO1-I2','PAF1-I6','CTR9-I7','CDC73-I2']
y=np.array([classes.index(g) if g in classes else -1 for g in rg])
train=Xr_s[np.array([g in classes for g in rg])]
ytrain=np.array([classes.index(g) for g in rg if g in classes])
lda=LinearDiscriminantAnalysis(solver='lsqr', shrinkage='auto')
lda.fit(train,ytrain)
pred=lda.predict(Xq_s)
proba=lda.predict_proba(Xq_s)
print('LDA trained on ref PAF1 complex')
for j,g in enumerate(qg):
    if g in ['guide14','guide28','guide10','guide12','guide30','guide16','guide20','guide17','guide9'] or proba[j].max()>0.15:
        print(g, 'pred', classes[pred[j]], 'proba', dict(zip(classes,np.round(proba[j],3))))
# Print top candidates for LEO1 by proba
print('\\nTop LEO1 candidates')
order=np.argsort(proba[:,0])[::-1]
for j in order[:15]: print(qg[j], proba[j,0], dict(zip(classes,np.round(proba[j],3))))
PY"
