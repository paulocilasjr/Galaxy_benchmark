/bin/bash -lc "python - <<'PY'
import anndata as ad, numpy as np
from collections import Counter
# original labels
orig=[x.strip() for x in open('k562_guide_like_labels.txt') if x.strip()]
selected=set(['CDC73-I2','CPSF1-I3','CPSF2-I1','CPSF3-I3','CPSF3L-I8','CPSF4-I8','CPSF6-I5','CPSF7-I1','CSTF1-I5','CSTF2-I4','CSTF3-I5','CTR9-I7','FIP1L1-I1','LEO1-I2','NUDT21-I3','PABPC1-I1','PABPN1-I2','PAF1-I6','PAPOLA-I3','PCF11-I5','PHF3-I5','RBBP6-I4','RPAP2-I2','RPRD1A-I5','RPRD1B-I1','SCAF1-I2','SCAF4-I4','SCAF8-I3','SCAF11-I3','SF3A1-I5','SRSF3-I1','SYMPK-I3','THOC5-I1']+[f'NT-I{i}' for i in range(1,11)])
orig_sel=[g for g in orig if g in selected]
q=ad.read_h5ad('perturb.seq.align.q1.query.h5ad')
qseq=q.obs['guide'].astype(str).tolist()
# greedy subsequence match of NT labels only, preserving order, using selected orig
positions=[]
oi=0
for qi,g in enumerate(qseq):
    if g.startswith('NT-'):
        while oi<len(orig_sel) and orig_sel[oi]"'!=g:
            oi+=1
        if oi>=len(orig_sel):
            print('"'failed at qi',qi,g,'oi',oi); break
        positions.append((qi,oi,g)); oi+=1
else:
    print('matched all',len(positions),'of',sum(g.startswith('NT-') for g in qseq),'final oi',oi,'len orig_sel',len(orig_sel))
print('orig_sel counts',Counter(orig_sel).most_common())
print('query counts',Counter(qseq).most_common())
PY"
