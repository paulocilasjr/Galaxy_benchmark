/bin/bash -lc "python - <<'PY'
import gzip, time
fn='GSE269596_CPA_K562.Rds.gz'
offset=9310329000
count=1000000
chunk=16*1024*1024
print('skipping to metadata')
t=time.time()
with gzip.open(fn,'rb') as f1, gzip.GzipFile(fileobj=f1, mode='rb') as f2:
    skipped=0
    while skipped<offset:
        data=f2.read(min(chunk, offset-skipped))
        if not data: raise EOFError
        skipped+=len(data)
    seg=f2.read(count)
open('k562_meta_segment.bin','wb').write(seg)
print('wrote',len(seg),'time',time.time()-t)
PY
strings -n 3 k562_meta_segment.bin | head -300"
