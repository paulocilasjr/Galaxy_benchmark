/bin/bash -lc "python - <<'PY'
import urllib.request, json
url='https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/traces?recursive=true'
data=json.load(urllib.request.urlopen(url, timeout=60))
print('len',len(data))
paths=[x.get('path') for x in data]
for s in ['compbio','perturb','align','agy']:
    print(s, [p for p in paths if s in p.lower()][:20])
PY"
