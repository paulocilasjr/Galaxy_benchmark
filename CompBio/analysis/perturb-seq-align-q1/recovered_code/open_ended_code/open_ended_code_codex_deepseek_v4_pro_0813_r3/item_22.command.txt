/bin/bash -lc "python - <<'PY'
import urllib.request, json
for path in ['tasks','traces']:
    url=f'https://huggingface.co/api/datasets/amanutej/trustworthy-biology-agents-traces/tree/main/{path}?recursive=true'
    data=json.load(urllib.request.urlopen(url, timeout=60))
    print('path',path,'len',len(data))
    hits=[x.get('path') for x in data if 'perturb' in x.get('path','').lower() or 'align-q1' in x.get('path','').lower()]
    print('hits',hits[:50])
    # print first few
    for x in data[:20]: print(x.get('path'), x.get('type'))
PY"
