/bin/bash -lc "python - <<'PY'
import h5py, numpy as np

def dec(a): return np.array([x.decode() if isinstance(x,(bytes,bytearray)) else str(x) for x in a])
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as fr, h5py.File('perturb.seq.align.q1.query.h5ad','r') as fq:
    vr=dec(fr['var/_index'][()]); vq=dec(fq['var/_index'][()])
print('len', len(vr), len(vq), 'equal order', np.array_equal(vr,vq), 'same set', set(vr)==set(vq), 'intersection', len(set(vr)&set(vq)))
print('ref first20', vr[:20])
print('query first20', vq[:20])
for gene in ['PABPC1','NUDT21','LEO1']:
    print(gene, 'ref idx', np.where(vr==gene)[0][:5], 'query idx', np.where(vq==gene)[0][:5])
    print(' query contains case-insens?', [x for x in vq if x.upper()==gene.upper()][:10])
    print(' ref contains case-insens?', [x for x in vr if x.upper()==gene.upper()][:10])
print('examples ref not query', list(sorted(set(vr)-set(vq)))[:20])
print('examples query not ref', list(sorted(set(vq)-set(vr)))[:20])
PY"
