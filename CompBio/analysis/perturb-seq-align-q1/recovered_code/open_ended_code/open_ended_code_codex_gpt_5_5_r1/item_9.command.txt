/bin/bash -lc "python - <<'PY'
import h5py, collections

def cat_col(f, name):
    g=f['obs'][name]
    cats=[x.decode() if isinstance(x, bytes) else str(x) for x in g['categories'][()]]
    codes=g['codes'][()]
    return [cats[int(c)] if c>=0 else None for c in codes], cats
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as f:
    guides, guide_cats=cat_col(f,'guide')
    targets, target_cats=cat_col(f,'target_gene')
    print('guide categories')
    print('\\n'.join(guide_cats))
    print('\\ntarget categories')
    print('\\n'.join(target_cats))
    pairs=collections.Counter(zip(guides, targets))
    print('\\nPairs by guide:')
    for (g,t),n in sorted(pairs.items()):
        print(f'{g}\\t{t}\\t{n}')
    print('\\nRequested:')
    for gene in ['PABPC1','NUDT21','LEO1']:
        gs=sorted({g for (g,t),n in pairs.items() if t==gene})
        print(gene, gs)
with h5py.File('perturb.seq.align.q1.query.h5ad','r') as f:
    qguides,qcats=cat_col(f,'guide')
    print('\\nquery guide categories')
    print('\\n'.join(qcats))
PY"
