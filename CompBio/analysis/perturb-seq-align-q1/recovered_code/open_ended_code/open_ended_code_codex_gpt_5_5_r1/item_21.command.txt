/bin/bash -lc "python - <<'PY'
import h5py, numpy as np, collections

def dec(a): return np.array([x.decode() if isinstance(x,(bytes,bytearray)) else str(x) for x in a])
with h5py.File('perturb.seq.align.q1.ref.h5ad','r') as f:
    trcats=dec(f['obs/target_gene/categories'][()])
    vr=dec(f['var/_index'][()])
with h5py.File('perturb.seq.align.q1.query.h5ad','r') as f:
    vq=dec(f['var/_index'][()])
print('targets in query var:', [t for t in trcats if t in set(vq)])
print('targets missing query var:', [t for t in trcats if t not in set(vq)])
print('ref-only count/list:', sorted(set(vr)-set(vq)))
print('query-only count/list:', sorted(set(vq)-set(vr)))
PY"
