/bin/bash -lc "python - <<'PY'
import h5py

def show(name, obj):
    depth=name.count('/')
    if depth<=2:
        typ='Group' if isinstance(obj,h5py.Group) else f'Dataset shape={obj.shape} dtype={obj.dtype}'
        attrs={k: obj.attrs[k] for k in obj.attrs.keys()}
        print(name, typ, attrs)

for fn in ['perturb.seq.align.q1.ref.h5ad','perturb.seq.align.q1.query.h5ad']:
    print('\\nFILE', fn)
    with h5py.File(fn,'r') as f:
        print('root keys', list(f.keys()))
        print('root attrs', dict(f.attrs))
        f.visititems(show)
PY"
