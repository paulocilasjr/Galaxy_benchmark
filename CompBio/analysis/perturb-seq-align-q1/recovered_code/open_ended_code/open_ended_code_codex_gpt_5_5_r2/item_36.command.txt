from scipy.sparse import csc_matrix

def dec(a): return np.array([x.decode() if isinstance(x,bytes) else str(x) for x in a])
def load(fn):
    with h5py.File(fn,'r') as f:

def lognorm_dense(X, cols):
    X=X[:,cols].tocsr().astype(np.float32)
    # IMPORTANT lib should use all genes, not selected; approximate with selected? pass all normalized first? This uses selected only, so not ideal.
    return X
# implement all-gene lib normalization then select and densify
def lognorm_select_dense(X, cols):
    X=X.copy().tocsr().astype(np.float32)
    lib=np.asarray(X.sum(1)).ravel().astype(np.float32); scale=np.divide(1e4,lib,out=np.zeros_like(lib),where=lib>0)
    X.data *= np.repeat(scale,np.diff(X.indptr)); np.log1p(X.data,out=X.data)
    return X[:,cols].toarray().astype(np.float32)
refX,refg,refc,refv=load('perturb.seq.align.q1.ref.h5ad'); qX,qg,qc,qv=load('perturb.seq.align.q1.query.h5ad')
