def prep(Xn, guides):
    cats=[g for g in sorted(set(guides)) if not g.startswith('NT-')]
    nts=[g for g in sorted(set(guides)) if g.startswith('NT-')]

        idx=np.where(guides==g)[0]

        M[g]=mu-mu_nt
        # Welch t-stat

        T[g]=t
    return cats,M,T
